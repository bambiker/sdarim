איוב פרק יא לפי החלוקה לסדרים

פרק יא בספר איוב (20 פסוקים) נמצא באיוב סדר ב (פסוקים קלב–קמט) ובאיוב סדר ג (פסוקים א–ב).

איוב סדר ב

(א) הִנֵּה־ זֹ֭את חֲקַרְנ֥וּהָ כֶּֽן־ הִ֑יא שְׁ֝מָעֶ֗נָּה וְאַתָּ֥ה דַֽע־ לָֽךְ׃ (ב) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (ג) ל֗וּ שָׁק֣וֹל יִשָּׁקֵ֣ל כַּעְשִׂ֑י (והיתי) וְ֝הַוָּתִ֗י בְּֽמֹאזְנַ֥יִם יִשְׂאוּ־ יָֽחַד׃ (ד) כִּֽי־ עַתָּ֗ה מֵח֣וֹל יַמִּ֣ים יִכְבָּ֑ד עַל־ כֵּ֝֗ן דְּבָרַ֥י לָֽעוּ׃ (ה) כִּ֤י חִצֵּ֪י שַׁדַּ֡י עִמָּדִ֗י אֲשֶׁ֣ר חֲ֭מָתָם שֹׁתָ֣ה רוּחִ֑י בִּעוּתֵ֖י אֱל֣וֹהַּ יַֽעַרְכֽוּנִי׃ (ו) הֲיִֽנְהַק־ פֶּ֥רֶא עֲלֵי־ דֶ֑שֶׁא אִ֥ם יִגְעֶה־ שּׁ֝֗וֹר עַל־ בְּלִילֽוֹ׃ (ז) הֲיֵאָכֵ֣ל תָּ֭פֵל מִבְּלִי־ מֶ֑לַח אִם־ יֶשׁ־ טַ֝֗עַם בְּרִ֣יר חַלָּמֽוּת׃ (ח) מֵאֲנָ֣ה לִנְגּ֣וֹעַ נַפְשִׁ֑י הֵ֝֗מָּה כִּדְוֵ֥י לַחְמִֽי׃ (ט) מִֽי־ יִ֭תֵּן תָּב֣וֹא שֶֽׁאֱלָתִ֑י וְ֝תִקְוָתִ֗י יִתֵּ֥ן אֱלֽוֹהַּ׃ (י) וְיֹאֵ֣ל אֱ֭לוֹהַּ וִֽידַכְּאֵ֑נִי יַתֵּ֥ר יָ֝ד֗וֹ וִֽיבַצְּעֵֽנִי׃ (יא) וּ֥תְהִי ע֨וֹד ׀ נֶ֘חָ֤מָתִ֗י וַאֲסַלְּדָ֣ה בְ֭חִילָה לֹ֣א יַחְמ֑וֹל כִּי־ לֹ֥א כִ֝חַ֗דְתִּי אִמְרֵ֥י קָדֽוֹשׁ׃ (יב) מַה־ כֹּחִ֥י כִֽי־ אֲיַחֵ֑ל וּמַה־ קִּ֝צִּ֗י כִּֽי־ אַאֲרִ֥יךְ נַפְשִֽׁי׃ (יג) אִם־ כֹּ֣חַ אֲבָנִ֣ים כֹּחִ֑י אִֽם־ בְּשָׂרִ֥י נָחֽוּשׁ׃ (יד) הַאִ֬ם אֵ֣ין עֶזְרָתִ֣י בִ֑י וְ֝תֻשִׁיָּ֗ה נִדְּחָ֥ה מִמֶּֽנִּי׃ (טו) לַמָּ֣ס מֵרֵעֵ֣הוּ חָ֑סֶד וְיִרְאַ֖ת שַׁדַּ֣י יַעֲזֽוֹב׃ (טז) אַ֭חַי בָּגְד֣וּ כְמוֹ־ נָ֑חַל כַּאֲפִ֖יק נְחָלִ֣ים יַעֲבֹֽרוּ׃ (יז) הַקֹּדְרִ֥ים מִנִּי־ קָ֑רַח עָ֝לֵ֗ימוֹ יִתְעַלֶּם־ שָֽׁלֶג׃ (יח) בְּ֭עֵת יְזֹרְב֣וּ נִצְמָ֑תוּ בְּ֝חֻמּ֗וֹ נִדְעֲכ֥וּ מִמְּקוֹמָֽם׃ (יט) יִ֭לָּ֣פְתוּ אָרְח֣וֹת דַּרְכָּ֑ם יַעֲל֖וּ בַתֹּ֣הוּ וְיֹאבֵֽדוּ׃ (כ) הִ֭בִּיטוּ אָרְח֣וֹת תֵּמָ֑א הֲלִיכֹ֥ת שְׁ֝בָ֗א קִוּוּ־ לָֽמוֹ׃ (כא) בֹּ֥שׁוּ כִּֽי־ בָטָ֑ח בָּ֥אוּ עָ֝דֶ֗יהָ וַיֶּחְפָּֽרוּ׃ (כב) כִּֽי־ עַ֭תָּה הֱיִ֣יתֶם (לא) ל֑וֹ תִּֽרְא֥וּ חֲ֝תַ֗ת וַתִּירָֽאוּ׃ (כג) הֲ‍ֽכִי־ אָ֭מַרְתִּי הָ֣בוּ לִ֑י וּ֝מִכֹּחֲכֶ֗ם שִׁחֲד֥וּ בַעֲדִֽי׃ (כד) וּמַלְּט֥וּנִי מִיַּד־ צָ֑ר וּמִיַּ֖ד עָרִיצִ֣ים תִּפְדּֽוּנִי׃ (כה) ה֭וֹרוּנִי וַאֲנִ֣י אַחֲרִ֑ישׁ וּמַה־ שָּׁ֝גִ֗יתִי הָבִ֥ינוּ לִֽי׃ (כו) מַה־ נִּמְרְצ֥וּ אִמְרֵי־ יֹ֑שֶׁר וּמַה־ יּוֹכִ֖יחַ הוֹכֵ֣חַ מִכֶּֽם׃ (כז) הַלְהוֹכַ֣ח מִלִּ֣ים תַּחְשֹׁ֑בוּ וּ֝לְר֗וּחַ אִמְרֵ֥י נֹאָֽשׁ׃ (כח) אַף־ עַל־ יָת֥וֹם תַּפִּ֑ילוּ וְ֝תִכְר֗וּ עַל־ רֵֽיעֲכֶֽם׃ (כט) וְ֭עַתָּה הוֹאִ֣ילוּ פְנוּ־ בִ֑י וְעַל־ פְּ֝נֵיכֶ֗ם אִם־ אֲכַזֵּֽב׃ (ל) שֻֽׁבוּ־ נָ֭א אַל־ תְּהִ֣י עַוְלָ֑ה (ושבי) וְשׁ֥וּבוּ ע֝וֹד צִדְקִי־ בָֽהּ׃ (לא) הֲיֵשׁ־ בִּלְשׁוֹנִ֥י עַוְלָ֑ה אִם־ חִ֝כִּ֗י לֹא־ יָבִ֥ין הַוּֽוֹת׃ (לב) הֲלֹא־ צָבָ֣א לֶאֱנ֣וֹשׁ (על)־ עֲלֵי־ אָ֑רֶץ וְכִימֵ֖י שָׂכִ֣יר יָמָֽיו׃ (לג) כְּעֶ֥בֶד יִשְׁאַף־ צֵ֑ל וּ֝כְשָׂכִ֗יר יְקַוֶּ֥ה פָעֳלֽוֹ׃ (לד) כֵּ֤ן הָנְחַ֣לְתִּי לִ֭י יַרְחֵי־ שָׁ֑וְא וְלֵיל֥וֹת עָ֝מָ֗ל מִנּוּ־ לִֽי׃ (לה) אִם־ שָׁכַ֗בְתִּי וְאָמַ֗רְתִּי מָתַ֣י אָ֭קוּם וּמִדַּד־ עָ֑רֶב וְשָׂבַ֖עְתִּי נְדֻדִ֣ים עֲדֵי־ נָֽשֶׁף׃ (לו) לָ֘בַ֤שׁ בְּשָׂרִ֣י רִ֭מָּה (וגיש) וְג֣וּשׁ עָפָ֑ר עוֹרִ֥י רָ֝גַ֗ע וַיִּמָּאֵֽס׃ (לז) יָמַ֣י קַ֭לּוּ מִנִּי־ אָ֑רֶג וַ֝יִּכְל֗וּ בְּאֶ֣פֶס תִּקְוָֽה׃ (לח) זְ֭כֹר כִּי־ ר֣וּחַ חַיָּ֑י לֹא־ תָשׁ֥וּב עֵ֝ינִ֗י לִרְא֥וֹת טֽוֹב׃ (לט) לֹֽא־ תְ֭שׁוּרֵנִי עֵ֣ין רֹ֑אִי עֵינֶ֖יךָ בִּ֣י וְאֵינֶֽנִּי׃ (מ) כָּלָ֣ה עָ֭נָן וַיֵּלַ֑ךְ כֵּ֥ן יוֹרֵ֥ד שְׁ֝א֗וֹל לֹ֣א יַעֲלֶֽה׃ (מא) לֹא־ יָשׁ֣וּב ע֣וֹד לְבֵית֑וֹ וְלֹא־ יַכִּירֶ֖נּוּ ע֣וֹד מְקֹמֽוֹ׃ (מב) גַּם־ אֲנִי֮ לֹ֤א אֶחֱשָׂ֫ךְ פִּ֥י אֲ‍ֽ֭דַבְּרָה בְּצַ֣ר רוּחִ֑י אָ֝שִׂ֗יחָה בְּמַ֣ר נַפְשִֽׁי׃ (מג) הֲ‍ֽיָם־ אָ֭נִי אִם־ תַּנִּ֑ין כִּֽי־ תָשִׂ֖ים עָלַ֣י מִשְׁמָֽר׃ (מד) כִּֽי־ אָ֭מַרְתִּי תְּנַחֲמֵ֣נִי עַרְשִׂ֑י יִשָּׂ֥א בְ֝שִׂיחִ֗י מִשְׁכָּבִֽי׃ (מה) וְחִתַּתַּ֥נִי בַחֲלֹמ֑וֹת וּֽמֵחֶזְיֹנ֥וֹת תְּבַעֲתַֽנִּי׃ (מו) וַתִּבְחַ֣ר מַחֲנָ֣ק נַפְשִׁ֑י מָ֝֗וֶת מֵֽעַצְמוֹתָֽי׃ (מז) מָ֭אַסְתִּי לֹא־ לְעֹלָ֣ם אֶֽחְיֶ֑ה חֲדַ֥ל מִ֝מֶּ֗נִּי כִּי־ הֶ֥בֶל יָמָֽי׃ (מח) מָֽה־ אֱ֭נוֹשׁ כִּ֣י תְגַדְּלֶ֑נּוּ וְכִי־ תָשִׁ֖ית אֵלָ֣יו לִבֶּֽךָ׃ (מט) וַתִּפְקְדֶ֥נּוּ לִבְקָרִ֑ים לִ֝רְגָעִ֗ים תִּבְחָנֶֽנּוּ׃ (נ) כַּ֭מָּה לֹא־ תִשְׁעֶ֣ה מִמֶּ֑נִּי לֹֽא־ תַ֝רְפֵּ֗נִי עַד־ בִּלְעִ֥י רֻקִּֽי׃ (נא) חָטָ֡אתִי מָ֤ה אֶפְעַ֨ל ׀ לָךְ֮ נֹצֵ֪ר הָאָ֫דָ֥ם לָ֤מָה שַׂמְתַּ֣נִי לְמִפְגָּ֣ע לָ֑ךְ וָאֶהְיֶ֖ה עָלַ֣י לְמַשָּֽׂא׃ (נב) וּמֶ֤ה ׀ לֹא־ תִשָּׂ֣א פִשְׁעִי֮ וְתַעֲבִ֪יר אֶת־ עֲוֺ֫נִ֥י כִּֽי־ עַ֭תָּה לֶעָפָ֣ר אֶשְׁכָּ֑ב וְשִׁ֖חֲרְתַּ֣נִי וְאֵינֶֽנִּי׃ (נג) וַ֭יַּעַן בִּלְדַּ֥ד הַשּׁוּחִ֗י וַיֹאמַֽר׃ (נד) עַד־ אָ֥ן תְּמַלֶּל־ אֵ֑לֶּה וְר֥וּחַ כַּ֝בִּיר אִמְרֵי־ פִֽיךָ׃ (נה) הַ֭אֵל יְעַוֵּ֣ת מִשְׁפָּ֑ט וְאִם־ שַׁ֝דַּ֗י יְעַוֵּֽת־ צֶֽדֶק׃ (נו) אִם־ בָּנֶ֥יךָ חָֽטְאוּ־ ל֑וֹ וַֽ֝יְשַׁלְּחֵ֗ם בְּיַד־ פִּשְׁעָֽם׃ (נז) אִם־ אַ֭תָּה תְּשַׁחֵ֣ר אֶל־ אֵ֑ל וְאֶל־ שַׁ֝דַּ֗י תִּתְחַנָּֽן׃ (נח) אִם־ זַ֥ךְ וְיָשָׁ֗ר אָ֥תָּה כִּי־ עַ֭תָּה יָעִ֣יר עָלֶ֑יךָ וְ֝שִׁלַּ֗ם נְוַ֣ת צִדְקֶֽךָ׃ (נט) וְהָיָ֣ה רֵאשִׁיתְךָ֣ מִצְעָ֑ר וְ֝אַחֲרִיתְךָ֗ יִשְׂגֶּ֥ה מְאֹֽד׃ (ס) כִּֽי־ שְׁאַל־ נָ֭א לְדֹ֣ר רִישׁ֑וֹן וְ֝כוֹנֵ֗ן לְחֵ֣קֶר אֲבוֹתָֽם׃ (סא) כִּֽי־ תְמ֣וֹל אֲ֭נַחְנוּ וְלֹ֣א נֵדָ֑ע כִּ֤י צֵ֖ל יָמֵ֣ינוּ עֲלֵי־ אָֽרֶץ׃ (סב) הֲלֹא־ הֵ֣ם י֭וֹרוּךָ יֹ֣אמְרוּ לָ֑ךְ וּ֝מִלִּבָּ֗ם יוֹצִ֥אוּ מִלִּֽים׃ (סג) הֲיִֽגְאֶה־ גֹּ֭מֶא בְּלֹ֣א בִצָּ֑ה יִשְׂגֶּה־ אָ֥חוּ בְלִי־ מָֽיִם׃ (סד) עֹדֶ֣נּוּ בְ֭אִבּוֹ לֹ֣א יִקָּטֵ֑ף וְלִפְנֵ֖י כָל־ חָצִ֣יר יִיבָֽשׁ׃ (סה) כֵּ֗ן אָ֭רְחוֹת כָּל־ שֹׁ֣כְחֵי אֵ֑ל וְתִקְוַ֖ת חָנֵ֣ף תֹּאבֵֽד׃ (סו) אֲשֶׁר־ יָק֥וֹט כִּסְל֑וֹ וּבֵ֥ית עַ֝כָּבִ֗ישׁ מִבְטַחֽוֹ׃ (סז) יִשָּׁעֵ֣ן עַל־ בֵּ֭יתוֹ וְלֹ֣א יַעֲמֹ֑ד יַחֲזִ֥יק בּ֝֗וֹ וְלֹ֣א יָקֽוּם׃ (סח) רָטֹ֣ב ה֭וּא לִפְנֵי־ שָׁ֑מֶשׁ וְעַ֥ל גַּ֝נָּת֗וֹ יֹֽנַקְתּ֥וֹ תֵצֵֽא׃ (סט) עַל־ גַּ֭ל שָֽׁרָשָׁ֣יו יְסֻבָּ֑כוּ בֵּ֖ית אֲבָנִ֣ים יֶחֱזֶֽה׃ (ע) אִם־ יְבַלְּעֶ֥נּוּ מִמְּקוֹמ֑וֹ וְכִ֥חֶשׁ בּ֝֗וֹ לֹ֣א רְאִיתִֽיךָ׃ (עא) הֶן־ ה֭וּא צָפַ֣נְתָּ מִּשָּׂ֑כֶל עַל־ כֵּ֝֗ן לֹ֣א תְרֹמֵֽם׃ (קלט) לְ֭חֵלֶק יַגִּ֣יד רֵעִ֑ים וְעֵינֵ֖י בָנָ֣יו תִּכְלֶֽנָה׃ (קמ) וְֽ֭הִצִּגַנִי לִמְשֹׁ֣ל עַמִּ֑ים וְתֹ֖פֶת לְפָנִ֣ים אֶֽהְיֶֽה׃ (קמא) וַתֵּ֣כַהּ מִכַּ֣עַשׂ עֵינִ֑י וִֽיצֻרַ֖י כַּצֵּ֣ל כֻּלָּֽם׃ (קמב) יָשֹׁ֣מּוּ יְשָׁרִ֣ים עַל־ זֹ֑את וְ֝נָקִ֗י עַל־ חָנֵ֥ף יִתְעֹרָֽר׃
