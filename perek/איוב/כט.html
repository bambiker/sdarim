איוב פרק כט לפי החלוקה לסדרים

פרק כט בספר איוב (25 פסוקים) נמצא באיוב סדר ה (פסוקים קטו–קכז) ובאיוב סדר ו (פסוקים א–יב).

איוב סדר ה

(א) יְֽמַלֵּ֥ט אִֽי־ נָקִ֑י וְ֝נִמְלַ֗ט בְּבֹ֣ר כַּפֶּֽיךָ׃ (ב) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (ג) גַּם־ הַ֭יּוֹם מְרִ֣י שִׂחִ֑י יָ֝דִ֗י כָּבְדָ֥ה עַל־ אַנְחָתִֽי׃ (ד) מִֽי־ יִתֵּ֣ן יָ֭דַעְתִּי וְאֶמְצָאֵ֑הוּ אָ֝ב֗וֹא עַד־ תְּכוּנָתֽוֹ׃ (ה) אֶעֶרְכָ֣ה לְפָנָ֣יו מִשְׁפָּ֑ט וּ֝פִ֗י אֲמַלֵּ֥א תוֹכָחֽוֹת׃ (ו) אֵ֭דְעָה מִלִּ֣ים יַעֲנֵ֑נִי וְ֝אָבִ֗ינָה מַה־ יֹּ֥אמַר לִֽי׃ (ז) הַבְּרָב־ כֹּ֭חַ יָרִ֣יב עִמָּדִ֑י לֹ֥א אַךְ־ ה֝֗וּא יָשִׂ֥ם בִּֽי׃ (ח) שָׁ֗ם יָ֭שָׁר נוֹכָ֣ח עִמּ֑וֹ וַאֲפַלְּטָ֥ה לָ֝נֶ֗צַח מִשֹּׁפְטִֽי׃ (ט) הֵ֤ן קֶ֣דֶם אֶהֱלֹ֣ךְ וְאֵינֶ֑נּוּ וְ֝אָח֗וֹר וְֽלֹא־ אָבִ֥ין לֽוֹ׃ (י) שְׂמֹ֣אול בַּעֲשֹׂת֣וֹ וְלֹא־ אָ֑חַז יַעְטֹ֥ף יָ֝מִ֗ין וְלֹ֣א אֶרְאֶֽה׃ (יא) כִּֽי־ יָ֭דַע דֶּ֣רֶךְ עִמָּדִ֑י בְּ֝חָנַ֗נִי כַּזָּהָ֥ב אֵצֵֽא׃ (יב) בַּ֭אֲשֻׁרוֹ אָחֲזָ֣ה רַגְלִ֑י דַּרְכּ֖וֹ שָׁמַ֣רְתִּי וְלֹא־ אָֽט׃ (יג) מִצְוַ֣ת שְׂ֭פָתָיו וְלֹ֣א אָמִ֑ישׁ מֵ֝חֻקִּ֗י צָפַ֥נְתִּי אִמְרֵי־ פִֽיו׃ (יד) וְה֣וּא בְ֭אֶחָד וּמִ֣י יְשִׁיבֶ֑נּוּ וְנַפְשׁ֖וֹ אִוְּתָ֣ה וַיָּֽעַשׂ׃ (טו) כִּ֭י יַשְׁלִ֣ים חֻקִּ֑י וְכָהֵ֖נָּה רַבּ֣וֹת עִמּֽוֹ׃ (טז) עַל־ כֵּ֭ן מִפָּנָ֣יו אֶבָּהֵ֑ל אֶ֝תְבּוֹנֵ֗ן וְאֶפְחַ֥ד מִמֶּֽנּוּ׃ (יז) וְ֭אֵל הֵרַ֣ךְ לִבִּ֑י וְ֝שַׁדַּ֗י הִבְהִילָֽנִי׃ (יח) כִּֽי־ לֹ֣א נִ֭צְמַתִּי מִפְּנֵי־ חֹ֑שֶׁךְ וּ֝מִפָּנַ֗י כִּסָּה־ אֹֽפֶל׃ (יט) מַדּ֗וּעַ מִ֭שַּׁדַּי לֹא־ נִצְפְּנ֣וּ עִתִּ֑ים (וידעו) וְ֝יֹדְעָ֗יו לֹא־ חָ֥זוּ יָמָֽיו׃ (כ) גְּבֻל֥וֹת יַשִּׂ֑יגוּ עֵ֥דֶר גָּ֝זְל֗וּ וַיִּרְעֽוּ׃ (כא) חֲמ֣וֹר יְתוֹמִ֣ים יִנְהָ֑גוּ יַ֝חְבְּל֗וּ שׁ֣וֹר אַלְמָנָֽה׃ (כב) יַטּ֣וּ אֶבְיוֹנִ֣ים מִדָּ֑רֶךְ יַ֥חַד חֻ֝בְּא֗וּ עֲנִיֵּי־ אָֽרֶץ׃ (כג) הֵ֤ן פְּרָאִ֨ים ׀ בַּֽמִּדְבָּ֗ר יָצְא֣וּ בְּ֭פָעֳלָם מְשַׁחֲרֵ֣י לַטָּ֑רֶף עֲרָבָ֥ה ל֥וֹ לֶ֝֗חֶם לַנְּעָרִֽים׃ (כד) בַּ֭שָּׂדֶה בְּלִיל֣וֹ (יקצירו) יִקְצ֑וֹרוּ וְכֶ֖רֶם רָשָׁ֣ע יְלַקֵּֽשׁוּ׃ (כה) עָר֣וֹם יָ֭לִינוּ מִבְּלִ֣י לְב֑וּשׁ וְאֵ֥ין כְּ֝ס֗וּת בַּקָּרָֽה׃ (כו) מִזֶּ֣רֶם הָרִ֣ים יִרְטָ֑בוּ וּֽמִבְּלִ֥י מַ֝חְסֶ֗ה חִבְּקוּ־ צֽוּר׃ (כז) יִ֭גְזְלוּ מִשֹּׁ֣ד יָת֑וֹם וְֽעַל־ עָנִ֥י יַחְבֹּֽלוּ׃ (כח) עָר֣וֹם הִ֭לְּכוּ בְּלִ֣י לְב֑וּשׁ וּ֝רְעֵבִ֗ים נָ֣שְׂאוּ עֹֽמֶר׃ (כט) בֵּין־ שׁוּרֹתָ֥ם יַצְהִ֑ירוּ יְקָבִ֥ים דָּ֝רְכ֗וּ וַיִּצְמָֽאוּ׃ (ל) מֵ֘עִ֤יר מְתִ֨ים ׀ יִנְאָ֗קוּ וְנֶֽפֶשׁ־ חֲלָלִ֥ים תְּשַׁוֵּ֑עַ וֶ֝אֱל֗וֹהַּ לֹא־ יָשִׂ֥ים תִּפְלָֽה׃ (לא) הֵ֤מָּה ׀ הָיוּ֮ בְּֽמֹרְדֵ֫י־ א֥וֹר לֹֽא־ הִכִּ֥ירוּ דְרָכָ֑יו וְלֹ֥א יָ֝שְׁב֗וּ בִּנְתִיבֹתָֽיו׃ (לב) לָא֡וֹר יָ֘ק֤וּם רוֹצֵ֗חַ יִֽקְטָל־ עָנִ֥י וְאֶבְי֑וֹן וּ֝בַלַּ֗יְלָה יְהִ֣י כַגַּנָּֽב׃ (לג) וְעֵ֤ין נֹאֵ֨ף ׀ שָׁ֤מְרָֽה נֶ֣שֶׁף לֵ֭אמֹר לֹא־ תְשׁוּרֵ֣נִי עָ֑יִן וְסֵ֖תֶר פָּנִ֣ים יָשִֽׂים׃ (לד) חָתַ֥ר בַּחֹ֗שֶׁךְ בָּ֫תִּ֥ים יוֹמָ֥ם חִתְּמוּ־ לָ֗מוֹ לֹא־ יָ֥דְעוּ אֽוֹר׃ (לה) כִּ֤י יַחְדָּ֨ו ׀ בֹּ֣קֶר לָ֣מוֹ צַלְמָ֑וֶת כִּֽי־ יַ֝כִּ֗יר בַּלְה֥וֹת צַלְמָֽוֶת׃ (לו) קַֽל־ ה֤וּא ׀ עַל־ פְּנֵי־ מַ֗יִם תְּקֻלַּ֣ל חֶלְקָתָ֣ם בָּאָ֑רֶץ לֹֽא־ יִ֝פְנֶה דֶּ֣רֶךְ כְּרָמִֽים׃ (לז) צִיָּ֤ה גַם־ חֹ֗ם יִגְזְל֥וּ מֵֽימֵי־ שֶׁ֗לֶג שְׁא֣וֹל חָטָֽאוּ׃ (לח) יִשְׁכָּ֘חֵ֤הוּ רֶ֨חֶם ׀ מְתָ֘ק֤וֹ רִמָּ֗ה ע֥וֹד לֹֽא־ יִזָּכֵ֑ר וַתִּשָּׁבֵ֖ר כָּעֵ֣ץ עַוְלָֽה׃ (לט) רֹעֶ֣ה עֲ֭קָרָה לֹ֣א תֵלֵ֑ד וְ֝אַלְמָנָ֗ה לֹ֣א יְיֵטִֽיב׃ (מ) וּמָשַׁ֣ךְ אַבִּירִ֣ים בְּכֹח֑וֹ יָ֝ק֗וּם וְֽלֹא־ יַאֲמִ֥ין בַּֽחַיִּֽין׃ (מא) יִתֶּן־ ל֣וֹ לָ֭בֶטַח וְיִשָּׁעֵ֑ן וְ֝עֵינֵ֗יהוּ עַל־ דַּרְכֵיהֶֽם׃ (מב) ר֤וֹמּוּ מְּעַ֨ט ׀ וְֽאֵינֶ֗נּוּ וְֽהֻמְּכ֗וּ כַּכֹּ֥ל יִקָּפְצ֑וּן וּכְרֹ֖אשׁ שִׁבֹּ֣לֶת יִמָּֽלוּ׃ (מג) וְאִם־ לֹ֣א אֵ֭פוֹ מִ֣י יַכְזִיבֵ֑נִי וְיָשֵׂ֥ם לְ֝אַ֗ל מִלָּתִֽי׃ (מד) וַ֭יַּעַן בִּלְדַּ֥ד הַשֻּׁחִ֗י וַיֹּאמַֽר׃ (מה) הַמְשֵׁ֣ל וָפַ֣חַד עִמּ֑וֹ עֹשֶׂ֥ה שָׁ֝ל֗וֹם בִּמְרוֹמָֽיו׃ (מו) הֲיֵ֣שׁ מִ֭סְפָּר לִגְדוּדָ֑יו וְעַל־ מִ֝֗י לֹא־ יָק֥וּם אוֹרֵֽהוּ׃ (מז) וּמַה־ יִּצְדַּ֣ק אֱנ֣וֹשׁ עִם־ אֵ֑ל וּמַה־ יִּ֝זְכֶּ֗ה יְל֣וּד אִשָּֽׁה׃ (מח) הֵ֣ן עַד־ יָ֭רֵחַ וְלֹ֣א יַאֲהִ֑יל וְ֝כוֹכָבִ֗ים לֹא־ זַכּ֥וּ בְעֵינָֽיו׃ (מט) אַ֭ף כִּֽי־ אֱנ֣וֹשׁ רִמָּ֑ה וּבֶן־ אָ֝דָ֗ם תּוֹלֵעָֽה׃ (נ) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (נא) מֶה־ עָזַ֥רְתָּ לְלֹא־ כֹ֑חַ ה֝וֹשַׁ֗עְתָּ זְר֣וֹעַ לֹא־ עֹֽז׃ (נב) מַה־ יָּ֭עַצְתָּ לְלֹ֣א חָכְמָ֑ה וְ֝תוּשִׁיָּ֗ה לָרֹ֥ב הוֹדָֽעְתָּ׃ (נג) אֶת־ מִ֭י הִגַּ֣דְתָּ מִלִּ֑ין וְנִשְׁמַת־ מִ֝י יָצְאָ֥ה מִמֶּֽךָּ׃ (נד) הָרְפָאִ֥ים יְחוֹלָ֑לוּ מִתַּ֥חַת מַ֝֗יִם וְשֹׁכְנֵיהֶֽם׃ (נה) עָר֣וֹם שְׁא֣וֹל נֶגְדּ֑וֹ וְאֵ֥ין כְּ֝ס֗וּת לָֽאֲבַדּֽוֹן׃ (נו) נֹטֶ֣ה צָפ֣וֹן עַל־ תֹּ֑הוּ תֹּ֥לֶה אֶ֝֗רֶץ עַל־ בְּלִי־ מָֽה׃ (נז) צֹרֵֽר־ מַ֥יִם בְּעָבָ֑יו וְלֹא־ נִבְקַ֖ע עָנָ֣ן תַּחְתָּֽם׃ (נח) מְאַחֵ֥ז פְּנֵי־ כִסֵּ֑ה פַּרְשֵׁ֖ז עָלָ֣יו עֲנָנֽוֹ׃ (נט) חֹֽק־ חָ֭ג עַל־ פְּנֵי־ מָ֑יִם עַד־ תַּכְלִ֖ית א֣וֹר עִם־ חֹֽשֶׁךְ׃ (ס) עַמּוּדֵ֣י שָׁמַ֣יִם יְרוֹפָ֑פוּ וְ֝יִתְמְה֗וּ מִגַּעֲרָתֽוֹ׃ (סא) בְּ֭כֹחוֹ רָגַ֣ע הַיָּ֑ם (ובתובנתו) וּ֝בִתְבוּנָת֗וֹ מָ֣חַץ רָֽהַב׃ (סב) בְּ֭רוּחוֹ שָׁמַ֣יִם שִׁפְרָ֑ה חֹֽלֲלָ֥ה יָ֝ד֗וֹ נָחָ֥שׁ בָּרִֽיחַ׃ (סג) הֶן־ אֵ֤לֶּה ׀ קְצ֬וֹת (דרכו) דְּרָכָ֗יו וּמַה־ שֵּׁ֣מֶץ דָּ֭בָר נִשְׁמַע־ בּ֑וֹ וְרַ֥עַם (גבורתו) גְּ֝בוּרוֹתָ֗יו מִ֣י יִתְבּוֹנָֽן׃ (סד) וַיֹּ֣סֶף אִ֭יּוֹב שְׂאֵ֥ת מְשָׁל֗וֹ וַיֹּאמַֽר׃ (סה) חַי־ אֵ֭ל הֵסִ֣יר מִשְׁפָּטִ֑י וְ֝שַׁדַּ֗י הֵמַ֥ר נַפְשִֽׁי׃ (סו) כִּֽי־ כָל־ ע֣וֹד נִשְׁמָתִ֣י בִ֑י וְר֖וּחַ אֱל֣וֹהַּ בְּאַפִּֽי׃ (סז) אִם־ תְּדַבֵּ֣רְנָה שְׂפָתַ֣י עַוְלָ֑ה וּ֝לְשׁוֹנִ֗י אִם־ יֶהְגֶּ֥ה רְמִיָּֽה׃ (סח) חָלִ֣ילָה לִּי֮ אִם־ אַצְדִּ֪יק אֶ֫תְכֶ֥ם עַד־ אֶגְוָ֑ע לֹא־ אָסִ֖יר תֻּמָּתִ֣י מִמֶּֽנִּי׃ (סט) בְּצִדְקָתִ֣י הֶ֭חֱזַקְתִּי וְלֹ֣א הַבּוּזִ֗י וַיֹּ֫אמַ֥ר צָ֘עִ֤יר אֲנִ֣י לְ֭יָמִים וְאַתֶּ֣ם יְשִׁישִׁ֑ים עַל־ כֵּ֖ן זָחַ֥לְתִּי וָֽאִירָ֓א ׀ מֵחַוֺּ֖ת דֵּעִ֣י אֶתְכֶֽם׃ (צ) אָ֭מַרְתִּי יָמִ֣ים יְדַבֵּ֑רוּ וְרֹ֥ב שָׁ֝נִ֗ים יֹדִ֥יעוּ חָכְמָֽה׃ (צא) אָ֭כֵן רֽוּחַ־ הִ֣יא בֶאֱנ֑וֹשׁ וְנִשְׁמַ֖ת שַׁדַּ֣י תְּבִינֵֽם׃ (צב) לֹֽא־ רַבִּ֥ים יֶחְכָּ֑מוּ וּ֝זְקֵנִ֗ים יָבִ֥ינוּ מִשְׁפָּֽט׃ (צג) לָכֵ֣ן אָ֭מַרְתִּי שִׁמְעָה־ לִּ֑י אֲחַוֶּ֖ה דֵּעִ֣י אַף־ אָֽנִי׃ (צד) הֵ֤ן הוֹחַ֨לְתִּי לְֽדִבְרֵיכֶ֗ם אָ֭זִין עַד־ תְּב֥וּנֹֽתֵיכֶ֑ם עַֽד־ תַּחְקְר֥וּן מִלִּֽין׃ (צה) וְעָֽדֵיכֶ֗ם אֶתְבּ֫וֹנָ֥ן וְהִנֵּ֤ה אֵ֣ין לְאִיּ֣וֹב מוֹכִ֑יחַ עוֹנֶ֖ה אֲמָרָ֣יו מִכֶּֽם׃ (צו) פֶּן־ תֹּ֣֭אמְרוּ מָצָ֣אנוּ חָכְמָ֑ה אֵ֖ל יִדְּפֶ֣נּוּ לֹא־ אִֽישׁ׃ (צז) וְלֹא־ עָרַ֣ךְ אֵלַ֣י מִלִּ֑ין וּ֝בְאִמְרֵיכֶ֗ם לֹ֣א אֲשִׁיבֶֽנּוּ׃ (צח) חַ֭תּוּ לֹא־ עָ֣נוּ ע֑וֹד הֶעְתִּ֖יקוּ מֵהֶ֣ם מִלִּֽים׃ (צט) וְ֭הוֹחַלְתִּי כִּי־ לֹ֣א יְדַבֵּ֑רוּ כִּ֥י עָ֝מְד֗וּ לֹא־ עָ֥נוּ עֽוֹד׃ (ק) אַעֲנֶ֣ה אַף־ אֲנִ֣י חֶלְקִ֑י אֲחַוֶּ֖ה דֵעִ֣י אַף־ אָֽנִי׃ (קא) כִּ֭י מָלֵ֣תִי מִלִּ֑ים הֱ֝צִיקַ֗תְנִי ר֣וּחַ בִּטְנִֽי׃ (קב) הִנֵּֽה־ בִטְנִ֗י כְּיַ֥יִן לֹא־ יִפָּתֵ֑חַ כְּאֹב֥וֹת חֲ֝דָשִׁ֗ים יִבָּקֵֽעַ׃ (קג) אֲדַבְּרָ֥ה וְיִֽרְוַֽח־ לִ֑י אֶפְתַּ֖ח שְׂפָתַ֣י וְאֶֽעֱנֶֽה׃ (קד) אַל־ נָ֭א אֶשָּׂ֣א פְנֵי־ אִ֑ישׁ וְאֶל־ אָ֝דָ֗ם לֹ֣א אֲכַנֶּֽה׃ (קה) כִּ֤י לֹ֣א יָדַ֣עְתִּי אֲכַנֶּ֑ה כִּ֝מְעַ֗ט יִשָּׂאֵ֥נִי עֹשֵֽׂנִי׃ (קו) וְֽאוּלָ֗ם שְׁמַֽע־ נָ֣א אִיּ֣וֹב מִלָּ֑י וְֽכָל־ דְּבָרַ֥י הַאֲזִֽינָה׃ (קז) הִנֵּה־ נָ֭א פָּתַ֣חְתִּי פִ֑י דִּבְּרָ֖ה לְשׁוֹנִ֣י בְחִכִּֽי׃ (קח) יֹֽשֶׁר־ לִבִּ֥י אֲמָרָ֑י וְדַ֥עַת שְׂ֝פָתַ֗י בָּר֥וּר מִלֵּֽלוּ׃ (קט) רֽוּחַ־ אֵ֥ל עָשָׂ֑תְנִי וְנִשְׁמַ֖ת שַׁדַּ֣י תְּחַיֵּֽנִי׃ (קי) אִם־ תּוּכַ֥ל הֲשִׁיבֵ֑נִי עֶרְכָ֥ה לְ֝פָנַ֗י הִתְיַצָּֽבָה׃ (קיא) הֵן־ אֲנִ֣י כְפִ֣יךָ לָאֵ֑ל מֵ֝חֹ֗מֶר קֹרַ֥צְתִּי גַם־ אָֽנִי׃ (קיב) הִנֵּ֣ה אֵ֭מָתִי לֹ֣א תְבַעֲתֶ֑ךָּ וְ֝אַכְפִּ֗י עָלֶ֥יךָ לֹא־ יִכְבָּֽד׃ (קיג) אַ֭ךְ אָמַ֣רְתָּ בְאָזְנָ֑י וְק֖וֹל מִלִּ֣ין אֶשְׁמָֽע׃ (קיד) זַ֥ךְ אֲנִ֗י בְּֽלִ֫י פָ֥שַׁע חַ֥ף אָנֹכִ֑י וְלֹ֖א עָוֺ֣ן לִֽי׃ (קטו) הֵ֣ן תְּ֭נוּאוֹת עָלַ֣י יִמְצָ֑א יַחְשְׁבֵ֖נִי לְאוֹיֵ֣ב לֽוֹ׃ (קטז) יָשֵׂ֣ם בַּסַּ֣ד רַגְלָ֑י יִ֝שְׁמֹ֗ר כָּל־ אָרְחֹתָֽי׃ (קיז) הֶן־ זֹ֣את לֹא־ צָדַ֣קְתָּ אֶעֱנֶ֑ךָּ כִּֽי־ יִרְבֶּ֥ה אֱ֝ל֗וֹהַ מֵאֱנֽוֹשׁ׃ (קיח) מַ֭דּוּעַ אֵלָ֣יו רִיב֑וֹתָ כִּ֥י כָל־ דְּ֝בָרָ֗יו לֹ֣א־ יַעֲנֶֽה׃ (קיט) כִּֽי־ בְאַחַ֥ת יְדַבֶּר־ אֵ֑ל וּ֝בִשְׁתַּ֗יִם לֹ֣א יְשׁוּרֶֽנָּה׃ (קכ) בַּחֲל֤וֹם ׀ חֶזְי֬וֹן לַ֗יְלָה בִּנְפֹ֣ל תַּ֭רְדֵּמָה עַל־ אֲנָשִׁ֑ים בִּ֝תְנוּמ֗וֹת עֲלֵ֣י מִשְׁכָּֽב׃ (קכא) אָ֣ז יִ֭גְלֶה אֹ֣זֶן אֲנָשִׁ֑ים וּבְמֹ֖סָרָ֣ם יַחְתֹּֽם׃ (קכב) לְ֭הָסִיר אָדָ֣ם מַעֲשֶׂ֑ה וְגֵוָ֖ה מִגֶּ֣בֶר יְכַסֶּֽה׃ (קכג) יַחְשֹׂ֣ךְ נַ֭פְשׁוֹ מִנִּי־ שָׁ֑חַת וְ֝חַיָּת֗וֹ מֵעֲבֹ֥ר בַּשָּֽׁלַח׃ (קכד) וְהוּכַ֣ח בְּ֭מַכְאוֹב עַל־ מִשְׁכָּב֑וֹ (וריב) וְר֖וֹב עֲצָמָ֣יו אֵתָֽן׃ (קכה) וְזִֽהֲמַ֣תּוּ חַיָּת֣וֹ לָ֑חֶם וְ֝נַפְשׁ֗וֹ מַאֲכַ֥ל תַּאֲוָֽה׃ (קכו) יִ֣כֶל בְּשָׂר֣וֹ מֵרֹ֑אִי (ושפי) וְשֻׁפּ֥וּ עַ֝צְמוֹתָ֗יו לֹ֣א רֻאּֽוּ׃ (קכז) וַתִּקְרַ֣ב לַשַּׁ֣חַת נַפְשׁ֑וֹ וְ֝חַיָּת֗וֹ לַֽמְמִתִֽים׃ (קכח) אִם־ יֵ֤שׁ עָלָ֨יו ׀ מַלְאָ֗ךְ מֵלִ֗יץ אֶחָ֥ד מִנִּי־ אָ֑לֶף לְהַגִּ֖יד לְאָדָ֣ם יָשְׁרֽוֹ׃ (קכט) וַיְחֻנֶּ֗נּוּ וַיֹּ֗אמֶר פְּ֭דָעֵהוּ מֵרֶ֥דֶת שָׁ֗חַת מָצָ֥אתִי כֹֽפֶר׃ (קל) רֻֽטֲפַ֣שׁ בְּשָׂר֣וֹ מִנֹּ֑עַר יָ֝שׁ֗וּב לִימֵ֥י עֲלוּמָֽיו׃ (קלא) יֶעְתַּ֤ר אֶל־ אֱל֨וֹהַּ ׀ וַיִּרְצֵ֗הוּ וַיַּ֣רְא פָּ֭נָיו בִּתְרוּעָ֑ה וַיָּ֥שֶׁב לֶ֝אֱנ֗וֹשׁ צִדְקָתֽוֹ׃ (קלב) יָשֹׁ֤ר ׀ עַל־ אֲנָשִׁ֗ים וַיֹּ֗אמֶר חָ֭טָאתִי וְיָשָׁ֥ר הֶעֱוֵ֗יתִי וְלֹא־ שָׁ֥וָה לִֽי׃ (קלג) פָּדָ֣ה (נפשי) נַ֭פְשׁוֹ מֵעֲבֹ֣ר בַּשָּׁ֑חַת (וחיתי) וְ֝חַיָּתוֹ בָּא֥וֹר תִּרְאֶֽה׃ (קלד) הֶן־ כָּל־ אֵ֭לֶּה יִפְעַל־ אֵ֑ל פַּעֲמַ֖יִם שָׁל֣וֹשׁ עִם־ גָּֽבֶר׃ (קלה) לְהָשִׁ֣יב נַ֭פְשׁוֹ מִנִּי־ שָׁ֑חַת לֵ֝א֗וֹר בְּא֣וֹר הַֽחַיִּים׃ (קלו) הַקְשֵׁ֖ב אִיּ֥וֹב שְֽׁמַֽע־ לִ֑י הַ֝חֲרֵ֗שׁ וְאָנֹכִ֥י אֲדַבֵּֽר׃ (קלז) אִם־ יֵשׁ־ מִלִּ֥ין הֲשִׁיבֵ֑נִי דַּ֝בֵּ֗ר כִּֽי־ חָפַ֥צְתִּי
