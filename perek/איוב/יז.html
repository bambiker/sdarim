איוב פרק יז לפי החלוקה לסדרים

פרק יז בספר איוב (16 פסוקים) נמצא באיוב סדר ג (פסוקים קלה–קמב) ובאיוב סדר ד (פסוקים א–ח).

איוב סדר ג

(א) וְֽ֭רָבַצְתָּ וְאֵ֣ין מַחֲרִ֑יד וְחִלּ֖וּ פָנֶ֣יךָ רַבִּֽים׃ (ב) וְעֵינֵ֥י רְשָׁעִ֗ים תִּ֫כְלֶ֥ינָה וּ֭מָנוֹס אָבַ֣ד מִנְהֶ֑ם וְ֝תִקְוָתָ֗ם מַֽפַּח־ נָֽפֶשׁ׃ (ג) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (ד) אָ֭מְנָם כִּ֣י אַתֶּם־ עָ֑ם וְ֝עִמָּכֶ֗ם תָּמ֥וּת חָכְמָֽה׃ (ה) גַּם־ לִ֤י לֵבָ֨ב ׀ כְּֽמוֹכֶ֗ם לֹא־ נֹפֵ֣ל אָנֹכִ֣י מִכֶּ֑ם וְאֶת־ מִי־ אֵ֥ין כְּמוֹ־ אֵֽלֶּה׃ (ו) שְׂחֹ֤ק לְרֵעֵ֨הוּ ׀ אֶֽהְיֶ֗ה קֹרֵ֣א לֶ֭אֱלוֹהַּ וַֽיַּעֲנֵ֑הוּ שְׂ֝ח֗וֹק צַדִּ֥יק תָּמִֽים׃ (ז) לַפִּ֣יד בּ֭וּז לְעַשְׁתּ֣וּת שַׁאֲנָ֑ן נָ֝כ֗וֹן לְמ֣וֹעֲדֵי רָֽגֶל׃ (ח) יִשְׁלָ֤יוּ אֹֽהָלִ֨ים ׀ לְשֹׁ֥דְדִ֗ים וּֽ֭בַטֻּחוֹת לְמַרְגִּ֣יזֵי אֵ֑ל לַאֲשֶׁ֤ר הֵבִ֖יא אֱל֣וֹהַּ בְּיָדֽוֹ׃ (ט) וְֽאוּלָ֗ם שְׁאַל־ נָ֣א בְהֵמ֣וֹת וְתֹרֶ֑ךָּ וְע֥וֹף הַ֝שָּׁמַ֗יִם וְיַגֶּד־ לָֽךְ׃ (י) א֤וֹ שִׂ֣יחַ לָאָ֣רֶץ וְתֹרֶ֑ךָּ וִֽיסַפְּר֥וּ לְ֝ךָ֗ דְּגֵ֣י הַיָּֽם׃ (יא) מִ֭י לֹא־ יָדַ֣ע בְּכָל־ אֵ֑לֶּה כִּ֥י יַד־ יְ֝הוָה עָ֣שְׂתָה זֹּֽאת׃ (יב) אֲשֶׁ֣ר בְּ֭יָדוֹ נֶ֣פֶשׁ כָּל־ חָ֑י וְ֝ר֗וּחַ כָּל־ בְּשַׂר־ אִֽישׁ׃ (יג) הֲלֹא־ אֹ֭זֶן מִלִּ֣ין תִּבְחָ֑ן וְ֝חֵ֗ךְ אֹ֣כֶל יִטְעַם־ לֽוֹ׃ (יד) בִּֽישִׁישִׁ֥ים חָכְמָ֑ה וְאֹ֖רֶךְ יָמִ֣ים תְּבוּנָֽה׃ (טו) עִ֭מּוֹ חָכְמָ֣ה וּגְבוּרָ֑ה ל֝֗וֹ עֵצָ֥ה וּתְבוּנָֽה׃ (טז) הֵ֣ן יַ֭הֲרוֹס וְלֹ֣א יִבָּנֶ֑ה יִסְגֹּ֥ר עַל־ אִ֝֗ישׁ וְלֹ֣א יִפָּתֵֽחַ׃ (יז) הֵ֤ן יַעְצֹ֣ר בַּמַּ֣יִם וְיִבָ֑שׁוּ וִֽ֝ישַׁלְּחֵ֗ם וְיַ֖הַפְכוּ אָֽרֶץ׃ (יח) עִ֭מּוֹ עֹ֣ז וְתֽוּשִׁיָּ֑ה ל֝֗וֹ שֹׁגֵ֥ג וּמַשְׁגֶּֽה׃ (יט) מוֹלִ֣יךְ יוֹעֲצִ֣ים שׁוֹלָ֑ל וְֽשֹׁפְטִ֥ים יְהוֹלֵֽל׃ (כ) מוּסַ֣ר מְלָכִ֣ים פִּתֵּ֑חַ וַיֶּאְסֹ֥ר אֵ֝ז֗וֹר בְּמָתְנֵיהֶֽם׃ (כא) מוֹלִ֣יךְ כֹּהֲנִ֣ים שׁוֹלָ֑ל וְאֵֽתָנִ֣ים יְסַלֵּֽף׃ (כב) מֵסִ֣יר שָׂ֭פָה לְנֶאֱמָנִ֑ים וְטַ֖עַם זְקֵנִ֣ים יִקָּֽח׃ (כג) שׁוֹפֵ֣ךְ בּ֭וּז עַל־ נְדִיבִ֑ים וּמְזִ֖יחַ אֲפִיקִ֣ים רִפָּֽה׃ (כד) מְגַלֶּ֣ה עֲ֭מֻקוֹת מִנִּי־ חֹ֑שֶׁךְ וַיֹּצֵ֖א לָא֣וֹר צַלְמָֽוֶת׃ (כה) מַשְׂגִּ֣יא לַ֭גּוֹיִם וַֽיְאַבְּדֵ֑ם שֹׁטֵ֥חַ לַ֝גּוֹיִ֗ם וַיַּנְחֵֽם׃ (כו) מֵסִ֗יר לֵ֭ב רָאשֵׁ֣י עַם־ הָאָ֑רֶץ וַ֝יַּתְעֵ֗ם בְּתֹ֣הוּ לֹא־ דָֽרֶךְ׃ (כז) יְמַֽשְׁשׁוּ־ חֹ֥שֶׁךְ וְלֹא־ א֑וֹר וַ֝יַּתְעֵ֗ם כַּשִּׁכּֽוֹר׃ (כח) הֶן־ כֹּ֭ל רָאֲתָ֣ה עֵינִ֑י שָֽׁמְעָ֥ה אָ֝זְנִ֗י וַתָּ֥בֶן לָֽהּ׃ (כט) כְּֽ֭דַעְתְּכֶם יָדַ֣עְתִּי גַם־ אָ֑נִי לֹא־ נֹפֵ֖ל אָנֹכִ֣י מִכֶּֽם׃ (ל) אוּלָ֗ם אֲ֭נִי אֶל־ שַׁדַּ֣י אֲדַבֵּ֑ר וְהוֹכֵ֖חַ אֶל־ אֵ֣ל אֶחְפָּֽץ׃ (לא) וְֽאוּלָ֗ם אַתֶּ֥ם טֹֽפְלֵי־ שָׁ֑קֶר רֹפְאֵ֖י אֱלִ֣ל כֻּלְּכֶֽם׃ (לב) מִֽי־ יִ֭תֵּן הַחֲרֵ֣שׁ תַּחֲרִישׁ֑וּן וּתְהִ֖י לָכֶ֣ם לְחָכְמָֽה׃ (לג) שִׁמְעוּ־ נָ֥א תוֹכַחְתִּ֑י וְרִב֖וֹת שְׂפָתַ֣י הַקְשִֽׁיבוּ׃ (לד) הַ֭לְאֵל תְּדַבְּר֣וּ עַוְלָ֑ה וְ֝ל֗וֹ תְּֽדַבְּר֥וּ רְמִיָּֽה׃ (לה) הֲפָנָ֥יו תִּשָּׂא֑וּן אִם־ לָאֵ֥ל תְּרִיבֽוּן׃ (לו) הֲ֭טוֹב כִּֽי־ יַחְקֹ֣ר אֶתְכֶ֑ם אִם־ כְּהָתֵ֥ל בֶּ֝אֱנ֗וֹשׁ תְּהָתֵ֥לּוּ בֽוֹ׃ (לז) הוֹכֵ֣חַ יוֹכִ֣יחַ אֶתְכֶ֑ם אִם־ בַּ֝סֵּ֗תֶר פָּנִ֥ים תִּשָּׂאֽוּן׃ (לח) הֲלֹ֣א שְׂ֭אֵתוֹ תְּבַעֵ֣ת אֶתְכֶ֑ם וּ֝פַחְדּ֗וֹ יִפֹּ֥ל עֲלֵיכֶֽם׃ (לט) זִֽ֭כְרֹנֵיכֶם מִשְׁלֵי־ אֵ֑פֶר לְגַבֵּי־ חֹ֝֗מֶר גַּבֵּיכֶֽם׃ (מ) הַחֲרִ֣ישׁוּ מִ֭מֶּנִּי וַאֲדַבְּרָה־ אָ֑נִי וְיַעֲבֹ֖ר עָלַ֣י מָֽה׃ (מא) עַל־ מָ֤ה ׀ אֶשָּׂ֣א בְשָׂרִ֣י בְשִׁנָּ֑י וְ֝נַפְשִׁ֗י אָשִׂ֥ים בְּכַפִּֽי׃ (מב) הֵ֣ן יִ֭קְטְלֵנִי (לא) ל֣וֹ אֲיַחֵ֑ל אַךְ־ דְּ֝רָכַ֗י אֶל־ פָּנָ֥יו אוֹכִֽיחַ׃ (מג) גַּם־ הוּא־ לִ֥י לִֽישׁוּעָ֑ה כִּי־ לֹ֥א לְ֝פָנָ֗יו חָנֵ֥ף יָבֽוֹא׃ (מד) שִׁמְע֣וּ שָׁ֭מוֹעַ מִלָּתִ֑י וְ֝אַֽחֲוָתִ֗י בְּאָזְנֵיכֶֽם׃ (מה) הִנֵּה־ נָ֭א עָרַ֣כְתִּי מִשְׁפָּ֑ט יָ֝דַ֗עְתִּי כִּֽי־ אֲנִ֥י אֶצְדָּֽק׃ (מו) מִי־ ה֭וּא יָרִ֣יב עִמָּדִ֑י כִּֽי־ עַתָּ֖ה אַחֲרִ֣ישׁ וְאֶגְוָֽע׃ (מז) אַךְ־ שְׁ֭תַּיִם אַל־ תַּ֣עַשׂ עִמָּדִ֑י אָ֥ז מִ֝פָּנֶ֗יךָ לֹ֣א אֶסָּתֵֽר׃ (מח) כַּ֭פְּךָ מֵעָלַ֣י הַרְחַ֑ק וְ֝אֵ֥מָתְךָ֗ אַֽל־ תְּבַעֲתַֽנִּי׃ (מט) וּ֭קְרָא וְאָנֹכִ֣י אֶֽעֱנֶ֑ה אֽוֹ־ אֲ֝דַבֵּ֗ר וַהֲשִׁיבֵֽנִי׃ (נ) כַּמָּ֣ה לִ֭י עֲוֺנ֣וֹת וְחַטָּא֑וֹת פִּֽשְׁעִ֥י וְ֝חַטָּאתִ֗י הֹדִיעֵֽנִי׃ (נא) לָֽמָּה־ פָנֶ֥יךָ תַסְתִּ֑יר וְתַחְשְׁבֵ֖נִי לְאוֹיֵ֣ב לָֽךְ׃ (נב) הֶעָלֶ֣ה נִדָּ֣ף תַּעֲר֑וֹץ וְאֶת־ קַ֖שׁ יָבֵ֣שׁ תִּרְדֹּֽף׃ (נג) כִּֽי־ תִכְתֹּ֣ב עָלַ֣י מְרֹר֑וֹת וְ֝תוֹרִישֵׁ֗נִי עֲוֺנ֥וֹת נְעוּרָֽי׃ (נד) וְתָ֘שֵׂ֤ם בַּסַּ֨ד ׀ רַגְלַ֗י וְתִשְׁמ֥וֹר כָּל־ אָרְחוֹתָ֑י עַל־ שָׁרְשֵׁ֥י רַ֝גְלַ֗י תִּתְחַקֶּֽה׃ (נה) וְ֭הוּא כְּרָקָ֣ב יִבְלֶ֑ה כְּ֝בֶ֗גֶד אֲכָ֣לוֹ עָֽשׁ׃ (נו) אָ֭דָם יְל֣וּד אִשָּׁ֑ה קְצַ֥ר יָ֝מִ֗ים וּֽשְׂבַֽע־ רֹֽגֶז׃ (נז) כְּצִ֣יץ יָ֭צָא וַיִּמָּ֑ל וַיִּבְרַ֥ח כַּ֝צֵּ֗ל וְלֹ֣א יַעֲמֽוֹד׃ (נח) אַף־ עַל־ זֶ֭ה פָּקַ֣חְתָּ עֵינֶ֑ךָ וְאֹ֘תִ֤י תָבִ֖יא בְמִשְׁפָּ֣ט עִמָּֽךְ׃ (נט) מִֽי־ יִתֵּ֣ן טָ֭הוֹר מִטָּמֵ֗א לֹ֣א אֶחָֽד׃ (ס) אִ֥ם חֲרוּצִ֨ים ׀ יָמָ֗יו מִֽסְפַּר־ חֳדָשָׁ֥יו אִתָּ֑ךְ (חקו) חֻקָּ֥יו עָ֝שִׂ֗יתָ וְלֹ֣א יַעֲבֽוֹר׃ (סא) שְׁעֵ֣ה מֵעָלָ֣יו וְיֶחְדָּ֑ל עַד־ יִ֝רְצֶ֗ה כְּשָׂכִ֥יר יוֹמֽוֹ׃ (סב) כִּ֤י יֵ֥שׁ לָעֵ֗ץ תִּ֫קְוָ֥ה אִֽם־ יִ֭כָּרֵת וְע֣וֹד יַחֲלִ֑יף וְ֝יֹֽנַקְתּ֗וֹ לֹ֣א תֶחְדָּֽל׃ (סג) אִם־ יַזְקִ֣ין בָּאָ֣רֶץ שָׁרְשׁ֑וֹ וּ֝בֶעָפָ֗ר יָמ֥וּת גִּזְעֽוֹ׃ (סד) מֵרֵ֣יחַ מַ֣יִם יַפְרִ֑חַ וְעָשָׂ֖ה קָצִ֣יר כְּמוֹ־ נָֽטַע׃ (סה) וְגֶ֣בֶר יָ֭מוּת וַֽיֶּחֱלָ֑שׁ וַיִּגְוַ֖ע אָדָ֣ם וְאַיּֽוֹ׃ (סו) אָֽזְלוּ־ מַ֭יִם מִנִּי־ יָ֑ם וְ֝נָהָ֗ר יֶחֱרַ֥ב וְיָבֵֽשׁ׃ (סז) וְאִ֥ישׁ שָׁכַ֗ב וְֽלֹא־ יָ֫ק֥וּם עַד־ בִּלְתִּ֣י שָׁ֭מַיִם לֹ֣א יָקִ֑יצוּ וְלֹֽא־ יֵ֝עֹ֗רוּ מִשְּׁנָתָֽם׃ (סח) מִ֤י יִתֵּ֨ן ׀ בִּשְׁא֬וֹל תַּצְפִּנֵ֗נִי תַּ֭סְתִּירֵנִי עַד־ שׁ֣וּב אַפֶּ֑ךָ תָּ֤שִׁ֥ית לִ֖י חֹ֣ק וְתִזְכְּרֵֽנִי׃ (סט) אִם־ יָמ֥וּת גֶּ֗בֶר הֲיִ֫חְיֶ֥ה כָּל־ יְמֵ֣י צְבָאִ֣י אֲיַחֵ֑ל עַד־ בּ֝֗וֹא חֲלִיפָתִֽי׃ (ע) תִּ֭קְרָא וְאָנֹכִ֣י אֶֽעֱנֶ֑ךָּ לְֽמַעֲשֵׂ֖ה יָדֶ֣יךָ תִכְסֹֽף׃ (עא) כִּֽי־ עַ֭תָּה צְעָדַ֣י תִּסְפּ֑וֹר לֹֽא־ מ֖וֹרָשֵׁ֣י לְבָבִֽי׃ (ד) לַ֭יְלָה לְי֣וֹם יָשִׂ֑ימוּ א֝֗וֹר קָר֥וֹב מִפְּנֵי־ חֹֽשֶׁךְ׃ (ה) אִם־ אֲ֭קַוֶּה שְׁא֣וֹל בֵּיתִ֑י בַּ֝חֹ֗שֶׁךְ רִפַּ֥דְתִּי יְצוּעָֽי׃ (ו) לַשַּׁ֣חַת קָ֭רָאתִי אָ֣בִי אָ֑תָּה אִמִּ֥י וַ֝אֲחֹתִ֗י לָֽרִמָּֽה׃ (ז) וְ֭אַיֵּה אֵפ֣וֹ תִקְוָתִ֑י וְ֝תִקְוָתִ֗י מִ֣י יְשׁוּרֶֽנָּה׃ (ח) בַּדֵּ֣י שְׁאֹ֣ל תֵּרַ֑דְנָה אִם־ יַ֖חַד עַל־ עָפָ֣ר נָֽחַת׃ סוף פרק יז (ט) וַ֭יַּעַן בִּלְדַּ֥ד הַשֻּׁחִ֗י וַיֹּאמַֽר׃ (י) עַד־ אָ֤נָה ׀ תְּשִׂימ֣וּן קִנְצֵ֣י לְמִלִּ֑ין תָּ֝בִ֗ינוּ וְאַחַ֥ר נְדַבֵּֽר׃ (יא) מַ֭דּוּעַ נֶחְשַׁ֣בְנוּ כַבְּהֵמָ֑ה נִ֝טְמִ֗ינוּ בְּעֵינֵיכֶֽם׃ (יב) טֹֽרֵ֥ף נַפְשׁ֗וֹ בְּאַ֫פּ֥וֹ הַ֭לְמַעַנְךָ תֵּעָ֣זַב אָ֑רֶץ וְיֶעְתַּק־ צ֝֗וּר מִמְּקֹמֽוֹ׃ (יג) גַּ֤ם א֣וֹר רְשָׁעִ֣ים יִדְעָ֑ךְ וְלֹֽא־ יִ֝גַּ֗הּ שְׁבִ֣יב אִשּֽׁוֹ׃ (יד) א֭וֹר חָשַׁ֣ךְ בְּאָהֳל֑וֹ וְ֝נֵר֗וֹ עָלָ֥יו יִדְעָֽךְ׃ (טו) יֵֽ֭צְרוּ צַעֲדֵ֣י אוֹנ֑וֹ וְֽתַשְׁלִיכֵ֥הוּ עֲצָתֽוֹ׃ (טז) כִּֽי־ שֻׁלַּ֣ח בְּרֶ֣שֶׁת בְּרַגְלָ֑יו וְעַל־ שְׂ֝בָכָ֗ה יִתְהַלָּֽךְ׃ (יז) יֹאחֵ֣ז בְּעָקֵ֣ב פָּ֑ח יַחֲזֵ֖ק עָלָ֣יו צַמִּֽים׃ (יח) טָמ֣וּן בָּאָ֣רֶץ חַבְל֑וֹ וּ֝מַלְכֻּדְתּ֗וֹ עֲלֵ֣י נָתִֽיב׃ (יט) סָ֭בִיב בִּֽעֲתֻ֣הוּ בַלָּה֑וֹת וֶהֱפִיצֻ֥הוּ לְרַגְלָֽיו׃ (כ) יְהִי־ רָעֵ֥ב אֹנ֑וֹ וְ֝אֵ֗יד נָכ֥וֹן לְצַלְעֽוֹ׃ (כא) יֹ֭אכַל בַּדֵּ֣י עוֹר֑וֹ יֹאכַ֥ל בַּ֝דָּ֗יו בְּכ֣וֹר מָֽוֶת׃ (כב) יִנָּתֵ֣ק מֵ֭אָהֳלוֹ מִבְטַח֑וֹ וְ֝תַצְעִדֵ֗הוּ לְמֶ֣לֶךְ בַּלָּהֽוֹת׃ (כג) תִּשְׁכּ֣וֹן בְּ֭אָהֳלוֹ מִבְּלִי־ ל֑וֹ יְזֹרֶ֖ה עַל־ נָוֵ֣הוּ גָפְרִֽית׃ (כד) מִ֭תַּחַת שָֽׁרָשָׁ֣יו יִבָ֑שׁוּ וּ֝מִמַּ֗עַל יִמַּ֥ל קְצִירֽוֹ׃ (כה) זִֽכְרוֹ־ אָ֭בַד מִנִּי־ אָ֑רֶץ וְלֹא־ שֵׁ֥ם ל֝֗וֹ עַל־ פְּנֵי־ חֽוּץ׃ (כו) יֶ֭הְדְּפֻהוּ מֵא֣וֹר אֶל־ חֹ֑שֶׁךְ וּֽמִתֵּבֵ֥ל יְנִדֻּֽהוּ׃ (כז) לֹ֘א נִ֤ין ל֣וֹ וְלֹא־ נֶ֣כֶד בְּעַמּ֑וֹ וְאֵ֥ין שָׂ֝רִ֗יד בִּמְגוּרָֽיו׃ (כח) עַל־ י֭וֹמוֹ נָשַׁ֣מּוּ אַחֲרֹנִ֑ים וְ֝קַדְמֹנִ֗ים אָ֣חֲזוּ שָֽׂעַר׃ (כט) אַךְ־ אֵ֭לֶּה מִשְׁכְּנ֣וֹת עַוָּ֑ל וְ֝זֶ֗ה מְק֣וֹם לֹא־ יָדַֽע־ אֵֽל׃ (ל) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (לא) עַד־ אָ֭נָה תּוֹגְי֣וּן נַפְשִׁ֑י וּֽתְדַכְּאוּנַ֥נִי בְמִלִּֽים׃ (לב) זֶ֤ה עֶ֣שֶׂר פְּ֭עָמִים תַּכְלִימ֑וּנִי לֹֽא־ תֵ֝בֹ֗שׁוּ תַּהְכְּרוּ־ לִֽי׃ (לג) וְאַף־ אָמְנָ֥ם שָׁגִ֑יתִי אִ֝תִּ֗י תָּלִ֥ין מְשׁוּגָתִֽי׃ (לד) אִם־ אָ֭מְנָם עָלַ֣י תַּגְדִּ֑ילוּ וְתוֹכִ֥יחוּ עָ֝לַ֗י חֶרְפָּתִּֽי׃ (לה) דְּֽעוּ־ אֵ֭פוֹ כִּי־ אֱל֣וֹהַּ עִוְּתָ֑נִי וּ֝מְצוּד֗וֹ עָלַ֥י הִקִּֽיף׃ (לו) הֵ֤ן אֶצְעַ֣ק חָ֭מָס וְלֹ֣א אֵעָנֶ֑ה אֲ֝שַׁוַּ֗ע וְאֵ֣ין מִשְׁפָּֽט׃ (לז) אָרְחִ֣י גָ֭דַר וְלֹ֣א אֶעֱב֑וֹר וְעַ֥ל נְ֝תִיבוֹתַ֗י חֹ֣שֶׁךְ יָשִֽׂים׃ (לח) כְּ֭בוֹדִי מֵעָלַ֣י הִפְשִׁ֑יט וַ֝יָּ֗סַר עֲטֶ֣רֶת רֹאשִֽׁי׃ (לט) יִתְּצֵ֣נִי סָ֭בִיב וָאֵלַ֑ךְ וַיַּסַּ֥ע כָּ֝עֵ֗ץ תִּקְוָתִֽי׃ (מ) וַיַּ֣חַר עָלַ֣י אַפּ֑וֹ וַיַּחְשְׁבֵ֖נִי ל֣וֹ כְצָרָֽיו׃ (מא) יַ֤חַד ׀ יָ֘בֹ֤אוּ גְדוּדָ֗יו וַיָּסֹ֣לּוּ עָלַ֣י דַּרְכָּ֑ם וַיַּחֲנ֖וּ סָבִ֣יב לְאָהֳלִֽי׃ (מב) אַ֭חַי מֵעָלַ֣י הִרְחִ֑יק וְ֝יֹדְעַ֗י אַךְ־ זָ֥רוּ מִמֶּֽנִּי׃ (מג) חָדְל֥וּ קְרוֹבָ֑י וּֽמְיֻדָּעַ֥י שְׁכֵחֽוּנִי׃ (מד) גָּ֘רֵ֤י בֵיתִ֣י וְ֭אַמְהֹתַי לְזָ֣ר תַּחְשְׁבֻ֑נִי נָ֝כְרִ֗י הָיִ֥יתִי בְעֵינֵיהֶֽם׃ (מה) לְעַבְדִּ֣י קָ֭רָאתִי וְלֹ֣א יַעֲנֶ֑ה בְּמוֹ־ פִ֝֗י אֶתְחַנֶּן־ לֽוֹ׃ (מו) ר֭וּחִֽי זָ֣רָה לְאִשְׁתִּ֑י וְ֝חַנֹּתִ֗י לִבְנֵ֥י בִטְנִֽי׃ (מז) גַּם־ עֲ֭וִילִים מָ֣אֲסוּ בִ֑י אָ֝ק֗וּמָה וַיְדַבְּרוּ־ בִֽי׃ (מח) תִּֽ֭עֲבוּנִי כָּל־ מְתֵ֣י סוֹדִ֑י וְזֶֽה־ אָ֝הַ֗בְתִּי נֶהְפְּכוּ־ בִֽי׃ (מט) בְּעוֹרִ֣י וּ֭בִבְשָׂרִי דָּבְקָ֣ה עַצְמִ֑י וָ֝אֶתְמַלְּטָ֗ה בְּע֣וֹר שִׁנָּֽי׃ (נ) חָנֻּ֬נִי חָנֻּ֣נִי אַתֶּ֣ם רֵעָ֑י כִּ֥י יַד־ אֱ֝ל֗וֹהַּ נָ֣גְעָה בִּֽי׃ (נא) לָ֭מָּה תִּרְדְּפֻ֣נִי כְמוֹ־ אֵ֑ל וּ֝מִבְּשָׂרִ֗י לֹ֣א תִשְׂבָּֽעוּ׃ (נב) מִֽי־ יִתֵּ֣ן אֵ֭פוֹ וְיִכָּתְב֣וּן מִלָּ֑י מִֽי־ יִתֵּ֖ן בַּסֵּ֣פֶר וְיֻחָֽקוּ׃ (נג) בְּעֵט־ בַּרְזֶ֥ל וְעֹפָ֑רֶת לָ֝עַ֗ד בַּצּ֥וּר יֵחָצְבֽוּן׃ (נד) וַאֲנִ֣י יָ֭דַעְתִּי גֹּ֣אֲלִי חָ֑י וְ֝אַחֲר֗וֹן עַל־ עָפָ֥ר יָקֽוּם׃ (נה) וְאַחַ֣ר ע֭וֹרִֽי נִקְּפוּ־ זֹ֑את וּ֝מִבְּשָׂרִ֗י אֶֽחֱזֶ֥ה אֱלֽוֹהַּ׃ (נו) אֲשֶׁ֤ר אֲנִ֨י ׀ אֶֽחֱזֶה־ לִּ֗י וְעֵינַ֣י רָא֣וּ וְלֹא־ זָ֑ר כָּל֖וּ כִלְיֹתַ֣י בְּחֵקִֽי׃ (נז) כִּ֣י תֹ֭אמְרוּ מַה־ נִּרְדָּף־ ל֑וֹ וְשֹׁ֥רֶשׁ דָּ֝בָ֗ר נִמְצָא־ בִֽי׃ (נח) גּ֤וּרוּ לָכֶ֨ם ׀ מִפְּנֵי־ חֶ֗רֶב כִּֽי־ חֵ֭מָה עֲוֺנ֣וֹת חָ֑רֶב לְמַ֖עַן תֵּדְע֣וּן (שדין) שַׁדּֽוּן׃ (נט) וַ֭יַּעַן צֹפַ֥ר הַנַּֽעֲמָתִ֗י וַיֹּאמַֽר׃ (ס) לָ֭כֵן שְׂעִפַּ֣י יְשִׁיב֑וּנִי וּ֝בַעֲב֗וּר ח֣וּשִׁי בִֽי׃ (סא) מוּסַ֣ר כְּלִמָּתִ֣י אֶשְׁמָ֑ע וְ֝ר֗וּחַ מִֽבִּינָתִ֥י יַעֲנֵֽנִי׃ (סב) הֲזֹ֣את יָ֭דַעְתָּ מִנִּי־ עַ֑ד מִנִּ֤י שִׂ֖ים אָדָ֣ם עֲלֵי־ אָֽרֶץ׃ (סג) כִּ֤י רִנְנַ֣ת רְ֭שָׁעִים מִקָּר֑וֹב וְשִׂמְחַ֖ת חָנֵ֣ף עֲדֵי־ רָֽגַע׃ (סד) אִם־ יַעֲלֶ֣ה לַשָּׁמַ֣יִם שִׂיא֑וֹ וְ֝רֹאשׁ֗וֹ לָעָ֥ב יַגִּֽיעַ׃ (סה) כְּֽ֭גֶלֲלוֹ לָנֶ֣צַח יֹאבֵ֑ד רֹ֝אָ֗יו יֹאמְר֥וּ אַיּֽוֹ׃ (סו) כַּחֲל֣וֹם יָ֭עוּף וְלֹ֣א יִמְצָא֑וּהוּ וְ֝יֻדַּ֗ד כְּחֶזְי֥וֹן לָֽיְלָה׃ (סז) עַ֣יִן שְׁ֭זָפַתּוּ וְלֹ֣א תוֹסִ֑יף וְלֹא־ ע֝֗וֹד תְּשׁוּרֶ֥נּוּ מְקוֹמֽוֹ׃ (סח) בָּ֭נָיו יְרַצּ֣וּ דַלִּ֑ים וְ֝יָדָ֗יו תָּשֵׁ֥בְנָה אוֹנֽוֹ׃ (סט) עַ֭צְמוֹתָיו מָלְא֣וּ (עלומו) עֲלוּמָ֑יו וְ֝עִמּ֗וֹ עַל־ עָפָ֥ר תִּשְׁכָּֽב׃ (ע) אִם־ תַּמְתִּ֣יק בְּפִ֣יו רָעָ֑ה יַ֝כְחִידֶ֗נָּה תַּ֣חַת לְשׁוֹנֽוֹ׃ (עא) יַחְמֹ֣ל עָ֭לֶיהָ וְלֹ֣א יַֽעַזְבֶ֑נָּה וְ֝יִמְנָעֶ֗נָּה רִ֭צַּץ עָזַ֣ב דַּלִּ֑ים בַּ֥יִת גָּ֝זַ֗ל וְלֹ֣א יִבֶנֵֽהוּ׃ (עח) כִּ֤י ׀ לֹא־ יָדַ֣ע שָׁלֵ֣ו בְּבִטְנ֑וֹ בַּ֝חֲמוּד֗וֹ לֹ֣א יְמַלֵּֽט׃ (עט) אֵין־ שָׂרִ֥יד לְאָכְל֑וֹ עַל־ כֵּ֝֗ן לֹא־ יָחִ֥יל טוּבֽוֹ׃ (פ) בִּמְלֹ֣אות שִׂ֭פְקוֹ יֵ֣צֶר ל֑וֹ כָּל־ יַ֖ד עָמֵ֣ל תְּבוֹאֶֽנּוּ׃ (פא) יְהִ֤י ׀ לְמַלֵּ֬א בִטְנ֗וֹ יְֽשַׁלַּח־ בּ֭וֹ חֲר֣וֹן אַפּ֑וֹ וְיַמְטֵ֥ר עָ֝לֵ֗ימוֹ בִּלְחוּמֽוֹ׃ (פב) יִ֭בְרַח מִנֵּ֣שֶׁק בַּרְזֶ֑ל תַּ֝חְלְפֵ֗הוּ קֶ֣שֶׁת נְחוּשָֽׁה׃ (פג) שָׁלַף֮ וַיֵּצֵ֪א מִגֵּ֫וָ֥ה וּ֭בָרָק מִֽמְּרֹרָת֥וֹ יַהֲלֹ֗ךְ עָלָ֥יו אֵמִֽים׃ (פד) כָּל־ חֹשֶׁךְ֮ טָמ֪וּן לִצְפּ֫וּנָ֥יו תְּ֭אָכְלֵהוּ אֵ֣שׁ לֹֽא־ נֻפָּ֑ח יֵ֖רַע שָׂרִ֣יד בְּאָהֳלֽוֹ׃ (פה) יְגַלּ֣וּ שָׁמַ֣יִם עֲוֺנ֑וֹ וְ֝אֶ֗רֶץ מִתְקוֹמָ֘מָ֥ה לֽוֹ׃ (פו) יִ֭גֶל יְב֣וּל בֵּית֑וֹ נִ֝גָּר֗וֹת בְּי֣וֹם אַפּֽוֹ׃ (פז) זֶ֤ה ׀ חֵֽלֶק־ אָדָ֣ם רָ֭שָׁע מֵאֱלֹהִ֑ים וְנַחֲלַ֖ת אִמְר֣וֹ מֵאֵֽל׃ (פח) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (פט) שִׁמְע֣וּ שָׁ֭מוֹעַ מִלָּתִ֑י וּתְהִי־ זֹ֝֗את תַּנְח֥וּמֹֽתֵיכֶֽם׃ (צ) שָׂ֭אוּנִי וְאָנֹכִ֣י אֲדַבֵּ֑ר וְאַחַ֖ר דַּבְּרִ֣י תַלְעִֽיג׃ (צא) הֶ֭אָנֹכִי לְאָדָ֣ם שִׂיחִ֑י וְאִם־ מַ֝דּ֗וּעַ לֹא־ תִקְצַ֥ר רוּחִֽי׃ (צב) פְּנוּ־ אֵלַ֥י וְהָשַׁ֑מּוּ וְשִׂ֖ימוּ יָ֣ד עַל־ פֶּֽה׃ (צג) וְאִם־ זָכַ֥רְתִּי וְנִבְהָ֑לְתִּי וְאָחַ֥ז בְּ֝שָׂרִ֗י פַּלָּצֽוּת׃ (צד) מַדּ֣וּעַ רְשָׁעִ֣ים יִחְי֑וּ עָ֝תְק֗וּ גַּם־ גָּ֥בְרוּ חָֽיִל׃ (צה) זַרְעָ֤ם נָכ֣וֹן לִפְנֵיהֶ֣ם עִמָּ֑ם וְ֝צֶאֱצָאֵיהֶ֗ם לְעֵינֵיהֶֽם׃ (צו) בָּתֵּיהֶ֣ם שָׁל֣וֹם מִפָּ֑חַד וְלֹ֤א שֵׁ֖בֶט אֱל֣וֹהַּ עֲלֵיהֶֽם׃ (צז) שׁוֹר֣וֹ עִ֭בַּר וְלֹ֣א יַגְעִ֑ל תְּפַלֵּ֥ט פָּ֝רָת֗וֹ וְלֹ֣א תְשַׁכֵּֽל׃ (צח) יְשַׁלְּח֣וּ כַ֭צֹּאן עֲוִילֵיהֶ֑ם וְ֝יַלְדֵיהֶ֗ם יְרַקֵּדֽוּן׃ (צט) יִ֭שְׂאוּ כְּתֹ֣ף וְכִנּ֑וֹר וְ֝יִשְׂמְח֗וּ לְק֣וֹל עוּגָֽב׃ (ק) (יבלו) יְכַלּ֣וּ בַטּ֣וֹב יְמֵיהֶ֑ם וּ֝בְרֶ֗גַע שְׁא֣וֹל יֵחָֽתּוּ׃ (קא) וַיֹּאמְר֣וּ לָ֭אֵל ס֣וּר מִמֶּ֑נּוּ וְדַ֥עַת דְּ֝רָכֶ֗יךָ לֹ֣א חָפָֽצְנוּ׃ (קב) מַה־ שַׁדַּ֥י כִּֽי־ נַֽעַבְדֶ֑נּוּ וּמַה־ נּ֝וֹעִ֗יל כִּ֣י נִפְגַּע־ בּֽוֹ׃ (קג) הֵ֤ן לֹ֣א בְיָדָ֣ם טוּבָ֑ם עֲצַ֥ת רְ֝שָׁעִ֗ים רָ֣חֲקָה מֶֽנִּי׃ (קד) כַּמָּ֤ה ׀ נֵר־ רְשָׁ֘עִ֤ים יִדְעָ֗ךְ וְיָבֹ֣א עָלֵ֣ימוֹ אֵידָ֑ם חֲ֝בָלִ֗ים יְחַלֵּ֥ק בְּאַפּֽוֹ׃ (קה) יִהְי֗וּ כְּתֶ֥בֶן לִפְנֵי־ ר֑וּחַ וּ֝כְמֹ֗ץ גְּנָבַ֥תּוּ סוּפָֽה׃ (קו) אֱל֗וֹהַּ יִצְפֹּן־ לְבָנָ֥יו אוֹנ֑וֹ יְשַׁלֵּ֖ם אֵלָ֣יו וְיֵדָֽע׃ (קז) יִרְא֣וּ (עינו) עֵינָ֣יו כִּיד֑וֹ וּמֵחֲמַ֖ת שַׁדַּ֣י יִשְׁתֶּֽה׃ (קח) כִּ֤י מַה־ חֶפְצ֣וֹ בְּבֵית֣וֹ אַחֲרָ֑יו וּמִסְפַּ֖ר חֳדָשָׁ֣יו חֻצָּֽצוּ׃ (קט) הַלְאֵ֥ל יְלַמֶּד־ דָּ֑עַת וְ֝ה֗וּא רָמִ֥ים יִשְׁפּֽוֹט׃ (קי) זֶ֗ה יָ֭מוּת בְּעֶ֣צֶם תֻּמּ֑וֹ כֻּ֝לּ֗וֹ שַׁלְאֲנַ֥ן וְשָׁלֵֽיו׃ (קיא) עֲ֭טִינָיו מָלְא֣וּ חָלָ֑ב וּמֹ֖חַ עַצְמוֹתָ֣יו יְשֻׁקֶּֽה׃ (קיב) וְזֶ֗ה יָ֭מוּת בְּנֶ֣פֶשׁ מָרָ֑ה וְלֹֽא־ אָ֝כַ֗ל בַּטּוֹבָֽה׃ (קיג) יַ֭חַד עַל־ עָפָ֣ר יִשְׁכָּ֑בוּ וְ֝רִמָּ֗ה תְּכַסֶּ֥ה עֲלֵיהֶֽם׃ (קיד) הֵ֣ן יָ֭דַעְתִּי מַחְשְׁבֽוֹתֵיכֶ֑ם וּ֝מְזִמּ֗וֹת עָלַ֥י תַּחְמֹֽסוּ׃ (קטו) כִּ֤י תֹֽאמְר֗וּ אַיֵּ֥ה בֵית־ נָדִ֑יב וְ֝אַיֵּ֗ה אֹ֤הֶל ׀ מִשְׁכְּנ֬וֹת רְשָׁעִֽים׃ (קטז) הֲלֹ֣א שְׁ֭אֶלְתֶּם ע֣וֹבְרֵי דָ֑רֶךְ וְ֝אֹתֹתָ֗ם לֹ֣א תְנַכֵּֽרוּ׃ (קיז) כִּ֤י לְי֣וֹם אֵ֭יד יֵחָ֣שֶׂךְ רָ֑ע לְי֖וֹם עֲבָר֣וֹת יוּבָֽלוּ׃ (קיח) מִֽי־ יַגִּ֣יד עַל־ פָּנָ֣יו דַּרְכּ֑וֹ וְהֽוּא־ עָ֝שָׂ֗ה מִ֣י יְשַׁלֶּם־ לֽוֹ׃ (קיט) וְ֭הוּא לִקְבָר֣וֹת יוּבָ֑ל וְֽעַל־ גָּדִ֥ישׁ יִשְׁקֽוֹד׃ (קכ) מָֽתְקוּ־ ל֗וֹ רִגְבֵ֫י נָ֥חַל וְ֭אַחֲרָיו כָּל־ אָדָ֣ם יִמְשׁ֑וֹךְ וּ֝לְפָנָ֗יו אֵ֣ין מִסְפָּֽר׃ (קכא) וְ֭אֵיךְ תְּנַחֲמ֣וּנִי הָ֑בֶל וּ֝תְשֽׁוּבֹתֵיכֶ֗ם נִשְׁאַר־ מָֽעַל׃ (קכב) וַ֭יַּעַן אֱלִיפַ֥ז הַֽתֵּמָנִ֗י וַיֹּאמַֽר׃ (קכג) הַלְאֵ֥ל יִסְכָּן־ גָּ֑בֶר כִּֽי־ יִסְכֹּ֖ן עָלֵ֣ימוֹ מַשְׂכִּֽיל׃ (קכד) הַחֵ֣פֶץ לְ֭שַׁדַּי כִּ֣י תִצְדָּ֑ק וְאִם־ בֶּ֝֗צַע כִּֽי־ תַתֵּ֥ם דְּרָכֶֽיךָ׃ (קכה) הֲֽ֭מִיִּרְאָ֣תְךָ יֹכִיחֶ֑ךָ יָב֥וֹא עִ֝מְּךָ֗ בַּמִּשְׁפָּֽט׃ (קכו) הֲלֹ֣א רָעָֽתְךָ֣ רַבָּ֑ה וְאֵֽין־ קֵ֝֗ץ לַעֲוֺנֹתֶֽיךָ׃ (קכז) כִּֽי־ תַחְבֹּ֣ל אַחֶ֣יךָ חִנָּ֑ם וּבִגְדֵ֖י עֲרוּמִּ֣ים תַּפְשִֽׁיט׃ (קכח) לֹא־ מַ֭יִם עָיֵ֣ף תַּשְׁקֶ֑ה וּ֝מֵרָעֵ֗ב תִּֽמְנַֽע־ לָֽחֶם׃ (קכט) וְאִ֣ישׁ זְ֭רוֹעַ ל֣וֹ הָאָ֑רֶץ וּנְשׂ֥וּא פָ֝נִ֗ים יֵ֣שֶׁב בָּֽהּ׃ (קל) אַ֭לְמָנוֹת שִׁלַּ֣חְתָּ רֵיקָ֑ם וּזְרֹע֖וֹת יְתֹמִ֣ים יְדֻכָּֽא׃ (קלא) עַל־ כֵּ֭ן סְבִיבוֹתֶ֣יךָ פַחִ֑ים וִֽ֝יבַהֶלְךָ פַּ֣חַד פִּתְאֹֽם׃ (קלב) אוֹ־ חֹ֥שֶׁךְ לֹֽא־ תִרְאֶ֑ה וְֽשִׁפְעַת־ מַ֥יִם תְּכַסֶּֽךָּ׃ (קלג) הֲ‍ֽלֹא־ אֱ֭לוֹהַּ גֹּ֣בַהּ שָׁמָ֑יִם וּרְאֵ֤ה רֹ֖אשׁ כּוֹכָבִ֣ים כִּי־ רָֽמּוּ׃ (קלד) וְֽ֭אָמַרְתָּ מַה־ יָּ֣דַֽע אֵ֑ל הַבְעַ֖ד עֲרָפֶ֣ל יִשְׁפּֽוֹט׃ (קלה) עָבִ֣ים סֵֽתֶר־ ל֭וֹ וְלֹ֣א יִרְאֶ֑ה וְח֥וּג שָׁ֝מַ֗יִם יִתְהַלָּֽךְ׃ (קלו) הַאֹ֣רַח עוֹלָ֣ם תִּשְׁמֹ֑ר אֲשֶׁ֖ר דָּרְכ֣וּ מְתֵי־ אָֽוֶן׃ (קלז) אֲשֶֽׁר־ קֻמְּט֥וּ וְלֹא־ עֵ֑ת נָ֝הָ֗ר יוּצַ֥ק יְסוֹדָֽם׃ (קלח) הָאֹמְרִ֣ים לָ֭אֵל ס֣וּר מִמֶּ֑נּוּ וּמַה־ יִּפְעַ֖ל שַׁדַּ֣י לָֽמוֹ׃ (קלט) וְה֤וּא מִלֵּ֣א בָתֵּיהֶ֣ם ט֑וֹב וַעֲצַ֥ת רְ֝שָׁעִ֗ים רָ֣חֲקָה מֶֽנִּי׃ (קמ) יִרְא֣וּ צַדִּיקִ֣ים וְיִשְׂמָ֑חוּ וְ֝נָקִ֗י יִלְעַג־ לָֽמוֹ׃ (קמא) אִם־ לֹ֣א נִכְחַ֣ד קִימָ֑נוּ וְ֝יִתְרָ֗ם אָ֣כְלָה אֵֽשׁ׃ (קמב) הַסְכֶּן־ נָ֣א עִמּ֑וֹ וּשְׁלם בָּ֝הֶ֗ם תְּֽבוֹאַתְךָ֥ טוֹבָֽה׃ (קמג) קַח־ נָ֣א מִפִּ֣יו תּוֹרָ֑ה וְשִׂ֥ים אֲ֝מָרָ֗יו בִּלְבָבֶֽךָ׃ (קמד) אִם־ תָּשׁ֣וּב עַד־ שַׁ֭דַּי תִּבָּנֶ֑ה תַּרְחִ֥יק עַ֝וְלָ֗ה מֵאָהֳלֶֽךָ׃ (קמה) וְשִׁית־ עַל־ עָפָ֥ר בָּ֑צֶר וּבְצ֖וּר נְחָלִ֣ים אוֹפִֽיר׃ (קמו) וְהָיָ֣ה שַׁדַּ֣י בְּצָרֶ֑יךָ וְכֶ֖סֶף תּוֹעָפ֣וֹת לָֽךְ׃ (קמז) כִּי־ אָ֭ז עַל־ שַׁדַּ֣י תִּתְעַנָּ֑ג וְתִשָּׂ֖א אֶל־ אֱל֣וֹהַּ פָּנֶֽיךָ׃ (קמח) תַּעְתִּ֣יר אֵ֭לָיו וְיִשְׁמָעֶ֑ךָּ וּנְדָרֶ֥יךָ תְשַׁלֵּֽם׃ (קמט) וְֽתִגְזַר־ א֭וֹמֶר וְיָ֣קָם לָ֑ךְ וְעַל־ דְּ֝רָכֶ֗יךָ נָ֣גַֽהּ אֽוֹר׃ (קנ) כִּֽי־ הִ֭שְׁפִּילוּ וַתֹּ֣אמֶר גֵּוָ֑ה וְשַׁ֖ח עֵינַ֣יִם יוֹשִֽׁעַ׃
