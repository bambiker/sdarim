איוב פרק לג לפי החלוקה לסדרים

פרק לג בספר איוב (33 פסוקים) נמצא באיוב סדר ו (פסוקים קו–קלז) ובאיוב סדר ז (פסוקים א–א).

איוב סדר ו

(א) צֶ֣דֶק לָ֭בַשְׁתִּי וַיִּלְבָּשֵׁ֑נִי כִּֽמְעִ֥יל וְ֝צָנִ֗יף מִשְׁפָּטִֽי׃ (ב) עֵינַ֣יִם הָ֭יִיתִי לַֽעִוֵּ֑ר וְרַגְלַ֖יִם לַפִּסֵּ֣חַ אָֽנִי׃ (ג) אָ֣ב אָ֭נֹכִֽי לָֽאֶבְיוֹנִ֑ים וְרִ֖ב לֹא־ יָדַ֣עְתִּי אֶחְקְרֵֽהוּ׃ (ד) וָֽ֭אֲשַׁבְּרָה מְתַלְּע֣וֹת עַוָּ֑ל וּ֝מִשִּׁנָּ֗יו אַשְׁלִ֥יךְ טָֽרֶף׃ (ה) וָ֭אֹמַר עִם־ קִנִּ֣י אֶגְוָ֑ע וְ֝כַח֗וֹל אַרְבֶּ֥ה יָמִֽים׃ (ו) שָׁרְשִׁ֣י פָת֣וּחַ אֱלֵי־ מָ֑יִם וְ֝טַ֗ל יָלִ֥ין בִּקְצִירִֽי׃ (ז) כְּ֭בוֹדִי חָדָ֣שׁ עִמָּדִ֑י וְ֝קַשְׁתִּ֗י בְּיָדִ֥י תַחֲלִֽיף׃ (ח) לִֽי־ שָׁמְע֥וּ וְיִחֵ֑לּוּ וְ֝יִדְּמ֗וּ לְמ֣וֹ עֲצָתִֽי׃ (ט) אַחֲרֵ֣י דְ֭בָרִי לֹ֣א יִשְׁנ֑וּ וְ֝עָלֵ֗ימוֹ תִּטֹּ֥ף מִלָּתִֽי׃ (י) וְיִֽחֲל֣וּ כַמָּטָ֣ר לִ֑י וּ֝פִיהֶ֗ם פָּעֲר֥וּ לְמַלְקֽוֹשׁ׃ (יא) אֶשְׂחַ֣ק אֲ֭לֵהֶם לֹ֣א יַאֲמִ֑ינוּ וְא֥וֹר פָּ֝נַ֗י לֹ֣א יַפִּילֽוּן׃ (יב) אֶֽבֲחַ֣ר דַּרְכָּם֮ וְאֵשֵׁ֪ב רֹ֥אשׁ וְ֭אֶשְׁכּוֹן כְּמֶ֣לֶךְ בַּגְּד֑וּד כַּאֲשֶׁ֖ר אֲבֵלִ֣ים יְנַחֵֽם׃ (יג) וְעַתָּ֤ה ׀ שָֽׂחֲק֣וּ עָלַי֮ צְעִירִ֥ים מִמֶּ֗נִּי לְיָ֫מִ֥ים אֲשֶׁר־ מָאַ֥סְתִּי אֲבוֹתָ֑ם לָ֝שִׁ֗ית עִם־ כַּלְבֵ֥י צֹאנִֽי׃ (יד) גַּם־ כֹּ֣חַ יְ֭דֵיהֶם לָ֣מָּה לִּ֑י עָ֝לֵ֗ימוֹ אָ֣בַד כָּֽלַח׃ (טו) בְּחֶ֥סֶר וּבְכָפָ֗ן גַּ֫לְמ֥וּד הַֽעֹרְקִ֥ים צִיָּ֑ה אֶ֝֗מֶשׁ שׁוֹאָ֥ה וּמְשֹׁאָֽה׃ (טז) הַקֹּטְפִ֣ים מַלּ֣וּחַ עֲלֵי־ שִׂ֑יחַ וְשֹׁ֖רֶשׁ רְתָמִ֣ים לַחְמָֽם׃ (יז) מִן־ גֵּ֥ו יְגֹרָ֑שׁוּ יָרִ֥יעוּ עָ֝לֵ֗ימוֹ כַּגַּנָּֽב׃ (יח) בַּעֲר֣וּץ נְחָלִ֣ים לִשְׁכֹּ֑ן חֹרֵ֖י עָפָ֣ר וְכֵפִֽים׃ (יט) בֵּין־ שִׂיחִ֥ים יִנְהָ֑קוּ תַּ֖חַת חָר֣וּל יְסֻפָּֽחוּ׃ (כ) בְּֽנֵי־ נָ֭בָל גַּם־ בְּנֵ֣י בְלִי־ שֵׁ֑ם נִ֝כְּא֗וּ מִן־ הָאָֽרֶץ׃ (כא) וְ֭עַתָּה נְגִינָתָ֣ם הָיִ֑יתִי וָאֱהִ֖י לָהֶ֣ם לְמִלָּֽה׃ (כב) תִּֽ֭עֲבוּנִי רָ֣חֲקוּ מֶ֑נִּי וּ֝מִפָּנַ֗י לֹא־ חָ֥שְׂכוּ רֹֽק׃ (כג) כִּֽי־ (יתרו) יִתְרִ֣י פִ֭תַּח וַיְעַנֵּ֑נִי וְ֝רֶ֗סֶן מִפָּנַ֥י שִׁלֵּֽחוּ׃ (כד) עַל־ יָמִין֮ פִּרְחַ֪ח יָ֫ק֥וּמוּ רַגְלַ֥י שִׁלֵּ֑חוּ וַיָּסֹ֥לּוּ עָ֝לַ֗י אָרְח֥וֹת אֵידָֽם׃ (כה) נָתְס֗וּ נְֽתִיבָ֫תִ֥י לְהַוָּתִ֥י יֹעִ֑ילוּ לֹ֖א עֹזֵ֣ר לָֽמוֹ׃ (כו) כְּפֶ֣רֶץ רָחָ֣ב יֶאֱתָ֑יוּ תַּ֥חַת שֹׁ֝אָ֗ה הִתְגַּלְגָּֽלוּ׃ (כז) הָהְפַּ֥ךְ עָלַ֗י בַּלָּ֫ה֥וֹת תִּרְדֹּ֣ף כָּ֭רוּחַ נְדִבָתִ֑י וּ֝כְעָ֗ב עָבְרָ֥ה יְשֻׁעָתִֽי׃ (כח) וְעַתָּ֗ה עָ֭לַי תִּשְׁתַּפֵּ֣ךְ נַפְשִׁ֑י יֹ֭אחֲז֣וּנִי יְמֵי־ עֹֽנִי׃ (כט) לַ֗יְלָה עֲ֭צָמַי נִקַּ֣ר מֵעָלָ֑י וְ֝עֹרְקַ֗י לֹ֣א יִשְׁכָּבֽוּן׃ (ל) בְּרָב־ כֹּ֭חַ יִתְחַפֵּ֣שׂ לְבוּשִׁ֑י כְּפִ֖י כֻתָּנְתִּ֣י יַֽאַזְרֵֽנִי׃ (לא) הֹרָ֥נִי לַחֹ֑מֶר וָ֝אֶתְמַשֵּׁ֗ל כֶּעָפָ֥ר וָאֵֽפֶר׃ (לב) אֲשַׁוַּ֣ע אֵ֭לֶיךָ וְלֹ֣א תַעֲנֵ֑נִי עָ֝מַ֗דְתִּי וַתִּתְבֹּ֥נֶן בִּֽי׃ (לג) תֵּהָפֵ֣ךְ לְאַכְזָ֣ר לִ֑י בְּעֹ֖צֶם יָדְךָ֣ תִשְׂטְמֵֽנִי׃ (לד) תִּשָּׂאֵ֣נִי אֶל־ ר֭וּחַ תַּרְכִּיבֵ֑נִי וּ֝תְמֹגְגֵ֗נִי (תשוה) תּוּשִׁיָּֽה׃ (לה) כִּֽי־ יָ֭דַעְתִּי מָ֣וֶת תְּשִׁיבֵ֑נִי וּבֵ֖ית מוֹעֵ֣ד לְכָל־ חָֽי׃ (לו) אַ֣ךְ לֹא־ בְ֭עִי יִשְׁלַח־ יָ֑ד אִם־ בְּ֝פִיד֗וֹ לָהֶ֥ן שֽׁוּעַ׃ (לז) אִם־ לֹ֣א בָ֭כִיתִי לִקְשֵׁה־ י֑וֹם עָֽגְמָ֥ה נַ֝פְשִׁ֗י לָאֶבְיֽוֹן׃ (לח) כִּ֤י ט֣וֹב קִ֭וִּיתִי וַיָּ֣בֹא רָ֑ע וַֽאֲיַחֲלָ֥ה לְ֝א֗וֹר וַיָּ֥בֹא אֹֽפֶל׃ (לט) מֵעַ֖י רֻתְּח֥וּ וְלֹא־ דָ֗מּוּ קִדְּמֻ֥נִי יְמֵי־ עֹֽנִי׃ (מ) קֹדֵ֣ר הִ֭לַּכְתִּי בְּלֹ֣א חַמָּ֑ה קַ֖מְתִּי בַקָּהָ֣ל אֲשַׁוֵּֽעַ׃ (מא) אָ֭ח הָיִ֣יתִי לְתַנִּ֑ים וְ֝רֵ֗עַ לִבְנ֥וֹת יַעֲנָֽה׃ (מב) ע֭וֹרִי שָׁחַ֣ר מֵעָלָ֑י וְעַצְמִי־ חָ֝֗רָה מִנִּי־ חֹֽרֶב׃ (מג) וַיְהִ֣י לְ֭אֵבֶל כִּנֹּרִ֑י וְ֝עֻגָבִ֗י לְק֣וֹל בֹּכִֽים׃ (מד) בְּ֭רִית כָּרַ֣תִּי לְעֵינָ֑י וּמָ֥ה אֶ֝תְבּוֹנֵ֗ן עַל־ בְּתוּלָֽה׃ (מה) וּמֶ֤ה ׀ חֵ֣לֶק אֱל֣וֹהַּ מִמָּ֑עַל וְֽנַחֲלַ֥ת שַׁ֝דַּ֗י מִמְּרֹמִֽים׃ (מו) הֲלֹא־ אֵ֥יד לְעַוָּ֑ל וְ֝נֵ֗כֶר לְפֹ֣עֲלֵי אָֽוֶן׃ (מז) הֲלֹא־ ה֭וּא יִרְאֶ֣ה דְרָכָ֑י וְֽכָל־ צְעָדַ֥י יִסְפּֽוֹר׃ (מח) אִם־ הָלַ֥כְתִּי עִם־ שָׁ֑וְא וַתַּ֖חַשׁ עַל־ מִרְמָ֣ה רַגְלִֽי׃ (מט) יִשְׁקְלֵ֥נִי בְמֹאזְנֵי־ צֶ֑דֶק וְיֵדַ֥ע אֱ֝ל֗וֹהַּ תֻּמָּתִֽי׃ (נ) אִ֥ם תִּטֶּ֣ה אַשֻּׁרִי֮ מִנִּ֪י הַ֫דָּ֥רֶךְ וְאַחַ֣ר עֵ֭ינַי הָלַ֣ךְ לִבִּ֑י וּ֝בְכַפַּ֗י דָּ֣בַק מֻאֽוּם׃ (נא) אֶ֭זְרְעָה וְאַחֵ֣ר יֹאכֵ֑ל וְֽצֶאֱצָאַ֥י יְשֹׁרָֽשׁוּ׃ (נב) אִם־ נִפְתָּ֣ה לִ֭בִּי עַל־ אִשָּׁ֑ה וְעַל־ פֶּ֖תַח רֵעִ֣י אָרָֽבְתִּי׃ (נג) תִּטְחַ֣ן לְאַחֵ֣ר אִשְׁתִּ֑י וְ֝עָלֶ֗יהָ יִכְרְע֥וּן אֲחֵרִֽין׃ (נד) כִּי־ (הוא) הִ֥יא זִמָּ֑ה (והיא) וְ֝ה֗וּא עָוֺ֥ן פְּלִילִֽים׃ (נה) כִּ֤י אֵ֣שׁ הִ֭יא עַד־ אֲבַדּ֣וֹן תֹּאכֵ֑ל וּֽבְכָל־ תְּב֖וּאָתִ֣י תְשָׁרֵֽשׁ׃ (נו) אִם־ אֶמְאַ֗ס מִשְׁפַּ֣ט עַ֭בְדִּי וַאֲמָתִ֑י בְּ֝רִבָ֗ם עִמָּדִֽי׃ (נז) וּמָ֣ה אֶֽ֭עֱשֶׂה כִּֽי־ יָק֣וּם אֵ֑ל וְכִֽי־ יִ֝פְקֹ֗ד מָ֣ה אֲשִׁיבֶֽנּוּ׃ (נח) הֲ‍ֽ֝לֹא־ בַ֭בֶּטֶן עֹשֵׂ֣נִי עָשָׂ֑הוּ וַ֝יְכֻנֶ֗נּוּ בָּרֶ֥חֶם אֶחָֽד׃ (נט) אִם־ אֶ֭מְנַע מֵחֵ֣פֶץ דַּלִּ֑ים וְעֵינֵ֖י אַלְמָנָ֣ה אֲכַלֶּֽה׃ (ס) וְאֹכַ֣ל פִּתִּ֣י לְבַדִּ֑י וְלֹא־ אָכַ֖ל יָת֣וֹם מִמֶּֽנָּה׃ (סא) כִּ֣י מִ֭נְּעוּרַי גְּדֵלַ֣נִי כְאָ֑ב וּמִבֶּ֖טֶן אִמִּ֣י אַנְחֶֽנָּה׃ (סב) אִם־ אֶרְאֶ֣ה א֭וֹבֵד מִבְּלִ֣י לְב֑וּשׁ וְאֵ֥ין כְּ֝ס֗וּת לָאֶבְיֽוֹן׃ (סג) אִם־ לֹ֣א בֵרֲכ֣וּנִי (חלצו) חֲלָצָ֑יו וּמִגֵּ֥ז כְּ֝בָשַׂי יִתְחַמָּֽם׃ (סד) אִם־ הֲנִיפ֣וֹתִי עַל־ יָת֣וֹם יָדִ֑י כִּֽי־ אֶרְאֶ֥ה בַ֝שַּׁ֗עַר עֶזְרָתִֽי׃ (סה) כְּ֭תֵפִי מִשִּׁכְמָ֣ה תִפּ֑וֹל וְ֝אֶזְרֹעִ֗י מִקָּנָ֥ה תִשָּׁבֵֽר׃ (סו) כִּ֤י פַ֣חַד אֵ֭לַי אֵ֣יד אֵ֑ל וּ֝מִשְּׂאֵת֗וֹ לֹ֣א אוּכָֽל׃ (סז) אִם־ שַׂ֣מְתִּי זָהָ֣ב כִּסְלִ֑י וְ֝לַכֶּ֗תֶם אָמַ֥רְתִּי מִבְטַחִֽי׃ (סח) אִם־ אֶ֭שְׂמַח כִּי־ רַ֣ב חֵילִ֑י וְכִֽי־ כַ֝בִּ֗יר מָצְאָ֥ה יָדִֽי׃ (סט) אִם־ אֶרְאֶ֣ה א֖וֹר כִּ֣י יָהֵ֑ל וְ֝יָרֵ֗חַ יָקָ֥ר הֹלֵֽךְ׃ (ע) אֹ֭זֶן מִלִּ֣ין תִּבְחָ֑ן וְ֝חֵ֗ךְ יִטְעַ֥ם לֶאֱכֹֽל׃ (ה) מִשְׁפָּ֥ט נִבְחֲרָה־ לָּ֑נוּ נֵדְעָ֖ה בֵינֵ֣ינוּ מַה־ טּֽוֹב׃ (ו) כִּֽי־ אָ֭מַר אִיּ֣וֹב צָדַ֑קְתִּי וְ֝אֵ֗ל הֵסִ֥יר מִשְׁפָּטִֽי׃ (ז) עַל־ יָשִׂ֣ים אֵלָ֣יו לִבּ֑וֹ רוּח֥וֹ וְ֝נִשְׁמָת֗וֹ אֵלָ֥יו יֶאֱסֹֽף׃ (טז) יִגְוַ֣ע כָּל־ בָּשָׂ֣ר יָ֑חַד וְ֝אָדָ֗ם עַל־ עָפָ֥ר יָשֽׁוּב׃ (יז) וְאִם־ בִּ֥ינָה שִׁמְעָה־ זֹּ֑את הַ֝אֲזִ֗ינָה לְק֣וֹל מִלָּֽי׃ (יח) הַאַ֬ף שׂוֹנֵ֣א מִשְׁפָּ֣ט יַחֲב֑וֹשׁ וְאִם־ צַדִּ֖יק כַּבִּ֣יר תַּרְשִֽׁיעַ׃ (יט) הַאֲמֹ֣ר לְמֶ֣לֶךְ בְּלִיָּ֑עַל רָ֝שָׁ֗ע אֶל־ נְדִיבִֽים׃ (כ) אֲשֶׁ֤ר לֹֽא־ נָשָׂ֨א ׀ פְּנֵ֥י שָׂרִ֗ים בַּמִּשְׁפָּֽט׃ (כה) יָרֹ֣עַ כַּבִּירִ֣ים לֹא־ חֵ֑קֶר וַיַּעֲמֵ֖ד אֲחֵרִ֣ים תַּחְתָּֽם׃ (כו) לָכֵ֗ן יַ֭כִּיר מַעְבָּֽדֵיהֶ֑ם וְהָ֥פַךְ לַ֝֗יְלָה וְיִדַּכָּֽאוּ׃ (כז) תַּֽחַת־ רְשָׁעִ֥ים סְפָקָ֗ם וּמִ֥י יַרְשִׁ֗עַ וְיַסְתֵּ֣ר פָּ֭נִים וּמִ֣י יְשׁוּרֶ֑נּוּ וְעַל־ גּ֖וֹי וְעַל־ אָדָ֣ם יָֽחַד׃ (לא) מִ֭מְּלֹךְ אָדָ֥ם חָנֵ֗ף מִמֹּ֥קְשֵׁי עָֽם׃ (לב) כִּֽי־ אֶל־ אֵ֭ל הֶאָמַ֥ר נָשָׂ֗אתִי לֹ֣א אֶחְבֹּֽל׃ (לג) בִּלְעֲדֵ֣י אֶ֭חֱזֶה אַתָּ֣ה הֹרֵ֑נִי אִֽם־ עָ֥וֶל פָּ֝עַ֗לְתִּי לֹ֣א אֹסִֽיף׃ (לד) הַֽמֵעִמְּךָ֬ יְשַׁלְמֶ֨נָּה כִּֽי־ מָאַ֗סְתָּ כִּי־ אַתָּ֣ה תִבְחַ֣ר וְלֹא־ אָ֑נִי וּֽמַה־ יָדַ֥עְתָּ דַבֵּֽר׃ (לה) אַנְשֵׁ֣י לֵ֭בָב יֹ֣אמְרוּ לִ֑י וְגֶ֥בֶר חָ֝כָ֗ם שֹׁמֵ֥עַֽ לִֽי׃ (לו) אִ֭יּוֹב לֹא־ בְדַ֣עַת יְדַבֵּ֑ר וּ֝דְבָרָ֗יו לֹ֣א בְהַשְׂכֵּֽיל׃ (לז) אָבִ֗י יִבָּחֵ֣ן אִיּ֣וֹב עַד־ נֶ֑צַח עַל־ תְּ֝שֻׁבֹ֗ת בְּאַנְשֵׁי־ אָֽוֶן׃ (לח) כִּ֥י יֹ֘סִ֤יף עַֽל־ חַטָּאת֣וֹ פֶ֭שַׁע בֵּינֵ֣ינוּ יִסְפּ֑וֹק וְיֶ֖רֶב אֲמָרָ֣יו לָאֵֽל׃ (לט) וַיַּ֥עַן אֱלִיה֗וּ וַיֹּאמַֽר׃ (מ) הֲ֭זֹאת חָשַׁ֣בְתָּ לְמִשְׁפָּ֑ט אָ֝מַ֗רְתָּ צִדְקִ֥י מֵאֵֽל׃ (מא) כִּֽי־ תֹ֭אמַר מַה־ יִּסְכָּן־ לָ֑ךְ מָֽה־ אֹ֝עִ֗יל מֵֽחַטָּאתִֽי׃ (מב) אֲ֭נִי אֲשִֽׁיבְךָ֣ מִלִּ֑ין וְֽאֶת־ רֵעֶ֥יךָ עִמָּֽךְ׃ (מג) הַבֵּ֣ט שָׁמַ֣יִם יִ֝לָּכְד֗וּן בְּחַבְלֵי־ עֹֽנִי׃ (סג) וַיַּגֵּ֣ד לָהֶ֣ם פָּעֳלָ֑ם וּ֝פִשְׁעֵיהֶ֗ם כִּ֣י יִתְגַּבָּֽרוּ׃ (סד) וַיִּ֣גֶל אָ֭זְנָם לַמּוּסָ֑ר וַ֝יֹּ֗אמֶר כִּֽי־ יְשֻׁב֥וּן מֵאָֽוֶן׃ (סה) אִֽם־ יִשְׁמְע֗וּ וְֽיַ֫עֲבֹ֥דוּ יְכַלּ֣וּ יְמֵיהֶ֣ם בַּטּ֑וֹב וּ֝שְׁנֵיהֶ֗ם בַּנְּעִימִֽים׃ (סו) וְאִם־ לֹ֣א יִ֭שְׁמְעוּ בְּשֶׁ֣לַח יַעֲבֹ֑רוּ וְ֝יִגְוְע֗וּ כִּבְלִי־ דָֽעַת׃ (סז) וְֽחַנְפֵי־ לֵ֭ב כֹּ֝֗פֶר אַל־ יַטֶּֽךָּ׃ (עג) הֲיַעֲרֹ֣ךְ שׁ֭וּעֲךָ לֹ֣א בְצָ֑ר וְ֝כֹ֗ל מַאֲמַצֵּי־ כֹֽחַ׃ (עד) אַל־ תִּשְׁאַ֥ף הַלָּ֑יְלָה לַעֲל֖וֹת עַמִּ֣ים תַּחְתָּֽם׃ (עה) הִ֭שָּׁמֶר אַל־ תֵּ֣פֶן אֶל־ אָ֑וֶן כִּֽי־ עַל־ זֶ֝֗ה בָּחַ֥רְתָּ מֵעֹֽנִי׃ (עו) הֶן־ אֵ֭ל יַשְׂגִּ֣יב בְּכֹח֑וֹ מִ֖י כָמֹ֣הוּ מוֹרֶֽה׃ (עז) מִֽי־ פָקַ֣ד עָלָ֣יו דַּרְכּ֑וֹ וּמִֽי־ אָ֝מַ֗ר פָּעַ֥לְתָּ עַוְלָֽה׃ (עח) זְ֭כֹר כִּֽי־ תַשְׂגִּ֣יא פָעֳל֑וֹ אֲשֶׁ֖ר שֹׁרְר֣וּ אֲנָשִֽׁים׃ (עט) כָּל־ אָדָ֥ם חָֽזוּ־ ב֑וֹ אֱ֝נ֗וֹשׁ יַבִּ֥יט מֵרָחֽוֹק׃ (פ) הֶן־ אֵ֣ל שַׂ֭גִּיא וְלֹ֣א נֵדָ֑ע מִסְפַּ֖ר שָׁנָ֣יו וְלֹא־ חֵֽקֶר׃ (פא) כִּ֭י יְגָרַ֣ע נִטְפֵי־ מָ֑יִם יָזֹ֖קּוּ מָטָ֣ר לְאֵדֽוֹ׃ (פב) אֲשֶֽׁר־ יִזְּל֥וּ שְׁחָקִ֑ים יִ֝רְעֲפ֗וּ עֲלֵ֤י ׀ אָדָ֬ם רָֽב׃ (פג) אַ֣ף אִם־ יָ֭בִין מִפְרְשֵׂי־ עָ֑ב תְּ֝שֻׁא֗וֹת סֻכָּתֽוֹ׃ (פד) הֵן־ פָּרַ֣שׂ עָלָ֣יו אוֹר֑וֹ וְשָׁרְשֵׁ֖י הַיָּ֣ם כִּסָּֽה׃ (פה) כִּי־ בָ֭ם יָדִ֣ין עַמִּ֑ים יִֽתֶּן־ אֹ֥כֶל לְמַכְבִּֽיר׃ (פו) עַל־ כַּפַּ֥יִם כִּסָּה־ א֑וֹר וַיְצַ֖ו עָלֶ֣יהָ בְמַפְגִּֽיעַ׃ (פז) יַגִּ֣יד עָלָ֣יו רֵע֑וֹ מִ֝קְנֶ֗ה אַ֣ף עַל־ עוֹלֶֽה׃ (פח) אַף־ לְ֭זֹאת יֶחֱרַ֣ד לִבִּ֑י וְ֝יִתַּ֗ר מִמְּקוֹמֽוֹ׃ (פט) שִׁמְע֤וּ שָׁמ֣וֹעַ בְּרֹ֣גֶז קֹל֑וֹ וְ֝הֶ֗גֶה מִפִּ֥יו יֵצֵֽא׃ (צ) תַּֽחַת־ כָּל־ הַשָּׁמַ֥יִם יִשְׁרֵ֑הוּ וְ֝אוֹר֗וֹ עַל־ כַּנְפ֥וֹת הָאָֽרֶץ׃ (צא) אַחֲרָ֤יו ׀ יִשְׁאַג־ ק֗וֹל יַ֭רְעֵם בְּק֣וֹל גְּאוֹנ֑וֹ וְלֹ֥א יְ֝עַקְּבֵ֗ם כִּֽי־ יִשָּׁמַ֥ע קוֹלֽוֹ׃ (צב) יַרְעֵ֤ם אֵ֣ל בְּ֭קוֹלוֹ נִפְלָא֑וֹת עֹשֶׂ֥ה גְ֝דֹל֗וֹת וְלֹ֣א נֵדָֽע׃ (צג) כִּ֤י לַשֶּׁ֨לַג ׀ יֹאמַ֗ר הֱוֵ֫א אָ֥רֶץ וְגֶ֥שֶׁם מָטָ֑ר וְ֝גֶ֗שֶׁם מִטְר֥וֹת עֻזּֽוֹ׃ (צד) בְּיַד־ כָּל־ אָדָ֥ם יַחְתּ֑וֹם לָ֝דַ֗עַת כָּל־ אַנְשֵׁ֥י מַעֲשֵֽׂהוּ׃ (צה) וַתָּבֹ֣א חַיָּ֣ה בְמוֹ־ אָ֑רֶב וּבִמְע֖וֹנֹתֶ֣יהָ תִשְׁכֹּֽן׃ (צו) מִן־ הַ֭חֶדֶר תָּב֣וֹא סוּפָ֑ה וּֽמִמְּזָרִ֥ים קָרָֽה׃ (צז) מִנִּשְׁמַת־ אֵ֥ל יִתֶּן־ קָ֑רַח וְרֹ֖חַב מַ֣יִם בְּמוּצָֽק׃ (צח) אַף־ בְּ֭רִי יַטְרִ֣יחַ עָ֑ב יָ֝פִ֗יץ עֲנַ֣ן אוֹרֽוֹ׃ (צט) וְה֤וּא מְסִבּ֨וֹת ׀ מִתְהַפֵּ֣ךְ (בתחבולתו) בְּתַחְבּוּלֹתָ֣יו לְפָעֳלָ֑ם כֹּ֖ל אֲשֶׁ֥ר יְצַוֵּ֓ם ׀ עַל־ פְּנֵ֖י תֵבֵ֣ל אָֽרְצָה׃ (ק) אִם־ לְשֵׁ֥בֶט אִם־ לְאַרְצ֑וֹ אִם־ לְ֝חֶ֗סֶד יַמְצִאֵֽהוּ׃ (קא) הַאֲזִ֣ינָה זֹּ֣את אִיּ֑וֹב עֲ֝מֹ֗ד וְהִתְבּוֹנֵ֤ן ׀ נִפְלְא֬וֹת אֵֽל׃ (קב) הֲ֭תֵדַע בְּשׂוּם־ אֱל֣וֹהַּ עֲלֵיהֶ֑ם וְ֝הוֹפִ֗יעַ א֣וֹר עֲנָנֽוֹ׃ (קג) הֲ֭תֵדַע עַל־ מִפְלְשֵׂי־ עָ֑ב מִ֝פְלְא֗וֹת תְּמִ֣ים דֵּעִֽים׃ (קד) אֲשֶׁר־ בְּגָדֶ֥יךָ חַמִּ֑ים בְּהַשְׁקִ֥ט אֶ֝֗רֶץ מִדָּרֽוֹם׃ (קה) תַּרְקִ֣יעַ עִ֭מּוֹ לִשְׁחָקִ֑ים חֲ֝זָקִ֗ים כִּרְאִ֥י מוּצָֽק׃ (קו) ה֭וֹדִיעֵנוּ מַה־ נֹּ֣אמַר ל֑וֹ לֹ֥א־ נַ֝עֲרֹ֗ךְ מִפְּנֵי־ חֹֽשֶׁךְ׃ (קז) הַֽיְסֻפַּר־ ל֭וֹ כִּ֣י אֲדַבֵּ֑ר אִֽם־ אָ֥מַר אִ֝֗ישׁ כִּ֣י יְבֻלָּֽע׃ (קח) וְעַתָּ֤ה ׀ לֹ֘א רָ֤אוּ א֗וֹר בָּהִ֣יר ה֭וּא בַּשְּׁחָקִ֑ים וְר֥וּחַ עָ֝בְרָ֗ה וַֽתְּטַהֲרֵֽם׃ (קט) מִ֭צָּפוֹן זָהָ֣ב יֶֽאֱתֶ֑ה עַל־ אֱ֝ל֗וֹהַּ נ֣וֹרָא הֽוֹד׃ (קי) שַׁדַּ֣י לֹֽא־ מְ֭צָאנֻהוּ שַׂגִּיא־ כֹ֑חַ וּמִשְׁפָּ֥ט וְרֹב־ צְ֝דָקָ֗ה לֹ֣א יְעַנֶּֽה׃ (קיא) לָ֭כֵן יְרֵא֣וּהוּ אֲנָשִׁ֑ים לֹֽא־ יִ֝רְאֶ֗ה כָּל־ חַכְמֵי־ לֵֽב׃ (קיב) וַיַּֽעַן־ יְהוָ֣ה אֶת־ אִ֭יּוֹב (מנ הסערה) מִ֥ן ׀ הַסְּעָרָ֗ה וַיֹּאמַֽר׃ (קיג) מִ֤י זֶ֨ה ׀ מַחְשִׁ֖יךְ עֵצָ֥ה בְמִלִּ֗ין בְּֽלִי־ דָֽעַת׃ (קיד) אֱזָר־ נָ֣א כְגֶ֣בֶר חֲלָצֶ֑יךָ וְ֝אֶשְׁאָלְךָ֗ וְהוֹדִיעֵֽנִי׃ (קטו) אֵיפֹ֣ה הָ֭יִיתָ בְּיָסְדִי־ אָ֑רֶץ הַ֝גֵּ֗ד אִם־ יָדַ֥עְתָּ בִינָֽה׃ (קטז) מִי־ שָׂ֣ם מְ֭מַדֶּיהָ כִּ֣י תֵדָ֑ע א֤וֹ מִֽי־ נָטָ֖ה עָלֶ֣יהָ קָּֽו׃ (קיז) עַל־ מָ֭ה אֲדָנֶ֣יהָ הָטְבָּ֑עוּ א֥וֹ מִֽי־ יָ֝רָ֗ה אֶ֣בֶן פִּנָּתָֽהּ׃ (קיח) בְּרָן־ יַ֭חַד כּ֣וֹכְבֵי בֹ֑קֶר וַ֝יָּרִ֗יעוּ כָּל־ בְּנֵ֥י אֱלֹהִֽים׃ (קיט) וַיָּ֣סֶךְ בִּדְלָתַ֣יִם יָ֑ם בְּ֝גִיח֗וֹ מֵרֶ֥חֶם יֵצֵֽא׃ (קכ) בְּשׂוּמִ֣י עָנָ֣ן לְבֻשׁ֑וֹ וַ֝עֲרָפֶ֗ל חֲתֻלָּתֽוֹ׃ (קכא) וָאֶשְׁבֹּ֣ר עָלָ֣יו חֻקִּ֑י וָֽ֝אָשִׂ֗ים בְּרִ֣יחַ וּדְלָתָֽיִם׃ (קכב) וָאֹמַ֗ר עַד־ פֹּ֣ה תָ֭בוֹא וְלֹ֣א תֹסִ֑יף וּפֹ֥א־ יָ֝שִׁ֗ית בִּגְא֥וֹן גַּלֶּֽיךָ׃ (קכג) הְֽ֭מִיָּמֶיךָ צִוִּ֣יתָ בֹּ֑קֶר (ידעתה שחר) יִדַּ֖עְתָּה הַשַּׁ֣חַר מְקֹמֽוֹ׃ (קכד) לֶ֭אֱחֹז בְּכַנְפ֣וֹת הָאָ֑רֶץ וְיִנָּעֲר֖וּ רְשָׁ מִמֶּֽנָּה׃ (קכה) תִּ֭תְהַפֵּךְ כְּחֹ֣מֶר חוֹתָ֑ם וְ֝יִֽתְיַצְּב֗וּ כְּמ֣וֹ לְבֽוּשׁ׃ (קכו) וְיִמָּנַ֣ע מֵרְשָׁ אוֹרָ֑ם וּזְר֥וֹעַ רָ֝מָ֗ה תִּשָּׁבֵֽר׃ (קכז) הֲ֭בָאתָ עַד־ נִבְכֵי־ יָ֑ם וּבְחֵ֥קֶר תְּ֝ה֗וֹם הִתְהַלָּֽכְתָּ׃ (קכח) הֲנִגְל֣וּ לְ֭ךָ שַׁעֲרֵי־ מָ֑וֶת וְשַׁעֲרֵ֖י צַלְמָ֣וֶת תִּרְאֶֽה׃ (קכט) הִ֭תְבֹּנַנְתָּ עַד־ רַחֲבֵי־ אָ֑רֶץ הַ֝גֵּ֗ד אִם־ יָדַ֥עְתָּ כֻלָּֽהּ׃ (קל) אֵי־ זֶ֣ה הַ֭דֶּרֶךְ יִשְׁכָּן־ א֑וֹר וְ֝חֹ֗שֶׁךְ אֵי־ זֶ֥ה מְקֹמֽוֹ׃ (קלא) כִּ֣י תִ֭קָּחֶנּוּ אֶל־ גְּבוּל֑וֹ וְכִֽי־ תָ֝בִ֗ין נְתִיב֥וֹת בֵּיתֽוֹ׃ (קלב) יָ֭דַעְתָּ כִּי־ אָ֣ז תִּוָּלֵ֑ד וּמִסְפַּ֖ר יָמֶ֣יךָ רַבִּֽים׃ (קלג) הֲ֭בָאתָ אֶל־ אֹצְר֣וֹת שָׁ֑לֶג וְאֹצְר֖וֹת בָּרָ֣ד תִּרְאֶֽה׃ (קלד) אֲשֶׁר־ חָשַׂ֥כְתִּי לְעֶת־ צָ֑ר לְי֥וֹם קְ֝רָ֗ב וּמִלְחָמָֽה׃ (קלה) אֵי־ זֶ֣ה הַ֭דֶּרֶךְ יֵחָ֣לֶק א֑וֹר יָפֵ֖ץ קָדִ֣ים עֲלֵי־ אָֽרֶץ׃ (קלו) מִֽי־ פִלַּ֣ג לַשֶּׁ֣טֶף תְּעָלָ֑ה וְ֝דֶ֗רֶךְ לַחֲזִ֥יז קֹלֽוֹת׃ (קלז) לְ֭הַמְטִיר עַל־ אֶ֣רֶץ לֹא־ אִ֑ישׁ מִ֝דְבָּ֗ר לֹא־ אָדָ֥ם בּֽוֹ׃ (קלח) לְהַשְׂבִּ֣יעַ שֹׁ֭אָה וּמְשֹׁאָ֑ה וּ֝לְהַצְמִ֗יחַ מֹ֣צָא דֶֽשֶׁא׃ (קלט) הֲיֵשׁ־ לַמָּטָ֥ר אָ֑ב א֥וֹ מִי־ ה֝וֹלִ֗יד אֶגְלֵי־ טָֽל׃ (קמ) מִבֶּ֣טֶן מִ֭י יָצָ֣א הַקָּ֑רַח וּכְפֹ֥ר שָׁ֝מַיִם מִ֣י יְלָדֽוֹ׃ (קמא) כָּ֭אֶבֶן מַ֣יִם יִתְחַבָּ֑אוּ וּפְנֵ֥י תְ֝ה֗וֹם יִתְלַכָּֽדוּ׃ (קמב) הַֽ֭תְקַשֵּׁר מַעֲדַנּ֣וֹת כִּימָ֑ה אֽוֹ־ מֹשְׁכ֖וֹת כְּסִ֣יל תְּפַתֵּֽחַ׃ (קמג) הֲתֹצִ֣יא מַזָּר֣וֹת בְּעִתּ֑וֹ וְ֝עַ֗יִשׁ עַל־ בָּנֶ֥יהָ תַנְחֵֽם׃ (קמד) הֲ֭יָדַעְתָּ חֻקּ֣וֹת שָׁמָ֑יִם אִם־ תָּשִׂ֖ים מִשְׁטָר֣וֹ בָאָֽרֶץ׃ (קמה) הֲתָרִ֣ים לָעָ֣ב קוֹלֶ֑ךָ וְֽשִׁפְעַת־ מַ֥יִם תְּכַסֶּֽךָּ׃
