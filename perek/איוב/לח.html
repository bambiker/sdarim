איוב פרק לח לפי החלוקה לסדרים

פרק לח בספר איוב (41 פסוקים) נמצא באיוב סדר ז (פסוקים קיב–קמה) ובאיוב סדר ח (פסוקים א–ז).

איוב סדר ז

(א) אִם־ אַ֭יִן אַתָּ֥ה שְֽׁמַֽע־ לִ֑י הַ֝חֲרֵ֗שׁ וַאֲאַלֶּפְךָ֥ חָכְמָֽה׃ (ב) וַיַּ֥עַן אֱלִיה֗וּא וַיֹּאמַֽר׃ (ג) שִׁמְע֣וּ חֲכָמִ֣ים מִלָּ֑י וְ֝יֹדְעִ֗ים הַאֲזִ֥ינוּ לִֽי׃ (ד) כִּי־ אֹ֭זֶן מִלִּ֣ין תִּבְחָ֑ן וְ֝חֵ֗ךְ יִטְעַ֥ם לֶאֱכֹֽל׃ (ה) מִשְׁפָּ֥ט נִבְחֲרָה־ לָּ֑נוּ נֵדְעָ֖ה בֵינֵ֣ינוּ מַה־ טּֽוֹב׃ (ו) כִּֽי־ אָ֭מַר אִיּ֣וֹב צָדַ֑קְתִּי וְ֝אֵ֗ל הֵסִ֥יר מִשְׁפָּטִֽי׃ (ז) עַל־ מִשְׁפָּטִ֥י אֲכַזֵּ֑ב אָנ֖וּשׁ חִצִּ֣י בְלִי־ פָֽשַׁע׃ (ח) מִי־ גֶ֥בֶר כְּאִיּ֑וֹב יִֽשְׁתֶּה־ לַּ֥עַג כַּמָּֽיִם׃ (ט) וְאָרַ֣ח לְ֭חֶבְרָה עִם־ פֹּ֣עֲלֵי אָ֑וֶן וְ֝לָלֶ֗כֶת עִם־ אַנְשֵׁי־ רֶֽשַׁע׃ (י) כִּֽי־ אָ֭מַר לֹ֣א יִסְכָּן־ גָּ֑בֶר בִּ֝רְצֹת֗וֹ עִם־ אֱלֹהִֽים׃ (יא) לָכֵ֤ן ׀ אַ֥נֲשֵׁ֥י לֵבָ֗ב שִׁמְע֫וּ לִ֥י חָלִ֖לָה לָאֵ֥ל מֵרֶ֗שַׁע וְשַׁדַּ֥י מֵעָֽוֶל׃ (יב) כִּ֤י פֹ֣עַל אָ֭דָם יְשַׁלֶּם־ ל֑וֹ וּֽכְאֹ֥רַח אִ֝֗ישׁ יַמְצִאֶֽנּוּ׃ (יג) אַף־ אָמְנָ֗ם אֵ֥ל לֹֽא־ יַרְשִׁ֑יעַ וְ֝שַׁדַּ֗י לֹֽא־ יְעַוֵּ֥ת מִשְׁפָּֽט׃ (יד) מִֽי־ פָקַ֣ד עָלָ֣יו אָ֑רְצָה וּמִ֥י שָׂ֝֗ם תֵּבֵ֥ל כֻּלָּֽהּ׃ (טו) אִם־ יָשִׂ֣ים אֵלָ֣יו לִבּ֑וֹ רוּח֥וֹ וְ֝נִשְׁמָת֗וֹ אֵלָ֥יו יֶאֱסֹֽף׃ (טז) יִגְוַ֣ע כָּל־ בָּשָׂ֣ר יָ֑חַד וְ֝אָדָ֗ם עַל־ עָפָ֥ר יָשֽׁוּב׃ (יז) וְאִם־ בִּ֥ינָה שִׁמְעָה־ זֹּ֑את הַ֝אֲזִ֗ינָה לְק֣וֹל מִלָּֽי׃ (יח) הַאַ֬ף שׂוֹנֵ֣א מִשְׁפָּ֣ט יַחֲב֑וֹשׁ וְאִם־ צַדִּ֖יק כַּבִּ֣יר תַּרְשִֽׁיעַ׃ (יט) הַאֲמֹ֣ר לְמֶ֣לֶךְ בְּלִיָּ֑עַל רָ֝שָׁ֗ע אֶל־ נְדִיבִֽים׃ (כ) אֲשֶׁ֤ר לֹֽא־ נָשָׂ֨א ׀ פְּנֵ֥י שָׂרִ֗ים וְלֹ֣א נִכַּר־ שׁ֭וֹעַ לִפְנֵי־ דָ֑ל כִּֽי־ מַעֲשֵׂ֖ה יָדָ֣יו כֻּלָּֽם׃ (כא) רֶ֤גַע ׀ יָמֻתוּ֮ וַחֲצ֪וֹת לָ֥יְלָה יְגֹעֲשׁ֣וּ עָ֣ם וְיַעֲבֹ֑רוּ וְיָסִ֥ירוּ אַ֝בִּ֗יר לֹ֣א בְיָֽד׃ (כב) כִּי־ עֵ֭ינָיו עַל־ דַּרְכֵי־ אִ֑ישׁ וְֽכָל־ צְעָדָ֥יו יִרְאֶֽה׃ (כג) אֵֽין־ חֹ֭שֶׁךְ וְאֵ֣ין צַלְמָ֑וֶת לְהִסָּ֥תֶר שָׁ֝֗ם פֹּ֣עֲלֵי אָֽוֶן׃ (כד) כִּ֤י לֹ֣א עַל־ אִ֭ישׁ יָשִׂ֣ים ע֑וֹד לַהֲלֹ֥ךְ אֶל־ אֵ֝֗ל בַּמִּשְׁפָּֽט׃ (כה) יָרֹ֣עַ כַּבִּירִ֣ים לֹא־ חֵ֑קֶר וַיַּעֲמֵ֖ד אֲחֵרִ֣ים תַּחְתָּֽם׃ (כו) לָכֵ֗ן יַ֭כִּיר מַעְבָּֽדֵיהֶ֑ם וְהָ֥פַךְ לַ֝֗יְלָה וְיִדַּכָּֽאוּ׃ (כז) תַּֽחַת־ רְשָׁעִ֥ים סְפָקָ֗ם בִּמְק֥וֹם רֹאִֽים׃ (כח) אֲשֶׁ֣ר עַל־ כֵּ֭ן סָ֣רוּ מֵֽאַחֲרָ֑יו וְכָל־ דְּ֝רָכָ֗יו לֹ֣א הִשְׂכִּֽילוּ׃ (כט) לְהָבִ֣יא עָ֭לָיו צַֽעֲקַת־ דָּ֑ל וְצַעֲקַ֖ת עֲנִיִּ֣ים יִשְׁמָֽע׃ (ל) וְה֤וּא יַשְׁקִ֨ט ׀ וּמִ֥י יַרְשִׁ֗עַ וְיַסְתֵּ֣ר פָּ֭נִים וּמִ֣י יְשׁוּרֶ֑נּוּ וְעַל־ גּ֖וֹי וְעַל־ אָדָ֣ם יָֽחַד׃ (לא) מִ֭מְּלֹךְ אָדָ֥ם חָנֵ֗ף מִמֹּ֥קְשֵׁי עָֽם׃ (לב) כִּֽי־ אֶל־ אֵ֭ל הֶאָמַ֥ר נָשָׂ֗אתִי לֹ֣א אֶחְבֹּֽל׃ (לג) בִּלְעֲדֵ֣י אֶ֭חֱזֶה אַתָּ֣ה הֹרֵ֑נִי אִֽם־ עָ֥וֶל פָּ֝עַ֗לְתִּי לֹ֣א אֹסִֽיף׃ (לד) הַֽמֵעִמְּךָ֬ יְשַׁלְמֶ֨נָּה כִּֽי־ מָאַ֗סְתָּ כִּי־ אַתָּ֣ה תִבְחַ֣ר וְלֹא־ אָ֑נִי וּֽמַה־ יָדַ֥עְתָּ דַבֵּֽר׃ (לה) אַנְשֵׁ֣י לֵ֭בָב יֹ֣אמְרוּ לִ֑י וְגֶ֥בֶר חָ֝כָ֗ם שֹׁמֵ֥עַֽ לִֽי׃ (לו) אִ֭יּוֹב לֹא־ בְדַ֣עַת יְדַבֵּ֑ר וּ֝דְבָרָ֗יו לֹ֣א בְהַשְׂכֵּֽיל׃ (לז) אָבִ֗י יִבָּחֵ֣ן אִיּ֣וֹב עַד־ נֶ֑צַח עַל־ תְּ֝שֻׁבֹ֗ת בְּאַנְשֵׁי־ אָֽוֶן׃ (לח) כִּ֥י יֹ֘סִ֤יף עַֽל־ חַטָּאת֣וֹ פֶ֭שַׁע בֵּינֵ֣ינוּ יִסְפּ֑וֹק וְיֶ֖רֶב אֲמָרָ֣יו לָאֵֽל׃ (לט) וַיַּ֥עַן אֱלִיה֗וּ וַיֹּאמַֽר׃ (מ) הֲ֭זֹאת חָשַׁ֣בְתָּ לְמִשְׁפָּ֑ט אָ֝מַ֗רְתָּ צִדְקִ֥י מֵאֵֽל׃ (מא) כִּֽי־ תֹ֭אמַר מַה־ יִּסְכָּן־ לָ֑ךְ מָֽה־ אֹ֝עִ֗יל מֵֽחַטָּאתִֽי׃ (מב) אֲ֭נִי אֲשִֽׁיבְךָ֣ מִלִּ֑ין וְֽאֶת־ רֵעֶ֥יךָ עִמָּֽךְ׃ (מג) הַבֵּ֣ט שָׁמַ֣יִם וּרְאֵ֑ה וְשׁ֥וּר שְׁ֝חָקִ֗ים גָּבְה֥וּ מִמֶּֽךָּ׃ (מד) אִם־ חָ֭טָאתָ מַה־ תִּפְעָל־ בּ֑וֹ וְרַבּ֥וּ פְ֝שָׁעֶ֗יךָ מַה־ תַּעֲשֶׂה־ לּֽוֹ׃ (מה) אִם־ צָ֭דַקְתָּ מַה־ תִּתֶּן־ ל֑וֹ א֥וֹ מַה־ מִיָּדְךָ֥ יִקָּֽח׃ (מו) לְאִישׁ־ כָּמ֥וֹךָ רִשְׁעֶ֑ךָ וּלְבֶן־ אָ֝דָ֗ם צִדְקָתֶֽךָ׃ (מז) מֵ֭רֹב עֲשׁוּקִ֣ים יַזְעִ֑יקוּ יְשַׁוְּע֖וּ מִזְּר֣וֹעַ רַבִּֽים׃ (מח) וְֽלֹא־ אָמַ֗ר אַ֭יֵּה אֱל֣וֹהַּ עֹשָׂ֑י נֹתֵ֖ן זְמִר֣וֹת בַּלָּֽיְלָה׃ (מט) מַ֭לְּפֵנוּ מִבַּהֲמ֣וֹת אָ֑רֶץ וּמֵע֖וֹף הַשָּׁמַ֣יִם יְחַכְּמֵֽנוּ׃ (נ) שָׁ֣ם יִ֭צְעֲקוּ וְלֹ֣א יַעֲנֶ֑ה מִ֝פְּנֵ֗י גְּא֣וֹן רָעִֽים׃ (נא) אַךְ־ שָׁ֭וְא לֹא־ יִשְׁמַ֥ע ׀ אֵ֑ל וְ֝שַׁדַּ֗י לֹ֣א יְשׁוּרֶֽנָּה׃ (נב) אַ֣ף כִּֽי־ תֹ֭אמַר לֹ֣א תְשׁוּרֶ֑נּוּ דִּ֥ין לְ֝פָנָ֗יו וּתְח֥וֹלֵֽל לֽוֹ׃ (נג) וְעַתָּ֗ה כִּי־ אַ֭יִן פָּקַ֣ד אַפּ֑וֹ וְלֹֽא־ יָדַ֖ע בַּפַּ֣שׁ מְאֹֽד׃ (נד) וְ֭אִיּוֹב הֶ֣בֶל יִפְצֶה־ פִּ֑יהוּ בִּבְלִי־ דַ֝֗עַת מִלִּ֥ין יַכְבִּֽר׃ (נה) וַיֹּ֥סֶף אֱלִיה֗וּא וַיֹּאמַֽר׃ (נו) כַּתַּר־ לִ֣י זְ֭עֵיר וַאֲחַוֶּ֑ךָּ כִּ֤י ע֖וֹד לֶאֱל֣וֹהַּ מִלִּֽים׃ (נז) אֶשָּׂ֣א דֵ֭עִי לְמֵרָח֑וֹק וּ֝לְפֹעֲלִ֗י אֶֽתֵּֽן־ צֶֽדֶק׃ (נח) כִּֽי־ אָ֭מְנָם לֹא־ שֶׁ֣קֶר מִלָּ֑י תְּמִ֖ים דֵּע֣וֹת עִמָּֽךְ׃ (נט) הֶן־ אֵ֣ל כַּ֭בִּיר וְלֹ֣א יִמְאָ֑ס כַּ֝בִּ֗יר כֹּ֣חַֽ לֵֽב׃ (ס) לֹא־ יְחַיֶּ֥ה רָשָׁ֑ע וּמִשְׁפַּ֖ט עֲנִיִּ֣ים יִתֵּֽן׃ (סא) לֹֽא־ יִגְרַ֥ע מִצַּדִּ֗יק עֵ֫ינָ֥יו וְאֶת־ מְלָכִ֥ים לַכִּסֵּ֑א וַיֹּשִׁיבֵ֥ם לָ֝נֶ֗צַח וַיִּגְבָּֽהוּ׃ (סב) וְאִם־ אֲסוּרִ֥ים בַּזִּקִּ֑ים יִ֝לָּכְד֗וּן בְּחַבְלֵי־ עֹֽנִי׃ (סג) וַיַּגֵּ֣ד לָהֶ֣ם פָּעֳלָ֑ם וּ֝פִשְׁעֵיהֶ֗ם כִּ֣י יִתְגַּבָּֽרוּ׃ (סד) וַיִּ֣גֶל אָ֭זְנָם לַמּוּסָ֑ר וַ֝יֹּ֗אמֶר כִּֽי־ יְשֻׁב֥וּן מֵאָֽוֶן׃ (סה) אִֽם־ יִשְׁמְע֗וּ וְֽיַ֫עֲבֹ֥דוּ יְכַלּ֣וּ יְמֵיהֶ֣ם בַּטּ֑וֹב וּ֝שְׁנֵיהֶ֗ם בַּנְּעִימִֽים׃ (סו) וְאִם־ לֹ֣א יִ֭שְׁמְעוּ בְּשֶׁ֣לַח יַעֲבֹ֑רוּ וְ֝יִגְוְע֗וּ כִּבְלִי־ דָֽעַת׃ (סז) וְֽחַנְפֵי־ לֵ֭ב יָשִׂ֣ימוּ אָ֑ף לֹ֥א יְ֝שַׁוְּע֗וּ כִּ֣י אֲסָרָֽם׃ (סח) תָּמֹ֣ת בַּנֹּ֣עַר נַפְשָׁ֑ם וְ֝חַיָּתָ֗ם בַּקְּדֵשִֽׁים׃ (סט) יְחַלֵּ֣ץ עָנִ֣י בְעָנְי֑וֹ וְיִ֖גֶל בַּלַּ֣חַץ אָזְנָֽם׃ (ע) וְאַ֤ף הֲסִיתְךָ֨ ׀ שָׁ֝מַ֗יִם מִ֣י יַשְׁכִּֽיב׃ (ד) בְּצֶ֣קֶת עָ֭פָר לַמּוּצָ֑ק וּרְגָבִ֥ים יְדֻבָּֽקוּ׃ (ה) הֲתָצ֣וּד לְלָבִ֣יא טָ֑רֶף וְחַיַּ֖ת כְּפִירִ֣ים תְּמַלֵּֽא׃ (ו) כִּי־ יָשֹׁ֥חוּ בַמְּעוֹנ֑וֹת יֵשְׁב֖וּ בַסֻּכָּ֣ה לְמוֹ־ אָֽרֶב׃ (ז) מִ֤י יָכִ֥ין לָעֹרֵ֗ב צֵ֫יד֥וֹ כִּֽי־ (ילדו) יְ֭לָדָיו אֶל־ אֵ֣ל יְשַׁוֵּ֑עוּ יִ֝תְע֗וּ לִבְלִי־ אֹֽכֶל׃ סוף פרק לח (ח) הֲיָדַ֗עְתָּ עֵ֭ת לֶ֣דֶת יַעֲלֵי־ סָ֑לַע חֹלֵ֖ל אַיָּל֣וֹת תִּשְׁמֹֽר׃ (ט) תִּסְפֹּ֣ר יְרָחִ֣ים תְּמַלֶּ֑אנָה וְ֝יָדַ֗עְתָּ עֵ֣ת לִדְתָּֽנָה׃ (י) תִּ֭כְרַעְנָה יַלְדֵיהֶ֣ן תְּפַלַּ֑חְנָה חֶבְלֵיהֶ֥ם תְּשַׁלַּֽחְנָה׃ (יא) יַחְלְמ֣וּ בְ֭נֵיהֶם יִרְבּ֣וּ בַבָּ֑ר יָ֝צְא֗וּ וְלֹא־ שָׁ֥בוּ לָֽמוֹ׃ (יב) מִֽי־ שִׁלַּ֣ח פֶּ֣רֶא חָפְשִׁ֑י וּמֹסְר֥וֹת עָ֝ר֗וֹד מִ֣י פִתֵּֽחַ׃ (יג) אֲשֶׁר־ שַׂ֣מְתִּי עֲרָבָ֣ה בֵית֑וֹ וּֽמִשְׁכְּנוֹתָ֥יו מְלֵֽחָה׃ (יד) יִ֭שְׂחַק לַהֲמ֣וֹן קִרְיָ֑ה תְּשֻׁא֥וֹת נ֝וֹגֵ֗שׂ לֹ֣א יִשְׁמָֽע׃ (טו) יְת֣וּר הָרִ֣ים מִרְעֵ֑הוּ וְאַחַ֖ר כָּל־ יָר֣וֹק יִדְרֽוֹשׁ׃ (טז) הֲיֹ֣אבֶה רֵּ֣ים עָבְדֶ֑ךָ אִם־ יָ֝לִ֗ין עַל־ אֲבוּסֶֽךָ׃ (יז) הֲ‍ֽתִקְשָׁר־ רֵ֭ים בְּתֶ֣לֶם עֲבֹת֑וֹ אִם־ יְשַׂדֵּ֖ד עֲמָקִ֣ים אַחֲרֶֽיךָ׃ (יח) הֲֽתִבְטַח־ בּ֭וֹ כִּי־ רַ֣ב כֹּח֑וֹ וְתַעֲזֹ֖ב אֵלָ֣יו יְגִיעֶֽךָ׃ (יט) הֲתַאֲמִ֣ין בּ֭וֹ כִּי־ (ישוב) יָשִׁ֣יב זַרְעֶ֑ךָ וְֽגָרְנְךָ֥ יֶאֱסֹֽף׃ (כ) כְּנַף־ רְנָנִ֥ים נֶעֱלָ֑סָה אִם־ אֶ֝בְרָ֗ה חֲסִידָ֥ה וְנֹצָֽה׃ (כא) כִּֽי־ תַעֲזֹ֣ב לָאָ֣רֶץ בֵּצֶ֑יהָ וְֽעַל־ עָפָ֥ר תְּחַמֵּֽם׃ (כב) וַ֭תִּשְׁכַּח כִּי־ רֶ֣גֶל תְּזוּרֶ֑הָ וְחַיַּ֖ת הַשָּׂדֶ֣ה תְּדוּשֶֽׁהָ׃ (כג) הִקְשִׁ֣יחַ בָּנֶ֣יהָ לְּלֹא־ לָ֑הּ לְרִ֖יק יְגִיעָ֣הּ בְּלִי־ פָֽחַד׃ (כד) כִּֽי־ הִשָּׁ֣הּ אֱל֣וֹהַּ חָכְמָ֑ה וְלֹא־ חָ֥לַק לָ֝֗הּ בַּבִּינָֽה׃ (כה) כָּ֭עֵת בַּמָּר֣וֹם תַּמְרִ֑יא תִּֽשְׂחַ֥ק לַ֝סּ֗וּס וּלְרֹֽכְבֽוֹ׃ (כו) הֲתִתֵּ֣ן לַסּ֣וּס גְּבוּרָ֑ה הֲתַלְבִּ֖ישׁ צַוָּאר֣וֹ רַעְמָֽה׃ (כז) הְֽ֭תַרְעִישֶׁנּוּ כָּאַרְבֶּ֑ה ה֖וֹד נַחְר֣וֹ אֵימָֽה׃ (כח) יַחְפְּר֣וּ בָ֭עֵמֶק וְיָשִׂ֣ישׂ בְּכֹ֑חַ יֵ֝צֵ֗א לִקְרַאת־ נָֽשֶׁק׃ (כט) יִשְׂחַ֣ק לְ֭פַחַד וְלֹ֣א יֵחָ֑ת וְלֹֽא־ יָ֝שׁ֗וּב מִפְּנֵי־ חָֽרֶב׃ (ל) עָ֭לָיו תִּרְנֶ֣ה אַשְׁפָּ֑ה לַ֖הַב חֲנִ֣ית וְכִידֽוֹן׃ (לא) בְּרַ֣עַשׁ וְ֭רֹגֶז יְגַמֶּא־ אָ֑רֶץ וְלֹֽא־ יַ֝אֲמִ֗ין כִּי־ ק֥וֹל שׁוֹפָֽר׃ (לב) בְּדֵ֤י שֹׁפָ֨ר ׀ יֹ֘אמַ֤ר הֶאָ֗ח וּֽ֭מֵרָחוֹק יָרִ֣יחַ מִלְחָמָ֑ה רַ֥עַם שָׂ֝רִים וּתְרוּעָֽה׃ (לג) הֲ‍ֽ֭מִבִּינָ֣תְךָ יַֽאֲבֶר־ נֵ֑ץ יִפְרֹ֖שׂ (כנפו) כְּנָפָ֣יו לְתֵימָֽן׃ (לד) אִם־ עַל־ פִּ֭יךָ יַגְבִּ֣יהַּ נָ֑שֶׁר וְ֝כִ֗י יָרִ֥ים קִנּֽוֹ׃ (לה) סֶ֣לַע יִ֭שְׁכֹּן וְיִתְלֹנָ֑ן עַֽל־ שֶׁן־ סֶ֝֗לַע וּמְצוּדָֽה׃ (לו) מִשָּׁ֥ם חָֽפַר־ אֹ֑כֶל לְ֝מֵרָח֗וֹק עֵינָ֥יו יַבִּֽיטוּ׃ (לז) (ואפרחו) וְאֶפְרֹחָ֥יו יְעַלְעוּ־ דָ֑ם וּבַאֲשֶׁ֥ר חֲ֝לָלִ֗ים שָׁ֣ם הֽוּא׃ (לח) וַיַּ֖עַן יְהוָ֥ה אֶת־ אִיּ֗וֹב וַיֹּאמַֽר׃ (לט) הֲ֭רֹב עִם־ שַׁדַּ֣י יִסּ֑וֹר מוֹכִ֖יחַ אֱל֣וֹהַּ יַעֲנֶֽנָּה׃ (מ) וַיַּ֖עַן אִיּ֥וֹב אֶת־ יְהוָ֗ה וַיֹּאמַֽר׃ (מא) הֵ֣ן קַ֭לֹּתִי מָ֣ה אֲשִׁיבֶ֑ךָּ יָ֝דִ֗י שַׂ֣מְתִּי לְמוֹ־ פִֽי׃ (מב) אַחַ֣ת דִּ֭בַּרְתִּי וְלֹ֣א אֶֽעֱנֶ֑ה וּ֝שְׁתַּ֗יִם וְלֹ֣א אוֹסִֽיף׃ (מג) וַיַּֽעַן־ יְהוָ֣ה אֶת־ אִ֭יּוֹב (מנ סערה) מִ֥ן ׀ סְעָרָ֗ה וַיֹּאמַֽר׃ (מד) אֱזָר־ נָ֣א כְגֶ֣בֶר חֲלָצֶ֑יךָ אֶ֝שְׁאָלְךָ֗ וְהוֹדִיעֵֽנִי׃ (מה) הַ֭אַף תָּפֵ֣ר מִשְׁפָּטִ֑י תַּ֝רְשִׁיעֵ֗נִי לְמַ֣עַן תִּצְדָּֽק׃ (מו) וְאִם־ זְר֖וֹעַ כָּאֵ֥ל ׀ לָ֑ךְ וּ֝בְק֗וֹל כָּמֹ֥הוּ תַרְעֵֽם׃ (מז) עֲדֵ֥ה נָ֣א גָֽא֣וֹן וָגֹ֑בַהּ וְה֖וֹד וְהָדָ֣ר תִּלְבָּֽשׁ׃ (מח) הָ֭פֵץ עֶבְר֣וֹת אַפֶּ֑ךָ וּרְאֵ֥ה כָל־ גֵּ֝אֶ֗ה וְהַשְׁפִּילֵֽהוּ׃ (מט) רְאֵ֣ה כָל־ גֵּ֭אֶה הַכְנִיעֵ֑הוּ וַהֲדֹ֖ךְ רְשָׁעִ֣ים תַּחְתָּֽם׃ (נ) טָמְנֵ֣ם בֶּעָפָ֣ר יָ֑חַד פְּ֝נֵיהֶ֗ם חֲבֹ֣שׁ בַּטָּמֽוּן׃ (נא) וְגַם־ אֲנִ֥י אוֹדֶ֑ךָּ כִּֽי־ תוֹשִׁ֖עַ לְךָ֣ יְמִינֶֽךָ׃ (נב) הִנֵּה־ נָ֣א בְ֭הֵמוֹת אֲשֶׁר־ עָשִׂ֣יתִי עִמָּ֑ךְ חָ֝צִ֗יר כַּבָּקָ֥ר יֹאכֵֽל׃ (נג) הִנֵּה־ נָ֣א כֹח֣וֹ בְמָתְנָ֑יו וְ֝אֹנ֗וֹ בִּשְׁרִירֵ֥י בִטְנֽוֹ׃ (נד) יַחְפֹּ֣ץ זְנָב֣וֹ כְמוֹ־ אָ֑רֶז גִּידֵ֖י (פחדו) פַחֲדָ֣יו יְשֹׂרָֽגוּ׃ (נה) עֲ֭צָמָיו אֲפִיקֵ֣י נְחוּשָׁ֑ה גְּ֝רָמָ֗יו כִּמְטִ֥יל בַּרְזֶֽל׃ (נו) ה֭וּא רֵאשִׁ֣ית דַּרְכֵי־ אֵ֑ל הָ֝עֹשׂוֹ יַגֵּ֥שׁ חַרְבּֽוֹ׃ (נז) כִּֽי־ ב֭וּל הָרִ֣ים יִשְׂאוּ־ ל֑וֹ וְֽכָל־ חַיַּ֥ת הַ֝שָּׂדֶ֗ה יְשַֽׂחֲקוּ־ שָֽׁם׃ (נח) תַּֽחַת־ צֶאֱלִ֥ים יִשְׁכָּ֑ב בְּסֵ֖תֶר קָנֶ֣ה וּבִצָּֽה׃ (נט) יְסֻכֻּ֣הוּ צֶאֱלִ֣ים צִֽלֲל֑וֹ יְ֝סֻבּ֗וּהוּ עַרְבֵי־ נָֽחַל׃ (ס) הֵ֤ן יַעֲשֹׁ֣ק נָ֭הָר לֹ֣א יַחְפּ֑וֹז יִבְטַ֓ח ׀ כִּֽי־ יָגִ֖יחַ יַרְדֵּ֣ן אֶל־ פִּֽיהוּ׃ (סא) בְּעֵינָ֥יו יִקָּחֶ֑נּוּ בְּ֝מֽוֹקְשִׁ֗ים יִנְקָב־ אָֽף׃ (סב) תִּמְשֹׁ֣ךְ לִוְיָתָ֣ן בְּחַכָּ֑ה וּ֝בְחֶ֗בֶל תַּשְׁקִ֥יעַ לְשֹׁנֽוֹ׃ (סג) הֲתָשִׂ֣ים אַגְמ֣וֹן בְּאַפּ֑וֹ וּ֝בְח֗וֹחַ תִּקּ֥וֹב לֶֽחֱיוֹ׃ (סד) הֲיַרְבֶּ֣ה אֵ֭לֶיךָ תַּחֲנוּנִ֑ים אִם־ יְדַבֵּ֖ר אֵלֶ֣יךָ רַכּֽוֹת׃ (סה) הֲיִכְרֹ֣ת בְּרִ֣ית עִמָּ֑ךְ תִּ֝קָּחֶ֗נּוּ לְעֶ֣בֶד עוֹלָֽם׃ (סו) הַֽתְשַׂחֶק־ בּ֭וֹ כַּצִּפּ֑וֹר וְ֝תִקְשְׁרֶ֗נּוּ לְנַעֲרוֹתֶֽיךָ׃ (סז) יִכְר֣וּ עָ֭לָיו חַבָּרִ֑ים יֶ֝חֱצ֗וּהוּ בֵּ֣ין כְּֽנַעֲנִֽים׃ (סח) הַֽתְמַלֵּ֣א בְשֻׂכּ֣וֹת עוֹר֑וֹ וּבְצִלְצַ֖ל דָּגִ֣ים רֹאשֽׁוֹ׃ (סט) שִׂים־ עָלָ֥יו כַּפֶּ֑ךָ זְכֹ֥ר מִ֝לְחָמָ֗ה אַל־ תּוֹסַֽף׃ (ע) הֵן־ תֹּחַלְתּ֥וֹ נִכְזָ֑בָה הֲגַ֖ם אֶל־ מַרְאָ֣יו יֻטָֽל׃ (עא) לֹֽא־ אַ֭כְזָר כִּ֣י יְעוּרֶ֑נּוּ וּמִ֥י ה֝֗וּא לְפָנַ֥י יִתְיַצָּֽב׃ (עב) מִ֣י הִ֭קְדִּימַנִי וַאֲשַׁלֵּ֑ם תַּ֖חַת כָּל־ הַשָּׁמַ֣יִם לִי־ הֽוּא׃ (עג) (לא)־ לֽוֹ־ אַחֲרִ֥ישׁ בַּדָּ֑יו וּדְבַר־ גְּ֝בוּר֗וֹת וְחִ֣ין עֶרְכּֽוֹ׃ (עד) מִֽי־ גִ֭לָּה פְּנֵ֣י לְבוּשׁ֑וֹ בְּכֶ֥פֶל רִ֝סְנ֗וֹ מִ֣י יָבֽוֹא׃ (עה) דַּלְתֵ֣י פָ֭נָיו מִ֣י פִתֵּ֑חַ סְבִיב֖וֹת שִׁנָּ֣יו אֵימָֽה׃ (עו) גַּ֭אֲוָה אֲפִיקֵ֣י מָֽגִנִּ֑ים סָ֝ג֗וּר חוֹתָ֥ם צָֽר׃ (עז) אֶחָ֣ד בְּאֶחָ֣ד יִגַּ֑שׁוּ וְ֝ר֗וּחַ לֹא־ יָב֥וֹא בֵֽינֵיהֶֽם׃ (עח) אִישׁ־ בְּאָחִ֥יהוּ יְדֻבָּ֑קוּ יִ֝תְלַכְּד֗וּ וְלֹ֣א יִתְפָּרָֽדוּ׃ (עט) עֲ‍ֽ֭טִישֹׁתָיו תָּ֣הֶל א֑וֹר וְ֝עֵינָ֗יו כְּעַפְעַפֵּי־ שָֽׁחַר׃ (פ) מִ֭פִּיו לַפִּידִ֣ים יַהֲלֹ֑כוּ כִּיד֥וֹדֵי אֵ֝֗שׁ יִתְמַלָּֽטוּ׃ (פא) מִ֭נְּחִירָיו יֵצֵ֣א עָשָׁ֑ן כְּד֖וּד נָפ֣וּחַ וְאַגְמֹֽן׃ (פב) נַ֭פְשׁוֹ גֶּחָלִ֣ים תְּלַהֵ֑ט וְ֝לַ֗הַב מִפִּ֥יו יֵצֵֽא׃ (פג) בְּֽ֭צַוָּארוֹ יָלִ֣ין עֹ֑ז וּ֝לְפָנָ֗יו תָּד֥וּץ דְּאָבָֽה׃ (פד) מַפְּלֵ֣י בְשָׂר֣וֹ דָבֵ֑קוּ יָצ֥וּק עָ֝לָ֗יו בַּל־ יִמּֽוֹט׃ (פה) לִ֭בּוֹ יָצ֣וּק כְּמוֹ־ אָ֑בֶן וְ֝יָצ֗וּק כְּפֶ֣לַח תַּחְתִּֽית׃ (פו) מִ֭שֵּׂתוֹ יָג֣וּרוּ אֵלִ֑ים מִ֝שְּׁבָרִ֗ים יִתְחַטָּֽאוּ׃ (פז) מַשִּׂיגֵ֣הוּ חֶ֭רֶב בְּלִ֣י תָק֑וּם חֲנִ֖ית מַסָּ֣ע וְשִׁרְיָֽה׃ (פח) יַחְשֹׁ֣ב לְתֶ֣בֶן בַּרְזֶ֑ל לְעֵ֖ץ רִקָּב֣וֹן נְחוּשָֽׁה׃ (פט) לֹֽא־ יַבְרִיחֶ֥נּוּ בֶן־ קָ֑שֶׁת לְ֝קַ֗שׁ נֶהְפְּכוּ־ ל֥וֹ אַבְנֵי־ קָֽלַע׃ (צ) כְּ֭קַשׁ נֶחְשְׁב֣וּ תוֹתָ֑ח וְ֝יִשְׂחַ֗ק לְרַ֣עַשׁ כִּידֽוֹן׃ (צא) תַּ֭חְתָּיו חַדּ֣וּדֵי חָ֑רֶשׂ יִרְפַּ֖ד חָר֣וּץ עֲלֵי־ טִֽיט׃ (צב) יַרְתִּ֣יחַ כַּסִּ֣יר מְצוּלָ֑ה יָ֝֗ם יָשִׂ֥ים כַּמֶּרְקָחָֽה׃ (צג) אַ֭חֲרָיו יָאִ֣יר נָתִ֑יב יַחְשֹׁ֖ב תְּה֣וֹם לְשֵׂיבָֽה׃ (צד) אֵֽין־ עַל־ עָפָ֥ר מָשְׁל֑וֹ הֶ֝עָשׂ֗וּ לִבְלִי־ חָֽת׃ (צה) אֵֽת־ כָּל־ גָּבֹ֥הַּ יִרְאֶ֑ה ה֝֗וּא מֶ֣לֶךְ עַל־ כָּל־ בְּנֵי־ שָֽׁחַץ׃ (צו) וַיַּ֖עַן אִיּ֥וֹב אֶת־ יְהוָ֗ה וַיֹּאמַֽר׃ (צז) (ידעת) יָ֭דַעְתִּי כִּי־ כֹ֣ל תּוּכָ֑ל וְלֹא־ יִבָּצֵ֖ר מִמְּךָ֣ מְזִמָּֽה׃ (צח) מִ֤י זֶ֨ה ׀ מַעְלִ֥ים עֵצָ֗ה בְּֽלִ֫י דָ֥עַת לָכֵ֣ן הִ֭גַּדְתִּי וְלֹ֣א אָבִ֑ין נִפְלָא֥וֹת מִ֝מֶּ֗נִּי וְלֹ֣א אֵדָֽע׃ (צט) שְֽׁמַֽע־ נָ֭א וְאָנֹכִ֣י אֲדַבֵּ֑ר אֶ֝שְׁאָלְךָ֗ וְהוֹדִיעֵֽנִי׃ (ק) לְשֵֽׁמַע־ אֹ֥זֶן שְׁמַעְתִּ֑יךָ וְ֝עַתָּ֗ה עֵינִ֥י רָאָֽתְךָ׃ (קא) עַל־ כֵּ֭ן אֶמְאַ֣ס וְנִחַ֑מְתִּי עַל־ עָפָ֥ר וָאֵֽפֶר׃ (קב) וַיְהִ֗י אַחַ֨ר דִּבֶּ֧ר יְהוָ֛ה אֶת־ הַדְּבָרִ֥ים הָאֵ֖לֶּה אֶל־ אִיּ֑וֹב וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ אֱלִיפַ֣ז הַתֵּֽימָנִ֗י חָרָ֨ה אַפִּ֤י בְךָ֙ וּבִשְׁנֵ֣י רֵעֶ֔יךָ כִּ֠י לֹ֣א דִבַּרְתֶּ֥ם אֵלַ֛י נְכוֹנָ֖ה כְּעַבְדִּ֥י אִיּֽוֹב׃ (קג) וְעַתָּ֡ה קְחֽוּ־ לָכֶ֣ם שִׁבְעָֽה־ פָרִים֩ וְשִׁבְעָ֨ה אֵילִ֜ים וּלְכ֣וּ ׀ אֶל־ עַבְדִּ֣י אִיּ֗וֹב וְהַעֲלִיתֶ֤ם עוֹלָה֙ בַּֽעַדְכֶ֔ם וְאִיּ֣וֹב עַבְדִּ֔י יִתְפַּלֵּ֖ל עֲלֵיכֶ֑ם כִּ֧י אִם־ פָּנָ֣יו אֶשָּׂ֗א לְבִלְתִּ֞י עֲשׂ֤וֹת עִמָּכֶם֙ נְבָלָ֔ה כִּ֠י לֹ֣א דִבַּרְתֶּ֥ם אֵלַ֛י נְכוֹנָ֖ה כְּעַבְדִּ֥י אִיּֽוֹב׃ (קד) וַיֵּלְכוּ֩ אֱלִיפַ֨ז הַתֵּֽימָנִ֜י וּבִלְדַּ֣ד הַשּׁוּחִ֗י צֹפַר֙ הַנַּ֣עֲמָתִ֔י וַֽיַּעֲשׂ֔וּ כַּאֲשֶׁ֛ר דִּבֶּ֥ר אֲלֵיהֶ֖ם יְהוָ֑ה וַיִּשָּׂ֥א יְהוָ֖ה אֶת־ פְּנֵ֥י אִיּֽוֹב׃ (קה) וַֽיהוָ֗ה שָׁ֚ב אֶת־ (שבית) שְׁב֣וּת אִיּ֔וֹב בְּהִֽתְפַּֽלְל֖וֹ בְּעַ֣ד רֵעֵ֑הוּ וַ֧יֹּסֶף יְהוָ֛ה אֶת־ כָּל־ אֲשֶׁ֥ר לְאִיּ֖וֹב לְמִשְׁנֶֽה׃ (קו) וַיָּבֹ֣אוּ אֵ֠לָיו כָּל־ אֶחָ֨יו וְכָל־ (אחיתיו) אַחְיוֹתָ֜יו וְכָל־ יֹדְעָ֣יו לְפָנִ֗ים וַיֹּאכְל֨וּ עִמּ֣וֹ לֶחֶם֮ בְּבֵיתוֹ֒ וַיָּנֻ֤דוּ לוֹ֙ וַיְנַחֲמ֣וּ אֹת֔וֹ עַ֚ל כָּל־ הָ֣רָעָ֔ה אֲשֶׁר־ הֵבִ֥יא יְהוָ֖ה עָלָ֑יו וַיִּתְּנוּ־ ל֗וֹ אִ֚ישׁ קְשִׂיטָ֣ה אֶחָ֔ת וְאִ֕ישׁ נֶ֥זֶם זָהָ֖ב אֶחָֽד׃ (קז) וַֽיהוָ֗ה בֵּרַ֛ךְ אֶת־ אַחֲרִ֥ית אִיּ֖וֹב מֵרֵאשִׁת֑וֹ וַֽיְהִי־ ל֡וֹ אַרְבָּעָה֩ עָשָׂ֨ר אֶ֜לֶף צֹ֗אן וְשֵׁ֤שֶׁת אֲלָפִים֙ גְּמַלִּ֔ים וְאֶֽלֶף־ צֶ֥מֶד בָּקָ֖ר וְאֶ֥לֶף אֲתוֹנֽוֹת׃ (קח) וַֽיְהִי־ ל֛וֹ שִׁבְעָ֥נָה בָנִ֖ים וְשָׁל֥וֹשׁ בָּנֽוֹת׃ (קט) וַיִּקְרָ֤א שֵׁם־ הָֽאַחַת֙ יְמִימָ֔ה וְשֵׁ֥ם הַשֵּׁנִ֖ית קְצִיעָ֑ה וְשֵׁ֥ם הַשְּׁלִישִׁ֖ית קֶ֥רֶן הַפּֽוּךְ׃ (קי) וְלֹ֨א נִמְצָ֜א נָשִׁ֥ים יָפ֛וֹת כִּבְנ֥וֹת אִיּ֖וֹב בְּכָל־ הָאָ֑רֶץ וַיִּתֵּ֨ן לָהֶ֧ם אֲבִיהֶ֛ם נַחֲלָ֖ה בְּת֥וֹךְ אֲחֵיהֶֽם׃ (קיא) וַיְחִ֤י אִיּוֹב֙ אַֽחֲרֵי־ זֹ֔את מֵאָ֥ה וְאַרְבָּעִ֖ים שָׁנָ֑ה (וירא) וַיִּרְאֶ֗ה אֶת־ בָּנָיו֙ וְאֶת־ בְּנֵ֣י בָנָ֔יו אַרְבָּעָ֖ה דֹּרֽוֹת׃ (קיב) וַיָּ֣מָת אִיּ֔וֹב זָקֵ֖ן וּשְׂבַ֥ע יָמִֽים׃
