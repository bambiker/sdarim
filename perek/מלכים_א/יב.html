מלכים א פרק יב לפי החלוקה לסדרים

פרק יב בספר מלכים א (33 פסוקים) נמצא במלכים סדר י (פסוקים יז–לט) ובמלכים סדר יא (פסוקים א–י).

מלכים סדר י

(א) וְהָאִ֥ישׁ יָרָבְעָ֖ם גִּבּ֣וֹר חָ֑יִל וַיַּ֨רְא שְׁלֹמֹ֜ה אֶת־ הַנַּ֗עַר כִּֽי־ עֹשֵׂ֤ה מְלָאכָה֙ ה֔וּא וַיַּפְקֵ֣ד אֹת֔וֹ לְכָל־ סֵ֖בֶל בֵּ֥ית יוֹסֵֽף׃ (ב) וַֽיְהִי֙ בָּעֵ֣ת הַהִ֔יא וְיָֽרָבְעָ֖ם יָצָ֣א מִירוּשָׁלִָ֑ם וַיִּמְצָ֣א אֹת֡וֹ אֲחִיָּה֩ הַשִּׁילֹנִ֨י הַנָּבִ֜יא בַּדֶּ֗רֶךְ וְה֤וּא מִתְכַּסֶּה֙ בְּשַׂלְמָ֣ה חֲדָשָׁ֔ה וּשְׁנֵיהֶ֥ם לְבַדָּ֖ם בַּשָּׂדֶֽה׃ (ג) וַיִּתְפֹּ֣שׂ אֲחִיָּ֔ה בַּשַּׂלְמָ֥ה הַחֲדָשָׁ֖ה אֲשֶׁ֣ר עָלָ֑יו וַיִּ֨קְרָעֶ֔הָ שְׁנֵ֥ים עָשָׂ֖ר קְרָעִֽים׃ (ד) וַיֹּ֙אמֶר֙ לְיָֽרָבְעָ֔ם קַח־ לְךָ֖ עֲשָׂרָ֣ה קְרָעִ֑ים כִּ֣י כֹה֩ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֣י יִשְׂרָאֵ֗ל הִנְנִ֨י קֹרֵ֤עַ אֶת־ הַמַּמְלָכָה֙ מִיַּ֣ד שְׁלֹמֹ֔ה וְנָתַתִּ֣י לְךָ֔ אֵ֖ת עֲשָׂרָ֥ה הַשְּׁבָטִֽים׃ (ה) וְהַשֵּׁ֥בֶט הָאֶחָ֖ד יִֽהְיֶה־ לּ֑וֹ לְמַ֣עַן ׀ עַבְדִּ֣י דָוִ֗ד וּלְמַ֙עַן֙ יְר֣וּשָׁלִַ֔ם הָעִיר֙ אֲשֶׁ֣ר בָּחַ֣רְתִּי בָ֔הּ מִכֹּ֖ל שִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (ו) יַ֣עַן ׀ אֲשֶׁ֣ר עֲזָב֗וּנִי וַיִּֽשְׁתַּחֲווּ֮ לְעַשְׁתֹּרֶת֮ אֱלֹהֵ֣י צִֽדֹנִין֒ לִכְמוֹשׁ֙ אֱלֹהֵ֣י מוֹאָ֔ב וּלְמִלְכֹּ֖ם אֱלֹהֵ֣י בְנֵֽי־ עַמּ֑וֹן וְלֹֽא־ הָלְכ֣וּ בִדְרָכַ֗י לַעֲשׂ֨וֹת הַיָּשָׁ֧ר בְּעֵינַ֛י וְחֻקֹּתַ֥י וּמִשְׁפָּטַ֖י כְּדָוִ֥ד אָבִֽיו׃ (ז) וְלֹֽא־ אֶקַּ֥ח אֶת־ כָּל־ הַמַּמְלָכָ֖ה מִיָּד֑וֹ כִּ֣י ׀ נָשִׂ֣יא אֲשִׁתֶ֗נּוּ כֹּ֚ל יְמֵ֣י חַיָּ֔יו לְמַ֨עַן דָּוִ֤ד עַבְדִּי֙ אֲשֶׁ֣ר בָּחַ֣רְתִּי אֹת֔וֹ אֲשֶׁ֥ר שָׁמַ֖ר מִצְוֺתַ֥י וְחֻקֹּתָֽי׃ (ח) וְלָקַחְתִּ֥י הַמְּלוּכָ֖ה מִיַּ֣ד בְּנ֑וֹ וּנְתַתִּ֣יהָ לְּךָ֔ אֵ֖ת עֲשֶׂ֥רֶת הַשְּׁבָטִֽים׃ (ט) וְלִבְנ֖וֹ אֶתֵּ֣ן שֵֽׁבֶט־ אֶחָ֑ד לְמַ֣עַן הֱיֽוֹת־ נִ֣יר לְדָֽוִיד־ עַ֠בְדִּי כָּֽל־ הַיָּמִ֤ים ׀ לְפָנַי֙ בִּיר֣וּשָׁלִַ֔ם הָעִיר֙ אֲשֶׁ֣ר בָּחַ֣רְתִּי לִ֔י לָשׂ֥וּם שְׁמִ֖י שָֽׁם׃ (י) וְאֹתְךָ֣ אֶקַּ֔ח וּמָ֣לַכְתָּ֔ בְּכֹ֥ל אֲשֶׁר־ תְּאַוֶּ֖ה נַפְשֶׁ֑ךָ וְהָיִ֥יתָ מֶּ֖לֶךְ עַל־ יִשְׂרָאֵֽל׃ (יא) וְהָיָ֗ה אִם־ תִּשְׁמַע֮ אֶת־ כָּל־ אֲשֶׁ֣ר אֲצַוֶּךָ֒ וְהָלַכְתָּ֣ בִדְרָכַ֗י וְעָשִׂ֨יתָ הַיָּשָׁ֤ר בְּעֵינַי֙ לִשְׁמ֤וֹר חֻקּוֹתַי֙ וּמִצְוֺתַ֔י כַּאֲשֶׁ֥ר עָשָׂ֖ה דָּוִ֣ד עַבְדִּ֑י וְהָיִ֣יתִי עִמָּ֗ךְ וּבָנִ֨יתִי לְךָ֤ בַֽיִת־ נֶאֱמָן֙ כַּאֲשֶׁ֣ר בָּנִ֣יתִי לְדָוִ֔ד וְנָתַתִּ֥י לְךָ֖ אֶת־ יִשְׂרָאֵֽל׃ (יב) וַֽאעַנֶּ֛ה אֶת־ זֶ֥רַע דָּוִ֖ד לְמַ֣עַן זֹ֑את אַ֖ךְ לֹ֥א כָל־ הַיָּמִֽים׃ (יג) וַיְבַקֵּ֥שׁ שְׁלֹמֹ֖ה לְהָמִ֣ית אֶת־ יָרָבְעָ֑ם וַיָּ֣קָם יָרָבְעָ֗ם וַיִּבְרַ֤ח מִצְרַ֙יִם֙ אֶל־ שִׁישַׁ֣ק מֶֽלֶךְ־ מִצְרַ֔יִם וַיְהִ֥י בְמִצְרַ֖יִם עַד־ מ֥וֹת שְׁלֹמֹֽה׃ (יד) וְיֶ֨תֶר דִּבְרֵ֧י שְׁלֹמֹ֛ה וְכָל־ אֲשֶׁ֥ר עָשָׂ֖ה וְחָכְמָת֑וֹ הֲלֽוֹא־ הֵ֣ם כְּתֻבִ֔ים עַל־ סֵ֖פֶר דִּבְרֵ֥י שְׁלֹמֹֽה׃ (טו) וְהַיָּמִ֗ים אֲשֶׁר֩ מָלַ֨ךְ שְׁלֹמֹ֤ה בִירוּשָׁלִַ֙ם֙ עַל־ כָּל־ יִשְׂרָאֵ֔ל אַרְבָּעִ֖ים שָׁנָֽה׃ (טז) וַיִּשְׁכַּ֤ב שְׁלֹמֹה֙ עִם־ אֲבֹתָ֔יו וַיִּ֨קָּבֵ֔ר בְּעִ֖יר דָּוִ֣ד אָבִ֑יו וַיִּמְלֹ֛ךְ רְחַבְעָ֥ם בְּנ֖וֹ תַּחְתָּֽיו׃ תחילת פרק יב (יז) וַיֵּ֥לֶךְ רְחַבְעָ֖ם שְׁכֶ֑ם כִּ֥י שְׁכֶ֛ם בָּ֥א כָל־ יִשְׂרָאֵ֖ל לְהַמְלִ֥יךְ אֹתֽוֹ׃ (יח) וַיְהִ֞י כִּשְׁמֹ֣עַ ׀ יָרָבְעָ֣ם בֶּן־ נְבָ֗ט וְהוּא֙ עוֹדֶ֣נּוּ בְמִצְרַ֔יִם אֲשֶׁ֣ר בָּרַ֔ח מִפְּנֵ֖י הַמֶּ֣לֶךְ שְׁלֹמֹ֑ה וַיֵּ֥שֶׁב יָרָבְעָ֖ם בְּמִצְרָֽיִם׃ (יט) וַֽיִּשְׁלְחוּ֙ וַיִּקְרְאוּ־ ל֔וֹ (ויבאו) וַיָּבֹ֥א יָרָבְעָ֖ם וְכָל־ קְהַ֣ל יִשְׂרָאֵ֑ל וַֽיְדַבְּר֔וּ אֶל־ רְחַבְעָ֖ם לֵאמֹֽר׃ (כ) אָבִ֖יךָ הִקְשָׁ֣ה אֶת־ עֻלֵּ֑נוּ וְאַתָּ֡ה עַתָּ֣ה הָקֵל֩ מֵעֲבֹדַ֨ת אָבִ֜יךָ הַקָּשָׁ֗ה וּמֵעֻלּ֧וֹ הַכָּבֵ֛ד אֲשֶׁר־ נָתַ֥ן עָלֵ֖ינוּ וְנַעַבְדֶֽךָּ׃ (כא) וַיֹּ֣אמֶר אֲלֵיהֶ֗ם לְכ֥וּ עֹ֛ד שְׁלֹשָׁ֥ה יָמִ֖ים וְשׁ֣וּבוּ אֵלָ֑י וַיֵּלְכ֖וּ הָעָֽם׃ (כב) וַיִּוָּעַ֞ץ הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ הַזְּקֵנִים֙ אֲשֶׁר־ הָי֣וּ עֹמְדִ֗ים אֶת־ פְּנֵי֙ שְׁלֹמֹ֣ה אָבִ֔יו בִּֽהְיֹת֥וֹ חַ֖י לֵאמֹ֑ר אֵ֚יךְ אַתֶּ֣ם נֽוֹעָצִ֔ים לְהָשִׁ֥יב אֶת־ הָֽעָם־ הַזֶּ֖ה דָּבָֽר׃ (כג) (וידבר) וַיְדַבְּר֨וּ אֵלָ֜יו לֵאמֹ֗ר אִם־ הַ֠יּוֹם תִּֽהְיֶה־ עֶ֜בֶד לָעָ֤ם הַזֶּה֙ וַֽעֲבַדְתָּ֔ם וַעֲנִיתָ֕ם וְדִבַּרְתָּ֥ אֲלֵיהֶ֖ם דְּבָרִ֣ים טוֹבִ֑ים וְהָי֥וּ לְךָ֛ עֲבָדִ֖ים כָּל־ הַיָּמִֽים׃ (כד) וַֽיַּעֲזֹ֛ב אֶת־ עֲצַ֥ת הַזְּקֵנִ֖ים אֲשֶׁ֣ר יְעָצֻ֑הוּ וַיִּוָּעַ֗ץ אֶת־ הַיְלָדִים֙ אֲשֶׁ֣ר גָּדְל֣וּ אִתּ֔וֹ אֲשֶׁ֥ר הָעֹמְדִ֖ים לְפָנָֽיו׃ (כה) וַיֹּ֣אמֶר אֲלֵיהֶ֗ם מָ֚ה אַתֶּ֣ם נֽוֹעָצִ֔ים וְנָשִׁ֥יב דָּבָ֖ר אֶת־ הָעָ֣ם הַזֶּ֑ה אֲשֶׁ֨ר דִּבְּר֤וּ אֵלַי֙ לֵאמֹ֔ר הָקֵל֙ מִן־ הָעֹ֔ל אֲשֶׁר־ נָתַ֥ן אָבִ֖יךָ עָלֵֽינוּ׃ (כו) וַיְדַבְּר֣וּ אֵלָ֗יו הַיְלָדִים֙ אֲשֶׁ֨ר גָּדְל֣וּ אִתּוֹ֮ לֵאמֹר֒ כֹּֽה־ תֹאמַ֣ר לָעָ֣ם הַזֶּ֡ה אֲשֶׁר֩ דִּבְּר֨וּ אֵלֶ֜יךָ לֵאמֹ֗ר אָבִ֙יךָ֙ הִכְבִּ֣יד אֶת־ עֻלֵּ֔נוּ וְאַתָּ֖ה הָקֵ֣ל מֵעָלֵ֑ינוּ כֹּ֚ה תְּדַבֵּ֣ר אֲלֵיהֶ֔ם קָֽטָנִּ֥י עָבָ֖ה מִמָּתְנֵ֥י אָבִֽי׃ (כז) וְעַתָּ֗ה אָבִי֙ הֶעְמִ֤יס עֲלֵיכֶם֙ עֹ֣ל כָּבֵ֔ד וַאֲנִ֖י אוֹסִ֣יף עַֽל־ עֻלְּכֶ֑ם אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֕י אֲיַסֵּ֥ר אֶתְכֶ֖ם בָּעַקְרַבִּֽים׃ (כח) (ויבו) וַיָּב֨וֹא יָרָבְעָ֧ם וְכָל־ הָעָ֛ם אֶל־ רְחַבְעָ֖ם בַּיּ֣וֹם הַשְּׁלִישִׁ֑י כַּאֲשֶׁ֨ר דִּבֶּ֤ר הַמֶּ֙לֶךְ֙ לֵאמֹ֔ר שׁ֥וּבוּ אֵלַ֖י בַּיּ֥וֹם הַשְּׁלִישִֽׁי׃ (כט) וַיַּ֧עַן הַמֶּ֛לֶךְ אֶת־ הָעָ֖ם קָשָׁ֑ה וַֽיַּעֲזֹ֛ב אֶת־ עֲצַ֥ת הַזְּקֵנִ֖ים אֲשֶׁ֥ר יְעָצֻֽהוּ׃ (ל) וַיְדַבֵּ֣ר אֲלֵיהֶ֗ם כַּעֲצַ֤ת הַיְלָדִים֙ לֵאמֹ֔ר אָבִי֙ הִכְבִּ֣יד אֶֽת־ עֻלְּכֶ֔ם וַאֲנִ֖י אֹסִ֣יף עַֽל־ עֻלְּכֶ֑ם אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֕י אֲיַסֵּ֥ר אֶתְכֶ֖ם בָּעַקְרַבִּֽים׃ (לא) וְלֹֽא־ שָׁמַ֥ע הַמֶּ֖לֶךְ אֶל־ הָעָ֑ם כִּֽי־ הָיְתָ֤ה סִבָּה֙ מֵעִ֣ם יְהוָ֔ה לְמַ֜עַן הָקִ֣ים אֶת־ דְּבָר֗וֹ אֲשֶׁ֨ר דִּבֶּ֤ר יְהוָה֙ בְּיַד֙ אֲחִיָּ֣ה הַשִּׁילֹנִ֔י אֶל־ יָרָבְעָ֖ם בֶּן־ נְבָֽט׃ (לב) וַיַּ֣רְא כָּל־ יִשְׂרָאֵ֗ל כִּ֠י לֹֽא־ שָׁמַ֣ע הַמֶּלֶךְ֮ אֲלֵיהֶם֒ וַיָּשִׁ֣בוּ הָעָ֣ם אֶת־ הַמֶּ֣לֶךְ דָּבָ֣ר ׀ לֵאמֹ֡ר מַה־ לָּנוּ֩ חֵ֨לֶק בְּדָוִ֜ד וְלֹֽא־ נַחֲלָ֣ה בְּבֶן־ יִשַׁ֗י לְאֹהָלֶ֙יךָ֙ יִשְׂרָאֵ֔ל עַתָּ֕ה רְאֵ֥ה בֵיתְךָ֖ דָּוִ֑ד וַיֵּ֥לֶךְ יִשְׂרָאֵ֖ל לְאֹהָלָֽיו׃ (לג) וּבְנֵ֣י יִשְׂרָאֵ֔ל הַיֹּשְׁבִ֖ים בְּעָרֵ֣י יְהוּדָ֑ה וַיִּמְלֹ֥ךְ עֲלֵיהֶ֖ם רְחַבְעָֽם׃ (לד) וַיִּשְׁלַ֞ח הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ אֲדֹרָם֙ אֲשֶׁ֣ר עַל־ הַמַּ֔ס וַיִּרְגְּמ֨וּ כָל־ יִשְׂרָאֵ֥ל בּ֛וֹ אֶ֖בֶן וַיָּמֹ֑ת לְחַטָּ֑את וַיֵּלְכ֥וּ הָעָ֛ם לִפְנֵ֥י הָאֶחָ֖ד עַד־ דָּֽן׃ (ח) וַיַּ֖עַשׂ אֶת־ בֵּ֣ית בָּמ֑וֹת וַיַּ֤עַשׂ כֹּֽהֲנִים֙ מִקְצ֣וֹת הָעָ֔ם אֲשֶׁ֥ר לֹֽא־ הָי֖וּ מִבְּנֵ֥י לֵוִֽי׃ (ט) וַיַּ֣עַשׂ יָרָבְעָ֣ם וַיֹּ֣אמֶר אֵלָ֗יו הַאַתָּ֧ה אִישׁ־ הָאֱלֹהִ֛ים אֲשֶׁר־ בָּ֥אתָ מִֽיהוּדָ֖ה וַיֹּ֥אמֶר אָֽנִי׃ (כה) וַיֹּ֣אמֶר אֵלָ֔יו לֵ֥ךְ אִתִּ֖י הַבָּ֑יְתָה וֶאֱכֹ֖ל לָֽחֶם׃ (כו) וַיֹּ֗אמֶר לֹ֥א אוּכַ֛ל לָשׁ֥וּב אִתָּ֖ךְ וְלָב֣וֹא אִתָּ֑ךְ וְלֹֽא־ אֹ֣כַל לֶ֗חֶם וְלֹֽא־ אֶשְׁתֶּ֤ה אִתְּךָ֙ מַ֔יִם בַּמָּק֖וֹם הַזֶּֽה׃ (כז) כִּֽי־ דָבָ֤ר אֵלַי֙ בִּדְבַ֣ר יְהוָ֔ה לֹֽא־ תֹאכַ֣ל לֶ֔חֶם וְלֹֽא־ תִשְׁתֶּ֥ה שָׁ֖ם מָ֑יִם לֹא־ תָשׁ֣וּב לָלֶ֔כֶת בַּדֶּ֖רֶךְ אֲשֶׁר־ הָלַ֥כְתָּ בָּֽהּ׃ (כח) וַיֹּ֣אמֶר ל֗וֹ גַּם־ אֲנִ֣י נָבִיא֮ כָּמוֹךָ֒ וּמַלְאָ֡ךְ דִּבֶּ֣ר אֵלַי֩ בִּדְבַ֨ר יְהוָ֜ה לֵאמֹ֗ר הֲשִׁבֵ֤הוּ אִתְּךָ֙ אֶל־ בֵּיתֶ֔ךָ וְיֹ֥אכַל לֶ֖חֶם וְיֵ֣שְׁתְּ מָ֑יִם כִּחֵ֖שׁ לֽוֹ׃ (כט) וַיָּ֣שָׁב אִתּ֗וֹ וַיֹּ֥אכַל לֶ֛חֶם בְּבֵית֖וֹ וַיֵּ֥שְׁתְּ מָֽיִם׃ (ל) וַיְהִ֕י הֵ֥ם יֹשְׁבִ֖ים אֶל־ הַשֻּׁלְחָ֑ן וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ הַנָּבִ֖יא אֲשֶׁ֥ר הֱשִׁיבֽוֹ׃ (לא) וַיִּקְרָ֞א אֶל־ אִ֣ישׁ הָאֱלֹהִ֗ים אֲשֶׁר־ בָּ֤א מִֽיהוּדָה֙ לֵאמֹ֔ר כֹּ֖ה אָמַ֣ר יְהוָ֑ה יַ֗עַן כִּ֤י מָרִ֙יתָ֙ פִּ֣י יְהוָ֔ה וְלֹ֤א שָׁמַ֙רְתָּ֙ אֶת־ הַמִּצְוָ֔ה אֲשֶׁ֥ר צִוְּךָ֖ יְהוָ֥ה אֱלֹהֶֽיךָ׃ (לב) וַתָּ֗שָׁב וַתֹּ֤אכַל לֶ֙חֶם֙ וַתֵּ֣שְׁתְּ מַ֔יִם בַּמָּקוֹם֙ אֲשֶׁ֣ר דִּבֶּ֣ר אֵלֶ֔יךָ אַל־ תֹּ֥אכַל לֶ֖חֶם וְאַל־ תֵּ֣שְׁתְּ מָ֑יִם לֹֽא־ תָב֥וֹא נִבְלָתְךָ֖ אֶל־ קֶ֥בֶר אֲבֹתֶֽיךָ׃ (לג) וַיְהִ֗י אַחֲרֵ֛י אָכְל֥וֹ לֶ֖חֶם וְאַחֲרֵ֣י שְׁתוֹת֑וֹ וַיַּחֲבָשׁ־ ל֣וֹ הַחֲמ֔וֹר לַנָּבִ֖יא אֲשֶׁ֥ר הֱשִׁיבֽוֹ׃ (לד) וַיֵּ֕לֶךְ וַיִּמְצָאֵ֧הוּ אַרְיֵ֛ה בַּדֶּ֖רֶךְ וַיְמִיתֵ֑הוּ וַתְּהִ֤י נִבְלָתוֹ֙ מֻשְׁלֶ֣כֶת בַּדֶּ֔רֶךְ וְהַחֲמוֹר֙ עֹמֵ֣ד אֶצְלָ֔הּ וְהָ֣אַרְיֵ֔ה עֹמֵ֖ד אֵ֥צֶל הַנְּבֵלָֽה׃ (לה) וְהִנֵּ֧ה אֲנָשִׁ֣ים עֹבְרִ֗ים וַיִּרְא֤וּ אֶת־ הַנְּבֵלָה֙ מֻשְׁלֶ֣כֶת בַּדֶּ֔רֶךְ וְאֶת־ הָ֣אַרְיֵ֔ה עֹמֵ֖ד אֵ֣צֶל הַנְּבֵלָ֑ה וַיָּבֹ֙אוּ֙ וַיְדַבְּר֣וּ בָעִ֔יר אֲשֶׁ֛ר הַנָּבִ֥יא הַזָּקֵ֖ן יֹשֵׁ֥ב בָּֽהּ׃ (לו) וַיִּשְׁמַ֣ע הַנָּבִיא֮ אֲשֶׁ֣ר הֱשִׁיב֣וֹ מִן־ הַדֶּרֶךְ֒ וַיֹּ֙אמֶר֙ אִ֣ישׁ הָאֱלֹהִ֣ים ה֔וּא אֲשֶׁ֥ר מָרָ֖ה אֶת־ פִּ֣י יְהוָ֑ה וַיִּתְּנֵ֨הוּ יְהוָ֜ה לָאַרְיֵ֗ה וַֽיִּשְׁבְּרֵ֙הוּ֙ וַיְמִתֵ֔הוּ כִּדְבַ֥ר יְהוָ֖ה אֲשֶׁ֥ר דִּבֶּר־ לֽוֹ׃ (לז) וַיְדַבֵּ֤ר אֶל־ בָּנָיו֙ לֵאמֹ֔ר חִבְשׁוּ־ לִ֖י אֶֽת־ הַחֲמ֑וֹר וַֽיַּחֲבֹֽשׁוּ׃ (לח) וַיֵּ֗לֶךְ וַיִּמְצָ֤א אֶת־ נִבְלָתוֹ֙ מֻשְׁלֶ֣כֶת בַּדֶּ֔רֶךְ וַֽחֲמוֹר֙ וְהָ֣אַרְיֵ֔ה עֹמְדִ֖ים אֵ֣צֶל הַנְּבֵלָ֑ה לֹֽא־ אָכַ֤ל הָֽאַרְיֵה֙ אֶת־ הַנְּבֵלָ֔ה וְלֹ֥א שָׁבַ֖ר אֶֽת־ הַחֲמֽוֹר׃ (לט) וַיִּשָּׂ֨א הַנָּבִ֜יא אֶת־ נִבְלַ֧ת אִישׁ־ הָאֱלֹהִ֛ים וַיַּנִּחֵ֥הוּ אֶֽל־ הַחֲמ֖וֹר וַיְשִׁיבֵ֑הוּ וַיָּבֹ֗א אֶל־ עִיר֙ הַנָּבִ֣יא הַזָּקֵ֔ן לִסְפֹּ֖ד וּלְקָבְרֽוֹ׃ (מ) וַיַּנַּ֥ח אֶת־ נִבְלָת֖וֹ בְּקִבְר֑וֹ וַיִּסְפְּד֥וּ עָלָ֖יו ה֥וֹי אָחִֽי׃
