מלכים א פרק ד לפי החלוקה לסדרים

פרק ד בספר מלכים א (20 פסוקים) נמצא במלכים סדר ג (פסוקים לא–מט) ובמלכים סדר ד (פסוקים א–א).

מלכים סדר ג

(א) וְהַמֶּ֥לֶךְ שְׁלֹמֹ֖ה בָּר֑וּךְ וְכִסֵּ֣א דָוִ֗ד יִהְיֶ֥ה נָכ֛וֹן לִפְנֵ֥י יְהוָ֖ה עַד־ עוֹלָֽם׃ (ב) וַיְצַ֣ו הַמֶּ֗לֶךְ אֶת־ בְּנָיָ֙הוּ֙ בֶּן־ יְה֣וֹיָדָ֔ע וַיֵּצֵ֕א וַיִּפְגַּע־ בּ֖וֹ וַיָּמֹ֑ת וְהַמַּמְלָכָ֥ה נָכ֖וֹנָה בְּיַד־ שְׁלֹמֹֽה׃ (ג) וַיִּתְחַתֵּ֣ן שְׁלֹמֹ֔ה אֶת־ פַּרְעֹ֖ה מֶ֣לֶךְ מִצְרָ֑יִם וַיִּקַּ֣ח אֶת־ בַּת־ פַּרְעֹ֗ה וַיְבִיאֶ֙הָ֙ אֶל־ עִ֣יר דָּוִ֔ד עַ֣ד כַּלֹּת֗וֹ לִבְנ֤וֹת אֶת־ בֵּיתוֹ֙ וְאֶת־ בֵּ֣ית יְהוָ֔ה וְאֶת־ חוֹמַ֥ת יְרוּשָׁלִַ֖ם סָבִֽיב׃ (ד) רַ֣ק הָעָ֔ם מְזַבְּחִ֖ים בַּבָּמ֑וֹת כִּ֠י לֹא־ נִבְנָ֥ה בַ֙יִת֙ לְשֵׁ֣ם יְהוָ֔ה עַ֖ד הַיָּמִ֥ים הָהֵֽם׃ (ה) וַיֶּאֱהַ֤ב שְׁלֹמֹה֙ אֶת־ יְהוָ֔ה לָלֶ֕כֶת בְּחֻקּ֖וֹת דָּוִ֣ד אָבִ֑יו רַ֚ק בַּבָּמ֔וֹת ה֥וּא מְזַבֵּ֖חַ וּמַקְטִֽיר׃ (ו) וַיֵּ֨לֶךְ הַמֶּ֤לֶךְ גִּבְעֹ֙נָה֙ לִזְבֹּ֣חַ שָׁ֔ם כִּ֥י הִ֖יא הַבָּמָ֣ה הַגְּדוֹלָ֑ה אֶ֤לֶף עֹלוֹת֙ יַעֲלֶ֣ה שְׁלֹמֹ֔ה עַ֖ל הַמִּזְבֵּ֥חַ הַהֽוּא׃ (ז) בְּגִבְע֗וֹן נִרְאָ֧ה יְהֹוָ֛ה אֶל־ שְׁלֹמֹ֖ה בַּחֲל֣וֹם הַלָּ֑יְלָה וַיֹּ֣אמֶר אֱלֹהִ֔ים שְׁאַ֖ל מָ֥ה אֶתֶּן־ לָֽךְ׃ (ח) וַיֹּ֣אמֶר שְׁלֹמֹ֗ה אַתָּ֨ה עָשִׂ֜יתָ עִם־ עַבְדְּךָ֙ דָוִ֣ד אָבִי֮ חֶ֣סֶד גָּדוֹל֒ כַּאֲשֶׁר֩ הָלַ֨ךְ לְפָנֶ֜יךָ בֶּאֱמֶ֧ת וּבִצְדָקָ֛ה וּבְיִשְׁרַ֥ת לֵבָ֖ב עִמָּ֑ךְ וַתִּשְׁמָר־ ל֗וֹ אֶת־ הַחֶ֤סֶד הַגָּדוֹל֙ הַזֶּ֔ה וַתִּתֶּן־ ל֥וֹ בֵ֛ן יֹשֵׁ֥ב עַל־ כִּסְא֖וֹ כַּיּ֥וֹם הַזֶּֽה׃ (ט) וְעַתָּה֙ יְהוָ֣ה אֱלֹהָ֔י אַתָּה֙ הִמְלַ֣כְתָּ אֶֽת־ עַבְדְּךָ֔ תַּ֖חַת דָּוִ֣ד אָבִ֑י וְאָֽנֹכִי֙ נַ֣עַר קָטֹ֔ן לֹ֥א אֵדַ֖ע צֵ֥את וָבֹֽא׃ (י) וְעַ֨בְדְּךָ֔ בְּת֥וֹךְ עַמְּךָ֖ אֲשֶׁ֣ר בָּחָ֑רְתָּ עַם־ רָ֕ב אֲשֶׁ֧ר לֹֽא־ יִמָּנֶ֛ה וְלֹ֥א יִסָּפֵ֖ר מֵרֹֽב׃ (יא) וְנָתַתָּ֨ לְעַבְדְּךָ֜ לֵ֤ב שֹׁמֵ֙עַ֙ לִשְׁפֹּ֣ט אֶֽת־ עַמְּךָ֔ לְהָבִ֖ין בֵּֽין־ ט֣וֹב לְרָ֑ע כִּ֣י מִ֤י יוּכַל֙ לִשְׁפֹּ֔ט אֶת־ עַמְּךָ֥ הַכָּבֵ֖ד הַזֶּֽה׃ (יב) וַיִּיטַ֥ב הַדָּבָ֖ר בְּעֵינֵ֣י אֲדֹנָ֑י כִּ֚י שָׁאַ֣ל שְׁלֹמֹ֔ה אֶת־ הַדָּבָ֖ר הַזֶּֽה׃ (יג) וַיֹּ֨אמֶר אֱלֹהִ֜ים אֵלָ֗יו יַעַן֩ אֲשֶׁ֨ר שָׁאַ֜לְתָּ אֶת־ הַדָּבָ֣ר הַזֶּ֗ה וְלֹֽא־ שָׁאַ֨לְתָּ לְּךָ֜ יָמִ֣ים רַבִּ֗ים וְלֹֽא־ שָׁאַ֤לְתָּ לְּךָ֙ עֹ֔שֶׁר וְלֹ֥א שָׁאַ֖לְתָּ נֶ֣פֶשׁ אֹיְבֶ֑יךָ וְשָׁאַ֧לְתָּ לְּךָ֛ הָבִ֖ין לִשְׁמֹ֥עַ מִשְׁפָּֽט׃ (יד) הִנֵּ֥ה עָשִׂ֖יתִי כִּדְבָרֶ֑יךָ הִנֵּ֣ה ׀ נָתַ֣תִּי לְךָ֗ לֵ֚ב חָכָ֣ם וְנָב֔וֹן אֲשֶׁ֤ר כָּמ֙וֹךָ֙ לֹא־ הָיָ֣ה לְפָנֶ֔יךָ וְאַחֲרֶ֖יךָ לֹא־ יָק֥וּם כָּמֽוֹךָ׃ (טו) וְגַ֨ם אֲשֶׁ֤ר לֹֽא־ שָׁאַ֙לְתָּ֙ נָתַ֣תִּי לָ֔ךְ גַּם־ עֹ֖שֶׁר גַּם־ כָּב֑וֹד אֲ֠שֶׁר לֹא־ הָיָ֨ה כָמ֥וֹךָ אִ֛ישׁ בַּמְּלָכִ֖ים כָּל־ יָמֶֽיךָ׃ (טז) וְאִ֣ם ׀ תֵּלֵ֣ךְ בִּדְרָכַ֗י לִשְׁמֹ֤ר חֻקַּי֙ וּמִצְוֺתַ֔י כַּאֲשֶׁ֥ר הָלַ֖ךְ דָּוִ֣יד אָבִ֑יךָ וְהַאַרַכְתִּ֖י אֶת־ יָמֶֽיךָ׃ (יז) וַיִּקַ֥ץ שְׁלֹמֹ֖ה וְהִנֵּ֣ה חֲל֑וֹם וַיָּב֨וֹא יְרוּשָׁלִַ֜ם וַֽיַּעֲמֹ֣ד ׀ לִפְנֵ֣י ׀ אֲר֣וֹן בְּרִית־ אֲדֹנָ֗י וַיַּ֤עַל עֹלוֹת֙ וַיַּ֣עַשׂ שְׁלָמִ֔ים וַיַּ֥עַשׂ מִשְׁתֶּ֖ה לְכָל־ עֲבָדָֽיו׃ (יח) אָ֣ז תָּבֹ֗אנָה שְׁתַּ֛יִם נָשִׁ֥ים זֹנ֖וֹת אֶל־ הַמֶּ֑לֶךְ וַֽתַּעֲמֹ֖דְנָה לְפָנָֽיו׃ (יט) וַתֹּ֜אמֶר הָאִשָּׁ֤ה הָֽאַחַת֙ בִּ֣י אֲדֹנִ֔י אֲנִי֙ וְהָאִשָּׁ֣ה הַזֹּ֔את יֹשְׁבֹ֖ת בְּבַ֣יִת אֶחָ֑ד וָאֵלֵ֥ד עִמָּ֖הּ בַּבָּֽיִת׃ (כ) וַיְהִ֞י בַּיּ֤וֹם הַשְּׁלִישִׁי֙ לְלִדְתִּ֔י וַתֵּ֖לֶד גַּם־ הָאִשָּׁ֣ה הַזֹּ֑את וַאֲנַ֣חְנוּ יַחְדָּ֗ו אֵֽין־ זָ֤ר אִתָּ֙נוּ֙ בַּבַּ֔יִת זוּלָתִ֥י שְׁתַּֽיִם־ אֲנַ֖חְנוּ בַּבָּֽיִת׃ (כא) וַיָּ֛מָת בֶּן־ הָאִשָּׁ֥ה הַזֹּ֖את לָ֑יְלָה אֲשֶׁ֥ר שָׁכְבָ֖ה עָלָֽיו׃ (כב) וַתָּקָם֩ בְּת֨וֹךְ הַלַּ֜יְלָה וַתִּקַּ֧ח אֶת־ בְּנִ֣י מֵֽאֶצְלִ֗י וַאֲמָֽתְךָ֙ יְשֵׁנָ֔ה וַתַּשְׁכִּיבֵ֖הוּ בְּחֵיקָ֑הּ וְאֶת־ בְּנָ֥הּ הַמֵּ֖ת הִשְׁכִּ֥יבָה בְחֵיקִֽי׃ (כג) וָאָקֻ֥ם בַּבֹּ֛קֶר לְהֵינִ֥יק אֶת־ בְּנִ֖י וְהִנֵּה־ מֵ֑ת וָאֶתְבּוֹנֵ֤ן אֵלָיו֙ בַּבֹּ֔קֶר וְהִנֵּ֛ה לֹֽא־ הָיָ֥ה בְנִ֖י אֲשֶׁ֥ר יָלָֽדְתִּי׃ (כד) וַתֹּאמֶר֩ הָאִשָּׁ֨ה הָאַחֶ֜רֶת לֹ֣א כִ֗י בְּנִ֤י הַחַי֙ וּבְנֵ֣ךְ הַמֵּ֔ת וְזֹ֤את אֹמֶ֙רֶת֙ לֹ֣א כִ֔י בְּנֵ֥ךְ הַמֵּ֖ת וּבְנִ֣י הֶחָ֑י וַתְּדַבֵּ֖רְנָה לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כה) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ זֹ֣את אֹמֶ֔רֶת זֶה־ בְּנִ֥י הַחַ֖י וּבְנֵ֣ךְ הַמֵּ֑ת וְזֹ֤את אֹמֶ֙רֶת֙ לֹ֣א כִ֔י בְּנֵ֥ךְ הַמֵּ֖ת וּבְנִ֥י הֶחָֽי׃ (כו) וַיֹּ֥אמֶר הַמֶּ֖לֶךְ קְח֣וּ לִי־ חָ֑רֶב וַיָּבִ֥אוּ הַחֶ֖רֶב לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כז) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ גִּזְר֛וּ אֶת־ הַיֶּ֥לֶד הַחַ֖י לִשְׁנָ֑יִם וּתְנ֤וּ אֶֽת־ הַחֲצִי֙ לְאַחַ֔ת וְאֶֽת־ הַחֲצִ֖י לְאֶחָֽת׃ (כח) וַתֹּ֣אמֶר הָאִשָּׁה֩ אֲשֶׁר־ בְּנָ֨הּ הַחַ֜י אֶל־ הַמֶּ֗לֶךְ כִּֽי־ נִכְמְר֣וּ רַחֲמֶיהָ֮ עַל־ בְּנָהּ֒ וַתֹּ֣אמֶר ׀ בִּ֣י אֲדֹנִ֗י תְּנוּ־ לָהּ֙ אֶת־ הַיָּל֣וּד הַחַ֔י וְהָמֵ֖ת אַל־ תְּמִיתֻ֑הוּ וְזֹ֣את אֹמֶ֗רֶת גַּם־ לִ֥י גַם־ לָ֛ךְ לֹ֥א יִהְיֶ֖ה גְּזֹֽרוּ׃ (כט) וַיַּ֨עַן הַמֶּ֜לֶךְ וַיֹּ֗אמֶר תְּנוּ־ לָהּ֙ אֶת־ הַיָּל֣וּד הַחַ֔י וְהָמֵ֖ת לֹ֣א תְמִיתֻ֑הוּ הִ֖יא אִמּֽוֹ׃ (ל) וַיִּשְׁמְע֣וּ כָל־ יִשְׂרָאֵ֗ל אֶת־ הַמִּשְׁפָּט֙ אֲשֶׁ֣ר שָׁפַ֣ט הַמֶּ֔לֶךְ וַיִּֽרְא֖וּ מִפְּנֵ֣י הַמֶּ֑לֶךְ כִּ֣י רָא֔וּ כִּֽי־ חָכְמַ֧ת אֱלֹהִ֛ים בְּקִרְבּ֖וֹ לַעֲשׂ֥וֹת מִשְׁפָּֽט׃ תחילת פרק ד (לא) וַֽיְהִי֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה מֶ֖לֶךְ עַל־ כָּל־ יִשְׂרָאֵֽל׃ (לב) וְאֵ֥לֶּה הַשָּׂרִ֖ים אֲשֶׁר־ ל֑וֹ עֲזַרְיָ֥הוּ בֶן־ צָד֖וֹק הַכֹּהֵֽן׃ (לג) אֱלִיחֹ֧רֶף וַאֲחִיָּ֛ה בְּנֵ֥י שִׁישָׁ֖א סֹפְרִ֑ים יְהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד הַמַּזְכִּֽיר׃ (לד) וּבְנָיָ֥הוּ בֶן־ יְהוֹיָדָ֖ע עַל־ הַצָּבָ֑א וְצָד֥וֹק וְאֶבְיָתָ֖ר כֹּהֲנִֽים׃ (לה) וַעֲזַרְיָ֥הוּ בֶן־ נָתָ֖ן עַל־ הַנִּצָּבִ֑ים וְזָב֧וּד בֶּן־ נָתָ֛ן כֹּהֵ֖ן רֵעֶ֥ה הַמֶּֽלֶךְ׃ (לו) וַאֲחִישָׁ֖ר עַל־ הַבָּ֑יִת וַאֲדֹנִירָ֥ם בֶּן־ עַבְדָּ֖א עַל־ הַמַּֽס׃ (לז) וְלִשְׁלֹמֹ֞ה שְׁנֵים־ עָשָׂ֤ר נִצָּבִים֙ עַל־ כָּל־ יִשְׂרָאֵ֔ל וְכִלְכְּל֥וּ אֶת־ הַמֶּ֖לֶךְ וְאֶת־ בֵּית֑וֹ חֹ֧דֶשׁ בַּשָּׁנָ֛ה יִהְיֶ֥ה עַל־ (אחד) הָאֶחָ֖ד לְכַלְכֵּֽל׃ (לח) וְאֵ֣לֶּה שְׁמוֹתָ֔ם בֶּן־ ח֖וּר בְּהַ֥ר אֶפְרָֽיִם׃ (לט) בֶּן־ דֶּ֛קֶר בְּמָקַ֥ץ וּבְשַֽׁעַלְבִ֖ים וּבֵ֣ית שָׁ֑מֶשׁ וְאֵיל֖וֹן בֵּ֥ית חָנָֽן׃ (מ) בֶּן־ חֶ֖סֶד בָּֽאֲרֻבּ֑וֹת ל֥וֹ שֹׂכֹ֖ה וְכָל־ אֶ֥רֶץ חֵֽפֶר׃ (מא) בֶּן־ פְּלִשְׁתִּ֔ים וְעַ֖ד גְּב֣וּל מִצְרָ֑יִם מַגִּשִׁ֥ים מִנְחָ֛ה וְעֹבְדִ֥ים אֶת־ שְׁלֹמֹ֖ה כָּל־ יְמֵ֥י חַיָּֽיו׃ (ג) וַיְהִ֥י לֶֽחֶם־ שְׁלֹמֹ֖ה לְי֣וֹם אֶחָ֑ד שְׁלֹשִׁ֥ים כֹּר֙ סֹ֔לֶת וְשִׁשִּׁ֥ים כֹּ֖ר קָֽמַח׃ (ד) עֲשָׂרָ֨ה בָקָ֜ר בְּרִאִ֗ים וְעֶשְׂרִ֥ים בָּקָ֛ר רְעִ֖י וּמֵ֣אָה צֹ֑אן לְ֠בַד מֵֽאַיָּ֤ל וּצְבִי֙ וְיַחְמ֔וּר וּבַרְבֻּרִ֖ים אֲבוּסִֽים׃ (ה) כִּי־ ה֞וּא רֹדֶ֣ה ׀ בְּכָל־ עֵ֣בֶר הַנָּהָ֗ר מִתִּפְסַח֙ וְעַד־ עַזָּ֔ה בְּכָל־ מַלְכֵ֖י עֵ֣בֶר הַנָּהָ֑ר וְשָׁל֗וֹם הָ֥יָה ל֛וֹ מִכָּל־ עֲבָרָ֖יו מִסָּבִֽיב׃ (ו) וַיֵּשֶׁב֩ יְהוּדָ֨ה וְיִשְׂרָאֵ֜ל לָבֶ֗טַח אִ֣ישׁ תַּ֤חַת גַּפְנוֹ֙ וְתַ֣חַת תְּאֵֽנָת֔וֹ מִדָּ֖ן וְעַד־ בְּאֵ֣ר שָׁ֑בַע כֹּ֖ל יְמֵ֥י שְׁלֹמֹֽה׃ (ז) וַיְהִ֣י לִשְׁלֹמֹ֗ה אַרְבָּעִ֥ים אֶ֛לֶף אֻרְוֺ֥ת סוּסִ֖ים לְמֶרְכָּב֑וֹ וּשְׁנֵים־ עָשָׂ֥ר אֶ֖לֶף פָּרָשִֽׁים׃ (ח) וְכִלְכְּלוּ֩ הַנִּצָּבִ֨ים הָאֵ֜לֶּה אֶת־ הַמֶּ֣לֶךְ שְׁלֹמֹ֗ה וְאֵ֧ת כָּל־ הַקָּרֵ֛ב אֶל־ שֻׁלְחַ֥ן הַמֶּֽלֶךְ־ שְׁלֹמֹ֖ה אִ֣ישׁ חָדְשׁ֑וֹ לֹ֥א יְעַדְּר֖וּ דָּבָֽר׃ (ט) וְהַשְּׂעֹרִ֣ים וְהַתֶּ֔בֶן לַסּוּסִ֖ים וְלָרָ֑כֶשׁ יָבִ֗אוּ אֶל־ הַמָּקוֹם֙ אֲשֶׁ֣ר יִֽהְיֶה־ שָּׁ֔ם אִ֖ישׁ כְּמִשְׁפָּטֽוֹ׃ (י) וַיִּתֵּן֩ אֱלֹהִ֨ים חָכְמָ֧ה לִשְׁלֹמֹ֛ה וּתְבוּנָ֖ה הַרְבֵּ֣ה מְאֹ֑ד וְרֹ֣חַב לֵ֔ב כַּח֕וֹל אֲשֶׁ֖ר עַל־ שְׂפַ֥ת הַיָּֽם׃ (יא) וַתֵּ֙רֶב֙ חָכְמַ֣ת שְׁלֹמֹ֔ה מֵֽחָכְמַ֖ת כָּל־ בְּנֵי־ קֶ֑דֶם וּמִכֹּ֖ל חָכְמַ֥ת מִצְרָֽיִם׃ (יב) וַיֶּחְכַּם֮ מִכָּל־ הָֽאָדָם֒ מֵאֵיתָ֣ן הָאֶזְרָחִ֗י וְהֵימָ֧ן הַמֶּ֧לֶךְ שְׁלֹמֹ֛ה מַ֖ס מִכָּל־ יִשְׂרָאֵ֑ל וַיְהִ֣י הַמַּ֔ס שְׁלֹשִׁ֥ים אֶ֖לֶף אִֽישׁ׃ (כט) וַיִּשְׁלָחֵ֣ם לְבָנ֗וֹנָה עֲשֶׂ֨רֶת אֲלָפִ֤ים בַּחֹ֙דֶשׁ֙ חֲלִיפ֔וֹת חֹ֚דֶשׁ יִהְי֣וּ בַלְּבָנ֔וֹן שְׁנַ֥יִם חֳדָשִׁ֖ים בְּבֵית֑וֹ וַאֲדֹנִירָ֖ם עַל־ הַמַּֽס׃ (ל) וַיְהִ֧י לִשְׁלֹמֹ֛ה שִׁבְעִ֥ים אֶ֖לֶף נֹשֵׂ֣א סַבָּ֑ל וּשְׁמֹנִ֥ים אֶ֖לֶף חֹצֵ֥ב בָּהָֽר׃ (לא) לְ֠בַד מִשָּׂרֵ֨י הַנִּצָּבִ֤ים לִשְׁלֹמֹה֙ אֲשֶׁ֣ר עַל־ הַמְּלָאכָ֔ה שְׁלֹ֥שֶׁת אֲלָפִ֖ים וּשְׁלֹ֣שׁ מֵא֑וֹת הָרֹדִ֣ים בָּעָ֔ם הָעֹשִׂ֖ים בַּמְּלָאכָֽה׃ (לב) וַיְצַ֣ו הַמֶּ֡לֶךְ וַיַּסִּעוּ֩ אֲבָנִ֨ים גְּדֹל֜וֹת אֲבָנִ֧ים יְקָר֛וֹת לְיַסֵּ֥ד הַבָּ֖יִת אַבְנֵ֥י גָזִֽית׃ (לג) וַֽיִּפְסְל֞וּ בֹּנֵ֧י שְׁלֹמֹ֛ה וּבֹנֵ֥י חִיר֖וֹם וְהַגִּבְלִ֑ים וַיָּכִ֛ינוּ הָעֵצִ֥ים וְהָאֲבָנִ֖ים לִבְנ֥וֹת הַבָּֽיִת׃ (לד) וַיְהִ֣י בִשְׁמוֹנִ֣ים שָׁנָ֣ה וְאַרְבַּ֣ע מֵא֣וֹת שָׁנָ֡ה לְצֵ֣את בְּנֵֽי־ יִשְׂרָאֵ֣ל מֵאֶֽרֶץ־ מִצְרַיִם֩ בַּשָּׁנָ֨ה הָרְבִיעִ֜ית בְּחֹ֣דֶשׁ זִ֗ו ה֚וּא הַחֹ֣דֶשׁ הַשֵּׁנִ֔י לִמְלֹ֥ךְ שְׁלֹמֹ֖ה עַל־ יִשְׂרָאֵ֑ל וַיִּ֥בֶן הַבַּ֖יִת לַיהוָֽה׃ (לה) וְהַבַּ֗יִת אֲשֶׁ֨ר בָּנָ֜ה הַמֶּ֤לֶךְ שְׁלֹמֹה֙ לַֽיהוָ֔ה שִׁשִּֽׁים־ אַמָּ֥ה אָרְכּ֖וֹ וְעֶשְׂרִ֤ים רָחְבּ֑וֹ וּשְׁלֹשִׁ֥ים אַמָּ֖ה קוֹמָתֽוֹ׃ (לו) וְהָאוּלָ֗ם עַל־ פְּנֵי֙ הֵיכַ֣ל הַבַּ֔יִת עֶשְׂרִ֣ים אַמָּה֙ אָרְכּ֔וֹ עַל־ פְּנֵ֖י רֹ֣חַב הַבָּ֑יִת עֶ֧שֶׂר בָּאַמָּ֛ה רָחְבּ֖וֹ עַל־ פְּנֵ֥י הַבָּֽיִת׃ (לז) וַיַּ֣עַשׂ לַבָּ֔יִת חַלּוֹנֵ֖י שְׁקֻפִ֥ים אֲטֻמִֽים׃ (לח) וַיִּבֶן֩ עַל־ קִ֨יר הַבַּ֤יִת (יצוע) יָצִ֙יעַ֙ סָבִ֔יב אֶת־ קִיר֤וֹת הַבַּ֙יִת֙ סָבִ֔יב לַֽהֵיכָ֖ל וְלַדְּבִ֑יר וַיַּ֥עַשׂ צְלָע֖וֹת סָבִֽיב׃ (לט) (היצוע) הַיָּצִ֨יעַ הַתַּחְתֹּנָ֜ה חָמֵ֧שׁ בָּאַמָּ֣ה רָחְבָּ֗הּ וְהַתִּֽיכֹנָה֙ שֵׁ֤שׁ בָּֽאַמָּה֙ רָחְבָּ֔הּ וְהַ֨שְּׁלִישִׁ֔ית שֶׁ֥בַע בָּאַמָּ֖ה רָחְבָּ֑הּ כִּ֡י מִגְרָעוֹת֩ נָתַ֨ן לַבַּ֤יִת סָבִיב֙ ח֔וּצָה לְבִלְתִּ֖י אֲחֹ֥ז בְּקִֽירוֹת־ הַבָּֽיִת׃ (מ) וְהַבַּ֙יִת֙ בְּהִבָּ֣נֹת֔וֹ אֶֽבֶן־ שְׁלֵמָ֥ה מַסָּ֖ע נִבְנָ֑ה וּמַקָּב֤וֹת וְהַגַּרְזֶן֙ כָּל־ כְּלִ֣י בַרְזֶ֔ל לֹֽא־ נִשְׁמַ֥ע בַּבַּ֖יִת בְּהִבָּנֹתֽוֹ׃ (מא) פֶּ֗תַח הַצֵּלָע֙ הַתִּ֣יכֹנָ֔ה אֶל־ כֶּ֥תֶף הַבַּ֖יִת הַיְמָנִ֑ית וּבְלוּלִּ֗ים יַֽעֲלוּ֙ עַל־ הַתִּ֣יכֹנָ֔ה וּמִן־ הַתִּֽיכֹנָ֖ה אֶל־ הַשְּׁלִשִֽׁים׃ (מב) וַיִּ֥בֶן אֶת־ הַבַּ֖יִת וַיְכַלֵּ֑הוּ וַיִּסְפֹּ֤ן אֶת־ הַבַּ֙יִת֙ גֵּבִ֔ים וּשְׂדֵרֹ֖ת בָּאֲרָזִֽים׃ (מג) וַיִּ֤בֶן אֶת־ (היצוע) הַיָּצִ֙יעַ֙ עַל־ כָּל־ הַבַּ֔יִת חָמֵ֥שׁ אַמּ֖וֹת קֽוֹמָת֑וֹ וַיֶּאֱחֹ֥ז אֶת־ הַבַּ֖יִת בַּעֲצֵ֥י אֲרָזִֽים׃ (מד) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ שְׁלֹמֹ֖ה לֵאמֹֽר׃ (מה) הַבַּ֨יִת הַזֶּ֜ה אֲשֶׁר־ אַתָּ֣ה בֹנֶ֗ה אִם־ תֵּלֵ֤ךְ בְּחֻקֹּתַי֙ וְאֶת־ מִשְׁפָּטַ֣י תַּֽעֲשֶׂ֔ה וְשָׁמַרְתָּ֥ אֶת־ כָּל־ מִצְוֺתַ֖י לָלֶ֣כֶת בָּהֶ֑ם וַהֲקִמֹתִ֤י אֶת־ דְּבָרִי֙ אִתָּ֔ךְ אֲשֶׁ֥ר דִּבַּ֖רְתִּי אֶל־ דָּוִ֥ד אָבִֽיךָ׃
