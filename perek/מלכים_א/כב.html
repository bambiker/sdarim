מלכים א פרק כב לפי החלוקה לסדרים

פרק כב בספר מלכים א (54 פסוקים) נמצא במלכים סדר יח (פסוקים יג–נד) ובמלכים סדר יט (פסוקים א–יב).

מלכים סדר יח

(א) ק֣וּם רֵ֗ד לִקְרַ֛את אַחְאָ֥ב מֶֽלֶךְ־ יִשְׂרָאֵ֖ל אֲשֶׁ֣ר בְּשֹׁמְר֑וֹן הִנֵּה֙ בְּכֶ֣רֶם נָב֔וֹת אֲשֶׁר־ יָ֥רַד שָׁ֖ם לְרִשְׁתּֽוֹ׃ (ב) וְדִבַּרְתָּ֨ אֵלָ֜יו לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה הֲרָצַ֖חְתָּ וְגַם־ יָרָ֑שְׁתָּ וְדִבַּרְתָּ֨ אֵלָ֜יו לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה בִּמְק֗וֹם אֲשֶׁ֨ר לָקְק֤וּ הַכְּלָבִים֙ אֶת־ דַּ֣ם נָב֔וֹת יָלֹ֧קּוּ הַכְּלָבִ֛ים אֶת־ דָּמְךָ֖ גַּם־ אָֽתָּה׃ (ג) וַיֹּ֤אמֶר אַחְאָב֙ אֶל־ אֵ֣לִיָּ֔הוּ הַֽמְצָאתַ֖נִי אֹיְבִ֑י וַיֹּ֣אמֶר מָצָ֔אתִי יַ֚עַן הִתְמַכֶּרְךָ֔ לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֥י יְהוָֽה׃ (ד) הִנְנִ֨י (מבי) מֵבִ֤יא אֵלֶ֙יךָ֙ רָעָ֔ה וּבִעַרְתִּ֖י אַחֲרֶ֑יךָ וְהִכְרַתִּ֤י לְאַחְאָב֙ מַשְׁתִּ֣ין בְּקִ֔יר וְעָצ֥וּר וְעָז֖וּב בְּיִשְׂרָאֵֽל׃ (ה) וְנָתַתִּ֣י אֶת־ בֵּיתְךָ֗ כְּבֵית֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט וּכְבֵ֖ית בַּעְשָׁ֣א בֶן־ אֲחִיָּ֑ה אֶל־ הַכַּ֙עַס֙ אֲשֶׁ֣ר הִכְעַ֔סְתָּ וַֽתַּחֲטִ֖א אֶת־ יִשְׂרָאֵֽל׃ (ו) וְגַ֨ם־ לְאִיזֶ֔בֶל דִּבֶּ֥ר יְהוָ֖ה לֵאמֹ֑ר הַכְּלָבִ֛ים יֹאכְל֥וּ אֶת־ אִיזֶ֖בֶל בְּחֵ֥ל יִזְרְעֶֽאל׃ (ז) הַמֵּ֤ת לְאַחְאָב֙ בָּעִ֔יר יֹאכְל֖וּ הַכְּלָבִ֑ים וְהַמֵּת֙ בַּשָּׂדֶ֔ה יֹאכְל֖וּ ע֥וֹף הַשָּׁמָֽיִם׃ (ח) רַ֚ק לֹֽא־ הָיָ֣ה כְאַחְאָ֔ב אֲשֶׁ֣ר הִתְמַכֵּ֔ר לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה אֲשֶׁר־ הֵסַ֥תָּה אֹת֖וֹ אִיזֶ֥בֶל אִשְׁתּֽוֹ׃ (ט) וַיַּתְעֵ֣ב מְאֹ֔ד לָלֶ֖כֶת אַחֲרֵ֣י הַגִּלֻּלִ֑ים כְּכֹל֙ אֲשֶׁ֣ר עָשׂ֣וּ הָאֱמֹרִ֔י אֲשֶׁר֙ הוֹרִ֣ישׁ יְהוָ֔ה מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (י) וַיְהִי֩ כִשְׁמֹ֨עַ אַחְאָ֜ב אֶת־ הַדְּבָרִ֤ים הָאֵ֙לֶּה֙ וַיִּקְרַ֣ע בְּגָדָ֔יו וַיָּֽשֶׂם־ שַׂ֥ק עַל־ בְּשָׂר֖וֹ וַיָּצ֑וֹם וַיִּשְׁכַּ֣ב בַּשָּׂ֔ק וַיְהַלֵּ֖ךְ אַֽט׃ (יא) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ אֵלִיָּ֥הוּ הַתִּשְׁבִּ֖י לֵאמֹֽר׃ (יב) הֲ‍ֽרָאִ֔יתָ כִּֽי־ נִכְנַ֥ע אַחְאָ֖ב מִלְּפָנָ֑י יַ֜עַן כִּֽי־ נִכְנַ֣ע מִפָּנַ֗י לֹֽא־ (אבי) אָבִ֤יא הָֽרָעָה֙ בְּיָמָ֔יו בִּימֵ֣י בְנ֔וֹ אָבִ֥יא הָרָעָ֖ה עַל־ בֵּיתֽוֹ׃ תחילת פרק כב (יג) וַיֵּשְׁב֖וּ שָׁלֹ֣שׁ שָׁנִ֑ים אֵ֚ין מִלְחָמָ֔ה בֵּ֥ין אֲרָ֖ם וּבֵ֥ין יִשְׂרָאֵֽל׃ (יד) וַיְהִ֖י בַּשָּׁנָ֣ה הַשְּׁלִישִׁ֑ית וַיֵּ֛רֶד יְהוֹשָׁפָ֥ט מֶֽלֶךְ־ יְהוּדָ֖ה אֶל־ מֶ֥לֶךְ יִשְׂרָאֵֽל׃ (טו) וַיֹּ֤אמֶר מֶֽלֶךְ־ יִשְׂרָאֵל֙ אֶל־ עֲבָדָ֔יו הַיְדַעְתֶּ֕ם כִּֽי־ לָ֖נוּ רָמֹ֣ת גִּלְעָ֑ד וַאֲנַ֣חְנוּ מַחְשִׁ֔ים מִקַּ֣חַת אֹתָ֔הּ מִיַּ֖ד מֶ֥לֶךְ אֲרָֽם׃ (טז) וַיֹּ֙אמֶר֙ אֶל־ יְה֣וֹשָׁפָ֔ט הֲתֵלֵ֥ךְ אִתִּ֛י לַמִּלְחָמָ֖ה רָמֹ֣ת גִּלְעָ֑ד וַיֹּ֤אמֶר יְהֽוֹשָׁפָט֙ אֶל־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל כָּמ֧וֹנִי כָמ֛וֹךָ כְּעַמִּ֥י כְעַמֶּ֖ךָ כְּסוּסַ֥י כְּסוּסֶֽיךָ׃ (יז) וַיֹּ֥אמֶר יְהוֹשָׁפָ֖ט אֶל־ מֶ֣לֶךְ יִשְׂרָאֵ֑ל דְּרָשׁ־ נָ֥א כַיּ֖וֹם אֶת־ דְּבַ֥ר יְהוָֽה׃ (יח) וַיִּקְבֹּ֨ץ מֶֽלֶךְ־ יִשְׂרָאֵ֥ל אֶֽת־ הַנְּבִיאִים֮ כְּאַרְבַּ֣ע מֵא֣וֹת אִישׁ֒ וַיֹּ֣אמֶר אֲלֵהֶ֗ם הַאֵלֵ֞ךְ עַל־ רָמֹ֥ת גִּלְעָ֛ד לַמִּלְחָמָ֖ה אִם־ אֶחְדָּ֑ל וַיֹּאמְר֣וּ עֲלֵ֔ה וְיִתֵּ֥ן אֲדֹנָ֖י בְּיַ֥ד הַמֶּֽלֶךְ׃ (יט) וַיֹּ֙אמֶר֙ יְה֣וֹשָׁפָ֔ט הַאֵ֨ין פֹּ֥ה נָבִ֛יא לַיהוָ֖ה ע֑וֹד וְנִדְרְשָׁ֖ה מֵאוֹתֽוֹ׃ (כ) וַיֹּ֣אמֶר מֶֽלֶךְ־ יִשְׂרָאֵ֣ל ׀ אֶֽל־ יְהוֹשָׁפָ֡ט ע֣וֹד אִישׁ־ אֶחָ֡ד לִדְרֹשׁ֩ אֶת־ יְהוָ֨ה מֵאֹת֜וֹ וַאֲנִ֣י שְׂנֵאתִ֗יו כִּ֠י לֹֽא־ יִתְנַבֵּ֨א עָלַ֥י טוֹב֙ כִּ֣י אִם־ רָ֔ע מִיכָ֖יְהוּ בֶּן־ יִמְלָ֑ה וַיֹּ֙אמֶר֙ יְה֣וֹשָׁפָ֔ט אַל־ יֹאמַ֥ר הַמֶּ֖לֶךְ כֵּֽן׃ (כא) וַיִּקְרָא֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל אֶל־ סָרִ֖יס אֶחָ֑ד וַיֹּ֕אמֶר מַהֲרָ֖ה מִיכָ֥יְהוּ בֶן־ יִמְלָֽה׃ (כב) וּמֶ֣לֶךְ יִשְׂרָאֵ֡ל וִֽיהוֹשָׁפָ֣ט מֶֽלֶךְ־ יְהוּדָ֡ה יֹשְׁבִים֩ אִ֨ישׁ עַל־ כִּסְא֜וֹ מְלֻבָּשִׁ֤ים בְּגָדִים֙ בְּגֹ֔רֶן פֶּ֖תַח שַׁ֣עַר שֹׁמְר֑וֹן וְכָ֨ל־ הַנְּבִיאִ֔ים מִֽתְנַבְּאִ֖ים לִפְנֵיהֶֽם׃ (כג) וַיַּ֥עַשׂ ל֛וֹ צִדְקִיָּ֥ה בֶֽן־ כְּנַעֲנָ֖ה קַרְנֵ֣י בַרְזֶ֑ל וַיֹּ֙אמֶר֙ כֹּֽה־ אָמַ֣ר יְהוָ֔ה בְּאֵ֛לֶּה תְּנַגַּ֥ח אֶת־ אֲרָ֖ם עַד־ כַּלֹּתָֽם׃ (כד) וְכָל־ הַנְּבִאִ֔ים נִבְּאִ֥ים כֵּ֖ן לֵאמֹ֑ר עֲלֵ֞ה רָמֹ֤ת גִּלְעָד֙ וְהַצְלַ֔ח וְנָתַ֥ן יְהוָ֖ה בְּיַ֥ד הַמֶּֽלֶךְ׃ (כה) וְהַמַּלְאָ֞ךְ אֲשֶׁר־ הָלַ֣ךְ ׀ לִקְרֹ֣א מִיכָ֗יְהוּ דִּבֶּ֤ר אֵלָיו֙ לֵאמֹ֔ר הִנֵּה־ נָ֞א דִּבְרֵ֧י הַנְּבִיאִ֛ים פֶּֽה־ אֶחָ֥ד ט֖וֹב אֶל־ הַמֶּ֑לֶךְ יְהִֽי־ נָ֣א (דבריך) דְבָרְךָ֗ כִּדְבַ֛ר אַחַ֥ד מֵהֶ֖ם וְדִבַּ֥רְתָּ טּֽוֹב׃ (כו) וַיֹּ֖אמֶר מִיכָ֑יְהוּ חַי־ יְהוָ֕ה כִּ֠י אֶת־ אֲשֶׁ֨ר יֹאמַ֧ר יְהוָ֛ה אֵלַ֖י אֹת֥וֹ אֲדַבֵּֽר׃ (כז) וַיָּבוֹא֮ אֶל־ הַמֶּלֶךְ֒ וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֵלָ֗יו מִיכָ֙יְהוּ֙ הֲנֵלֵ֞ךְ אֶל־ רָמֹ֥ת גִּלְעָ֛ד לַמִּלְחָמָ֖ה אִם־ נֶחְדָּ֑ל וַיֹּ֤אמֶר אֵלָיו֙ עֲלֵ֣ה וְהַצְלַ֔ח וְנָתַ֥ן יְהוָ֖ה בְּיַ֥ד הַמֶּֽלֶךְ׃ (כח) וַיֹּ֤אמֶר אֵלָיו֙ הַמֶּ֔לֶךְ עַד־ כַּמֶּ֥ה פְעָמִ֖ים אֲנִ֣י מַשְׁבִּעֶ֑ךָ אֲ֠שֶׁר לֹֽא־ תְדַבֵּ֥ר אֵלַ֛י רַק־ אֱמֶ֖ת בְּשֵׁ֥ם יְהוָֽה׃ (כט) וַיֹּ֗אמֶר רָאִ֤יתִי אֶת־ כָּל־ יִשְׂרָאֵל֙ נְפֹצִ֣ים אֶל־ הֶהָרִ֔ים כַּצֹּ֕אן אֲשֶׁ֥ר אֵין־ לָהֶ֖ם רֹעֶ֑ה וַיֹּ֤אמֶר יְהוָה֙ לֹֽא־ אֲדֹנִ֣ים לָאֵ֔לֶּה יָשׁ֥וּבוּ אִישׁ־ לְבֵית֖וֹ בְּשָׁלֽוֹם׃ (ל) וַיֹּ֥אמֶר מֶֽלֶךְ־ יִשְׂרָאֵ֖ל אֶל־ יְהוֹשָׁפָ֑ט הֲלוֹא֙ אָמַ֣רְתִּי אֵלֶ֔יךָ לֽוֹא־ יִתְנַבֵּ֥א עָלַ֛י ט֖וֹב כִּ֥י אִם־ רָֽע׃ (לא) וַיֹּ֕אמֶר לָכֵ֖ן שְׁמַ֣ע דְּבַר־ יְהוָ֑ה רָאִ֤יתִי אֶת־ יְהוָה֙ יֹשֵׁ֣ב עַל־ כִּסְא֔וֹ וְכָל־ צְבָ֤א הַשָּׁמַ֙יִם֙ עֹמֵ֣ד עָלָ֔יו מִימִינ֖וֹ וּמִשְּׂמֹאלֽוֹ׃ (לב) וַיֹּ֣אמֶר יְהוָ֗ה מִ֤י יְפַתֶּה֙ אֶת־ אַחְאָ֔ב וְיַ֕עַל וְיִפֹּ֖ל בְּרָמֹ֣ת גִּלְעָ֑ד וַיֹּ֤אמֶר זֶה֙ בְּכֹ֔ה וְזֶ֥ה אֹמֵ֖ר בְּכֹֽה׃ (לג) וַיֵּצֵ֣א הָר֗וּחַ וַֽיַּעֲמֹד֙ לִפְנֵ֣י יְהוָ֔ה וַיֹּ֖אמֶר אֲנִ֣י אֲפַתֶּ֑נּוּ וַיֹּ֧אמֶר יְהוָ֛ה אֵלָ֖יו בַּמָּֽה׃ (לד) וַיֹּ֗אמֶר אֵצֵא֙ וְהָיִ֙יתִי֙ ר֣וּחַ שֶׁ֔קֶר בְּפִ֖י כָּל־ נְבִיאָ֑יו וַיֹּ֗אמֶר תְּפַתֶּה֙ וְגַם־ תּוּכָ֔ל צֵ֖א וַעֲשֵׂה־ כֵֽן׃ (לה) וְעַתָּ֗ה הִנֵּ֨ה נָתַ֤ן יְהוָה֙ ר֣וּחַ שֶׁ֔קֶר בְּפִ֖י כָּל־ נְבִיאֶ֣יךָ אֵ֑לֶּה וַֽיהוָ֔ה דִּבֶּ֥ר עָלֶ֖יךָ רָעָֽה׃ (לו) וַיִּגַּשׁ֙ צִדְקִיָּ֣הוּ בֶֽן־ כְּנַעֲנָ֔ה וַיַּכֶּ֥ה אֶת־ מִיכָ֖יְהוּ עַל־ הַלֶּ֑חִי וַיֹּ֕אמֶר אֵי־ זֶ֨ה עָבַ֧ר רֽוּחַ־ יְהוָ֛ה מֵאִתִּ֖י לְדַבֵּ֥ר אוֹתָֽךְ׃ (לז) וַיֹּ֣אמֶר מִיכָ֔יְהוּ הִנְּךָ֥ רֹאֶ֖ה בַּיּ֣וֹם הַה֑וּא אֲשֶׁ֥ר תָּבֹ֛א חֶ֥דֶר בְּחֶ֖דֶר לְהֵחָבֵֽה׃ (לח) וַיֹּ֙אמֶר֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל קַ֚ח אֶת־ מִיכָ֔יְהוּ וַהֲשִׁיבֵ֖הוּ אֶל־ אָמֹ֣ן שַׂר־ הָעִ֑יר וְאֶל־ מִיכָ֔יְהוּ אִם־ שׁ֤וֹב תָּשׁוּב֙ בְּשָׁל֔וֹם לֹֽא־ דִבֶּ֥ר יְהוָ֖ה בִּ֑י וַיֹּ֕אמֶר שִׁמְע֖וּ עַמִּ֥ים כֻּלָּֽם׃ (מא) וַיַּ֧עַל מֶֽלֶךְ־ יִשְׂרָאֵ֛ל וְיהוֹשָׁפָ֥ט מֶֽלֶךְ־ יְהוּדָ֖ה רָמֹ֥ת גִּלְעָֽד׃ (מב) וַיֹּאמֶר֩ מֶ֨לֶךְ יִשְׂרָאֵ֜ל אֶל־ יְהוֹשָׁפָ֗ט הִתְחַפֵּשׂ֙ וָבֹ֣א בַמִּלְחָמָ֔ה וְאַתָּ֖ה לְבַ֣שׁ בְּגָדֶ֑יךָ וַיִּתְחַפֵּשׂ֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיָּב֖וֹא בַּמִּלְחָמָֽה׃ (מג) וּמֶ֣לֶךְ אֲרָ֡ם צִוָּ֣ה אֶת־ שָׂרֵי֩ הָרֶ֨כֶב אֲשֶׁר־ ל֜וֹ שְׁלֹשִׁ֤ים וּשְׁנַ֙יִם֙ לֵאמֹ֔ר לֹ֚א תִּלָּ֣חֲמ֔וּ אֶת־ קָטֹ֖ן וְאֶת־ גָּד֑וֹל כִּ֛י אִֽם־ אֶת־ מֶ֥לֶךְ יִשְׂרָאֵ֖ל לְבַדּֽוֹ׃ (מד) וַיְהִ֡י כִּֽי־ לֹא־ מֶ֥לֶךְ יִשְׂרָאֵ֖ל ה֑וּא וַיָּשׁ֖וּבוּ מֵאַחֲרָֽיו׃ (מו) וְאִ֗ישׁ מָשַׁ֤ךְ בַּקֶּ֙שֶׁת֙ לְתֻמּ֔וֹ וַיַּכֶּה֙ אֶת־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל בֵּ֥ין הַדְּבָקִ֖ים וּבֵ֣ין הַשִּׁרְיָ֑ן וַיֹּ֣אמֶר לְרַכָּב֗וֹ הֲפֹ֥ךְ יָדְךָ֛ וְהוֹצִיאֵ֥נִי מִן־ הַֽמַּחֲנֶ֖ה כִּ֥י הָחֳלֵֽיתִי׃ (מז) וַתַּעֲלֶ֤ה הַמִּלְחָמָה֙ בַּיּ֣וֹם הַה֔וּא וְהַמֶּ֗לֶךְ הָיָ֧ה מָעֳמָ֛ד בַּמֶּרְכָּבָ֖ה נֹ֣כַח אֲרָ֑ם וַיָּ֣מָת בָּעֶ֔רֶב וַיִּ֥צֶק דַּֽם־ הַמַּכָּ֖ה אֶל־ חֵ֥יק הָרָֽכֶב׃ (מח) וַיַּעֲבֹ֤ר הָרִנָּה֙ בַּֽמַּחֲנֶ֔ה כְּבֹ֥א הַשֶּׁ֖מֶשׁ לֵאמֹ֑ר אִ֥ישׁ אֶל־ עִיר֖וֹ וְאִ֥ישׁ אֶל־ אַרְצֽוֹ׃ (מט) וַיָּ֣מָת הַמֶּ֔לֶךְ וַיָּב֖וֹא שֹׁמְר֑וֹן וַיִּקְבְּר֥וּ אֶת־ הַמֶּ֖לֶךְ בְּשֹׁמְרֽוֹן׃ (נ) וַיִּשְׁטֹ֨ף אֶת־ הָרֶ֜כֶב עַ֣ל ׀ בְּרֵכַ֣ת שֹׁמְר֗וֹן וַיָּלֹ֤קּוּ הַכְּלָבִים֙ אֶת־ דָּמ֔וֹ וְהַזֹּנ֖וֹת רָחָ֑צוּ כִּדְבַ֥ר יְהוָ֖ה אֲשֶׁ֥ר דִּבֵּֽר׃ (נא) וְיֶתֶר֩ דִּבְרֵ֨י אַחְאָ֜ב וְכָל־ אֲשֶׁ֣ר עָשָׂ֗ה וּבֵ֤ית הַשֵּׁן֙ אֲשֶׁ֣ר בָּנָ֔ה וְכָל־ הֶעָרִ֖ים אֲשֶׁ֣ר בָּנָ֑ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (נב) וַיִּשְׁכַּ֥ב אַחְאָ֖ב עִם־ אֲבֹתָ֑יו וַיִּמְלֹ֛ךְ אֲחַזְיָ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ (נג) וִיהֽוֹשָׁפָט֙ בֶּן־ אָסָ֔א מָלַ֖ךְ עַל־ יְהוּדָ֑ה בִּשְׁנַ֣ת אַרְבַּ֔ע לְאַחְאָ֖ב מֶ֥לֶךְ יִשְׂרָאֵֽל׃ (נד) יְהוֹשָׁפָ֗ט בֶּן־ שְׁלֹשִׁ֨ים וְחָמֵ֤שׁ שָׁנָה֙ בְּמָלְכ֔וֹ וְעֶשְׂרִ֤ים וְחָמֵשׁ֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ עֲזוּבָ֖ה בַּת־ שִׁלְחִֽי׃

(א) וַיֵּ֗לֶךְ בְּכָל־ דֶּ֛רֶךְ אָסָ֥א אָבִ֖יו לֹא־ סָ֣ר מִמֶּ֑נּוּ לַעֲשׂ֥וֹת הַיָּשָׁ֖ר בְּעֵינֵ֥י יְהוָֽה׃ (ב) אַ֥ךְ הַבָּמ֖וֹת לֹֽא־ סָ֑רוּ ע֥וֹד הָעָ֛ם מְזַבְּחִ֥ים וּֽמְקַטְּרִ֖ים בַּבָּמֽוֹת׃ (ג) וַיַּשְׁלֵ֥ם יְהוֹשָׁפָ֖ט עִם־ מֶ֥לֶךְ יִשְׂרָאֵֽל׃ (ד) וְיֶ֨תֶר דִּבְרֵ֧י יְהוֹשָׁפָ֛ט וּגְבוּרָת֥וֹ אֲשֶׁר־ עָשָׂ֖ה וַאֲשֶׁ֣ר נִלְחָ֑ם הֲלֹֽא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (ה) וְיֶ֙תֶר֙ הַקָּדֵ֔שׁ אֲשֶׁ֣ר נִשְׁאַ֔ר בִּימֵ֖י אָסָ֣א אָבִ֑יו בִּעֵ֖ר מִן־ הָאָֽרֶץ׃ (ו) וּמֶ֥לֶךְ אֵ֛ין בֶּאֱד֖וֹם נִצָּ֥ב מֶֽלֶךְ׃ (ז) יְהוֹשָׁפָ֡ט (עשר) עָשָׂה֩ עֲבָדַ֛י עִם־ עֲבָדֶ֖יךָ בָּאֳנִיּ֑וֹת וְלֹ֥א אָבָ֖ה יְהוֹשָׁפָֽט׃ (ט) וַיִּשְׁכַּ֤ב יְהֽוֹשָׁפָט֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵר֙ עִם־ אֲבֹתָ֔יו בְּעִ֖יר דָּוִ֣ד אָבִ֑יו וַיִּמְלֹ֛ךְ יְהוֹרָ֥ם בְּנ֖וֹ וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיֵּ֗לֶךְ בְּדֶ֤רֶךְ אָבִיו֙ וּבְדֶ֣רֶךְ אִמּ֔וֹ וּבְדֶ֙רֶךְ֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (יב) וַֽיַּעֲבֹד֙ אֶת־ הַבַּ֔עַל וַיִּֽשְׁתַּחֲוֶ֖ה ל֑וֹ וַיַּכְעֵ֗ס אֶת־ יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה אָבִֽיו׃ סוף פרק כב (יג) וַיִּפְשַׁ֤ע מוֹאָב֙ בְּיִשְׂרָאֵ֔ל אַחֲרֵ֖י מ֥וֹת אַחְאָֽב׃ (יד) וַיִּפֹּ֨ל זֶֽה׃ (טו) וּמַלְאַ֣ךְ יְהוָ֗ה דִּבֶּר֙ אֶל־ אֵלִיָּ֣ה הַתִּשְׁבִּ֔י ק֣וּם עֲלֵ֔ה לִקְרַ֖את מַלְאֲכֵ֣י מֶֽלֶךְ־ שֹׁמְר֑וֹן וְדַבֵּ֣ר אֲלֵהֶ֔ם הַֽמִבְּלִ֤י אֵין־ אֱלֹהִים֙ בְּיִשְׂרָאֵ֔ל אַתֶּם֙ הֹֽלְכִ֔ים לִדְרֹ֕שׁ בְּבַ֥עַל זְב֖וּב אֱלֹהֵ֥י עֶקְרֽוֹן׃ (טז) וְלָכֵן֙ כֹּֽה־ אָמַ֣ר יְהוָ֔ה הַמִּטָּ֞ה אֲשֶׁר־ עָלִ֥יתָ שָּׁ֛ם לֹֽא־ תֵרֵ֥ד מִמֶּ֖נָּה כִּ֣י מ֣וֹת תָּמ֑וּת וַיֵּ֖לֶךְ אֵלִיָּֽה׃ (יז) וַיָּשׁ֥וּבוּ הַמַּלְאָכִ֖ים אֵלָ֑יו וַיֹּ֥אמֶר אֲלֵיהֶ֖ם מַה־ זֶּ֥ה שַׁבְתֶּֽם׃ (יח) וַיֹּאמְר֨וּ אֵלָ֜יו אִ֣ישׁ ׀ עָלָ֣ה לִקְרָאתֵ֗נוּ וַיֹּ֣אמֶר אֵלֵינוּ֮ לְכ֣וּ שׁוּבוּ֮ אֶל־ הַמֶּ֣לֶךְ אֲשֶׁר־ שָׁלַ֣ח אֶתְכֶם֒ וְדִבַּרְתֶּ֣ם אֵלָ֗יו כֹּ֚ה אָמַ֣ר יְהוָ֔ה הַֽמִבְּלִ֤י אֵין־ אֱלֹהִים֙ בְּיִשְׂרָאֵ֔ל אַתָּ֣ה שֹׁלֵ֔חַ לִדְרֹ֕שׁ בְּבַ֥עַל זְב֖וּב אֱלֹהֵ֣י עֶקְר֑וֹן לָ֠כֵן הַמִּטָּ֞ה אֲשֶׁר־ עָלִ֥יתָ שָּׁ֛ם לֹֽא־ תֵרֵ֥ד מִמֶּ֖נָּה כִּֽי־ מ֥וֹת תָּמֽוּת׃ (יט) וַיְדַבֵּ֣ר אֲלֵהֶ֔ם מֶ֚ה מִשְׁפַּ֣ט הָאִ֔ישׁ אֲשֶׁ֥ר עָלָ֖ה לִקְרַאתְכֶ֑ם וַיְדַבֵּ֣ר אֲלֵיכֶ֔ם אֶת־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (כ) וַיֹּאמְר֣וּ אֵלָ֗יו אִ֚ישׁ בַּ֣עַל שֵׂעָ֔ר וְאֵז֥וֹר ע֖וֹר אָז֣וּר בְּמָתְנָ֑יו וַיֹּאמַ֕ר אֵלִיָּ֥ה הַתִּשְׁבִּ֖י הֽוּא׃ (כא) וַיִּשְׁלַ֥ח אֵלָ֛יו שַׂר־ חֲמִשִּׁ֖ים וַחֲמִשָּׁ֑יו וַיַּ֣עַל אֵלָ֗יו וְהִנֵּה֙ יֹשֵׁב֙ עַל־ רֹ֣אשׁ הָהָ֔ר וַיְדַבֵּ֣ר אֵלָ֔יו אִ֚ישׁ הָֽאֱלֹהִ֔ים הַמֶּ֥לֶךְ דִּבֶּ֖ר רֵֽדָה׃ (כב) וַיַּעֲנֶ֣ה אֵלִיָּ֗הוּ וַיְדַבֵּר֮ אֶל־ שַׂ֣ר הַחֲמִשִּׁים֒ וְאִם־ אִ֤ישׁ אֱלֹהִים֙ אָ֔נִי תֵּ֤רֶד אֵשׁ֙ מִן־ הַשָּׁמַ֔יִם וְתֹאכַ֥ל אֹתְךָ֖ וְאֶת־ חֲמִשֶּׁ֑יךָ וַתֵּ֤רֶד אֵשׁ֙ מִן־ הַשָּׁמַ֔יִם וַתֹּ֥אכַל אֹת֖וֹ וְאֶת־ חֲמִשָּֽׁיו׃ (כג) וַיָּ֜שָׁב וַיִּשְׁלַ֥ח אֵלָ֛יו שַׂר־ חֲמִשִּׁ֥ים אַחֵ֖ר וַחֲמִשָּׁ֑יו וַיַּ֙עַן֙ וַיְדַבֵּ֣ר אֵלָ֔יו אִ֚ישׁ הָאֱלֹהִ֔ים כֹּֽה־ אָמַ֥ר הַמֶּ֖לֶךְ מְהֵרָ֥ה רֵֽדָה׃ (כד) וַיַּ֣עַן אֵלִיָּה֮ וַיְדַבֵּ֣ר אֲלֵיהֶם֒ אִם־ אִ֤ישׁ הָֽאֱלֹהִים֙ אָ֔נִי תֵּ֤רֶד אֵשׁ֙ מִן־ הַשָּׁמַ֔יִם וְתֹאכַ֥ל אֹתְךָ֖ וְאֶת־ חֲמִשֶּׁ֑יךָ וַתֵּ֤רֶד אֵשׁ־ אֱלֹהִים֙ מִן־ הַשָּׁמַ֔יִם וַתֹּ֥אכַל אֹת֖וֹ וְאֶת־ חֲמִשָּֽׁיו׃ (כה) וַיָּ֗שָׁב וַיִּשְׁלַ֛ח שַׂר־ חֲמִשִּׁ֥ים שְׁלִשִׁ֖ים וַחֲמִשָּׁ֑יו וַיַּ֡עַל וַיָּבֹא֩ שַׂר־ הַחֲמִשִּׁ֨ים הַשְּׁלִישִׁ֜י וַיִּכְרַ֥ע עַל־ בִּרְכָּ֣יו ׀ לְנֶ֣גֶד אֵלִיָּ֗הוּ וַיִּתְחַנֵּ֤ן אֵלָיו֙ וַיְדַבֵּ֣ר אֵלָ֔יו אִ֚ישׁ הָֽאֱלֹהִ֔ים תִּֽיקַר־ נָ֣א נַפְשִׁ֗י וְנֶ֨פֶשׁ עֲבָדֶ֥יךָֽ אֵ֛לֶּה חֲמִשִּׁ֖ים בְּעֵינֶֽיךָ׃ (כו) הִ֠נֵּה יָ֤רְדָה אֵשׁ֙ מִן־ הַשָּׁמַ֔יִם וַ֠תֹּאכַל אֶת־ שְׁנֵ֞י שָׂרֵ֧י הַחֲמִשִּׁ֛ים הָרִאשֹׁנִ֖ים וְאֶת־ חֲמִשֵּׁיהֶ֑ם וְעַתָּ֕ה תִּיקַ֥ר נַפְשִׁ֖י בְּעֵינֶֽיךָ׃ (כז) וַיְדַבֵּ֞ר מַלְאַ֤ךְ יְהוָה֙ אֶל־ אֵ֣לִיָּ֔הוּ רֵ֣ד אוֹת֔וֹ אַל־ תִּירָ֖א מִפָּנָ֑יו וַיָּ֛קָם וַיֵּ֥רֶד אוֹת֖וֹ אֶל־ הַמֶּֽלֶךְ׃ (כח) וַיְדַבֵּ֨ר אֵלָ֜יו כֹּֽה־ אָמַ֣ר יְהוָ֗ה יַ֜עַן אֲשֶׁר־ שָׁלַ֣חְתָּ מַלְאָכִים֮ לִדְרֹשׁ֮ בְּבַ֣עַל זְבוּב֮ אֱלֹהֵ֣י עֶקְרוֹן֒ הַֽמִבְּלִ֤י אֵין־ אֱלֹהִים֙ בְּיִשְׂרָאֵ֔ל לִדְרֹ֖שׁ בִּדְבָר֑וֹ לָ֠כֵן הַמִּטָּ֞ה אֲשֶׁר־ עָלִ֥יתָ שָּׁ֛ם לֹֽא־ תֵרֵ֥ד מִמֶּ֖נָּה כִּֽי־ מ֥וֹת תָּמֽוּת׃ (כט) וַיָּ֜מָת כִּדְבַ֥ר יְהוָ֣ה ׀ אֲשֶׁר־ דִּבֶּ֣ר אֵלִיָּ֗הוּ וַיִּמְלֹ֤ךְ יְהוֹרָם֙ תַּחְתָּ֔יו בִּשְׁנַ֣ת שְׁתַּ֔יִם לִיהוֹרָ֥ם בֶּן־ יְהוֹשָׁפָ֖ט מֶ֣לֶךְ יְהוּדָ֑ה כִּ֛י לֹֽא־ הָ֥יָה ל֖וֹ בֵּֽן׃ (ל) וְיֶ֛תֶר דִּבְרֵ֥י אֲחַזְיָ֖הוּ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֽוֹא־ הֵ֣מָּה כְתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (לא) וַיְהִ֗י בְּהַעֲל֤וֹת יְהוָה֙ אֶת־ אֵ֣לִיָּ֔הוּ בַּֽסְעָרָ֖ה הַשָּׁמָ֑יִם וַיֵּ֧לֶךְ אֵלִיָּ֛הוּ וֶאֱלִישָׁ֖ע מִן־ הַגִּלְגָּֽל׃ (לב) וַיֹּאמֶר֩ אֵלִיָּ֨הוּ אֶל־ אֱלִישָׁ֜ע שֵֽׁב־ נָ֣א פֹ֗ה כִּ֤י יְהוָה֙ שְׁלָחַ֣נִי עַד־ בֵּֽית־ אֵ֔ל וַיֹּ֣אמֶר אֱלִישָׁ֔ע חַי־ יְהוָ֥ה וְחֵֽי־ נַפְשְׁךָ֖ אִם־ אֶעֶזְבֶ֑ךָּ וַיֵּרְד֖וּ בֵּֽית־ אֵֽל׃ (לג) וַיֵּצְא֨וּ בְנֵֽי־ הַנְּבִיאִ֥ים אֲשֶׁר־ בֵּֽית־ אֵל֮ אֶל־ אֱלִישָׁע֒ וַיֹּאמְר֣וּ אֵלָ֔יו הֲיָדַ֕עְתָּ כִּ֣י הַיּ֗וֹם יְהוָ֛ה לֹקֵ֥חַ אֶת־ אֲדֹנֶ֖יךָ מֵעַ֣ל רֹאשֶׁ֑ךָ וַיֹּ֛אמֶר גַּם־ אֲנִ֥י יָדַ֖עְתִּי הֶחֱשֽׁוּ׃ (לד) וַיֹּאמֶר֩ ל֨וֹ אֵלִיָּ֜הוּ אֱלִישָׁ֣ע ׀ שֵֽׁב־ נָ֣א פֹ֗ה כִּ֤י יְהוָה֙ שְׁלָחַ֣נִי יְרִיח֔וֹ וַיֹּ֕אמֶר חַי־ יְהוָ֥ה וְחֵֽי־ נַפְשְׁךָ֖ אִם־ אֶעֶזְבֶ֑ךָּ וַיָּבֹ֖אוּ יְרִיחֽוֹ׃ (לה) וַיִּגְּשׁ֨וּ בְנֵֽי־ הַנְּבִיאִ֥ים אֲשֶׁר־ בִּֽירִיחוֹ֮ אֶל־ אֱלִישָׁע֒ וַיֹּאמְר֣וּ אֵלָ֔יו הֲיָדַ֕עְתָּ כִּ֣י הַיּ֗וֹם יְהוָ֛ה לֹקֵ֥חַ אֶת־ אֲדֹנֶ֖יךָ מֵעַ֣ל רֹאשֶׁ֑ךָ וַיֹּ֛אמֶר גַּם־ אֲנִ֥י יָדַ֖עְתִּי הֶחֱשֽׁוּ׃ (לו) וַיֹּאמֶר֩ ל֨וֹ אֵלִיָּ֜הוּ שֵֽׁב־ נָ֣א פֹ֗ה כִּ֤י יְהוָה֙ שְׁלָחַ֣נִי הַיַּרְדֵּ֔נָה וַיֹּ֕אמֶר חַי־ יְהוָ֥ה וְחֵֽי־ נַפְשְׁךָ֖ אִם־ אֶעֶזְבֶ֑ךָּ וַיֵּלְכ֖וּ שְׁנֵיהֶֽם׃ (לז) וַחֲמִשִּׁ֨ים אִ֜ישׁ מִבְּנֵ֤י הַנְּבִיאִים֙ הָֽלְכ֔וּ וַיַּעַמְד֥וּ מִנֶּ֖גֶד מֵרָח֑וֹק וּשְׁנֵיהֶ֖ם עָמְד֥וּ עַל־ הַיַּרְדֵּֽן׃ (לח) וַיִּקַּח֩ אֵלִיָּ֨הוּ אֶת־ אַדַּרְתּ֤וֹ וַיִּגְלֹם֙ וַיַּכֶּ֣ה אֶת־ הַמַּ֔יִם וַיֵּחָצ֖וּ הֵ֣נָּה וָהֵ֑נָּה וַיַּעַבְר֥וּ שְׁנֵיהֶ֖ם בֶּחָרָבָֽה׃ (לט) וַיְהִ֣י כְעָבְרָ֗ם וְאֵ֨לִיָּ֜הוּ אָמַ֤ר אֶל־ אֱלִישָׁע֙ שְׁאַל֙ מָ֣ה אֶֽעֱשֶׂה־ לָּ֔ךְ בְּטֶ֖רֶם אֶלָּקַ֣ח מֵעִמָּ֑ךְ וַיֹּ֣אמֶר אֱלִישָׁ֔ע וִֽיהִי־ נָ֛א פִּֽי־ שְׁנַ֥יִם בְּרוּחֲךָ֖ אֵלָֽי׃ (מ) וַיֹּ֖אמֶר הִקְשִׁ֣יתָ לִשְׁא֑וֹל אִם־ תִּרְאֶ֨ה אֹתִ֜י לֻקָּ֤ח מֵֽאִתָּךְ֙ יְהִֽי־ לְךָ֣ כֵ֔ן וְאִם־ אַ֖יִן לֹ֥א יִהְיֶֽה׃ (מא) וַיְהִ֗י הֵ֣מָּה הֹלְכִ֤ים הָלוֹךְ֙ וְדַבֵּ֔ר וְהִנֵּ֤ה רֶֽכֶב־ אֵשׁ֙ וְס֣וּסֵי אֵ֔שׁ וַיַּפְרִ֖דוּ בֵּ֣ין שְׁנֵיהֶ֑ם וַיַּ֙עַל֙ אֵ֣לִיָּ֔הוּ בַּֽסְעָרָ֖ה הַשָּׁמָֽיִם׃ (מב) וֶאֱלִישָׁ֣ע רֹאֶ֗ה וְה֤וּא מְצַעֵק֙ אָבִ֣י ׀ אָבִ֗י רֶ֤כֶב יִשְׂרָאֵל֙ וּפָ֣רָשָׁ֔יו וְלֹ֥א רָאָ֖הוּ ע֑וֹד וַֽיַּחֲזֵק֙ בִּבְגָדָ֔יו וַיִּקְרָעֵ֖ם לִשְׁנַ֥יִם קְרָעִֽים׃ (מג) וַיָּ֙רֶם֙ אֶת־ אַדֶּ֣רֶת אֵלִיָּ֔הוּ אֲשֶׁ֥ר נָפְלָ֖ה מֵעָלָ֑יו וַיָּ֥שָׁב וַֽיַּעֲמֹ֖ד עַל־ שְׂפַ֥ת הַיַּרְדֵּֽן׃ (מד) וַיִּקַּח֩ אֶת־ אַדֶּ֨רֶת אֵלִיָּ֜הוּ אֲשֶׁר־ נָפְלָ֤ה מֵֽעָלָיו֙ וַיַּכֶּ֣ה אֶת־ הַמַּ֔יִם וַיֹּאמַ֕ר אַיֵּ֕ה יְהוָ֖ה אֱלֹהֵ֣י אֵלִיָּ֑הוּ אַף־ ה֣וּא ׀ וַיַּכֶּ֣ה אֶת־ הַמַּ֗יִם וַיֵּֽחָצוּ֙ הֵ֣נָּה וָהֵ֔נָּה וַֽיַּעֲבֹ֖ר אֱלִישָֽׁע׃
