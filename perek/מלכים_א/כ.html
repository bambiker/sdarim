מלכים א פרק כ לפי החלוקה לסדרים

פרק כ בספר מלכים א (43 פסוקים) נמצא במלכים סדר טז (פסוקים ל–מא) ובמלכים סדר יז (פסוקים א–לא).

מלכים סדר טז

(א) וַיַּרְא֙ כָּל־ הָעָ֔ם וַֽיִּפְּל֖וּ עַל־ פְּנֵיהֶ֑ם וַיֹּ֣אמְר֔וּ יְהוָה֙ ה֣וּא הָאֱלֹהִ֔ים יְהוָ֖ה ה֥וּא הָאֱלֹהִֽים׃ (ב) וַיֹּאמֶר֩ אֵלִיָּ֨הוּ לָהֶ֜ם תִּפְשׂ֣וּ ׀ אֶת־ נְבִיאֵ֣י הַבַּ֗עַל אִ֛ישׁ אַל־ יִמָּלֵ֥ט מֵהֶ֖ם וַֽיִּתְפְּשׂ֑וּם וַיּוֹרִדֵ֤ם אֵלִיָּ֙הוּ֙ אֶל־ נַ֣חַל קִישׁ֔וֹן וַיִּשְׁחָטֵ֖ם שָֽׁם׃ (ג) וַיֹּ֤אמֶר אֵלִיָּ֙הוּ֙ לְאַחְאָ֔ב עֲלֵ֖ה אֱכֹ֣ל וּשְׁתֵ֑ה כִּי־ ק֖וֹל הֲמ֥וֹן הַגָּֽשֶׁם׃ (ד) וַיַּעֲלֶ֥ה אַחְאָ֖ב לֶאֱכֹ֣ל וְלִשְׁתּ֑וֹת וְאֵ֨לִיָּ֜הוּ עָלָ֨ה אֶל־ רֹ֤אשׁ הַכַּרְמֶל֙ וַיִּגְהַ֣ר אַ֔רְצָה וַיָּ֥שֶׂם פָּנָ֖יו בֵּ֥ין (ברכו) בִּרְכָּֽיו׃ (ה) וַיֹּ֣אמֶר אֶֽל־ נַעֲר֗וֹ עֲלֵֽה־ נָא֙ הַבֵּ֣ט דֶּֽרֶךְ־ יָ֔ם וַיַּ֙עַל֙ וַיַּבֵּ֔ט וַיֹּ֖אמֶר אֵ֣ין מְא֑וּמָה וַיֹּ֕אמֶר שֻׁ֖ב שֶׁ֥בַע פְּעָמִֽים׃ (ו) וַֽיְהִי֙ בַּשְּׁבִעִ֔ית וַיֹּ֗אמֶר הִנֵּה־ עָ֛ב קְטַנָּ֥ה כְּכַף־ אִ֖ישׁ עֹלָ֣ה מִיָּ֑ם וַיֹּ֗אמֶר עֲלֵ֨ה אֱמֹ֤ר אֶל־ אַחְאָב֙ אֱסֹ֣ר וָרֵ֔ד וְלֹ֥א יַעַצָרְכָ֖ה הַגָּֽשֶׁם׃ (ז) וַיְהִ֣י ׀ עַד־ כֹּ֣ה וְעַד־ כֹּ֗ה וְהַשָּׁמַ֙יִם֙ הִֽתְקַדְּרוּ֙ עָבִ֣ים וְר֔וּחַ וַיְהִ֖י גֶּ֣שֶׁם גָּד֑וֹל וַיִּרְכַּ֥ב אַחְאָ֖ב וַיֵּ֥לֶךְ יִזְרְעֶֽאלָה׃ (ח) וְיַד־ יְהוָ֗ה הָֽיְתָה֙ אֶל־ אֵ֣לִיָּ֔הוּ וַיְשַׁנֵּ֖ס מָתְנָ֑יו וַיָּ֙רָץ֙ לִפְנֵ֣י אַחְאָ֔ב עַד־ בֹּאֲכָ֖ה יִזְרְעֶֽאלָה׃ (ט) וַיַּגֵּ֤ד אַחְאָב֙ לְאִיזֶ֔בֶל אֵ֛ת כָּל־ אֲשֶׁ֥ר עָשָׂ֖ה אֵלִיָּ֑הוּ וְאֵ֨ת כָּל־ אֲשֶׁ֥ר הָרַ֛ג אֶת־ כָּל־ הַנְּבִיאִ֖ים בֶּחָֽרֶב׃ (י) וַתִּשְׁלַ֤ח אִיזֶ֙בֶל֙ מַלְאָ֔ךְ אֶל־ אֵלִיָּ֖הוּ לֵאמֹ֑ר כֹּֽה־ יַעֲשׂ֤וּן אֱלֹהִים֙ וְכֹ֣ה יוֹסִפ֔וּן כִּֽי־ כָעֵ֤ת מָחָר֙ אָשִׂ֣ים אֶֽת־ נַפְשְׁךָ֔ כְּנֶ֖פֶשׁ אַחַ֥ד מֵהֶֽם׃ (יא) וַיַּ֗רְא וַיָּ֙קָם֙ וַיֵּ֣לֶךְ אֶל־ נַפְשׁ֔וֹ וַיָּבֹ֕א בְּאֵ֥ר שֶׁ֖בַע אֲשֶׁ֣ר לִֽיהוּדָ֑ה וַיַּנַּ֥ח אֶֽת־ נַעֲר֖וֹ שָֽׁם׃ (יב) וְהֽוּא־ הָלַ֤ךְ בַּמִּדְבָּר֙ דֶּ֣רֶךְ י֔וֹם וַיָּבֹ֕א וַיֵּ֕שֶׁב תַּ֖חַת רֹ֣תֶם (אחת) אֶחָ֑ד וַיִּשְׁאַ֤ל אֶת־ נַפְשׁוֹ֙ לָמ֔וּת וַיֹּ֣אמֶר ׀ רַ֗ב עַתָּ֤ה יְהוָה֙ קַ֣ח נַפְשִׁ֔י כִּֽי־ לֹא־ ט֥וֹב אָנֹכִ֖י מֵאֲבֹתָֽי׃ (יג) וַיִּשְׁכַּב֙ וַיִּישַׁ֔ן תַּ֖חַת רֹ֣תֶם אֶחָ֑ד וְהִנֵּֽה־ זֶ֤ה מַלְאָךְ֙ נֹגֵ֣עַ בּ֔וֹ וַיֹּ֥אמֶר ל֖וֹ ק֥וּם אֱכֽוֹל׃ (יד) וַיַּבֵּ֕ט וְהִנֵּ֧ה מְרַאֲשֹׁתָ֛יו עֻגַ֥ת רְצָפִ֖ים וְצַפַּ֣חַת מָ֑יִם וַיֹּ֣אכַל וַיֵּ֔שְׁתְּ וַיָּ֖שָׁב וַיִּשְׁכָּֽב׃ (טו) וַיָּשָׁב֩ מַלְאַ֨ךְ יְהוָ֤ה ׀ שֵׁנִית֙ וַיִּגַּע־ בּ֔וֹ וַיֹּ֖אמֶר ק֣וּם אֱכֹ֑ל כִּ֛י רַ֥ב מִמְּךָ֖ הַדָּֽרֶךְ׃ (טז) וַיָּ֖קָם וַיֹּ֣אכַל וַיִּשְׁתֶּ֑ה וַיֵּ֜לֶךְ בְּכֹ֣חַ ׀ הָאֲכִילָ֣ה הַהִ֗יא אַרְבָּעִ֥ים יוֹם֙ וְאַרְבָּעִ֣ים לַ֔יְלָה עַ֛ד הַ֥ר הָאֱלֹהִ֖ים חֹרֵֽב׃ (יז) וַיָּבֹא־ שָׁ֥ם אֶל־ הַמְּעָרָ֖ה וַיָּ֣לֶן שָׁ֑ם וְהִנֵּ֤ה דְבַר־ יְהוָה֙ אֵלָ֔יו וַיֹּ֣אמֶר ל֔וֹ מַה־ לְּךָ֥ פֹ֖ה אֵלִיָּֽהוּ׃ (יח) וַיֹּאמֶר֩ קַנֹּ֨א קִנֵּ֜אתִי לַיהוָ֣ה ׀ אֱלֹהֵ֣י צְבָא֗וֹת כִּֽי־ עָזְב֤וּ בְרִֽיתְךָ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶת־ מִזְבְּחֹתֶ֣יךָ הָרָ֔סוּ וְאֶת־ נְבִיאֶ֖יךָ הָרְג֣וּ בֶחָ֑רֶב וָֽאִוָּתֵ֤ר אֲנִי֙ לְבַדִּ֔י וַיְבַקְשׁ֥וּ אֶת־ נַפְשִׁ֖י לְקַחְתָּֽהּ׃ (יט) וַיֹּ֗אמֶר צֵ֣א וְעָמַדְתָּ֣ בָהָר֮ לִפְנֵ֣י יְהוָה֒ וְהִנֵּ֧ה יְהוָ֣ה עֹבֵ֗ר וְר֣וּחַ גְּדוֹלָ֡ה וְחָזָ֞ק מְפָרֵק֩ הָרִ֨ים וּמְשַׁבֵּ֤ר סְלָעִים֙ לִפְנֵ֣י יְהוָ֔ה לֹ֥א בָר֖וּחַ יְהוָ֑ה וְאַחַ֤ר הָר֨וּחַ רַ֔עַשׁ לֹ֥א בָרַ֖עַשׁ יְהוָֽה׃ (כ) וְאַחַ֤ר הָרַ֙עַשׁ֙ אֵ֔שׁ לֹ֥א בָאֵ֖שׁ יְהוָ֑ה וְאַחַ֣ר הָאֵ֔שׁ ק֖וֹל דְּמָמָ֥ה דַקָּֽה׃ (כא) וַיְהִ֣י ׀ כִּשְׁמֹ֣עַ אֵלִיָּ֗הוּ וַיָּ֤לֶט פָּנָיו֙ בְּאַדַּרְתּ֔וֹ וַיֵּצֵ֕א וַֽיַּעֲמֹ֖ד פֶּ֣תַח הַמְּעָרָ֑ה וְהִנֵּ֤ה אֵלָיו֙ ק֔וֹל וַיֹּ֕אמֶר מַה־ לְּךָ֥ פֹ֖ה אֵלִיָּֽהוּ׃ (כב) וַיֹּאמֶר֩ קַנֹּ֨א קִנֵּ֜אתִי לַיהוָ֣ה ׀ אֱלֹהֵ֣י צְבָא֗וֹת כִּֽי־ עָזְב֤וּ בְרִֽיתְךָ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶת־ מִזְבְּחֹתֶ֣יךָ הָרָ֔סוּ וְאֶת־ נְבִיאֶ֖יךָ הָרְג֣וּ בֶחָ֑רֶב וָאִוָּתֵ֤ר אֲנִי֙ לְבַדִּ֔י וַיְבַקְשׁ֥וּ אֶת־ נַפְשִׁ֖י לְקַחְתָּֽהּ׃ (כג) וַיֹּ֤אמֶר יְהוָה֙ אֵלָ֔יו לֵ֛ךְ שׁ֥וּב לְדַרְכְּךָ֖ מִדְבַּ֣רָה דַמָּ֑שֶׂק וּבָ֗אתָ וּמָשַׁחְתָּ֧ אֶת־ חֲזָאֵ֛ל לְמֶ֖לֶךְ עַל־ אֲרָֽם׃ (כד) וְאֵת֙ יֵה֣וּא בֶן־ נִמְשִׁ֔י תִּמְשַׁ֥ח לְמֶ֖לֶךְ עַל־ יִשְׂרָאֵ֑ל וְאֶת־ אֱלִישָׁ֤ע בֶּן־ שָׁפָט֙ מֵאָבֵ֣ל מְחוֹלָ֔ה תִּמְשַׁ֥ח לְנָבִ֖יא תַּחְתֶּֽיךָ׃ (כה) וְהָיָ֗ה הַנִּמְלָ֛ט מֵחֶ֥רֶב חֲזָאֵ֖ל יָמִ֣ית יֵה֑וּא וְהַנִּמְלָ֛ט מֵחֶ֥רֶב יֵה֖וּא יָמִ֥ית אֱלִישָֽׁע׃ (כו) וְהִשְׁאַרְתִּ֥י בְיִשְׂרָאֵ֖ל שִׁבְעַ֣ת אֲלָפִ֑ים כָּל־ הַבִּרְכַּ֗יִם אֲשֶׁ֤ר לֹֽא־ כָֽרְעוּ֙ לַבַּ֔עַל וְכָ֨ל־ הַפֶּ֔ה אֲשֶׁ֥ר לֹֽא־ נָשַׁ֖ק לֽוֹ׃ (כז) וַיֵּ֣לֶךְ מִ֠שָּׁם וַיִּמְצָ֞א אֶת־ אֱלִישָׁ֤ע בֶּן־ שָׁפָט֙ וְה֣וּא חֹרֵ֔שׁ שְׁנֵים־ עָשָׂ֤ר צְמָדִים֙ לְפָנָ֔יו וְה֖וּא בִּשְׁנֵ֣ים הֶעָשָׂ֑ר וַיַּעֲבֹ֤ר אֵלִיָּ֙הוּ֙ אֵלָ֔יו וַיַּשְׁלֵ֥ךְ אַדַּרְתּ֖וֹ אֵלָֽיו׃ (כח) וַיַּעֲזֹ֣ב אֶת־ הַבָּקָ֗ר וַיָּ֙רָץ֙ אַחֲרֵ֣י אֵֽלִיָּ֔הוּ וַיֹּ֗אמֶר אֶשְּׁקָה־ נָּא֙ לְאָבִ֣י וּלְאִמִּ֔י וְאֵלְכָ֖ה אַחֲרֶ֑יךָ וַיֹּ֤אמֶר לוֹ֙ לֵ֣ךְ שׁ֔וּב כִּ֥י מֶה־ עָשִׂ֖יתִי לָֽךְ׃ (כט) וַיָּ֨שָׁב מֵאַחֲרָ֜יו וַיִּקַּ֣ח אֶת־ צֶ֧מֶד הַבָּקָ֣ר וַיִּזְבָּחֵ֗הוּ וּבִכְלִ֤י הַבָּקָר֙ בִּשְּׁלָ֣ם הַבָּשָׂ֔ר וַיִּתֵּ֥ן לָעָ֖ם וַיֹּאכֵ֑לוּ וַיָּ֗קָם וַיֵּ֛לֶךְ אַחֲרֵ֥י אֵלִיָּ֖הוּ וַיְשָׁרְתֵֽהוּ׃ תחילת פרק כ (ל) וּבֶן־ הֲדַ֣ד מֶֽלֶךְ־ אֲרָ֗ם קָבַץ֙ אֶת־ כָּל־ חֵיל֔וֹ וּשְׁלֹשִׁ֨ים וּשְׁנַ֥יִם מֶ֛לֶךְ אִתּ֖וֹ וְס֣וּס וָרָ֑כֶב וַיַּ֗עַל וַיָּ֙צַר֙ עַל־ שֹׁ֣מְר֔וֹן וַיִּלָּ֖חֶם בָּֽהּ׃ (לא) וַיִּשְׁלַ֧ח מַלְאָכִ֛ים אֶל־ אַחְאָ֥ב מֶֽלֶךְ־ יִשְׂרָאֵ֖ל הָעִֽירָה׃ (לב) וַיֹּ֣אמֶר ל֗וֹ כֹּ֚ה אָמַ֣ר בֶּן־ הֲדַ֔ד כַּסְפְּךָ֥ וּֽזְהָבְךָ֖ לִֽי־ ה֑וּא וְנָשֶׁ֧יךָ וּבָנֶ֛יךָ הַטּוֹבִ֖ים לִי־ הֵֽם׃ (לג) וַיַּ֤עַן מֶֽלֶךְ־ יִשְׂרָאֵל֙ וַיֹּ֔אמֶר כִּדְבָרְךָ֖ אֲדֹנִ֣י הַמֶּ֑לֶךְ לְךָ֥ אֲנִ֖י וְכָל־ אֲשֶׁר־ לִֽי׃ (לד) וַיָּשֻׁ֙בוּ֙ הַמַּלְאָכִ֔ים וַיֹּ֣אמְר֔וּ כֹּֽה־ אָמַ֥ר בֶּן־ הֲדַ֖ד לֵאמֹ֑ר כִּֽי־ שָׁלַ֤חְתִּי אֵלֶ֙יךָ֙ לֵאמֹ֔ר כַּסְפְּךָ֧ וּזְהָבְךָ֛ וְנָשֶׁ֥יךָ וּבָנֶ֖יךָ לִ֥י תִתֵּֽן׃ (לה) כִּ֣י ׀ אִם־ כָּעֵ֣ת מָחָ֗ר אֶשְׁלַ֤ח אֶת־ עֲבָדַי֙ אֵלֶ֔יךָ וְחִפְּשׂוּ֙ בַּֽצָּהֳרָ֑יִם וּבֶן־ הֲדַד֩ שֹׁתֶ֨ה שִׁכּ֜וֹר בַּסֻּכּ֗וֹת ה֧וּא וְהַמְּלָכִ֛ים שְׁלֹשִֽׁים־ וּשְׁנַ֥יִם מֶ֖לֶךְ עֹזֵ֥ר אֹתֽוֹ׃ (ה) וַיֵּצְא֗וּ נַעֲרֵ֛י שָׂרֵ֥י הַמְּדִינ֖וֹת בָּרִֽאשֹׁנָ֑ה וַיִּשְׁלַ֣ח בֶּן־ הֲדַ֗ד וַיַּגִּ֤ידוּ לוֹ֙ לֵאמֹ֔ר אֲנָשִׁ֕ים יָצְא֖וּ מִשֹּׁמְרֽוֹן׃ (ו) וַיֹּ֛אמֶר אִם־ לְשָׁל֥וֹם יָצָ֖אוּ תִּפְשׂ֣וּם חַיִּ֑ים וְאִ֧ם לְמִלְחָמָ֛ה יָצָ֖אוּ חַיִּ֥ים תִּפְשֽׂוּם׃ (ז) וְאֵ֙לֶּה֙ יָצְא֣וּ מִן־ הָעִ֔יר נַעֲרֵ֖י שָׂרֵ֣י הַמְּדִינ֑וֹת וְהַחַ֖יִל אֲשֶׁ֥ר אַחֲרֵיהֶֽם׃ (ח) וַיַּכּוּ֙ אִ֣ישׁ אִישׁ֔וֹ וַיָּנֻ֣סוּ אֲרָ֔ם וַֽיִּרְדְּפֵ֖ם יִשְׂרָאֵ֑ל וַיִּמָּלֵ֗ט בֶּן־ הֶעָרִ֣ים אֲשֶׁר־ לָֽקַח־ אָבִי֩ מֵאֵ֨ת אָבִ֜יךָ אָשִׁ֗יב וְ֠חוּצוֹת תָּשִׂ֨ים לְךָ֤ בְדַמֶּ֙שֶׂק֙ כַּאֲשֶׁר־ שָׂ֤ם אָבִי֙ בְּשֹׁ֣מְר֔וֹן וַאֲנִ֖י בַּבְּרִ֣ית אֲשַׁלְּחֶ֑ךָּ וַיִּכְרָת־ ל֥וֹ בְרִ֖ית וַֽיְשַׁלְּחֵֽהוּ׃ (כג) וְאִ֨ישׁ אֶחָ֜ד מִבְּנֵ֣י הַנְּבִיאִ֗ים אָמַ֧ר אֶל־ רֵעֵ֛הוּ בִּדְבַ֥ר יְהוָ֖ה הַכֵּ֣ינִי נָ֑א וַיְמָאֵ֥ן הָאִ֖ישׁ לְהַכֹּתֽוֹ׃ (כד) וַיֹּ֣אמֶר ל֗וֹ יַ֚עַן אֲשֶׁ֤ר לֹֽא־ שָׁמַ֙עְתָּ֙ בְּק֣וֹל יְהוָ֔ה הִנְּךָ֤ הוֹלֵךְ֙ מֵֽאִתִּ֔י וְהִכְּךָ֖ הָאַרְיֵ֑ה וַיֵּ֙לֶךְ֙ מֵֽאֶצְל֔וֹ וַיִּמְצָאֵ֥הוּ הָאַרְיֵ֖ה וַיַּכֵּֽהוּ׃ (כה) וַיִּמְצָא֙ אִ֣ישׁ אַחֵ֔ר וַיֹּ֖אמֶר הַכֵּ֣ינִי נָ֑א וַיַּכֵּ֥הוּ הָאִ֖ישׁ הַכֵּ֥ה וּפָצֹֽעַ׃ (כו) וַיֵּ֙לֶךְ֙ הַנָּבִ֔יא וַיַּעֲמֹ֥ד לַמֶּ֖לֶךְ עַל־ הַדָּ֑רֶךְ וַיִּתְחַפֵּ֥שׂ בָּאֲפֵ֖ר עַל־ עֵינָֽיו׃ (כז) וַיְהִ֤י הַמֶּ֙לֶךְ֙ עֹבֵ֔ר וְה֖וּא צָעַ֣ק אֶל־ הַמֶּ֑לֶךְ וַיֹּ֜אמֶר עַבְדְּךָ֣ ׀ יָצָ֣א בְקֶֽרֶב־ הַמִּלְחָמָ֗ה וְהִנֵּֽה־ אִ֨ישׁ סָ֜ר וַיָּבֵ֧א אֵלַ֣י אִ֗ישׁ וַיֹּ֙אמֶר֙ שְׁמֹר֙ אֶת־ הָאִ֣ישׁ הַזֶּ֔ה אִם־ הִפָּקֵד֙ יִפָּקֵ֔ד וְהָיְתָ֤ה נַפְשְׁךָ֙ תַּ֣חַת נַפְשׁ֔וֹ א֥וֹ כִכַּר־ כֶּ֖סֶף תִּשְׁקֽוֹל׃ (כח) וַיְהִ֣י עַבְדְּךָ֗ עֹשֵׂ֥ה הֵ֛נָּה וָהֵ֖נָּה וְה֣וּא אֵינֶ֑נּוּ וַיֹּ֨אמֶר אֵלָ֧יו מֶֽלֶךְ־ יִשְׂרָאֵ֛ל כֵּ֥ן מִשְׁפָּטֶ֖ךָ אַתָּ֥ה חָרָֽצְתָּ׃ (כט) וַיְמַהֵ֕ר וַיָּ֙סַר֙ אֶת־ הָ֣אֲפֵ֔ר (מעל) מֵעֲלֵ֖י עֵינָ֑יו וַיַּכֵּ֤ר אֹתוֹ֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל כִּ֥י מֵֽהַנְּבִאִ֖ים הֽוּא׃ (ל) וַיֹּ֣אמֶר אֵלָ֗יו כֹּ֚ה אָמַ֣ר יְהוָ֔ה יַ֛עַן שִׁלַּ֥חְתָּ אֶת־ אִישׁ־ חֶרְמִ֖י מִיָּ֑ד וְהָיְתָ֤ה נַפְשְׁךָ֙ תַּ֣חַת נַפְשׁ֔וֹ וְעַמְּךָ֖ תַּ֥חַת עַמּֽוֹ׃ (לא) וַיֵּ֧לֶךְ מֶֽלֶךְ־ יִשְׂרָאֵ֛ל עַל־ בֵּית֖וֹ סַ֣ר וְזָעֵ֑ף וַיָּבֹ֖א שֹׁמְרֽוֹנָה׃ סוף פרק כ (לב) וַיְהִ֗י אַחַר֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה כֶּ֧רֶם הָיָ֛ה לְנָב֥וֹת הַיִּזְרְעֵאלִ֖י אֲשֶׁ֣ר בְּיִזְרְעֶ֑אל אֵ֚צֶל הֵיכַ֣ל אַחְאָ֔ב מֶ֖לֶךְ שֹׁמְרֽוֹן׃ (לג) וַיְדַבֵּ֣ר אַחְאָ֣ב אֶל־ נָב֣וֹת ׀ לֵאמֹר֩ ׀ תְּנָה־ לִּ֨י אֶֽת־ כַּרְמְךָ֜ וִֽיהִי־ לִ֣י לְגַן־ יָרָ֗ק כִּ֣י ה֤וּא קָרוֹב֙ אֵ֣צֶל בֵּיתִ֔י וְאֶתְּנָ֤ה לְךָ֙ תַּחְתָּ֔יו כֶּ֖רֶם ט֣וֹב מִמֶּ֑נּוּ אִ֚ם ט֣וֹב בְּעֵינֶ֔יךָ אֶתְּנָה־ לְךָ֥ כֶ֖סֶף מְחִ֥יר זֶֽה׃ (לד) וַיֹּ֥אמֶר נָב֖וֹת אֶל־ אַחְאָ֑ב חָלִ֤ילָה לִּי֙ מֵֽיהוָ֔ה מִתִּתִּ֛י אֶת־ נַחֲלַ֥ת אֲבֹתַ֖י לָֽךְ׃ (לה) וַיָּבֹא֩ אַחְאָ֨ב אֶל־ בֵּית֜וֹ סַ֣ר וְזָעֵ֗ף עַל־ הַדָּבָר֙ אֲשֶׁר־ דִּבֶּ֣ר אֵלָ֗יו נָבוֹת֙ הַיִּזְרְעֵאלִ֔י וַיֹּ֕אמֶר לֹֽא־ אֶתֵּ֥ן לְךָ֖ אֶת־ נַחֲלַ֣ת אֲבוֹתָ֑י וַיִּשְׁכַּב֙ עַל־ מִטָּת֔וֹ וַיַּסֵּ֥ב אֶת־ פָּנָ֖יו וְלֹֽא־ אָ֥כַל לָֽחֶם׃ (לו) וַתָּבֹ֥א אֵלָ֖יו אִיזֶ֣בֶל אִשְׁתּ֑וֹ וַתְּדַבֵּ֣ר אֵלָ֗יו מַה־ זֶּה֙ רוּחֲךָ֣ סָרָ֔ה וְאֵינְךָ֖ אֹכֵ֥ל לָֽחֶם׃ (לז) וַיְדַבֵּ֣ר אֵלֶ֗יהָ כִּֽי־ אֲ֠דַבֵּר אֶל־ נָב֨וֹת הַיִּזְרְעֵאלִ֜י וָאֹ֣מַר ל֗וֹ תְּנָה־ לִּ֤י אֶֽת־ כַּרְמְךָ֙ בְּכֶ֔סֶף א֚וֹ אִם־ חָפֵ֣ץ אַתָּ֔ה אֶתְּנָה־ לְךָ֥ כֶ֖רֶם תַּחְתָּ֑יו וַיֹּ֕אמֶר לֹֽא־ אֶתֵּ֥ן לְךָ֖ אֶת־ כַּרְמִֽי׃ (לח) וַתֹּ֤אמֶר אֵלָיו֙ אִיזֶ֣בֶל אִשְׁתּ֔וֹ אַתָּ֕ה עַתָּ֛ה תַּעֲשֶׂ֥ה מְלוּכָ֖ה עַל־ יִשְׂרָאֵ֑ל ק֤וּם אֱכָל־ לֶ֙חֶם֙ וְיִטַ֣ב לִבֶּ֔ךָ אֲנִי֙ אֶתֵּ֣ן לְךָ֔ אֶת־ כֶּ֖רֶם נָב֥וֹת הַיִּזְרְעֵאלִֽי׃ (לט) וַתִּכְתֹּ֤ב סְפָרִים֙ בְּשֵׁ֣ם אַחְאָ֔ב וַתַּחְתֹּ֖ם בְּחֹתָמ֑וֹ וַתִּשְׁלַ֣ח (הספרים) סְפָרִ֗ים אֶל־ הַזְקֵנִ֤ים וְאֶל־ הַֽחֹרִים֙ אֲשֶׁ֣ר בְּעִיר֔וֹ הַיֹּשְׁבִ֖ים אֶת־ נָבֽוֹת׃ (מ) וַתִּכְתֹּ֥ב בַּסְּפָרִ֖ים לֵאמֹ֑ר קִֽרְאוּ־ צ֔וֹם וְהוֹשִׁ֥יבוּ אֶת־ נָב֖וֹת בְּרֹ֥אשׁ הָעָֽם׃ (מא) וְ֠הוֹשִׁיבוּ שְׁנַ֨יִם אֲנָשִׁ֥ים בְּנֵֽי־ בְלִיַּעַל֮ נֶגְדּוֹ֒ וִיעִדֻ֣הוּ לֵאמֹ֔ר בֵּרַ֥כְתָּ אֱלֹהִ֖ים וָמֶ֑לֶךְ וְהוֹצִיאֻ֥הוּ וְסִקְלֻ֖הוּ וְיָמֹֽת׃ (מב) וַיַּעֲשׂוּ֩ אַנְשֵׁ֨י עִיר֜וֹ הַזְּקֵנִ֣ים וְהַחֹרִ֗ים אֲשֶׁ֤ר הַיֹּֽשְׁבִים֙ בְּעִיר֔וֹ כַּאֲשֶׁ֛ר שָׁלְחָ֥ה אֲלֵיהֶ֖ם אִיזָ֑בֶל כַּאֲשֶׁ֤ר כָּתוּב֙ בַּסְּפָרִ֔ים אֲשֶׁ֥ר שָׁלְחָ֖ה אֲלֵיהֶֽם׃ (מג) קָרְא֖וּ צ֑וֹם וְהֹשִׁ֥יבוּ אֶת־ נָב֖וֹת בְּרֹ֥אשׁ הָעָֽם׃ (מד) וַ֠יָּבֹאוּ שְׁנֵ֨י הָאֲנָשִׁ֥ים בְּנֵֽי־ בְלִיַּעַל֮ וַיֵּשְׁב֣וּ נֶגְדּוֹ֒ וַיְעִדֻהוּ֩ אַנְשֵׁ֨י הַבְּלִיַּ֜עַל אֶת־ נָב֗וֹת נֶ֤גֶד הָעָם֙ לֵאמֹ֔ר בֵּרַ֥ךְ נָב֛וֹת אֱלֹהִ֖ים וָמֶ֑לֶךְ וַיֹּצִאֻ֙הוּ֙ מִח֣וּץ לָעִ֔יר וַיִּסְקְלֻ֥הוּ בָאֲבָנִ֖ים וַיָּמֹֽת׃ (מה) וַֽיִּשְׁלְח֖וּ אֶל־ אִיזֶ֣בֶל לֵאמֹ֑ר סֻקַּ֥ל נָב֖וֹת וַיָּמֹֽת׃ (מו) וַֽיְהִי֙ כִּשְׁמֹ֣עַ אִיזֶ֔בֶל כִּֽי־ סֻקַּ֥ל נָב֖וֹת וַיָּמֹ֑ת וַתֹּ֨אמֶר אִיזֶ֜בֶל אֶל־ אַחְאָ֗ב ק֣וּם רֵ֞שׁ אֶת־ כֶּ֣רֶם ׀ נָב֣וֹת הַיִּזְרְעֵאלִ֗י אֲשֶׁ֤ר מֵאֵן֙ לָתֶת־ לְךָ֣ בְכֶ֔סֶף כִּ֣י אֵ֥ין נָב֛וֹת חַ֖י כִּי־ מֵֽת׃ (מז) וַיְהִ֛י כִּשְׁמֹ֥עַ אַחְאָ֖ב כִּ֣י מֵ֣ת נָב֑וֹת וַיָּ֣קָם אַחְאָ֗ב לָרֶ֛דֶת אֶל־ כֶּ֛רֶם נָב֥וֹת הַיִּזְרְעֵאלִ֖י לְרִשְׁתּֽוֹ׃ (מח) וַיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ אֵלִיָּ֥הוּ הַתִּשְׁבִּ֖י לֵאמֹֽר׃
