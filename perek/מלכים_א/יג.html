מלכים א פרק יג לפי החלוקה לסדרים

פרק יג בספר מלכים א (34 פסוקים) נמצא במלכים סדר יא (פסוקים יא–מ) ובמלכים סדר יב (פסוקים א–ד).

מלכים סדר יא

(א) כֹּ֣ה אָמַ֣ר יְהוָ֡ה לֹֽא־ תַעֲלוּ֩ וְלֹא־ תִלָּ֨חֲמ֜וּן עִם־ אֲחֵיכֶ֣ם בְּנֵֽי־ יִשְׂרָאֵ֗ל שׁ֚וּבוּ אִ֣ישׁ לְבֵית֔וֹ כִּ֧י מֵאִתִּ֛י נִהְיָ֖ה הַדָּבָ֣ר הַזֶּ֑ה וַיִּשְׁמְעוּ֙ אֶת־ דְּבַ֣ר יְהוָ֔ה וַיָּשֻׁ֥בוּ לָלֶ֖כֶת כִּדְבַ֥ר יְהוָֽה׃ (ב) וַיִּ֨בֶן יָרָבְעָ֧ם אֶת־ שְׁכֶ֛ם בְּהַ֥ר אֶפְרַ֖יִם וַיֵּ֣שֶׁב בָּ֑הּ וַיֵּצֵ֣א מִשָּׁ֔ם וַיִּ֖בֶן אֶת־ פְּנוּאֵֽל׃ (ג) וַיֹּ֥אמֶר יָרָבְעָ֖ם בְּלִבּ֑וֹ עַתָּ֛ה תָּשׁ֥וּב הַמַּמְלָכָ֖ה לְבֵ֥ית דָּוִֽד׃ (ד) אִֽם־ יַעֲלֶ֣ה ׀ הָעָ֣ם הַזֶּ֗ה לַעֲשׂ֨וֹת זְבָחִ֤ים בְּבֵית־ יְהוָה֙ בִּיר֣וּשָׁלִַ֔ם וְ֠שָׁב לֵ֣ב הָעָ֤ם הַזֶּה֙ אֶל־ אֲדֹ֣נֵיהֶ֔ם אֶל־ רְחַבְעָ֖ם מֶ֣לֶךְ יְהוּדָ֑ה וַהֲרָגֻ֕נִי וְשָׁ֖בוּ אֶל־ רְחַבְעָ֥ם מֶֽלֶךְ־ יְהוּדָֽה׃ (ה) וַיִּוָּעַ֣ץ הַמֶּ֔לֶךְ וַיַּ֕עַשׂ שְׁנֵ֖י עֶגְלֵ֣י זָהָ֑ב וַיֹּ֣אמֶר אֲלֵהֶ֗ם רַב־ לָכֶם֙ מֵעֲל֣וֹת יְרוּשָׁלִַ֔ם הִנֵּ֤ה אֱלֹהֶ֙יךָ֙ יִשְׂרָאֵ֔ל אֲשֶׁ֥ר הֶעֱל֖וּךָ מֵאֶ֥רֶץ מִצְרָֽיִם׃ (ו) וַיָּ֥שֶׂם אֶת־ הָאֶחָ֖ד בְּבֵֽית־ אֵ֑ל וְאֶת־ הָאֶחָ֖ד נָתַ֥ן בְּדָֽן׃ (ז) וַיְהִ֛י הַדָּבָ֥ר הַזֶּ֖ה לְחַטָּ֑את וַיֵּלְכ֥וּ הָעָ֛ם לִפְנֵ֥י הָאֶחָ֖ד עַד־ דָּֽן׃ (ח) וַיַּ֖עַשׂ אֶת־ בֵּ֣ית בָּמ֑וֹת וַיַּ֤עַשׂ כֹּֽהֲנִים֙ מִקְצ֣וֹת הָעָ֔ם אֲשֶׁ֥ר לֹֽא־ הָי֖וּ מִבְּנֵ֥י לֵוִֽי׃ (ט) וַיַּ֣עַשׂ יָרָבְעָ֣ם ׀ חָ֡ג בַּחֹ֣דֶשׁ הַשְּׁמִינִ֣י בַּחֲמִשָּֽׁה־ עָשָׂר֩ י֨וֹם ׀ לַחֹ֜דֶשׁ כֶּחָ֣ג ׀ אֲשֶׁ֣ר בִּיהוּדָ֗ה וַיַּ֙עַל֙ עַל־ הַמִּזְבֵּ֔חַ כֵּ֤ן עָשָׂה֙ בְּבֵֽית־ אֵ֔ל לְזַבֵּ֖חַ לָעֲגָלִ֣ים אֲשֶׁר־ עָשָׂ֑ה וְהֶעֱמִיד֙ בְּבֵ֣ית אֵ֔ל אֶת־ כֹּהֲנֵ֥י הַבָּמ֖וֹת אֲשֶׁ֥ר עָשָֽׂה׃ (י) וַיַּ֜עַל עַֽל־ הַמִּזְבֵּ֣חַ ׀ אֲשֶׁר־ עָשָׂ֣ה בְּבֵֽית־ אֵ֗ל בַּחֲמִשָּׁ֨ה עָשָׂ֥ר יוֹם֙ בַּחֹ֣דֶשׁ הַשְּׁמִינִ֔י בַּחֹ֖דֶשׁ אֲשֶׁר־ בָּדָ֣א (מלבד) מִלִּבּ֑וֹ וַיַּ֤עַשׂ חָג֙ לִבְנֵ֣י יִשְׂרָאֵ֔ל וַיַּ֥עַל עַל־ הַמִּזְבֵּ֖חַ לְהַקְטִֽיר׃ תחילת פרק יג (יא) וְהִנֵּ֣ה ׀ אִ֣ישׁ אֱלֹהִ֗ים בָּ֧א מִיהוּדָ֛ה בִּדְבַ֥ר יְהוָ֖ה אֶל־ בֵּֽית־ אֵ֑ל וְיָרָבְעָ֛ם עֹמֵ֥ד עַל־ הַמִּזְבֵּ֖חַ לְהַקְטִֽיר׃ (יב) וַיִּקְרָ֤א עַל־ הַמִּזְבֵּ֙חַ֙ בִּדְבַ֣ר יְהוָ֔ה וַיֹּ֙אמֶר֙ מִזְבֵּ֣חַ מִזְבֵּ֔חַ כֹּ֖ה אָמַ֣ר יְהוָ֑ה הִנֵּֽה־ בֵ֞ן נוֹלָ֤ד לְבֵית־ דָּוִד֙ יֹאשִׁיָּ֣הוּ שְׁמ֔וֹ וְזָבַ֣ח עָלֶ֗יךָ אֶת־ כֹּהֲנֵ֤י הַבָּמוֹת֙ הַמַּקְטִרִ֣ים עָלֶ֔יךָ וְעַצְמ֥וֹת אָדָ֖ם יִשְׂרְפ֥וּ עָלֶֽיךָ׃ (יג) וְנָתַן֩ בַּיּ֨וֹם הַה֤וּא מוֹפֵת֙ לֵאמֹ֔ר זֶ֣ה הַמּוֹפֵ֔ת אֲשֶׁ֖ר דִּבֶּ֣ר יְהוָ֑ה הִנֵּ֤ה הַמִּזְבֵּ֙חַ֙ נִקְרָ֔ע וְנִשְׁפַּ֖ךְ הַדֶּ֥שֶׁן אֲשֶׁר־ עָלָֽיו׃ (יד) וַיְהִי֩ כִשְׁמֹ֨עַ הַמֶּ֜לֶךְ אֶת־ דְּבַ֣ר אִישׁ־ הָאֱלֹהִ֗ים אֲשֶׁ֨ר קָרָ֤א עַל־ הַמִּזְבֵּ֙חַ֙ בְּבֵֽית־ אֵ֔ל וַיִּשְׁלַ֨ח יָרָבְעָ֧ם אֶת־ יָד֛וֹ מֵעַ֥ל הַמִּזְבֵּ֖חַ לֵאמֹ֣ר ׀ תִּפְשֻׂ֑הוּ וַתִּיבַ֤שׁ יָדוֹ֙ אֲשֶׁ֣ר שָׁלַ֣ח עָלָ֔יו וְלֹ֥א יָכֹ֖ל לַהֲשִׁיבָ֥הּ אֵלָֽיו׃ (טו) וְהַמִּזְבֵּ֣חַ נִקְרָ֔ע וַיִּשָּׁפֵ֥ךְ הַדֶּ֖שֶׁן מִן־ הַמִּזְבֵּ֑חַ כַּמּוֹפֵ֗ת אֲשֶׁ֥ר נָתַ֛ן אִ֥ישׁ הָאֱלֹהִ֖ים בִּדְבַ֥ר יְהוָֽה׃ (טז) וַיַּ֨עַן הַמֶּ֜לֶךְ וַיֹּ֣אמֶר ׀ אֶל־ אִ֣ישׁ הָאֱלֹהִ֗ים חַל־ נָ֞א אֶת־ פְּנֵ֨י יְהוָ֤ה אֱלֹהֶ֙יךָ֙ וְהִתְפַּלֵּ֣ל בַּעֲדִ֔י וְתָשֹׁ֥ב יָדִ֖י אֵלָ֑י וַיְחַ֤ל אִישׁ־ הָֽאֱלֹהִים֙ אֶת־ פְּנֵ֣י יְהוָ֔ה וַתָּ֤שָׁב יַד־ הַמֶּ֙לֶךְ֙ אֵלָ֔יו וַתְּהִ֖י כְּבָרִֽאשֹׁנָֽה׃ (יז) וַיְדַבֵּ֤ר הַמֶּ֙לֶךְ֙ אֶל־ אִ֣ישׁ הָאֱלֹהִ֔ים בֹּאָה־ אִתִּ֥י הַבַּ֖יְתָה וּֽסְעָ֑דָה וְאֶתְּנָ֥ה לְךָ֖ מַתָּֽת׃ (יח) וַיֹּ֤אמֶר אִישׁ־ הָֽאֱלֹהִים֙ אֶל־ הַמֶּ֔לֶךְ אִם־ תִּתֶּן־ לִי֙ אֶת־ חֲצִ֣י בֵיתֶ֔ךָ לֹ֥א אָבֹ֖א עִמָּ֑ךְ וְלֹֽא־ אֹ֤כַל לֶ֙חֶם֙ וְלֹ֣א אֶשְׁתֶּה־ מַּ֔יִם בַּמָּק֖וֹם הַזֶּֽה׃ (יט) כִּֽי־ כֵ֣ן ׀ צִוָּ֣ה אֹתִ֗י בִּדְבַ֤ר יְהוָה֙ לֵאמֹ֔ר לֹא־ תֹ֥אכַל לֶ֖חֶם וְלֹ֣א תִשְׁתֶּה־ מָּ֑יִם וְלֹ֣א תָשׁ֔וּב בַּדֶּ֖רֶךְ אֲשֶׁ֥ר הָלָֽכְתָּ׃ (כ) וַיֵּ֖לֶךְ בְּדֶ֣רֶךְ אַחֵ֑ר וְלֹֽא־ שָׁ֣ב בַּדֶּ֔רֶךְ אֲשֶׁ֛ר בָּ֥א בָ֖הּ אֶל־ בֵּֽית־ אֵֽל׃ (כא) וְנָבִ֤יא אֶחָד֙ זָקֵ֔ן יֹשֵׁ֖ב בְּבֵֽית־ אֵ֑ל וַיָּב֣וֹא בְנ֡וֹ וַיְסַפֶּר־ ל֣וֹ אֶת־ כָּל־ הַמַּעֲשֶׂ֣ה אֲשֶׁר־ עָשָׂה֩ אִישׁ־ הָאֱלֹהִ֨ים ׀ הַיּ֜וֹם בְּבֵֽית־ אֵ֗ל אֶת־ הַדְּבָרִים֙ אֲשֶׁ֣ר דִּבֶּ֣ר אֶל־ הַמֶּ֔לֶךְ וַֽיְסַפְּר֖וּם לַאֲבִיהֶֽם׃ (כב) וַיְדַבֵּ֤ר אֲלֵהֶם֙ אֲבִיהֶ֔ם אֵֽי־ זֶ֥ה הַדֶּ֖רֶךְ הָלָ֑ךְ וַיִּרְא֣וּ בָנָ֗יו אֶת־ הַדֶּ֙רֶךְ֙ אֲשֶׁ֤ר הָלַךְ֙ אִ֣ישׁ הָאֱלֹהִ֔ים אֲשֶׁר־ בָּ֖א מִיהוּדָֽה׃ (כג) וַיֹּ֙אמֶר֙ אֶל־ בָּנָ֔יו חִבְשׁוּ־ לִ֖י הַחֲמ֑וֹר וַיַּחְבְּשׁוּ־ ל֣וֹ הַחֲמ֔וֹר וַיִּרְכַּ֖ב עָלָֽיו׃ (כד) וַיֵּ֗לֶךְ אַֽחֲרֵי֙ אִ֣ישׁ הָאֱלֹהִ֔ים וַיִּ֨מְצָאֵ֔הוּ יֹשֵׁ֖ב תַּ֣חַת הָאֵלָ֑ה וַיֹּ֣אמֶר אֵלָ֗יו הַאַתָּ֧ה אִישׁ־ הָאֱלֹהִ֛ים אֲשֶׁר־ בָּ֥אתָ מִֽיהוּדָ֖ה וַיֹּ֥אמֶר אָֽנִי׃ (כה) וַיֹּ֣אמֶר אֵלָ֔יו לֵ֥ךְ אִתִּ֖י הַבָּ֑יְתָה וֶאֱכֹ֖ל לָֽחֶם׃ (כו) וַיֹּ֗אמֶר לֹ֥א אוּכַ֛ל לָשׁ֥וּב אִתָּ֖ךְ וְלָב֣וֹא אִתָּ֑ךְ וְלֹֽא־ אֹ֣כַל לֶ֗חֶם וְלֹֽא־ אֶשְׁתֶּ֤ה אִתְּךָ֙ מַ֔יִם בַּמָּק֖וֹם הַזֶּֽה׃ (כז) כִּֽי־ דָבָ֤ר אֵלַי֙ בִּדְבַ֣ר יְהוָ֔ה לֹֽא־ תֹאכַ֣ל לֶ֔חֶם וְלֹֽא־ תִשְׁתֶּ֥ה שָׁ֖ם מָ֑יִם לֹא־ תָשׁ֣וּב לָלֶ֔כֶת בַּדֶּ֖רֶךְ אֲשֶׁר־ הָלַ֥כְתָּ בָּֽהּ׃ (כח) וַיֹּ֣אמֶר ל֗וֹ גַּם־ אֲנִ֣י נָבִיא֮ כָּמוֹךָ֒ וּמַלְאָ֡ךְ דִּבֶּ֣ר אֵלַי֩ בִּדְבַ֨ר יְהוָ֜ה לֵאמֹ֗ר הֲשִׁבֵ֤הוּ אִתְּךָ֙ אֶל־ בֵּיתֶ֔ךָ וְיֹ֥אכַל לֶ֖חֶם וְיֵ֣שְׁתְּ מָ֑יִם כִּחֵ֖שׁ לֽוֹ׃ (כט) וַיָּ֣שָׁב אִתּ֗וֹ וַיֹּ֥אכַל לֶ֛חֶם בְּבֵית֖וֹ וַיֵּ֥שְׁתְּ מָֽיִם׃ (ל) וַיְהִ֕י הֵ֥ם יֹשְׁבִ֖ים אֶל־ הַשֻּׁלְחָ֑ן וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ הַנָּבִ֖יא אֲשֶׁ֥ר הֱשִׁיבֽוֹ׃ (לא) וַיִּקְרָ֞א אֶל־ אִ֣ישׁ הָאֱלֹהִ֗ים אֲשֶׁר־ בָּ֤א מִֽיהוּדָה֙ לֵאמֹ֔ר כֹּ֖ה אָמַ֣ר יְהוָ֑ה יַ֗עַן כִּ֤י מָרִ֙יתָ֙ פִּ֣י יְהוָ֔ה וְלֹ֤א שָׁמַ֙רְתָּ֙ אֶת־ הַמִּצְוָ֔ה אֲשֶׁ֥ר צִוְּךָ֖ יְהוָ֥ה אֱלֹהֶֽיךָ׃ (לב) וַתָּ֗שָׁב וַתֹּ֤אכַל לֶ֙חֶם֙ וַתֵּ֣שְׁתְּ מַ֔יִם בַּמָּקוֹם֙ אֲשֶׁ֣ר דִּבֶּ֣ר אֵלֶ֔יךָ אַל־ תֹּ֥אכַל לֶ֖חֶם וְאַל־ תֵּ֣שְׁתְּ מָ֑יִם לֹֽא־ תָב֥וֹא נִבְלָתְךָ֖ אֶל־ קֶ֥בֶר אֲבֹתֶֽיךָ׃ (לג) וַיְהִ֗י אַחֲרֵ֛י אָכְל֥וֹ לֶ֖חֶם וְאַחֲרֵ֣י שְׁתוֹת֑וֹ וַיַּחֲבָשׁ־ ל֣וֹ הַחֲמ֔וֹר לַנָּבִ֖יא אֲשֶׁ֥ר הֱשִׁיבֽוֹ׃ (לד) וַיֵּ֕לֶךְ וַיִּמְצָאֵ֧הוּ אֶת־ הַנְּבֵלָה֙ מֻשְׁלֶ֣כֶת בַּדֶּ֔רֶךְ וְאֶת־ הָ֣אַרְיֵ֔ה עֹמֵ֖ד אֵ֣צֶל הַנְּבֵלָ֑ה וַיָּבֹ֙אוּ֙ וַיְדַבְּר֣וּ בָעִ֔יר אֲשֶׁ֛ר הַנָּבִ֥יא הַזָּקֵ֖ן יֹשֵׁ֥ב בָּֽהּ׃ (לו) וַיִּשְׁמַ֣ע הַנָּבִיא֮ אֲשֶׁ֣ר הֱשִׁיב֣וֹ מִן־ הַדֶּרֶךְ֒ וַיֹּ֙אמֶר֙ אִ֣ישׁ הָאֱלֹהִ֣ים ה֔וּא אֲשֶׁ֥ר מָרָ֖ה אֶת־ פִּ֣י יְהוָ֑ה וַיִּתְּנֵ֨הוּ יְהוָ֜ה לָאַרְיֵ֗ה וַֽיִּשְׁבְּרֵ֙הוּ֙ וַיְמִתֵ֔הוּ כִּדְבַ֥ר יְהוָ֖ה אֲשֶׁ֥ר דִּבֶּר־ לֽוֹ׃ (לז) וַיְדַבֵּ֤ר אֶל־ בָּנָיו֙ לֵאמֹ֔ר חִבְשׁוּ־ לִ֖י אֶֽת־ הַחֲמ֑וֹר וַֽיַּחֲבֹֽשׁוּ׃ (לח) וַיֵּ֗לֶךְ וַיִּמְצָ֤א אֶת־ נִבְלָתוֹ֙ מֻשְׁלֶ֣כֶת בַּדֶּ֔רֶךְ וַֽחֲמוֹר֙ וְהָ֣אַרְיֵ֔ה עֹמְדִ֖ים אֵ֣צֶל הַנְּבֵלָ֑ה לֹֽא־ אָכַ֤ל הָֽאַרְיֵה֙ אֶת־ הַנְּבֵלָ֔ה וְלֹ֥א שָׁבַ֖ר אֶֽת־ הַחֲמֽוֹר׃ (לט) וַיִּשָּׂ֨א הַנָּבִ֜יא אֶת־ נִבְלַ֧ת אִישׁ־ הָאֱלֹהִ֛ים וַיַּנִּחֵ֥הוּ אֶֽל־ הַחֲמ֖וֹר וַיְשִׁיבֵ֑הוּ וַיָּבֹ֗א אֶל־ עִיר֙ הַנָּבִ֣יא הַזָּקֵ֔ן לִסְפֹּ֖ד וּלְקָבְרֽוֹ׃ (מ) וַיַּנַּ֥ח אֶת־ נִבְלָת֖וֹ בְּקִבְר֑וֹ וַיִּסְפְּד֥וּ עָלָ֖יו ה֥וֹי אָחִֽי׃

(א) וַיְהִי֮ אַחֲרֵ֣י קָבְר֣וֹ אֹתוֹ֒ וַיֹּ֤אמֶר אֶל־ בָּנָיו֙ לֵאמֹ֔ר בְּמוֹתִי֙ וּקְבַרְתֶּ֣ם אֹתִ֔י בַּקֶּ֕בֶר אֲשֶׁ֛ר אִ֥ישׁ הָאֱלֹהִ֖ים קָב֣וּר בּ֑וֹ אֵ֚צֶל עַצְמֹתָ֔יו הַנִּ֖יחוּ אֶת־ עַצְמֹתָֽי׃ (ב) כִּי֩ הָיֹ֨ה יִהְיֶ֜ה הַדָּבָ֗ר אֲשֶׁ֤ר קָרָא֙ בִּדְבַ֣ר יְהוָ֔ה עַל־ הַמִּזְבֵּ֖חַ אֲשֶׁ֣ר בְּבֵֽית־ אֵ֑ל וְעַל֙ כָּל־ בָּתֵּ֣י הַבָּמ֔וֹת אֲשֶׁ֖ר בְּעָרֵ֥י שֹׁמְרֽוֹן׃ (ג) אַחַר֙ בַּדָּבָ֣ר הַזֶּ֔ה לְחַטַּ֖את בֵּ֣ית יָרָבְעָ֑ם וּלְהַכְחִיד֙ וּלְהַשְׁמִ֔יד מֵעַ֖ל פְּנֵ֥י הָאֲדָמָֽה׃ סוף פרק יג (ה) בָּעֵ֣ת הַהִ֔יא חָלָ֖ה אֲבִיָּ֥ה בֶן־ יָרָבְעָֽם׃ (ו) וַיֹּ֨אמֶר יָרָבְעָ֜ם הַזֶּֽה׃ (ז) וְלָקַ֣חַתְּ בְּ֠יָדֵךְ עֲשָׂרָ֨ה לֶ֧חֶם וְנִקֻּדִ֛ים וּבַקְבֻּ֥ק דְּבַ֖שׁ וּבָ֣את אֵלָ֑יו ה֚וּא יַגִּ֣יד לָ֔ךְ מַה־ יִּֽהְיֶ֖ה לַנָּֽעַר׃ (ח) וַתַּ֤עַשׂ כֵּן֙ אֵ֣שֶׁת יָרָבְעָ֔ם יָֽרָבְעָ֔ם אֲשֶׁ֥ר נִלְחַ֖ם וַאֲשֶׁ֣ר מָלָ֑ךְ הִנָּ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כד) וְהַיָּמִים֙ אֲשֶׁ֣ר מָלַ֣ךְ יָרָבְעָ֔ם עֶשְׂרִ֥ים וּשְׁתַּ֖יִם שָׁנָ֑ה וַיִּשְׁכַּב֙ עִם־ אֲבֹתָ֔יו וַיִּמְלֹ֛ךְ נָדָ֥ב בְּנ֖וֹ תַּחְתָּֽיו׃ (כה) וּרְחַבְעָם֙ בֶּן־ שְׁלֹמֹ֔ה מָלַ֖ךְ בִּֽיהוּדָ֑ה בֶּן־ אַרְבָּעִ֣ים וְאַחַ֣ת שָׁנָה֩ רְחַבְעָ֨ם בְּמָלְכ֜וֹ וּֽשֲׁבַ֨ע עֶשְׂרֵ֥ה שָׁנָ֣ה ׀ מָלַ֣ךְ בִּירוּשָׁלִַ֗ם הָ֠עִיר אֲשֶׁר־ בָּחַ֨ר יְהוָ֜ה לָשׂ֨וּם אֶת־ שְׁמ֥וֹ שָׁם֙ מִכֹּל֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל וְשֵׁ֣ם אִמּ֔וֹ נַעֲמָ֖ה הָעַמֹּנִֽית׃ (כו) וַיַּ֧עַשׂ יְהוּדָ֛ה הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיְקַנְא֣וּ אֹת֗וֹ מִכֹּל֙ אֲשֶׁ֣ר עָשׂ֣וּ אֲבֹתָ֔ם בְּחַטֹּאתָ֖ם אֲשֶׁ֥ר חָטָֽאוּ׃ (כז) וַיִּבְנ֨וּ גַם־ הֵ֧מָּה לָהֶ֛ם בָּמ֥וֹת וּמַצֵּב֖וֹת וַאֲשֵׁרִ֑ים עַ֚ל כָּל־ גִּבְעָ֣ה גְבֹהָ֔ה וְתַ֖חַת כָּל־ עֵ֥ץ רַעֲנָֽן׃ (כח) וְגַם־ קָדֵ֖שׁ הָיָ֣ה בָאָ֑רֶץ עָשׂ֗וּ כְּכֹל֙ הַתּוֹעֲבֹ֣ת הַגּוֹיִ֔ם אֲשֶׁר֙ הוֹרִ֣ישׁ יְהוָ֔ה מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (כט) וַיְהִ֛י בַּשָּׁנָ֥ה הַחֲמִישִׁ֖ית לַמֶּ֣לֶךְ רְחַבְעָ֑ם עָלָ֛ה (שושק) שִׁישַׁ֥ק מֶֽלֶךְ־ מִצְרַ֖יִם עַל־ יְרוּשָׁלִָֽם׃ (ל) וַיִּקַּ֞ח אֶת־ אֹצְר֣וֹת בֵּית־ יְהוָ֗ה וְאֶת־ אֽוֹצְרוֹת֙ בֵּ֣ית הַמֶּ֔לֶךְ וְאֶת־ הַכֹּ֖ל לָקָ֑ח וַיִּקַּח֙ אֶת־ כָּל־ מָגִנֵּ֣י הַזָּהָ֔ב אֲשֶׁ֥ר עָשָׂ֖ה שְׁלֹמֹֽה׃ (לא) וַיַּ֨עַשׂ הַמֶּ֤לֶךְ רְחַבְעָם֙ תַּחְתָּ֔ם מָגִנֵּ֖י נְחֹ֑שֶׁת וְהִפְקִ֗יד עַל־ יַד֙ שָׂרֵ֣י הָרָצִ֔ים הַשֹּׁ֣מְרִ֔ים פֶּ֖תַח בֵּ֥ית הַמֶּֽלֶךְ׃ (לב) וַיְהִ֛י מִדֵּי־ בֹ֥א הַמֶּ֖לֶךְ בֵּ֣ית יְהוָ֑ה יִשָּׂאוּם֙ הָֽרָצִ֔ים וֶהֱשִׁיב֖וּם אֶל־ תָּ֥א הָרָצִֽים׃ (לג) וְיֶ֛תֶר דִּבְרֵ֥י רְחַבְעָ֖ם וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣מָּה כְתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (לד) וּמִלְחָמָ֨ה הָיְתָ֧ה בֵין־ רְחַבְעָ֛ם וּבֵ֥ין יָרָבְעָ֖ם כָּל־ הַיָּמִֽים׃ (לה) וַיִּשְׁכַּ֨ב רְחַבְעָ֜ם עִם־ אֲבֹתָ֗יו וַיִּקָּבֵ֤ר עִם־ אֲבֹתָיו֙ בְּעִ֣יר דָּוִ֔ד וְשֵׁ֣ם אִמּ֔וֹ נַעֲמָ֖ה הָעַמֹּנִ֑ית וַיִּמְלֹ֛ךְ אֲבִיָּ֥ם בְּנ֖וֹ תַּחְתָּֽיו׃ (לו) וּבִשְׁנַת֙ שְׁמֹנֶ֣ה עֶשְׂרֵ֔ה לַמֶּ֖לֶךְ יָרָבְעָ֣ם בֶּן־ נְבָ֑ט מָלַ֥ךְ אֲבִיָּ֖ם עַל־ יְהוּדָֽה׃ (לז) שָׁלֹ֣שׁ שָׁנִ֔ים מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ מַעֲכָ֖ה בַּת־ אֲבִישָׁלֽוֹם׃ (לח) וַיֵּ֕לֶךְ בְּכָל־ חַטֹּ֥אות אָבִ֖יו אֲשֶׁר־ עָשָׂ֣ה לְפָנָ֑יו וְלֹא־ הָיָ֨ה לְבָב֤וֹ שָׁלֵם֙ עִם־ יְהוָ֣ה אֱלֹהָ֔יו כִּלְבַ֖ב דָּוִ֥ד אָבִֽיו׃ (לט) כִּ֚י לְמַ֣עַן דָּוִ֔ד נָתַן֩ יְהוָ֨ה אֱלֹהָ֥יו ל֛וֹ נִ֖יר בִּירוּשָׁלִָ֑ם לְהָקִ֤ים אֶת־ בְּנוֹ֙ אַחֲרָ֔יו וּֽלְהַעֲמִ֖יד אֶת־ יְרוּשָׁלִָֽם׃ (מ) אֲשֶׁ֨ר עָשָׂ֥ה דָוִ֛ד אֶת־ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה וְלֹֽא־ סָ֞ר מִכֹּ֣ל אֲשֶׁר־ צִוָּ֗הוּ כֹּ֚ל יְמֵ֣י חַיָּ֔יו רַ֕ק בִּדְבַ֖ר אוּרִיָּ֥ה הַחִתִּֽי׃ (מא) וּמִלְחָמָ֨ה הָיְתָ֧ה בֵין־ רְחַבְעָ֛ם וּבֵ֥ין יָרָבְעָ֖ם כָּל־ יְמֵ֥י חַיָּֽיו׃ (מב) וְיֶ֨תֶר דִּבְרֵ֤י אֲבִיָּם֙ וְכָל־ אֲשֶׁ֣ר עָשָׂ֔ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֣י יְהוּדָ֑ה וּמִלְחָמָ֥ה הָיְתָ֛ה בֵּ֥ין אֲבִיָּ֖ם וּבֵ֥ין יָרָבְעָֽם׃
