מלכים א פרק טו לפי החלוקה לסדרים

פרק טו בספר מלכים א (34 פסוקים) נמצא במלכים סדר יב (פסוקים לו–מב) ובמלכים סדר יג (פסוקים א–כז).

מלכים סדר יב

(א) וַיְהִי֮ אַחֲרֵ֣י קָבְר֣וֹ אֹתוֹ֒ וַיֹּ֤אמֶר אֶל־ בָּנָיו֙ לֵאמֹ֔ר בְּמוֹתִי֙ וּקְבַרְתֶּ֣ם אֹתִ֔י בַּקֶּ֕בֶר אֲשֶׁ֛ר אִ֥ישׁ הָאֱלֹהִ֖ים קָב֣וּר בּ֑וֹ אֵ֚צֶל עַצְמֹתָ֔יו הַנִּ֖יחוּ אֶת־ עַצְמֹתָֽי׃ (ב) כִּי֩ הָיֹ֨ה יִהְיֶ֜ה הַדָּבָ֗ר אֲשֶׁ֤ר קָרָא֙ בִּדְבַ֣ר יְהוָ֔ה עַל־ הַמִּזְבֵּ֖חַ אֲשֶׁ֣ר בְּבֵֽית־ אֵ֑ל וְעַל֙ כָּל־ בָּתֵּ֣י הַבָּמ֔וֹת אֲשֶׁ֖ר בְּעָרֵ֥י שֹׁמְרֽוֹן׃ (ג) אַחַר֙ הַדָּבָ֣ר הַזֶּ֔ה לֹֽא־ שָׁ֥ב יָרָבְעָ֖ם מִדַּרְכּ֣וֹ הָרָעָ֑ה וַ֠יָּשָׁב וַיַּ֜עַשׂ מִקְצ֤וֹת הָעָם֙ כֹּהֲנֵ֣י בָמ֔וֹת הֶֽחָפֵץ֙ יְמַלֵּ֣א אֶת־ יָד֔וֹ וִיהִ֖י כֹּהֲנֵ֥י בָמֽוֹת׃ (ד) וַיְהִי֙ בַּדָּבָ֣ר הַזֶּ֔ה לְחַטַּ֖את בֵּ֣ית יָרָבְעָ֑ם וּלְהַכְחִיד֙ וּלְהַשְׁמִ֔יד מֵעַ֖ל פְּנֵ֥י הָאֲדָמָֽה׃ (ה) בָּעֵ֣ת הַהִ֔יא חָלָ֖ה אֲבִיָּ֥ה בֶן־ יָרָבְעָֽם׃ (ו) וַיֹּ֨אמֶר יָרָבְעָ֜ם לְאִשְׁתּ֗וֹ ק֤וּמִי נָא֙ וְהִשְׁתַּנִּ֔ית וְלֹ֣א יֵֽדְע֔וּ כִּי־ (אתי) אַ֖תְּ אֵ֣שֶׁת יָרָבְעָ֑ם וְהָלַ֣כְתְּ שִׁלֹ֗ה הִנֵּה־ שָׁם֙ אֲחִיָּ֣ה הַנָּבִ֔יא הֽוּא־ דִבֶּ֥ר עָלַ֛י לְמֶ֖לֶךְ עַל־ הָעָ֥ם הַזֶּֽה׃ (ז) וְלָקַ֣חַתְּ בְּ֠יָדֵךְ עֲשָׂרָ֨ה לֶ֧חֶם וְנִקֻּדִ֛ים וּבַקְבֻּ֥ק דְּבַ֖שׁ וּבָ֣את אֵלָ֑יו ה֚וּא יַגִּ֣יד לָ֔ךְ מַה־ יִּֽהְיֶ֖ה לַנָּֽעַר׃ (ח) וַתַּ֤עַשׂ כֵּן֙ אֵ֣שֶׁת יָרָבְעָ֔ם וַתָּ֙קָם֙ וַתֵּ֣לֶךְ שִׁלֹ֔ה וַתָּבֹ֖א בֵּ֣ית אֲחִיָּ֑ה וַאֲחִיָּ֙הוּ֙ לֹֽא־ יָכֹ֣ל לִרְא֔וֹת כִּ֛י קָ֥מוּ עֵינָ֖יו מִשֵּׂיבֽוֹ׃ (ט) וַיהוָ֞ה אָמַ֣ר אֶל־ אֲחִיָּ֗הוּ הִנֵּ֣ה אֵ֣שֶׁת יָרָבְעָ֡ם בָּאָ֣ה לִדְרֹשׁ֩ דָּבָ֨ר מֵעִמְּךָ֤ אֶל־ בְּנָהּ֙ כִּֽי־ חֹלֶ֣ה ה֔וּא כָּזֹ֥ה וְכָזֶ֖ה תְּדַבֵּ֣ר אֵלֶ֑יהָ וִיהִ֣י כְבֹאָ֔הּ וְהִ֖יא מִתְנַכֵּרָֽה׃ (י) וַיְהִי֩ כִשְׁמֹ֨עַ אֲחִיָּ֜הוּ אֶת־ ק֤וֹל רַגְלֶ֙יהָ֙ בָּאָ֣ה בַפֶּ֔תַח וַיֹּ֕אמֶר בֹּ֖אִי אֵ֣שֶׁת יָרָבְעָ֑ם לָ֣מָּה זֶּ֗ה אַ֚תְּ מִתְנַכֵּרָ֔ה וְאָ֣נֹכִ֔י שָׁל֥וּחַ אֵלַ֖יִךְ קָשָֽׁה׃ (יא) לְכִ֞י אִמְרִ֣י לְיָרָבְעָ֗ם כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל יַ֛עַן אֲשֶׁ֥ר הֲרִימֹתִ֖יךָ מִתּ֣וֹךְ הָעָ֑ם וָאֶתֶּנְךָ֣ נָגִ֔יד עַ֖ל עַמִּ֥י יִשְׂרָאֵֽל׃ (יב) וָאֶקְרַ֤ע אֶת־ הַמַּמְלָכָה֙ מִבֵּ֣ית דָּוִ֔ד וָאֶתְּנֶ֖הָ לָ֑ךְ וְלֹֽא־ הָיִ֜יתָ כְּעַבְדִּ֣י דָוִ֗ד אֲשֶׁר֩ שָׁמַ֨ר מִצְוֺתַ֜י וַאֲשֶׁר־ הָלַ֤ךְ אַחֲרַי֙ בְּכָל־ לְבָב֔וֹ לַעֲשׂ֕וֹת רַ֖ק הַיָּשָׁ֥ר בְּעֵינָֽי׃ (יג) וַתָּ֣רַע לַעֲשׂ֔וֹת מִכֹּ֖ל אֲשֶׁר־ הָי֣וּ לְפָנֶ֑יךָ וַתֵּ֡לֶךְ וַתַּעֲשֶׂה־ לְּךָ֩ אֱלֹהִ֨ים אֲחֵרִ֤ים וּמַסֵּכוֹת֙ לְהַכְעִיסֵ֔נִי וְאֹתִ֥י הִשְׁלַ֖כְתָּ אַחֲרֵ֥י גַוֶּֽךָ׃ (יד) לָכֵ֗ן הִנְנִ֨י מֵבִ֤יא רָעָה֙ אֶל־ בֵּ֣ית יָרָבְעָ֔ם וְהִכְרַתִּ֤י לְיָֽרָבְעָם֙ מַשְׁתִּ֣ין בְּקִ֔יר עָצ֥וּר וְעָז֖וּב בְּיִשְׂרָאֵ֑ל וּבִֽעַרְתִּי֙ אַחֲרֵ֣י בֵית־ יָרָבְעָ֔ם כַּאֲשֶׁ֛ר יְבַעֵ֥ר הַגָּלָ֖ל עַד־ תֻּמּֽוֹ׃ (טו) הַמֵּ֨ת לְיָֽרָבְעָ֤ם בָּעִיר֙ יֹאכְל֣וּ הַכְּלָבִ֔ים וְהַמֵּת֙ בַּשָּׂדֶ֔ה יֹאכְל֖וּ ע֣וֹף הַשָּׁמָ֑יִם כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (טז) וְאַ֥תְּ ק֖וּמִי לְכִ֣י לְבֵיתֵ֑ךְ בְּבֹאָ֥ה רַגְלַ֛יִךְ הָעִ֖ירָה וּמֵ֥ת הַיָּֽלֶד׃ (יז) וְסָֽפְדוּ־ ל֤וֹ כָל־ יִשְׂרָאֵל֙ וְקָבְר֣וּ אֹת֔וֹ כִּֽי־ זֶ֣ה לְבַדּ֔וֹ יָבֹ֥א לְיָרָבְעָ֖ם אֶל־ קָ֑בֶר יַ֣עַן נִמְצָא־ ב֞וֹ דָּבָ֣ר ט֗וֹב אֶל־ יְהוָ֛ה אֱלֹהֵ֥י יִשְׂרָאֵ֖ל בְּבֵ֥ית יָרָבְעָֽם׃ (יח) וְהֵקִים֩ יְהוָ֨ה ל֥וֹ מֶ֙לֶךְ֙ עַל־ יִשְׂרָאֵ֔ל אֲשֶׁ֥ר יַכְרִ֛ית אֶת־ בֵּ֥ית יָרָבְעָ֖ם זֶ֣ה הַיּ֑וֹם וּמֶ֖ה גַּם־ עָֽתָּה׃ (יט) וְהִכָּ֨ה יְהוָ֜ה אֶת־ יִשְׂרָאֵ֗ל כַּאֲשֶׁ֨ר יָנ֣וּד הַקָּנֶה֮ בַּמַּיִם֒ וְנָתַ֣שׁ אֶת־ יִשְׂרָאֵ֗ל מֵ֠עַל הָאֲדָמָ֨ה הַטּוֹבָ֤ה הַזֹּאת֙ אֲשֶׁ֤ר נָתַן֙ לַאֲב֣וֹתֵיהֶ֔ם וְזֵרָ֖ם מֵעֵ֣בֶר לַנָּהָ֑ר יַ֗עַן אֲשֶׁ֤ר עָשׂוּ֙ אֶת־ אֲשֵׁ֣רֵיהֶ֔ם מַכְעִיסִ֖ים אֶת־ יְהוָֽה׃ (כ) וְיִתֵּ֖ן אֶת־ יִשְׂרָאֵ֑ל בִּגְלַ֞ל חַטֹּ֤אות יָֽרָבְעָם֙ אֲשֶׁ֣ר חָטָ֔א וַאֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (כא) וַתָּ֙קָם֙ אֵ֣שֶׁת יָרָבְעָ֔ם וַתֵּ֖לֶךְ וַתָּבֹ֣א תִרְצָ֑תָה הִ֛יא בָּאָ֥ה בְסַף־ הַבַּ֖יִת וְהַנַּ֥עַר מֵֽת׃ (כב) וַיִּקְבְּר֥וּ אֹת֛וֹ וַיִּסְפְּדוּ־ ל֖וֹ כָּל־ יִשְׂרָאֵ֑ל כִּדְבַ֤ר יְהוָה֙ אֲשֶׁ֣ר דִּבֶּ֔ר בְּיַד־ עַבְדּ֖וֹ אֲחִיָּ֥הוּ הַנָּבִֽיא׃ (כג) וְיֶ֙תֶר֙ דִּבְרֵ֣י יָֽרָבְעָ֔ם אֲשֶׁ֥ר נִלְחַ֖ם וַאֲשֶׁ֣ר מָלָ֑ךְ הִנָּ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כד) וְהַיָּמִים֙ אֲשֶׁ֣ר מָלַ֣ךְ יָרָבְעָ֔ם עֶשְׂרִ֥ים וּשְׁתַּ֖יִם שָׁנָ֑ה וַיִּשְׁכַּב֙ עִם־ אֲבֹתָ֔יו וַיִּמְלֹ֛ךְ נָדָ֥ב בְּנ֖וֹ תַּחְתָּֽיו׃ (כה) וּרְחַבְעָם֙ בֶּן־ שְׁלֹמֹ֔ה מָלַ֖ךְ בִּֽיהוּדָ֑ה בֶּן־ אַרְבָּעִ֣ים וְאַחַ֣ת שָׁנָה֩ רְחַבְעָ֨ם בְּמָלְכ֜וֹ וּֽשֲׁבַ֨ע עֶשְׂרֵ֥ה שָׁנָ֣ה ׀ מָלַ֣ךְ בִּירוּשָׁלִַ֗ם הָ֠עִיר אֲשֶׁר־ בָּחַ֨ר יְהוָ֜ה לָשׂ֨וּם אֶת־ שְׁמ֥וֹ שָׁם֙ מִכֹּל֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל וְשֵׁ֣ם אִמּ֔וֹ נַעֲמָ֖ה הָעַמֹּנִֽית׃ (כו) וַיַּ֧עַשׂ יְהוּדָ֛ה הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיְקַנְא֣וּ אֹת֗וֹ מִכֹּל֙ אֲשֶׁ֣ר עָשׂ֣וּ אֲבֹתָ֔ם בְּחַטֹּאתָ֖ם אֲשֶׁ֥ר חָטָֽאוּ׃ (כז) וַיִּבְנ֨וּ גַם־ הֵ֧מָּה לָהֶ֛ם בָּמ֥וֹת וּמַצֵּב֖וֹת וַאֲשֵׁרִ֑ים עַ֚ל כָּל־ גִּבְעָ֣ה גְבֹהָ֔ה וְתַ֖חַת כָּל־ עֵ֥ץ רַעֲנָֽן׃ (כח) וְגַם־ קָדֵ֖שׁ הָיָ֣ה בָאָ֑רֶץ עָשׂ֗וּ כְּכֹל֙ הַתּוֹעֲבֹ֣ת הַגּוֹיִ֔ם אֲשֶׁר֙ הוֹרִ֣ישׁ יְהוָ֔ה מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (כט) וַיְהִ֛י בַּשָּׁנָ֥ה הַחֲמִישִׁ֖ית לַמֶּ֣לֶךְ רְחַבְעָ֑ם עָלָ֛ה (שושק) שִׁישַׁ֥ק מֶֽלֶךְ־ מִצְרַ֖יִם עַל־ יְרוּשָׁלִָֽם׃ (ל) וַיִּקַּ֞ח אֶת־ אֹצְר֣וֹת בֵּית־ יְהוָ֗ה וְאֶת־ אֽוֹצְרוֹת֙ בֵּ֣ית הַמֶּ֔לֶךְ וְאֶת־ הַכֹּ֖ל לָקָ֑ח וַיִּקַּח֙ אֶת־ כָּל־ מָגִנֵּ֣י הַזָּהָ֔ב אֲשֶׁ֥ר עָשָׂ֖ה שְׁלֹמֹֽה׃ (לא) וַיַּ֨עַשׂ הַמֶּ֤לֶךְ רְחַבְעָם֙ תַּחְתָּ֔ם מָגִנֵּ֖י נְחֹ֑שֶׁת וְהִפְקִ֗יד עַל־ יַד֙ שָׂרֵ֣י הָרָצִ֔ים הַשֹּׁ֣מְרִ֔ים פֶּ֖תַח בֵּ֥ית הַמֶּֽלֶךְ׃ (לב) וַיְהִ֛י מִדֵּי־ בֹ֥א הַמֶּ֖לֶךְ בֵּ֣ית יְהוָ֑ה יִשָּׂאוּם֙ הָֽרָצִ֔ים וֶהֱשִׁיב֖וּם אֶל־ תָּ֥א הָרָצִֽים׃ (לג) וְיֶ֛תֶר דִּבְרֵ֥י רְחַבְעָ֖ם וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣מָּה כְתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (לד) וּמִלְחָמָ֨ה הָיְתָ֧ה בֵין־ רְחַבְעָ֛ם וּבֵ֥ין יָרָבְעָ֖ם כָּל־ הַיָּמִֽים׃ (לה) וַיִּשְׁכַּ֨ב רְחַבְעָ֜ם עִם־ אֲבֹתָ֗יו הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיֵּ֙לֶךְ֙ בְּדֶ֣רֶךְ יָרָבְעָ֔ם וּ֨בְחַטָּאת֔וֹ אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ סוף פרק טו (כח) וַיְהִ֤י דְבַר־ יְהוָה֙ אֶל־ יֵה֣וּא בֶן־ חֲנָ֔נִי עַל־ בַּעְשָׁ֖א לֵאמֹֽר׃ (כט) יַ֗עַן אֲשֶׁ֤ר הֲרִימֹתִ֙יךָ֙ מִן־ הֶ֣עָפָ֔ר וָאֶתֶּנְךָ֣ נָגִ֔יד עַ֖ל עַמִּ֣י יִשְׂרָאֵ֑ל וַתֵּ֣לֶךְ ׀ בְּדֶ֣רֶךְ יָרָבְעָ֗ם וַֽתַּחֲטִא֙ אֶת־ עַמִּ֣י יִשְׂרָאֵ֔ל לְהַכְעִיסֵ֖נִי בְּחַטֹּאתָֽם׃ (ל) הִנְנִ֥י מַבְעִ֛יר אַחֲרֵ֥י בַעְשָׁ֖א וְאַחֲרֵ֣י בֵית֑וֹ וְנָֽתַתִּי֙ אֶת־ בֵּ֣יתְךָ֔ כְּבֵ֖ית יָרָבְעָ֥ם בֶּן־ נְבָֽט׃ (לא) הַמֵּ֤ת לְבַעְשָׁא֙ בָּעִ֔יר יֹֽאכְל֖וּ הַכְּלָבִ֑ים וְהַמֵּ֥ת לוֹ֙ בַּשָּׂדֶ֔ה יֹאכְל֖וּ ע֥וֹף הַשָּׁמָֽיִם׃ (לב) וְיֶ֨תֶר דִּבְרֵ֥י בַעְשָׁ֛א וַאֲשֶׁ֥ר עָשָׂ֖ה וּגְבֽוּרָת֑וֹ הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (לג) וַיִּשְׁכַּ֤ב בַּעְשָׁא֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֖ר בְּתִרְצָ֑ה וַיִּמְלֹ֛ךְ אֵלָ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ (לד) וְגַ֡ם בְּיַד־ יֵה֨וּא בֶן־ חֲנָ֜נִי הַנָּבִ֗יא דְּבַר־ יְהוָ֡ה הָיָה֩ אֶל־ בַּעְשָׁ֨א וְאֶל־ בֵּית֜וֹ וְעַ֥ל כָּל־ הָרָעָ֣ה ׀ אֲשֶׁר־ עָשָׂ֣ה ׀ בְּעֵינֵ֣י יְהוָ֗ה לְהַכְעִיסוֹ֙ בְּמַעֲשֵׂ֣ה יָדָ֔יו לִהְי֖וֹת כְּבֵ֣ית יָרָבְעָ֑ם וְעַ֥ל אֲשֶׁר־ הִכָּ֖ה אֹתֽוֹ׃ (לה) בִּשְׁנַ֨ת עֶשְׂרִ֤ים וָשֵׁשׁ֙ שָׁנָ֔ה לְאָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ אֵלָ֨ה בֶן־ בַּעְשָׁ֧א עַל־ יִשְׂרָאֵ֛ל בְּתִרְצָ֖ה שְׁנָתָֽיִם׃ (לו) וַיִּקְשֹׁ֤ר עָלָיו֙ עַבְדּ֣וֹ זִמְרִ֔י שַׂ֖ר מַחֲצִ֣ית הָרָ֑כֶב וְה֤וּא בְתִרְצָה֙ שֹׁתֶ֣ה שִׁכּ֔וֹר בֵּ֣ית אַרְצָ֔א אֲשֶׁ֥ר עַל־ הַבַּ֖יִת בְּתִרְצָֽה׃ (לז) וַיָּבֹ֤א זִמְרִי֙ וַיַּכֵּ֣הוּ וַיְמִיתֵ֔הוּ בִּשְׁנַת֙ עֶשְׂרִ֣ים וָשֶׁ֔בַע לְאָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה וַיִּמְלֹ֖ךְ תַּחְתָּֽיו׃ (לח) וַיְהִ֨י בְמָלְכ֜וֹ כְּשִׁבְתּ֣וֹ עַל־ כִּסְא֗וֹ הִכָּה֙ אֶת־ כָּל־ בֵּ֣ית בַּעְשָׁ֔א לֹֽא־ הִשְׁאִ֥יר ל֖וֹ מַשְׁתִּ֣ין בְּקִ֑יר וְגֹאֲלָ֖יו וְרֵעֵֽהוּ׃ (לט) וַיַּשְׁמֵ֣ד זִמְרִ֔י אֵ֖ת כָּל־ בֵּ֣ית בַּעְשָׁ֑א כִּדְבַ֤ר יְהוָה֙ אֲשֶׁ֣ר דִּבֶּ֣ר אֶל־ בַּעְשָׁ֔א בְּיַ֖ד יֵה֥וּא הַנָּבִֽיא׃ (מ) אֶ֚ל כָּל־ חַטֹּ֣אות בַּעְשָׁ֔א וְחַטֹּ֖אות אֵלָ֣ה בְנ֑וֹ אֲשֶׁ֣ר חָטְא֗וּ וַאֲשֶׁ֤ר הֶחֱטִ֙יאוּ֙ אֶת־ יִשְׂרָאֵ֔ל לְהַכְעִ֗יס אֶת־ יְהוָ֛ה אֱלֹהֵ֥י יִשְׂרָאֵ֖ל בְּהַבְלֵיהֶֽם׃ (מא) וְיֶ֛תֶר דִּבְרֵ֥י אֵלָ֖ה וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃
