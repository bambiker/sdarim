מלכים א פרק ו לפי החלוקה לסדרים

פרק ו בספר מלכים א (38 פסוקים) נמצא במלכים סדר ד (פסוקים לד–מה) ובמלכים סדר ה (פסוקים א–כו).

מלכים סדר ד

(א) יְהוּדָ֤ה וְיִשְׂרָאֵל֙ רַבִּ֔ים כַּח֥וֹל אֲשֶׁר־ עַל־ הַיָּ֖ם לָרֹ֑ב אֹכְלִ֥ים וְשֹׁתִ֖ים וּשְׂמֵחִֽים׃ (ב) וּשְׁלֹמֹ֗ה הָיָ֤ה מוֹשֵׁל֙ בְּכָל־ הַמַּמְלָכ֔וֹת מִן־ הַנָּהָר֙ אֶ֣רֶץ פְּלִשְׁתִּ֔ים וְעַ֖ד גְּב֣וּל מִצְרָ֑יִם מַגִּשִׁ֥ים מִנְחָ֛ה וְעֹבְדִ֥ים אֶת־ שְׁלֹמֹ֖ה כָּל־ יְמֵ֥י חַיָּֽיו׃ (ג) וַיְהִ֥י לֶֽחֶם־ שְׁלֹמֹ֖ה לְי֣וֹם אֶחָ֑ד שְׁלֹשִׁ֥ים כֹּר֙ סֹ֔לֶת וְשִׁשִּׁ֥ים כֹּ֖ר קָֽמַח׃ (ד) עֲשָׂרָ֨ה בָקָ֜ר בְּרִאִ֗ים וְעֶשְׂרִ֥ים בָּקָ֛ר רְעִ֖י וּמֵ֣אָה צֹ֑אן לְ֠בַד מֵֽאַיָּ֤ל וּצְבִי֙ וְיַחְמ֔וּר וּבַרְבֻּרִ֖ים אֲבוּסִֽים׃ (ה) כִּי־ ה֞וּא רֹדֶ֣ה ׀ בְּכָל־ עֵ֣בֶר הַנָּהָ֗ר מִתִּפְסַח֙ וְעַד־ עַזָּ֔ה בְּכָל־ מַלְכֵ֖י עֵ֣בֶר הַנָּהָ֑ר וְשָׁל֗וֹם הָ֥יָה ל֛וֹ מִכָּל־ עֲבָרָ֖יו מִסָּבִֽיב׃ (ו) וַיֵּשֶׁב֩ יְהוּדָ֨ה וְיִשְׂרָאֵ֜ל לָבֶ֗טַח אִ֣ישׁ תַּ֤חַת גַּפְנוֹ֙ וְתַ֣חַת תְּאֵֽנָת֔וֹ מִדָּ֖ן וְעַד־ בְּאֵ֣ר שָׁ֑בַע כֹּ֖ל יְמֵ֥י שְׁלֹמֹֽה׃ (ז) וַיְהִ֣י לִשְׁלֹמֹ֗ה אַרְבָּעִ֥ים אֶ֛לֶף אֻרְוֺ֥ת סוּסִ֖ים לְמֶרְכָּב֑וֹ וּשְׁנֵים־ עָשָׂ֥ר אֶ֖לֶף פָּרָשִֽׁים׃ (ח) וְכִלְכְּלוּ֩ הַנִּצָּבִ֨ים הָאֵ֜לֶּה אֶת־ הַמֶּ֣לֶךְ שְׁלֹמֹ֗ה וְאֵ֧ת כָּל־ הַקָּרֵ֛ב אֶל־ שֻׁלְחַ֥ן הַמֶּֽלֶךְ־ שְׁלֹמֹ֖ה אִ֣ישׁ חָדְשׁ֑וֹ לֹ֥א יְעַדְּר֖וּ דָּבָֽר׃ (ט) וְהַשְּׂעֹרִ֣ים וְהַתֶּ֔בֶן לַסּוּסִ֖ים וְלָרָ֑כֶשׁ יָבִ֗אוּ אֶל־ הַמָּקוֹם֙ אֲשֶׁ֣ר יִֽהְיֶה־ שָּׁ֔ם אִ֖ישׁ כְּמִשְׁפָּטֽוֹ׃ (י) וַיִּתֵּן֩ אֱלֹהִ֨ים חָכְמָ֧ה לִשְׁלֹמֹ֛ה וּתְבוּנָ֖ה הַרְבֵּ֣ה מְאֹ֑ד וְרֹ֣חַב לֵ֔ב כַּח֕וֹל אֲשֶׁ֖ר עַל־ שְׂפַ֥ת הַיָּֽם׃ (יא) וַתֵּ֙רֶב֙ חָכְמַ֣ת שְׁלֹמֹ֔ה מֵֽחָכְמַ֖ת כָּל־ בְּנֵי־ קֶ֑דֶם וּמִכֹּ֖ל חָכְמַ֥ת מִצְרָֽיִם׃ (יב) וַיֶּחְכַּם֮ מִכָּל־ הָֽאָדָם֒ מֵאֵיתָ֣ן הָאֶזְרָחִ֗י וְהֵימָ֧ן וְכַלְכֹּ֛ל וְדַרְדַּ֖ע בְּנֵ֣י מָח֑וֹל וַיְהִֽי־ שְׁמ֥וֹ בְכָֽל־ הַגּוֹיִ֖ם סָבִֽיב׃ (יג) וַיְדַבֵּ֕ר שְׁלֹ֥שֶׁת אֲלָפִ֖ים מָשָׁ֑ל וַיְהִ֥י שִׁיר֖וֹ חֲמִשָּׁ֥ה וָאָֽלֶף׃ (יד) וַיְדַבֵּר֮ עַל־ הָֽעֵצִים֒ מִן־ הָאֶ֙רֶז֙ אֲשֶׁ֣ר בַּלְּבָנ֔וֹן וְעַד֙ הָאֵז֔וֹב אֲשֶׁ֥ר יֹצֵ֖א בַּקִּ֑יר וַיְדַבֵּר֙ עַל־ הַבְּהֵמָ֣ה וְעַל־ הָע֔וֹף וְעַל־ הָרֶ֖מֶשׂ וְעַל־ הַדָּגִֽים׃ (טו) וַיָּבֹ֙אוּ֙ מִכָּל־ הָ֣עַמִּ֔ים לִשְׁמֹ֕עַ אֵ֖ת חָכְמַ֣ת שְׁלֹמֹ֑ה מֵאֵת֙ כָּל־ מַלְכֵ֣י הָאָ֔רֶץ אֲשֶׁ֥ר שָׁמְע֖וּ אֶת־ חָכְמָתֽוֹ׃ (טז) וַ֠יִּשְׁלַח חִירָ֨ם מֶֽלֶךְ־ צ֤וֹר אֶת־ עֲבָדָיו֙ אֶל־ שְׁלֹמֹ֔ה כִּ֣י שָׁמַ֔ע כִּ֥י אֹת֛וֹ מָשְׁח֥וּ לְמֶ֖לֶךְ תַּ֣חַת אָבִ֑יהוּ כִּ֣י אֹהֵ֗ב הָיָ֥ה חִירָ֛ם לְדָוִ֖ד כָּל־ הַיָּמִֽים׃ (יז) וַיִּשְׁלַ֣ח שְׁלֹמֹ֔ה אֶל־ חִירָ֖ם לֵאמֹֽר׃ (יח) אַתָּ֨ה יָדַ֜עְתָּ אֶת־ דָּוִ֣ד אָבִ֗י כִּ֣י לֹ֤א יָכֹל֙ לִבְנ֣וֹת בַּ֗יִת לְשֵׁם֙ יְהוָ֣ה אֱלֹהָ֔יו מִפְּנֵ֥י הַמִּלְחָמָ֖ה אֲשֶׁ֣ר סְבָבֻ֑הוּ עַ֤ד תֵּת־ יְהוָה֙ אֹתָ֔ם תַּ֖חַת כַּפּ֥וֹת (רגלו) רַגְלָֽי׃ (יט) וְעַתָּ֕ה הֵנִ֨יחַ יְהוָ֧ה אֱלֹהַ֛י לִ֖י מִסָּבִ֑יב אֵ֣ין שָׂטָ֔ן וְאֵ֖ין פֶּ֥גַע רָֽע׃ (כ) וְהִנְנִ֣י אֹמֵ֔ר לִבְנ֣וֹת בַּ֔יִת לְשֵׁ֖ם יְהוָ֣ה אֱלֹהָ֑י כַּאֲשֶׁ֣ר ׀ דִּבֶּ֣ר יְהוָ֗ה אֶל־ דָּוִ֤ד אָבִי֙ לֵאמֹ֔ר בִּנְךָ֗ אֲשֶׁ֨ר אֶתֵּ֤ן תַּחְתֶּ֙יךָ֙ עַל־ כִּסְאֶ֔ךָ הֽוּא־ יִבְנֶ֥ה הַבַּ֖יִת לִשְׁמִֽי׃ (כא) וְעַתָּ֡ה צַוֵּה֩ וְיִכְרְתוּ־ לִ֨י אֲרָזִ֜ים מִן־ הַלְּבָנ֗וֹן וַֽעֲבָדַי֙ יִהְי֣וּ עִם־ עֲבָדֶ֔יךָ וּשְׂכַ֤ר עֲבָדֶ֙יךָ֙ אֶתֵּ֣ן לְךָ֔ כְּכֹ֖ל אֲשֶׁ֣ר תֹּאמֵ֑ר כִּ֣י ׀ אַתָּ֣ה יָדַ֗עְתָּ כִּ֣י אֵ֥ין בָּ֛נוּ אִ֛ישׁ יֹדֵ֥עַ לִכְרָת־ עֵצִ֖ים כַּצִּדֹנִֽים׃ (כב) וַיְהִ֞י כִּשְׁמֹ֧עַ חִירָ֛ם אֶת־ דִּבְרֵ֥י שְׁלֹמֹ֖ה וַיִּשְׂמַ֣ח מְאֹ֑ד וַיֹּ֗אמֶר בָּר֤וּךְ יְהוָה֙ הַיּ֔וֹם אֲשֶׁ֨ר נָתַ֤ן לְדָוִד֙ בֵּ֣ן חָכָ֔ם עַל־ הָעָ֥ם הָרָ֖ב הַזֶּֽה׃ (כג) וַיִּשְׁלַ֤ח חִירָם֙ אֶל־ שְׁלֹמֹ֣ה לֵאמֹ֔ר שָׁמַ֕עְתִּי אֵ֥ת אֲשֶׁר־ שָׁלַ֖חְתָּ אֵלָ֑י אֲנִ֤י אֶֽעֱשֶׂה֙ אֶת־ כָּל־ חֶפְצְךָ֔ בַּעֲצֵ֥י אֲרָזִ֖ים וּבַעֲצֵ֥י בְרוֹשִֽׁים׃ (כד) עֲ֠בָדַי יֹרִ֨דוּ מִן־ הַלְּבָנ֜וֹן יָ֗מָּה וַ֠אֲנִי אֲשִׂימֵ֨ם דֹּבְר֤וֹת בַּיָּם֙ עַֽד־ הַמָּק֞וֹם אֲשֶׁר־ תִּשְׁלַ֥ח אֵלַ֛י וְנִפַּצְתִּ֥ים שָׁ֖ם וְאַתָּ֣ה תִשָּׂ֑א וְאַתָּה֙ תַּעֲשֶׂ֣ה אֶת־ חֶפְצִ֔י לָתֵ֖ת לֶ֥חֶם בֵּיתִֽי׃ (כה) וַיְהִ֨י חִיר֜וֹם נֹתֵ֣ן לִשְׁלֹמֹ֗ה עֲצֵ֧י אֲרָזִ֛ים וַעֲצֵ֥י בְרוֹשִׁ֖ים כָּל־ חֶפְצֽוֹ׃ (כו) וּשְׁלֹמֹה֩ נָתַ֨ן לְחִירָ֜ם עֶשְׂרִים֩ אֶ֨לֶף כֹּ֤ר חִטִּים֙ מַכֹּ֣לֶת לְבֵית֔וֹ וְעֶשְׂרִ֥ים כֹּ֖ר שֶׁ֣מֶן כָּתִ֑ית כֹּֽה־ יִתֵּ֧ן שְׁלֹמֹ֛ה לְחִירָ֖ם שָׁנָ֥ה בְשָׁנָֽה׃ (כז) וַיהוָ֗ה נָתַ֤ן חָכְמָה֙ לִשְׁלֹמֹ֔ה כַּאֲשֶׁ֖ר דִּבֶּר־ ל֑וֹ וַיְהִ֣י שָׁלֹ֗ם בֵּ֤ין חִירָם֙ וּבֵ֣ין שְׁלֹמֹ֔ה וַיִּכְרְת֥וּ בְרִ֖ית שְׁנֵיהֶֽם׃ (כח) וַיַּ֨עַל הַמֶּ֧לֶךְ שְׁלֹמֹ֛ה מַ֖ס מִכָּל־ יִשְׂרָאֵ֑ל וַיְהִ֣י הַמַּ֔ס שְׁלֹשִׁ֥ים אֶ֖לֶף אִֽישׁ׃ (כט) וַיִּשְׁלָחֵ֣ם לְבָנ֗וֹנָה עֲשֶׂ֨רֶת אֲלָפִ֤ים בַּחֹ֙דֶשׁ֙ חֲלִיפ֔וֹת חֹ֚דֶשׁ יִהְי֣וּ בַלְּבָנ֔וֹן שְׁנַ֥יִם חֳדָשִׁ֖ים בְּבֵית֑וֹ וַאֲדֹנִירָ֖ם עַל־ הַמַּֽס׃ (ל) וַיְהִ֧י לִשְׁלֹמֹ֛ה שִׁבְעִ֥ים אֶ֖לֶף נֹשֵׂ֣א סַבָּ֑ל וּשְׁמֹנִ֥ים אֶ֖לֶף חֹצֵ֥ב בָּהָֽר׃ (לא) לְ֠בַד מִשָּׂרֵ֨י הַנִּצָּבִ֤ים לִשְׁלֹמֹה֙ אֲשֶׁ֣ר עַל־ הַמְּלָאכָ֔ה שְׁלֹ֥שֶׁת אֲלָפִ֖ים וּשְׁלֹ֣שׁ מֵא֑וֹת הָרֹדִ֣ים בָּעָ֔ם הָעֹשִׂ֖ים בַּמְּלָאכָֽה׃ (לב) וַיְצַ֣ו הַמֶּ֡לֶךְ וַיַּסִּעוּ֩ אֲבָנִ֨ים גְּדֹל֜וֹת אֲבָנִ֧ים יְקָר֛וֹת לְיַסֵּ֥ד הַבָּ֖יִת אַבְנֵ֥י גָזִֽית׃ (לג) וַֽיִּפְסְל֞וּ בֹּנֵ֧י שְׁלֹמֹ֛ה וּבֹנֵ֥י חִיר֖וֹם וְהַגִּבְלִ֑ים וַיָּכִ֛ינוּ הָעֵצִ֥ים וְהָאֲבָנִ֖ים לִבְנ֥וֹת הַבָּֽיִת׃ תחילת פרק ו (לד) וַיְהִ֣י בִשְׁמוֹנִ֣ים שָׁנָ֣ה וְאַרְבַּ֣ע מֵא֣וֹת שָׁנָ֡ה לְצֵ֣את בְּנֵֽי־ יִשְׂרָאֵ֣ל מֵאֶֽרֶץ־ מִצְרַיִם֩ בַּשָּׁנָ֨ה הָרְבִיעִ֜ית בְּחֹ֣דֶשׁ זִ֗ו ה֚וּא הַחֹ֣דֶשׁ הַשֵּׁנִ֔י לִמְלֹ֥ךְ שְׁלֹמֹ֖ה עַל־ יִשְׂרָאֵ֑ל וַיִּ֥בֶן הַבַּ֖יִת לַיהוָֽה׃ (לה) וְהַבַּ֗יִת אֲשֶׁ֨ר בָּנָ֜ה הַמֶּ֤לֶךְ שְׁלֹמֹה֙ לַֽיהוָ֔ה שִׁשִּֽׁים־ אַמָּ֥ה אָרְכּ֖וֹ וְעֶשְׂרִ֤ים רָחְבּ֑וֹ וּשְׁלֹשִׁ֥ים אַמָּ֖ה קוֹמָתֽוֹ׃ (לו) וְהָאוּלָ֗ם עַל־ פְּנֵי֙ הֵיכַ֣ל הַבַּ֔יִת עֶשְׂרִ֣ים אַמָּה֙ אָרְכּ֔וֹ עַל־ פְּנֵ֖י רֹ֣חַב הַבָּ֑יִת עֶ֧שֶׂר בָּאַמָּ֛ה רָחְבּ֖וֹ עַל־ פְּנֵ֥י הַבָּֽיִת׃ (לז) וַיַּ֣עַשׂ לַבָּ֔יִת חַלּוֹנֵ֖י שְׁקֻפִ֥ים אֲטֻמִֽים׃ (לח) וַיִּבֶן֩ עַל־ קִ֨יר הַבַּ֤יִת (יצוע) יָצִ֙יעַ֙ סָבִ֔יב אֶת־ קִיר֤וֹת הַבַּ֙יִת֙ סָבִ֔יב לַֽהֵיכָ֖ל וְלַדְּבִ֑יר וַיַּ֥עַשׂ צְלָע֖וֹת סָבִֽיב׃ (לט) (היצוע) הַיָּצִ֨יעַ הַתַּחְתֹּנָ֜ה חָמֵ֧שׁ פְּעָמִֽים׃ (לב) וְאֵ֨ת אוּלָ֤ם הָֽעַמּוּדִים֙ עָשָׂ֔ה חֲמִשִּׁ֤ים אַמָּה֙ אָרְכּ֔וֹ וּשְׁלֹשִׁ֥ים אַמָּ֖ה רָחְבּ֑וֹ וְאוּלָם֙ עַל־ פְּנֵיהֶ֔ם וְעַמֻּדִ֥ים וְעָ֖ב עַל־ פְּנֵיהֶֽם׃ (לג) וְאוּלָ֤ם הַכִּסֵּא֙ אֲשֶׁ֣ר יִשְׁפָּט־ שָׁ֔ם אֻלָ֥ם הַמִּשְׁפָּ֖ט עָשָׂ֑ה וְסָפ֣וּן בָּאֶ֔רֶז מֵהַקַּרְקַ֖ע עַד־ הַקַּרְקָֽע׃ (לד) וּבֵיתוֹ֩ אֲשֶׁר־ יֵ֨שֶׁב שָׁ֜ם חָצֵ֣ר הָאַחֶ֗רֶת מִבֵּית֙ לָֽאוּלָ֔ם כַּמַּֽעֲשֶׂ֥ה הַזֶּ֖ה הָיָ֑ה וּבַ֜יִת יַעֲשֶׂ֤ה לְבַת־ פַּרְעֹה֙ אֲשֶׁ֣ר לָקַ֣ח שְׁלֹמֹ֔ה כָּאוּלָ֖ם הַזֶּֽה׃ (לה) כָּל־ אֵ֜לֶּה אֲבָנִ֤ים יְקָרֹת֙ כְּמִדֹּ֣ת גָּזִ֔ית מְגֹרָר֥וֹת בַּמְּגֵרָ֖ה מִבַּ֣יִת וּמִח֑וּץ וּמִמַּסָּד֙ עַד־ הַטְּפָח֔וֹת וּמִח֖וּץ עַד־ הֶחָצֵ֥ר הַגְּדוֹלָֽה׃ (לו) וּמְיֻסָּ֕ד אֲבָנִ֥ים יְקָר֖וֹת אֲבָנִ֣ים גְּדֹל֑וֹת אַבְנֵי֙ עֶ֣שֶׂר אַמּ֔וֹת וְאַבְנֵ֖י שְׁמֹנֶ֥ה אַמּֽוֹת׃ (לז) וּמִלְמַ֗עְלָה אֲבָנִ֧ים יְקָר֛וֹת כְּמִדּ֥וֹת גָּזִ֖ית וָאָֽרֶז׃ (לח) וְחָצֵ֨ר הַגְּדוֹלָ֜ה סָבִ֗יב שְׁלֹשָׁה֙ טוּרִ֣ים גָּזִ֔ית וְט֖וּר כְּרֻתֹ֣ת אֲרָזִ֑ים וְלַחֲצַ֧ר בֵּית־ יְהוָ֛ה הַפְּנִימִ֖ית וּלְאֻלָ֥ם הַבָּֽיִת׃ (לט) וַיִּשְׁלַח֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה וַיִּקַּ֥ח אֶת־ חִירָ֖ם מִצֹּֽר׃ (מ) בֶּן־ אִשָּׁה֩ אַלְמָנָ֨ה ה֜וּא מִמַּטֵּ֣ה נַפְתָּלִ֗י וְאָבִ֣יו אִישׁ־ צֹרִי֮ חֹרֵ֣שׁ נְחֹשֶׁת֒ וַ֠יִּמָּלֵא אֶת־ הַחָכְמָ֤ה וְאֶת־ הַתְּבוּנָה֙ וְאֶת־ הַדַּ֔עַת לַעֲשׂ֥וֹת כָּל־ מְלָאכָ֖ה בַּנְּחֹ֑שֶׁת וַיָּבוֹא֙ אֶל־ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה וַיַּ֖עַשׂ אֶת־ כָּל־ מְלַאכְתּֽוֹ׃ (מא) וַיָּ֛צַר אֶת־ שְׁנֵ֥י הָעַמּוּדִ֖ים נְחֹ֑שֶׁת שְׁמֹנֶ֨ה עֶשְׂרֵ֜ה אַמָּ֗ה קוֹמַת֙ הָעַמּ֣וּד הָאֶחָ֔ד וְחוּט֙ שְׁתֵּים־ עֶשְׂרֵ֣ה אַמָּ֔ה יָסֹ֖ב אֶת־ הָעַמּ֥וּד הַשֵּׁנִֽי׃ (מב) וּשְׁתֵּ֨י כֹתָרֹ֜ת עָשָׂ֗ה לָתֵ֛ת עַל־ רָאשֵׁ֥י הָֽעַמּוּדִ֖ים מֻצַ֣ק נְחֹ֑שֶׁת חָמֵ֣שׁ אַמּ֗וֹת קוֹמַת֙ הַכֹּתֶ֣רֶת הָאֶחָ֔ת וְחָמֵ֣שׁ אַמּ֔וֹת קוֹמַ֖ת הַכֹּתֶ֥רֶת הַשֵּׁנִֽית׃ (מג) שְׂבָכִ֞ים מַעֲשֵׂ֣ה שְׂבָכָ֗ה גְּדִלִים֙ מַעֲשֵׂ֣ה שַׁרְשְׁר֔וֹת לַכֹּ֣תָרֹ֔ת אֲשֶׁ֖ר עַל־ רֹ֣אשׁ הָעַמּוּדִ֑ים שִׁבְעָה֙ לַכֹּתֶ֣רֶת הָאֶחָ֔ת וְשִׁבְעָ֖ה לַכֹּתֶ֥רֶת הַשֵּׁנִֽית׃ (מד) וַיַּ֖עַשׂ אֶת־ הָעַמּוּדִ֑ים וּשְׁנֵי֩ טוּרִ֨ים סָבִ֜יב עַל־ הַשְּׂבָכָ֣ה הָאֶחָ֗ת לְכַסּ֤וֹת אֶת־ הַכֹּֽתָרֹת֙ אֲשֶׁר֙ עַל־ רֹ֣אשׁ הָֽרִמֹּנִ֔ים וְכֵ֣ן עָשָׂ֔ה לַכֹּתֶ֖רֶת הַשֵּׁנִֽית׃ (מה) וְכֹֽתָרֹ֗ת אֲשֶׁר֙ עַל־ רֹ֣אשׁ הָעַמּוּדִ֔ים מַעֲשֵׂ֖ה שׁוּשַׁ֣ן בָּֽאוּלָ֑ם אַרְבַּ֖ע אַמּֽוֹת׃ (מו) וְכֹתָרֹ֗ת עַל־ שְׁנֵי֙ הָֽעַמּוּדִ֔ים גַּם־ מִמַּ֙עַל֙ מִלְּעֻמַּ֣ת הַבֶּ֔טֶן אֲשֶׁ֖ר לְעֵ֣בֶר (שבכה) הַשְּׂבָכָ֑ה וְהָרִמּוֹנִ֤ים מָאתַ֙יִם֙ טֻרִ֣ים סָבִ֔יב עַ֖ל הַכֹּתֶ֥רֶת הַשֵּׁנִֽית׃
