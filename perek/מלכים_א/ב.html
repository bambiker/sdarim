מלכים א פרק ב לפי החלוקה לסדרים

פרק ב בספר מלכים א (46 פסוקים) נמצא במלכים סדר ב (פסוקים ז–נ) ובמלכים סדר ג (פסוקים א–ב).

מלכים סדר ב

(א) וְגַם־ כָּ֖כָה אָמַ֣ר הַמֶּ֑לֶךְ בָּר֨וּךְ יְהוָ֜ה אֱלֹהֵ֣י יִשְׂרָאֵ֗ל אֲשֶׁ֨ר נָתַ֥ן הַיּ֛וֹם יֹשֵׁ֥ב עַל־ כִּסְאִ֖י וְעֵינַ֥י רֹאֽוֹת׃ (ב) וַיֶּֽחֶרְדוּ֙ וַיָּקֻ֔מוּ כָּל־ הַ֨קְּרֻאִ֔ים אֲשֶׁ֖ר לַאֲדֹנִיָּ֑הוּ וַיֵּלְכ֖וּ אִ֥ישׁ לְדַרְכּֽוֹ׃ (ג) וַאֲדֹ֣נִיָּ֔הוּ יָרֵ֖א מִפְּנֵ֣י שְׁלֹמֹ֑ה וַיָּ֣קָם וַיֵּ֔לֶךְ וַֽיַּחֲזֵ֖ק בְּקַרְנ֥וֹת הַמִּזְבֵּֽחַ׃ (ד) וַיֻּגַּ֤ד לִשְׁלֹמֹה֙ לֵאמֹ֔ר הִנֵּה֙ אֲדֹ֣נִיָּ֔הוּ יָרֵ֖א אֶת־ הַמֶּ֣לֶךְ שְׁלֹמֹ֑ה וְ֠הִנֵּה אָחַ֞ז בְּקַרְנ֤וֹת הַמִּזְבֵּ֙חַ֙ לֵאמֹ֔ר יִשָּֽׁבַֽע־ לִ֤י כַיּוֹם֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה אִם־ יָמִ֥ית אֶת־ עַבְדּ֖וֹ בֶּחָֽרֶב׃ (ה) וַיֹּ֣אמֶר שְׁלֹמֹ֔ה אִ֚ם יִהְיֶ֣ה לְבֶן־ חַ֔יִל לֹֽא־ יִפֹּ֥ל מִשַּׂעֲרָת֖וֹ אָ֑רְצָה וְאִם־ רָעָ֥ה תִמָּצֵא־ ב֖וֹ וָמֵֽת׃ (ו) וַיִּשְׁלַ֞ח הַמֶּ֣לֶךְ שְׁלֹמֹ֗ה וַיֹּרִדֻ֙הוּ֙ מֵעַ֣ל הַמִּזְבֵּ֔חַ וַיָּבֹ֕א וַיִּשְׁתַּ֖חוּ לַמֶּ֣לֶךְ שְׁלֹמֹ֑ה וַיֹּֽאמֶר־ ל֥וֹ שְׁלֹמֹ֖ה לֵ֥ךְ לְבֵיתֶֽךָ׃ תחילת פרק ב (ז) וַיִּקְרְב֥וּ יְמֵֽי־ דָוִ֖ד לָמ֑וּת וַיְצַ֛ו אֶת־ שְׁלֹמֹ֥ה בְנ֖וֹ לֵאמֹֽר׃ (ח) אָנֹכִ֣י הֹלֵ֔ךְ בְּדֶ֖רֶךְ כָּל־ הָאָ֑רֶץ וְחָזַקְתָּ֖ וְהָיִ֥יתָֽ לְאִֽישׁ׃ (ט) וְשָׁמַרְתָּ֞ אֶת־ מִשְׁמֶ֣רֶת ׀ יְהוָ֣ה אֱלֹהֶ֗יךָ לָלֶ֤כֶת בִּדְרָכָיו֙ לִשְׁמֹ֨ר חֻקֹּתָ֤יו מִצְוֺתָיו֙ וּמִשְׁפָּטָ֣יו וְעֵדְוֺתָ֔יו כַּכָּת֖וּב בְּתוֹרַ֣ת מֹשֶׁ֑ה לְמַ֣עַן תַּשְׂכִּ֗יל אֵ֚ת כָּל־ אֲשֶׁ֣ר תַּֽעֲשֶׂ֔ה וְאֵ֛ת כָּל־ אֲשֶׁ֥ר תִּפְנֶ֖ה שָֽׁם׃ (י) לְמַעַן֩ יָקִ֨ים יְהוָ֜ה אֶת־ דְּבָר֗וֹ אֲשֶׁ֨ר דִּבֶּ֣ר עָלַי֮ לֵאמֹר֒ אִם־ יִשְׁמְר֨וּ בָנֶ֜יךָ אֶת־ דַּרְכָּ֗ם לָלֶ֤כֶת לְפָנַי֙ בֶּאֱמֶ֔ת בְּכָל־ לְבָבָ֖ם וּבְכָל־ נַפְשָׁ֑ם לֵאמֹ֕ר לֹֽא־ יִכָּרֵ֤ת לְךָ֙ אִ֔ישׁ מֵעַ֖ל כִּסֵּ֥א יִשְׂרָאֵֽל׃ (יא) וְגַ֣ם אַתָּ֣ה יָדַ֡עְתָּ אֵת֩ אֲשֶׁר־ עָ֨שָׂה לִ֜י יוֹאָ֣ב בֶּן־ צְרוּיָ֗ה אֲשֶׁ֣ר עָשָׂ֣ה לִשְׁנֵֽי־ שָׂרֵ֣י צִבְא֣וֹת יִ֠שְׂרָאֵל לְאַבְנֵ֨ר בֶּן־ נֵ֜ר וְלַעֲמָשָׂ֤א בֶן־ יֶ֙תֶר֙ וַיַּ֣הַרְגֵ֔ם וַיָּ֥שֶׂם דְּמֵֽי־ מִלְחָמָ֖ה בְּשָׁלֹ֑ם וַיִּתֵּ֞ן דְּמֵ֣י מִלְחָמָ֗ה בַּחֲגֹֽרָתוֹ֙ אֲשֶׁ֣ר בְּמָתְנָ֔יו וּֽבְנַעֲל֖וֹ אֲשֶׁ֥ר בְּרַגְלָֽיו׃ (יב) וְעָשִׂ֖יתָ כְּחָכְמָתֶ֑ךָ וְלֹֽא־ תוֹרֵ֧ד שֵׂיבָת֛וֹ בְּשָׁלֹ֖ם שְׁאֹֽל׃ (יג) וְלִבְנֵ֨י בַרְזִלַּ֤י הַגִּלְעָדִי֙ תַּֽעֲשֶׂה־ חֶ֔סֶד וְהָי֖וּ בְּאֹכְלֵ֣י שֻׁלְחָנֶ֑ךָ כִּי־ כֵן֙ קָרְב֣וּ אֵלַ֔י בְּבָרְחִ֕י מִפְּנֵ֖י אַבְשָׁל֥וֹם אָחִֽיךָ׃ (יד) וְהִנֵּ֣ה עִ֠מְּךָ שִֽׁמְעִ֨י בֶן־ גֵּרָ֥א בֶן־ הַיְמִינִי֮ מִבַּחֻרִים֒ וְה֤וּא קִֽלְלַ֙נִי֙ קְלָלָ֣ה נִמְרֶ֔צֶת בְּי֖וֹם לֶכְתִּ֣י מַחֲנָ֑יִם וְהֽוּא־ יָרַ֤ד לִקְרָאתִי֙ הַיַּרְדֵּ֔ן וָאֶשָּׁ֨בַֽע ל֤וֹ בַֽיהוָה֙ לֵאמֹ֔ר אִם־ אֲמִֽיתְךָ֖ בֶּחָֽרֶב׃ (טו) וְעַתָּה֙ אַל־ תְּנַקֵּ֔הוּ כִּ֛י אִ֥ישׁ חָכָ֖ם אָ֑תָּה וְיָֽדַעְתָּ֙ אֵ֣ת אֲשֶׁ֣ר תַּֽעֲשֶׂה־ לּ֔וֹ וְהוֹרַדְתָּ֧ אֶת־ שֵׂיבָת֛וֹ בְּדָ֖ם שְׁאֽוֹל׃ (טז) וַיִּשְׁכַּ֥ב דָּוִ֖ד עִם־ אֲבֹתָ֑יו וַיִּקָּבֵ֖ר בְּעִ֥יר דָּוִֽד׃ (יז) וְהַיָּמִ֗ים אֲשֶׁ֨ר מָלַ֤ךְ דָּוִד֙ עַל־ יִשְׂרָאֵ֔ל אַרְבָּעִ֖ים שָׁנָ֑ה בְּחֶבְר֤וֹן מָלַךְ֙ שֶׁ֣בַע שָׁנִ֔ים וּבִירוּשָׁלִַ֣ם מָלַ֔ךְ שְׁלֹשִׁ֥ים וְשָׁלֹ֖שׁ שָׁנִֽים׃ (יח) וּשְׁלֹמֹ֕ה יָשַׁ֕ב עַל־ כִּסֵּ֖א דָּוִ֣ד אָבִ֑יו וַתִּכֹּ֥ן מַלְכֻת֖וֹ מְאֹֽד׃ (יט) וַיָּבֹ֞א אֲדֹנִיָּ֣הוּ בֶן־ חַגֵּ֗ית אֶל־ בַּת־ שֶׁ֙בַע֙ אֵם־ שְׁלֹמֹ֔ה וַתֹּ֖אמֶר הֲשָׁל֣וֹם בֹּאֶ֑ךָ וַיֹּ֖אמֶר שָׁלֽוֹם׃ (כ) וַיֹּ֕אמֶר דָּבָ֥ר לִ֖י אֵלָ֑יִךְ וַתֹּ֖אמֶר דַּבֵּֽר׃ (כא) וַיֹּ֗אמֶר אַ֤תְּ יָדַ֙עַתְּ֙ כִּי־ לִי֙ הָיְתָ֣ה הַמְּלוּכָ֔ה וְעָלַ֞י שָׂ֧מוּ כָֽל־ יִשְׂרָאֵ֛ל פְּנֵיהֶ֖ם לִמְלֹ֑ךְ וַתִּסֹּ֤ב הַמְּלוּכָה֙ וַתְּהִ֣י לְאָחִ֔י כִּ֥י מֵיְהוָ֖ה הָ֥יְתָה לּֽוֹ׃ (כב) וְעַתָּ֗ה שְׁאֵלָ֤ה אַחַת֙ אָֽנֹכִי֙ שֹׁאֵ֣ל מֵֽאִתָּ֔ךְ אַל־ תָּשִׁ֖בִי אֶת־ פָּנָ֑י וַתֹּ֥אמֶר אֵלָ֖יו דַּבֵּֽר׃ (כג) וַיֹּ֗אמֶר אִמְרִי־ נָא֙ לִשְׁלֹמֹ֣ה הַמֶּ֔לֶךְ כִּ֥י לֹֽא־ יָשִׁ֖יב אֶת־ פָּנָ֑יִךְ וְיִתֶּן־ לִ֛י אֶת־ אֲבִישַׁ֥ג הַשּׁוּנַמִּ֖ית לְאִשָּֽׁה׃ (כד) וַתֹּ֥אמֶר בַּת־ שֶׁ֖בַע ט֑וֹב אָנֹכִ֕י אֲדַבֵּ֥ר עָלֶ֖יךָ אֶל־ הַמֶּֽלֶךְ׃ (כה) וַתָּבֹ֤א בַת־ שֶׁ֙בַע֙ אֶל־ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה לְדַבֶּר־ ל֖וֹ עַל־ אֲדֹנִיָּ֑הוּ וַיָּקָם֩ הַמֶּ֨לֶךְ לִקְרָאתָ֜הּ וַיִּשְׁתַּ֣חוּ לָ֗הּ וַיֵּ֙שֶׁב֙ עַל־ כִּסְא֔וֹ וַיָּ֤שֶׂם כִּסֵּא֙ לְאֵ֣ם הַמֶּ֔לֶךְ וַתֵּ֖שֶׁב לִֽימִינֽוֹ׃ (כו) וַתֹּ֗אמֶר שְׁאֵלָ֨ה אַחַ֤ת קְטַנָּה֙ אָֽנֹכִי֙ שֹׁאֶ֣לֶת מֵֽאִתָּ֔ךְ אַל־ תָּ֖שֶׁב אֶת־ פָּנָ֑י וַיֹּֽאמֶר־ לָ֤הּ הַמֶּ֙לֶךְ֙ שַׁאֲלִ֣י אִמִּ֔י כִּ֥י לֹֽא־ אָשִׁ֖יב אֶת־ פָּנָֽיִךְ׃ (כז) וַתֹּ֕אמֶר יֻתַּ֖ן אֶת־ אֲבִישַׁ֣ג הַשֻּׁנַמִּ֑ית לַאֲדֹנִיָּ֥הוּ אָחִ֖יךָ לְאִשָּֽׁה׃ (כח) וַיַּעַן֩ הַמֶּ֨לֶךְ שְׁלֹמֹ֜ה וַיֹּ֣אמֶר לְאִמּ֗וֹ וְלָמָה֩ אַ֨תְּ שֹׁאֶ֜לֶת אֶת־ אֲבִישַׁ֤ג הַשֻּׁנַמִּית֙ לַאֲדֹ֣נִיָּ֔הוּ וְשַֽׁאֲלִי־ לוֹ֙ אֶת־ הַמְּלוּכָ֔ה כִּ֛י ה֥וּא אָחִ֖י הַגָּד֣וֹל מִמֶּ֑נִּי וְלוֹ֙ וּלְאֶבְיָתָ֣ר הַכֹּהֵ֔ן וּלְיוֹאָ֖ב בֶּן־ צְרוּיָֽה׃ (כט) וַיִּשָּׁבַע֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה בַּֽיהוָ֖ה לֵאמֹ֑ר כֹּ֣ה יַֽעֲשֶׂה־ לִּ֤י אֱלֹהִים֙ וְכֹ֣ה יוֹסִ֔יף כִּ֣י בְנַפְשׁ֔וֹ דִּבֶּר֙ אֲדֹ֣נִיָּ֔הוּ אֶת־ הַדָּבָ֖ר הַזֶּֽה׃ (ל) וְעַתָּ֗ה חַי־ יְהוָה֙ אֲשֶׁ֣ר הֱכִינַ֗נִי (ויושיביני) וַיּֽוֹשִׁיבַ֙נִי֙ עַל־ כִּסֵּא֙ דָּוִ֣ד אָבִ֔י וַאֲשֶׁ֧ר עָֽשָׂה־ לִ֛י בַּ֖יִת כַּאֲשֶׁ֣ר דִּבֵּ֑ר כִּ֣י הַיּ֔וֹם יוּמַ֖ת אֲדֹנִיָּֽהוּ׃ (לא) וַיִּשְׁלַח֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה בְּיַ֖ד בְּנָיָ֣הוּ בֶן־ יְהוֹיָדָ֑ע וַיִּפְגַּע־ בּ֖וֹ וַיָּמֹֽת׃ (לב) וּלְאֶבְיָתָ֨ר הַכֹּהֵ֜ן אָמַ֣ר הַמֶּ֗לֶךְ עֲנָתֹת֙ לֵ֣ךְ עַל־ שָׂדֶ֔יךָ כִּ֛י אִ֥ישׁ מָ֖וֶת אָ֑תָּה וּבַיּ֨וֹם הַזֶּ֜ה לֹ֣א אֲמִיתֶ֗ךָ כִּֽי־ נָשָׂ֜אתָ אֶת־ אֲר֨וֹן אֲדֹנָ֤י יְהֹוִה֙ לִפְנֵי֙ דָּוִ֣ד אָבִ֔י וְכִ֣י הִתְעַנִּ֔יתָ בְּכֹ֥ל אֲשֶֽׁר־ הִתְעַנָּ֖ה אָבִֽי׃ (לג) וַיְגָ֤רֶשׁ שְׁלֹמֹה֙ אֶת־ אֶבְיָתָ֔ר מִהְי֥וֹת כֹּהֵ֖ן לַֽיהוָ֑ה לְמַלֵּא֙ אֶת־ דְּבַ֣ר יְהוָ֔ה אֲשֶׁ֥ר דִּבֶּ֛ר עַל־ בֵּ֥ית עֵלִ֖י בְּשִׁלֹֽה׃ (לד) וְהַשְּׁמֻעָה֙ בָּ֣אָה עַד־ יוֹאָ֔ב כִּ֣י יוֹאָ֗ב נָטָה֙ אַחֲרֵ֣י אֲדֹנִיָּ֔ה וְאַחֲרֵ֥י אַבְשָׁל֖וֹם לֹ֣א נָטָ֑ה וַיָּ֤נָס יוֹאָב֙ אֶל־ אֹ֣הֶל יְהוָ֔ה וַֽיַּחֲזֵ֖ק נֶ֣פֶשׁ אֹיְבֶ֑יךָ וְשָׁאַ֧לְתָּ לְּךָ֛ הָבִ֖ין לִשְׁמֹ֥עַ מִשְׁפָּֽט׃ (יד) הִנֵּ֥ה עָשִׂ֖יתִי כִּדְבָרֶ֑יךָ הִנֵּ֣ה ׀ נָתַ֣תִּי לְךָ֗ לֵ֚ב חָכָ֣ם וְנָב֔וֹן אֲשֶׁ֤ר כָּמ֙וֹךָ֙ לֹא־ הָיָ֣ה לְפָנֶ֔יךָ וְאַחֲרֶ֖יךָ לֹא־ יָק֥וּם כָּמֽוֹךָ׃ (טו) וְגַ֨ם אֲשֶׁ֤ר לֹֽא־ שָׁאַ֙לְתָּ֙ נָתַ֣תִּי לָ֔ךְ גַּם־ עֹ֖שֶׁר גַּם־ כָּב֑וֹד אֲ֠שֶׁר לֹא־ הָיָ֨ה כָמ֥וֹךָ אִ֛ישׁ בַּמְּלָכִ֖ים כָּל־ יָמֶֽיךָ׃ (טז) וְאִ֣ם ׀ תֵּלֵ֣ךְ בִּדְרָכַ֗י לִשְׁמֹ֤ר חֻקַּי֙ וּמִצְוֺתַ֔י כַּאֲשֶׁ֥ר הָלַ֖ךְ דָּוִ֣יד אָבִ֑יךָ וְהַאַרַכְתִּ֖י אֶת־ יָמֶֽיךָ׃ (יז) וַיִּקַ֥ץ שְׁלֹמֹ֖ה וְהִנֵּ֣ה חֲל֑וֹם וַיָּב֨וֹא יְרוּשָׁלִַ֜ם וַֽיַּעֲמֹ֣ד ׀ לִפְנֵ֣י ׀ אֲר֣וֹן בְּרִית־ אֲדֹנָ֗י וַיַּ֤עַל עֹלוֹת֙ וַיַּ֣עַשׂ שְׁלָמִ֔ים וַיַּ֥עַשׂ מִשְׁתֶּ֖ה לְכָל־ עֲבָדָֽיו׃ (יח) אָ֣ז תָּבֹ֗אנָה שְׁתַּ֛יִם נָשִׁ֥ים זֹנ֖וֹת אֶל־ הַמֶּ֑לֶךְ וַֽתַּעֲמֹ֖דְנָה לְפָנָֽיו׃ (יט) וַתֹּ֜אמֶר הָאִשָּׁ֤ה הָֽאַחַת֙ בִּ֣י אֲדֹנִ֔י אֲנִי֙ וְהָאִשָּׁ֣ה הַזֹּ֔את יֹשְׁבֹ֖ת בְּבַ֣יִת אֶחָ֑ד וָאֵלֵ֥ד עִמָּ֖הּ בַּבָּֽיִת׃ (כ) וַיְהִ֞י בַּיּ֤וֹם הַשְּׁלִישִׁי֙ לְלִדְתִּ֔י וַתֵּ֖לֶד גַּם־ הָאִשָּׁ֣ה הַזֹּ֑את וַאֲנַ֣חְנוּ יַחְדָּ֗ו אֵֽין־ זָ֤ר אִתָּ֙נוּ֙ בַּבַּ֔יִת זוּלָתִ֥י שְׁתַּֽיִם־ אֲנַ֖חְנוּ בַּבָּֽיִת׃ (כא) וַיָּ֛מָת בֶּן־ הָאִשָּׁ֥ה הַזֹּ֖את לָ֑יְלָה אֲשֶׁ֥ר שָׁכְבָ֖ה עָלָֽיו׃ (כב) וַתָּקָם֩ בְּת֨וֹךְ הַלַּ֜יְלָה וַתִּקַּ֧ח אֶת־ בְּנִ֣י מֵֽאֶצְלִ֗י וַאֲמָֽתְךָ֙ יְשֵׁנָ֔ה וַתַּשְׁכִּיבֵ֖הוּ בְּחֵיקָ֑הּ וְאֶת־ בְּנָ֥הּ הַמֵּ֖ת הִשְׁכִּ֥יבָה בְחֵיקִֽי׃ (כג) וָאָקֻ֥ם בַּבֹּ֛קֶר לְהֵינִ֥יק אֶת־ בְּנִ֖י וְהִנֵּה־ מֵ֑ת וָאֶתְבּוֹנֵ֤ן אֵלָיו֙ בַּבֹּ֔קֶר וְהִנֵּ֛ה לֹֽא־ הָיָ֥ה בְנִ֖י אֲשֶׁ֥ר יָלָֽדְתִּי׃ (כד) וַתֹּאמֶר֩ הָאִשָּׁ֨ה הָאַחֶ֜רֶת לֹ֣א כִ֗י בְּנִ֤י הַחַי֙ וּבְנֵ֣ךְ הַמֵּ֔ת וְזֹ֤את אֹמֶ֙רֶת֙ לֹ֣א כִ֔י בְּנֵ֥ךְ הַמֵּ֖ת וּבְנִ֣י הֶחָ֑י וַתְּדַבֵּ֖רְנָה לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כה) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ זֹ֣את אֹמֶ֔רֶת זֶה־ בְּנִ֥י הַחַ֖י וּבְנֵ֣ךְ הַמֵּ֑ת וְזֹ֤את אֹמֶ֙רֶת֙ לֹ֣א כִ֔י בְּנֵ֥ךְ הַמֵּ֖ת וּבְנִ֥י הֶחָֽי׃ (כו) וַיֹּ֥אמֶר הַמֶּ֖לֶךְ קְח֣וּ לִי־ חָ֑רֶב וַיָּבִ֥אוּ הַחֶ֖רֶב לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כז) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ גִּזְר֛וּ אֶת־ הַיֶּ֥לֶד הַחַ֖י לִשְׁנָ֑יִם וּתְנ֤וּ אֶֽת־ הַחֲצִי֙ לְאַחַ֔ת וְאֶֽת־ הַחֲצִ֖י לְאֶחָֽת׃ (כח) וַתֹּ֣אמֶר הָאִשָּׁה֩ אֲשֶׁר־ בְּנָ֨הּ הַחַ֜י אֶל־ הַמֶּ֗לֶךְ כִּֽי־ נִכְמְר֣וּ רַחֲמֶיהָ֮ עַל־ בְּנָהּ֒ וַתֹּ֣אמֶר ׀ בִּ֣י אֲדֹנִ֗י תְּנוּ־ לָהּ֙ אֶת־ הַיָּל֣וּד הַחַ֔י וְהָמֵ֖ת אַל־ תְּמִיתֻ֑הוּ וְזֹ֣את אֹמֶ֗רֶת גַּם־ לִ֥י גַם־ לָ֛ךְ לֹ֥א יִהְיֶ֖ה גְּזֹֽרוּ׃ (כט) וַיַּ֨עַן הַמֶּ֜לֶךְ וַיֹּ֗אמֶר תְּנוּ־ לָהּ֙ אֶת־ הַיָּל֣וּד הַחַ֔י וְהָמֵ֖ת לֹ֣א תְמִיתֻ֑הוּ הִ֖יא אִמּֽוֹ׃ (ל) וַיִּשְׁמְע֣וּ כָל־ יִשְׂרָאֵ֗ל אֶת־ הַמִּשְׁפָּט֙ אֲשֶׁ֣ר שָׁפַ֣ט הַמֶּ֔לֶךְ וַיִּֽרְא֖וּ מִפְּנֵ֣י הַמֶּ֑לֶךְ כִּ֣י רָא֔וּ כִּֽי־ חָכְמַ֧ת אֱלֹהִ֛ים בְּקִרְבּ֖וֹ לַעֲשׂ֥וֹת מִשְׁפָּֽט׃ (לא) וַֽיְהִי֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה מֶ֖לֶךְ עַל־ כָּל־ יִשְׂרָאֵֽל׃ (לב) וְאֵ֥לֶּה הַשָּׂרִ֖ים אֲשֶׁר־ ל֑וֹ עֲזַרְיָ֥הוּ בֶן־ צָד֖וֹק הַכֹּהֵֽן׃ (לג) אֱלִיחֹ֧רֶף וַאֲחִיָּ֛ה בְּנֵ֥י שִׁישָׁ֖א סֹפְרִ֑ים יְהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד הַמַּזְכִּֽיר׃ (לד) וּבְנָיָ֥הוּ בֶן־ יְהוֹיָדָ֖ע עַל־ הַצָּבָ֑א וְצָד֥וֹק וְאֶבְיָתָ֖ר כֹּהֲנִֽים׃ (לה) וַעֲזַרְיָ֥הוּ בֶן־ נָתָ֖ן עַל־ הַנִּצָּבִ֑ים וְזָב֧וּד בֶּן־ נָתָ֛ן כֹּהֵ֖ן רֵעֶ֥ה הַמֶּֽלֶךְ׃ (לו) וַאֲחִישָׁ֖ר עַל־ הַבָּ֑יִת וַאֲדֹנִירָ֥ם בֶּן־ עַבְדָּ֖א עַל־ הַמַּֽס׃ (לז) וְלִשְׁלֹמֹ֞ה שְׁנֵים־ עָשָׂ֤ר נִצָּבִים֙ עַל־ כָּל־ יִשְׂרָאֵ֔ל וְכִלְכְּל֥וּ אֶת־ הַמֶּ֖לֶךְ וְאֶת־ בֵּית֑וֹ חֹ֧דֶשׁ בַּשָּׁנָ֛ה יִהְיֶ֥ה עַל־ (אחד) הָאֶחָ֖ד לְכַלְכֵּֽל׃ (לח) וְאֵ֣לֶּה שְׁמוֹתָ֔ם בֶּן־ ח֖וּר בְּהַ֥ר אֶפְרָֽיִם׃ (לט) בֶּן־ דֶּ֛קֶר בְּמָקַ֥ץ וּבְשַֽׁעַלְבִ֖ים וּבֵ֣ית שָׁ֑מֶשׁ וְאֵיל֖וֹן בֵּ֥ית חָנָֽן׃ (מ) בֶּן־ חֶ֖סֶד בָּֽאֲרֻבּ֑וֹת ל֥וֹ שֹׂכֹ֖ה וְכָל־ אֶ֥רֶץ חֵֽפֶר׃ (מא) בֶּן־ אֲבִֽינָדָ֖ב כָּל־ נָ֣פַת דֹּ֑אר טָפַת֙ בַּת־ שְׁלֹמֹ֔ה הָ֥יְתָה לּ֖וֹ לְאִשָּֽׁה׃ (מב) בַּֽעֲנָא֙ בֶּן־ אֲחִיל֔וּד תַּעְנַ֖ךְ וּמְגִדּ֑וֹ וְכָל־ בֵּ֣ית שְׁאָ֡ן אֲשֶׁר֩ אֵ֨צֶל צָרְתַ֜נָה מִתַּ֣חַת לְיִזְרְעֶ֗אל מִבֵּ֤ית שְׁאָן֙ עַ֚ד אָבֵ֣ל מְחוֹלָ֔ה עַ֖ד מֵעֵ֥בֶר לְיָקְמֳעָֽם׃ (מג) בֶּן־ גֶּ֖בֶר בְּרָמֹ֣ת גִּלְעָ֑ד ל֡וֹ חַוֺּת֩ יָאִ֨יר בֶּן־ מְנַשֶּׁ֜ה אֲשֶׁ֣ר בַּגִּלְעָ֗ד ל֚וֹ חֶ֤בֶל אַרְגֹּב֙ אֲשֶׁ֣ר בַּבָּשָׁ֔ן שִׁשִּׁים֙ עָרִ֣ים גְּדֹל֔וֹת חוֹמָ֖ה וּבְרִ֥יחַ נְחֹֽשֶׁת׃ (מד) אֲחִֽינָדָ֥ב בֶּן־ עִדֹּ֖א מַחֲנָֽיְמָה׃ (מה) אֲחִימַ֖עַץ בְּנַפְתָּלִ֑י גַּם־ ה֗וּא לָקַ֛ח אֶת־ בָּשְׂמַ֥ת בַּת־ שְׁלֹמֹ֖ה לְאִשָּֽׁה׃ (מו) בַּֽעֲנָא֙ בֶּן־ חוּשָׁ֔י בְּאָשֵׁ֖ר וּבְעָלֽוֹת׃ (מז) יְהוֹשָׁפָ֥ט בֶּן־ פָּר֖וּחַ בְּיִשָׂשכָֽר׃ (מח) שִׁמְעִ֥י בֶן־ אֵלָ֖א בְּבִנְיָמִֽן׃ (מט) גֶּ֥בֶר בֶּן־ אֻרִ֖י בְּאֶ֣רֶץ גִּלְעָ֑ד אֶ֜רֶץ סִיח֣וֹן ׀ מֶ֣לֶךְ הָאֱמֹרִ֗י וְעֹג֙ מֶ֣לֶךְ הַבָּשָׁ֔ן וּנְצִ֥יב אֶחָ֖ד אֲשֶׁ֥ר בָּאָֽרֶץ׃
