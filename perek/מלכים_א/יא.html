מלכים א פרק יא לפי החלוקה לסדרים

פרק יא בספר מלכים א (43 פסוקים) נמצא במלכים סדר ט (פסוקים כב–מח) ובמלכים סדר י (פסוקים א–טז).

מלכים סדר ט

(א) יְהִ֨י יְהוָ֤ה אֱלֹהֶ֙יךָ֙ בָּר֔וּךְ אֲשֶׁר֙ חָפֵ֣ץ בְּךָ֔ לְתִתְּךָ֖ עַל־ כִּסֵּ֣א יִשְׂרָאֵ֑ל בְּאַהֲבַ֨ת יְהוָ֤ה אֶת־ יִשְׂרָאֵל֙ לְעֹלָ֔ם וַיְשִֽׂימְךָ֣ לְמֶ֔לֶךְ לַעֲשׂ֥וֹת מִשְׁפָּ֖ט וּצְדָקָֽה׃ (ב) וַתִּתֵּ֨ן לַמֶּ֜לֶךְ מֵאָ֥ה וְעֶשְׂרִ֣ים ׀ כִּכַּ֣ר זָהָ֗ב וּבְשָׂמִ֛ים הַרְבֵּ֥ה מְאֹ֖ד וְאֶ֣בֶן יְקָרָ֑ה לֹא־ בָא֩ כַבֹּ֨שֶׂם הַה֥וּא עוֹד֙ לָרֹ֔ב אֲשֶׁר־ נָתְנָ֥ה מַֽלְכַּת־ שְׁבָ֖א לַמֶּ֥לֶךְ שְׁלֹמֹֽה׃ (ג) וְגַם֙ אֳנִ֣י חִירָ֔ם אֲשֶׁר־ נָשָׂ֥א זָהָ֖ב מֵאוֹפִ֑יר הֵבִ֨יא מֵאֹפִ֜יר עֲצֵ֧י אַלְמֻגִּ֛ים הַרְבֵּ֥ה מְאֹ֖ד וְאֶ֥בֶן יְקָרָֽה׃ (ד) וַיַּ֣עַשׂ הַ֠מֶּלֶךְ אֶת־ עֲצֵ֨י הָאַלְמֻגִּ֜ים מִסְעָ֤ד לְבֵית־ יְהוָה֙ וּלְבֵ֣ית הַמֶּ֔לֶךְ וְכִנֹּר֥וֹת וּנְבָלִ֖ים לַשָּׁרִ֑ים לֹ֣א בָֽא־ כֵ֞ן עֲצֵ֤י אַלְמֻגִּים֙ וְלֹ֣א נִרְאָ֔ה עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (ה) וְהַמֶּ֨לֶךְ שְׁלֹמֹ֜ה נָתַ֣ן לְמַֽלְכַּת־ שְׁבָ֗א אֶת־ כָּל־ חֶפְצָהּ֙ אֲשֶׁ֣ר שָׁאָ֔לָה מִלְּבַד֙ אֲשֶׁ֣ר נָֽתַן־ לָ֔הּ כְּיַ֖ד הַמֶּ֣לֶךְ שְׁלֹמֹ֑ה וַתֵּ֛פֶן וַתֵּ֥לֶךְ לְאַרְצָ֖הּ הִ֥יא וַעֲבָדֶֽיהָ׃ (ו) וַֽיְהִי֙ מִשְׁקַ֣ל הַזָּהָ֔ב אֲשֶׁר־ בָּ֥א לִשְׁלֹמֹ֖ה בְּשָׁנָ֣ה אֶחָ֑ת שֵׁ֥שׁ מֵא֛וֹת שִׁשִּׁ֥ים וָשֵׁ֖שׁ כִּכַּ֥ר זָהָֽב׃ (ז) לְבַד֙ מֵאַנְשֵׁ֣י הַתָּרִ֔ים וּמִסְחַ֖ר הָרֹכְלִ֑ים וְכָל־ מַלְכֵ֥י הָעֶ֖רֶב וּפַח֥וֹת הָאָֽרֶץ׃ (ח) וַיַּ֨עַשׂ הַמֶּ֧לֶךְ שְׁלֹמֹ֛ה מָאתַ֥יִם צִנָּ֖ה זָהָ֣ב שָׁח֑וּט שֵׁשׁ־ מֵא֣וֹת זָהָ֔ב יַעֲלֶ֖ה עַל־ הַצִּנָּ֥ה הָאֶחָֽת׃ (ט) וּשְׁלֹשׁ־ מֵא֤וֹת מָֽגִנִּים֙ זָהָ֣ב שָׁח֔וּט שְׁלֹ֤שֶׁת מָנִים֙ זָהָ֔ב יַעֲלֶ֖ה עַל־ הַמָּגֵ֣ן הָאֶחָ֑ת וַיִּתְּנֵ֣ם הַמֶּ֔לֶךְ בֵּ֖ית יַ֥עַר הַלְּבָנֽוֹן׃ (י) וַיַּ֧עַשׂ הַמֶּ֛לֶךְ כִּסֵּא־ שֵׁ֖ן גָּד֑וֹל וַיְצַפֵּ֖הוּ זָהָ֥ב מוּפָֽז׃ (יא) שֵׁ֧שׁ מַעֲל֣וֹת לַכִּסֵּ֗ה וְרֹאשׁ־ עָגֹ֤ל לַכִּסֵּה֙ מֵאַֽחֲרָ֔יו וְיָדֹ֛ת מִזֶּ֥ה וּמִזֶּ֖ה אֶל־ מְק֣וֹם הַשָּׁ֑בֶת וּשְׁנַ֣יִם אֲרָי֔וֹת עֹמְדִ֖ים אֵ֥צֶל הַיָּדֽוֹת׃ (יב) וּשְׁנֵ֧ים עָשָׂ֣ר אֲרָיִ֗ים עֹמְדִ֥ים שָׁ֛ם עַל־ שֵׁ֥שׁ הַֽמַּעֲל֖וֹת מִזֶּ֣ה וּמִזֶּ֑ה לֹֽא־ נַעֲשָׂ֥ה כֵ֖ן לְכָל־ מַמְלָכֽוֹת׃ (יג) וְ֠כֹל כְּלֵ֞י מַשְׁקֵ֨ה הַמֶּ֤לֶךְ שְׁלֹמֹה֙ זָהָ֔ב וְכֹ֗ל כְּלֵ֛י בֵּֽית־ יַ֥עַר הַלְּבָנ֖וֹן זָהָ֣ב סָג֑וּר אֵ֣ין כֶּ֗סֶף לֹ֥א נֶחְשָׁ֛ב בִּימֵ֥י שְׁלֹמֹ֖ה לִמְאֽוּמָה׃ (יד) כִּי֩ אֳנִ֨י תַרְשִׁ֤ישׁ לַמֶּ֙לֶךְ֙ בַּיָּ֔ם עִ֖ם אֳנִ֣י חִירָ֑ם אַחַת֩ לְשָׁלֹ֨שׁ שָׁנִ֜ים תָּב֣וֹא ׀ אֳנִ֣י תַרְשִׁ֗ישׁ נֹֽשְׂאֵת֙ זָהָ֣ב וָכֶ֔סֶף שֶׁנְהַבִּ֥ים וְקֹפִ֖ים וְתֻכִּיִּֽים׃ (טו) וַיִּגְדַּל֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה מִכֹּ֖ל מַלְכֵ֣י הָאָ֑רֶץ לְעֹ֖שֶׁר וּלְחָכְמָֽה׃ (טז) וְכָ֨ל־ הָאָ֔רֶץ מְבַקְשִׁ֖ים אֶת־ פְּנֵ֣י שְׁלֹמֹ֑ה לִשְׁמֹ֙עַ֙ אֶת־ חָכְמָת֔וֹ אֲשֶׁר־ נָתַ֥ן אֱלֹהִ֖ים בְּלִבּֽוֹ׃ (יז) וְהֵ֣מָּה מְבִאִ֣ים אִ֣ישׁ מִנְחָת֡וֹ כְּלֵ֣י כֶסֶף֩ וּכְלֵ֨י זָהָ֤ב וּשְׂלָמוֹת֙ וְנֵ֣שֶׁק וּבְשָׂמִ֔ים סוּסִ֖ים וּפְרָדִ֑ים דְּבַר־ שָׁנָ֖ה בְּשָׁנָֽה׃ (יח) וַיֶּאֱסֹ֣ף שְׁלֹמֹה֮ רֶ֣כֶב וּפָרָשִׁים֒ וַיְהִי־ ל֗וֹ אֶ֤לֶף וְאַרְבַּע־ מֵאוֹת֙ רֶ֔כֶב וּשְׁנֵים־ עָשָׂ֥ר אֶ֖לֶף פָּֽרָשִׁ֑ים וַיַּנְחֵם֙ בְּעָרֵ֣י הָרֶ֔כֶב וְעִם־ הַמֶּ֖לֶךְ בִּירוּשָׁלִָֽם׃ (יט) וַיִּתֵּ֨ן הַמֶּ֧לֶךְ אֶת־ הַכֶּ֛סֶף בִּירוּשָׁלִַ֖ם כָּאֲבָנִ֑ים וְאֵ֣ת הָאֲרָזִ֗ים נָתַ֛ן כַּשִּׁקְמִ֥ים אֲשֶׁר־ בַּשְּׁפֵלָ֖ה לָרֹֽב׃ (כ) וּמוֹצָ֧א הַסּוּסִ֛ים אֲשֶׁ֥ר לִשְׁלֹמֹ֖ה מִמִּצְרָ֑יִם וּמִקְוֵ֕ה סֹחֲרֵ֣י הַמֶּ֔לֶךְ יִקְח֥וּ מִקְוֵ֖ה בִּמְחִֽיר׃ (כא) וַֽ֠תַּעֲלֶה וַתֵּצֵ֨א מֶרְכָּבָ֤ה מִמִּצְרַ֙יִם֙ בְּשֵׁ֣שׁ מֵא֣וֹת כֶּ֔סֶף וְס֖וּס בַּחֲמִשִּׁ֣ים וּמֵאָ֑ה וְ֠כֵן לְכָל־ מַלְכֵ֧י הַחִתִּ֛ים וּלְמַלְכֵ֥י אֲרָ֖ם בְּיָדָ֥ם יֹצִֽאוּ׃ תחילת פרק יא (כב) וְהַמֶּ֣לֶךְ שְׁלֹמֹ֗ה אָהַ֞ב נָשִׁ֧ים נָכְרִיּ֛וֹת רַבּ֖וֹת וְאֶת־ בַּת־ פַּרְעֹ֑ה מוֹאֲבִיּ֤וֹת עַמֳּנִיּוֹת֙ אֲדֹ֣מִיֹּ֔ת צֵדְנִיֹּ֖ת חִתִּיֹּֽת׃ (כג) מִן־ הַגּוֹיִ֗ם אֲשֶׁ֣ר אָֽמַר־ יְהוָה֩ אֶל־ בְּנֵ֨י יִשְׂרָאֵ֜ל לֹֽא־ תָבֹ֣אוּ בָהֶ֗ם וְהֵם֙ לֹא־ יָבֹ֣אוּ בָכֶ֔ם אָכֵן֙ יַטּ֣וּ אֶת־ לְבַבְכֶ֔ם אַחֲרֵ֖י אֱלֹהֵיהֶ֑ם בָּהֶ֛ם דָּבַ֥ק שְׁלֹמֹ֖ה לְאַהֲבָֽה׃ (כד) וַיְהִי־ ל֣וֹ נָשִׁ֗ים שָׂרוֹת֙ שְׁבַ֣ע מֵא֔וֹת וּפִֽלַגְשִׁ֖ים שְׁלֹ֣שׁ מֵא֑וֹת וַיַּטּ֥וּ נָשָׁ֖יו אֶת־ לִבּֽוֹ׃ (כה) וַיְהִ֗י לְעֵת֙ זִקְנַ֣ת שְׁלֹמֹ֔ה נָשָׁיו֙ הִטּ֣וּ אֶת־ לְבָב֔וֹ אַחֲרֵ֖י אֱלֹהִ֣ים אֲחֵרִ֑ים וְלֹא־ הָיָ֨ה לְבָב֤וֹ שָׁלֵם֙ עִם־ יְהוָ֣ה אֱלֹהָ֔יו כִּלְבַ֖ב דָּוִ֥יד אָבִֽיו׃ (כו) וַיֵּ֣לֶךְ שְׁלֹמֹ֔ה אַחֲרֵ֣י עַשְׁתֹּ֔רֶת אֱלֹהֵ֖י צִדֹנִ֑ים וְאַחֲרֵ֣י מִלְכֹּ֔ם שִׁקֻּ֖ץ עַמֹּנִֽים׃ (כז) וַיַּ֧עַשׂ שְׁלֹמֹ֛ה הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וְלֹ֥א מִלֵּ֛א אַחֲרֵ֥י יְהוָ֖ה כְּדָוִ֥ד אָבִֽיו׃ (כח) אָז֩ יִבְנֶ֨ה שְׁלֹמֹ֜ה בָּמָ֗ה לִכְמוֹשׁ֙ שִׁקֻּ֣ץ מוֹאָ֔ב בָּהָ֕ר אֲשֶׁ֖ר עַל־ פְּנֵ֣י יְרוּשָׁלִָ֑ם וּלְמֹ֕לֶךְ שִׁקֻּ֖ץ בְּנֵ֥י עַמּֽוֹן׃ (כט) וְכֵ֣ן עָשָׂ֔ה לְכָל־ נָשָׁ֖יו הַנָּכְרִיּ֑וֹת מַקְטִיר֥וֹת וּֽמְזַבְּח֖וֹת לֵאלֹהֵיהֶֽן׃ (ל) וַיִּתְאַנַּ֥ף יְהוָ֖ה בִּשְׁלֹמֹ֑ה כִּֽי־ נָטָ֣ה לְבָב֗וֹ מֵעִ֤ם יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הַנִּרְאָ֥ה אֵלָ֖יו פַּעֲמָֽיִם׃ (לא) וְצִוָּ֤ה אֵלָיו֙ עַל־ הַדָּבָ֣ר הַזֶּ֔ה לְבִ֨לְתִּי־ לֶ֔כֶת אַחֲרֵ֖י אֱלֹהִ֣ים אֲחֵרִ֑ים וְלֹ֣א שָׁמַ֔ר אֵ֥ת אֲשֶׁר־ צִוָּ֖ה יְהוָֽה׃ (לב) וַיֹּ֨אמֶר יְהוָ֜ה לִשְׁלֹמֹ֗ה יַ֚עַן אֲשֶׁ֣ר הָֽיְתָה־ זֹּ֣את עִמָּ֔ךְ וְלֹ֤א שָׁמַ֙רְתָּ֙ בְּרִיתִ֣י וְחֻקֹּתַ֔י אֲשֶׁ֥ר צִוִּ֖יתִי עָלֶ֑יךָ קָרֹ֨עַ אֶקְרַ֤ע אֶת־ הַמַּמְלָכָה֙ מֵֽעָלֶ֔יךָ וּנְתַתִּ֖יהָ לְעַבְדֶּֽךָ׃ (לג) אַךְ־ בְּיָמֶ֙יךָ֙ לֹ֣א אֶעֱשֶׂ֔נָּה לְמַ֖עַן דָּוִ֣ד אָבִ֑יךָ מִיַּ֥ד בִּנְךָ֖ אֶקְרָעֶֽנָּה׃ (לד) רַ֤ק אֶת־ כָּל־ הַמַּמְלָכָה֙ לֹ֣א אֶקְרָ֔ע שֵׁ֥בֶט אֶחָ֖ד אֶתֵּ֣ן לִבְנֶ֑ךָ לְמַ֙עַן֙ דָּוִ֣ד עַבְדִּ֔י וּלְמַ֥עַן יְרוּשָׁלִַ֖ם אֲשֶׁ֥ר בָּחָֽרְתִּי׃ (לה) וַיָּ֨קֶם יְהוָ֤ה שָׂטָן֙ לִשְׁלֹמֹ֔ה אֵ֖ת הֲדַ֣ד הָאֲדֹמִ֑י מִזֶּ֧רַע הַמֶּ֛לֶךְ ה֖וּא בֶּאֱדֽוֹם׃ (לו) וַיְהִ֗י בִּֽהְי֤וֹת דָּוִד֙ אֶת־ אֱד֔וֹם בַּעֲל֗וֹת יוֹאָב֙ שַׂ֣ר הַצָּבָ֔א לְקַבֵּ֖ר אֶת־ הַחֲלָלִ֑ים וַיַּ֥ךְ כָּל־ זָכָ֖ר בֶּאֱדֽוֹם׃ (לז) כִּ֣י שֵׁ֧שֶׁת חֳדָשִׁ֛ים יָֽשַׁב־ שָׁ֥ם יוֹאָ֖ב וְכָל־ יִשְׂרָאֵ֑ל עַד־ הִכְרִ֥ית כָּל־ זָכָ֖ר בֶּאֱדֽוֹם׃ (לח) וַיִּבְרַ֣ח אֲדַ֡ד הוּא֩ וַאֲנָשִׁ֨ים אֲדֹמִיִּ֜ים מֵעַבְדֵ֥י אָבִ֛יו אִתּ֖וֹ לָב֣וֹא מִצְרָ֑יִם וַהֲדַ֖ד נַ֥עַר קָטָֽן׃ (לט) וַיָּקֻ֙מוּ֙ מִמִּדְיָ֔ן וַיָּבֹ֖אוּ פָּארָ֑ן וַיִּקְחוּ֩ אֲנָשִׁ֨ים עִמָּ֜ם מִפָּארָ֗ן וַיָּבֹ֤אוּ מִצְרַ֙יִם֙ אֶל־ פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם וַיִּתֶּן־ ל֣וֹ בַ֗יִת וְלֶ֙חֶם֙ אָ֣מַר ל֔וֹ וְאֶ֖רֶץ נָ֥תַן לֽוֹ׃ (מ) וַיִּמְצָא֙ הֲדַ֥ד חֵ֛ן בְּעֵינֵ֥י פַרְעֹ֖ה מְאֹ֑ד וַיִּתֶּן־ ל֤וֹ אִשָּׁה֙ אֶת־ אֲח֣וֹת אִשְׁתּ֔וֹ אֲח֖וֹת תַּחְפְּנֵ֥יס הַגְּבִירָֽה׃ (מא) וַתֵּ֨לֶד ל֜וֹ אֲח֣וֹת תַּחְפְּנֵ֗יס אֵ֚ת גְּנֻבַ֣ת בְּנ֔וֹ וַתִּגְמְלֵ֣הוּ תַחְפְּנֵ֔ס בְּת֖וֹךְ בֵּ֣ית פַּרְעֹ֑ה וַיְהִ֤י גְנֻבַת֙ בֵּ֣ית פַּרְעֹ֔ה בְּת֖וֹךְ בְּנֵ֥י פַרְעֹֽה׃ (מב) וַהֲדַ֞ד שָׁמַ֣ע בְּמִצְרַ֗יִם כִּֽי־ שָׁכַ֤ב דָּוִד֙ עִם־ אֲבֹתָ֔יו וְכִי־ מֵ֖ת יוֹאָ֣ב שַֽׂר־ הַצָּבָ֑א וַיֹּ֤אמֶר הֲדַד֙ אֶל־ פַּרְעֹ֔ה שַׁלְּחֵ֖נִי וְאֵלֵ֥ךְ אֶל־ אַרְצִֽי׃ (מג) וַיֹּ֧אמֶר ל֣וֹ פַרְעֹ֗ה כִּ֠י מָה־ אַתָּ֤ה חָסֵר֙ עִמִּ֔י וְהִנְּךָ֥ מְבַקֵּ֖שׁ לָלֶ֣כֶת אֶל־ אַרְצֶ֑ךָ וַיֹּ֣אמֶר ׀ לֹ֔א כִּ֥י שַׁלֵּ֖חַ תְּשַׁלְּחֵֽנִי׃ (מד) וַיָּ֨קֶם אֱלֹהִ֥ים לוֹ֙ שָׂטָ֔ן אֶת־ רְז֖וֹן בֶּן־ אֶלְיָדָ֑ע אֲשֶׁ֣ר בָּרַ֗ח מֵאֵ֛ת הֲדַדְעֶ֥זֶר מֶֽלֶךְ־ צוֹבָ֖ה אֲדֹנָֽיו׃ (מה) וַיִּקְבֹּ֤ץ עָלָיו֙ אֲנָשִׁ֔ים וַיְהִ֣י שַׂר־ גְּד֔וּד בַּהֲרֹ֥ג דָּוִ֖ד אֹתָ֑ם וַיֵּלְכ֤וּ דַמֶּ֙שֶׂק֙ וַיֵּ֣שְׁבוּ בָ֔הּ וַֽיִּמְלְכ֖וּ בְּדַמָּֽשֶׂק׃ (מו) וַיְהִ֨י שָׂטָ֤ן לְיִשְׂרָאֵל֙ כָּל־ יְמֵ֣י שְׁלֹמֹ֔ה וְאֶת־ הָרָעָ֖ה אֲשֶׁ֣ר הֲדָ֑ד וַיָּ֙קָץ֙ בְּיִשְׂרָאֵ֔ל וַיִּמְלֹ֖ךְ עַל־ אֲרָֽם׃ (מז) וְיָרָבְעָם֩ בֶּן־ נְבָ֨ט אֶפְרָתִ֜י מִן־ הַצְּרֵדָ֗ה וְשֵׁ֤ם אִמּוֹ֙ צְרוּעָה֙ אִשָּׁ֣ה אַלְמָנָ֔ה עֶ֖בֶד לִשְׁלֹמֹ֑ה וַיָּ֥רֶם יָ֖ד בַּמֶּֽלֶךְ׃ (מח) וְזֶ֣ה הַדָּבָ֔ר אֲשֶׁר־ הֵרִ֥ים יָ֖ד בַּמֶּ֑לֶךְ שְׁלֹמֹה֙ בָּנָ֣ה אֶת־ הַמִּלּ֔וֹא סָגַ֕ר אֶת־ פֶּ֕רֶץ עִ֖יר דָּוִ֥ד אָבִֽיו׃

(א) וְהָאִ֥ישׁ יָרָבְעָ֖ם גִּבּ֣וֹר חָ֑יִל וַיַּ֨רְא שְׁלֹמֹ֜ה אֶת־ הַנַּ֗עַר כִּֽי־ עֹשֵׂ֤ה מְלָאכָה֙ ה֔וּא וַיַּפְקֵ֣ד אֹת֔וֹ לְכָל־ סֵ֖בֶל בֵּ֥ית יוֹסֵֽף׃ (ב) וַֽיְהִי֙ בָּעֵ֣ת הַהִ֔יא וְיָֽרָבְעָ֖ם יָצָ֣א מִירוּשָׁלִָ֑ם וַיִּמְצָ֣א אֹת֡וֹ אֲחִיָּה֩ הַשִּׁילֹנִ֨י הַנָּבִ֜יא בַּדֶּ֗רֶךְ וְה֤וּא מִתְכַּסֶּה֙ בְּשַׂלְמָ֣ה חֲדָשָׁ֔ה וּשְׁנֵיהֶ֥ם לְבַדָּ֖ם בַּשָּׂדֶֽה׃ (ג) וַיִּתְפֹּ֣שׂ אֲחִיָּ֔ה בַּשַּׂלְמָ֥ה הַחֲדָשָׁ֖ה אֲשֶׁ֣ר עָלָ֑יו וַיִּ֨קְרָעֶ֔הָ שְׁנֵ֥ים עָשָׂ֖ר קְרָעִֽים׃ (ד) וַיֹּ֙אמֶר֙ לְיָֽרָבְעָ֔ם קַח־ לְךָ֖ עֲשָׂרָ֣ה קְרָעִ֑ים כִּ֣י כֹה֩ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֣י יִשְׂרָאֵ֗ל הִנְנִ֨י קֹרֵ֤עַ אֶת־ הַמַּמְלָכָה֙ מִיַּ֣ד שְׁלֹמֹ֔ה וְנָתַתִּ֣י לְךָ֔ אֵ֖ת עֲשָׂרָ֥ה הַשְּׁבָטִֽים׃ (ה) וְהַשֵּׁ֥בֶט הָאֶחָ֖ד יִֽהְיֶה־ לּ֑וֹ לְמַ֣עַן ׀ עַבְדִּ֣י דָוִ֗ד וּלְמַ֙עַן֙ יְר֣וּשָׁלִַ֔ם הָעִיר֙ אֲשֶׁ֣ר בָּחַ֣רְתִּי בָ֔הּ מִכֹּ֖ל שִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (ו) יַ֣עַן ׀ אֲשֶׁ֣ר עֲזָב֗וּנִי וַיִּֽשְׁתַּחֲווּ֮ לְעַשְׁתֹּרֶת֮ אֱלֹהֵ֣י צִֽדֹנִין֒ לִכְמוֹשׁ֙ אֱלֹהֵ֣י מוֹאָ֔ב וּלְמִלְכֹּ֖ם אֱלֹהֵ֣י בְנֵֽי־ עַמּ֑וֹן וְלֹֽא־ הָלְכ֣וּ בִדְרָכַ֗י לַעֲשׂ֨וֹת הַיָּשָׁ֧ר בְּעֵינַ֛י וְחֻקֹּתַ֥י וּמִשְׁפָּטַ֖י כְּדָוִ֥ד אָבִֽיו׃ (ז) וְלֹֽא־ אֶקַּ֥ח אֶת־ כָּל־ הַמַּמְלָכָ֖ה מִיָּד֑וֹ כִּ֣י ׀ נָשִׂ֣יא אֲשִׁתֶ֗נּוּ כֹּ֚ל יְמֵ֣י חַיָּ֔יו לְמַ֨עַן דָּוִ֤ד עַבְדִּי֙ אֲשֶׁ֣ר בָּחַ֣רְתִּי אֹת֔וֹ אֲשֶׁ֥ר שָׁמַ֖ר מִצְוֺתַ֥י וְחֻקֹּתָֽי׃ (ח) וְלָקַחְתִּ֥י הַמְּלוּכָ֖ה מִיַּ֣ד מֵעֲבֹדַ֨ת אָבִ֜יךָ הַקָּשָׁ֗ה וּמֵעֻלּ֧וֹ הַכָּבֵ֛ד אֲשֶׁר־ נָתַ֥ן עָלֵ֖ינוּ וְנַעַבְדֶֽךָּ׃ (כא) וַיֹּ֣אמֶר אֲלֵיהֶ֗ם לְכ֥וּ עֹ֛ד שְׁלֹשָׁ֥ה יָמִ֖ים וְשׁ֣וּבוּ אֵלָ֑י וַיֵּלְכ֖וּ הָעָֽם׃ (כב) וַיִּוָּעַ֞ץ הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ הַזְּקֵנִים֙ אֲשֶׁר־ הָי֣וּ עֹמְדִ֗ים אֶת־ פְּנֵי֙ שְׁלֹמֹ֣ה אָבִ֔יו בִּֽהְיֹת֥וֹ חַ֖י לֵאמֹ֑ר אֵ֚יךְ אַתֶּ֣ם נֽוֹעָצִ֔ים לְהָשִׁ֥יב אֶת־ הָֽעָם־ הַזֶּ֖ה דָּבָֽר׃ (כג) (וידבר) וַיְדַבְּר֨וּ אֵלָ֜יו לֵאמֹ֗ר אִם־ הַ֠יּוֹם תִּֽהְיֶה־ עֶ֜בֶד לָעָ֤ם הַזֶּה֙ וַֽעֲבַדְתָּ֔ם וַעֲנִיתָ֕ם וְדִבַּרְתָּ֥ אֲלֵיהֶ֖ם דְּבָרִ֣ים טוֹבִ֑ים וְהָי֥וּ לְךָ֛ עֲבָדִ֖ים כָּל־ הַיָּמִֽים׃ (כד) וַֽיַּעֲזֹ֛ב אֶת־ עֲצַ֥ת הַזְּקֵנִ֖ים אֲשֶׁ֣ר יְעָצֻ֑הוּ וַיִּוָּעַ֗ץ אֶת־ הַיְלָדִים֙ אֲשֶׁ֣ר גָּדְל֣וּ אִתּ֔וֹ אֲשֶׁ֥ר הָעֹמְדִ֖ים לְפָנָֽיו׃ (כה) וַיֹּ֣אמֶר אֲלֵיהֶ֗ם מָ֚ה אַתֶּ֣ם נֽוֹעָצִ֔ים וְנָשִׁ֥יב דָּבָ֖ר אֶת־ הָעָ֣ם הַזֶּ֑ה אֲשֶׁ֨ר דִּבְּר֤וּ אֵלַי֙ לֵאמֹ֔ר הָקֵל֙ מִן־ הָעֹ֔ל אֲשֶׁר־ נָתַ֥ן אָבִ֖יךָ עָלֵֽינוּ׃ (כו) וַיְדַבְּר֣וּ אֵלָ֗יו הַיְלָדִים֙ אֲשֶׁ֨ר גָּדְל֣וּ אִתּוֹ֮ לֵאמֹר֒ כֹּֽה־ תֹאמַ֣ר לָעָ֣ם הַזֶּ֡ה אֲשֶׁר֩ דִּבְּר֨וּ אֵלֶ֜יךָ לֵאמֹ֗ר אָבִ֙יךָ֙ הִכְבִּ֣יד אֶת־ עֻלֵּ֔נוּ וְאַתָּ֖ה הָקֵ֣ל מֵעָלֵ֑ינוּ כֹּ֚ה תְּדַבֵּ֣ר אֲלֵיהֶ֔ם קָֽטָנִּ֥י עָבָ֖ה מִמָּתְנֵ֥י אָבִֽי׃ (כז) וְעַתָּ֗ה אָבִי֙ הֶעְמִ֤יס עֲלֵיכֶם֙ עֹ֣ל כָּבֵ֔ד וַאֲנִ֖י אוֹסִ֣יף עַֽל־ עֻלְּכֶ֑ם אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֕י אֲיַסֵּ֥ר אֶתְכֶ֖ם בָּעַקְרַבִּֽים׃ (כח) (ויבו) וַיָּב֨וֹא יָרָבְעָ֧ם וְכָל־ הָעָ֛ם אֶל־ רְחַבְעָ֖ם בַּיּ֣וֹם הַשְּׁלִישִׁ֑י כַּאֲשֶׁ֨ר דִּבֶּ֤ר הַמֶּ֙לֶךְ֙ לֵאמֹ֔ר שׁ֥וּבוּ אֵלַ֖י בַּיּ֥וֹם הַשְּׁלִישִֽׁי׃ (כט) וַיַּ֧עַן הַמֶּ֛לֶךְ אֶת־ הָעָ֖ם קָשָׁ֑ה וַֽיַּעֲזֹ֛ב אֶת־ עֲצַ֥ת הַזְּקֵנִ֖ים אֲשֶׁ֥ר יְעָצֻֽהוּ׃ (ל) וַיְדַבֵּ֣ר אֲלֵיהֶ֗ם כַּעֲצַ֤ת הַיְלָדִים֙ לֵאמֹ֔ר אָבִי֙ הִכְבִּ֣יד אֶֽת־ עֻלְּכֶ֔ם וַאֲנִ֖י אֹסִ֣יף עַֽל־ עֻלְּכֶ֑ם אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֕י אֲיַסֵּ֥ר אֶתְכֶ֖ם בָּעַקְרַבִּֽים׃ (לא) וְלֹֽא־ שָׁמַ֥ע הַמֶּ֖לֶךְ אֶל־ הָעָ֑ם כִּֽי־ הָיְתָ֤ה סִבָּה֙ מֵעִ֣ם יְהוָ֔ה לְמַ֜עַן הָקִ֣ים אֶת־ דְּבָר֗וֹ אֲשֶׁ֨ר דִּבֶּ֤ר יְהוָה֙ בְּיַד֙ אֲחִיָּ֣ה הַשִּׁילֹנִ֔י אֶל־ יָרָבְעָ֖ם בֶּן־ נְבָֽט׃ (לב) וַיַּ֣רְא כָּל־ יִשְׂרָאֵ֗ל כִּ֠י לֹֽא־ שָׁמַ֣ע הַמֶּלֶךְ֮ אֲלֵיהֶם֒ וַיָּשִׁ֣בוּ הָעָ֣ם אֶת־ הַמֶּ֣לֶךְ דָּבָ֣ר ׀ לֵאמֹ֡ר מַה־ לָּנוּ֩ חֵ֨לֶק בְּדָוִ֜ד וְלֹֽא־ נַחֲלָ֣ה בְּבֶן־ יִשַׁ֗י לְאֹהָלֶ֙יךָ֙ יִשְׂרָאֵ֔ל עַתָּ֕ה רְאֵ֥ה בֵיתְךָ֖ דָּוִ֑ד וַיֵּ֥לֶךְ יִשְׂרָאֵ֖ל לְאֹהָלָֽיו׃ (לג) וּבְנֵ֣י יִשְׂרָאֵ֔ל הַיֹּשְׁבִ֖ים בְּעָרֵ֣י יְהוּדָ֑ה וַיִּמְלֹ֥ךְ עֲלֵיהֶ֖ם רְחַבְעָֽם׃ (לד) וַיִּשְׁלַ֞ח הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ אֲדֹרָם֙ אֲשֶׁ֣ר עַל־ הַמַּ֔ס וַיִּרְגְּמ֨וּ כָל־ יִשְׂרָאֵ֥ל בּ֛וֹ אֶ֖בֶן וַיָּמֹ֑ת וְהַמֶּ֣לֶךְ רְחַבְעָ֗ם הִתְאַמֵּץ֙ לַעֲל֣וֹת בַּמֶּרְכָּבָ֔ה לָנ֖וּס יְרוּשָׁלִָֽם׃ (לה) וַיִּפְשְׁע֤וּ יִשְׂרָאֵל֙ בְּבֵ֣ית דָּוִ֔ד עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (לו) וַיְהִ֞י כִּשְׁמֹ֤עַ כָּל־ יִשְׂרָאֵל֙ כִּֽי־ שָׁ֣ב יָרָבְעָ֔ם וַֽיִּשְׁלְח֗וּ וַיִּקְרְא֤וּ אֹתוֹ֙ אֶל־ הָ֣עֵדָ֔ה וַיַּמְלִ֥יכוּ אֹת֖וֹ עַל־ כָּל־ יִשְׂרָאֵ֑ל לֹ֤א הָיָה֙ אַחֲרֵ֣י בֵית־ דָּוִ֔ד זוּלָתִ֥י שֵֽׁבֶט־ יְהוּדָ֖ה לְבַדּֽוֹ׃ (לז) (ויבאו) וַיָּבֹ֣א רְחַבְעָם֮ יְרוּשָׁלִַם֒ וַיַּקְהֵל֩ אֶת־ כָּל־ בֵּ֨ית יְהוּדָ֜ה וְאֶת־ שֵׁ֣בֶט בִּנְיָמִ֗ן מֵאָ֨ה וּשְׁמֹנִ֥ים אֶ֛לֶף בָּח֖וּר עֹשֵׂ֣ה מִלְחָמָ֑ה לְהִלָּחֵם֙ עִם־ בֵּ֣ית יִשְׂרָאֵ֔ל לְהָשִׁיב֙ אֶת־ הַמְּלוּכָ֔ה לִרְחַבְעָ֖ם בֶּן־ שְׁלֹמֹֽה׃ (לח) וַיְהִי֙ דְּבַ֣ר הָֽאֱלֹהִ֔ים אֶל־ שְׁמַעְיָ֥ה אִישׁ־ הָאֱלֹהִ֖ים לֵאמֹֽר׃ (לט) אֱמֹ֗ר אֶל־ רְחַבְעָ֤ם בֶּן־ שְׁלֹמֹה֙ מֶ֣לֶךְ יְהוּדָ֔ה וְאֶל־ כָּל־ בֵּ֥ית יְהוּדָ֖ה וּבִנְיָמִ֑ין וְיֶ֥תֶר הָעָ֖ם לֵאמֹֽר׃
