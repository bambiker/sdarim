מלכים א פרק טז לפי החלוקה לסדרים

פרק טז בספר מלכים א (34 פסוקים) נמצא במלכים סדר יג (פסוקים כח–מא) ובמלכים סדר יד (פסוקים א–כ).

מלכים סדר יג

(א) וַיִּשְׁכַּ֤ב אֲבִיָּם֙ עִם־ אֲבֹתָ֔יו וַיִּקְבְּר֥וּ אֹת֖וֹ בְּעִ֣יר דָּוִ֑ד וַיִּמְלֹ֛ךְ אָסָ֥א בְנ֖וֹ תַּחְתָּֽיו׃ (ב) וּבִשְׁנַ֣ת עֶשְׂרִ֔ים לְיָרָבְעָ֖ם מֶ֣לֶךְ יִשְׂרָאֵ֑ל מָלַ֥ךְ אָסָ֖א מֶ֥לֶךְ יְהוּדָֽה׃ (ג) וְאַרְבָּעִ֤ים וְאַחַת֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ מַעֲכָ֖ה בַּת־ אֲבִישָׁלֽוֹם׃ (ד) וַיַּ֧עַשׂ אָסָ֛א הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה כְּדָוִ֖ד אָבִֽיו׃ (ה) וַיַּעֲבֵ֥ר הַקְּדֵשִׁ֖ים מִן־ הָאָ֑רֶץ וַיָּ֙סַר֙ אֶת־ כָּל־ הַגִּלֻּלִ֔ים אֲשֶׁ֥ר עָשׂ֖וּ אֲבֹתָֽיו׃ (ו) וְגַ֣ם ׀ אֶת־ מַעֲכָ֣ה אִמּ֗וֹ וַיְסִרֶ֙הָ֙ מִגְּבִירָ֔ה אֲשֶׁר־ עָשְׂתָ֥ה מִפְלֶ֖צֶת לָאֲשֵׁרָ֑ה וַיִּכְרֹ֤ת אָסָא֙ אֶת־ מִפְלַצְתָּ֔הּ וַיִּשְׂרֹ֖ף בְּנַ֥חַל קִדְרֽוֹן׃ (ז) וְהַבָּמ֖וֹת לֹא־ סָ֑רוּ רַ֣ק לְבַב־ אָסָ֗א הָיָ֥ה שָׁלֵ֛ם עִם־ יְהוָ֖ה כָּל־ יָמָֽיו׃ (ח) וַיָּבֵא֙ אֶת־ קָדְשֵׁ֣י אָבִ֔יו (וקדשו) וְקָדְשֵׁ֖י בֵּ֣ית יְהוָ֑ה כֶּ֥סֶף וְזָהָ֖ב וְכֵלִֽים׃ (ט) וּמִלְחָמָ֨ה הָיְתָ֜ה בֵּ֣ין אָסָ֗א וּבֵ֛ין בַּעְשָׁ֥א מֶֽלֶךְ־ יִשְׂרָאֵ֖ל כָּל־ יְמֵיהֶֽם׃ (י) וַיַּ֨עַל בַּעְשָׁ֤א מֶֽלֶךְ־ יִשְׂרָאֵל֙ עַל־ יְהוּדָ֔ה וַיִּ֖בֶן אֶת־ הָרָמָ֑ה לְבִלְתִּ֗י תֵּ֚ת יֹצֵ֣א וָבָ֔א לְאָסָ֖א מֶ֥לֶךְ יְהוּדָֽה׃ (יא) וַיִּקַּ֣ח אָ֠סָא אֶת־ כָּל־ הַכֶּ֨סֶף וְהַזָּהָ֜ב הַֽנּוֹתָרִ֣ים ׀ בְּאוֹצְר֣וֹת בֵּית־ יְהוָ֗ה וְאֶת־ אֽוֹצְרוֹת֙ בֵּ֣ית (מלך) הַמֶּ֔לֶךְ וַֽיִּתְּנֵ֖ם בְּיַד־ עֲבָדָ֑יו וַיִּשְׁלָחֵ֞ם הַמֶּ֣לֶךְ אָסָ֗א אֶל־ בֶּן־ הֲ֠דַד בֶּן־ טַבְרִמֹּ֤ן בֶּן־ חֶזְיוֹן֙ מֶ֣לֶךְ אֲרָ֔ם הַיֹּשֵׁ֥ב בְּדַמֶּ֖שֶׂק לֵאמֹֽר׃ (יב) בְּרִית֙ בֵּינִ֣י וּבֵינֶ֔ךָ בֵּ֥ין אָבִ֖י וּבֵ֣ין אָבִ֑יךָ הִנֵּה֩ שָׁלַ֨חְתִּֽי לְךָ֥ שֹׁ֙חַד֙ כֶּ֣סֶף וְזָהָ֔ב לֵ֣ךְ הָפֵ֗רָה אֶת־ בְּרִֽיתְךָ֙ אֶת־ בַּעְשָׁ֣א מֶֽלֶךְ־ יִשְׂרָאֵ֔ל וְיַעֲלֶ֖ה מֵעָלָֽי׃ (יג) וַיִּשְׁמַ֨ע בֶּן־ הֲדַ֜ד אֶל־ הַמֶּ֣לֶךְ אָסָ֗א וַ֠יִּשְׁלַח אֶת־ שָׂרֵ֨י הַחֲיָלִ֤ים אֲשֶׁר־ לוֹ֙ עַל־ עָרֵ֣י יִשְׂרָאֵ֔ל וַיַּךְ֙ אֶת־ עִיּ֣וֹן וְאֶת־ דָּ֔ן וְאֵ֖ת אָבֵ֣ל בֵּֽית־ מַעֲכָ֑ה וְאֵת֙ כָּל־ כִּנְר֔וֹת עַ֖ל כָּל־ אֶ֥רֶץ נַפְתָּלִֽי׃ (יד) וַֽיְהִי֙ כִּשְׁמֹ֣עַ בַּעְשָׁ֔א וַיֶּחְדַּ֕ל מִבְּנ֖וֹת אֶת־ הָֽרָמָ֑ה וַיֵּ֖שֶׁב בְּתִרְצָֽה׃ (טו) וְהַמֶּ֨לֶךְ אָסָ֜א הִשְׁמִ֤יעַ אֶת־ כָּל־ יְהוּדָה֙ אֵ֣ין נָקִ֔י וַיִּשְׂא֞וּ אֶת־ אַבְנֵ֤י הָֽרָמָה֙ וְאֶת־ עֵצֶ֔יהָ אֲשֶׁ֥ר בָּנָ֖ה בַּעְשָׁ֑א וַיִּ֤בֶן בָּם֙ הַמֶּ֣לֶךְ אָסָ֔א אֶת־ גֶּ֥בַע בִּנְיָמִ֖ן וְאֶת־ הַמִּצְפָּֽה׃ (טז) וְיֶ֣תֶר כָּל־ דִּבְרֵֽי־ אָ֠סָא וְכָל־ גְּב֨וּרָת֜וֹ וְכָל־ אֲשֶׁ֣ר עָשָׂ֗ה וְהֶֽעָרִים֙ אֲשֶׁ֣ר בָּנָ֔ה הֲלֹֽא־ הֵ֣מָּה כְתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֣י יְהוּדָ֑ה רַ֚ק לְעֵ֣ת זִקְנָת֔וֹ חָלָ֖ה אֶת־ רַגְלָֽיו׃ (יז) וַיִּשְׁכַּ֤ב אָסָא֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵר֙ עִם־ אֲבֹתָ֔יו בְּעִ֖יר דָּוִ֣ד אָבִ֑יו וַיִּמְלֹ֛ךְ יְהוֹשָׁפָ֥ט בְּנ֖וֹ תַּחְתָּֽיו׃ (יח) וְנָדָ֣ב בֶּן־ יָרָבְעָ֗ם מָלַךְ֙ עַל־ יִשְׂרָאֵ֔ל בִּשְׁנַ֣ת שְׁתַּ֔יִם לְאָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה וַיִּמְלֹ֥ךְ עַל־ יִשְׂרָאֵ֖ל שְׁנָתָֽיִם׃ (יט) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיֵּ֙לֶךְ֙ בְּדֶ֣רֶךְ אָבִ֔יו וּ֨בְחַטָּאת֔וֹ אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (כ) וַיִּקְשֹׁ֨ר עָלָ֜יו בַּעְשָׁ֤א בֶן־ אֲחִיָּה֙ לְבֵ֣ית יִשָּׂשכָ֔ר וַיַּכֵּ֣הוּ בַעְשָׁ֔א בְּגִבְּת֖וֹן אֲשֶׁ֣ר לַפְּלִשְׁתִּ֑ים וְנָדָב֙ וְכָל־ יִשְׂרָאֵ֔ל צָרִ֖ים עַֽל־ גִּבְּתֽוֹן׃ (כא) וַיְמִתֵ֣הוּ בַעְשָׁ֔א בִּשְׁנַ֣ת שָׁלֹ֔שׁ לְאָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה וַיִּמְלֹ֖ךְ תַּחְתָּֽיו׃ (כב) וַיְהִ֣י כְמָלְכ֗וֹ הִכָּה֙ אֶת־ כָּל־ בֵּ֣ית יָרָבְעָ֔ם לֹֽא־ הִשְׁאִ֧יר כָּל־ נְשָׁמָ֛ה לְיָרָבְעָ֖ם עַד־ הִשְׁמִד֑וֹ כִּדְבַ֣ר יְהוָ֔ה אֲשֶׁ֣ר דִּבֶּ֔ר בְּיַד־ עַבְדּ֖וֹ אֲחִיָּ֥ה הַשִּׁילֹנִֽי׃ (כג) עַל־ חַטֹּ֤אות יָרָבְעָם֙ אֲשֶׁ֣ר חָטָ֔א וַאֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵ֑ל בְּכַעְס֕וֹ אֲשֶׁ֣ר הִכְעִ֔יס אֶת־ יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (כד) וְיֶ֛תֶר דִּבְרֵ֥י נָדָ֖ב וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כה) וּמִלְחָמָ֨ה הָיְתָ֜ה בֵּ֣ין אָסָ֗א וּבֵ֛ין בַּעְשָׁ֥א מֶֽלֶךְ־ יִשְׂרָאֵ֖ל כָּל־ יְמֵיהֶֽם׃ (כו) בִּשְׁנַ֣ת שָׁלֹ֔שׁ לְאָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ בַּעְשָׁ֨א בֶן־ אֲחִיָּ֤ה עַל־ כָּל־ יִשְׂרָאֵל֙ בְּתִרְצָ֔ה עֶשְׂרִ֥ים וְאַרְבַּ֖ע שָׁנָֽה׃ (כז) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיֵּ֙לֶךְ֙ בְּדֶ֣רֶךְ יָרָבְעָ֔ם וּ֨בְחַטָּאת֔וֹ אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ תחילת פרק טז (כח) וַיְהִ֤י דְבַר־ יְהוָה֙ אֶל־ יֵה֣וּא בֶן־ חֲנָ֔נִי עַל־ בַּעְשָׁ֖א לֵאמֹֽר׃ (כט) יַ֗עַן אֲשֶׁ֤ר הֲרִימֹתִ֙יךָ֙ מִן־ הֶ֣עָפָ֔ר וָאֶתֶּנְךָ֣ נָגִ֔יד עַ֖ל עַמִּ֣י יִשְׂרָאֵ֑ל וַתֵּ֣לֶךְ ׀ בְּדֶ֣רֶךְ יָרָבְעָ֗ם וַֽתַּחֲטִא֙ אֶת־ עַמִּ֣י יִשְׂרָאֵ֔ל לְהַכְעִיסֵ֖נִי בְּחַטֹּאתָֽם׃ (ל) הִנְנִ֥י מַבְעִ֛יר אַחֲרֵ֥י בַעְשָׁ֖א וְאַחֲרֵ֣י בֵית֑וֹ וְנָֽתַתִּי֙ אֶת־ בֵּ֣יתְךָ֔ כְּבֵ֖ית יָרָבְעָ֥ם בֶּן־ נְבָֽט׃ (לא) הַמֵּ֤ת לְבַעְשָׁא֙ בָּעִ֔יר יֹֽאכְל֖וּ הַכְּלָבִ֑ים וְהַמֵּ֥ת לוֹ֙ בַּשָּׂדֶ֔ה יֹאכְל֖וּ ע֥וֹף הַשָּׁמָֽיִם׃ (לב) וְיֶ֨תֶר דִּבְרֵ֥י בַעְשָׁ֛א וַאֲשֶׁ֥ר עָשָׂ֖ה וּגְבֽוּרָת֑וֹ הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (לג) וַיִּשְׁכַּ֤ב בַּעְשָׁא֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֖ר בְּתִרְצָ֑ה וַיִּמְלֹ֛ךְ אֵלָ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ (לד) וְגַ֡ם בְּיַד־ יֵה֨וּא בֶן־ חֲנָ֜נִי הַנָּבִ֗יא דְּבַר־ יְהוָ֡ה הָיָה֩ אֶל־ בַּעְשָׁ֨א וְאֶל־ בֵּית֜וֹ וְעַ֥ל כָּל־ הָרָעָ֣ה ׀ אֲשֶׁר־ עָשָׂ֣ה ׀ בְּעֵינֵ֣י יְהוָ֗ה לְהַכְעִיסוֹ֙ בְּמַעֲשֵׂ֣ה יָדָ֔יו לִהְי֖וֹת כְּבֵ֣ית יָרָבְעָ֑ם וְעַ֥ל אֲשֶׁר־ הִכָּ֖ה אֹתֽוֹ׃ (לה) בִּשְׁנַ֨ת עֶשְׂרִ֤ים וָשֵׁשׁ֙ שָׁנָ֔ה לְאָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ אֵלָ֨ה בֶן־ בַּעְשָׁ֧א עַל־ יִשְׂרָאֵ֛ל בְּתִרְצָ֖ה שְׁנָתָֽיִם׃ (לו) וַיִּקְשֹׁ֤ר עָלָיו֙ עַבְדּ֣וֹ זִמְרִ֔י שַׂ֖ר מַחֲצִ֣ית הָרָ֑כֶב וְה֤וּא בְתִרְצָה֙ שֹׁתֶ֣ה שִׁכּ֔וֹר בֵּ֣ית אַרְצָ֔א אֲשֶׁ֥ר עַל־ הַבַּ֖יִת בְּתִרְצָֽה׃ (לז) וַיָּבֹ֤א זִמְרִי֙ וַיַּכֵּ֣הוּ וַיְמִיתֵ֔הוּ בִּשְׁנַת֙ עֶשְׂרִ֣ים וָשֶׁ֔בַע לְאָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה וַיִּמְלֹ֖ךְ תַּחְתָּֽיו׃ (לח) וַיְהִ֨י בְמָלְכ֜וֹ כְּשִׁבְתּ֣וֹ עַל־ כִּסְא֗וֹ הִכָּה֙ אֶת־ כָּל־ בֵּ֣ית בַּעְשָׁ֔א לֹֽא־ הִשְׁאִ֥יר ל֖וֹ מַשְׁתִּ֣ין בְּקִ֑יר וְגֹאֲלָ֖יו וְרֵעֵֽהוּ׃ (לט) וַיַּשְׁמֵ֣ד זִמְרִ֔י אֵ֖ת כָּל־ בֵּ֣ית בַּעְשָׁ֑א כִּדְבַ֤ר יְהוָה֙ אֲשֶׁ֣ר דִּבֶּ֣ר אֶל־ בַּעְשָׁ֔א בְּיַ֖ד יֵה֥וּא הַנָּבִֽיא׃ (מ) אֶ֚ל כָּל־ חַטֹּ֣אות בַּעְשָׁ֔א וְחַטֹּ֖אות אֵלָ֣ה בְנ֑וֹ אֲשֶׁ֣ר (א) בִּשְׁנַת֩ עֶשְׂרִ֨ים וָשֶׁ֜בַע שָׁנָ֗ה לְאָסָא֙ מֶ֣לֶךְ יְהוּדָ֔ה מָלַ֥ךְ זִמְרִ֛י שִׁבְעַ֥ת יָמִ֖ים בְּתִרְצָ֑ה וְהָעָ֣ם חֹנִ֔ים עַֽל־ גִּבְּת֖וֹן אֲשֶׁ֥ר לַפְּלִשְׁתִּֽים׃ (ב) וַיִּשְׁמַ֤ע הָעָם֙ הַחֹנִ֣ים לֵאמֹ֔ר קָשַׁ֣ר זִמְרִ֔י וְגַ֖ם הִכָּ֣ה אֶת־ הַמֶּ֑לֶךְ וַיַּמְלִ֣כוּ כָֽל־ יִ֠שְׂרָאֵל אֶת־ עָמְרִ֨י שַׂר־ צָבָ֧א עַל־ יִשְׂרָאֵ֛ל בַּיּ֥וֹם הַה֖וּא בַּֽמַּחֲנֶֽה׃ (ג) וַיַּעֲלֶ֥ה עָלָ֧יו אֶת־ בֵּֽית־ מֶ֛לֶךְ בָּאֵ֖שׁ וַיָּמֹֽת׃ (ה) עַל־ (חטאתו) חַטֹּאתָיו֙ אֲשֶׁ֣ר חָטָ֔א לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לָלֶ֙כֶת֙ בְּדֶ֣רֶךְ יָרָבְעָ֔ם וּבְחַטָּאתוֹ֙ אֲשֶׁ֣ר עָשָׂ֔ה יִשְׂרָאֵ֖ל לַחֵ֑צִי חֲצִ֨י הָעָ֜ם הָ֠יָה אַחֲרֵ֨י תִבְנִ֤י בֶן־ גִּינַת֙ לְהַמְלִיכ֔וֹ וְהַחֲצִ֖י אַחֲרֵ֥י עָמְרִֽי׃ (ח) וַיֶּחֱזַ֤ק הָעָם֙ אֲשֶׁ֣ר אַחֲרֵ֣י עָמְרִ֔י אֶת־ הָעָ֕ם אֲשֶׁ֥ר אַחֲרֵ֖י תִּבְנִ֣י בֶן־ גִּינַ֑ת וַיָּ֣מָת תִּבְנִ֔י וַיִּמְלֹ֖ךְ עָמְרִֽי׃ (ט) בִּשְׁנַת֩ שְׁלֹשִׁ֨ים וְאַחַ֜ת שָׁנָ֗ה לְאָסָא֙ מֶ֣לֶךְ יְהוּדָ֔ה מָלַ֤ךְ עָמְרִי֙ עַל־ יִשְׂרָאֵ֔ל שְׁתֵּ֥ים עֶשְׂרֵ֖ה שָׁנָ֑ה בְּתִרְצָ֖ה מָלַ֥ךְ שֵׁשׁ־ שָׁנִֽים׃ (י) וַיִּ֜קֶן אֶת־ הָהָ֥ר שֹׁמְר֛וֹן מֵ֥אֶת שֶׁ֖מֶר בְּכִכְּרַ֣יִם כָּ֑סֶף וַיִּ֙בֶן֙ אֶת־ הָהָ֔ר וַיִּקְרָ֗א אֶת־ שֵׁ֤ם הָעִיר֙ אֲשֶׁ֣ר בָּנָ֔ה עַ֣ל שֶׁם־ שֶׁ֔מֶר אֲדֹנֵ֖י הָהָ֥ר שֹׁמְרֽוֹן׃ (יא) וַיַּעֲשֶׂ֥ה עָמְרִ֛י הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיָּ֕רַע מִכֹּ֖ל אֲשֶׁ֥ר לְפָנָֽיו׃ (יב) וַיֵּ֗לֶךְ בְּכָל־ דֶּ֙רֶךְ֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט (ובחטאתיו) וּשְׁתַּ֖יִם שָׁנָֽה׃ (טז) וַיַּ֨עַשׂ אַחְאָ֧ב בֶּן־ עָמְרִ֛י הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה מִכֹּ֖ל אֲשֶׁ֥ר לְפָנָֽיו׃ (יז) וַיְהִי֙ הֲנָקֵ֣ל לֶכְתּ֔וֹ בְּחַטֹּ֖אות יָרָבְעָ֣ם בֶּן־ נְבָ֑ט וַיִּקַּ֨ח הָאֵ֖לֶּה טַ֣ל וּמָטָ֑ר כִּ֖י אִם־ לְפִ֥י דְבָרִֽי׃ (כב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלָ֥יו לֵאמֹֽר׃ (כג) לֵ֣ךְ מִזֶּ֔ה וּפָנִ֥יתָ לְּךָ֖ קֵ֑דְמָה וְנִסְתַּרְתָּ֙ בְּנַ֣חַל כְּרִ֔ית אֲשֶׁ֖ר עַל־ פְּנֵ֥י לְכַלְכְּלֶֽךָ׃ (ל) וַיָּ֣קָם ׀ וַיֵּ֣לֶךְ צָרְפַ֗תָה וַיָּבֹא֙ אֶל־ פֶּ֣תַח הָעִ֔יר וְהִנֵּֽה־ שָׁ֛ם אִשָּׁ֥ה אַלְמָנָ֖ה מְקֹשֶׁ֣שֶׁת עֵצִ֑ים וַיִּקְרָ֤א אֵלֶ֙יהָ֙ וַיֹּאמַ֔ר קְחִי־ נָ֨א לִ֧י מְעַט־ מַ֛יִם בַּכְּלִ֖י וְאֶשְׁתֶּֽה׃ (לא) וַתֵּ֖לֶךְ לָקַ֑חַת וַיִּקְרָ֤א אֵלֶ֙יהָ֙ וַיֹּאמַ֔ר לִֽקְחִי־ נָ֥א לִ֛י פַּת־ לֶ֖חֶם בְּיָדֵֽךְ׃ (לב) וַתֹּ֗אמֶר חַי־ יְהוָ֤ה אֱלֹהֶ֙יךָ֙ אִם־ יֶשׁ־ לִ֣י מָע֔וֹג כִּ֣י אִם־ מְלֹ֤א כַף־ קֶ֙מַח֙ בַּכַּ֔ד וּמְעַט־ שֶׁ֖מֶן בַּצַּפָּ֑חַת וְהִנְנִ֨י מְקֹשֶׁ֜שֶׁת שְׁנַ֣יִם עֵצִ֗ים וּבָ֙אתִי֙ וַעֲשִׂיתִ֙יהוּ֙ לִ֣י וְלִבְנִ֔י וַאֲכַלְנֻ֖הוּ וָמָֽתְנוּ׃ (לג) וַיֹּ֨אמֶר אֵלֶ֤יהָ אֵלִיָּ֙הוּ֙ אַל־ תִּ֣ירְאִ֔י בֹּ֖אִי עֲשִׂ֣י כִדְבָרֵ֑ךְ אַ֣ךְ עֲשִׂי־ לִ֣י מִ֠שָּׁם עֻגָ֨ה קְטַנָּ֤ה בָרִאשֹׁנָה֙ וְהוֹצֵ֣אתְ לִ֔י וְלָ֣ךְ וְלִבְנֵ֔ךְ תַּעֲשִׂ֖י בָּאַחֲרֹנָֽה׃ (לד) כִּ֣י כֹה֩ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֣י יִשְׂרָאֵ֗ל כַּ֤ד הַקֶּ֙מַח֙ לֹ֣א תִכְלָ֔ה וְצַפַּ֥חַת הַשֶּׁ֖מֶן לֹ֣א תֶחְסָ֑ר עַ֠ד י֧וֹם (תתן)־ תֵּת־ יְהוָ֛ה גֶּ֖שֶׁם עַל־ פְּנֵ֥י הָאֲדָמָֽה׃ (לה) וַתֵּ֥לֶךְ וַתַּעֲשֶׂ֖ה כִּדְבַ֣ר אֵלִיָּ֑הוּ וַתֹּ֧אכַל (הוא)־ (והיא) הִֽיא־ וָה֛וּא וּבֵיתָ֖הּ יָמִֽים׃ (לו) כַּ֤ד הַקֶּ֙מַח֙ לֹ֣א כָלָ֔תָה וְצַפַּ֥חַת הַשֶּׁ֖מֶן לֹ֣א חָסֵ֑ר כִּדְבַ֣ר יְהוָ֔ה אֲשֶׁ֥ר דִּבֶּ֖ר בְּיַ֥ד אֵלִיָּֽהוּ׃ (לז) וַיְהִ֗י אַחַר֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה חָלָ֕ה בֶּן־ הָאִשָּׁ֖ה בַּעֲלַ֣ת הַבָּ֑יִת וַיְהִ֤י חָלְיוֹ֙ חָזָ֣ק מְאֹ֔ד עַ֛ד אֲשֶׁ֥ר לֹא־ נֽוֹתְרָה־ בּ֖וֹ נְשָׁמָֽה׃ (לח) וַתֹּ֙אמֶר֙ אֶל־ אֵ֣לִיָּ֔הוּ מַה־ לִּ֥י וָלָ֖ךְ אִ֣ישׁ הָאֱלֹהִ֑ים בָּ֧אתָ אֵלַ֛י לְהַזְכִּ֥יר אֶת־ עֲוֺנִ֖י וּלְהָמִ֥ית אֶת־ בְּנִֽי׃ (לט) וַיֹּ֥אמֶר אֵלֶ֖יהָ תְּנִֽי־ לִ֣י אֶת־ בְּנֵ֑ךְ וַיִּקָּחֵ֣הוּ מֵחֵיקָ֗הּ וַֽיַּעֲלֵ֙הוּ֙ אֶל־ הָעֲלִיָּ֗ה אֲשֶׁר־ הוּא֙ יֹשֵׁ֣ב שָׁ֔ם וַיַּשְׁכִּבֵ֖הוּ עַל־ מִטָּתֽוֹ׃ (מ) וַיִּקְרָ֥א אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר יְהוָ֣ה אֱלֹהָ֔י הֲ֠גַם עַל־ הָאַלְמָנָ֞ה אֲשֶׁר־ אֲנִ֨י מִתְגּוֹרֵ֥ר עִמָּ֛הּ הֲרֵע֖וֹתָ לְהָמִ֥ית אֶת־ בְּנָֽהּ׃ (מא) וַיִּתְמֹדֵ֤ד עַל־ הַיֶּ֙לֶד֙ שָׁלֹ֣שׁ פְּעָמִ֔ים וַיִּקְרָ֥א אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר יְהוָ֣ה אֱלֹהָ֔י תָּ֥שָׁב נָ֛א נֶֽפֶשׁ־ הַיֶּ֥לֶד הַזֶּ֖ה עַל־ קִרְבּֽוֹ׃ (מב) וַיִּשְׁמַ֥ע יְהוָ֖ה בְּק֣וֹל אֵלִיָּ֑הוּ וַתָּ֧שָׁב נֶֽפֶשׁ־ הַיֶּ֛לֶד עַל־ קִרְבּ֖וֹ וַיֶּֽחִי׃ (מג) וַיִּקַּ֨ח אֵלִיָּ֜הוּ אֶת־ הַיֶּ֗לֶד וַיֹּרִדֵ֤הוּ מִן־ הָעֲלִיָּה֙ הַבַּ֔יְתָה וַֽיִּתְּנֵ֖הוּ לְאִמּ֑וֹ וַיֹּ֙אמֶר֙ אֵ֣לִיָּ֔הוּ רְאִ֖י חַ֥י בְּנֵֽךְ׃
