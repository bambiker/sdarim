מלכים א פרק כא לפי החלוקה לסדרים

פרק כא בספר מלכים א (29 פסוקים) נמצא במלכים סדר יז (פסוקים לב–מח) ובמלכים סדר יח (פסוקים א–יב).

מלכים סדר יז

(א) וְהִנֵּ֣ה ׀ נָבִ֣יא אֶחָ֗ד נִגַּשׁ֮ אֶל־ אַחְאָ֣ב מֶֽלֶךְ־ יִשְׂרָאֵל֒ וַיֹּ֗אמֶר כֹּ֚ה אָמַ֣ר יְהוָ֔ה הְֽרָאִ֔יתָ אֵ֛ת כָּל־ הֶהָמ֥וֹן הַגָּד֖וֹל הַזֶּ֑ה הִנְנִ֨י נֹתְנ֤וֹ בְיָֽדְךָ֙ הַיּ֔וֹם וְיָדַעְתָּ֖ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (ב) וַיֹּ֤אמֶר אַחְאָב֙ בְּמִ֔י וַיֹּ֙אמֶר֙ כֹּֽה־ אָמַ֣ר יְהוָ֔ה בְּנַעֲרֵ֖י שָׂרֵ֣י הַמְּדִינ֑וֹת וַיֹּ֛אמֶר מִֽי־ יֶאְסֹ֥ר הַמִּלְחָמָ֖ה וַיֹּ֥אמֶר אָֽתָּה׃ (ג) וַיִּפְקֹ֗ד אֶֽת־ נַעֲרֵי֙ שָׂרֵ֣י הַמְּדִינ֔וֹת וַיִּהְי֕וּ מָאתַ֖יִם שְׁנַ֣יִם וּשְׁלֹשִׁ֑ים וְאַחֲרֵיהֶ֗ם פָּקַ֧ד אֶת־ כָּל־ הָעָ֛ם כָּל־ בְּנֵ֥י יִשְׂרָאֵ֖ל שִׁבְעַ֥ת אֲלָפִֽים׃ (ד) וַיֵּצְא֖וּ בַּֽצָּהֳרָ֑יִם וּבֶן־ הֲדַד֩ שֹׁתֶ֨ה שִׁכּ֜וֹר בַּסֻּכּ֗וֹת ה֧וּא וְהַמְּלָכִ֛ים שְׁלֹשִֽׁים־ וּשְׁנַ֥יִם מֶ֖לֶךְ עֹזֵ֥ר אֹתֽוֹ׃ (ה) וַיֵּצְא֗וּ נַעֲרֵ֛י שָׂרֵ֥י הַמְּדִינ֖וֹת בָּרִֽאשֹׁנָ֑ה וַיִּשְׁלַ֣ח בֶּן־ הֲדַ֗ד וַיַּגִּ֤ידוּ לוֹ֙ לֵאמֹ֔ר אֲנָשִׁ֕ים יָצְא֖וּ מִשֹּׁמְרֽוֹן׃ (ו) וַיֹּ֛אמֶר אִם־ לְשָׁל֥וֹם יָצָ֖אוּ תִּפְשׂ֣וּם חַיִּ֑ים וְאִ֧ם לְמִלְחָמָ֛ה יָצָ֖אוּ חַיִּ֥ים תִּפְשֽׂוּם׃ (ז) וְאֵ֙לֶּה֙ יָצְא֣וּ מִן־ הָעִ֔יר נַעֲרֵ֖י שָׂרֵ֣י הַמְּדִינ֑וֹת וְהַחַ֖יִל אֲשֶׁ֥ר אַחֲרֵיהֶֽם׃ (ח) וַיַּכּוּ֙ אִ֣ישׁ אִישׁ֔וֹ וַיָּנֻ֣סוּ אֲרָ֔ם וַֽיִּרְדְּפֵ֖ם יִשְׂרָאֵ֑ל וַיִּמָּלֵ֗ט בֶּן־ הֲדַד֙ מֶ֣לֶךְ אֲרָ֔ם עַל־ ס֖וּס וּפָרָשִֽׁים׃ (ט) וַיֵּצֵא֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיַּ֥ךְ אֶת־ הַסּ֖וּס וְאֶת־ הָרָ֑כֶב וְהִכָּ֥ה בַאֲרָ֖ם מַכָּ֥ה גְדוֹלָֽה׃ (י) וַיִּגַּ֤שׁ הַנָּבִיא֙ אֶל־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיֹּ֤אמֶר לוֹ֙ לֵ֣ךְ הִתְחַזַּ֔ק וְדַ֥ע וּרְאֵ֖ה אֵ֣ת אֲשֶֽׁר־ תַּעֲשֶׂ֑ה כִּ֚י לִתְשׁוּבַ֣ת הַשָּׁנָ֔ה מֶ֥לֶךְ אֲרָ֖ם עֹלֶ֥ה עָלֶֽיךָ׃ (יא) וְעַבְדֵ֨י מֶֽלֶךְ־ אֲרָ֜ם אָמְר֣וּ אֵלָ֗יו אֱלֹהֵ֤י הָרִים֙ אֱלֹ֣הֵיהֶ֔ם עַל־ כֵּ֖ן חָזְק֣וּ מִמֶּ֑נּוּ וְאוּלָ֗ם נִלָּחֵ֤ם אִתָּם֙ בַּמִּישׁ֔וֹר אִם־ לֹ֥א נֶחֱזַ֖ק מֵהֶֽם׃ (יב) וְאֶת־ הַדָּבָ֥ר הַזֶּ֖ה עֲשֵׂ֑ה הָסֵ֤ר הַמְּלָכִים֙ אִ֣ישׁ מִמְּקֹמ֔וֹ וְשִׂ֥ים פַּח֖וֹת תַּחְתֵּיהֶֽם׃ (יג) וְאַתָּ֣ה תִֽמְנֶה־ לְךָ֣ ׀ חַ֡יִל כַּחַיִל֩ הַנֹּפֵ֨ל מֵאוֹתָ֜ךְ וְס֣וּס כַּסּ֣וּס ׀ וְרֶ֣כֶב כָּרֶ֗כֶב וְנִֽלָּחֲמָ֤ה אוֹתָם֙ בַּמִּישׁ֔וֹר אִם־ לֹ֥א נֶחֱזַ֖ק מֵהֶ֑ם וַיִּשְׁמַ֥ע לְקֹלָ֖ם וַיַּ֥עַשׂ כֵּֽן׃ (יד) וַֽיְהִי֙ לִתְשׁוּבַ֣ת הַשָּׁנָ֔ה וַיִּפְקֹ֥ד בֶּן־ הֲדַ֖ד אֶת־ אֲרָ֑ם וַיַּ֣עַל אֲפֵ֔קָה לַמִּלְחָמָ֖ה עִם־ יִשְׂרָאֵֽל׃ (טו) וּבְנֵ֣י יִשְׂרָאֵ֗ל הָתְפָּקְדוּ֙ וְכָלְכְּל֔וּ וַיֵּלְכ֖וּ לִקְרָאתָ֑ם וַיַּחֲנ֨וּ בְנֵֽי־ יִשְׂרָאֵ֜ל נֶגְדָּ֗ם כִּשְׁנֵי֙ חֲשִׂפֵ֣י עִזִּ֔ים וַאֲרָ֖ם מִלְא֥וּ אֶת־ הָאָֽרֶץ׃ (טז) וַיִּגַּ֞שׁ אִ֣ישׁ הָאֱלֹהִ֗ים וַיֹּאמֶר֮ אֶל־ מֶ֣לֶךְ יִשְׂרָאֵל֒ וַיֹּ֜אמֶר כֹּֽה־ אָמַ֣ר יְהוָ֗ה יַ֠עַן אֲשֶׁ֨ר אָמְר֤וּ אֲרָם֙ אֱלֹהֵ֤י הָרִים֙ יְהוָ֔ה וְלֹֽא־ אֱלֹהֵ֥י עֲמָקִ֖ים ה֑וּא וְ֠נָתַתִּי אֶת־ כָּל־ הֶהָמ֨וֹן הַגָּ֤דוֹל הַזֶּה֙ בְּיָדֶ֔ךָ וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (יז) וַֽיַּחֲנ֧וּ אֵ֦לֶּה נֹ֥כַח אֵ֖לֶּה שִׁבְעַ֣ת יָמִ֑ים וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁבִיעִ֗י וַתִּקְרַב֙ הַמִּלְחָמָ֔ה וַיַּכּ֨וּ בְנֵֽי־ יִשְׂרָאֵ֧ל אֶת־ אֲרָ֛ם מֵאָה־ אֶ֥לֶף רַגְלִ֖י בְּי֥וֹם אֶחָֽד׃ (יח) וַיָּנֻ֨סוּ הַנּוֹתָרִ֥ים ׀ אֲפֵקָה֮ אֶל־ הָעִיר֒ וַתִּפֹּל֙ הַחוֹמָ֔ה עַל־ עֶשְׂרִ֨ים וְשִׁבְעָ֥ה אֶ֛לֶף אִ֖ישׁ הַנּוֹתָרִ֑ים וּבֶן־ הֲדַ֣ד נָ֔ס וַיָּבֹ֥א אֶל־ הָעִ֖יר חֶ֥דֶר בְּחָֽדֶר׃ (יט) וַיֹּאמְר֣וּ אֵלָיו֮ עֲבָדָיו֒ הִנֵּֽה־ נָ֣א שָׁמַ֔עְנוּ כִּ֗י מַלְכֵי֙ בֵּ֣ית יִשְׂרָאֵ֔ל כִּֽי־ מַלְכֵ֥י חֶ֖סֶד הֵ֑ם נָשִׂ֣ימָה נָּא֩ שַׂקִּ֨ים בְּמָתְנֵ֜ינוּ וַחֲבָלִ֣ים בְּרֹאשֵׁ֗נוּ וְנֵצֵא֙ אֶל־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל אוּלַ֖י יְחַיֶּ֥ה אֶת־ נַפְשֶֽׁךָ׃ (כ) וַיַּחְגְּרוּ֩ שַׂקִּ֨ים בְּמָתְנֵיהֶ֜ם וַחֲבָלִ֣ים בְּרָאשֵׁיהֶ֗ם וַיָּבֹ֙אוּ֙ אֶל־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיֹּ֣אמְר֔וּ עַבְדְּךָ֧ בֶן־ הֲדַ֛ד אָמַ֖ר תְּחִֽי־ נָ֣א נַפְשִׁ֑י וַיֹּ֛אמֶר הַעוֹדֶ֥נּוּ חַ֖י אָחִ֥י הֽוּא׃ (כא) וְהָאֲנָשִׁים֩ יְנַחֲשׁ֨וּ וַֽיְמַהֲר֜וּ וַיַּחְלְט֣וּ הֲמִמֶּ֗נּוּ וַיֹּֽאמְרוּ֙ אָחִ֣יךָ בֶן־ הֲדַ֔ד וַיֹּ֖אמֶר בֹּ֣אוּ קָחֻ֑הוּ וַיֵּצֵ֤א אֵלָיו֙ בֶּן־ הֲדַ֔ד וַֽיַּעֲלֵ֖הוּ עַל־ הַמֶּרְכָּבָֽה׃ (כב) וַיֹּ֣אמֶר אֵלָ֡יו הֶעָרִ֣ים אֲשֶׁר־ לָֽקַח־ אָבִי֩ מֵאֵ֨ת אָבִ֜יךָ אָשִׁ֗יב וְ֠חוּצוֹת תָּשִׂ֨ים לְךָ֤ בְדַמֶּ֙שֶׂק֙ כַּאֲשֶׁר־ שָׂ֤ם אָבִי֙ בְּשֹׁ֣מְר֔וֹן וַאֲנִ֖י בַּבְּרִ֣ית אֲשַׁלְּחֶ֑ךָּ וַיִּכְרָת־ ל֥וֹ בְרִ֖ית וַֽיְשַׁלְּחֵֽהוּ׃ (כג) וְאִ֨ישׁ אֶחָ֜ד מִבְּנֵ֣י הַנְּבִיאִ֗ים אָמַ֧ר אֶל־ רֵעֵ֛הוּ בִּדְבַ֥ר יְהוָ֖ה הַכֵּ֣ינִי נָ֑א וַיְמָאֵ֥ן הָאִ֖ישׁ לְהַכֹּתֽוֹ׃ (כד) וַיֹּ֣אמֶר ל֗וֹ יַ֚עַן אֲשֶׁ֤ר לֹֽא־ שָׁמַ֙עְתָּ֙ בְּק֣וֹל יְהוָ֔ה הִנְּךָ֤ הוֹלֵךְ֙ מֵֽאִתִּ֔י וְהִכְּךָ֖ הָאַרְיֵ֑ה וַיֵּ֙לֶךְ֙ מֵֽאֶצְל֔וֹ וַיִּמְצָאֵ֥הוּ הָאַרְיֵ֖ה וַיַּכֵּֽהוּ׃ (כה) וַיִּמְצָא֙ אִ֣ישׁ אַחֵ֔ר וַיֹּ֖אמֶר הַכֵּ֣ינִי נָ֑א וַיַּכֵּ֥הוּ הָאִ֖ישׁ הַכֵּ֥ה וּפָצֹֽעַ׃ (כו) וַיֵּ֙לֶךְ֙ הַנָּבִ֔יא וַיַּעֲמֹ֥ד לַמֶּ֖לֶךְ עַל־ הַדָּ֑רֶךְ וַיִּתְחַפֵּ֥שׂ בָּאֲפֵ֖ר עַל־ עֵינָֽיו׃ (כז) וַיְהִ֤י הַמֶּ֙לֶךְ֙ עֹבֵ֔ר וְה֖וּא צָעַ֣ק אֶל־ הַמֶּ֑לֶךְ וַיֹּ֜אמֶר עַבְדְּךָ֣ ׀ יָצָ֣א בְקֶֽרֶב־ הַמִּלְחָמָ֗ה וְהִנֵּֽה־ אִ֨ישׁ סָ֜ר וַיָּבֵ֧א אֵלַ֣י אִ֗ישׁ וַיֹּ֙אמֶר֙ שְׁמֹר֙ אֶת־ הָאִ֣ישׁ הַזֶּ֔ה אִם־ הִפָּקֵד֙ יִפָּקֵ֔ד וְהָיְתָ֤ה נַפְשְׁךָ֙ תַּ֣חַת נַפְשׁ֔וֹ א֥וֹ כִכַּר־ כֶּ֖סֶף תִּשְׁקֽוֹל׃ (כח) וַיְהִ֣י עַבְדְּךָ֗ עֹשֵׂ֥ה הֵ֛נָּה וָהֵ֖נָּה וְה֣וּא אֵינֶ֑נּוּ וַיֹּ֨אמֶר אֵלָ֧יו מֶֽלֶךְ־ יִשְׂרָאֵ֛ל כֵּ֥ן מִשְׁפָּטֶ֖ךָ אַתָּ֥ה חָרָֽצְתָּ׃ (כט) וַיְמַהֵ֕ר וַיָּ֙סַר֙ אֶת־ הָ֣אֲפֵ֔ר (מעל) מֵעֲלֵ֖י עֵינָ֑יו וַיַּכֵּ֤ר אֹתוֹ֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל כִּ֥י מֵֽהַנְּבִאִ֖ים הֽוּא׃ (ל) וַיֹּ֣אמֶר אֵלָ֗יו כֹּ֚ה אָמַ֣ר יְהוָ֔ה יַ֛עַן שִׁלַּ֥חְתָּ אֶת־ אִישׁ־ חֶרְמִ֖י מִיָּ֑ד וְהָיְתָ֤ה נַפְשְׁךָ֙ תַּ֣חַת נַפְשׁ֔וֹ וְעַמְּךָ֖ תַּ֥חַת עַמּֽוֹ׃ (לא) וַיֵּ֧לֶךְ מֶֽלֶךְ־ יִשְׂרָאֵ֛ל עַל־ בֵּית֖וֹ סַ֣ר וְזָעֵ֑ף וַיָּבֹ֖א שֹׁמְרֽוֹנָה׃ תחילת פרק כא (לב) וַיְהִ֗י אַחַר֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה כֶּ֧רֶם הָיָ֛ה לְנָב֥וֹת הַיִּזְרְעֵאלִ֖י אֲשֶׁ֣ר בְּיִזְרְעֶ֑אל אֵ֚צֶל הֵיכַ֣ל אַחְאָ֔ב מֶ֖לֶךְ שֹׁמְרֽוֹן׃ (לג) מֵאוֹתֽוֹ׃ (כ) וַיֹּ֣אמֶר מֶֽלֶךְ־ יִשְׂרָאֵ֣ל ׀ אֶֽל־ יְהוֹשָׁפָ֡ט ע֣וֹד אִישׁ־ אֶחָ֡ד לִדְרֹשׁ֩ אֶת־ יְהוָ֨ה מֵאֹת֜וֹ וַאֲנִ֣י שְׂנֵאתִ֗יו כִּ֠י לֹֽא־ יִתְנַבֵּ֨א עָלַ֥י טוֹב֙ כִּ֣י אִם־ רָ֔ע מִיכָ֖יְהוּ בֶּן־ יִמְלָ֑ה וַיֹּ֙אמֶר֙ יְה֣וֹשָׁפָ֔ט אַל־ יֹאמַ֥ר הַמֶּ֖לֶךְ כֵּֽן׃ (כא) וַיִּקְרָא֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל אֶל־ סָרִ֖יס אֶחָ֑ד וַיֹּ֕אמֶר מַהֲרָ֖ה מִיכָ֥יְהוּ בֶן־ יִמְלָֽה׃ (כב) וּמֶ֣לֶךְ יִשְׂרָאֵ֡ל וִֽיהוֹשָׁפָ֣ט מֶֽלֶךְ־ יְהוּדָ֡ה יֹשְׁבִים֩ אִ֨ישׁ עַל־ כִּסְא֜וֹ מְלֻבָּשִׁ֤ים בְּגָדִים֙ בְּגֹ֔רֶן פֶּ֖תַח שַׁ֣עַר שֹׁמְר֑וֹן וְכָ֨ל־ הַנְּבִיאִ֔ים מִֽתְנַבְּאִ֖ים לִפְנֵיהֶֽם׃ (כג) וַיַּ֥עַשׂ ל֛וֹ צִדְקִיָּ֥ה בֶֽן־ כְּנַעֲנָ֖ה קַרְנֵ֣י בַרְזֶ֑ל וַיֹּ֙אמֶר֙ כֹּֽה־ אָמַ֣ר יְהוָ֔ה בְּאֵ֛לֶּה תְּנַגַּ֥ח אֶת־ אֲרָ֖ם עַד־ כַּלֹּתָֽם׃ (כד) וְכָל־ הַנְּבִאִ֔ים נִבְּאִ֥ים כֵּ֖ן לֵאמֹ֑ר עֲלֵ֞ה רָמֹ֤ת גִּלְעָד֙ וְהַצְלַ֔ח וְנָתַ֥ן יְהוָ֖ה בְּיַ֥ד הַמֶּֽלֶךְ׃ (כה) וְהַמַּלְאָ֞ךְ אֲשֶׁר־ הָלַ֣ךְ ׀ לִקְרֹ֣א מִיכָ֗יְהוּ דִּבֶּ֤ר אֵלָיו֙ לֵאמֹ֔ר הִנֵּה־ נָ֞א דִּבְרֵ֧י הַנְּבִיאִ֛ים פֶּֽה־ אֶחָ֥ד ט֖וֹב אֶל־ הַמֶּ֑לֶךְ יְהִֽי־ נָ֣א (דבריך) דְבָרְךָ֗ כִּדְבַ֛ר אַחַ֥ד מֵהֶ֖ם וְדִבַּ֥רְתָּ טּֽוֹב׃ (כו) וַיֹּ֖אמֶר מִיכָ֑יְהוּ חַי־ יְהוָ֕ה כִּ֠י אֶת־ אֲשֶׁ֨ר יֹאמַ֧ר יְהוָ֛ה אֵלַ֖י אֹת֥וֹ אֲדַבֵּֽר׃ (כז) וַיָּבוֹא֮ אֶל־ הַמֶּלֶךְ֒ וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֵלָ֗יו מִיכָ֙יְהוּ֙ הֲנֵלֵ֞ךְ אֶל־ רָמֹ֥ת גִּלְעָ֛ד לַמִּלְחָמָ֖ה אִם־ נֶחְדָּ֑ל וַיֹּ֤אמֶר אֵלָיו֙ עֲלֵ֣ה וְהַצְלַ֔ח וְנָתַ֥ן יְהוָ֖ה בְּיַ֥ד הַמֶּֽלֶךְ׃ (כח) וַיֹּ֤אמֶר אֵלָיו֙ הַמֶּ֔לֶךְ עַד־ כַּמֶּ֥ה פְעָמִ֖ים אֲנִ֣י מַשְׁבִּעֶ֑ךָ אֲ֠שֶׁר לֹֽא־ תְדַבֵּ֥ר אֵלַ֛י רַק־ אֱמֶ֖ת בְּשֵׁ֥ם יְהוָֽה׃ (כט) וַיֹּ֗אמֶר רָאִ֤יתִי אֶת־ כָּל־ יִשְׂרָאֵל֙ נְפֹצִ֣ים אֶל־ הֶהָרִ֔ים כַּצֹּ֕אן אֲשֶׁ֥ר אֵין־ לָהֶ֖ם רֹעֶ֑ה וַיֹּ֤אמֶר יְהוָה֙ לֹֽא־ אֲדֹנִ֣ים לָאֵ֔לֶּה יָשׁ֥וּבוּ אִישׁ־ לְבֵית֖וֹ בְּשָׁלֽוֹם׃ (ל) וַיֹּ֥אמֶר מֶֽלֶךְ־ יִשְׂרָאֵ֖ל אֶל־ יְהוֹשָׁפָ֑ט הֲלוֹא֙ אָמַ֣רְתִּי אֵלֶ֔יךָ לֽוֹא־ יִתְנַבֵּ֥א עָלַ֛י ט֖וֹב כִּ֥י אִם־ רָֽע׃ (לא) וַיֹּ֕אמֶר לָכֵ֖ן שְׁמַ֣ע דְּבַר־ יְהוָ֑ה רָאִ֤יתִי אֶת־ יְהוָה֙ יֹשֵׁ֣ב עַל־ כִּסְא֔וֹ וְכָל־ צְבָ֤א הַשָּׁמַ֙יִם֙ עֹמֵ֣ד עָלָ֔יו מִימִינ֖וֹ וּמִשְּׂמֹאלֽוֹ׃ (לב) וַיֹּ֣אמֶר יְהוָ֗ה מִ֤י יְפַתֶּה֙ אֶת־ אַחְאָ֔ב וְיַ֕עַל וְיִפֹּ֖ל בְּרָמֹ֣ת גִּלְעָ֑ד וַיֹּ֤אמֶר זֶה֙ בְּכֹ֔ה וְזֶ֥ה אֹמֵ֖ר בְּכֹֽה׃ (לג) וַיֵּצֵ֣א הָר֗וּחַ וַֽיַּעֲמֹד֙ לִפְנֵ֣י יְהוָ֔ה וַיֹּ֖אמֶר אֲנִ֣י אֲפַתֶּ֑נּוּ וַיֹּ֧אמֶר יְהוָ֛ה אֵלָ֖יו בַּמָּֽה׃ (לד) וַיֹּ֗אמֶר אֵצֵא֙ וְהָיִ֙יתִי֙ ר֣וּחַ שֶׁ֔קֶר בְּפִ֖י כָּל־ נְבִיאָ֑יו וַיֹּ֗אמֶר תְּפַתֶּה֙ וְגַם־ תּוּכָ֔ל צֵ֖א וַעֲשֵׂה־ כֵֽן׃ (לה) וְעַתָּ֗ה הִנֵּ֨ה נָתַ֤ן יְהוָה֙ ר֣וּחַ שֶׁ֔קֶר בְּפִ֖י כָּל־ נְבִיאֶ֣יךָ אֵ֑לֶּה וַֽיהוָ֔ה דִּבֶּ֥ר עָלֶ֖יךָ רָעָֽה׃ (לו) וַיִּגַּשׁ֙ צִדְקִיָּ֣הוּ בֶֽן־ כְּנַעֲנָ֔ה וַיַּכֶּ֥ה אֶת־ מִיכָ֖יְהוּ עַל־ הַלֶּ֑חִי וַיֹּ֕אמֶר אֵי־ זֶ֨ה עָבַ֧ר רֽוּחַ־ יְהוָ֛ה מֵאִתִּ֖י לְדַבֵּ֥ר אוֹתָֽךְ׃ (לז) וַיֹּ֣אמֶר מִיכָ֔יְהוּ הִנְּךָ֥ רֹאֶ֖ה בַּיּ֣וֹם הַה֑וּא אֲשֶׁ֥ר תָּבֹ֛א חֶ֥דֶר בְּחֶ֖דֶר לְהֵחָבֵֽה׃ (לח) וַיֹּ֙אמֶר֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל קַ֚ח אֶת־ מִיכָ֔יְהוּ וַהֲשִׁיבֵ֖הוּ אֶל־ אָמֹ֣ן שַׂר־ הָעִ֑יר וְאֶל־ יוֹאָ֖שׁ בֶּן־ הַמֶּֽלֶךְ׃ (לט) וְאָמַרְתָּ֗ כֹּ֚ה אָמַ֣ר הַמֶּ֔לֶךְ שִׂ֥ימוּ אֶת־ זֶ֖ה בֵּ֣ית הַכֶּ֑לֶא וְהַאֲכִילֻ֨הוּ לֶ֤חֶם לַ֙חַץ֙ וּמַ֣יִם לַ֔חַץ עַ֖ד בֹּאִ֥י בְשָׁלֽוֹם׃ (מ) וַיֹּ֣אמֶר מִיכָ֔יְהוּ אִם־ שׁ֤וֹב תָּשׁוּב֙ בְּשָׁל֔וֹם לֹֽא־ דִבֶּ֥ר יְהוָ֖ה בִּ֑י וַיֹּ֕אמֶר שִׁמְע֖וּ עַמִּ֥ים כֻּלָּֽם׃ (מא) וַיַּ֧עַל מֶֽלֶךְ־ יִשְׂרָאֵ֛ל וְיהוֹשָׁפָ֥ט מֶֽלֶךְ־ יְהוּדָ֖ה רָמֹ֥ת גִּלְעָֽד׃ (מב) וַיֹּאמֶר֩ מֶ֨לֶךְ יִשְׂרָאֵ֜ל אֶל־ יְהוֹשָׁפָ֗ט הִתְחַפֵּשׂ֙ וָבֹ֣א בַמִּלְחָמָ֔ה וְאַתָּ֖ה לְבַ֣שׁ בְּגָדֶ֑יךָ וַיִּתְחַפֵּשׂ֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיָּב֖וֹא בַּמִּלְחָמָֽה׃ (מג) וּמֶ֣לֶךְ אֲרָ֡ם צִוָּ֣ה אֶת־ שָׂרֵי֩ הָרֶ֨כֶב אֲשֶׁר־ ל֜וֹ שְׁלֹשִׁ֤ים וּשְׁנַ֙יִם֙ לֵאמֹ֔ר לֹ֚א תִּלָּ֣חֲמ֔וּ אֶת־ קָטֹ֖ן וְאֶת־ גָּד֑וֹל כִּ֛י אִֽם־ אֶת־ מֶ֥לֶךְ יִשְׂרָאֵ֖ל לְבַדּֽוֹ׃ (מד) וַיְהִ֡י כִּרְאוֹת֩ שָׂרֵ֨י הָרֶ֜כֶב אֶת־ יְהוֹשָׁפָ֗ט וְהֵ֤מָּה אָֽמְרוּ֙ אַ֣ךְ מֶֽלֶךְ־ יִשְׂרָאֵ֣ל ה֔וּא וַיָּסֻ֥רוּ עָלָ֖יו לְהִלָּחֵ֑ם וַיִּזְעַ֖ק יְהוֹשָׁפָֽט׃ (מה) וַיְהִ֗י כִּרְאוֹת֙ שָׂרֵ֣י הָרֶ֔כֶב כִּֽי־ לֹא־ מֶ֥לֶךְ יִשְׂרָאֵ֖ל ה֑וּא וַיָּשׁ֖וּבוּ מֵאַחֲרָֽיו׃ (מו) וְאִ֗ישׁ מָשַׁ֤ךְ בַּקֶּ֙שֶׁת֙ לְתֻמּ֔וֹ וַיַּכֶּה֙ אֶת־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל בֵּ֥ין הַדְּבָקִ֖ים וּבֵ֣ין הַשִּׁרְיָ֑ן וַיֹּ֣אמֶר לְרַכָּב֗וֹ הֲפֹ֥ךְ יָדְךָ֛ וְהוֹצִיאֵ֥נִי מִן־ הַֽמַּחֲנֶ֖ה כִּ֥י הָחֳלֵֽיתִי׃ (מז) וַתַּעֲלֶ֤ה הַמִּלְחָמָה֙ בַּיּ֣וֹם הַה֔וּא וְהַמֶּ֗לֶךְ הָיָ֧ה מָעֳמָ֛ד בַּמֶּרְכָּבָ֖ה נֹ֣כַח אֲרָ֑ם וַיָּ֣מָת בָּעֶ֔רֶב וַיִּ֥צֶק דַּֽם־ הַמַּכָּ֖ה אֶל־ חֵ֥יק הָרָֽכֶב׃ (מח) וַיַּעֲבֹ֤ר הָרִנָּה֙ בַּֽמַּחֲנֶ֔ה כְּבֹ֥א הַשֶּׁ֖מֶשׁ לֵאמֹ֑ר אִ֥ישׁ אֶל־ עִיר֖וֹ וְאִ֥ישׁ אֶל־ אַרְצֽוֹ׃ (מט) וַיָּ֣מָת הַמֶּ֔לֶךְ וַיָּב֖וֹא שֹׁמְר֑וֹן וַיִּקְבְּר֥וּ אֶת־ הַמֶּ֖לֶךְ בְּשֹׁמְרֽוֹן׃ (נ) וַיִּשְׁטֹ֨ף אֶת־ הָרֶ֜כֶב עַ֣ל ׀ בְּרֵכַ֣ת שֹׁמְר֗וֹן וַיָּלֹ֤קּוּ הַכְּלָבִים֙ אֶת־ דָּמ֔וֹ וְהַזֹּנ֖וֹת רָחָ֑צוּ כִּדְבַ֥ר יְהוָ֖ה אֲשֶׁ֥ר דִּבֵּֽר׃ (נא) וְיֶתֶר֩ דִּבְרֵ֨י אַחְאָ֜ב וְכָל־ אֲשֶׁ֣ר עָשָׂ֗ה וּבֵ֤ית הַשֵּׁן֙ אֲשֶׁ֣ר בָּנָ֔ה וְכָל־ הֶעָרִ֖ים אֲשֶׁ֣ר בָּנָ֑ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (נב) וַיִּשְׁכַּ֥ב אַחְאָ֖ב עִם־ אֲבֹתָ֑יו וַיִּמְלֹ֛ךְ אֲחַזְיָ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ (נג) וִיהֽוֹשָׁפָט֙ בֶּן־ אָסָ֔א מָלַ֖ךְ עַל־ יְהוּדָ֑ה בִּשְׁנַ֣ת אַרְבַּ֔ע לְאַחְאָ֖ב מֶ֥לֶךְ יִשְׂרָאֵֽל׃ (נד) יְהוֹשָׁפָ֗ט בֶּן־ שְׁלֹשִׁ֨ים וְחָמֵ֤שׁ שָׁנָה֙ בְּמָלְכ֔וֹ וְעֶשְׂרִ֤ים וְחָמֵשׁ֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ עֲזוּבָ֖ה בַּת־ שִׁלְחִֽי׃
