מלכים א פרק י לפי החלוקה לסדרים

פרק י בספר מלכים א (29 פסוקים) נמצא במלכים סדר ח (פסוקים לח–מה) ובמלכים סדר ט (פסוקים א–כא).

מלכים סדר ח

(א) לְהַטּ֥וֹת לְבָבֵ֖נוּ אֵלָ֑יו לָלֶ֣כֶת בְּכָל־ דְּרָכָ֗יו וְלִשְׁמֹ֨ר מִצְוֺתָ֤יו וְחֻקָּיו֙ וּמִשְׁפָּטָ֔יו אֲשֶׁ֥ר צִוָּ֖ה אֶת־ אֲבֹתֵֽינוּ׃ (ב) וְיִֽהְי֨וּ דְבָרַ֜י אֵ֗לֶּה אֲשֶׁ֤ר הִתְחַנַּ֙נְתִּי֙ לִפְנֵ֣י יְהוָ֔ה קְרֹבִ֛ים אֶל־ יְהוָ֥ה אֱלֹהֵ֖ינוּ יוֹמָ֣ם וָלָ֑יְלָה לַעֲשׂ֣וֹת ׀ מִשְׁפַּ֣ט עַבְדּ֗וֹ וּמִשְׁפַּ֛ט עַמּ֥וֹ יִשְׂרָאֵ֖ל דְּבַר־ י֥וֹם בְּיוֹמֽוֹ׃ (ג) לְמַ֗עַן דַּ֚עַת כָּל־ עַמֵּ֣י הָאָ֔רֶץ כִּ֥י יְהוָ֖ה ה֣וּא הָאֱלֹהִ֑ים אֵ֖ין עֽוֹד׃ (ד) וְהָיָ֤ה לְבַבְכֶם֙ שָׁלֵ֔ם עִ֖ם יְהוָ֣ה אֱלֹהֵ֑ינוּ לָלֶ֧כֶת בְּחֻקָּ֛יו וְלִשְׁמֹ֥ר מִצְוֺתָ֖יו כַּיּ֥וֹם הַזֶּֽה׃ (ה) וְֽהַמֶּ֔לֶךְ וְכָל־ יִשְׂרָאֵ֖ל עִמּ֑וֹ זֹבְחִ֥ים זֶ֖בַח לִפְנֵ֥י יְהוָֽה׃ (ו) וַיִּזְבַּ֣ח שְׁלֹמֹ֗ה אֵ֣ת זֶ֣בַח הַשְּׁלָמִים֮ אֲשֶׁ֣ר זָבַ֣ח לַיהוָה֒ בָּקָ֗ר עֶשְׂרִ֤ים וּשְׁנַ֙יִם֙ אֶ֔לֶף וְצֹ֕אן מֵאָ֥ה וְעֶשְׂרִ֖ים אָ֑לֶף וַֽיַּחְנְכוּ֙ אֶת־ בֵּ֣ית יְהוָ֔ה הַמֶּ֖לֶךְ וְכָל־ בְּנֵ֥י יִשְׂרָאֵֽל׃ (ז) בַּיּ֣וֹם הַה֗וּא קִדַּ֨שׁ הַמֶּ֜לֶךְ אֶת־ תּ֣וֹךְ הֶחָצֵ֗ר אֲשֶׁר֙ לִפְנֵ֣י בֵית־ יְהוָ֔ה כִּי־ עָ֣שָׂה שָׁ֗ם אֶת־ הָֽעֹלָה֙ וְאֶת־ הַמִּנְחָ֔ה וְאֵ֖ת חֶלְבֵ֣י הַשְּׁלָמִ֑ים כִּֽי־ מִזְבַּ֤ח הַנְּחֹ֙שֶׁת֙ אֲשֶׁ֣ר לִפְנֵ֣י יְהוָ֔ה קָטֹ֗ן מֵֽהָכִיל֙ אֶת־ הָעֹלָ֣ה וְאֶת־ הַמִּנְחָ֔ה וְאֵ֖ת חֶלְבֵ֥י הַשְּׁלָמִֽים׃ (ח) וַיַּ֣עַשׂ שְׁלֹמֹ֣ה בָֽעֵת־ הַהִ֣יא ׀ אֶת־ הֶחָ֡ג וְכָל־ יִשְׂרָאֵ֣ל עִמּוֹ֩ קָהָ֨ל גָּד֜וֹל מִלְּב֥וֹא חֲמָ֣ת ׀ עַד־ נַ֣חַל מִצְרַ֗יִם לִפְנֵי֙ יְהוָ֣ה אֱלֹהֵ֔ינוּ שִׁבְעַ֥ת יָמִ֖ים וְשִׁבְעַ֣ת יָמִ֑ים אַרְבָּעָ֥ה עָשָׂ֖ר יֽוֹם׃ (ט) בַּיּ֤וֹם הַשְּׁמִינִי֙ שִׁלַּ֣ח אֶת־ הָעָ֔ם וַֽיְבָרֲכ֖וּ אֶת־ הַמֶּ֑לֶךְ וַיֵּלְכ֣וּ לְאָהֳלֵיהֶ֗ם שְׂמֵחִים֙ וְט֣וֹבֵי לֵ֔ב עַ֣ל כָּל־ הַטּוֹבָ֗ה אֲשֶׁ֨ר עָשָׂ֤ה יְהוָה֙ לְדָוִ֣ד עַבְדּ֔וֹ וּלְיִשְׂרָאֵ֖ל עַמּֽוֹ׃ (י) וַיְהִי֙ כְּכַלּ֣וֹת שְׁלֹמֹ֔ה לִבְנ֥וֹת אֶת־ בֵּית־ יְהוָ֖ה וְאֶת־ בֵּ֣ית הַמֶּ֑לֶךְ וְאֵת֙ כָּל־ חֵ֣שֶׁק שְׁלֹמֹ֔ה אֲשֶׁ֥ר חָפֵ֖ץ לַעֲשֽׂוֹת׃ (יא) וַיֵּרָ֧א יְהוָ֛ה אֶל־ שְׁלֹמֹ֖ה שֵׁנִ֑ית כַּאֲשֶׁ֛ר נִרְאָ֥ה אֵלָ֖יו בְּגִבְעֽוֹן׃ (יב) וַיֹּ֨אמֶר יְהוָ֜ה אֵלָ֗יו שָׁ֠מַעְתִּי אֶת־ תְּפִלָּתְךָ֣ וְאֶת־ תְּחִנָּתְךָ֮ אֲשֶׁ֣ר הִתְחַנַּ֣נְתָּה לְפָנַי֒ הִקְדַּ֗שְׁתִּי אֶת־ הַבַּ֤יִת הַזֶּה֙ אֲשֶׁ֣ר בָּנִ֔תָה לָשֽׂוּם־ שְׁמִ֥י שָׁ֖ם עַד־ עוֹלָ֑ם וְהָי֨וּ עֵינַ֧י וְלִבִּ֛י שָׁ֖ם כָּל־ הַיָּמִֽים׃ (יג) וְאַתָּ֞ה אִם־ תֵּלֵ֣ךְ לְפָנַ֗י כַּאֲשֶׁ֨ר הָלַ֜ךְ דָּוִ֤ד אָבִ֙יךָ֙ בְּתָם־ לֵבָ֣ב וּבְיֹ֔שֶׁר לַעֲשׂ֕וֹת כְּכֹ֖ל אֲשֶׁ֣ר צִוִּיתִ֑יךָ חֻקַּ֥י וּמִשְׁפָּטַ֖י תִּשְׁמֹֽר׃ (יד) וַהֲקִ֨מֹתִ֜י אֶת־ כִּסֵּ֧א מַֽמְלַכְתְּךָ֛ עַל־ יִשְׂרָאֵ֖ל לְעֹלָ֑ם כַּאֲשֶׁ֣ר דִּבַּ֗רְתִּי עַל־ דָּוִ֤ד אָבִ֙יךָ֙ לֵאמֹ֔ר לֹֽא־ יִכָּרֵ֤ת לְךָ֙ אִ֔ישׁ מֵעַ֖ל כִּסֵּ֥א יִשְׂרָאֵֽל׃ (טו) אִם־ שׁ֨וֹב תְּשֻׁב֜וּן אַתֶּ֤ם וּבְנֵיכֶם֙ מֵֽאַחֲרַ֔י וְלֹ֤א תִשְׁמְרוּ֙ מִצְוֺתַ֣י חֻקֹּתַ֔י אֲשֶׁ֥ר נָתַ֖תִּי לִפְנֵיכֶ֑ם וַהֲלַכְתֶּ֗ם וַעֲבַדְתֶּם֙ אֱלֹהִ֣ים אֲחֵרִ֔ים וְהִשְׁתַּחֲוִיתֶ֖ם לָהֶֽם׃ (טז) וְהִכְרַתִּ֣י אֶת־ יִשְׂרָאֵ֗ל מֵעַ֨ל פְּנֵ֤י הָאֲדָמָה֙ אֲשֶׁ֣ר נָתַ֣תִּי לָהֶ֔ם וְאֶת־ הַבַּ֙יִת֙ אֲשֶׁ֣ר הִקְדַּ֣שְׁתִּי לִשְׁמִ֔י אֲשַׁלַּ֖ח מֵעַ֣ל פָּנָ֑י וְהָיָ֧ה יִשְׂרָאֵ֛ל לְמָשָׁ֥ל וְלִשְׁנִינָ֖ה בְּכָל־ הָעַמִּֽים׃ (יז) וְהַבַּ֤יִת הַזֶּה֙ יִהְיֶ֣ה עֶלְי֔וֹן כָּל־ עֹבֵ֥ר עָלָ֖יו יִשֹּׁ֣ם וְשָׁרָ֑ק וְאָמְר֗וּ עַל־ מֶ֨ה עָשָׂ֤ה יְהוָה֙ כָּ֔כָה לָאָ֥רֶץ הַזֹּ֖את וְלַבַּ֥יִת הַזֶּֽה׃ (יח) וְאָמְר֗וּ עַל֩ אֲשֶׁ֨ר עָזְב֜וּ אֶת־ יְהוָ֣ה אֱלֹהֵיהֶ֗ם אֲשֶׁ֨ר הוֹצִ֣יא אֶת־ אֲבֹתָם֮ מֵאֶ֣רֶץ מִצְרַיִם֒ וַֽיַּחֲזִ֙קוּ֙ בֵּאלֹהִ֣ים אֲחֵרִ֔ים (וישתחו) וַיִּשְׁתַּחֲו֥וּ לָהֶ֖ם וַיַּעַבְדֻ֑ם עַל־ כֵּ֗ן הֵבִ֤יא יְהוָה֙ עֲלֵיהֶ֔ם אֵ֥ת כָּל־ הָרָעָ֖ה הַזֹּֽאת׃ (יט) וַיְהִ֗י מִקְצֵה֙ עֶשְׂרִ֣ים שָׁנָ֔ה אֲשֶׁר־ בָּנָ֥ה שְׁלֹמֹ֖ה אֶת־ שְׁנֵ֣י הַבָּתִּ֑ים אֶת־ בֵּ֥ית יְהוָ֖ה וְאֶת־ בֵּ֥ית הַמֶּֽלֶךְ׃ (כ) חִירָ֣ם מֶֽלֶךְ־ צֹ֠ר נִשָּׂ֨א אֶת־ שְׁלֹמֹ֜ה בַּעֲצֵי֩ אֲרָזִ֨ים וּבַעֲצֵ֧י בְרוֹשִׁ֛ים וּבַזָּהָ֖ב לְכָל־ חֶפְצ֑וֹ אָ֡ז יִתֵּן֩ הַמֶּ֨לֶךְ שְׁלֹמֹ֤ה לְחִירָם֙ עֶשְׂרִ֣ים עִ֔יר בְּאֶ֖רֶץ הַגָּלִֽיל׃ (כא) וַיֵּצֵ֤א חִירָם֙ מִצֹּ֔ר לִרְאוֹת֙ אֶת־ הֶ֣עָרִ֔ים אֲשֶׁ֥ר נָתַן־ ל֖וֹ שְׁלֹמֹ֑ה וְלֹ֥א יָשְׁר֖וּ בְּעֵינָֽיו׃ (כב) וַיֹּ֕אמֶר מָ֚ה הֶעָרִ֣ים הָאֵ֔לֶּה אֲשֶׁר־ נָתַ֥תָּה לִּ֖י אָחִ֑י וַיִּקְרָ֤א לָהֶם֙ אֶ֣רֶץ כָּב֔וּל עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כג) וַיִּשְׁלַ֥ח חִירָ֖ם לַמֶּ֑לֶךְ מֵאָ֥ה וְעֶשְׂרִ֖ים כִּכַּ֥ר זָהָֽב׃ (כד) וְזֶ֨ה דְבַר־ הַמַּ֜ס אֲשֶֽׁר־ הֶעֱלָ֣ה ׀ הַמֶּ֣לֶךְ שְׁלֹמֹ֗ה לִבְנוֹת֩ אֶת־ בֵּ֨ית יְהוָ֤ה וְאֶת־ בֵּיתוֹ֙ וְאֶת־ הַמִּלּ֔וֹא וְאֵ֖ת חוֹמַ֣ת יְרוּשָׁלִָ֑ם וְאֶת־ חָצֹ֥ר וְאֶת־ מְגִדּ֖וֹ וְאֶת־ גָּֽזֶר׃ (כה) פַּרְעֹ֨ה מֶֽלֶךְ־ מִצְרַ֜יִם עָלָ֗ה וַיִּלְכֹּ֤ד אֶת־ גֶּ֙זֶר֙ וַיִּשְׂרְפָ֣הּ בָּאֵ֔שׁ וְאֶת־ הַֽכְּנַעֲנִ֛י הַיֹּשֵׁ֥ב בָּעִ֖יר הָרָ֑ג וַֽיִּתְּנָהּ֙ שִׁלֻּחִ֔ים לְבִתּ֖וֹ אֵ֥שֶׁת שְׁלֹמֹֽה׃ (כו) וַיִּ֤בֶן שְׁלֹמֹה֙ אֶת־ גָּ֔זֶר וְאֶת־ בֵּ֥ית חֹרֹ֖ן תַּחְתּֽוֹן׃ (כז) וְאֶֽת־ בַּעֲלָ֛ת וְאֶת־ (תמר) תַּדְמֹ֥ר בַּמִּדְבָּ֖ר בָּאָֽרֶץ׃ (כח) וְאֵ֨ת כָּל־ עָרֵ֤י הַֽמִּסְכְּנוֹת֙ אֲשֶׁ֣ר הָי֣וּ לִשְׁלֹמֹ֔ה וְאֵת֙ עָרֵ֣י הָרֶ֔כֶב וְאֵ֖ת עָרֵ֣י הַפָּרָשִׁ֑ים וְאֵ֣ת ׀ חֵ֣שֶׁק שְׁלֹמֹ֗ה אֲשֶׁ֤ר חָשַׁק֙ לִבְנ֤וֹת בִּירוּשָׁלִַ֙ם֙ וּבַלְּבָנ֔וֹן וּבְכֹ֖ל אֶ֥רֶץ מֶמְשַׁלְתּֽוֹ׃ (כט) כָּל־ הָ֠עָם הַנּוֹתָ֨ר מִן־ הָאֱמֹרִ֜י הַחִתִּ֤י הַפְּרִזִּי֙ הַחִוִּ֣י וְהַיְבוּסִ֔י אֲשֶׁ֛ר לֹֽא־ מִבְּנֵ֥י יִשְׂרָאֵ֖ל הֵֽמָּה׃ (ל) בְּנֵיהֶ֗ם אֲשֶׁ֨ר נֹתְר֤וּ אַחֲרֵיהֶם֙ בָּאָ֔רֶץ אֲשֶׁ֧ר לֹֽא־ יָכְל֛וּ בְּנֵ֥י יִשְׂרָאֵ֖ל לְהַֽחֲרִימָ֑ם וַיַּעֲלֵ֤ם שְׁלֹמֹה֙ לְמַס־ עֹבֵ֔ד עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (לא) וּמִבְּנֵי֙ יִשְׂרָאֵ֔ל לֹֽא־ נָתַ֥ן שְׁלֹמֹ֖ה עָ֑בֶד כִּי־ הֵ֞ם אַנְשֵׁ֣י הַמִּלְחָמָ֗ה וַעֲבָדָיו֙ וְשָׂרָ֣יו וְשָׁלִשָׁ֔יו וְשָׂרֵ֥י רִכְבּ֖וֹ וּפָרָשָֽׁיו׃ (לב) אֵ֣לֶּה ׀ שָׂרֵ֣י הַנִּצָּבִ֗ים אֲשֶׁ֤ר עַל־ הַמְּלָאכָה֙ לִשְׁלֹמֹ֔ה חֲמִשִּׁ֖ים וַחֲמֵ֣שׁ מֵא֑וֹת הָרֹדִ֣ים בָּעָ֔ם הָעֹשִׂ֖ים בַּמְּלָאכָֽה׃ (לג) אַ֣ךְ בַּת־ פַּרְעֹ֗ה עָֽלְתָה֙ מֵעִ֣יר דָּוִ֔ד אֶל־ בֵּיתָ֖הּ אֲשֶׁ֣ר בָּֽנָה־ לָ֑הּ אָ֖ז בָּנָ֥ה אֶת־ הַמִּלּֽוֹא׃ (לד) וְהֶעֱלָ֣ה שְׁלֹמֹ֡ה שָׁלֹשׁ֩ פְּעָמִ֨ים בַּשָּׁנָ֜ה עֹל֣וֹת וּשְׁלָמִ֗ים עַל־ הַמִּזְבֵּ֙חַ֙ אֲשֶׁ֣ר בָּנָ֣ה לַיהוָ֔ה וְהַקְטֵ֣יר אִתּ֔וֹ אֲשֶׁ֖ר לִפְנֵ֣י יְהוָ֑ה וְשִׁלַּ֖ם אֶת־ הַבָּֽיִת׃ (לה) וָאֳנִ֡י עָשָׂה֩ הַמֶּ֨לֶךְ שְׁלֹמֹ֜ה בְּעֶצְיֽוֹן־ גֶּ֨בֶר אֲשֶׁ֧ר אֶת־ אֵל֛וֹת עַל־ שְׂפַ֥ת יַם־ ס֖וּף בְּאֶ֥רֶץ אֱדֽוֹם׃ (לו) וַיִּשְׁלַ֨ח חִירָ֤ם בָּֽאֳנִי֙ אֶת־ עֲבָדָ֔יו אַנְשֵׁ֣י אֳנִיּ֔וֹת יֹדְעֵ֖י הַיָּ֑ם עִ֖ם עַבְדֵ֥י שְׁלֹמֹֽה׃ (לז) וַיָּבֹ֣אוּ אוֹפִ֔ירָה וַיִּקְח֤וּ מִשָּׁם֙ זָהָ֔ב אַרְבַּע־ מֵא֥וֹת וְעֶשְׂרִ֖ים כִּכָּ֑ר וַיָּבִ֖אוּ אֶל־ הַמֶּ֥לֶךְ שְׁלֹמֹֽה׃ תחילת פרק י (לח) וּמַֽלְכַּת־ שְׁבָ֗א שֹׁמַ֛עַת אֶת־ שֵׁ֥מַע שְׁלֹמֹ֖ה לְשֵׁ֣ם יְהוָ֑ה וַתָּבֹ֥א לְנַסֹּת֖וֹ בְּחִידֽוֹת׃ (לט) וַתָּבֹ֣א יְרוּשָׁלְַ֗מָה בְּחַיִל֮ כָּבֵ֣ד מְאֹד֒ גְּ֠מַלִּים נֹשְׂאִ֨ים בְּשָׂמִ֧ים וְזָהָ֛ב רַב־ מְאֹ֖ד וְאֶ֣בֶן יְקָרָ֑ה וַתָּבֹא֙ אֶל־ שְׁלֹמֹ֔ה וַתְּדַבֵּ֣ר אֵלָ֔יו אֵ֛ת כָּל־ אֲשֶׁ֥ר הָיָ֖ה עִם־ לְבָבָֽהּ׃ (מ) וַיַּגֶּד־ לָ֥הּ שְׁלֹמֹ֖ה אֶת־ כָּל־ דְּבָרֶ֑יהָ לֹֽא־ הָיָ֤ה דָּבָר֙ נֶעְלָ֣ם מִן־ הַמֶּ֔לֶךְ אֲשֶׁ֧ר לֹ֦א הִגִּ֖יד לָֽהּ׃ (מא) וַתֵּ֙רֶא֙ מַֽלְכַּת־ שְׁבָ֔א אֵ֖ת כָּל־ חָכְמַ֣ת שְׁלֹמֹ֑ה וְהַבַּ֖יִת אֲשֶׁ֥ר בָּנָֽה׃ (מב) וּמַאֲכַ֣ל שֻׁלְחָנ֡וֹ וּמוֹשַׁ֣ב עֲבָדָיו֩ וּמַעֲמַ֨ד (משרתו) מְשָׁרְתָ֜יו וּמַלְבֻּֽשֵׁיהֶם֙ וּמַשְׁקָ֔יו וְעֹ֣לָת֔וֹ אֲשֶׁ֥ר יַעֲלֶ֖ה בֵּ֣ית יְהוָ֑ה וְלֹא־ הָ֥יָה בָ֛הּ ע֖וֹד רֽוּחַ׃ (מג) וַתֹּ֙אמֶר֙ אֶל־ הַמֶּ֔לֶךְ אֱמֶת֙ הָיָ֣ה הַדָּבָ֔ר אֲשֶׁ֥ר שָׁמַ֖עְתִּי בְּאַרְצִ֑י עַל־ דְּבָרֶ֖יךָ וְעַל־ חָכְמָתֶֽךָ׃ (מד) וְלֹֽא־ הֶאֱמַ֣נְתִּי לַדְּבָרִ֗ים עַ֤ד אֲשֶׁר־ בָּ֙אתִי֙ וַתִּרְאֶ֣ינָה עֵינַ֔י וְהִנֵּ֥ה לֹֽא־ הֻגַּד־ לִ֖י הַחֵ֑צִי הוֹסַ֤פְתָּ חָכְמָה֙ וָט֔וֹב אֶל־ הַשְּׁמוּעָ֖ה אֲשֶׁ֥ר שָׁמָֽעְתִּי׃ (מה) אַשְׁרֵ֣י אֲנָשֶׁ֔יךָ אַשְׁרֵ֖י עֲבָדֶ֣יךָ אֵ֑לֶּה הָֽעֹמְדִ֤ים לְפָנֶ֙יךָ֙ תָּמִ֔יד הַשֹּׁמְעִ֖ים אֶת־ חָכְמָתֶֽךָ׃

מלכים סדר ט

(א) יְהִ֨י יְהוָ֤ה אֱלֹהֶ֙יךָ֙ בָּר֔וּךְ אֲשֶׁר֙ חָפֵ֣ץ בְּךָ֔ לְתִתְּךָ֖ עַל־ כִּסֵּ֣א יִשְׂרָאֵ֑ל בְּאַהֲבַ֨ת יְהוָ֤ה אֶת־ יִשְׂרָאֵל֙ לְעֹלָ֔ם וַיְשִֽׂימְךָ֣ לְמֶ֔לֶךְ לַעֲשׂ֥וֹת מִשְׁפָּ֖ט וּצְדָקָֽה׃ (ב) וַתִּתֵּ֨ן לַמֶּ֜לֶךְ מֵאָ֥ה וְעֶשְׂרִ֣ים ׀ כִּכַּ֣ר זָהָ֗ב וּבְשָׂמִ֛ים הַרְבֵּ֥ה מְאֹ֖ד וְאֶ֣בֶן יְקָרָ֑ה לֹא־ בָא֩ כַבֹּ֨שֶׂם הַה֥וּא עוֹד֙ לָרֹ֔ב אֲשֶׁר־ נָתְנָ֥ה מַֽלְכַּת־ שְׁבָ֖א לַמֶּ֥לֶךְ שְׁלֹמֹֽה׃ (ג) וְגַם֙ אֳנִ֣י חִירָ֔ם אֲשֶׁר־ נָשָׂ֥א זָהָ֖ב מֵאוֹפִ֑יר הֵבִ֨יא מֵאֹפִ֜יר עֲצֵ֧י אַלְמֻגִּ֛ים הַרְבֵּ֥ה מְאֹ֖ד וְאֶ֥בֶן יְקָרָֽה׃ (ד) וַיַּ֣עַשׂ הַ֠מֶּלֶךְ אֶת־ עֲצֵ֨י הָאַלְמֻגִּ֜ים מִסְעָ֤ד לְבֵית־ יְהוָה֙ וּלְבֵ֣ית הַמֶּ֔לֶךְ וְכִנֹּר֥וֹת וּנְבָלִ֖ים לַשָּׁרִ֑ים לֹ֣א בָֽא־ כֵ֞ן עֲצֵ֤י אַלְמֻגִּים֙ וְלֹ֣א נִרְאָ֔ה עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (ה) וְהַמֶּ֨לֶךְ שְׁלֹמֹ֜ה נָתַ֣ן לְמַֽלְכַּת־ שְׁבָ֗א אֶת־ כָּל־ חֶפְצָהּ֙ אֲשֶׁ֣ר שָׁאָ֔לָה מִלְּבַד֙ אֲשֶׁ֣ר נָֽתַן־ לָ֔הּ כְּיַ֖ד הַמֶּ֣לֶךְ שְׁלֹמֹ֑ה וַתֵּ֛פֶן וַתֵּ֥לֶךְ לְאַרְצָ֖הּ הִ֥יא וַעֲבָדֶֽיהָ׃ (ו) וַֽיְהִי֙ מִשְׁקַ֣ל הַזָּהָ֔ב אֲשֶׁר־ בָּ֥א לִשְׁלֹמֹ֖ה בְּשָׁנָ֣ה אֶחָ֑ת שֵׁ֥שׁ מֵא֛וֹת שִׁשִּׁ֥ים וָשֵׁ֖שׁ כִּכַּ֥ר זָהָֽב׃ (ז) לְבַד֙ מֵאַנְשֵׁ֣י הַתָּרִ֔ים וּמִסְחַ֖ר הָרֹכְלִ֑ים וְכָל־ מַלְכֵ֥י הָעֶ֖רֶב וּפַח֥וֹת הָאָֽרֶץ׃ (ח) וַיַּ֨עַשׂ הַמֶּ֧לֶךְ שְׁלֹמֹ֛ה מָאתַ֥יִם צִנָּ֖ה זָהָ֣ב שָׁח֑וּט שֵׁשׁ־ מֵא֣וֹת זָהָ֔ב יַעֲלֶ֖ה עַל־ הַצִּנָּ֥ה הָאֶחָֽת׃ (ט) וּשְׁלֹשׁ־ מֵא֤וֹת מָֽגִנִּים֙ זָהָ֣ב שָׁח֔וּט שְׁלֹ֤שֶׁת מָנִים֙ זָהָ֔ב יַעֲלֶ֖ה עַל־ הַמָּגֵ֣ן הָאֶחָ֑ת וַיִּתְּנֵ֣ם הַמֶּ֔לֶךְ בֵּ֖ית יַ֥עַר הַלְּבָנֽוֹן׃ (י) וַיַּ֧עַשׂ הַמֶּ֛לֶךְ כִּסֵּא־ שֵׁ֖ן גָּד֑וֹל וַיְצַפֵּ֖הוּ זָהָ֥ב מוּפָֽז׃ (יא) שֵׁ֧שׁ מַעֲל֣וֹת לַכִּסֵּ֗ה וְרֹאשׁ־ עָגֹ֤ל לַכִּסֵּה֙ מֵאַֽחֲרָ֔יו וְיָדֹ֛ת מִזֶּ֥ה וּמִזֶּ֖ה אֶל־ מְק֣וֹם הַשָּׁ֑בֶת וּשְׁנַ֣יִם אֲרָי֔וֹת עֹמְדִ֖ים אֵ֥צֶל הַיָּדֽוֹת׃ (יב) וּשְׁנֵ֧ים עָשָׂ֣ר אֲרָיִ֗ים עֹמְדִ֥ים שָׁ֛ם עַל־ שֵׁ֥שׁ הַֽמַּעֲל֖וֹת מִזֶּ֣ה וּמִזֶּ֑ה לֹֽא־ נַעֲשָׂ֥ה כֵ֖ן לְכָל־ מַמְלָכֽוֹת׃ (יג) וְ֠כֹל כְּלֵ֞י מַשְׁקֵ֨ה הַמֶּ֤לֶךְ שְׁלֹמֹה֙ זָהָ֔ב וְכֹ֗ל כְּלֵ֛י בֵּֽית־ יַ֥עַר הַלְּבָנ֖וֹן זָהָ֣ב סָג֑וּר אֵ֣ין כֶּ֗סֶף לֹ֥א נֶחְשָׁ֛ב בִּימֵ֥י שְׁלֹמֹ֖ה לִמְאֽוּמָה׃ (יד) כִּי֩ אֳנִ֨י תַרְשִׁ֤ישׁ לַמֶּ֙לֶךְ֙ בַּיָּ֔ם עִ֖ם אֳנִ֣י חִירָ֑ם אַחַת֩ לְשָׁלֹ֨שׁ שָׁנִ֜ים תָּב֣וֹא ׀ אֳנִ֣י תַרְשִׁ֗ישׁ נֹֽשְׂאֵת֙ זָהָ֣ב וָכֶ֔סֶף שֶׁנְהַבִּ֥ים וְקֹפִ֖ים וְתֻכִּיִּֽים׃ (טו) וַיִּגְדַּל֙ הַמֶּ֣לֶךְ שְׁלֹמֹ֔ה מִכֹּ֖ל מַלְכֵ֣י הָאָ֑רֶץ לְעֹ֖שֶׁר וּלְחָכְמָֽה׃ (טז) וְכָ֨ל־ הָאָ֔רֶץ מְבַקְשִׁ֖ים אֶת־ פְּנֵ֣י שְׁלֹמֹ֑ה לִשְׁמֹ֙עַ֙ אֶת־ חָכְמָת֔וֹ אֲשֶׁר־ נָתַ֥ן אֱלֹהִ֖ים בְּלִבּֽוֹ׃ (יז) וְהֵ֣מָּה מְבִאִ֣ים אִ֣ישׁ מִנְחָת֡וֹ כְּלֵ֣י כֶסֶף֩ וּכְלֵ֨י זָהָ֤ב וּשְׂלָמוֹת֙ וְנֵ֣שֶׁק וּבְשָׂמִ֔ים סוּסִ֖ים וּפְרָדִ֑ים דְּבַר־ שָׁנָ֖ה בְּשָׁנָֽה׃ (יח) וַיֶּאֱסֹ֣ף שְׁלֹמֹה֮ רֶ֣כֶב וּפָרָשִׁים֒ וַיְהִי־ ל֗וֹ אֶ֤לֶף וְאַרְבַּע־ מֵאוֹת֙ רֶ֔כֶב וּשְׁנֵים־ עָשָׂ֥ר אֶ֖לֶף פָּֽרָשִׁ֑ים וַיַּנְחֵם֙ בְּעָרֵ֣י הָרֶ֔כֶב וְעִם־ הַמֶּ֖לֶךְ בִּירוּשָׁלִָֽם׃ (יט) וַיִּתֵּ֨ן הַמֶּ֧לֶךְ אֶת־ הַכֶּ֛סֶף בִּירוּשָׁלִַ֖ם כָּאֲבָנִ֑ים וְאֵ֣ת הָאֲרָזִ֗ים נָתַ֛ן כַּשִּׁקְמִ֥ים אֲשֶׁר־ בַּשְּׁפֵלָ֖ה לָרֹֽב׃ (כ) וּמוֹצָ֧א הַסּוּסִ֛ים אֲשֶׁ֥ר לִשְׁלֹמֹ֖ה מִמִּצְרָ֑יִם וּמִקְוֵ֕ה סֹחֲרֵ֣י הַמֶּ֔לֶךְ יִקְח֥וּ מִקְוֵ֖ה בִּמְחִֽיר׃ (כא) וַֽ֠תַּעֲלֶה וַתֵּצֵ֨א מֶרְכָּבָ֤ה מִמִּצְרַ֙יִם֙ בְּשֵׁ֣שׁ מֵא֣וֹת כֶּ֔סֶף וְס֖וּס בַּחֲמִשִּׁ֣ים וּמֵאָ֑ה וְ֠כֵן לְכָל־ מַלְכֵ֧י הַחִתִּ֛ים וּלְמַלְכֵ֥י אֲרָ֖ם בְּיָדָ֥ם יֹצִֽאוּ׃ סוף פרק י (כב) וְהַמֶּ֣לֶךְ שְׁלֹמֹ֗ה אָהַ֞ב נָשִׁ֧ים נָכְרִיּ֛וֹת רַבּ֖וֹת וְאֶת־ בַּת־ פַּרְעֹ֑ה מוֹאֲבִיּ֤וֹת עַמֳּנִיּוֹת֙ אֲדֹ֣מִיֹּ֔ת צֵדְנִיֹּ֖ת חִתִּיֹּֽת׃ (כג) מִן־ הַגּוֹיִ֗ם אֲשֶׁ֣ר אָֽמַר־ יְהוָה֩ אֶל־ בְּנֵ֨י יִשְׂרָאֵ֜ל לֹֽא־ תָבֹ֣אוּ בָהֶ֗ם וְהֵם֙ לֹא־ יָבֹ֣אוּ בָכֶ֔ם אָכֵן֙ יַטּ֣וּ אֶת־ לְבַבְכֶ֔ם אַחֲרֵ֖י אֱלֹהֵיהֶ֑ם בָּהֶ֛ם דָּבַ֥ק שְׁלֹמֹ֖ה לְאַהֲבָֽה׃ (כד) וַיְהִי־ ל֣וֹ נָשִׁ֗ים שָׂרוֹת֙ שְׁבַ֣ע מֵא֔וֹת וּפִֽלַגְשִׁ֖ים שְׁלֹ֣שׁ מֵא֑וֹת וַיַּטּ֥וּ נָשָׁ֖יו אֶת־ לִבּֽוֹ׃ (כה) וַיְהִ֗י לְעֵת֙ זִקְנַ֣ת שְׁלֹמֹ֔ה נָשָׁיו֙ הִטּ֣וּ אֶת־ לְבָב֔וֹ אַחֲרֵ֖י אֱלֹהִ֣ים אֲחֵרִ֑ים וְלֹא־ הָיָ֨ה לְבָב֤וֹ שָׁלֵם֙ עִם־ יְהוָ֣ה אֱלֹהָ֔יו כִּלְבַ֖ב דָּוִ֥יד אָבִֽיו׃ (כו) וַיֵּ֣לֶךְ שְׁלֹמֹ֔ה אַחֲרֵ֣י עַשְׁתֹּ֔רֶת אֱלֹהֵ֖י צִדֹנִ֑ים וְאַחֲרֵ֣י מִלְכֹּ֔ם שִׁקֻּ֖ץ עַמֹּנִֽים׃ (כז) וַיַּ֧עַשׂ שְׁלֹמֹ֛ה הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וְלֹ֥א מִלֵּ֛א אַחֲרֵ֥י יְהוָ֖ה כְּדָוִ֥ד אָבִֽיו׃ (כח) אָז֩ יִבְנֶ֨ה שְׁלֹמֹ֜ה בָּמָ֗ה לִכְמוֹשׁ֙ שִׁקֻּ֣ץ מוֹאָ֔ב בָּהָ֕ר אֲשֶׁ֖ר עַל־ פְּנֵ֣י יְרוּשָׁלִָ֑ם וּלְמֹ֕לֶךְ שִׁקֻּ֖ץ בְּנֵ֥י עַמּֽוֹן׃ (כט) וְכֵ֣ן עָשָׂ֔ה לְכָל־ נָשָׁ֖יו הַנָּכְרִיּ֑וֹת מַקְטִיר֥וֹת וּֽמְזַבְּח֖וֹת לֵאלֹהֵיהֶֽן׃ (ל) וַיִּתְאַנַּ֥ף יְהוָ֖ה בִּשְׁלֹמֹ֑ה כִּֽי־ נָטָ֣ה לְבָב֗וֹ מֵעִ֤ם יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הַנִּרְאָ֥ה אֵלָ֖יו פַּעֲמָֽיִם׃ (לא) וְצִוָּ֤ה אֵלָיו֙ עַל־ הַדָּבָ֣ר הַזֶּ֔ה לְבִ֨לְתִּי־ לֶ֔כֶת אַחֲרֵ֖י אֱלֹהִ֣ים אֲחֵרִ֑ים וְלֹ֣א שָׁמַ֔ר אֵ֥ת אֲשֶׁר־ צִוָּ֖ה יְהוָֽה׃ (לב) וַיֹּ֨אמֶר יְהוָ֜ה לִשְׁלֹמֹ֗ה יַ֚עַן אֲשֶׁ֣ר הָֽיְתָה־ זֹּ֣את עִמָּ֔ךְ וְלֹ֤א שָׁמַ֙רְתָּ֙ בְּרִיתִ֣י וְחֻקֹּתַ֔י אֲשֶׁ֥ר צִוִּ֖יתִי עָלֶ֑יךָ קָרֹ֨עַ אֶקְרַ֤ע אֶת־ הַמַּמְלָכָה֙ מֵֽעָלֶ֔יךָ וּנְתַתִּ֖יהָ לְעַבְדֶּֽךָ׃ (לג) אַךְ־ בְּיָמֶ֙יךָ֙ לֹ֣א אֶעֱשֶׂ֔נָּה לְמַ֖עַן דָּוִ֣ד אָבִ֑יךָ מִיַּ֥ד בִּנְךָ֖ אֶקְרָעֶֽנָּה׃ (לד) רַ֤ק אֶת־ כָּל־ הַמַּמְלָכָה֙ לֹ֣א אֶקְרָ֔ע שֵׁ֥בֶט אֶחָ֖ד אֶתֵּ֣ן לִבְנֶ֑ךָ לְמַ֙עַן֙ דָּוִ֣ד עַבְדִּ֔י וּלְמַ֥עַן יְרוּשָׁלִַ֖ם אֲשֶׁ֥ר בָּחָֽרְתִּי׃ (לה) וַיָּ֨קֶם יְהוָ֤ה שָׂטָן֙ לִשְׁלֹמֹ֔ה אֵ֖ת הֲדַ֣ד הָאֲדֹמִ֑י מִזֶּ֧רַע הַמֶּ֛לֶךְ ה֖וּא בֶּאֱדֽוֹם׃ (לו) וַיְהִ֗י בִּֽהְי֤וֹת דָּוִד֙ אֶת־ אֱד֔וֹם בַּעֲל֗וֹת יוֹאָב֙ שַׂ֣ר הַצָּבָ֔א לְקַבֵּ֖ר אֶת־ הַחֲלָלִ֑ים וַיַּ֥ךְ כָּל־ זָכָ֖ר בֶּאֱדֽוֹם׃ (לז) כִּ֣י שֵׁ֧שֶׁת חֳדָשִׁ֛ים יָֽשַׁב־ שָׁ֥ם יוֹאָ֖ב וְכָל־ יִשְׂרָאֵ֑ל עַד־ הִכְרִ֥ית כָּל־ זָכָ֖ר בֶּאֱדֽוֹם׃ (לח) וַיִּבְרַ֣ח אֲדַ֡ד הוּא֩ וַאֲנָשִׁ֨ים אֲדֹמִיִּ֜ים מֵעַבְדֵ֥י אָבִ֛יו אִתּ֖וֹ לָב֣וֹא מִצְרָ֑יִם וַהֲדַ֖ד נַ֥עַר קָטָֽן׃ (לט) וַיָּקֻ֙מוּ֙ מִמִּדְיָ֔ן וַיָּבֹ֖אוּ פָּארָ֑ן וַיִּקְחוּ֩ אֲנָשִׁ֨ים עִמָּ֜ם מִפָּארָ֗ן וַיָּבֹ֤אוּ מִצְרַ֙יִם֙ אֶל־ פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם וַיִּתֶּן־ ל֣וֹ בַ֗יִת וְלֶ֙חֶם֙ אָ֣מַר ל֔וֹ וְאֶ֖רֶץ נָ֥תַן לֽוֹ׃ (מ) וַיִּמְצָא֙ הֲדַ֥ד חֵ֛ן בְּעֵינֵ֥י פַרְעֹ֖ה מְאֹ֑ד וַיִּתֶּן־ ל֤וֹ אִשָּׁה֙ אֶת־ אֲח֣וֹת אִשְׁתּ֔וֹ אֲח֖וֹת תַּחְפְּנֵ֥יס הַגְּבִירָֽה׃ (מא) וַתֵּ֨לֶד ל֜וֹ אֲח֣וֹת תַּחְפְּנֵ֗יס אֵ֚ת גְּנֻבַ֣ת בְּנ֔וֹ וַתִּגְמְלֵ֣הוּ תַחְפְּנֵ֔ס בְּת֖וֹךְ בֵּ֣ית פַּרְעֹ֑ה וַיְהִ֤י גְנֻבַת֙ בֵּ֣ית פַּרְעֹ֔ה בְּת֖וֹךְ בְּנֵ֥י פַרְעֹֽה׃ (מב) וַהֲדַ֞ד שָׁמַ֣ע בְּמִצְרַ֗יִם כִּֽי־ שָׁכַ֤ב דָּוִד֙ עִם־ אֲבֹתָ֔יו וְכִי־ מֵ֖ת יוֹאָ֣ב שַֽׂר־ הַצָּבָ֑א וַיֹּ֤אמֶר הֲדַד֙ אֶל־ פַּרְעֹ֔ה שַׁלְּחֵ֖נִי וְאֵלֵ֥ךְ אֶל־ אַרְצִֽי׃ (מג) וַיֹּ֧אמֶר ל֣וֹ פַרְעֹ֗ה כִּ֠י מָה־ אַתָּ֤ה חָסֵר֙ עִמִּ֔י וְהִנְּךָ֥ מְבַקֵּ֖שׁ לָלֶ֣כֶת אֶל־ אַרְצֶ֑ךָ וַיֹּ֣אמֶר ׀ לֹ֔א כִּ֥י שַׁלֵּ֖חַ תְּשַׁלְּחֵֽנִי׃ (מד) וַיָּ֨קֶם אֱלֹהִ֥ים לוֹ֙ שָׂטָ֔ן אֶת־ רְז֖וֹן בֶּן־ אֶלְיָדָ֑ע אֲשֶׁ֣ר בָּרַ֗ח מֵאֵ֛ת הֲדַדְעֶ֥זֶר מֶֽלֶךְ־ צוֹבָ֖ה אֲדֹנָֽיו׃ (מה) וַיִּקְבֹּ֤ץ עָלָיו֙ אֲנָשִׁ֔ים וַיְהִ֣י שַׂר־ גְּד֔וּד בַּהֲרֹ֥ג דָּוִ֖ד אֹתָ֑ם וַיֵּלְכ֤וּ דַמֶּ֙שֶׂק֙ וַיֵּ֣שְׁבוּ בָ֔הּ וַֽיִּמְלְכ֖וּ בְּדַמָּֽשֶׂק׃ (מו) וַיְהִ֨י שָׂטָ֤ן לְיִשְׂרָאֵל֙ כָּל־ יְמֵ֣י שְׁלֹמֹ֔ה וְאֶת־ הָרָעָ֖ה אֲשֶׁ֣ר הֲדָ֑ד וַיָּ֙קָץ֙ בְּיִשְׂרָאֵ֔ל וַיִּמְלֹ֖ךְ עַל־ אֲרָֽם׃ (מז) וְיָרָבְעָם֩ בֶּן־ נְבָ֨ט אֶפְרָתִ֜י מִן־ הַצְּרֵדָ֗ה וְשֵׁ֤ם אִמּוֹ֙ צְרוּעָה֙ אִשָּׁ֣ה אַלְמָנָ֔ה עֶ֖בֶד לִשְׁלֹמֹ֑ה וַיָּ֥רֶם יָ֖ד בַּמֶּֽלֶךְ׃ (מח) וְזֶ֣ה הַדָּבָ֔ר אֲשֶׁר־ הֵרִ֥ים יָ֖ד בַּמֶּ֑לֶךְ שְׁלֹמֹה֙ בָּנָ֣ה אֶת־ הַמִּלּ֔וֹא סָגַ֕ר אֶת־ פֶּ֕רֶץ עִ֖יר דָּוִ֥ד אָבִֽיו׃
