יהושע פרק יט לפי החלוקה לסדרים

פרק יט בספר יהושע (51 פסוקים) נמצא ביהושע סדר יא (פסוקים ב–נא) וביהושע סדר יב (פסוקים א–א).

יהושע סדר יא

(א) וְצֵלַ֡ע הָאֶ֜לֶף וְהַיְבוּסִ֨י הִ֤יא יְרֽוּשָׁלִַ֙ם֙ גִּבְעַ֣ת קִרְיַ֔ת עָרִ֥ים אַרְבַּֽע־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶ֑ן זֹ֛את נַֽחֲלַ֥ת בְּנֵֽי־ בִנְיָמִ֖ן לְמִשְׁפְּחֹתָֽם׃ תחילת פרק יט (ב) וַיֵּצֵ֞א הַגּוֹרָ֤ל הַשֵּׁנִי֙ לְשִׁמְע֔וֹן לְמַטֵּ֥ה בְנֵֽי־ שִׁמְע֖וֹן לְמִשְׁפְּחוֹתָ֑ם וַֽיְהִי֙ נַֽחֲלָתָ֔ם בְּת֖וֹךְ נַחֲלַ֥ת בְּנֵֽי־ יְהוּדָֽה׃ (ג) וַיְהִ֥י לָהֶ֖ם בְּנַֽחֲלָתָ֑ם בְּאֵֽר־ שֶׁ֥בַע וְשֶׁ֖בַע וּמוֹלָדָֽה׃ (ד) וַחֲצַ֥ר שׁוּעָ֛ל וּבָלָ֖ה וָעָֽצֶם׃ (ה) וְאֶלְתּוֹלַ֥ד וּבְת֖וּל וְחָרְמָֽה׃ (ו) וְצִֽקְלַ֥ג וּבֵית־ הַמַּרְכָּב֖וֹת וַחֲצַ֥ר סוּסָֽה׃ (ז) וּבֵ֥ית לְבָא֖וֹת וְשָֽׁרוּחֶ֑ן עָרִ֥ים שְׁלֹשׁ־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (ח) עַ֥יִן ׀ רִמּ֖וֹן וָעֶ֣תֶר וְעָשָׁ֑ן עָרִ֥ים אַרְבַּ֖ע וְחַצְרֵיהֶֽן׃ (ט) וְכָל־ הַֽחֲצֵרִ֗ים אֲשֶׁ֤ר סְבִיבוֹת֙ הֶֽעָרִ֣ים הָאֵ֔לֶּה עַד־ בַּֽעֲלַ֥ת בְּאֵ֖ר רָ֣אמַת נֶ֑גֶב זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵֽי־ שִׁמְע֖וֹן לְמִשְׁפְּחֹתָֽם׃ (י) מֵחֶ֙בֶל֙ בְּנֵ֣י יְהוּדָ֔ה נַחֲלַ֖ת בְּנֵ֣י שִׁמְע֑וֹן כִּֽי־ הָיָ֞ה חֵ֤לֶק בְּנֵֽי־ יְהוּדָה֙ רַ֣ב מֵהֶ֔ם וַיִּנְחֲל֥וּ בְנֵֽי־ שִׁמְע֖וֹן בְּת֥וֹךְ נַחֲלָתָֽם׃ (יא) וַיַּ֙עַל֙ הַגּוֹרָ֣ל הַשְּׁלִישִׁ֔י לִבְנֵ֥י זְבוּלֻ֖ן לְמִשְׁפְּחֹתָ֑ם וַיְהִ֛י גְּב֥וּל נַחֲלָתָ֖ם עַד־ שָׂרִֽיד׃ (יב) וְעָלָ֨ה גְבוּלָ֧ם ׀ לַיָּ֛מָּה וּמַרְעֲלָ֖ה וּפָגַ֣ע בְּדַבָּ֑שֶׁת וּפָגַע֙ אֶל־ הַנַּ֔חַל אֲשֶׁ֖ר עַל־ פְּנֵ֥י יָקְנְעָֽם׃ (יג) וְשָׁ֣ב מִשָּׂרִ֗יד קֵ֚דְמָה מִזְרַ֣ח הַשֶּׁ֔מֶשׁ עַל־ גְּב֥וּל כִּסְלֹ֖ת תָּבֹ֑ר וְיָצָ֥א אֶל־ הַדָּֽבְרַ֖ת וְעָלָ֥ה יָפִֽיעַ׃ (יד) וּמִשָּׁ֤ם עָבַר֙ קֵ֣דְמָה מִזְרָ֔חָה גִּתָּ֥ה חֵ֖פֶר עִתָּ֣ה קָצִ֑ין וְיָצָ֛א רִמּ֥וֹן הַמְּתֹאָ֖ר הַנֵּעָֽה׃ (טו) וְנָסַ֤ב אֹתוֹ֙ הַגְּב֔וּל מִצְּפ֖וֹן חַנָּתֹ֑ן וְהָיוּ֙ תֹּֽצְאֹתָ֔יו גֵּ֖י יִפְתַּח־ אֵֽל׃ (טז) וְקַטָּ֤ת וְנַֽהֲלָל֙ וְשִׁמְר֔וֹן וְיִדְאֲלָ֖ה וּבֵ֣ית לָ֑חֶם עָרִ֥ים שְׁתֵּים־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (יז) זֹ֛את נַחֲלַ֥ת בְּנֵֽי־ זְבוּלֻ֖ן לְמִשְׁפְּחוֹתָ֑ם הֶֽעָרִ֥ים הָאֵ֖לֶּה וְחַצְרֵיהֶֽן׃ (יח) לְיִ֨שָּׂשכָ֔ר יָצָ֖א הַגּוֹרָ֣ל הָֽרְבִיעִ֑י לִבְנֵ֥י יִשָּׂשכָ֖ר לְמִשְׁפְּחוֹתָֽם׃ (יט) וַיְהִ֖י גְּבוּלָ֑ם יִזְרְעֶ֥אלָה וְהַכְּסוּלֹ֖ת וְשׁוּנֵֽם׃ (כ) וַחֲפָרַ֥יִם וְשִׁיאֹ֖ן וַאֲנָחֲרַֽת׃ (כא) וְהָֽרַבִּ֥ית וְקִשְׁי֖וֹן וָאָֽבֶץ׃ (כב) וְרֶ֧מֶת וְעֵין־ גַּנִּ֛ים וְעֵ֥ין חַדָּ֖ה וּבֵ֥ית פַּצֵּֽץ׃ (כג) וּפָגַע֩ הַגְּב֨וּל בְּתָב֤וֹר (ושחצומה) וְשַׁחֲצִ֙ימָה֙ וּבֵ֣ית שֶׁ֔מֶשׁ וְהָי֛וּ תֹּצְא֥וֹת גְּבוּלָ֖ם הַיַּרְדֵּ֑ן עָרִ֥ים שֵׁשׁ־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (כד) זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵֽי־ יִשָּׂשכָ֖ר לְמִשְׁפְּחֹתָ֑ם הֶעָרִ֖ים וְחַצְרֵיהֶֽן׃ (כה) וַיֵּצֵא֙ הַגּוֹרָ֣ל הַֽחֲמִישִׁ֔י לְמַטֵּ֥ה בְנֵֽי־ אָשֵׁ֖ר לְמִשְׁפְּחוֹתָֽם׃ (כו) וַיְהִ֖י גְּבוּלָ֑ם חֶלְקַ֥ת וַחֲלִ֖י וָבֶ֥טֶן וְאַכְשָֽׁף׃ (כז) וְאַֽלַמֶּ֥לֶךְ וְעַמְעָ֖ד וּמִשְׁאָ֑ל וּפָגַ֤ע בְּכַרְמֶל֙ הַיָּ֔מָּה וּבְשִׁיח֖וֹר לִבְנָֽת׃ (כח) וְשָׁ֨ב מִזְרַ֣ח הַשֶּׁמֶשׁ֮ בֵּ֣ית דָּגֹן֒ וּפָגַ֣ע בִּ֠זְבֻלוּן וּבְגֵ֨י יִפְתַּח־ אֵ֥ל צָפ֛וֹנָה בֵּ֥ית הָעֵ֖מֶק וּנְעִיאֵ֑ל וְיָצָ֥א אֶל־ כָּב֖וּל מִשְּׂמֹֽאל׃ (כט) וְעֶבְרֹ֥ן וּרְחֹ֖ב וְחַמּ֣וֹן וְקָנָ֑ה עַ֖ד צִיד֥וֹן רַבָּֽה׃ (ל) וְשָׁ֤ב הַגְּבוּל֙ הָֽרָמָ֔ה וְעַד־ עִ֖יר מִבְצַר־ צֹ֑ר וְשָׁ֤ב הַגְּבוּל֙ חֹסָ֔ה (ויהיו) וְהָי֧וּ תֹצְאֹתָ֛יו הַיָּ֖מָּה מֵחֶ֥בֶל אַכְזִֽיבָה׃ (לא) וְעֻמָ֥ה וַאֲפֵ֖ק וּרְחֹ֑ב עָרִ֛ים עֶשְׂרִ֥ים וּשְׁתַּ֖יִם וְחַצְרֵיהֶֽן׃ (לב) זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵֽי־ אָשֵׁ֖ר לְמִשְׁפְּחֹתָ֑ם הֶעָרִ֥ים הָאֵ֖לֶּה וְחַצְרֵיהֶֽן׃ (לג) לִבְנֵ֣י נַפְתָּלִ֔י יָצָ֖א הַגּוֹרָ֣ל הַשִּׁשִּׁ֑י לִבְנֵ֥י נַפְתָּלִ֖י לְמִשְׁפְּחֹתָֽם׃ (לד) וַיְהִ֣י גְבוּלָ֗ם מֵחֵ֨לֶף מֵֽאֵל֜וֹן בְּצַעֲנַנִּ֗ים וַאֲדָמִ֥י הַנֶּ֛קֶב וְיַבְנְאֵ֖ל עַד־ לַקּ֑וּם וַיְהִ֥י תֹצְאֹתָ֖יו הַיַּרְדֵּֽן׃ (לה) וְשָׁ֨ב הַגְּב֥וּל יָ֙מָּה֙ אַזְנ֣וֹת תָּב֔וֹר וְיָצָ֥א מִשָּׁ֖ם חוּקֹ֑קָה וּפָגַ֨ע בִּזְבֻל֜וּן מִנֶּ֗גֶב וּבְאָשֵׁר֙ פָּגַ֣ע מִיָּ֔ם וּבִ֣יהוּדָ֔ה הַיַּרְדֵּ֖ן מִזְרַ֥ח הַשָּֽׁמֶשׁ׃ (לו) וְעָרֵ֖י מִבְצָ֑ר הַצִּדִּ֣ים צֵ֔ר וְחַמַּ֖ת רַקַּ֥ת וְכִנָּֽרֶת׃ (לז) וַאֲדָמָ֥ה וְהָרָמָ֖ה וְחָצֽוֹר׃ (לח) וְקֶ֥דֶשׁ וְאֶדְרֶ֖עִי וְעֵ֥ין חָצֽוֹר׃ (לט) וְיִרְאוֹן֙ וּמִגְדַּל־ אֵ֔ל חֳרֵ֥ם וּבֵית־ עֲנָ֖ת וּבֵ֣ית שָׁ֑מֶשׁ עָרִ֥ים תְּשַֽׁע־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (מ) זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵֽי־ נַפְתָּלִ֖י לְמִשְׁפְּחֹתָ֑ם הֶעָרִ֖ים וְחַצְרֵיהֶֽן׃ (מא) לְמַטֵּ֥ה בְנֵי־ דָ֖ן לְמִשְׁפְּחֹתָ֑ם יָצָ֖א הַגּוֹרָ֥ל הַשְּׁבִיעִֽי׃ (מב) וַיְהִ֖י גְּב֣וּל נַחֲלָתָ֑ם צָרְעָ֥ה וְאֶשְׁתָּא֖וֹל וְעִ֥יר שָֽׁמֶשׁ׃ (מג) וְשַֽׁעֲלַבִּ֥ין וְאַיָּל֖וֹן וְיִתְלָֽה׃ (מד) וְאֵיל֥וֹן וְתִמְנָ֖תָה וְעֶקְרֽוֹן׃ (מה) וְאֶלְתְּקֵ֥ה וְגִבְּת֖וֹן וּבַעֲלָֽת׃ (מו) וִיהֻ֥ד וּבְנֵֽי־ בְרַ֖ק וְגַת־ רִמּֽוֹן׃ (מז) וּמֵ֥י הַיַּרְק֖וֹן וְהָֽרַקּ֑וֹן עִֽם־ הַגְּב֖וּל מ֥וּל יָפֽוֹ׃ (מח) וַיֵּצֵ֥א גְבוּל־ בְּנֵי־ דָ֖ן מֵהֶ֑ם וַיַּעֲל֣וּ בְנֵֽי־ דָ֠ן וַיִּלָּחֲמ֨וּ עִם־ לֶ֜שֶׁם וַיִּלְכְּד֥וּ אוֹתָ֣הּ ׀ וַיַּכּ֧וּ אוֹתָ֣הּ לְפִי־ חֶ֗רֶב וַיִּֽרְשׁ֤וּ אוֹתָהּ֙ וַיֵּ֣שְׁבוּ בָ֔הּ וַיִּקְרְא֤וּ לְלֶ֙שֶׁם֙ דָּ֔ן כְּשֵׁ֖ם דָּ֥ן אֲבִיהֶֽם׃ (מט) זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵי־ דָ֖ן לְמִשְׁפְּחֹתָ֑ם הֶֽעָרִ֥ים הָאֵ֖לֶּה וְחַצְרֵיהֶֽן׃ (נ) וַיְכַלּ֥וּ לִנְחֹל־ אֶת־ הָאָ֖רֶץ לִגְבֽוּלֹתֶ֑יהָ וַיִּתְּנ֨וּ בְנֵי־ יִשְׂרָאֵ֧ל נַחֲלָ֛ה לִיהוֹשֻׁ֥עַ בִּן־ נ֖וּן בְּתוֹכָֽם׃ (נא) עַל־ פִּ֨י יְהוָ֜ה נָ֣תְנוּ ל֗וֹ אֶת־ הָעִיר֙ אֲשֶׁ֣ר שָׁאָ֔ל אֶת־ תִּמְנַת־ סֶ֖רַח בְּהַ֣ר אֶפְרָ֑יִם וַיִּבְנֶ֥ה אֶת־ הָעִ֖יר וַיֵּ֥שֶׁב בָּֽהּ׃

יהושע סדר יב

(א) אֵ֣לֶּה הַנְּחָלֹ֡ת אֲשֶׁ֣ר נִחֲל֣וּ אֶלְעָזָ֣ר הַכֹּהֵ֣ן ׀ וִיהוֹשֻׁ֪עַ בִּן־ נ֟וּן וְרָאשֵׁ֣י הָֽאָב֣וֹת לְמַטּוֹת֩ בְּנֵי־ יִשְׂרָאֵ֨ל ׀ בְּגוֹרָ֤ל ׀ בְּשִׁלֹה֙ לִפְנֵ֣י יְהוָ֔ה פֶּ֖תַח אֹ֣הֶל מוֹעֵ֑ד וַיְכַלּ֕וּ מֵֽחַלֵּ֖ק אֶת־ הָאָֽרֶץ׃ סוף פרק יט (ב) וַיְדַבֵּ֣ר יְהוָ֔ה אֶל־ יְהוֹשֻׁ֖עַ לֵאמֹֽר׃ (ג) דַּבֵּ֛ר אֶל־ בְּנֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֑ר תְּנ֤וּ לָכֶם֙ אֶת־ עָרֵ֣י הַמִּקְלָ֔ט אֲשֶׁר־ דִּבַּ֥רְתִּי אֲלֵיכֶ֖ם בְּיַד־ מֹשֶֽׁה׃ (ד) לָנ֥וּס שָׁ֙מָּה֙ רוֹצֵ֔חַ מַכֵּה־ נֶ֥פֶשׁ בִּשְׁגָגָ֖ה בִּבְלִי־ דָ֑עַת וְהָי֤וּ לָכֶם֙ לְמִקְלָ֔ט מִגֹּאֵ֖ל הַדָּֽם׃ (ה) וְנָ֞ס אֶל־ אַחַ֣ת ׀ מֵהֶעָרִ֣ים מִגְרָשֶֽׁהָ׃ (לט) אֶת־ יַרְמוּת֙ וְאֶת־ מִגְרָשֶׁ֔הָ אֶת־ עֵ֥ין גַּנִּ֖ים וְאֶת־ מִגְרָשֶׁ֑הָ עָרִ֖ים אַרְבַּֽע׃ (מ) וּמִמַּטֵּ֣ה אָשֵׁ֔ר אֶת־ מִשְׁאָ֖ל וְאֶת־ מִגְרָשֶׁ֑הָ אֶת־ עַבְדּ֖וֹן וְאֶת־ מִגְרָשֶֽׁהָ׃ (מא) אֶת־ חֶלְקָת֙ וְאֶת־ מִגְרָשֶׁ֔הָ וְאֶת־ רְחֹ֖ב וְאֶת־ מִגְרָשֶׁ֑הָ עָרִ֖ים אַרְבַּֽע׃ (מב) וּמִמַּטֵּ֨ה נַפְתָּלִ֜י אֶת־ עִ֣יר ׀ מִקְלַ֣ט הָֽרֹצֵ֗חַ אֶת־ קֶ֨דֶשׁ בַּגָּלִ֤יל וְאֶת־ מִגְרָשֶׁ֙הָ֙ וְאֶת־ חַמֹּ֥ת דֹּאר֙ וְאֶת־ מִגְרָשֶׁ֔הָ וְאֶת־ קַרְתָּ֖ן וְאֶת־ מִגְרָשֶׁ֑הָ עָרִ֖ים שָׁלֹֽשׁ׃ (מג) כָּל־ עָרֵ֥י הַגֵּרְשֻׁנִּ֖י לְמִשְׁפְּחֹתָ֑ם שְׁלֹשׁ־ עֶשְׂרֵ֥ה עִ֖יר וּמִגְרְשֵׁיהֶֽן׃ (מד) וּלְמִשְׁפְּח֣וֹת בְּנֵֽי־ מְרָרִי֮ הַלְוִיִּ֣ם הַנּוֹתָרִים֒ מֵאֵת֙ מַטֵּ֣ה זְבוּלֻ֔ן אֶֽת־ יָקְנְעָ֖ם וְאֶת־ מִגְרָשֶׁ֑הָ אֶת־ קַרְתָּ֖ה וְאֶת־ מִגְרָשֶֽׁהָ׃ (מה) אֶת־ דִּמְנָה֙ וְאֶת־ מִגְרָשֶׁ֔הָ אֶֽת־ נַהֲלָ֖ל וְאֶת־ מִגְרָשֶׁ֑הָ עָרִ֖ים אַרְבַּֽע׃ (מו) וּמִמַּטֵּ֣ה רְאוּבֵ֔ן אֶת־ בֶּ֖צֶר וְאֶת־ מִגְרָשֶׁ֑הָ וְאֶת־ יַ֖הְצָה וְאֶת־ מִגְרָשֶֽׁהָ׃ (מז) אֶת־ קְדֵמוֹת֙ וְאֶת־ מִגְרָשֶׁ֔הָ וְאֶת־ מֵיפָ֖עַת וְאֶת־ מִגְרָשֶׁ֑הָ עָרִ֖ים אַרְבַּֽע׃ (מח) וּמִמַּטֵּה־ גָ֗ד אֶת־ עִיר֙ מִקְלַ֣ט הָרֹצֵ֔חַ אֶת־ רָמֹ֥ת בַּגִּלְעָ֖ד וְאֶת־ מִגְרָשֶׁ֑הָ וְאֶֽת־ מַחֲנַ֖יִם וְאֶת־ מִגְרָשֶֽׁהָ׃ (מט) אֶת־ חֶשְׁבּוֹן֙ וְאֶת־ מִגְרָשֶׁ֔הָ אֶת־ יַעְזֵ֖ר וְאֶת־ מִגְרָשֶׁ֑הָ כָּל־ עָרִ֖ים אַרְבַּֽע׃ (נ) כָּל־ הֶ֨עָרִ֜ים לִבְנֵ֤י מְרָרִי֙ לְמִשְׁפְּחֹתָ֔ם הַנּוֹתָרִ֖ים מִמִּשְׁפְּח֣וֹת הַלְוִיִּ֑ם וַיְהִי֙ גּוֹרָלָ֔ם עָרִ֖ים שְׁתֵּ֥ים עֶשְׂרֵֽה׃
