יהושע פרק יד לפי החלוקה לסדרים

פרק יד בספר יהושע (15 פסוקים) נמצא ביהושע סדר ח (פסוקים לד–מז) וביהושע סדר ט (פסוקים א–א).

יהושע סדר ח

(א) וִיהוֹשֻׁ֣עַ זָקֵ֔ן בָּ֖א בַּיָּמִ֑ים וַיֹּ֨אמֶר יְהוָ֜ה אֵלָ֗יו אַתָּ֤ה זָקַ֙נְתָּה֙ בָּ֣אתָ בַיָּמִ֔ים וְהָאָ֛רֶץ נִשְׁאֲרָ֥ה הַרְבֵּֽה־ מְאֹ֖ד לְרִשְׁתָּֽהּ׃ (ב) זֹ֥את הָאָ֖רֶץ הַנִּשְׁאָ֑רֶת כָּל־ גְּלִיל֥וֹת הַפְּלִשְׁתִּ֖ים וְכָל־ הַגְּשׁוּרִֽי׃ (ג) מִֽן־ הַשִּׁיח֞וֹר אֲשֶׁ֣ר ׀ עַל־ פְּנֵ֣י מִצְרַ֗יִם וְעַ֨ד גְּב֤וּל עֶקְרוֹן֙ צָפ֔וֹנָה לַֽכְּנַעֲנִ֖י תֵּחָשֵׁ֑ב חֲמֵ֣שֶׁת ׀ סַרְנֵ֣י פְלִשְׁתִּ֗ים הָעַזָּתִ֤י וְהָאַשְׁדּוֹדִי֙ הָאֶשְׁקְלוֹנִ֣י הַגִּתִּ֔י וְהָעֶקְרוֹנִ֖י וְהָעַוִּֽים׃ (ד) מִתֵּימָ֞ן כָּל־ אֶ֣רֶץ הַֽכְּנַעֲנִ֗י וּמְעָרָ֛ה אֲשֶׁ֥ר לַצִּידֹנִ֖ים עַד־ אֲפֵ֑קָה עַ֖ד גְּב֥וּל הָאֱמֹרִֽי׃ (ה) וְהָאָ֣רֶץ הַגִּבְלִ֗י וְכָל־ הַלְּבָנוֹן֙ מִזְרַ֣ח הַשֶּׁ֔מֶשׁ מִבַּ֣עַל גָּ֔ד תַּ֖חַת הַר־ חֶרְמ֑וֹן עַ֖ד לְב֥וֹא חֲמָֽת׃ (ו) כָּל־ יֹשְׁבֵ֣י הָ֠הָר מִֽן־ הַלְּבָנ֞וֹן עַד־ מִשְׂרְפֹ֥ת מַ֙יִם֙ כָּל־ צִ֣ידֹנִ֔ים אָֽנֹכִי֙ אוֹרִישֵׁ֔ם מִפְּנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל רַ֠ק הַפִּלֶ֤הָ לְיִשְׂרָאֵל֙ בְּֽנַחֲלָ֔ה כַּאֲשֶׁ֖ר צִוִּיתִֽיךָ׃ (ז) וְעַתָּ֗ה חַלֵּ֞ק אֶת־ הָאָ֧רֶץ הַזֹּ֛את בְּנַחֲלָ֖ה לְתִשְׁעַ֣ת הַשְּׁבָטִ֑ים וַחֲצִ֖י הַשֵּׁ֥בֶט הַֽמְנַשֶּֽׁה׃ (ח) עִמּ֗וֹ הָרֽאוּבֵנִי֙ וְהַגָּדִ֔י לָקְח֖וּ נַחֲלָתָ֑ם אֲשֶׁר֩ נָתַ֨ן לָהֶ֜ם מֹשֶׁ֗ה בְּעֵ֤בֶר הַיַּרְדֵּן֙ מִזְרָ֔חָה כַּאֲשֶׁר֙ נָתַ֣ן לָהֶ֔ם מֹשֶׁ֖ה עֶ֥בֶד יְהוָֽה׃ (ט) מֵעֲרוֹעֵ֡ר אֲשֶׁר֩ עַל־ שְׂפַת־ נַ֨חַל אַרְנ֜וֹן וְהָעִ֨יר אֲשֶׁ֧ר בְּתוֹךְ־ הַנַּ֛חַל וְכָל־ הַמִּישֹׁ֥ר מֵידְבָ֖א עַד־ דִּיבֽוֹן׃ (י) וְכֹ֗ל עָרֵי֙ סִיחוֹן֙ מֶ֣לֶךְ הָאֱמֹרִ֔י אֲשֶׁ֥ר מָלַ֖ךְ בְּחֶשְׁבּ֑וֹן עַד־ גְּב֖וּל בְּנֵ֥י עַמּֽוֹן׃ (יא) וְהַגִּלְעָ֞ד וּגְב֧וּל הַגְּשׁוּרִ֣י וְהַמַּעֲכָתִ֗י וְכֹ֨ל הַ֥ר חֶרְמ֛וֹן וְכָל־ הַבָּשָׁ֖ן עַד־ סַלְכָֽה׃ (יב) כָּל־ מַמְלְכ֥וּת עוֹג֙ בַּבָּשָׁ֔ן אֲשֶׁר־ מָלַ֥ךְ בְּעַשְׁתָּר֖וֹת וּבְאֶדְרֶ֑עִי ה֤וּא נִשְׁאַר֙ מִיֶּ֣תֶר הָרְפָאִ֔ים וַיַּכֵּ֥ם מֹשֶׁ֖ה וַיֹּרִשֵֽׁם׃ (יג) וְלֹ֤א הוֹרִ֙ישׁוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶת־ הַגְּשׁוּרִ֖י וְאֶת־ הַמַּעֲכָתִ֑י וַיֵּ֨שֶׁב גְּשׁ֤וּר וּמַֽעֲכָת֙ בְּקֶ֣רֶב יִשְׂרָאֵ֔ל עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (יד) רַ֚ק לְשֵׁ֣בֶט הַלֵּוִ֔י לֹ֥א נָתַ֖ן נַחֲלָ֑ה אִשֵּׁ֨י יְהוָ֜ה אֱלֹהֵ֤י יִשְׂרָאֵל֙ ה֣וּא נַחֲלָת֔וֹ כַּאֲשֶׁ֖ר דִּבֶּר־ לֽוֹ׃ (טו) וַיִּתֵּ֣ן מֹשֶׁ֔ה לְמַטֵּ֥ה בְנֵֽי־ רְאוּבֵ֖ן לְמִשְׁפְּחֹתָֽם׃ (טז) וַיְהִ֨י לָהֶ֜ם הַגְּב֗וּל מֵעֲרוֹעֵ֡ר אֲשֶׁר֩ עַל־ שְׂפַת־ נַ֨חַל אַרְנ֜וֹן וְהָעִ֨יר אֲשֶׁ֧ר בְּתוֹךְ־ הַנַּ֛חַל וְכָל־ הַמִּישֹׁ֖ר עַל־ מֵידְבָֽא׃ (יז) חֶשְׁבּ֥וֹן וְכָל־ עָרֶ֖יהָ אֲשֶׁ֣ר בַּמִּישֹׁ֑ר דִּיבוֹן֙ וּבָמ֣וֹת בַּ֔עַל וּבֵ֖ית בַּ֥עַל מְעֽוֹן׃ (יח) וְיַ֥הְצָה וּקְדֵמֹ֖ת וּמֵפָֽעַת׃ (יט) וְקִרְיָתַ֣יִם וְשִׂבְמָ֔ה וְצֶ֥רֶת הַשַּׁ֖חַר בְּהַ֥ר הָעֵֽמֶק׃ (כ) וּבֵ֥ית פְּע֛וֹר וְאַשְׁדּ֥וֹת הַפִּסְגָּ֖ה וּבֵ֥ית הַיְשִׁמֽוֹת׃ (כא) וְכֹל֙ עָרֵ֣י הַמִּישֹׁ֔ר וְכָֽל־ מַמְלְכ֗וּת סִיחוֹן֙ מֶ֣לֶךְ הָאֱמֹרִ֔י אֲשֶׁ֥ר מָלַ֖ךְ בְּחֶשְׁבּ֑וֹן אֲשֶׁר֩ הִכָּ֨ה מֹשֶׁ֜ה אֹת֣וֹ ׀ וְאֶת־ נְשִׂיאֵ֣י מִדְיָ֗ן אֶת־ אֱוִ֤י וְאֶת־ רֶ֙קֶם֙ וְאֶת־ צ֤וּר וְאֶת־ חוּר֙ וְאֶת־ רֶ֔בַע נְסִיכֵ֣י סִיח֔וֹן יֹשְׁבֵ֖י הָאָֽרֶץ׃ (כב) וְאֶת־ בִּלְעָ֥ם בֶּן־ בְּע֖וֹר הַקּוֹסֵ֑ם הָרְג֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל בַּחֶ֖רֶב אֶל־ חַלְלֵיהֶֽם׃ (כג) וַיְהִ֗י גְּבוּל֙ בְּנֵ֣י רְאוּבֵ֔ן הַיַּרְדֵּ֖ן וּגְב֑וּל זֹ֣את נַחֲלַ֤ת בְּנֵֽי־ רְאוּבֵן֙ לְמִשְׁפְּחֹתָ֔ם הֶעָרִ֖ים וְחַצְרֵיהֶֽן׃ (כד) וַיִּתֵּ֤ן מֹשֶׁה֙ לְמַטֵּה־ גָ֔ד לִבְנֵי־ גָ֖ד לְמִשְׁפְּחֹתָֽם׃ (כה) וַיְהִ֤י לָהֶם֙ הַגְּב֔וּל יַעְזֵר֙ וְכָל־ עָרֵ֣י הַגִּלְעָ֔ד וַחֲצִ֕י אֶ֖רֶץ בְּנֵ֣י עַמּ֑וֹן עַד־ עֲרוֹעֵ֕ר אֲשֶׁ֖ר עַל־ פְּנֵ֥י רַבָּֽה׃ (כו) וּמֵחֶשְׁבּ֛וֹן עַד־ רָמַ֥ת הַמִּצְפֶּ֖ה וּבְטֹנִ֑ים וּמִֽמַּחֲנַ֖יִם עַד־ גְּב֥וּל לִדְבִֽר׃ (כז) וּבָעֵ֡מֶק בֵּ֣ית הָרָם֩ וּבֵ֨ית נִמְרָ֜ה וְסֻכּ֣וֹת וְצָפ֗וֹן יֶ֚תֶר מַמְלְכ֗וּת סִיחוֹן֙ מֶ֣לֶךְ חֶשְׁבּ֔וֹן הַיַּרְדֵּ֖ן וּגְבֻ֑ל עַד־ קְצֵה֙ יָם־ כִּנֶּ֔רֶת עֵ֥בֶר הַיַּרְדֵּ֖ן מִזְרָֽחָה׃ (כח) זֹ֛את נַחֲלַ֥ת בְּנֵי־ גָ֖ד לְמִשְׁפְּחֹתָ֑ם הֶעָרִ֖ים וְחַצְרֵיהֶֽם׃ (כט) וַיִּתֵּ֣ן מֹשֶׁ֔ה לַחֲצִ֖י שֵׁ֣בֶט מְנַשֶּׁ֑ה וַיְהִ֗י לַחֲצִ֛י מַטֵּ֥ה בְנֵֽי־ מְנַשֶּׁ֖ה לְמִשְׁפְּחוֹתָֽם׃ (ל) וַיְהִ֣י גְבוּלָ֗ם מִמַּחֲנַ֨יִם כָּֽל־ הַבָּשָׁ֜ן כָּֽל־ מַמְלְכ֣וּת ׀ ע֣וֹג מֶֽלֶךְ־ הַבָּשָׁ֗ן וְכָל־ חַוֺּ֥ת יָאִ֛יר אֲשֶׁ֥ר בַּבָּשָׁ֖ן שִׁשִּׁ֥ים עִֽיר׃ (לא) וַחֲצִ֤י הַגִּלְעָד֙ וְעַשְׁתָּר֣וֹת וְאֶדְרֶ֔עִי עָרֵ֛י מַמְלְכ֥וּת ע֖וֹג בַּבָּשָׁ֑ן לִבְנֵ֤י מָכִיר֙ בֶּן־ מְנַשֶּׁ֔ה לַחֲצִ֥י בְנֵֽי־ מָכִ֖יר לְמִשְׁפְּחוֹתָֽם׃ (לב) אֵ֕לֶּה אֲשֶׁר־ נִחַ֥ל מֹשֶׁ֖ה בְּעַֽרְב֣וֹת מוֹאָ֑ב מֵעֵ֛בֶר לְיַרְדֵּ֥ן יְרִיח֖וֹ מִזְרָֽחָה׃ (לג) וּלְשֵׁ֙בֶט֙ הַלֵּוִ֔י לֹֽא־ נָתַ֥ן מֹשֶׁ֖ה נַחֲלָ֑ה יְהוָ֞ה אֱלֹהֵ֤י יִשְׂרָאֵל֙ ה֣וּא נַחֲלָתָ֔ם כַּאֲשֶׁ֖ר דִּבֶּ֥ר לָהֶֽם׃ תחילת פרק יד (לד) וְאֵ֛לֶּה אֲשֶׁר־ נָחֲל֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל בְּאֶ֣רֶץ כְּנָ֑עַן אֲשֶׁ֨ר נִֽחֲל֜וּ אוֹתָ֗ם אֶלְעָזָ֤ר הַכֹּהֵן֙ וִיהוֹשֻׁ֣עַ בִּן־ נ֔וּן וְרָאשֵׁ֛י אֲב֥וֹת הַמַּטּ֖וֹת לִבְנֵ֥י יִשְׂרָאֵֽל׃ (לה) בְּגוֹרַ֖ל נַחֲלָתָ֑ם כַּאֲשֶׁ֨ר צִוָּ֤ה יְהוָה֙ בְּיַד־ מֹשֶׁ֔ה לְתִשְׁעַ֥ת הַמַּטּ֖וֹת וַחֲצִ֥י הַמַּטֶּֽה׃ (לו) כִּֽי־ נָתַ֨ן מֹשֶׁ֜ה נַחֲלַ֨ת שְׁנֵ֤י הַמַּטּוֹת֙ וַחֲצִ֣י הַמַּטֶּ֔ה מֵעֵ֖בֶר לַיַּרְדֵּ֑ן וְלַ֨לְוִיִּ֔ם לֹֽא־ נָתַ֥ן נַחֲלָ֖ה בְּתוֹכָֽם׃ (לז) כִּֽי־ הָי֧וּ בְנֵֽי־ יוֹסֵ֛ף שְׁנֵ֥י מַטּ֖וֹת מְנַשֶּׁ֣ה וְאֶפְרָ֑יִם וְלֹֽא־ נָתְנוּ֩ חֵ֨לֶק לַלְוִיִּ֜ם בָּאָ֗רֶץ כִּ֤י אִם־ עָרִים֙ לָשֶׁ֔בֶת וּמִ֨גְרְשֵׁיהֶ֔ם לְמִקְנֵיהֶ֖ם וּלְקִנְיָנָֽם׃ (לח) כַּאֲשֶׁ֨ר צִוָּ֤ה יְהוָה֙ אֶת־ מֹשֶׁ֔ה כֵּ֥ן עָשׂ֖וּ בְּנֵ֣י יִשְׂרָאֵ֑ל וַֽיַּחְלְק֖וּ אֶת־ הָאָֽרֶץ׃ (לט) וַיִּגְּשׁ֨וּ בְנֵֽי־ יְהוּדָ֤ה אֶל־ יְהוֹשֻׁ֙עַ֙ בַּגִּלְגָּ֔ל וַיֹּ֣אמֶר אֵלָ֔יו כָּלֵ֥ב בֶּן־ יְפֻנֶּ֖ה הַקְּנִזִּ֑י אַתָּ֣ה יָדַ֡עְתָּ אֶֽת־ הַדָּבָר֩ אֲשֶׁר־ דִּבֶּ֨ר יְהוָ֜ה אֶל־ מֹשֶׁ֣ה אִישׁ־ הָאֱלֹהִ֗ים עַ֧ל אֹדוֹתַ֛י וְעַ֥ל אֹדוֹתֶ֖יךָ בְּקָדֵ֥שׁ בַּרְנֵֽעַ׃ (מ) בֶּן־ אַרְבָּעִ֨ים שָׁנָ֜ה אָנֹכִ֗י בִּ֠שְׁלֹחַ מֹשֶׁ֨ה עֶֽבֶד־ יְהוָ֥ה וְחַצְרֵיהֶֽן׃ (לד) בַּשְּׁפֵלָ֑ה אֶשְׁתָּא֥וֹל וְצָרְעָ֖ה וְאַשְׁנָֽה׃ (לה) וְזָנ֙וֹחַ֙ וְעֵ֣ין גַּנִּ֔ים תַּפּ֖וּחַ וְהָעֵינָֽם׃ (לו) יַרְמוּת֙ וַעֲדֻלָּ֔ם שׂוֹכֹ֖ה וַעֲזֵקָֽה׃ (לז) וְשַׁעֲרַ֙יִם֙ וַעֲדִיתַ֔יִם וְהַגְּדֵרָ֖ה וּגְדֵרֹתָ֑יִם עָרִ֥ים אַרְבַּֽע־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (לח) צְנָ֥ן וַחֲדָשָׁ֖ה וּמִגְדַּל־ גָּֽד׃ (לט) וְדִלְעָ֥ן וְהַמִּצְפֶּ֖ה וְיָקְתְאֵֽל׃ (מ) לָכִ֥ישׁ וּבָצְקַ֖ת וְעֶגְלֽוֹן׃ (מא) וְכַבּ֥וֹן וְלַחְמָ֖ס וְכִתְלִֽישׁ׃ (מב) וּגְדֵר֕וֹת בֵּית־ דָּג֥וֹן וְנַעֲמָ֖ה וּמַקֵּדָ֑ה עָרִ֥ים שֵׁשׁ־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (מג) לִבְנָ֥ה וָעֶ֖תֶר וְעָשָֽׁן׃ (מד) וְיִפְתָּ֥ח וְאַשְׁנָ֖ה וּנְצִֽיב׃ (מה) וּקְעִילָ֥ה וְאַכְזִ֖יב וּמָֽרֵאשָׁ֑ה עָרִ֥ים תֵּ֖שַׁע וְחַצְרֵיהֶֽן׃ (מו) עֶקְר֥וֹן וּבְנֹתֶ֖יהָ וַחֲצֵרֶֽיהָ׃ (מז) מֵעֶקְר֖וֹן וָיָ֑מָּה כֹּ֛ל אֲשֶׁר־ עַל־ יַ֥ד אַשְׁדּ֖וֹד וְחַצְרֵיהֶֽן׃ (מח) אַשְׁדּ֞וֹד בְּנוֹתֶ֣יהָ וַחֲצֵרֶ֗יהָ עַזָּ֥ה בְּנוֹתֶ֥יהָ וַחֲצֵרֶ֖יהָ עַד־ נַ֣חַל מִצְרָ֑יִם וְהַיָּ֥ם (הגבול) הַגָּד֖וֹל וּגְבֽוּל׃ (מט) וּבָהָ֑ר שָׁמִ֥יר וְיַתִּ֖יר וְשׂוֹכֹֽה׃ (נ) וְדַנָּ֥ה וְקִרְיַת־ סַנָּ֖ה הִ֥יא דְבִֽר׃ (נא) וַעֲנָ֥ב וְאֶשְׁתְּמֹ֖ה וְעָנִֽים׃ (נב) וְגֹ֥שֶׁן וְחֹלֹ֖ן וְגִלֹ֑ה עָרִ֥ים אַֽחַת־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (נג) אֲרַ֥ב וְרוּמָ֖ה וְאֶשְׁעָֽן׃ (נד) (וינים) וְיָנ֥וּם וּבֵית־ תַּפּ֖וּחַ וַאֲפֵֽקָה׃ (נה) וְחֻמְטָ֗ה וְקִרְיַ֥ת אַרְבַּ֛ע הִ֥יא חֶבְר֖וֹן וְצִיעֹ֑ר עָרִ֥ים תֵּ֖שַׁע וְחַצְרֵיהֶֽן׃ (נו) מָע֥וֹן ׀ כַּרְמֶ֖ל וָזִ֥יף וְיוּטָּֽה׃ (נז) וְיִזְרְעֶ֥אל וְיָקְדְעָ֖ם וְזָנֽוֹחַ׃ (נח) הַקַּ֖יִן גִּבְעָ֣ה וְתִמְנָ֑ה עָרִ֥ים עֶ֖שֶׂר וְחַצְרֵיהֶֽן׃ (נט) חַלְח֥וּל בֵּֽית־ צ֖וּר וּגְדֽוֹר׃ (ס) וּמַעֲרָ֥ת וּבֵית־ עֲנ֖וֹת וְאֶלְתְּקֹ֑ן עָרִ֥ים שֵׁ֖שׁ וְחַצְרֵיהֶֽן׃ (סא) קִרְיַת־ בַּ֗עַל הִ֛יא קִרְיַ֥ת יְעָרִ֖ים וְהָֽרַבָּ֑ה עָרִ֥ים שְׁתַּ֖יִם וְחַצְרֵיהֶֽן׃ (סב) בַּמִּדְבָּ֑ר בֵּ֚ית הָעֲרָבָ֔ה מִדִּ֖ין וּסְכָכָֽה׃ (סג) וְהַנִּבְשָׁ֥ן וְעִיר־ הַמֶּ֖לַח וְעֵ֣ין גֶּ֑דִי עָרִ֥ים שֵׁ֖שׁ וְחַצְרֵיהֶֽן׃ (סד) וְאֶת־ הַיְבוּסִי֙ יוֹשְׁבֵ֣י יְרֽוּשָׁלִַ֔ם לֹֽא־ (יוכלו) יָכְל֥וּ בְנֵֽי־ יְהוּדָ֖ה לְהֽוֹרִישָׁ֑ם וַיֵּ֨שֶׁב הַיְבוּסִ֜י אֶת־ בְּנֵ֤י יְהוּדָה֙ בִּיר֣וּשָׁלִַ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (סה) וַיֵּצֵ֨א הַגּוֹרָ֜ל לִבְנֵ֤י יוֹסֵף֙ מִיַּרְדֵּ֣ן יְרִיח֔וֹ לְמֵ֥י יְרִיח֖וֹ מִזְרָ֑חָה הַמִּדְבָּ֗ר עֹלֶ֧ה מִירִיח֛וֹ בָּהָ֖ר בֵּֽית־ אֵֽל׃ (סו) וְיָצָ֥א מִבֵּֽית־ אֵ֖ל ל֑וּזָה וְעָבַ֛ר אֶל־ גְּב֥וּל הָאַרְכִּ֖י עֲטָרֽוֹת׃ (סז) וְיָֽרַד־ יָ֜מָּה אֶל־ גְּב֣וּל הַיַּפְלֵטִ֗י עַ֣ד גְּב֧וּל בֵּית־ חוֹרֹ֛ן תַּחְתּ֖וֹן וְעַד־ גָּ֑זֶר וְהָי֥וּ (תצאתו) תֹצְאֹתָ֖יו יָֽמָּה׃ (סח) וַיִּנְחֲל֥וּ בְנֵי־ יוֹסֵ֖ף מְנַשֶּׁ֥ה וְאֶפְרָֽיִם׃ (סט) וַיְהִ֛י גְּב֥וּל בְּנֵֽי־ אֶפְרַ֖יִם לְמִשְׁפְּחֹתָ֑ם וַיְהִ֞י גְּב֤וּל נַחֲלָתָם֙ מִזְרָ֔חָה עַטְר֣וֹת אַדָּ֔ר עַד־ בֵּ֥ית חוֹרֹ֖ן עֶלְיֽוֹן׃ (ע) וְיָצָ֨א הַגְּב֜וּל הַיָּ֗מָּה הַֽמִּכְמְתָת֙ מִצָּפ֔וֹן וְנָסַ֧ב הַגְּב֛וּל מִזְרָ֖חָה תַּאֲנַ֣ת שִׁלֹ֑ה וְעָבַ֣ר אוֹת֔וֹ מִמִּזְרַ֖ח יָנֽוֹחָה׃ (עא) וְיָרַ֥ד מִיָּנ֖וֹחָה עֲטָר֣וֹת וְנַעֲרָ֑תָה וּפָגַע֙ בִּֽירִיח֔וֹ וְיָצָ֖א הַיַּרְדֵּֽן׃ (עב) מִתַּפּ֜וּחַ יֵלֵ֨ךְ הַגְּב֥וּל יָ֙מָּה֙ נַ֣חַל קָנָ֔ה וְהָי֥וּ תֹצְאֹתָ֖יו הַיָּ֑מָּה זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵי־ אֶפְרַ֖יִם לְמִשְׁפְּחֹתָֽם׃ (עג) וְהֶעָרִ֗ים הַמִּבְדָּלוֹת֙ לִבְנֵ֣י אֶפְרַ֔יִם בְּת֖וֹךְ נַחֲלַ֣ת בְּנֵֽי־ מְנַשֶּׁ֑ה כָּֽל־ הֶעָרִ֖ים וְחַצְרֵיהֶֽן׃ (עד) וְלֹ֣א הוֹרִ֔ישׁוּ אֶת־ הַֽכְּנַעֲנִ֖י הַיּוֹשֵׁ֣ב בְּגָ֑זֶר וַיֵּ֨שֶׁב הַֽכְּנַעֲנִ֜י בְּקֶ֤רֶב אֶפְרַ֙יִם֙ עַד־ הַיּ֣וֹם הַזֶּ֔ה וַיְהִ֖י לְמַס־ עֹבֵֽד׃ (עה) וַיְהִ֤י הַגּוֹרָל֙ לְמַטֵּ֣ה מְנַשֶּׁ֔ה כִּי־ ה֖וּא בְּכ֣וֹר יוֹסֵ֑ף לְמָכִיר֩ בְּכ֨וֹר מְנַשֶּׁ֜ה אֲבִ֣י הַגִּלְעָ֗ד כִּ֣י ה֤וּא הָיָה֙ אִ֣ישׁ מִלְחָמָ֔ה וַֽיְהִי־ ל֖וֹ הַגִּלְעָ֥ד וְהַבָּשָֽׁן׃ (עו) וַ֠יְהִי לִבְנֵ֨י מְנַשֶּׁ֥ה הַנּוֹתָרִים֮ לְמִשְׁפְּחֹתָם֒ לִבְנֵ֨י אֲבִיעֶ֜זֶר וְלִבְנֵי־ חֵ֗לֶק וְלִבְנֵ֤י אַשְׂרִיאֵל֙ וְלִבְנֵי־ שֶׁ֔כֶם וְלִבְנֵי־ חֵ֖פֶר וְלִבְנֵ֣י שְׁמִידָ֑ע אֵ֠לֶּה בְּנֵ֨י מְנַשֶּׁ֧ה בֶּן־ יוֹסֵ֛ף הַזְּכָרִ֖ים לְמִשְׁפְּחֹתָֽם׃ (עז) וְלִצְלָפְחָד֩ בֶּן־ חֵ֨פֶר בֶּן־ גִּלְעָ֜ד בֶּן־ מָכִ֣יר בֶּן־ מְנַשֶּׁ֗ה לֹא־ הָ֥יוּ ל֛וֹ בָּנִ֖ים כִּ֣י אִם־ בָּנ֑וֹת וְאֵ֙לֶּה֙ שְׁמ֣וֹת בְּנֹתָ֔יו מַחְלָ֣ה וְנֹעָ֔ה
