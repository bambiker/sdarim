יהושע פרק יז לפי החלוקה לסדרים

פרק יז בספר יהושע (18 פסוקים) נמצא ביהושע סדר ט (פסוקים עה–עז) וביהושע סדר י (פסוקים א–טו).

יהושע סדר ט

(א) וְשֵׁ֨ם חֶבְר֤וֹן לְפָנִים֙ קִרְיַ֣ת אַרְבַּ֔ע הָאָדָ֧ם הַגָּד֛וֹל בָּעֲנָקִ֖ים ה֑וּא וְהָאָ֥רֶץ שָׁקְטָ֖ה מִמִּלְחָמָֽה׃ (ב) וַיְהִ֣י הַגּוֹרָ֗ל לְמַטֵּ֛ה בְּנֵ֥י יְהוּדָ֖ה לְמִשְׁפְּחֹתָ֑ם אֶל־ גְּב֨וּל אֱד֧וֹם מִדְבַּר־ צִ֛ן נֶ֖גְבָּה מִקְצֵ֥ה תֵימָֽן׃ (ג) וַיְהִ֤י לָהֶם֙ גְּב֣וּל נֶ֔גֶב מִקְצֵ֖ה יָ֣ם הַמֶּ֑לַח מִן־ הַלָּשֹׁ֖ן הַפֹּנֶ֥ה נֶֽגְבָּה׃ (ד) וְ֠יָצָא אֶל־ מִנֶּ֜גֶב לְמַעֲלֵ֤ה עַקְרַבִּים֙ וְעָ֣בַר צִ֔נָה וְעָלָ֥ה מִנֶּ֖גֶב לְקָדֵ֣שׁ בַּרְנֵ֑עַ וְעָבַ֤ר חֶצְרוֹן֙ וְעָלָ֣ה אַדָּ֔רָה וְנָסַ֖ב הַקַּרְקָֽעָה׃ (ה) וְעָבַ֣ר עַצְמ֗וֹנָה וְיָצָא֙ נַ֣חַל מִצְרַ֔יִם (והיה) וְהָי֛וּ תֹּצְא֥וֹת הַגְּב֖וּל יָ֑מָּה זֶה־ יִהְיֶ֥ה לָכֶ֖ם גְּב֥וּל נֶֽגֶב׃ (ו) וּגְב֥וּל קֵ֙דְמָה֙ יָ֣ם הַמֶּ֔לַח עַד־ קְצֵ֖ה הַיַּרְדֵּ֑ן וּגְב֞וּל לִפְאַ֤ת צָפ֙וֹנָה֙ מִלְּשׁ֣וֹן הַיָּ֔ם מִקְצֵ֖ה הַיַּרְדֵּֽן׃ (ז) וְעָלָ֤ה הַגְּבוּל֙ בֵּ֣ית חָגְלָ֔ה וְעָבַ֕ר מִצְּפ֖וֹן לְבֵ֣ית הָעֲרָבָ֑ה וְעָלָ֣ה הַגְּב֔וּל אֶ֥בֶן בֹּ֖הַן בֶּן־ רְאוּבֵֽן׃ (ח) וְעָלָ֨ה הַגְּב֥וּל ׀ דְּבִרָה֮ מֵעֵ֣מֶק עָכוֹר֒ וְצָפ֜וֹנָה פֹּנֶ֣ה אֶל־ הַגִּלְגָּ֗ל אֲשֶׁר־ נֹ֙כַח֙ לְמַעֲלֵ֣ה אֲדֻמִּ֔ים אֲשֶׁ֥ר מִנֶּ֖גֶב לַנָּ֑חַל וְעָבַ֤ר הַגְּבוּל֙ אֶל־ מֵי־ עֵ֣ין שֶׁ֔מֶשׁ וְהָי֥וּ תֹצְאֹתָ֖יו אֶל־ עֵ֥ין רֹגֵֽל׃ (ט) וְעָלָ֨ה הַגְּב֜וּל גֵּ֣י בֶן־ הִנֹּ֗ם אֶל־ כֶּ֤תֶף הַיְבוּסִי֙ מִנֶּ֔גֶב הִ֖יא יְרֽוּשָׁלִָ֑ם וְעָלָ֨ה הַגְּב֜וּל אֶל־ רֹ֣אשׁ הָהָ֗ר אֲ֠שֶׁר עַל־ פְּנֵ֤י גֵֽי־ הִנֹּם֙ יָ֔מָּה אֲשֶׁ֛ר בִּקְצֵ֥ה עֵֽמֶק־ רְפָאִ֖ים צָפֹֽנָה׃ (י) וְתָאַ֨ר הַגְּב֜וּל מֵרֹ֣אשׁ הָהָ֗ר אֶל־ מַעְיַן֙ מֵ֣י נֶפְתּ֔וֹחַ וְיָצָ֖א אֶל־ עָרֵ֣י הַר־ עֶפְר֑וֹן וְתָאַ֤ר הַגְּבוּל֙ בַּעֲלָ֔ה הִ֖יא קִרְיַ֥ת יְעָרִֽים׃ (יא) וְנָסַב֩ הַגְּב֨וּל מִבַּעֲלָ֥ה יָ֙מָּה֙ אֶל־ הַ֣ר שֵׂעִ֔יר וְעָבַ֕ר אֶל־ כֶּ֧תֶף הַר־ יְעָרִ֛ים מִצָּפ֖וֹנָה הִ֣יא כְסָל֑וֹן וְיָרַ֥ד בֵּֽית־ שֶׁ֖מֶשׁ וְעָבַ֥ר תִּמְנָֽה׃ (יב) וְיָצָ֨א הַגְּב֜וּל אֶל־ כֶּ֣תֶף עֶקְרוֹן֮ צָפוֹנָה֒ וְתָאַ֤ר הַגְּבוּל֙ שִׁכְּר֔וֹנָה וְעָבַ֥ר הַר־ הַֽבַּעֲלָ֖ה וְיָצָ֣א יַבְנְאֵ֑ל וְהָי֛וּ תֹּצְא֥וֹת הַגְּב֖וּל יָֽמָּה׃ (יג) וּגְב֣וּל יָ֔ם הַיָּ֥מָּה הַגָּד֖וֹל וּגְב֑וּל זֶ֠ה גְּב֧וּל בְּנֵֽי־ יְהוּדָ֛ה סָבִ֖יב לְמִשְׁפְּחֹתָֽם׃ (יד) וּלְכָלֵ֣ב בֶּן־ יְפֻנֶּ֗ה נָ֤תַן חֵ֙לֶק֙ בְּת֣וֹךְ בְּנֵֽי־ יְהוּדָ֔ה אֶל־ פִּ֥י יְהוָ֖ה לִֽיהוֹשֻׁ֑עַ אֶת־ קִרְיַ֥ת אַרְבַּ֛ע אֲבִ֥י הָעֲנָ֖ק הִ֥יא חֶבְרֽוֹן׃ (טו) וַיֹּ֤רֶשׁ מִשָּׁם֙ כָּלֵ֔ב אֶת־ שְׁלוֹשָׁ֖ה בְּנֵ֣י הָעֲנָ֑ק אֶת־ שֵׁשַׁ֤י וְאֶת־ אֲחִימַן֙ וְאֶת־ תַּלְמַ֔י יְלִידֵ֖י הָעֲנָֽק׃ (טז) וַיַּ֣עַל מִשָּׁ֔ם אֶל־ יֹשְׁבֵ֖י דְּבִ֑ר וְשֵׁם־ דְּבִ֥ר לְפָנִ֖ים קִרְיַת־ סֵֽפֶר׃ (יז) וַיֹּ֣אמֶר כָּלֵ֔ב אֲשֶׁר־ יַכֶּ֥ה אֶת־ קִרְיַת־ סֵ֖פֶר וּלְכָדָ֑הּ וְנָתַ֥תִּי ל֛וֹ אֶת־ עַכְסָ֥ה בִתִּ֖י לְאִשָּֽׁה׃ (יח) וַֽיִּלְכְּדָ֛הּ עָתְנִיאֵ֥ל בֶּן־ קְנַ֖ז אֲחִ֣י כָלֵ֑ב וַיִּתֶּן־ ל֛וֹ אֶת־ עַכְסָ֥ה בִתּ֖וֹ לְאִשָּֽׁה׃ (יט) וַיְהִ֣י בְּבוֹאָ֗הּ וַתְּסִיתֵ֙הוּ֙ לִשְׁא֤וֹל מֵֽאֵת־ אָבִ֙יהָ֙ שָׂדֶ֔ה וַתִּצְנַ֖ח מֵעַ֣ל הַחֲמ֑וֹר וַיֹּֽאמֶר־ לָ֥הּ כָּלֵ֖ב מַה־ לָּֽךְ׃ (כ) וַתֹּ֜אמֶר תְּנָה־ לִּ֣י בְרָכָ֗ה כִּ֣י אֶ֤רֶץ הַנֶּ֙גֶב֙ נְתַתָּ֔נִי וְנָתַתָּ֥ה לִ֖י גֻּלֹּ֣ת מָ֑יִם וַיִּתֶּן־ לָ֗הּ אֵ֚ת גֻּלֹּ֣ת עִלִּיּ֔וֹת וְאֵ֖ת גֻּלֹּ֥ת תַּחְתִּיּֽוֹת׃ (כא) זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵי־ יְהוּדָ֖ה לְמִשְׁפְּחֹתָֽם׃ (כב) וַיִּֽהְי֣וּ הֶעָרִ֗ים מִקְצֵה֙ לְמַטֵּ֣ה בְנֵֽי־ יְהוּדָ֔ה אֶל־ גְּב֥וּל אֱד֖וֹם בַּנֶּ֑גְבָּה קַבְצְאֵ֥ל וְעֵ֖דֶר וְיָגֽוּר׃ (כג) וְקִינָ֥ה וְדִֽימוֹנָ֖ה וְעַדְעָדָֽה׃ (כד) וְקֶ֥דֶשׁ וְחָצ֖וֹר וְיִתְנָֽן׃ (כה) זִ֥יף וָטֶ֖לֶם וּבְעָלֽוֹת׃ (כו) וְחָצ֤וֹר ׀ חֲדַתָּה֙ וּקְרִיּ֔וֹת חֶצְר֖וֹן הִ֥יא חָצֽוֹר׃ (כז) אֲמָ֥ם וּשְׁמַ֖ע וּמוֹלָדָֽה׃ (כח) וַחֲצַ֥ר גַּדָּ֛ה וְחֶשְׁמ֖וֹן וּבֵ֥ית פָּֽלֶט׃ (כט) וַחֲצַ֥ר שׁוּעָ֛ל וּבְאֵ֥ר שֶׁ֖בַע וּבִזְיוֹתְיָֽה׃ (ל) בַּעֲלָ֥ה וְעִיִּ֖ים וָעָֽצֶם׃ (לא) וְאֶלְתּוֹלַ֥ד וּכְסִ֖יל וְחָרְמָֽה׃ (לב) וְצִֽקְלַ֥ג וּמַדְמַנָּ֖ה וְסַנְסַנָּֽה׃ (לג) וּלְבָא֥וֹת וְשִׁלְחִ֖ים וְעַ֣יִן וְרִמּ֑וֹן כָּל־ עָרִ֛ים עֶשְׂרִ֥ים וָתֵ֖שַׁע וְחַצְרֵיהֶֽן׃ (לד) בַּשְּׁפֵלָ֑ה אֶשְׁתָּא֥וֹל וְצָרְעָ֖ה וְאַשְׁנָֽה׃ (לה) וְזָנ֙וֹחַ֙ וְעֵ֣ין גַּנִּ֔ים תַּפּ֖וּחַ וְהָעֵינָֽם׃ (לו) יַרְמוּת֙ וַעֲדֻלָּ֔ם שׂוֹכֹ֖ה וַעֲזֵקָֽה׃ (לז) וְשַׁעֲרַ֙יִם֙ וַעֲדִיתַ֔יִם וְהַגְּדֵרָ֖ה וּגְדֵרֹתָ֑יִם עָרִ֥ים אַרְבַּֽע־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (לח) צְנָ֥ן וַחֲדָשָׁ֖ה וּמִגְדַּל־ גָּֽד׃ (לט) וְדִלְעָ֥ן וְהַמִּצְפֶּ֖ה וְיָקְתְאֵֽל׃ (מ) לָכִ֥ישׁ וּבָצְקַ֖ת וְעֶגְלֽוֹן׃ (מא) וְכַבּ֥וֹן וְלַחְמָ֖ס וְכִתְלִֽישׁ׃ (מב) וּגְדֵר֕וֹת בֵּית־ דָּג֥וֹן וְנַעֲמָ֖ה וּמַקֵּדָ֑ה עָרִ֥ים שֵׁשׁ־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (מג) לִבְנָ֥ה וָעֶ֖תֶר וְעָשָֽׁן׃ (מד) וְיִפְתָּ֥ח וְאַשְׁנָ֖ה וּנְצִֽיב׃ (מה) וּקְעִילָ֥ה וְאַכְזִ֖יב וּמָֽרֵאשָׁ֑ה עָרִ֥ים תֵּ֖שַׁע וְחַצְרֵיהֶֽן׃ (מו) עֶקְר֥וֹן וּבְנֹתֶ֖יהָ וַחֲצֵרֶֽיהָ׃ (מז) מֵעֶקְר֖וֹן וָיָ֑מָּה כֹּ֛ל אֲשֶׁר־ עַל־ יַ֥ד אַשְׁדּ֖וֹד וְחַצְרֵיהֶֽן׃ (מח) אַשְׁדּ֞וֹד בְּנוֹתֶ֣יהָ וַחֲצֵרֶ֗יהָ עַזָּ֥ה בְּנוֹתֶ֥יהָ וַחֲצֵרֶ֖יהָ עַד־ נַ֣חַל מִצְרָ֑יִם וְהַיָּ֥ם (הגבול) הַגָּד֖וֹל וּגְבֽוּל׃ (מט) וּבָהָ֑ר שָׁמִ֥יר וְיַתִּ֖יר וְשׂוֹכֹֽה׃ (נ) וְדַנָּ֥ה וְקִרְיַת־ סַנָּ֖ה הִ֥יא דְבִֽר׃ (נא) וַעֲנָ֥ב וְאֶשְׁתְּמֹ֖ה וְעָנִֽים׃ (נב) וְגֹ֥שֶׁן וְחֹלֹ֖ן וְגִלֹ֑ה עָרִ֥ים אַֽחַת־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (נג) אֲרַ֥ב וְרוּמָ֖ה וְאֶשְׁעָֽן׃ (נד) (וינים) וְיָנ֥וּם וּבֵית־ תַּפּ֖וּחַ וַאֲפֵֽקָה׃ (נה) וְחֻמְטָ֗ה וְקִרְיַ֥ת אַרְבַּ֛ע הִ֥יא חֶבְר֖וֹן וְצִיעֹ֑ר עָרִ֥ים תֵּ֖שַׁע וְחַצְרֵיהֶֽן׃ (נו) מָע֥וֹן ׀ כַּרְמֶ֖ל וָזִ֥יף וְיוּטָּֽה׃ (נז) וְיִזְרְעֶ֥אל וְיָקְדְעָ֖ם וְזָנֽוֹחַ׃ (נח) הַקַּ֖יִן גִּבְעָ֣ה וְתִמְנָ֑ה עָרִ֥ים עֶ֖שֶׂר וְחַצְרֵיהֶֽן׃ (נט) חַלְח֥וּל בֵּֽית־ צ֖וּר וּגְדֽוֹר׃ (ס) וּמַעֲרָ֥ת וּבֵית־ עֲנ֖וֹת וְאֶלְתְּקֹ֑ן עָרִ֥ים שֵׁ֖שׁ וְחַצְרֵיהֶֽן׃ (סא) קִרְיַת־ בַּ֗עַל הִ֛יא קִרְיַ֥ת יְעָרִ֖ים וְהָֽרַבָּ֑ה עָרִ֥ים שְׁתַּ֖יִם וְחַצְרֵיהֶֽן׃ (סב) בַּמִּדְבָּ֑ר בֵּ֚ית הָעֲרָבָ֔ה מִדִּ֖ין וּסְכָכָֽה׃ (סג) וְהַנִּבְשָׁ֥ן וְעִיר־ הַמֶּ֖לַח וְעֵ֣ין גֶּ֑דִי עָרִ֥ים שֵׁ֖שׁ וְחַצְרֵיהֶֽן׃ (סד) וְאֶת־ הַיְבוּסִי֙ יוֹשְׁבֵ֣י יְרֽוּשָׁלִַ֔ם לֹֽא־ (יוכלו) יָכְל֥וּ בְנֵֽי־ יְהוּדָ֖ה לְהֽוֹרִישָׁ֑ם וַיֵּ֨שֶׁב הָאָ֔רֶץ אֲשֶׁר֙ נָתַ֣ן לָכֶ֔ם יְהוָ֖ה אֱלֹהֵ֥י אֲבֽוֹתֵיכֶֽם׃ (יט) הָב֥וּ לָכֶ֛ם שְׁלֹשָׁ֥ה אֲנָשִׁ֖ים לַשָּׁ֑בֶט וְאֶשְׁלָחֵ֗ם וְיָקֻ֜מוּ וְיִֽתְהַלְּכ֥וּ בָאָ֛רֶץ וְיִכְתְּב֥וּ אוֹתָ֛הּ לְפִ֥י נַֽחֲלָתָ֖ם וְיָבֹ֥אוּ אֵלָֽי׃ (כ) וְהִֽתְחַלְּק֥וּ אֹתָ֖הּ לְשִׁבְעָ֣ה חֲלָקִ֑ים יְהוּדָ֞ה יַעֲמֹ֤ד עַל־ גְּבוּלוֹ֙ מִנֶּ֔גֶב וּבֵ֥ית יוֹסֵ֛ף יַעַמְד֥וּ עַל־ גְּבוּלָ֖ם מִצָּפֽוֹן׃ (כא) וְאַתֶּ֞ם תִּכְתְּב֤וּ אֶת־ הָאָ֙רֶץ֙ שִׁבְעָ֣ה חֲלָקִ֔ים וַֽהֲבֵאתֶ֥ם אֵלַ֖י הֵ֑נָּה וְיָרִ֨יתִי לָכֶ֤ם גּוֹרָל֙ פֹּ֔ה לִפְנֵ֖י יְהוָ֥ה אֱלֹהֵֽינוּ׃ (כב) כִּ֠י אֵֽין־ חֵ֤לֶק לַלְוִיִּם֙ בְּקִרְבְּכֶ֔ם כִּֽי־ כְהֻנַּ֥ת יְהוָ֖ה נַחֲלָת֑וֹ וְגָ֡ד וּרְאוּבֵ֡ן וַחֲצִי֩ שֵׁ֨בֶט הַֽמְנַשֶּׁ֜ה לָקְח֣וּ נַחֲלָתָ֗ם מֵעֵ֤בֶר לַיַּרְדֵּן֙ מִזְרָ֔חָה אֲשֶׁר֙ נָתַ֣ן לָהֶ֔ם מֹשֶׁ֖ה עֶ֥בֶד יְהוָֽה׃ (כג) וַיָּקֻ֥מוּ הָאֲנָשִׁ֖ים וַיֵּלֵ֑כוּ וַיְצַ֣ו יְהוֹשֻׁ֡עַ אֶת־ הַהֹלְכִים֩ לִכְתֹּ֨ב אֶת־ הָאָ֜רֶץ לֵאמֹ֗ר לְ֠כוּ וְהִתְהַלְּכ֨וּ בָאָ֜רֶץ וְכִתְב֤וּ אוֹתָהּ֙ וְשׁ֣וּבוּ אֵלַ֔י וּ֠פֹה אַשְׁלִ֨יךְ לָכֶ֥ם גּוֹרָ֛ל לִפְנֵ֥י יְהוָ֖ה בְּשִׁלֹֽה׃ (כד) וַיֵּלְכ֤וּ הָֽאֲנָשִׁים֙ וַיַּעַבְר֣וּ בָאָ֔רֶץ וַיִּכְתְּב֧וּהָ לֶֽעָרִ֛ים לְשִׁבְעָ֥ה חֲלָקִ֖ים עַל־ סֵ֑פֶר וַיָּבֹ֧אוּ אֶל־ יְהוֹשֻׁ֛עַ אֶל־ הַֽמַּחֲנֶ֖ה שִׁלֹֽה׃ (כה) וַיַּשְׁלֵךְ֩ לָהֶ֨ם יְהוֹשֻׁ֧עַ גּוֹרָ֛ל בְּשִׁלֹ֖ה לִפְנֵ֣י יְהוָ֑ה וַיְחַלֶּק־ שָׁ֨ם יְהוֹשֻׁ֧עַ אֶת־ הָאָ֛רֶץ לִבְנֵ֥י יִשְׂרָאֵ֖ל כְּמַחְלְקֹתָֽם׃ (כו) וַיַּ֗עַל גּוֹרַ֛ל מַטֵּ֥ה בְנֵֽי־ בִנְיָמִ֖ן לְמִשְׁפְּחֹתָ֑ם וַיֵּצֵא֙ גְּב֣וּל גּֽוֹרָלָ֔ם בֵּ֚ין בְּנֵ֣י יְהוּדָ֔ה וּבֵ֖ין בְּנֵ֥י יוֹסֵֽף׃ (כז) וַיְהִ֨י לָהֶ֧ם הַגְּב֛וּל לִפְאַ֥ת צָפ֖וֹנָה מִן־ הַיַּרְדֵּ֑ן וְעָלָ֣ה הַגְּבוּל֩ אֶל־ כֶּ֨תֶף יְרִיח֜וֹ מִצָּפ֗וֹן וְעָלָ֤ה בָהָר֙ יָ֔מָּה (והיה) וְהָיוּ֙ תֹּֽצְאֹתָ֔יו מִדְבַּ֖רָה בֵּ֥ית אָֽוֶן׃ (כח) וְעָבַר֩ מִשָּׁ֨ם הַגְּב֜וּל ל֗וּזָה אֶל־ כֶּ֤תֶף ל֙וּזָה֙ נֶ֔גְבָּה הִ֖יא בֵּֽית־ אֵ֑ל וְיָרַ֤ד הַגְּבוּל֙ עַטְר֣וֹת אַדָּ֔ר עַל־ הָהָ֕ר אֲשֶׁ֛ר מִנֶּ֥גֶב לְבֵית־ חֹר֖וֹן תַּחְתּֽוֹן׃ (כט) וְתָאַ֣ר הַגְּבוּל֩ וְנָסַ֨ב לִפְאַת־ יָ֜ם נֶ֗גְבָּה מִן־ הָהָר֙ אֲשֶׁ֨ר עַל־ פְּנֵ֥י בֵית־ חֹרוֹן֮ נֶגְבָּה֒ (והיה) וְהָי֣וּ תֹֽצְאֹתָ֗יו אֶל־ קִרְיַת־ בַּ֙עַל֙ הִ֚יא קִרְיַ֣ת יְעָרִ֔ים עִ֖יר בְּנֵ֣י יְהוּדָ֑ה זֹ֖את פְּאַת־ יָֽם׃ (ל) וּפְאַת־ נֶ֕גְבָּה מִקְצֵ֖ה קִרְיַ֣ת יְעָרִ֑ים וְיָצָ֤א הַגְּבוּל֙ יָ֔מָּה וְיָצָ֕א אֶל־ מַעְיַ֖ן מֵ֥י נֶפְתּֽוֹחַ׃ (לא) וְיָרַ֨ד הַגְּב֜וּל אֶל־ קְצֵ֣ה הָהָ֗ר אֲשֶׁר֙ עַל־ פְּנֵי֙ גֵּ֣י בֶן־ הִנֹּ֔ם אֲשֶׁ֛ר בְּעֵ֥מֶק רְפָאִ֖ים צָפ֑וֹנָה וְיָרַד֩ גֵּ֨י הִנֹּ֜ם אֶל־ כֶּ֤תֶף הַיְבוּסִי֙ נֶ֔גְבָּה וְיָרַ֖ד עֵ֥ין רֹגֵֽל׃ (לב) וְתָאַ֣ר מִצָּפ֗וֹן וְיָצָא֙ עֵ֣ין שֶׁ֔מֶשׁ וְיָצָא֙ אֶל־ גְּלִיל֔וֹת אֲשֶׁר־ נֹ֖כַח מַעֲלֵ֣ה אֲדֻמִּ֑ים וְיָרַ֕ד אֶ֥בֶן בֹּ֖הַן בֶּן־ רְאוּבֵֽן׃ (לג) וְעָבַ֛ר אֶל־ כֶּ֥תֶף מוּל־ הָֽעֲרָבָ֖ה צָפ֑וֹנָה וְיָרַ֖ד הָעֲרָבָֽתָה׃ (לד) וְעָבַ֨ר הַגְּב֜וּל אֶל־ כֶּ֣תֶף בֵּית־ חָגְלָה֮ צָפוֹנָה֒ (והיה) וְהָי֣וּ ׀ (תצאותיו) תֹּצְא֣וֹת הַגְּב֗וּל אֶל־ לְשׁ֤וֹן יָם־ הַמֶּ֙לַח֙ צָפ֔וֹנָה אֶל־ קְצֵ֥ה הַיַּרְדֵּ֖ן נֶ֑גְבָּה זֶ֖ה גְּב֥וּל נֶֽגֶב׃ (לה) וְהַיַּרְדֵּ֥ן יִגְבֹּל־ אֹת֖וֹ לִפְאַת־ קֵ֑דְמָה זֹ֡את נַחֲלַת֩ בְּנֵ֨י בִנְיָמִ֧ן לִגְבֽוּלֹתֶ֛יהָ סָבִ֖יב לְמִשְׁפְּחֹתָֽם׃ (לו) וְהָי֣וּ הֶֽעָרִ֗ים לְמַטֵּ֛ה בְּנֵ֥י בִנְיָמִ֖ן לְמִשְׁפְּחֽוֹתֵיהֶ֑ם יְרִיח֥וֹ וּבֵית־ חָגְלָ֖ה וְעֵ֥מֶק קְצִֽיץ׃ (לז) וּבֵ֧ית הָֽעֲרָבָ֛ה וּצְמָרַ֖יִם וּבֵֽית־ אֵֽל׃ (לח) וְהָעַוִּ֥ים וְהַפָּרָ֖ה וְעָפְרָֽה׃ (לט) וּכְפַ֧ר (העמני) הָֽעַמֹּנָ֛ה וְהָֽעָפְנִ֖י וָגָ֑בַע עָרִ֥ים שְׁתֵּים־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (מ) גִּבְע֥וֹן וְהָֽרָמָ֖ה וּבְאֵרֽוֹת׃ (מא) וְהַמִּצְפֶּ֥ה וְהַכְּפִירָ֖ה וְהַמֹּצָֽה׃ (מב) וְרֶ֥קֶם וְיִרְפְּאֵ֖ל וְתַרְאֲלָֽה׃
