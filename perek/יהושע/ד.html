יהושע פרק ד לפי החלוקה לסדרים

פרק ד בספר יהושע (24 פסוקים) נמצא ביהושע סדר ב (פסוקים יב–לד) וביהושע סדר ג (פסוקים א–א).

יהושע סדר ב

(א) וַיֹּ֤אמֶר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ הַיּ֣וֹם הַזֶּ֗ה אָחֵל֙ גַּדֶּלְךָ֔ בְּעֵינֵ֖י כָּל־ יִשְׂרָאֵ֑ל אֲשֶׁר֙ יֵֽדְע֔וּן כִּ֗י כַּאֲשֶׁ֥ר הָיִ֛יתִי עִם־ מֹשֶׁ֖ה אֶהְיֶ֥ה עִמָּֽךְ׃ (ב) וְאַתָּ֗ה תְּצַוֶּה֙ אֶת־ הַכֹּ֣הֲנִ֔ים נֹשְׂאֵ֥י אֲרֽוֹן־ הַבְּרִ֖ית לֵאמֹ֑ר כְּבֹאֲכֶ֗ם עַד־ קְצֵה֙ מֵ֣י הַיַּרְדֵּ֔ן בַּיַּרְדֵּ֖ן תַּעֲמֹֽדוּ׃ (ג) וַיֹּ֥אמֶר יְהוֹשֻׁ֖עַ אֶל־ בְּנֵ֣י יִשְׂרָאֵ֑ל גֹּ֣שׁוּ הֵ֔נָּה וְשִׁמְע֕וּ אֶת־ דִּבְרֵ֖י יְהוָ֥ה אֱלֹֽהֵיכֶֽם׃ (ד) וַיֹּ֣אמֶר יְהוֹשֻׁ֔עַ בְּזֹאת֙ תֵּֽדְע֔וּן כִּ֛י אֵ֥ל חַ֖י בְּקִרְבְּכֶ֑ם וְהוֹרֵ֣שׁ יוֹרִ֣ישׁ מִ֠פְּנֵיכֶם אֶת־ הַכְּנַעֲנִ֨י וְאֶת־ הַחִתִּ֜י וְאֶת־ הַחִוִּ֗י וְאֶת־ הַפְּרִזִּי֙ וְאֶת־ הַגִּרְגָּשִׁ֔י וְהָאֱמֹרִ֖י וְהַיְבוּסִֽי׃ (ה) הִנֵּה֙ אֲר֣וֹן הַבְּרִ֔ית אֲד֖וֹן כָּל־ הָאָ֑רֶץ עֹבֵ֥ר לִפְנֵיכֶ֖ם בַּיַּרְדֵּֽן׃ (ו) וְעַתָּ֗ה קְח֤וּ לָכֶם֙ שְׁנֵ֣י עָשָׂ֣ר אִ֔ישׁ מִשִּׁבְטֵ֖י יִשְׂרָאֵ֑ל אִישׁ־ אֶחָ֥ד אִישׁ־ אֶחָ֖ד לַשָּֽׁבֶט׃ (ז) וְהָיָ֡ה כְּנ֣וֹחַ כַּפּ֣וֹת רַגְלֵ֣י הַכֹּהֲנִ֡ים נֹשְׂאֵי֩ אֲר֨וֹן יְהוָ֜ה אֲד֤וֹן כָּל־ הָאָ֙רֶץ֙ בְּמֵ֣י הַיַּרְדֵּ֔ן מֵ֤י הַיַּרְדֵּן֙ יִכָּ֣רֵת֔וּן הַמַּ֥יִם הַיֹּרְדִ֖ים מִלְמָ֑עְלָה וְיַעַמְד֖וּ נֵ֥ד אֶחָֽד׃ (ח) וַיְהִ֗י בִּנְסֹ֤עַ הָעָם֙ מֵאָ֣הֳלֵיהֶ֔ם לַעֲבֹ֖ר אֶת־ הַיַּרְדֵּ֑ן וְהַכֹּהֲנִ֗ים נֹֽשְׂאֵ֛י הָאָר֥וֹן הַבְּרִ֖ית לִפְנֵ֥י הָעָֽם׃ (ט) וּכְב֞וֹא נֹשְׂאֵ֤י הָֽאָרוֹן֙ עַד־ הַיַּרְדֵּ֔ן וְרַגְלֵ֤י הַכֹּֽהֲנִים֙ נֹשְׂאֵ֣י הָֽאָר֔וֹן נִטְבְּל֖וּ בִּקְצֵ֣ה הַמָּ֑יִם וְהַיַּרְדֵּ֗ן מָלֵא֙ עַל־ כָּל־ גְּדוֹתָ֔יו כֹּ֖ל יְמֵ֥י קָצִֽיר׃ (י) וַיַּעַמְד֡וּ הַמַּיִם֩ הַיֹּרְדִ֨ים מִלְמַ֜עְלָה קָ֣מוּ נֵד־ אֶחָ֗ד הַרְחֵ֨ק מְאֹ֜ד (באדם) מֵֽאָדָ֤ם הָעִיר֙ אֲשֶׁר֙ מִצַּ֣ד צָֽרְתָ֔ן וְהַיֹּרְדִ֗ים עַ֣ל יָ֧ם הָעֲרָבָ֛ה יָם־ הַמֶּ֖לַח תַּ֣מּוּ נִכְרָ֑תוּ וְהָעָ֥ם עָבְר֖וּ נֶ֥גֶד יְרִיחֽוֹ׃ (יא) וַיַּעַמְד֣וּ הַכֹּהֲנִ֡ים נֹ֠שְׂאֵי הָאָר֨וֹן בְּרִית־ יְהוָ֜ה בֶּחָֽרָבָ֛ה בְּת֥וֹךְ הַיַּרְדֵּ֖ן הָכֵ֑ן וְכָל־ יִשְׂרָאֵ֗ל עֹֽבְרִים֙ בֶּחָ֣רָבָ֔ה עַ֤ד אֲשֶׁר־ תַּ֙מּוּ֙ כָּל־ הַגּ֔וֹי לַעֲבֹ֖ר אֶת־ הַיַּרְדֵּֽן׃ תחילת פרק ד (יב) וַיְהִי֙ כַּאֲשֶׁר־ תַּ֣מּוּ כָל־ הַגּ֔וֹי לַעֲב֖וֹר אֶת־ הַיַּרְדֵּ֑ן וַיֹּ֣אמֶר יְהוָ֔ה אֶל־ יְהוֹשֻׁ֖עַ לֵאמֹֽר׃ (יג) קְח֤וּ לָכֶם֙ מִן־ הָעָ֔ם שְׁנֵ֥ים עָשָׂ֖ר אֲנָשִׁ֑ים אִישׁ־ אֶחָ֥ד אִישׁ־ אֶחָ֖ד מִשָּֽׁבֶט׃ (יד) וְצַוּ֣וּ אוֹתָם֮ לֵאמֹר֒ שְׂאֽוּ־ לָכֶ֨ם מִזֶּ֜ה מִתּ֣וֹךְ הַיַּרְדֵּ֗ן מִמַּצַּב֙ רַגְלֵ֣י הַכֹּהֲנִ֔ים הָכִ֖ין שְׁתֵּים־ עֶשְׂרֵ֣ה אֲבָנִ֑ים וְהַעֲבַרְתֶּ֤ם אוֹתָם֙ עִמָּכֶ֔ם וְהִנַּחְתֶּ֣ם אוֹתָ֔ם בַּמָּל֕וֹן אֲשֶׁר־ תָּלִ֥ינוּ ב֖וֹ הַלָּֽיְלָה׃ (טו) וַיִּקְרָ֣א יְהוֹשֻׁ֗עַ אֶל־ שְׁנֵ֤ים הֶֽעָשָׂר֙ אִ֔ישׁ אֲשֶׁ֥ר הֵכִ֖ין מִבְּנֵ֣י יִשְׂרָאֵ֑ל אִישׁ־ אֶחָ֥ד אִישׁ־ אֶחָ֖ד מִשָּֽׁבֶט׃ (טז) וַיֹּ֤אמֶר לָהֶם֙ יְהוֹשֻׁ֔עַ עִ֠בְרוּ לִפְנֵ֨י אֲר֧וֹן יְהוָ֛ה אֱלֹֽהֵיכֶ֖ם אֶל־ תּ֣וֹךְ הַיַּרְדֵּ֑ן וְהָרִ֨ימוּ לָכֶ֜ם אִ֣ישׁ אֶ֤בֶן אַחַת֙ עַל־ שִׁכְמ֔וֹ לְמִסְפַּ֖ר שִׁבְטֵ֥י בְנֵי־ יִשְׂרָאֵֽל׃ (יז) לְמַ֗עַן תִּֽהְיֶ֛ה זֹ֥את א֖וֹת בְּקִרְבְּכֶ֑ם כִּֽי־ יִשְׁאָל֨וּן בְּנֵיכֶ֤ם מָחָר֙ לֵאמֹ֔ר מָ֛ה הָאֲבָנִ֥ים הָאֵ֖לֶּה לָכֶֽם׃ (יח) וַאֲמַרְתֶּ֣ם לָהֶ֗ם אֲשֶׁ֨ר נִכְרְת֜וּ מֵימֵ֤י הַיַּרְדֵּן֙ מִפְּנֵי֙ אֲר֣וֹן בְּרִית־ יְהוָ֔ה בְּעָבְרוֹ֙ בַּיַּרְדֵּ֔ן נִכְרְת֖וּ מֵ֣י הַיַּרְדֵּ֑ן וְ֠הָיוּ הָאֲבָנִ֨ים הָאֵ֧לֶּה לְזִכָּר֛וֹן לִבְנֵ֥י יִשְׂרָאֵ֖ל עַד־ עוֹלָֽם׃ (יט) וַיַּעֲשׂוּ־ כֵ֣ן בְּנֵי־ יִשְׂרָאֵ֘ל כַּאֲשֶׁ֣ר צִוָּ֣ה יְהוֹשֻׁעַ֒ וַיִּשְׂא֡וּ שְׁתֵּֽי־ עֶשְׂרֵ֨ה אֲבָנִ֜ים מִתּ֣וֹךְ הַיַּרְדֵּ֗ן כַּאֲשֶׁ֨ר דִּבֶּ֤ר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ לְמִסְפַּ֖ר שִׁבְטֵ֣י בְנֵֽי־ יִשְׂרָאֵ֑ל וַיַּעֲבִר֤וּם עִמָּם֙ אֶל־ הַמָּל֔וֹן וַיַּנִּח֖וּם שָֽׁם׃ (כ) וּשְׁתֵּ֧ים עֶשְׂרֵ֣ה אֲבָנִ֗ים הֵקִ֣ים יְהוֹשֻׁעַ֮ בְּת֣וֹךְ הַיַּרְדֵּן֒ תַּ֗חַת מַצַּב֙ רַגְלֵ֣י הַכֹּהֲנִ֔ים נֹשְׂאֵ֖י אֲר֣וֹן הַבְּרִ֑ית וַיִּ֣הְיוּ שָׁ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כא) וְהַכֹּהֲנִ֞ים נֹשְׂאֵ֣י הָאָר֗וֹן עֹמְדִים֮ בְּת֣וֹךְ הַיַּרְדֵּן֒ עַ֣ד תֹּ֣ם כָּֽל־ הַ֠דָּבָר אֲשֶׁר־ צִוָּ֨ה יְהוָ֤ה אֶת־ יְהוֹשֻׁ֙עַ֙ לְדַבֵּ֣ר אֶל־ הָעָ֔ם כְּכֹ֛ל אֲשֶׁר־ צִוָּ֥ה מֹשֶׁ֖ה אֶת־ יְהוֹשֻׁ֑עַ וַיְמַהֲר֥וּ הָעָ֖ם וַֽיַּעֲבֹֽרוּ׃ (כב) וַיְהִ֛י כַּֽאֲשֶׁר־ תַּ֥ם כָּל־ הָעָ֖ם לַֽעֲב֑וֹר וַיַּעֲבֹ֧ר אֲרוֹן־ יְהוָ֛ה וְהַכֹּהֲנִ֖ים לִפְנֵ֥י הָעָֽם׃ (כג) וַ֠יַּעַבְרוּ בְּנֵי־ רְאוּבֵ֨ן וּבְנֵי־ גָ֜ד וַחֲצִ֨י שֵׁ֤בֶט הַֽמְנַשֶּׁה֙ חֲמֻשִׁ֔ים לִפְנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל כַּאֲשֶׁ֛ר דִּבֶּ֥ר אֲלֵיהֶ֖ם מֹשֶֽׁה׃ (כד) כְּאַרְבָּעִ֥ים אֶ֖לֶף חֲלוּצֵ֣י הַצָּבָ֑א עָבְר֞וּ לִפְנֵ֤י יְהוָה֙ לַמִּלְחָמָ֔ה אֶ֖ל עַֽרְב֥וֹת יְרִיחֽוֹ׃ (כה) בַּיּ֣וֹם הַה֗וּא גִּדַּ֤ל יְהוָה֙ אֶת־ יְהוֹשֻׁ֔עַ בְּעֵינֵ֖י כָּל־ יִשְׂרָאֵ֑ל וַיִּֽרְא֣וּ אֹת֔וֹ כַּאֲשֶׁ֛ר יָרְא֥וּ אֶת־ מֹשֶׁ֖ה כָּל־ יְמֵ֥י חַיָּֽיו׃ (כו) וַיֹּ֣אמֶר יְהוָ֔ה אֶל־ יְהוֹשֻׁ֖עַ לֵאמֹֽר׃ (כז) צַוֵּה֙ אֶת־ הַכֹּ֣הֲנִ֔ים נֹשְׂאֵ֖י אֲר֣וֹן הָעֵד֑וּת וְיַעֲל֖וּ מִן־ הַיַּרְדֵּֽן׃ (כח) וַיְצַ֣ו יְהוֹשֻׁ֔עַ אֶת־ הַכֹּהֲנִ֖ים לֵאמֹ֑ר עֲל֖וּ מִן־ הַיַּרְדֵּֽן׃ (כט) וַ֠יְהִי (בעלות) כַּעֲל֨וֹת הַכֹּהֲנִ֜ים נֹשְׂאֵ֨י אֲר֤וֹן בְּרִית־ יְהוָה֙ מִתּ֣וֹךְ הַיַּרְדֵּ֔ן נִתְּק֗וּ כַּפּוֹת֙ רַגְלֵ֣י הַכֹּהֲנִ֔ים אֶ֖ל הֶחָרָבָ֑ה וַיָּשֻׁ֤בוּ מֵֽי־ הַיַּרְדֵּן֙ לִמְקוֹמָ֔ם וַיֵּלְכ֥וּ כִתְמוֹל־ שִׁלְשׁ֖וֹם עַל־ כָּל־ גְּדוֹתָֽיו׃ (ל) וְהָעָ֗ם עָלוּ֙ מִן־ הַיַּרְדֵּ֔ן בֶּעָשׂ֖וֹר לַחֹ֣דֶשׁ הָרִאשׁ֑וֹן וַֽיַּחֲנוּ֙ בַּגִּלְגָּ֔ל בִּקְצֵ֖ה מִזְרַ֥ח יְרִיחֽוֹ׃ (לא) וְאֵת֩ שְׁתֵּ֨ים עֶשְׂרֵ֤ה הָֽאֲבָנִים֙ הָאֵ֔לֶּה אֲשֶׁ֥ר לָקְח֖וּ מִן־ הַיַּרְדֵּ֑ן הֵקִ֥ים יְהוֹשֻׁ֖עַ בַּגִּלְגָּֽל׃ (לב) וַיֹּ֛אמֶר אֶל־ בְּנֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֑ר אֲשֶׁר֩ יִשְׁאָל֨וּן בְּנֵיכֶ֤ם מָחָר֙ אֶת־ אֲבוֹתָ֣ם לֵאמֹ֔ר מָ֖ה הָאֲבָנִ֥ים הָאֵֽלֶּה׃ (לג) וְהוֹדַעְתֶּ֖ם אֶת־ בְּנֵיכֶ֣ם לֵאמֹ֑ר בַּיַּבָּשָׁה֙ עָבַ֣ר יִשְׂרָאֵ֔ל אֶת־ הַיַּרְדֵּ֖ן הַזֶּֽה׃ (לד) אֲשֶׁר־ הוֹבִישׁ֩ יְהוָ֨ה אֱלֹהֵיכֶ֜ם אֶת־ מֵ֧י הַיַּרְדֵּ֛ן מִפְּנֵיכֶ֖ם עַֽד־ עָבְרְכֶ֑ם כַּאֲשֶׁ֣ר עָשָׂה֩ יְהוָ֨ה אֱלֹהֵיכֶ֧ם לְיַם־ ס֛וּף אֲשֶׁר־ הוֹבִ֥ישׁ מִפָּנֵ֖ינוּ עַד־ עָבְרֵֽנוּ׃

יהושע סדר ג

(א) לְ֠מַעַן דַּ֜עַת כָּל־ עַמֵּ֤י הָאָ֙רֶץ֙ אֶת־ יַ֣ד יְהוָ֔ה כִּ֥י חֲזָקָ֖ה הִ֑יא לְמַ֧עַן יְרָאתֶ֛ם אֶת־ יְהוָ֥ה אֱלֹהֵיכֶ֖ם כָּל־ הַיָּמִֽים׃ סוף פרק ד (ב) וַיְהִ֣י כִשְׁמֹ֣עַ כָּל־ מַלְכֵ֣י הָאֱמֹרִ֡י אֲשֶׁר֩ בְּעֵ֨בֶר הַיַּרְדֵּ֜ן יָ֗מָּה וְכָל־ מַלְכֵ֤י הַֽכְּנַעֲנִי֙ אֲשֶׁ֣ר עַל־ הַיָּ֔ם אֵ֠ת אֲשֶׁר־ הוֹבִ֨ישׁ יְהוָ֜ה אֶת־ מֵ֧י הַיַּרְדֵּ֛ן מִפְּנֵ֥י בְנֵֽי־ יִשְׂרָאֵ֖ל עַד־ (עברנו) עָבְרָ֑ם וַיִּמַּ֣ס לְבָבָ֗ם וְלֹא־ הָ֨יָה בָ֥ם עוֹד֙ ר֔וּחַ מִפְּנֵ֖י בְּנֵֽי־ יִשְׂרָאֵֽל׃ (ג) בָּעֵ֣ת הַהִ֗יא אָמַ֤ר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ עֲשֵׂ֥ה לְךָ֖ חַֽרְב֣וֹת צֻרִ֑ים וְשׁ֛וּב מֹ֥ל אֶת־ בְּנֵֽי־ יִשְׂרָאֵ֖ל שֵׁנִֽית׃ (ד) וַיַּעַשׂ־ ל֥וֹ יְהוֹשֻׁ֖עַ חַֽרְב֣וֹת צֻרִ֑ים וַיָּ֙מָל֙ אֶת־ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶל־ גִּבְעַ֖ת הָעֲרָלֽוֹת׃ (ה) וְזֶ֥ה הַדָּבָ֖ר אֲשֶׁר־ מָ֣ל יְהוֹשֻׁ֑עַ כָּל־ הָעָ֣ם הַיֹּצֵא֩ מִמִּצְרַ֨יִם הַזְּכָרִ֜ים כֹּ֣ל ׀ אַנְשֵׁ֣י הַמִּלְחָמָ֗ה מֵ֤תוּ בַמִּדְבָּר֙ בַּדֶּ֔רֶךְ בְּצֵאתָ֖ם מִמִּצְרָֽיִם׃ (ו) כִּֽי־ מֻלִ֣ים הָי֔וּ כָּל־ הָעָ֖ם הַיֹּֽצְאִ֑ים וְכָל־ הָ֠עָם הַיִּלֹּדִ֨ים בַּמִּדְבָּ֥ר בַּדֶּ֛רֶךְ בְּצֵאתָ֥ם מִמִּצְרַ֖יִם לֹא־ מָֽלוּ׃ (ז) כִּ֣י ׀ אַרְבָּעִ֣ים שָׁנָ֗ה הָלְכ֣וּ בְנֵֽי־ יִשְׂרָאֵל֮ בַּמִּדְבָּר֒ עַד־ תֹּ֨ם כָּל־ הַגּ֜וֹי אַנְשֵׁ֤י הַמִּלְחָמָה֙ הַיֹּצְאִ֣ים מִמִּצְרַ֔יִם אֲשֶׁ֥ר לֹֽא־ שָׁמְע֖וּ בְּק֣וֹל יְהוָ֑ה אֲשֶׁ֨ר נִשְׁבַּ֤ע יְהוָה֙ לָהֶ֔ם לְבִלְתִּ֞י הַרְאוֹתָ֣ם אֶת־ הָאָ֗רֶץ אֲשֶׁר֩ נִשְׁבַּ֨ע יְהוָ֤ה לַֽאֲבוֹתָם֙ לָ֣תֶת לָ֔נוּ אֶ֛רֶץ זָבַ֥ת חָלָ֖ב וּדְבָֽשׁ׃ (ח) וְאֶת־ בְּנֵיהֶם֙ הֵקִ֣ים תַּחְתָּ֔ם אֹתָ֖ם מָ֣ל יְהוֹשֻׁ֑עַ כִּי־ עֲרֵלִ֣ים הָי֔וּ כִּ֛י לֹא־ מָ֥לוּ אוֹתָ֖ם בַּדָּֽרֶךְ׃ (ט) וַיְהִ֛י כַּאֲשֶׁר־ תַּ֥מּוּ כָל־ הַגּ֖וֹי לְהִמּ֑וֹל וַיֵּשְׁב֥וּ תַחְתָּ֛ם בַּֽמַּחֲנֶ֖ה עַ֥ד חֲיוֹתָֽם׃ (י) וַיֹּ֤אמֶר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ הַיּ֗וֹם גַּלּ֛וֹתִי אֶת־ חֶרְפַּ֥ת מִצְרַ֖יִם מֵעֲלֵיכֶ֑ם וַיִּקְרָ֞א שֵׁ֣ם הַמָּק֤וֹם הַהוּא֙ גִּלְגָּ֔ל עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (יא) וַיַּחֲנ֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל בַּגִּלְגָּ֑ל וַיַּעֲשׂ֣וּ אֶת־ הַפֶּ֡סַח בְּאַרְבָּעָה֩ עָשָׂ֨ר י֥וֹם לַחֹ֛דֶשׁ בָּעֶ֖רֶב בְּעַֽרְב֥וֹת יְרִיחֽוֹ׃ (יב) וַיֹּ֨אכְל֜וּ מֵעֲב֥וּר הָאָ֛רֶץ מִמָּֽחֳרַ֥ת הַפֶּ֖סַח מַצּ֣וֹת וְקָל֑וּי בְּעֶ֖צֶם הַיּ֥וֹם הַזֶּֽה׃ (יג) וַיִּשְׁבֹּ֨ת הַמָּ֜ן מִֽמָּחֳרָ֗ת בְּאָכְלָם֙ מֵעֲב֣וּר הָאָ֔רֶץ וְלֹא־ הָ֥יָה ע֛וֹד לִבְנֵ֥י יִשְׂרָאֵ֖ל מָ֑ן וַיֹּאכְל֗וּ מִתְּבוּאַת֙ אֶ֣רֶץ כְּנַ֔עַן בַּשָּׁנָ֖ה הַהִֽיא׃ (יד) וַיְהִ֗י בִּֽהְי֣וֹת יְהוֹשֻׁעַ֮ בִּירִיחוֹ֒ וַיִּשָּׂ֤א עֵינָיו֙ וַיַּ֔רְא וְהִנֵּה־ אִישׁ֙ עֹמֵ֣ד לְנֶגְדּ֔וֹ וְחַרְבּ֥וֹ שְׁלוּפָ֖ה בְּיָד֑וֹ וַיֵּ֨לֶךְ יְהוֹשֻׁ֤עַ אֵלָיו֙ וַיֹּ֣אמֶר ל֔וֹ הֲלָ֥נוּ אַתָּ֖ה אִם־ לְצָרֵֽינוּ׃ (טו) וַיֹּ֣אמֶר ׀ לֹ֗א כִּ֛י אֲנִ֥י שַׂר־ צְבָֽא־ יְהוָ֖ה עַתָּ֣ה בָ֑אתִי וַיִּפֹּל֩ יְהוֹשֻׁ֨עַ אֶל־ פָּנָ֥יו אַ֙רְצָה֙ וַיִּשְׁתָּ֔חוּ וַיֹּ֣אמֶר ל֔וֹ מָ֥ה אֲדֹנִ֖י מְדַבֵּ֥ר אֶל־ עַבְדּֽוֹ׃ (טז) וַיֹּאמֶר֩ שַׂר־ צְבָ֨א יְהוָ֜ה אֶל־ יְהוֹשֻׁ֗עַ שַׁל־ נַֽעַלְךָ֙ מֵעַ֣ל רַגְלֶ֔ךָ כִּ֣י הַמָּק֗וֹם אֲשֶׁ֥ר אַתָּ֛ה עֹמֵ֥ד עָלָ֖יו קֹ֣דֶשׁ ה֑וּא וַיַּ֥עַשׂ יְהוֹשֻׁ֖עַ כֵּֽן׃ (יז) וִֽירִיחוֹ֙ סֹגֶ֣רֶת וּמְסֻגֶּ֔רֶת מִפְּנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל אֵ֥ין יוֹצֵ֖א וְאֵ֥ין בָּֽא׃ (יח) וַיֹּ֤אמֶר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ רְאֵה֙ נָתַ֣תִּי בְיָֽדְךָ֔ אֶת־ יְרִיח֖וֹ וְאֶת־ מַלְכָּ֑הּ גִּבּוֹרֵ֖י הֶחָֽיִל׃ (יט) וְסַבֹּתֶ֣ם אֶת־ הָעִ֗יר כֹּ֚ל אַנְשֵׁ֣י הַמִּלְחָמָ֔ה הַקֵּ֥יף אֶת־ הָעִ֖יר פַּ֣עַם אֶחָ֑ת כֹּ֥ה תַעֲשֶׂ֖ה שֵׁ֥שֶׁת יָמִֽים׃ (כ) וְשִׁבְעָ֣ה כֹהֲנִ֡ים יִשְׂאוּ֩ שִׁבְעָ֨ה שׁוֹפְר֤וֹת הַיּֽוֹבְלִים֙ לִפְנֵ֣י הָאָר֔וֹן וּבַיּוֹם֙ הַשְּׁבִיעִ֔י תָּסֹ֥בּוּ אֶת־ הָעִ֖יר שֶׁ֣בַע פְּעָמִ֑ים וְהַכֹּ֣הֲנִ֔ים יִתְקְע֖וּ בַּשּׁוֹפָרֽוֹת׃ (כא) וְהָיָ֞ה בִּמְשֹׁ֣ךְ ׀ בְּקֶ֣רֶן הַיּוֹבֵ֗ל (בשמעכם) כְּשָׁמְעֲכֶם֙ אֶת־ ק֣וֹל הַשּׁוֹפָ֔ר יָרִ֥יעוּ כָל־ הָעָ֖ם תְּרוּעָ֣ה גְדוֹלָ֑ה וְנָ֨פְלָ֜ה חוֹמַ֤ת הָעִיר֙ תַּחְתֶּ֔יהָ וְעָל֥וּ הָעָ֖ם אִ֥ישׁ נֶגְדּֽוֹ׃ (כב) וַיִּקְרָ֞א יְהוֹשֻׁ֤עַ בִּן־ נוּן֙ אֶל־ הַכֹּ֣הֲנִ֔ים וַיֹּ֣אמֶר אֲלֵהֶ֔ם שְׂא֖וּ אֶת־ אֲר֣וֹן הַבְּרִ֑ית וְשִׁבְעָ֣ה כֹֽהֲנִ֗ים יִשְׂאוּ֙ שִׁבְעָ֤ה שֽׁוֹפְרוֹת֙ יוֹבְלִ֔ים לִפְנֵ֖י אֲר֥וֹן יְהוָֽה׃ (כג) (ויאמרו) וַיֹּ֙אמֶר֙ אֶל־ הָעָ֔ם עִבְר֖וּ וְסֹ֣בּוּ אֶת־ הָעִ֑יר וְהֶ֣חָל֔וּץ יַעֲבֹ֕ר לִפְנֵ֖י אֲר֥וֹן יְהוָֽה׃ (כד) וַיְהִ֗י כֶּאֱמֹ֣ר יְהוֹשֻׁעַ֮ אֶל־ הָעָם֒ וְשִׁבְעָ֣ה הַכֹּהֲנִ֡ים נֹשְׂאִים֩ שִׁבְעָ֨ה שׁוֹפְר֤וֹת הַיּֽוֹבְלִים֙ לִפְנֵ֣י יְהוָ֔ה עָבְר֕וּ וְתָקְע֖וּ בַּשּֽׁוֹפָר֑וֹת וַֽאֲרוֹן֙ בְּרִ֣ית יְהוָ֔ה הֹלֵ֖ךְ אַחֲרֵיהֶֽם׃ (כה) וְהֶחָל֣וּץ הֹלֵ֔ךְ לִפְנֵי֙ הַכֹּ֣הֲנִ֔ים (תקעו) תֹּקְעֵ֖י הַשּֽׁוֹפָר֑וֹת וְהַֽמְאַסֵּ֗ף הֹלֵךְ֙ אַחֲרֵ֣י הָאָר֔וֹן הָל֖וֹךְ וְתָק֥וֹעַ בַּשּׁוֹפָרֽוֹת׃ (כו) וְאֶת־ הָעָם֩ צִוָּ֨ה יְהוֹשֻׁ֜עַ לֵאמֹ֗ר לֹ֤א תָרִ֙יעוּ֙ וְלֹֽא־ תַשְׁמִ֣יעוּ אֶת־ קוֹלְכֶ֔ם וְלֹא־ יֵצֵ֥א מִפִּיכֶ֖ם דָּבָ֑ר עַ֠ד י֣וֹם אָמְרִ֧י אֲלֵיכֶ֛ם הָרִ֖יעוּ וַהֲרִיעֹתֶֽם׃ (כז) וַיַּסֵּ֤ב אֲרוֹן־ יְהוָה֙ אֶת־ הָעִ֔יר הַקֵּ֖ף פַּ֣עַם אֶחָ֑ת וַיָּבֹ֙אוּ֙ הַֽמַּחֲנֶ֔ה וַיָּלִ֖ינוּ בַּֽמַּחֲנֶֽה׃ (כח) וַיַּשְׁכֵּ֥ם יְהוֹשֻׁ֖עַ בַּבֹּ֑קֶר וַיִּשְׂא֥וּ הַכֹּהֲנִ֖ים אֶת־ אֲר֥וֹן יְהוָֽה׃ (כט) וְשִׁבְעָ֣ה הַכֹּהֲנִ֡ים נֹשְׂאִים֩ שִׁבְעָ֨ה שׁוֹפְר֜וֹת הַיֹּבְלִ֗ים לִפְנֵי֙ אֲר֣וֹן יְהוָ֔ה הֹלְכִ֣ים הָל֔וֹךְ וְתָקְע֖וּ בַּשּׁוֹפָר֑וֹת וְהֶחָלוּץ֙ הֹלֵ֣ךְ לִפְנֵיהֶ֔ם וְהַֽמְאַסֵּ֗ף הֹלֵךְ֙ אַֽחֲרֵי֙ אֲר֣וֹן יְהוָ֔ה (הולך) הָל֖וֹךְ וְתָק֥וֹעַ בַּשּׁוֹפָרֽוֹת׃ (ל) וַיָּסֹ֨בּוּ אֶת־ הָעִ֜יר בַּיּ֤וֹם הַשֵּׁנִי֙ פַּ֣עַם אַחַ֔ת וַיָּשֻׁ֖בוּ הַֽמַּחֲנֶ֑ה כֹּ֥ה עָשׂ֖וּ שֵׁ֥שֶׁת יָמִֽים׃ (לא) וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁבִיעִ֗י וַיַּשְׁכִּ֙מוּ֙ כַּעֲל֣וֹת הַשַּׁ֔חַר וַיָּסֹ֧בּוּ אֶת־ הָעִ֛יר כַּמִּשְׁפָּ֥ט הַזֶּ֖ה שֶׁ֣בַע פְּעָמִ֑ים רַ֚ק בַּיּ֣וֹם הַה֔וּא סָבְב֥וּ אֶת־ הָעִ֖יר שֶׁ֥בַע פְּעָמִֽים׃ (לב) וַיְהִי֙ בַּפַּ֣עַם הַשְּׁבִיעִ֔ית תָּקְע֥וּ הַכֹּהֲנִ֖ים בַּשּׁוֹפָר֑וֹת וַיֹּ֨אמֶר יְהוֹשֻׁ֤עַ אֶל־ הָעָם֙ הָרִ֔יעוּ כִּֽי־ נָתַ֧ן יְהוָ֛ה לָכֶ֖ם אֶת־ הָעִֽיר׃ (לג) וְהָיְתָ֨ה הָעִ֥יר חֵ֛רֶם הִ֥יא וְכָל־ אֲשֶׁר־ בָּ֖הּ לַֽיהוָ֑ה רַק֩ רָחָ֨ב הַזּוֹנָ֜ה תִּֽחְיֶ֗ה הִ֚יא וְכָל־ אֲשֶׁ֣ר אִתָּ֣הּ בַּבַּ֔יִת כִּ֣י הֶחְבְּאַ֔תָה אֶת־ הַמַּלְאָכִ֖ים אֲשֶׁ֥ר שָׁלָֽחְנוּ׃ (לד) וְרַק־ אַתֶּם֙ שִׁמְר֣וּ מִן־ הַחֵ֔רֶם פֶּֽן־ תַּחֲרִ֖ימוּ וּלְקַחְתֶּ֣ם מִן־ הַחֵ֑רֶם וְשַׂמְתֶּ֞ם אֶת־ מַחֲנֵ֤ה יִשְׂרָאֵל֙ לְחֵ֔רֶם וַעֲכַרְתֶּ֖ם אוֹתֽוֹ׃ (לה) וְכֹ֣ל ׀ כֶּ֣סֶף וְזָהָ֗ב וּכְלֵ֤י נְחֹ֙שֶׁת֙ וּבַרְזֶ֔ל קֹ֥דֶשׁ ה֖וּא לַֽיהוָ֑ה אוֹצַ֥ר יְהוָ֖ה יָבֽוֹא׃ (לו) וַיָּ֣רַע הָעָ֔ם וַֽיִּתְקְע֖וּ בַּשֹּֽׁפָר֑וֹת וַיְהִי֩ כִשְׁמֹ֨עַ הָעָ֜ם אֶת־ ק֣וֹל הַשּׁוֹפָ֗ר וַיָּרִ֤יעוּ הָעָם֙ תְּרוּעָ֣ה גְדוֹלָ֔ה וַתִּפֹּ֨ל הַֽחוֹמָ֜ה תַּחְתֶּ֗יהָ וַיַּ֨עַל הָעָ֤ם הָעִ֙ירָה֙ אִ֣ישׁ נֶגְדּ֔וֹ וַֽיִּלְכְּד֖וּ אֶת־ הָעִֽיר׃ (לז) וַֽיַּחֲרִ֙ימוּ֙ אֶת־ כָּל־ אֲשֶׁ֣ר בָּעִ֔יר מֵאִישׁ֙ וְעַד־ אִשָּׁ֔ה מִנַּ֖עַר וְעַד־ זָקֵ֑ן וְעַ֨ד שׁ֥וֹר וָשֶׂ֛ה וַחֲמ֖וֹר לְפִי־ חָֽרֶב׃ (לח) וְלִשְׁנַ֨יִם הָאֲנָשִׁ֜ים הַֽמְרַגְּלִ֤ים אֶת־ הָאָ֙רֶץ֙ אָמַ֣ר יְהוֹשֻׁ֔עַ בֹּ֖אוּ בֵּית־ הָאִשָּׁ֣ה הַזּוֹנָ֑ה וְהוֹצִ֨יאוּ מִשָּׁ֤ם אֶת־ הָֽאִשָּׁה֙ וְאֶת־ כָּל־ אֲשֶׁר־ לָ֔הּ כַּאֲשֶׁ֥ר נִשְׁבַּעְתֶּ֖ם לָֽהּ׃ (לט) וַיָּבֹ֜אוּ הַנְּעָרִ֣ים הַֽמְרַגְּלִ֗ים וַיֹּצִ֡יאוּ אֶת־ רָ֠חָב וְאֶת־ אָבִ֨יהָ וְאֶת־ אִמָּ֤הּ וְאֶת־ אַחֶ֙יהָ֙ וְאֶת־ כָּל־ אֲשֶׁר־ לָ֔הּ וְאֵ֥ת כָּל־ מִשְׁפְּחוֹתֶ֖יהָ הוֹצִ֑יאוּ וַיַּ֨נִּיח֔וּם מִח֖וּץ לְמַחֲנֵ֥ה יִשְׂרָאֵֽל׃ (מ) וְהָעִ֛יר שָׂרְפ֥וּ בָאֵ֖שׁ וְכָל־ אֲשֶׁר־ בָּ֑הּ רַ֣ק ׀ הַכֶּ֣סֶף וְהַזָּהָ֗ב וּכְלֵ֤י הַנְּחֹ֙שֶׁת֙ וְהַבַּרְזֶ֔ל נָתְנ֖וּ אוֹצַ֥ר בֵּית־ יְהוָֽה׃ (מא) וְֽאֶת־ רָחָ֣ב הַ֠זּוֹנָה וְאֶת־ בֵּ֨ית אָבִ֤יהָ וְאֶת־ כָּל־ אֲשֶׁר־ לָהּ֙ הֶחֱיָ֣ה יְהוֹשֻׁ֔עַ וַתֵּ֙שֶׁב֙ בְּקֶ֣רֶב יִשְׂרָאֵ֔ל עַ֖ד הַיּ֣וֹם הַזֶּ֑ה כִּ֤י הֶחְבִּ֙יאָה֙ אֶת־ הַמַּלְאָכִ֔ים אֲשֶׁר־ שָׁלַ֥ח יְהוֹשֻׁ֖עַ לְרַגֵּ֥ל אֶת־ יְרִיחֽוֹ׃ (מב) וַיַּשְׁבַּ֣ע יְהוֹשֻׁ֔עַ בָּעֵ֥ת הַהִ֖יא לֵאמֹ֑ר אָר֨וּר הָאִ֜ישׁ לִפְנֵ֣י יְהוָ֗ה אֲשֶׁ֤ר יָקוּם֙ וּבָנָ֞ה אֶת־ הָעִ֤יר הַזֹּאת֙ אֶת־ יְרִיח֔וֹ בִּבְכֹר֣וֹ יְיַסְּדֶ֔נָּה וּבִצְעִיר֖וֹ יַצִּ֥יב דְּלָתֶֽיהָ׃
