יהושע פרק י לפי החלוקה לסדרים

פרק י בספר יהושע (43 פסוקים) נמצא ביהושע סדר ה (פסוקים לא–לז) וביהושע סדר ו (פסוקים א–לד) וביהושע סדר ז (פסוקים א–ב).

יהושע סדר ה

(א) וְכָל־ יִשְׂרָאֵ֡ל וּזְקֵנָ֡יו וְשֹׁטְרִ֣ים ׀ וְשֹׁפְטָ֡יו עֹמְדִ֣ים מִזֶּ֣ה ׀ וּמִזֶּ֣ה ׀ לָאָר֡וֹן נֶגֶד֩ הַכֹּהֲנִ֨ים הַלְוִיִּ֜ם נֹשְׂאֵ֣י ׀ אֲר֣וֹן בְּרִית־ יְהוָ֗ה כַּגֵּר֙ כָּֽאֶזְרָ֔ח חֶצְיוֹ֙ אֶל־ מ֣וּל הַר־ גְּרִזִ֔ים וְהַֽחֶצְי֖וֹ אֶל־ מ֣וּל הַר־ עֵיבָ֑ל כַּאֲשֶׁ֨ר צִוָּ֜ה מֹשֶׁ֣ה עֶֽבֶד־ יְהוָ֗ה לְבָרֵ֛ךְ אֶת־ הָעָ֥ם יִשְׂרָאֵ֖ל בָּרִאשֹׁנָֽה׃ (ב) וְאַֽחֲרֵי־ כֵ֗ן קָרָא֙ אֶת־ כָּל־ דִּבְרֵ֣י הַתּוֹרָ֔ה הַבְּרָכָ֖ה וְהַקְּלָלָ֑ה כְּכָל־ הַכָּת֖וּב בְּסֵ֥פֶר הַתּוֹרָֽה׃ (ג) לֹֽא־ הָיָ֣ה דָבָ֔ר מִכֹּ֖ל אֲשֶׁר־ צִוָּ֣ה מֹשֶׁ֑ה אֲשֶׁ֨ר לֹֽא־ קָרָ֜א יְהוֹשֻׁ֗עַ נֶ֣גֶד כָּל־ קְהַ֤ל יִשְׂרָאֵל֙ וְהַנָּשִׁ֣ים וְהַטַּ֔ף וְהַגֵּ֖ר הַהֹלֵ֥ךְ בְּקִרְבָּֽם׃ (ד) וַיְהִ֣י כִשְׁמֹ֣עַ כָּֽל־ הַמְּלָכִ֡ים אֲשֶׁר֩ בְּעֵ֨בֶר הַיַּרְדֵּ֜ן בָּהָ֣ר וּבַשְּׁפֵלָ֗ה וּבְכֹל֙ ח֚וֹף הַיָּ֣ם הַגָּד֔וֹל אֶל־ מ֖וּל הַלְּבָנ֑וֹן הַֽחִתִּי֙ וְהָ֣אֱמֹרִ֔י הַֽכְּנַעֲנִי֙ הַפְּרִזִּ֔י הַחִוִּ֖י וְהַיְבוּסִֽי׃ (ה) וַיִּֽתְקַבְּצ֣וּ יַחְדָּ֔ו לְהִלָּחֵ֥ם עִם־ יְהוֹשֻׁ֖עַ וְעִם־ יִשְׂרָאֵ֑ל פֶּ֖ה אֶחָֽד׃ (ו) וְיֹשְׁבֵ֨י גִבְע֜וֹן שָׁמְע֗וּ אֵת֩ אֲשֶׁ֨ר עָשָׂ֧ה יְהוֹשֻׁ֛עַ לִֽירִיח֖וֹ וְלָעָֽי׃ (ז) וַיַּעֲשׂ֤וּ גַם־ הֵ֙מָּה֙ בְּעָרְמָ֔ה וַיֵּלְכ֖וּ וַיִּצְטַיָּ֑רוּ וַיִּקְח֞וּ שַׂקִּ֤ים בָּלִים֙ לַחֲמ֣וֹרֵיהֶ֔ם וְנֹאד֥וֹת יַ֙יִן֙ בָּלִ֔ים וּמְבֻקָּעִ֖ים וּמְצֹרָרִֽים׃ (ח) וּנְעָל֨וֹת בָּל֤וֹת וּמְטֻלָּאוֹת֙ בְּרַגְלֵיהֶ֔ם וּשְׂלָמ֥וֹת בָּל֖וֹת עֲלֵיהֶ֑ם וְכֹל֙ לֶ֣חֶם צֵידָ֔ם יָבֵ֖שׁ הָיָ֥ה נִקֻּדִֽים׃ (ט) וַיֵּלְכ֧וּ אֶל־ יְהוֹשֻׁ֛עַ אֶל־ הַֽמַּחֲנֶ֖ה הַגִּלְגָּ֑ל וַיֹּאמְר֨וּ אֵלָ֜יו וְאֶל־ אִ֣ישׁ יִשְׂרָאֵ֗ל מֵאֶ֤רֶץ רְחוֹקָה֙ בָּ֔אנוּ וְעַתָּ֖ה כִּרְתוּ־ לָ֥נוּ בְרִֽית׃ (י) (ויאמרו) וַיֹּ֥אמֶר אִֽישׁ־ יִשְׂרָאֵ֖ל אֶל־ הַחִוִּ֑י אוּלַ֗י בְּקִרְבִּי֙ אַתָּ֣ה יוֹשֵׁ֔ב וְאֵ֖יךְ (אכרות)־ אֶֽכְרָת־ לְךָ֥ בְרִֽית׃ (יא) וַיֹּאמְר֥וּ אֶל־ יְהוֹשֻׁ֖עַ עֲבָדֶ֣יךָ אֲנָ֑חְנוּ וַיֹּ֨אמֶר אֲלֵהֶ֧ם יְהוֹשֻׁ֛עַ מִ֥י אַתֶּ֖ם וּמֵאַ֥יִן תָּבֹֽאוּ׃ (יב) וַיֹּאמְר֣וּ אֵלָ֗יו מֵאֶ֨רֶץ רְחוֹקָ֤ה מְאֹד֙ בָּ֣אוּ עֲבָדֶ֔יךָ לְשֵׁ֖ם יְהוָ֣ה אֱלֹהֶ֑יךָ כִּֽי־ שָׁמַ֣עְנוּ שָׁמְע֔וֹ וְאֵ֛ת כָּל־ אֲשֶׁ֥ר עָשָׂ֖ה בְּמִצְרָֽיִם׃ (יג) וְאֵ֣ת ׀ כָּל־ אֲשֶׁ֣ר עָשָׂ֗ה לִשְׁנֵי֙ מַלְכֵ֣י הָאֱמֹרִ֔י אֲשֶׁ֖ר בְּעֵ֣בֶר הַיַּרְדֵּ֑ן לְסִיחוֹן֙ מֶ֣לֶךְ חֶשְׁבּ֔וֹן וּלְע֥וֹג מֶֽלֶךְ־ הַבָּשָׁ֖ן אֲשֶׁ֥ר בְּעַשְׁתָּרֽוֹת׃ (יד) וַיֹּאמְר֣וּ אֵלֵ֡ינוּ זְֽקֵינֵינוּ֩ וְכָל־ יֹשְׁבֵ֨י אַרְצֵ֜נוּ לֵאמֹ֗ר קְח֨וּ בְיֶדְכֶ֤ם צֵידָה֙ לַדֶּ֔רֶךְ וּלְכ֖וּ לִקְרָאתָ֑ם וַאֲמַרְתֶּ֤ם אֲלֵיהֶם֙ עַבְדֵיכֶ֣ם אֲנַ֔חְנוּ וְעַתָּ֖ה כִּרְתוּ־ לָ֥נוּ בְרִֽית׃ (טו) זֶ֣ה ׀ לַחְמֵ֗נוּ חָ֞ם הִצְטַיַּ֤דְנוּ אֹתוֹ֙ מִבָּ֣תֵּ֔ינוּ בְּי֥וֹם צֵאתֵ֖נוּ לָלֶ֣כֶת אֲלֵיכֶ֑ם וְעַתָּה֙ הִנֵּ֣ה יָבֵ֔שׁ וְהָיָ֖ה נִקֻּדִֽים׃ (טז) וְאֵ֨לֶּה נֹאד֤וֹת הַיַּ֙יִן֙ אֲשֶׁ֣ר מִלֵּ֣אנוּ חֲדָשִׁ֔ים וְהִנֵּ֖ה הִתְבַּקָּ֑עוּ וְאֵ֤לֶּה שַׂלְמוֹתֵ֙ינוּ֙ וּנְעָלֵ֔ינוּ בָּל֕וּ מֵרֹ֥ב הַדֶּ֖רֶךְ מְאֹֽד׃ (יז) וַיִּקְח֥וּ הָֽאֲנָשִׁ֖ים מִצֵּידָ֑ם וְאֶת־ פִּ֥י יְהוָ֖ה לֹ֥א שָׁאָֽלוּ׃ (יח) וַיַּ֨עַשׂ לָהֶ֤ם יְהוֹשֻׁ֙עַ֙ שָׁל֔וֹם וַיִּכְרֹ֥ת לָהֶ֛ם בְּרִ֖ית לְחַיּוֹתָ֑ם וַיִּשָּׁבְע֣וּ לָהֶ֔ם נְשִׂיאֵ֖י הָעֵדָֽה׃ (יט) וַיְהִ֗י מִקְצֵה֙ שְׁלֹ֣שֶׁת יָמִ֔ים אַחֲרֵ֕י אֲשֶׁר־ כָּרְת֥וּ לָהֶ֖ם בְּרִ֑ית וַֽיִּשְׁמְע֗וּ כִּי־ קְרֹבִ֥ים הֵם֙ אֵלָ֔יו וּבְקִרְבּ֖וֹ הֵ֥ם יֹשְׁבִֽים׃ (כ) וַיִּסְע֣וּ בְנֵֽי־ יִשְׂרָאֵ֗ל וַיָּבֹ֛אוּ אֶל־ עָרֵיהֶ֖ם בַּיּ֣וֹם הַשְּׁלִישִׁ֑י וְעָרֵיהֶם֙ גִּבְע֣וֹן וְהַכְּפִירָ֔ה וּבְאֵר֖וֹת וְקִרְיַ֥ת יְעָרִֽים׃ (כא) וְלֹ֤א הִכּוּם֙ בְּנֵ֣י יִשְׂרָאֵ֔ל כִּֽי־ נִשְׁבְּע֤וּ לָהֶם֙ נְשִׂיאֵ֣י הָֽעֵדָ֔ה בַּֽיהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וַיִּלֹּ֥נוּ כָל־ הָעֵדָ֖ה עַל־ הַנְּשִׂיאִֽים׃ (כב) וַיֹּאמְר֤וּ כָל־ הַנְּשִׂיאִים֙ אֶל־ כָּל־ הָ֣עֵדָ֔ה אֲנַ֙חְנוּ֙ נִשְׁבַּ֣עְנוּ לָהֶ֔ם בַּֽיהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וְעַתָּ֕ה לֹ֥א נוּכַ֖ל לִנְגֹּ֥עַ בָּהֶֽם׃ (כג) זֹ֛את נַעֲשֶׂ֥ה לָהֶ֖ם וְהַחֲיֵ֣ה אוֹתָ֑ם וְלֹֽא־ יִֽהְיֶ֤ה עָלֵ֙ינוּ֙ קֶ֔צֶף עַל־ הַשְּׁבוּעָ֖ה אֲשֶׁר־ נִשְׁבַּ֥עְנוּ לָהֶֽם׃ (כד) וַיֹּאמְר֧וּ אֲלֵיהֶ֛ם הַנְּשִׂיאִ֖ים יִֽחְי֑וּ וַ֠יִּֽהְיוּ חֹטְבֵ֨י עֵצִ֤ים וְשֹֽׁאֲבֵי־ מַ֙יִם֙ לְכָל־ הָ֣עֵדָ֔ה כַּאֲשֶׁ֛ר דִּבְּר֥וּ לָהֶ֖ם הַנְּשִׂיאִֽים׃ (כה) וַיִּקְרָ֤א לָהֶם֙ יְהוֹשֻׁ֔עַ וַיְדַבֵּ֥ר אֲלֵיהֶ֖ם לֵאמֹ֑ר לָמָּה֩ רִמִּיתֶ֨ם אֹתָ֜נוּ לֵאמֹ֗ר רְחוֹקִ֨ים אֲנַ֤חְנוּ מִכֶּם֙ מְאֹ֔ד וְאַתֶּ֖ם בְּקִרְבֵּ֥נוּ יֹשְׁבִֽים׃ (כו) וְעַתָּ֖ה אֲרוּרִ֣ים אַתֶּ֑ם וְלֹֽא־ יִכָּרֵ֨ת מִכֶּ֜ם עֶ֗בֶד וְחֹטְבֵ֥י עֵצִ֛ים וְשֹֽׁאֲבֵי־ מַ֖יִם לְבֵ֥ית אֱלֹהָֽי׃ (כז) וַיַּעֲנ֨וּ אֶת־ יְהוֹשֻׁ֜עַ וַיֹּאמְר֗וּ כִּי֩ הֻגֵּ֨ד הֻגַּ֤ד לַעֲבָדֶ֙יךָ֙ אֵת֩ אֲשֶׁ֨ר צִוָּ֜ה יְהוָ֤ה אֱלֹהֶ֙יךָ֙ אֶת־ מֹשֶׁ֣ה עַבְדּ֔וֹ לָתֵ֤ת לָכֶם֙ אֶת־ כָּל־ הָאָ֔רֶץ וּלְהַשְׁמִ֛יד אֶת־ כָּל־ יֹשְׁבֵ֥י הָאָ֖רֶץ מִפְּנֵיכֶ֑ם וַנִּירָ֨א מְאֹ֤ד לְנַפְשֹׁתֵ֙ינוּ֙ מִפְּנֵיכֶ֔ם וַֽנַּעֲשֵׂ֖ה אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (כח) וְעַתָּ֖ה הִנְנ֣וּ בְיָדֶ֑ךָ כַּטּ֨וֹב וְכַיָּשָׁ֧ר בְּעֵינֶ֛יךָ לַעֲשׂ֥וֹת לָ֖נוּ עֲשֵֽׂה׃ (כט) וַיַּ֥עַשׂ לָהֶ֖ם כֵּ֑ן וַיַּצֵּ֥ל אוֹתָ֛ם מִיַּ֥ד בְּנֵֽי־ יִשְׂרָאֵ֖ל וְלֹ֥א הֲרָגֽוּם׃ (ל) וַיִּתְּנֵ֨ם יְהוֹשֻׁ֜עַ בַּיּ֣וֹם הַה֗וּא חֹטְבֵ֥י עֵצִ֛ים וְשֹׁ֥אֲבֵי מַ֖יִם לָֽעֵדָ֑ה וּלְמִזְבַּ֤ח יְהוָה֙ עַד־ הַיּ֣וֹם הַזֶּ֔ה אֶל־ הַמָּק֖וֹם אֲשֶׁ֥ר יִבְחָֽר׃ תחילת פרק י (לא) וַיְהִי֩ כִשְׁמֹ֨עַ אֲדֹֽנִי־ צֶ֜דֶק מֶ֣לֶךְ יְרוּשָׁלִַ֗ם כִּֽי־ לָכַ֨ד יְהוֹשֻׁ֣עַ אֶת־ הָעַי֮ וַיַּחֲרִימָהּ֒ כַּאֲשֶׁ֨ר עָשָׂ֤ה לִֽירִיחוֹ֙ וּלְמַלְכָּ֔הּ כֵּן־ עָשָׂ֥ה לָעַ֖י וּלְמַלְכָּ֑הּ וְכִ֨י הִשְׁלִ֜ימוּ יֹשְׁבֵ֤י גִבְעוֹן֙ אֶת־ יִשְׂרָאֵ֔ל וַיִּֽהְי֖וּ בְּקִרְבָּֽם׃ (לב) וַיִּֽירְא֣וּ מְאֹ֔ד כִּ֣י עִ֤יר גְּדוֹלָה֙ גִּבְע֔וֹן כְּאַחַ֖ת עָרֵ֣י הַמַּמְלָכָ֑ה וְכִ֨י הִ֤יא גְדוֹלָה֙ מִן־ לְנַחֲלָ֧ה לְיִשְׂרָאֵ֛ל כְּמַחְלְקֹתָ֖ם לְשִׁבְטֵיהֶ֑ם וְהָאָ֥רֶץ שָׁקְטָ֖ה מִמִּלְחָמָֽה׃ (כו) וְאֵ֣לֶּה ׀ מַלְכֵ֣י הָאָ֗רֶץ אֲשֶׁ֨ר הִכּ֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ וַיִּֽרְשׁ֣וּ אֶת־ אַרְצָ֔ם בְּעֵ֥בֶר הַיַּרְדֵּ֖ן מִזְרְחָ֣ה הַשָּׁ֑מֶשׁ מִנַּ֤חַל אַרְנוֹן֙ עַד־ הַ֣ר חֶרְמ֔וֹן וְכָל־ הָעֲרָבָ֖ה מִזְרָֽחָה׃ (כז) סִיחוֹן֙ מֶ֣לֶךְ הָאֱמֹרִ֔י הַיּוֹשֵׁ֖ב בְּחֶשְׁבּ֑וֹן מֹשֵׁ֡ל מֵעֲרוֹעֵ֡ר אֲשֶׁר֩ עַל־ שְׂפַת־ נַ֨חַל אַרְנ֜וֹן וְת֤וֹךְ הַנַּ֙חַל֙ וַחֲצִ֣י הַגִּלְעָ֔ד וְעַד֙ יַבֹּ֣ק הַנַּ֔חַל גְּב֖וּל בְּנֵ֥י עַמּֽוֹן׃ (כח) וְהָעֲרָבָה֩ עַד־ יָ֨ם כִּנְר֜וֹת מִזְרָ֗חָה וְ֠עַד יָ֣ם הָעֲרָבָ֤ה יָם־ הַמֶּ֙לַח֙ מִזְרָ֔חָה דֶּ֖רֶךְ בֵּ֣ית הַיְשִׁמ֑וֹת וּמִ֨תֵּימָ֔ן תַּ֖חַת אַשְׁדּ֥וֹת הַפִּסְגָּֽה׃ (כט) וּגְב֗וּל ע֚וֹג מֶ֣לֶךְ הַבָּשָׁ֔ן מִיֶּ֖תֶר הָרְפָאִ֑ים הַיּוֹשֵׁ֥ב בְּעַשְׁתָּר֖וֹת וּבְאֶדְרֶֽעִי׃ (ל) וּ֠מֹשֵׁל בְּהַ֨ר חֶרְמ֤וֹן וּבְסַלְכָה֙ וּבְכָל־ הַבָּשָׁ֔ן עַד־ גְּב֥וּל הַגְּשׁוּרִ֖י וְהַמַּעֲכָתִ֑י וַחֲצִי֙ הַגִּלְעָ֔ד גְּב֖וּל סִיח֥וֹן מֶֽלֶךְ־ חֶשְׁבּֽוֹן׃ (לא) מֹשֶׁ֧ה עֶֽבֶד־ יְהוָ֛ה וּבְנֵ֥י יִשְׂרָאֵ֖ל הִכּ֑וּם וַֽ֠יִּתְּנָהּ מֹשֶׁ֨ה עֶֽבֶד־ יְהוָ֜ה יְרֻשָּׁ֗ה לָרֻֽאוּבֵנִי֙ וְלַגָּדִ֔י וְלַחֲצִ֖י שֵׁ֥בֶט הַֽמְנַשֶּֽׁה׃ (לב) וְאֵ֣לֶּה מַלְכֵ֣י הָאָ֡רֶץ אֲשֶׁר֩ הִכָּ֨ה יְהוֹשֻׁ֜עַ וּבְנֵ֣י יִשְׂרָאֵ֗ל בְּעֵ֤בֶר הַיַּרְדֵּן֙ יָ֔מָּה מִבַּ֤עַל גָּד֙ בְּבִקְעַ֣ת הַלְּבָנ֔וֹן וְעַד־ הָהָ֥ר הֶחָלָ֖ק הָעֹלֶ֣ה שֵׂעִ֑ירָה וַיִּתְּנָ֨הּ יְהוֹשֻׁ֜עַ לְשִׁבְטֵ֧י יִשְׂרָאֵ֛ל יְרֻשָּׁ֖ה כְּמַחְלְקֹתָֽם׃ (לג) בָּהָ֣ר וּבַשְּׁפֵלָ֗ה וּבָֽעֲרָבָה֙ וּבָ֣אֲשֵׁד֔וֹת וּבַמִּדְבָּ֖ר וּבַנֶּ֑גֶב הַֽחִתִּי֙ הָֽאֱמֹרִ֔י וְהַֽכְּנַעֲנִי֙ הַפְּרִזִּ֔י הַחִוִּ֖י וְהַיְבוּסִֽי׃ (לד) מֶ֥לֶךְ יְרִיח֖וֹ אֶחָ֑ד מֶ֧לֶךְ הָעַ֛י אֲשֶׁר־ מִצַּ֥ד בֵּֽית־ אֵ֖ל אֶחָֽד׃ (לה) מֶ֤לֶךְ יְרוּשָׁלִַ֙ם֙ אֶחָ֔ד מֶ֥לֶךְ חֶבְר֖וֹן אֶחָֽד׃ (לו) מֶ֤לֶךְ יַרְמוּת֙ אֶחָ֔ד מֶ֥לֶךְ לָכִ֖ישׁ אֶחָֽד׃ (לז) מֶ֤לֶךְ עֶגְלוֹן֙ אֶחָ֔ד מֶ֥לֶךְ גֶּ֖זֶר אֶחָֽד׃ (לח) מֶ֤לֶךְ דְּבִר֙ אֶחָ֔ד מֶ֥לֶךְ גֶּ֖דֶר אֶחָֽד׃ (לט) מֶ֤לֶךְ חָרְמָה֙ אֶחָ֔ד מֶ֥לֶךְ עֲרָ֖ד אֶחָֽד׃ (מ) מֶ֤לֶךְ לִבְנָה֙ אֶחָ֔ד מֶ֥לֶךְ עֲדֻלָּ֖ם אֶחָֽד׃ (מא) מֶ֤לֶךְ מַקֵּדָה֙ אֶחָ֔ד מֶ֥לֶךְ בֵּֽית־ אֵ֖ל אֶחָֽד׃ (מב) מֶ֤לֶךְ תַּפּ֙וּחַ֙ אֶחָ֔ד מֶ֥לֶךְ חֵ֖פֶר אֶחָֽד׃ (מג) מֶ֤לֶךְ אֲפֵק֙ אֶחָ֔ד מֶ֥לֶךְ לַשָּׁר֖וֹן אֶחָֽד׃ (מד) מֶ֤לֶךְ מָדוֹן֙ אֶחָ֔ד מֶ֥לֶךְ חָצ֖וֹר אֶחָֽד׃ (מה) מֶ֣לֶךְ שִׁמְר֤וֹן מְראוֹן֙ אֶחָ֔ד מֶ֥לֶךְ אַכְשָׁ֖ף אֶחָֽד׃ (מו) מֶ֤לֶךְ תַּעְנַךְ֙ אֶחָ֔ד מֶ֥לֶךְ מְגִדּ֖וֹ אֶחָֽד׃ (מז) מֶ֤לֶךְ קֶ֙דֶשׁ֙ אֶחָ֔ד מֶֽלֶךְ־ יָקְנֳעָ֥ם לַכַּרְמֶ֖ל אֶחָֽד׃ (מח) מֶ֥לֶךְ דּ֛וֹר לְנָפַ֥ת דּ֖וֹר אֶחָ֑ד מֶֽלֶךְ־ גּוֹיִ֥ם לְגִלְגָּ֖ל אֶחָֽד׃ (מט) מֶ֥לֶךְ תִּרְצָ֖ה אֶחָ֑ד כָּל־ מְלָכִ֖ים שְׁלֹשִׁ֥ים
