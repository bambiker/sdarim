יהושע פרק יח לפי החלוקה לסדרים

פרק יח בספר יהושע (28 פסוקים) נמצא ביהושע סדר י (פסוקים טז–מב) וביהושע סדר יא (פסוקים א–א).

יהושע סדר י

(א) וַתִּקְרַ֡בְנָה לִפְנֵי֩ אֶלְעָזָ֨ר הַכֹּהֵ֜ן וְלִפְנֵ֣י ׀ יְהוֹשֻׁ֣עַ בִּן־ נ֗וּן וְלִפְנֵ֤י הַנְּשִׂיאִים֙ לֵאמֹ֔ר יְהוָה֙ צִוָּ֣ה אֶת־ מֹשֶׁ֔ה לָֽתֶת־ לָ֥נוּ נַחֲלָ֖ה בְּת֣וֹךְ אַחֵ֑ינוּ וַיִּתֵּ֨ן לָהֶ֜ם אֶל־ פִּ֤י יְהוָה֙ נַֽחֲלָ֔ה בְּת֖וֹךְ אֲחֵ֥י אֲבִיהֶֽן׃ (ב) וַיִּפְּל֥וּ חַבְלֵֽי־ מְנַשֶּׁ֖ה עֲשָׂרָ֑ה לְבַ֞ד מֵאֶ֤רֶץ הַגִּלְעָד֙ וְהַבָּשָׁ֔ן אֲשֶׁ֖ר מֵעֵ֥בֶר לַיַּרְדֵּֽן׃ (ג) כִּ֚י בְּנ֣וֹת מְנַשֶּׁ֔ה נָחֲל֥וּ נַחֲלָ֖ה בְּת֣וֹךְ בָּנָ֑יו וְאֶ֙רֶץ֙ הַגִּלְעָ֔ד הָיְתָ֥ה לִבְנֵֽי־ מְנַשֶּׁ֖ה הַנּֽוֹתָרִֽים׃ (ד) וַיְהִ֤י גְבוּל־ מְנַשֶּׁה֙ מֵֽאָשֵׁ֔ר הַֽמִּכְמְתָ֔ת אֲשֶׁ֖ר עַל־ פְּנֵ֣י שְׁכֶ֑ם וְהָלַ֤ךְ הַגְּבוּל֙ אֶל־ הַיָּמִ֔ין אֶל־ יֹשְׁבֵ֖י עֵ֥ין תַּפּֽוּחַ׃ (ה) לִמְנַשֶּׁ֕ה הָיְתָ֖ה אֶ֣רֶץ תַּפּ֑וּחַ וְתַפּ֛וּחַ אֶל־ גְּב֥וּל מְנַשֶּׁ֖ה לִבְנֵ֥י אֶפְרָֽיִם׃ (ו) וְיָרַ֣ד הַגְּבוּל֩ נַ֨חַל קָנָ֜ה נֶ֣גְבָּה לַנַּ֗חַל עָרִ֤ים הָאֵ֙לֶּה֙ לְאֶפְרַ֔יִם בְּת֖וֹךְ עָרֵ֣י מְנַשֶּׁ֑ה וּגְב֤וּל מְנַשֶּׁה֙ מִצְּפ֣וֹן לַנַּ֔חַל וַיְהִ֥י תֹצְאֹתָ֖יו הַיָּֽמָּה׃ (ז) נֶ֣גְבָּה לְאֶפְרַ֗יִם וְצָפ֙וֹנָה֙ לִמְנַשֶּׁ֔ה וַיְהִ֥י הַיָּ֖ם גְּבוּל֑וֹ וּבְאָשֵׁר֙ יִפְגְּע֣וּן מִצָּפ֔וֹן וּבְיִשָּׂשכָ֖ר מִמִּזְרָֽח׃ (ח) וַיְהִ֨י לִמְנַשֶּׁ֜ה בְּיִשָּׂשכָ֣ר וּבְאָשֵׁ֗ר בֵּית־ שְׁאָ֣ן וּ֠בְנוֹתֶיהָ וְיִבְלְעָ֨ם וּבְנוֹתֶ֜יהָ וְֽאֶת־ יֹשְׁבֵ֧י דֹ֣אר וּבְנוֹתֶ֗יהָ וְיֹשְׁבֵ֤י עֵֽין־ דֹּר֙ וּבְנֹתֶ֔יהָ וְיֹשְׁבֵ֤י תַעְנַךְ֙ וּבְנֹתֶ֔יהָ וְיֹשְׁבֵ֥י מְגִדּ֖וֹ וּבְנוֹתֶ֑יהָ שְׁלֹ֖שֶׁת הַנָּֽפֶת׃ (ט) וְלֹ֤א יָכְלוּ֙ בְּנֵ֣י מְנַשֶּׁ֔ה לְהוֹרִ֖ישׁ אֶת־ הֶֽעָרִ֣ים הָאֵ֑לֶּה וַיּ֙וֹאֶל֙ הַֽכְּנַעֲנִ֔י לָשֶׁ֖בֶת בָּאָ֥רֶץ הַזֹּֽאת׃ (י) וַֽיְהִ֗י כִּ֤י חָֽזְקוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל וַיִּתְּנ֥וּ אֶת־ הַֽכְּנַעֲנִ֖י לָמַ֑ס וְהוֹרֵ֖שׁ לֹ֥א הוֹרִישֽׁוֹ׃ (יא) וַֽיְדַבְּרוּ֙ בְּנֵ֣י יוֹסֵ֔ף אֶת־ יְהוֹשֻׁ֖עַ לֵאמֹ֑ר מַדּוּעַ֩ נָתַ֨תָּה לִּ֜י נַחֲלָ֗ה גּוֹרָ֤ל אֶחָד֙ וְחֶ֣בֶל אֶחָ֔ד וַֽאֲנִ֣י עַם־ רָ֔ב עַ֥ד אֲשֶׁר־ עַד־ כֹּ֖ה בֵּֽרְכַ֥נִי יְהוָֽה׃ (יב) וַיֹּ֨אמֶר אֲלֵיהֶ֜ם יְהוֹשֻׁ֗עַ אִם־ עַם־ רַ֤ב אַתָּה֙ עֲלֵ֣ה לְךָ֣ הַיַּ֔עְרָה וּבֵרֵאתָ֤ לְךָ֙ שָׁ֔ם בְּאֶ֥רֶץ הַפְּרִזִּ֖י וְהָֽרְפָאִ֑ים כִּֽי־ אָ֥ץ לְךָ֖ הַר־ אֶפְרָֽיִם׃ (יג) וַיֹּֽאמְרוּ֙ בְּנֵ֣י יוֹסֵ֔ף לֹֽא־ יִמָּ֥צֵא לָ֖נוּ הָהָ֑ר וְרֶ֣כֶב בַּרְזֶ֗ל בְּכָל־ הַֽכְּנַעֲנִי֙ הַיֹּשֵׁ֣ב בְּאֶֽרֶץ־ הָעֵ֔מֶק לַֽאֲשֶׁ֤ר בְּבֵית־ שְׁאָן֙ וּבְנוֹתֶ֔יהָ וְלַֽאֲשֶׁ֖ר בְּעֵ֥מֶק יִזְרְעֶֽאל׃ (יד) וַיֹּ֤אמֶר יְהוֹשֻׁ֙עַ֙ אֶל־ בֵּ֣ית יוֹסֵ֔ף לְאֶפְרַ֥יִם וְלִמְנַשֶּׁ֖ה לֵאמֹ֑ר עַם־ רַ֣ב אַתָּ֗ה וְכֹ֤חַ גָּדוֹל֙ לָ֔ךְ לֹֽא־ יִהְיֶ֥ה לְךָ֖ גּוֹרָ֥ל אֶחָֽד׃ (טו) כִּ֣י הַ֤ר יִֽהְיֶה־ לָּךְ֙ כִּֽי־ יַ֣עַר ה֔וּא וּבֵ֣רֵאת֔וֹ וְהָיָ֥ה לְךָ֖ תֹּֽצְאֹתָ֑יו כִּֽי־ תוֹרִ֣ישׁ אֶת־ הַֽכְּנַעֲנִ֗י כִּ֣י רֶ֤כֶב בַּרְזֶל֙ ל֔וֹ כִּ֥י חָזָ֖ק הֽוּא׃ תחילת פרק יח (טז) וַיִּקָּ֨הֲל֜וּ כָּל־ עֲדַ֤ת בְּנֵֽי־ יִשְׂרָאֵל֙ שִׁלֹ֔ה וַיַּשְׁכִּ֥ינוּ שָׁ֖ם אֶת־ אֹ֣הֶל מוֹעֵ֑ד וְהָאָ֥רֶץ נִכְבְּשָׁ֖ה לִפְנֵיהֶֽם׃ (יז) וַיִּוָּֽתְרוּ֙ בִּבְנֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֥ר לֹֽא־ חָלְק֖וּ אֶת־ נַֽחֲלָתָ֑ם שִׁבְעָ֖ה שְׁבָטִֽים׃ (יח) וַיֹּ֥אמֶר יְהוֹשֻׁ֖עַ אֶל־ בְּנֵ֣י יִשְׂרָאֵ֑ל עַד־ אָ֙נָה֙ אַתֶּ֣ם מִתְרַפִּ֔ים לָבוֹא֙ לָרֶ֣שֶׁת אֶת־ הָאָ֔רֶץ אֲשֶׁר֙ נָתַ֣ן לָכֶ֔ם יְהוָ֖ה אֱלֹהֵ֥י אֲבֽוֹתֵיכֶֽם׃ (יט) הָב֥וּ לָכֶ֛ם שְׁלֹשָׁ֥ה אֲנָשִׁ֖ים לַשָּׁ֑בֶט וְאֶשְׁלָחֵ֗ם וְיָקֻ֜מוּ וְיִֽתְהַלְּכ֥וּ בָאָ֛רֶץ וְיִכְתְּב֥וּ אוֹתָ֛הּ לְפִ֥י נַֽחֲלָתָ֖ם וְיָבֹ֥אוּ אֵלָֽי׃ (כ) וְהִֽתְחַלְּק֥וּ אֹתָ֖הּ לְשִׁבְעָ֣ה חֲלָקִ֑ים יְהוּדָ֞ה יַעֲמֹ֤ד עַל־ גְּבוּלוֹ֙ מִנֶּ֔גֶב וּבֵ֥ית יוֹסֵ֛ף יַעַמְד֥וּ עַל־ גְּבוּלָ֖ם מִצָּפֽוֹן׃ (כא) וְאַתֶּ֞ם תִּכְתְּב֤וּ אֶת־ הָאָ֙רֶץ֙ שִׁבְעָ֣ה חֲלָקִ֔ים וַֽהֲבֵאתֶ֥ם אֵלַ֖י הֵ֑נָּה וְיָרִ֨יתִי לָכֶ֤ם גּוֹרָל֙ פֹּ֔ה לִפְנֵ֖י יְהוָ֥ה אֱלֹהֵֽינוּ׃ (כב) כִּ֠י אֵֽין־ חֵ֤לֶק לַלְוִיִּם֙ בְּקִרְבְּכֶ֔ם כִּֽי־ כְהֻנַּ֥ת יְהוָ֖ה נַחֲלָת֑וֹ וְגָ֡ד וּרְאוּבֵ֡ן וַחֲצִי֩ שֵׁ֨בֶט הַֽמְנַשֶּׁ֜ה לָקְח֣וּ נַחֲלָתָ֗ם מֵעֵ֤בֶר לַיַּרְדֵּן֙ מִזְרָ֔חָה אֲשֶׁר֙ נָתַ֣ן לָהֶ֔ם מֹשֶׁ֖ה עֶ֥בֶד יְהוָֽה׃ (כג) וַיָּקֻ֥מוּ הָאֲנָשִׁ֖ים וַיֵּלֵ֑כוּ וַיְצַ֣ו יְהוֹשֻׁ֡עַ אֶת־ הַהֹלְכִים֩ לִכְתֹּ֨ב אֶת־ הָאָ֜רֶץ לֵאמֹ֗ר לְ֠כוּ וְהִתְהַלְּכ֨וּ בָאָ֜רֶץ וְכִתְב֤וּ אוֹתָהּ֙ וְשׁ֣וּבוּ אֵלַ֔י וּ֠פֹה אַשְׁלִ֨יךְ לָכֶ֥ם גּוֹרָ֛ל לִפְנֵ֥י יְהוָ֖ה בְּשִׁלֹֽה׃ (כד) וַיֵּלְכ֤וּ הָֽאֲנָשִׁים֙ וַיַּעַבְר֣וּ בָאָ֔רֶץ וַיִּכְתְּב֧וּהָ לֶֽעָרִ֛ים לְשִׁבְעָ֥ה חֲלָקִ֖ים עַל־ סֵ֑פֶר וַיָּבֹ֧אוּ אֶל־ יְהוֹשֻׁ֛עַ אֶל־ הַֽמַּחֲנֶ֖ה שִׁלֹֽה׃ (כה) וַיַּשְׁלֵךְ֩ לָהֶ֨ם יְהוֹשֻׁ֧עַ גּוֹרָ֛ל בְּשִׁלֹ֖ה לִפְנֵ֣י יְהוָ֑ה וַיְחַלֶּק־ שָׁ֨ם יְהוֹשֻׁ֧עַ אֶת־ הָאָ֛רֶץ לִבְנֵ֥י יִשְׂרָאֵ֖ל כְּמַחְלְקֹתָֽם׃ (כו) וַיַּ֗עַל גּוֹרַ֛ל מַטֵּ֥ה בְנֵֽי־ בִנְיָמִ֖ן לְמִשְׁפְּחֹתָ֑ם וַיֵּצֵא֙ גְּב֣וּל גּֽוֹרָלָ֔ם בֵּ֚ין בְּנֵ֣י יְהוּדָ֔ה וּבֵ֖ין בְּנֵ֥י יוֹסֵֽף׃ (כז) וַיְהִ֨י לָהֶ֧ם הַגְּב֛וּל לִפְאַ֥ת צָפ֖וֹנָה מִן־ הַיַּרְדֵּ֑ן וְעָלָ֣ה הַגְּבוּל֩ אֶל־ כֶּ֨תֶף יְרִיח֜וֹ מִצָּפ֗וֹן וְעָלָ֤ה בָהָר֙ יָ֔מָּה (והיה) וְהָיוּ֙ תֹּֽצְאֹתָ֔יו מִדְבַּ֖רָה בֵּ֥ית אָֽוֶן׃ (כח) וְעָבַר֩ מִשָּׁ֨ם הַגְּב֜וּל ל֗וּזָה אֶל־ כֶּ֤תֶף ל֙וּזָה֙ נֶ֔גְבָּה הִ֖יא בֵּֽית־ אֵ֑ל וְיָרַ֤ד הַגְּבוּל֙ עַטְר֣וֹת אַדָּ֔ר עַל־ הָהָ֕ר אֲשֶׁ֛ר מִנֶּ֥גֶב לְבֵית־ חֹר֖וֹן תַּחְתּֽוֹן׃ (כט) וְתָאַ֣ר הַגְּבוּל֩ וְנָסַ֨ב לִפְאַת־ יָ֜ם נֶ֗גְבָּה מִן־ הָהָר֙ אֲשֶׁ֨ר עַל־ פְּנֵ֥י בֵית־ חֹרוֹן֮ נֶגְבָּה֒ (והיה) וְהָי֣וּ תֹֽצְאֹתָ֗יו אֶל־ קִרְיַת־ בַּ֙עַל֙ הִ֚יא קִרְיַ֣ת יְעָרִ֔ים עִ֖יר בְּנֵ֣י יְהוּדָ֑ה זֹ֖את פְּאַת־ יָֽם׃ (ל) וּפְאַת־ נֶ֕גְבָּה מִקְצֵ֖ה קִרְיַ֣ת יְעָרִ֑ים וְיָצָ֤א הַגְּבוּל֙ יָ֔מָּה וְיָצָ֕א אֶל־ מַעְיַ֖ן מֵ֥י נֶפְתּֽוֹחַ׃ (לא) וְיָרַ֨ד הַגְּב֜וּל אֶל־ קְצֵ֣ה הָהָ֗ר אֲשֶׁר֙ עַל־ פְּנֵי֙ גֵּ֣י בֶן־ הִנֹּ֔ם אֲשֶׁ֛ר בְּעֵ֥מֶק רְפָאִ֖ים צָפ֑וֹנָה וְיָרַד֩ גֵּ֨י הִנֹּ֜ם אֶל־ כֶּ֤תֶף הַיְבוּסִי֙ נֶ֔גְבָּה וְיָרַ֖ד עֵ֥ין רֹגֵֽל׃ (לב) וְתָאַ֣ר מִצָּפ֗וֹן וְיָצָא֙ עֵ֣ין שֶׁ֔מֶשׁ וְיָצָא֙ אֶל־ גְּלִיל֔וֹת אֲשֶׁר־ נֹ֖כַח מַעֲלֵ֣ה הַגּוֹרָ֤ל הַשֵּׁנִי֙ לְשִׁמְע֔וֹן לְמַטֵּ֥ה בְנֵֽי־ שִׁמְע֖וֹן לְמִשְׁפְּחוֹתָ֑ם וַֽיְהִי֙ נַֽחֲלָתָ֔ם בְּת֖וֹךְ נַחֲלַ֥ת בְּנֵֽי־ יְהוּדָֽה׃ (ג) וַיְהִ֥י לָהֶ֖ם בְּנַֽחֲלָתָ֑ם בְּאֵֽר־ שֶׁ֥בַע עָרִ֥ים שְׁלֹשׁ־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (ח) עַ֥יִן ׀ רִמּ֖וֹן וָעֶ֣תֶר וְעָשָׁ֑ן עָרִ֥ים אַרְבַּ֖ע וְחַצְרֵיהֶֽן׃ (ט) וְכָל־ הַֽחֲצֵרִ֗ים אֲשֶׁ֤ר סְבִיבוֹת֙ הֶֽעָרִ֣ים הָאֵ֔לֶּה עַד־ בַּֽעֲלַ֥ת בְּאֵ֖ר רָ֣אמַת נֶ֑גֶב זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵֽי־ שִׁמְע֖וֹן לְמִשְׁפְּחֹתָֽם׃ (י) מֵחֶ֙בֶל֙ בְּנֵ֣י יְהוּדָ֔ה נַחֲלַ֖ת בְּנֵ֣י שִׁמְע֑וֹן כִּֽי־ הָיָ֞ה חֵ֤לֶק בְּנֵֽי־ יְהוּדָה֙ רַ֣ב מֵהֶ֔ם וַיִּנְחֲל֥וּ בְנֵֽי־ שִׁמְע֖וֹן בְּת֥וֹךְ נַחֲלָתָֽם׃ (יא) וַיַּ֙עַל֙ הַגּוֹרָ֣ל הַשְּׁלִישִׁ֔י לִבְנֵ֥י זְבוּלֻ֖ן לְמִשְׁפְּחֹתָ֑ם וַיְהִ֛י גְּב֥וּל נַחֲלָתָ֖ם עַד־ שָׂרִֽיד׃ (יב) וְעָלָ֨ה גְבוּלָ֧ם יִפְתַּח־ אֵ֥ל צָפ֛וֹנָה בֵּ֥ית הָעֵ֖מֶק וּנְעִיאֵ֑ל וְיָצָ֥א אֶל־ כָּב֖וּל מִשְּׂמֹֽאל׃ (כט) וְעֶבְרֹ֥ן וּרְחֹ֖ב וְחַמּ֣וֹן וְקָנָ֑ה עַ֖ד צִיד֥וֹן רַבָּֽה׃ (ל) וְשָׁ֤ב הַגְּבוּל֙ הָֽרָמָ֔ה וְעַד־ זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵֽי־ נַפְתָּלִ֖י לְמִשְׁפְּחֹתָ֑ם הֶעָרִ֖ים וְחַצְרֵיהֶֽן׃ (מא) לְמַטֵּ֥ה בְנֵי־ דָ֖ן לְמִשְׁפְּחֹתָ֑ם יָצָ֖א הַגּוֹרָ֥ל הַשְּׁבִיעִֽי׃ (מב) וַיְהִ֖י גְּב֣וּל נַחֲלָתָ֑ם צָרְעָ֥ה וְאֶשְׁתָּא֖וֹל וְעִ֥יר שָֽׁמֶשׁ׃ (מג) וְשַֽׁעֲלַבִּ֥ין וְאַיָּל֖וֹן וְיִתְלָֽה׃ (מד) וְאֵיל֥וֹן וְתִמְנָ֖תָה וְעֶקְרֽוֹן׃ (מה) וְאֶלְתְּקֵ֥ה וְגִבְּת֖וֹן וּבַעֲלָֽת׃ (מו) וִיהֻ֥ד וּבְנֵֽי־ בְרַ֖ק וְגַת־ רִמּֽוֹן׃ (מז) וּמֵ֥י הַיַּרְק֖וֹן וְהָֽרַקּ֑וֹן עִֽם־ הַגְּב֖וּל מ֥וּל יָפֽוֹ׃ (מח) וַיֵּצֵ֥א גְבוּל־ בְּנֵי־ דָ֖ן מֵהֶ֑ם וַיַּעֲל֣וּ בְנֵֽי־ דָ֠ן וַיִּלָּחֲמ֨וּ עִם־ לֶ֜שֶׁם וַיִּלְכְּד֥וּ אוֹתָ֣הּ ׀ וַיַּכּ֧וּ אוֹתָ֣הּ לְפִי־ חֶ֗רֶב וַיִּֽרְשׁ֤וּ אוֹתָהּ֙ וַיֵּ֣שְׁבוּ בָ֔הּ וַיִּקְרְא֤וּ לְלֶ֙שֶׁם֙ דָּ֔ן כְּשֵׁ֖ם דָּ֥ן אֲבִיהֶֽם׃ (מט) זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵי־ דָ֖ן לְמִשְׁפְּחֹתָ֑ם הֶֽעָרִ֥ים הָאֵ֖לֶּה וְחַצְרֵיהֶֽן׃ (נ) וַיְכַלּ֥וּ לִנְחֹל־ אֶת־ הָאָ֖רֶץ לִגְבֽוּלֹתֶ֑יהָ וַיִּתְּנ֨וּ בְנֵי־ יִשְׂרָאֵ֧ל נַחֲלָ֛ה לִיהוֹשֻׁ֥עַ בִּן־ נ֖וּן בְּתוֹכָֽם׃ (נא) עַל־ פִּ֨י יְהוָ֜ה נָ֣תְנוּ ל֗וֹ אֶת־ הָעִיר֙ אֲשֶׁ֣ר שָׁאָ֔ל אֶת־ תִּמְנַת־ סֶ֖רַח בְּהַ֣ר אֶפְרָ֑יִם וַיִּבְנֶ֥ה אֶת־ הָעִ֖יר וַיֵּ֥שֶׁב בָּֽהּ׃
