חגי פרק ב לפי החלוקה לסדרים

פרק ב בספר חגי (23 פסוקים) נמצא בתרי עשר סדר טו (פסוקים יז–לח) ובתרי עשר סדר טז (פסוקים א–א).

תרי עשר סדר טו

(א) בָּעֵ֤ת הַהִיא֙ אָבִ֣יא אֶתְכֶ֔ם וּבָעֵ֖ת קַבְּצִ֣י אֶתְכֶ֑ם כִּֽי־ אֶתֵּ֨ן אֶתְכֶ֜ם לְשֵׁ֣ם וְלִתְהִלָּ֗ה בְּכֹל֙ עַמֵּ֣י הָאָ֔רֶץ בְּשׁוּבִ֧י אֶת־ שְׁבוּתֵיכֶ֛ם לְעֵינֵיכֶ֖ם אָמַ֥ר יְהוָֽה׃ (ב) בִּשְׁנַ֤ת שְׁתַּ֙יִם֙ לְדָרְיָ֣וֶשׁ הַמֶּ֔לֶךְ בַּחֹ֙דֶשׁ֙ הַשִּׁשִּׁ֔י בְּי֥וֹם אֶחָ֖ד לַחֹ֑דֶשׁ הָיָ֨ה דְבַר־ יְהוָ֜ה בְּיַד־ חַגַּ֣י הַנָּבִ֗יא אֶל־ זְרֻבָּבֶ֤ל בֶּן־ שְׁאַלְתִּיאֵל֙ פַּחַ֣ת יְהוּדָ֔ה וְאֶל־ יְהוֹשֻׁ֧עַ בֶּן־ יְהוֹצָדָ֛ק הַכֹּהֵ֥ן הַגָּד֖וֹל לֵאמֹֽר׃ (ג) כֹּ֥ה אָמַ֛ר יְהוָ֥ה צְבָא֖וֹת לֵאמֹ֑ר הָעָ֤ם הַזֶּה֙ אָֽמְר֔וּ לֹ֥א עֶת־ בֹּ֛א עֶת־ בֵּ֥ית יְהוָ֖ה לְהִבָּנֽוֹת׃ (ד) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה בְּיַד־ חַגַּ֥י הַנָּבִ֖יא לֵאמֹֽר׃ (ה) הַעֵ֤ת לָכֶם֙ אַתֶּ֔ם לָשֶׁ֖בֶת בְּבָתֵּיכֶ֣ם סְפוּנִ֑ים וְהַבַּ֥יִת הַזֶּ֖ה חָרֵֽב׃ (ו) וְעַתָּ֕ה כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָא֑וֹת שִׂ֥ימוּ לְבַבְכֶ֖ם עַל־ דַּרְכֵיכֶֽם׃ (ז) זְרַעְתֶּ֨ם הַרְבֵּ֜ה וְהָבֵ֣א מְעָ֗ט אָכ֤וֹל וְאֵין־ לְשָׂבְעָה֙ שָׁת֣וֹ וְאֵין־ לְשָׁכְרָ֔ה לָב֖וֹשׁ וְאֵין־ לְחֹ֣ם ל֑וֹ וְהַ֨מִּשְׂתַּכֵּ֔ר מִשְׂתַּכֵּ֖ר אֶל־ צְר֥וֹר נָקֽוּב׃ (ח) כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָא֑וֹת שִׂ֥ימוּ לְבַבְכֶ֖ם עַל־ דַּרְכֵיכֶֽם׃ (ט) עֲל֥וּ הָהָ֛ר וַהֲבֵאתֶ֥ם עֵ֖ץ וּבְנ֣וּ הַבָּ֑יִת וְאֶרְצֶה־ בּ֥וֹ (ואכבד) וְאֶכָּבְדָ֖ה אָמַ֥ר יְהוָֽה׃ (י) פָּנֹ֤ה אֶל־ הַרְבֵּה֙ וְהִנֵּ֣ה לִמְעָ֔ט וַהֲבֵאתֶ֥ם הַבַּ֖יִת וְנָפַ֣חְתִּי ב֑וֹ יַ֣עַן מֶ֗ה נְאֻם֙ יְהוָ֣ה צְבָא֔וֹת יַ֗עַן בֵּיתִי֙ אֲשֶׁר־ ה֣וּא חָרֵ֔ב וְאַתֶּ֥ם רָצִ֖ים אִ֥ישׁ לְבֵיתֽוֹ׃ (יא) עַל־ כֵּ֣ן עֲלֵיכֶ֔ם כָּלְא֥וּ שָמַ֖יִם מִטָּ֑ל וְהָאָ֖רֶץ כָּלְאָ֥ה יְבוּלָֽהּ׃ (יב) וָאֶקְרָ֨א חֹ֜רֶב עַל־ הָאָ֣רֶץ וְעַל־ הֶהָרִ֗ים וְעַל־ הַדָּגָן֙ וְעַל־ הַתִּיר֣וֹשׁ וְעַל־ הַיִּצְהָ֔ר וְעַ֛ל אֲשֶׁ֥ר תּוֹצִ֖יא הָאֲדָמָ֑ה וְעַל־ הָֽאָדָם֙ וְעַל־ הַבְּהֵמָ֔ה וְעַ֖ל כָּל־ יְגִ֥יעַ כַּפָּֽיִם׃ (יג) וַיִּשְׁמַ֣ע זְרֻבָּבֶ֣ל ׀ בֶּֽן־ שַׁלְתִּיאֵ֡ל וִיהוֹשֻׁ֣עַ בֶּן־ יְהוֹצָדָק֩ הַכֹּהֵ֨ן הַגָּד֜וֹל וְכֹ֣ל ׀ שְׁאֵרִ֣ית הָעָ֗ם בְּקוֹל֙ יְהוָ֣ה אֱלֹֽהֵיהֶ֔ם וְעַל־ דִּבְרֵי֙ חַגַּ֣י הַנָּבִ֔יא כַּאֲשֶׁ֥ר שְׁלָח֖וֹ יְהוָ֣ה אֱלֹהֵיהֶ֑ם וַיִּֽירְא֥וּ הָעָ֖ם מִפְּנֵ֥י יְהוָֽה׃ (יד) וַ֠יֹּאמֶר חַגַּ֞י מַלְאַ֧ךְ יְהוָ֛ה בְּמַלְאֲכ֥וּת יְהוָ֖ה לָעָ֣ם לֵאמֹ֑ר אֲנִ֥י אִתְּכֶ֖ם נְאֻם־ יְהוָֽה׃ (טו) וַיָּ֣עַר יְהוָ֡ה אֶת־ רוּחַ֩ זְרֻבָּבֶ֨ל בֶּן־ שַׁלְתִּיאֵ֜ל פַּחַ֣ת יְהוּדָ֗ה וְאֶת־ ר֙וּחַ֙ יְהוֹשֻׁ֤עַ בֶּן־ יְהוֹצָדָק֙ הַכֹּהֵ֣ן הַגָּד֔וֹל וְֽאֶת־ ר֔וּחַ כֹּ֖ל שְׁאֵרִ֣ית הָעָ֑ם וַיָּבֹ֙אוּ֙ וַיַּעֲשׂ֣וּ מְלָאכָ֔ה בְּבֵית־ יְהוָ֥ה צְבָא֖וֹת אֱלֹהֵיהֶֽם׃ (טז) בְּי֨וֹם עֶשְׂרִ֧ים וְאַרְבָּעָ֛ה לַחֹ֖דֶשׁ בַּשִּׁשִּׁ֑י בִּשְׁנַ֥ת שְׁתַּ֖יִם לְדָרְיָ֥וֶשׁ הַמֶּֽלֶךְ׃ תחילת פרק ב (יז) בַּשְּׁבִיעִ֕י בְּעֶשְׂרִ֥ים וְאֶחָ֖ד לַחֹ֑דֶשׁ הָיָה֙ דְּבַר־ יְהוָ֔ה בְּיַד־ חַגַּ֥י הַנָּבִ֖יא לֵאמֹֽר׃ (יח) אֱמָר־ נָ֗א אֶל־ זְרֻבָּבֶ֤ל בֶּן־ שַׁלְתִּיאֵל֙ פַּחַ֣ת יְהוּדָ֔ה וְאֶל־ יְהוֹשֻׁ֥עַ בֶּן־ יְהוֹצָדָ֖ק הַכֹּהֵ֣ן הַגָּד֑וֹל וְאֶל־ שְׁאֵרִ֥ית הָעָ֖ם לֵאמֹֽר׃ (יט) מִ֤י בָכֶם֙ הַנִּשְׁאָ֔ר אֲשֶׁ֤ר רָאָה֙ אֶת־ הַבַּ֣יִת הַזֶּ֔ה בִּכְבוֹד֖וֹ הָרִאשׁ֑וֹן וּמָ֨ה אַתֶּ֜ם רֹאִ֤ים אֹתוֹ֙ עַ֔תָּה הֲל֥וֹא כָמֹ֛הוּ כְּאַ֖יִן בְּעֵינֵיכֶֽם׃ (כ) וְעַתָּ֣ה חֲזַ֣ק זְרֻבָּבֶ֣ל ׀ נְאֻם־ יְהוָ֡ה וַחֲזַ֣ק יְהוֹשֻׁ֣עַ בֶּן־ יְהוֹצָדָק֩ הַכֹּהֵ֨ן הַגָּד֜וֹל וַחֲזַ֨ק כָּל־ עַ֥ם הָאָ֛רֶץ נְאֻם־ יְהוָ֖ה וַֽעֲשׂ֑וּ כִּֽי־ אֲנִ֣י אִתְּכֶ֔ם נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כא) אֶֽת־ הַדָּבָ֞ר אֲשֶׁר־ כָּרַ֤תִּי אִתְּכֶם֙ בְּצֵאתְכֶ֣ם מִמִּצְרַ֔יִם וְרוּחִ֖י עֹמֶ֣דֶת בְּתוֹכְכֶ֑ם אַל־ תִּירָֽאוּ׃ (כב) כִּ֣י כֹ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת ע֥וֹד אַחַ֖ת מְעַ֣ט הִ֑יא וַאֲנִ֗י מַרְעִישׁ֙ אֶת־ הַשָּׁמַ֣יִם וְאֶת־ הָאָ֔רֶץ וְאֶת־ הַיָּ֖ם וְאֶת־ הֶחָרָבָֽה׃ (כג) וְהִרְעַשְׁתִּי֙ אֶת־ כָּל־ הַגּוֹיִ֔ם וּבָ֖אוּ חֶמְדַּ֣ת כָּל־ הַגּוֹיִ֑ם וּמִלֵּאתִ֞י אֶת־ הַבַּ֤יִת הַזֶּה֙ כָּב֔וֹד אָמַ֖ר יְהוָ֥ה צְבָאֽוֹת׃ (כד) לִ֥י הַכֶּ֖סֶף וְלִ֣י הַזָּהָ֑ב נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כה) גָּד֣וֹל יִֽהְיֶ֡ה כְּבוֹד֩ הַבַּ֨יִת הַזֶּ֤ה הָאַֽחֲרוֹן֙ מִן־ הָ֣רִאשׁ֔וֹן אָמַ֖ר יְהוָ֣ה צְבָא֑וֹת וּבַמָּק֤וֹם הַזֶּה֙ אֶתֵּ֣ן שָׁל֔וֹם נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כו) בְּעֶשְׂרִ֤ים וְאַרְבָּעָה֙ לַתְּשִׁיעִ֔י בִּשְׁנַ֥ת שְׁתַּ֖יִם לְדָרְיָ֑וֶשׁ הָיָה֙ דְּבַר־ יְהוָ֔ה אֶל־ חַגַּ֥י הַנָּבִ֖יא לֵאמֹֽר׃ (כז) כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָא֑וֹת שְׁאַל־ נָ֧א אֶת־ הַכֹּהֲנִ֛ים תּוֹרָ֖ה לֵאמֹֽר׃ (כח) הֵ֣ן ׀ יִשָּׂא־ אִ֨ישׁ בְּשַׂר־ קֹ֜דֶשׁ בִּכְנַ֣ף בִּגְד֗וֹ וְנָגַ֣ע בִּ֠כְנָפוֹ אֶל־ הַלֶּ֨חֶם וְאֶל־ הַנָּזִ֜יד וְאֶל־ הַיַּ֧יִן וְאֶל־ שֶׁ֛מֶן וְאֶל־ כָּל־ מַאֲכָ֖ל הֲיִקְדָּ֑שׁ וַיַּעֲנ֧וּ הַכֹּהֲנִ֛ים וַיֹּאמְר֖וּ לֹֽא׃ (כט) וַיֹּ֣אמֶר חַגַּ֔י אִם־ יִגַּ֧ע טְמֵא־ נֶ֛פֶשׁ בְּכָל־ אֵ֖לֶּה הֲיִטְמָ֑א וַיַּעֲנ֧וּ הַכֹּהֲנִ֛ים וַיֹּאמְר֖וּ יִטְמָֽא׃ (ל) וַיַּ֨עַן חַגַּ֜י וַיֹּ֗אמֶר כֵּ֣ן הָֽעָם־ הַ֠זֶּה וְכֵן־ הַגּ֨וֹי הַזֶּ֤ה לְפָנַי֙ נְאֻם־ יְהוָ֔ה וְכֵ֖ן כָּל־ מַעֲשֵׂ֣ה יְדֵיהֶ֑ם וַאֲשֶׁ֥ר יַקְרִ֛יבוּ שָׁ֖ם טָמֵ֥א הֽוּא׃ (לא) וְעַתָּה֙ שִֽׂימוּ־ נָ֣א לְבַבְכֶ֔ם מִן־ הַיּ֥וֹם הַזֶּ֖ה וָמָ֑עְלָה מִטֶּ֧רֶם שֽׂוּם־ אֶ֛בֶן אֶל־ אֶ֖בֶן בְּהֵיכַ֥ל יְהוָֽה׃ (לב) מִֽהְיוֹתָ֥ם בָּא֙ אֶל־ עֲרֵמַ֣ת עֶשְׂרִ֔ים וְהָיְתָ֖ה עֲשָׂרָ֑ה בָּ֣א אֶל־ הַיֶּ֗קֶב לַחְשֹׂף֙ חֲמִשִּׁ֣ים פּוּרָ֔ה וְהָיְתָ֖ה עֶשְׂרִֽים׃ (לג) הִכֵּ֨יתִי אֶתְכֶ֜ם בַּשִּׁדָּפ֤וֹן וּבַיֵּֽרָקוֹן֙ וּבַבָּרָ֔ד אֵ֖ת כָּל־ מַעֲשֵׂ֣ה יְדֵיכֶ֑ם וְאֵין־ אֶתְכֶ֥ם אֵלַ֖י נְאֻם־ יְהוָֽה׃ (לד) שִׂימוּ־ נָ֣א לְבַבְכֶ֔ם מִן־ הַיּ֥וֹם הַזֶּ֖ה וָמָ֑עְלָה מִיּוֹם֩ עֶשְׂרִ֨ים וְאַרְבָּעָ֜ה לַתְּשִׁיעִ֗י לְמִן־ הַיּ֛וֹם אֲשֶׁר־ יֻסַּ֥ד הֵֽיכַל־ יְהוָ֖ה שִׂ֥ימוּ לְבַבְכֶֽם׃ (לה) הַע֤וֹד הַזֶּ֙רַע֙ בַּמְּגוּרָ֔ה וְעַד־ הַגֶּ֨פֶן וְהַתְּאֵנָ֧ה וְהָרִמּ֛וֹן וְעֵ֥ץ הַזַּ֖יִת לֹ֣א נָשָׂ֑א מִן־ הַיּ֥וֹם הַזֶּ֖ה אֲבָרֵֽךְ׃ (לו) וַיְהִ֨י דְבַר־ יְהוָ֤ה ׀ שֵׁנִית֙ אֶל־ חַגַּ֔י בְּעֶשְׂרִ֧ים וְאַרְבָּעָ֛ה לַחֹ֖דֶשׁ לֵאמֹֽר׃ (לז) אֱמֹ֕ר אֶל־ זְרֻבָּבֶ֥ל פַּֽחַת־ יְהוּדָ֖ה לֵאמֹ֑ר אֲנִ֣י מַרְעִ֔ישׁ אֶת־ הַשָּׁמַ֖יִם וְאֶת־ הָאָֽרֶץ׃ (לח) וְהָֽפַכְתִּי֙ כִּסֵּ֣א מַמְלָכ֔וֹת וְהִ֨שְׁמַדְתִּ֔י חֹ֖זֶק מַמְלְכ֣וֹת הַגּוֹיִ֑ם וְהָפַכְתִּ֤י מֶרְכָּבָה֙ וְרֹ֣כְבֶ֔יהָ וְיָרְד֤וּ סוּסִים֙ וְרֹ֣כְבֵיהֶ֔ם אִ֖ישׁ בְּחֶ֥רֶב אָחִֽיו׃

(א) בַּיּ֣וֹם הַה֣וּא נְאֻם־ יְהוָ֣ה צְבָא֡וֹת אֶ֠קָּחֲךָ זְרֻבָּבֶ֨ל בֶּן־ שְׁאַלְתִּיאֵ֤ל עַבְדִּי֙ נְאֻם־ יְהוָ֔ה וְשַׂמְתִּ֖יךָ כַּֽחוֹתָ֑ם כִּֽי־ בְךָ֣ בָחַ֔רְתִּי נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ סוף פרק ב (ב) בַּחֹ֙דֶשׁ֙ הַשְּׁמִינִ֔י בִּשְׁנַ֥ת שְׁתַּ֖יִם לְדָרְיָ֑וֶשׁ הָיָ֣ה דְבַר־ יְהוָ֗ה אֶל־ זְכַרְיָה֙ בֶּן־ בֶּ֣רֶכְיָ֔ה בֶּן־ עִדּ֥וֹ הַנָּבִ֖יא לֵאמֹֽר׃ (ג) קָצַ֧ף יְהוָ֛ה עַל־ אֲבֽוֹתֵיכֶ֖ם קָֽצֶף׃ (ד) וְאָמַרְתָּ֣ אֲלֵהֶ֗ם כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת שׁ֣וּבוּ אֵלַ֔י נְאֻ֖ם יְהוָ֣ה צְבָא֑וֹת וְאָשׁ֣וּב אֲלֵיכֶ֔ם אָמַ֖ר יְהוָ֥ה צְבָאֽוֹת׃ (ה) אַל־ תִּהְי֣וּ כַאֲבֹֽתֵיכֶ֡ם אֲשֶׁ֣ר קָרְאֽוּ־ אֲלֵיהֶם֩ הַנְּבִיאִ֨ים הָרִֽאשֹׁנִ֜ים לֵאמֹ֗ר כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת שׁ֤וּבוּ נָא֙ מִדַּרְכֵיכֶ֣ם הָרָעִ֔ים (ומעליליכם) וּמַֽעֲלְלֵיכֶ֖ם הָֽרָעִ֑ים וְלֹ֥א שָׁמְע֛וּ וְלֹֽא־ הִקְשִׁ֥יבוּ אֵלַ֖י נְאֻם־ יְהוָֽה׃ (ו) אֲבֽוֹתֵיכֶ֖ם אַיֵּה־ הֵ֑ם וְהַ֨נְּבִאִ֔ים הַלְעוֹלָ֖ם יִֽחְיֽוּ׃ (ז) אַ֣ךְ ׀ דְּבָרַ֣י וְחֻקַּ֗י אֲשֶׁ֤ר צִוִּ֙יתִי֙ אֶת־ עֲבָדַ֣י הַנְּבִיאִ֔ים הֲל֥וֹא הִשִּׂ֖יגוּ אֲבֹתֵיכֶ֑ם וַיָּשׁ֣וּבוּ וַיֹּאמְר֗וּ כַּאֲשֶׁ֨ר זָמַ֜ם יְהוָ֤ה צְבָאוֹת֙ לַעֲשׂ֣וֹת לָ֔נוּ כִּדְרָכֵ֙ינוּ֙ וּכְמַ֣עֲלָלֵ֔ינוּ כֵּ֖ן עָשָׂ֥ה אִתָּֽנוּ׃ (ח) בְּיוֹם֩ עֶשְׂרִ֨ים וְאַרְבָּעָ֜ה לְעַשְׁתֵּֽי־ עָשָׂ֥ר חֹ֙דֶשׁ֙ הוּא־ חֹ֣דֶשׁ שְׁבָ֔ט בִּשְׁנַ֥ת שְׁתַּ֖יִם לְדָרְיָ֑וֶשׁ הָיָ֣ה דְבַר־ יְהוָ֗ה אֶל־ זְכַרְיָה֙ בֶּן־ בֶּ֣רֶכְיָ֔הוּ בֶּן־ עִדּ֥וֹא הַנָּבִ֖יא לֵאמֹֽר׃ (ט) רָאִ֣יתִי ׀ הַלַּ֗יְלָה וְהִנֵּה־ אִישׁ֙ רֹכֵב֙ עַל־ ס֣וּס אָדֹ֔ם וְה֣וּא עֹמֵ֔ד בֵּ֥ין הַהֲדַסִּ֖ים אֲשֶׁ֣ר בַּמְּצֻלָ֑ה וְאַחֲרָיו֙ סוּסִ֣ים אֲדֻמִּ֔ים שְׂרֻקִּ֖ים וּלְבָנִֽים׃ (י) וָאֹמַ֖ר מָה־ אֵ֣לֶּה אֲדֹנִ֑י וַיֹּ֣אמֶר אֵלַ֗י הַמַּלְאָךְ֙ הַדֹּבֵ֣ר בִּ֔י אֲנִ֥י אַרְאֶ֖ךָּ מָה־ הֵ֥מָּה אֵֽלֶּה׃ (יא) וַיַּ֗עַן הָאִ֛ישׁ הָעֹמֵ֥ד בֵּין־ הַהַדַסִּ֖ים וַיֹּאמַ֑ר אֵ֚לֶּה אֲשֶׁ֣ר שָׁלַ֣ח יְהוָ֔ה לְהִתְהַלֵּ֖ךְ בָּאָֽרֶץ׃ (יב) וַֽיַּעֲנ֞וּ אֶת־ מַלְאַ֣ךְ יְהוָ֗ה הָֽעֹמֵד֙ בֵּ֣ין הַהֲדַסִּ֔ים וַיֹּאמְר֖וּ הִתְהַלַּ֣כְנוּ בָאָ֑רֶץ וְהִנֵּ֥ה כָל־ הָאָ֖רֶץ יֹשֶׁ֥בֶת וְשֹׁקָֽטֶת׃ (יג) וַיַּ֣עַן מַלְאַךְ־ יְהוָה֮ וַיֹּאמַר֒ יְהוָ֣ה צְבָא֔וֹת עַד־ מָתַ֗י אַתָּה֙ לֹֽא־ תְרַחֵ֣ם אֶת־ יְרוּשָׁלִַ֔ם וְאֵ֖ת עָרֵ֣י יְהוּדָ֑ה אֲשֶׁ֣ר זָעַ֔מְתָּה זֶ֖ה שִׁבְעִ֥ים שָׁנָֽה׃ (יד) וַיַּ֣עַן יְהוָ֗ה אֶת־ הַמַּלְאָ֛ךְ הַדֹּבֵ֥ר בִּ֖י דְּבָרִ֣ים טוֹבִ֑ים דְּבָרִ֖ים נִחֻמִֽים׃ (טו) וַיֹּ֣אמֶר אֵלַ֗י הַמַּלְאָךְ֙ הַדֹּבֵ֣ר בִּ֔י קְרָ֣א לֵאמֹ֔ר כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָא֑וֹת קִנֵּ֧אתִי לִירוּשָׁלִַ֛ם וּלְצִיּ֖וֹן קִנְאָ֥ה גְדוֹלָֽה׃ (טז) וְקֶ֤צֶף גָּדוֹל֙ אֲנִ֣י קֹצֵ֔ף עַל־ הַגּוֹיִ֖ם הַשַּֽׁאֲנַנִּ֑ים אֲשֶׁ֤ר אֲנִי֙ קָצַ֣פְתִּי מְּעָ֔ט וְהֵ֖מָּה עָזְר֥וּ לְרָעָֽה׃ (יז) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה שַׁ֤בְתִּי לִירוּשָׁלִַ֙ם֙ בְּֽרַחֲמִ֔ים בֵּיתִי֙ יִבָּ֣נֶה בָּ֔הּ נְאֻ֖ם יְהוָ֣ה צְבָא֑וֹת (וקוה) וְקָ֥ו יִנָּטֶ֖ה עַל־ יְרוּשָׁלִָֽם׃ (יח) ע֣וֹד ׀ קְרָ֣א לֵאמֹ֗ר כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת ע֛וֹד תְּפוּצֶ֥ינָה עָרַ֖י מִטּ֑וֹב וְנִחַ֨ם יְהוָ֥ה עוֹד֙ אֶת־ צִיּ֔וֹן וּבָחַ֥ר ע֖וֹד בִּירוּשָׁלִָֽם׃ (יט) וָאֶשָּׂ֥א אֶת־ עֵינַ֖י וָאֵ֑רֶא וְהִנֵּ֖ה אַרְבַּ֥ע קְרָנֽוֹת׃ (כ) וָאֹמַ֗ר יְהוָ֤ה בְּךָ֙ הַשָּׂטָ֔ן וְיִגְעַ֤ר יְהוָה֙ בְּךָ֔ הַבֹּחֵ֖ר בִּירֽוּשָׁלִָ֑ם הֲל֧וֹא זֶ֦ה א֖וּד מֻצָּ֥ל מֵאֵֽשׁ׃ (לח) וִיהוֹשֻׁ֕עַ הָיָ֥ה לָבֻ֖שׁ בְּגָדִ֣ים צוֹאִ֑ים וְעֹמֵ֖ד לִפְנֵ֥י הַמַּלְאָֽךְ׃ (לט) וַיַּ֣עַן וַיֹּ֗אמֶר אֶל־ הָעֹמְדִ֤ים לְפָנָיו֙ לֵאמֹ֔ר הָסִ֛ירוּ הַבְּגָדִ֥ים הַצֹּאִ֖ים מֵעָלָ֑יו וַיֹּ֣אמֶר אֵלָ֗יו רְאֵ֨ה הֶעֱבַ֤רְתִּי מֵעָלֶ֙יךָ֙ עֲוֺנֶ֔ךָ וְהַלְבֵּ֥שׁ אֹתְךָ֖ מַחֲלָצֽוֹת׃ (מ) וָאֹמַ֕ר יָשִׂ֛ימוּ צָנִ֥יף טָה֖וֹר עַל־ רֹאשׁ֑וֹ וַיָּשִׂימוּ֩ הַצָּנִ֨יף הַטָּה֜וֹר עַל־ רֹאשׁ֗וֹ וַיַּלְבִּשֻׁ֙הוּ֙ בְּגָדִ֔ים וּמַלְאַ֥ךְ יְהוָ֖ה עֹמֵֽד׃ (מא) וַיָּ֙עַד֙ מַלְאַ֣ךְ יְהוָ֔ה בִּיהוֹשֻׁ֖עַ לֵאמֹֽר׃ (מב) כֹּה־ אָמַ֞ר יְהוָ֣ה צְבָא֗וֹת אִם־ בִּדְרָכַ֤י תֵּלֵךְ֙ וְאִ֣ם אֶת־ מִשְׁמַרְתִּ֣י תִשְׁמֹ֔ר וְגַם־ אַתָּה֙ תָּדִ֣ין אֶת־ בֵּיתִ֔י וְגַ֖ם תִּשְׁמֹ֣ר אֶת־ חֲצֵרָ֑י וְנָתַתִּ֤י לְךָ֙ מַהְלְכִ֔ים בֵּ֥ין הָעֹמְדִ֖ים הָאֵֽלֶּה׃ (מג) שְֽׁמַֽע־ נָ֞א יְהוֹשֻׁ֣עַ ׀ הַכֹּהֵ֣ן הַגָּד֗וֹל אַתָּה֙ וְרֵעֶ֙יךָ֙ הַיֹּשְׁבִ֣ים לְפָנֶ֔יךָ כִּֽי־ אַנְשֵׁ֥י מוֹפֵ֖ת הֵ֑מָּה כִּֽי־ הִנְנִ֥י מֵבִ֛יא אֶת־ עַבְדִּ֖י צֶֽמַח׃ (מד) כִּ֣י ׀ הִנֵּ֣ה הָאֶ֗בֶן אֲשֶׁ֤ר נָתַ֙תִּי֙ לִפְנֵ֣י יְהוֹשֻׁ֔עַ עַל־ אֶ֥בֶן אַחַ֖ת שִׁבְעָ֣ה עֵינָ֑יִם הִנְנִ֧י מְפַתֵּ֣חַ פִּתֻּחָ֗הּ נְאֻם֙ יְהוָ֣ה צְבָא֔וֹת וּמַשְׁתִּ֛י אֶת־ עֲוֺ֥ן הָאָֽרֶץ־ הַהִ֖יא בְּי֥וֹם אֶחָֽד׃ (מה) בַּיּ֣וֹם הַה֗וּא נְאֻם֙ יְהוָ֣ה צְבָא֔וֹת תִּקְרְא֖וּ אִ֣ישׁ לְרֵעֵ֑הוּ אֶל־ תַּ֥חַת גֶּ֖פֶן וְאֶל־ תַּ֥חַת תְּאֵנָֽה׃ (מו) וַיָּ֕שָׁב הַמַּלְאָ֖ךְ הַדֹּבֵ֣ר בִּ֑י וַיְעִירֵ֕נִי כְּאִ֖ישׁ אֲשֶׁר־ יֵע֥וֹר מִשְּׁנָתֽוֹ׃
