תהילים פרק סז לפי החלוקה לסדרים

פרק סז בספר תהילים (8 פסוקים) נמצא בתהילים סדר ח (פסוקים קכה–קלא) ובתהילים סדר ט (פסוקים א–א).

תהילים סדר ח

(א) ר֣וּמָה עַל־ שָׁמַ֣יִם אֱלֹהִ֑ים עַ֖ל כָּל־ הָאָ֣רֶץ כְּבוֹדֶֽךָ׃ (ב) לַמְנַצֵּ֥חַ אַל־ תַּשְׁחֵ֗ת לְדָוִ֥ד מִכְתָּֽם׃ (ג) הַֽאֻמְנָ֗ם אֵ֣לֶם צֶ֭דֶק תְּדַבֵּר֑וּן מֵישָׁרִ֥ים תִּ֝שְׁפְּט֗וּ בְּנֵ֣י אָדָֽם׃ (ד) אַף־ בְּלֵב֮ עוֹלֹ֪ת תִּפְעָ֫ל֥וּן בָּאָ֡רֶץ חֲמַ֥ס יְ֝דֵיכֶ֗ם תְּפַלֵּֽסֽוּן׃ (ה) זֹ֣רוּ רְשָׁעִ֣ים מֵרָ֑חֶם תָּע֥וּ מִ֝בֶּ֗טֶן דֹּבְרֵ֥י כָזָֽב׃ (ו) חֲמַת־ לָ֗מוֹ כִּדְמ֥וּת חֲמַת־ נָחָ֑שׁ כְּמוֹ־ פֶ֥תֶן חֵ֝רֵ֗שׁ יַאְטֵ֥ם אָזְנֽוֹ׃ (ז) אֲשֶׁ֣ר לֹא־ יִ֭שְׁמַע לְק֣וֹל מְלַחֲשִׁ֑ים חוֹבֵ֖ר חֲבָרִ֣ים מְחֻכָּֽם׃ (ח) אֱ‍ֽלֹהִ֗ים הֲרָס־ שִׁנֵּ֥ימוֹ בְּפִ֑ימוֹ מַלְתְּע֥וֹת כְּ֝פִירִ֗ים נְתֹ֣ץ ׀ יְהוָֽה׃ (ט) יִמָּאֲס֣וּ כְמוֹ־ מַ֭יִם יִתְהַלְּכוּ־ לָ֑מוֹ יִדְרֹ֥ךְ (חצו) חִ֝צָּ֗יו כְּמ֣וֹ יִתְמֹלָֽלוּ׃ (י) כְּמ֣וֹ שַׁ֭בְּלוּל תֶּ֣מֶס יַהֲלֹ֑ךְ נֵ֥פֶל אֵ֝֗שֶׁת בַּל־ חָ֥זוּ שָֽׁמֶשׁ׃ (יא) בְּטֶ֤רֶם יָבִ֣ינוּ סִּֽירֹתֵיכֶ֣ם אָטָ֑ד כְּמוֹ־ חַ֥י כְּמוֹ־ חָ֝ר֗וֹן יִשְׂעָרֶֽנּוּ׃ (יב) יִשְׂמַ֣ח צַ֭דִּיק כִּי־ חָזָ֣ה נָקָ֑ם פְּעָמָ֥יו יִ֝רְחַ֗ץ בְּדַ֣ם הָרָשָֽׁע׃ (יג) וְיֹאמַ֣ר אָ֭דָם אַךְ־ פְּרִ֣י לַצַּדִּ֑יק אַ֥ךְ יֵשׁ־ אֱ֝לֹהִ֗ים שֹׁפְטִ֥ים בָּאָֽרֶץ׃ (יד) לַמְנַצֵּ֣חַ אַל־ תַּשְׁחֵת֮ לְדָוִ֪ד מִ֫כְתָּ֥ם בִּשְׁלֹ֥חַ שָׁא֑וּל וַֽיִּשְׁמְר֥וּ אֶת־ הַ֝בַּ֗יִת לַהֲמִיתֽוֹ׃ (טו) הַצִּילֵ֖נִי מֵאֹיְבַ֥י ׀ אֱלֹהָ֑י מִּמִתְקוֹמְמַ֥י תְּשַׂגְּבֵֽנִי׃ (טז) הַ֭צִּילֵנִי מִפֹּ֣עֲלֵי אָ֑וֶן וּֽמֵאַנְשֵׁ֥י דָ֝מִ֗ים הוֹשִׁיעֵֽנִי׃ (יז) כִּ֤י הִנֵּ֪ה אָֽרְב֡וּ לְנַפְשִׁ֗י יָג֣וּרוּ עָלַ֣י עַזִ֑ים לֹא־ פִשְׁעִ֖י וְלֹא־ חַטָּאתִ֣י יְהוָֽה׃ (יח) בְּֽלִי־ עָ֭וֺן יְרוּצ֣וּן וְיִכּוֹנָ֑נוּ ע֖וּרָה לִקְרָאתִ֣י וּרְאֵה׃ (יט) וְאַתָּ֤ה יְהוָֽה־ אֱלֹהִ֥ים ׀ צְבָא֡וֹת אֱלֹ֘הֵ֤י יִשְׂרָאֵ֗ל הָקִ֗יצָה לִפְקֹ֥ד כָּֽל־ הַגּוֹיִ֑ם אַל־ תָּחֹ֨ן כָּל־ בֹּ֖גְדֵי אָ֣וֶן סֶֽלָה׃ (כ) יָשׁ֣וּבוּ לָ֭עֶרֶב יֶהֱמ֥וּ כַכָּ֗לֶב וִיס֥וֹבְבוּ עִֽיר׃ (כא) הִנֵּ֤ה ׀ יַבִּ֘יע֤וּן בְּפִיהֶ֗ם חֲ֭רָבוֹת בְּשִׂפְתוֹתֵיהֶ֑ם כִּי־ מִ֥י שֹׁמֵֽעַ׃ (כב) וְאַתָּ֣ה יְ֭הוָה תִּשְׂחַק־ לָ֑מוֹ תִּ֝לְעַ֗ג לְכָל־ גּוֹיִֽם׃ (כג) עֻ֭זּוֹ אֵלֶ֣יךָ אֶשְׁמֹ֑רָה כִּֽי־ אֱ֝לֹהִ֗ים מִשְׂגַּבִּֽי׃ (כד) אֱלֹהֵ֣י (חסדו) חַסְדִּ֣י יְקַדְּמֵ֑נִי אֱ֝לֹהִ֗ים יַרְאֵ֥נִי בְשֹׁרְרָֽי׃ (כה) אַל־ תַּהַרְגֵ֤ם ׀ פֶּֽן־ יִשְׁכְּח֬וּ עַמִּ֗י הֲנִיעֵ֣מוֹ בְ֭חֵילְךָ וְהוֹרִידֵ֑מוֹ מָֽגִנֵּ֣נוּ אֲדֹנָֽי׃ (כו) חַטַּאת־ פִּ֗ימוֹ דְּֽבַר־ שְׂפָ֫תֵ֥ימוֹ וְיִלָּכְד֥וּ בִגְאוֹנָ֑ם וּמֵאָלָ֖ה וּמִכַּ֣חַשׁ יְסַפֵּֽרוּ׃ (כז) כַּלֵּ֥ה בְחֵמָה֮ כַּלֵּ֪ה וְֽאֵ֫ינֵ֥מוֹ וְֽיֵדְע֗וּ כִּֽי־ אֱ֭לֹהִים מֹשֵׁ֣ל בְּיַעֲקֹ֑ב לְאַפְסֵ֖י הָאָ֣רֶץ סֶֽלָה׃ (כח) וְיָשׁ֣וּבוּ לָ֭עֶרֶב יֶהֱמ֥וּ כַכָּ֗לֶב וִיס֥וֹבְבוּ עִֽיר׃ (כט) הֵ֭מָּה (ינועון) יְנִיע֣וּן לֶאֱכֹ֑ל אִם־ לֹ֥א יִ֝שְׂבְּע֗וּ וַיָּלִֽינוּ׃ (ל) וַאֲנִ֤י ׀ אָשִׁ֣יר עֻזֶּךָ֮ וַאֲרַנֵּ֥ן לַבֹּ֗קֶר חַ֫סְדֶּ֥ךָ כִּֽי־ הָיִ֣יתָ מִשְׂגָּ֣ב לִ֑י וּ֝מָנ֗וֹס בְּי֣וֹם צַר־ לִֽי׃ (לא) עֻ֭זִּי אֵלֶ֣יךָ אֲזַמֵּ֑רָה כִּֽי־ אֱלֹהִ֥ים מִ֝שְׂגַּבִּ֗י אֱלֹהֵ֥י חַסְדִּֽי׃ (לב) לַ֭מְנַצֵּחַ עַל־ שׁוּשַׁ֣ן עֵד֑וּת מִכְתָּ֖ם לְדָוִ֣ד לְלַמֵּֽד׃ (לג) בְּהַצּוֹת֨וֹ ׀ אֶ֥ת אֲרַ֣ם נַהֲרַיִם֮ וְאֶת־ אֲרַ֪ם צ֫וֹבָ֥ה וַיָּ֤שָׁב יוֹאָ֗ב וַיַּ֣ךְ אֶת־ אֱד֣וֹם בְּגֵיא־ מֶ֑לַח שְׁנֵ֖ים עָשָׂ֣ר אָֽלֶף׃ (לד) אֱ֭לֹהִים זְנַחְתָּ֣נוּ פְרַצְתָּ֑נוּ אָ֝נַ֗פְתָּ תְּשׁ֣וֹבֵ֥ב לָֽנוּ׃ (לה) הִרְעַ֣שְׁתָּה אֶ֣רֶץ פְּצַמְתָּ֑הּ רְפָ֖ה שְׁבָרֶ֣יהָ כִי־ מָֽטָה׃ (לו) הִרְאִ֣יתָה עַמְּךָ֣ קָשָׁ֑ה הִ֝שְׁקִיתָ֗נוּ יַ֣יִן תַּרְעֵלָֽה׃ (לז) נָ֘תַ֤תָּה לִּירֵאֶ֣יךָ נֵּ֭ס לְהִתְנוֹסֵ֑ס מִ֝פְּנֵ֗י קֹ֣שֶׁט סֶֽלָה׃ (לח) לְ֭מַעַן יֵחָלְצ֣וּן יְדִידֶ֑יךָ הוֹשִׁ֖יעָה יְמִֽינְךָ֣ (ועננו) וַעֲנֵֽנִי׃ (לט) אֱלֹהִ֤ים ׀ דִּבֶּ֥ר בְּקָדְשׁ֗וֹ אֶ֫עְלֹ֥זָה אֲחַלְּקָ֥ה שְׁכֶ֑ם וְעֵ֖מֶק סֻכּ֣וֹת אֲמַדֵּֽד׃ (מ) לִ֤י גִלְעָ֨ד ׀ וְלִ֬י מְנַשֶּׁ֗ה וְ֭אֶפְרַיִם מָע֣וֹז רֹאשִׁ֑י יְ֝הוּדָ֗ה מְחֹֽקְקִי׃ (מא) מוֹאָ֤ב ׀ סִ֬יר רַחְצִ֗י עַל־ אֱ֭דוֹם אַשְׁלִ֣יךְ נַעֲלִ֑י עָ֝לַ֗י פְּלֶ֣שֶׁת הִתְרֹעָֽעִֽי׃ (מב) מִ֣י יֹ֭בִלֵנִי עִ֣יר מָצ֑וֹר מִ֖י נָחַ֣נִי עַד־ אֱדֽוֹם׃ (מג) הֲלֹֽא־ אַתָּ֣ה אֱלֹהִ֣ים זְנַחְתָּ֑נוּ וְֽלֹא־ תֵצֵ֥א אֱ֝לֹהִ֗ים בְּצִבְאוֹתֵֽינוּ׃ (מד) הָֽבָה־ לָּ֣נוּ עֶזְרָ֣ת מִצָּ֑ר וְ֝שָׁ֗וְא תְּשׁוּעַ֥ת אָדָם׃ (מה) בֵּֽאלֹהִ֥ים נַעֲשֶׂה־ חָ֑יִל וְ֝ה֗וּא יָב֥וּס צָרֵֽינוּ׃ (מו) לַמְנַצֵּ֬חַ ׀ עַֽל־ נְגִינַ֬ת לְדָוִֽד׃ (מז) שִׁמְעָ֣ה אֱ֭לֹהִים רִנָּתִ֑י הַ֝קְשִׁ֗יבָה תְּפִלָּתִֽי׃ (מח) מִקְצֵ֤ה הָאָ֨רֶץ ׀ אֵלֶ֣יךָ אֶ֭קְרָא בַּעֲטֹ֣ף לִבִּ֑י בְּצוּר־ יָר֖וּם מִמֶּ֣נִּי תַנְחֵֽנִי׃ (מט) כִּֽי־ הָיִ֣יתָ מַחְסֶ֣ה לִ֑י מִגְדַּל־ עֹ֝֗ז מִפְּנֵ֥י אוֹיֵֽב׃ (נ) אָג֣וּרָה בְ֭אָהָלְךָ עוֹלָמִ֑ים אֶֽחֱסֶ֨ה בְסֵ֖תֶר כְּנָפֶ֣יךָ סֶּֽלָה׃ (נא) כִּֽי־ אַתָּ֣ה אֱ֭לֹהִים שָׁמַ֣עְתָּ לִנְדָרָ֑י נָתַ֥תָּ יְ֝רֻשַּׁ֗ת יִרְאֵ֥י שְׁמֶֽךָ׃ (נב) יָמִ֣ים עַל־ יְמֵי־ מֶ֣לֶךְ תּוֹסִ֑יף שְׁ֝נוֹתָ֗יו כְּמוֹ־ דֹ֥ר וָדֹֽר׃ (נג) יֵשֵׁ֣ב ע֭וֹלָם לִפְנֵ֣י אֱלֹהִ֑ים חֶ֥סֶד וֶ֝אֱמֶ֗ת מַ֣ן יִנְצְרֻֽהוּ׃ (נד) כֵּ֤ן אֲזַמְּרָ֣ה שִׁמְךָ֣ לָעַ֑ד לְֽשַׁלְּמִ֥י נְדָרַ֗י י֣וֹם ׀ יֽוֹם׃ (נה) לַמְנַצֵּ֥חַ עַֽל־ יְדוּת֗וּן מִזְמ֥וֹר לְדָוִֽד׃ (נו) אַ֣ךְ אֶל־ אֱ֭לֹהִים דּֽוּמִיָּ֣ה נַפְשִׁ֑י מִ֝מֶּ֗נּוּ יְשׁוּעָתִֽי׃ (נז) אַךְ־ ה֣וּא צ֭וּרִי וִֽישׁוּעָתִ֑י מִ֝שְׂגַּבִּ֗י לֹא־ אֶמּ֥וֹט רַבָּֽה׃ (נח) עַד־ אָ֤נָה ׀ תְּהֽוֹתְת֣וּ עַל אִישׁ֮ תְּרָצְּח֪וּ כֻ֫לְּכֶ֥ם כְּקִ֥יר נָט֑וּי גָּ֝דֵ֗ר הַדְּחוּיָֽה׃ (נט) אַ֤ךְ מִשְּׂאֵת֨וֹ ׀ יָעֲצ֣וּ לְהַדִּיחַ֮ יִרְצ֪וּ כָ֫זָ֥ב בְּפִ֥יו יְבָרֵ֑כוּ וּ֝בְקִרְבָּ֗ם יְקַלְלוּ־ סֶֽלָה׃ (ס) אַ֣ךְ לֵ֭אלֹהִים דּ֣וֹמִּי נַפְשִׁ֑י כִּי־ מִ֝מֶּ֗נּוּ תִּקְוָתִֽי׃ (סא) אַךְ־ ה֣וּא לִפְנֵי־ שֶׁמֶשׁ֮ (ינין) יִנּ֪וֹן שְׁ֫מ֥וֹ וְיִתְבָּ֥רְכוּ ב֑וֹ כָּל־ גּוֹיִ֥ם יְאַשְּׁרֽוּהוּ׃ (קכב) בָּר֤וּךְ ׀ יְהוָ֣ה אֱ֭לֹהִים אֱלֹהֵ֣י יִשְׂרָאֵ֑ל עֹשֵׂ֖ה נִפְלָא֣וֹת לְבַדּֽוֹ׃ (קכג) וּבָר֤וּךְ ׀ שֵׁ֥ם כְּבוֹד֗וֹ לְע֫וֹלָ֥ם וְיִמָּלֵ֣א כְ֭בוֹדוֹ אֶת־ כֹּ֥ל הָאָ֗רֶץ אָ֘מֵ֥ן ׀ וְאָמֵֽן׃
