תהילים סדר ד

(א) יְֽהוָ֗ה עֹ֭ז לְעַמּ֣וֹ יִתֵּ֑ן יְהוָ֓ה ׀ יְבָרֵ֖ךְ אֶת־ עַמּ֣וֹ בַשָּׁלֽוֹם׃ (ב) מִזְמ֡וֹר שִׁיר־ חֲנֻכַּ֖ת הַבַּ֣יִת לְדָוִֽד׃ (ג) אֲרוֹמִמְךָ֣ יְ֭הוָה כִּ֣י דִלִּיתָ֑נִי וְלֹא־ שִׂמַּ֖חְתָּ אֹיְבַ֣י לִֽי׃ (ד) יְהוָ֥ה אֱלֹהָ֑י שִׁוַּ֥עְתִּי אֵ֝לֶ֗יךָ וַתִּרְפָּאֵֽנִי׃ (ה) יְֽהוָ֗ה הֶֽעֱלִ֣יתָ מִן־ שְׁא֣וֹל נַפְשִׁ֑י חִ֝יִּיתַ֗נִי (מיורדי)־ מִיָּֽרְדִי־ בֽוֹר׃ (ו) זַמְּר֣וּ לַיהוָ֣ה חֲסִידָ֑יו וְ֝הוֹד֗וּ לְזֵ֣כֶר קָדְשֽׁוֹ׃ (ז) כִּ֤י רֶ֨גַע ׀ בְּאַפּוֹ֮ חַיִּ֪ים בִּרְצ֫וֹנ֥וֹ בָּ֭עֶרֶב יָלִ֥ין בֶּ֗כִי וְלַבֹּ֥קֶר רִנָּֽה׃ (ח) וַ֭אֲנִי אָמַ֣רְתִּי בְשַׁלְוִ֑י בַּל־ אֶמּ֥וֹט לְעוֹלָֽם׃ (ט) יְֽהוָ֗ה בִּרְצוֹנְךָ֮ הֶעֱמַ֪דְתָּה לְֽהַרְרִ֫י עֹ֥ז הִסְתַּ֥רְתָּ פָנֶ֗יךָ הָיִ֥יתִי נִבְהָֽל׃ (י) אֵלֶ֣יךָ יְהוָ֣ה אֶקְרָ֑א וְאֶל־ אֲ֝דֹנָ֗י אֶתְחַנָּֽן׃ (יא) מַה־ בֶּ֥צַע בְּדָמִי֮ בְּרִדְתִּ֪י אֶ֫ל־ שָׁ֥חַת הֲיוֹדְךָ֥ עָפָ֑ר הֲיַגִּ֥יד אֲמִתֶּֽךָ׃ (יב) שְׁמַע־ יְהוָ֥ה וְחָנֵּ֑נִי יְ֝הוָה הֱ‍ֽיֵה־ עֹזֵ֥ר לִֽי׃ (יג) הָפַ֣כְתָּ מִסְפְּדִי֮ לְמָח֪וֹל לִ֥י פִּתַּ֥חְתָּ שַׂקִּ֑י וַֽתְּאַזְּרֵ֥נִי שִׂמְחָֽה׃ (יד) לְמַ֤עַן ׀ יְזַמֶּרְךָ֣ כָ֭בוֹד וְלֹ֣א יִדֹּ֑ם יְהוָ֥ה אֱ֝לֹהַ֗י לְעוֹלָ֥ם אוֹדֶֽךָּ׃ (טו) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (טז) בְּךָ֖ יְהוָ֣ה חָ֭סִיתִי אַל־ אֵב֣וֹשָׁה לְעוֹלָ֑ם בְּצִדְקָתְךָ֥ פַלְּטֵֽנִי׃ (יז) הַטֵּ֤ה אֵלַ֨י ׀ אָזְנְךָ֮ מְהֵרָ֪ה הַצִּ֫ילֵ֥נִי הֱיֵ֤ה לִ֨י ׀ לְֽצוּר־ מָ֭עוֹז לְבֵ֥ית מְצוּד֗וֹת לְהוֹשִׁיעֵֽנִי׃ (יח) כִּֽי־ סַלְעִ֣י וּמְצוּדָתִ֣י אָ֑תָּה וּלְמַ֥עַן שִׁ֝מְךָ֗ תַּֽנְחֵ֥נִי וּֽתְנַהֲלֵֽנִי׃ (יט) תּוֹצִיאֵ֗נִי מֵרֶ֣שֶׁת ז֭וּ טָ֣מְנוּ לִ֑י כִּֽי־ אַ֝תָּה מָֽעוּזִּֽי׃ (כ) בְּיָדְךָ֮ אַפְקִ֪יד ר֫וּחִ֥י פָּדִ֖יתָה אוֹתִ֥י יְהוָ֗ה אֵ֣ל אֱמֶֽת׃ (כא) שָׂנֵ֗אתִי הַשֹּׁמְרִ֥ים הַבְלֵי־ שָׁ֑וְא וַ֝אֲנִ֗י אֶל־ יְהוָ֥ה בָּטָֽחְתִּי׃ (כב) אָגִ֥ילָה וְאֶשְׂמְחָ֗ה בְּחַ֫סְדֶּ֥ךָ אֲשֶׁ֣ר רָ֭אִיתָ אֶת־ עָנְיִ֑י יָ֝דַ֗עְתָּ בְּצָר֥וֹת נַפְשִֽׁי׃ (כג) וְלֹ֣א הִ֭סְגַּרְתַּנִי בְּיַד־ אוֹיֵ֑ב הֶֽעֱמַ֖דְתָּ בַמֶּרְחָ֣ב רַגְלָֽי׃ (כד) חָנֵּ֥נִי יְהוָה֮ כִּ֤י צַ֫ר־ לִ֥י עָשְׁשָׁ֖ה בְכַ֥עַס עֵינִ֗י נַפְשִׁ֥י וּבִטְנִֽי׃ (כה) כִּ֤י כָל֪וּ בְיָג֡וֹן חַיַּי֮ וּשְׁנוֹתַ֪י בַּאֲנָ֫חָ֥ה כָּשַׁ֣ל בַּעֲוֺנִ֣י כֹחִ֑י וַעֲצָמַ֥י עָשֵֽׁשׁוּ׃ (כו) מִכָּל־ צֹרְרַ֨י הָיִ֪יתִי חֶרְפָּ֡ה וְלִשֲׁכֵנַ֨י ׀ מְאֹד֮ וּפַ֪חַד לִֽמְיֻדָּ֫עָ֥י רֹאַ֥י בַּח֑וּץ נָדְד֥וּ מִמֶּֽנִּי׃ (כז) נִ֭שְׁכַּחְתִּי כְּמֵ֣ת מִלֵּ֑ב הָ֝יִ֗יתִי כִּכְלִ֥י אֹבֵֽד׃ (כח) כִּ֤י שָׁמַ֨עְתִּי ׀ דִּבַּ֥ת רַבִּים֮ מָג֪וֹר מִסָּ֫בִ֥יב בְּהִוָּסְדָ֣ם יַ֣חַד עָלַ֑י לָקַ֖חַת נַפְשִׁ֣י זָמָֽמוּ׃ (כט) וַאֲנִ֤י ׀ עָלֶ֣יךָ בָטַ֣חְתִּי יְהוָ֑ה אָ֝מַ֗רְתִּי אֱלֹהַ֥י אָֽתָּה׃ (ל) בְּיָדְךָ֥ עִתֹּתָ֑י הַצִּ֘ילֵ֤נִי מִיַּד־ א֝וֹיְבַ֗י וּמֵרֹדְפָֽי׃ (לא) הָאִ֣ירָה פָ֭נֶיךָ עַל־ עַבְדֶּ֑ךָ ה֖וֹשִׁיעֵ֣נִי בְחַסְדֶּֽךָ׃ (לב) יְֽהוָ֗ה אַל־ אֵ֭בוֹשָׁה כִּ֣י קְרָאתִ֑יךָ יֵבֹ֥שׁוּ רְ֝שָׁעִ֗ים יִדְּמ֥וּ לִשְׁאֽוֹל׃ (לג) תֵּ֥אָלַ֗מְנָה שִׂפְתֵ֫י שָׁ֥קֶר הַדֹּבְר֖וֹת עַל־ צַדִּ֥יק עָתָ֗ק בְּגַאֲוָ֥ה וָבֽוּז׃ (לד) מָ֤ה רַֽב־ טוּבְךָ֮ אֲשֶׁר־ צָפַ֪נְתָּ לִּֽירֵ֫אֶ֥יךָ פָּ֭עַלְתָּ לַחֹסִ֣ים בָּ֑ךְ נֶ֝֗גֶד בְּנֵ֣י אָדָם׃ (לה) תַּסְתִּירֵ֤ם ׀ בְּסֵ֥תֶר פָּנֶיךָ֮ מֵֽרֻכְסֵ֫י אִ֥ישׁ תִּצְפְּנֵ֥ם בְּסֻכָּ֗ה מֵרִ֥יב לְשֹׁנֽוֹת׃ (לו) בָּר֥וּךְ יְהוָ֑ה כִּ֥י הִפְלִ֘יא חַסְדּ֥וֹ לִ֝֗י בְּעִ֣יר מָצֽוֹר׃ (לז) וַאֲנִ֤י ׀ אָ֘מַ֤רְתִּי בְחָפְזִ֗י נִגְרַזְתִּי֮ מִנֶּ֪גֶד עֵ֫ינֶ֥יךָ אָכֵ֗ן שָׁ֭מַעְתָּ ק֥וֹל תַּחֲנוּנַ֗י בְּשַׁוְּעִ֥י אֵלֶֽיךָ׃ (לח) אֶֽהֱב֥וּ אֶת־ יְהוָ֗ה כָּֽל־ חֲסִ֫ידָ֥יו אֱ֭מוּנִים נֹצֵ֣ר יְהוָ֑ה וּמְשַׁלֵּ֥ם עַל־ יֶ֝֗תֶר עֹשֵׂ֥ה גַאֲוָֽה׃ (לט) חִ֭זְקוּ וְיַאֲמֵ֣ץ לְבַבְכֶ֑ם כָּל־ הַ֝מְיַחֲלִ֗ים לַיהוָֽה׃ (מ) לְדָוִ֗ד מַ֫שְׂכִּ֥יל אַשְׁרֵ֥י נְֽשׂוּי־ פֶּ֗שַׁע כְּס֣וּי חֲטָאָֽה׃ (מא) אַ֥שְֽׁרֵי אָדָ֗ם לֹ֤א יַחְשֹׁ֬ב יְהוָ֣ה ל֣וֹ עָוֺ֑ן וְאֵ֖ין בְּרוּח֣וֹ רְמִיָּה׃ (מב) כִּֽי־ הֶ֭חֱרַשְׁתִּי בָּל֣וּ עֲצָמָ֑י בְּ֝שַׁאֲגָתִ֗י כָּל־ הַיּֽוֹם׃ (מג) כִּ֤י ׀ יוֹמָ֣ם וָלַיְלָה֮ תִּכְבַּ֥ד עָלַ֗י יָ֫דֶ֥ךָ נֶהְפַּ֥ךְ לְשַׁדִּ֑י בְּחַרְבֹ֖נֵי קַ֣יִץ סֶֽלָה׃ (מד) חַטָּאתִ֨י אוֹדִ֪יעֲךָ֡ וַעֲוֺ֘נִ֤י לֹֽא־ כִסִּ֗יתִי אָמַ֗רְתִּי אוֹדֶ֤ה עֲלֵ֣י פְ֭שָׁעַי לַיהוָ֑ה וְאַתָּ֨ה נָ֘שָׂ֤אתָ עֲוֺ֖ן חַטָּאתִ֣י סֶֽלָה׃ (מה) עַל־ זֹ֡את יִתְפַּלֵּ֬ל כָּל־ חָסִ֨יד ׀ אֵלֶיךָ֮ לְעֵ֪ת מְ֫צֹ֥א רַ֗ק לְ֭שֵׁטֶף מַ֣יִם רַבִּ֑ים אֵ֝לָ֗יו לֹ֣א יַגִּֽיעוּ׃ (מו) אַתָּ֤ה ׀ סֵ֥תֶר לִי֮ מִצַּ֪ר תִּ֫צְּרֵ֥נִי רָנֵּ֥י פַלֵּ֑ט תְּס֖וֹבְבֵ֣נִי סֶֽלָה׃ (מז) אַשְׂכִּֽילְךָ֨ ׀ וְֽאוֹרְךָ֗ בְּדֶֽרֶךְ־ ז֥וּ תֵלֵ֑ךְ אִֽיעֲצָ֖ה עָלֶ֣יךָ עֵינִֽי׃ (מח) אַל־ תִּֽהְי֤וּ ׀ כְּס֥וּס כְּפֶרֶד֮ אֵ֤ין הָ֫בִ֥ין בְּמֶֽתֶג־ וָרֶ֣סֶן עֶדְי֣וֹ לִבְל֑וֹם בַּ֝֗ל קְרֹ֣ב אֵלֶֽיךָ׃ (מט) רַבִּ֥ים מַכְאוֹבִ֗ים לָרָ֫שָׁ֥ע וְהַבּוֹטֵ֥חַ בַּיהוָ֑ה חֶ֝֗סֶד יְסוֹבְבֶֽנּוּ׃ (נ) שִׂמְח֬וּ בַֽיהוָ֣ה וְ֭גִילוּ צַדִּיקִ֑ים וְ֝הַרְנִ֗ינוּ כָּל־ יִשְׁרֵי־ לֵֽב׃ (נא) רַנְּנ֣וּ צַ֭דִּיקִים בַּֽיהוָ֑ה לַ֝יְשָׁרִ֗ים נָאוָ֥ה תְהִלָּֽה׃ (נב) הוֹד֣וּ לַיהוָ֣ה בְּכִנּ֑וֹר בְּנֵ֥בֶל עָ֝שׂ֗וֹר זַמְּרוּ־ לֽוֹ׃ (נג) שִֽׁירוּ־ ל֭וֹ שִׁ֣יר חָדָ֑שׁ הֵיטִ֥יבוּ נַ֝גֵּ֗ן בִּתְרוּעָֽה׃ (נד) כִּֽי־ יָשָׁ֥ר דְּבַר־ יְהוָ֑ה וְכָל־ מַ֝עֲשֵׂ֗הוּ בֶּאֱמוּנָֽה׃ (נה) אֹ֭הֵב צְדָקָ֣ה וּמִשְׁפָּ֑ט חֶ֥סֶד יְ֝הוָ֗ה מָלְאָ֥ה הָאָֽרֶץ׃ (נו) בִּדְבַ֣ר יְ֭הוָה שָׁמַ֣יִם נַעֲשׂ֑וּ וּבְר֥וּחַ פִּ֝֗יו כָּל־ צְבָאָֽם׃ (נז) כֹּנֵ֣ס כַּ֭נֵּד מֵ֣י הַיָּ֑ם נֹתֵ֖ן בְּאֹצָר֣וֹת תְּהוֹמֽוֹת׃ (נח) יִֽירְא֣וּ מֵ֭יְהוָה כָּל־ הָאָ֑רֶץ מִמֶּ֥נּוּ יָ֝ג֗וּרוּ כָּל־ יֹשְׁבֵ֥י תֵבֵֽל׃ (נט) כִּ֤י ה֣וּא אָמַ֣ר וַיֶּ֑הִי בְּעֹ֖ד רָשָׁ֣ע לְנֶגְדִּֽי׃ (פ) נֶאֱלַ֣מְתִּי ד֭וּמִיָּה הֶחֱשֵׁ֣יתִי מִטּ֑וֹב וּכְאֵבִ֥י נֶעְכָּֽר׃ (פא) חַם־ לִבִּ֨י ׀ בְּקִרְבִּ֗י בַּהֲגִיגִ֥י תִבְעַר־ אֵ֑שׁ דִּ֝בַּ֗רְתִּי בִּלְשֽׁוֹנִי׃ (פב) הוֹדִ֘יעֵ֤נִי יְהוָ֨ה ׀ קִצִּ֗י וּמִדַּ֣ת יָמַ֣י מַה־ הִ֑יא אֵ֝דְעָ֗ה מֶה־ חָדֵ֥ל אָֽנִי׃ (פג) הִנֵּ֤ה טְפָח֨וֹת ׀ נָ֘תַ֤תָּה יָמַ֗י וְחֶלְדִּ֣י כְאַ֣יִן נֶגְדֶּ֑ךָ אַ֥ךְ כָּֽל־ הֶ֥בֶל כָּל־ אָ֝דָ֗ם נִצָּ֥ב סֶֽלָה׃ (פד) אַךְ־ בְּצֶ֤לֶם ׀ יִֽתְהַלֶּךְ־ אִ֗ישׁ אַךְ־ הֶ֥בֶל יֶהֱמָי֑וּן יִ֝צְבֹּ֗ר וְֽלֹא־ יֵדַ֥ע מִי־ אֹסְפָֽם׃ (פה) וְעַתָּ֣ה מַה־ קִּוִּ֣יתִי אֲדֹנָ֑י תּ֝וֹחַלְתִּ֗י לְךָ֣ הִֽיא׃ (פו) מִכָּל־ פְּשָׁעַ֥י הַצִּילֵ֑נִי חֶרְפַּ֥ת נָ֝בָ֗ל אַל־ תְּשִׂימֵֽנִי׃ (פז) נֶ֭אֱלַמְתִּי לֹ֣א אֶפְתַּח־ פִּ֑י כִּ֖י אַתָּ֣ה עָשִֽׂיתָ׃ (פח) הָסֵ֣ר מֵעָלַ֣י נִגְעֶ֑ךָ מִתִּגְרַ֥ת יָ֝דְךָ֗ אֲנִ֣י כָלִֽיתִי׃ (פט) בְּֽתוֹכָ֘ח֤וֹת עַל־ עָוֺ֨ן ׀ יִסַּ֬רְתָּ אִ֗ישׁ וַתֶּ֣מֶס כָּעָ֣שׁ חֲמוּד֑וֹ אַ֤ךְ הֶ֖בֶל כָּל־ אָדָ֣ם סֶֽלָה׃ (צ) שִֽׁמְעָ֥ה־ תְפִלָּתִ֨י ׀ יְהוָ֡ה וְשַׁוְעָתִ֨י ׀ הַאֲזִינָה֮ אֶֽל־ דִּמְעָתִ֗י אַֽל־ תֶּ֫חֱרַ֥שׁ כִּ֤י גֵ֣ר אָנֹכִ֣י עִמָּ֑ךְ תּ֝וֹשָׁ֗ב כְּכָל־ אֲבוֹתָֽי׃ (צא) הָשַׁ֣ע מִמֶּ֣נִּי וְאַבְלִ֑יגָה בְּטֶ֖רֶם אֵלֵ֣ךְ וְאֵינֶֽנִּי׃ (צב) לַ֝מְנַצֵּ֗חַ לְדָוִ֥ד מִזְמֽוֹר׃ (צג) קַוֺּ֣ה קִוִּ֣יתִי יְהוָ֑ה וַיֵּ֥ט אֵ֝לַ֗י וַיִּשְׁמַ֥ע שַׁוְעָתִֽי׃ (צד) וַיַּעֲלֵ֤נִי ׀ מִבּ֥וֹר שָׁאוֹן֮ מִטִּ֪יט הַיָּ֫וֵ֥ן וַיָּ֖קֶם עַל־ סֶ֥לַע רַגְלַ֗י כּוֹנֵ֥ן אֲשֻׁרָֽי׃ (צה) וַיִּתֵּ֬ן בְּפִ֨י ׀ שִׁ֥יר חָדָשׁ֮ תְּהִלָּ֪ה לֵֽאלֹ֫הֵ֥ינוּ יִרְא֣וּ רַבִּ֣ים וְיִירָ֑אוּ וְ֝יִבְטְח֗וּ בַּיהוָֽה׃ (צו) אַ֥שְֽׁרֵי הַגֶּ֗בֶר אֲשֶׁר־ שָׂ֣ם יְ֭הֹוָה מִבְטַח֑וֹ וְֽלֹא־ פָנָ֥ה אֶל־ רְ֝הָבִ֗ים וְשָׂטֵ֥י כָזָֽב׃ (צז) רַבּ֤וֹת עָשִׂ֨יתָ ׀ אַתָּ֤ה ׀ יְהוָ֣ה אֱלֹהַי֮ נִֽפְלְאֹתֶ֥יךָ וּמַחְשְׁבֹתֶ֗יךָ אֵ֫לֵ֥ינוּ אֵ֤ין ׀ עֲרֹ֬ךְ אֵלֶ֗יךָ אַגִּ֥ידָה וַאֲדַבֵּ֑רָה עָ֝צְמ֗וּ מִסַּפֵּֽר׃ (צח) זֶ֤בַח וּמִנְחָ֨ה ׀ לֹֽא־ חָפַ֗צְתָּ אָ֭זְנַיִם כָּרִ֣יתָ לִּ֑י עוֹלָ֥ה וַ֝חֲטָאָ֗ה לֹ֣א שָׁאָֽלְתָּ׃ (צט) אָ֣ז אָ֭מַרְתִּי הִנֵּה־ בָ֑אתִי בִּמְגִלַּת־ סֵ֝֗פֶר כָּת֥וּב עָלָֽי׃ (ק) לַֽעֲשֽׂוֹת־ רְצוֹנְךָ֣ אֱלֹהַ֣י חָפָ֑צְתִּי וְ֝ת֥וֹרָתְךָ֗ בְּת֣וֹךְ מֵעָֽי׃ (קא) בִּשַּׂ֤רְתִּי צֶ֨דֶק ׀ בְּקָ֘הָ֤ל רָ֗ב הִנֵּ֣ה שְׂ֭פָתַי לֹ֣א אֶכְלָ֑א יְ֝הוָ֗ה אַתָּ֥ה יָדָֽעְתָּ׃ (קב) צִדְקָתְךָ֬ לֹא־ כִסִּ֨יתִי ׀ בְּת֬וֹךְ לִבִּ֗י אֱמוּנָתְךָ֣ וּתְשׁוּעָתְךָ֣ אָמָ֑רְתִּי לֹא־ כִחַ֥דְתִּי חַסְדְּךָ֥ וַ֝אֲמִתְּךָ֗ לְקָהָ֥ל רָֽב׃ (קג) אַתָּ֤ה יְהוָ֗ה לֹא־ תִכְלָ֣א רַחֲמֶ֣יךָ מִמֶּ֑נִּי חַסְדְּךָ֥ וַ֝אֲמִתְּךָ֗ תָּמִ֥יד יִצְּרֽוּנִי׃ (קד) כִּ֤י אָפְפ֥וּ־ עָלַ֨י ׀ רָע֡וֹת עַד־ אֵ֬ין מִסְפָּ֗ר הִשִּׂיג֣וּנִי עֲ֭וֺנֹתַי וְלֹא־ יָכֹ֣לְתִּי לִרְא֑וֹת עָצְמ֥וּ מִשַּֽׂעֲר֥וֹת רֹ֝אשִׁ֗י וְלִבִּ֥י עֲזָבָֽנִי׃ (קה) רְצֵ֣ה יְ֭הוָה לְהַצִּילֵ֑נִי יְ֝הוָ֗ה לְעֶזְרָ֥תִי חֽוּשָׁה׃ (קו) יֵ֘בֹ֤שׁוּ וְיַחְפְּר֨וּ ׀ יַחַד֮ מְבַקְשֵׁ֥י נַפְשִׁ֗י לִסְפּ֫וֹתָ֥הּ יִסֹּ֣גוּ אָ֭חוֹר וְיִכָּלְמ֑וּ חֲ֝פֵצֵ֗י רָעָתִֽי׃ (קז) יָ֭שֹׁמּוּ עַל־ עֵ֣קֶב בָּשְׁתָּ֑ם הָאֹמְרִ֥ים לִ֝֗י הֶ֘אָ֥ח ׀ הֶאָֽח׃ (קח) יָ֘שִׂ֤ישׂוּ וְיִשְׂמְח֨וּ ׀ בְּךָ֗ כָּֽל־ מְבַ֫קְשֶׁ֥יךָ יֹאמְר֣וּ תָ֭מִיד יִגְדַּ֣ל יְהוָ֑ה אֹֽ֝הֲבֵ֗י תְּשׁוּעָתֶֽךָ׃ (קט) וַאֲנִ֤י ׀ עָנִ֣י וְאֶבְיוֹן֮ אֲדֹנָ֪י יַחֲשָׁ֫ב לִ֥י עֶזְרָתִ֣י וּמְפַלְטִ֣י אַ֑תָּה אֱ֝לֹהַ֗י אַל־ תְּאַחַֽר׃ (קי) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (קיא) אַ֭שְׁרֵי מַשְׂכִּ֣יל אֶל־ דָּ֑ל בְּי֥וֹם רָ֝עָ֗ה יְֽמַלְּטֵ֥הוּ יְהוָֽה׃ (קיב) יְהוָ֤ה ׀ יִשְׁמְרֵ֣הוּ וִֽ֭יחַיֵּהוּ (יאשר) וְאֻשַּׁ֣ר בָּאָ֑רֶץ וְאַֽל־ תִּ֝תְּנֵ֗הוּ בְּנֶ֣פֶשׁ אֹיְבָֽיו׃ (קיג) יְֽהוָ֗ה יִ֭סְעָדֶנּוּ עַל־ עֶ֣רֶשׂ דְּוָ֑י כָּל־ מִ֝שְׁכָּב֗וֹ הָפַ֥כְתָּ בְחָלְיֽוֹ׃ (קיד) אֲ‍ֽנִי־ אָ֭מַרְתִּי יְהוָ֣ה חָנֵּ֑נִי רְפָאָ֥ה נַ֝פְשִׁ֗י כִּי־ חָטָ֥אתִי לָֽךְ׃ (קטו) אוֹיְבַ֗י יֹאמְר֣וּ רַ֣ע לִ֑י מָתַ֥י יָ֝מ֗וּת וְאָבַ֥ד שְׁמֽוֹ׃ (קטז) וְאִם־ בָּ֤א לִרְא֨וֹת ׀ שָׁ֤וְא יְדַבֵּ֗ר לִבּ֗וֹ יִקְבָּץ־ אָ֥וֶן ל֑וֹ יֵצֵ֖א לַח֣וּץ יְדַבֵּֽר׃ (קיז) יַ֗חַד עָלַ֣י יִ֭תְלַחֲשׁוּ כָּל־ שֹׂנְאָ֑י עָלַ֓י ׀ יַחְשְׁב֖וּ רָעָ֣ה לִֽי׃ (קיח) דְּֽבַר־ בְּ֭לִיַּעַל יָצ֣וּק בּ֑וֹ וַאֲשֶׁ֥ר שָׁ֝כַ֗ב לֹא־ יוֹסִ֥יף לָקֽוּם׃ (קיט) גַּם־ אִ֤ישׁ שְׁלוֹמִ֨י ׀ אֲשֶׁר־ בָּטַ֣חְתִּי ב֭וֹ אוֹכֵ֣ל לַחְמִ֑י הִגְדִּ֖יל עָלַ֣י עָקֵֽב׃ (קכ) וְאַתָּ֤ה יְהוָ֗ה חָנֵּ֥נִי וַהֲקִימֵ֑נִי וַֽאֲשַׁלְּמָ֥ה לָהֶֽם׃ (קכא) בְּזֹ֣את יָ֭דַעְתִּי כִּֽי־ חָפַ֣צְתָּ בִּ֑י כִּ֤י לֹֽא־ יָרִ֖יעַ אֹיְבִ֣י עָלָֽי׃ (קכב) וַאֲנִ֗י בְּ֭תֻמִּי תָּמַ֣כְתָּ בִּ֑י וַתַּצִּיבֵ֖נִי לְפָנֶ֣יךָ לְעוֹלָֽם׃
