תהילים פרק ג לפי החלוקה לסדרים

פרק ג בספר תהילים (9 פסוקים) נמצא בתהילים סדר א (פסוקים יט–כז).

תהילים סדר א

(א) אַ֥שְֽׁרֵי־ הָאִ֗ישׁ אֲשֶׁ֤ר ׀ לֹ֥א הָלַךְ֮ בַּעֲצַ֪ת רְשָׁ֫עִ֥ים וּבְדֶ֣רֶךְ חַ֭טָּאִים לֹ֥א עָמָ֑ד וּבְמוֹשַׁ֥ב לֵ֝צִ֗ים לֹ֣א יָשָֽׁב׃ (ב) כִּ֤י אִ֥ם בְּתוֹרַ֥ת יְהוָ֗ה חֶ֫פְצ֥וֹ וּֽבְתוֹרָת֥וֹ יֶהְגֶּ֗ה יוֹמָ֥ם וָלָֽיְלָה׃ (ג) וְֽהָיָ֗ה כְּעֵץ֮ שָׁת֪וּל עַֽל־ פַּלְגֵ֫י מָ֥יִם אֲשֶׁ֤ר פִּרְי֨וֹ ׀ יִתֵּ֬ן בְּעִתּ֗וֹ וְעָלֵ֥הוּ לֹֽא־ יִבּ֑וֹל וְכֹ֖ל אֲשֶׁר־ יַעֲשֶׂ֣ה יַצְלִֽיחַ׃ (ד) לֹא־ כֵ֥ן הָרְשָׁעִ֑ים כִּ֥י אִם־ כַּ֝מֹּ֗ץ אֲ‍ֽשֶׁר־ תִּדְּפֶ֥נּוּ רֽוּחַ׃ (ה) עַל־ כֵּ֤ן ׀ לֹא־ יָקֻ֣מוּ רְ֭שָׁעִים בַּמִּשְׁפָּ֑ט וְ֝חַטָּאִ֗ים בַּעֲדַ֥ת צַדִּיקִֽים׃ (ו) כִּֽי־ יוֹדֵ֣עַ יְ֭הוָה דֶּ֣רֶךְ צַדִּיקִ֑ים וְדֶ֖רֶךְ רְשָׁעִ֣ים תֹּאבֵֽד׃ (ז) לָ֭מָּה רָגְשׁ֣וּ גוֹיִ֑ם וּ֝לְאֻמִּ֗ים יֶהְגּוּ־ רִֽיק׃ (ח) יִ֥תְיַצְּב֨וּ ׀ מַלְכֵי־ אֶ֗רֶץ וְרוֹזְנִ֥ים נֽוֹסְדוּ־ יָ֑חַד עַל־ יְ֝הוָה וְעַל־ מְשִׁיחֽוֹ׃ (ט) נְֽ֭נַתְּקָה אֶת־ מֽוֹסְרוֹתֵ֑ימוֹ וְנַשְׁלִ֖יכָה מִמֶּ֣נּוּ עֲבֹתֵֽימוֹ׃ (י) יוֹשֵׁ֣ב בַּשָּׁמַ֣יִם יִשְׂחָ֑ק אֲ֝דֹנָ֗י יִלְעַג־ לָֽמוֹ׃ (יא) אָ֤ז יְדַבֵּ֣ר אֵלֵ֣ימוֹ בְאַפּ֑וֹ וּֽבַחֲרוֹנ֥וֹ יְבַהֲלֵֽמוֹ׃ (יב) וַ֭אֲנִי נָסַ֣כְתִּי מַלְכִּ֑י עַל־ צִ֝יּ֗וֹן הַר־ קָדְשִֽׁי׃ (יג) אֲסַפְּרָ֗ה אֶֽ֫ל חֹ֥ק יְֽהוָ֗ה אָמַ֘ר אֵלַ֥י בְּנִ֥י אַ֑תָּה אֲ֝נִ֗י הַיּ֥וֹם יְלִדְתִּֽיךָ׃ (יד) שְׁאַ֤ל מִמֶּ֗נִּי וְאֶתְּנָ֣ה ג֭וֹיִם נַחֲלָתֶ֑ךָ וַ֝אֲחֻזָּתְךָ֗ אַפְסֵי־ אָֽרֶץ׃ (טו) תְּ֭רֹעֵם בְּשֵׁ֣בֶט בַּרְזֶ֑ל כִּכְלִ֖י יוֹצֵ֣ר תְּנַפְּצֵֽם׃ (טז) וְ֭עַתָּה מְלָכִ֣ים הַשְׂכִּ֑ילוּ הִ֝וָּסְר֗וּ שֹׁ֣פְטֵי אָֽרֶץ׃ (יז) עִבְד֣וּ אֶת־ יְהוָ֣ה בְּיִרְאָ֑ה וְ֝גִ֗ילוּ בִּרְעָדָֽה׃ (יח) נַשְּׁקוּ־ בַ֡ר פֶּן־ יֶאֱנַ֤ף ׀ וְתֹ֬אבְדוּ דֶ֗רֶךְ כִּֽי־ יִבְעַ֣ר כִּמְעַ֣ט אַפּ֑וֹ אַ֝שְׁרֵ֗י כָּל־ ח֥וֹסֵי בֽוֹ׃ תחילת פרק ג (יט) מִזְמ֥וֹר לְדָוִ֑ד בְּ֝בָרְח֗וֹ מִפְּנֵ֤י ׀ אַבְשָׁל֬וֹם בְּנֽוֹ׃ (כ) יְ֭הוָה מָֽה־ רַבּ֣וּ צָרָ֑י רַ֝בִּ֗ים קָמִ֥ים עָלָֽי׃ (כא) רַבִּים֮ אֹמְרִ֪ים לְנַ֫פְשִׁ֥י אֵ֤ין יְֽשׁוּעָ֓תָה לּ֬וֹ בֵֽאלֹהִ֬ים סֶֽלָה׃ (כב) וְאַתָּ֣ה יְ֭הוָה מָגֵ֣ן בַּעֲדִ֑י כְּ֝בוֹדִ֗י וּמֵרִ֥ים רֹאשִֽׁי׃ (כג) ק֭וֹלִי אֶל־ יְהוָ֣ה אֶקְרָ֑א וַיַּֽעֲנֵ֨נִי מֵהַ֖ר קָדְשׁ֣וֹ סֶֽלָה׃ (כד) אֲנִ֥י שָׁכַ֗בְתִּי וָֽאִ֫ישָׁ֥נָה הֱקִיצ֑וֹתִי כִּ֖י יְהוָ֣ה יִסְמְכֵֽנִי׃ (כה) לֹֽא־ אִ֭ירָא מֵרִבְב֥וֹת עָ֑ם אֲשֶׁ֥ר סָ֝בִ֗יב שָׁ֣תוּ עָלָֽי׃ (כו) ק֘וּמָ֤ה יְהוָ֨ה ׀ הוֹשִׁ֘יעֵ֤נִי אֱלֹהַ֗י כִּֽי־ הִכִּ֣יתָ אֶת־ כָּל־ אֹיְבַ֣י לֶ֑חִי שִׁנֵּ֖י רְשָׁעִ֣ים שִׁבַּֽרְתָּ׃ (כז) לַיהוָ֥ה הַיְשׁוּעָ֑ה עַֽל־ עַמְּךָ֖ בִרְכָתֶ֣ךָ סֶּֽלָה׃ סוף פרק ג (כח) לַמְנַצֵּ֥חַ בִּנְגִינ֗וֹת מִזְמ֥וֹר לְדָוִֽד׃ (כט) בְּקָרְאִ֡י עֲנֵ֤נִי ׀ אֱלֹ֘הֵ֤י צִדְקִ֗י בַּ֭צָּר הִרְחַ֣בְתָּ לִּ֑י חָ֝נֵּ֗נִי וּשְׁמַ֥ע תְּפִלָּתִֽי׃ (ל) בְּנֵ֥י אִ֡ישׁ עַד־ מֶ֬ה כְבוֹדִ֣י לִ֭כְלִמָּה תֶּאֱהָב֣וּן רִ֑יק תְּבַקְשׁ֖וּ כָזָ֣ב סֶֽלָה׃ (לא) וּדְע֗וּ כִּֽי־ הִפְלָ֣ה יְ֭הוָה חָסִ֣יד ל֑וֹ יְהוָ֥ה יִ֝שְׁמַ֗ע בְּקָרְאִ֥י אֵלָֽיו׃ (לב) רִגְז֗וּ וְֽאַל־ תֶּ֫חֱטָ֥אוּ אִמְר֣וּ בִ֭לְבַבְכֶם עַֽל־ מִשְׁכַּבְכֶ֗ם וְדֹ֣מּוּ סֶֽלָה׃ (לג) זִבְח֥וּ זִבְחֵי־ צֶ֑דֶק וּ֝בִטְח֗וּ אֶל־ יְהוָֽה׃ (לד) רַבִּ֥ים אֹמְרִים֮ מִֽי־ יַרְאֵ֪נ֫וּ ט֥וֹב נְֽסָה־ עָ֭לֵינוּ א֨וֹר פָּנֶ֬יךָ יְהוָֽה׃ (לה) נָתַ֣תָּה שִׂמְחָ֣ה בְלִבִּ֑י מֵעֵ֬ת דְּגָנָ֖ם וְתִֽירוֹשָׁ֣ם רָֽבּוּ׃ (לו) בְּשָׁל֣וֹם יַחְדָּו֮ אֶשְׁכְּבָ֪ה וְאִ֫ישָׁ֥ן כִּֽי־ אַתָּ֣ה יְהוָ֣ה לְבָדָ֑ד לָ֝בֶ֗טַח תּוֹשִׁיבֵֽנִי׃ (לז) לַמְנַצֵּ֥חַ אֶֽל־ הַנְּחִיל֗וֹת מִזְמ֥וֹר לְדָוִֽד׃ (לח) אֲמָרַ֖י הַאֲזִ֥ינָה ׀ יְהוָ֗ה בִּ֣ינָה הֲגִֽיגִי׃ (לט) הַקְשִׁ֤יבָה ׀ לְק֬וֹל שַׁוְעִ֗י מַלְכִּ֥י וֵאלֹהָ֑י כִּֽי־ אֵ֝לֶ֗יךָ אֶתְפַּלָּֽל׃ (מ) יְֽהוָ֗ה בֹּ֭קֶר תִּשְׁמַ֣ע קוֹלִ֑י בֹּ֥קֶר אֶֽעֱרָךְ־ לְ֝ךָ֗ וַאֲצַפֶּֽה׃ (מא) כִּ֤י ׀ לֹ֤א אֵֽל־ חָפֵ֘ץ רֶ֥שַׁע ׀ אָ֑תָּה לֹ֖א יְגֻרְךָ֣ רָֽע׃ (מב) לֹֽא־ יִתְיַצְּב֣וּ הֽ֭וֹלְלִים לְנֶ֣גֶד עֵינֶ֑יךָ שָׂ֝נֵ֗אתָ כָּל־ פֹּ֥עֲלֵי אָֽוֶן׃ (מג) תְּאַבֵּד֮ דֹּבְרֵ֪י כָ֫זָ֥ב אִישׁ־ דָּמִ֥ים וּמִרְמָ֗ה יְתָ֘עֵ֥ב ׀ יְהוָֽה׃ (מד) וַאֲנִ֗י בְּרֹ֣ב חַ֭סְדְּךָ אָב֣וֹא בֵיתֶ֑ךָ אֶשְׁתַּחֲוֶ֥ה אֶל־ הֵֽיכַל־ קָ֝דְשְׁךָ֗ בְּיִרְאָתֶֽךָ׃ (מה) יְהוָ֤ה ׀ נְחֵ֬נִי בְצִדְקָתֶ֗ךָ לְמַ֥עַן שׁוֹרְרָ֑י (הושר) הַיְשַׁ֖ר לְפָנַ֣י דַּרְכֶּֽךָ׃ (מו) כִּ֤י אֵ֪ין בְּפִ֡יהוּ נְכוֹנָה֮ קִרְבָּ֪ם הַ֫וּ֥וֹת קֶֽבֶר־ פָּת֥וּחַ גְּרוֹנָ֑ם לְ֝שׁוֹנָ֗ם יַחֲלִֽיקוּן׃ (מז) הַֽאֲשִׁימֵ֨ם ׀ אֱ‍ֽלֹהִ֗ים יִפְּלוּ֮ מִֽמֹּעֲצ֪וֹתֵ֫יהֶ֥ם בְּרֹ֣ב פִּ֭שְׁעֵיהֶם הַדִּיחֵ֑מוֹ כִּי־ מָ֥רוּ בָֽךְ׃ (מח) וְיִשְׂמְח֨וּ כָל־ ח֪וֹסֵי בָ֡ךְ לְעוֹלָ֣ם יְ֭רַנֵּנוּ וְתָסֵ֣ךְ עָלֵ֑ימוֹ וְֽיַעְלְצ֥וּ בְ֝ךָ֗ אֹהֲבֵ֥י שְׁמֶֽךָ׃ (מט) כִּֽי־ אַתָּה֮ תְּבָרֵ֪ךְ צַ֫דִּ֥יק יְהוָ֑ה כַּ֝צִּנָּ֗ה רָצ֥וֹן תַּעְטְרֶֽנּוּ׃ (נ) לַמְנַצֵּ֣חַ בִּ֭נְגִינוֹת עַֽל־ הַשְּׁמִינִ֗ית מִזְמ֥וֹר לְדָוִֽד׃ (נא) יְֽהוָ֗ה אַל־ בְּאַפְּךָ֥ תוֹכִיחֵ֑נִי וְֽאַל־ בַּחֲמָתְךָ֥ תְיַסְּרֵֽנִי׃ (נב) חָנֵּ֥נִי יְהוָה֮ כִּ֤י אֻמְלַ֫ל אָ֥נִי רְפָאֵ֥נִי יְהוָ֑ה כִּ֖י נִבְהֲל֣וּ עֲצָמָֽי׃ (נג) וְ֭נַפְשִׁי נִבְהֲלָ֣ה מְאֹ֑ד (ואת) וְאַתָּ֥ה יְ֝הוָ֗ה עַד־ מָתָֽי׃ (נד) שׁוּבָ֣ה יְ֭הוָה חַלְּצָ֣ה נַפְשִׁ֑י ה֝וֹשִׁיעֵ֗נִי לְמַ֣עַן חַסְדֶּֽךָ׃ (נה) כִּ֤י אֵ֣ין בַּמָּ֣וֶת זִכְרֶ֑ךָ בִּ֝שְׁא֗וֹל מִ֣י יֽוֹדֶה־ לָּֽךְ׃ (נו) יָגַ֤עְתִּי ׀ בְּֽאַנְחָתִ֗י אַשְׂחֶ֣ה בְכָל־ לַ֭יְלָה מִטָּתִ֑י בְּ֝דִמְעָתִ֗י עַרְשִׂ֥י אַמְסֶֽה׃ (נז) עָֽשְׁשָׁ֣ה מִכַּ֣עַס עֵינִ֑י עָֽ֝תְקָ֗ה בְּכָל־ צוֹרְרָֽי׃ (נח) ס֣וּרוּ מִ֭מֶּנִּי כָּל־ פֹּ֣עֲלֵי אָ֑וֶן כִּֽי־ שָׁמַ֥ע יְ֝הוָ֗ה ק֣וֹל בִּכְיִֽי׃ (נט) שָׁמַ֣ע יְ֭הוָה תְּחִנָּתִ֑י יְ֝הוָ֗ה תְּֽפִלָּתִ֥י יִקָּֽח׃ (ס) יֵבֹ֤שׁוּ ׀ וְיִבָּהֲל֣וּ מְ֭אֹד כָּל־ אֹיְבָ֑י יָ֝שֻׁ֗בוּ יֵבֹ֥שׁוּ רָֽגַע׃ (סא) שִׁגָּי֗וֹן לְדָ֫וִ֥ד אֲשֶׁר־ שָׁ֥ר לַיהוָ֑ה עַל־ דִּבְרֵי־ כ֝֗וּשׁ בֶּן־ יְמִינִֽי׃ (סב) יְהוָ֣ה אֱ֭לֹהַי בְּךָ֣ חָסִ֑יתִי הוֹשִׁיעֵ֥נִי מִכָּל־ רֹ֝דְפַ֗י וְהַצִּילֵֽנִי׃ (סג) פֶּן־ יִטְרֹ֣ף כְּאַרְיֵ֣ה נַפְשִׁ֑י פֹּ֝רֵ֗ק וְאֵ֣ין מַצִּֽיל׃ (סד) יְהוָ֣ה אֱ֭לֹהַי אִם־ עָשִׂ֣יתִי זֹ֑את אִֽם־ יֶשׁ־ עָ֥וֶל בְּכַפָּֽי׃ (סה) אִם־ גָּ֭מַלְתִּי שֽׁוֹלְמִ֥י רָ֑ע וָאֲחַלְּצָ֖ה צוֹרְרִ֣י לִ֝בָּ֗ם תַּקְשִׁ֥יב אָזְנֶֽךָ׃ (קכז) לִשְׁפֹּ֥ט יָת֗וֹם וָ֫דָ֥ךְ בַּל־ יוֹסִ֥יף ע֑וֹד לַעֲרֹ֥ץ אֱ֝נ֗וֹשׁ מִן־ הָאָֽרֶץ׃ (קכח) לַמְנַצֵּ֗חַ לְדָ֫וִ֥ד בַּֽיהוָ֨ה ׀ חָסִ֗יתִי אֵ֭יךְ תֹּאמְר֣וּ לְנַפְשִׁ֑י (נודו) נ֝֗וּדִי הַרְכֶ֥ם צִפּֽוֹר׃ (קכט) כִּ֤י הִנֵּ֪ה הָרְשָׁעִ֡ים יִדְרְכ֬וּן קֶ֗שֶׁת כּוֹנְנ֣וּ חִצָּ֣ם עַל־ יֶ֑תֶר לִיר֥וֹת בְּמוֹ־ אֹ֝֗פֶל לְיִשְׁרֵי־ לֵֽב׃ (קל) כִּ֣י הַ֭שָּׁתוֹת יֵֽהָרֵס֑וּן צַ֝דִּ֗יק מַה־ פָּעָֽל׃ (קלא) יְהוָ֤ה ׀ בְּֽהֵ֘יכַ֤ל קָדְשׁ֗וֹ יְהוָה֮ בַּשָּׁמַ֪יִם כִּ֫סְא֥וֹ עֵינָ֥יו יֶחֱז֑וּ עַפְעַפָּ֥יו יִ֝בְחֲנ֗וּ בְּנֵ֣י אָדָֽם׃ (קלב) יְהוָה֮ צַדִּ֪יק יִ֫בְחָ֥ן וְ֭רָשָׁע וְאֹהֵ֣ב חָמָ֑ס שָֽׂנְאָ֥ה נַפְשֽׁוֹ׃ (קלג) יַמְטֵ֥ר עַל־ רְשָׁעִ֗ים פַּ֫חִ֥ים אֵ֣שׁ וְ֭גָפְרִית וְר֥וּחַ זִלְעָפ֗וֹת מְנָ֣ת כּוֹסָֽם׃
