תהילים סדר י

(א) כָּלּ֥וּ תְפִלּ֑וֹת דָּ֝וִ֗ד בֶּן־ יִשָֽׁי׃ (ב) מִזְמ֗וֹר לְאָ֫סָ֥ף אַ֤ךְ ט֭וֹב לְיִשְׂרָאֵ֥ל אֱלֹהִ֗ים לְבָרֵ֥י לֵבָֽב׃ (ג) וַאֲנִ֗י כִּ֭מְעַט (נטוי) נָטָ֣יוּ רַגְלָ֑י כְּ֝אַ֗יִן (שפכה) שֻׁפְּכ֥וּ אֲשֻׁרָֽי׃ (ד) כִּֽי־ קִ֭נֵּאתִי בַּֽהוֹלְלִ֑ים שְׁל֖וֹם רְשָׁעִ֣ים אֶרְאֶֽה׃ (ה) כִּ֤י אֵ֖ין חַרְצֻבּ֥וֹת לְמוֹתָ֗ם וּבָרִ֥יא אוּלָֽם׃ (ו) בַּעֲמַ֣ל אֱנ֣וֹשׁ אֵינֵ֑מוֹ וְעִם־ אָ֝דָ֗ם לֹ֣א יְנֻגָּֽעוּ׃ (ז) לָ֭כֵן עֲנָקַ֣תְמוֹ גַאֲוָ֑ה יַעֲטָף־ שִׁ֝֗ית חָמָ֥ס לָֽמוֹ׃ (ח) יָ֭צָא מֵחֵ֣לֶב עֵינֵ֑מוֹ עָ֝בְר֗וּ מַשְׂכִּיּ֥וֹת לֵבָֽב׃ (ט) יָמִ֤יקוּ ׀ וִידַבְּר֣וּ בְרָ֣ע עֹ֑שֶׁק מִמָּר֥וֹם יְדַבֵּֽרוּ׃ (י) שַׁתּ֣וּ בַשָּׁמַ֣יִם פִּיהֶ֑ם וּ֝לְשׁוֹנָ֗ם תִּֽהֲלַ֥ךְ בָּאָֽרֶץ׃ (יא) לָכֵ֤ן ׀ (ישיב) יָשׁ֣וּב עַמּ֣וֹ הֲלֹ֑ם וּמֵ֥י מָ֝לֵ֗א יִמָּ֥צוּ לָֽמוֹ׃ (יב) וְֽאָמְר֗וּ אֵיכָ֥ה יָדַֽע־ אֵ֑ל וְיֵ֖שׁ דֵּעָ֣ה בְעֶלְיֽוֹן׃ (יג) הִנֵּה־ אֵ֥לֶּה רְשָׁעִ֑ים וְשַׁלְוֵ֥י ע֝וֹלָ֗ם הִשְׂגּוּ־ חָֽיִל׃ (יד) אַךְ־ רִ֭יק זִכִּ֣יתִי לְבָבִ֑י וָאֶרְחַ֖ץ בְּנִקָּי֣וֹן כַּפָּֽי׃ (טו) וָאֱהִ֣י נָ֭גוּעַ כָּל־ הַיּ֑וֹם וְ֝תוֹכַחְתִּ֗י לַבְּקָרִֽים׃ (טז) אִם־ אָ֭מַרְתִּי אֲסַפְּרָ֥ה כְמ֑וֹ הִנֵּ֤ה ד֭וֹר בָּנֶ֣יךָ בָגָֽדְתִּי׃ (יז) וָֽ֭אֲחַשְּׁבָה לָדַ֣עַת זֹ֑את עָמָ֖ל (היא) ה֣וּא בְעֵינָֽי׃ (יח) עַד־ אָ֭בוֹא אֶל־ מִקְדְּשֵׁי־ אֵ֑ל אָ֝בִ֗ינָה לְאַחֲרִיתָֽם׃ (יט) אַ֣ךְ בַּ֭חֲלָקוֹת תָּשִׁ֣ית לָ֑מוֹ הִ֝פַּלְתָּ֗ם לְמַשּׁוּאֽוֹת׃ (כ) אֵ֤יךְ הָי֣וּ לְשַׁמָּ֣ה כְרָ֑גַע סָ֥פוּ תַ֝֗מּוּ מִן־ בַּלָּהֽוֹת׃ (כא) כַּחֲל֥וֹם מֵהָקִ֑יץ אֲ֝דֹנָי בָּעִ֤יר ׀ צַלְמָ֬ם תִּבְזֶֽה׃ (כב) כִּ֭י יִתְחַמֵּ֣ץ לְבָבִ֑י וְ֝כִלְיוֹתַ֗י אֶשְׁתּוֹנָֽן׃ (כג) וַאֲנִי־ בַ֭עַר וְלֹ֣א אֵדָ֑ע בְּ֝הֵמ֗וֹת הָיִ֥יתִי עִמָּֽךְ׃ (כד) וַאֲנִ֣י תָמִ֣יד עִמָּ֑ךְ אָ֝חַ֗זְתָּ בְּיַד־ יְמִינִֽי׃ (כה) בַּעֲצָתְךָ֥ תַנְחֵ֑נִי וְ֝אַחַ֗ר כָּב֥וֹד תִּקָּחֵֽנִי׃ (כו) מִי־ לִ֥י בַשָּׁמָ֑יִם וְ֝עִמְּךָ֗ לֹא־ חָפַ֥צְתִּי בָאָֽרֶץ׃ (כז) כָּלָ֥ה שְׁאֵרִ֗י וּלְבָ֫בִ֥י צוּר־ לְבָבִ֥י וְחֶלְקִ֗י אֱלֹהִ֥ים לְעוֹלָֽם׃ (כח) כִּֽי־ הִנֵּ֣ה רְחֵקֶ֣יךָ יֹאבֵ֑דוּ הִ֝צְמַ֗תָּה כָּל־ זוֹנֶ֥ה מִמֶּֽךָּ׃ (כט) וַאֲנִ֤י ׀ קִֽרֲבַ֥ת אֱלֹהִ֗ים לִ֫י־ ט֥וֹב שַׁתִּ֤י ׀ בַּאדֹנָ֣י יְהֹוִ֣ה מַחְסִ֑י לְ֝סַפֵּ֗ר כָּל־ מַלְאֲכוֹתֶֽיךָ׃ (ל) מַשְׂכִּ֗יל לְאָ֫סָ֥ף לָמָ֣ה אֱ֭לֹהִים זָנַ֣חְתָּ לָנֶ֑צַח יֶעְשַׁ֥ן אַ֝פְּךָ֗ בְּצֹ֣אן מַרְעִיתֶֽךָ׃ (לא) זְכֹ֤ר עֲדָתְךָ֨ ׀ קָ֘נִ֤יתָ קֶּ֗דֶם גָּ֭אַלְתָּ שֵׁ֣בֶט נַחֲלָתֶ֑ךָ הַר־ צִ֝יּ֗וֹן זֶ֤ה ׀ שָׁכַ֬נְתָּ בּֽוֹ׃ (לב) הָרִ֣ימָה פְ֭עָמֶיךָ לְמַשֻּׁא֣וֹת נֶ֑צַח כָּל־ הֵרַ֖ע אוֹיֵ֣ב בַּקֹּֽדֶשׁ׃ (לג) שָׁאֲג֣וּ צֹ֭רְרֶיךָ בְּקֶ֣רֶב מוֹעֲדֶ֑ךָ שָׂ֖מוּ אוֹתֹתָ֣ם אֹתֽוֹת׃ (לד) יִ֭וָּדַע כְּמֵבִ֣יא לְמָ֑עְלָה בִּֽסֲבָךְ־ עֵ֝֗ץ קַרְדֻּמּֽוֹת׃ (לה) (ועת) וְ֭עַתָּה פִּתּוּחֶ֣יהָ יָּ֑חַד בְּכַשִּׁ֥יל וְ֝כֵֽילַפֹּ֗ת יַהֲלֹמֽוּן׃ (לו) שִׁלְח֣וּ בָ֭אֵשׁ מִקְדָּשֶׁ֑ךָ לָ֝אָ֗רֶץ חִלְּל֥וּ מִֽשְׁכַּן־ שְׁמֶֽךָ׃ (לז) אָמְר֣וּ בְ֭לִבָּם נִינָ֣ם יָ֑חַד שָׂרְפ֖וּ כָל־ מוֹעֲדֵי־ אֵ֣ל בָּאָֽרֶץ׃ (לח) אֽוֹתֹתֵ֗ינוּ לֹ֥א רָ֫אִ֥ינוּ אֵֽין־ ע֥וֹד נָבִ֑יא וְלֹֽא־ אִ֝תָּ֗נוּ יֹדֵ֥עַ עַד־ מָֽה׃ (לט) עַד־ מָתַ֣י אֱ֭לֹהִים יְחָ֣רֶף צָ֑ר יְנָ֘אֵ֤ץ אוֹיֵ֖ב שִׁמְךָ֣ לָנֶֽצַח׃ (מ) לָ֤מָּה תָשִׁ֣יב יָ֭דְךָ וִֽימִינֶ֑ךָ מִקֶּ֖רֶב (חוקך) חֵֽיקְךָ֣ כַלֵּֽה׃ (מא) וֵ֭אלֹהִים מַלְכִּ֣י מִקֶּ֑דֶם פֹּעֵ֥ל יְ֝שׁוּע֗וֹת בְּקֶ֣רֶב הָאָֽרֶץ׃ (מב) אַתָּ֤ה פוֹרַ֣רְתָּ בְעָזְּךָ֣ יָ֑ם שִׁבַּ֖רְתָּ רָאשֵׁ֥י תַ֝נִּינִ֗ים עַל־ הַמָּֽיִם׃ (מג) אַתָּ֣ה רִ֭צַּצְתָּ רָאשֵׁ֣י לִוְיָתָ֑ן תִּתְּנֶ֥נּוּ מַ֝אֲכָ֗ל לְעָ֣ם לְצִיִּֽים׃ (מד) אַתָּ֣ה בָ֭קַעְתָּ מַעְיָ֣ן וָנָ֑חַל אַתָּ֥ה ה֝וֹבַ֗שְׁתָּ נַהֲר֥וֹת אֵיתָֽן׃ (מה) לְךָ֣ י֭וֹם אַף־ לְךָ֥ לָ֑יְלָה אַתָּ֥ה הֲ֝כִינ֗וֹתָ מָא֥וֹר וָשָֽׁמֶשׁ׃ (מו) אַתָּ֣ה הִ֭צַּבְתָּ כָּל־ גְּבוּל֣וֹת אָ֑רֶץ קַ֥יִץ וָ֝חֹ֗רֶף אַתָּ֥ה יְצַרְתָּם׃ (מז) זְכָר־ זֹ֗את א֭וֹיֵב חֵרֵ֣ף ׀ יְהוָ֑ה וְעַ֥ם נָ֝בָ֗ל נִֽאֲצ֥וּ שְׁמֶֽךָ׃ (מח) אַל־ תִּתֵּ֣ן לְ֭חַיַּת נֶ֣פֶשׁ תּוֹרֶ֑ךָ חַיַּ֥ת עֲ֝נִיֶּ֗יךָ אַל־ תִּשְׁכַּ֥ח לָנֶֽצַח׃ (מט) הַבֵּ֥ט לַבְּרִ֑ית כִּ֥י מָלְא֥וּ מַחֲשַׁכֵּי־ אֶ֝֗רֶץ נְא֣וֹת חָמָֽס׃ (נ) אַל־ יָשֹׁ֣ב דַּ֣ךְ נִכְלָ֑ם עָנִ֥י וְ֝אֶבְי֗וֹן יְֽהַלְל֥וּ שְׁמֶֽךָ׃ (נא) קוּמָ֣ה אֱ֭לֹהִים רִיבָ֣ה רִיבֶ֑ךָ זְכֹ֥ר חֶרְפָּתְךָ֥ מִנִּי־ נָ֝בָ֗ל כָּל־ הַיּֽוֹם׃ (נב) אַל־ תִּ֭שְׁכַּח ק֣וֹל צֹרְרֶ֑יךָ שְׁא֥וֹן קָ֝מֶ֗יךָ עֹלֶ֥ה תָמִֽיד׃ (נג) לַמְנַצֵּ֥חַ אַל־ תַּשְׁחֵ֑ת מִזְמ֖וֹר לְאָסָ֣ף שִֽׁיר׃ (נד) ה֘וֹדִ֤ינוּ לְּךָ֨ ׀ אֱ‍ֽלֹהִ֗ים ה֭וֹדִינוּ וְקָר֣וֹב שְׁמֶ֑ךָ סִ֝פְּר֗וּ נִפְלְאוֹתֶֽיךָ׃ (נה) כִּ֭י אֶקַּ֣ח מוֹעֵ֑ד אֲ֝נִ֗י מֵישָׁרִ֥ים אֶשְׁפֹּֽט׃ (נו) נְֽמֹגִ֗ים אֶ֥רֶץ וְכָל־ יֹשְׁבֶ֑יהָ אָנֹכִ֨י תִכַּ֖נְתִּי עַמּוּדֶ֣יהָ סֶּֽלָה׃ (נז) אָמַ֣רְתִּי לַֽ֭הוֹלְלִים אַל־ תָּהֹ֑לּוּ וְ֝לָרְשָׁעִ֗ים אַל־ תָּרִ֥ימוּ קָֽרֶן׃ (נח) אַל־ תָּרִ֣ימוּ לַמָּר֣וֹם קַרְנְכֶ֑ם תְּדַבְּר֖וּ בְצַוָּ֣אר עָתָֽק׃ (נט) כִּ֤י לֹ֣א מִ֭מּוֹצָא וּמִֽמַּעֲרָ֑ב וְ֝לֹ֗א מִמִּדְבַּ֥ר הָרִֽים׃ (ס) כִּֽי־ אֱלֹהִ֥ים שֹׁפֵ֑ט זֶ֥ה יַ֝שְׁפִּ֗יל וְזֶ֣ה יָרִֽים׃ (סא) כִּ֤י כ֪וֹס בְּֽיַד־ יְהוָ֡ה וְיַ֤יִן חָמַ֨ר ׀ מָ֥לֵא מֶסֶךְ֮ וַיַּגֵּ֪ר מִ֫זֶּ֥ה אַךְ־ שְׁ֭מָרֶיהָ יִמְצ֣וּ יִשְׁתּ֑וּ כֹּ֝֗ל רִשְׁעֵי־ אָֽרֶץ׃ (סב) וַ֭אֲנִי אַגִּ֣יד לְעֹלָ֑ם אֲ֝זַמְּרָ֗ה לֵאלֹהֵ֥י יַעֲקֹֽב׃ (סג) וְכָל־ קַרְנֵ֣י רְשָׁעִ֣ים אֲגַדֵּ֑עַ תְּ֝רוֹמַ֗מְנָה קַֽרְנ֥וֹת צַדִּֽיק׃ תחילת פרק עו (סד) לַמְנַצֵּ֥חַ בִּנְגִינֹ֑ת מִזְמ֖וֹר לְאָסָ֣ף שִֽׁיר׃ (סה) נוֹדָ֣ע בִּֽיהוּדָ֣ה אֱלֹהִ֑ים בְּ֝יִשְׂרָאֵ֗ל גָּד֥וֹל שְׁמֽוֹ׃ (סו) וַיְהִ֣י בְשָׁלֵ֣ם סֻכּ֑וֹ וּמְע֖וֹנָת֣וֹ בְצִיּֽוֹן׃ (סז) שָׁ֭מָּה שִׁבַּ֣ר רִשְׁפֵי־ קָ֑שֶׁת מָגֵ֬ן וְחֶ֖רֶב וּמִלְחָמָ֣ה סֶֽלָה׃ (סח) נָ֭אוֹר אַתָּ֥ה אַדִּ֗יר מֵֽהַרְרֵי־ טָֽרֶף׃ (סט) אֶשְׁתּוֹלְל֨וּ ׀ אַבִּ֣ירֵי לֵ֭ב נָמ֣וּ שְׁנָתָ֑ם וְלֹא־ מָצְא֖וּ כָל־ אַנְשֵׁי־ חַ֣יִל יְדֵיהֶֽם׃ (ע) מִ֭גַּעֲרָ֣תְךָ אֱלֹהֵ֣י יַעֲקֹ֑ב נִ֝רְדָּ֗ם וְרֶ֣כֶב וָסֽוּס׃ (עא) אַתָּ֤ה ׀ נ֥וֹרָא אַ֗תָּה וּמִֽי־ יַעֲמֹ֥ד לְפָנֶ֗יךָ מֵאָ֥ז אַפֶּֽךָ׃ (עב) מִ֭שָּׁמַיִם הִשְׁמַ֣עְתָּ דִּ֑ין אֶ֖רֶץ יָֽרְאָ֣ה וְשָׁקָֽטָה׃ (עג) בְּקוּם־ לַמִּשְׁפָּ֥ט אֱלֹהִ֑ים לְהוֹשִׁ֖יעַ כָּל־ עַנְוֵי־ אֶ֣רֶץ סֶֽלָה׃ (עד) כִּֽי־ חֲמַ֣ת אָדָ֣ם תּוֹדֶ֑ךָּ שְׁאֵרִ֖ית חֵמֹ֣ת תַּחְגֹּֽר׃ (עה) נִֽדֲר֣וּ וְשַׁלְּמוּ֮ לַיהוָ֪ה אֱ‍ֽלֹהֵ֫יכֶ֥ם כָּל־ סְבִיבָ֑יו יוֹבִ֥ילוּ שַׁ֝֗י לַמּוֹרָֽא׃ (עו) יִ֭בְצֹר ר֣וּחַ נְגִידִ֑ים נ֝וֹרָ֗א לְמַלְכֵי־ אָֽרֶץ׃ סוף פרק עו (עז) לַמְנַצֵּ֥חַ עַֽל־ (ידיתון) יְדוּת֗וּן לְאָסָ֥ף מִזְמֽוֹר׃ (עח) קוֹלִ֣י אֶל־ אֱלֹהִ֣ים וְאֶצְעָ֑קָה קוֹלִ֥י אֶל־ אֱ֝לֹהִ֗ים וְהַאֲזִ֥ין אֵלָֽי׃ (עט) בְּי֥וֹם צָרָתִי֮ אֲדֹנָ֪י דָּ֫רָ֥שְׁתִּי יָדִ֤י ׀ לַ֣יְלָה נִ֭גְּרָה וְלֹ֣א תָפ֑וּג מֵאֲנָ֖ה הִנָּחֵ֣ם נַפְשִֽׁי׃ (פ) אֶזְכְּרָ֣ה אֱלֹהִ֣ים וְאֶֽהֱמָיָ֑ה אָשִׂ֓יחָה ׀ וְתִתְעַטֵּ֖ף רוּחִ֣י סֶֽלָה׃ (פא) אָ֭חַזְתָּ שְׁמֻר֣וֹת עֵינָ֑י נִ֝פְעַ֗מְתִּי וְלֹ֣א אֲדַבֵּֽר׃ (פב) חִשַּׁ֣בְתִּי יָמִ֣ים מִקֶּ֑דֶם שְׁ֝נ֗וֹת עוֹלָמִֽים׃ (פג) אֶֽזְכְּרָ֥ה נְגִינָתִ֗י בַּ֫לָּ֥יְלָה עִם־ לְבָבִ֥י אָשִׂ֑יחָה וַיְחַפֵּ֥שׂ רוּחִֽי׃ (פד) הַֽ֭לְעוֹלָמִים יִזְנַ֥ח ׀ אֲדֹנָ֑י וְלֹֽא־ יֹסִ֖יף לִרְצ֣וֹת עֽוֹד׃ (פה) הֶאָפֵ֣ס לָנֶ֣צַח חַסְדּ֑וֹ גָּ֥מַר אֹ֝֗מֶר לְדֹ֣ר וָדֹֽר׃ (פו) הֲשָׁכַ֣ח חַנּ֣וֹת אֵ֑ל אִם־ קָפַ֥ץ בְּ֝אַ֗ף רַחֲמָ֥יו סֶֽלָה׃ (פז) וָ֭אֹמַר חַלּ֣וֹתִי הִ֑יא שְׁ֝נ֗וֹת יְמִ֣ין עֶלְיֽוֹן׃ (פח) (אזכיר) אֶזְכּ֥וֹר מַֽעַלְלֵי־ יָ֑הּ כִּֽי־ אֶזְכְּרָ֖ה מִקֶּ֣דֶם פִּלְאֶֽךָ׃ (פט) וְהָגִ֥יתִי בְכָל־ פָּעֳלֶ֑ךָ וּֽבַעֲלִ֖ילוֹתֶ֣יךָ אָשִֽׂיחָה׃ (צ) אֱ֭לֹהִים בַּקֹּ֣דֶשׁ דַּרְכֶּ֑ךָ מִי־ אֵ֥ל גָּ֝ד֗וֹל כֵּֽאלֹהִֽים׃ (צא) אַתָּ֣ה הָ֭אֵל עֹ֣שֵׂה פֶ֑לֶא הוֹדַ֖עְתָּ בָעַמִּ֣ים עֻזֶּֽךָ׃ (צב) גָּאַ֣לְתָּ בִּזְר֣וֹעַ עַמֶּ֑ךָ בְּנֵי־ יַעֲקֹ֖ב וְיוֹסֵ֣ף סֶֽלָה׃ (צג) רָ֘א֤וּךָ מַּ֨יִם ׀ אֱ‍ֽלֹהִ֗ים רָא֣וּךָ מַּ֣יִם יָחִ֑ילוּ אַ֝֗ף יִרְגְּז֥וּ תְהֹמֽוֹת׃ (צד) זֹ֤רְמוּ מַ֨יִם ׀ עָב֗וֹת ק֭וֹל נָתְנ֣וּ שְׁחָקִ֑ים אַף־ חֲ֝צָצֶ֗יךָ יִתְהַלָּֽכוּ׃ (צה) ק֤וֹל רַעַמְךָ֨ ׀ בַּגַּלְגַּ֗ל הֵאִ֣ירוּ בְרָקִ֣ים תֵּבֵ֑ל רָגְזָ֖ה וַתִּרְעַ֣שׁ הָאָֽרֶץ׃ (צו) בַּיָּ֤ם דַּרְכֶּ֗ךָ (ושביליך) וּֽ֭שְׁבִֽילְךָ בְּמַ֣יִם רַבִּ֑ים וְ֝עִקְּבוֹתֶ֗יךָ לֹ֣א נֹדָֽעוּ׃ (צז) נָחִ֣יתָ כַצֹּ֣אן עַמֶּ֑ךָ בְּֽיַד־ מֹשֶׁ֥ה וְאַהֲרֹֽן׃ (צח) מַשְׂכִּ֗יל לְאָ֫סָ֥ף הַאֲזִ֣ינָה עַ֭מִּי תּוֹרָתִ֑י הַטּ֥וּ אָ֝זְנְכֶ֗ם לְאִמְרֵי־ פִֽי׃ (צט) אֶפְתְּחָ֣ה בְמָשָׁ֣ל פִּ֑י אַבִּ֥יעָה חִ֝יד֗וֹת מִנִּי־ קֶֽדֶם׃ (ק) אֲשֶׁ֣ר שָׁ֭מַעְנוּ וַנֵּדָעֵ֑ם וַ֝אֲבוֹתֵ֗ינוּ סִפְּרוּ־ לָֽנוּ׃ (קא) לֹ֤א נְכַחֵ֨ד ׀ מִבְּנֵיהֶ֗ם לְד֥וֹר אַחֲר֗וֹן מְֽ֭סַפְּרִים תְּהִלּ֣וֹת יְהוָ֑ה וֶעֱזוּז֥וֹ וְ֝נִפְלְאוֹתָ֗יו אֲשֶׁ֣ר עָשָֽׂה׃ (קב) וַיָּ֤קֶם עֵד֨וּת ׀ בְּֽיַעֲקֹ֗ב וְתוֹרָה֮ שָׂ֤ם בְּיִשְׂרָ֫אֵ֥ל אֲשֶׁ֣ר צִ֭וָּה אֶת־ אֲבוֹתֵ֑ינוּ לְ֝הוֹדִיעָ֗ם לִבְנֵיהֶֽם׃ (קג) לְמַ֤עַן יֵדְע֨וּ ׀ דּ֣וֹר אַ֭חֲרוֹן בָּנִ֣ים יִוָּלֵ֑דוּ יָ֝קֻ֗מוּ וִֽיסַפְּר֥וּ לִבְנֵיהֶֽם׃ (קד) וְיָשִׂ֥ימוּ בֵֽאלֹהִ֗ים כִּ֫סְלָ֥ם וְלֹ֣א יִ֭שְׁכְּחוּ מַֽעַלְלֵי־ אֵ֑ל וּמִצְוֺתָ֥יו יִנְצֹֽרוּ׃ (קה) וְלֹ֤א יִהְי֨וּ ׀ כַּאֲבוֹתָ֗ם דּוֹר֮ סוֹרֵ֪ר וּמֹ֫רֶ֥ה דּ֭וֹר לֹא־ הֵכִ֣ין לִבּ֑וֹ וְלֹא־ נֶאֶמְנָ֖ה אֶת־ אֵ֣ל רוּחֽוֹ׃ (קו) בְּֽנֵי־ אֶפְרַ֗יִם נוֹשְׁקֵ֥י רוֹמֵי־ קָ֑שֶׁת הָ֝פְכ֗וּ בְּי֣וֹם קְרָֽב׃ (קז) לֹ֣א שָׁ֭מְרוּ בְּרִ֣ית אֱלֹהִ֑ים וּ֝בְתוֹרָת֗וֹ מֵאֲנ֥וּ לָלֶֽכֶת׃ (קח) וַיִּשְׁכְּח֥וּ עֲלִילוֹתָ֑יו וְ֝נִפְלְאוֹתָ֗יו אֲשֶׁ֣ר הֶרְאָֽם׃ (קט) נֶ֣גֶד אֲ֭בוֹתָם עָ֣שָׂה פֶ֑לֶא בְּאֶ֖רֶץ מִצְרַ֣יִם שְׂדֵה־ צֹֽעַן׃ (קי) בָּ֣קַע יָ֭ם וַיַּֽעֲבִירֵ֑ם וַֽיַּצֶּב־ מַ֥יִם כְּמוֹ־ נֵֽד׃ (קיא) וַיַּנְחֵ֣ם בֶּעָנָ֣ן יוֹמָ֑ם וְכָל־ הַ֝לַּ֗יְלָה בְּא֣וֹר אֵֽשׁ׃ (קיב) יְבַקַּ֣ע צֻ֭רִים בַּמִּדְבָּ֑ר וַ֝יַּ֗שְׁקְ כִּתְהֹמ֥וֹת רַבָּֽה׃ (קיג) וַיּוֹצִ֣א נוֹזְלִ֣ים מִסָּ֑לַע וַיּ֖וֹרֶד כַּנְּהָר֣וֹת מָֽיִם׃ (קיד) וַיּוֹסִ֣יפוּ ע֭וֹד לַחֲטֹא־ ל֑וֹ לַֽמְר֥וֹת עֶ֝לְי֗וֹן בַּצִּיָּֽה׃ (קטו) וַיְנַסּוּ־ אֵ֥ל בִּלְבָבָ֑ם לִֽשְׁאָל־ אֹ֥כֶל לְנַפְשָֽׁם׃ (קטז) וַֽיְדַבְּר֗וּ בֵּֽאלֹ֫הִ֥ים אָ֭מְרוּ הֲי֣וּכַל אֵ֑ל לַעֲרֹ֥ךְ שֻׁ֝לְחָ֗ן בַּמִּדְבָּֽר׃ (קיז) הֵ֤ן הִכָּה־ צ֨וּר ׀ וַיָּז֣וּבוּ מַיִם֮ וּנְחָלִ֪ים יִ֫שְׁטֹ֥פוּ הֲגַם־ לֶ֭חֶם י֣וּכַל תֵּ֑ת אִם־ יָכִ֖ין שְׁאֵ֣ר לְעַמּֽוֹ׃ (קיח) לָכֵ֤ן ׀ שָׁמַ֥ע יְהוָ֗ה וַֽיִּתְעַבָּ֥ר וְ֭אֵשׁ נִשְּׂקָ֣ה בְיַעֲקֹ֑ב וְגַם־ אַ֝֗ף עָלָ֥ה בְיִשְׂרָאֵֽל׃ (קיט) כִּ֤י לֹ֣א הֶ֭אֱמִינוּ בֵּאלֹהִ֑ים וְלֹ֥א בָ֝טְח֗וּ בִּֽישׁוּעָתֽוֹ׃ (קכ) וַיְצַ֣ו שְׁחָקִ֣ים מִמָּ֑עַל וְדַלְתֵ֖י שָׁמַ֣יִם פָּתָֽח׃ (קכא) וַיַּמְטֵ֬ר עֲלֵיהֶ֣ם מָ֣ן לֶאֱכֹ֑ל וּדְגַן־ שָׁ֝מַ֗יִם נָ֣תַן לָֽמוֹ׃ (קכב) לֶ֣חֶם אַ֭בִּירִים אָ֣כַל אִ֑ישׁ צֵידָ֬ה שָׁלַ֖ח לָהֶ֣ם לָשֹֽׂבַע׃ (קכג) יַסַּ֣ע קָ֭דִים בַּשָּׁמָ֑יִם וַיְנַהֵ֖ג בְּעֻזּ֣וֹ תֵימָֽן׃ (קכד) וַיַּמְטֵ֬ר עֲלֵיהֶ֣ם כֶּעָפָ֣ר שְׁאֵ֑ר וּֽכְח֥וֹל יַ֝מִּ֗ים ע֣וֹף כָּנָֽף׃ (קכה) וַ֭יַּפֵּל בְּקֶ֣רֶב מַחֲנֵ֑הוּ סָ֝בִ֗יב לְמִשְׁכְּנֹתָֽיו׃ (קכו) וַיֹּאכְל֣וּ וַיִּשְׂבְּע֣וּ מְאֹ֑ד וְ֝תַֽאֲוָתָ֗ם יָבִ֥א לָהֶֽם׃ (קכז) לֹא־ זָר֥וּ מִתַּאֲוָתָ֑ם ע֝֗וֹד אָכְלָ֥ם בְּפִיהֶֽם׃ (קכח) וְאַ֤ף אֱלֹהִ֨ים ׀ עָ֘לָ֤ה בָהֶ֗ם וַֽ֭יַּהֲרֹג בְּמִשְׁמַנֵּיהֶ֑ם וּבַחוּרֵ֖י יִשְׂרָאֵ֣ל הִכְרִֽיעַ׃ (קכט) בְּכָל־ זֹ֭את חָֽטְאוּ־ ע֑וֹד וְלֹֽא־ הֶ֝אֱמִ֗ינוּ בְּנִפְלְאוֹתָֽיו׃ (קל) וַיְכַל־ בַּהֶ֥בֶל יְמֵיהֶ֑ם וּ֝שְׁנוֹתָ֗ם בַּבֶּהָלָֽה׃ (קלא) אִם־ הֲרָגָ֥ם וּדְרָשׁ֑וּהוּ וְ֝שָׁ֗בוּ וְשִֽׁחֲרוּ־ אֵֽל׃ (קלב) וַֽ֭יִּזְכְּרוּ כִּֽי־ אֱלֹהִ֣ים צוּרָ֑ם וְאֵ֥ל עֶ֝לְיוֹן גֹּאֲלָֽם׃ (קלג) וַיְפַתּ֥וּהוּ בְּפִיהֶ֑ם
