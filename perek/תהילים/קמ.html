תהילים סדר יח

(א) וּרְאֵֽה־ בָנִ֥ים לְבָנֶ֑יךָ שָׁ֝ל֗וֹם עַל־ יִשְׂרָאֵֽל׃ (ב) שִׁ֗יר הַֽמַּ֫עֲל֥וֹת רַ֭בַּת צְרָר֣וּנִי מִנְּעוּרַ֑י יֹֽאמַר־ נָ֝א יִשְׂרָאֵֽל׃ (ג) רַ֭בַּת צְרָר֣וּנִי מִנְּעוּרָ֑י גַּ֝ם לֹא־ יָ֥כְלוּ לִֽי׃ (ד) עַל־ גַּ֭בִּי חָרְשׁ֣וּ חֹרְשִׁ֑ים הֶ֝אֱרִ֗יכוּ (למענותם) לְמַעֲנִיתָֽם׃ (ה) יְהוָ֥ה צַדִּ֑יק קִ֝צֵּ֗ץ עֲב֣וֹת רְשָׁעִֽים׃ (ו) יֵ֭בֹשׁוּ וְיִסֹּ֣גוּ אָח֑וֹר כֹּ֝֗ל שֹׂנְאֵ֥י צִיּֽוֹן׃ (ז) יִ֭הְיוּ כַּחֲצִ֣יר גַּגּ֑וֹת שֶׁקַּדְמַ֖ת שָׁלַ֣ף יָבֵֽשׁ׃ (ח) שֶׁלֹּ֤א מִלֵּ֖א כַפּ֥וֹ קוֹצֵ֗ר וְחִצְנ֥וֹ מְעַמֵּֽר׃ (ט) וְלֹ֤א אָֽמְר֨וּ ׀ הָעֹבְרִ֗ים בִּרְכַּֽת־ יְהוָ֥ה אֲלֵיכֶ֑ם בֵּרַ֥כְנוּ אֶ֝תְכֶ֗ם בְּשֵׁ֣ם יְהוָֽה׃ (י) שִׁ֥יר הַֽמַּעֲל֑וֹת מִמַּעֲמַקִּ֖ים קְרָאתִ֣יךָ יְהוָֽה׃ (יא) אֲדֹנָי֮ שִׁמְעָ֪ה בְק֫וֹלִ֥י תִּהְיֶ֣ינָה אָ֭זְנֶיךָ קַשֻּׁב֑וֹת לְ֝ק֗וֹל תַּחֲנוּנָֽי׃ (יב) אִם־ עֲוֺנ֥וֹת תִּשְׁמָר־ יָ֑הּ אֲ֝דֹנָ֗י מִ֣י יַעֲמֹֽד׃ (יג) כִּֽי־ עִמְּךָ֥ הַסְּלִיחָ֑ה לְ֝מַ֗עַן תִּוָּרֵֽא׃ (יד) קִוִּ֣יתִי יְ֭הוָה קִוְּתָ֣ה נַפְשִׁ֑י וְֽלִדְבָר֥וֹ הוֹחָֽלְתִּי׃ (טו) נַפְשִׁ֥י לַֽאדֹנָ֑י מִשֹּׁמְרִ֥ים לַ֝בֹּ֗קֶר שֹׁמְרִ֥ים לַבֹּֽקֶר׃ (טז) יַחֵ֥ל יִשְׂרָאֵ֗ל אֶל־ יְה֫וָה כִּֽי־ עִם־ יְהוָ֥ה הַחֶ֑סֶד וְהַרְבֵּ֖ה עִמּ֣וֹ פְדֽוּת׃ (יז) וְ֭הוּא יִפְדֶּ֣ה אֶת־ יִשְׂרָאֵ֑ל מִ֝כֹּ֗ל עֲוֺנֹתָֽיו׃ (יח) שִׁ֥יר הַֽמַּֽעֲל֗וֹת לְדָ֫וִ֥ד יְהוָ֤ה ׀ לֹא־ גָבַ֣הּ לִ֭בִּי וְלֹא־ רָמ֣וּ עֵינַ֑י וְלֹֽא־ הִלַּ֓כְתִּי ׀ בִּגְדֹל֖וֹת וּבְנִפְלָא֣וֹת מִמֶּֽנִּי׃ (יט) אִם־ לֹ֤א שִׁוִּ֨יתִי ׀ וְדוֹמַ֗מְתִּי נַ֫פְשִׁ֥י כְּ֭גָמֻל עֲלֵ֣י אִמּ֑וֹ כַּגָּמֻ֖ל עָלַ֣י נַפְשִֽׁי׃ (כ) יַחֵ֣ל יִ֭שְׂרָאֵל אֶל־ יְהוָ֑ה מֵֽ֝עַתָּ֗ה וְעַד־ עוֹלָֽם׃ (כא) שִׁ֗יר הַֽמַּ֫עֲל֥וֹת זְכוֹר־ יְהוָ֥ה לְדָוִ֑ד אֵ֝ת כָּל־ עֻנּוֹתֽוֹ׃ (כב) אֲשֶׁ֣ר נִ֭שְׁבַּע לַיהוָ֑ה נָ֝דַ֗ר לַאֲבִ֥יר יַעֲקֹֽב׃ (כג) אִם־ אָ֭בֹא בְּאֹ֣הֶל בֵּיתִ֑י אִם־ אֶ֝עֱלֶ֗ה עַל־ עֶ֥רֶשׂ יְצוּעָֽי׃ (כד) אִם־ אֶתֵּ֣ן שְׁנַ֣ת לְעֵינָ֑י לְֽעַפְעַפַּ֥י תְּנוּמָֽה׃ (כה) עַד־ אֶמְצָ֣א מָ֭קוֹם לַיהוָ֑ה מִ֝שְׁכָּנ֗וֹת לַאֲבִ֥יר יַעֲקֹֽב׃ (כו) הִנֵּֽה־ שְׁמַֽעֲנ֥וּהָ בְאֶפְרָ֑תָה מְ֝צָאנ֗וּהָ בִּשְׂדֵי־ יָֽעַר׃ (כז) נָב֥וֹאָה לְמִשְׁכְּנוֹתָ֑יו נִ֝שְׁתַּחֲוֶ֗ה לַהֲדֹ֥ם רַגְלָֽיו׃ (כח) קוּמָ֣ה יְ֭הוָה לִמְנוּחָתֶ֑ךָ אַ֝תָּ֗ה וַאֲר֥וֹן עֻזֶּֽךָ׃ (כט) כֹּהֲנֶ֥יךָ יִלְבְּשׁוּ־ צֶ֑דֶק וַחֲסִידֶ֥יךָ יְרַנֵּֽנוּ׃ (ל) בַּ֭עֲבוּר דָּוִ֣ד עַבְדֶּ֑ךָ אַל־ תָּ֝שֵׁ֗ב פְּנֵ֣י מְשִׁיחֶֽךָ׃ (לא) נִשְׁבַּֽע־ יְהוָ֨ה ׀ לְדָוִ֡ד אֱמֶת֮ לֹֽא־ יָשׁ֪וּב מִ֫מֶּ֥נָּה מִפְּרִ֥י בִטְנְךָ֑ אָ֝שִׁ֗ית לְכִסֵּא־ לָֽךְ׃ (לב) אִֽם־ יִשְׁמְר֬וּ בָנֶ֨יךָ ׀ בְּרִיתִי֮ וְעֵדֹתִ֥י ז֗וֹ אֲלַ֫מְּדֵ֥ם גַּם־ בְּנֵיהֶ֥ם עֲדֵי־ עַ֑ד יֵ֝שְׁב֗וּ לְכִסֵּא־ לָֽךְ׃ (לג) כִּֽי־ בָחַ֣ר יְהוָ֣ה בְּצִיּ֑וֹן אִ֝וָּ֗הּ לְמוֹשָׁ֥ב לֽוֹ׃ (לד) זֹאת־ מְנוּחָתִ֥י עֲדֵי־ עַ֑ד פֹּֽה־ אֵ֝שֵׁ֗ב כִּ֣י אִוִּתִֽיהָ׃ (לה) צֵ֭ידָהּ בָּרֵ֣ךְ אֲבָרֵ֑ךְ אֶ֝בְיוֹנֶ֗יהָ אַשְׂבִּ֥יעַֽ לָֽחֶם׃ (לו) וְֽ֭כֹהֲנֶיהָ אַלְבִּ֣ישׁ יֶ֑שַׁע וַ֝חֲסִידֶ֗יהָ רַנֵּ֥ן יְרַנֵּֽנוּ׃ (לז) שָׁ֤ם אַצְמִ֣יחַ קֶ֣רֶן לְדָוִ֑ד עָרַ֥כְתִּי נֵ֝֗ר לִמְשִׁיחִֽי׃ (לח) א֭וֹיְבָיו אַלְבִּ֣ישׁ בֹּ֑שֶׁת וְ֝עָלָ֗יו יָצִ֥יץ נִזְרֽוֹ׃ (לט) שִׁ֥יר הַֽמַּעֲל֗וֹת לְדָ֫וִ֥ד הִנֵּ֣ה מַה־ טּ֭וֹב וּמַה־ נָּעִ֑ים שֶׁ֖בֶת אַחִ֣ים גַּם־ יָֽחַד׃ (מ) כַּשֶּׁ֤מֶן הַטּ֨וֹב ׀ עַל־ הָרֹ֗אשׁ יֹרֵ֗ד עַֽל־ הַזָּקָ֥ן זְקַֽן־ אַהֲרֹ֑ן שֶׁ֝יֹּרֵ֗ד עַל־ פִּ֥י מִדּוֹתָֽיו׃ (מא) כְּטַל־ חֶרְמ֗וֹן שֶׁיֹּרֵד֮ עַל־ הַרְרֵ֪י צִ֫יּ֥וֹן כִּ֤י שָׁ֨ם ׀ צִוָּ֣ה יְ֭הוָה אֶת־ הַבְּרָכָ֑ה חַ֝יִּ֗ים עַד־ הָעוֹלָֽם׃ (מב) שִׁ֗יר הַֽמַּ֫עֲל֥וֹת הִנֵּ֤ה ׀ בָּרֲכ֣וּ אֶת־ יְ֭הוָה כָּל־ עַבְדֵ֣י יְהוָ֑ה הָעֹמְדִ֥ים בְּבֵית־ יְ֝הוָ֗ה בַּלֵּילֽוֹת׃ (מג) שְׂאֽוּ־ יְדֵכֶ֥ם קֹ֑דֶשׁ וּ֝בָרֲכוּ אֶת־ יְהוָֽה׃ (מד) יְבָרֶכְךָ֣ יְ֭הוָה מִצִּיּ֑וֹן עֹ֝שֵׂ֗ה שָׁמַ֥יִם וָאָֽרֶץ׃ (מה) הַ֥לְלוּ יָ֨הּ ׀ הַֽ֭לְלוּ אֶת־ שֵׁ֣ם יְהוָ֑ה הַֽ֝לְלוּ עַבְדֵ֥י יְהוָֽה׃ (מו) שֶׁ֣֭עֹֽמְדִים בְּבֵ֣ית יְהוָ֑ה בְּ֝חַצְר֗וֹת בֵּ֣ית אֱלֹהֵֽינוּ׃ (מז) הַֽ֭לְלוּ־ יָהּ כִּי־ ט֣וֹב יְהוָ֑ה זַמְּר֥וּ לִ֝שְׁמ֗וֹ כִּ֣י נָעִֽים׃ (מח) כִּֽי־ יַעֲקֹ֗ב בָּחַ֣ר ל֣וֹ יָ֑הּ יִ֝שְׂרָאֵ֗ל לִסְגֻלָּתֽוֹ׃ (מט) כִּ֤י אֲנִ֣י יָ֭דַעְתִּי כִּי־ גָד֣וֹל יְהוָ֑ה וַ֝אֲדֹנֵ֗ינוּ מִכָּל־ אֱלֹהִֽים׃ (נ) כֹּ֤ל אֲשֶׁר־ חָפֵ֥ץ יְהוָ֗ה עָ֫שָׂ֥ה בַּשָּׁמַ֥יִם וּבָאָ֑רֶץ בַּ֝יַּמִּ֗ים וְכָל־ תְּהוֹמֽוֹת׃ (נא) מַֽעֲלֶ֣ה נְשִׂאִים֮ מִקְצֵ֪ה הָ֫אָ֥רֶץ בְּרָקִ֣ים לַמָּטָ֣ר עָשָׂ֑ה מֽוֹצֵא־ ר֝וּחַ מֵאֽוֹצְרוֹתָֽיו׃ (נב) שֶֽׁ֭הִכָּה בְּכוֹרֵ֣י מִצְרָ֑יִם מֵ֝אָדָ֗ם עַד־ בְּהֵמָֽה׃ (נג) שָׁלַ֤ח ׀ אֹת֣וֹת וּ֭מֹפְתִים בְּתוֹכֵ֣כִי מִצְרָ֑יִם בְּ֝פַרְעֹ֗ה וּבְכָל־ עֲבָדָֽיו׃ (נד) שֶֽׁ֭הִכָּה גּוֹיִ֣ם רַבִּ֑ים וְ֝הָרַ֗ג מְלָכִ֥ים עֲצוּמִֽים׃ (נה) לְסִיח֤וֹן ׀ מֶ֤לֶךְ הָאֱמֹרִ֗י וּ֭לְעוֹג מֶ֣לֶךְ הַבָּשָׁ֑ן וּ֝לְכֹ֗ל מַמְלְכ֥וֹת כְּנָֽעַן׃ (נו) וְנָתַ֣ן אַרְצָ֣ם נַחֲלָ֑ה נַ֝חֲלָ֗ה לְיִשְׂרָאֵ֥ל עַמּֽוֹ׃ (נז) יְ֭הוָה שִׁמְךָ֣ לְעוֹלָ֑ם יְ֝הוָ֗ה זִכְרְךָ֥ לְדֹר־ וָדֹֽר׃ (נח) כִּֽי־ יָדִ֣ין יְהוָ֣ה עַמּ֑וֹ וְעַל־ עֲ֝בָדָ֗יו יִתְנֶחָֽם׃ (נט) עֲצַבֵּ֣י הַ֭גּוֹיִם כֶּ֣סֶף וְזָהָ֑ב מַ֝עֲשֵׂ֗ה יְדֵ֣י אָדָֽם׃ (ס) פֶּֽה־ לָ֭הֶם וְלֹ֣א יְדַבֵּ֑רוּ עֵינַ֥יִם לָ֝הֶ֗ם וְלֹ֣א יִרְאֽוּ׃ (סא) אָזְנַ֣יִם לָ֭הֶם וְלֹ֣א יַאֲזִ֑ינוּ אַ֝֗ף אֵין־ יֶשׁ־ ר֥וּחַ בְּפִיהֶֽם׃ (סב) כְּ֭מוֹהֶם יִהְי֣וּ עֹשֵׂיהֶ֑ם כֹּ֭ל אֲשֶׁר־ בֹּטֵ֣חַ בָּהֶֽם׃ (סג) בֵּ֣ית יִ֭שְׂרָאֵל בָּרֲכ֣וּ אֶת־ יְהוָ֑ה בֵּ֥ית אַ֝הֲרֹ֗ן בָּרֲכ֥וּ אֶת־ יְהוָֽה׃ (סד) בֵּ֣ית הַ֭לֵּוִי בָּרֲכ֣וּ אֶת־ יְהוָ֑ה יִֽרְאֵ֥י יְ֝הוָ֗ה בָּרֲכ֥וּ אֶת־ יְהוָֽה׃ (סה) בָּ֘ר֤וּךְ יְהוָ֨ה ׀ מִצִּיּ֗וֹן שֹׁ֘כֵ֤ן יְֽרוּשָׁלִָ֗ם הַֽלְלוּ־ יָֽהּ׃ (סו) הוֹד֣וּ לַיהוָ֣ה כִּי־ ט֑וֹב כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (סז) ה֭וֹדוּ לֵֽאלֹהֵ֣י הָאֱלֹהִ֑ים כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (סח) ה֭וֹדוּ לַאֲדֹנֵ֣י הָאֲדֹנִ֑ים כִּ֖י בִּרְקִ֥יעַ עֻזּֽוֹ׃ (קכב) הַֽלְל֥וּהוּ בִגְבוּרֹתָ֑יו הַֽ֝לְל֗וּהוּ כְּרֹ֣ב גֻּדְלֽוֹ׃ (קכג) הַֽ֭לְלוּהוּ בְּתֵ֣קַע שׁוֹפָ֑ר הַֽ֝לְל֗וּהוּ בְּנֵ֣בֶל וְכִנּֽוֹר׃ (קכד) הַֽ֭לְלוּהוּ בְתֹ֣ף וּמָח֑וֹל הַֽ֝לְל֗וּהוּ בְּמִנִּ֥ים וְעוּגָֽב׃ (קכה) הַֽלְל֥וּהוּ בְצִלְצְלֵי־ שָׁ֑מַע הַֽ֝לְל֗וּהוּ בְּֽצִלְצְלֵ֥י תְרוּעָֽה׃ (קכו) כֹּ֣ל הַ֭נְּשָׁמָה תְּהַלֵּ֥ל יָ֗הּ הַֽלְלוּ־ יָֽהּ׃
