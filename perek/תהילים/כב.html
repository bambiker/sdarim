תהילים פרק כב לפי החלוקה לסדרים

פרק כב בספר תהילים (32 פסוקים) נמצא בתהילים סדר ג (פסוקים טז–מז).

תהילים סדר ג

(א) יְהוָ֥ה הוֹשִׁ֑יעָה הַ֝מֶּ֗לֶךְ יַעֲנֵ֥נוּ בְיוֹם־ קָרְאֵֽנוּ׃ (ב) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (ג) יְֽהוָ֗ה בְּעָזְּךָ֥ יִשְׂמַח־ מֶ֑לֶךְ וּ֝בִישׁ֥וּעָתְךָ֗ מַה־ (יגיל) יָּ֥גֶל מְאֹֽד׃ (ד) תַּאֲוַ֣ת לִ֭בּוֹ נָתַ֣תָּה לּ֑וֹ וַאֲרֶ֥שֶׁת שְׂ֝פָתָ֗יו בַּל־ מָנַ֥עְתָּ סֶּֽלָה׃ (ה) כִּֽי־ תְ֭קַדְּמֶנּוּ בִּרְכ֣וֹת ט֑וֹב תָּשִׁ֥ית לְ֝רֹאשׁ֗וֹ עֲטֶ֣רֶת פָּֽז׃ (ו) חַיִּ֤ים ׀ שָׁאַ֣ל מִ֭מְּךָ נָתַ֣תָּה לּ֑וֹ אֹ֥רֶךְ יָ֝מִ֗ים עוֹלָ֥ם וָעֶֽד׃ (ז) גָּד֣וֹל כְּ֭בוֹדוֹ בִּישׁוּעָתֶ֑ךָ ה֥וֹד וְ֝הָדָר תְּשַׁוֶּ֥ה עָלָֽיו׃ (ח) כִּֽי־ תְשִׁיתֵ֣הוּ בְרָכ֣וֹת לָעַ֑ד תְּחַדֵּ֥הוּ בְ֝שִׂמְחָ֗ה אֶת־ פָּנֶֽיךָ׃ (ט) כִּֽי־ הַ֭מֶּלֶךְ בֹּטֵ֣חַ בַּיהוָ֑ה וּבְחֶ֥סֶד עֶ֝לְי֗וֹן בַּל־ יִמּֽוֹט׃ (י) תִּמְצָ֣א יָ֭דְךָ לְכָל־ אֹיְבֶ֑יךָ יְ֝מִֽינְךָ תִּמְצָ֥א שֹׂנְאֶֽיךָ׃ (יא) תְּשִׁיתֵ֤מוֹ ׀ כְּתַנּ֥וּר אֵשׁ֮ לְעֵ֪ת פָּ֫נֶ֥יךָ יְ֭הוָה בְּאַפּ֣וֹ יְבַלְּעֵ֑ם וְֽתֹאכְלֵ֥ם אֵֽשׁ׃ (יב) פִּ֭רְיָמוֹ מֵאֶ֣רֶץ תְּאַבֵּ֑ד וְ֝זַרְעָ֗ם מִבְּנֵ֥י אָדָֽם׃ (יג) כִּי־ נָט֣וּ עָלֶ֣יךָ רָעָ֑ה חָֽשְׁב֥וּ מְ֝זִמָּ֗ה בַּל־ יוּכָֽלוּ׃ (יד) כִּ֭י תְּשִׁיתֵ֣מוֹ שֶׁ֑כֶם בְּ֝מֵֽיתָרֶ֗יךָ תְּכוֹנֵ֥ן עַל־ פְּנֵיהֶֽם׃ (טו) ר֣וּמָה יְהוָ֣ה בְּעֻזֶּ֑ךָ נָשִׁ֥ירָה וּֽ֝נְזַמְּרָה גְּבוּרָתֶֽךָ׃ תחילת פרק כב (טז) לַ֭מְנַצֵּחַ עַל־ אַיֶּ֥לֶת הַשַּׁ֗חַר מִזְמ֥וֹר לְדָוִֽד׃ (יז) אֵלִ֣י אֵ֭לִי לָמָ֣ה עֲזַבְתָּ֑נִי רָח֥וֹק מִֽ֝ישׁוּעָתִ֗י דִּבְרֵ֥י שַׁאֲגָתִֽי׃ (יח) אֱ‍ֽלֹהַ֗י אֶקְרָ֣א י֭וֹמָם וְלֹ֣א תַעֲנֶ֑ה וְ֝לַ֗יְלָה וְֽלֹא־ דֽוּמִיָּ֥ה לִֽי׃ (יט) וְאַתָּ֥ה קָד֑וֹשׁ י֝וֹשֵׁ֗ב תְּהִלּ֥וֹת יִשְׂרָאֵֽל׃ (כ) בְּ֭ךָ בָּטְח֣וּ אֲבֹתֵ֑ינוּ בָּ֝טְח֗וּ וַֽתְּפַלְּטֵֽמוֹ׃ (כא) אֵלֶ֣יךָ זָעֲק֣וּ וְנִמְלָ֑טוּ בְּךָ֖ בָטְח֣וּ וְלֹא־ בֽוֹשׁוּ׃ (כב) וְאָנֹכִ֣י תוֹלַ֣עַת וְלֹא־ אִ֑ישׁ חֶרְפַּ֥ת אָ֝דָ֗ם וּבְז֥וּי עָֽם׃ (כג) כָּל־ רֹ֭אַי יַלְעִ֣גוּ לִ֑י יַפְטִ֥ירוּ בְ֝שָׂפָ֗ה יָנִ֥יעוּ רֹֽאשׁ׃ (כד) גֹּ֣ל אֶל־ יְהוָ֣ה יְפַלְּטֵ֑הוּ יַ֝צִּילֵ֗הוּ כִּ֘י חָ֥פֵֽץ בּֽוֹ׃ (כה) כִּֽי־ אַתָּ֣ה גֹחִ֣י מִבָּ֑טֶן מַ֝בְטִיחִ֗י עַל־ שְׁדֵ֥י אִמִּֽי׃ (כו) עָ֭לֶיךָ הָשְׁלַ֣כְתִּי מֵרָ֑חֶם מִבֶּ֥טֶן אִ֝מִּ֗י אֵ֣לִי אָֽתָּה׃ (כז) אַל־ תִּרְחַ֣ק מִ֭מֶּנִּי כִּי־ צָרָ֣ה קְרוֹבָ֑ה כִּי־ אֵ֥ין עוֹזֵֽר׃ (כח) סְ֭בָבוּנִי פָּרִ֣ים רַבִּ֑ים אַבִּירֵ֖י בָשָׁ֣ן כִּתְּרֽוּנִי׃ (כט) פָּצ֣וּ עָלַ֣י פִּיהֶ֑ם אַ֝רְיֵ֗ה טֹרֵ֥ף וְשֹׁאֵֽג׃ (ל) כַּמַּ֥יִם נִשְׁפַּכְתִּי֮ וְהִתְפָּֽרְד֗וּ כָּֽל־ עַצְמ֫וֹתָ֥י הָיָ֣ה לִ֭בִּי כַּדּוֹנָ֑ג נָ֝מֵ֗ס בְּת֣וֹךְ מֵעָֽי׃ (לא) יָ֘בֵ֤שׁ כַּחֶ֨רֶשׂ ׀ כֹּחִ֗י וּ֭לְשׁוֹנִי מֻדְבָּ֣ק מַלְקוֹחָ֑י וְֽלַעֲפַר־ מָ֥וֶת תִּשְׁפְּתֵֽנִי׃ (לב) כִּ֥י סְבָב֗וּנִי כְּלָ֫בִ֥ים עֲדַ֣ת מְ֭רֵעִים הִקִּיפ֑וּנִי כָּ֝אֲרִ֗י יָדַ֥י וְרַגְלָֽי׃ (לג) אֲסַפֵּ֥ר כָּל־ עַצְמוֹתָ֑י הֵ֥מָּה יַ֝בִּ֗יטוּ יִרְאוּ־ בִֽי׃ (לד) יְחַלְּק֣וּ בְגָדַ֣י לָהֶ֑ם וְעַל־ לְ֝בוּשִׁ֗י יַפִּ֥ילוּ גוֹרָֽל׃ (לה) וְאַתָּ֣ה יְ֭הוָה אַל־ תִּרְחָ֑ק אֱ֝יָלוּתִ֗י לְעֶזְרָ֥תִי חֽוּשָׁה׃ (לו) הַצִּ֣ילָה מֵחֶ֣רֶב נַפְשִׁ֑י מִיַּד־ כֶּ֝֗לֶב יְחִידָתִֽי׃ (לז) ה֭וֹשִׁיעֵנִי מִפִּ֣י אַרְיֵ֑ה וּמִקַּרְנֵ֖י רֵמִ֣ים עֲנִיתָֽנִי׃ (לח) אֲסַפְּרָ֣ה שִׁמְךָ֣ לְאֶחָ֑י בְּת֖וֹךְ קָהָ֣ל אֲהַלְלֶֽךָּ׃ (לט) יִרְאֵ֤י יְהוָ֨ה ׀ הַֽלְל֗וּהוּ כָּל־ זֶ֣רַע יַעֲקֹ֣ב כַּבְּד֑וּהוּ וְג֥וּרוּ מִ֝מֶּ֗נּוּ כָּל־ זֶ֥רַע יִשְׂרָאֵֽל׃ (מ) כִּ֤י לֹֽא־ בָזָ֨ה וְלֹ֪א שִׁקַּ֡ץ עֱנ֬וּת עָנִ֗י וְלֹא־ הִסְתִּ֣יר פָּנָ֣יו מִמֶּ֑נּוּ וּֽבְשַׁוְּע֖וֹ אֵלָ֣יו שָׁמֵֽעַ׃ (מא) מֵ֥אִתְּךָ֗ תְֽהִלָּ֫תִ֥י בְּקָהָ֥ל רָ֑ב נְדָרַ֥י אֲ֝שַׁלֵּ֗ם נֶ֣גֶד יְרֵאָֽיו׃ (מב) יֹאכְל֬וּ עֲנָוִ֨ים ׀ וְיִשְׂבָּ֗עוּ יְהַֽלְל֣וּ יְ֭הוָה דֹּ֣רְשָׁ֑יו יְחִ֖י לְבַבְכֶ֣ם לָעַֽד׃ (מג) יִזְכְּר֤וּ ׀ וְיָשֻׁ֣בוּ אֶל־ יְ֭הוָה כָּל־ אַפְסֵי־ אָ֑רֶץ וְיִֽשְׁתַּחֲו֥וּ לְ֝פָנֶ֗יךָ כָּֽל־ מִשְׁפְּח֥וֹת גּוֹיִֽם׃ (מד) כִּ֣י לַ֭יהוָה הַמְּלוּכָ֑ה וּ֝מֹשֵׁ֗ל בַּגּוֹיִֽם׃ (מה) אָכְל֬וּ וַיִּֽשְׁתַּחֲוּ֨וּ ׀ כָּֽל־ דִּשְׁנֵי־ אֶ֗רֶץ לְפָנָ֣יו יִ֭כְרְעוּ כָּל־ יוֹרְדֵ֣י עָפָ֑ר וְ֝נַפְשׁ֗וֹ לֹ֣א חִיָּֽה׃ (מו) זֶ֥רַע יַֽעַבְדֶ֑נּוּ יְסֻפַּ֖ר לַֽאדֹנָ֣י לַדּֽוֹר׃ (מז) יָ֭בֹאוּ וְיַגִּ֣ידוּ צִדְקָת֑וֹ לְעַ֥ם נ֝וֹלָ֗ד כִּ֣י עָשָֽׂה׃ סוף פרק כב (מח) מִזְמ֥וֹר לְדָוִ֑ד יְהוָ֥ה רֹ֝עִ֗י לֹ֣א אֶחְסָֽר׃ (מט) בִּנְא֣וֹת דֶּ֭שֶׁא יַרְבִּיצֵ֑נִי עַל־ מֵ֖י מְנֻח֣וֹת יְנַהֲלֵֽנִי׃ (נ) נַפְשִׁ֥י יְשׁוֹבֵ֑ב יַֽנְחֵ֥נִי בְמַעְגְּלֵי־ צֶ֝֗דֶק לְמַ֣עַן שְׁמֽוֹ׃ (נא) גַּ֤ם כִּֽי־ אֵלֵ֨ךְ בְּגֵ֪יא צַלְמָ֡וֶת לֹא־ אִ֘ירָ֤א רָ֗ע כִּי־ אַתָּ֥ה עִמָּדִ֑י שִׁבְטְךָ֥ וּ֝מִשְׁעַנְתֶּ֗ךָ הֵ֣מָּה יְנַֽחֲמֻֽנִי׃ (נב) תַּעֲרֹ֬ךְ לְפָנַ֨י ׀ שֻׁלְחָ֗ן נֶ֥גֶד צֹרְרָ֑י דִּשַּׁ֖נְתָּ בַשֶּׁ֥מֶן רֹ֝אשִׁ֗י כּוֹסִ֥י רְוָיָֽה׃ (נג) אַ֤ךְ ׀ ט֤וֹב וָחֶ֣סֶד יִ֭רְדְּפוּנִי כָּל־ יְמֵ֣י חַיָּ֑י וְשַׁבְתִּ֥י בְּבֵית־ יְ֝הוָ֗ה לְאֹ֣רֶךְ יָמִֽים׃ (נד) לְדָוִ֗ד מִ֫זְמ֥וֹר לַֽ֭יהוָה הָאָ֣רֶץ וּמְלוֹאָ֑הּ תֵּ֝בֵ֗ל וְיֹ֣שְׁבֵי בָֽהּ׃ (נה) כִּי־ ה֭וּא עַל־ יַמִּ֣ים יְסָדָ֑הּ וְעַל־ נְ֝הָר֗וֹת יְכוֹנְנֶֽהָ׃ (נו) מִֽי־ יַעֲלֶ֥ה בְהַר־ יְהוָ֑ה וּמִי־ יָ֝קוּם בִּמְק֥וֹם קָדְשֽׁוֹ׃ (נז) נְקִ֥י כַפַּ֗יִם וּֽבַר־ לֵ֫בָ֥ב אֲשֶׁ֤ר ׀ לֹא־ נָשָׂ֣א לַשָּׁ֣וְא נַפְשִׁ֑י וְלֹ֖א נִשְׁבַּ֣ע לְמִרְמָֽה׃ (נח) יִשָּׂ֣א בְ֭רָכָה מֵאֵ֣ת יְהוָ֑ה וּ֝צְדָקָ֗ה מֵאֱלֹהֵ֥י יִשְׁעֽוֹ׃ (נט) זֶ֭ה דּ֣וֹר (דרשו) דֹּרְשָׁ֑יו מְבַקְשֵׁ֨י פָנֶ֖יךָ יַעֲקֹ֣ב סֶֽלָה׃ (ס) שְׂא֤וּ שְׁעָרִ֨ים ׀ רָֽאשֵׁיכֶ֗ם וְֽ֭הִנָּשְׂאוּ פִּתְחֵ֣י עוֹלָ֑ם וְ֝יָב֗וֹא מֶ֣לֶךְ הַכָּבֽוֹד׃ (סא) מִ֥י זֶה֮ מֶ֤לֶךְ הַכָּ֫ב֥וֹד יְ֭הוָה עִזּ֣וּז וְגִבּ֑וֹר יְ֝הוָ֗ה גִּבּ֥וֹר מִלְחָמָֽה׃ (סב) שְׂא֤וּ שְׁעָרִ֨ים ׀ רָֽאשֵׁיכֶ֗ם וּ֭שְׂאוּ פִּתְחֵ֣י עוֹלָ֑ם וְ֝יָבֹא מֶ֣לֶךְ הַכָּבֽוֹד׃ (סג) מִ֤י ה֣וּא זֶה֮ מֶ֤לֶךְ הַכָּ֫ב֥וֹד יְהוָ֥ה צְבָא֑וֹת ה֤וּא מֶ֖לֶךְ הַכָּב֣וֹד סֶֽלָה׃ (סד) לְדָוִ֡ד אֵלֶ֥יךָ יְ֝הוָ֗ה נַפְשִׁ֥י אֶשָּֽׂא׃ (סה) אֱ‍ֽלֹהַ֗י בְּךָ֣ בָ֭טַחְתִּי אַל־ אֵב֑וֹשָׁה אַל־ יַֽעַלְצ֖וּ אֹיְבַ֣י לִֽי׃ (סו) גַּ֣ם כָּל־ קֹ֭וֶיךָ לֹ֣א יֵבֹ֑שׁוּ יֵ֝בֹ֗שׁוּ הַבּוֹגְדִ֥ים רֵיקָֽם׃ (סז) דְּרָכֶ֣יךָ יְ֭הוָה הוֹדִיעֵ֑נִי אֹ֖רְחוֹתֶ֣יךָ לַמְּדֵֽנִי׃ הִרְעִ֑ים יְ֝הוָ֗ה עַל־ מַ֥יִם רַבִּֽים׃ (קכד) קוֹל־ יְהוָ֥ה בַּכֹּ֑חַ ק֥וֹל יְ֝הוָ֗ה בֶּהָדָֽר׃ (קכה) ק֣וֹל יְ֭הוָה שֹׁבֵ֣ר אֲרָזִ֑ים וַיְשַׁבֵּ֥ר יְ֝הוָ֗ה אֶת־ אַרְזֵ֥י הַלְּבָנֽוֹן׃ (קכו) וַיַּרְקִידֵ֥ם כְּמוֹ־ עֵ֑גֶל לְבָנ֥וֹן וְ֝שִׂרְיֹ֗ן כְּמ֣וֹ בֶן־ רְאֵמִֽים׃ (קכז) קוֹל־ יְהוָ֥ה חֹצֵ֗ב לַהֲב֥וֹת אֵֽשׁ׃ (קכח) ק֣וֹל יְ֭הוָה יָחִ֣יל מִדְבָּ֑ר יָחִ֥יל יְ֝הוָ֗ה מִדְבַּ֥ר קָדֵֽשׁ׃ (קכט) ק֤וֹל יְהוָ֨ה ׀ יְחוֹלֵ֣ל אַיָּלוֹת֮ וַֽיֶּחֱשֹׂ֪ף יְעָ֫ר֥וֹת וּבְהֵיכָל֑וֹ כֻּ֝לּ֗וֹ אֹמֵ֥ר כָּבֽוֹד׃ (קל) יְ֭הוָה לַמַּבּ֣וּל יָשָׁ֑ב וַיֵּ֥שֶׁב יְ֝הוָ֗ה מֶ֣לֶךְ לְעוֹלָֽם׃
