תהילים פרק כ לפי החלוקה לסדרים

פרק כ בספר תהילים (10 פסוקים) נמצא בתהילים סדר ב (פסוקים קכא–קכט) ובתהילים סדר ג (פסוקים א–א).

תהילים סדר ב

(א) כִּֽי־ צַדִּ֣יק יְ֭הוָה צְדָק֣וֹת אָהֵ֑ב יָ֝שָׁ֗ר יֶחֱז֥וּ פָנֵֽימוֹ׃ (ב) לַמְנַצֵּ֥חַ עַֽל־ הַשְּׁמִינִ֗ית מִזְמ֥וֹר לְדָוִֽד׃ (ג) הוֹשִׁ֣יעָה יְ֭הוָה כִּי־ גָמַ֣ר חָסִ֑יד כִּי־ פַ֥סּוּ אֱ֝מוּנִ֗ים מִבְּנֵ֥י אָדָֽם׃ (ד) שָׁ֤וְא ׀ יְֽדַבְּרוּ֮ אִ֤ישׁ אֶת־ רֵ֫עֵ֥הוּ שְׂפַ֥ת חֲלָק֑וֹת בְּלֵ֖ב וָלֵ֣ב יְדַבֵּֽרוּ׃ (ה) יַכְרֵ֣ת יְ֭הוָה כָּל־ שִׂפְתֵ֣י חֲלָק֑וֹת לָ֝שׁ֗וֹן מְדַבֶּ֥רֶת גְּדֹלֽוֹת׃ (ו) אֲשֶׁ֤ר אָֽמְר֨וּ ׀ לִלְשֹׁנֵ֣נוּ נַ֭גְבִּיר שְׂפָתֵ֣ינוּ אִתָּ֑נוּ מִ֖י אָד֣וֹן לָֽנוּ׃ (ז) מִשֹּׁ֥ד עֲנִיִּים֮ מֵאַנְקַ֪ת אֶבְי֫וֹנִ֥ים עַתָּ֣ה אָ֭קוּם יֹאמַ֣ר יְהוָ֑ה אָשִׁ֥ית בְּ֝יֵ֗שַׁע יָפִ֥יחַֽ לֽוֹ׃ (ח) אִֽמֲר֣וֹת יְהוָה֮ אֲמָר֪וֹת טְהֹ֫ר֥וֹת כֶּ֣סֶף צָ֭רוּף בַּעֲלִ֣יל לָאָ֑רֶץ מְ֝זֻקָּ֗ק שִׁבְעָתָֽיִם׃ (ט) אַתָּֽה־ יְהוָ֥ה תִּשְׁמְרֵ֑ם תִּצְּרֶ֓נּוּ ׀ מִן־ הַדּ֖וֹר ז֣וּ לְעוֹלָֽם׃ (י) סָבִ֗יב רְשָׁעִ֥ים יִתְהַלָּכ֑וּן כְּרֻ֥ם זֻ֝לּ֗וּת לִבְנֵ֥י אָדָֽם׃ (יא) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (יב) עַד־ אָ֣נָה יְ֭הוָה תִּשְׁכָּחֵ֣נִי נֶ֑צַח עַד־ אָ֓נָה ׀ תַּסְתִּ֖יר אֶת־ פָּנֶ֣יךָ מִמֶּֽנִּי׃ (יג) עַד־ אָ֨נָה אָשִׁ֪ית עֵצ֡וֹת בְּנַפְשִׁ֗י יָג֣וֹן בִּלְבָבִ֣י יוֹמָ֑ם עַד־ אָ֓נָה ׀ יָר֖וּם אֹיְבִ֣י עָלָֽי׃ (יד) הַבִּ֣יטָֽה עֲ֭נֵנִי יְהוָ֣ה אֱלֹהָ֑י הָאִ֥ירָה עֵ֝ינַ֗י פֶּן־ אִישַׁ֥ן הַמָּֽוֶת׃ (טו) פֶּן־ יֹאמַ֣ר אֹיְבִ֣י יְכָלְתִּ֑יו צָרַ֥י יָ֝גִ֗ילוּ כִּ֣י אֶמּֽוֹט׃ (טז) וַאֲנִ֤י ׀ בְּחַסְדְּךָ֣ בָטַחְתִּי֮ יָ֤גֵ֥ל לִבִּ֗י בִּֽישׁוּעָ֫תֶ֥ךָ אָשִׁ֥ירָה לַיהוָ֑ה כִּ֖י גָמַ֣ל עָלָֽי׃ (יז) לַמְנַצֵּ֗חַ לְדָ֫וִ֥ד אָ֘מַ֤ר נָבָ֣ל בְּ֭לִבּוֹ אֵ֣ין אֱלֹהִ֑ים הִֽשְׁחִ֗יתוּ הִֽתְעִ֥יבוּ עֲלִילָ֗ה אֵ֣ין עֹֽשֵׂה־ טֽוֹב׃ (יח) יְֽהוָ֗ה מִשָּׁמַיִם֮ הִשְׁקִ֪יף עַֽל־ בְּנֵי־ אָ֫דָ֥ם לִ֭רְאוֹת הֲיֵ֣שׁ מַשְׂכִּ֑יל דֹּ֝רֵשׁ אֶת־ אֱלֹהִֽים׃ (יט) הַכֹּ֥ל סָר֮ יַחְדָּ֪ו נֶ֫אֱלָ֥חוּ אֵ֤ין עֹֽשֵׂה־ ט֑וֹב אֵ֝֗ין גַּם־ אֶחָֽד׃ (כ) הֲלֹ֥א יָדְעוּ֮ כָּל־ פֹּ֪עֲלֵ֫י אָ֥וֶן אֹכְלֵ֣י עַ֭מִּי אָ֣כְלוּ לֶ֑חֶם יְ֝הוָ֗ה לֹ֣א קָרָֽאוּ׃ (כא) שָׁ֤ם ׀ פָּ֣חֲדוּ פָ֑חַד כִּֽי־ אֱ֝לֹהִ֗ים בְּד֣וֹר צַדִּֽיק׃ (כב) עֲצַת־ עָנִ֥י תָבִ֑ישׁוּ כִּ֖י יְהוָ֣ה מַחְסֵֽהוּ׃ (כג) מִ֥י יִתֵּ֣ן מִצִּיּוֹן֮ יְשׁוּעַ֪ת יִשְׂרָ֫אֵ֥ל בְּשׁ֣וּב יְ֭הוָה שְׁב֣וּת עַמּ֑וֹ יָגֵ֥ל יַ֝עֲקֹ֗ב יִשְׂמַ֥ח יִשְׂרָֽאֵל׃ (כד) מִזְמ֗וֹר לְדָ֫וִ֥ד יְ֭הֹוָה מִי־ יָג֣וּר בְּאָהֳלֶ֑ךָ מִֽי־ יִ֝שְׁכֹּ֗ן בְּהַ֣ר קָדְשֶֽׁךָ׃ (כה) הוֹלֵ֣ךְ תָּ֭מִים וּפֹעֵ֥ל צֶ֑דֶק וְדֹבֵ֥ר אֱ֝מֶ֗ת בִּלְבָבֽוֹ׃ (כו) לֹֽא־ רָגַ֨ל ׀ עַל־ לְשֹׁנ֗וֹ לֹא־ עָשָׂ֣ה לְרֵעֵ֣הוּ רָעָ֑ה וְ֝חֶרְפָּ֗ה לֹא־ נָשָׂ֥א עַל־ קְרֹֽבוֹ׃ (כז) נִבְזֶ֤ה ׀ בְּֽעֵ֘ינָ֤יו נִמְאָ֗ס וְאֶת־ יִרְאֵ֣י יְהוָ֣ה יְכַבֵּ֑ד נִשְׁבַּ֥ע לְ֝הָרַ֗ע וְלֹ֣א יָמִֽר׃ (כח) כַּסְפּ֤וֹ ׀ לֹא־ נָתַ֣ן בְּנֶשֶׁךְ֮ וְשֹׁ֥חַד עַל־ נָקִ֗י לֹ֥א לָ֫קָ֥ח עֹֽשֵׂה־ אֵ֑לֶּה לֹ֖א יִמּ֣וֹט לְעוֹלָֽם׃ (כט) מִכְתָּ֥ם לְדָוִ֑ד שָֽׁמְרֵ֥נִי אֵ֝֗ל כִּֽי־ חָסִ֥יתִי בָֽךְ׃ (ל) אָמַ֣רְתְּ לַֽ֭יהוָה אֲדֹנָ֣י אָ֑תָּה ט֝וֹבָתִ֗י בַּל־ עָלֶֽיךָ׃ (לא) לִ֭קְדוֹשִׁים אֲשֶׁר־ בָּאָ֣רֶץ הֵ֑מָּה וְ֝אַדִּירֵ֗י כָּל־ חֶפְצִי־ בָֽם׃ (לב) יִרְבּ֥וּ עַצְּבוֹתָם֮ אַחֵ֪ר מָ֫הָ֥רוּ בַּל־ אַסִּ֣יךְ נִסְכֵּיהֶ֣ם מִדָּ֑ם וּֽבַל־ אֶשָּׂ֥א אֶת־ שְׁ֝מוֹתָ֗ם עַל־ שְׂפָתָֽי׃ (לג) יְֽהוָ֗ה מְנָת־ חֶלְקִ֥י וְכוֹסִ֑י אַ֝תָּ֗ה תּוֹמִ֥יךְ גּוֹרָלִֽי׃ (לד) חֲבָלִ֣ים נָֽפְלוּ־ לִ֭י בַּנְּעִמִ֑ים אַף־ נַ֝חֲלָ֗ת שָֽׁפְרָ֥ה עָלָֽי׃ (לה) אֲבָרֵ֗ךְ אֶת־ יְ֭הוָה אֲשֶׁ֣ר יְעָצָ֑נִי אַף־ לֵ֝יל֗וֹת יִסְּר֥וּנִי כִלְיוֹתָֽי׃ (לו) שִׁוִּ֬יתִי יְהוָ֣ה לְנֶגְדִּ֣י תָמִ֑יד כִּ֥י מִֽ֝ימִינִ֗י בַּל־ אֶמּֽוֹט׃ (לז) לָכֵ֤ן ׀ שָׂמַ֣ח לִ֭בִּי וַיָּ֣גֶל כְּבוֹדִ֑י אַף־ בְּ֝שָׂרִ֗י יִשְׁכֹּ֥ן לָבֶֽטַח׃ (לח) כִּ֤י ׀ לֹא־ תַעֲזֹ֣ב נַפְשִׁ֣י לִשְׁא֑וֹל לֹֽא־ תִתֵּ֥ן חֲ֝סִידְךָ֗ לִרְא֥וֹת שָֽׁחַת׃ (לט) תּֽוֹדִיעֵנִי֮ אֹ֤רַח חַ֫יִּ֥ים שֹׂ֣בַע שְׂ֭מָחוֹת אֶת־ פָּנֶ֑יךָ נְעִמ֖וֹת בִּימִינְךָ֣ נֶֽצַח׃ (מ) תְּפִלָּ֗ה לְדָ֫וִ֥ד שִׁמְעָ֤ה יְהוָ֨ה ׀ צֶ֗דֶק הַקְשִׁ֥יבָה רִנָּתִ֗י הַאֲזִ֥ינָה תְפִלָּתִ֑י בְּ֝לֹ֗א שִׂפְתֵ֥י מִרְמָֽה׃ (מא) מִ֭לְּפָנֶיךָ מִשְׁפָּטִ֣י יֵצֵ֑א עֵ֝ינֶ֗יךָ תֶּחֱזֶ֥ינָה מֵישָׁרִֽים׃ (מב) בָּ֘חַ֤נְתָּ לִבִּ֨י ׀ פָּ֘קַ֤דְתָּ לַּ֗יְלָה צְרַפְתַּ֥נִי בַל־ תִּמְצָ֑א זַ֝מֹּתִ֗י בַּל־ יַעֲבָר־ פִּֽי׃ (מג) לִפְעֻלּ֣וֹת אָ֭דָם בִּדְבַ֣ר שְׂפָתֶ֑יךָ אֲנִ֥י שָׁ֝מַ֗רְתִּי אָרְח֥וֹת פָּרִֽיץ׃ (מד) תָּמֹ֣ךְ אֲ֭שֻׁרַי בְּמַעְגְּלוֹתֶ֑יךָ בַּל־ נָמ֥וֹטּוּ פְעָמָֽי׃ (מה) אֲנִֽי־ קְרָאתִ֣יךָ כִֽי־ תַעֲנֵ֣נִי אֵ֑ל הַֽט־ אָזְנְךָ֥ לִ֝֗י שְׁמַ֣ע אִמְרָתִֽי׃ (מו) הַפְלֵ֣ה חֲ֭סָדֶיךָ מוֹשִׁ֣יעַ חוֹסִ֑ים מִ֝מִּתְקוֹמְמִ֗ים בִּֽימִינֶֽךָ׃ (מז) שָׁ֭מְרֵנִי כְּאִישׁ֣וֹן בַּת־ עָ֑יִן בְּצֵ֥ל כְּ֝נָפֶ֗יךָ תַּסְתִּירֵֽנִי׃ (מח) מִפְּנֵ֣י רְ֭שָׁעִים ז֣וּ שַׁדּ֑וּנִי אֹיְבַ֥י בְּ֝נֶ֗פֶשׁ יַקִּ֥יפוּ עָלָֽי׃ (מט) חֶלְבָּ֥מוֹ סָּגְר֑וּ פִּ֝֗ימוֹ דִּבְּר֥וּ בְגֵאֽוּת׃ (נ) אַ֭שֻּׁרֵינוּ עַתָּ֣ה (סבבוני) סְבָב֑וּנוּ עֵינֵיהֶ֥ם יָ֝שִׁ֗יתוּ לִנְט֥וֹת בָּאָֽרֶץ׃ (נא) דִּמְיֹנ֗וֹ כְּ֭אַרְיֵה יִכְס֣וֹף לִטְר֑וֹף וְ֝כִכְפִ֗יר יֹשֵׁ֥ב בְּמִסְתָּרִֽים׃ (נב) קוּמָ֤ה יְהוָ֗ה קַדְּמָ֣ה פָ֭נָיו הַכְרִיעֵ֑הוּ פַּלְּטָ֥ה נַ֝פְשִׁ֗י מֵרָשָׁ֥ע חַרְבֶּֽךָ׃ (נג) מִֽמְתִ֥ים יָדְךָ֨ ׀ יְהוָ֡ה מִֽמְתִ֬ים מֵחֶ֗לֶד חֶלְקָ֥ם בַּֽחַיִּים֮ (וצפינך) וּֽצְפוּנְךָ֮ תְּמַלֵּ֪א בִ֫טְנָ֥ם יִשְׂבְּע֥וּ בָנִ֑ים וְהִנִּ֥יחוּ יִ֝תְרָ֗ם לְעוֹלְלֵיהֶֽם׃ (נד) אֲנִ֗י בְּ֭צֶדֶק אֶחֱזֶ֣ה פָנֶ֑יךָ אֶשְׂבְּעָ֥ה בְ֝הָקִ֗יץ תְּמוּנָתֶֽךָ׃ (נה) לַמְנַצֵּ֤חַ ׀ לְעֶ֥בֶד יְהוָ֗ה לְדָ֫וִ֥ד אֲשֶׁ֤ר דִּבֶּ֨ר ׀ לַיהוָ֗ה אֶת־ דִּ֭בְרֵי הַשִּׁירָ֣ה הַזֹּ֑את בְּי֤וֹם הִֽצִּיל־ יְהוָ֘ה אוֹת֥וֹ מִכַּ֥ף כָּל־ אֹ֝יְבָ֗יו וּמִיַּ֥ד לַֽ֭יהוָה בְּנֵ֣י אֵלִ֑ים הָב֥וּ לַ֝יהוָ֗ה כָּב֥וֹד וָעֹֽז׃ (קכב) הָב֣וּ לַֽ֭יהוָה כְּב֣וֹד שְׁמ֑וֹ הִשְׁתַּחֲו֥וּ לַ֝יהוָ֗ה בְּהַדְרַת־ קֹֽדֶשׁ׃ (קכג) ק֥וֹל יְהוָ֗ה עַל־ הַ֫מָּ֥יִם אֵֽל־ הַכָּב֥וֹד הִרְעִ֑ים יְ֝הוָ֗ה עַל־ מַ֥יִם רַבִּֽים׃ (קכד) קוֹל־ יְהוָ֥ה בַּכֹּ֑חַ ק֥וֹל יְ֝הוָ֗ה בֶּהָדָֽר׃ (קכה) ק֣וֹל יְ֭הוָה שֹׁבֵ֣ר אֲרָזִ֑ים וַיְשַׁבֵּ֥ר יְ֝הוָ֗ה אֶת־ אַרְזֵ֥י הַלְּבָנֽוֹן׃ (קכו) וַיַּרְקִידֵ֥ם כְּמוֹ־ עֵ֑גֶל לְבָנ֥וֹן וְ֝שִׂרְיֹ֗ן כְּמ֣וֹ בֶן־ רְאֵמִֽים׃ (קכז) קוֹל־ יְהוָ֥ה חֹצֵ֗ב לַהֲב֥וֹת אֵֽשׁ׃ (קכח) ק֣וֹל יְ֭הוָה יָחִ֣יל מִדְבָּ֑ר יָחִ֥יל יְ֝הוָ֗ה מִדְבַּ֥ר קָדֵֽשׁ׃ (קכט) ק֤וֹל יְהוָ֨ה ׀ יְחוֹלֵ֣ל אַיָּלוֹת֮ וַֽיֶּחֱשֹׂ֪ף יְעָ֫ר֥וֹת וּבְהֵיכָל֑וֹ כֻּ֝לּ֗וֹ אֹמֵ֥ר כָּבֽוֹד׃ (קל) יְ֭הוָה לַמַּבּ֣וּל יָשָׁ֑ב וַיֵּ֥שֶׁב יְ֝הוָ֗ה מֶ֣לֶךְ לְעוֹלָֽם׃
