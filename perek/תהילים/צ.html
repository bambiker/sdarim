פרק צ בספר תהילים (17 פסוקים) נמצא בתהילים סדר יב (פסוקים קיב–קכז) ובתהילים סדר יג (פסוקים א–א).

תהילים סדר יב

(א) יְהוָ֥ה צְבָא֑וֹת אַֽשְׁרֵ֥י אָ֝דָ֗ם בֹּטֵ֥חַ בָּֽךְ׃ (ב) לַמְנַצֵּ֬חַ ׀ לִבְנֵי־ קֹ֬רַח מִזְמֽוֹר׃ (ג) רָצִ֣יתָ יְהוָ֣ה אַרְצֶ֑ךָ שַׁ֝֗בְתָּ (שבות) שְׁבִ֣ית יַעֲקֹֽב׃ (ד) נָ֭שָׂאתָ עֲוֺ֣ן עַמֶּ֑ךָ כִּסִּ֖יתָ כָל־ חַטָּאתָ֣ם סֶֽלָה׃ (ה) אָסַ֥פְתָּ כָל־ עֶבְרָתֶ֑ךָ הֱ֝שִׁיב֗וֹתָ מֵחֲר֥וֹן אַפֶּֽךָ׃ (ו) שׁ֭וּבֵנוּ אֱלֹהֵ֣י יִשְׁעֵ֑נוּ וְהָפֵ֖ר כַּֽעַסְךָ֣ עִמָּֽנוּ׃ (ז) הַלְעוֹלָ֥ם תֶּֽאֱנַף־ בָּ֑נוּ תִּמְשֹׁ֥ךְ אַ֝פְּךָ֗ לְדֹ֣ר וָדֹֽר׃ (ח) הֲֽלֹא־ אַ֭תָּה תָּשׁ֣וּב תְּחַיֵּ֑נוּ וְ֝עַמְּךָ֗ יִשְׂמְחוּ־ בָֽךְ׃ (ט) הַרְאֵ֣נוּ יְהוָ֣ה חַסְדֶּ֑ךָ וְ֝יֶשְׁעֲךָ֗ תִּתֶּן־ לָֽנוּ׃ (י) אֶשְׁמְעָ֗ה מַה־ יְדַבֵּר֮ הָאֵ֪ל ׀ יְה֫וָ֥ה כִּ֤י ׀ יְדַבֵּ֬ר שָׁל֗וֹם אֶל־ עַמּ֥וֹ וְאֶל־ חֲסִידָ֑יו וְֽאַל־ יָשׁ֥וּבוּ לְכִסְלָֽה׃ (יא) אַ֤ךְ ׀ קָר֣וֹב לִירֵאָ֣יו יִשְׁע֑וֹ לִשְׁכֹּ֖ן כָּב֣וֹד בְּאַרְצֵֽנוּ׃ (יב) חֶֽסֶד־ וֶאֱמֶ֥ת נִפְגָּ֑שׁוּ צֶ֖דֶק וְשָׁל֣וֹם נָשָֽׁקוּ׃ (יג) אֱ֭מֶת מֵאֶ֣רֶץ תִּצְמָ֑ח וְ֝צֶ֗דֶק מִשָּׁמַ֥יִם נִשְׁקָֽף׃ (יד) גַּם־ יְ֭הוָה יִתֵּ֣ן הַטּ֑וֹב וְ֝אַרְצֵ֗נוּ תִּתֵּ֥ן יְבוּלָֽהּ׃ (טו) צֶ֭דֶק לְפָנָ֣יו יְהַלֵּ֑ךְ וְיָשֵׂ֖ם לְדֶ֣רֶךְ פְּעָמָֽיו׃ (טז) תְּפִלָּ֗ה לְדָ֫וִ֥ד הַטֵּֽה־ יְהוָ֣ה אָזְנְךָ֣ עֲנֵ֑נִי כִּֽי־ עָנִ֖י וְאֶבְי֣וֹן אָֽנִי׃ (יז) שָֽׁמְרָ֣ה נַפְשִׁי֮ כִּֽי־ חָסִ֪יד אָ֥נִי הוֹשַׁ֣ע עַ֭בְדְּךָ אַתָּ֣ה אֱלֹהַ֑י הַבּוֹטֵ֥חַ אֵלֶֽיךָ׃ (יח) חָנֵּ֥נִי אֲדֹנָ֑י כִּ֥י אֵלֶ֥יךָ אֶ֝קְרָ֗א כָּל־ הַיּֽוֹם׃ (יט) שַׂ֭מֵּחַ נֶ֣פֶשׁ עַבְדֶּ֑ךָ כִּ֥י אֵלֶ֥יךָ אֲ֝דֹנָ֗י נַפְשִׁ֥י אֶשָּֽׂא׃ (כ) כִּֽי־ אַתָּ֣ה אֲ֭דֹנָי ט֣וֹב וְסַלָּ֑ח וְרַב־ חֶ֝֗סֶד לְכָל־ קֹרְאֶֽיךָ׃ (כא) הַאֲזִ֣ינָה יְ֭הוָה תְּפִלָּתִ֑י וְ֝הַקְשִׁ֗יבָה בְּק֣וֹל תַּחֲנוּנוֹתָֽי׃ (כב) בְּי֣וֹם צָ֭רָתִ֥י אֶקְרָאֶ֗ךָּ כִּ֣י תַעֲנֵֽנִי׃ (כג) אֵין־ כָּמ֖וֹךָ בָאֱלֹהִ֥ים ׀ אֲדֹנָ֗י וְאֵ֣ין כְּֽמַעֲשֶֽׂיךָ׃ (כד) כָּל־ גּוֹיִ֤ם ׀ אֲשֶׁ֥ר עָשִׂ֗יתָ יָב֤וֹאוּ ׀ וְיִשְׁתַּחֲו֣וּ לְפָנֶ֣יךָ אֲדֹנָ֑י וִֽיכַבְּד֣וּ לִשְׁמֶֽךָ׃ (כה) כִּֽי־ גָד֣וֹל אַ֭תָּה וְעֹשֵׂ֣ה נִפְלָא֑וֹת אַתָּ֖ה אֱלֹהִ֣ים לְבַדֶּֽךָ׃ (כו) ה֘וֹרֵ֤נִי יְהוָ֨ה ׀ דַּרְכֶּ֗ךָ אֲהַלֵּ֥ךְ בַּאֲמִתֶּ֑ךָ יַחֵ֥ד לְ֝בָבִ֗י לְיִרְאָ֥ה שְׁמֶֽךָ׃ (כז) אוֹדְךָ֤ ׀ אֲדֹנָ֣י אֱ֭לֹהַי בְּכָל־ לְבָבִ֑י וַאֲכַבְּדָ֖ה שִׁמְךָ֣ לְעוֹלָֽם׃ (כח) כִּֽי־ חַ֭סְדְּךָ גָּד֣וֹל עָלָ֑י וְהִצַּ֥לְתָּ נַ֝פְשִׁ֗י מִשְּׁא֥וֹל תַּחְתִּיָּֽה׃ (כט) אֱלֹהִ֤ים ׀ זֵ֘דִ֤ים קָֽמוּ־ עָלַ֗י וַעֲדַ֣ת עָ֭רִיצִים בִּקְשׁ֣וּ נַפְשִׁ֑י וְלֹ֖א שָׂמ֣וּךָ לְנֶגְדָּֽם׃ (ל) וְאַתָּ֣ה אֲ֭דֹנָי אֵל־ רַח֣וּם וְחַנּ֑וּן אֶ֥רֶךְ אַ֝פַּ֗יִם וְרַב־ חֶ֥סֶד וֶאֱמֶֽת׃ (לא) פְּנֵ֥ה אֵלַ֗י וְחָ֫נֵּ֥נִי תְּנָֽה־ עֻזְּךָ֥ לְעַבְדֶּ֑ךָ וְ֝הוֹשִׁ֗יעָה לְבֶן־ אֲמָתֶֽךָ׃ (לב) עֲשֵֽׂה־ עִמִּ֥י א֗וֹת לְט֫וֹבָ֥ה וְיִרְא֣וּ שֹׂנְאַ֣י וְיֵבֹ֑שׁוּ כִּֽי־ אַתָּ֥ה יְ֝הוָ֗ה עֲזַרְתַּ֥נִי וְנִחַמְתָּֽנִי׃ (לג) לִבְנֵי־ קֹ֭רַח מִזְמ֣וֹר שִׁ֑יר יְ֝סוּדָת֗וֹ בְּהַרְרֵי־ קֹֽדֶשׁ׃ (לד) אֹהֵ֣ב יְ֭הוָה שַׁעֲרֵ֣י צִיּ֑וֹן מִ֝כֹּ֗ל מִשְׁכְּנ֥וֹת יַעֲקֹֽב׃ (לה) נִ֭כְבָּדוֹת מְדֻבָּ֣ר בָּ֑ךְ עִ֖יר הָאֱלֹהִ֣ים סֶֽלָה׃ (לו) אַזְכִּ֤יר ׀ רַ֥הַב וּבָבֶ֗ל לְֽיֹ֫דְעָ֥י הִנֵּ֤ה פְלֶ֣שֶׁת וְצ֣וֹר עִם־ כּ֑וּשׁ זֶ֝֗ה יֻלַּד־ שָֽׁם׃ (לז) וּֽלֲצִיּ֨וֹן ׀ יֵאָמַ֗ר אִ֣ישׁ וְ֭אִישׁ יֻלַּד־ בָּ֑הּ וְה֖וּא יְכוֹנְנֶ֣הָ עֶלְיֽוֹן׃ (לח) יְֽהוָ֗ה יִ֭סְפֹּר בִּכְת֣וֹב עַמִּ֑ים זֶ֖ה יֻלַּד־ שָׁ֣ם סֶֽלָה׃ (לט) וְשָׁרִ֥ים כְּחֹלְלִ֑ים כָּֽל־ מַעְיָנַ֥י בָּֽךְ׃ (מ) שִׁ֥יר מִזְמ֗וֹר לִבְנֵ֫י קֹ֥רַח לַמְנַצֵּ֣חַ עַל־ מָחֲלַ֣ת לְעַנּ֑וֹת מַ֝שְׂכִּ֗יל לְהֵימָ֥ן הָאֶזְרָחִֽי׃ (מא) יְ֭הוָה אֱלֹהֵ֣י יְשׁוּעָתִ֑י יוֹם־ צָעַ֖קְתִּי בַלַּ֣יְלָה נֶגְדֶּֽךָ׃ (מב) תָּב֣וֹא לְ֭פָנֶיךָ תְּפִלָּתִ֑י הַטֵּֽה־ אָ֝זְנְךָ֗ לְרִנָּתִֽי׃ (מג) כִּֽי־ שָֽׂבְעָ֣ה בְרָע֣וֹת נַפְשִׁ֑י וְחַיַּ֗י לִשְׁא֥וֹל הִגִּֽיעוּ׃ (מד) נֶ֭חְשַׁבְתִּי עִם־ י֣וֹרְדֵי ב֑וֹר הָ֝יִ֗יתִי כְּגֶ֣בֶר אֵֽין־ אֱיָֽל׃ (מה) בַּמֵּתִ֗ים חָ֫פְשִׁ֥י כְּמ֤וֹ חֲלָלִ֨ים ׀ שֹׁ֥כְבֵי קֶ֗בֶר אֲשֶׁ֤ר לֹ֣א זְכַרְתָּ֣ם ע֑וֹד וְ֝הֵ֗מָּה מִיָּדְךָ֥ נִגְזָֽרוּ׃ (מו) שַׁ֭תַּנִי בְּב֣וֹר תַּחְתִּיּ֑וֹת בְּ֝מַחֲשַׁכִּ֗ים בִּמְצֹלֽוֹת׃ (מז) עָ֭לַי סָמְכָ֣ה חֲמָתֶ֑ךָ וְכָל־ מִ֝שְׁבָּרֶ֗יךָ עִנִּ֥יתָ סֶּֽלָה׃ (מח) הִרְחַ֥קְתָּ מְיֻדָּעַ֗י מִ֫מֶּ֥נִּי שַׁתַּ֣נִי תוֹעֵב֣וֹת לָ֑מוֹ כָּ֝לֻ֗א וְלֹ֣א אֵצֵֽא׃ (מט) עֵינִ֥י דָאֲבָ֗ה מִנִּ֫י עֹ֥נִי קְרָאתִ֣יךָ יְהוָ֣ה בְּכָל־ י֑וֹם שִׁטַּ֖חְתִּי אֵלֶ֣יךָ כַפָּֽי׃ (נ) הֲלַמֵּתִ֥ים תַּעֲשֶׂה־ פֶּ֑לֶא אִם־ רְ֝פָאִ֗ים יָק֤וּמוּ ׀ יוֹד֬וּךָ סֶּֽלָה׃ (נא) הַיְסֻפַּ֣ר בַּקֶּ֣בֶר חַסְדֶּ֑ךָ אֱ֝מֽוּנָתְךָ֗ בָּאֲבַדּֽוֹן׃ (נב) הֲיִוָּדַ֣ע בַּחֹ֣שֶׁךְ פִּלְאֶ֑ךָ וְ֝צִדְקָתְךָ֗ בְּאֶ֣רֶץ נְשִׁיָּֽה׃ (נג) וַאֲנִ֤י ׀ אֵלֶ֣יךָ יְהוָ֣ה שִׁוַּ֑עְתִּי וּ֝בַבֹּ֗קֶר תְּֽפִלָּתִ֥י תְקַדְּמֶֽךָּ׃ (נד) לָמָ֣ה יְ֭הוָה תִּזְנַ֣ח נַפְשִׁ֑י תַּסְתִּ֖יר פָּנֶ֣יךָ מִמֶּֽנִּי׃ (נה) עָ֘נִ֤י אֲנִ֣י וְגֹוֵ֣עַ מִנֹּ֑עַר נָשָׂ֖אתִי אֵמֶ֣יךָ אָפֽוּנָה׃ (נו) עָ֭לַי עָבְר֣וּ חֲרוֹנֶ֑יךָ בִּ֝עוּתֶ֗יךָ צִמְּתוּתֻֽנִי׃ (נז) סַבּ֣וּנִי כַ֭מַּיִם כָּל־ הַיּ֑וֹם הִקִּ֖יפוּ עָלַ֣י יָֽחַד׃ (נח) הִרְחַ֣קְתָּ מִ֭מֶּנִּי אֹהֵ֣ב וָרֵ֑עַ מְֽיֻדָּעַ֥י מַחְשָֽׁךְ׃ (נט) מַ֝שְׂכִּ֗יל לְאֵיתָ֥ן הָֽאֶזְרָחִֽי׃ (ס) חַֽסְדֵ֣י יְ֭הוָה עוֹלָ֣ם אָשִׁ֑ירָה לְדֹ֥ר וָדֹ֓ר ׀ אוֹדִ֖יעַ אֱמוּנָתְךָ֣ בְּפִֽי׃ (סא) כִּֽי־ אָמַ֗רְתִּי ע֭וֹלָם חֶ֣סֶד יִבָּנֶ֑ה שָׁמַ֓יִם ׀ תָּכִ֖ן אֱמוּנָתְךָ֣ בָהֶֽם׃ (סב) כָּרַ֣תִּֽי בְ֭רִית לִבְחִירִ֑י נִ֝שְׁבַּ֗עְתִּי לְדָוִ֥ד עַבְדִּֽי׃ (סג) עַד־ ע֭וֹלָם אָכִ֣ין זַרְעֶ֑ךָ וּבָנִ֨יתִי לְדֹר־ וָד֖וֹר כִּסְאֲךָ֣ סֶֽלָה׃ (סד) וְי֘וֹד֤וּ שָׁמַ֣יִם פִּלְאֲךָ֣ לַ֭יהוָה כְּב֣וֹד שְׁמ֑וֹ שְׂאֽוּ־ מִ֝נְחָ֗ה וּבֹ֥אוּ לְחַצְרוֹתָֽיו׃ (פא) הִשְׁתַּחֲו֣וּ לַ֭יהוָה בְּהַדְרַת־ קֹ֑דֶשׁ חִ֥ילוּ מִ֝פָּנָ֗יו כָּל־ הָאָֽרֶץ׃ (פב) אִמְר֤וּ בַגּוֹיִ֨ם ׀ יְה֘וָ֤ה מָלָ֗ךְ אַף־ תִּכּ֣וֹן תֵּ֭בֵל בַּל־ תִּמּ֑וֹט יָדִ֥ין עַ֝מִּ֗ים בְּמֵישָׁרִֽים׃ (פג) יִשְׂמְח֣וּ הַ֭שָּׁמַיִם וְתָגֵ֣ל הָאָ֑רֶץ יִֽרְעַ֥ם הַ֝יָּ֗ם וּמְלֹאֽוֹ׃ (פד) יַעֲלֹ֣ז שָׂ֭דַי וְכָל־ אֲשֶׁר־ בּ֑וֹ אָ֥ז יְ֝רַנְּנ֗וּ כָּל־ עֲצֵי־ יָֽעַר׃ (פה) לִפְנֵ֤י יְהוָ֨ה ׀ כִּ֬י בָ֗א כִּ֥י בָא֮ לִשְׁפֹּ֪ט הָ֫אָ֥רֶץ יִשְׁפֹּֽט־ תֵּבֵ֥ל בְּצֶ֑דֶק וְ֝עַמִּ֗ים בֶּאֱמוּנָתֽוֹ׃ (פו) יְהוָ֣ה מָ֭לָךְ תָּגֵ֣ל הָאָ֑רֶץ יִ֝שְׂמְח֗וּ אִיִּ֥ים רַבִּֽים׃ (פז) עָנָ֣ן וַעֲרָפֶ֣ל סְבִיבָ֑יו צֶ֥דֶק וּ֝מִשְׁפָּ֗ט מְכ֣וֹן כִּסְאֽוֹ׃ (פח) אֵ֭שׁ לְפָנָ֣יו תֵּלֵ֑ךְ וּתְלַהֵ֖ט סָבִ֣יב צָרָֽיו׃ (פט) הֵאִ֣ירוּ בְרָקָ֣יו תֵּבֵ֑ל רָאֲתָ֖ה וַתָּחֵ֣ל הָאָֽרֶץ׃ (צ) הָרִ֗ים כַּדּוֹנַ֗ג נָ֭מַסּוּ מִלִּפְנֵ֣י יְהוָ֑ה מִ֝לִּפְנֵ֗י אֲד֣וֹן כָּל־ הָאָֽרֶץ׃ (צא) הִגִּ֣ידוּ הַשָּׁמַ֣יִם צִדְק֑וֹ וְרָא֖וּ כָל־ הָעַמִּ֣ים כְּבוֹדֽוֹ׃ (צב) יֵבֹ֤שׁוּ ׀ כָּל־ עֹ֬בְדֵי פֶ֗סֶל הַמִּֽתְהַלְלִ֥ים בָּאֱלִילִ֑ים הִשְׁתַּחֲווּ־ ל֝וֹ כָּל־ אֱלֹהִֽים׃ (צג) שָׁמְעָ֬ה וַתִּשְׂמַ֨ח ׀ צִיּ֗וֹן וַ֭תָּגֵלְנָה בְּנ֣וֹת יְהוּדָ֑ה לְמַ֖עַן מִשְׁפָּטֶ֣יךָ יְהוָֽה׃ (צד) כִּֽי־ אַתָּ֤ה יְהוָ֗ה עֶלְי֥וֹן עַל־ כָּל־ הָאָ֑רֶץ מְאֹ֥ד נַ֝עֲלֵ֗יתָ עַל־ כָּל־ אֱלֹהִֽים׃ (צה) אֹהֲבֵ֥י יְהוָ֗ה שִׂנְא֫וּ רָ֥ע שֹׁ֭מֵר נַפְשׁ֣וֹת חֲסִידָ֑יו מִיַּ֥ד רְ֝שָׁעִ֗ים יַצִּילֵֽם׃ (צו) א֭וֹר זָרֻ֣עַ לַצַּדִּ֑יק וּֽלְיִשְׁרֵי־ לֵ֥ב שִׂמְחָֽה׃ (צז) שִׂמְח֣וּ צַ֭דִּיקִים בַּֽיהוָ֑ה וְ֝הוֹד֗וּ לְזֵ֣כֶר קָדְשֽׁוֹ׃ (צח) מִזְמ֡וֹר שִׁ֤ירוּ לַֽיהוָ֨ה ׀ שִׁ֣יר חָ֭דָשׁ כִּֽי־ נִפְלָא֣וֹת עָשָׂ֑ה הוֹשִֽׁיעָה־ לּ֥וֹ יְ֝מִינ֗וֹ וּזְר֥וֹעַ קָדְשֽׁוֹ׃ (צט) הוֹדִ֣יעַ יְ֭הוָה יְשׁוּעָת֑וֹ לְעֵינֵ֥י הַ֝גּוֹיִ֗ם גִּלָּ֥ה צִדְקָתֽוֹ׃ (ק) זָ֘כַ֤ר חַסְדּ֨וֹ ׀ וֶֽאֱֽמוּנָתוֹ֮ לְבֵ֪ית יִשְׂרָ֫אֵ֥ל רָא֥וּ כָל־ אַפְסֵי־ אָ֑רֶץ אֵ֝֗ת יְשׁוּעַ֥ת אֱלֹהֵֽינוּ׃ (קא) הָרִ֣יעוּ לַֽ֭יהוָה כָּל־ הָאָ֑רֶץ פִּצְח֖וּ וְרַנְּנ֣וּ וְזַמֵּֽרוּ׃ (קב) זַמְּר֣וּ לַיהוָ֣ה בְּכִנּ֑וֹר בְּ֝כִנּ֗וֹר וְק֣וֹל זִמְרָֽה׃ (קג) בַּ֭חֲצֹ֣צְרוֹת וְק֣וֹל שׁוֹפָ֑ר הָ֝רִ֗יעוּ לִפְנֵ֤י ׀ הַמֶּ֬לֶךְ יְהוָֽה׃ (קד) יִרְעַ֣ם הַ֭יָּם וּמְלֹא֑וֹ תֵּ֝בֵ֗ל וְיֹ֣שְׁבֵי בָֽהּ׃ (קה) נְהָר֥וֹת יִמְחֲאוּ־ כָ֑ף יַ֝֗חַד הָרִ֥ים יְרַנֵּֽנוּ׃ (קו) לִֽפְֽנֵי־ יְהוָ֗ה כִּ֥י בָא֮ לִשְׁפֹּ֪ט הָ֫אָ֥רֶץ יִשְׁפֹּֽט־ תֵּבֵ֥ל בְּצֶ֑דֶק וְ֝עַמִּ֗ים בְּמֵישָׁרִֽים׃ (קז) יְהוָ֣ה מָ֭לָךְ יִרְגְּז֣וּ עַמִּ֑ים יֹשֵׁ֥ב כְּ֝רוּבִ֗ים תָּנ֥וּט הָאָֽרֶץ׃ (קח) יְ֭הוָה בְּצִיּ֣וֹן גָּד֑וֹל וְרָ֥ם ה֝֗וּא עַל־ כָּל־ הָֽעַמִּֽים׃ (קט) יוֹד֣וּ שִׁ֭מְךָ גָּד֥וֹל וְנוֹרָ֗א קָד֥וֹשׁ הֽוּא׃ (קי) וְעֹ֥ז מֶלֶךְ֮ מִשְׁפָּ֪ט אָ֫הֵ֥ב אַ֭תָּה כּוֹנַ֣נְתָּ מֵישָׁרִ֑ים מִשְׁפָּ֥ט וּ֝צְדָקָ֗ה בְּיַעֲקֹ֤ב ׀ אַתָּ֬ה עָשִֽׂיתָ׃ (קיא) רֽוֹמְמ֡וּ יְה֘וָ֤ה אֱלֹהֵ֗ינוּ וְֽ֭הִשְׁתַּחֲווּ לַהֲדֹ֥ם רַגְלָ֗יו קָד֥וֹשׁ הֽוּא׃ (קיב) מֹ֘שֶׁ֤ה וְאַהֲרֹ֨ן ׀ בְּֽכֹהֲנָ֗יו וּ֭שְׁמוּאֵל בְּקֹרְאֵ֣י שְׁמ֑וֹ קֹרִ֥אים אֶל־ יְ֝הוָ֗ה וְה֣וּא יַעֲנֵֽם׃ (קיג) בְּעַמּ֣וּד עָ֭נָן יְדַבֵּ֣ר אֲלֵיהֶ֑ם שָׁמְר֥וּ עֵ֝דֹתָ֗יו וְחֹ֣ק נָֽתַן־ לָֽמוֹ׃ (קיד) יְהוָ֣ה אֱלֹהֵינוּ֮ אַתָּ֪ה עֲנִ֫יתָ֥ם אֵ֣ל נֹ֭שֵׂא הָיִ֣יתָ לָהֶ֑ם וְ֝נֹקֵ֗ם עַל־ עֲלִילוֹתָֽם׃ (קטו) רֽוֹמְמ֡וּ יְה֘וָ֤ה אֱלֹהֵ֗ינוּ וְ֭הִֽשְׁתַּחֲווּ לְהַ֣ר קָדְשׁ֑וֹ כִּֽי־ קָ֝ד֗וֹשׁ יְהוָ֥ה אֱלֹהֵֽינוּ׃ (קטז) מִזְמ֥וֹר לְתוֹדָ֑ה הָרִ֥יעוּ לַ֝יהוָ֗ה כָּל־ הָאָֽרֶץ׃ (קיז) עִבְד֣וּ אֶת־ יְהוָ֣ה בְּשִׂמְחָ֑ה בֹּ֥אוּ לְ֝פָנָ֗יו בִּרְנָנָֽה׃ (קיח) דְּע֗וּ כִּֽי־ יְהוָה֮ ה֤וּא אֱלֹ֫הִ֥ים הֽוּא־ עָ֭שָׂנוּ (ולא) וְל֣וֹ אֲנַ֑חְנוּ עַ֝מּ֗וֹ וְצֹ֣אן מַרְעִיתֽוֹ׃ (קיט) בֹּ֤אוּ שְׁעָרָ֨יו ׀ בְּתוֹדָ֗ה חֲצֵרֹתָ֥יו בִּתְהִלָּ֑ה הֽוֹדוּ־ ל֝֗וֹ בָּרֲכ֥וּ שְׁמֽוֹ׃ (קכ) כִּי־ ט֣וֹב יְ֭הֹוָה לְעוֹלָ֣ם חַסְדּ֑וֹ וְעַד־ דֹּ֥ר וָ֝דֹ֗ר אֱמוּנָתֽוֹ׃
