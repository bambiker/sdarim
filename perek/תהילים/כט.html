תהילים פרק כט לפי החלוקה לסדרים

פרק כט בספר תהילים (11 פסוקים) נמצא בתהילים סדר ג (פסוקים קכא–קל) ובתהילים סדר ד (פסוקים א–א).

תהילים סדר ג

(א) יְהוָ֥ה הוֹשִׁ֑יעָה הַ֝מֶּ֗לֶךְ יַעֲנֵ֥נוּ בְיוֹם־ קָרְאֵֽנוּ׃ (ב) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (ג) יְֽהוָ֗ה בְּעָזְּךָ֥ יִשְׂמַח־ מֶ֑לֶךְ וּ֝בִישׁ֥וּעָתְךָ֗ מַה־ (יגיל) יָּ֥גֶל מְאֹֽד׃ (ד) תַּאֲוַ֣ת לִ֭בּוֹ נָתַ֣תָּה לּ֑וֹ וַאֲרֶ֥שֶׁת שְׂ֝פָתָ֗יו בַּל־ מָנַ֥עְתָּ סֶּֽלָה׃ (ה) כִּֽי־ תְ֭קַדְּמֶנּוּ בִּרְכ֣וֹת ט֑וֹב תָּשִׁ֥ית לְ֝רֹאשׁ֗וֹ עֲטֶ֣רֶת פָּֽז׃ (ו) חַיִּ֤ים ׀ שָׁאַ֣ל מִ֭מְּךָ נָתַ֣תָּה לּ֑וֹ אֹ֥רֶךְ יָ֝מִ֗ים עוֹלָ֥ם וָעֶֽד׃ (ז) גָּד֣וֹל כְּ֭בוֹדוֹ בִּישׁוּעָתֶ֑ךָ ה֥וֹד וְ֝הָדָר תְּשַׁוֶּ֥ה עָלָֽיו׃ (ח) כִּֽי־ תְשִׁיתֵ֣הוּ בְרָכ֣וֹת לָעַ֑ד תְּחַדֵּ֥הוּ בְ֝שִׂמְחָ֗ה אֶת־ פָּנֶֽיךָ׃ (ט) כִּֽי־ הַ֭מֶּלֶךְ בֹּטֵ֣חַ בַּיהוָ֑ה וּבְחֶ֥סֶד עֶ֝לְי֗וֹן בַּל־ יִמּֽוֹט׃ (י) תִּמְצָ֣א יָ֭דְךָ לְכָל־ אֹיְבֶ֑יךָ יְ֝מִֽינְךָ תִּמְצָ֥א שֹׂנְאֶֽיךָ׃ (יא) תְּשִׁיתֵ֤מוֹ ׀ כְּתַנּ֥וּר אֵשׁ֮ לְעֵ֪ת פָּ֫נֶ֥יךָ יְ֭הוָה בְּאַפּ֣וֹ יְבַלְּעֵ֑ם וְֽתֹאכְלֵ֥ם אֵֽשׁ׃ (יב) פִּ֭רְיָמוֹ מֵאֶ֣רֶץ תְּאַבֵּ֑ד וְ֝זַרְעָ֗ם מִבְּנֵ֥י אָדָֽם׃ (יג) כִּי־ נָט֣וּ עָלֶ֣יךָ רָעָ֑ה חָֽשְׁב֥וּ מְ֝זִמָּ֗ה בַּל־ יוּכָֽלוּ׃ (יד) כִּ֭י תְּשִׁיתֵ֣מוֹ שֶׁ֑כֶם בְּ֝מֵֽיתָרֶ֗יךָ תְּכוֹנֵ֥ן עַל־ פְּנֵיהֶֽם׃ (טו) ר֣וּמָה יְהוָ֣ה בְּעֻזֶּ֑ךָ נָשִׁ֥ירָה וּֽ֝נְזַמְּרָה גְּבוּרָתֶֽךָ׃ (טז) לַ֭מְנַצֵּחַ עַל־ אַיֶּ֥לֶת הַשַּׁ֗חַר מִזְמ֥וֹר לְדָוִֽד׃ (יז) אֵלִ֣י אֵ֭לִי לָמָ֣ה עֲזַבְתָּ֑נִי רָח֥וֹק מִֽ֝ישׁוּעָתִ֗י דִּבְרֵ֥י שַׁאֲגָתִֽי׃ (יח) אֱ‍ֽלֹהַ֗י אֶקְרָ֣א י֭וֹמָם וְלֹ֣א תַעֲנֶ֑ה וְ֝לַ֗יְלָה וְֽלֹא־ דֽוּמִיָּ֥ה לִֽי׃ (יט) וְאַתָּ֥ה קָד֑וֹשׁ י֝וֹשֵׁ֗ב תְּהִלּ֥וֹת יִשְׂרָאֵֽל׃ (כ) בְּ֭ךָ בָּטְח֣וּ אֲבֹתֵ֑ינוּ בָּ֝טְח֗וּ וַֽתְּפַלְּטֵֽמוֹ׃ (כא) אֵלֶ֣יךָ זָעֲק֣וּ וְנִמְלָ֑טוּ בְּךָ֖ בָטְח֣וּ וְלֹא־ בֽוֹשׁוּ׃ (כב) וְאָנֹכִ֣י תוֹלַ֣עַת וְלֹא־ אִ֑ישׁ חֶרְפַּ֥ת אָ֝דָ֗ם וּבְז֥וּי עָֽם׃ (כג) כָּל־ רֹ֭אַי יַלְעִ֣גוּ לִ֑י יַפְטִ֥ירוּ בְ֝שָׂפָ֗ה יָנִ֥יעוּ רֹֽאשׁ׃ (כד) גֹּ֣ל אֶל־ יְהוָ֣ה יְפַלְּטֵ֑הוּ יַ֝צִּילֵ֗הוּ כִּ֘י חָ֥פֵֽץ בּֽוֹ׃ (כה) כִּֽי־ אַתָּ֣ה גֹחִ֣י מִבָּ֑טֶן מַ֝בְטִיחִ֗י עַל־ שְׁדֵ֥י אִמִּֽי׃ (כו) עָ֭לֶיךָ הָשְׁלַ֣כְתִּי מֵרָ֑חֶם מִבֶּ֥טֶן אִ֝מִּ֗י אֵ֣לִי אָֽתָּה׃ (כז) אַל־ תִּרְחַ֣ק מִ֭מֶּנִּי כִּי־ צָרָ֣ה קְרוֹבָ֑ה כִּי־ אֵ֥ין עוֹזֵֽר׃ (כח) סְ֭בָבוּנִי פָּרִ֣ים רַבִּ֑ים אַבִּירֵ֖י בָשָׁ֣ן כִּתְּרֽוּנִי׃ (כט) פָּצ֣וּ עָלַ֣י פִּיהֶ֑ם אַ֝רְיֵ֗ה טֹרֵ֥ף וְשֹׁאֵֽג׃ (ל) כַּמַּ֥יִם נִשְׁפַּכְתִּי֮ וְהִתְפָּֽרְד֗וּ כָּֽל־ עַצְמ֫וֹתָ֥י הָיָ֣ה לִ֭בִּי כַּדּוֹנָ֑ג נָ֝מֵ֗ס בְּת֣וֹךְ מֵעָֽי׃ (לא) יָ֘בֵ֤שׁ כַּחֶ֨רֶשׂ ׀ כֹּחִ֗י וּ֭לְשׁוֹנִי מֻדְבָּ֣ק מַלְקוֹחָ֑י וְֽלַעֲפַר־ מָ֥וֶת תִּשְׁפְּתֵֽנִי׃ (לב) כִּ֥י סְבָב֗וּנִי כְּלָ֫בִ֥ים עֲדַ֣ת מְ֭רֵעִים הִקִּיפ֑וּנִי כָּ֝אֲרִ֗י יָדַ֥י וְרַגְלָֽי׃ (לג) אֲסַפֵּ֥ר כָּל־ עַצְמוֹתָ֑י הֵ֥מָּה יַ֝בִּ֗יטוּ יִרְאוּ־ בִֽי׃ (לד) יְחַלְּק֣וּ בְגָדַ֣י לָהֶ֑ם וְעַל־ לְ֝בוּשִׁ֗י יַפִּ֥ילוּ גוֹרָֽל׃ (לה) וְאַתָּ֣ה יְ֭הוָה אַל־ תִּרְחָ֑ק אֱ֝יָלוּתִ֗י לְעֶזְרָ֥תִי חֽוּשָׁה׃ (לו) הַצִּ֣ילָה מֵחֶ֣רֶב נַפְשִׁ֑י מִיַּד־ כֶּ֝֗לֶב יְחִידָתִֽי׃ (לז) ה֭וֹשִׁיעֵנִי מִפִּ֣י אַרְיֵ֑ה וּמִקַּרְנֵ֖י רֵמִ֣ים עֲנִיתָֽנִי׃ (לח) אֲסַפְּרָ֣ה שִׁמְךָ֣ לְאֶחָ֑י בְּת֖וֹךְ קָהָ֣ל אֲהַלְלֶֽךָּ׃ (לט) יִרְאֵ֤י יְהוָ֨ה ׀ הַֽלְל֗וּהוּ כָּל־ זֶ֣רַע יַעֲקֹ֣ב כַּבְּד֑וּהוּ וְג֥וּרוּ מִ֝מֶּ֗נּוּ כָּל־ זֶ֥רַע יִשְׂרָאֵֽל׃ (מ) כִּ֤י לֹֽא־ בָזָ֨ה וְלֹ֪א שִׁקַּ֡ץ עֱנ֬וּת עָנִ֗י וְלֹא־ הִסְתִּ֣יר פָּנָ֣יו מִמֶּ֑נּוּ וּֽבְשַׁוְּע֖וֹ אֵלָ֣יו שָׁמֵֽעַ׃ (מא) מֵ֥אִתְּךָ֗ תְֽהִלָּ֫תִ֥י בְּקָהָ֥ל רָ֑ב נְדָרַ֥י אֲ֝שַׁלֵּ֗ם נֶ֣גֶד יְרֵאָֽיו׃ (מב) יֹאכְל֬וּ עֲנָוִ֨ים ׀ וְיִשְׂבָּ֗עוּ יְהַֽלְל֣וּ יְ֭הוָה דֹּ֣רְשָׁ֑יו יְחִ֖י לְבַבְכֶ֣ם לָעַֽד׃ (מג) יִזְכְּר֤וּ ׀ וְיָשֻׁ֣בוּ אֶל־ יְ֭הוָה כָּל־ אַפְסֵי־ אָ֑רֶץ וְיִֽשְׁתַּחֲו֥וּ לְ֝פָנֶ֗יךָ כָּֽל־ מִשְׁפְּח֥וֹת גּוֹיִֽם׃ (מד) כִּ֣י לַ֭יהוָה הַמְּלוּכָ֑ה וּ֝מֹשֵׁ֗ל בַּגּוֹיִֽם׃ (מה) אָכְל֬וּ וַיִּֽשְׁתַּחֲוּ֨וּ ׀ כָּֽל־ דִּשְׁנֵי־ אֶ֗רֶץ לְפָנָ֣יו יִ֭כְרְעוּ כָּל־ יוֹרְדֵ֣י עָפָ֑ר וְ֝נַפְשׁ֗וֹ לֹ֣א חִיָּֽה׃ (מו) זֶ֥רַע יַֽעַבְדֶ֑נּוּ יְסֻפַּ֖ר לַֽאדֹנָ֣י לַדּֽוֹר׃ (מז) יָ֭בֹאוּ וְיַגִּ֣ידוּ צִדְקָת֑וֹ לְעַ֥ם נ֝וֹלָ֗ד כִּ֣י עָשָֽׂה׃ (מח) מִזְמ֥וֹר לְדָוִ֑ד יְהוָ֥ה רֹ֝עִ֗י לֹ֣א אֶחְסָֽר׃ (מט) בִּנְא֣וֹת דֶּ֭שֶׁא יַרְבִּיצֵ֑נִי עַל־ מֵ֖י מְנֻח֣וֹת יְנַהֲלֵֽנִי׃ (נ) נַפְשִׁ֥י יְשׁוֹבֵ֑ב יַֽנְחֵ֥נִי בְמַעְגְּלֵי־ צֶ֝֗דֶק לְמַ֣עַן שְׁמֽוֹ׃ (נא) גַּ֤ם כִּֽי־ אֵלֵ֨ךְ בְּגֵ֪יא צַלְמָ֡וֶת לֹא־ אִ֘ירָ֤א רָ֗ע כִּי־ אַתָּ֥ה עִמָּדִ֑י שִׁבְטְךָ֥ וּ֝מִשְׁעַנְתֶּ֗ךָ הֵ֣מָּה יְנַֽחֲמֻֽנִי׃ (נב) תַּעֲרֹ֬ךְ לְפָנַ֨י ׀ שֻׁלְחָ֗ן נֶ֥גֶד צֹרְרָ֑י דִּשַּׁ֖נְתָּ בַשֶּׁ֥מֶן רֹ֝אשִׁ֗י כּוֹסִ֥י רְוָיָֽה׃ (נג) אַ֤ךְ ׀ ט֤וֹב וָחֶ֣סֶד יִ֭רְדְּפוּנִי כָּל־ יְמֵ֣י חַיָּ֑י וְשַׁבְתִּ֥י בְּבֵית־ יְ֝הוָ֗ה לְאֹ֣רֶךְ יָמִֽים׃ (נד) לְדָוִ֗ד מִ֫זְמ֥וֹר לַֽ֭יהוָה הָאָ֣רֶץ וּמְלוֹאָ֑הּ תֵּ֝בֵ֗ל וְיֹ֣שְׁבֵי בָֽהּ׃ (נה) כִּי־ ה֭וּא עַל־ יַמִּ֣ים יְסָדָ֑הּ וְעַל־ נְ֝הָר֗וֹת יְכוֹנְנֶֽהָ׃ (נו) מִֽי־ יַעֲלֶ֥ה בְהַר־ יְהוָ֑ה וּמִי־ יָ֝קוּם בִּמְק֥וֹם קָדְשֽׁוֹ׃ (נז) נְקִ֥י כַפַּ֗יִם וּֽבַר־ לֵ֫בָ֥ב אֲשֶׁ֤ר ׀ לֹא־ נָשָׂ֣א לַשָּׁ֣וְא נַפְשִׁ֑י וְלֹ֖א נִשְׁבַּ֣ע לְמִרְמָֽה׃ (נח) יִשָּׂ֣א בְ֭רָכָה מֵאֵ֣ת יְהוָ֑ה וּ֝צְדָקָ֗ה מֵאֱלֹהֵ֥י יִשְׁעֽוֹ׃ (נט) זֶ֭ה דּ֣וֹר (דרשו) דֹּרְשָׁ֑יו מְבַקְשֵׁ֨י פָנֶ֖יךָ יַעֲקֹ֣ב סֶֽלָה׃ (ס) שְׂא֤וּ שְׁעָרִ֨ים ׀ רָֽאשֵׁיכֶ֗ם וְֽ֭הִנָּשְׂאוּ פִּתְחֵ֣י עוֹלָ֑ם וְ֝יָב֗וֹא מֶ֣לֶךְ הַכָּבֽוֹד׃ (סא) מִ֥י זֶה֮ מֶ֤לֶךְ הַכָּ֫ב֥וֹד יְ֭הוָה עִזּ֣וּז וְגִבּ֑וֹר יְ֝הוָ֗ה גִּבּ֥וֹר מִלְחָמָֽה׃ (סב) שְׂא֤וּ שְׁעָרִ֨ים ׀ רָֽאשֵׁיכֶ֗ם וּ֭שְׂאוּ פִּתְחֵ֣י עוֹלָ֑ם וְ֝יָבֹא מֶ֣לֶךְ הֽוּא־ צִ֝וָּ֗ה וַֽיַּעֲמֹֽד׃ (ס) יְֽהוָ֗ה הֵפִ֥יר עֲצַת־ גּוֹיִ֑ם הֵ֝נִ֗יא מַחְשְׁב֥וֹת עַמִּֽים׃ (סא) עֲצַ֣ת יְ֭הוָה לְעוֹלָ֣ם תַּעֲמֹ֑ד מַחְשְׁב֥וֹת לִ֝בּ֗וֹ לְדֹ֣ר וָדֹֽר׃ (סב) אַשְׁרֵ֣י הַ֭גּוֹי אֲשֶׁר־ יְהוָ֣ה אֱלֹהָ֑יו הָעָ֓ם ׀ בָּחַ֖ר לְנַחֲלָ֣ה לֽוֹ׃ (סג) מִ֭שָּׁמַיִם הִבִּ֣יט יְהוָ֑ה רָ֝אָ֗ה אֶֽת־ כָּל־ בְּנֵ֥י הָאָדָֽם׃ (סד) מִֽמְּכוֹן־ שִׁבְתּ֥וֹ הִשְׁגִּ֑יחַ אֶ֖ל כָּל־ יֹשְׁבֵ֣י הָאָֽרֶץ׃ (סה) הַיֹּצֵ֣ר יַ֣חַד לִבָּ֑ם הַ֝מֵּבִ֗ין אֶל־ כָּל־ מַעֲשֵׂיהֶֽם׃ (סו) אֵֽין־ הַ֭מֶּלֶךְ נוֹשָׁ֣ע בְּרָב־ חָ֑יִל גִּ֝בּ֗וֹר לֹֽא־ יִנָּצֵ֥ל בְּרָב־ כֹּֽחַ׃ (סז) שֶׁ֣קֶר הַ֭סּוּס לִתְשׁוּעָ֑ה וּבְרֹ֥ב חֵ֝יל֗וֹ לֹ֣א יְמַלֵּֽט׃ (סח) הִנֵּ֤ה עֵ֣ין יְ֭הוָה אֶל־ יְרֵאָ֑יו לַֽמְיַחֲלִ֥ים לְחַסְדּֽוֹ׃ (סט) לְהַצִּ֣יל מִמָּ֣וֶת נַפְשָׁ֑ם וּ֝לְחַיּוֹתָ֗ם בָּרָעָֽב׃ (ע) נַ֭פְשֵׁנוּ חִכְּתָ֣ה לַֽיהוָ֑ה עֶזְרֵ֖נוּ וּמָגִנֵּ֣נוּ הֽוּא׃ (עא) כִּי־ ב֭וֹ יִשְׂמַ֣ח לִבֵּ֑נוּ כִּ֤י בְשֵׁ֖ם קָדְשׁ֣וֹ בָטָֽחְנוּ׃ (עב) יְהִֽי־ חַסְדְּךָ֣ יְהוָ֣ה עָלֵ֑ינוּ כַּ֝אֲשֶׁ֗ר יִחַ֥לְנוּ לָֽךְ׃ (עג) לְדָוִ֗ד בְּשַׁנּוֹת֣וֹ אֶת־ טַ֭עְמוֹ לִפְנֵ֣י אֲבִימֶ֑לֶךְ וַֽ֝יְגָרֲשֵׁ֗הוּ וַיֵּלַֽךְ׃ (עד) אֲבָרֲכָ֣ה אֶת־ יְהוָ֣ה בְּכָל־ עֵ֑ת תָּ֝מִ֗יד תְּֽהִלָּת֥וֹ בְּפִֽי׃ (עה) בַּ֭יהוָה תִּתְהַלֵּ֣ל נַפְשִׁ֑י יִשְׁמְע֖וּ עֲנָוִ֣ים וְיִשְׂמָֽחוּ׃ (עו) גַּדְּל֣וּ לַיהוָ֣ה אִתִּ֑י וּנְרוֹמְמָ֖ה שְׁמ֣וֹ יַחְדָּֽו׃ (עז) דָּרַ֣שְׁתִּי אֶת־ יְהוָ֣ה וְעָנָ֑נִי וּמִכָּל־ מְ֝גוּרוֹתַ֗י הִצִּילָֽנִי׃ (עח) הִבִּ֣יטוּ אֵלָ֣יו וְנָהָ֑רוּ וּ֝פְנֵיהֶ֗ם אַל־ יֶחְפָּֽרוּ׃ (עט) זֶ֤ה עָנִ֣י קָ֭רָא וַיהוָ֣ה שָׁמֵ֑עַ וּמִכָּל־ צָ֝רוֹתָ֗יו הוֹשִׁיעֽוֹ׃ (פ) חֹנֶ֤ה מַלְאַךְ־ יְהוָ֓ה סָ֘בִ֤יב לִֽירֵאָ֗יו וַֽיְחַלְּצֵֽם׃ (פא) טַעֲמ֣וּ וּ֭רְאוּ כִּי־ ט֣וֹב יְהוָ֑ה אַֽשְׁרֵ֥י הַ֝גֶּ֗בֶר יֶחֱסֶה־ בּֽוֹ׃ (פב) יְר֣אוּ אֶת־ יְהוָ֣ה קְדֹשָׁ֑יו כִּי־ אֵ֥ין מַ֝חְס֗וֹר לִירֵאָֽיו׃ (פג) כְּ֭פִירִים רָשׁ֣וּ וְרָעֵ֑בוּ וְדֹרְשֵׁ֥י יְ֝הוָ֗ה לֹא־ יַחְסְר֥וּ כָל־ טֽוֹב׃ (פד) לְֽכוּ־ בָ֭נִים שִׁמְעוּ־ לִ֑י יִֽרְאַ֥ת יְ֝הוָ֗ה אֲלַמֶּדְכֶֽם׃ (פה) מִֽי־ הָ֭אִישׁ הֶחָפֵ֣ץ חַיִּ֑ים אֹהֵ֥ב יָ֝מִ֗ים לִרְא֥וֹת טֽוֹב׃ (פו) נְצֹ֣ר לְשׁוֹנְךָ֣ מֵרָ֑ע וּ֝שְׂפָתֶ֗יךָ מִדַּבֵּ֥ר מִרְמָֽה׃ (פז) ס֣וּר מֵ֭רָע וַעֲשֵׂה־ ט֑וֹב בַּקֵּ֖שׁ שָׁל֣וֹם וְרָדְפֵֽהוּ׃ (פח) עֵינֵ֣י יְ֭הוָה אֶל־ צַדִּיקִ֑ים וְ֝אָזְנָ֗יו אֶל־ שַׁוְעָתָֽם׃ (פט) פְּנֵ֣י יְ֭הוָה בְּעֹ֣שֵׂי רָ֑ע לְהַכְרִ֖ית מֵאֶ֣רֶץ זִכְרָֽם׃ (צ) צָעֲק֣וּ וַיהוָ֣ה שָׁמֵ֑עַ וּמִכָּל־ צָ֝רוֹתָ֗ם הִצִּילָֽם׃ (צא) קָר֣וֹב יְ֭הוָה לְנִשְׁבְּרֵי־ לֵ֑ב וְֽאֶת־ דַּכְּאֵי־ ר֥וּחַ יוֹשִֽׁיעַ׃ (צב) רַ֭בּוֹת רָע֣וֹת צַדִּ֑יק וּ֝מִכֻּלָּ֗ם יַצִּילֶ֥נּוּ יְהוָֽה׃ (צג) שֹׁמֵ֥ר כָּל־ עַצְמוֹתָ֑יו אַחַ֥ת מֵ֝הֵ֗נָּה לֹ֣א נִשְׁבָּֽרָה׃ (צד) תְּמוֹתֵ֣ת רָשָׁ֣ע רָעָ֑ה וְשֹׂנְאֵ֖י צַדִּ֣יק יֶאְשָֽׁמוּ׃ (צה) פּוֹדֶ֣ה יְ֭הוָה נֶ֣פֶשׁ עֲבָדָ֑יו וְלֹ֥א יֶ֝אְשְׁמ֗וּ כָּֽל־ הַחֹסִ֥ים בּֽוֹ׃ (צו) לְדָוִ֨ד ׀ רִיבָ֣ה יְ֭הוָה אֶת־ יְרִיבַ֑י לְ֝חַ֗ם אֶת־ לֹֽחֲמָֽי׃ (צז) הַחֲזֵ֣ק מָגֵ֣ן וְצִנָּ֑ה וְ֝ק֗וּמָה בְּעֶזְרָתִֽי׃ (צח) וְהָ֘רֵ֤ק חֲנִ֣ית וּ֭סְגֹר לִקְרַ֣את רֹדְפָ֑י אֱמֹ֥ר לְ֝נַפְשִׁ֗י יְֽשֻׁעָתֵ֥ךְ אָֽנִי׃ (צט) יֵבֹ֣שׁוּ וְיִכָּלְמוּ֮ מְבַקְשֵׁ֪י נַ֫פְשִׁ֥י יִסֹּ֣גוּ אָח֣וֹר וְיַחְפְּר֑וּ חֹ֝שְׁבֵ֗י רָעָתִֽי׃ (ק) יִֽהְי֗וּ כְּמֹ֥ץ לִפְנֵי־ ר֑וּחַ וּמַלְאַ֖ךְ יְהוָ֣ה דּוֹחֶֽה׃ (קא) יְֽהִי־ דַרְכָּ֗ם חֹ֥שֶׁךְ וַחֲלַקְלַקּ֑וֹת וּמַלְאַ֥ךְ יְ֝הוָ֗ה רֹדְפָֽם׃ (קב) כִּֽי־ חִנָּ֣ם טָֽמְנוּ־ לִ֭י שַׁ֣חַת רִשְׁתָּ֑ם חִ֝נָּ֗ם חָפְר֥וּ לְנַפְשִֽׁי׃ (קג) תְּבוֹאֵ֣הוּ שׁוֹאָה֮ לֹֽא־ יֵ֫דָ֥ע וְרִשְׁתּ֣וֹ אֲשֶׁר־ טָמַ֣ן תִּלְכְּד֑וֹ בְּ֝שׁוֹאָ֗ה יִפָּל־ בָּֽהּ׃ (קד) וְ֭נַפְשִׁי תָּגִ֣יל בַּיהוָ֑ה תָּ֝שִׂישׂ בִּישׁוּעָתֽוֹ׃ (קה) כָּ֥ל עַצְמוֹתַ֨י ׀ תֹּאמַרְנָה֮ יְהוָ֗ה מִ֥י כָ֫מ֥וֹךָ מַצִּ֣יל עָ֭נִי מֵחָזָ֣ק מִמֶּ֑נּוּ וְעָנִ֥י וְ֝אֶבְי֗וֹן מִגֹּזְלֽוֹ׃ (קו) יְ֭קוּמוּן עֵדֵ֣י חָמָ֑ס אֲשֶׁ֥ר לֹא־ יָ֝דַ֗עְתִּי יִשְׁאָלֽוּנִי׃ (קז) יְשַׁלְּמ֣וּנִי רָ֭עָה תַּ֥חַת טוֹבָ֗ה שְׁכ֣וֹל לְנַפְשִֽׁי׃ (קח) וַאֲנִ֤י ׀ בַּחֲלוֹתָ֡ם לְב֬וּשִׁי שָׂ֗ק עִנֵּ֣יתִי בַצּ֣וֹם נַפְשִׁ֑י וּ֝תְפִלָּתִ֗י עַל־ חֵיקִ֥י תָשֽׁוּב׃ (קט) כְּרֵֽעַ־ כְּאָ֣ח לִ֭י הִתְהַלָּ֑כְתִּי כַּאֲבֶל־ אֵ֝֗ם קֹדֵ֥ר שַׁחֽוֹתִי׃ (קי) וּבְצַלְעִי֮ שָׂמְח֪וּ וְֽנֶאֱ֫סָ֥פוּ נֶאֶסְפ֬וּ עָלַ֣י נֵ֭כִים וְלֹ֣א יָדַ֑עְתִּי קָֽרְע֥וּ וְלֹא־ דָֽמּוּ׃ (קיא) בְּ֭חַנְפֵי לַעֲגֵ֣י מָע֑וֹג חָרֹ֖ק עָלַ֣י שִׁנֵּֽימוֹ׃ (קיב) אֲדֹנָי֮ כַּמָּ֪ה תִּ֫רְאֶ֥ה הָשִׁ֣יבָה נַ֭פְשִׁי מִשֹּׁאֵיהֶ֑ם מִ֝כְּפִירִ֗ים יְחִידָתִֽי׃ (קיג) א֭וֹדְךָ בְּקָהָ֣ל רָ֑ב בְּעַ֖ם עָצ֣וּם אֲהַֽלְלֶֽךָּ׃ (קיד) אַֽל־ יִשְׂמְחוּ־ לִ֣י אֹיְבַ֣י שֶׁ֑קֶר שֹׂנְאַ֥י חִ֝נָּ֗ם יִקְרְצוּ־ עָֽיִן׃ (קטו) כִּ֤י לֹ֥א שָׁל֗וֹם יְדַ֫בֵּ֥רוּ וְעַ֥ל רִגְעֵי־ אֶ֑רֶץ דִּבְרֵ֥י מִ֝רְמוֹת יַחֲשֹׁבֽוּן׃ (קטז) וַיַּרְחִ֥יבוּ עָלַ֗י פִּ֫יהֶ֥ם אָ֭מְרוּ הֶאָ֣ח ׀ הֶאָ֑ח רָאֲתָ֥ה עֵינֵֽינוּ׃ (קיז) רָאִ֣יתָה יְ֭הוָה אַֽל־ תֶּחֱרַ֑שׁ אֲ֝דֹנָ֗י אֲל־ תִּרְחַ֥ק מִמֶּֽנִּי׃ (קיח) הָעִ֣ירָה וְ֭הָקִיצָה לְמִשְׁפָּטִ֑י אֱלֹהַ֖י וַֽאדֹנָ֣י לְרִיבִֽי׃ (קיט) שָׁפְטֵ֣נִי כְ֭צִדְקְךָ יְהוָ֥ה אֱלֹהָ֗י וְאַל־ יִשְׂמְחוּ־ לִֽי׃ (קכ) אַל־ יֹאמְר֣וּ בְ֭לִבָּם הֶאָ֣ח נַפְשֵׁ֑נוּ אַל־ יֹ֝אמְר֗וּ בִּֽלַּעֲנֽוּהוּ׃ (קכא) יֵ֘בֹ֤שׁוּ וְיַחְפְּר֨וּ ׀ יַחְדָּו֮ שְׂמֵחֵ֪י רָעָ֫תִ֥י יִֽלְבְּשׁוּ־ בֹ֥שֶׁת וּכְלִמָּ֑ה הַֽמַּגְדִּילִ֥ים עָלָֽי׃ (קכב) יָרֹ֣נּוּ
