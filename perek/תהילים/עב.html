פרק עב בספר תהילים (20 פסוקים) נמצא בתהילים סדר ט (פסוקים קה–קכג) ובתהילים סדר י (פסוקים א–א).

תהילים סדר ט

(א) יְבָרְכֵ֥נוּ אֱלֹהִ֑ים וְיִֽירְא֥וּ אֹ֝ת֗וֹ כָּל־ אַפְסֵי־ אָֽרֶץ׃ (ב) לַמְנַצֵּ֥חַ לְדָוִ֗ד מִזְמ֥וֹר שִֽׁיר׃ (ג) יָק֣וּם אֱ֭לֹהִים יָפ֣וּצוּ אוֹיְבָ֑יו וְיָנ֥וּסוּ מְ֝שַׂנְאָ֗יו מִפָּנָֽיו׃ (ד) כְּהִנְדֹּ֥ף עָשָׁ֗ן תִּ֫נְדֹּ֥ף כְּהִמֵּ֣ס דּ֭וֹנַג מִפְּנֵי־ אֵ֑שׁ יֹאבְד֥וּ רְ֝שָׁעִ֗ים מִפְּנֵ֥י אֱלֹהִֽים׃ (ה) וְֽצַדִּיקִ֗ים יִשְׂמְח֣וּ יַֽ֭עַלְצוּ לִפְנֵ֥י אֱלֹהִ֗ים וְיָשִׂ֥ישׂוּ בְשִׂמְחָֽה׃ (ו) שִׁ֤ירוּ ׀ לֵֽאלֹהִים֮ זַמְּר֪וּ שְׁ֫מ֥וֹ סֹ֡לּוּ לָרֹכֵ֣ב בָּ֭עֲרָבוֹת בְּיָ֥הּ שְׁמ֗וֹ וְעִלְז֥וּ לְפָנָֽיו׃ (ז) אֲבִ֣י יְ֭תוֹמִים וְדַיַּ֣ן אַלְמָנ֑וֹת אֱ֝לֹהִ֗ים בִּמְע֥וֹן קָדְשֽׁוֹ׃ (ח) אֱלֹהִ֤ים ׀ מ֘וֹשִׁ֤יב יְחִידִ֨ים ׀ בַּ֗יְתָה מוֹצִ֣יא אֲ֭סִירִים בַּכּוֹשָׁר֑וֹת אַ֥ךְ ס֝וֹרֲרִ֗ים שָׁכְנ֥וּ צְחִיחָֽה׃ (ט) אֱ‍ֽלֹהִ֗ים בְּ֭צֵאתְךָ לִפְנֵ֣י עַמֶּ֑ךָ בְּצַעְדְּךָ֖ בִֽישִׁימ֣וֹן סֶֽלָה׃ (י) אֶ֤רֶץ רָעָ֨שָׁה ׀ אַף־ שָׁמַ֣יִם נָטְפוּ֮ מִפְּנֵ֪י אֱלֹ֫הִ֥ים זֶ֥ה סִינַ֑י מִפְּנֵ֥י אֱ֝לֹהִ֗ים אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (יא) גֶּ֣שֶׁם נְ֭דָבוֹת תָּנִ֣יף אֱלֹהִ֑ים נַחֲלָתְךָ֥ וְ֝נִלְאָ֗ה אַתָּ֥ה כֽוֹנַנְתָּֽהּ׃ (יב) חַיָּתְךָ֥ יָֽשְׁבוּ־ בָ֑הּ תָּ֤כִ֥ין בְּטוֹבָתְךָ֖ לֶעָנִ֣י אֱלֹהִֽים׃ (יג) אֲדֹנָ֥י יִתֶּן־ אֹ֑מֶר הַֽ֝מְבַשְּׂר֗וֹת צָבָ֥א רָֽב׃ (יד) מַלְכֵ֣י צְ֭בָאוֹת יִדֹּד֣וּן יִדֹּד֑וּן וּנְוַת בַּ֝֗יִת תְּחַלֵּ֥ק שָׁלָֽל׃ (טו) אִֽם־ תִּשְׁכְּבוּן֮ בֵּ֪ין שְׁפַ֫תָּ֥יִם כַּנְפֵ֣י י֭וֹנָה נֶחְפָּ֣ה בַכֶּ֑סֶף וְ֝אֶבְרוֹתֶ֗יהָ בִּֽירַקְרַ֥ק חָרֽוּץ׃ (טז) בְּפָ֘רֵ֤שׂ שַׁדַּ֓י מְלָ֘כִ֤ים בָּ֗הּ תַּשְׁלֵ֥ג בְּצַלְמֽוֹן׃ (יז) הַר־ אֱ֭לֹהִים הַר־ בָּשָׁ֑ן הַ֥ר גַּ֝בְנֻנִּ֗ים הַר־ בָּשָֽׁן׃ (יח) לָ֤מָּה ׀ תְּֽרַצְּדוּן֮ הָרִ֪ים גַּבְנֻ֫נִּ֥ים הָהָ֗ר חָמַ֣ד אֱלֹהִ֣ים לְשִׁבְתּ֑וֹ אַף־ יְ֝הוָ֗ה יִשְׁכֹּ֥ן לָנֶֽצַח׃ (יט) רֶ֤כֶב אֱלֹהִ֗ים רִבֹּתַ֣יִם אַלְפֵ֣י שִׁנְאָ֑ן אֲדֹנָ֥י בָ֝֗ם סִינַ֥י בַּקֹּֽדֶשׁ׃ (כ) עָ֘לִ֤יתָ לַמָּר֨וֹם ׀ שָׁ֘בִ֤יתָ שֶּׁ֗בִי לָקַ֣חְתָּ מַ֭תָּנוֹת בָּאָדָ֑ם וְאַ֥ף ס֝וֹרְרִ֗ים לִשְׁכֹּ֤ן ׀ יָ֬הּ אֱלֹהִֽים׃ (כא) בָּ֤ר֣וּךְ אֲדֹנָי֮ י֤וֹם ׀ י֥וֹם יַֽעֲמָס־ לָ֗נוּ הָ֘אֵ֤ל יְֽשׁוּעָתֵ֬נוּ סֶֽלָה׃ (כב) הָ֤אֵ֣ל ׀ לָנוּ֮ אֵ֤ל לְֽמוֹשָׁ֫ע֥וֹת וְלֵיהוִ֥ה אֲדֹנָ֑י לַ֝מָּ֗וֶת תּוֹצָאֽוֹת׃ (כג) אַךְ־ אֱלֹהִ֗ים יִמְחַץ֮ רֹ֤אשׁ אֹ֫יְבָ֥יו קָדְקֹ֥ד שֵׂעָ֑ר מִ֝תְהַלֵּ֗ךְ בַּאֲשָׁמָֽיו׃ (כד) אָמַ֣ר אֲ֭דֹנָי מִבָּשָׁ֣ן אָשִׁ֑יב אָ֝שִׁ֗יב מִֽמְּצֻל֥וֹת יָֽם׃ (כה) לְמַ֤עַן ׀ תִּֽמְחַ֥ץ רַגְלְךָ֗ בְּ֫דָ֥ם לְשׁ֥וֹן כְּלָבֶ֑יךָ מֵאֹיְבִ֥ים מִנֵּֽהוּ׃ (כו) רָא֣וּ הֲלִיכוֹתֶ֣יךָ אֱלֹהִ֑ים הֲלִ֘יכ֤וֹת אֵלִ֖י מַלְכִּ֣י בַקֹּֽדֶשׁ׃ (כז) קִדְּמ֣וּ שָׁ֭רִים אַחַ֣ר נֹגְנִ֑ים בְּת֥וֹךְ עֲ֝לָמ֗וֹת תּוֹפֵפֽוֹת׃ (כח) בְּֽ֭מַקְהֵלוֹת בָּרְכ֣וּ אֱלֹהִ֑ים יְ֝הוָ֗ה מִמְּק֥וֹר יִשְׂרָאֵֽל׃ (כט) שָׁ֤ם בִּנְיָמִ֨ן ׀ צָעִ֡יר רֹדֵ֗ם שָׂרֵ֣י יְ֭הוּדָה רִגְמָתָ֑ם שָׂרֵ֥י זְ֝בֻל֗וּן שָׂרֵ֥י נַפְתָּלִֽי׃ (ל) צִוָּ֥ה אֱלֹהֶ֗יךָ עֻ֫זֶּ֥ךָ עוּזָּ֥ה אֱלֹהִ֑ים ז֝֗וּ פָּעַ֥לְתָּ לָּֽנוּ׃ (לא) מֵֽ֭הֵיכָלֶךָ עַל־ יְרוּשָׁלִָ֑ם לְךָ֤ יוֹבִ֖ילוּ מְלָכִ֣ים שָֽׁי׃ (לב) גְּעַ֨ר חַיַּ֪ת קָנֶ֡ה עֲדַ֤ת אַבִּירִ֨ים ׀ בְּעֶגְלֵ֬י עַמִּ֗ים מִתְרַפֵּ֥ס בְּרַצֵּי־ כָ֑סֶף בִּזַּ֥ר עַ֝מִּ֗ים קְרָב֥וֹת יֶחְפָּֽצוּ׃ (לג) יֶאֱתָ֣יוּ חַ֭שְׁמַנִּים מִנִּ֣י מִצְרָ֑יִם כּ֥וּשׁ תָּרִ֥יץ יָ֝דָ֗יו לֵאלֹהִֽים׃ (לד) מַמְלְכ֣וֹת הָ֭אָרֶץ שִׁ֣ירוּ לֵאלֹהִ֑ים זַמְּר֖וּ אֲדֹנָ֣י סֶֽלָה׃ (לה) לָ֭רֹכֵב בִּשְׁמֵ֣י שְׁמֵי־ קֶ֑דֶם הֵ֥ן יִתֵּ֥ן בְּ֝קוֹלוֹ ק֣וֹל עֹֽז׃ (לו) תְּנ֥וּ עֹ֗ז לֵֽאלֹ֫הִ֥ים עַֽל־ יִשְׂרָאֵ֥ל גַּאֲוָת֑וֹ וְ֝עֻזּ֗וֹ בַּשְּׁחָקִֽים׃ (לז) נ֤וֹרָ֥א אֱלֹהִ֗ים מִֽמִּקְדָּ֫שֶׁ֥יךָ אֵ֤ל יִשְׂרָאֵ֗ל ה֤וּא נֹתֵ֨ן ׀ עֹ֖ז וְתַעֲצֻמ֥וֹת לָעָ֗ם בָּר֥וּךְ אֱלֹהִֽים׃ (לח) לַמְנַצֵּ֬חַ עַֽל־ שׁוֹשַׁנִּ֬ים לְדָוִֽד׃ (לט) הוֹשִׁיעֵ֥נִי אֱלֹהִ֑ים כִּ֤י בָ֖אוּ מַ֣יִם עַד־ נָֽפֶשׁ׃ (מ) טָבַ֤עְתִּי ׀ בִּיוֵ֣ן מְ֭צוּלָה וְאֵ֣ין מָעֳמָ֑ד בָּ֥אתִי בְמַעֲמַקֵּי־ מַ֝֗יִם וְשִׁבֹּ֥לֶת שְׁטָפָֽתְנִי׃ (מא) יָגַ֣עְתִּי בְקָרְאִי֮ נִחַ֪ר גְּר֫וֹנִ֥י כָּל֥וּ עֵינַ֑י מְ֝יַחֵ֗ל לֵאלֹהָֽי׃ (מב) רַבּ֤וּ ׀ מִשַּׂעֲר֣וֹת רֹאשִׁי֮ שֹׂנְאַ֪י חִ֫נָּ֥ם עָצְמ֣וּ מַ֭צְמִיתַי אֹיְבַ֣י שֶׁ֑קֶר אֲשֶׁ֥ר לֹא־ גָ֝זַ֗לְתִּי אָ֣ז אָשִֽׁיב׃ (מג) אֱ‍ֽלֹהִ֗ים אַתָּ֣ה יָ֭דַעְתָּ לְאִוַּלְתִּ֑י וְ֝אַשְׁמוֹתַ֗י מִמְּךָ֥ לֹא־ נִכְחָֽדוּ׃ (מד) אַל־ יֵ֘בֹ֤שׁוּ בִ֨י ׀ קֹוֶיךָ֮ אֲדֹנָ֥י יְהוִ֗ה צְבָ֫א֥וֹת אַל־ יִכָּ֣לְמוּ בִ֣י מְבַקְשֶׁ֑יךָ אֱ֝לֹהֵ֗י יִשְׂרָאֵֽל׃ (מה) כִּֽי־ עָ֭לֶיךָ נָשָׂ֣אתִי חֶרְפָּ֑ה כִּסְּתָ֖ה כְלִמָּ֣ה פָנָֽי׃ (מו) מ֭וּזָר הָיִ֣יתִי לְאֶחָ֑י וְ֝נָכְרִ֗י לִבְנֵ֥י אִמִּֽי׃ (מז) כִּֽי־ קִנְאַ֣ת בֵּיתְךָ֣ אֲכָלָ֑תְנִי וְחֶרְפּ֥וֹת ח֝וֹרְפֶ֗יךָ נָפְל֥וּ עָלָֽי׃ (מח) וָאֶבְכֶּ֣ה בַצּ֣וֹם נַפְשִׁ֑י וַתְּהִ֖י לַחֲרָפ֣וֹת לִֽי׃ (מט) וָאֶתְּנָ֣ה לְבוּשִׁ֣י שָׂ֑ק וָאֱהִ֖י לָהֶ֣ם לְמָשָֽׁל׃ (נ) יָשִׂ֣יחוּ בִ֭י יֹ֣שְׁבֵי שָׁ֑עַר וּ֝נְגִינ֗וֹת שׁוֹתֵ֥י שֵׁכָֽר׃ (נא) וַאֲנִ֤י תְפִלָּתִֽי־ לְךָ֨ ׀ יְהוָ֡ה עֵ֤ת רָצ֗וֹן אֱלֹהִ֥ים בְּרָב־ חַסְדֶּ֑ךָ עֲ֝נֵ֗נִי בֶּאֱמֶ֥ת יִשְׁעֶֽךָ׃ (נב) הַצִּילֵ֣נִי מִ֭טִּיט וְאַל־ אֶטְבָּ֑עָה אִנָּצְלָ֥ה מִ֝שֹּֽׂנְאַ֗י וּמִמַּֽעֲמַקֵּי־ מָֽיִם׃ (נג) אַל־ תִּשְׁטְפֵ֤נִי ׀ שִׁבֹּ֣לֶת מַ֭יִם וְאַל־ תִּבְלָעֵ֣נִי מְצוּלָ֑ה וְאַל־ תֶּאְטַר־ עָלַ֖י בְּאֵ֣ר פִּֽיהָ׃ (נד) עֲנֵ֣נִי יְ֭הוָה כִּי־ ט֣וֹב חַסְדֶּ֑ךָ כְּרֹ֥ב רַ֝חֲמֶ֗יךָ פְּנֵ֣ה אֵלָֽי׃ (נה) וְאַל־ תַּסְתֵּ֣ר פָּ֭נֶיךָ מֵֽעַבְדֶּ֑ךָ כִּֽי־ צַר־ לִ֝֗י מַהֵ֥ר עֲנֵֽנִי׃ (נו) קָרְבָ֣ה אֶל־ נַפְשִׁ֣י גְאָלָ֑הּ לְמַ֖עַן אֹיְבַ֣י פְּדֵֽנִי׃ (נז) אַתָּ֤ה יָדַ֗עְתָּ חֶרְפָּתִ֣י וּ֭בָשְׁתִּי וּכְלִמָּתִ֑י נֶ֝גְדְּךָ֗ כָּל־ צוֹרְרָֽי׃ (נח) חֶרְפָּ֤ה ׀ שָֽׁבְרָ֥ה לַיהוָ֪ה אֱ‍ֽלֹהֵ֫יכֶ֥ם כָּל־ סְבִיבָ֑יו יוֹבִ֥ילוּ שַׁ֝֗י לַמּוֹרָֽא׃ (עו) יִ֭בְצֹר ר֣וּחַ נְגִידִ֑ים נ֝וֹרָ֗א לְמַלְכֵי־ אָֽרֶץ׃ (עז) לַמְנַצֵּ֥חַ עַֽל־ (ידיתון) יְדוּת֗וּן לְאָסָ֥ף מִזְמֽוֹר׃ (עח) קוֹלִ֣י אֶל־ אֱלֹהִ֣ים וְאֶצְעָ֑קָה קוֹלִ֥י אֶל־ אֱ֝לֹהִ֗ים וְהַאֲזִ֥ין אֵלָֽי׃ (עט) בְּי֥וֹם צָרָתִי֮ אֲדֹנָ֪י דָּ֫רָ֥שְׁתִּי יָדִ֤י ׀ לַ֣יְלָה נִ֭גְּרָה וְלֹ֣א תָפ֑וּג מֵאֲנָ֖ה הִנָּחֵ֣ם נַפְשִֽׁי׃ (פ) אֶזְכְּרָ֣ה אֱלֹהִ֣ים וְאֶֽהֱמָיָ֑ה אָשִׂ֓יחָה ׀ וְתִתְעַטֵּ֖ף רוּחִ֣י סֶֽלָה׃ (פא) אָ֭חַזְתָּ שְׁמֻר֣וֹת עֵינָ֑י נִ֝פְעַ֗מְתִּי וְלֹ֣א אֲדַבֵּֽר׃ (פב) חִשַּׁ֣בְתִּי יָמִ֣ים מִקֶּ֑דֶם שְׁ֝נ֗וֹת עוֹלָמִֽים׃ (פג) אֶֽזְכְּרָ֥ה נְגִינָתִ֗י בַּ֫לָּ֥יְלָה עִם־ לְבָבִ֥י אָשִׂ֑יחָה וַיְחַפֵּ֥שׂ רוּחִֽי׃ (פד) הַֽ֭לְעוֹלָמִים יִזְנַ֥ח ׀ אֲדֹנָ֑י וְלֹֽא־ יֹסִ֖יף לִרְצ֣וֹת עֽוֹד׃ (פה) הֶאָפֵ֣ס לָנֶ֣צַח חַסְדּ֑וֹ גָּ֥מַר אֹ֝֗מֶר לְדֹ֣ר וָדֹֽר׃ (פו) הֲשָׁכַ֣ח חַנּ֣וֹת אֵ֑ל אִם־ קָפַ֥ץ בְּ֝אַ֗ף רַחֲמָ֥יו סֶֽלָה׃ (פז) וָ֭אֹמַר חַלּ֣וֹתִי הִ֑יא שְׁ֝נ֗וֹת יְמִ֣ין עֶלְיֽוֹן׃ (פח) (אזכיר) אֶזְכּ֥וֹר מַֽעַלְלֵי־ יָ֑הּ כִּֽי־ אֶזְכְּרָ֖ה מִקֶּ֣דֶם פִּלְאֶֽךָ׃ (פט) וְהָגִ֥יתִי בְכָל־ פָּעֳלֶ֑ךָ וּֽבַעֲלִ֖ילוֹתֶ֣יךָ אָשִֽׂיחָה׃ (צ) אֱ֭לֹהִים בַּקֹּ֣דֶשׁ דַּרְכֶּ֑ךָ מִי־ אֵ֥ל גָּ֝ד֗וֹל כֵּֽאלֹהִֽים׃ (צא) אַתָּ֣ה הָ֭אֵל עֹ֣שֵׂה פֶ֑לֶא הוֹדַ֖עְתָּ בָעַמִּ֣ים עֻזֶּֽךָ׃ (צב) גָּאַ֣לְתָּ בִּזְר֣וֹעַ עַמֶּ֑ךָ בְּנֵי־ יַעֲקֹ֖ב וְיוֹסֵ֣ף סֶֽלָה׃ (צג) רָ֘א֤וּךָ מַּ֨יִם ׀ אֱ‍ֽלֹהִ֗ים רָא֣וּךָ מַּ֣יִם יָחִ֑ילוּ אַ֝֗ף יִרְגְּז֥וּ תְהֹמֽוֹת׃ (צד) זֹ֤רְמוּ מַ֨יִם ׀ עָב֗וֹת ק֭וֹל נָתְנ֣וּ שְׁחָקִ֑ים אַף־ חֲ֝צָצֶ֗יךָ יִתְהַלָּֽכוּ׃ (צה) ק֤וֹל רַעַמְךָ֨ ׀ בַּגַּלְגַּ֗ל הֵאִ֣ירוּ בְרָקִ֣ים תֵּבֵ֑ל רָגְזָ֖ה וַתִּרְעַ֣שׁ הָאָֽרֶץ׃ (צו) בַּיָּ֤ם דַּרְכֶּ֗ךָ (ושביליך) וּֽ֭שְׁבִֽילְךָ בְּמַ֣יִם רַבִּ֑ים וְ֝עִקְּבוֹתֶ֗יךָ לֹ֣א נֹדָֽעוּ׃ (צז) נָחִ֣יתָ כַצֹּ֣אן עַמֶּ֑ךָ בְּֽיַד־ מֹשֶׁ֥ה וְאַהֲרֹֽן׃ (צח) מַשְׂכִּ֗יל לְאָ֫סָ֥ף הַאֲזִ֣ינָה עַ֭מִּי תּוֹרָתִ֑י הַטּ֥וּ אָ֝זְנְכֶ֗ם לְאִמְרֵי־ פִֽי׃ (צט) אֶפְתְּחָ֣ה בְמָשָׁ֣ל פִּ֑י אַבִּ֥יעָה חִ֝יד֗וֹת מִנִּי־ קֶֽדֶם׃ (ק) אֲשֶׁ֣ר שָׁ֭מַעְנוּ וַנֵּדָעֵ֑ם וַ֝אֲבוֹתֵ֗ינוּ סִפְּרוּ־ לָֽנוּ׃ (קא) לֹ֤א נְכַחֵ֨ד ׀ מִבְּנֵיהֶ֗ם לְד֥וֹר אַחֲר֗וֹן מְֽ֭סַפְּרִים תְּהִלּ֣וֹת יְהוָ֑ה וֶעֱזוּז֥וֹ וְ֝נִפְלְאוֹתָ֗יו אֲשֶׁ֣ר עָשָֽׂה׃ (קב) וַיָּ֤קֶם עֵד֨וּת ׀ בְּֽיַעֲקֹ֗ב וְתוֹרָה֮ שָׂ֤ם בְּיִשְׂרָ֫אֵ֥ל אֲשֶׁ֣ר צִ֭וָּה אֶת־ אֲבוֹתֵ֑ינוּ לְ֝הוֹדִיעָ֗ם לִבְנֵיהֶֽם׃ (קג) לְמַ֤עַן יֵדְע֨וּ ׀ דּ֣וֹר אַ֭חֲרוֹן בָּנִ֣ים יִוָּלֵ֑דוּ יָ֝קֻ֗מוּ וִֽיסַפְּר֥וּ לִבְנֵיהֶֽם׃ (קד) וְיָשִׂ֥ימוּ בֵֽאלֹהִ֗ים כִּ֫סְלָ֥ם וְלֹ֣א יִ֭שְׁכְּחוּ מַֽעַלְלֵי־ אֵ֑ל וּמִצְוֺתָ֥יו יִנְצֹֽרוּ׃ (קה) וְלֹ֤א יִהְי֨וּ ׀ כַּאֲבוֹתָ֗ם דּוֹר֮ סוֹרֵ֪ר וּמֹ֫רֶ֥ה דּ֭וֹר לֹא־ הֵכִ֣ין לִבּ֑וֹ וְלֹא־ נֶאֶמְנָ֖ה אֶת־ אֵ֣ל רוּחֽוֹ׃ (קו) בְּֽנֵי־ אֶפְרַ֗יִם נוֹשְׁקֵ֥י רוֹמֵי־ קָ֑שֶׁת הָ֝פְכ֗וּ בְּי֣וֹם קְרָֽב׃ (קז) לֹ֣א שָׁ֭מְרוּ בְּרִ֣ית אֱלֹהִ֑ים וּ֝בְתוֹרָת֗וֹ מֵאֲנ֥וּ לָלֶֽכֶת׃ (קח) וַיִּשְׁכְּח֥וּ עֲלִילוֹתָ֑יו וְ֝נִפְלְאוֹתָ֗יו אֲשֶׁ֣ר הֶרְאָֽם׃ (קט) נֶ֣גֶד אֲ֭בוֹתָם עָ֣שָׂה פֶ֑לֶא בְּאֶ֖רֶץ מִצְרַ֣יִם שְׂדֵה־ צֹֽעַן׃ (קי) בָּ֣קַע יָ֭ם וַיַּֽעֲבִירֵ֑ם וַֽיַּצֶּב־ מַ֥יִם כְּמוֹ־ נֵֽד׃ (קיא) וַיַּנְחֵ֣ם בֶּעָנָ֣ן יוֹמָ֑ם וְכָל־ הַ֝לַּ֗יְלָה בְּא֣וֹר אֵֽשׁ׃ (קיב) יְבַקַּ֣ע צֻ֭רִים בַּמִּדְבָּ֑ר וַ֝יַּ֗שְׁקְ כִּתְהֹמ֥וֹת רַבָּֽה׃ (קיג) וַיּוֹצִ֣א נוֹזְלִ֣ים מִסָּ֑לַע וַיּ֖וֹרֶד כַּנְּהָר֣וֹת מָֽיִם׃ (קיד) וַיּוֹסִ֣יפוּ ע֭וֹד לַחֲטֹא־ ל֑וֹ לַֽמְר֥וֹת עֶ֝לְי֗וֹן בַּצִּיָּֽה׃ (קטו) וַיְנַסּוּ־ אֵ֥ל בִּלְבָבָ֑ם לִֽשְׁאָל־ אֹ֥כֶל לְנַפְשָֽׁם׃ (קטז) וַֽיְדַבְּר֗וּ בֵּֽאלֹ֫הִ֥ים אָ֭מְרוּ הֲי֣וּכַל אֵ֑ל לַעֲרֹ֥ךְ שֻׁ֝לְחָ֗ן בַּמִּדְבָּֽר׃ (קיז) הֵ֤ן הִכָּה־ צ֨וּר ׀ וַיָּז֣וּבוּ מַיִם֮ וּנְחָלִ֪ים יִ֫שְׁטֹ֥פוּ הֲגַם־ לֶ֭חֶם י֣וּכַל תֵּ֑ת אִם־ יָכִ֖ין שְׁאֵ֣ר לָהֶ֣ם לָשֹֽׂבַע׃ (קכג) יַסַּ֣ע קָ֭דִים בַּשָּׁמָ֑יִם וַיְנַהֵ֖ג בְּעֻזּ֣וֹ תֵימָֽן׃ (קכד) וַיַּמְטֵ֬ר עֲלֵיהֶ֣ם כֶּעָפָ֣ר שְׁאֵ֑ר וּֽכְח֥וֹל יַ֝מִּ֗ים ע֣וֹף כָּנָֽף׃ (קכה) וַ֭יַּפֵּל בְּקֶ֣רֶב עָ֘לָ֤ה בָהֶ֗ם וַֽ֭יַּהֲרֹג בְּמִשְׁמַנֵּיהֶ֑ם וּבַחוּרֵ֖י יִשְׂרָאֵ֣ל הִכְרִֽיעַ׃ (קכט) בְּכָל־ זֹ֭את חָֽטְאוּ־ ע֑וֹד וְלֹֽא־ הֶ֝אֱמִ֗ינוּ בְּנִפְלְאוֹתָֽיו׃ (קל) וַיְכַל־ בַּהֶ֥בֶל יְמֵיהֶ֑ם
