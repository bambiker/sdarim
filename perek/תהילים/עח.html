תהילים סדר י

(א) כָּלּ֥וּ תְפִלּ֑וֹת דָּ֝וִ֗ד בֶּן־ יִשָֽׁי׃ (ב) מִזְמ֗וֹר לְאָ֫סָ֥ף אַ֤ךְ ט֭וֹב לְיִשְׂרָאֵ֥ל אֱלֹהִ֗ים לְבָרֵ֥י לֵבָֽב׃ (ג) וַאֲנִ֗י כִּ֭מְעַט (נטוי) נָטָ֣יוּ רַגְלָ֑י כְּ֝אַ֗יִן (שפכה) שֻׁפְּכ֥וּ אֲשֻׁרָֽי׃ (ד) כִּֽי־ קִ֭נֵּאתִי בַּֽהוֹלְלִ֑ים שְׁל֖וֹם רְשָׁעִ֣ים אֶרְאֶֽה׃ (ה) כִּ֤י אֵ֖ין חַרְצֻבּ֥וֹת לְמוֹתָ֗ם וּבָרִ֥יא אוּלָֽם׃ (ו) בַּעֲמַ֣ל אֱנ֣וֹשׁ אֵינֵ֑מוֹ וְעִם־ אָ֝דָ֗ם לֹ֣א יְנֻגָּֽעוּ׃ (ז) לָ֭כֵן עֲנָקַ֣תְמוֹ גַאֲוָ֑ה יַעֲטָף־ שִׁ֝֗ית חָמָ֥ס לָֽמוֹ׃ (ח) יָ֭צָא מֵחֵ֣לֶב עֵינֵ֑מוֹ עָ֝בְר֗וּ מַשְׂכִּיּ֥וֹת לֵבָֽב׃ (ט) יָמִ֤יקוּ ׀ וִידַבְּר֣וּ בְרָ֣ע עֹ֑שֶׁק מִמָּר֥וֹם יְדַבֵּֽרוּ׃ (י) שַׁתּ֣וּ בַשָּׁמַ֣יִם פִּיהֶ֑ם וּ֝לְשׁוֹנָ֗ם תִּֽהֲלַ֥ךְ בָּאָֽרֶץ׃ (יא) לָכֵ֤ן ׀ (ישיב) יָשׁ֣וּב עַמּ֣וֹ הֲלֹ֑ם וּמֵ֥י מָ֝לֵ֗א יִמָּ֥צוּ לָֽמוֹ׃ (יב) וְֽאָמְר֗וּ אֵיכָ֥ה יָדַֽע־ אֵ֑ל וְיֵ֖שׁ דֵּעָ֣ה בְעֶלְיֽוֹן׃ (יג) הִנֵּה־ אֵ֥לֶּה רְשָׁעִ֑ים וְשַׁלְוֵ֥י ע֝וֹלָ֗ם הִשְׂגּוּ־ חָֽיִל׃ (יד) אַךְ־ רִ֭יק זִכִּ֣יתִי לְבָבִ֑י וָאֶרְחַ֖ץ בְּנִקָּי֣וֹן כַּפָּֽי׃ (טו) וָאֱהִ֣י נָ֭גוּעַ כָּל־ הַיּ֑וֹם וְ֝תוֹכַחְתִּ֗י לַבְּקָרִֽים׃ (טז) אִם־ אָ֭מַרְתִּי אֲסַפְּרָ֥ה כְמ֑וֹ הִנֵּ֤ה ד֭וֹר בָּנֶ֣יךָ בָגָֽדְתִּי׃ (יז) וָֽ֭אֲחַשְּׁבָה לָדַ֣עַת זֹ֑את עָמָ֖ל (היא) ה֣וּא בְעֵינָֽי׃ (יח) עַד־ אָ֭בוֹא אֶל־ מִקְדְּשֵׁי־ אֵ֑ל אָ֝בִ֗ינָה לְאַחֲרִיתָֽם׃ (יט) אַ֣ךְ בַּ֭חֲלָקוֹת תָּשִׁ֣ית לָ֑מוֹ הִ֝פַּלְתָּ֗ם לְמַשּׁוּאֽוֹת׃ (כ) אֵ֤יךְ הָי֣וּ לְשַׁמָּ֣ה כְרָ֑גַע סָ֥פוּ תַ֝֗מּוּ מִן־ בַּלָּהֽוֹת׃ (כא) כַּחֲל֥וֹם מֵהָקִ֑יץ אֲ֝דֹנָי בָּעִ֤יר ׀ צַלְמָ֬ם תִּבְזֶֽה׃ (כב) כִּ֭י יִתְחַמֵּ֣ץ לְבָבִ֑י וְ֝כִלְיוֹתַ֗י אֶשְׁתּוֹנָֽן׃ (כג) וַאֲנִי־ בַ֭עַר וְלֹ֣א אֵדָ֑ע בְּ֝הֵמ֗וֹת הָיִ֥יתִי עִמָּֽךְ׃ (כד) וַאֲנִ֣י תָמִ֣יד עִמָּ֑ךְ אָ֝חַ֗זְתָּ בְּיַד־ יְמִינִֽי׃ (כה) בַּעֲצָתְךָ֥ תַנְחֵ֑נִי וְ֝אַחַ֗ר כָּב֥וֹד תִּקָּחֵֽנִי׃ (כו) מִי־ לִ֥י בַשָּׁמָ֑יִם וְ֝עִמְּךָ֗ לֹא־ חָפַ֥צְתִּי בָאָֽרֶץ׃ (כז) כָּלָ֥ה שְׁאֵרִ֗י וּלְבָ֫בִ֥י צוּר־ לְבָבִ֥י וְחֶלְקִ֗י אֱלֹהִ֥ים לְעוֹלָֽם׃ (כח) כִּֽי־ הִנֵּ֣ה רְחֵקֶ֣יךָ יֹאבֵ֑דוּ הִ֝צְמַ֗תָּה כָּל־ זוֹנֶ֥ה מִמֶּֽךָּ׃ (כט) וַאֲנִ֤י ׀ קִֽרֲבַ֥ת אֱלֹהִ֗ים לִ֫י־ ט֥וֹב שַׁתִּ֤י ׀ בַּאדֹנָ֣י יְהֹוִ֣ה מַחְסִ֑י לְ֝סַפֵּ֗ר כָּל־ מַלְאֲכוֹתֶֽיךָ׃ (ל) מַשְׂכִּ֗יל לְאָ֫סָ֥ף לָמָ֣ה אֱ֭לֹהִים זָנַ֣חְתָּ לָנֶ֑צַח יֶעְשַׁ֥ן אַ֝פְּךָ֗ בְּצֹ֣אן מַרְעִיתֶֽךָ׃ (לא) זְכֹ֤ר עֲדָתְךָ֨ ׀ קָ֘נִ֤יתָ קֶּ֗דֶם גָּ֭אַלְתָּ שֵׁ֣בֶט נַחֲלָתֶ֑ךָ הַר־ צִ֝יּ֗וֹן זֶ֤ה ׀ שָׁכַ֬נְתָּ בּֽוֹ׃ (לב) הָרִ֣ימָה פְ֭עָמֶיךָ לְמַשֻּׁא֣וֹת נֶ֑צַח כָּל־ הֵרַ֖ע אוֹיֵ֣ב בַּקֹּֽדֶשׁ׃ (לג) שָׁאֲג֣וּ צֹ֭רְרֶיךָ בְּקֶ֣רֶב מוֹעֲדֶ֑ךָ שָׂ֖מוּ אוֹתֹתָ֣ם אֹתֽוֹת׃ (לד) יִ֭וָּדַע כְּמֵבִ֣יא לְמָ֑עְלָה בִּֽסֲבָךְ־ עֵ֝֗ץ קַרְדֻּמּֽוֹת׃ (לה) (ועת) וְ֭עַתָּה פִּתּוּחֶ֣יהָ יָּ֑חַד בְּכַשִּׁ֥יל וְ֝כֵֽילַפֹּ֗ת יַהֲלֹמֽוּן׃ (לו) שִׁלְח֣וּ בָ֭אֵשׁ מִקְדָּשֶׁ֑ךָ לָ֝אָ֗רֶץ חִלְּל֥וּ מִֽשְׁכַּן־ שְׁמֶֽךָ׃ (לז) אָמְר֣וּ בְ֭לִבָּם נִינָ֣ם יָ֑חַד שָׂרְפ֖וּ כָל־ מוֹעֲדֵי־ אֵ֣ל בָּאָֽרֶץ׃ (לח) אֽוֹתֹתֵ֗ינוּ לֹ֥א רָ֫אִ֥ינוּ אֵֽין־ ע֥וֹד נָבִ֑יא וְלֹֽא־ אִ֝תָּ֗נוּ יֹדֵ֥עַ עַד־ מָֽה׃ (לט) עַד־ מָתַ֣י אֱ֭לֹהִים יְחָ֣רֶף צָ֑ר יְנָ֘אֵ֤ץ אוֹיֵ֖ב שִׁמְךָ֣ לָנֶֽצַח׃ (מ) לָ֤מָּה תָשִׁ֣יב יָ֭דְךָ וִֽימִינֶ֑ךָ מִקֶּ֖רֶב (חוקך) חֵֽיקְךָ֣ כַלֵּֽה׃ (מא) וֵ֭אלֹהִים מַלְכִּ֣י מִקֶּ֑דֶם פֹּעֵ֥ל יְ֝שׁוּע֗וֹת בְּקֶ֣רֶב הָאָֽרֶץ׃ (מב) אַתָּ֤ה פוֹרַ֣רְתָּ בְעָזְּךָ֣ יָ֑ם שִׁבַּ֖רְתָּ רָאשֵׁ֥י תַ֝נִּינִ֗ים עַל־ הַמָּֽיִם׃ (מג) אַתָּ֣ה רִ֭צַּצְתָּ רָאשֵׁ֣י לִוְיָתָ֑ן תִּתְּנֶ֥נּוּ מַ֝אֲכָ֗ל לְעָ֣ם לְצִיִּֽים׃ (מד) אַתָּ֣ה בָ֭קַעְתָּ מַעְיָ֣ן וָנָ֑חַל אַתָּ֥ה ה֝וֹבַ֗שְׁתָּ נַהֲר֥וֹת אֵיתָֽן׃ (מה) לְךָ֣ י֭וֹם אַף־ לְךָ֥ לָ֑יְלָה אַתָּ֥ה הֲ֝כִינ֗וֹתָ מָא֥וֹר וָשָֽׁמֶשׁ׃ (מו) אַתָּ֣ה הִ֭צַּבְתָּ כָּל־ גְּבוּל֣וֹת אָ֑רֶץ קַ֥יִץ וָ֝חֹ֗רֶף אַתָּ֥ה יְצַרְתָּם׃ (מז) זְכָר־ זֹ֗את א֭וֹיֵב חֵרֵ֣ף ׀ יְהוָ֑ה וְעַ֥ם נָ֝בָ֗ל נִֽאֲצ֥וּ שְׁמֶֽךָ׃ (מח) אַל־ תִּתֵּ֣ן לְ֭חַיַּת נֶ֣פֶשׁ תּוֹרֶ֑ךָ חַיַּ֥ת עֲ֝נִיֶּ֗יךָ אַל־ תִּשְׁכַּ֥ח לָנֶֽצַח׃ (מט) הַבֵּ֥ט לַבְּרִ֑ית כִּ֥י מָלְא֥וּ מַחֲשַׁכֵּי־ אֶ֝֗רֶץ נְא֣וֹת חָמָֽס׃ (נ) אַל־ יָשֹׁ֣ב דַּ֣ךְ נִכְלָ֑ם עָנִ֥י וְ֝אֶבְי֗וֹן יְֽהַלְל֥וּ שְׁמֶֽךָ׃ (נא) קוּמָ֣ה אֱ֭לֹהִים רִיבָ֣ה רִיבֶ֑ךָ זְכֹ֥ר חֶרְפָּתְךָ֥ מִנִּי־ נָ֝בָ֗ל כָּל־ הַיּֽוֹם׃ (נב) אַל־ תִּ֭שְׁכַּח ק֣וֹל צֹרְרֶ֑יךָ שְׁא֥וֹן קָ֝מֶ֗יךָ עֹלֶ֥ה תָמִֽיד׃ (נג) לַמְנַצֵּ֥חַ אַל־ תַּשְׁחֵ֑ת מִזְמ֖וֹר לְאָסָ֣ף שִֽׁיר׃ (נד) ה֘וֹדִ֤ינוּ לְּךָ֨ ׀ אֱ‍ֽלֹהִ֗ים ה֭וֹדִינוּ וְקָר֣וֹב שְׁמֶ֑ךָ סִ֝פְּר֗וּ נִפְלְאוֹתֶֽיךָ׃ (נה) כִּ֭י אֶקַּ֣ח מוֹעֵ֑ד אֲ֝נִ֗י מֵישָׁרִ֥ים אֶשְׁפֹּֽט׃ (נו) נְֽמֹגִ֗ים אֶ֥רֶץ וְכָל־ יֹשְׁבֶ֑יהָ אָנֹכִ֨י תִכַּ֖נְתִּי עַמּוּדֶ֣יהָ סֶּֽלָה׃ (נז) אָמַ֣רְתִּי לַֽ֭הוֹלְלִים אַל־ תָּהֹ֑לּוּ וְ֝לָרְשָׁעִ֗ים אַל־ תָּרִ֥ימוּ קָֽרֶן׃ (נח) אַל־ תָּרִ֣ימוּ לַמָּר֣וֹם קַרְנְכֶ֑ם תְּדַבְּר֖וּ בְצַוָּ֣אר עָתָֽק׃ (נט) כִּ֤י לֹ֣א מִ֭מּוֹצָא וּמִֽמַּעֲרָ֑ב וְ֝לֹ֗א מִמִּדְבַּ֥ר הָרִֽים׃ (ס) כִּֽי־ אֱלֹהִ֥ים שֹׁפֵ֑ט זֶ֥ה יַ֝שְׁפִּ֗יל וְזֶ֣ה יָרִֽים׃ (סא) כִּ֤י כ֪וֹס בְּֽיַד־ יְהוָ֡ה וְיַ֤יִן חָמַ֨ר ׀ מָ֥לֵא מֶסֶךְ֮ וַיַּגֵּ֪ר מִ֫זֶּ֥ה אַךְ־ שְׁ֭מָרֶיהָ יִמְצ֣וּ יִשְׁתּ֑וּ כֹּ֝֗ל רִשְׁעֵי־ אָֽרֶץ׃ (סב) וַ֭אֲנִי אַגִּ֣יד לְעֹלָ֑ם אֲ֝זַמְּרָ֗ה לֵאלֹהֵ֥י יַעֲקֹֽב׃ (סג) וְכָל־ קַרְנֵ֣י רְשָׁעִ֣ים אֲגַדֵּ֑עַ תְּ֝רוֹמַ֗מְנָה קַֽרְנ֥וֹת צַדִּֽיק׃ (סד) לַמְנַצֵּ֥חַ בִּנְגִינֹ֑ת מִזְמ֖וֹר לְאָסָ֣ף שִֽׁיר׃ (סה) נוֹדָ֣ע בִּֽיהוּדָ֣ה אֱלֹהִ֑ים בְּ֝יִשְׂרָאֵ֗ל גָּד֥וֹל שְׁמֽוֹ׃ (סו) וַיְהִ֣י בְשָׁלֵ֣ם סֻכּ֑וֹ וּמְע֖וֹנָת֣וֹ בְצִיּֽוֹן׃ (סז) שָׁ֭מָּה שִׁבַּ֣ר רִשְׁפֵי־ קָ֑שֶׁת מָגֵ֬ן לֵאלֹהִ֣ים עוּזֵּ֑נוּ הָ֝רִ֗יעוּ לֵאלֹהֵ֥י יַעֲקֹֽב׃ (עא) שְֽׂאוּ־ זִ֭מְרָה וּתְנוּ־ תֹ֑ף כִּנּ֖וֹר נָעִ֣ים עִם־ נָֽבֶל׃ (עב) תִּקְע֣וּ בַחֹ֣דֶשׁ שׁוֹפָ֑ר בַּ֝כֵּ֗סֶה לְי֣וֹם חַגֵּֽנוּ׃ (עג) כִּ֤י חֹ֣ק לְיִשְׂרָאֵ֣ל ה֑וּא מִ֝שְׁפָּ֗ט לֵאלֹהֵ֥י יַעֲקֹֽב׃ (עד) עֵ֤דוּת ׀ בִּֽיה֘וֹסֵ֤ף שָׂמ֗וֹ בְּ֭צֵאתוֹ עַל־ אֶ֣רֶץ מִצְרָ֑יִם שְׂפַ֖ת לֹא־ יָדַ֣עְתִּי אֶשְׁמָֽע׃ (עה) הֲסִיר֣וֹתִי מִסֵּ֣בֶל שִׁכְמ֑וֹ כַּ֝פָּ֗יו מִדּ֥וּד תַּעֲבֹֽרְנָה׃ (עו) בַּצָּרָ֥ה קָרָ֗אתָ וָאֲחַ֫לְּצֶ֥ךָּ אֶ֭עֶנְךָ בְּסֵ֣תֶר רַ֑עַם אֶבְחָֽנְךָ֨ עַל־ מֵ֖י מְרִיבָ֣ה סֶֽלָה׃ (עז) שְׁמַ֣ע עַ֭מִּי וְאָעִ֣ידָה בָּ֑ךְ יִ֝שְׂרָאֵ֗ל אִם־ תִּֽשְׁמַֽע־ לִֽי׃ (עח) לֹֽא־ יִהְיֶ֣ה בְ֭ךָ אֵ֣ל זָ֑ר וְלֹ֥א תִ֝שְׁתַּחֲוֶ֗ה לְאֵ֣ל נֵכָֽר׃ (עט) אָנֹכִ֨י ׀ יְה֘וָ֤ה אֱלֹהֶ֗יךָ הַֽ֭מַּעַלְךָ מֵאֶ֣רֶץ מִצְרָ֑יִם הַרְחֶב־ פִּ֝֗יךָ וַאֲמַלְאֵֽהוּ׃ (פ) וְלֹא־ שָׁמַ֣ע עַמִּ֣י לְקוֹלִ֑י וְ֝יִשְׂרָאֵ֗ל לֹא־ אָ֥בָה לִֽי׃ (פא) וָֽ֭אֲשַׁלְּחֵהוּ בִּשְׁרִיר֣וּת לִבָּ֑ם יֵ֝לְכ֗וּ בְּֽמוֹעֲצוֹתֵיהֶֽם׃ (פב) ל֗וּ עַ֭מִּי שֹׁמֵ֣עַֽ לִ֑י יִ֝שְׂרָאֵ֗ל בִּדְרָכַ֥י יְהַלֵּֽכוּ׃ (פג) כִּ֭מְעַט אוֹיְבֵיהֶ֣ם אַכְנִ֑יעַ וְעַ֥ל צָ֝רֵיהֶ֗ם אָשִׁ֥יב יָדִֽי׃ (פד) מְשַׂנְאֵ֣י יְ֭הוָה יְכַֽחֲשׁוּ־ ל֑וֹ וִיהִ֖י עִתָּ֣ם לְעוֹלָֽם׃ (פה) וַֽ֭יַּאֲכִילֵהוּ מֵחֵ֣לֶב חִטָּ֑ה וּ֝מִצּ֗וּר דְּבַ֣שׁ אַשְׂבִּיעֶֽךָ׃ (פו) מִזְמ֗וֹר לְאָ֫סָ֥ף אֱ‍ֽלֹהִ֗ים נִצָּ֥ב בַּעֲדַת־ אֵ֑ל בְּקֶ֖רֶב אֱלֹהִ֣ים יִשְׁפֹּֽט׃ (פז) עַד־ מָתַ֥י תִּשְׁפְּטוּ־ עָ֑וֶל וּפְנֵ֥י רְ֝שָׁעִ֗ים תִּשְׂאוּ־ סֶֽלָה׃ (פח) שִׁפְטוּ־ דַ֥ל וְיָת֑וֹם עָנִ֖י וָרָ֣שׁ הַצְדִּֽיקוּ׃ (פט) פַּלְּטוּ־ דַ֥ל וְאֶבְי֑וֹן מִיַּ֖ד רְשָׁעִ֣ים הַצִּֽילוּ׃ (צ) לֹ֤א יָֽדְע֨וּ ׀ וְלֹ֥א יָבִ֗ינוּ בַּחֲשֵׁכָ֥ה יִתְהַלָּ֑כוּ יִ֝מּ֗וֹטוּ כָּל־ מ֥וֹסְדֵי אָֽרֶץ׃ (צא) אֲ‍ֽנִי־ אָ֭מַרְתִּי אֱלֹהִ֣ים אַתֶּ֑ם וּבְנֵ֖י עֶלְי֣וֹן כֻּלְּכֶֽם׃ (צב) אָ֭כֵן כְּאָדָ֣ם תְּמוּת֑וּן וּכְאַחַ֖ד הַשָּׂרִ֣ים תִּפֹּֽלוּ׃ (צג) קוּמָ֣ה אֱ֭לֹהִים שָׁפְטָ֣ה הָאָ֑רֶץ כִּֽי־ אַתָּ֥ה תִ֝נְחַ֗ל בְּכָל־ הַגּוֹיִֽם׃ (צד) שִׁ֖יר מִזְמ֣וֹר לְאָסָֽף׃ (צה) אֱלֹהִ֥ים אַל־ דֳּמִי־ לָ֑ךְ אַל־ תֶּחֱרַ֖שׁ וְאַל־ תִּשְׁקֹ֣ט אֵֽל׃ (צו) כִּֽי־ הִנֵּ֣ה א֭וֹיְבֶיךָ יֶהֱמָי֑וּן וּ֝מְשַׂנְאֶ֗יךָ נָ֣שְׂאוּ רֹֽאשׁ׃ (צז) עַֽל־ עַ֭מְּךָ יַעֲרִ֣ימוּ ס֑וֹד וְ֝יִתְיָעֲצ֗וּ עַל־ צְפוּנֶֽיךָ׃ (צח) אָמְר֗וּ לְ֭כוּ וְנַכְחִידֵ֣ם מִגּ֑וֹי וְלֹֽא־ יִזָּכֵ֖ר שֵֽׁם־ יִשְׂרָאֵ֣ל עֽוֹד׃ (צט) כִּ֤י נוֹעֲצ֣וּ לֵ֣ב יַחְדָּ֑ו עָ֝לֶ֗יךָ בְּרִ֣ית יִכְרֹֽתוּ׃ (ק) אָהֳלֵ֣י אֱ֭דוֹם וְיִשְׁמְעֵאלִ֗ים מוֹאָ֥ב וְהַגְרִֽים׃ (קא) גְּבָ֣ל וְ֭עַמּוֹן וַעֲמָלֵ֑ק פְּ֝לֶ֗שֶׁת עִם־ יֹ֥שְׁבֵי צֽוֹר׃ (קב) גַּם־ אַ֭שּׁוּר נִלְוָ֣ה עִמָּ֑ם הָ֤י֥וּ זְר֖וֹעַ לִבְנֵי־ ל֣וֹט סֶֽלָה׃ (קג) עֲשֵֽׂה־ לָהֶ֥ם כְּמִדְיָ֑ן כְּֽסִֽיסְרָ֥א כְ֝יָבִ֗ין בְּנַ֣חַל קִישֽׁוֹן׃ (קד) נִשְׁמְד֥וּ בְֽעֵין־ דֹּ֑אר הָ֥יוּ דֹּ֝֗מֶן לָאֲדָמָֽה׃ (קה) שִׁיתֵ֣מוֹ נְ֭דִיבֵמוֹ כְּעֹרֵ֣ב וְכִזְאֵ֑ב וּֽכְזֶ֥בַח וּ֝כְצַלְמֻנָּ֗ע כָּל־ נְסִיכֵֽמוֹ׃ (קו) אֲשֶׁ֣ר אָ֭מְרוּ נִ֣ירֲשָׁה לָּ֑נוּ אֵ֝֗ת נְא֣וֹת אֱלֹהִֽים׃ (קז) אֱ‍ֽלֹהַ֗י שִׁיתֵ֥מוֹ כַגַּלְגַּ֑ל כְּ֝קַ֗שׁ לִפְנֵי־ רֽוּחַ׃ (קח) כְּאֵ֥שׁ תִּבְעַר־ יָ֑עַר וּ֝כְלֶהָבָ֗ה תְּלַהֵ֥ט הָרִֽים׃ (קט) כֵּ֭ן תִּרְדְּפֵ֣ם בְּסַעֲרֶ֑ךָ וּבְסוּפָתְךָ֥ תְבַהֲלֵֽם׃ (קי) מַלֵּ֣א פְנֵיהֶ֣ם קָל֑וֹן וִֽיבַקְשׁ֖וּ שִׁמְךָ֣ יְהוָֽה׃ (קיא) יֵבֹ֖שׁוּ וְיִבָּהֲל֥וּ עֲדֵי־ עַ֗ד וְֽיַחְפְּר֥וּ וְיֹאבֵֽדוּ׃ (קיב) וְֽיֵדְע֗וּ כִּֽי־ אַתָּ֬ה שִׁמְךָ֣ יְהוָ֣ה לְבַדֶּ֑ךָ עֶ֝לְי֗וֹן עַל־ כָּל־ הָאָֽרֶץ׃ (קיג) לַמְנַצֵּ֥חַ עַֽל־ הַגִּתִּ֑ית לִבְנֵי־ קֹ֥רַח מִזְמֽוֹר׃ (קיד) מַה־ יְּדִיד֥וֹת מִשְׁכְּנוֹתֶ֗יךָ יְהוָ֥ה צְבָאֽוֹת׃ (קטו) נִכְסְפָ֬ה וְגַם־ כָּלְתָ֨ה ׀ נַפְשִׁי֮ לְחַצְר֪וֹת יְה֫וָ֥ה לִבִּ֥י וּבְשָׂרִ֑י יְ֝רַנְּנ֗וּ אֶ֣ל אֵֽל־ חָֽי׃ (קטז) גַּם־ צִפּ֨וֹר ׀ מָ֪צְאָה בַ֡יִת וּדְר֤וֹר ׀ קֵ֥ן לָהּ֮ אֲשֶׁר־ שָׁ֪תָה אֶפְרֹ֫חֶ֥יהָ אֶֽת־ מִ֭זְבְּחוֹתֶיךָ יְהוָ֣ה צְבָא֑וֹת מַ֝לְכִּ֗י וֵאלֹהָֽי׃ (קיז) אַ֭שְׁרֵי יוֹשְׁבֵ֣י בֵיתֶ֑ךָ ע֝֗וֹד יְֽהַלְל֥וּךָ סֶּֽלָה׃ (קיח) אַשְׁרֵ֣י אָ֭דָם עֽוֹז־ ל֥וֹ בָ֑ךְ מְ֝סִלּ֗וֹת בִּלְבָבָֽם׃ (קיט) עֹבְרֵ֤י ׀ בְּעֵ֣מֶק הַ֭בָּכָא מַעְיָ֣ן יְשִׁית֑וּהוּ גַּם־ בְּ֝רָכ֗וֹת יַעְטֶ֥ה מוֹרֶֽה׃ (קכ) יֵ֭לְכוּ מֵחַ֣יִל אֶל־ חָ֑יִל יֵרָאֶ֖ה אֶל־ אֱלֹהִ֣ים בְּצִיּֽוֹן׃ (קכא) יְה֘וָ֤ה אֱלֹהִ֣ים צְ֭בָאוֹת שִׁמְעָ֣ה תְפִלָּתִ֑י הַאֲזִ֨ינָה אֱלֹהֵ֖י יַעֲקֹ֣ב סֶֽלָה׃ (קכב) מָ֭גִנֵּנוּ רְאֵ֣ה אֱלֹהִ֑ים וְ֝הַבֵּ֗ט פְּנֵ֣י מְשִׁיחֶֽךָ׃ (קכג) כִּ֤י טֽוֹב־ י֥וֹם בַּחֲצֵרֶ֗יךָ מֵ֫אָ֥לֶף בָּחַ֗רְתִּי הִ֭סְתּוֹפֵף בְּבֵ֣ית אֱלֹהַ֑י מִ֝דּ֗וּר בְּאָהֳלֵי־ רֶֽשַׁע׃ (קכד) כִּ֤י שֶׁ֨מֶשׁ ׀ וּמָגֵן֮ יְהוָ֪ה אֱלֹ֫הִ֥ים חֵ֣ן וְ֭כָבוֹד יִתֵּ֣ן יְהוָ֑ה לֹ֥א יִמְנַע־ ט֝֗וֹב לַֽהֹלְכִ֥ים בְּתָמִֽים׃
