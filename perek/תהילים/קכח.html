תהילים פרק קכח לפי החלוקה לסדרים

פרק קכח בספר תהילים (6 פסוקים) נמצא בתהילים סדר יז (פסוקים קנח–קסב) ובתהילים סדר יח (פסוקים א–א).

תהילים סדר יז

(א) טֽוֹב־ לִ֥י תֽוֹרַת־ פִּ֑יךָ מֵ֝אַלְפֵ֗י זָהָ֥ב וָכָֽסֶף׃ (ב) יָדֶ֣יךָ עָ֭שׂוּנִי וַֽיְכוֹנְנ֑וּנִי הֲ֝בִינֵ֗נִי וְאֶלְמְדָ֥ה מִצְוֺתֶֽיךָ׃ (ג) יְ֭רֵאֶיךָ יִרְא֣וּנִי וְיִשְׂמָ֑חוּ כִּ֖י לִדְבָרְךָ֣ יִחָֽלְתִּי׃ (ד) יָדַ֣עְתִּי יְ֭הוָה כִּי־ צֶ֣דֶק מִשְׁפָּטֶ֑יךָ וֶ֝אֱמוּנָ֗ה עִנִּיתָֽנִי׃ (ה) יְהִי־ נָ֣א חַסְדְּךָ֣ לְנַחֲמֵ֑נִי כְּאִמְרָתְךָ֥ לְעַבְדֶּֽךָ׃ (ו) יְבֹא֣וּנִי רַחֲמֶ֣יךָ וְאֶֽחְיֶ֑ה כִּי־ תֽ֝וֹרָתְךָ֗ שַֽׁעֲשֻׁעָֽי׃ (ז) יֵבֹ֣שׁוּ זֵ֭דִים כִּי־ שֶׁ֣קֶר עִוְּת֑וּנִי אֲ֝נִ֗י אָשִׂ֥יחַ בְּפִקּוּדֶֽיךָ׃ (ח) יָשׁ֣וּבוּ לִ֣י יְרֵאֶ֑יךָ (וידעו) וְ֝יֹדְעֵ֗י עֵדֹתֶֽיךָ׃ (ט) יְהִֽי־ לִבִּ֣י תָמִ֣ים בְּחֻקֶּ֑יךָ לְ֝מַ֗עַן לֹ֣א אֵבֽוֹשׁ׃ (י) כָּלְתָ֣ה לִתְשׁוּעָתְךָ֣ נַפְשִׁ֑י לִדְבָרְךָ֥ יִחָֽלְתִּי׃ (יא) כָּל֣וּ עֵ֭ינַי לְאִמְרָתֶ֑ךָ לֵ֝אמֹ֗ר מָתַ֥י תְּֽנַחֲמֵֽנִי׃ (יב) כִּֽי־ הָ֭יִיתִי כְּנֹ֣אד בְּקִיט֑וֹר חֻ֝קֶּ֗יךָ לֹ֣א שָׁכָֽחְתִּי׃ (יג) כַּמָּ֥ה יְמֵֽי־ עַבְדֶּ֑ךָ מָתַ֬י תַּעֲשֶׂ֖ה בְרֹדְפַ֣י מִשְׁפָּֽט׃ (יד) כָּֽרוּ־ לִ֣י זֵדִ֣ים שִׁיח֑וֹת אֲ֝שֶׁ֗ר לֹ֣א כְתוֹרָתֶֽךָ׃ (טו) כָּל־ מִצְוֺתֶ֥יךָ אֱמוּנָ֑ה שֶׁ֖קֶר רְדָפ֣וּנִי עָזְרֵֽנִי׃ (טז) כִּ֭מְעַט כִּלּ֣וּנִי בָאָ֑רֶץ וַ֝אֲנִ֗י לֹא־ עָזַ֥בְתִּי פִקֻּודֶֽיךָ׃ (יז) כְּחַסְדְּךָ֥ חַיֵּ֑נִי וְ֝אֶשְׁמְרָ֗ה עֵד֥וּת פִּֽיךָ׃ (יח) לְעוֹלָ֥ם יְהוָ֑ה דְּ֝בָרְךָ֗ נִצָּ֥ב בַּשָּׁמָֽיִם׃ (יט) לְדֹ֣ר וָ֭דֹר אֱמֽוּנָתֶ֑ךָ כּוֹנַ֥נְתָּ אֶ֝֗רֶץ וַֽתַּעֲמֹֽד׃ (כ) לְֽ֭מִשְׁפָּטֶיךָ עָמְד֣וּ הַיּ֑וֹם כִּ֖י הַכֹּ֣ל עֲבָדֶֽיךָ׃ (כא) לוּלֵ֣י ת֭וֹרָתְךָ שַׁעֲשֻׁעָ֑י אָ֝֗ז אָבַ֥דְתִּי בְעָנְיִֽי׃ (כב) לְ֭עוֹלָם לֹא־ אֶשְׁכַּ֣ח פִּקּוּדֶ֑יךָ כִּ֥י בָ֝֗ם חִיִּיתָֽנִי׃ (כג) לְֽךָ־ אֲ֭נִי הוֹשִׁיעֵ֑נִי כִּ֖י פִקּוּדֶ֣יךָ דָרָֽשְׁתִּי׃ (כד) לִ֤י קִוּ֣וּ רְשָׁעִ֣ים לְאַבְּדֵ֑נִי עֵ֝דֹתֶ֗יךָ אֶתְבּוֹנָֽן׃ (כה) לְֽכָל תִּ֭כְלָה רָאִ֣יתִי קֵ֑ץ רְחָבָ֖ה מִצְוָתְךָ֣ מְאֹֽד׃ (כו) מָֽה־ אָהַ֥בְתִּי תוֹרָתֶ֑ךָ כָּל־ הַ֝יּ֗וֹם הִ֣יא שִׂיחָתִֽי׃ (כז) מֵ֭אֹ֣יְבַי תְּחַכְּמֵ֣נִי מִצְוֺתֶ֑ךָ כִּ֖י לְעוֹלָ֣ם הִיא־ לִֽי׃ (כח) מִכָּל־ מְלַמְּדַ֥י הִשְׂכַּ֑לְתִּי כִּ֥י עֵ֝דְוֺתֶ֗יךָ שִׂ֣יחָה לִֽֿי׃ (כט) מִזְּקֵנִ֥ים אֶתְבּוֹנָ֑ן כִּ֖י פִקּוּדֶ֣יךָ נָצָֽרְתִּי׃ (ל) מִכָּל־ אֹ֣רַח רָ֭ע כָּלִ֣אתִי רַגְלָ֑י לְ֝מַ֗עַן אֶשְׁמֹ֥ר דְּבָרֶֽךָ׃ (לא) מִמִּשְׁפָּטֶ֥יךָ לֹא־ סָ֑רְתִּי כִּֽי־ אַ֝תָּ֗ה הוֹרֵתָֽנִי׃ (לב) מַה־ נִּמְלְצ֣וּ לְ֭חִכִּי אִמְרָתֶ֗ךָ מִדְּבַ֥שׁ לְפִֽי׃ (לג) מִפִּקּוּדֶ֥יךָ אֶתְבּוֹנָ֑ן עַל־ כֵּ֝֗ן שָׂנֵ֤אתִי ׀ כָּל־ אֹ֬רַח שָֽׁקֶר׃ (לד) נֵר־ לְרַגְלִ֥י דְבָרֶ֑ךָ וְ֝א֗וֹר לִנְתִיבָתִֽי׃ (לה) נִשְׁבַּ֥עְתִּי וָאֲקַיֵּ֑מָה לִ֝שְׁמֹ֗ר מִשְׁפְּטֵ֥י צִדְקֶֽךָ׃ (לו) נַעֲנֵ֥יתִי עַד־ מְאֹ֑ד יְ֝הוָ֗ה חַיֵּ֥נִי כִדְבָרֶֽךָ׃ (לז) נִדְב֣וֹת פִּ֭י רְצֵה־ נָ֣א יְהוָ֑ה וּֽמִשְׁפָּטֶ֥יךָ לַמְּדֵֽנִי׃ (לח) נַפְשִׁ֣י בְכַפִּ֣י תָמִ֑יד וְ֝תֽוֹרָתְךָ֗ לֹ֣א שָׁכָֽחְתִּי׃ (לט) נָתְנ֬וּ רְשָׁעִ֣ים פַּ֣ח לִ֑י וּ֝מִפִּקּוּדֶ֗יךָ לֹ֣א תָעִֽיתִי׃ (מ) נָחַ֣לְתִּי עֵדְוֺתֶ֣יךָ לְעוֹלָ֑ם כִּֽי־ שְׂשׂ֖וֹן לִבִּ֣י הֵֽמָּה׃ (מא) נָטִ֣יתִי לִ֭בִּי לַעֲשׂ֥וֹת חֻקֶּ֗יךָ לְעוֹלָ֥ם עֵֽקֶב׃ (מב) סֵעֲפִ֥ים שָׂנֵ֑אתִי וְֽתוֹרָתְךָ֥ אָהָֽבְתִּי׃ (מג) סִתְרִ֣י וּמָגִנִּ֣י אָ֑תָּה לִדְבָרְךָ֥ יִחָֽלְתִּי׃ (מד) סֽוּרוּ־ מִמֶּ֥נִּי מְרֵעִ֑ים וְ֝אֶצְּרָ֗ה מִצְוֺ֥ת אֱלֹהָֽי׃ (מה) סָמְכֵ֣נִי כְאִמְרָתְךָ֣ וְאֶֽחְיֶ֑ה וְאַל־ תְּ֝בִישֵׁ֗נִי מִשִּׂבְרִֽי׃ (מו) סְעָדֵ֥נִי וְאִוָּשֵׁ֑עָה וְאֶשְׁעָ֖ה בְחֻקֶּ֣יךָ תָמִֽיד׃ (מז) סָ֭לִיתָ כָּל־ שׁוֹגִ֣ים מֵחֻקֶּ֑יךָ כִּי־ שֶׁ֝֗קֶר תַּרְמִיתָֽם׃ (מח) סִגִ֗ים הִשְׁבַּ֥תָּ כָל־ רִשְׁעֵי־ אָ֑רֶץ לָ֝כֵ֗ן אָהַ֥בְתִּי עֵדֹתֶֽיךָ׃ (מט) סָמַ֣ר מִפַּחְדְּךָ֣ בְשָׂרִ֑י וּֽמִמִּשְׁפָּטֶ֥יךָ יָרֵֽאתִי׃ (נ) עָ֭שִׂיתִי מִשְׁפָּ֣ט וָצֶ֑דֶק בַּל־ תַּ֝נִּיחֵ֗נִי לְעֹֽשְׁקָֽי׃ (נא) עֲרֹ֣ב עַבְדְּךָ֣ לְט֑וֹב אַֽל־ יַעַשְׁקֻ֥נִי זֵדִֽים׃ (נב) עֵ֭ינַי כָּל֣וּ לִֽישׁוּעָתֶ֑ךָ וּלְאִמְרַ֥ת צִדְקֶֽךָ׃ (נג) עֲשֵׂ֖ה עִם־ עַבְדְּךָ֥ כְחַסְדֶּ֗ךָ וְחֻקֶּ֥יךָ לַמְּדֵֽנִי׃ (נד) עַבְדְּךָ־ אָ֥נִי הֲבִינֵ֑נִי וְ֝אֵדְעָ֗ה עֵדֹתֶֽיךָ׃ (נה) עֵ֭ת לַעֲשׂ֣וֹת לַיהוָ֑ה הֵ֝פֵ֗רוּ תּוֹרָתֶֽךָ׃ (נו) עַל־ כֵּ֭ן אָהַ֣בְתִּי מִצְוֺתֶ֑יךָ מִזָּהָ֥ב וּמִפָּֽז׃ (נז) עַל־ כֵּ֤ן ׀ כָּל־ פִּקּ֣וּדֵי כֹ֣ל יִשָּׁ֑רְתִּי כָּל־ אֹ֖רַח שֶׁ֣קֶר שָׂנֵֽאתִי׃ (נח) פְּלָא֥וֹת עֵדְוֺתֶ֑יךָ עַל־ כֵּ֝֗ן נְצָרָ֥תַם נַפְשִֽׁי׃ (נט) פֵּ֖תַח דְּבָרֶ֥יךָ יָאִ֗יר מֵבִ֥ין פְּתָיִֽים׃ (ס) פִּֽי־ פָ֭עַרְתִּי וָאֶשְׁאָ֑פָה כִּ֖י לְמִצְוֺתֶ֣יךָ יָאָֽבְתִּי׃ (סא) פְּנֵה־ אֵלַ֥י וְחָנֵּ֑נִי כְּ֝מִשְׁפָּ֗ט לְאֹהֲבֵ֥י שְׁמֶֽךָ׃ (סב) פְּ֭עָמַי הָכֵ֣ן בְּאִמְרָתֶ֑ךָ וְֽאַל־ תַּשְׁלֶט־ בִּ֥י כָל־ אָֽוֶן׃ (סג) פְּ֭דֵנִי מֵעֹ֣שֶׁק אָדָ֑ם וְ֝אֶשְׁמְרָ֗ה פִּקּוּדֶֽיךָ׃ (סד) פָּ֭נֶיךָ הָאֵ֣ר בְּעַבְדֶּ֑ךָ וְ֝לַמְּדֵ֗נִי אֶת־ חֻקֶּֽיךָ׃ (סה) פַּלְגֵי־ מַ֭יִם יָרְד֣וּ עֵינָ֑י עַ֝֗ל לֹא־ שָׁמְר֥וּ תוֹרָתֶֽךָ׃ (סו) צַדִּ֣יק אַתָּ֣ה יְהוָ֑ה וְ֝יָשָׁ֗ר מִשְׁפָּטֶֽיךָ׃ (סז) צִ֭וִּיתָ צֶ֣דֶק עֵדֹתֶ֑יךָ וֶֽאֱמוּנָ֥ה מְאֹֽד׃ (סח) צִמְּתַ֥תְנִי קִנְאָתִ֑י כִּֽי־ שָׁכְח֖וּ דְבָרֶ֣יךָ צָרָֽי׃ (סט) צְרוּפָ֖ה אִמְרָתְךָ֥ מְאֹ֗ד וְֽעַבְדְּךָ֥ אֲהֵבָֽהּ׃ (ע) צָעִ֣יר אָנֹכִ֣י וְנִבְזֶ֑ה פִּ֝קֻּדֶ֗יךָ לֹ֣א שָׁכָֽחְתִּי׃ (עא) צִדְקָתְךָ֣ צֶ֣דֶק לְעוֹלָ֑ם וְֽתוֹרָתְךָ֥ אֱמֶֽת׃ (עב) צַר־ וּמָצ֥וֹק מְצָא֑וּנִי מִ֝צְוֺתֶ֗יךָ שַׁעֲשֻׁעָֽי׃ (עג) צֶ֖דֶק עֵדְוֺתֶ֥יךָ לְעוֹלָ֗ם הֲבִינֵ֥נִי וְאֶחְיֶֽה׃ (עד) קָרָ֣אתִי בְכָל־ לֵ֭ב עֲנֵ֥נִי יְהוָ֗ה חֻקֶּ֥יךָ אֶצֹּֽרָה׃ (עה) קְרָאתִ֥יךָ הוֹשִׁיעֵ֑נִי וְ֝אֶשְׁמְרָ֗ה עֵדֹתֶֽיךָ׃ (עו) קִדַּ֣מְתִּי בַ֭נֶּשֶׁף וָאֲשַׁוֵּ֑עָה (לדבריך) לִדְבָרְךָ֥ יִחָֽלְתִּי׃ (עז) גָּלְמִ֤י ׀ רָ֘א֤וּ עֵינֶ֗יךָ וְעַֽל־ סִפְרְךָ֮ כֻּלָּ֪ם יִכָּ֫תֵ֥בוּ יָמִ֥ים יֻצָּ֑רוּ (ולא) וְל֖וֹ אֶחָ֣ד בָּהֶֽם׃ (קכה) וְלִ֗י מַה־ יָּקְר֣וּ רֵעֶ֣יךָ אֵ֑ל מֶ֥ה עָ֝צְמוּ רָאשֵׁיהֶֽם׃ (קכו) אֶ֭סְפְּרֵם מֵח֣וֹל יִרְבּ֑וּן הֱ֝קִיצֹ֗תִי וְעוֹדִ֥י עִמָּֽךְ׃ (קכז) אִם־ תִּקְטֹ֖ל אֱל֥וֹהַּ ׀ רָשָׁ֑ע וְאַנְשֵׁ֥י דָ֝מִ֗ים ס֣וּרוּ מֶֽנִּי׃ (קכח) אֲשֶׁ֣ר יֹ֭אמְרֻךָ לִמְזִמָּ֑ה נָשֻׂ֖א לַשָּׁ֣וְא עָרֶֽיךָ׃ (קכט) הֲלֽוֹא־ מְשַׂנְאֶ֖יךָ יְהוָ֥ה ׀ אֶשְׂנָ֑א וּ֝בִתְקוֹמְמֶ֗יךָ אֶתְקוֹטָֽט׃ (קל) תַּכְלִ֣ית שִׂנְאָ֣ה שְׂנֵאתִ֑ים לְ֝אוֹיְבִ֗ים הָ֣יוּ לִֽי׃ (קלא) חָקְרֵ֣נִי אֵ֭ל וְדַ֣ע לְבָבִ֑י בְּ֝חָנֵ֗נִי וְדַ֣ע שַׂרְעַפָּֽי׃ (קלב) וּרְאֵ֗ה אִם־ דֶּֽרֶךְ־ עֹ֥צֶב בִּ֑י וּ֝נְחֵ֗נִי בְּדֶ֣רֶךְ עוֹלָֽם׃ (קלג) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (קלד) חַלְּצֵ֣נִי יְ֭הוָה מֵאָדָ֣ם רָ֑ע מֵאִ֖ישׁ חֲמָסִ֣ים תִּנְצְרֵֽנִי׃ (קלה) אֲשֶׁ֤ר חָשְׁב֣וּ רָע֣וֹת בְּלֵ֑ב כָּל־ י֝֗וֹם יָג֥וּרוּ מִלְחָמֽוֹת׃ (קלו) שָֽׁנֲנ֣וּ לְשׁוֹנָם֮ כְּֽמוֹ־ נָ֫חָ֥שׁ חֲמַ֥ת עַכְשׁ֑וּב תַּ֖חַת שְׂפָתֵ֣ימוֹ סֶֽלָה׃ (קלז) שָׁמְרֵ֤נִי יְהוָ֨ה ׀ מִ֘ידֵ֤י רָשָׁ֗ע מֵאִ֣ישׁ חֲמָסִ֣ים תִּנְצְרֵ֑נִי אֲשֶׁ֥ר חָ֝שְׁב֗וּ לִדְח֥וֹת פְּעָמָֽי׃ (קלח) טָֽמְנֽוּ־ גֵאִ֨ים ׀ פַּ֡ח לִ֗י וַחֲבָלִ֗ים פָּ֣רְשׂוּ רֶ֭שֶׁת לְיַד־ מַעְגָּ֑ל מֹקְשִׁ֖ים שָֽׁתוּ־ לִ֣י סֶֽלָה׃ (קלט) אָמַ֣רְתִּי לַ֭יהוָה אֵ֣לִי אָ֑תָּה הַאֲזִ֥ינָה יְ֝הוָ֗ה ק֣וֹל תַּחֲנוּנָֽי׃ (קמ) יְהֹוִ֣ה אֲ֭דֹנָי עֹ֣ז יְשׁוּעָתִ֑י סַכֹּ֥תָה לְ֝רֹאשִׁ֗י בְּי֣וֹם נָֽשֶׁק׃ (קמא) אַל־ תִּתֵּ֣ן יְ֭הוָה מַאֲוַיֵּ֣י רָשָׁ֑ע זְמָמ֥וֹ אַל־ תָּ֝פֵ֗ק יָר֥וּמוּ סֶֽלָה׃ (קמב) רֹ֥אשׁ מְסִבָּ֑י עֲמַ֖ל שְׂפָתֵ֣ימוֹ (יכסומו) יְכַסֵּֽמוֹ׃ (קמג) (ימיטו) יִמּ֥וֹטוּ עֲלֵיהֶ֗ם גֶּֽחָ֫לִ֥ים בָּאֵ֥שׁ יַפִּלֵ֑ם בְּ֝מַהֲמֹר֗וֹת בַּֽל־ יָקֽוּמוּ׃ (קמד) אִ֥ישׁ לָשׁוֹן֮ בַּל־ יִכּ֪וֹן בָּ֫אָ֥רֶץ אִישׁ־ חָמָ֥ס רָ֑ע יְ֝צוּדֶ֗נּוּ לְמַדְחֵפֹֽת׃ (קמה) (ידעת) יָדַ֗עְתִּי כִּֽי־ יַעֲשֶׂ֣ה יְ֭הוָה דִּ֣ין עָנִ֑י מִ֝שְׁפַּ֗ט אֶבְיֹנִֽים׃
