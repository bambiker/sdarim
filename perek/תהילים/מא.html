פרק מא בספר תהילים (14 פסוקים) נמצא בתהילים סדר ה (פסוקים קי–קכב) ובתהילים סדר ו (פסוקים א–א).

תהילים סדר ה

(א) וּ֭לְשׁוֹנִי תֶּהְגֶּ֣ה צִדְקֶ֑ךָ כָּל־ הַ֝יּוֹם תְּהִלָּתֶֽךָ׃ (ב) לַמְנַצֵּ֬חַ ׀ לְעֶֽבֶד־ יְהוָ֬ה לְדָוִֽד׃ (ג) נְאֻֽם־ פֶּ֣שַׁע לָ֭רָשָׁע בְּקֶ֣רֶב לִבִּ֑י אֵֽין־ פַּ֥חַד אֱ֝לֹהִ֗ים לְנֶ֣גֶד עֵינָֽיו׃ (ד) כִּֽי־ הֶחֱלִ֣יק אֵלָ֣יו בְּעֵינָ֑יו לִמְצֹ֖א עֲוֺנ֣וֹ לִשְׂנֹֽא׃ (ה) דִּבְרֵי־ פִ֭יו אָ֣וֶן וּמִרְמָ֑ה חָדַ֖ל לְהַשְׂכִּ֣יל לְהֵיטִֽיב׃ (ו) אָ֤וֶן ׀ יַחְשֹׁ֗ב עַֽל־ מִשְׁכָּ֫ב֥וֹ יִ֭תְיַצֵּב עַל־ דֶּ֣רֶךְ לֹא־ ט֑וֹב רָ֝֗ע לֹ֣א יִמְאָֽס׃ (ז) יְ֭הוָה בְּהַשָּׁמַ֣יִם חַסְדֶּ֑ךָ אֱ֝מֽוּנָתְךָ֗ עַד־ שְׁחָקִֽים׃ (ח) צִדְקָֽתְךָ֨ ׀ כְּֽהַרְרֵי־ אֵ֗ל מִ֭שְׁפָּטֶךָ תְּה֣וֹם רַבָּ֑ה אָ֤דָֽם־ וּבְהֵמָ֖ה תוֹשִׁ֣יעַ יְהוָֽה׃ (ט) מַה־ יָּקָ֥ר חַסְדְּךָ֗ אֱלֹ֫הִ֥ים וּבְנֵ֥י אָדָ֑ם בְּצֵ֥ל כְּ֝נָפֶ֗יךָ יֶחֱסָיֽוּן׃ (י) יִ֭רְוְיֻן מִדֶּ֣שֶׁן בֵּיתֶ֑ךָ וְנַ֖חַל עֲדָנֶ֣יךָ תַשְׁקֵֽם׃ (יא) כִּֽי־ עִ֭מְּךָ מְק֣וֹר חַיִּ֑ים בְּ֝אוֹרְךָ֗ נִרְאֶה־ אֽוֹר׃ (יב) מְשֹׁ֣ךְ חַ֭סְדְּךָ לְיֹדְעֶ֑יךָ וְ֝צִדְקָֽתְךָ֗ לְיִשְׁרֵי־ לֵֽב׃ (יג) אַל־ תְּ֭בוֹאֵנִי רֶ֣גֶל גַּאֲוָ֑ה וְיַד־ רְ֝שָׁעִ֗ים אַל־ תְּנִדֵֽנִי׃ (יד) שָׁ֣ם נָ֭פְלוּ פֹּ֣עֲלֵי אָ֑וֶן דֹּ֝ח֗וּ וְלֹא־ יָ֥כְלוּ קֽוּם׃ (טו) לְדָוִ֨ד ׀ אַל־ תִּתְחַ֥ר בַּמְּרֵעִ֑ים אַל־ תְּ֝קַנֵּ֗א בְּעֹשֵׂ֥י עַוְלָֽה׃ (טז) כִּ֣י כֶ֭חָצִיר מְהֵרָ֣ה יִמָּ֑לוּ וּכְיֶ֥רֶק דֶּ֝֗שֶׁא יִבּוֹלֽוּן׃ (יז) בְּטַ֣ח בַּֽ֭יהוָה וַעֲשֵׂה־ ט֑וֹב שְׁכָן־ אֶ֝֗רֶץ וּרְעֵ֥ה אֱמוּנָֽה׃ (יח) וְהִתְעַנַּ֥ג עַל־ יְהוָ֑ה וְיִֽתֶּן־ לְ֝ךָ֗ מִשְׁאֲלֹ֥ת לִבֶּֽךָ׃ (יט) גּ֣וֹל עַל־ יְהוָ֣ה דַּרְכֶּ֑ךָ וּבְטַ֥ח עָ֝לָ֗יו וְה֣וּא יַעֲשֶֽׂה׃ (כ) וְהוֹצִ֣יא כָא֣וֹר צִדְקֶ֑ךָ וּ֝מִשְׁפָּטֶ֗ךָ כַּֽצָּהֳרָֽיִם׃ (כא) דּ֤וֹם ׀ לַיהוָה֮ וְהִתְח֪וֹלֵ֫ל ל֥וֹ אַל־ תִּ֭תְחַר בְּמַצְלִ֣יחַ דַּרְכּ֑וֹ בְּ֝אִ֗ישׁ עֹשֶׂ֥ה מְזִמּֽוֹת׃ (כב) הֶ֣רֶף מֵ֭אַף וַעֲזֹ֣ב חֵמָ֑ה אַל־ תִּ֝תְחַ֗ר אַךְ־ לְהָרֵֽעַ׃ (כג) כִּֽי־ מְ֭רֵעִים יִכָּרֵת֑וּן וְקֹוֵ֥י יְ֝הוָ֗ה הֵ֣מָּה יִֽירְשׁוּ־ אָֽרֶץ׃ (כד) וְע֣וֹד מְ֭עַט וְאֵ֣ין רָשָׁ֑ע וְהִתְבּוֹנַ֖נְתָּ עַל־ מְקוֹמ֣וֹ וְאֵינֶֽנּוּ׃ (כה) וַעֲנָוִ֥ים יִֽירְשׁוּ־ אָ֑רֶץ וְ֝הִתְעַנְּג֗וּ עַל־ רֹ֥ב שָׁלֽוֹם׃ (כו) זֹמֵ֣ם רָ֭שָׁע לַצַּדִּ֑יק וְחֹרֵ֖ק עָלָ֣יו שִׁנָּֽיו׃ (כז) אֲדֹנָ֥י יִשְׂחַק־ ל֑וֹ כִּֽי־ רָ֝אָ֗ה כִּֽי־ יָבֹ֥א יוֹמֽוֹ׃ (כח) חֶ֤רֶב ׀ פָּֽתְח֣וּ רְשָׁעִים֮ וְדָרְכ֪וּ קַ֫שְׁתָּ֥ם לְ֭הַפִּיל עָנִ֣י וְאֶבְי֑וֹן לִ֝טְב֗וֹחַ יִשְׁרֵי־ דָֽרֶךְ׃ (כט) חַ֭רְבָּם תָּב֣וֹא בְלִבָּ֑ם וְ֝קַשְּׁתוֹתָ֗ם תִּשָּׁבַֽרְנָה׃ (ל) טוֹב־ מְ֭עַט לַצַּדִּ֑יק מֵ֝הֲמ֗וֹן רְשָׁעִ֥ים רַבִּֽים׃ (לא) כִּ֤י זְרוֹע֣וֹת רְ֭שָׁעִים תִּשָּׁבַ֑רְנָה וְסוֹמֵ֖ךְ צַדִּיקִ֣ים יְהוָֽה׃ (לב) יוֹדֵ֣עַ יְ֭הוָה יְמֵ֣י תְמִימִ֑ם וְ֝נַחֲלָתָ֗ם לְעוֹלָ֥ם תִּהְיֶֽה׃ (לג) לֹֽא־ יֵ֭בֹשׁוּ בְּעֵ֣ת רָעָ֑ה וּבִימֵ֖י רְעָב֣וֹן יִשְׂבָּֽעוּ׃ (לד) כִּ֤י רְשָׁעִ֨ים ׀ יֹאבֵ֗דוּ וְאֹיְבֵ֣י יְ֭הוָה כִּיקַ֣ר כָּרִ֑ים כָּל֖וּ בֶעָשָׁ֣ן כָּֽלוּ׃ (לה) לֹוֶ֣ה רָ֭שָׁע וְלֹ֣א יְשַׁלֵּ֑ם וְ֝צַדִּ֗יק חוֹנֵ֥ן וְנוֹתֵֽן׃ (לו) כִּ֣י מְ֭בֹרָכָיו יִ֣ירְשׁוּ אָ֑רֶץ וּ֝מְקֻלָּלָ֗יו יִכָּרֵֽתוּ׃ (לז) מֵ֭יְהוָה מִֽצְעֲדֵי־ גֶ֥בֶר כּוֹנָ֗נוּ וְדַרְכּ֥וֹ יֶחְפָּֽץ׃ (לח) כִּֽי־ יִפֹּ֥ל לֹֽא־ יוּטָ֑ל כִּֽי־ יְ֝הוָ֗ה סוֹמֵ֥ךְ יָדֽוֹ׃ (לט) נַ֤עַר ׀ הָיִ֗יתִי גַּם־ זָ֫קַ֥נְתִּי וְֽלֹא־ רָ֭אִיתִי צַדִּ֣יק נֶעֱזָ֑ב וְ֝זַרְע֗וֹ מְבַקֶּשׁ־ לָֽחֶם׃ (מ) כָּל־ הַ֭יּוֹם חוֹנֵ֣ן וּמַלְוֶ֑ה וְ֝זַרְע֗וֹ לִבְרָכָֽה׃ (מא) ס֣וּר מֵ֭רָע וַעֲשֵׂה־ ט֗וֹב וּשְׁכֹ֥ן לְעוֹלָֽם׃ (מב) כִּ֤י יְהוָ֨ה ׀ אֹ֘הֵ֤ב מִשְׁפָּ֗ט וְלֹא־ יַעֲזֹ֣ב אֶת־ חֲ֭סִידָיו לְעוֹלָ֣ם נִשְׁמָ֑רוּ וְזֶ֖רַע רְשָׁעִ֣ים נִכְרָֽת׃ (מג) צַדִּיקִ֥ים יִֽירְשׁוּ־ אָ֑רֶץ וְיִשְׁכְּנ֖וּ לָעַ֣ד עָלֶֽיהָ׃ (מד) פִּֽי־ צַ֭דִּיק יֶהְגֶּ֣ה חָכְמָ֑ה וּ֝לְשׁוֹנ֗וֹ תְּדַבֵּ֥ר מִשְׁפָּֽט׃ (מה) תּוֹרַ֣ת אֱלֹהָ֣יו בְּלִבּ֑וֹ לֹ֖א תִמְעַ֣ד אֲשֻׁרָיו׃ (מו) צוֹפֶ֣ה רָ֭שָׁע לַצַּדִּ֑יק וּ֝מְבַקֵּ֗שׁ לַהֲמִיתוֹ׃ (מז) יְ֭הוָה לֹא־ יַעַזְבֶ֣נּוּ בְיָד֑וֹ וְלֹ֥א יַ֝רְשִׁיעֶ֗נּוּ בְּהִשָּׁפְטֽוֹ׃ (מח) קַוֵּ֤ה אֶל־ יְהוָ֨ה ׀ וּשְׁמֹ֬ר דַּרְכּ֗וֹ וִֽ֭ירוֹמִמְךָ לָרֶ֣שֶׁת אָ֑רֶץ בְּהִכָּרֵ֖ת רְשָׁעִ֣ים תִּרְאֶֽה׃ (מט) רָ֭אִיתִי רָשָׁ֣ע עָרִ֑יץ וּ֝מִתְעָרֶ֗ה כְּאֶזְרָ֥ח רַעֲנָֽן׃ (נ) וַ֭יַּֽעֲבֹר וְהִנֵּ֣ה אֵינֶ֑נּוּ וָֽ֝אֲבַקְשֵׁ֗הוּ וְלֹ֣א נִמְצָֽא׃ (נא) שְׁמָר־ תָּ֭ם וּרְאֵ֣ה יָשָׁ֑ר כִּֽי־ אַחֲרִ֖ית לְאִ֣ישׁ שָׁלֽוֹם׃ (נב) וּֽ֭פֹשְׁעִים נִשְׁמְד֣וּ יַחְדָּ֑ו אַחֲרִ֖ית רְשָׁעִ֣ים נִכְרָֽתָה׃ (נג) וּתְשׁוּעַ֣ת צַ֭דִּיקִים מֵיְהוָ֑ה מָֽ֝עוּזָּ֗ם בְּעֵ֣ת צָרָֽה׃ (נד) וַֽיַּעְזְרֵ֥ם יְהוָ֗ה וַֽיְפַ֫לְּטֵ֥ם יְפַלְּטֵ֣ם מֵ֭רְשָׁעִים וְיוֹשִׁיעֵ֑ם כִּי־ חָ֥סוּ בֽוֹ׃ (נה) מִזְמ֖וֹר לְדָוִ֣ד לְהַזְכִּֽיר׃ (נו) יְֽהוָ֗ה אַל־ בְּקֶצְפְּךָ֥ תוֹכִיחֵ֑נִי וּֽבַחֲמָתְךָ֥ תְיַסְּרֵֽנִי׃ (נז) כִּֽי־ חִ֭צֶּיךָ נִ֣חֲתוּ בִ֑י וַתִּנְחַ֖ת עָלַ֣י יָדֶֽךָ׃ (נח) אֵין־ מְתֹ֣ם בִּ֭בְשָׂרִי מִפְּנֵ֣י זַעְמֶ֑ךָ אֵין־ שָׁל֥וֹם בַּ֝עֲצָמַ֗י מִפְּנֵ֥י חַטָּאתִֽי׃ (נט) כִּ֣י עֲ֭וֺנֹתַי עָבְר֣וּ רֹאשִׁ֑י כְּמַשָּׂ֥א כָ֝בֵ֗ד יִכְבְּד֥וּ מִמֶּֽנִּי׃ (ס) הִבְאִ֣ישׁוּ נָ֭מַקּוּ חַבּוּרֹתָ֑י מִ֝פְּנֵ֗י אִוַּלְתִּֽי׃ (סא) נַעֲוֵ֣יתִי שַׁחֹ֣תִי עַד־ מְאֹ֑ד כָּל־ הַ֝יּ֗וֹם קֹדֵ֥ר הִלָּֽכְתִּי׃ (סב) כִּֽי־ כְ֭סָלַי מָלְא֣וּ נִקְלֶ֑ה וְאֵ֥ין מְ֝תֹ֗ם בִּבְשָׂרִֽי׃ (סג) נְפוּג֣וֹתִי וְנִדְכֵּ֣יתִי עַד־ מְאֹ֑ד שָׁ֝אַ֗גְתִּי מִֽנַּהֲמַ֥ת לִבִּֽי׃ (סד) אֲ‍ֽדֹנָי נֶגְדְּךָ֥ כָל־ תַּאֲוָתִ֑י וְ֝אַנְחָתִ֗י מִמְּךָ֥ לֹא־ נִסְתָּֽרָה׃ (סה) לִבִּ֣י סְ֭חַרְחַר עֲזָבַ֣נִי כֹחִ֑י וְֽאוֹר־ עֵינַ֥י גַּם־ הֵ֝֗ם אֵ֣ין אִתִּֽי׃ (סו) אֹֽהֲבַ֨י ׀ וְרֵעַ֗י מִנֶּ֣גֶד נִגְעִ֣י יַעֲמֹ֑דוּ וּ֝קְרוֹבַ֗י מֵרָחֹ֥ק עָמָֽדוּ׃ (סז) וַיְנַקְשׁ֤וּ ׀ מְבַקְשֵׁ֬י עֶזְרָ֥ה בְ֝צָר֗וֹת נִמְצָ֥א מְאֹֽד׃ (סו) עַל־ כֵּ֣ן לֹא־ נִ֭ירָא בְּהָמִ֣יר אָ֑רֶץ וּבְמ֥וֹט הָ֝רִ֗ים בְּלֵ֣ב יַמִּֽים׃ (סז) יֶהֱמ֣וּ יֶחְמְר֣וּ מֵימָ֑יו יִֽרְעֲשֽׁוּ־ הָרִ֖ים בְּגַאֲוָת֣וֹ סֶֽלָה׃ (סח) נָהָ֗ר פְּלָגָ֗יו יְשַׂמְּח֥וּ עִיר־ אֱלֹהִ֑ים קְ֝דֹ֗שׁ מִשְׁכְּנֵ֥י עֶלְיֽוֹן׃ (סט) אֱלֹהִ֣ים בְּ֭קִרְבָּהּ בַּל־ תִּמּ֑וֹט יַעְזְרֶ֥הָ אֱ֝לֹהִ֗ים לִפְנ֥וֹת בֹּֽקֶר׃ (ע) הָמ֣וּ ג֭וֹיִם מָ֣טוּ מַמְלָכ֑וֹת נָתַ֥ן בְּ֝קוֹל֗וֹ תָּמ֥וּג אָֽרֶץ׃ (עא) יְהוָ֣ה צְבָא֣וֹת עִמָּ֑נוּ מִשְׂגָּֽב־ לָ֝נוּ אֱלֹהֵ֖י יַעֲקֹ֣ב סֶֽלָה׃ (עב) לְֽכוּ־ חֲ֭זוּ מִפְעֲל֣וֹת יְהוָ֑ה אֲשֶׁר־ שָׂ֖ם שַׁמּ֣וֹת בָּאָֽרֶץ׃ (עג) מַשְׁבִּ֥ית מִלְחָמוֹת֮ עַד־ קְצֵ֪ה הָ֫אָ֥רֶץ קֶ֣שֶׁת יְ֭שַׁבֵּר וְקִצֵּ֣ץ חֲנִ֑ית עֲ֝גָל֗וֹת יִשְׂרֹ֥ף בָּאֵֽשׁ׃ (עד) הַרְפּ֣וּ וּ֭דְעוּ כִּי־ אָנֹכִ֣י אֱלֹהִ֑ים אָר֥וּם בַּ֝גּוֹיִ֗ם אָר֥וּם בָּאָֽרֶץ׃ (עה) יְהוָ֣ה צְבָא֣וֹת עִמָּ֑נוּ מִשְׂגָּֽב־ לָ֝נוּ אֱלֹהֵ֖י יַעֲקֹ֣ב סֶֽלָה׃ (עו) לַמְנַצֵּ֬חַ ׀ לִבְנֵי־ קֹ֬רַח מִזְמֽוֹר׃ (עז) כָּֽל־ הָ֭עַמִּים תִּקְעוּ־ כָ֑ף הָרִ֥יעוּ לֵ֝אלֹהִ֗ים בְּק֣וֹל רִנָּֽה׃ (עח) כִּֽי־ יְהוָ֣ה עֶלְי֣וֹן נוֹרָ֑א מֶ֥לֶךְ גָּ֝דוֹל עַל־ כָּל־ הָאָֽרֶץ׃ (עט) יַדְבֵּ֣ר עַמִּ֣ים תַּחְתֵּ֑ינוּ וּ֝לְאֻמִּ֗ים תַּ֣חַת רַגְלֵֽינוּ׃ (פ) יִבְחַר־ לָ֥נוּ אֶת־ נַחֲלָתֵ֑נוּ אֶ֥ת גְּא֨וֹן יַעֲקֹ֖ב אֲשֶׁר־ אָהֵ֣ב סֶֽלָה׃ (פא) עָלָ֣ה אֱ֭לֹהִים בִּתְרוּעָ֑ה יְ֝הֹוָ֗ה בְּק֣וֹל שׁוֹפָֽר׃ (פב) זַמְּר֣וּ אֱלֹהִ֣ים זַמֵּ֑רוּ זַמְּר֖וּ לְמַלְכֵּ֣נוּ זַמֵּֽרוּ׃ (פג) כִּ֤י מֶ֖לֶךְ כָּל־ הָאָ֥רֶץ אֱלֹהִ֗ים זַמְּר֥וּ מַשְׂכִּֽיל׃ (פד) מָלַ֣ךְ אֱ֭לֹהִים עַל־ גּוֹיִ֑ם אֱ֝לֹהִ֗ים יָשַׁ֤ב ׀ עַל־ כִּסֵּ֬א קָדְשֽׁוֹ׃ (פה) נְדִ֘יבֵ֤י עַמִּ֨ים ׀ נֶאֱסָ֗פוּ עַם֮ אֱלֹהֵ֪י אַבְרָ֫הָ֥ם כִּ֣י לֵֽ֭אלֹהִים מָֽגִנֵּי־ אֶ֗רֶץ מְאֹ֣ד נַעֲלָֽה׃ (פו) שִׁ֥יר מִ֝זְמוֹר לִבְנֵי־ קֹֽרַח׃ (פז) גָּ֘ד֤וֹל יְהוָ֣ה וּמְהֻלָּ֣ל מְאֹ֑ד בְּעִ֥יר אֱ֝לֹהֵ֗ינוּ הַר־ קָדְשֽׁוֹ׃ (פח) יְפֵ֥ה נוֹף֮ מְשׂ֪וֹשׂ כָּל־ הָ֫אָ֥רֶץ הַר־ צִ֭יּוֹן יַרְכְּתֵ֣י צָפ֑וֹן קִ֝רְיַ֗ת מֶ֣לֶךְ רָֽב׃ (פט) אֱלֹהִ֥ים בְּאַרְמְנוֹתֶ֗יהָ נוֹדַ֥ע לְמִשְׂגָּֽב׃ (צ) כִּֽי־ הִנֵּ֣ה הַ֭מְּלָכִים נֽוֹעֲד֑וּ עָבְר֥וּ יַחְדָּֽו׃ (צא) הֵ֣מָּה רָ֭אוּ כֵּ֣ן תָּמָ֑הוּ נִבְהֲל֥וּ נֶחְפָּֽזוּ׃ (צב) רְ֭עָדָה אֲחָזָ֣תַם שָׁ֑ם חִ֝֗יל כַּיּוֹלֵֽדָה׃ (צג) בְּר֥וּחַ קָדִ֑ים תְּ֝שַׁבֵּ֗ר אֳנִיּ֥וֹת תַּרְשִֽׁישׁ׃ (צד) כַּאֲשֶׁ֤ר שָׁמַ֨עְנוּ ׀ כֵּ֤ן רָאִ֗ינוּ בְּעִיר־ יְהוָ֣ה צְ֭בָאוֹת בְּעִ֣יר אֱלֹהֵ֑ינוּ אֱלֹ֘הִ֤ים יְכוֹנְנֶ֖הָ עַד־ עוֹלָ֣ם סֶֽלָה׃ (צה) דִּמִּ֣ינוּ אֱלֹהִ֣ים חַסְדֶּ֑ךָ בְּ֝קֶ֗רֶב הֵיכָלֶֽךָ׃ (צו) כְּשִׁמְךָ֤ אֱלֹהִ֗ים כֵּ֣ן תְּ֭הִלָּתְךָ עַל־ קַצְוֵי־ אֶ֑רֶץ צֶ֝֗דֶק מָלְאָ֥ה יְמִינֶֽךָ׃ (צז) יִשְׂמַ֤ח ׀ הַר־ צִיּ֗וֹן תָּ֭גֵלְנָה בְּנ֣וֹת יְהוּדָ֑ה לְ֝מַ֗עַן מִשְׁפָּטֶֽיךָ׃ (צח) סֹ֣בּוּ צִ֭יּוֹן וְהַקִּיפ֑וּהָ סִ֝פְר֗וּ מִגְדָּלֶֽיהָ׃ (צט) שִׁ֤יתוּ לִבְּכֶ֨ם ׀ לְֽחֵילָ֗ה פַּסְּג֥וּ אַרְמְנוֹתֶ֑יהָ לְמַ֥עַן תְּ֝סַפְּר֗וּ לְד֣וֹר אַחֲרֽוֹן׃ (ק) כִּ֤י זֶ֨ה ׀ אֱלֹהִ֣ים אֱ֭לֹהֵינוּ עוֹלָ֣ם וָעֶ֑ד ה֖וּא יְנַהֲגֵ֣נוּ עַל־ מֽוּת׃ (קא) לַמְנַצֵּ֬חַ ׀ לִבְנֵי־ קֹ֬רַח מִזְמֽוֹר׃ (קב) שִׁמְעוּ־ זֹ֭את כָּל־ הָֽעַמִּ֑ים הַ֝אֲזִ֗ינוּ כָּל־ יֹ֥שְׁבֵי חָֽלֶד׃ (קג) גַּם־ בְּנֵ֣י אָ֭דָם גַּם־ בְּנֵי־ אִ֑ישׁ יַ֝֗חַד עָשִׁ֥יר וְאֶבְיֽוֹן׃ (קד) פִּ֭י יְדַבֵּ֣ר חָכְמ֑וֹת וְהָג֖וּת לִבִּ֣י תְבוּנֽוֹת׃ (קה) אַטֶּ֣ה לְמָשָׁ֣ל אָזְנִ֑י אֶפְתַּ֥ח בְּ֝כִנּ֗וֹר חִידָתִֽי׃ (קו) לָ֣מָּה אִ֭ירָא בִּ֣ימֵי רָ֑ע עֲוֺ֖ן עֲקֵבַ֣י יְסוּבֵּֽנִי׃ (קז) הַבֹּטְחִ֥ים עַל־ חֵילָ֑ם וּבְרֹ֥ב עָ֝שְׁרָ֗ם יִתְהַלָּֽלוּ׃ (קח) אָ֗ח לֹא־ פָדֹ֣ה יִפְדֶּ֣ה אִ֑ישׁ לֹא־ יִתֵּ֖ן לֵאלֹהִ֣ים כָּפְרֽוֹ׃ (קט) וְ֭יֵקַר פִּדְי֥וֹן נַפְשָׁ֗ם וְחָדַ֥ל לְעוֹלָֽם׃ (קי) וִֽיחִי־ ע֥וֹד לָנֶ֑צַח לֹ֖א יִרְאֶ֣ה הַשָּֽׁחַת׃ (קיא) כִּ֤י יִרְאֶ֨ה ׀ חֲכָ֘מִ֤ים יָמ֗וּתוּ יַ֤חַד כְּסִ֣יל וָבַ֣עַר יֹאבֵ֑דוּ וְעָזְב֖וּ לַאֲחֵרִ֣ים חֵילָֽם׃ (קיב) קִרְבָּ֤ם בָּתֵּ֨ימוֹ ׀ לְֽעוֹלָ֗ם מִ֭שְׁכְּנֹתָם לְדֹ֣ר וָדֹ֑ר קָֽרְא֥וּ בִ֝שְׁמוֹתָ֗ם עֲלֵ֣י אֲדָמֽוֹת׃ (קיג) וְאָדָ֣ם בִּ֭יקָר בַּל־ יָלִ֑ין נִמְשַׁ֖ל כַּבְּהֵמ֣וֹת נִדְמֽוּ׃ (קיד) זֶ֣ה דַ֭רְכָּם כֵּ֣סֶל לָ֑מוֹ וְאַחֲרֵיהֶ֓ם ׀ בְּפִיהֶ֖ם יִרְצ֣וּ סֶֽלָה׃ (קטו) כַּצֹּ֤אן ׀ לִֽשְׁא֣וֹל שַׁתּוּ֮ מָ֤וֶת יִ֫רְעֵ֥ם וַיִּרְדּ֘וּ בָ֤ם יְשָׁרִ֨ים ׀ לַבֹּ֗קֶר (וצירם) וְ֭צוּרָם לְבַלּ֥וֹת שְׁא֗וֹל מִזְּבֻ֥ל לֽוֹ׃ (קטז) אַךְ־ אֱלֹהִ֗ים יִפְדֶּ֣ה נַ֭פְשִׁי מִֽיַּד־ שְׁא֑וֹל כִּ֖י יִקָּחֵ֣נִי סֶֽלָה׃ (קיז) אַל־ תִּ֭ירָא כִּֽי־ יַעֲשִׁ֣ר אִ֑ישׁ כִּֽי־ יִ֝רְבֶּה כְּב֣וֹד בֵּיתֽוֹ׃ (קיח) כִּ֤י לֹ֣א בְ֭מוֹתוֹ יִקַּ֣ח הַכֹּ֑ל לֹא־ יֵרֵ֖ד אַחֲרָ֣יו כְּבוֹדֽוֹ׃
