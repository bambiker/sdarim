שופטים פרק ג לפי החלוקה לסדרים

פרק ג בספר שופטים (31 פסוקים) נמצא בשופטים סדר ב (פסוקים יח–מז) ובשופטים סדר ג (פסוקים א–א).

שופטים סדר ב

(א) וַיַּעַבְד֤וּ הָעָם֙ אֶת־ יְהוָ֔ה כֹּ֖ל יְמֵ֣י יְהוֹשֻׁ֑עַ וְכֹ֣ל ׀ יְמֵ֣י הַזְּקֵנִ֗ים אֲשֶׁ֨ר הֶאֱרִ֤יכוּ יָמִים֙ אַחֲרֵ֣י יְהוֹשׁ֔וּעַ אֲשֶׁ֣ר רָא֗וּ אֵ֣ת כָּל־ מַעֲשֵׂ֤ה יְהוָה֙ הַגָּד֔וֹל אֲשֶׁ֥ר עָשָׂ֖ה לְיִשְׂרָאֵֽל׃ (ב) וַיָּ֛מָת יְהוֹשֻׁ֥עַ בִּן־ נ֖וּן עֶ֣בֶד יְהוָ֑ה בֶּן־ מֵאָ֥ה וָעֶ֖שֶׂר שָׁנִֽים׃ (ג) וַיִּקְבְּר֤וּ אוֹתוֹ֙ בִּגְב֣וּל נַחֲלָת֔וֹ בְּתִמְנַת־ חֶ֖רֶס בְּהַ֣ר אֶפְרָ֑יִם מִצְּפ֖וֹן לְהַר־ גָּֽעַשׁ׃ (ד) וְגַם֙ כָּל־ הַדּ֣וֹר הַה֔וּא נֶאֶסְפ֖וּ אֶל־ אֲבוֹתָ֑יו וַיָּקָם֩ דּ֨וֹר אַחֵ֜ר אַחֲרֵיהֶ֗ם אֲשֶׁ֤ר לֹא־ יָֽדְעוּ֙ אֶת־ יְהוָ֔ה וְגַם֙ אֶת־ הַֽמַּעֲשֶׂ֔ה אֲשֶׁ֥ר עָשָׂ֖ה לְיִשְׂרָאֵֽל׃ (ה) וַיַּעֲשׂ֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶת־ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיַּעַבְד֖וּ אֶת־ הַבְּעָלִֽים׃ (ו) וַיַּעַזְב֞וּ אֶת־ יְהוָ֣ה ׀ אֱלֹהֵ֣י אֲבוֹתָ֗ם הַמּוֹצִ֣יא אוֹתָם֮ מֵאֶ֣רֶץ מִצְרַיִם֒ וַיֵּלְכ֞וּ אַחֲרֵ֣י ׀ אֱלֹהִ֣ים אֲחֵרִ֗ים מֵאֱלֹהֵ֤י הָֽעַמִּים֙ אֲשֶׁר֙ סְבִיב֣וֹתֵיהֶ֔ם וַיִּֽשְׁתַּחֲו֖וּ לָהֶ֑ם וַיַּכְעִ֖סוּ אֶת־ יְהוָֽה׃ (ז) וַיַּעַזְב֖וּ אֶת־ יְהוָ֑ה וַיַּעַבְד֥וּ לַבַּ֖עַל וְלָעַשְׁתָּרֽוֹת׃ (ח) וַיִּֽחַר־ אַ֤ף יְהוָה֙ בְּיִשְׂרָאֵ֔ל וַֽיִּתְּנֵם֙ בְּיַד־ שֹׁסִ֔ים וַיָּשֹׁ֖סּוּ אוֹתָ֑ם וַֽיִּמְכְּרֵ֞ם בְּיַ֤ד אֽוֹיְבֵיהֶם֙ מִסָּבִ֔יב וְלֹֽא־ יָכְל֣וּ ע֔וֹד לַעֲמֹ֖ד לִפְנֵ֥י אוֹיְבֵיהֶֽם׃ (ט) בְּכֹ֣ל ׀ אֲשֶׁ֣ר יָצְא֗וּ יַד־ יְהוָה֙ הָיְתָה־ בָּ֣ם לְרָעָ֔ה כַּֽאֲשֶׁר֙ דִּבֶּ֣ר יְהוָ֔ה וְכַאֲשֶׁ֛ר נִשְׁבַּ֥ע יְהוָ֖ה לָהֶ֑ם וַיֵּ֥צֶר לָהֶ֖ם מְאֹֽד׃ (י) וַיָּ֥קֶם יְהוָ֖ה שֹֽׁפְטִ֑ים וַיּ֣וֹשִׁיע֔וּם מִיַּ֖ד שֹׁסֵיהֶֽם׃ (יא) וְגַ֤ם אֶל־ שֹֽׁפְטֵיהֶם֙ לֹ֣א שָׁמֵ֔עוּ כִּ֣י זָנ֗וּ אַֽחֲרֵי֙ אֱלֹהִ֣ים אֲחֵרִ֔ים וַיִּֽשְׁתַּחֲו֖וּ לָהֶ֑ם סָ֣רוּ מַהֵ֗ר מִן־ הַדֶּ֜רֶךְ אֲשֶׁ֨ר הָלְכ֧וּ אֲבוֹתָ֛ם לִשְׁמֹ֥עַ מִצְוֺת־ יְהוָ֖ה לֹא־ עָ֥שׂוּ כֵֽן׃ (יב) וְכִֽי־ הֵקִ֨ים יְהוָ֥ה ׀ לָהֶם֮ שֹֽׁפְטִים֒ וְהָיָ֤ה יְהוָה֙ עִם־ הַשֹּׁפֵ֔ט וְהֽוֹשִׁיעָם֙ מִיַּ֣ד אֹֽיְבֵיהֶ֔ם כֹּ֖ל יְמֵ֣י הַשּׁוֹפֵ֑ט כִּֽי־ יִנָּחֵ֤ם יְהוָה֙ מִנַּֽאֲקָתָ֔ם מִפְּנֵ֥י לֹחֲצֵיהֶ֖ם וְדֹחֲקֵיהֶֽם׃ (יג) וְהָיָ֣ה ׀ בְּמ֣וֹת הַשּׁוֹפֵ֗ט יָשֻׁ֙בוּ֙ וְהִשְׁחִ֣יתוּ מֵֽאֲבוֹתָ֔ם לָלֶ֗כֶת אַֽחֲרֵי֙ אֱלֹהִ֣ים אֲחֵרִ֔ים לְעָבְדָ֖ם וּלְהִשְׁתַּחֲוֺ֣ת לָהֶ֑ם לֹ֤א הִפִּ֙ילוּ֙ מִמַּ֣עַלְלֵיהֶ֔ם וּמִדַּרְכָּ֖ם הַקָּשָֽׁה׃ (יד) וַיִּֽחַר־ אַ֥ף יְהוָ֖ה בְּיִשְׂרָאֵ֑ל וַיֹּ֗אמֶר יַעַן֩ אֲשֶׁ֨ר עָבְר֜וּ הַגּ֣וֹי הַזֶּ֗ה אֶת־ בְּרִיתִי֙ אֲשֶׁ֣ר צִוִּ֣יתִי אֶת־ אֲבוֹתָ֔ם וְלֹ֥א שָׁמְע֖וּ לְקוֹלִֽי׃ (טו) גַּם־ אֲנִי֙ לֹ֣א אוֹסִ֔יף לְהוֹרִ֥ישׁ אִ֖ישׁ מִפְּנֵיהֶ֑ם מִן־ הַגּוֹיִ֛ם אֲשֶׁר־ עָזַ֥ב יְהוֹשֻׁ֖עַ וַיָּמֹֽת׃ (טז) לְמַ֛עַן נַסּ֥וֹת בָּ֖ם אֶת־ יִשְׂרָאֵ֑ל הֲשֹׁמְרִ֣ים הֵם֩ אֶת־ דֶּ֨רֶךְ יְהוָ֜ה לָלֶ֣כֶת בָּ֗ם כַּאֲשֶׁ֛ר שָׁמְר֥וּ אֲבוֹתָ֖ם אִם־ לֹֽא׃ (יז) וַיַּנַּ֤ח יְהוָה֙ אֶת־ הַגּוֹיִ֣ם הָאֵ֔לֶּה לְבִלְתִּ֥י הוֹרִישָׁ֖ם מַהֵ֑ר וְלֹ֥א נְתָנָ֖ם בְּיַד־ יְהוֹשֻֽׁעַ׃ תחילת פרק ג (יח) וְאֵ֤לֶּה הַגּוֹיִם֙ אֲשֶׁ֣ר הִנִּ֣יחַ יְהוָ֔ה לְנַסּ֥וֹת בָּ֖ם אֶת־ יִשְׂרָאֵ֑ל אֵ֚ת כָּל־ אֲשֶׁ֣ר לֹֽא־ יָדְע֔וּ אֵ֖ת כָּל־ מִלְחֲמ֥וֹת כְּנָֽעַן׃ (יט) רַ֗ק לְמַ֙עַן֙ דַּ֚עַת דֹּר֣וֹת בְּנֵֽי־ יִשְׂרָאֵ֔ל לְלַמְּדָ֖ם מִלְחָמָ֑ה רַ֥ק אֲשֶׁר־ לְפָנִ֖ים לֹ֥א יְדָעֽוּם׃ (כ) חֲמֵ֣שֶׁת ׀ סַרְנֵ֣י פְלִשְׁתִּ֗ים וְכָל־ הַֽכְּנַעֲנִי֙ וְהַצִּ֣ידֹנִ֔י וְהַ֣חִוִּ֔י יֹשֵׁ֖ב הַ֣ר הַלְּבָנ֑וֹן מֵהַר֙ בַּ֣עַל חֶרְמ֔וֹן עַ֖ד לְב֥וֹא חֲמָֽת׃ (כא) וַֽיִּהְי֕וּ לְנַסּ֥וֹת בָּ֖ם אֶת־ יִשְׂרָאֵ֑ל לָדַ֗עַת הֲיִשְׁמְעוּ֙ אֶת־ מִצְוֺ֣ת יְהוָ֔ה אֲשֶׁר־ צִוָּ֥ה אֶת־ אֲבוֹתָ֖ם בְּיַד־ מֹשֶֽׁה׃ (כב) וּבְנֵ֣י יִשְׂרָאֵ֔ל יָשְׁב֖וּ בְּקֶ֣רֶב הַֽכְּנַעֲנִ֑י הַחִתִּ֤י וְהָֽאֱמֹרִי֙ וְהַפְּרִזִּ֔י וְהַחִוִּ֖י וְהַיְבוּסִֽי׃ (כג) וַיִּקְח֨וּ אֶת־ בְּנוֹתֵיהֶ֤ם לָהֶם֙ לְנָשִׁ֔ים וְאֶת־ בְּנוֹתֵיהֶ֖ם נָתְנ֣וּ לִבְנֵיהֶ֑ם וַיַּעַבְד֖וּ אֶת־ אֱלֹהֵיהֶֽם׃ (כד) וַיַּעֲשׂ֨וּ בְנֵי־ יִשְׂרָאֵ֤ל אֶת־ הָרַע֙ בְּעֵינֵ֣י יְהוָ֔ה וַֽיִּשְׁכְּח֖וּ אֶת־ יְהוָ֣ה אֱלֹֽהֵיהֶ֑ם וַיַּעַבְד֥וּ אֶת־ הַבְּעָלִ֖ים וְאֶת־ הָאֲשֵׁרֽוֹת׃ (כה) וַיִּֽחַר־ אַ֤ף יְהוָה֙ בְּיִשְׂרָאֵ֔ל וַֽיִּמְכְּרֵ֗ם בְּיַד֙ כּוּשַׁ֣ן רִשְׁעָתַ֔יִם מֶ֖לֶךְ אֲרַ֣ם נַהֲרָ֑יִם וַיַּעַבְד֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶת־ כּוּשַׁ֥ן רִשְׁעָתַ֖יִם שְׁמֹנֶ֥ה שָׁנִֽים׃ (כו) וַיִּזְעֲק֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ אֶל־ יְהוָ֔ה וַיָּ֨קֶם יְהוָ֥ה מוֹשִׁ֛יעַ לִבְנֵ֥י יִשְׂרָאֵ֖ל וַיּֽוֹשִׁיעֵ֑ם אֵ֚ת עָתְנִיאֵ֣ל בֶּן־ קְנַ֔ז אֲחִ֥י כָלֵ֖ב הַקָּטֹ֥ן מִמֶּֽנּוּ׃ (כז) וַתְּהִ֨י עָלָ֥יו רֽוּחַ־ יְהוָה֮ וַיִּשְׁפֹּ֣ט אֶת־ יִשְׂרָאֵל֒ וַיֵּצֵא֙ לַמִּלְחָמָ֔ה וַיִּתֵּ֤ן יְהוָה֙ בְּיָד֔וֹ אֶת־ כּוּשַׁ֥ן רִשְׁעָתַ֖יִם מֶ֣לֶךְ אֲרָ֑ם וַתָּ֣עָז יָד֔וֹ עַ֖ל כּוּשַׁ֥ן רִשְׁעָתָֽיִם׃ (כח) וַתִּשְׁקֹ֥ט הָאָ֖רֶץ אַרְבָּעִ֣ים שָׁנָ֑ה וַיָּ֖מָת עָתְנִיאֵ֥ל בֶּן־ קְנַֽז׃ (כט) וַיֹּסִ֙פוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיְחַזֵּ֨ק יְהוָ֜ה אֶת־ עֶגְל֤וֹן מֶֽלֶךְ־ מוֹאָב֙ עַל־ יִשְׂרָאֵ֔ל עַ֛ל כִּֽי־ עָשׂ֥וּ אֶת־ הָרַ֖ע בְּעֵינֵ֥י יְהוָֽה׃ (ל) וַיֶּאֱסֹ֣ף אֵלָ֔יו אֶת־ בְּנֵ֥י עַמּ֖וֹן וַעֲמָלֵ֑ק וַיֵּ֗לֶךְ וַיַּךְ֙ אֶת־ יִשְׂרָאֵ֔ל וַיִּֽירְשׁ֖וּ אֶת־ עִ֥יר הַתְּמָרִֽים׃ (לא) וַיַּעַבְד֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ אֶת־ עֶגְל֣וֹן מֶֽלֶךְ־ מוֹאָ֔ב שְׁמוֹנֶ֥ה עֶשְׂרֵ֖ה שָׁנָֽה׃ (לב) וַיִּזְעֲק֣וּ בְנֵֽי־ יִשְׂרָאֵל֮ אֶל־ יְהוָה֒ וַיָּקֶם֩ יְהוָ֨ה לָהֶ֜ם מוֹשִׁ֗יעַ אֶת־ אֵה֤וּד בֶּן־ גֵּרָא֙ בֶּן־ הַיְמִינִ֔י אִ֥ישׁ אִטֵּ֖ר יַד־ יְמִינ֑וֹ וַיִּשְׁלְח֨וּ בְנֵי־ יִשְׂרָאֵ֤ל בְּיָדוֹ֙ מִנְחָ֔ה לְעֶגְל֖וֹן מֶ֥לֶךְ מוֹאָֽב׃ (לג) וַיַּעַשׂ֩ ל֨וֹ אֵה֜וּד חֶ֗רֶב וְלָ֛הּ שְׁנֵ֥י פֵי֖וֹת גֹּ֣מֶד אָרְכָּ֑הּ וַיַּחְגֹּ֤ר אוֹתָהּ֙ מִתַּ֣חַת לְמַדָּ֔יו עַ֖ל יֶ֥רֶךְ יְמִינֽוֹ׃ (לד) וַיַּקְרֵב֙ אֶת־ הַמִּנְחָ֔ה לְעֶגְל֖וֹן מֶ֣לֶךְ מוֹאָ֑ב וְעֶגְל֕וֹן אִ֥ישׁ בָּרִ֖יא מְאֹֽד׃ (לה) וַֽיְהִי֙ כַּאֲשֶׁ֣ר כִּלָּ֔ה לְהַקְרִ֖יב אֶת־ הַמִּנְחָ֑ה וַיְשַׁלַּח֙ אֶת־ הָעָ֔ם נֹשְׂאֵ֖י הַמִּנְחָֽה׃ (לו) וְה֣וּא שָׁ֗ב מִן־ הַפְּסִילִים֙ אֲשֶׁ֣ר אֶת־ הַגִּלְגָּ֔ל וַיֹּ֕אמֶר דְּבַר־ סֵ֥תֶר לִ֛י אֵלֶ֖יךָ הַמֶּ֑לֶךְ וַיֹּ֣אמֶר הָ֔ס וַיֵּֽצְאוּ֙ מֵֽעָלָ֔יו כָּל־ הָעֹמְדִ֖ים עָלָֽיו׃ (לז) וְאֵה֣וּד ׀ בָּ֣א אֵלָ֗יו וְהֽוּא־ יֹ֠שֵׁב בַּעֲלִיַּ֨ת הַמְּקֵרָ֤ה אֲשֶׁר־ לוֹ֙ לְבַדּ֔וֹ וַיֹּ֣אמֶר אֵה֔וּד דְּבַר־ אֱלֹהִ֥ים לִ֖י אֵלֶ֑יךָ וַיָּ֖קָם מֵעַ֥ל הַכִּסֵּֽא׃ (לח) וַיִּשְׁלַ֤ח אֵהוּד֙ אֶת־ יַ֣ד שְׂמֹאל֔וֹ וַיִּקַּח֙ אֶת־ הַחֶ֔רֶב מֵעַ֖ל יֶ֣רֶךְ יְמִינ֑וֹ וַיִּתְקָעֶ֖הָ בְּבִטְנֽוֹ׃ (לט) וַיָּבֹ֨א גַֽם־ הַנִּצָּ֜ב אַחַ֣ר הַלַּ֗הַב וַיִּסְגֹּ֤ר הַחֵ֙לֶב֙ בְּעַ֣ד הַלַּ֔הַב כִּ֣י לֹ֥א שָׁלַ֛ף הַחֶ֖רֶב מִבִּטְנ֑וֹ וַיֵּצֵ֖א הַֽפַּרְשְׁדֹֽנָה׃ (מ) וַיֵּצֵ֥א אֵה֖וּד הַֽמִּסְדְּר֑וֹנָה וַיִּסְגֹּ֞ר דַּלְת֧וֹת הָעַלִיָּ֛ה בַּעֲד֖וֹ וְנָעָֽל׃ (מא) וְה֤וּא יָצָא֙ וַעֲבָדָ֣יו בָּ֔אוּ וַיִּרְא֕וּ וְהִנֵּ֛ה דַּלְת֥וֹת הָעֲלִיָּ֖ה נְעֻל֑וֹת וַיֹּ֣אמְר֔וּ אַ֣ךְ מֵסִ֥יךְ ה֛וּא אֶת־ רַגְלָ֖יו בַּחֲדַ֥ר הַמְּקֵרָֽה׃ (מב) וַיָּחִ֣ילוּ עַד־ בּ֔וֹשׁ וְהְנֵּ֛ה אֵינֶ֥נּוּ פֹתֵ֖חַ דַּלְת֣וֹת הָֽעֲלִיָּ֑ה וַיִּקְח֤וּ אֶת־ הַמַּפְתֵּ֙חַ֙ וַיִּפְתָּ֔חוּ וְהִנֵּה֙ אֲדֹ֣נֵיהֶ֔ם נֹפֵ֥ל אַ֖רְצָה מֵֽת׃ (מג) וְאֵה֥וּד נִמְלַ֖ט עַ֣ד הִֽתְמַהְמְהָ֑ם וְהוּא֙ עָבַ֣ר אֶת־ הַפְּסִילִ֔ים וַיִּמָּלֵ֖ט הַשְּׂעִירָֽתָה׃ (מד) וַיְהִ֣י בְּבוֹא֔וֹ וַיִּתְקַ֥ע בַּשּׁוֹפָ֖ר בְּהַ֣ר אֶפְרָ֑יִם וַיֵּרְד֨וּ עִמּ֧וֹ בְנֵֽי־ יִשְׂרָאֵ֛ל מִן־ הָהָ֖ר וְה֥וּא לִפְנֵיהֶֽם׃ (מה) וַיֹּ֤אמֶר אֲלֵהֶם֙ רִדְפ֣וּ אַחֲרַ֔י כִּֽי־ נָתַ֨ן יְהוָ֧ה אֶת־ אֹיְבֵיכֶ֛ם אֶת־ מוֹאָ֖ב בְּיֶדְכֶ֑ם וַיֵּרְד֣וּ אַחֲרָ֗יו וַֽיִּלְכְּד֞וּ אֶת־ מַעְבְּר֤וֹת הַיַּרְדֵּן֙ לְמוֹאָ֔ב וְלֹֽא־ נָתְנ֥וּ אִ֖ישׁ לַעֲבֹֽר׃ (מו) וַיַּכּ֨וּ אֶת־ מוֹאָ֜ב בָּעֵ֣ת הַהִ֗יא כַּעֲשֶׂ֤רֶת אֲלָפִים֙ אִ֔ישׁ כָּל־ שָׁמֵ֖ן וְכָל־ אִ֣ישׁ חָ֑יִל וְלֹ֥א נִמְלַ֖ט אִֽישׁ׃ (מז) וַתִּכָּנַ֤ע מוֹאָב֙ בַּיּ֣וֹם הַה֔וּא תַּ֖חַת יַ֣ד יִשְׂרָאֵ֑ל וַתִּשְׁקֹ֥ט הָאָ֖רֶץ שְׁמוֹנִ֥ים שָׁנָֽה׃

(א) וְאַחֲרָ֤יו הָיָה֙ שַׁמְגַּ֣ר בֶּן־ עֲנָ֔ת וַיַּ֤ךְ אֶת־ פְּלִשְׁתִּים֙ שֵֽׁשׁ־ מֵא֣וֹת אִ֔ישׁ בְּמַלְמַ֖ד הַבָּקָ֑ר וַיֹּ֥שַׁע גַּם־ ה֖וּא אֶת־ יִשְׂרָאֵֽל׃ סוף פרק ג (ב) וַיֹּסִ֙פוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וְאֵה֖וּד מֵֽת׃ (ג) וַיִּמְכְּרֵ֣ם יְהוָ֗ה בְּיַד֙ יָבִ֣ין מֶֽלֶךְ־ כְּנַ֔עַן אֲשֶׁ֥ר מָלַ֖ךְ בְּחָצ֑וֹר וְשַׂר־ צְבָאוֹ֙ סִֽיסְרָ֔א וְה֥וּא יוֹשֵׁ֖ב בַּחֲרֹ֥שֶׁת הַגּוֹיִֽם׃ (ד) וַיִּצְעֲק֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל אֶל־ יְהוָ֑ה כִּ֠י תְּשַׁ֨ע מֵא֤וֹת רֶֽכֶב־ בַּרְזֶל֙ ל֔וֹ וְ֠הוּא לָחַ֞ץ אֶת־ בְּנֵ֧י יִשְׂרָאֵ֛ל בְּחָזְקָ֖ה עֶשְׂרִ֥ים שָׁנָֽה׃ (ה) וּדְבוֹרָה֙ אִשָּׁ֣ה נְבִיאָ֔ה אֵ֖שֶׁת לַפִּיד֑וֹת הִ֛יא שֹׁפְטָ֥ה אֶת־ יִשְׂרָאֵ֖ל בָּעֵ֥ת הַהִֽיא׃ (ו) וְ֠הִיא יוֹשֶׁ֨בֶת תַּֽחַת־ תֹּ֜מֶר דְּבוֹרָ֗ה בֵּ֧ין הָרָמָ֛ה וּבֵ֥ין בֵּֽית־ אֵ֖ל בְּהַ֣ר אֶפְרָ֑יִם וַיַּעֲל֥וּ אֵלֶ֛יהָ בְּנֵ֥י יִשְׂרָאֵ֖ל לַמִּשְׁפָּֽט׃ (ז) וַתִּשְׁלַ֗ח וַתִּקְרָא֙ לְבָרָ֣ק בֶּן־ אֲבִינֹ֔עַם מִקֶּ֖דֶשׁ נַפְתָּלִ֑י וַתֹּ֨אמֶר אֵלָ֜יו הֲלֹ֥א צִוָּ֣ה ׀ יְהוָ֣ה אֱלֹהֵֽי־ יִשְׂרָאֵ֗ל לֵ֤ךְ וּמָֽשַׁכְתָּ֙ בְּהַ֣ר תָּב֔וֹר וְלָקַחְתָּ֣ עִמְּךָ֗ עֲשֶׂ֤רֶת אֲלָפִים֙ אִ֔ישׁ מִבְּנֵ֥י נַפְתָּלִ֖י וּמִבְּנֵ֥י זְבֻלֽוּן׃ (ח) וּמָשַׁכְתִּ֨י אֵלֶ֜יךָ אֶל־ נַ֣חַל קִישׁ֗וֹן אֶת־ סִֽיסְרָא֙ שַׂר־ צְבָ֣א יָבִ֔ין וְאֶת־ רִכְבּ֖וֹ וְאֶת־ הֲמוֹנ֑וֹ וּנְתַתִּ֖יהוּ בְּיָדֶֽךָ׃ (ט) וַיֹּ֤אמֶר אֵלֶ֙יהָ֙ בָּרָ֔ק אִם־ תֵּלְכִ֥י עִמִּ֖י וְהָלָ֑כְתִּי וְאִם־ לֹ֥א תֵלְכִ֛י עִמִּ֖י לֹ֥א אֵלֵֽךְ׃ (י) וַתֹּ֜אמֶר הָלֹ֧ךְ אֵלֵ֣ךְ עִמָּ֗ךְ אֶ֚פֶס כִּי֩ לֹ֨א תִֽהְיֶ֜ה תִּֽפְאַרְתְּךָ֗ עַל־ הַדֶּ֙רֶךְ֙ אֲשֶׁ֣ר אַתָּ֣ה הוֹלֵ֔ךְ כִּ֣י בְֽיַד־ אִשָּׁ֔ה יִמְכֹּ֥ר יְהוָ֖ה אֶת־ סִֽיסְרָ֑א וַתָּ֧קָם דְּבוֹרָ֛ה וַתֵּ֥לֶך עִם־ בָּרָ֖ק קֶֽדְשָׁה׃ (יא) וַיַּזְעֵ֨ק בָּרָ֜ק אֶת־ זְבוּלֻ֤ן וְאֶת־ נַפְתָּלִי֙ קֶ֔דְשָׁה וַיַּ֣עַל בְּרַגְלָ֔יו עֲשֶׂ֥רֶת אַלְפֵ֖י אִ֑ישׁ וַתַּ֥עַל עִמּ֖וֹ דְּבוֹרָֽה׃ (יב) וְחֶ֤בֶר הַקֵּינִי֙ נִפְרָ֣ד מִקַּ֔יִן מִבְּנֵ֥י חֹבָ֖ב חֹתֵ֣ן מֹשֶׁ֑ה וַיֵּ֣ט אָהֳל֔וֹ עַד־ אֵל֥וֹן (בצענים) בְּצַעֲנַנִּ֖ים אֲשֶׁ֥ר אֶת־ קֶֽדֶשׁ׃ (יג) וַיַּגִּ֖דוּ לְסִֽיסְרָ֑א כִּ֥י עָלָ֛ה בָּרָ֥ק בֶּן־ אֲבִינֹ֖עַם הַר־ תָּבֽוֹר׃ (יד) וַיַּזְעֵ֨ק סִֽיסְרָ֜א אֶת־ כָּל־ רִכְבּ֗וֹ תְּשַׁ֤ע מֵאוֹת֙ רֶ֣כֶב בַּרְזֶ֔ל וְאֶת־ כָּל־ הָעָ֖ם אֲשֶׁ֣ר אִתּ֑וֹ מֵחֲרֹ֥שֶׁת הַגּוֹיִ֖ם אֶל־ נַ֥חַל קִישֽׁוֹן׃ (טו) וַתֹּאמֶר֩ דְּבֹרָ֨ה אֶל־ בָּרָ֜ק ק֗וּם כִּ֣י זֶ֤ה הַיּוֹם֙ אֲשֶׁר֩ נָתַ֨ן יְהוָ֤ה אֶת־ סִֽיסְרָא֙ בְּיָדֶ֔ךָ הֲלֹ֥א יְהוָ֖ה יָצָ֣א לְפָנֶ֑יךָ וַיֵּ֤רֶד בָּרָק֙ מֵהַ֣ר תָּב֔וֹר וַעֲשֶׂ֧רֶת אֲלָפִ֛ים אִ֖ישׁ אַחֲרָֽיו׃ (טז) וַיָּ֣הָם יְ֠הוָה אֶת־ סִֽיסְרָ֨א וְאֶת־ כָּל־ הָרֶ֧כֶב וְאֶת־ כָּל־ הַֽמַּחֲנֶ֛ה לְפִי־ חֶ֖רֶב לִפְנֵ֣י בָרָ֑ק וַיֵּ֧רֶד סִֽיסְרָ֛א מֵעַ֥ל הַמֶּרְכָּבָ֖ה וַיָּ֥נָס בְּרַגְלָֽיו׃ (יז) וּבָרָ֗ק רָדַ֞ף אַחֲרֵ֤י הָרֶ֙כֶב֙ וְאַחֲרֵ֣י הַֽמַּחֲנֶ֔ה עַ֖ד חֲרֹ֣שֶׁת הַגּוֹיִ֑ם וַיִּפֹּ֞ל כָּל־ מַחֲנֵ֤ה סִֽיסְרָא֙ לְפִי־ חֶ֔רֶב לֹ֥א נִשְׁאַ֖ר עַד־ אֶחָֽד׃ (יח) וְסִֽיסְרָא֙ נָ֣ס בְּרַגְלָ֔יו אֶל־ אֹ֣הֶל יָעֵ֔ל אֵ֖שֶּׁת חֶ֣בֶר הַקֵּינִ֑י כִּ֣י שָׁל֗וֹם בֵּ֚ין יָבִ֣ין מֶֽלֶךְ־ חָצ֔וֹר וּבֵ֕ין בֵּ֖ית חֶ֥בֶר הַקֵּינִֽי׃ (יט) וַתֵּצֵ֣א יָעֵל֮ לִקְרַ֣את סִֽיסְרָא֒ וַתֹּ֣אמֶר אֵלָ֗יו סוּרָ֧ה אֲדֹנִ֛י סוּרָ֥ה אֵלַ֖י אַל־ תִּירָ֑א וַיָּ֤סַר אֵלֶ֙יהָ֙ הָאֹ֔הֱלָה וַתְּכַסֵּ֖הוּ בַּשְּׂמִיכָֽה׃ (כ) וַיֹּ֧אמֶר אֵלֶ֛יהָ הַשְׁקִינִי־ נָ֥א מְעַט־ מַ֖יִם כִּ֣י צָמֵ֑אתִי וַתִּפְתַּ֞ח אֶת־ נֹ֧אוד הֶחָלָ֛ב וַתַּשְׁקֵ֖הוּ וַתְּכַסֵּֽהוּ׃ (כא) וַיֹּ֣אמֶר אֵלֶ֔יהָ עֲמֹ֖ד פֶּ֣תַח הָאֹ֑הֶל וְהָיָה֩ אִם־ אִ֨ישׁ יָב֜וֹא וּשְׁאֵלֵ֗ךְ וְאָמַ֛ר הֲיֵֽשׁ־ פֹּ֥ה אִ֖ישׁ וְאָמַ֥רְתְּ אָֽיִן׃ (כב) וַתִּקַּ֣ח יָעֵ֣ל אֵֽשֶׁת־ חֶ֠בֶר אֶת־ יְתַ֨ד הָאֹ֜הֶל וַתָּ֧שֶׂם אֶת־ הַמַּקֶּ֣בֶת בְּיָדָ֗הּ וַתָּב֤וֹא אֵלָיו֙ בַּלָּ֔אט וַתִּתְקַ֤ע אֶת־ הַיָּתֵד֙ בְּרַקָּת֔וֹ וַתִּצְנַ֖ח בָּאָ֑רֶץ וְהֽוּא־ נִרְדָּ֥ם וַיָּ֖עַף וַיָּמֹֽת׃ (כג) וְהִנֵּ֣ה בָרָק֮ רֹדֵ֣ף אֶת־ סִֽיסְרָא֒ וַתֵּצֵ֤א יָעֵל֙ לִקְרָאת֔וֹ וַתֹּ֣אמֶר ל֔וֹ לֵ֣ךְ וְאַרְאֶ֔ךָּ אֶת־ הָאִ֖ישׁ אֲשֶׁר־ אַתָּ֣ה מְבַקֵּ֑שׁ וַיָּבֹ֣א אֵלֶ֔יהָ וְהִנֵּ֤ה סִֽיסְרָא֙ נֹפֵ֣ל מֵ֔ת וְהַיָּתֵ֖ד בְּרַקָּתֽוֹ׃ (כד) וַיַּכְנַ֤ע אֱלֹהִים֙ בַּיּ֣וֹם הַה֔וּא אֵ֖ת יָבִ֣ין מֶֽלֶךְ־ כְּנָ֑עַן לִפְנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (כה) וַתֵּ֜לֶךְ יַ֤ד בְּנֵֽי־ יִשְׂרָאֵל֙ הָל֣וֹךְ וְקָשָׁ֔ה עַ֖ל יָבִ֣ין מֶֽלֶךְ־ כְּנָ֑עַן עַ֚ד אֲשֶׁ֣ר הִכְרִ֔יתוּ אֵ֖ת יָבִ֥ין מֶֽלֶךְ־ כְּנָֽעַן׃ (כו) וַתָּ֣שַׁר דְּבוֹרָ֔ה וּבָרָ֖ק בֶּן־ אֲבִינֹ֑עַם בַּיּ֥וֹם הַה֖וּא לֵאמֹֽר׃ (כז) בִּפְרֹ֤עַ פְּרָעוֹת֙ בְּיִשְׂרָאֵ֔ל בְּהִתְנַדֵּ֖ב עָ֑ם בָּרֲכ֖וּ יְהוָֽה׃ (כח) שִׁמְע֣וּ מְלָכִ֔ים הַאֲזִ֖ינוּ רֹֽזְנִ֑ים אָֽנֹכִ֗י לַֽיהוָה֙ אָנֹכִ֣י אָשִׁ֔ירָה אֲזַמֵּ֕ר לַֽיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (כט) יְהוָ֗ה בְּצֵאתְךָ֤ מִשֵּׂעִיר֙ בְּצַעְדְּךָ֙ מִשְּׂדֵ֣ה אֱד֔וֹם אֶ֣רֶץ רָעָ֔שָׁה גַּם־ שָׁמַ֖יִם נָטָ֑פוּ גַּם־ עָבִ֖ים נָ֥טְפוּ מָֽיִם׃ (ל) הָרִ֥ים נָזְל֖וּ מִפְּנֵ֣י יְהוָ֑ה זֶ֣ה סִינַ֔י מִפְּנֵ֕י יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (לא) בִּימֵ֞י שַׁמְגַּ֤ר בֶּן־ עֲנָת֙ בִּימֵ֣י יָעֵ֔ל חָדְל֖וּ אֳרָח֑וֹת וְהֹלְכֵ֣י נְתִיב֔וֹת יֵלְכ֕וּ אֳרָח֖וֹת עֲקַלְקַלּֽוֹת׃ (לב) חָדְל֧וּ פְרָז֛וֹן בְּיִשְׂרָאֵ֖ל חָדֵ֑לּוּ עַ֤ד שַׁקַּ֙מְתִּי֙ דְּבוֹרָ֔ה שַׁקַּ֥מְתִּי אֵ֖ם בְּיִשְׂרָאֵֽל׃ (לג) יִבְחַר֙ אֱלֹהִ֣ים חֲדָשִׁ֔ים אָ֖ז לָחֶ֣ם שְׁעָרִ֑ים מָגֵ֤ן אִם־ יֵֽרָאֶה֙ וָרֹ֔מַח בְּאַרְבָּעִ֥ים אֶ֖לֶף בְּיִשְׂרָאֵֽל׃ (לד) לִבִּי֙ לְחוֹקְקֵ֣י יִשְׂרָאֵ֔ל הַמִּֽתְנַדְּבִ֖ים בָּעָ֑ם בָּרֲכ֖וּ יְהוָֽה׃ (לה) רֹכְבֵי֩ אֲתֹנ֨וֹת צְחֹר֜וֹת יֹשְׁבֵ֧י עַל־ מִדִּ֛ין וְהֹלְכֵ֥י עַל־ דֶּ֖רֶךְ שִֽׂיחוּ׃ (לו) מִקּ֣וֹל מְחַֽצְצִ֗ים בֵּ֚ין מַשְׁאַבִּ֔ים שָׁ֤ם יְתַנּוּ֙ צִדְק֣וֹת יְהוָ֔ה צִדְקֹ֥ת פִּרְזֹנ֖וֹ בְּיִשְׂרָאֵ֑ל אָ֛ז יָרְד֥וּ לַשְּׁעָרִ֖ים עַם־ יְהוָֽה׃ (לז) עוּרִ֤י עוּרִי֙ דְּבוֹרָ֔ה ע֥וּרִי ע֖וּרִי דַּבְּרִי־ שִׁ֑יר ק֥וּם בָּרָ֛ק וּֽשֲׁבֵ֥ה שֶׁבְיְךָ֖ בֶּן־ אֲבִינֹֽעַם׃ (לח) אָ֚ז יְרַ֣ד שָׂרִ֔יד לְאַדִּירִ֖ים עָ֑ם יְהוָ֕ה יְרַד־ לִ֖י בַּגִּבּוֹרִֽים׃ (לט) מִנִּ֣י אֶפְרַ֗יִם שָׁרְשָׁם֙ בַּעֲמָלֵ֔ק אַחֲרֶ֥יךָ בִנְיָמִ֖ין בַּֽעֲמָמֶ֑יךָ מִנִּ֣י מָכִ֗יר יָֽרְדוּ֙ מְחֹ֣קְקִ֔ים וּמִ֨זְּבוּלֻ֔ן מֹשְׁכִ֖ים בְּשֵׁ֥בֶט סֹפֵֽר׃ (מ) וְשָׂרַ֤י בְּיִשָּׂשכָר֙ עִם־ דְּבֹרָ֔ה וְיִשָּׂשכָר֙ כֵּ֣ן בָּרָ֔ק בָּעֵ֖מֶק שֻׁלַּ֣ח בְּרַגְלָ֑יו בִּפְלַגּ֣וֹת רְאוּבֵ֔ן גְּדֹלִ֖ים חִקְקֵי־ לֵֽב׃ (מא) לָ֣מָּה יָשַׁ֗בְתָּ בֵּ֚ין הַֽמִּשְׁפְּתַ֔יִם לִשְׁמֹ֖עַ שְׁרִק֣וֹת עֲדָרִ֑ים לִפְלַגּ֣וֹת רְאוּבֵ֔ן גְּדוֹלִ֖ים חִקְרֵי־ לֵֽב׃ (מב) גִּלְעָ֗ד בְּעֵ֤בֶר הַיַּרְדֵּן֙ שָׁכֵ֔ן וְדָ֕ן לָ֥מָּה יָג֖וּר אֳנִיּ֑וֹת אָשֵׁ֗ר יָשַׁב֙ לְח֣וֹף יַמִּ֔ים וְעַ֥ל מִפְרָצָ֖יו יִשְׁכּֽוֹן׃ (מג) זְבֻל֗וּן עַ֣ם חֵרֵ֥ף נַפְשׁ֛וֹ לָמ֖וּת וְנַפְתָּלִ֑י עַ֖ל מְרוֹמֵ֥י שָׂדֶֽה׃ (מד) בָּ֤אוּ מְלָכִים֙ נִלְחָ֔מוּ אָ֤ז נִלְחֲמוּ֙ מַלְכֵ֣י כְנַ֔עַן בְּתַעְנַ֖ךְ עַל־ מֵ֣י מְגִדּ֑וֹ בֶּ֥צַע כֶּ֖סֶף לֹ֥א לָקָֽחוּ׃ (מה) מִן־ שָׁמַ֖יִם נִלְחָ֑מוּ הַכּֽוֹכָבִים֙ מִמְּסִלּוֹתָ֔ם נִלְחֲמ֖וּ עִם־ סִיסְרָֽא׃ (מו) נַ֤חַל קִישׁוֹן֙ גְּרָפָ֔ם נַ֥חַל קְדוּמִ֖ים נַ֣חַל קִישׁ֑וֹן תִּדְרְכִ֥י נַפְשִׁ֖י עֹֽז׃ (מז) אָ֥ז הָלְמ֖וּ עִקְּבֵי־ ס֑וּס מִֽדַּהֲר֖וֹת דַּהֲר֥וֹת אַבִּירָֽיו׃ (מח) א֣וֹרוּ מֵר֗וֹז אָמַר֙ מַלְאַ֣ךְ יְהוָ֔ה אֹ֥רוּ אָר֖וֹר יֹשְׁבֶ֑יהָ כִּ֤י לֹֽא־ בָ֙אוּ֙ לְעֶזְרַ֣ת יְהוָ֔ה לְעֶזְרַ֥ת יְהוָ֖ה בַּגִּבּוֹרִֽים׃ (מט) תְּבֹרַךְ֙ מִנָּשִׁ֔ים יָעֵ֕ל אֵ֖שֶׁת חֶ֣בֶר הַקֵּינִ֑י מִנָּשִׁ֥ים בָּאֹ֖הֶל תְּבֹרָֽךְ׃ (נ) מַ֥יִם שָׁאַ֖ל חָלָ֣ב נָתָ֑נָה בְּסֵ֥פֶל אַדִּירִ֖ים הִקְרִ֥יבָה חֶמְאָֽה׃ (נא) יָדָהּ֙ לַיָּתֵ֣ד תִּשְׁלַ֔חְנָה וִֽימִינָ֖הּ לְהַלְמ֣וּת עֲמֵלִ֑ים וְהָלְמָ֤ה סִֽיסְרָא֙ מָחֲקָ֣ה רֹאשׁ֔וֹ וּמָחֲצָ֥ה וְחָלְפָ֖ה רַקָּתֽוֹ׃ (נב) בֵּ֣ין רַגְלֶ֔יהָ כָּרַ֥ע נָפַ֖ל שָׁכָ֑ב בֵּ֤ין רַגְלֶ֙יהָ֙ כָּרַ֣ע נָפָ֔ל בַּאֲשֶׁ֣ר כָּרַ֔ע שָׁ֖ם נָפַ֥ל שָׁדֽוּד׃ (נג) בְּעַד֩ הַחַלּ֨וֹן נִשְׁקְפָ֧ה וַתְּיַבֵּ֛ב אֵ֥ם סִֽיסְרָ֖א בְּעַ֣ד הָֽאֶשְׁנָ֑ב מַדּ֗וּעַ בֹּשֵׁ֤שׁ רִכְבּוֹ֙ לָב֔וֹא מַדּ֣וּעַ אֶֽחֱר֔וּ פַּעֲמֵ֖י מַרְכְּבוֹתָֽיו׃ (נד) חַכְמ֥וֹת שָׂרוֹתֶ֖יהָ תַּעֲנֶ֑ינָּה אַף־ הִ֕יא תָּשִׁ֥יב אֲמָרֶ֖יהָ לָֽהּ׃ (נה) הֲלֹ֨א יִמְצְא֜וּ יְחַלְּק֣וּ שָׁלָ֗ל רַ֤חַם רַחֲמָתַ֙יִם֙ לְרֹ֣אשׁ גֶּ֔בֶר שְׁלַ֤ל צְבָעִים֙ לְסִ֣יסְרָ֔א שְׁלַ֥ל צְבָעִ֖ים רִקְמָ֑ה צֶ֥בַע רִקְמָתַ֖יִם לְצַוְּארֵ֥י שָׁלָֽל׃
