שופטים פרק כ לפי החלוקה לסדרים

פרק כ בספר שופטים (48 פסוקים) נמצא בשופטים סדר יג (פסוקים יב–לז) ובשופטים סדר יד (פסוקים א–כב).

שופטים סדר יג

(א) וַיֹּ֨אמֶר הָאִ֤ישׁ הַזָּקֵן֙ שָׁל֣וֹם לָ֔ךְ רַ֥ק כָּל־ מַחְסוֹרְךָ֖ עָלָ֑י רַ֥ק בָּרְח֖וֹב אַל־ תָּלַֽן׃ (ב) וַיְבִיאֵ֣הוּ לְבֵית֔וֹ וַיָּ֖בָול לַחֲמוֹרִ֑ים וַֽיִּרְחֲצוּ֙ רַגְלֵיהֶ֔ם וַיֹּאכְל֖וּ וַיִּשְׁתּֽוּ׃ (ג) הֵמָּה֮ מֵיטִיבִ֣ים אֶת־ לִבָּם֒ וְהִנֵּה֩ אַנְשֵׁ֨י הָעִ֜יר אַנְשֵׁ֣י בְנֵֽי־ בְלִיַּ֗עַל נָסַ֙בּוּ֙ אֶת־ הַבַּ֔יִת מִֽתְדַּפְּקִ֖ים עַל־ הַדָּ֑לֶת וַיֹּאמְר֗וּ אֶל־ הָ֠אִישׁ בַּ֣עַל הַבַּ֤יִת הַזָּקֵן֙ לֵאמֹ֔ר הוֹצֵ֗א אֶת־ הָאִ֛ישׁ אֲשֶׁר־ בָּ֥א אֶל־ בֵּיתְךָ֖ וְנֵדָעֶֽנּוּ׃ (ד) וַיֵּצֵ֣א אֲלֵיהֶ֗ם הָאִישׁ֙ בַּ֣עַל הַבַּ֔יִת וַיֹּ֣אמֶר אֲלֵהֶ֔ם אַל־ אַחַ֖י אַל־ תָּרֵ֣עוּ נָ֑א אַ֠חֲרֵי אֲשֶׁר־ בָּ֞א הָאִ֤ישׁ הַזֶּה֙ אַל־ בֵּיתִ֔י אַֽל־ תַּעֲשׂ֖וּ אֶת־ הַנְּבָלָ֥ה הַזֹּֽאת׃ (ה) הִנֵּה֩ בִתִּ֨י הַבְּתוּלָ֜ה וּפִֽילַגְשֵׁ֗הוּ אוֹצִֽיאָה־ נָּ֤א אוֹתָם֙ וְעַנּ֣וּ אוֹתָ֔ם וַעֲשׂ֣וּ לָהֶ֔ם הַטּ֖וֹב בְּעֵינֵיכֶ֑ם וְלָאִ֤ישׁ הַזֶּה֙ לֹ֣א תַעֲשׂ֔וּ דְּבַ֖ר הַנְּבָלָ֥ה הַזֹּֽאת׃ (ו) וְלֹֽא־ אָב֤וּ הָאֲנָשִׁים֙ לִשְׁמֹ֣עַֽ ל֔וֹ וַיַּחֲזֵ֤ק הָאִישׁ֙ בְּפִ֣ילַגְשׁ֔וֹ וַיֹּצֵ֥א אֲלֵיהֶ֖ם הַח֑וּץ וַיֵּדְע֣וּ א֠וֹתָהּ וַיִּֽתְעַלְּלוּ־ בָ֤הּ כָּל־ הַלַּ֙יְלָה֙ עַד־ הַבֹּ֔קֶר וַֽיְשַׁלְּח֖וּהָ (בעלות) כַּעֲל֥וֹת הַשָּֽׁחַר׃ (ז) וַתָּבֹ֥א הָאִשָּׁ֖ה לִפְנ֣וֹת הַבֹּ֑קֶר וַתִּפֹּ֞ל פֶּ֧תַח בֵּית־ הָאִ֛ישׁ אֲשֶׁר־ אֲדוֹנֶ֥יהָ שָּׁ֖ם עַד־ הָאֽוֹר׃ (ח) וַיָּ֨קָם אֲדֹנֶ֜יהָ בַּבֹּ֗קֶר וַיִּפְתַּח֙ דַּלְת֣וֹת הַבַּ֔יִת וַיֵּצֵ֖א לָלֶ֣כֶת לְדַרְכּ֑וֹ וְהִנֵּ֧ה הָאִשָּׁ֣ה פִֽילַגְשׁ֗וֹ נֹפֶ֙לֶת֙ פֶּ֣תַח הַבַּ֔יִת וְיָדֶ֖יהָ עַל־ הַסַּֽף׃ (ט) וַיֹּ֧אמֶר אֵלֶ֛יהָ ק֥וּמִי וְנֵלֵ֖כָה וְאֵ֣ין עֹנֶ֑ה וַיִּקָּחֶ֙הָ֙ עַֽל־ הַחֲמ֔וֹר וַיָּ֣קָם הָאִ֔ישׁ וַיֵּ֖לֶךְ לִמְקֹמֽוֹ׃ (י) וַיָּבֹ֣א אֶל־ בֵּית֗וֹ וַיִּקַּ֤ח אֶת־ הַֽמַּאֲכֶ֙לֶת֙ וַיַּחֲזֵ֣ק בְּפִֽילַגְשׁ֔וֹ וַֽיְנַתְּחֶ֙הָ֙ לַעֲצָמֶ֔יהָ לִשְׁנֵ֥ים עָשָׂ֖ר נְתָחִ֑ים וַֽיְשַׁלְּחֶ֔הָ בְּכֹ֖ל גְּב֥וּל יִשְׂרָאֵֽל׃ (יא) וְהָיָ֣ה כָל־ הָרֹאֶ֗ה וְאָמַר֙ לֹֽא־ נִהְיְתָ֤ה וְלֹֽא־ נִרְאֲתָה֙ כָּזֹ֔את לְמִיּ֞וֹם עֲל֤וֹת בְּנֵֽי־ יִשְׂרָאֵל֙ מֵאֶ֣רֶץ מִצְרַ֔יִם עַ֖ד הַיּ֣וֹם הַזֶּ֑ה שִֽׂימוּ־ לָכֶ֥ם עָלֶ֖יהָ עֻ֥צוּ וְדַבֵּֽרוּ׃ תחילת פרק כ (יב) וַיֵּצְאוּ֮ כָּל־ בְּנֵ֣י יִשְׂרָאֵל֒ וַתִּקָּהֵ֨ל הָעֵדָ֜ה כְּאִ֣ישׁ אֶחָ֗ד לְמִדָּן֙ וְעַד־ בְּאֵ֣ר שֶׁ֔בַע וְאֶ֖רֶץ הַגִּלְעָ֑ד אֶל־ יְהוָ֖ה הַמִּצְפָּֽה׃ (יג) וַיִּֽתְיַצְּב֞וּ פִּנּ֣וֹת כָּל־ הָעָ֗ם כֹּ֚ל שִׁבְטֵ֣י יִשְׂרָאֵ֔ל בִּקְהַ֖ל עַ֣ם הָאֱלֹהִ֑ים אַרְבַּ֨ע מֵא֥וֹת אֶ֛לֶף אִ֥ישׁ רַגְלִ֖י שֹׁ֥לֵֽף חָֽרֶב׃ (יד) וַֽיִּשְׁמְעוּ֙ בְּנֵ֣י בִנְיָמִ֔ן כִּֽי־ עָל֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל הַמִּצְפָּ֑ה וַיֹּֽאמְרוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל דַּבְּר֕וּ אֵיכָ֥ה נִהְיְתָ֖ה הָרָעָ֥ה הַזֹּֽאת׃ (טו) וַיַּ֜עַן הָאִ֣ישׁ הַלֵּוִ֗י אִ֛ישׁ הָאִשָּׁ֥ה הַנִּרְצָחָ֖ה וַיֹּאמַ֑ר הַגִּבְעָ֙תָה֙ אֲשֶׁ֣ר לְבִנְיָמִ֔ן בָּ֛אתִי אֲנִ֥י וּפִֽילַגְשִׁ֖י לָלֽוּן׃ (טז) וַיָּקֻ֤מוּ עָלַי֙ בַּעֲלֵ֣י הַגִּבְעָ֔ה וַיָּסֹ֧בּוּ עָלַ֛י אֶת־ הַבַּ֖יִת לָ֑יְלָה אוֹתִי֙ דִּמּ֣וּ לַהֲרֹ֔ג וְאֶת־ פִּילַגְשִׁ֥י עִנּ֖וּ וַתָּמֹֽת׃ (יז) וָֽאֹחֵ֤ז בְּפִֽילַגְשִׁי֙ וָֽאֲנַתְּחֶ֔הָ וָֽאֲשַׁלְּחֶ֔הָ בְּכָל־ שְׂדֵ֖ה נַחֲלַ֣ת יִשְׂרָאֵ֑ל כִּ֥י עָשׂ֛וּ זִמָּ֥ה וּנְבָלָ֖ה בְּיִשְׂרָאֵֽל׃ (יח) הִנֵּ֥ה כֻלְּכֶ֖ם בְּנֵ֣י יִשְׂרָאֵ֑ל הָב֥וּ לָכֶ֛ם דָּבָ֥ר וְעֵצָ֖ה הֲלֹֽם׃ (יט) וַיָּ֙קָם֙ כָּל־ הָעָ֔ם כְּאִ֥ישׁ אֶחָ֖ד לֵאמֹ֑ר לֹ֤א נֵלֵךְ֙ אִ֣ישׁ לְאָהֳל֔וֹ וְלֹ֥א נָס֖וּר אִ֥ישׁ לְבֵיתֽוֹ׃ (כ) וְעַתָּ֕ה זֶ֣ה הַדָּבָ֔ר אֲשֶׁ֥ר נַעֲשֶׂ֖ה לַגִּבְעָ֑ה עָלֶ֖יהָ בְּגוֹרָֽל׃ (כא) וְלָקַ֣חְנוּ עֲשָׂרָה֩ אֲנָשִׁ֨ים לַמֵּאָ֜ה לְכֹ֣ל ׀ שִׁבְטֵ֣י יִשְׂרָאֵ֗ל וּמֵאָ֤ה לָאֶ֙לֶף֙ וְאֶ֣לֶף לָרְבָבָ֔ה לָקַ֥חַת צֵדָ֖ה לָעָ֑ם לַעֲשׂ֗וֹת לְבוֹאָם֙ לְגֶ֣בַע בִּנְיָמִ֔ן כְּכָל־ הַ֨נְּבָלָ֔ה אֲשֶׁ֥ר עָשָׂ֖ה בְּיִשְׂרָאֵֽל׃ (כב) וַיֵּֽאָסֵ֞ף כָּל־ אִ֤ישׁ יִשְׂרָאֵל֙ אֶל־ הָעִ֔יר כְּאִ֥ישׁ אֶחָ֖ד חֲבֵרִֽים׃ (כג) וַֽיִּשְׁלְח֞וּ שִׁבְטֵ֤י יִשְׂרָאֵל֙ אֲנָשִׁ֔ים בְּכָל־ שִׁבְטֵ֥י בִנְיָמִ֖ן לֵאמֹ֑ר מָ֚ה הָרָעָ֣ה הַזֹּ֔את אֲשֶׁ֥ר נִהְיְתָ֖ה בָּכֶֽם׃ (כד) וְעַתָּ֡ה תְּנוּ֩ אֶת־ הָאֲנָשִׁ֨ים בְּנֵֽי־ בְלִיַּ֜עַל אֲשֶׁ֤ר בַּגִּבְעָה֙ וּנְמִיתֵ֔ם וּנְבַעֲרָ֥ה רָעָ֖ה מִיִּשְׂרָאֵ֑ל וְלֹ֤א אָבוּ֙ בְּנֵ֣י בִּנְיָמִ֔ן לִשְׁמֹ֕עַ בְּק֖וֹל אֲחֵיהֶ֥ם בְּנֵֽי־ יִשְׂרָאֵֽל׃ (כה) וַיֵּאָסְפ֧וּ בְנֵֽי־ בִנְיָמִ֛ן מִן־ הֶעָרִ֖ים הַגִּבְעָ֑תָה לָצֵ֥את לַמִּלְחָמָ֖ה עִם־ בְּנֵ֥י יִשְׂרָאֵֽל׃ (כו) וַיִּתְפָּֽקְדוּ֩ בְנֵ֨י בִנְיָמִ֜ן בַּיּ֤וֹם הַהוּא֙ מֵהֶ֣עָרִ֔ים עֶשְׂרִ֨ים וְשִׁשָּׁ֥ה אֶ֛לֶף אִ֖ישׁ שֹׁ֣לֵֽף חָ֑רֶב לְ֠בַד מִיֹּשְׁבֵ֤י הַגִּבְעָה֙ הִתְפָּ֣קְד֔וּ שְׁבַ֥ע מֵא֖וֹת אִ֥ישׁ בָּחֽוּר׃ (כז) מִכֹּ֣ל ׀ הָעָ֣ם הַזֶּ֗ה שְׁבַ֤ע מֵאוֹת֙ אִ֣ישׁ בָּח֔וּר אִטֵּ֖ר יַד־ יְמִינ֑וֹ כָּל־ זֶ֗ה קֹלֵ֧עַ בָּאֶ֛בֶן אֶל־ הַֽשַּׂעֲרָ֖ה וְלֹ֥א יַחֲטִֽא׃ (כח) וְאִ֨ישׁ יִשְׂרָאֵ֜ל הִתְפָּֽקְד֗וּ לְבַד֙ מִבִּנְיָמִ֔ן אַרְבַּ֨ע מֵא֥וֹת אֶ֛לֶף אִ֖ישׁ שֹׁ֣לֵֽף חָ֑רֶב כָּל־ זֶ֖ה אִ֥ישׁ מִלְחָמָֽה׃ (כט) וַיָּקֻ֜מוּ וַיַּעֲל֣וּ בֵֽית־ אֵל֮ וַיִּשְׁאֲל֣וּ בֵאלֹהִים֒ וַיֹּֽאמְרוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל מִ֚י יַעֲלֶה־ לָּ֣נוּ בַתְּחִלָּ֔ה לַמִּלְחָמָ֖ה עִם־ בְּנֵ֣י בִנְיָמִ֑ן וַיֹּ֥אמֶר יְהוָ֖ה יְהוּדָ֥ה בַתְּחִלָּֽה׃ (ל) וַיָּק֥וּמוּ בְנֵֽי־ יִשְׂרָאֵ֖ל בַּבֹּ֑קֶר וַיַּֽחֲנ֖וּ עַל־ הַגִּבְעָֽה׃ (לא) וַיֵּצֵא֙ אִ֣ישׁ יִשְׂרָאֵ֔ל לַמִּלְחָמָ֖ה עִם־ בִּנְיָמִ֑ן וַיַּעַרְכ֨וּ אִתָּ֧ם אִֽישׁ־ יִשְׂרָאֵ֛ל מִלְחָמָ֖ה אֶל־ הַגִּבְעָֽה׃ (לב) וַיֵּצְא֥וּ בְנֵֽי־ בִנְיָמִ֖ן מִן־ הַגִּבְעָ֑ה וַיַּשְׁחִ֨יתוּ בְיִשְׂרָאֵ֜ל בַּיּ֣וֹם הַה֗וּא שְׁנַ֨יִם וְעֶשְׂרִ֥ים אֶ֛לֶף אִ֖ישׁ אָֽרְצָה׃ (לג) וַיִּתְחַזֵּ֥ק הָעָ֖ם אִ֣ישׁ יִשְׂרָאֵ֑ל וַיֹּסִ֙פוּ֙ לַעֲרֹ֣ךְ מִלְחָמָ֔ה בַּמָּק֕וֹם אֲשֶׁר־ עָ֥רְכוּ שָׁ֖ם בַּיּ֥וֹם הָרִאשֽׁוֹן׃ (לד) וַיַּעֲל֣וּ בְנֵֽי־ יִשְׂרָאֵ֗ל וַיִּבְכּ֣וּ לִפְנֵֽי־ יְהוָה֮ עַד־ הָעֶרֶב֒ וַיִּשְׁאֲל֤וּ בַֽיהוָה֙ לֵאמֹ֔ר הַאוֹסִ֗יף לָגֶ֙שֶׁת֙ לַמִּלְחָמָ֔ה עִם־ בְּנֵ֥י בִנְיָמִ֖ן אָחִ֑י וַיֹּ֥אמֶר יְהוָ֖ה עֲל֥וּ אֵלָֽיו׃ (לה) וַיִּקְרְב֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶל־ בְּנֵ֥י בִנְיָמִ֖ן בַּיּ֥וֹם (א) וַיִּשְׁאֲל֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל בַּֽיהוָ֑ה וְשָׁ֗ם אֲרוֹן֙ בְּרִ֣ית הָאֱלֹהִ֔ים בַּיָּמִ֖ים הָהֵֽם׃ (ב) וּ֠פִינְחָס בֶּן־ אֶלְעָזָ֨ר בֶּֽן־ אַהֲרֹ֜ן עֹמֵ֣ד ׀ לְפָנָ֗יו בַּיָּמִ֣ים הָהֵם֮ לֵאמֹר֒ הַאוֹסִ֨ף ע֜וֹד לָצֵ֧את לַמִּלְחָמָ֛ה עִם־ בְּנֵֽי־ בִנְיָמִ֥ן אָחִ֖י אִם־ אֶחְדָּ֑ל וַיֹּ֤אמֶר יְהוָה֙ עֲל֔וּ כִּ֥י מָחָ֖ר אֶתְּנֶ֥נּוּ בְיָדֶֽךָ׃ (ג) וַיָּ֤שֶׂם יִשְׂרָאֵל֙ אֹֽרְבִ֔ים אֶל־ הַגִּבְעָ֖ה סָבִֽיב׃ (ד) וַיַּעֲל֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶל־ בְּנֵ֥י בִנְיָמִ֖ן בַּיּ֣וֹם הַשְּׁלִישִׁ֑י וַיַּעַרְכ֥וּ אֶל־ הַגִּבְעָ֖ה כְּפַ֥עַם בְּפָֽעַם׃ (ה) וַיֵּצְא֤וּ בְנֵֽי־ בִנְיָמִן֙ לִקְרַ֣את הָעָ֔ם הָנְתְּק֖וּ מִן־ הָעִ֑יר וַיָּחֵ֡לּוּ לְהַכּוֹת֩ מֵהָעָ֨ם חֲלָלִ֜ים כְּפַ֣עַם ׀ בְּפַ֗עַם בַּֽמְסִלּוֹת֙ אֲשֶׁ֨ר אַחַ֜ת עֹלָ֣ה בֵֽית־ אֵ֗ל וְאַחַ֤ת גִּבְעָ֙תָה֙ בַּשָּׂדֶ֔ה כִּשְׁלֹשִׁ֥ים אִ֖ישׁ בְּיִשְׂרָאֵֽל׃ (ו) וַיֹּֽאמְרוּ֙ בְּנֵ֣י בִנְיָמִ֔ן נִגָּפִ֥ים הֵ֛ם לְפָנֵ֖ינוּ כְּבָרִאשֹׁנָ֑ה וּבְנֵ֧י יִשְׂרָאֵ֣ל אָמְר֗וּ נָנ֙וּסָה֙ וּֽנְתַקְּנֻ֔הוּ מִן־ הָעִ֖יר אֶל־ הַֽמְסִלּֽוֹת׃ (ז) וְכֹ֣ל ׀ אִ֣ישׁ יִשְׂרָאֵ֗ל קָ֚מוּ מִמְּקוֹמ֔וֹ וַיַּעַרְכ֖וּ בְּבַ֣עַל תָּמָ֑ר וְאֹרֵ֧ב יִשְׂרָאֵ֛ל מֵגִ֥יחַ מִמְּקֹמ֖וֹ מִמַּֽעֲרֵה־ גָֽבַע׃ (ח) וַיָּבֹאוּ֩ מִנֶּ֨גֶד לַגִּבְעָ֜ה עֲשֶׂרֶת֩ אֲלָפִ֨ים אִ֤ישׁ בָּחוּר֙ מִכָּל־ יִשְׂרָאֵ֔ל וְהַמִּלְחָמָ֖ה כָּבֵ֑דָה וְהֵם֙ לֹ֣א יָדְע֔וּ כִּֽי־ נֹגַ֥עַת עֲלֵיהֶ֖ם הָרָעָֽה׃ (ט) וַיִּגֹּ֨ף יְהוָ֥ה ׀ אֶֽת־ בִּנְיָמִן֮ לִפְנֵ֣י יִשְׂרָאֵל֒ וַיַּשְׁחִיתוּ֩ בְנֵ֨י יִשְׂרָאֵ֤ל בְּבִנְיָמִן֙ בַּיּ֣וֹם הַה֔וּא עֶשְׂרִ֨ים וַחֲמִשָּׁ֥ה אֶ֛לֶף וּמֵאָ֖ה אִ֑ישׁ כָּל־ אֵ֖לֶּה שֹׁ֥לֵף חָֽרֶב׃ (י) וַיִּרְא֥וּ בְנֵֽי־ בִנְיָמִ֖ן כִּ֣י נִגָּ֑פוּ וַיִּתְּנ֨וּ אִֽישׁ־ יִשְׂרָאֵ֤ל מָקוֹם֙ לְבִנְיָמִ֔ן כִּ֤י בָֽטְחוּ֙ אֶל־ הָ֣אֹרֵ֔ב אֲשֶׁר שָׂ֖מוּ אֶל־ הַגִּבְעָֽה׃ (יא) וְהָאֹרֵ֣ב הֵחִ֔ישׁוּ וַֽיִּפְשְׁט֖וּ אֶל־ הַגִּבְעָ֑ה וַיִּמְשֹׁךְ֙ הָאֹרֵ֔ב וַיַּ֥ךְ אֶת־ כָּל־ הָעִ֖יר לְפִי־ חָֽרֶב׃ (יב) וְהַמּוֹעֵ֗ד הָיָ֛ה לְאִ֥ישׁ יִשְׂרָאֵ֖ל עִם־ הָאֹרֵ֑ב הֶ֕רֶב לְהַעֲלוֹתָ֛ם מַשְׂאַ֥ת הֶעָשָׁ֖ן מִן־ הָעִֽיר׃ (יג) וַיַּהֲפֹ֥ךְ אִֽישׁ־ יִשְׂרָאֵ֖ל בַּמִּלְחָמָ֑ה וּבִנְיָמִ֡ן הֵחֵל֩ לְהַכּ֨וֹת חֲלָלִ֤ים בְּאִֽישׁ־ יִשְׂרָאֵל֙ כִּשְׁלֹשִׁ֣ים אִ֔ישׁ כִּ֣י אָמְר֔וּ אַךְ֩ נִגּ֨וֹף נִגָּ֥ף הוּא֙ לְפָנֵ֔ינוּ כַּמִּלְחָמָ֖ה הָרִאשֹׁנָֽה׃ (יד) וְהַמַּשְׂאֵ֗ת הֵחֵ֛לָּה לַעֲל֥וֹת מִן־ הָעִ֖יר עַמּ֣וּד עָשָׁ֑ן וַיִּ֤פֶן בִּנְיָמִן֙ אַחֲרָ֔יו וְהִנֵּ֛ה עָלָ֥ה כְלִיל־ הָעִ֖יר הַשָּׁמָֽיְמָה׃ (טו) וְאִ֤ישׁ יִשְׂרָאֵל֙ הָפַ֔ךְ מֵהֶ֣עָרִ֔ים מַשְׁחִיתִ֥ים אוֹת֖וֹ בְּתוֹכֽוֹ׃ (יז) כִּתְּר֤וּ אֶת־ בִּנְיָמִן֙ הִרְדִיפֻ֔הוּ מְנוּחָ֖ה הִדְרִיכֻ֑הוּ עַ֛ד נֹ֥כַח הַגִּבְעָ֖ה מִמִּזְרַח־ שָֽׁמֶשׁ׃ (יח) וַֽיִּפְּלוּ֙ מִבִּנְיָמִ֔ן מִבְּנוֹתֵ֖ינוּ לְנָשִֽׁים׃ (ל) וַיֹּ֣אמְר֔וּ מִ֗י אֶחָד֙ מִשִּׁבְטֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֛ר לֹֽא־ עָלָ֥ה אֶל־ יְהוָ֖ה הַמִּצְפָּ֑ה וְ֠הִנֵּה לֹ֣א בָא־ אִ֧ישׁ אֶל־ הַֽמַּחֲנֶ֛ה מִיָּבֵ֥ישׁ גִּלְעָ֖ד אֶל־ הַקָּהָֽל׃ (לא) וַיִּתְפָּקֵ֖ד הָעָ֑ם וְהִנֵּ֤ה אֵֽין־ שָׁם֙ אִ֔ישׁ מִיּוֹשְׁבֵ֖י יָבֵ֥שׁ גִּלְעָֽד׃ (לב) וַיִּשְׁלְחוּ־ שָׁ֣ם הָעֵדָ֗ה שְׁנֵים־ עָשָׂ֥ר אֶ֛לֶף אִ֖ישׁ מִבְּנֵ֣י הֶחָ֑יִל וַיְצַוּ֨וּ אוֹתָ֜ם לֵאמֹ֗ר לְ֠כוּ וְהִכִּיתֶ֞ם אֶת־ יוֹשְׁבֵ֨י יָבֵ֤שׁ גִּלְעָד֙ לְפִי־ חֶ֔רֶב וְהַנָּשִׁ֖ים וְהַטָּֽף׃ (לג) וְזֶ֥ה הַדָּבָ֖ר אֲשֶׁ֣ר תַּעֲשׂ֑וּ כָּל־ זָכָ֗ר וְכָל־ אִשָּׁ֛ה יֹדַ֥עַת מִשְׁכַּב־ זָכָ֖ר תַּחֲרִֽימוּ׃ (לד) וַֽיִּמְצְא֞וּ מִיּוֹשְׁבֵ֣י ׀ יָבֵ֣ישׁ גִּלְעָ֗ד אַרְבַּ֤ע מֵאוֹת֙ נַעֲרָ֣ה בְתוּלָ֔ה אֲשֶׁ֧ר לֹֽא־ יָדְעָ֛ה אִ֖ישׁ לְמִשְׁכַּ֣ב זָכָ֑ר וַיָּבִ֨יאוּ אוֹתָ֤ם אֶל־ הַֽמַּחֲנֶה֙ שִׁלֹ֔ה אֲשֶׁ֖ר בְּאֶ֥רֶץ כְּנָֽעַן׃ (לה) וַֽיִּשְׁלְחוּ֙ כָּל־ הָ֣עֵדָ֔ה וַֽיְדַבְּרוּ֙ אֶל־ בְּנֵ֣י בִנְיָמִ֔ן אֲשֶׁ֖ר בְּסֶ֣לַע רִמּ֑וֹן וַיִּקְרְא֥וּ לָהֶ֖ם שָׁלֽוֹם׃ (לו) וַיָּ֤שָׁב בִּנְיָמִן֙ בָּעֵ֣ת הַהִ֔יא וַיִּתְּנ֤וּ לָהֶם֙ הַנָּשִׁ֔ים אֲשֶׁ֣ר חִיּ֔וּ מִנְּשֵׁ֖י יָבֵ֣שׁ גִּלְעָ֑ד וְלֹֽא־ מָצְא֥וּ לָהֶ֖ם כֵּֽן׃ (לז) וְהָעָ֥ם נִחָ֖ם לְבִנְיָמִ֑ן כִּֽי־ עָשָׂ֧ה יְהוָ֛ה פֶּ֖רֶץ בְּשִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (לח) וַיֹּֽאמְר֨וּ זִקְנֵ֣י הָעֵדָ֔ה מַה־ נַּעֲשֶׂ֥ה לַנּוֹתָרִ֖ים לְנָשִׁ֑ים כִּֽי־ נִשְׁמְדָ֥ה מִבִּנְיָמִ֖ן אִשָּֽׁה׃ (לט) וַיֹּ֣אמְר֔וּ יְרֻשַּׁ֥ת פְּלֵיטָ֖ה לְבִנְיָמִ֑ן וְלֹֽא־ יִמָּחֶ֥ה שֵׁ֖בֶט מִיִּשְׂרָאֵֽל׃ (מ) וַאֲנַ֗חְנוּ לֹ֥א נוּכַ֛ל לָתֵת־ לָהֶ֥ם נָשִׁ֖ים מִבְּנוֹתֵ֑ינוּ כִּֽי־ נִשְׁבְּע֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ לֵאמֹ֔ר אָר֕וּר נֹתֵ֥ן אִשָּׁ֖ה לְבִנְיָמִֽן׃ (מא) וַיֹּאמְר֡וּ הִנֵּה֩ חַג־ יְהוָ֨ה בְּשִׁל֜וֹ מִיָּמִ֣ים ׀ יָמִ֗ימָה אֲשֶׁ֞ר מִצְּפ֤וֹנָה לְבֵֽית־ אֵל֙ מִזְרְחָ֣ה הַשֶּׁ֔מֶשׁ לִמְסִלָּ֔ה הָעֹלָ֥ה מִבֵּֽית־ אֵ֖ל שְׁכֶ֑מָה וּמִנֶּ֖גֶב לִלְבוֹנָֽה׃ (מב) (ויצו) וַיְצַוּ֕וּ אֶת־ בְּנֵ֥י בִנְיָמִ֖ן לֵאמֹ֑ר לְכ֖וּ וַאֲרַבְתֶּ֥ם בַּכְּרָמִֽים׃ (מג) וּרְאִיתֶ֗ם וְ֠הִנֵּה אִם־ יֵ֨צְא֥וּ בְנוֹת־ שִׁילוֹ֮ לָח֣וּל בַּמְּחֹלוֹת֒ וִֽיצָאתֶם֙ מִן־ הַכְּרָמִ֔ים וַחֲטַפְתֶּ֥ם לָכֶ֛ם אִ֥ישׁ אִשְׁתּ֖וֹ מִבְּנ֣וֹת שִׁיל֑וֹ וַהֲלַכְתֶּ֖ם אֶ֥רֶץ בִּנְיָמִֽן׃ (מד) וְהָיָ֡ה כִּֽי־ יָבֹ֣אוּ אֲבוֹתָם֩ א֨וֹ אֲחֵיהֶ֜ם (לרוב) לָרִ֣יב ׀ אֵלֵ֗ינוּ וְאָמַ֤רְנוּ אֲלֵיהֶם֙ חָנּ֣וּנוּ אוֹתָ֔ם כִּ֣י לֹ֥א לָקַ֛חְנוּ אִ֥ישׁ אִשְׁתּ֖וֹ בַּמִּלְחָמָ֑ה כִּ֣י לֹ֥א אַתֶּ֛ם נְתַתֶּ֥ם לָהֶ֖ם כָּעֵ֥ת תֶּאְשָֽׁמוּ׃ (מה) וַיַּֽעֲשׂוּ־ כֵן֙ בְּנֵ֣י בִנְיָמִ֔ן וַיִּשְׂא֤וּ נָשִׁים֙ לְמִסְפָּרָ֔ם מִן־ הַמְּחֹלְל֖וֹת אֲשֶׁ֣ר גָּזָ֑לוּ וַיֵּלְכ֗וּ וַיָּשׁ֙וּבוּ֙ אֶל־ נַ֣חֲלָתָ֔ם וַיִּבְנוּ֙ אֶת־ הֶ֣עָרִ֔ים וַיֵּשְׁב֖וּ בָּהֶֽם׃ (מו) וַיִּתְהַלְּכ֨וּ מִשָּׁ֤ם בְּנֵֽי־ יִשְׂרָאֵל֙ בָּעֵ֣ת הַהִ֔יא אִ֥ישׁ לְשִׁבְט֖וֹ וּלְמִשְׁפַּחְתּ֑וֹ וַיֵּצְא֣וּ מִשָּׁ֔ם אִ֖ישׁ לְנַחֲלָתֽוֹ׃ (מז) בַּיָּמִ֣ים הָהֵ֔ם אֵ֥ין מֶ֖לֶךְ בְּיִשְׂרָאֵ֑ל אִ֛ישׁ הַיָּשָׁ֥ר בְּעֵינָ֖יו יַעֲשֶֽׂה׃
