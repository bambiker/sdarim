שופטים פרק יג לפי החלוקה לסדרים

פרק יג בספר שופטים (25 פסוקים) נמצא בשופטים סדר ט (פסוקים כה–מז) ובשופטים סדר י (פסוקים א–ב).

שופטים סדר ט

(א) וַיַּעֲבֹ֥ר יִפְתָּ֛ח אֶל־ בְּנֵ֥י עַמּ֖וֹן לְהִלָּ֣חֶם בָּ֑ם וַיִתְּנֵ֥ם יְהוָ֖ה בְּיָדֽוֹ׃ (ב) וַיַּכֵּ֡ם מֵעֲרוֹעֵר֩ וְעַד־ בּוֹאֲךָ֨ מִנִּ֜ית עֶשְׂרִ֣ים עִ֗יר וְעַד֙ אָבֵ֣ל כְּרָמִ֔ים מַכָּ֖ה גְּדוֹלָ֣ה מְאֹ֑ד וַיִּכָּֽנְעוּ֙ בְּנֵ֣י עַמּ֔וֹן מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (ג) וַיָּבֹ֨א יִפְתָּ֣ח הַמִּצְפָּה֮ אֶל־ בֵּיתוֹ֒ וְהִנֵּ֤ה בִתּוֹ֙ יֹצֵ֣את לִקְרָאת֔וֹ בְתֻפִּ֖ים וּבִמְחֹל֑וֹת וְרַק֙ הִ֣יא יְחִידָ֔ה אֵֽין־ ל֥וֹ מִמֶּ֛נּוּ בֵּ֖ן אוֹ־ בַֽת׃ (ד) וַיְהִי֩ כִרְאוֹת֨וֹ אוֹתָ֜הּ וַיִּקְרַ֣ע אֶת־ בְּגָדָ֗יו וַיֹּ֙אמֶר֙ אֲהָ֤הּ בִּתִּי֙ הַכְרֵ֣עַ הִכְרַעְתִּ֔נִי וְאַ֖תְּ הָיִ֣יתְ בְּעֹֽכְרָ֑י וְאָנֹכִ֗י פָּצִ֤יתִי־ פִי֙ אֶל־ יְהוָ֔ה וְלֹ֥א אוּכַ֖ל לָשֽׁוּב׃ (ה) וַתֹּ֣אמֶר אֵלָ֗יו אָבִי֙ פָּצִ֤יתָה אֶת־ פִּ֙יךָ֙ אֶל־ יְהוָ֔ה עֲשֵׂ֣ה לִ֔י כַּאֲשֶׁ֖ר יָצָ֣א מִפִּ֑יךָ אַחֲרֵ֡י אֲשֶׁ֣ר עָשָׂה֩ לְךָ֙ יְהוָ֧ה נְקָמ֛וֹת מֵאֹיְבֶ֖יךָ מִבְּנֵ֥י עַמּֽוֹן׃ (ו) וַתֹּ֙אמֶר֙ אֶל־ אָבִ֔יהָ יֵעָ֥שֶׂה לִּ֖י הַדָּבָ֣ר הַזֶּ֑ה הַרְפֵּ֨ה מִמֶּ֜נִּי שְׁנַ֣יִם חֳדָשִׁ֗ים וְאֵֽלְכָה֙ וְיָרַדְתִּ֣י עַל־ הֶֽהָרִ֔ים וְאֶבְכֶּה֙ עַל־ בְּתוּלַ֔י אָנֹכִ֖י (ורעיתי) וְרֵעוֹתָֽי׃ (ז) וַיֹּ֣אמֶר לֵ֔כִי וַיִּשְׁלַ֥ח אוֹתָ֖הּ שְׁנֵ֣י חֳדָשִׁ֑ים וַתֵּ֤לֶךְ הִיא֙ וְרֵ֣עוֹתֶ֔יהָ וַתֵּ֥בְךְּ עַל־ בְּתוּלֶ֖יהָ עַל־ הֶהָרִֽים׃ (ח) וַיְהִ֞י מִקֵּ֣ץ ׀ שְׁנַ֣יִם חֳדָשִׁ֗ים וַתָּ֙שָׁב֙ אֶל־ אָבִ֔יהָ וַיַּ֣עַשׂ לָ֔הּ אֶת־ נִדְר֖וֹ אֲשֶׁ֣ר נָדָ֑ר וְהִיא֙ לֹא־ יָדְעָ֣ה אִ֔ישׁ וַתְּהִי־ חֹ֖ק בְּיִשְׂרָאֵֽל׃ (ט) מִיָּמִ֣ים ׀ יָמִ֗ימָה תֵּלַ֙כְנָה֙ בְּנ֣וֹת יִשְׂרָאֵ֔ל לְתַנּ֕וֹת לְבַת־ יִפְתָּ֖ח הַגִּלְעָדִ֑י אַרְבַּ֥עַת יָמִ֖ים בַּשָּׁנָֽה׃ (י) וַיִּצָּעֵק֙ אִ֣ישׁ אֶפְרַ֔יִם וַֽיַּעֲבֹ֖ר צָפ֑וֹנָה וַיֹּאמְר֨וּ לְיִפְתָּ֜ח מַדּ֣וּעַ ׀ עָבַ֣רְתָּ ׀ לְהִלָּחֵ֣ם בִּבְנֵי־ עַמּ֗וֹן וְלָ֙נוּ֙ לֹ֤א קָרָ֙אתָ֙ לָלֶ֣כֶת עִמָּ֔ךְ בֵּיתְךָ֕ נִשְׂרֹ֥ף עָלֶ֖יךָ בָּאֵֽשׁ׃ (יא) וַיֹּ֤אמֶר יִפְתָּח֙ אֲלֵיהֶ֔ם אִ֣ישׁ רִ֗יב הָיִ֛יתִי אֲנִ֛י וְעַמִּ֥י וּבְנֵֽי־ עַמּ֖וֹן מְאֹ֑ד וָאֶזְעַ֣ק אֶתְכֶ֔ם וְלֹֽא־ הוֹשַׁעְתֶּ֥ם אוֹתִ֖י מִיָּדָֽם׃ (יב) וָֽאֶרְאֶ֞ה כִּֽי־ אֵינְךָ֣ מוֹשִׁ֗יע וָאָשִׂ֨ימָה נַפְשִׁ֤י בְכַפִּי֙ וָֽאֶעְבְּרָה֙ אֶל־ בְּנֵ֣י עַמּ֔וֹן וַיִּתְּנֵ֥ם יְהוָ֖ה בְּיָדִ֑י וְלָמָ֞ה עֲלִיתֶ֥ם אֵלַ֛י הַיּ֥וֹם הַזֶּ֖ה לְהִלָּ֥חֶם בִּֽי׃ (יג) וַיִּקְבֹּ֤ץ יִפְתָּח֙ אֶת־ כָּל־ אַנְשֵׁ֣י גִלְעָ֔ד וַיִּלָּ֖חֶם אֶת־ אֶפְרָ֑יִם וַיַּכּוּ֩ אַנְשֵׁ֨י גִלְעָ֜ד אֶת־ אֶפְרַ֗יִם כִּ֤י אָמְרוּ֙ פְּלִיטֵ֤י אֶפְרַ֙יִם֙ אַתֶּ֔ם גִּלְעָ֕ד בְּת֥וֹךְ אֶפְרַ֖יִם בְּת֥וֹךְ מְנַשֶּֽׁה׃ (יד) וַיִּלְכֹּ֥ד גִּלְעָ֛ד אֶֽת־ מַעְבְּר֥וֹת הַיַּרְדֵּ֖ן לְאֶפְרָ֑יִם וְֽ֠הָיָה כִּ֣י יֹאמְר֞וּ פְּלִיטֵ֤י אֶפְרַ֙יִם֙ אֶעֱבֹ֔רָה וַיֹּ֨אמְרוּ ל֧וֹ אַנְשֵֽׁי־ גִלְעָ֛ד הַֽאֶפְרָתִ֥י אַ֖תָּה וַיֹּ֥אמֶֽר ׀ לֹֽא׃ (טו) וַיֹּ֣אמְרוּ לוֹ֩ אֱמָר־ נָ֨א שִׁבֹּ֜לֶת וַיֹּ֣אמֶר סִבֹּ֗לֶת וְלֹ֤א יָכִין֙ לְדַבֵּ֣ר כֵּ֔ן וַיֹּאחֲז֣וּ אוֹת֔וֹ וַיִּשְׁחָט֖וּהוּ אֶל־ מַעְבְּר֣וֹת הַיַּרְדֵּ֑ן וַיִּפֹּ֞ל בָּעֵ֤ת הַהִיא֙ מֵֽאֶפְרַ֔יִם אַרְבָּעִ֥ים וּשְׁנַ֖יִם אָֽלֶף׃ (טז) וַיִּשְׁפֹּ֥ט יִפְתָּ֛ח אֶת־ יִשְׂרָאֵ֖ל שֵׁ֣שׁ שָׁנִ֑ים וַיָּ֗מָת יִפְתָּח֙ הַגִּלְעָדִ֔י וַיִּקָּבֵ֖ר בְּעָרֵ֥י גִלְעָֽד׃ (יז) וַיִּשְׁפֹּ֤ט אַֽחֲרָיו֙ אֶת־ יִשְׂרָאֵ֔ל אִבְצָ֖ן מִבֵּ֥ית לָֽחֶם׃ (יח) וַיְהִי־ ל֞וֹ שְׁלֹשִׁ֣ים בָּנִ֗ים וּשְׁלֹשִׁ֤ים בָּנוֹת֙ שִׁלַּ֣ח הַח֔וּצָה וּשְׁלֹשִׁ֣ים בָּנ֔וֹת הֵבִ֥יא לְבָנָ֖יו מִן־ הַח֑וּץ וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל שֶׁ֥בַע שָׁנִֽים׃ (יט) וַיָּ֣מָת אִבְצָ֔ן וַיִּקָּבֵ֖ר בְּבֵ֥ית לָֽחֶם׃ (כ) וַיִּשְׁפֹּ֤ט אַֽחֲרָיו֙ אֶת־ יִשְׂרָאֵ֔ל אֵיל֖וֹן הַזְּבֽוּלֹנִ֑י וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל עֶ֥שֶׂר שָׁנִֽים׃ (כא) וַיָּ֖מָת אֵל֣וֹן הַזְּבֽוּלֹנִ֑י וַיִּקָּבֵ֥ר בְּאַיָּל֖וֹן בְּאֶ֥רֶץ זְבוּלֻֽן׃ (כב) וַיִּשְׁפֹּ֥ט אַחֲרָ֖יו אֶת־ יִשְׂרָאֵ֑ל עַבְדּ֥וֹן בֶּן־ הִלֵּ֖ל הַפִּרְעָתוֹנִֽי׃ (כג) וַיְהִי־ ל֞וֹ אַרְבָּעִ֣ים בָּנִ֗ים וּשְׁלֹשִׁים֙ בְּנֵ֣י בָנִ֔ים רֹכְבִ֖ים עַל־ שִׁבְעִ֣ים עֲיָרִ֑ם וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל שְׁמֹנֶ֥ה שָׁנִֽים׃ (כד) וַיָּ֛מָת עַבְדּ֥וֹן בֶּן־ הִלֵּ֖ל הַפִּרְעָתוֹנִ֑י וַיִּקָּבֵ֤ר בְּפִרְעָתוֹן֙ בְּאֶ֣רֶץ אֶפְרַ֔יִם בְּהַ֖ר הָעֲמָלֵקִֽי׃ תחילת פרק יג (כה) וַיֹּסִ֙פוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיִּתְּנֵ֧ם יְהוָ֛ה בְּיַד־ פְּלִשְׁתִּ֖ים אַרְבָּעִ֥ים שָׁנָֽה׃ (כו) וַיְהִי֩ אִ֨ישׁ אֶחָ֧ד מִצָּרְעָ֛ה מִמִּשְׁפַּ֥חַת הַדָּנִ֖י וּשְׁמ֣וֹ מָנ֑וֹחַ וְאִשְׁתּ֥וֹ עֲקָרָ֖ה וְלֹ֥א יָלָֽדָה׃ (כז) וַיֵּרָ֥א מַלְאַךְ־ יְהוָ֖ה אֶל־ הָאִשָּׁ֑ה וַיֹּ֣אמֶר אֵלֶ֗יהָ הִנֵּה־ נָ֤א אַתְּ־ עֲקָרָה֙ וְלֹ֣א יָלַ֔דְתְּ וְהָרִ֖ית וְיָלַ֥דְתְּ בֵּֽן׃ (כח) וְעַתָּה֙ הִשָּׁ֣מְרִי נָ֔א וְאַל־ תִּשְׁתִּ֖י יַ֣יִן וְשֵׁכָ֑ר וְאַל־ תֹּאכְלִ֖י כָּל־ טָמֵֽא׃ (כט) כִּי֩ הִנָּ֨ךְ הָרָ֜ה וְיֹלַ֣דְתְּ בֵּ֗ן וּמוֹרָה֙ לֹא־ יַעֲלֶ֣ה עַל־ רֹאשׁ֔וֹ כִּֽי־ נְזִ֧יר אֱלֹהִ֛ים יִהְיֶ֥ה הַנַּ֖עַר מִן־ הַבָּ֑טֶן וְה֗וּא יָחֵ֛ל לְהוֹשִׁ֥יעַ אֶת־ יִשְׂרָאֵ֖ל מִיַּ֥ד פְּלִשְׁתִּֽים׃ (ל) וַתָּבֹ֣א הָאִשָּׁ֗ה וַתֹּ֣אמֶר לְאִישָׁהּ֮ לֵאמֹר֒ אִ֤ישׁ הָאֱלֹהִים֙ בָּ֣א אֵלַ֔י וּמַרְאֵ֕הוּ כְּמַרְאֵ֛ה מַלְאַ֥ךְ הָאֱלֹהִ֖ים נוֹרָ֣א מְאֹ֑ד וְלֹ֤א שְׁאִלְתִּ֙יהוּ֙ אֵֽי־ מִזֶּ֣ה ה֔וּא וְאֶת־ שְׁמ֖וֹ לֹֽא־ הִגִּ֥יד לִֽי׃ (לא) וַיֹּ֣אמֶר לִ֔י הִנָּ֥ךְ הָרָ֖ה וְיֹלַ֣דְתְּ בֵּ֑ן וְעַתָּ֞ה אַל־ תִּשְׁתִּ֣י ׀ יַ֣יִן וְשֵׁכָ֗ר וְאַל־ תֹּֽאכְלִי֙ כָּל־ טֻמְאָ֔ה כִּֽי־ נְזִ֤יר אֱלֹהִים֙ יִהְיֶ֣ה הַנַּ֔עַר מִן־ הַבֶּ֖טֶן עַד־ י֥וֹם מוֹתֽוֹ׃ (לב) וַיֶּעְתַּ֥ר מָנ֛וֹחַ אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר בִּ֣י אֲדוֹנָ֔י אִ֣ישׁ הָאֱלֹהִ֞ים אֲשֶׁ֣ר שָׁלַ֗חְתָּ יָבוֹא־ נָ֥א עוֹד֙ אֵלֵ֔ינוּ וְיוֹרֵ֕נוּ מַֽה־ נַּעֲשֶׂ֖ה לַנַּ֥עַר הַיּוּלָּֽד׃ (לג) וַיִּשְׁמַ֥ע הָאֱלֹהִ֖ים בְּק֣וֹל מָנ֑וֹחַ וַיָּבֹ֣א מַלְאַךְ֩ הָאֱלֹהִ֨ים ע֜וֹד אֶל־ הָאִשָּׁ֗ה וְהִיא֙ יוֹשֶׁ֣בֶת בַּשָּׂדֶ֔ה וּמָנ֥וֹחַ אִישָׁ֖הּ אֵ֥ין עִמָּֽהּ׃ (לד) וַתְּמַהֵר֙ הָֽאִשָּׁ֔ה וַתָּ֖רָץ וַתַּגֵּ֣ד לְאִישָׁ֑הּ וַתֹּ֣אמֶר אֵלָ֔יו הִנֵּ֨ה נִרְאָ֤ה אֵלַי֙ הָאִ֔ישׁ אֲשֶׁר־ בָּ֥א בַיּ֖וֹם אֵלָֽי׃ (לה) וַיָּ֛קָם וַיֵּ֥לֶךְ מָנ֖וֹחַ אַחֲרֵ֣י אִשְׁתּ֑וֹ וַיָּבֹא֙ אֶל־ הָאִ֔ישׁ וַיֹּ֣אמֶר ל֗וֹ הַאַתָּ֥ה הָאִ֛ישׁ אֲשֶׁר־ דִּבַּ֥רְתָּ אֶל־ הָאִשָּׁ֖ה וַיֹּ֥אמֶר אָֽנִי׃ (לו) לָקַ֣חַת אִשָּׁ֔ה מִפְּלִשְׁתִּ֖ים הָעֲרֵלִ֑ים וַיֹּ֨אמֶר שִׁמְשׁ֤וֹן אֶל־ אָבִיו֙ אוֹתָ֣הּ קַֽח־ לִ֔י כִּֽי־ הִ֖יא יָשְׁרָ֥ה בְעֵינָֽי׃ (ו) וְאָבִ֨יו וְאִמּ֜וֹ לֹ֣א יָדְע֗וּ כִּ֤י מֵיְהוָה֙ הִ֔יא וַיַּ֥ךְ מֵהֶ֣ם ׀ שְׁלֹשִׁ֣ים אִ֗ישׁ וַיִּקַּח֙ אֶת־ חֲלִ֣יצוֹתָ֔ם וַיִּתֵּן֙ הַחֲלִיפ֔וֹת לְמַגִּידֵ֖י הַחִידָ֑ה וַיִּ֣חַר אַפּ֔וֹ וַיַּ֖עַל בֵּ֥ית אָבִֽיהוּ׃ (כב) וַתְּהִ֖י אֵ֣שֶׁת שִׁמְשׁ֑וֹן לְמֵ֣רֵעֵ֔הוּ אֲשֶׁ֥ר רֵעָ֖ה לֽוֹ׃ (כג) וַיְהִ֨י מִיָּמִ֜ים בִּימֵ֣י קְצִיר־ חִטִּ֗ים וַיִּפְקֹ֨ד שִׁמְשׁ֤וֹן אֶת־ אִשְׁתּוֹ֙ בִּגְדִ֣י עִזִּ֔ים וַיֹּ֕אמֶר אָבֹ֥אָה אֶל־ אִשְׁתִּ֖י הֶחָ֑דְרָה וְלֹֽא־ נְתָנ֥וֹ אָבִ֖יהָ לָבֽוֹא׃ (כד) וַיֹּ֣אמֶר אָבִ֗יהָ אָמֹ֤ר אָמַ֙רְתִּי֙ כִּי־ שָׂנֹ֣א שְׂנֵאתָ֔הּ וָאֶתְּנֶ֖נָּה לְמֵרֵעֶ֑ךָ הֲלֹ֨א אֲחֹתָ֤הּ הַקְּטַנָּה֙ טוֹבָ֣ה מִמֶּ֔נָּה תְּהִי־ נָ֥א לְךָ֖ תַּחְתֶּֽיהָ׃ (כה) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן נִקֵּ֥יתִי הַפַּ֖עַם מִפְּלִשְׁתִּ֑ים כִּֽי־ עֹשֶׂ֥ה אֲנִ֛י עִמָּ֖ם רָעָֽה׃ (כו) וַיֵּ֣לֶךְ שִׁמְשׁ֔וֹן וַיִּלְכֹּ֖ד שְׁלֹשׁ־ מֵא֣וֹת שׁוּעָלִ֑ים וַיִּקַּ֣ח לַפִּדִ֗ים וַיֶּ֤פֶן זָנָב֙ אֶל־ זָנָ֔ב וַיָּ֨שֶׂם לַפִּ֥יד אֶחָ֛ד בֵּין־ שְׁנֵ֥י הַזְּנָב֖וֹת בַּתָּֽוֶךְ׃ (כז) וַיַּבְעֶר־ אֵשׁ֙ בַּלַּפִּידִ֔ים וַיְשַׁלַּ֖ח בְּקָמ֣וֹת פְּלִשְׁתִּ֑ים וַיַּבְעֵ֛ר מִגָּדִ֥ישׁ וְעַד־ קָמָ֖ה וְעַד־ כֶּ֥רֶם זָֽיִת׃ (כח) וַיֹּאמְר֣וּ פְלִשְׁתִּים֮ מִ֣י עָ֣שָׂה זֹאת֒ וַיֹּאמְר֗וּ שִׁמְשׁוֹן֙ חֲתַ֣ן הַתִּמְנִ֔י כִּ֚י לָקַ֣ח אֶת־ אִשְׁתּ֔וֹ וַֽיִּתְּנָ֖הּ לְמֵרֵעֵ֑הוּ וַיַּעֲל֣וּ פְלִשְׁתִּ֔ים וַיִּשְׂרְפ֥וּ אוֹתָ֛הּ וְאֶת־ אָבִ֖יהָ בָּאֵֽשׁ׃ (כט) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן אִֽם־ תַּעֲשׂ֖וּן כָּזֹ֑את כִּ֛י אִם־ נִקַּ֥מְתִּי בָכֶ֖ם וְאַחַ֥ר אֶחְדָּֽל׃ (ל) וַיַּ֨ךְ אוֹתָ֥ם שׁ֛וֹק עַל־ יָרֵ֖ךְ מַכָּ֣ה גְדוֹלָ֑ה וַיֵּ֣רֶד וַיֵּ֔שֶׁב בִּסְעִ֖יף סֶ֥לַע עֵיטָֽם׃ (לא) וַיַּעֲל֣וּ פְלִשְׁתִּ֔ים וַֽיַּחֲנ֖וּ בִּיהוּדָ֑ה וַיִּנָּטְשׁ֖וּ בַּלֶּֽחִי׃ (לב) וַיֹּֽאמְרוּ֙ אִ֣ישׁ יְהוּדָ֔ה לָמָ֖ה עֲלִיתֶ֣ם עָלֵ֑ינוּ וַיֹּאמְר֗וּ לֶאֱס֤וֹר אֶת־ שִׁמְשׁוֹן֙ עָלִ֔ינוּ לַעֲשׂ֣וֹת ל֔וֹ כַּאֲשֶׁ֖ר עָ֥שָׂה לָֽנוּ׃ (לג) וַיֵּרְד֡וּ שְׁלֹשֶׁת֩ אֲלָפִ֨ים אִ֜ישׁ מִֽיהוּדָ֗ה אֶל־ סְעִיף֮ סֶ֣לַע עֵיטָם֒ וַיֹּאמְר֣וּ לְשִׁמְשׁ֗וֹן הֲלֹ֤א יָדַ֙עְתָּ֙ כִּֽי־ מֹשְׁלִ֥ים בָּ֙נוּ֙ פְּלִשְׁתִּ֔ים וּמַה־ זֹּ֖את עָשִׂ֣יתָ לָּ֑נוּ וַיֹּ֣אמֶר לָהֶ֔ם כַּאֲשֶׁר֙ עָ֣שׂוּ לִ֔י כֵּ֖ן עָשִׂ֥יתִי לָהֶֽם׃ (לד) וַיֹּ֤אמְרוּ לוֹ֙ לֶאֱסָרְךָ֣ יָרַ֔דְנוּ לְתִתְּךָ֖ בְּיַד־ פְּלִשְׁתִּ֑ים וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן הִשָּׁבְע֣וּ לִ֔י פֶּֽן־ תִּפְגְּע֥וּן בִּ֖י אַתֶּֽם׃ (לה) וַיֹּ֧אמְרוּ ל֣וֹ לֵאמֹ֗ר לֹ֚א כִּֽי־ אָסֹ֤ר נֶֽאֱסָרְךָ֙ וּנְתַנּ֣וּךָ בְיָדָ֔ם וְהָמֵ֖ת לֹ֣א נְמִיתֶ֑ךָ וַיַּאַסְרֻ֗הוּ בִּשְׁנַ֙יִם֙ עֲבֹתִ֣ים חֲדָשִׁ֔ים וַֽיַּעֲל֖וּהוּ מִן־ הַסָּֽלַע׃ (לו) הוּא־ בָ֣א עַד־ לֶ֔חִי וּפִלְשִׁתִּ֖ים הֵרִ֣יעוּ לִקְרָאת֑וֹ וַתִּצְלַ֨ח עָלָ֜יו ר֣וּחַ יְהוָ֗ה וַתִּהְיֶ֨ינָה הָעֲבֹתִ֜ים אֲשֶׁ֣ר עַל־ זְרוֹעוֹתָ֗יו כַּפִּשְׁתִּים֙ אֲשֶׁ֣ר בָּעֲר֣וּ בָאֵ֔שׁ וַיִּמַּ֥סּוּ אֱסוּרָ֖יו מֵעַ֥ל יָדָֽיו׃ (לז) וַיִּמְצָ֥א לְחִֽי־ חֲמ֖וֹר טְרִיָּ֑ה וַיִּשְׁלַ֤ח יָדוֹ֙ וַיִּקָּחֶ֔הָ וַיַּךְ־ בָּ֖הּ אֶ֥לֶף אִֽישׁ׃ (לח) וַיֹּ֣אמֶר שִׁמְשׁ֔וֹן בִּלְחִ֣י הַחֲמ֔וֹר חֲמ֖וֹר חֲמֹרָתָ֑יִם בִּלְחִ֣י הַחֲמ֔וֹר הִכֵּ֖יתִי אֶ֥לֶף אִֽישׁ׃ (לט) וַֽיְהִי֙ כְּכַלֹּת֣וֹ לְדַבֵּ֔ר וַיַּשְׁלֵ֥ךְ הַלְּחִ֖י מִיָּד֑וֹ וַיִּקְרָ֛א לַמָּק֥וֹם הַה֖וּא רָ֥מַת לֶֽחִי׃ (מ) וַיִּצְמָא֮ מְאֹד֒ וַיִּקְרָ֤א אֶל־ יְהוָה֙ וַיֹּאמַ֔ר אַתָּה֙ נָתַ֣תָּ בְיַֽד־ עַבְדְּךָ֔ אֶת־ הַתְּשׁוּעָ֥ה הַגְּדֹלָ֖ה הַזֹּ֑את וְעַתָּה֙ אָמ֣וּת בַּצָּמָ֔א וְנָפַלְתִּ֖י בְּיַ֥ד הָעֲרֵלִֽים׃ (מא) וַיִּבְקַ֨ע אֱלֹהִ֜ים אֶת־ הַמַּכְתֵּ֣שׁ אֲשֶׁר־ בַּלֶּ֗חִי וַיֵּצְא֨וּ מִמֶּ֤נּוּ מַ֙יִם֙ וַיֵּ֔שְׁתְּ וַתָּ֥שָׁב רוּח֖וֹ וַיֶּ֑חִי עַל־ כֵּ֣ן ׀ קָרָ֣א שְׁמָ֗הּ עֵ֤ין הַקּוֹרֵא֙ אֲשֶׁ֣ר בַּלֶּ֔חִי עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (מב) וַיִּשְׁפֹּ֧ט אֶת־ יִשְׂרָאֵ֛ל בִּימֵ֥י פְלִשְׁתִּ֖ים עֶשְׂרִ֥ים שָׁנָֽה׃ (מג) וַיֵּ֥לֶךְ שִׁמְשׁ֖וֹן עַזָּ֑תָה וַיַּרְא־ שָׁם֙ אִשָּׁ֣ה זוֹנָ֔ה וַיָּבֹ֖א אֵלֶֽיהָ׃ (מד) לַֽעַזָּתִ֣ים ׀ לֵאמֹ֗ר בָּ֤א שִׁמְשׁוֹן֙ הֵ֔נָּה וַיָּסֹ֛בּוּ וַיֶּאֶרְבוּ־ ל֥וֹ כָל־ הַלַּ֖יְלָה בְּשַׁ֣עַר הָעִ֑יר וַיִּתְחָרְשׁ֤וּ כָל־ הַלַּ֙יְלָה֙ לֵאמֹ֔ר עַד־ א֥וֹר הַבֹּ֖קֶר וַהֲרְגְנֻֽהוּ׃
