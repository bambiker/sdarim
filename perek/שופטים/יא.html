שופטים פרק יא לפי החלוקה לסדרים

פרק יא בספר שופטים (40 פסוקים) נמצא בשופטים סדר ח (פסוקים יט–מט) ובשופטים סדר ט (פסוקים א–ט).

שופטים סדר ח

(א) וַיָּקָם֩ אַחֲרֵ֨י אֲבִימֶ֜לֶךְ לְהוֹשִׁ֣יעַ אֶת־ יִשְׂרָאֵ֗ל תּוֹלָ֧ע בֶּן־ פּוּאָ֛ה בֶּן־ דּוֹד֖וֹ אִ֣ישׁ יִשָּׂשכָ֑ר וְהֽוּא־ יֹשֵׁ֥ב בְּשָׁמִ֖יר בְּהַ֥ר אֶפְרָֽיִם׃ (ב) וַיִּשְׁפֹּט֙ אֶת־ יִשְׂרָאֵ֔ל עֶשְׂרִ֥ים וְשָׁלֹ֖שׁ שָׁנָ֑ה וַיָּ֖מָת וַיִּקָּבֵ֥ר בְּשָׁמִֽיר׃ (ג) וַיָּ֣קָם אַחֲרָ֔יו יָאִ֖יר הַגִּלְעָדִ֑י וַיִּשְׁפֹּט֙ אֶת־ יִשְׂרָאֵ֔ל עֶשְׂרִ֥ים וּשְׁתַּ֖יִם שָׁנָֽה׃ (ד) וַֽיְהִי־ ל֞וֹ שְׁלֹשִׁ֣ים בָּנִ֗ים רֹֽכְבִים֙ עַל־ שְׁלֹשִׁ֣ים עֲיָרִ֔ים וּשְׁלֹשִׁ֥ים עֲיָרִ֖ים לָהֶ֑ם לָהֶ֞ם יִקְרְא֣וּ ׀ חַוֺּ֣ת יָאִ֗יר עַ֚ד הַיּ֣וֹם הַזֶּ֔ה אֲשֶׁ֖ר בְּאֶ֥רֶץ הַגִּלְעָֽד׃ (ה) וַיָּ֣מָת יָאִ֔יר וַיִּקָּבֵ֖ר בְּקָמֽוֹן׃ (ו) וַיֹּסִ֣פוּ ׀ בְּנֵ֣י יִשְׂרָאֵ֗ל לַעֲשׂ֣וֹת הָרַע֮ בְּעֵינֵ֣י יְהוָה֒ וַיַּעַבְד֣וּ אֶת־ הַבְּעָלִ֣ים וְאֶת־ הָעַשְׁתָּר֡וֹת וְאֶת־ אֱלֹהֵ֣י אֲרָם֩ וְאֶת־ אֱלֹהֵ֨י צִיד֜וֹן וְאֵ֣ת ׀ אֱלֹהֵ֣י מוֹאָ֗ב וְאֵת֙ אֱלֹהֵ֣י בְנֵי־ עַמּ֔וֹן וְאֵ֖ת אֱלֹהֵ֣י פְלִשְׁתִּ֑ים וַיַּעַזְב֥וּ אֶת־ יְהוָ֖ה וְלֹ֥א עֲבָדֽוּהוּ׃ (ז) וַיִּֽחַר־ אַ֥ף יְהוָ֖ה בְּיִשְׂרָאֵ֑ל וַֽיִּמְכְּרֵם֙ בְּיַד־ פְּלִשְׁתִּ֔ים וּבְיַ֖ד בְּנֵ֥י עַמּֽוֹן׃ (ח) וַֽיִּרְעֲצ֤וּ וַיְרֹֽצְצוּ֙ אֶת־ בְּנֵ֣י יִשְׂרָאֵ֔ל בַּשָּׁנָ֖ה הַהִ֑יא שְׁמֹנֶ֨ה עֶשְׂרֵ֜ה שָׁנָ֗ה אֶֽת־ כָּל־ בְּנֵ֤י יִשְׂרָאֵל֙ אֲשֶׁר֙ בְּעֵ֣בֶר הַיַּרְדֵּ֔ן בְּאֶ֥רֶץ הָאֱמֹרִ֖י אֲשֶׁ֥ר בַּגִּלְעָֽד׃ (ט) וַיַּעַבְר֤וּ בְנֵֽי־ עַמּוֹן֙ אֶת־ הַיַּרְדֵּ֔ן לְהִלָּחֵ֛ם גַּם־ בִּיהוּדָ֥ה וּבְבִנְיָמִ֖ין וּבְבֵ֣ית אֶפְרָ֑יִם וַתֵּ֥צֶר לְיִשְׂרָאֵ֖ל מְאֹֽד׃ (י) וַֽיִּזְעֲקוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶל־ יְהוָ֖ה לֵאמֹ֑ר חָטָ֣אנוּ לָ֔ךְ וְכִ֤י עָזַ֙בְנוּ֙ אֶת־ אֱלֹהֵ֔ינוּ וַֽנַּעֲבֹ֖ד אֶת־ הַבְּעָלִֽים׃ (יא) וַ֥יֹּאמֶר יְהוָ֖ה אֶל־ בְּנֵ֣י יִשְׂרָאֵ֑ל הֲלֹ֤א מִמִּצְרַ֙יִם֙ וּמִן־ הָ֣אֱמֹרִ֔י וּמִן־ בְּנֵ֥י עַמּ֖וֹן וּמִן־ פְּלִשְׁתִּֽים׃ (יב) וְצִידוֹנִ֤ים וַֽעֲמָלֵק֙ וּמָע֔וֹן לָחֲצ֖וּ אֶתְכֶ֑ם וַתִּצְעֲק֣וּ אֵלַ֔י וָאוֹשִׁ֥יעָה אֶתְכֶ֖ם מִיָּדָֽם׃ (יג) וְאַתֶּם֙ עֲזַבְתֶּ֣ם אוֹתִ֔י וַתַּעַבְד֖וּ אֱלֹהִ֣ים אֲחֵרִ֑ים לָכֵ֥ן לֹֽא־ אוֹסִ֖יף לְהוֹשִׁ֥יעַ אֶתְכֶֽם׃ (יד) לְכ֗וּ וְזַֽעֲקוּ֙ אֶל־ הָ֣אֱלֹהִ֔ים אֲשֶׁ֥ר בְּחַרְתֶּ֖ם בָּ֑ם הֵ֛מָּה יוֹשִׁ֥יעוּ לָכֶ֖ם בְּעֵ֥ת צָרַתְכֶֽם׃ (טו) וַיֹּאמְר֨וּ בְנֵי־ יִשְׂרָאֵ֤ל אֶל־ יְהוָה֙ חָטָ֔אנוּ עֲשֵׂה־ אַתָּ֣ה לָ֔נוּ כְּכָל־ הַטּ֖וֹב בְּעֵינֶ֑יךָ אַ֛ךְ הַצִּילֵ֥נוּ נָ֖א הַיּ֥וֹם הַזֶּֽה׃ (טז) וַיָּסִ֜ירוּ אֶת־ אֱלֹהֵ֤י הַנֵּכָר֙ מִקִּרְבָּ֔ם וַיַּעַבְד֖וּ אֶת־ יְהוָ֑ה וַתִּקְצַ֥ר נַפְשׁ֖וֹ בַּעֲמַ֥ל יִשְׂרָאֵֽל׃ (יז) וַיִּצָּֽעֲקוּ֙ בְּנֵ֣י עַמּ֔וֹן וַֽיַּחֲנ֖וּ בַּגִּלְעָ֑ד וַיֵּאָֽסְפוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל וַֽיַּחֲנ֖וּ בַּמִּצְפָּֽה׃ (יח) וַיֹּאמְר֨וּ הָעָ֜ם שָׂרֵ֤י גִלְעָד֙ אִ֣ישׁ אֶל־ רֵעֵ֔הוּ מִ֣י הָאִ֔ישׁ אֲשֶׁ֣ר יָחֵ֔ל לְהִלָּחֵ֖ם בִּבְנֵ֣י עַמּ֑וֹן יִֽהְיֶ֣ה לְרֹ֔אשׁ לְכֹ֖ל יֹשְׁבֵ֥י גִלְעָֽד׃ תחילת פרק יא (יט) וְיִפְתָּ֣ח הַגִּלְעָדִ֗י הָיָה֙ גִּבּ֣וֹר חַ֔יִל וְה֖וּא בֶּן־ אִשָּׁ֣ה זוֹנָ֑ה וַיּ֥וֹלֶד גִּלְעָ֖ד אֶת־ יִפְתָּֽח׃ (כ) וַתֵּ֧לֶד אֵֽשֶׁת־ גִּלְעָ֛ד ל֖וֹ בָּנִ֑ים וַיִּגְדְּל֨וּ בְֽנֵי־ הָאִשָּׁ֜ה וַיְגָרְשׁ֣וּ אֶת־ יִפְתָּ֗ח וַיֹּ֤אמְרוּ לוֹ֙ לֹֽא־ תִנְחַ֣ל בְּבֵית־ אָבִ֔ינוּ כִּ֛י בֶּן־ אִשָּׁ֥ה אַחֶ֖רֶת אָֽתָּה׃ (כא) וַיִּבְרַ֤ח יִפְתָּח֙ מִפְּנֵ֣י אֶחָ֔יו וַיֵּ֖שֶׁב בְּאֶ֣רֶץ ט֑וֹב וַיִּֽתְלַקְּט֤וּ אֶל־ יִפְתָּח֙ אֲנָשִׁ֣ים רֵיקִ֔ים וַיֵּצְא֖וּ עִמּֽוֹ׃ (כב) וַיְהִ֖י מִיָּמִ֑ים וַיִּלָּחֲמ֥וּ בְנֵֽי־ עַמּ֖וֹן עִם־ יִשְׂרָאֵֽל׃ (כג) וַיְהִ֕י כַּאֲשֶׁר־ נִלְחֲמ֥וּ בְנֵֽי־ עַמּ֖וֹן עִם־ יִשְׂרָאֵ֑ל וַיֵּֽלְכוּ֙ זִקְנֵ֣י גִלְעָ֔ד לָקַ֥חַת אֶת־ יִפְתָּ֖ח מֵאֶ֥רֶץ טֽוֹב׃ (כד) וַיֹּאמְר֣וּ לְיִפְתָּ֔ח לְכָ֕ה וְהָיִ֥יתָה לָּ֖נוּ לְקָצִ֑ין וְנִֽלָּחֲמָ֖ה בִּבְנֵ֥י עַמּֽוֹן׃ (כה) וַיֹּ֤אמֶר יִפְתָּח֙ לְזִקְנֵ֣י גִלְעָ֔ד הֲלֹ֤א אַתֶּם֙ שְׂנֵאתֶ֣ם אוֹתִ֔י וַתְּגָרְשׁ֖וּנִי מִבֵּ֣ית אָבִ֑י וּמַדּ֜וּעַ בָּאתֶ֤ם אֵלַי֙ עַ֔תָּה כַּאֲשֶׁ֖ר צַ֥ר לָכֶֽם׃ (כו) וַיֹּאמְרוּ֩ זִקְנֵ֨י גִלְעָ֜ד אֶל־ יִפְתָּ֗ח לָכֵן֙ עַתָּה֙ שַׁ֣בְנוּ אֵלֶ֔יךָ וְהָלַכְתָּ֣ עִמָּ֔נוּ וְנִלְחַמְתָּ֖ בִּבְנֵ֣י עַמּ֑וֹן וְהָיִ֤יתָ לָּ֙נוּ֙ לְרֹ֔אשׁ לְכֹ֖ל יֹשְׁבֵ֥י גִלְעָֽד׃ (כז) וַיֹּ֨אמֶר יִפְתָּ֜ח אֶל־ זִקְנֵ֣י גִלְעָ֗ד אִם־ מְשִׁיבִ֨ים אַתֶּ֤ם אוֹתִי֙ לְהִלָּחֵם֙ בִּבְנֵ֣י עַמּ֔וֹן וְנָתַ֧ן יְהוָ֛ה אוֹתָ֖ם לְפָנָ֑י אָנֹכִ֕י אֶהְיֶ֥ה לָכֶ֖ם לְרֹֽאשׁ׃ (כח) וַיֹּאמְר֥וּ זִקְנֵֽי־ גִלְעָ֖ד אֶל־ יִפְתָּ֑ח יְהוָ֗ה יִהְיֶ֤ה שֹׁמֵ֙עַ֙ בֵּֽינוֹתֵ֔ינוּ אִם־ לֹ֥א כִדְבָרְךָ֖ כֵּ֥ן נַעֲשֶֽׂה׃ (כט) וַיֵּ֤לֶךְ יִפְתָּח֙ עִם־ זִקְנֵ֣י גִלְעָ֔ד וַיָּשִׂ֨ימוּ הָעָ֥ם אוֹת֛וֹ עֲלֵיהֶ֖ם לְרֹ֣אשׁ וּלְקָצִ֑ין וַיְדַבֵּ֨ר יִפְתָּ֧ח אֶת־ כָּל־ דְּבָרָ֛יו לִפְנֵ֥י יְהוָ֖ה בַּמִּצְפָּֽה׃ (ל) וַיִּשְׁלַ֤ח יִפְתָּח֙ מַלְאָכִ֔ים אֶל־ מֶ֥לֶךְ בְּנֵֽי־ עַמּ֖וֹן לֵאמֹ֑ר מַה־ לִּ֣י וָלָ֔ךְ כִּֽי־ בָ֥אתָ אֵלַ֖י לְהִלָּחֵ֥ם בְּאַרְצִֽי׃ (לא) וַיֹּאמֶר֩ מֶ֨לֶךְ בְּנֵי־ עַמּ֜וֹן אֶל־ מַלְאֲכֵ֣י יִפְתָּ֗ח כִּֽי־ לָקַ֨ח יִשְׂרָאֵ֤ל אֶת־ אַרְצִי֙ בַּעֲלוֹת֣וֹ מִמִּצְרַ֔יִם מֵאַרְנ֥וֹן וְעַד־ הַיַּבֹּ֖ק וְעַד־ הַיַּרְדֵּ֑ן וְעַתָּ֕ה הָשִׁ֥יבָה אֶתְהֶ֖ן בְּשָׁלֽוֹם׃ (לב) וַיּ֥וֹסֶף ע֖וֹד יִפְתָּ֑ח וַיִּשְׁלַח֙ מַלְאָכִ֔ים אֶל־ מֶ֖לֶךְ בְּנֵ֥י עַמּֽוֹן׃ (לג) וַיֹּ֣אמֶר ל֔וֹ כֹּ֖ה אָמַ֣ר יִפְתָּ֑ח לֹֽא־ לָקַ֤ח יִשְׂרָאֵל֙ אֶת־ אֶ֣רֶץ מוֹאָ֔ב וְאֶת־ אֶ֖רֶץ בְּנֵ֥י עַמּֽוֹן׃ (לד) כִּ֖י בַּעֲלוֹתָ֣ם מִמִּצְרָ֑יִם וַיֵּ֨לֶךְ יִשְׂרָאֵ֤ל בַּמִּדְבָּר֙ עַד־ יַם־ ס֔וּף וַיָּבֹ֖א קָדֵֽשָׁה׃ (לה) וַיִּשְׁלַ֣ח יִשְׂרָאֵ֣ל מַלְאָכִ֣ים ׀ אֶל־ מֶלֶךְ֩ אֱד֨וֹם ׀ לֵאמֹ֜ר אֶעְבְּרָה־ נָּ֣א בְאַרְצֶ֗ךָ וְלֹ֤א שָׁמַע֙ מֶ֣לֶךְ אֱד֔וֹם וְגַ֨ם אֶל־ מֶ֧לֶךְ מוֹאָ֛ב שָׁלַ֖ח וְלֹ֣א אָבָ֑ה וַיֵּ֥שֶׁב יִשְׂרָאֵ֖ל בְּקָדֵֽשׁ׃ (לו) וַיֵּ֣לֶךְ בַּמִּדְבָּ֗ר וַיָּ֜סָב אֶת־ אֶ֤רֶץ אֱדוֹם֙ וְאֶת־ אֶ֣רֶץ מוֹאָ֔ב וַיָּבֹ֤א מִמִּזְרַח־ שֶׁ֙מֶשׁ֙ לְאֶ֣רֶץ מוֹאָ֔ב וַֽיַּחֲנ֖וּן בְּעֵ֣בֶר אַרְנ֑וֹן וְלֹא־ בָ֙אוּ֙ בִּגְב֣וּל מוֹאָ֔ב כִּ֥י אַרְנ֖וֹן גְּב֥וּל מוֹאָֽב׃ (לז) וַיִּשְׁלַ֤ח יִשְׂרָאֵל֙ מַלְאָכִ֔ים אֶל־ סִיח֥וֹן גְּדוֹלָ֣ה מְאֹ֑ד וַיִּכָּֽנְעוּ֙ בְּנֵ֣י עַמּ֔וֹן מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (ג) וַיָּבֹ֨א יִפְתָּ֣ח הַמִּצְפָּה֮ אֶל־ בֵּיתוֹ֒ וְהִנֵּ֤ה בִתּוֹ֙ יֹצֵ֣את לִקְרָאת֔וֹ בְתֻפִּ֖ים וּבִמְחֹל֑וֹת וַיִּשְׁפֹּ֤ט אַֽחֲרָיו֙ אֶת־ יִשְׂרָאֵ֔ל אִבְצָ֖ן מִבֵּ֥ית לָֽחֶם׃ (יח) וַיְהִי־ ל֞וֹ שְׁלֹשִׁ֣ים בָּנִ֗ים וּשְׁלֹשִׁ֤ים בָּנוֹת֙ שִׁלַּ֣ח הַח֔וּצָה וּשְׁלֹשִׁ֣ים בָּנ֔וֹת הֵבִ֥יא לְבָנָ֖יו מִן־ הַח֑וּץ וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל שֶׁ֥בַע שָׁנִֽים׃ (יט) וַיָּ֣מָת אִבְצָ֔ן וַיִּקָּבֵ֖ר בְּבֵ֥ית לָֽחֶם׃ (כ) וַיִּשְׁפֹּ֤ט אַֽחֲרָיו֙ אֶת־ יִשְׂרָאֵ֔ל אֵיל֖וֹן הַזְּבֽוּלֹנִ֑י וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל עֶ֥שֶׂר שָׁנִֽים׃ (כא) וַיָּ֖מָת אֵל֣וֹן הַזְּבֽוּלֹנִ֑י וַיִּקָּבֵ֥ר בְּאַיָּל֖וֹן בְּאֶ֥רֶץ זְבוּלֻֽן׃ (כב) וַיִּשְׁפֹּ֥ט אַחֲרָ֖יו אֶת־ יִשְׂרָאֵ֑ל עַבְדּ֥וֹן בֶּן־ הִלֵּ֖ל הַפִּרְעָתוֹנִֽי׃ (כג) וַיְהִי־ ל֞וֹ אַרְבָּעִ֣ים בָּנִ֗ים וּשְׁלֹשִׁים֙ בְּנֵ֣י בָנִ֔ים רֹכְבִ֖ים עַל־ שִׁבְעִ֣ים עֲיָרִ֑ם וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל שְׁמֹנֶ֥ה שָׁנִֽים׃ (כד) וַיָּ֛מָת עַבְדּ֥וֹן בֶּן־ הִלֵּ֖ל הַפִּרְעָתוֹנִ֑י וַיִּקָּבֵ֤ר בְּפִרְעָתוֹן֙ בְּאֶ֣רֶץ אֶפְרַ֔יִם בְּהַ֖ר הָעֲמָלֵקִֽי׃ (כה) וַיֹּסִ֙פוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיִּתְּנֵ֧ם יְהוָ֛ה בְּיַד־ פְּלִשְׁתִּ֖ים אַרְבָּעִ֥ים שָׁנָֽה׃ (כו) וַיְהִי֩ אִ֨ישׁ אֶחָ֧ד מִצָּרְעָ֛ה מִמִּשְׁפַּ֥חַת הַדָּנִ֖י וּשְׁמ֣וֹ מָנ֑וֹחַ וְאִשְׁתּ֥וֹ עֲקָרָ֖ה וְלֹ֥א יָלָֽדָה׃ (כז) וַיֵּרָ֥א מַלְאַךְ־ יְהוָ֖ה אֶל־ הָאִשָּׁ֑ה וַיֹּ֣אמֶר אֵלֶ֗יהָ הִנֵּה־ נָ֤א אַתְּ־ עֲקָרָה֙ וְלֹ֣א יָלַ֔דְתְּ וְהָרִ֖ית וְיָלַ֥דְתְּ בֵּֽן׃ (כח) וְעַתָּה֙ הִשָּׁ֣מְרִי נָ֔א וְאַל־ תִּשְׁתִּ֖י יַ֣יִן וְשֵׁכָ֑ר וְאַל־ תֹּאכְלִ֖י כָּל־ טָמֵֽא׃ (כט) כִּי֩ הִנָּ֨ךְ הָרָ֜ה וְיֹלַ֣דְתְּ בֵּ֗ן וּמוֹרָה֙ לֹא־ יַעֲלֶ֣ה עַל־ רֹאשׁ֔וֹ כִּֽי־ נְזִ֧יר אֱלֹהִ֛ים יִהְיֶ֥ה הַנַּ֖עַר מִן־ הַבָּ֑טֶן וְה֗וּא יָחֵ֛ל לְהוֹשִׁ֥יעַ אֶת־ יִשְׂרָאֵ֖ל מִיַּ֥ד פְּלִשְׁתִּֽים׃ (ל) וַתָּבֹ֣א הָאִשָּׁ֗ה וַתֹּ֣אמֶר לְאִישָׁהּ֮ לֵאמֹר֒ אִ֤ישׁ הָאֱלֹהִים֙ בָּ֣א אֵלַ֔י וּמַרְאֵ֕הוּ כְּמַרְאֵ֛ה מַלְאַ֥ךְ הָאֱלֹהִ֖ים נוֹרָ֣א מְאֹ֑ד וְלֹ֤א שְׁאִלְתִּ֙יהוּ֙ אֵֽי־ מִזֶּ֣ה ה֔וּא וְאֶת־ שְׁמ֖וֹ לֹֽא־ הִגִּ֥יד לִֽי׃ (לא) וַיֹּ֣אמֶר לִ֔י הִנָּ֥ךְ הָרָ֖ה וְיֹלַ֣דְתְּ בֵּ֑ן וְעַתָּ֞ה אַל־ תִּשְׁתִּ֣י ׀ יַ֣יִן וְשֵׁכָ֗ר וְאַל־ תֹּֽאכְלִי֙ כָּל־ טֻמְאָ֔ה כִּֽי־ נְזִ֤יר אֱלֹהִים֙ יִהְיֶ֣ה הַנַּ֔עַר מִן־ הַבֶּ֖טֶן עַד־ י֥וֹם מוֹתֽוֹ׃ (לב) וַיֶּעְתַּ֥ר מָנ֛וֹחַ אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר בִּ֣י אֲדוֹנָ֔י אִ֣ישׁ הָאֱלֹהִ֞ים אֲשֶׁ֣ר שָׁלַ֗חְתָּ יָבוֹא־ נָ֥א עוֹד֙ אֵלֵ֔ינוּ וְיוֹרֵ֕נוּ מַֽה־ נַּעֲשֶׂ֖ה לַנַּ֥עַר הַיּוּלָּֽד׃ (לג) וַיִּשְׁמַ֥ע הָאֱלֹהִ֖ים בְּק֣וֹל מָנ֑וֹחַ וַיָּבֹ֣א מַלְאַךְ֩ הָאֱלֹהִ֨ים ע֜וֹד אֶל־ הָאִשָּׁ֗ה וְהִיא֙ יוֹשֶׁ֣בֶת בַּשָּׂדֶ֔ה וּמָנ֥וֹחַ אִישָׁ֖הּ אֵ֥ין עִמָּֽהּ׃ (לד) וַתְּמַהֵר֙ הָֽאִשָּׁ֔ה וַתָּ֖רָץ וַתַּגֵּ֣ד לְאִישָׁ֑הּ וַתֹּ֣אמֶר אֵלָ֔יו הִנֵּ֨ה נִרְאָ֤ה אֵלַי֙ הָאִ֔ישׁ אֲשֶׁר־ בָּ֥א בַיּ֖וֹם אֵלָֽי׃ (לה) וַיָּ֛קָם וַיֵּ֥לֶךְ מָנ֖וֹחַ אַחֲרֵ֣י אִשְׁתּ֑וֹ וַיָּבֹא֙ אֶל־ הָאִ֔ישׁ וַיֹּ֣אמֶר ל֗וֹ הַאַתָּ֥ה הָאִ֛ישׁ אֲשֶׁר־ דִּבַּ֥רְתָּ אֶל־ הָאִשָּׁ֖ה וַיֹּ֥אמֶר אָֽנִי׃ (לו) וַיֹּ֣אמֶר מָנ֔וֹחַ עַתָּ֖ה יָבֹ֣א דְבָרֶ֑יךָ מַה־ יִּֽהְיֶ֥ה מִשְׁפַּט־ הַנַּ֖עַר וּמַעֲשֵֽׂהוּ׃ (לז) וַיֹּ֛אמֶר מַלְאַ֥ךְ יְהוָ֖ה אֶל־ מָנ֑וֹחַ מִכֹּ֛ל אֲשֶׁר־ אָמַ֥רְתִּי אֶל־ הָאִשָּׁ֖ה תִּשָּׁמֵֽר׃ (לח) מִכֹּ֣ל אֲשֶׁר־ יֵצֵא֩ מִגֶּ֨פֶן הַיַּ֜יִן לֹ֣א תֹאכַ֗ל וְיַ֤יִן וְשֵׁכָר֙ אַל־ תֵּ֔שְׁתְּ וְכָל־ טֻמְאָ֖ה אַל־ תֹּאכַ֑ל כֹּ֥ל אֲשֶׁר־ צִוִּיתִ֖יהָ תִּשְׁמֹֽר׃ (לט) וַיֹּ֥אמֶר מָנ֖וֹחַ אֶל־ מַלְאַ֣ךְ יְהוָ֑ה נַעְצְרָה־ נָּ֣א אוֹתָ֔ךְ וְנַעֲשֶׂ֥ה לְפָנֶ֖יךָ גְּדִ֥י עִזִּֽים׃ (מ) וַיֹּאמֶר֩ מַלְאַ֨ךְ יְהוָ֜ה אֶל־ מָנ֗וֹחַ אִם־ תַּעְצְרֵ֙נִי֙ לֹא־ אֹכַ֣ל בְּלַחְמֶ֔ךָ וְאִם־ תַּעֲשֶׂ֣ה עֹלָ֔ה לַיהוָ֖ה תַּעֲלֶ֑נָּה כִּ֚י לֹא־ יָדַ֣ע מָנ֔וֹחַ כִּֽי־ מַלְאַ֥ךְ יְהוָ֖ה הֽוּא׃ (מא) וַיֹּ֧אמֶר מָנ֛וֹחַ אֶל־ מַלְאַ֥ךְ יְהוָ֖ה מִ֣י שְׁמֶ֑ךָ כִּֽי־ יָבֹ֥א (דבריך) דְבָרְךָ֖ וְכִבַּדְנֽוּךָ׃ (מב) וַיֹּ֤אמֶר לּוֹ֙ מַלְאַ֣ךְ יְהוָ֔ה לָ֥מָּה זֶּ֖ה תִּשְׁאַ֣ל לִשְׁמִ֑י וְהוּא־ פֶ֛לִאי׃ (מג) וַיִּקַּ֨ח מָנ֜וֹחַ אֶת־ גְּדִ֤י הָעִזִּים֙ וְאֶת־ הַמִּנְחָ֔ה וַיַּ֥עַל עַל־ הַצּ֖וּר לַֽיהוָ֑ה וּמַפְלִ֣א לַעֲשׂ֔וֹת וּמָנ֥וֹחַ וְאִשְׁתּ֖וֹ רֹאִֽים׃ (מד) וַיְהִי֩ בַעֲל֨וֹת הַלַּ֜הַב מֵעַ֤ל הַמִּזְבֵּ֙חַ֙ הַשָּׁמַ֔יְמָה וַיַּ֥עַל מַלְאַךְ־ יְהוָ֖ה בְּלַ֣הַב הַמִּזְבֵּ֑חַ וּמָנ֤וֹחַ וְאִשְׁתּוֹ֙ רֹאִ֔ים וַיִּפְּל֥וּ עַל־ פְּנֵיהֶ֖ם אָֽרְצָה׃ (מה) וְלֹא־ יָ֤סַף עוֹד֙ מַלְאַ֣ךְ יְהוָ֔ה לְהֵרָאֹ֖ה אֶל־ מָנ֣וֹחַ וְאֶל־ אִשְׁתּ֑וֹ אָ֚ז יָדַ֣ע מָנ֔וֹחַ כִּֽי־ מַלְאַ֥ךְ יְהוָ֖ה הֽוּא׃ (מו) וַיֹּ֧אמֶר מָנ֛וֹחַ אֶל־ אִשְׁתּ֖וֹ מ֣וֹת נָמ֑וּת כִּ֥י אֱלֹהִ֖ים רָאִֽינוּ׃ (מז) וַתֹּ֧אמֶר ל֣וֹ אִשְׁתּ֗וֹ לוּ֩ חָפֵ֨ץ יְהוָ֤ה לַהֲמִיתֵ֙נוּ֙ לֹֽא־ לָקַ֤ח מִיָּדֵ֙נוּ֙ עֹלָ֣ה וּמִנְחָ֔ה וְלֹ֥א הֶרְאָ֖נוּ אֶת־ כָּל־ אֵ֑לֶּה וְכָעֵ֕ת לֹ֥א הִשְׁמִיעָ֖נוּ כָּזֹֽאת׃
