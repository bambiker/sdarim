שופטים פרק ה לפי החלוקה לסדרים

פרק ה בספר שופטים (31 פסוקים) נמצא בשופטים סדר ג (פסוקים כו–נה) ובשופטים סדר ד (פסוקים א–א).

שופטים סדר ג

(א) וְאַחֲרָ֤יו הָיָה֙ שַׁמְגַּ֣ר בֶּן־ עֲנָ֔ת וַיַּ֤ךְ אֶת־ פְּלִשְׁתִּים֙ שֵֽׁשׁ־ מֵא֣וֹת אִ֔ישׁ בְּמַלְמַ֖ד הַבָּקָ֑ר וַיֹּ֥שַׁע גַּם־ ה֖וּא אֶת־ יִשְׂרָאֵֽל׃ (ב) וַיֹּסִ֙פוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וְאֵה֖וּד מֵֽת׃ (ג) וַיִּמְכְּרֵ֣ם יְהוָ֗ה בְּיַד֙ יָבִ֣ין מֶֽלֶךְ־ כְּנַ֔עַן אֲשֶׁ֥ר מָלַ֖ךְ בְּחָצ֑וֹר וְשַׂר־ צְבָאוֹ֙ סִֽיסְרָ֔א וְה֥וּא יוֹשֵׁ֖ב בַּחֲרֹ֥שֶׁת הַגּוֹיִֽם׃ (ד) וַיִּצְעֲק֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל אֶל־ יְהוָ֑ה כִּ֠י תְּשַׁ֨ע מֵא֤וֹת רֶֽכֶב־ בַּרְזֶל֙ ל֔וֹ וְ֠הוּא לָחַ֞ץ אֶת־ בְּנֵ֧י יִשְׂרָאֵ֛ל בְּחָזְקָ֖ה עֶשְׂרִ֥ים שָׁנָֽה׃ (ה) וּדְבוֹרָה֙ אִשָּׁ֣ה נְבִיאָ֔ה אֵ֖שֶׁת לַפִּיד֑וֹת הִ֛יא שֹׁפְטָ֥ה אֶת־ יִשְׂרָאֵ֖ל בָּעֵ֥ת הַהִֽיא׃ (ו) וְ֠הִיא יוֹשֶׁ֨בֶת תַּֽחַת־ תֹּ֜מֶר דְּבוֹרָ֗ה בֵּ֧ין הָרָמָ֛ה וּבֵ֥ין בֵּֽית־ אֵ֖ל בְּהַ֣ר אֶפְרָ֑יִם וַיַּעֲל֥וּ אֵלֶ֛יהָ בְּנֵ֥י יִשְׂרָאֵ֖ל לַמִּשְׁפָּֽט׃ (ז) וַתִּשְׁלַ֗ח וַתִּקְרָא֙ לְבָרָ֣ק בֶּן־ אֲבִינֹ֔עַם מִקֶּ֖דֶשׁ נַפְתָּלִ֑י וַתֹּ֨אמֶר אֵלָ֜יו הֲלֹ֥א צִוָּ֣ה ׀ יְהוָ֣ה אֱלֹהֵֽי־ יִשְׂרָאֵ֗ל לֵ֤ךְ וּמָֽשַׁכְתָּ֙ בְּהַ֣ר תָּב֔וֹר וְלָקַחְתָּ֣ עִמְּךָ֗ עֲשֶׂ֤רֶת אֲלָפִים֙ אִ֔ישׁ מִבְּנֵ֥י נַפְתָּלִ֖י וּמִבְּנֵ֥י זְבֻלֽוּן׃ (ח) וּמָשַׁכְתִּ֨י אֵלֶ֜יךָ אֶל־ נַ֣חַל קִישׁ֗וֹן אֶת־ סִֽיסְרָא֙ שַׂר־ צְבָ֣א יָבִ֔ין וְאֶת־ רִכְבּ֖וֹ וְאֶת־ הֲמוֹנ֑וֹ וּנְתַתִּ֖יהוּ בְּיָדֶֽךָ׃ (ט) וַיֹּ֤אמֶר אֵלֶ֙יהָ֙ בָּרָ֔ק אִם־ תֵּלְכִ֥י עִמִּ֖י וְהָלָ֑כְתִּי וְאִם־ לֹ֥א תֵלְכִ֛י עִמִּ֖י לֹ֥א אֵלֵֽךְ׃ (י) וַתֹּ֜אמֶר הָלֹ֧ךְ אֵלֵ֣ךְ עִמָּ֗ךְ אֶ֚פֶס כִּי֩ לֹ֨א תִֽהְיֶ֜ה תִּֽפְאַרְתְּךָ֗ עַל־ הַדֶּ֙רֶךְ֙ אֲשֶׁ֣ר אַתָּ֣ה הוֹלֵ֔ךְ כִּ֣י בְֽיַד־ אִשָּׁ֔ה יִמְכֹּ֥ר יְהוָ֖ה אֶת־ סִֽיסְרָ֑א וַתָּ֧קָם דְּבוֹרָ֛ה וַתֵּ֥לֶך עִם־ בָּרָ֖ק קֶֽדְשָׁה׃ (יא) וַיַּזְעֵ֨ק בָּרָ֜ק אֶת־ זְבוּלֻ֤ן וְאֶת־ נַפְתָּלִי֙ קֶ֔דְשָׁה וַיַּ֣עַל בְּרַגְלָ֔יו עֲשֶׂ֥רֶת אַלְפֵ֖י אִ֑ישׁ וַתַּ֥עַל עִמּ֖וֹ דְּבוֹרָֽה׃ (יב) וְחֶ֤בֶר הַקֵּינִי֙ נִפְרָ֣ד מִקַּ֔יִן מִבְּנֵ֥י חֹבָ֖ב חֹתֵ֣ן מֹשֶׁ֑ה וַיֵּ֣ט אָהֳל֔וֹ עַד־ אֵל֥וֹן (בצענים) בְּצַעֲנַנִּ֖ים אֲשֶׁ֥ר אֶת־ קֶֽדֶשׁ׃ (יג) וַיַּגִּ֖דוּ לְסִֽיסְרָ֑א כִּ֥י עָלָ֛ה בָּרָ֥ק בֶּן־ אֲבִינֹ֖עַם הַר־ תָּבֽוֹר׃ (יד) וַיַּזְעֵ֨ק סִֽיסְרָ֜א אֶת־ כָּל־ רִכְבּ֗וֹ תְּשַׁ֤ע מֵאוֹת֙ רֶ֣כֶב בַּרְזֶ֔ל וְאֶת־ כָּל־ הָעָ֖ם אֲשֶׁ֣ר אִתּ֑וֹ מֵחֲרֹ֥שֶׁת הַגּוֹיִ֖ם אֶל־ נַ֥חַל קִישֽׁוֹן׃ (טו) וַתֹּאמֶר֩ דְּבֹרָ֨ה אֶל־ בָּרָ֜ק ק֗וּם כִּ֣י זֶ֤ה הַיּוֹם֙ אֲשֶׁר֩ נָתַ֨ן יְהוָ֤ה אֶת־ סִֽיסְרָא֙ בְּיָדֶ֔ךָ הֲלֹ֥א יְהוָ֖ה יָצָ֣א לְפָנֶ֑יךָ וַיֵּ֤רֶד בָּרָק֙ מֵהַ֣ר תָּב֔וֹר וַעֲשֶׂ֧רֶת אֲלָפִ֛ים אִ֖ישׁ אַחֲרָֽיו׃ (טז) וַיָּ֣הָם יְ֠הוָה אֶת־ סִֽיסְרָ֨א וְאֶת־ כָּל־ הָרֶ֧כֶב וְאֶת־ כָּל־ הַֽמַּחֲנֶ֛ה לְפִי־ חֶ֖רֶב לִפְנֵ֣י בָרָ֑ק וַיֵּ֧רֶד סִֽיסְרָ֛א מֵעַ֥ל הַמֶּרְכָּבָ֖ה וַיָּ֥נָס בְּרַגְלָֽיו׃ (יז) וּבָרָ֗ק רָדַ֞ף אַחֲרֵ֤י הָרֶ֙כֶב֙ וְאַחֲרֵ֣י הַֽמַּחֲנֶ֔ה עַ֖ד חֲרֹ֣שֶׁת הַגּוֹיִ֑ם וַיִּפֹּ֞ל כָּל־ מַחֲנֵ֤ה סִֽיסְרָא֙ לְפִי־ חֶ֔רֶב לֹ֥א נִשְׁאַ֖ר עַד־ אֶחָֽד׃ (יח) וְסִֽיסְרָא֙ נָ֣ס בְּרַגְלָ֔יו אֶל־ אֹ֣הֶל יָעֵ֔ל אֵ֖שֶּׁת חֶ֣בֶר הַקֵּינִ֑י כִּ֣י שָׁל֗וֹם בֵּ֚ין יָבִ֣ין מֶֽלֶךְ־ חָצ֔וֹר וּבֵ֕ין בֵּ֖ית חֶ֥בֶר הַקֵּינִֽי׃ (יט) וַתֵּצֵ֣א יָעֵל֮ לִקְרַ֣את סִֽיסְרָא֒ וַתֹּ֣אמֶר אֵלָ֗יו סוּרָ֧ה אֲדֹנִ֛י סוּרָ֥ה אֵלַ֖י אַל־ תִּירָ֑א וַיָּ֤סַר אֵלֶ֙יהָ֙ הָאֹ֔הֱלָה וַתְּכַסֵּ֖הוּ בַּשְּׂמִיכָֽה׃ (כ) וַיֹּ֧אמֶר אֵלֶ֛יהָ הַשְׁקִינִי־ נָ֥א מְעַט־ מַ֖יִם כִּ֣י צָמֵ֑אתִי וַתִּפְתַּ֞ח אֶת־ נֹ֧אוד הֶחָלָ֛ב וַתַּשְׁקֵ֖הוּ וַתְּכַסֵּֽהוּ׃ (כא) וַיֹּ֣אמֶר אֵלֶ֔יהָ עֲמֹ֖ד פֶּ֣תַח הָאֹ֑הֶל וְהָיָה֩ אִם־ אִ֨ישׁ יָב֜וֹא וּשְׁאֵלֵ֗ךְ וְאָמַ֛ר הֲיֵֽשׁ־ פֹּ֥ה אִ֖ישׁ וְאָמַ֥רְתְּ אָֽיִן׃ (כב) וַתִּקַּ֣ח יָעֵ֣ל אֵֽשֶׁת־ חֶ֠בֶר אֶת־ יְתַ֨ד הָאֹ֜הֶל וַתָּ֧שֶׂם אֶת־ הַמַּקֶּ֣בֶת בְּיָדָ֗הּ וַתָּב֤וֹא אֵלָיו֙ בַּלָּ֔אט וַתִּתְקַ֤ע אֶת־ הַיָּתֵד֙ בְּרַקָּת֔וֹ וַתִּצְנַ֖ח בָּאָ֑רֶץ וְהֽוּא־ נִרְדָּ֥ם וַיָּ֖עַף וַיָּמֹֽת׃ (כג) וְהִנֵּ֣ה בָרָק֮ רֹדֵ֣ף אֶת־ סִֽיסְרָא֒ וַתֵּצֵ֤א יָעֵל֙ לִקְרָאת֔וֹ וַתֹּ֣אמֶר ל֔וֹ לֵ֣ךְ וְאַרְאֶ֔ךָּ אֶת־ הָאִ֖ישׁ אֲשֶׁר־ אַתָּ֣ה מְבַקֵּ֑שׁ וַיָּבֹ֣א אֵלֶ֔יהָ וְהִנֵּ֤ה סִֽיסְרָא֙ נֹפֵ֣ל מֵ֔ת וְהַיָּתֵ֖ד בְּרַקָּתֽוֹ׃ (כד) וַיַּכְנַ֤ע אֱלֹהִים֙ בַּיּ֣וֹם הַה֔וּא אֵ֖ת יָבִ֣ין מֶֽלֶךְ־ כְּנָ֑עַן לִפְנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (כה) וַתֵּ֜לֶךְ יַ֤ד בְּנֵֽי־ יִשְׂרָאֵל֙ הָל֣וֹךְ וְקָשָׁ֔ה עַ֖ל יָבִ֣ין מֶֽלֶךְ־ כְּנָ֑עַן עַ֚ד אֲשֶׁ֣ר הִכְרִ֔יתוּ אֵ֖ת יָבִ֥ין מֶֽלֶךְ־ כְּנָֽעַן׃ תחילת פרק ה (כו) וַתָּ֣שַׁר דְּבוֹרָ֔ה וּבָרָ֖ק בֶּן־ אֲבִינֹ֑עַם בַּיּ֥וֹם הַה֖וּא לֵאמֹֽר׃ (כז) בִּפְרֹ֤עַ פְּרָעוֹת֙ בְּיִשְׂרָאֵ֔ל בְּהִתְנַדֵּ֖ב עָ֑ם בָּרֲכ֖וּ יְהוָֽה׃ (כח) שִׁמְע֣וּ מְלָכִ֔ים הַאֲזִ֖ינוּ רֹֽזְנִ֑ים אָֽנֹכִ֗י לַֽיהוָה֙ אָנֹכִ֣י אָשִׁ֔ירָה אֲזַמֵּ֕ר לַֽיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (כט) יְהוָ֗ה בְּצֵאתְךָ֤ מִשֵּׂעִיר֙ בְּצַעְדְּךָ֙ מִשְּׂדֵ֣ה אֱד֔וֹם אֶ֣רֶץ רָעָ֔שָׁה גַּם־ שָׁמַ֖יִם נָטָ֑פוּ גַּם־ עָבִ֖ים נָ֥טְפוּ מָֽיִם׃ (ל) הָרִ֥ים נָזְל֖וּ מִפְּנֵ֣י יְהוָ֑ה זֶ֣ה סִינַ֔י מִפְּנֵ֕י יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (לא) בִּימֵ֞י שַׁמְגַּ֤ר בֶּן־ עֲנָת֙ בִּימֵ֣י יָעֵ֔ל חָדְל֖וּ אֳרָח֑וֹת וְהֹלְכֵ֣י נְתִיב֔וֹת יֵלְכ֕וּ אֳרָח֖וֹת עֲקַלְקַלּֽוֹת׃ (לב) חָדְל֧וּ פְרָז֛וֹן בְּיִשְׂרָאֵ֖ל חָדֵ֑לּוּ עַ֤ד שַׁקַּ֙מְתִּי֙ דְּבוֹרָ֔ה שַׁקַּ֥מְתִּי אֵ֖ם בְּיִשְׂרָאֵֽל׃ (לג) יִבְחַר֙ אֱלֹהִ֣ים חֲדָשִׁ֔ים אָ֖ז לָחֶ֣ם שְׁעָרִ֑ים מָגֵ֤ן אִם־ יֵֽרָאֶה֙ וָרֹ֔מַח בְּאַרְבָּעִ֥ים אֶ֖לֶף בְּיִשְׂרָאֵֽל׃ (לד) לִבִּי֙ לְחוֹקְקֵ֣י יִשְׂרָאֵ֔ל הַמִּֽתְנַדְּבִ֖ים בָּעָ֑ם בָּרֲכ֖וּ יְהוָֽה׃ (לה) רֹכְבֵי֩ אֲתֹנ֨וֹת צְחֹר֜וֹת יֹשְׁבֵ֧י עַל־ מִדִּ֛ין וְהֹלְכֵ֥י עַל־ דֶּ֖רֶךְ שִֽׂיחוּ׃ (לו) מִקּ֣וֹל מְחַֽצְצִ֗ים בֵּ֚ין מַשְׁאַבִּ֔ים שָׁ֤ם יְתַנּוּ֙ צִדְק֣וֹת יְהוָ֔ה צִדְקֹ֥ת פִּרְזֹנ֖וֹ בְּיִשְׂרָאֵ֑ל אָ֛ז יָרְד֥וּ לַשְּׁעָרִ֖ים עַם־ יְהוָֽה׃ (לז) עוּרִ֤י עוּרִי֙ דְּבוֹרָ֔ה ע֥וּרִי ע֖וּרִי דַּבְּרִי־ שִׁ֑יר ק֥וּם בָּרָ֛ק וּֽשֲׁבֵ֥ה שֶׁבְיְךָ֖ בֶּן־ אֲבִינֹֽעַם׃ (לח) אָ֚ז יְרַ֣ד שָׂרִ֔יד לְאַדִּירִ֖ים עָ֑ם יְהוָ֕ה יְרַד־ לִ֖י בַּגִּבּוֹרִֽים׃ (לט) שָׂ֣ם בַּפָּר֑וּר וַיּוֹצֵ֥א אֵלָ֛יו אֶל־ תַּ֥חַת הָאֵלָ֖ה וַיַּגַּֽשׁ׃ (כא) וַיֹּ֨אמֶר אֵלָ֜יו מַלְאַ֣ךְ הָאֱלֹהִ֗ים קַ֣ח אֶת־ הַבָּשָׂ֤ר וְאֶת־ הַמַּצּוֹת֙ וְהַנַּח֙ אֶל־ הַסֶּ֣לַע הַלָּ֔ז וְאֶת־ הַמָּרַ֖ק שְׁפ֑וֹךְ וַיַּ֖עַשׂ כֵּֽן׃ (כב) וַיִּשְׁלַ֞ח מַלְאַ֣ךְ יְהוָ֗ה אֶת־ קְצֵ֤ה הַמִּשְׁעֶ֙נֶת֙ אֲשֶׁ֣ר בְּיָד֔וֹ וַיִּגַּ֥ע בַּבָּשָׂ֖ר וּבַמַּצּ֑וֹת וַתַּ֨עַל הָאֵ֜שׁ מִן־ הַצּ֗וּר וַתֹּ֤אכַל אֶת־ הַבָּשָׂר֙ וְאֶת־ הַמַּצּ֔וֹת וּמַלְאַ֣ךְ יְהוָ֔ה הָלַ֖ךְ מֵעֵינָֽיו׃ (כג) וַיַּ֣רְא גִּדְע֔וֹן כִּֽי־ מַלְאַ֥ךְ יְהוָ֖ה ה֑וּא וַיֹּ֣אמֶר גִּדְע֗וֹן אֲהָהּ֙ אֲדֹנָ֣י יְהוִ֔ה כִּֽי־ עַל־ כֵּ֤ן רָאִ֙יתִי֙ מַלְאַ֣ךְ יְהוָ֔ה פָּנִ֖ים אֶל־ פָּנִֽים׃ (כד) וַיֹּ֨אמֶר ל֧וֹ יְהוָ֛ה שָׁל֥וֹם לְךָ֖ אַל־ תִּירָ֑א לֹ֖א תָּמֽוּת׃ (כה) וַיִּבֶן֩ שָׁ֨ם גִּדְע֤וֹן מִזְבֵּ֙חַ֙ לַֽיהוָ֔ה וַיִּקְרָא־ ל֥וֹ יְהוָ֖ה שָׁל֑וֹם עַ֚ד הַיּ֣וֹם הַזֶּ֔ה עוֹדֶ֕נּוּ בְּעָפְרָ֖ת אֲבִ֥י הָעֶזְרִֽי׃ (כו) וַיְהִי֮ בַּלַּ֣יְלָה הַהוּא֒ וַיֹּ֧אמֶר ל֣וֹ יְהוָ֗ה קַ֤ח אֶת־ פַּר־ הַשּׁוֹר֙ אֲשֶׁ֣ר לְאָבִ֔יךָ וּפַ֥ר הַשֵּׁנִ֖י שֶׁ֣בַע שָׁנִ֑ים וְהָרַסְתָּ֗ אֶת־ מִזְבַּ֤ח הַבַּ֙עַל֙ אֲשֶׁ֣ר לְאָבִ֔יךָ וְאֶת־ הָאֲשֵׁרָ֥ה אֲשֶׁר־ עָלָ֖יו תִּכְרֹֽת׃ (כז) וּבָנִ֨יתָ מִזְבֵּ֜חַ לַיהוָ֣ה אֱלֹהֶ֗יךָ עַ֣ל רֹ֧אשׁ הַמָּע֛וֹז הַזֶּ֖ה בַּמַּֽעֲרָכָ֑ה וְלָֽקַחְתָּ֙ אֶת־ הַפָּ֣ר הַשֵּׁנִ֔י וְהַעֲלִ֣יתָ עוֹלָ֔ה בַּעֲצֵ֥י הָאֲשֵׁרָ֖ה אֲשֶׁ֥ר תִּכְרֹֽת׃ (כח) וַיִּקַּ֨ח גִּדְע֜וֹן עֲשָׂרָ֤ה אֲנָשִׁים֙ מֵֽעֲבָדָ֔יו וַיַּ֕עַשׂ כַּאֲשֶׁ֛ר דִּבֶּ֥ר אֵלָ֖יו יְהוָ֑ה וַיְהִ֡י כַּאֲשֶׁ֣ר יָרֵא֩ אֶת־ בֵּ֨ית אָבִ֜יו וְאֶת־ אַנְשֵׁ֥י הָעִ֛יר מֵעֲשׂ֥וֹת יוֹמָ֖ם וַיַּ֥עַשׂ לָֽיְלָה׃ (כט) וַיַּשְׁכִּ֜ימוּ אַנְשֵׁ֤י הָעִיר֙ בַּבֹּ֔קֶר וְהִנֵּ֤ה נֻתַּץ֙ מִזְבַּ֣ח הַבַּ֔עַל וְהָאֲשֵׁרָ֥ה אֲשֶׁר־ עָלָ֖יו כֹּרָ֑תָה וְאֵת֙ הַפָּ֣ר הַשֵּׁנִ֔י הֹֽעֲלָ֔ה עַל־ הַמִּזְבֵּ֖חַ הַבָּנֽוּי׃ (ל) וַיֹּֽאמְרוּ֙ אִ֣ישׁ אֶל־ רֵעֵ֔הוּ מִ֥י עָשָׂ֖ה הַדָּבָ֣ר הַזֶּ֑ה וַֽיִּדְרְשׁוּ֙ וַיְבַקְשׁ֔וּ וַיֹּ֣אמְר֔וּ גִּדְעוֹן֙ בֶּן־ יוֹאָ֔שׁ עָשָׂ֖ה הַדָּבָ֥ר הַזֶּֽה׃ (לא) וַיֹּ֨אמְר֜וּ אַנְשֵׁ֤י הָעִיר֙ אֶל־ יוֹאָ֔שׁ הוֹצֵ֥א אֶת־ בִּנְךָ֖ וְיָמֹ֑ת כִּ֤י נָתַץ֙ אֶת־ מִזְבַּ֣ח הַבַּ֔עַל וְכִ֥י כָרַ֖ת הָאֲשֵׁרָ֥ה אֲשֶׁר־ עָלָֽיו׃ (לב) וַיֹּ֣אמֶר יוֹאָ֡שׁ לְכֹל֩ אֲשֶׁר־ עָמְד֨וּ עָלָ֜יו הַאַתֶּ֣ם ׀ תְּרִיב֣וּן לַבַּ֗עַל אִם־ אַתֶּם֙ תּוֹשִׁיע֣וּן אוֹת֔וֹ אֲשֶׁ֨ר יָרִ֥יב ל֛וֹ יוּמַ֖ת עַד־ הַבֹּ֑קֶר אִם־ אֱלֹהִ֥ים הוּא֙ יָ֣רֶב ל֔וֹ כִּ֥י נָתַ֖ץ אֶֽת־ מִזְבְּחֽוֹ׃ (לג) וַיִּקְרָא־ ל֥וֹ בַיּוֹם־ הַה֖וּא יְרֻבַּ֣עַל לֵאמֹ֑ר יָ֤רֶב בּוֹ֙ הַבַּ֔עַל כִּ֥י נָתַ֖ץ אֶֽת־ מִזְבְּחֽוֹ׃ (לד) וְכָל־ מִדְיָ֧ן וַעֲמָלֵ֛ק וּבְנֵי־ קֶ֖דֶם נֶאֶסְפ֣וּ יַחְדָּ֑ו וַיַּעַבְר֥וּ וַֽיַּחֲנ֖וּ בְּעֵ֥מֶק יִזְרְעֶֽאל׃ (לה) וְר֣וּחַ יְהוָ֔ה לָבְשָׁ֖ה אֶת־ גִּדְע֑וֹן וַיִּתְקַע֙ בַּשּׁוֹפָ֔ר וַיִזָּעֵ֥ק אֲבִיעֶ֖זֶר אַחֲרָֽיו׃ (לו) וּמַלְאָכִים֙ שָׁלַ֣ח בְּכָל־ מְנַשֶּׁ֔ה וַיִזָּעֵ֥ק גַּם־ ה֖וּא אַחֲרָ֑יו וּמַלְאָכִ֣ים שָׁלַ֗ח בְּאָשֵׁ֤ר וּבִזְבֻלוּן֙ וּבְנַפְתָּלִ֔י וַֽיַּעֲל֖וּ לִקְרָאתָֽם׃ (לז) וַיֹּ֥אמֶר גִּדְע֖וֹן אֶל־ הָאֱלֹהִ֑ים אִם־ יֶשְׁךָ֞ מוֹשִׁ֧יעַ בְּיָדִ֛י אֶת־ יִשְׂרָאֵ֖ל כַּאֲשֶׁ֥ר דִּבַּֽרְתָּ׃ (לח) הִנֵּ֣ה אָנֹכִ֗י מַצִּ֛יג אֶת־ גִּזַּ֥ת הַצֶּ֖מֶר בַּגֹּ֑רֶן אִ֡ם טַל֩ יִהְיֶ֨ה עַֽל־ הַגִּזָּ֜ה לְבַדָּ֗הּ וְעַל־ כָּל־ הָאָ֙רֶץ֙ חֹ֔רֶב וְיָדַעְתִּ֗י כִּֽי־ תוֹשִׁ֧יעַ בְּיָדִ֛י אֶת־ יִשְׂרָאֵ֖ל כַּאֲשֶׁ֥ר דִּבַּֽרְתָּ׃ (לט) וַיְהִי־ כֵ֕ן וַיַּשְׁכֵּם֙ מִֽמָּחֳרָ֔ת וַיָּ֖זַר אֶת־ הַגִּזָּ֑ה וַיִּ֤מֶץ טַל֙ מִן־ הַגִּזָּ֔ה מְל֥וֹא הַסֵּ֖פֶל מָֽיִם׃ (מ) וַיֹּ֤אמֶר גִּדְעוֹן֙ אֶל־ הָ֣אֱלֹהִ֔ים אַל־ יִ֤חַר אַפְּךָ֙ בִּ֔י וַאֲדַבְּרָ֖ה אַ֣ךְ הַפָּ֑עַם אֲנַסֶּ֤ה נָּא־ רַק־ הַפַּ֙עַם֙ בַּגִּזָּ֔ה יְהִי־ נָ֨א חֹ֤רֶב אֶל־ הַגִּזָּה֙ לְבַדָּ֔הּ וְעַל־ כָּל־ הָאָ֖רֶץ יִֽהְיֶה־ טָּֽל׃
