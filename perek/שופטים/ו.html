שופטים פרק ו לפי החלוקה לסדרים

פרק ו בספר שופטים (40 פסוקים) נמצא בשופטים סדר ד (פסוקים ב–מ) ובשופטים סדר ה (פסוקים א–א).

שופטים סדר ד

(א) כֵּ֠ן יֹאבְד֤וּ כָל־ אוֹיְבֶ֙יךָ֙ יְהוָ֔ה וְאֹ֣הֲבָ֔יו כְּצֵ֥את הַשֶּׁ֖מֶשׁ בִּגְבֻרָת֑וֹ וַתִּשְׁקֹ֥ט הָאָ֖רֶץ אַרְבָּעִ֥ים שָׁנָֽה׃ תחילת פרק ו (ב) וַיַּעֲשׂ֧וּ בְנֵי־ יִשְׂרָאֵ֛ל הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיִּתְּנֵ֧ם יְהוָ֛ה בְּיַד־ מִדְיָ֖ן שֶׁ֥בַע שָׁנִֽים׃ (ג) וַתָּ֥עָז יַד־ מִדְיָ֖ן עַל־ יִשְׂרָאֵ֑ל מִפְּנֵ֨י מִדְיָ֜ן עָשֽׂוּ לָהֶ֣ם ׀ בְּנֵ֣י יִשְׂרָאֵ֗ל אֶת־ הַמִּנְהָרוֹת֙ אֲשֶׁ֣ר בֶּֽהָרִ֔ים וְאֶת־ הַמְּעָר֖וֹת וְאֶת־ הַמְּצָדֽוֹת׃ (ד) וְהָיָ֖ה אִם־ זָרַ֣ע יִשְׂרָאֵ֑ל וְעָלָ֨ה מִדְיָ֧ן וַֽעֲמָלֵ֛ק וּבְנֵי־ קֶ֖דֶם וְעָל֥וּ עָלָֽיו׃ (ה) וַיַּחֲנ֣וּ עֲלֵיהֶ֗ם וַיַּשְׁחִ֙יתוּ֙ אֶת־ יְב֣וּל הָאָ֔רֶץ עַד־ בּוֹאֲךָ֖ עַזָּ֑ה וְלֹֽא־ יַשְׁאִ֤ירוּ מִֽחְיָה֙ בְּיִשְׂרָאֵ֔ל וְשֶׂ֥ה וָשׁ֖וֹר וַחֲמֽוֹר׃ (ו) כִּ֡י הֵם֩ וּמִקְנֵיהֶ֨ם יַעֲל֜וּ וְאָהֳלֵיהֶ֗ם (יבאו) וּבָ֤אוּ כְדֵֽי־ אַרְבֶּה֙ לָרֹ֔ב וְלָהֶ֥ם וְלִגְמַלֵּיהֶ֖ם אֵ֣ין מִסְפָּ֑ר וַיָּבֹ֥אוּ בָאָ֖רֶץ לְשַׁחֲתָֽהּ׃ (ז) וַיִּדַּ֧ל יִשְׂרָאֵ֛ל מְאֹ֖ד מִפְּנֵ֣י מִדְיָ֑ן וַיִּזְעֲק֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל אֶל־ יְהוָֽה׃ (ח) וַיְהִ֕י כִּֽי־ זָעֲק֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל אֶל־ יְהוָ֑ה עַ֖ל אֹד֥וֹת מִדְיָֽן׃ (ט) וַיִּשְׁלַ֧ח יְהוָ֛ה אִ֥ישׁ נָבִ֖יא אֶל־ בְּנֵ֣י יִשְׂרָאֵ֑ל וַיֹּ֨אמֶר לָהֶ֜ם כֹּה־ אָמַ֥ר יְהוָ֣ה ׀ אֱלֹהֵ֣י יִשְׂרָאֵ֗ל אָנֹכִ֞י הֶעֱלֵ֤יתִי אֶתְכֶם֙ מִמִּצְרַ֔יִם וָאֹצִ֥יא אֶתְכֶ֖ם מִבֵּ֥ית עֲבָדִֽים׃ (י) וָאַצִּ֤ל אֶתְכֶם֙ מִיַּ֣ד מִצְרַ֔יִם וּמִיַּ֖ד כָּל־ לֹחֲצֵיכֶ֑ם וָאֲגָרֵ֤שׁ אוֹתָם֙ מִפְּנֵיכֶ֔ם וָאֶתְּנָ֥ה לָכֶ֖ם אֶת־ אַרְצָֽם׃ (יא) וָאֹמְרָ֣ה לָכֶ֗ם אֲנִי֙ יְהוָ֣ה אֱלֹהֵיכֶ֔ם לֹ֤א תִֽירְאוּ֙ אֶת־ אֱלֹהֵ֣י הָאֱמֹרִ֔י אֲשֶׁ֥ר אַתֶּ֖ם יוֹשְׁבִ֣ים בְּאַרְצָ֑ם וְלֹ֥א שְׁמַעְתֶּ֖ם בְּקוֹלִֽי׃ (יב) וַיָּבֹ֞א מַלְאַ֣ךְ יְהוָ֗ה וַיֵּ֙שֶׁב֙ תַּ֤חַת הָֽאֵלָה֙ אֲשֶׁ֣ר בְּעָפְרָ֔ה אֲשֶׁ֥ר לְיוֹאָ֖שׁ אֲבִ֣י הָֽעֶזְרִ֑י וְגִדְע֣וֹן בְּנ֗וֹ חֹבֵ֤ט חִטִּים֙ בַּגַּ֔ת לְהָנִ֖יס מִפְּנֵ֥י מִדְיָֽן׃ (יג) וַיֵּרָ֥א אֵלָ֖יו מַלְּאַ֣ךְ יְהוָ֑ה וַיֹּ֣אמֶר אֵלָ֔יו יְהוָ֥ה עִמְּךָ֖ גִּבּ֥וֹר הֶחָֽיִל׃ (יד) וַיֹּ֨אמֶר אֵלָ֤יו גִּדְעוֹן֙ בִּ֣י אֲדֹנִ֔י וְיֵ֤שׁ יְהוָה֙ עִמָּ֔נוּ וְלָ֥מָּה מְצָאַ֖תְנוּ כָּל־ זֹ֑את וְאַיֵּ֣ה כָֽל־ נִפְלְאֹתָ֡יו אֲשֶׁר֩ סִפְּרוּ־ לָ֨נוּ אֲבוֹתֵ֜ינוּ לֵאמֹ֗ר הֲלֹ֤א מִמִּצְרַ֙יִם֙ הֶעֱלָ֣נוּ יְהוָ֔ה וְעַתָּה֙ נְטָשָׁ֣נוּ יְהוָ֔ה וַֽיִּתְּנֵ֖נוּ בְּכַף־ מִדְיָֽן׃ (טו) וַיִּ֤פֶן אֵלָיו֙ יְהוָ֔ה וַיֹּ֗אמֶר לֵ֚ךְ בְּכֹחֲךָ֣ זֶ֔ה וְהוֹשַׁעְתָּ֥ אֶת־ יִשְׂרָאֵ֖ל מִכַּ֣ף מִדְיָ֑ן הֲלֹ֖א שְׁלַחְתִּֽיךָ׃ (טז) וַיֹּ֤אמֶר אֵלָיו֙ בִּ֣י אֲדֹנָ֔י בַּמָּ֥ה אוֹשִׁ֖יעַ אֶת־ יִשְׂרָאֵ֑ל הִנֵּ֤ה אַלְפִּי֙ הַדַּ֣ל בִּמְנַשֶּׁ֔ה וְאָנֹכִ֥י הַצָּעִ֖יר בְּבֵ֥ית אָבִֽי׃ (יז) וַיֹּ֤אמֶר אֵלָיו֙ יְהוָ֔ה כִּ֥י אֶהְיֶ֖ה עִמָּ֑ךְ וְהִכִּיתָ֥ אֶת־ מִדְיָ֖ן כְּאִ֥ישׁ אֶחָֽד׃ (יח) וַיֹּ֣אמֶר אֵלָ֔יו אִם־ נָ֛א מָצָ֥אתִי חֵ֖ן בְּעֵינֶ֑יךָ וְעָשִׂ֤יתָ לִּי֙ א֔וֹת שָׁאַתָּ֖ה מְדַבֵּ֥ר עִמִּֽי׃ (יט) אַל־ נָ֨א תָמֻ֤שׁ מִזֶּה֙ עַד־ בֹּאִ֣י אֵלֶ֔יךָ וְהֹֽצֵאתִי֙ אֶת־ מִנְחָתִ֔י וְהִנַּחְתִּ֖י לְפָנֶ֑יךָ וַיֹּאמַ֕ר אָנֹכִ֥י אֵשֵׁ֖ב עַ֥ד שׁוּבֶֽךָ׃ (כ) וְגִדְע֣וֹן בָּ֗א וַיַּ֤עַשׂ גְּדִֽי־ עִזִּים֙ וְאֵיפַת־ קֶ֣מַח מַצּ֔וֹת הַבָּשָׂר֙ שָׂ֣ם בַּסַּ֔ל וְהַמָּרַ֖ק שָׂ֣ם בַּפָּר֑וּר וַיּוֹצֵ֥א אֵלָ֛יו אֶל־ תַּ֥חַת הָאֵלָ֖ה וַיַּגַּֽשׁ׃ (כא) וַיֹּ֨אמֶר אֵלָ֜יו מַלְאַ֣ךְ הָאֱלֹהִ֗ים קַ֣ח אֶת־ הַבָּשָׂ֤ר וְאֶת־ הַמַּצּוֹת֙ וְהַנַּח֙ אֶל־ הַסֶּ֣לַע הַלָּ֔ז וְאֶת־ הַמָּרַ֖ק שְׁפ֑וֹךְ וַיַּ֖עַשׂ כֵּֽן׃ (כב) וַיִּשְׁלַ֞ח מַלְאַ֣ךְ יְהוָ֗ה אֶת־ קְצֵ֤ה הַמִּשְׁעֶ֙נֶת֙ אֲשֶׁ֣ר בְּיָד֔וֹ וַיִּגַּ֥ע בַּבָּשָׂ֖ר וּבַמַּצּ֑וֹת וַתַּ֨עַל הָאֵ֜שׁ מִן־ הַצּ֗וּר וַתֹּ֤אכַל אֶת־ הַבָּשָׂר֙ וְאֶת־ הַמַּצּ֔וֹת וּמַלְאַ֣ךְ יְהוָ֔ה הָלַ֖ךְ מֵעֵינָֽיו׃ (כג) וַיַּ֣רְא גִּדְע֔וֹן כִּֽי־ מַלְאַ֥ךְ יְהוָ֖ה ה֑וּא וַיֹּ֣אמֶר גִּדְע֗וֹן אֲהָהּ֙ אֲדֹנָ֣י יְהוִ֔ה כִּֽי־ עַל־ כֵּ֤ן רָאִ֙יתִי֙ מַלְאַ֣ךְ יְהוָ֔ה פָּנִ֖ים אֶל־ פָּנִֽים׃ (כד) וַיֹּ֨אמֶר ל֧וֹ יְהוָ֛ה שָׁל֥וֹם לְךָ֖ אַל־ תִּירָ֑א לֹ֖א תָּמֽוּת׃ (כה) וַיִּבֶן֩ שָׁ֨ם גִּדְע֤וֹן מִזְבֵּ֙חַ֙ לַֽיהוָ֔ה וַיִּקְרָא־ ל֥וֹ יְהוָ֖ה שָׁל֑וֹם עַ֚ד הַיּ֣וֹם הַזֶּ֔ה עוֹדֶ֕נּוּ בְּעָפְרָ֖ת אֲבִ֥י הָעֶזְרִֽי׃ (כו) וַיְהִי֮ בַּלַּ֣יְלָה הַהוּא֒ וַיֹּ֧אמֶר ל֣וֹ יְהוָ֗ה קַ֤ח אֶת־ פַּר־ הַשּׁוֹר֙ אֲשֶׁ֣ר לְאָבִ֔יךָ וּפַ֥ר הַשֵּׁנִ֖י שֶׁ֣בַע שָׁנִ֑ים וְהָרַסְתָּ֗ אֶת־ מִזְבַּ֤ח הַבַּ֙עַל֙ אֲשֶׁ֣ר לְאָבִ֔יךָ וְאֶת־ הָאֲשֵׁרָ֥ה אֲשֶׁר־ עָלָ֖יו תִּכְרֹֽת׃ (כז) וּבָנִ֨יתָ מִזְבֵּ֜חַ לַיהוָ֣ה אֱלֹהֶ֗יךָ עַ֣ל רֹ֧אשׁ הַמָּע֛וֹז הַזֶּ֖ה בַּמַּֽעֲרָכָ֑ה וְלָֽקַחְתָּ֙ אֶת־ הַפָּ֣ר הַשֵּׁנִ֔י וְהַעֲלִ֣יתָ עוֹלָ֔ה בַּעֲצֵ֥י הָאֲשֵׁרָ֖ה אֲשֶׁ֥ר תִּכְרֹֽת׃ (כח) וַיִּקַּ֨ח גִּדְע֜וֹן עֲשָׂרָ֤ה אֲנָשִׁים֙ מֵֽעֲבָדָ֔יו וַיַּ֕עַשׂ כַּאֲשֶׁ֛ר דִּבֶּ֥ר אֵלָ֖יו יְהוָ֑ה וַיְהִ֡י כַּאֲשֶׁ֣ר יָרֵא֩ אֶת־ בֵּ֨ית אָבִ֜יו וְאֶת־ אַנְשֵׁ֥י הָעִ֛יר מֵעֲשׂ֥וֹת יוֹמָ֖ם וַיַּ֥עַשׂ לָֽיְלָה׃ (כט) וַיַּשְׁכִּ֜ימוּ אַנְשֵׁ֤י הָעִיר֙ בַּבֹּ֔קֶר וְהִנֵּ֤ה נֻתַּץ֙ מִזְבַּ֣ח הַבַּ֔עַל וְהָאֲשֵׁרָ֥ה אֲשֶׁר־ עָלָ֖יו כֹּרָ֑תָה וְאֵת֙ הַפָּ֣ר הַשֵּׁנִ֔י הֹֽעֲלָ֔ה עַל־ הַמִּזְבֵּ֖חַ הַבָּנֽוּי׃ (ל) וַיֹּֽאמְרוּ֙ אִ֣ישׁ אֶל־ רֵעֵ֔הוּ מִ֥י עָשָׂ֖ה הַדָּבָ֣ר הַזֶּ֑ה וַֽיִּדְרְשׁוּ֙ וַיְבַקְשׁ֔וּ וַיֹּ֣אמְר֔וּ גִּדְעוֹן֙ בֶּן־ יוֹאָ֔שׁ עָשָׂ֖ה הַדָּבָ֥ר הַזֶּֽה׃ (לא) וַיֹּ֨אמְר֜וּ אַנְשֵׁ֤י הָעִיר֙ אֶל־ יוֹאָ֔שׁ הוֹצֵ֥א אֶת־ בִּנְךָ֖ וְיָמֹ֑ת כִּ֤י נָתַץ֙ אֶת־ מִזְבַּ֣ח הַבַּ֔עַל וְכִ֥י כָרַ֖ת הָאֲשֵׁרָ֥ה אֲשֶׁר־ עָלָֽיו׃ (לב) וַיֹּ֣אמֶר יוֹאָ֡שׁ לְכֹל֩ אֲשֶׁר־ עָמְד֨וּ עָלָ֜יו הַאַתֶּ֣ם ׀ תְּרִיב֣וּן לַבַּ֗עַל אִם־ אַתֶּם֙ תּוֹשִׁיע֣וּן אוֹת֔וֹ אֲשֶׁ֨ר יָרִ֥יב ל֛וֹ יוּמַ֖ת עַד־ הַבֹּ֑קֶר אִם־ אֱלֹהִ֥ים הוּא֙ יָ֣רֶב ל֔וֹ כִּ֥י נָתַ֖ץ אֶֽת־ מִזְבְּחֽוֹ׃ (לג) וַיִּקְרָא־ ל֥וֹ בַיּוֹם־ הַה֖וּא יְרֻבַּ֣עַל לֵאמֹ֑ר יָ֤רֶב בּוֹ֙ הַבַּ֔עַל כִּ֥י נָתַ֖ץ אֶֽת־ מִזְבְּחֽוֹ׃ (לד) וְכָל־ מִדְיָ֧ן וַעֲמָלֵ֛ק וּבְנֵי־ קֶ֖דֶם נֶאֶסְפ֣וּ יַחְדָּ֑ו וַיַּעַבְר֥וּ וַֽיַּחֲנ֖וּ בְּעֵ֥מֶק יִזְרְעֶֽאל׃ (לה) וְר֣וּחַ יְהוָ֔ה לָבְשָׁ֖ה אֶת־ גִּדְע֑וֹן וַיִּתְקַע֙ בַּשּׁוֹפָ֔ר וַיִזָּעֵ֥ק אֲבִיעֶ֖זֶר אַחֲרָֽיו׃ (לו) וּמַלְאָכִים֙ שָׁלַ֣ח בְּכָל־ מְנַשֶּׁ֔ה וַיִזָּעֵ֥ק גַּם־ ה֖וּא אַחֲרָ֑יו וּמַלְאָכִ֣ים שָׁלַ֗ח בְּאָשֵׁ֤ר חֶ֥רֶב אִ֛ישׁ בְּרֵעֵ֖הוּ וּבְכָל־ הַֽמַּחֲנֶ֑ה וַיָּ֨נָס הַֽמַּחֲנֶ֜ה עַד־ בֵּ֤ית הַשִּׁטָּה֙ צְֽרֵרָ֔תָה עַ֛ד שְׂפַת־ אָבֵ֥ל מְחוֹלָ֖ה עַל־ טַבָּֽת׃ (כד) וַיִּצָּעֵ֧ק אִֽישׁ־ יִשְׂרָאֵ֛ל מִנַּפְתָּלִ֥י וּמִן־ אָשֵׁ֖ר וּמִן־ כָּל־ מְנַשֶּׁ֑ה וַֽיִּרְדְּפ֖וּ אַחֲרֵ֥י מִדְיָֽן׃ (כה) וּמַלְאָכִ֡ים שָׁלַ֣ח גִּדְעוֹן֩ בְּכָל־ הַ֨ר אֶפְרַ֜יִם לֵאמֹ֗ר רְד֞וּ לִקְרַ֤את מִדְיָן֙ וְלִכְד֤וּ לָהֶם֙ אֶת־ הַמַּ֔יִם עַ֛ד בֵּ֥ית בָּרָ֖ה וְאֶת־ הַיַּרְדֵּ֑ן וַיִּצָּעֵ֞ק כָּל־ אִ֤ישׁ אֶפְרַ֙יִם֙ וַיִּלְכְּד֣וּ אֶת־ הַמַּ֔יִם עַ֛ד בֵּ֥ית בָּרָ֖ה וְאֶת־ הַיַּרְדֵּֽן׃ (כו) וַֽיִּלְכְּד֡וּ שְׁנֵֽי־ שָׂרֵ֨י מִדְיָ֜ן אֶת־ עֹרֵ֣ב וְאֶת־ זְאֵ֗ב וַיַּהַרְג֨וּ אֶת־ עוֹרֵ֤ב בְּצוּר־ עוֹרֵב֙ וְאֶת־ זְאֵב֙ הָרְג֣וּ בְיֶֽקֶב־ זְאֵ֔ב וַֽיִּרְדְּפ֖וּ אֶל־ מִדְיָ֑ן וְרֹאשׁ־ עֹרֵ֣ב וּזְאֵ֔ב הֵבִ֙יאוּ֙ אֶל־ גִּדְע֔וֹן מֵעֵ֖בֶר לַיַּרְדֵּֽן׃ (כז) וַיֹּאמְר֨וּ אֵלָ֜יו אִ֣ישׁ אֶפְרַ֗יִם מָֽה־ הַדָּבָ֤ר הַזֶּה֙ עָשִׂ֣יתָ לָּ֔נוּ לְבִלְתִּי֙ קְרֹ֣אות לָ֔נוּ כִּ֥י הָלַ֖כְתָּ לְהִלָּחֵ֣ם בְּמִדְיָ֑ן וַיְרִיב֥וּן אִתּ֖וֹ בְּחָזְקָֽה׃ (כח) וַיֹּ֣אמֶר אֲלֵיהֶ֔ם מֶה־ עָשִׂ֥יתִי עַתָּ֖ה כָּכֶ֑ם הֲל֗וֹא ט֛וֹב עֹלְל֥וֹת אֶפְרַ֖יִם מִבְצִ֥יר אֲבִיעֶֽזֶר׃
