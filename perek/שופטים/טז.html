שופטים פרק טז לפי החלוקה לסדרים

פרק טז בספר שופטים (31 פסוקים) נמצא בשופטים סדר י (פסוקים מג–מד) ובשופטים סדר יא (פסוקים א–כט).

שופטים סדר י

(א) וַתֵּ֤לֶד הָֽאִשָּׁה֙ בֵּ֔ן וַתִּקְרָ֥א אֶת־ שְׁמ֖וֹ שִׁמְשׁ֑וֹן וַיִּגְדַּ֣ל הַנַּ֔עַר וַֽיְבָרְכֵ֖הוּ יְהוָֽה׃ (ב) וַתָּ֙חֶל֙ ר֣וּחַ יְהוָ֔ה לְפַעֲמ֖וֹ בְּמַחֲנֵה־ דָ֑ן בֵּ֥ין צָרְעָ֖ה וּבֵ֥ין אֶשְׁתָּאֹֽל׃ (ג) וַיֵּ֥רֶד שִׁמְשׁ֖וֹן תִּמְנָ֑תָה וַיַּ֥רְא אִשָּׁ֛ה בְּתִמְנָ֖תָה מִבְּנ֥וֹת פְּלִשְׁתִּֽים׃ (ד) וַיַּ֗עַל וַיַּגֵּד֙ לְאָבִ֣יו וּלְאִמּ֔וֹ וַיֹּ֗אמֶר אִשָּׁ֛ה רָאִ֥יתִי בְתִמְנָ֖תָה מִבְּנ֣וֹת פְּלִשְׁתִּ֑ים וְעַתָּ֕ה קְחוּ־ אוֹתָ֥הּ לִּ֖י לְאִשָּֽׁה׃ (ה) וַיֹּ֨אמֶר ל֜וֹ אָבִ֣יו וְאִמּ֗וֹ הַאֵין֩ בִּבְנ֨וֹת אַחֶ֤יךָ וּבְכָל־ עַמִּי֙ אִשָּׁ֔ה כִּֽי־ אַתָּ֤ה הוֹלֵךְ֙ לָקַ֣חַת אִשָּׁ֔ה מִפְּלִשְׁתִּ֖ים הָעֲרֵלִ֑ים וַיֹּ֨אמֶר שִׁמְשׁ֤וֹן אֶל־ אָבִיו֙ אוֹתָ֣הּ קַֽח־ לִ֔י כִּֽי־ הִ֖יא יָשְׁרָ֥ה בְעֵינָֽי׃ (ו) וְאָבִ֨יו וְאִמּ֜וֹ לֹ֣א יָדְע֗וּ כִּ֤י מֵיְהוָה֙ הִ֔יא כִּי־ תֹאֲנָ֥ה הֽוּא־ מְבַקֵּ֖שׁ מִפְּלִשְׁתִּ֑ים וּבָעֵ֣ת הַהִ֔יא פְּלִשְׁתִּ֖ים מֹשְׁלִ֥ים בְּיִשְׂרָאֵֽל׃ (ז) וַיֵּ֧רֶד שִׁמְשׁ֛וֹן וְאָבִ֥יו וְאִמּ֖וֹ תִּמְנָ֑תָה וַיָּבֹ֙אוּ֙ עַד־ כַּרְמֵ֣י תִמְנָ֔תָה וְהִנֵּה֙ כְּפִ֣יר אֲרָי֔וֹת שֹׁאֵ֖ג לִקְרָאתֽוֹ׃ (ח) וַתִּצְלַ֨ח עָלָ֜יו ר֣וּחַ יְהוָ֗ה וַֽיְשַׁסְּעֵ֙הוּ֙ כְּשַׁסַּ֣ע הַגְּדִ֔י וּמְא֖וּמָה אֵ֣ין בְּיָד֑וֹ וְלֹ֤א הִגִּיד֙ לְאָבִ֣יו וּלְאִמּ֔וֹ אֵ֖ת אֲשֶׁ֥ר עָשָֽׂה׃ (ט) וַיֵּ֖רֶד וַיְדַבֵּ֣ר לָאִשָּׁ֑ה וַתִּישַׁ֖ר בְּעֵינֵ֥י שִׁמְשֽׁוֹן׃ (י) וַיָּ֤שָׁב מִיָּמִים֙ לְקַחְתָּ֔הּ וַיָּ֣סַר לִרְא֔וֹת אֵ֖ת מַפֶּ֣לֶת הָאַרְיֵ֑ה וְהִנֵּ֨ה עֲדַ֧ת דְּבוֹרִ֛ים בִּגְוִיַּ֥ת הָאַרְיֵ֖ה וּדְבָֽשׁ׃ (יא) וַיִּרְדֵּ֣הוּ אֶל־ כַּפָּ֗יו וַיֵּ֤לֶךְ הָלוֹךְ֙ וְאָכֹ֔ל וַיֵּ֙לֶךְ֙ אֶל־ אָבִ֣יו וְאֶל־ אִמּ֔וֹ וַיִּתֵּ֥ן לָהֶ֖ם וַיֹּאכֵ֑לוּ וְלֹֽא־ הִגִּ֣יד לָהֶ֔ם כִּ֛י מִגְּוִיַּ֥ת הָאַרְיֵ֖ה רָדָ֥ה הַדְּבָֽשׁ׃ (יב) וַיֵּ֥רֶד אָבִ֖יהוּ אֶל־ הָאִשָּׁ֑ה וַיַּ֨עַשׂ שָׁ֤ם שִׁמְשׁוֹן֙ מִשְׁתֶּ֔ה כִּ֛י כֵּ֥ן יַעֲשׂ֖וּ הַבַּחוּרִֽים׃ (יג) וַיְהִ֖י כִּרְאוֹתָ֣ם אוֹת֑וֹ וַיִּקְחוּ֙ שְׁלֹשִׁ֣ים מֵֽרֵעִ֔ים וַיִּהְי֖וּ אִתּֽוֹ׃ (יד) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן אָחֽוּדָה־ נָּ֥א לָכֶ֖ם חִידָ֑ה אִם־ הַגֵּ֣ד תַּגִּידוּ֩ אוֹתָ֨הּ לִ֜י שִׁבְעַ֨ת יְמֵ֤י הַמִּשְׁתֶּה֙ וּמְצָאתֶ֔ם וְנָתַתִּ֤י לָכֶם֙ שְׁלֹשִׁ֣ים סְדִינִ֔ים וּשְׁלֹשִׁ֖ים חֲלִפֹ֥ת בְּגָדִֽים׃ (טו) וְאִם־ לֹ֣א תוּכְלוּ֮ לְהַגִּ֣יד לִי֒ וּנְתַתֶּ֨ם אַתֶּ֥ם לִי֙ שְׁלֹשִׁ֣ים סְדִינִ֔ים וּשְׁלֹשִׁ֖ים חֲלִיפ֣וֹת בְּגָדִ֑ים וַיֹּ֣אמְרוּ ל֔וֹ ח֥וּדָה חִידָתְךָ֖ וְנִשְׁמָעֶֽנָּה׃ (טז) וַיֹּ֣אמֶר לָהֶ֗ם מֵהָֽאֹכֵל֙ יָצָ֣א מַאֲכָ֔ל וּמֵעַ֖ז יָצָ֣א מָת֑וֹק וְלֹ֥א יָכְל֛וּ לְהַגִּ֥יד הַחִידָ֖ה שְׁלֹ֥שֶׁת יָמִֽים׃ (יז) וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁבִיעִ֗י וַיֹּאמְר֤וּ לְאֵֽשֶׁת־ שִׁמְשׁוֹן֙ פַּתִּ֣י אֶת־ אִישֵׁ֗ךְ וְיַגֶּד־ לָ֙נוּ֙ אֶת־ הַ֣חִידָ֔ה פֶּן־ נִשְׂרֹ֥ף אוֹתָ֛ךְ וְאֶת־ בֵּ֥ית אָבִ֖יךְ בָּאֵ֑שׁ הַלְיָרְשֵׁ֕נוּ קְרָאתֶ֥ם לָ֖נוּ הֲלֹֽא׃ (יח) וַתֵּבְךְּ֩ אֵ֨שֶׁת שִׁמְשׁ֜וֹן עָלָ֗יו וַתֹּ֙אמֶר֙ רַק־ שְׂנֵאתַ֙נִי֙ וְלֹ֣א אֲהַבְתָּ֔נִי הַֽחִידָ֥ה חַ֙דְתָּ֙ לִבְנֵ֣י עַמִּ֔י וְלִ֖י לֹ֣א הִגַּ֑דְתָּה וַיֹּ֣אמֶר לָ֗הּ הִנֵּ֨ה לְאָבִ֧י וּלְאִמִּ֛י לֹ֥א הִגַּ֖דְתִּי וְלָ֥ךְ אַגִּֽיד׃ (יט) וַתֵּ֤בְךְּ עָלָיו֙ שִׁבְעַ֣ת הַיָּמִ֔ים אֲשֶׁר־ הָיָ֥ה לָהֶ֖ם הַמִּשְׁתֶּ֑ה וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁבִיעִ֗י וַיַּגֶּד־ לָהּ֙ כִּ֣י הֱצִיקַ֔תְהוּ וַתַּגֵּ֥ד הַחִידָ֖ה לִבְנֵ֥י עַמָּֽהּ׃ (כ) וַיֹּ֣אמְרוּ לוֹ֩ אַנְשֵׁ֨י הָעִ֜יר בַּיּ֣וֹם הַשְּׁבִיעִ֗י בְּטֶ֙רֶם֙ יָבֹ֣א הַחַ֔רְסָה מַה־ מָּת֣וֹק מִדְּבַ֔שׁ וּמֶ֥ה עַ֖ז מֵאֲרִ֑י וַיֹּ֣אמֶר לָהֶ֔ם לוּלֵא֙ חֲרַשְׁתֶּ֣ם בְּעֶגְלָתִ֔י לֹ֥א מְצָאתֶ֖ם חִידָתִֽי׃ (כא) וַתִּצְלַ֨ח עָלָ֜יו ר֣וּחַ יְהוָ֗ה וַיֵּ֨רֶד אַשְׁקְל֜וֹן וַיַּ֥ךְ מֵהֶ֣ם ׀ שְׁלֹשִׁ֣ים אִ֗ישׁ וַיִּקַּח֙ אֶת־ חֲלִ֣יצוֹתָ֔ם וַיִּתֵּן֙ הַחֲלִיפ֔וֹת לְמַגִּידֵ֖י הַחִידָ֑ה וַיִּ֣חַר אַפּ֔וֹ וַיַּ֖עַל בֵּ֥ית אָבִֽיהוּ׃ (כב) וַתְּהִ֖י אֵ֣שֶׁת שִׁמְשׁ֑וֹן לְמֵ֣רֵעֵ֔הוּ אֲשֶׁ֥ר רֵעָ֖ה לֽוֹ׃ (כג) וַיְהִ֨י מִיָּמִ֜ים בִּימֵ֣י קְצִיר־ חִטִּ֗ים וַיִּפְקֹ֨ד שִׁמְשׁ֤וֹן אֶת־ אִשְׁתּוֹ֙ בִּגְדִ֣י עִזִּ֔ים וַיֹּ֕אמֶר אָבֹ֥אָה אֶל־ אִשְׁתִּ֖י הֶחָ֑דְרָה וְלֹֽא־ נְתָנ֥וֹ אָבִ֖יהָ לָבֽוֹא׃ (כד) וַיֹּ֣אמֶר אָבִ֗יהָ אָמֹ֤ר אָמַ֙רְתִּי֙ כִּי־ שָׂנֹ֣א שְׂנֵאתָ֔הּ וָאֶתְּנֶ֖נָּה לְמֵרֵעֶ֑ךָ הֲלֹ֨א אֲחֹתָ֤הּ הַקְּטַנָּה֙ טוֹבָ֣ה מִמֶּ֔נָּה תְּהִי־ נָ֥א לְךָ֖ תַּחְתֶּֽיהָ׃ (כה) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן נִקֵּ֥יתִי הַפַּ֖עַם מִפְּלִשְׁתִּ֑ים כִּֽי־ עֹשֶׂ֥ה אֲנִ֛י עִמָּ֖ם רָעָֽה׃ (כו) וַיֵּ֣לֶךְ שִׁמְשׁ֔וֹן וַיִּלְכֹּ֖ד שְׁלֹשׁ־ מֵא֣וֹת שׁוּעָלִ֑ים וַיִּקַּ֣ח לַפִּדִ֗ים וַיֶּ֤פֶן זָנָב֙ אֶל־ זָנָ֔ב וַיָּ֨שֶׂם לַפִּ֥יד אֶחָ֛ד בֵּין־ שְׁנֵ֥י הַזְּנָב֖וֹת בַּתָּֽוֶךְ׃ (כז) וַיַּבְעֶר־ אֵשׁ֙ בַּלַּפִּידִ֔ים וַיְשַׁלַּ֖ח בְּקָמ֣וֹת פְּלִשְׁתִּ֑ים וַיַּבְעֵ֛ר מִגָּדִ֥ישׁ וְעַד־ קָמָ֖ה וְעַד־ כֶּ֥רֶם זָֽיִת׃ (כח) וַיֹּאמְר֣וּ פְלִשְׁתִּים֮ מִ֣י עָ֣שָׂה זֹאת֒ וַיֹּאמְר֗וּ שִׁמְשׁוֹן֙ חֲתַ֣ן הַתִּמְנִ֔י כִּ֚י לָקַ֣ח אֶת־ אִשְׁתּ֔וֹ וַֽיִּתְּנָ֖הּ לְמֵרֵעֵ֑הוּ וַיַּעֲל֣וּ פְלִשְׁתִּ֔ים וַיִּשְׂרְפ֥וּ אוֹתָ֛הּ וְאֶת־ אָבִ֖יהָ בָּאֵֽשׁ׃ (כט) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן אִֽם־ תַּעֲשׂ֖וּן כָּזֹ֑את כִּ֛י אִם־ נִקַּ֥מְתִּי בָכֶ֖ם וְאַחַ֥ר אֶחְדָּֽל׃ (ל) וַיַּ֨ךְ אוֹתָ֥ם שׁ֛וֹק עַל־ יָרֵ֖ךְ מַכָּ֣ה גְדוֹלָ֑ה וַיֵּ֣רֶד וַיֵּ֔שֶׁב בִּסְעִ֖יף סֶ֥לַע עֵיטָֽם׃ (לא) וַיַּעֲל֣וּ פְלִשְׁתִּ֔ים וַֽיַּחֲנ֖וּ בִּיהוּדָ֑ה וַיִּנָּטְשׁ֖וּ בַּלֶּֽחִי׃ (לב) וַיֹּֽאמְרוּ֙ אִ֣ישׁ יְהוּדָ֔ה לָמָ֖ה עֲלִיתֶ֣ם עָלֵ֑ינוּ וַיֹּאמְר֗וּ לֶאֱס֤וֹר אֶת־ שִׁמְשׁוֹן֙ עָלִ֔ינוּ לַעֲשׂ֣וֹת ל֔וֹ כַּאֲשֶׁ֖ר עָ֥שָׂה לָֽנוּ׃ (לג) וַיֵּרְד֡וּ שְׁלֹשֶׁת֩ אֲלָפִ֨ים אִ֜ישׁ מִֽיהוּדָ֗ה אֶל־ סְעִיף֮ סֶ֣לַע עֵיטָם֒ וַיֹּאמְר֣וּ לְשִׁמְשׁ֗וֹן הֲלֹ֤א יָדַ֙עְתָּ֙ כִּֽי־ מֹשְׁלִ֥ים בָּ֙נוּ֙ פְּלִשְׁתִּ֔ים וּמַה־ זֹּ֖את עָשִׂ֣יתָ לָּ֑נוּ וַיֹּ֣אמֶר לָהֶ֔ם כַּאֲשֶׁר֙ עָ֣שׂוּ לִ֔י כֵּ֖ן עָשִׂ֥יתִי לָהֶֽם׃ (לד) וַיֹּ֤אמְרוּ לוֹ֙ לֶאֱסָרְךָ֣ יָרַ֔דְנוּ לְתִתְּךָ֖ בְּיַד־ פְּלִשְׁתִּ֑ים וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן הִשָּׁבְע֣וּ לִ֔י פֶּֽן־ תִּפְגְּע֥וּן בִּ֖י אַתֶּֽם׃ (לה) וַיֹּ֧אמְרוּ ל֣וֹ לֵאמֹ֗ר לֹ֚א כִּֽי־ אָסֹ֤ר נֶֽאֱסָרְךָ֙ וּנְתַנּ֣וּךָ בְיָדָ֔ם וְהָמֵ֖ת לֹ֣א נְמִיתֶ֑ךָ וַיַּאַסְרֻ֗הוּ בִּשְׁנַ֙יִם֙ עֲבֹתִ֣ים חֲדָשִׁ֔ים וַֽיַּעֲל֖וּהוּ מִן־ הַסָּֽלַע׃ (לו) הוּא־ בָ֣א עַד־ לֶ֔חִי וּפִלְשִׁתִּ֖ים הֵרִ֣יעוּ לִקְרָאת֑וֹ וַתִּצְלַ֨ח עָלָ֜יו ר֣וּחַ יְהוָ֗ה וַתִּהְיֶ֨ינָה הָעֲבֹתִ֜ים אֲשֶׁ֣ר עַל־ זְרוֹעוֹתָ֗יו כַּפִּשְׁתִּים֙ אֲשֶׁ֣ר בָּעֲר֣וּ בָאֵ֔שׁ וַיִּמַּ֥סּוּ אֱסוּרָ֖יו מֵעַ֥ל יָדָֽיו׃ (לז) וַיִּמְצָ֥א לְחִֽי־ חֲמ֖וֹר טְרִיָּ֑ה וַיִּשְׁלַ֤ח יָדוֹ֙ וַיִּקָּחֶ֔הָ וַיַּךְ־ בָּ֖הּ אֶ֥לֶף אִֽישׁ׃ (לח) וַיֹּ֣אמֶר שִׁמְשׁ֔וֹן בִּלְחִ֣י הַחֲמ֔וֹר חֲמ֖וֹר חֲמֹרָתָ֑יִם בִּלְחִ֣י הַחֲמ֔וֹר הִכֵּ֖יתִי אֶ֥לֶף אִֽישׁ׃ (לט) וַֽיְהִי֙ כְּכַלֹּת֣וֹ לְדַבֵּ֔ר וַיַּשְׁלֵ֥ךְ הַלְּחִ֖י מִיָּד֑וֹ וַיִּקְרָ֛א לַמָּק֥וֹם הַה֖וּא רָ֥מַת לֶֽחִי׃ (מ) וַיִּצְמָא֮ מְאֹד֒ וַיִּקְרָ֤א אֶל־ יְהוָה֙ וַיֹּאמַ֔ר אַתָּה֙ נָתַ֣תָּ בְיַֽד־ עַבְדְּךָ֔ אֶת־ הַתְּשׁוּעָ֥ה הַגְּדֹלָ֖ה הַזֹּ֑את וְעַתָּה֙ אָמ֣וּת בַּצָּמָ֔א וְנָפַלְתִּ֖י בְּיַ֥ד הָעֲרֵלִֽים׃ (מא) וַיִּבְקַ֨ע אֱלֹהִ֜ים אֶת־ הַמַּכְתֵּ֣שׁ אֲשֶׁר־ בַּלֶּ֗חִי וַיֵּצְא֨וּ מִמֶּ֤נּוּ מַ֙יִם֙ וַיֵּ֔שְׁתְּ וַתָּ֥שָׁב רוּח֖וֹ וַיֶּ֑חִי עַל־ כֵּ֣ן ׀ קָרָ֣א שְׁמָ֗הּ עֵ֤ין הַקּוֹרֵא֙ אֲשֶׁ֣ר בַּלֶּ֔חִי עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (מב) וַיִּשְׁפֹּ֧ט אֶת־ יִשְׂרָאֵ֛ל בִּימֵ֥י פְלִשְׁתִּ֖ים עֶשְׂרִ֥ים שָׁנָֽה׃ תחילת פרק טז (מג) וַיֵּ֥לֶךְ שִׁמְשׁ֖וֹן עַזָּ֑תָה וַיַּרְא־ שָׁם֙ אִשָּׁ֣ה זוֹנָ֔ה וַיָּבֹ֖א אֵלֶֽיהָ׃ (מד) לַֽעַזָּתִ֣ים ׀ לֵאמֹ֗ר בָּ֤א שִׁמְשׁוֹן֙ הֵ֔נָּה וַיָּסֹ֛בּוּ וַיֶּאֶרְבוּ־ ל֥וֹ כָל־ הַלַּ֖יְלָה בְּשַׁ֣עַר הָעִ֑יר וַיִּתְחָרְשׁ֤וּ כָל־ הַלַּ֙יְלָה֙ לֵאמֹ֔ר עַד־ א֥וֹר הַבֹּ֖קֶר (א) וַיִּשְׁכַּ֣ב שִׁמְשׁוֹן֮ עַד־ חֲצִ֣י הַלַּיְלָה֒ וַיָּ֣קָם ׀ בַּחֲצִ֣י הַלַּ֗יְלָה וַיֶּאֱחֹ֞ז בְּדַלְת֤וֹת שַֽׁעַר־ הָעִיר֙ וּבִשְׁתֵּ֣י הַמְּזוּז֔וֹת וַיִּסָּעֵם֙ עִֽם־ הַבְּרִ֔יחַ וַיָּ֖שֶׂם עַל־ כְּתֵפָ֑יו וַֽיַּעֲלֵם֙ אֶל־ רֹ֣אשׁ הָהָ֔ר אֲשֶׁ֖ר עַל־ פְּנֵ֥י חֶבְרֽוֹן׃ (ב) וַֽיְהִי֙ אַחֲרֵי־ כֵ֔ן וַיֶּאֱהַ֥ב אִשָּׁ֖ה בְּנַ֣חַל שֹׂרֵ֑ק וּשְׁמָ֖הּ דְּלִילָֽה׃ (ג) וַיַּעֲל֨וּ אֵלֶ֜יהָ סַרְנֵ֣י פְלִשְׁתִּ֗ים וַיֹּ֨אמְרוּ לָ֜הּ פַּתִּ֣י אוֹת֗וֹ וּרְאִי֙ בַּמֶּה֙ כֹּח֣וֹ גָד֔וֹל וּבַמֶּה֙ נ֣וּכַל ל֔וֹ וַאֲסַרְנֻ֖הוּ לְעַנֹּת֑וֹ וַאֲנַ֙חְנוּ֙ נִתַּן־ לָ֔ךְ אִ֕ישׁ אֶ֥לֶף וּמֵאָ֖ה כָּֽסֶף׃ (ד) וַתֹּ֤אמֶר דְּלִילָה֙ אֶל־ שִׁמְשׁ֔וֹן הַגִּֽידָה־ נָּ֣א לִ֔י בַּמֶּ֖ה כֹּחֲךָ֣ גָד֑וֹל וּבַמֶּ֥ה תֵאָסֵ֖ר לְעַנּוֹתֶֽךָ׃ (ה) וַיֹּ֤אמֶר אֵלֶ֙יהָ֙ שִׁמְשׁ֔וֹן אִם־ יַאַסְרֻ֗נִי בְּשִׁבְעָ֛ה יְתָרִ֥ים לַחִ֖ים אֲשֶׁ֣ר לֹא־ חֹרָ֑בוּ וְחָלִ֥יתִי וְהָיִ֖יתִי כְּאַחַ֥ד הָאָדָֽם׃ (ו) וַיַּעֲלוּ־ לָ֞הּ סַרְנֵ֣י פְלִשְׁתִּ֗ים שִׁבְעָ֛ה יְתָרִ֥ים לַחִ֖ים אֲשֶׁ֣ר לֹא־ חֹרָ֑בוּ וַתַּאַסְרֵ֖הוּ בָּהֶֽם׃ (ז) וְהָאֹרֵ֗ב יֹשֵׁ֥ב לָהּ֙ בַּחֶ֔דֶר וַתֹּ֣אמֶר אֵלָ֔יו פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיְנַתֵּק֙ אֶת־ הַיְתָרִ֔ים כַּאֲשֶׁ֨ר יִנָּתֵ֤ק פְּתִֽיל־ הַנְּעֹ֙רֶת֙ בַּהֲרִיח֣וֹ אֵ֔שׁ וְלֹ֥א נוֹדַ֖ע כֹּחֽוֹ׃ (ח) וַתֹּ֤אמֶר דְּלִילָה֙ אֶל־ שִׁמְשׁ֔וֹן הִנֵּה֙ הֵתַ֣לְתָּ בִּ֔י וַתְּדַבֵּ֥ר אֵלַ֖י כְּזָבִ֑ים עַתָּה֙ הַגִּֽידָה־ נָּ֣א לִ֔י בַּמֶּ֖ה תֵּאָסֵֽר׃ (ט) וַיֹּ֣אמֶר אֵלֶ֔יהָ אִם־ אָס֤וֹר יַאַסְר֙וּנִי֙ בַּעֲבֹתִ֣ים חֲדָשִׁ֔ים אֲשֶׁ֛ר לֹֽא־ נַעֲשָׂ֥ה בָהֶ֖ם מְלָאכָ֑ה וְחָלִ֥יתִי וְהָיִ֖יתִי כְּאַחַ֥ד הָאָדָֽם׃ (י) וַתִּקַּ֣ח דְּלִילָה֩ עֲבֹתִ֨ים חֲדָשִׁ֜ים וַתַּאַסְרֵ֣הוּ בָהֶ֗ם וַתֹּ֤אמֶר אֵלָיו֙ פְּלִשְׁתִּ֤ים עָלֶ֙יךָ֙ שִׁמְשׁ֔וֹן וְהָאֹרֵ֖ב יֹשֵׁ֣ב בֶּחָ֑דֶר וַֽיְנַתְּקֵ֛ם מֵעַ֥ל זְרֹעֹתָ֖יו כַּחֽוּט׃ (יא) וַתֹּ֨אמֶר דְּלִילָ֜ה אֶל־ שִׁמְשׁ֗וֹן עַד־ הֵ֜נָּה הֵתַ֤לְתָּ בִּי֙ וַתְּדַבֵּ֤ר אֵלַי֙ כְּזָבִ֔ים הַגִּ֣ידָה לִּ֔י בַּמֶּ֖ה תֵּאָסֵ֑ר וַיֹּ֣אמֶר אֵלֶ֔יהָ אִם־ תַּאַרְגִ֗י אֶת־ שֶׁ֛בַע מַחְלְפ֥וֹת רֹאשִׁ֖י עִם־ הַמַּסָּֽכֶת׃ (יב) וַתִּתְקַע֙ בַּיָּתֵ֔ד וַתֹּ֣אמֶר אֵלָ֔יו פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיִּיקַץ֙ מִשְּׁנָת֔וֹ וַיִּסַּ֛ע אֶת־ הַיְתַ֥ד הָאֶ֖רֶג וְאֶת־ הַמַּסָּֽכֶת׃ (יג) וַתֹּ֣אמֶר אֵלָ֗יו אֵ֚יךְ תֹּאמַ֣ר אֲהַבְתִּ֔יךְ וְלִבְּךָ֖ אֵ֣ין אִתִּ֑י זֶ֣ה שָׁלֹ֤שׁ פְּעָמִים֙ הֵתַ֣לְתָּ בִּ֔י וְלֹא־ הִגַּ֣דְתָּ לִּ֔י בַּמֶּ֖ה כֹּחֲךָ֥ גָדֽוֹל׃ (יד) וַ֠יְהִי כִּֽי־ הֵצִ֨יקָה לּ֧וֹ בִדְבָרֶ֛יהָ כָּל־ הַיָּמִ֖ים וַתְּאַֽלֲצֵ֑הוּ וַתִּקְצַ֥ר נַפְשׁ֖וֹ לָמֽוּת׃ (טו) וַיַּגֶּד־ לָ֣הּ אֶת־ כָּל־ לִבּ֗וֹ וַיֹּ֤אמֶר לָהּ֙ מוֹרָה֙ לֹֽא־ עָלָ֣ה עַל־ רֹאשִׁ֔י כִּֽי־ נְזִ֧יר אֱלֹהִ֛ים אֲנִ֖י מִבֶּ֣טֶן אִמִּ֑י אִם־ גֻּלַּ֙חְתִּי֙ וְסָ֣ר מִמֶּ֣נִּי כֹחִ֔י וְחָלִ֥יתִי וְהָיִ֖יתִי כְּכָל־ הָאָדָֽם׃ (טז) וַתֵּ֣רֶא דְלִילָ֗ה כִּֽי־ הִגִּ֣יד לָהּ֮ אֶת־ כָּל־ לִבּוֹ֒ וַתִּשְׁלַ֡ח וַתִּקְרָא֩ לְסַרְנֵ֨י פְלִשְׁתִּ֤ים לֵאמֹר֙ עֲל֣וּ הַפַּ֔עַם כִּֽי־ הִגִּ֥יד (לה) לִ֖י אֶת־ כָּל־ לִבּ֑וֹ וְעָל֤וּ אֵלֶ֙יהָ֙ סַרְנֵ֣י פְלִשְׁתִּ֔ים וַיַּעֲל֥וּ הַכֶּ֖סֶף בְּיָדָֽם׃ (יז) וַתְּיַשְּׁנֵ֙הוּ֙ עַל־ בִּרְכֶּ֔יהָ וַתִּקְרָ֣א לָאִ֔ישׁ וַתְּגַלַּ֕ח אֶת־ שֶׁ֖בַע מַחְלְפ֣וֹת רֹאשׁ֑וֹ וַתָּ֙חֶל֙ לְעַנּוֹת֔וֹ וַיָּ֥סַר כֹּח֖וֹ מֵעָלָֽיו׃ (יח) וַתֹּ֕אמֶר פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיִּקַ֣ץ מִשְּׁנָת֗וֹ וַיֹּ֙אמֶר֙ אֵצֵ֞א כְּפַ֤עַם בְּפַ֙עַם֙ וְאִנָּעֵ֔ר וְהוּא֙ לֹ֣א יָדַ֔ע כִּ֥י יְהוָ֖ה סָ֥ר מֵעָלָֽיו׃ (יט) וַיֹּאחֲז֣וּהוּ פְלִשְׁתִּ֔ים וַֽיְנַקְּר֖וּ אֶת־ עֵינָ֑יו וַיּוֹרִ֨ידוּ אוֹת֜וֹ עַזָּ֗תָה וַיַּאַסְר֙וּהוּ֙ בַּֽנְחֻשְׁתַּ֔יִם וַיְהִ֥י טוֹחֵ֖ן בְּבֵ֥ית (האסירים) הָאֲסוּרִֽים׃ (כ) וַיָּ֧חֶל שְׂעַר־ רֹאשׁ֛וֹ לְצַמֵּ֖חַ כַּאֲשֶׁ֥ר גֻּלָּֽח׃ (כא) וְסַרְנֵ֣י פְלִשְׁתִּ֗ים נֶֽאֱסְפוּ֙ לִזְבֹּ֧חַ זֶֽבַח־ גָּד֛וֹל לְדָג֥וֹן אֱלֹהֵיהֶ֖ם וּלְשִׂמְחָ֑ה וַיֹּ֣אמְר֔וּ נָתַ֤ן אֱלֹהֵ֙ינוּ֙ בְּיָדֵ֔נוּ אֵ֖ת שִׁמְשׁ֥וֹן אוֹיְבֵֽינוּ׃ (כב) וַיִּרְא֤וּ אֹתוֹ֙ הָעָ֔ם וַֽיְהַלְל֖וּ אֶת־ אֱלֹהֵיהֶ֑ם כִּ֣י אָמְר֗וּ נָתַ֨ן אֱלֹהֵ֤ינוּ בְיָדֵ֙נוּ֙ אֶת־ א֣וֹיְבֵ֔נוּ וְאֵת֙ מַחֲרִ֣יב אַרְצֵ֔נוּ וַאֲשֶׁ֥ר הִרְבָּ֖ה אֶת־ חֲלָלֵֽינוּ׃ (כג) וַֽיְהִי֙ (כי טוב) כְּט֣וֹב לִבָּ֔ם וַיֹּ֣אמְר֔וּ קִרְא֥וּ לְשִׁמְשׁ֖וֹן וִישַֽׂחֶק־ לָ֑נוּ וַיִּקְרְא֨וּ לְשִׁמְשׁ֜וֹן מִבֵּ֣ית (האסירים) הָאֲסוּרִ֗ים וַיְצַחֵק֙ לִפְנֵיהֶ֔ם וַיַּעֲמִ֥ידוּ אוֹת֖וֹ בֵּ֥ין הָעַמּוּדִֽים׃ (כד) וַיֹּ֨אמֶר שִׁמְשׁ֜וֹן אֶל־ הַנַּ֨עַר הַמַּחֲזִ֣יק בְּיָדוֹ֮ הַנִּ֣יחָה אוֹתִי֒ (והימשני) וַהֲמִשֵׁ֙נִי֙ אֶת־ הָֽעַמֻּדִ֔ים אֲשֶׁ֥ר הַבַּ֖יִת נָכ֣וֹן עֲלֵיהֶ֑ם וְאֶשָּׁעֵ֖ן עֲלֵיהֶֽם׃ (כה) וְהַבַּ֗יִת מָלֵ֤א הָֽאֲנָשִׁים֙ וְהַנָּשִׁ֔ים וְשָׁ֕מָּה כֹּ֖ל סַרְנֵ֣י פְלִשְׁתִּ֑ים וְעַל־ הַגָּ֗ג כִּשְׁלֹ֤שֶׁת אֲלָפִים֙ אִ֣ישׁ וְאִשָּׁ֔ה הָרֹאִ֖ים בִּשְׂח֥וֹק שִׁמְשֽׁוֹן׃ (כו) וַיִּקְרָ֥א שִׁמְשׁ֛וֹן אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר אֲדֹנָ֣י יֱהֹוִ֡ה זָכְרֵ֣נִי נָא֩ וְחַזְּקֵ֨נִי נָ֜א אַ֣ךְ הַפַּ֤עַם הַזֶּה֙ הָאֱלֹהִ֔ים וְאִנָּקְמָ֧ה נְקַם־ אַחַ֛ת מִשְּׁתֵ֥י עֵינַ֖י מִפְּלִשְׁתִּֽים׃ (כז) וַיִּלְפֹּ֨ת שִׁמְשׁ֜וֹן אֶת־ שְׁנֵ֣י ׀ עַמּוּדֵ֣י הַתָּ֗וֶךְ אֲשֶׁ֤ר הַבַּ֙יִת֙ נָכ֣וֹן עֲלֵיהֶ֔ם וַיִּסָּמֵ֖ךְ עֲלֵיהֶ֑ם אֶחָ֥ד בִּימִינ֖וֹ וְאֶחָ֥ד בִּשְׂמֹאלֽוֹ׃ (כח) וַיֹּ֣אמֶר שִׁמְשׁ֗וֹן תָּמ֣וֹת נַפְשִׁי֮ עִם־ פְּלִשְׁתִּים֒ וַיֵּ֣ט בְּכֹ֔חַ וַיִּפֹּ֤ל הַבַּ֙יִת֙ עַל־ הַסְּרָנִ֔ים וְעַל־ כָּל־ הָעָ֖ם אֲשֶׁר־ בּ֑וֹ וַיִּהְי֤וּ הַמֵּתִים֙ אֲשֶׁ֣ר הֵמִ֣ית בְּמוֹת֔וֹ רַבִּ֕ים מֵאֲשֶׁ֥ר הֵמִ֖ית בְּחַיָּֽיו׃ (כט) וַיֵּרְד֨וּ אֶחָ֜יו וְכָל־ בֵּ֣ית אָבִיהוּ֮ וַיִּשְׂא֣וּ אֹתוֹ֒ וַֽיַּעֲל֣וּ ׀ וַיִּקְבְּר֣וּ אוֹת֗וֹ בֵּ֤ין צָרְעָה֙ וּבֵ֣ין אֶשְׁתָּאֹ֔ל בְּקֶ֖בֶר מָנ֣וֹחַ אָבִ֑יו וְה֛וּא שָׁפַ֥ט אֶת־ יִשְׂרָאֵ֖ל עֶשְׂרִ֥ים שָׁנָֽה׃ סוף פרק טז (ל) וַֽיְהִי־ אִ֥ישׁ מֵֽהַר־ אֶפְרָ֖יִם וּשְׁמ֥וֹ מִיכָֽיְהוּ׃ (לא) וַיֹּ֣אמֶר לְאִמּ֡וֹ אֶלֶף֩ וּמֵאָ֨ה הַכֶּ֜סֶף אֲשֶׁ֣ר לֻֽקַּֽח־ לָ֗ךְ (ואתי) וְאַ֤תְּ אָלִית֙ וְגַם֙ אָמַ֣רְתְּ בְּאָזְנַ֔י הִנֵּֽה־ הַכֶּ֥סֶף אִתִּ֖י אֲנִ֣י לְקַחְתִּ֑יו וַתֹּ֣אמֶר אִמּ֔וֹ בָּר֥וּךְ בְּנִ֖י לַיהוָֽה׃ (לב) וַיָּ֛שֶׁב אֶת־ אֶֽלֶף־ וּמֵאָ֥ה הַכֶּ֖סֶף לְאִמּ֑וֹ וַתֹּ֣אמֶר אִמּ֡וֹ הַקְדֵּ֣שׁ הִקְדַּ֣שְׁתִּי אֶת־ הַכֶּסֶף֩ לַיהוָ֨ה מִיָּדִ֜י לִבְנִ֗י לַֽעֲשׂוֹת֙ פֶּ֣סֶל וּמַסֵּכָ֔ה וְעַתָּ֖ה אֲשִׁיבֶ֥נּוּ לָֽךְ׃ (לג) וַיָּ֥שֶׁב אֶת־ הַכֶּ֖סֶף לְאִמּ֑וֹ וַתִּקַּ֣ח אִמּוֹ֩ מָאתַ֨יִם כֶּ֜סֶף וַתִּתְּנֵ֣הוּ לַצּוֹרֵ֗ף וַֽיַּעֲשֵׂ֙הוּ֙ פֶּ֣סֶל וּמַסֵּכָ֔ה וַיְהִ֖י בְּבֵ֥ית מִיכָֽיְהוּ׃ (לד) וְהָאִ֣ישׁ מִיכָ֔ה ל֖וֹ בֵּ֣ית אֱלֹהִ֑ים וַיַּ֤עַשׂ אֵפוֹד֙ וּתְרָפִ֔ים וַיְמַלֵּ֗א אֶת־ יַ֤ד אַחַד֙ מִבָּנָ֔יו וַיְהִי־ ל֖וֹ לְכֹהֵֽן׃ (לה) בַּיָּמִ֣ים הָהֵ֔ם אֵ֥ין מֶ֖לֶךְ בְּיִשְׂרָאֵ֑ל אִ֛ישׁ הַיָּשָׁ֥ר בְּעֵינָ֖יו יַעֲשֶֽׂה׃ (לו) וַיְהִי־ נַ֗עַר מִבֵּ֥ית לֶ֙חֶם֙ יְהוּדָ֔ה מִמִּשְׁפַּ֖חַת יְהוּדָ֑ה וְה֥וּא לֵוִ֖י וְה֥וּא גָֽר־ שָֽׁם׃ (לז) וַיֵּ֨לֶךְ הָאִ֜ישׁ מֵהָעִ֗יר מִבֵּ֥ית לֶ֙חֶם֙ יְהוּדָ֔ה לָג֖וּר בַּאֲשֶׁ֣ר יִמְצָ֑א וַיָּבֹ֧א הַר־ אֶפְרַ֛יִם עַד־ בֵּ֥ית מִיכָ֖ה לַעֲשׂ֥וֹת דַּרְכּֽוֹ׃ (לח) וַיֹּאמֶר־ ל֥וֹ מִיכָ֖ה מֵאַ֣יִן תָּב֑וֹא וַיֹּ֨אמֶר אֵלָ֜יו לֵוִ֣י אָנֹ֗כִי מִבֵּ֥ית לֶ֙חֶם֙ יְהוּדָ֔ה וְאָנֹכִ֣י הֹלֵ֔ךְ לָג֖וּר בַּאֲשֶׁ֥ר אֶמְצָֽא׃ (לט) וַיֹּאמֶר֩ ל֨וֹ מִיכָ֜ה שְׁבָ֣ה עִמָּדִ֗י וֶֽהְיֵה־ לִי֮ לְאָ֣ב וּלְכֹהֵן֒ וְאָנֹכִ֨י אֶֽתֶּן־ לְךָ֜ עֲשֶׂ֤רֶת כֶּ֙סֶף֙ לַיָּמִ֔ים וְעֵ֥רֶךְ בְּגָדִ֖ים וּמִחְיָתֶ֑ךָ וַיֵּ֖לֶךְ הַלֵּוִֽי׃ (מ) וַיּ֥וֹאֶל הַלֵּוִ֖י לָשֶׁ֣בֶת אֶת־ הָאִ֑ישׁ וַיְהִ֤י הַנַּ֙עַר֙ ל֔וֹ כְּאַחַ֖ד מִבָּנָֽיו׃ (מא) וַיְמַלֵּ֤א מִיכָה֙ אֶת־ יַ֣ד הַלֵּוִ֔י וַיְהִי־ ל֥וֹ הַנַּ֖עַר לְכֹהֵ֑ן וַיְהִ֖י בְּבֵ֥ית מִיכָֽה׃ (מב) וַיֹּ֣אמֶר מִיכָ֔ה עַתָּ֣ה יָדַ֔עְתִּי כִּֽי־ יֵיטִ֥יב יְהוָ֖ה לִ֑י כִּ֧י הָיָה־ לִ֛י הַלֵּוִ֖י לְכֹהֵֽן׃ (מג) בַּיָּמִ֣ים הָהֵ֔ם אֵ֥ין מֶ֖לֶךְ בְּיִשְׂרָאֵ֑ל וּבַיָּמִ֣ים הָהֵ֗ם שֵׁ֣בֶט הַדָּנִ֞י מְבַקֶּשׁ־ ל֤וֹ נַֽחֲלָה֙ לָשֶׁ֔בֶת כִּי֩ לֹֽא־ נָ֨פְלָה לּ֜וֹ עַד־ הַיּ֥וֹם הַה֛וּא בְּתוֹךְ־ שִׁבְטֵ֥י יִשְׂרָאֵ֖ל בְּנַחֲלָֽה׃ (מד) וַיִּשְׁלְח֣וּ בְנֵי־ דָ֣ן ׀ מִֽמִּשְׁפַּחְתָּ֡ם חֲמִשָּׁ֣ה אֲנָשִׁ֣ים מִקְצוֹתָם֩ אֲנָשִׁ֨ים בְּנֵי־ חַ֜יִל מִצָּרְעָ֣ה וּמֵֽאֶשְׁתָּאֹ֗ל לְרַגֵּ֤ל אֶת־ הָאָ֙רֶץ֙ וּלְחָקְרָ֔הּ וַיֹּאמְר֣וּ אֲלֵהֶ֔ם לְכ֖וּ חִקְר֣וּ אֶת־ הָאָ֑רֶץ וַיָּבֹ֤אוּ הַר־ אֶפְרַ֙יִם֙ עַד־ בֵּ֣ית מִיכָ֔ה וַיָּלִ֖ינוּ שָֽׁם׃ (מה) הֵ֚מָּה עִם־ בֵּ֣ית מִיכָ֔ה וְהֵ֣מָּה הִכִּ֔ירוּ אֶת־ ק֥וֹל הַנַּ֖עַר הַלֵּוִ֑י וַיָּס֣וּרוּ שָׁ֗ם וַיֹּ֤אמְרוּ לוֹ֙ מִֽי־ הֱבִיאֲךָ֣ הֲלֹ֔ם וּמָֽה־ אַתָּ֥ה עֹשֶׂ֛ה בָּזֶ֖ה וּמַה־ לְּךָ֥ פֹֽה׃ (מו) וַיֹּ֣אמֶר אֲלֵהֶ֔ם כָּזֹ֣ה וְכָזֶ֔ה עָ֥שָׂה לִ֖י מִיכָ֑ה וַיִּשְׂכְּרֵ֕נִי וָאֱהִי־ ל֖וֹ לְכֹהֵֽן׃ (מז) וַיֹּ֥אמְרוּ ל֖וֹ שְׁאַל־ נָ֣א בֵאלֹהִ֑ים וְנֵ֣דְעָ֔ה הֲתַצְלִ֣יחַ דַּרְכֵּ֔נוּ אֲשֶׁ֥ר אֲנַ֖חְנוּ הֹלְכִ֥ים עָלֶֽיהָ׃
