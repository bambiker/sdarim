שופטים פרק ט לפי החלוקה לסדרים

פרק ט בספר שופטים (57 פסוקים) נמצא בשופטים סדר ו (פסוקים לד–לט) ובשופטים סדר ז (פסוקים א–נא).

שופטים סדר ו

(א) בְּיֶדְכֶם֩ נָתַ֨ן אֱלֹהִ֜ים אֶת־ שָׂרֵ֤י מִדְיָן֙ אֶת־ עֹרֵ֣ב וְאֶת־ זְאֵ֔ב וּמַה־ יָּכֹ֖לְתִּי עֲשׂ֣וֹת כָּכֶ֑ם אָ֗ז רָפְתָ֤ה רוּחָם֙ מֵֽעָלָ֔יו בְּדַבְּר֖וֹ הַדָּבָ֥ר הַזֶּֽה׃ (ב) וַיָּבֹ֥א גִדְע֖וֹן הַיַּרְדֵּ֑נָה עֹבֵ֣ר ה֗וּא וּשְׁלֹשׁ־ מֵא֤וֹת הָאִישׁ֙ אֲשֶׁ֣ר אִתּ֔וֹ עֲיֵפִ֖ים וְרֹדְפִֽים׃ (ג) וַיֹּ֙אמֶר֙ לְאַנְשֵׁ֣י סֻכּ֔וֹת תְּנוּ־ נָא֙ כִּכְּר֣וֹת לֶ֔חֶם לָעָ֖ם אֲשֶׁ֣ר בְּרַגְלָ֑י כִּי־ עֲיֵפִ֣ים הֵ֔ם וְאָנֹכִ֗י רֹדֵ֛ף אַחֲרֵ֛י זֶ֥בַח וְצַלְמֻנָּ֖ע מַלְכֵ֥י מִדְיָֽן׃ (ד) וַיֹּ֙אמֶר֙ שָׂרֵ֣י סֻכּ֔וֹת הֲ֠כַף זֶ֧בַח וְצַלְמֻנָּ֛ע עַתָּ֖ה בְּיָדֶ֑ךָ כִּֽי־ נִתֵּ֥ן לִֽצְבָאֲךָ֖ לָֽחֶם׃ (ה) וַיֹּ֣אמֶר גִּדְע֔וֹן לָכֵ֗ן בְּתֵ֧ת יְהוָ֛ה אֶת־ זֶ֥בַח וְאֶת־ צַלְמֻנָּ֖ע בְּיָדִ֑י וְדַשְׁתִּי֙ אֶת־ בְּשַׂרְכֶ֔ם אֶת־ קוֹצֵ֥י הַמִּדְבָּ֖ר וְאֶת־ הַֽבַּרְקֳנִֽים׃ (ו) וַיַּ֤עַל מִשָּׁם֙ פְּנוּאֵ֔ל וַיְדַבֵּ֥ר אֲלֵיהֶ֖ם כָּזֹ֑את וַיַּעֲנ֤וּ אוֹתוֹ֙ אַנְשֵׁ֣י פְנוּאֵ֔ל כַּאֲשֶׁ֥ר עָנ֖וּ אַנְשֵׁ֥י סֻכּֽוֹת׃ (ז) וַיֹּ֛אמֶר גַּם־ לְאַנְשֵׁ֥י פְנוּאֵ֖ל לֵאמֹ֑ר בְּשׁוּבִ֣י בְשָׁל֔וֹם אֶתֹּ֖ץ אֶת־ הַמִּגְדָּ֥ל הַזֶּֽה׃ (ח) וְזֶ֨בַח וְצַלְמֻנָּ֜ע בַּקַּרְקֹ֗ר וּמַחֲנֵיהֶ֤ם עִמָּם֙ כַּחֲמֵ֤שֶׁת עָשָׂר֙ אֶ֔לֶף כֹּ֚ל הַנּ֣וֹתָרִ֔ים מִכֹּ֖ל מַחֲנֵ֣ה בְנֵי־ קֶ֑דֶם וְהַנֹּ֣פְלִ֔ים מֵאָ֨ה וְעֶשְׂרִ֥ים אֶ֛לֶף אִ֖ישׁ שֹׁ֥לֵֽף חָֽרֶב׃ (ט) וַיַּ֣עַל גִּדְע֗וֹן דֶּ֚רֶךְ הַשְּׁכוּנֵ֣י בָֽאֳהָלִ֔ים מִקֶּ֥דֶם לְנֹ֖בַח וְיָגְבֳּהָ֑ה וַיַּךְ֙ אֶת־ הַֽמַּחֲנֶ֔ה וְהַֽמַּחֲנֶ֖ה הָ֥יָה בֶֽטַח׃ (י) וַיָּנ֗וּסוּ זֶ֚בַח וְצַלְמֻנָּ֔ע וַיִּרְדֹּ֖ף אַחֲרֵיהֶ֑ם וַיִּלְכֹּ֞ד אֶת־ שְׁנֵ֣י ׀ מַלְכֵ֣י מִדְיָ֗ן אֶת־ זֶ֙בַח֙ וְאֶת־ צַלְמֻנָּ֔ע וְכָל־ הַֽמַּחֲנֶ֖ה הֶחֱרִֽיד׃ (יא) וַיָּ֛שָׁב גִּדְע֥וֹן בֶּן־ יוֹאָ֖שׁ מִן־ הַמִּלְחָמָ֑ה מִֽלְמַעֲלֵ֖ה הֶחָֽרֶס׃ (יב) וַיִּלְכָּד־ נַ֛עַר מֵאַנְשֵׁ֥י סֻכּ֖וֹת וַיִּשְׁאָלֵ֑הוּ וַיִּכְתֹּ֨ב אֵלָ֜יו אֶת־ שָׂרֵ֤י סֻכּוֹת֙ וְאֶת־ זְקֵנֶ֔יהָ שִׁבְעִ֥ים וְשִׁבְעָ֖ה אִֽישׁ׃ (יג) וַיָּבֹא֙ אֶל־ אַנְשֵׁ֣י סֻכּ֔וֹת וַיֹּ֕אמֶר הִנֵּ֖ה זֶ֣בַח וְצַלְמֻנָּ֑ע אֲשֶׁר֩ חֵרַפְתֶּ֨ם אוֹתִ֜י לֵאמֹ֗ר הֲ֠כַף זֶ֣בַח וְצַלְמֻנָּ֤ע עַתָּה֙ בְּיָדֶ֔ךָ כִּ֥י נִתֵּ֛ן לַאֲנָשֶׁ֥יךָ הַיְּעֵפִ֖ים לָֽחֶם׃ (יד) וַיִּקַּח֙ אֶת־ זִקְנֵ֣י הָעִ֔יר וְאֶת־ קוֹצֵ֥י הַמִּדְבָּ֖ר וְאֶת־ הַֽבַּרְקֳנִ֑ים וַיֹּ֣דַע בָּהֶ֔ם אֵ֖ת אַנְשֵׁ֥י סֻכּֽוֹת׃ (טו) וְאֶת־ מִגְדַּ֥ל פְּנוּאֵ֖ל נָתָ֑ץ וַֽיַּהֲרֹ֖ג אֶת־ אַנְשֵׁ֥י הָעִֽיר׃ (טז) וַיֹּ֗אמֶר אֶל־ זֶ֙בַח֙ וְאֶל־ צַלְמֻנָּ֔ע אֵיפֹה֙ הָאֲנָשִׁ֔ים אֲשֶׁ֥ר הֲרַגְתֶּ֖ם בְּתָב֑וֹר וַֽיֹּאמרוּ֙ כָּמ֣וֹךָ כְמוֹהֶ֔ם אֶחָ֕ד כְּתֹ֖אַר בְּנֵ֥י הַמֶּֽלֶךְ׃ (יז) וַיֹּאמַ֕ר אַחַ֥י בְּנֵֽי־ אִמִּ֖י הֵ֑ם חַי־ יְהוָ֗ה ל֚וּ הַחֲיִתֶ֣ם אוֹתָ֔ם לֹ֥א הָרַ֖גְתִּי אֶתְכֶֽם׃ (יח) וַיֹּ֙אמֶר֙ לְיֶ֣תֶר בְּכוֹר֔וֹ ק֖וּם הֲרֹ֣ג אוֹתָ֑ם וְלֹא־ שָׁלַ֨ף הַנַּ֤עַר חַרְבּוֹ֙ כִּ֣י יָרֵ֔א כִּ֥י עוֹדֶ֖נּוּ נָֽעַר׃ (יט) וַיֹּ֜אמֶר זֶ֣בַח וְצַלְמֻנָּ֗ע ק֤וּם אַתָּה֙ וּפְגַע־ בָּ֔נוּ כִּ֥י כָאִ֖ישׁ גְּבוּרָת֑וֹ וַיָּ֣קָם גִּדְע֗וֹן וַֽיַּהֲרֹג֙ אֶת־ זֶ֣בַח וְאֶת־ צַלְמֻנָּ֔ע וַיִּקַּח֙ אֶת־ הַשַּׂ֣הֲרֹנִ֔ים אֲשֶׁ֖ר בְּצַוְּארֵ֥י גְמַלֵּיהֶֽם׃ (כ) וַיֹּאמְר֤וּ אִֽישׁ־ יִשְׂרָאֵל֙ אֶל־ גִּדְע֔וֹן מְשָׁל־ בָּ֙נוּ֙ גַּם־ אַתָּ֔ה גַּם־ בִּנְךָ֖ גַּ֣ם בֶּן־ בְּנֶ֑ךָ כִּ֥י הוֹשַׁעְתָּ֖נוּ מִיַּ֥ד מִדְיָֽן׃ (כא) וַיֹּ֤אמֶר אֲלֵהֶם֙ גִּדְע֔וֹן לֹֽא־ אֶמְשֹׁ֤ל אֲנִי֙ בָּכֶ֔ם וְלֹֽא־ יִמְשֹׁ֥ל בְּנִ֖י בָּכֶ֑ם יְהוָ֖ה יִמְשֹׁ֥ל בָּכֶֽם׃ (כב) וַיֹּ֨אמֶר אֲלֵהֶ֜ם גִּדְע֗וֹן אֶשְׁאֲלָ֤ה מִכֶּם֙ שְׁאֵלָ֔ה וּתְנוּ־ לִ֕י אִ֖ישׁ נֶ֣זֶם שְׁלָל֑וֹ כִּֽי־ נִזְמֵ֤י זָהָב֙ לָהֶ֔ם כִּ֥י יִשְׁמְעֵאלִ֖ים הֵֽם׃ (כג) וַיֹּאמְר֖וּ נָת֣וֹן נִתֵּ֑ן וַֽיִּפְרְשׂוּ֙ אֶת־ הַשִּׂמְלָ֔ה וַיַּשְׁלִ֣יכוּ שָׁ֔מָּה אִ֖ישׁ נֶ֥זֶם שְׁלָלֽוֹ׃ (כד) וַיְהִ֗י מִשְׁקַ֞ל נִזְמֵ֤י הַזָּהָב֙ אֲשֶׁ֣ר שָׁאָ֔ל אֶ֥לֶף וּשְׁבַע־ מֵא֖וֹת זָהָ֑ב לְ֠בַד מִן־ הַשַּׂהֲרֹנִ֨ים וְהַנְּטִפ֜וֹת וּבִגְדֵ֣י הָאַרְגָּמָ֗ן שֶׁעַל֙ מַלְכֵ֣י מִדְיָ֔ן וּלְבַד֙ מִן־ הָ֣עֲנָק֔וֹת אֲשֶׁ֖ר בְּצַוְּארֵ֥י גְמַלֵּיהֶֽם׃ (כה) וַיַּעַשׂ֩ אוֹת֨וֹ גִדְע֜וֹן לְאֵפ֗וֹד וַיַּצֵּ֨ג אוֹת֤וֹ בְעִירוֹ֙ בְּעָפְרָ֔ה וַיִּזְנ֧וּ כָֽל־ יִשְׂרָאֵ֛ל אַחֲרָ֖יו שָׁ֑ם וַיְהִ֛י לְגִדְע֥וֹן וּלְבֵית֖וֹ לְמוֹקֵֽשׁ׃ (כו) וַיִּכָּנַ֣ע מִדְיָ֗ן לִפְנֵי֙ בְּנֵ֣י יִשְׂרָאֵ֔ל וְלֹ֥א יָסְפ֖וּ לָשֵׂ֣את רֹאשָׁ֑ם וַתִּשְׁקֹ֥ט הָאָ֛רֶץ אַרְבָּעִ֥ים שָׁנָ֖ה בִּימֵ֥י גִדְעֽוֹן׃ (כז) וַיֵּ֛לֶךְ יְרֻבַּ֥עַל בֶּן־ יוֹאָ֖שׁ וַיֵּ֥שֶׁב בְּבֵיתֽוֹ׃ (כח) וּלְגִדְע֗וֹן הָיוּ֙ שִׁבְעִ֣ים בָּנִ֔ים יֹצְאֵ֖י יְרֵכ֑וֹ כִּֽי־ נָשִׁ֥ים רַבּ֖וֹת הָ֥יוּ לֽוֹ׃ (כט) וּפִֽילַגְשׁוֹ֙ אֲשֶׁ֣ר בִּשְׁכֶ֔ם יָֽלְדָה־ לּ֥וֹ גַם־ הִ֖יא בֵּ֑ן וַיָּ֥שֶׂם אֶת־ שְׁמ֖וֹ אֲבִימֶֽלֶךְ׃ (ל) וַיָּ֛מָת גִּדְע֥וֹן בֶּן־ יוֹאָ֖שׁ בְּשֵׂיבָ֣ה טוֹבָ֑ה וַיִּקָּבֵ֗ר בְּקֶ֙בֶר֙ יוֹאָ֣שׁ אָבִ֔יו בְּעָפְרָ֖ה אֲבִ֥י הָֽעֶזְרִֽי׃ (לא) וַיְהִ֗י כַּֽאֲשֶׁר֙ מֵ֣ת גִּדְע֔וֹן וַיָּשׁ֙וּבוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל וַיִּזְנ֖וּ אַחֲרֵ֣י הַבְּעָלִ֑ים וַיָּשִׂ֧ימוּ לָהֶ֛ם בַּ֥עַל בְּרִ֖ית לֵאלֹהִֽים׃ (לב) וְלֹ֤א זָֽכְרוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶת־ יְהוָ֖ה אֱלֹהֵיהֶ֑ם הַמַּצִּ֥יל אוֹתָ֛ם מִיַּ֥ד כָּל־ אֹיְבֵיהֶ֖ם מִסָּבִֽיב׃ (לג) וְלֹֽא־ עָשׂ֣וּ חֶ֔סֶד עִם־ בֵּ֥ית יְרֻבַּ֖עַל גִּדְע֑וֹן כְּכָל־ הַטּוֹבָ֔ה אֲשֶׁ֥ר עָשָׂ֖ה עִם־ יִשְׂרָאֵֽל׃ תחילת פרק ט (לד) וַיֵּ֨לֶךְ אֲבִימֶ֤לֶךְ בֶּן־ יְרֻבַּ֙עַל֙ שְׁכֶ֔מָה אֶל־ אֲחֵ֖י אִמּ֑וֹ וַיְדַבֵּ֣ר אֲלֵיהֶ֔ם וְאֶל־ כָּל־ מִשְׁפַּ֛חַת בֵּית־ אֲבִ֥י אִמּ֖וֹ לֵאמֹֽר׃ (לה) דַּבְּרוּ־ נָ֞א בְּאָזְנֵ֨י כָל־ בַּעֲלֵ֣י שְׁכֶם֮ מַה־ טּ֣וֹב לָכֶם֒ הַמְשֹׁ֨ל בָּכֶ֜ם שִׁבְעִ֣ים אִ֗ישׁ כֹּ֚ל בְּנֵ֣י יְרֻבַּ֔עַל אִם־ מְשֹׁ֥ל בָּכֶ֖ם אִ֣ישׁ אֶחָ֑ד וּזְכַרְתֶּ֕ם כִּֽי־ עַצְמֵכֶם וּבְשַׂרְכֶ֖ם אָנִֽי׃ (לו) וַיְדַבְּר֨וּ אֲחֵֽי־ אִמּ֜וֹ עָלָ֗יו בְּאָזְנֵי֙ כָּל־ בַּעֲלֵ֣י שְׁכֶ֔ם אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיֵּ֤ט לִבָּם֙ אַחֲרֵ֣י אֲבִימֶ֔לֶךְ כִּ֥י אָמְר֖וּ אָחִ֥ינוּ הֽוּא׃ (לז) וַיִּתְּנוּ־ לוֹ֙ שִׁבְעִ֣ים כֶּ֔סֶף מִבֵּ֖יתּ בַ֣עַל בְּרִ֑ית וַיִּשְׂכֹּ֨ר בָּהֶ֜ם אֲבִימֶ֗לֶךְ אֲנָשִׁ֤ים רֵיקִים֙ וּפֹ֣חֲזִ֔ים וַיֵּלְכ֖וּ אַחֲרָֽיו׃ (לח) וַיָּבֹ֤א בֵית־ אָבִיו֙ עָפְרָ֔תָה וַֽיַּהֲרֹ֞ג אֶת־ אֶחָ֧יו בְּנֵֽי־ יְרֻבַּ֛עַל שִׁבְּעִ֥ים אִ֖ישׁ עַל־ אֶ֣בֶן אֶחָ֑ת שְׁכֶ֑ם וַיִּלָּ֖חֶם בַּאֲבִימֶֽלֶךְ׃ (לד) וַיִּרְדְּפֵ֣הוּ אֲבִימֶ֔לֶךְ וַיָּ֖נָס מִפָּנָ֑יו וַֽיִּפְּל֛וּ חֲלָלִ֥ים רַבִּ֖ים עַד־ פֶּ֥תַח הַשָּֽׁעַר׃ (לה) וַיֵּ֥שֶׁב אֲבִימֶ֖לֶךְ בָּארוּמָ֑ה וַיְגָ֧רֶשׁ זְבֻ֛ל אֶת־ גַּ֥עַל וְאֶת־ אֶחָ֖יו מִשֶּׁ֥בֶת בִּשְׁכֶֽם׃ (לו) וַֽיְהִי֙ מִֽמָּחֳרָ֔ת וַיֵּצֵ֥א הָעָ֖ם הַשָּׂדֶ֑ה וַיַּגִּ֖דוּ לַאֲבִימֶֽלֶךְ׃ (לז) וַיִּקַּ֣ח אֶת־ הָעָ֗ם וַֽיֶּחֱצֵם֙ לִשְׁלֹשָׁ֣ה רָאשִׁ֔ים וַיֶּאֱרֹ֖ב בַּשָּׂדֶ֑ה וַיַּ֗רְא וְהִנֵּ֤ה הָעָם֙ יֹצֵ֣א מִן־ הָעִ֔יר וַיָּ֥קָם עֲלֵיהֶ֖ם וַיַּכֵּֽם׃ (לח) וַאֲבִימֶ֗לֶךְ וְהָרָאשִׁים֙ אֲשֶׁ֣ר עִמּ֔וֹ פָּשְׁט֕וּ וַיַּ֣עַמְד֔וּ פֶּ֖תַח שַׁ֣עַר הָעִ֑יר וּשְׁנֵ֣י הָֽרָאשִׁ֗ים פָּֽשְׁט֛וּ עַֽל־ כָּל־ אֲשֶׁ֥ר בַּשָּׂדֶ֖ה וַיַּכּֽוּם׃ (לט) וַאֲבִימֶ֜לֶךְ נִלְחָ֣ם בָּעִ֗יר כֹּ֚ל הַיּ֣וֹם הַה֔וּא וַיִּלְכֹּד֙ אֶת־ הָעִ֔יר וְאֶת־ הָעָ֥ם אֲשֶׁר־ בָּ֖הּ הָרָ֑ג וַיִּתֹּץ֙ אֶת־ הָעִ֔יר וַיִּזְרָעֶ֖הָ מֶֽלַח׃ (מ) וַֽיִּשְׁמְע֔וּ כָּֽל־ בַּעֲלֵ֖י מִֽגְדַּל־ שְׁכֶ֑ם וַיָּבֹ֣אוּ אֶל־ צְרִ֔יחַ בֵּ֖ית אֵ֥ל בְּרִֽית׃ (מא) וַיֻּגַּ֖ד לַאֲבִימֶ֑לֶךְ כִּ֣י הִֽתְקַבְּצ֔וּ כָּֽל־ בַּעֲלֵ֖י מִֽגְדַּל־ שְׁכֶֽם׃ (מב) וַיַּ֨עַל אֲבִימֶ֜לֶךְ הַר־ צַלְמ֗וֹן הוּא֮ וְכָל־ הָעָ֣ם אֲשֶׁר־ אִתּוֹ֒ וַיִּקַּח֩ אֲבִימֶ֨לֶךְ אֶת־ הַקַּרְדֻּמּ֜וֹת בְּיָד֗וֹ וַיִּכְרֹת֙ שׂוֹכַ֣ת עֵצִ֔ים וַיִּ֨שָּׂאֶ֔הָ וַיָּ֖שֶׂם עַל־ שִׁכְמ֑וֹ וַיֹּ֜אמֶר אֶל־ הָעָ֣ם אֲשֶׁר־ עִמּ֗וֹ מָ֤ה רְאִיתֶם֙ עָשִׂ֔יתִי מַהֲר֖וּ עֲשׂ֥וּ כָמֽוֹנִי׃ (מג) וַיִּכְרְת֨וּ גַם־ כָּל־ הָעָ֜ם אִ֣ישׁ שׂוֹכֹ֗ה וַיֵּ֨לְכ֜וּ אַחֲרֵ֤י אֲבִימֶ֙לֶךְ֙ וַיָּשִׂ֣ימוּ עַֽל־ הַצְּרִ֔יחַ וַיַּצִּ֧יתוּ עֲלֵיהֶ֛ם אֶֽת־ הַצְּרִ֖יחַ בָּאֵ֑שׁ וַיָּמֻ֜תוּ גַּ֣ם כָּל־ אַנְשֵׁ֧י מִֽגְדַּל־ שְׁכֶ֛ם כְּאֶ֖לֶף אִ֥ישׁ וְאִשָּֽׁה׃ (מד) וַיֵּ֥לֶךְ אֲבִימֶ֖לֶךְ אֶל־ תֵּבֵ֑ץ וַיִּ֥חַן בְּתֵבֵ֖ץ וַֽיִּלְכְּדָֽהּ׃ (מה) וּמִגְדַּל־ עֹז֮ הָיָ֣ה בְתוֹךְ־ הָעִיר֒ וַיָּנֻ֨סוּ שָׁ֜מָּה כָּל־ הָאֲנָשִׁ֣ים וְהַנָּשִׁ֗ים וְכֹל֙ בַּעֲלֵ֣י הָעִ֔יר וַֽיִּסְגְּר֖וּ בַּעֲדָ֑ם וַֽיַּעֲל֖וּ עַל־ גַּ֥ג הַמִּגְדָּֽל׃ (מו) וַיָּבֹ֤א אֲבִימֶ֙לֶךְ֙ עַד־ הַמִּגְדָּ֔ל וַיִּלָּ֖חֶם בּ֑וֹ וַיִּגַּ֛שׁ עַד־ פֶּ֥תַח הַמִּגְדָּ֖ל לְשָׂרְפ֥וֹ בָאֵֽשׁ׃ (מז) וַתַּשְׁלֵ֞ךְ אִשָּׁ֥ה אַחַ֛ת פֶּ֥לַח רֶ֖כֶב עַל־ רֹ֣אשׁ אֲבִימֶ֑לֶךְ וַתָּ֖רִץ אֶת־ גֻּלְגָּלְתּֽוֹ׃ (מח) וַיִּקְרָ֨א מְהֵרָ֜ה אֶל־ הַנַּ֣עַר ׀ נֹשֵׂ֣א כֵלָ֗יו וַיֹּ֤אמֶר לוֹ֙ שְׁלֹ֤ף חַרְבְּךָ֙ וּמ֣וֹתְתֵ֔נִי פֶּן־ יֹ֥אמְרוּ לִ֖י אִשָּׁ֣ה הֲרָגָ֑תְהוּ וַיִּדְקְרֵ֥הוּ נַעֲר֖וֹ וַיָּמֹֽת׃ (מט) וַיִּרְא֥וּ אִֽישׁ־ יִשְׂרָאֵ֖ל כִּ֣י מֵ֣ת אֲבִימֶ֑לֶךְ וַיֵּלְכ֖וּ אִ֥ישׁ לִמְקֹמֽוֹ׃ (נ) וַיָּ֣שֶׁב אֱלֹהִ֔ים אֵ֖ת רָעַ֣ת אֲבִימֶ֑לֶךְ אֲשֶׁ֤ר עָשָׂה֙ לְאָבִ֔יו לַהֲרֹ֖ג אֶת־ שִׁבְעִ֥ים אֶחָֽיו׃ (נא) וְאֵ֗ת כָּל־ רָעַת֙ אַנְשֵׁ֣י שְׁכֶ֔ם הֵשִׁ֥יב אֱלֹהִ֖ים בְּרֹאשָׁ֑ם וַתָּבֹ֣א אֲלֵיהֶ֔ם קִֽלֲלַ֖ת יוֹתָ֥ם בֶּן־ יְרֻבָּֽעַל׃ סוף פרק ט
