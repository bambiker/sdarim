שופטים פרק יד לפי החלוקה לסדרים

פרק יד בספר שופטים (20 פסוקים) נמצא בשופטים סדר י (פסוקים ג–כב).

שופטים סדר י

(א) וַתֵּ֤לֶד הָֽאִשָּׁה֙ בֵּ֔ן וַתִּקְרָ֥א אֶת־ שְׁמ֖וֹ שִׁמְשׁ֑וֹן וַיִּגְדַּ֣ל הַנַּ֔עַר וַֽיְבָרְכֵ֖הוּ יְהוָֽה׃ (ב) וַתָּ֙חֶל֙ ר֣וּחַ יְהוָ֔ה לְפַעֲמ֖וֹ בְּמַחֲנֵה־ דָ֑ן בֵּ֥ין צָרְעָ֖ה וּבֵ֥ין אֶשְׁתָּאֹֽל׃ תחילת פרק יד (ג) וַיֵּ֥רֶד שִׁמְשׁ֖וֹן תִּמְנָ֑תָה וַיַּ֥רְא אִשָּׁ֛ה בְּתִמְנָ֖תָה מִבְּנ֥וֹת פְּלִשְׁתִּֽים׃ (ד) וַיַּ֗עַל וַיַּגֵּד֙ לְאָבִ֣יו וּלְאִמּ֔וֹ וַיֹּ֗אמֶר אִשָּׁ֛ה רָאִ֥יתִי בְתִמְנָ֖תָה מִבְּנ֣וֹת פְּלִשְׁתִּ֑ים וְעַתָּ֕ה קְחוּ־ אוֹתָ֥הּ לִּ֖י לְאִשָּֽׁה׃ (ה) וַיֹּ֨אמֶר ל֜וֹ אָבִ֣יו וְאִמּ֗וֹ הַאֵין֩ בִּבְנ֨וֹת אַחֶ֤יךָ וּבְכָל־ עַמִּי֙ אִשָּׁ֔ה כִּֽי־ אַתָּ֤ה הוֹלֵךְ֙ לָקַ֣חַת אִשָּׁ֔ה מִפְּלִשְׁתִּ֖ים הָעֲרֵלִ֑ים וַיֹּ֨אמֶר שִׁמְשׁ֤וֹן אֶל־ אָבִיו֙ אוֹתָ֣הּ קַֽח־ לִ֔י כִּֽי־ הִ֖יא יָשְׁרָ֥ה בְעֵינָֽי׃ (ו) וְאָבִ֨יו וְאִמּ֜וֹ לֹ֣א יָדְע֗וּ כִּ֤י מֵיְהוָה֙ הִ֔יא כִּי־ תֹאֲנָ֥ה הֽוּא־ מְבַקֵּ֖שׁ מִפְּלִשְׁתִּ֑ים וּבָעֵ֣ת הַהִ֔יא פְּלִשְׁתִּ֖ים מֹשְׁלִ֥ים בְּיִשְׂרָאֵֽל׃ (ז) וַיֵּ֧רֶד שִׁמְשׁ֛וֹן וְאָבִ֥יו וְאִמּ֖וֹ תִּמְנָ֑תָה וַיָּבֹ֙אוּ֙ עַד־ כַּרְמֵ֣י תִמְנָ֔תָה וְהִנֵּה֙ כְּפִ֣יר אֲרָי֔וֹת שֹׁאֵ֖ג לִקְרָאתֽוֹ׃ (ח) וַתִּצְלַ֨ח עָלָ֜יו ר֣וּחַ יְהוָ֗ה וַֽיְשַׁסְּעֵ֙הוּ֙ כְּשַׁסַּ֣ע הַגְּדִ֔י וּמְא֖וּמָה אֵ֣ין בְּיָד֑וֹ וְלֹ֤א הִגִּיד֙ לְאָבִ֣יו וּלְאִמּ֔וֹ אֵ֖ת אֲשֶׁ֥ר עָשָֽׂה׃ (ט) וַיֵּ֖רֶד וַיְדַבֵּ֣ר לָאִשָּׁ֑ה וַתִּישַׁ֖ר בְּעֵינֵ֥י שִׁמְשֽׁוֹן׃ (י) וַיָּ֤שָׁב מִיָּמִים֙ לְקַחְתָּ֔הּ וַיָּ֣סַר לִרְא֔וֹת אֵ֖ת מַפֶּ֣לֶת הָאַרְיֵ֑ה וְהִנֵּ֨ה עֲדַ֧ת דְּבוֹרִ֛ים בִּגְוִיַּ֥ת הָאַרְיֵ֖ה וּדְבָֽשׁ׃ (יא) וַיִּרְדֵּ֣הוּ אֶל־ כַּפָּ֗יו וַיֵּ֤לֶךְ הָלוֹךְ֙ וְאָכֹ֔ל וַיֵּ֙לֶךְ֙ אֶל־ אָבִ֣יו וְאֶל־ אִמּ֔וֹ וַיִּתֵּ֥ן לָהֶ֖ם וַיֹּאכֵ֑לוּ וְלֹֽא־ הִגִּ֣יד לָהֶ֔ם כִּ֛י מִגְּוִיַּ֥ת הָאַרְיֵ֖ה רָדָ֥ה הַדְּבָֽשׁ׃ (יב) וַיֵּ֥רֶד אָבִ֖יהוּ אֶל־ הָאִשָּׁ֑ה וַיַּ֨עַשׂ שָׁ֤ם שִׁמְשׁוֹן֙ מִשְׁתֶּ֔ה כִּ֛י כֵּ֥ן יַעֲשׂ֖וּ הַבַּחוּרִֽים׃ (יג) וַיְהִ֖י כִּרְאוֹתָ֣ם אוֹת֑וֹ וַיִּקְחוּ֙ שְׁלֹשִׁ֣ים מֵֽרֵעִ֔ים וַיִּהְי֖וּ אִתּֽוֹ׃ (יד) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן אָחֽוּדָה־ נָּ֥א לָכֶ֖ם חִידָ֑ה אִם־ הַגֵּ֣ד תַּגִּידוּ֩ אוֹתָ֨הּ לִ֜י שִׁבְעַ֨ת יְמֵ֤י הַמִּשְׁתֶּה֙ וּמְצָאתֶ֔ם וְנָתַתִּ֤י לָכֶם֙ שְׁלֹשִׁ֣ים סְדִינִ֔ים וּשְׁלֹשִׁ֖ים חֲלִפֹ֥ת בְּגָדִֽים׃ (טו) וְאִם־ לֹ֣א תוּכְלוּ֮ לְהַגִּ֣יד לִי֒ וּנְתַתֶּ֨ם אַתֶּ֥ם לִי֙ שְׁלֹשִׁ֣ים סְדִינִ֔ים וּשְׁלֹשִׁ֖ים חֲלִיפ֣וֹת בְּגָדִ֑ים וַיֹּ֣אמְרוּ ל֔וֹ ח֥וּדָה חִידָתְךָ֖ וְנִשְׁמָעֶֽנָּה׃ (טז) וַיֹּ֣אמֶר לָהֶ֗ם מֵהָֽאֹכֵל֙ יָצָ֣א מַאֲכָ֔ל וּמֵעַ֖ז יָצָ֣א מָת֑וֹק וְלֹ֥א יָכְל֛וּ לְהַגִּ֥יד הַחִידָ֖ה שְׁלֹ֥שֶׁת יָמִֽים׃ (יז) וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁבִיעִ֗י וַיֹּאמְר֤וּ לְאֵֽשֶׁת־ שִׁמְשׁוֹן֙ פַּתִּ֣י אֶת־ אִישֵׁ֗ךְ וְיַגֶּד־ לָ֙נוּ֙ אֶת־ הַ֣חִידָ֔ה פֶּן־ נִשְׂרֹ֥ף אוֹתָ֛ךְ וְאֶת־ בֵּ֥ית אָבִ֖יךְ בָּאֵ֑שׁ הַלְיָרְשֵׁ֕נוּ קְרָאתֶ֥ם לָ֖נוּ הֲלֹֽא׃ (יח) וַתֵּבְךְּ֩ אֵ֨שֶׁת שִׁמְשׁ֜וֹן עָלָ֗יו וַתֹּ֙אמֶר֙ רַק־ שְׂנֵאתַ֙נִי֙ וְלֹ֣א אֲהַבְתָּ֔נִי הַֽחִידָ֥ה חַ֙דְתָּ֙ לִבְנֵ֣י עַמִּ֔י וְלִ֖י לֹ֣א הִגַּ֑דְתָּה וַיֹּ֣אמֶר לָ֗הּ הִנֵּ֨ה לְאָבִ֧י וּלְאִמִּ֛י לֹ֥א הִגַּ֖דְתִּי וְלָ֥ךְ אַגִּֽיד׃ (יט) וַתֵּ֤בְךְּ עָלָיו֙ שִׁבְעַ֣ת הַיָּמִ֔ים אֲשֶׁר־ הָיָ֥ה לָהֶ֖ם הַמִּשְׁתֶּ֑ה וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁבִיעִ֗י וַיַּגֶּד־ לָהּ֙ כִּ֣י הֱצִיקַ֔תְהוּ וַתַּגֵּ֥ד הַחִידָ֖ה לִבְנֵ֥י עַמָּֽהּ׃ (כ) וַיֹּ֣אמְרוּ לוֹ֩ אַנְשֵׁ֨י הָעִ֜יר בַּיּ֣וֹם הַשְּׁבִיעִ֗י בְּטֶ֙רֶם֙ יָבֹ֣א הַחַ֔רְסָה מַה־ מָּת֣וֹק מִדְּבַ֔שׁ וּמֶ֥ה עַ֖ז מֵאֲרִ֑י וַיֹּ֣אמֶר לָהֶ֔ם לוּלֵא֙ חֲרַשְׁתֶּ֣ם בְּעֶגְלָתִ֔י לֹ֥א מְצָאתֶ֖ם חִידָתִֽי׃ (כא) וַתִּצְלַ֨ח עָלָ֜יו ר֣וּחַ יְהוָ֗ה וַיֵּ֨רֶד אַשְׁקְל֜וֹן וַיַּ֥ךְ מֵהֶ֣ם ׀ שְׁלֹשִׁ֣ים אִ֗ישׁ וַיִּקַּח֙ אֶת־ חֲלִ֣יצוֹתָ֔ם וַיִּתֵּן֙ הַחֲלִיפ֔וֹת לְמַגִּידֵ֖י הַחִידָ֑ה וַיִּ֣חַר אַפּ֔וֹ וַיַּ֖עַל בֵּ֥ית אָבִֽיהוּ׃ (כב) וַתְּהִ֖י אֵ֣שֶׁת שִׁמְשׁ֑וֹן לְמֵ֣רֵעֵ֔הוּ אֲשֶׁ֥ר רֵעָ֖ה לֽוֹ׃ סוף פרק יד (כג) וַיְהִ֨י מִיָּמִ֜ים בִּימֵ֣י קְצִיר־ חִטִּ֗ים וַיִּפְקֹ֨ד שִׁמְשׁ֤וֹן אֶת־ אִשְׁתּוֹ֙ בִּגְדִ֣י עִזִּ֔ים וַיֹּ֕אמֶר אָבֹ֥אָה אֶל־ אִשְׁתִּ֖י הֶחָ֑דְרָה וְלֹֽא־ נְתָנ֥וֹ אָבִ֖יהָ לָבֽוֹא׃ (כד) וַיֹּ֣אמֶר אָבִ֗יהָ אָמֹ֤ר אָמַ֙רְתִּי֙ כִּי־ שָׂנֹ֣א שְׂנֵאתָ֔הּ וָאֶתְּנֶ֖נָּה לְמֵרֵעֶ֑ךָ הֲלֹ֨א אֲחֹתָ֤הּ הַקְּטַנָּה֙ טוֹבָ֣ה מִמֶּ֔נָּה תְּהִי־ נָ֥א לְךָ֖ תַּחְתֶּֽיהָ׃ (כה) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן נִקֵּ֥יתִי הַפַּ֖עַם מִפְּלִשְׁתִּ֑ים כִּֽי־ עֹשֶׂ֥ה אֲנִ֛י עִמָּ֖ם רָעָֽה׃ (כו) וַיֵּ֣לֶךְ שִׁמְשׁ֔וֹן וַיִּלְכֹּ֖ד שְׁלֹשׁ־ מֵא֣וֹת שׁוּעָלִ֑ים וַיִּקַּ֣ח לַפִּדִ֗ים וַיֶּ֤פֶן זָנָב֙ אֶל־ זָנָ֔ב וַיָּ֨שֶׂם לַפִּ֥יד אֶחָ֛ד בֵּין־ שְׁנֵ֥י הַזְּנָב֖וֹת בַּתָּֽוֶךְ׃ (כז) וַיַּבְעֶר־ אֵשׁ֙ בַּלַּפִּידִ֔ים וַיְשַׁלַּ֖ח בְּקָמ֣וֹת פְּלִשְׁתִּ֑ים וַיַּבְעֵ֛ר מִגָּדִ֥ישׁ וְעַד־ קָמָ֖ה וְעַד־ כֶּ֥רֶם זָֽיִת׃ (כח) וַיֹּאמְר֣וּ פְלִשְׁתִּים֮ מִ֣י עָ֣שָׂה זֹאת֒ וַיֹּאמְר֗וּ שִׁמְשׁוֹן֙ חֲתַ֣ן הַתִּמְנִ֔י כִּ֚י לָקַ֣ח אֶת־ אִשְׁתּ֔וֹ וַֽיִּתְּנָ֖הּ לְמֵרֵעֵ֑הוּ וַיַּעֲל֣וּ פְלִשְׁתִּ֔ים וַיִּשְׂרְפ֥וּ אוֹתָ֛הּ וְאֶת־ אָבִ֖יהָ בָּאֵֽשׁ׃ (כט) וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן אִֽם־ תַּעֲשׂ֖וּן כָּזֹ֑את כִּ֛י אִם־ נִקַּ֥מְתִּי בָכֶ֖ם וְאַחַ֥ר אֶחְדָּֽל׃ (ל) וַיַּ֨ךְ אוֹתָ֥ם שׁ֛וֹק עַל־ יָרֵ֖ךְ מַכָּ֣ה גְדוֹלָ֑ה וַיֵּ֣רֶד וַיֵּ֔שֶׁב בִּסְעִ֖יף סֶ֥לַע עֵיטָֽם׃ (לא) וַיַּעֲל֣וּ פְלִשְׁתִּ֔ים וַֽיַּחֲנ֖וּ בִּיהוּדָ֑ה וַיִּנָּטְשׁ֖וּ בַּלֶּֽחִי׃ (לב) וַיֹּֽאמְרוּ֙ אִ֣ישׁ יְהוּדָ֔ה לָמָ֖ה עֲלִיתֶ֣ם עָלֵ֑ינוּ וַיֹּאמְר֗וּ לֶאֱס֤וֹר אֶת־ שִׁמְשׁוֹן֙ עָלִ֔ינוּ לַעֲשׂ֣וֹת ל֔וֹ כַּאֲשֶׁ֖ר עָ֥שָׂה לָֽנוּ׃ (לג) וַיֵּרְד֡וּ שְׁלֹשֶׁת֩ אֲלָפִ֨ים אִ֜ישׁ מִֽיהוּדָ֗ה אֶל־ סְעִיף֮ סֶ֣לַע עֵיטָם֒ וַיֹּאמְר֣וּ לְשִׁמְשׁ֗וֹן הֲלֹ֤א יָדַ֙עְתָּ֙ כִּֽי־ מֹשְׁלִ֥ים בָּ֙נוּ֙ פְּלִשְׁתִּ֔ים וּמַה־ זֹּ֖את עָשִׂ֣יתָ לָּ֑נוּ וַיֹּ֣אמֶר לָהֶ֔ם כַּאֲשֶׁר֙ עָ֣שׂוּ לִ֔י כֵּ֖ן עָשִׂ֥יתִי לָהֶֽם׃ (לד) וַיֹּ֤אמְרוּ לוֹ֙ לֶאֱסָרְךָ֣ יָרַ֔דְנוּ לְתִתְּךָ֖ בְּיַד־ פְּלִשְׁתִּ֑ים וַיֹּ֤אמֶר לָהֶם֙ שִׁמְשׁ֔וֹן הִשָּׁבְע֣וּ לִ֔י פֶּֽן־ תִּפְגְּע֥וּן בִּ֖י אַתֶּֽם׃ (לה) וַיֹּ֧אמְרוּ ל֣וֹ לֵאמֹ֗ר לֹ֚א כִּֽי־ אָסֹ֤ר נֶֽאֱסָרְךָ֙ וּנְתַנּ֣וּךָ בְיָדָ֔ם וְהָמֵ֖ת לֹ֣א נְמִיתֶ֑ךָ וַיַּאַסְרֻ֗הוּ בִּשְׁנַ֙יִם֙ עֲבֹתִ֣ים חֲדָשִׁ֔ים וַֽיַּעֲל֖וּהוּ מִן־ הַסָּֽלַע׃ (לו) הוּא־ בָ֣א עַד־ לֶ֔חִי וּפִלְשִׁתִּ֖ים הֵרִ֣יעוּ לִקְרָאת֑וֹ וַתִּצְלַ֨ח עָלָ֜יו ר֣וּחַ יְהוָ֗ה וַתִּהְיֶ֨ינָה הָעֲבֹתִ֜ים אֲשֶׁ֣ר עַל־ זְרוֹעוֹתָ֗יו כַּפִּשְׁתִּים֙ אֲשֶׁ֣ר בָּעֲר֣וּ בָאֵ֔שׁ וַיִּמַּ֥סּוּ אֱסוּרָ֖יו מֵעַ֥ל יָדָֽיו׃ (לז) וַיִּמְצָ֥א לְחִֽי־ חֲמ֖וֹר טְרִיָּ֑ה וַיִּשְׁלַ֤ח יָדוֹ֙ וַיִּקָּחֶ֔הָ וַיַּךְ־ בָּ֖הּ אֶ֥לֶף אִֽישׁ׃ (לח) וַיֹּ֣אמֶר שִׁמְשׁ֔וֹן בִּלְחִ֣י הַחֲמ֔וֹר חֲמ֖וֹר חֲמֹרָתָ֑יִם בִּלְחִ֣י הַחֲמ֔וֹר הִכֵּ֖יתִי אֶ֥לֶף אִֽישׁ׃ (לט) וַֽיְהִי֙ כְּכַלֹּת֣וֹ לְדַבֵּ֔ר וַיַּשְׁלֵ֥ךְ הַלְּחִ֖י מִיָּד֑וֹ וַיִּקְרָ֛א לַמָּק֥וֹם הַה֖וּא רָ֥מַת לֶֽחִי׃ (מ) וַיִּצְמָא֮ מְאֹד֒ וַיִּקְרָ֤א אֶל־ יְהוָה֙ וַיֹּאמַ֔ר אַתָּה֙ נָתַ֣תָּ בְיַֽד־ עַבְדְּךָ֔ אֶת־ הַתְּשׁוּעָ֥ה הַגְּדֹלָ֖ה הַזֹּ֑את וְעַתָּה֙ אָמ֣וּת בַּצָּמָ֔א וְנָפַלְתִּ֖י בְּיַ֥ד הָעֲרֵלִֽים׃ (מא) וַיִּבְקַ֨ע אֱלֹהִ֜ים אֶת־ הַמַּכְתֵּ֣שׁ אֲשֶׁר־ בַּלֶּ֗חִי וַיֵּצְא֨וּ מִמֶּ֤נּוּ מַ֙יִם֙ וַיֵּ֔שְׁתְּ וַתָּ֥שָׁב רוּח֖וֹ וַיֶּ֑חִי עַל־ כֵּ֣ן ׀ קָרָ֣א שְׁמָ֗הּ עֵ֤ין הַקּוֹרֵא֙ אֲשֶׁ֣ר בַּלֶּ֔חִי עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (מב) וַיִּשְׁפֹּ֧ט אֶת־ יִשְׂרָאֵ֛ל בִּימֵ֥י פְלִשְׁתִּ֖ים עֶשְׂרִ֥ים שָׁנָֽה׃ (מג) וַיֵּ֥לֶךְ שִׁמְשׁ֖וֹן עַזָּ֑תָה וַיַּרְא־ שָׁם֙ אִשָּׁ֣ה זוֹנָ֔ה וַיָּבֹ֖א אֵלֶֽיהָ׃ (מד) לַֽעַזָּתִ֣ים ׀ לֵאמֹ֗ר בָּ֤א שִׁמְשׁוֹן֙ הֵ֔נָּה וַיָּסֹ֛בּוּ וַיֶּאֶרְבוּ־ ל֥וֹ כָל־ הַלַּ֖יְלָה בְּשַׁ֣עַר הָעִ֑יר וַיִּתְחָרְשׁ֤וּ כָל־ הַלַּ֙יְלָה֙ לֵאמֹ֔ר עַד־ א֥וֹר הַבֹּ֖קֶר וַהֲרְגְנֻֽהוּ׃
