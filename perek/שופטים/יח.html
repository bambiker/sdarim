שופטים פרק יח לפי החלוקה לסדרים

פרק יח בספר שופטים (31 פסוקים) נמצא בשופטים סדר יא (פסוקים מג–מז) ובשופטים סדר יב (פסוקים א–כו).

שופטים סדר יא

(א) וַיִּשְׁכַּ֣ב שִׁמְשׁוֹן֮ עַד־ חֲצִ֣י הַלַּיְלָה֒ וַיָּ֣קָם ׀ בַּחֲצִ֣י הַלַּ֗יְלָה וַיֶּאֱחֹ֞ז בְּדַלְת֤וֹת שַֽׁעַר־ הָעִיר֙ וּבִשְׁתֵּ֣י הַמְּזוּז֔וֹת וַיִּסָּעֵם֙ עִֽם־ הַבְּרִ֔יחַ וַיָּ֖שֶׂם עַל־ כְּתֵפָ֑יו וַֽיַּעֲלֵם֙ אֶל־ רֹ֣אשׁ הָהָ֔ר אֲשֶׁ֖ר עַל־ פְּנֵ֥י חֶבְרֽוֹן׃ (ב) וַֽיְהִי֙ אַחֲרֵי־ כֵ֔ן וַיֶּאֱהַ֥ב אִשָּׁ֖ה בְּנַ֣חַל שֹׂרֵ֑ק וּשְׁמָ֖הּ דְּלִילָֽה׃ (ג) וַיַּעֲל֨וּ אֵלֶ֜יהָ סַרְנֵ֣י פְלִשְׁתִּ֗ים וַיֹּ֨אמְרוּ לָ֜הּ פַּתִּ֣י אוֹת֗וֹ וּרְאִי֙ בַּמֶּה֙ כֹּח֣וֹ גָד֔וֹל וּבַמֶּה֙ נ֣וּכַל ל֔וֹ וַאֲסַרְנֻ֖הוּ לְעַנֹּת֑וֹ וַאֲנַ֙חְנוּ֙ נִתַּן־ לָ֔ךְ אִ֕ישׁ אֶ֥לֶף וּמֵאָ֖ה כָּֽסֶף׃ (ד) וַתֹּ֤אמֶר דְּלִילָה֙ אֶל־ שִׁמְשׁ֔וֹן הַגִּֽידָה־ נָּ֣א לִ֔י בַּמֶּ֖ה כֹּחֲךָ֣ גָד֑וֹל וּבַמֶּ֥ה תֵאָסֵ֖ר לְעַנּוֹתֶֽךָ׃ (ה) וַיֹּ֤אמֶר אֵלֶ֙יהָ֙ שִׁמְשׁ֔וֹן אִם־ יַאַסְרֻ֗נִי בְּשִׁבְעָ֛ה יְתָרִ֥ים לַחִ֖ים אֲשֶׁ֣ר לֹא־ חֹרָ֑בוּ וְחָלִ֥יתִי וְהָיִ֖יתִי כְּאַחַ֥ד הָאָדָֽם׃ (ו) וַיַּעֲלוּ־ לָ֞הּ סַרְנֵ֣י פְלִשְׁתִּ֗ים שִׁבְעָ֛ה יְתָרִ֥ים לַחִ֖ים אֲשֶׁ֣ר לֹא־ חֹרָ֑בוּ וַתַּאַסְרֵ֖הוּ בָּהֶֽם׃ (ז) וְהָאֹרֵ֗ב יֹשֵׁ֥ב לָהּ֙ בַּחֶ֔דֶר וַתֹּ֣אמֶר אֵלָ֔יו פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיְנַתֵּק֙ אֶת־ הַיְתָרִ֔ים כַּאֲשֶׁ֨ר יִנָּתֵ֤ק פְּתִֽיל־ הַנְּעֹ֙רֶת֙ בַּהֲרִיח֣וֹ אֵ֔שׁ וְלֹ֥א נוֹדַ֖ע כֹּחֽוֹ׃ (ח) וַתֹּ֤אמֶר דְּלִילָה֙ אֶל־ שִׁמְשׁ֔וֹן הִנֵּה֙ הֵתַ֣לְתָּ בִּ֔י וַתְּדַבֵּ֥ר אֵלַ֖י כְּזָבִ֑ים עַתָּה֙ הַגִּֽידָה־ נָּ֣א לִ֔י בַּמֶּ֖ה תֵּאָסֵֽר׃ (ט) וַיֹּ֣אמֶר אֵלֶ֔יהָ אִם־ אָס֤וֹר יַאַסְר֙וּנִי֙ בַּעֲבֹתִ֣ים חֲדָשִׁ֔ים אֲשֶׁ֛ר לֹֽא־ נַעֲשָׂ֥ה בָהֶ֖ם מְלָאכָ֑ה וְחָלִ֥יתִי וְהָיִ֖יתִי כְּאַחַ֥ד הָאָדָֽם׃ (י) וַתִּקַּ֣ח דְּלִילָה֩ עֲבֹתִ֨ים חֲדָשִׁ֜ים וַתַּאַסְרֵ֣הוּ בָהֶ֗ם וַתֹּ֤אמֶר אֵלָיו֙ פְּלִשְׁתִּ֤ים עָלֶ֙יךָ֙ שִׁמְשׁ֔וֹן וְהָאֹרֵ֖ב יֹשֵׁ֣ב בֶּחָ֑דֶר וַֽיְנַתְּקֵ֛ם מֵעַ֥ל זְרֹעֹתָ֖יו כַּחֽוּט׃ (יא) וַתֹּ֨אמֶר דְּלִילָ֜ה אֶל־ שִׁמְשׁ֗וֹן עַד־ הֵ֜נָּה הֵתַ֤לְתָּ בִּי֙ וַתְּדַבֵּ֤ר אֵלַי֙ כְּזָבִ֔ים הַגִּ֣ידָה לִּ֔י בַּמֶּ֖ה תֵּאָסֵ֑ר וַיֹּ֣אמֶר אֵלֶ֔יהָ אִם־ תַּאַרְגִ֗י אֶת־ שֶׁ֛בַע מַחְלְפ֥וֹת רֹאשִׁ֖י עִם־ הַמַּסָּֽכֶת׃ (יב) וַתִּתְקַע֙ בַּיָּתֵ֔ד וַתֹּ֣אמֶר אֵלָ֔יו פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיִּיקַץ֙ מִשְּׁנָת֔וֹ וַיִּסַּ֛ע אֶת־ הַיְתַ֥ד הָאֶ֖רֶג וְאֶת־ הַמַּסָּֽכֶת׃ (יג) וַתֹּ֣אמֶר אֵלָ֗יו אֵ֚יךְ תֹּאמַ֣ר אֲהַבְתִּ֔יךְ וְלִבְּךָ֖ אֵ֣ין אִתִּ֑י זֶ֣ה שָׁלֹ֤שׁ פְּעָמִים֙ הֵתַ֣לְתָּ בִּ֔י וְלֹא־ הִגַּ֣דְתָּ לִּ֔י בַּמֶּ֖ה כֹּחֲךָ֥ גָדֽוֹל׃ (יד) וַ֠יְהִי כִּֽי־ הֵצִ֨יקָה לּ֧וֹ בִדְבָרֶ֛יהָ כָּל־ הַיָּמִ֖ים וַתְּאַֽלֲצֵ֑הוּ וַתִּקְצַ֥ר נַפְשׁ֖וֹ לָמֽוּת׃ (טו) וַיַּגֶּד־ לָ֣הּ אֶת־ כָּל־ לִבּ֗וֹ וַיֹּ֤אמֶר לָהּ֙ מוֹרָה֙ לֹֽא־ עָלָ֣ה עַל־ רֹאשִׁ֔י כִּֽי־ נְזִ֧יר אֱלֹהִ֛ים אֲנִ֖י מִבֶּ֣טֶן אִמִּ֑י אִם־ גֻּלַּ֙חְתִּי֙ וְסָ֣ר מִמֶּ֣נִּי כֹחִ֔י וְחָלִ֥יתִי וְהָיִ֖יתִי כְּכָל־ הָאָדָֽם׃ (טז) וַתֵּ֣רֶא דְלִילָ֗ה כִּֽי־ הִגִּ֣יד לָהּ֮ אֶת־ כָּל־ לִבּוֹ֒ וַתִּשְׁלַ֡ח וַתִּקְרָא֩ לְסַרְנֵ֨י פְלִשְׁתִּ֤ים לֵאמֹר֙ עֲל֣וּ הַפַּ֔עַם כִּֽי־ הִגִּ֥יד (לה) לִ֖י אֶת־ כָּל־ לִבּ֑וֹ וְעָל֤וּ אֵלֶ֙יהָ֙ סַרְנֵ֣י פְלִשְׁתִּ֔ים וַיַּעֲל֥וּ הַכֶּ֖סֶף בְּיָדָֽם׃ (יז) וַתְּיַשְּׁנֵ֙הוּ֙ עַל־ בִּרְכֶּ֔יהָ וַתִּקְרָ֣א לָאִ֔ישׁ וַתְּגַלַּ֕ח אֶת־ שֶׁ֖בַע מַחְלְפ֣וֹת רֹאשׁ֑וֹ וַתָּ֙חֶל֙ לְעַנּוֹת֔וֹ וַיָּ֥סַר כֹּח֖וֹ מֵעָלָֽיו׃ (יח) וַתֹּ֕אמֶר פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיִּקַ֣ץ מִשְּׁנָת֗וֹ וַיֹּ֙אמֶר֙ אֵצֵ֞א כְּפַ֤עַם בְּפַ֙עַם֙ וְאִנָּעֵ֔ר וְהוּא֙ לֹ֣א יָדַ֔ע כִּ֥י יְהוָ֖ה סָ֥ר מֵעָלָֽיו׃ (יט) וַיֹּאחֲז֣וּהוּ פְלִשְׁתִּ֔ים וַֽיְנַקְּר֖וּ אֶת־ עֵינָ֑יו וַיּוֹרִ֨ידוּ אוֹת֜וֹ עַזָּ֗תָה וַיַּאַסְר֙וּהוּ֙ בַּֽנְחֻשְׁתַּ֔יִם וַיְהִ֥י טוֹחֵ֖ן בְּבֵ֥ית (האסירים) הָאֲסוּרִֽים׃ (כ) וַיָּ֧חֶל שְׂעַר־ רֹאשׁ֛וֹ לְצַמֵּ֖חַ כַּאֲשֶׁ֥ר גֻּלָּֽח׃ (כא) וְסַרְנֵ֣י פְלִשְׁתִּ֗ים נֶֽאֱסְפוּ֙ לִזְבֹּ֧חַ זֶֽבַח־ גָּד֛וֹל לְדָג֥וֹן אֱלֹהֵיהֶ֖ם וּלְשִׂמְחָ֑ה וַיֹּ֣אמְר֔וּ נָתַ֤ן אֱלֹהֵ֙ינוּ֙ בְּיָדֵ֔נוּ אֵ֖ת שִׁמְשׁ֥וֹן אוֹיְבֵֽינוּ׃ (כב) וַיִּרְא֤וּ אֹתוֹ֙ הָעָ֔ם וַֽיְהַלְל֖וּ אֶת־ אֱלֹהֵיהֶ֑ם כִּ֣י אָמְר֗וּ נָתַ֨ן אֱלֹהֵ֤ינוּ בְיָדֵ֙נוּ֙ אֶת־ א֣וֹיְבֵ֔נוּ וְאֵת֙ מַחֲרִ֣יב אַרְצֵ֔נוּ וַאֲשֶׁ֥ר הִרְבָּ֖ה אֶת־ חֲלָלֵֽינוּ׃ (כג) וַֽיְהִי֙ (כי טוב) כְּט֣וֹב לִבָּ֔ם וַיֹּ֣אמְר֔וּ קִרְא֥וּ לְשִׁמְשׁ֖וֹן וִישַֽׂחֶק־ לָ֑נוּ וַיִּקְרְא֨וּ לְשִׁמְשׁ֜וֹן מִבֵּ֣ית (האסירים) הָאֲסוּרִ֗ים וַיְצַחֵק֙ לִפְנֵיהֶ֔ם וַיַּעֲמִ֥ידוּ אוֹת֖וֹ בֵּ֥ין הָעַמּוּדִֽים׃ (כד) וַיֹּ֨אמֶר שִׁמְשׁ֜וֹן אֶל־ הַנַּ֨עַר הַמַּחֲזִ֣יק בְּיָדוֹ֮ הַנִּ֣יחָה אוֹתִי֒ (והימשני) וַהֲמִשֵׁ֙נִי֙ אֶת־ הָֽעַמֻּדִ֔ים אֲשֶׁ֥ר הַבַּ֖יִת נָכ֣וֹן עֲלֵיהֶ֑ם וְאֶשָּׁעֵ֖ן עֲלֵיהֶֽם׃ (כה) וְהַבַּ֗יִת מָלֵ֤א הָֽאֲנָשִׁים֙ וְהַנָּשִׁ֔ים וְשָׁ֕מָּה כֹּ֖ל סַרְנֵ֣י פְלִשְׁתִּ֑ים וְעַל־ הַגָּ֗ג כִּשְׁלֹ֤שֶׁת אֲלָפִים֙ אִ֣ישׁ וְאִשָּׁ֔ה הָרֹאִ֖ים בִּשְׂח֥וֹק שִׁמְשֽׁוֹן׃ (כו) וַיִּקְרָ֥א שִׁמְשׁ֛וֹן אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר אֲדֹנָ֣י יֱהֹוִ֡ה זָכְרֵ֣נִי נָא֩ וְחַזְּקֵ֨נִי נָ֜א אַ֣ךְ הַפַּ֤עַם הַזֶּה֙ הָאֱלֹהִ֔ים וְאִנָּקְמָ֧ה נְקַם־ אַחַ֛ת מִשְּׁתֵ֥י עֵינַ֖י מִפְּלִשְׁתִּֽים׃ (כז) וַיִּלְפֹּ֨ת שִׁמְשׁ֜וֹן אֶת־ שְׁנֵ֣י ׀ עַמּוּדֵ֣י הַתָּ֗וֶךְ אֲשֶׁ֤ר הַבַּ֙יִת֙ נָכ֣וֹן עֲלֵיהֶ֔ם וַיִּסָּמֵ֖ךְ עֲלֵיהֶ֑ם אֶחָ֥ד בִּימִינ֖וֹ וְאֶחָ֥ד בִּשְׂמֹאלֽוֹ׃ (כח) וַיֹּ֣אמֶר שִׁמְשׁ֗וֹן תָּמ֣וֹת נַפְשִׁי֮ עִם־ פְּלִשְׁתִּים֒ וַיֵּ֣ט בְּכֹ֔חַ וַיִּפֹּ֤ל הַבַּ֙יִת֙ עַל־ הַסְּרָנִ֔ים וְעַל־ כָּל־ הָעָ֖ם אֲשֶׁר־ בּ֑וֹ וַיִּהְי֤וּ הַמֵּתִים֙ אֲשֶׁ֣ר הֵמִ֣ית בְּמוֹת֔וֹ רַבִּ֕ים מֵאֲשֶׁ֥ר הֵמִ֖ית בְּחַיָּֽיו׃ (כט) וַיֵּרְד֨וּ אֶחָ֜יו וְכָל־ בֵּ֣ית אָבִיהוּ֮ וַיִּשְׂא֣וּ אֹתוֹ֒ וַֽיַּעֲל֣וּ ׀ וַיִּקְבְּר֣וּ אוֹת֗וֹ בֵּ֤ין צָרְעָה֙ וּבֵ֣ין אֶשְׁתָּאֹ֔ל בְּקֶ֖בֶר מָנ֣וֹחַ אָבִ֑יו וְה֛וּא שָׁפַ֥ט אֶת־ יִשְׂרָאֵ֖ל עֶשְׂרִ֥ים שָׁנָֽה׃ (ל) וַֽיְהִי־ אִ֥ישׁ מֵֽהַר־ אֶפְרָ֖יִם וּשְׁמ֥וֹ מִיכָֽיְהוּ׃ (לא) וַיֹּ֣אמֶר לְאִמּ֡וֹ אֶלֶף֩ וּמֵאָ֨ה הַכֶּ֜סֶף אֲשֶׁ֣ר לֻֽקַּֽח־ לָ֗ךְ (ואתי) וְאַ֤תְּ אָלִית֙ וְגַם֙ אָמַ֣רְתְּ בְּאָזְנַ֔י הִנֵּֽה־ הַכֶּ֥סֶף אִתִּ֖י אֲנִ֣י לְקַחְתִּ֑יו וַתֹּ֣אמֶר אִמּ֔וֹ בָּר֥וּךְ בְּנִ֖י לַיהוָֽה׃ (לב) וַיָּ֛שֶׁב אֶת־ אֶֽלֶף־ וּמֵאָ֥ה הַכֶּ֖סֶף לְאִמּ֑וֹ וַתֹּ֣אמֶר אִמּ֡וֹ הַקְדֵּ֣שׁ הִקְדַּ֣שְׁתִּי אֶת־ הַטַּ֧ף וְאֶת־ הַמִּקְנֶ֛ה וְאֶת־ הַכְּבוּדָּ֖ה לִפְנֵיהֶֽם׃ (יז) הֵ֥מָּה הִרְחִ֖יקוּ מִבֵּ֣ית מִיכָ֑ה וְהָאֲנָשִׁ֗ים אֲשֶׁ֤ר בַּבָּתִּים֙ אֲשֶׁר֙ עִם־ בֵּ֣ית מִיכָ֔ה נִֽזְעֲק֔וּ וַיַּדְבִּ֖יקוּ אֶת־ בְּנֵי־ דָֽן׃ (יח) וַֽיִּקְרְאוּ֙ אֶל־ בְּנֵי־ דָ֔ן וַיַּסֵּ֖בּוּ פְּנֵיהֶ֑ם וַיֹּאמְר֣וּ לְמִיכָ֔ה מַה־ לְּךָ֖ כִּ֥י נִזְעָֽקְתָּ׃ (יט) וַיֹּ֡אמֶר אֶת־ אֱלֹהַי֩ אֲשֶׁר־ עָשִׂ֨יתִי לְקַחְתֶּ֧ם וְֽאֶת־ הַכֹּהֵ֛ן וַתֵּלְכ֖וּ וּמַה־ לִּ֣י ע֑וֹד וּמַה־ זֶּ֛ה תֹּאמְר֥וּ אֵלַ֖י מַה־ לָּֽךְ׃ (כ) וַיֹּאמְר֤וּ אֵלָיו֙ בְּנֵי־ דָ֔ן אַל־ תַּשְׁמַ֥ע קוֹלְךָ֖ עִמָּ֑נוּ פֶּֽן־ יִפְגְּע֣וּ בָכֶ֗ם אֲנָשִׁים֙ מָ֣רֵי נֶ֔פֶשׁ וְאָסַפְתָּ֥ה נַפְשְׁךָ֖ וְנֶ֥פֶשׁ בֵּיתֶֽךָ׃ (כא) וַיֵּלְכ֥וּ בְנֵי־ דָ֖ן לְדַרְכָּ֑ם וַיַּ֣רְא מִיכָ֗ה כִּי־ חֲזָקִ֥ים הֵ֙מָּה֙ מִמֶּ֔נּוּ וַיִּ֖פֶן וַיָּ֥שָׁב אֶל־ בֵּיתֽוֹ׃ (כב) וְהֵ֨מָּה לָקְח֜וּ אֵ֧ת אֲשֶׁר־ עָשָׂ֣ה מִיכָ֗ה וְֽאֶת־ הַכֹּהֵן֮ אֲשֶׁ֣ר הָיָה־ לוֹ֒ וַיָּבֹ֣אוּ עַל־ לַ֗יִשׁ עַל־ עַם֙ שֹׁקֵ֣ט וּבֹטֵ֔חַ וַיַּכּ֥וּ אוֹתָ֖ם לְפִי־ חָ֑רֶב וְאֶת־ הָעִ֖יר שָׂרְפ֥וּ בָאֵֽשׁ׃ (כג) וְאֵ֨ין מַצִּ֜יל כִּ֧י רְֽחוֹקָה־ הִ֣יא מִצִּיד֗וֹן וְדָבָ֤ר אֵין־ לָהֶם֙ עִם־ אָדָ֔ם וְהִ֕יא בָּעֵ֖מֶק אֲשֶׁ֣ר לְבֵית־ רְח֑וֹב וַיִּבְנ֥וּ אֶת־ הָעִ֖יר וַיֵּ֥שְׁבוּ בָֽהּ׃ (כד) וַיִּקְרְא֤וּ שֵׁם־ הָעִיר֙ דָּ֔ן בְּשֵׁם֙ דָּ֣ן אֲבִיהֶ֔ם אֲשֶׁ֥ר יוּלַּ֖ד לְיִשְׂרָאֵ֑ל וְאוּלָ֛ם לַ֥יִשׁ שֵׁם־ הָעִ֖יר לָרִאשֹׁנָֽה׃ (כה) וַיָּקִ֧ימוּ לָהֶ֛ם בְּנֵי־ דָ֖ן אֶת־ הַפָּ֑סֶל וִ֠יהוֹנָתָן בֶּן־ גֵּרְשֹׁ֨ם בֶּן־ מְ ה֣וּא וּבָנָ֗יו הָי֤וּ כֹהֲנִים֙ לְשֵׁ֣בֶט הַדָּנִ֔י עַד־ י֖וֹם גְּל֥וֹת הָאָֽרֶץ׃ (כו) וַיָּשִׂ֣ימוּ לָהֶ֔ם אֶת־ פֶּ֥סֶל מִיכָ֖ה אֲשֶׁ֣ר עָשָׂ֑ה כָּל־ יְמֵ֛י הֱי֥וֹת בֵּית־ הָאֱלֹהִ֖ים בְּשִׁלֹֽה׃ סוף פרק יח (כז) וַיְהִי֙ בַּיָּמִ֣ים הָהֵ֔ם וּמֶ֖לֶךְ אֵ֣ין בְּיִשְׂרָאֵ֑ל וַיְהִ֣י ׀ אִ֣ישׁ לֵוִ֗י גָּ֚ר בְּיַרְכְּתֵ֣י הַר־ אֶפְרַ֔יִם וַיִּֽקַּֽח־ לוֹ֙ אִשָּׁ֣ה פִילֶ֔גֶשׁ מִבֵּ֥ית לֶ֖חֶם יְהוּדָֽה׃ (כח) וַתִּזְנֶ֤ה עָלָיו֙ פִּֽילַגְשׁ֔וֹ וַתֵּ֤לֶךְ מֵֽאִתּוֹ֙ אֶל־ בֵּ֣ית אָבִ֔יהָ אֶל־ בֵּ֥ית לֶ֖חֶם יְהוּדָ֑ה וַתְּהִי־ שָׁ֕ם יָמִ֖ים אַרְבָּעָ֥ה חֳדָשִֽׁים׃ (כט) וַיָּ֨קָם אִישָׁ֜הּ וַיֵּ֣לֶךְ אַחֲרֶ֗יהָ לְדַבֵּ֤ר עַל־ לִבָּהּ֙ (להשיבו) לַהֲשִׁיבָ֔הּ וְנַעֲר֥וֹ עִמּ֖וֹ וְצֶ֣מֶד חֲמֹרִ֑ים וַתְּבִיאֵ֙הוּ֙ בֵּ֣ית אָבִ֔יהָ וַיִּרְאֵ֙הוּ֙ אֲבִ֣י הַֽנַּעֲרָ֔ה וַיִּשְׂמַ֖ח לִקְרָאתֽוֹ׃ (ל) וַיֶּחֱזַק־ בּ֤וֹ חֹֽתְנוֹ֙ אֲבִ֣י הַֽנַּעֲרָ֔ה וַיֵּ֥שֶׁב אִתּ֖וֹ שְׁלֹ֣שֶׁת יָמִ֑ים וַיֹּאכְלוּ֙ וַיִּשְׁתּ֔וּ וַיָּלִ֖ינוּ שָֽׁם׃ (לא) וַֽיְהִי֙ בַּיּ֣וֹם הָרְבִיעִ֔י וַיַּשְׁכִּ֥ימוּ בַבֹּ֖קֶר וַיָּ֣קָם לָלֶ֑כֶת וַיֹּאמֶר֩ אֲבִ֨י הַֽנַּעֲרָ֜ה אֶל־ חֲתָנ֗וֹ סְעָ֧ד לִבְּךָ֛ פַּת־ לֶ֖חֶם וְאַחַ֥ר תֵּלֵֽכוּ׃ (לב) וַיֵּשְׁב֗וּ וַיֹּאכְל֧וּ שְׁנֵיהֶ֛ם יַחְדָּ֖ו וַיִּשְׁתּ֑וּ וַיֹּ֜אמֶר אֲבִ֤י הַֽנַּעֲרָה֙ אֶל־ הָאִ֔ישׁ הֽוֹאֶל־ נָ֥א וְלִ֖ין וְיִטַ֥ב לִבֶּֽךָ׃ (לג) וַיָּ֥קָם הָאִ֖ישׁ לָלֶ֑כֶת וַיִּפְצַר־ בּוֹ֙ חֹתְנ֔וֹ וַיָּ֖שָׁב וַיָּ֥לֶן שָֽׁם׃ (לד) וַיַּשְׁכֵּ֨ם בַּבֹּ֜קֶר בַּיּ֣וֹם הַחֲמִישִׁי֮ לָלֶכֶת֒ וַיֹּ֣אמֶר ׀ אֲבִ֣י הַֽנַּעֲרָ֗ה סְעָד־ נָא֙ לְבָ֣בְךָ֔ וְהִֽתְמַהְמְה֖וּ עַד־ נְט֣וֹת הַיּ֑וֹם וַיֹּאכְל֖וּ שְׁנֵיהֶֽם׃ (לה) וַיָּ֤קָם הָאִישׁ֙ לָלֶ֔כֶת ה֥וּא וּפִילַגְשׁ֖וֹ וְנַעֲר֑וֹ וַיֹּ֣אמֶר ל֣וֹ חֹתְנ֣וֹ אֲבִ֣י הַֽנַּעֲרָ֡ה הִנֵּ֣ה נָא֩ רָפָ֨ה הַיּ֜וֹם לַעֲרֹ֗ב לִֽינוּ־ נָ֞א הִנֵּ֨ה חֲנ֤וֹת הַיּוֹם֙ לִ֥ין פֹּה֙ וְיִיטַ֣ב לְבָבֶ֔ךָ וְהִשְׁכַּמְתֶּ֤ם מָחָר֙ לְדַרְכְּכֶ֔ם וְהָלַכְתָּ֖ לְאֹהָלֶֽךָ׃ (לו) וְלֹֽא־ אָבָ֤ה הָאִישׁ֙ לָל֔וּן וַיָּ֣קָם וַיֵּ֗לֶךְ וַיָּבֹא֙ עַד־ נֹ֣כַח יְב֔וּס הִ֖יא יְרוּשָׁלִָ֑ם וְעִמּ֗וֹ צֶ֤מֶד חֲמוֹרִים֙ חֲבוּשִׁ֔ים וּפִילַגְשׁ֖וֹ עִמּֽוֹ׃ (לז) הֵ֣ם עִם־ יְב֔וּס וְהַיּ֖וֹם רַ֣ד מְאֹ֑ד וַיֹּ֨אמֶר הַנַּ֜עַר אֶל־ אֲדֹנָ֗יו לְכָה־ נָּ֛א וְנָס֛וּרָה אֶל־ עִֽיר־ הַיְבוּסִ֥י הַזֹּ֖את וְנָלִ֥ין בָּֽהּ׃ (לח) וַיֹּ֤אמֶר אֵלָיו֙ אֲדֹנָ֔יו לֹ֤א נָסוּר֙ אֶל־ עִ֣יר נָכְרִ֔י אֲשֶׁ֛ר לֹֽא־ מִבְּנֵ֥י יִשְׂרָאֵ֖ל הֵ֑נָּה וְעָבַ֖רְנוּ עַד־ גִּבְעָֽה׃ (לט) וַיֹּ֣אמֶר לְנַעֲר֔וֹ לְךָ֥ וְנִקְרְבָ֖ה בְּאַחַ֣ד הַמְּקֹמ֑וֹת וְלַ֥נּוּ בַגִּבְעָ֖ה א֥וֹ בָרָמָֽה׃ (מ) וַיַּעַבְר֖וּ וַיֵּלֵ֑כוּ וַתָּבֹ֤א לָהֶם֙ הַשֶּׁ֔מֶשׁ אֵ֥צֶל הַגִּבְעָ֖ה אֲשֶׁ֥ר לְבִנְיָמִֽן׃ (מא) וַיָּסֻ֣רוּ שָׁ֔ם לָב֖וֹא לָל֣וּן בַּגִּבְעָ֑ה וַיָּבֹ֗א וַיֵּ֙שֶׁב֙ בִּרְח֣וֹב הָעִ֔יר וְאֵ֥ין אִ֛ישׁ מְאַסֵּֽף־ אוֹתָ֥ם הַבַּ֖יְתָה לָלֽוּן׃ (מב) וְהִנֵּ֣ה ׀ אִ֣ישׁ זָקֵ֗ן בָּ֣א מִֽן־ מַעֲשֵׂ֤הוּ מִן־ הַשָּׂדֶה֙ בָּעֶ֔רֶב וְהָאִישׁ֙ מֵהַ֣ר אֶפְרַ֔יִם וְהוּא־ גָ֖ר בַּגִּבְעָ֑ה וְאַנְשֵׁ֥י הַמָּק֖וֹם בְּנֵ֥י יְמִינִֽי׃ (מג) וַיִּשָּׂ֣א עֵינָ֗יו וַיַּ֛רְא אֶת־ הָאִ֥ישׁ הָאֹרֵ֖חַ בִּרְחֹ֣ב הָעִ֑יר וַיֹּ֨אמֶר הָאִ֧ישׁ הַזָּקֵ֛ן אָ֥נָה תֵלֵ֖ךְ וּמֵאַ֥יִן תָּבֽוֹא׃ (מד) וַיֹּ֣אמֶר אֵלָ֗יו עֹבְרִ֨ים אֲנַ֜חְנוּ מִבֵּֽית־ לֶ֣חֶם יְהוּדָה֮ עַד־ יַרְכְּתֵ֣י הַר־ אֶפְרַיִם֒ מִשָּׁ֣ם אָנֹ֔כִי וָאֵלֵ֕ךְ עַד־ בֵּ֥ית לֶ֖חֶם יְהוּדָ֑ה וְאֶת־ בֵּ֤ית יְהוָה֙ אֲנִ֣י הֹלֵ֔ךְ וְאֵ֣ין אִ֔ישׁ מְאַסֵּ֥ף אוֹתִ֖י הַבָּֽיְתָה׃ (מה) וְגַם־ תֶּ֤בֶן גַּם־ מִסְפּוֹא֙ יֵ֣שׁ לַחֲמוֹרֵ֔ינוּ וְ֠גַם לֶ֣חֶם וָיַ֤יִן יֶשׁ־ לִי֙ וְלַֽאֲמָתֶ֔ךָ וְלַנַּ֖עַר עִם־ עֲבָדֶ֑יךָ אֵ֥ין מַחְס֖וֹר כָּל־ דָּבָֽר׃
