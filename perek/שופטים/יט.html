שופטים פרק יט לפי החלוקה לסדרים

פרק יט בספר שופטים (30 פסוקים) נמצא בשופטים סדר יב (פסוקים כז–מה) ובשופטים סדר יג (פסוקים א–יא).

שופטים סדר יב

(א) וַיֹּ֧אמֶר לָהֶ֛ם הַכֹּהֵ֖ן לְכ֣וּ לְשָׁל֑וֹם נֹ֣כַח יְהוָ֔ה דַּרְכְּכֶ֖ם אֲשֶׁ֥ר תֵּֽלְכוּ־ בָֽהּ׃ (ב) וַיֵּלְכוּ֙ חֲמֵ֣שֶׁת הָאֲנָשִׁ֔ים וַיָּבֹ֖אוּ לָ֑יְשָׁה וַיִּרְא֣וּ אֶת־ הָעָ֣ם אֲשֶׁר־ בְּקִרְבָּ֣הּ יוֹשֶֽׁבֶת־ לָ֠בֶטַח כְּמִשְׁפַּ֨ט צִדֹנִ֜ים שֹׁקֵ֣ט ׀ וּבֹטֵ֗חַ וְאֵין־ מַכְלִ֨ים דָּבָ֤ר בָּאָ֙רֶץ֙ יוֹרֵ֣שׁ עֶ֔צֶר וּרְחֹקִ֥ים הֵ֙מָּה֙ מִצִּ֣דֹנִ֔ים וְדָבָ֥ר אֵין־ לָהֶ֖ם עִם־ אָדָֽם׃ (ג) וַיָּבֹ֙אוּ֙ אֶל־ אֲחֵיהֶ֔ם צָרְעָ֖ה וְאֶשְׁתָּאֹ֑ל וַיֹּאמְר֥וּ לָהֶ֛ם אֲחֵיהֶ֖ם מָ֥ה אַתֶּֽם׃ (ד) וַיֹּאמְר֗וּ ק֚וּמָה וְנַעֲלֶ֣ה עֲלֵיהֶ֔ם כִּ֤י רָאִ֙ינוּ֙ אֶת־ הָאָ֔רֶץ וְהִנֵּ֥ה טוֹבָ֖ה מְאֹ֑ד וְאַתֶּ֣ם מַחְשִׁ֔ים אַל־ תֵּעָ֣צְל֔וּ לָלֶ֥כֶת לָבֹ֖א לָרֶ֥שֶׁת אֶת־ הָאָֽרֶץ׃ (ה) כְּבֹאֲכֶ֞ם תָּבֹ֣אוּ ׀ אֶל־ עַ֣ם בֹּטֵ֗חַ וְהָאָ֙רֶץ֙ רַחֲבַ֣ת יָדַ֔יִם כִּֽי־ נְתָנָ֥הּ אֱלֹהִ֖ים בְּיֶדְכֶ֑ם מָקוֹם֙ אֲשֶׁ֣ר אֵֽין־ שָׁ֣ם מַחְס֔וֹר כָּל־ דָּבָ֖ר אֲשֶׁ֥ר בָּאָֽרֶץ׃ (ו) וַיִּסְע֤וּ מִשָּׁם֙ מִמִּשְׁפַּ֣חַת הַדָּנִ֔י מִצָּרְעָ֖ה וּמֵאֶשְׁתָּאֹ֑ל שֵֽׁשׁ־ מֵא֣וֹת אִ֔ישׁ חָג֖וּר כְּלֵ֥י מִלְחָמָֽה׃ (ז) וַֽיַּעֲל֗וּ וַֽיַּחֲנ֛וּ בְּקִרְיַ֥ת יְעָרִ֖ים בִּֽיהוּדָ֑ה עַל־ כֵּ֡ן קָרְאוּ֩ לַמָּק֨וֹם הַה֜וּא מַחֲנֵה־ דָ֗ן עַ֚ד הַיּ֣וֹם הַזֶּ֔ה הִנֵּ֕ה אַחֲרֵ֖י קִרְיַ֥ת יְעָרִֽים׃ (ח) וַיַּעַבְר֥וּ מִשָּׁ֖ם הַר־ אֶפְרָ֑יִם וַיָּבֹ֖אוּ עַד־ בֵּ֥ית מִיכָֽה׃ (ט) וַֽיַּעֲנ֞וּ חֲמֵ֣שֶׁת הָאֲנָשִׁ֗ים הַהֹלְכִים֮ לְרַגֵּל֮ אֶת־ הָאָ֣רֶץ לַיִשׁ֒ וַיֹּֽאמְרוּ֙ אֶל־ אֲחֵיהֶ֔ם הַיְדַעְתֶּ֗ם כִּ֠י יֵ֣שׁ בַּבָּתִּ֤ים הָאֵ֙לֶּה֙ אֵפ֣וֹד וּתְרָפִ֔ים וּפֶ֖סֶל וּמַסֵּכָ֑ה וְעַתָּ֖ה דְּע֥וּ מַֽה־ תַּעֲשֽׂוּ׃ (י) וַיָּס֣וּרוּ שָׁ֔מָּה וַיָּבֹ֛אוּ אֶל־ בֵּֽית־ הַנַּ֥עַר הַלֵּוִ֖י בֵּ֣ית מִיכָ֑ה וַיִּשְׁאֲלוּ־ ל֖וֹ לְשָׁלֽוֹם׃ (יא) וְשֵׁשׁ־ מֵא֣וֹת אִ֗ישׁ חֲגוּרִים֙ כְּלֵ֣י מִלְחַמְתָּ֔ם נִצָּבִ֖ים פֶּ֣תַח הַשָּׁ֑עַר אֲשֶׁ֖ר מִבְּנֵי־ דָֽן׃ (יב) וַֽיַּעֲל֞וּ חֲמֵ֣שֶׁת הָאֲנָשִׁ֗ים הַהֹלְכִים֮ לְרַגֵּ֣ל אֶת־ הָאָרֶץ֒ בָּ֣אוּ שָׁ֔מָּה לָקְח֗וּ אֶת־ הַפֶּ֙סֶל֙ וְאֶת־ הָ֣אֵפ֔וֹד וְאֶת־ הַתְּרָפִ֖ים וְאֶת־ הַמַּסֵּכָ֑ה וְהַכֹּהֵ֗ן נִצָּב֙ פֶּ֣תַח הַשַּׁ֔עַר וְשֵׁשׁ־ מֵא֣וֹת הָאִ֔ישׁ הֶחָג֖וּר כְּלֵ֥י הַמִּלְחָמָֽה׃ (יג) וְאֵ֗לֶּה בָּ֚אוּ בֵּ֣ית מִיכָ֔ה וַיִּקְחוּ֙ אֶת־ פֶּ֣סֶל הָאֵפ֔וֹד וְאֶת־ הַתְּרָפִ֖ים וְאֶת־ הַמַּסֵּכָ֑ה וַיֹּ֤אמֶר אֲלֵיהֶם֙ הַכֹּהֵ֔ן מָ֥ה אַתֶּ֖ם עֹשִֽׂים׃ (יד) וַיֹּאמְרוּ֩ ל֨וֹ הַחֲרֵ֜שׁ שִֽׂים־ יָדְךָ֤ עַל־ פִּ֙יךָ֙ וְלֵ֣ךְ עִמָּ֔נוּ וֶֽהְיֵה־ לָ֖נוּ לְאָ֣ב וּלְכֹהֵ֑ן הֲט֣וֹב ׀ הֱיוֹתְךָ֣ כֹהֵ֗ן לְבֵית֙ אִ֣ישׁ אֶחָ֔ד א֚וֹ הֱיוֹתְךָ֣ כֹהֵ֔ן לְשֵׁ֥בֶט וּלְמִשְׁפָּחָ֖ה בְּיִשְׂרָאֵֽל׃ (טו) וַיִּיטַב֙ לֵ֣ב הַכֹּהֵ֔ן וַיִּקַּח֙ אֶת־ הָ֣אֵפ֔וֹד וְאֶת־ הַתְּרָפִ֖ים וְאֶת־ הַפָּ֑סֶל וַיָּבֹ֖א בְּקֶ֥רֶב הָעָֽם׃ (טז) וַיִּפְנ֖וּ וַיֵּלֵ֑כוּ וַיָּשִׂ֨ימוּ אֶת־ הַטַּ֧ף וְאֶת־ הַמִּקְנֶ֛ה וְאֶת־ הַכְּבוּדָּ֖ה לִפְנֵיהֶֽם׃ (יז) הֵ֥מָּה הִרְחִ֖יקוּ מִבֵּ֣ית מִיכָ֑ה וְהָאֲנָשִׁ֗ים אֲשֶׁ֤ר בַּבָּתִּים֙ אֲשֶׁר֙ עִם־ בֵּ֣ית מִיכָ֔ה נִֽזְעֲק֔וּ וַיַּדְבִּ֖יקוּ אֶת־ בְּנֵי־ דָֽן׃ (יח) וַֽיִּקְרְאוּ֙ אֶל־ בְּנֵי־ דָ֔ן וַיַּסֵּ֖בּוּ פְּנֵיהֶ֑ם וַיֹּאמְר֣וּ לְמִיכָ֔ה מַה־ לְּךָ֖ כִּ֥י נִזְעָֽקְתָּ׃ (יט) וַיֹּ֡אמֶר אֶת־ אֱלֹהַי֩ אֲשֶׁר־ עָשִׂ֨יתִי לְקַחְתֶּ֧ם וְֽאֶת־ הַכֹּהֵ֛ן וַתֵּלְכ֖וּ וּמַה־ לִּ֣י ע֑וֹד וּמַה־ זֶּ֛ה תֹּאמְר֥וּ אֵלַ֖י מַה־ לָּֽךְ׃ (כ) וַיֹּאמְר֤וּ אֵלָיו֙ בְּנֵי־ דָ֔ן אַל־ תַּשְׁמַ֥ע קוֹלְךָ֖ עִמָּ֑נוּ פֶּֽן־ יִפְגְּע֣וּ בָכֶ֗ם אֲנָשִׁים֙ מָ֣רֵי נֶ֔פֶשׁ וְאָסַפְתָּ֥ה נַפְשְׁךָ֖ וְנֶ֥פֶשׁ בֵּיתֶֽךָ׃ (כא) וַיֵּלְכ֥וּ בְנֵי־ דָ֖ן לְדַרְכָּ֑ם וַיַּ֣רְא מִיכָ֗ה כִּי־ חֲזָקִ֥ים הֵ֙מָּה֙ מִמֶּ֔נּוּ וַיִּ֖פֶן וַיָּ֥שָׁב אֶל־ בֵּיתֽוֹ׃ (כב) וְהֵ֨מָּה לָקְח֜וּ אֵ֧ת אֲשֶׁר־ עָשָׂ֣ה מִיכָ֗ה וְֽאֶת־ הַכֹּהֵן֮ אֲשֶׁ֣ר הָיָה־ לוֹ֒ וַיָּבֹ֣אוּ עַל־ לַ֗יִשׁ עַל־ עַם֙ שֹׁקֵ֣ט וּבֹטֵ֔חַ וַיַּכּ֥וּ אוֹתָ֖ם לְפִי־ חָ֑רֶב וְאֶת־ הָעִ֖יר שָׂרְפ֥וּ בָאֵֽשׁ׃ (כג) וְאֵ֨ין מַצִּ֜יל כִּ֧י רְֽחוֹקָה־ הִ֣יא מִצִּיד֗וֹן וְדָבָ֤ר אֵין־ לָהֶם֙ עִם־ אָדָ֔ם וְהִ֕יא בָּעֵ֖מֶק אֲשֶׁ֣ר לְבֵית־ רְח֑וֹב וַיִּבְנ֥וּ אֶת־ הָעִ֖יר וַיֵּ֥שְׁבוּ בָֽהּ׃ (כד) וַיִּקְרְא֤וּ שֵׁם־ הָעִיר֙ דָּ֔ן בְּשֵׁם֙ דָּ֣ן אֲבִיהֶ֔ם אֲשֶׁ֥ר יוּלַּ֖ד לְיִשְׂרָאֵ֑ל וְאוּלָ֛ם לַ֥יִשׁ שֵׁם־ הָעִ֖יר לָרִאשֹׁנָֽה׃ (כה) וַיָּקִ֧ימוּ לָהֶ֛ם בְּנֵי־ דָ֖ן אֶת־ הַפָּ֑סֶל וִ֠יהוֹנָתָן בֶּן־ גֵּרְשֹׁ֨ם בֶּן־ מְ ה֣וּא וּבָנָ֗יו הָי֤וּ כֹהֲנִים֙ לְשֵׁ֣בֶט הַדָּנִ֔י עַד־ י֖וֹם גְּל֥וֹת הָאָֽרֶץ׃ (כו) וַיָּשִׂ֣ימוּ לָהֶ֔ם אֶת־ פֶּ֥סֶל מִיכָ֖ה אֲשֶׁ֣ר עָשָׂ֑ה כָּל־ יְמֵ֛י הֱי֥וֹת בֵּית־ הָאֱלֹהִ֖ים בְּשִׁלֹֽה׃ תחילת פרק יט (כז) וַיְהִי֙ בַּיָּמִ֣ים הָהֵ֔ם וּמֶ֖לֶךְ אֵ֣ין בְּיִשְׂרָאֵ֑ל וַיְהִ֣י ׀ אִ֣ישׁ לֵוִ֗י גָּ֚ר בְּיַרְכְּתֵ֣י הַר־ אֶפְרַ֔יִם וַיִּֽקַּֽח־ לוֹ֙ אִשָּׁ֣ה פִילֶ֔גֶשׁ מִבֵּ֥ית לֶ֖חֶם יְהוּדָֽה׃ (כח) וַתִּזְנֶ֤ה עָלָיו֙ פִּֽילַגְשׁ֔וֹ וַתֵּ֤לֶךְ מֵֽאִתּוֹ֙ אֶל־ בֵּ֣ית אָבִ֔יהָ אֶל־ בֵּ֥ית לֶ֖חֶם יְהוּדָ֑ה וַתְּהִי־ שָׁ֕ם יָמִ֖ים אַרְבָּעָ֥ה חֳדָשִֽׁים׃ (כט) וַיָּ֨קָם אִישָׁ֜הּ וַיֵּ֣לֶךְ אַחֲרֶ֗יהָ לְדַבֵּ֤ר עַל־ לִבָּהּ֙ (להשיבו) לַהֲשִׁיבָ֔הּ וְנַעֲר֥וֹ עִמּ֖וֹ וְצֶ֣מֶד חֲמֹרִ֑ים וַתְּבִיאֵ֙הוּ֙ בֵּ֣ית אָבִ֔יהָ וַיִּרְאֵ֙הוּ֙ אֲבִ֣י הַֽנַּעֲרָ֔ה וַיִּשְׂמַ֖ח לִקְרָאתֽוֹ׃ (ל) וַיֶּחֱזַק־ בּ֤וֹ חֹֽתְנוֹ֙ אֲבִ֣י הַֽנַּעֲרָ֔ה וַיֵּ֥שֶׁב אִתּ֖וֹ שְׁלֹ֣שֶׁת יָמִ֑ים וַיֹּאכְלוּ֙ וַיִּשְׁתּ֔וּ וַיָּלִ֖ינוּ שָֽׁם׃ (לא) וַֽיְהִי֙ בַּיּ֣וֹם הָרְבִיעִ֔י וַיַּשְׁכִּ֥ימוּ בַבֹּ֖קֶר וַיָּ֣קָם לָלֶ֑כֶת וַיֹּאמֶר֩ אֲבִ֨י הַֽנַּעֲרָ֜ה אֶל־ חֲתָנ֗וֹ סְעָ֧ד לִבְּךָ֛ פַּת־ לֶ֖חֶם וְאַחַ֥ר תֵּלֵֽכוּ׃ (לב) וַיֵּשְׁב֗וּ וַיֹּאכְל֧וּ שְׁנֵיהֶ֛ם יַחְדָּ֖ו וַיִּשְׁתּ֑וּ וַיֹּ֜אמֶר אֲבִ֤י הַֽנַּעֲרָה֙ אֶל־ הָאִ֔ישׁ הֽוֹאֶל־ נָ֥א וְלִ֖ין וְיִטַ֥ב לִבֶּֽךָ׃ (לג) וַיָּ֥קָם הָאִ֖ישׁ לָלֶ֑כֶת וַיִּפְצַר־ בּוֹ֙ חֹתְנ֔וֹ וַיָּ֖שָׁב וַיָּ֥לֶן שָֽׁם׃ (לד) וַיַּשְׁכֵּ֨ם בַּבֹּ֜קֶר בַּיּ֣וֹם הַחֲמִישִׁי֮ לָלֶכֶת֒ וַיֹּ֣אמֶר ׀ אֲבִ֣י הַֽנַּעֲרָ֗ה סְעָד־ נָא֙ לְבָ֣בְךָ֔ וְהִֽתְמַהְמְה֖וּ עַד־ נְט֣וֹת הַיּ֑וֹם וַיֹּאכְל֖וּ שְׁנֵיהֶֽם׃ (לה) וַיָּ֤קָם הָאִישׁ֙ לָלֶ֔כֶת ה֥וּא וּפִילַגְשׁ֖וֹ וְנַעֲר֑וֹ וַיֹּ֣אמֶר ל֣וֹ חֹתְנ֣וֹ אֲבִ֣י הַֽנַּעֲרָ֡ה הִנֵּ֣ה נָא֩ רָפָ֨ה הַיּ֜וֹם לַעֲרֹ֗ב לִֽינוּ־ נָ֞א הִנֵּ֨ה חֲנ֤וֹת הַיּוֹם֙ לִ֥ין פֹּה֙ וְיִיטַ֣ב וַיֹּאכְל֖וּ וַיִּשְׁתּֽוּ׃ (ג) הֵמָּה֮ מֵיטִיבִ֣ים אֶת־ לִבָּם֒ וְהִנֵּה֩ אַנְשֵׁ֨י הָעִ֜יר אַנְשֵׁ֣י בְנֵֽי־ בְלִיַּ֗עַל נָסַ֙בּוּ֙ אֶת־ הַבַּ֔יִת מִֽתְדַּפְּקִ֖ים עַל־ הַדָּ֑לֶת וַיֹּאמְר֗וּ אֶל־ הָ֠אִישׁ בַּ֣עַל הַבַּ֤יִת הַזָּקֵן֙ לֵאמֹ֔ר הוֹצֵ֗א אֶת־ הָאִ֛ישׁ אֲשֶׁר־ בָּ֥א אֶל־ בֵּיתְךָ֖ וְנֵדָעֶֽנּוּ׃ (ד) וַיֵּצֵ֣א אֲלֵיהֶ֗ם הָאִישׁ֙ בַּ֣עַל הַבַּ֔יִת וַיֹּ֣אמֶר אֲלֵהֶ֔ם אַל־ אַחַ֖י אַל־ לָעָ֑ם לַעֲשׂ֗וֹת לְבוֹאָם֙ לְגֶ֣בַע בִּנְיָמִ֔ן כְּכָל־ הַ֨נְּבָלָ֔ה אֲשֶׁ֥ר עָשָׂ֖ה בְּיִשְׂרָאֵֽל׃ (כב) וַיֵּֽאָסֵ֞ף כָּל־ אִ֤ישׁ יִשְׂרָאֵל֙ אֶל־ הָעִ֔יר כְּאִ֥ישׁ אֶחָ֖ד חֲבֵרִֽים׃ (כג) וַֽיִּשְׁלְח֞וּ שִׁבְטֵ֤י יִשְׂרָאֵל֙ אֲנָשִׁ֔ים בְּכָל־ שִׁבְטֵ֥י בִנְיָמִ֖ן לֵאמֹ֑ר מָ֚ה הָרָעָ֣ה הַזֹּ֔את אֲשֶׁ֥ר נִהְיְתָ֖ה בָּכֶֽם׃ (כד) וְעַתָּ֡ה תְּנוּ֩ אֶת־ הָאֲנָשִׁ֨ים בְּנֵֽי־ בְלִיַּ֜עַל אֲשֶׁ֤ר בַּגִּבְעָה֙ וּנְמִיתֵ֔ם וּנְבַעֲרָ֥ה רָעָ֖ה מִיִּשְׂרָאֵ֑ל וְלֹ֤א אָבוּ֙ בְּנֵ֣י בִּנְיָמִ֔ן לִשְׁמֹ֕עַ בְּק֖וֹל אֲחֵיהֶ֥ם בְּנֵֽי־ יִשְׂרָאֵֽל׃ (כה) וַיֵּאָסְפ֧וּ בְנֵֽי־ בִנְיָמִ֛ן מִן־ הֶעָרִ֖ים הַגִּבְעָ֑תָה לָצֵ֥את לַמִּלְחָמָ֖ה עִם־ בְּנֵ֥י יִשְׂרָאֵֽל׃ (כו) וַיִּתְפָּֽקְדוּ֩ בְנֵ֨י בִנְיָמִ֜ן בַּיּ֤וֹם הַהוּא֙ מֵהֶ֣עָרִ֔ים עֶשְׂרִ֨ים וְשִׁשָּׁ֥ה אֶ֛לֶף אִ֖ישׁ שֹׁ֣לֵֽף חָ֑רֶב לְ֠בַד מִיֹּשְׁבֵ֤י הַגִּבְעָה֙ הִתְפָּ֣קְד֔וּ שְׁבַ֥ע מֵא֖וֹת אִ֥ישׁ בָּחֽוּר׃ (כז) מִכֹּ֣ל ׀ הָעָ֣ם הַזֶּ֗ה שְׁבַ֤ע מֵאוֹת֙ אִ֣ישׁ בָּח֔וּר אִטֵּ֖ר יַד־ יְמִינ֑וֹ כָּל־ זֶ֗ה קֹלֵ֧עַ בָּאֶ֛בֶן אֶל־ הַֽשַּׂעֲרָ֖ה וְלֹ֥א יַחֲטִֽא׃ (כח) וְאִ֨ישׁ יִשְׂרָאֵ֜ל הִתְפָּֽקְד֗וּ לְבַד֙ מִבִּנְיָמִ֔ן אַרְבַּ֨ע מֵא֥וֹת אֶ֛לֶף אִ֖ישׁ שֹׁ֣לֵֽף חָ֑רֶב כָּל־ זֶ֖ה אִ֥ישׁ מִלְחָמָֽה׃ (כט) וַיָּקֻ֜מוּ וַיַּעֲל֣וּ בֵֽית־ אֵל֮ וַיִּשְׁאֲל֣וּ בֵאלֹהִים֒ וַיֹּֽאמְרוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל מִ֚י יַעֲלֶה־ לָּ֣נוּ בַתְּחִלָּ֔ה לַמִּלְחָמָ֖ה עִם־ בְּנֵ֣י בִנְיָמִ֑ן וַיֹּ֥אמֶר יְהוָ֖ה יְהוּדָ֥ה בַתְּחִלָּֽה׃ (ל) וַיָּק֥וּמוּ בְנֵֽי־ יִשְׂרָאֵ֖ל בַּבֹּ֑קֶר וַיַּֽחֲנ֖וּ עַל־ הַגִּבְעָֽה׃ (לא) וַיֵּצֵא֙ אִ֣ישׁ יִשְׂרָאֵ֔ל לַמִּלְחָמָ֖ה עִם־ בִּנְיָמִ֑ן וַיַּעַרְכ֨וּ אִתָּ֧ם אִֽישׁ־ יִשְׂרָאֵ֛ל מִלְחָמָ֖ה אֶל־ הַגִּבְעָֽה׃ (לב) וַיֵּצְא֥וּ בְנֵֽי־ בִנְיָמִ֖ן מִן־ הַגִּבְעָ֑ה וַיַּשְׁחִ֨יתוּ בְיִשְׂרָאֵ֜ל בַּיּ֣וֹם הַה֗וּא שְׁנַ֨יִם וְעֶשְׂרִ֥ים אֶ֛לֶף אִ֖ישׁ אָֽרְצָה׃ (לג) וַיִּתְחַזֵּ֥ק הָעָ֖ם אִ֣ישׁ יִשְׂרָאֵ֑ל וַיֹּסִ֙פוּ֙ לַעֲרֹ֣ךְ מִלְחָמָ֔ה בַּמָּק֕וֹם אֲשֶׁר־ עָ֥רְכוּ שָׁ֖ם בַּיּ֥וֹם הָרִאשֽׁוֹן׃ (לד) וַיַּעֲל֣וּ בְנֵֽי־ יִשְׂרָאֵ֗ל וַיִּבְכּ֣וּ לִפְנֵֽי־ יְהוָה֮ עַד־ הָעֶרֶב֒ וַיִּשְׁאֲל֤וּ בַֽיהוָה֙ לֵאמֹ֔ר הַאוֹסִ֗יף לָגֶ֙שֶׁת֙ לַמִּלְחָמָ֔ה עִם־ בְּנֵ֥י בִנְיָמִ֖ן אָחִ֑י וַיֹּ֥אמֶר יְהוָ֖ה עֲל֥וּ אֵלָֽיו׃ (לה) וַיִּקְרְב֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶל־ בְּנֵ֥י בִנְיָמִ֖ן בַּיּ֥וֹם הַשֵּׁנִֽי׃ (לו) וַיֵּצֵא֩ בִנְיָמִ֨ן ׀ לִקְרָאתָ֥ם ׀ מִֽן־ הַגִּבְעָה֮ בַּיּ֣וֹם הַשֵּׁנִי֒ וַיַּשְׁחִיתוּ֩ בִבְנֵ֨י יִשְׂרָאֵ֜ל ע֗וֹד שְׁמֹנַ֨ת עָשָׂ֥ר אֶ֛לֶף אִ֖ישׁ אָ֑רְצָה כָּל־ אֵ֖לֶּה שֹׁ֥לְפֵי חָֽרֶב׃ (לז) וַיַּעֲל֣וּ כָל־ בְּנֵי֩ יִשְׂרָאֵ֨ל וְכָל־ הָעָ֜ם וַיָּבֹ֣אוּ בֵֽית־ אֵ֗ל וַיִּבְכּוּ֙ וַיֵּ֤שְׁבוּ שָׁם֙ לִפְנֵ֣י יְהוָ֔ה וַיָּצ֥וּמוּ בַיּוֹם־ הַה֖וּא עַד־ הָעָ֑רֶב וַֽיַּעֲל֛וּ עֹל֥וֹת וּשְׁלָמִ֖ים לִפְנֵ֥י יְהוָֽה׃
