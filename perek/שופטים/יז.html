שופטים פרק יז לפי החלוקה לסדרים

פרק יז בספר שופטים (13 פסוקים) נמצא בשופטים סדר יא (פסוקים ל–מב).

שופטים סדר יא

(א) וַיִּשְׁכַּ֣ב שִׁמְשׁוֹן֮ עַד־ חֲצִ֣י הַלַּיְלָה֒ וַיָּ֣קָם ׀ בַּחֲצִ֣י הַלַּ֗יְלָה וַיֶּאֱחֹ֞ז בְּדַלְת֤וֹת שַֽׁעַר־ הָעִיר֙ וּבִשְׁתֵּ֣י הַמְּזוּז֔וֹת וַיִּסָּעֵם֙ עִֽם־ הַבְּרִ֔יחַ וַיָּ֖שֶׂם עַל־ כְּתֵפָ֑יו וַֽיַּעֲלֵם֙ אֶל־ רֹ֣אשׁ הָהָ֔ר אֲשֶׁ֖ר עַל־ פְּנֵ֥י חֶבְרֽוֹן׃ (ב) וַֽיְהִי֙ אַחֲרֵי־ כֵ֔ן וַיֶּאֱהַ֥ב אִשָּׁ֖ה בְּנַ֣חַל שֹׂרֵ֑ק וּשְׁמָ֖הּ דְּלִילָֽה׃ (ג) וַיַּעֲל֨וּ אֵלֶ֜יהָ סַרְנֵ֣י פְלִשְׁתִּ֗ים וַיֹּ֨אמְרוּ לָ֜הּ פַּתִּ֣י אוֹת֗וֹ וּרְאִי֙ בַּמֶּה֙ כֹּח֣וֹ גָד֔וֹל וּבַמֶּה֙ נ֣וּכַל ל֔וֹ וַאֲסַרְנֻ֖הוּ לְעַנֹּת֑וֹ וַאֲנַ֙חְנוּ֙ נִתַּן־ לָ֔ךְ אִ֕ישׁ אֶ֥לֶף וּמֵאָ֖ה כָּֽסֶף׃ (ד) וַתֹּ֤אמֶר דְּלִילָה֙ אֶל־ שִׁמְשׁ֔וֹן הַגִּֽידָה־ נָּ֣א לִ֔י בַּמֶּ֖ה כֹּחֲךָ֣ גָד֑וֹל וּבַמֶּ֥ה תֵאָסֵ֖ר לְעַנּוֹתֶֽךָ׃ (ה) וַיֹּ֤אמֶר אֵלֶ֙יהָ֙ שִׁמְשׁ֔וֹן אִם־ יַאַסְרֻ֗נִי בְּשִׁבְעָ֛ה יְתָרִ֥ים לַחִ֖ים אֲשֶׁ֣ר לֹא־ חֹרָ֑בוּ וְחָלִ֥יתִי וְהָיִ֖יתִי כְּאַחַ֥ד הָאָדָֽם׃ (ו) וַיַּעֲלוּ־ לָ֞הּ סַרְנֵ֣י פְלִשְׁתִּ֗ים שִׁבְעָ֛ה יְתָרִ֥ים לַחִ֖ים אֲשֶׁ֣ר לֹא־ חֹרָ֑בוּ וַתַּאַסְרֵ֖הוּ בָּהֶֽם׃ (ז) וְהָאֹרֵ֗ב יֹשֵׁ֥ב לָהּ֙ בַּחֶ֔דֶר וַתֹּ֣אמֶר אֵלָ֔יו פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיְנַתֵּק֙ אֶת־ הַיְתָרִ֔ים כַּאֲשֶׁ֨ר יִנָּתֵ֤ק פְּתִֽיל־ הַנְּעֹ֙רֶת֙ בַּהֲרִיח֣וֹ אֵ֔שׁ וְלֹ֥א נוֹדַ֖ע כֹּחֽוֹ׃ (ח) וַתֹּ֤אמֶר דְּלִילָה֙ אֶל־ שִׁמְשׁ֔וֹן הִנֵּה֙ הֵתַ֣לְתָּ בִּ֔י וַתְּדַבֵּ֥ר אֵלַ֖י כְּזָבִ֑ים עַתָּה֙ הַגִּֽידָה־ נָּ֣א לִ֔י בַּמֶּ֖ה תֵּאָסֵֽר׃ (ט) וַיֹּ֣אמֶר אֵלֶ֔יהָ אִם־ אָס֤וֹר יַאַסְר֙וּנִי֙ בַּעֲבֹתִ֣ים חֲדָשִׁ֔ים אֲשֶׁ֛ר לֹֽא־ נַעֲשָׂ֥ה בָהֶ֖ם מְלָאכָ֑ה וְחָלִ֥יתִי וְהָיִ֖יתִי כְּאַחַ֥ד הָאָדָֽם׃ (י) וַתִּקַּ֣ח דְּלִילָה֩ עֲבֹתִ֨ים חֲדָשִׁ֜ים וַתַּאַסְרֵ֣הוּ בָהֶ֗ם וַתֹּ֤אמֶר אֵלָיו֙ פְּלִשְׁתִּ֤ים עָלֶ֙יךָ֙ שִׁמְשׁ֔וֹן וְהָאֹרֵ֖ב יֹשֵׁ֣ב בֶּחָ֑דֶר וַֽיְנַתְּקֵ֛ם מֵעַ֥ל זְרֹעֹתָ֖יו כַּחֽוּט׃ (יא) וַתֹּ֨אמֶר דְּלִילָ֜ה אֶל־ שִׁמְשׁ֗וֹן עַד־ הֵ֜נָּה הֵתַ֤לְתָּ בִּי֙ וַתְּדַבֵּ֤ר אֵלַי֙ כְּזָבִ֔ים הַגִּ֣ידָה לִּ֔י בַּמֶּ֖ה תֵּאָסֵ֑ר וַיֹּ֣אמֶר אֵלֶ֔יהָ אִם־ תַּאַרְגִ֗י אֶת־ שֶׁ֛בַע מַחְלְפ֥וֹת רֹאשִׁ֖י עִם־ הַמַּסָּֽכֶת׃ (יב) וַתִּתְקַע֙ בַּיָּתֵ֔ד וַתֹּ֣אמֶר אֵלָ֔יו פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיִּיקַץ֙ מִשְּׁנָת֔וֹ וַיִּסַּ֛ע אֶת־ הַיְתַ֥ד הָאֶ֖רֶג וְאֶת־ הַמַּסָּֽכֶת׃ (יג) וַתֹּ֣אמֶר אֵלָ֗יו אֵ֚יךְ תֹּאמַ֣ר אֲהַבְתִּ֔יךְ וְלִבְּךָ֖ אֵ֣ין אִתִּ֑י זֶ֣ה שָׁלֹ֤שׁ פְּעָמִים֙ הֵתַ֣לְתָּ בִּ֔י וְלֹא־ הִגַּ֣דְתָּ לִּ֔י בַּמֶּ֖ה כֹּחֲךָ֥ גָדֽוֹל׃ (יד) וַ֠יְהִי כִּֽי־ הֵצִ֨יקָה לּ֧וֹ בִדְבָרֶ֛יהָ כָּל־ הַיָּמִ֖ים וַתְּאַֽלֲצֵ֑הוּ וַתִּקְצַ֥ר נַפְשׁ֖וֹ לָמֽוּת׃ (טו) וַיַּגֶּד־ לָ֣הּ אֶת־ כָּל־ לִבּ֗וֹ וַיֹּ֤אמֶר לָהּ֙ מוֹרָה֙ לֹֽא־ עָלָ֣ה עַל־ רֹאשִׁ֔י כִּֽי־ נְזִ֧יר אֱלֹהִ֛ים אֲנִ֖י מִבֶּ֣טֶן אִמִּ֑י אִם־ גֻּלַּ֙חְתִּי֙ וְסָ֣ר מִמֶּ֣נִּי כֹחִ֔י וְחָלִ֥יתִי וְהָיִ֖יתִי כְּכָל־ הָאָדָֽם׃ (טז) וַתֵּ֣רֶא דְלִילָ֗ה כִּֽי־ הִגִּ֣יד לָהּ֮ אֶת־ כָּל־ לִבּוֹ֒ וַתִּשְׁלַ֡ח וַתִּקְרָא֩ לְסַרְנֵ֨י פְלִשְׁתִּ֤ים לֵאמֹר֙ עֲל֣וּ הַפַּ֔עַם כִּֽי־ הִגִּ֥יד (לה) לִ֖י אֶת־ כָּל־ לִבּ֑וֹ וְעָל֤וּ אֵלֶ֙יהָ֙ סַרְנֵ֣י פְלִשְׁתִּ֔ים וַיַּעֲל֥וּ הַכֶּ֖סֶף בְּיָדָֽם׃ (יז) וַתְּיַשְּׁנֵ֙הוּ֙ עַל־ בִּרְכֶּ֔יהָ וַתִּקְרָ֣א לָאִ֔ישׁ וַתְּגַלַּ֕ח אֶת־ שֶׁ֖בַע מַחְלְפ֣וֹת רֹאשׁ֑וֹ וַתָּ֙חֶל֙ לְעַנּוֹת֔וֹ וַיָּ֥סַר כֹּח֖וֹ מֵעָלָֽיו׃ (יח) וַתֹּ֕אמֶר פְּלִשְׁתִּ֥ים עָלֶ֖יךָ שִׁמְשׁ֑וֹן וַיִּקַ֣ץ מִשְּׁנָת֗וֹ וַיֹּ֙אמֶר֙ אֵצֵ֞א כְּפַ֤עַם בְּפַ֙עַם֙ וְאִנָּעֵ֔ר וְהוּא֙ לֹ֣א יָדַ֔ע כִּ֥י יְהוָ֖ה סָ֥ר מֵעָלָֽיו׃ (יט) וַיֹּאחֲז֣וּהוּ פְלִשְׁתִּ֔ים וַֽיְנַקְּר֖וּ אֶת־ עֵינָ֑יו וַיּוֹרִ֨ידוּ אוֹת֜וֹ עַזָּ֗תָה וַיַּאַסְר֙וּהוּ֙ בַּֽנְחֻשְׁתַּ֔יִם וַיְהִ֥י טוֹחֵ֖ן בְּבֵ֥ית (האסירים) הָאֲסוּרִֽים׃ (כ) וַיָּ֧חֶל שְׂעַר־ רֹאשׁ֛וֹ לְצַמֵּ֖חַ כַּאֲשֶׁ֥ר גֻּלָּֽח׃ (כא) וְסַרְנֵ֣י פְלִשְׁתִּ֗ים נֶֽאֱסְפוּ֙ לִזְבֹּ֧חַ זֶֽבַח־ גָּד֛וֹל לְדָג֥וֹן אֱלֹהֵיהֶ֖ם וּלְשִׂמְחָ֑ה וַיֹּ֣אמְר֔וּ נָתַ֤ן אֱלֹהֵ֙ינוּ֙ בְּיָדֵ֔נוּ אֵ֖ת שִׁמְשׁ֥וֹן אוֹיְבֵֽינוּ׃ (כב) וַיִּרְא֤וּ אֹתוֹ֙ הָעָ֔ם וַֽיְהַלְל֖וּ אֶת־ אֱלֹהֵיהֶ֑ם כִּ֣י אָמְר֗וּ נָתַ֨ן אֱלֹהֵ֤ינוּ בְיָדֵ֙נוּ֙ אֶת־ א֣וֹיְבֵ֔נוּ וְאֵת֙ מַחֲרִ֣יב אַרְצֵ֔נוּ וַאֲשֶׁ֥ר הִרְבָּ֖ה אֶת־ חֲלָלֵֽינוּ׃ (כג) וַֽיְהִי֙ (כי טוב) כְּט֣וֹב לִבָּ֔ם וַיֹּ֣אמְר֔וּ קִרְא֥וּ לְשִׁמְשׁ֖וֹן וִישַֽׂחֶק־ לָ֑נוּ וַיִּקְרְא֨וּ לְשִׁמְשׁ֜וֹן מִבֵּ֣ית (האסירים) הָאֲסוּרִ֗ים וַיְצַחֵק֙ לִפְנֵיהֶ֔ם וַיַּעֲמִ֥ידוּ אוֹת֖וֹ בֵּ֥ין הָעַמּוּדִֽים׃ (כד) וַיֹּ֨אמֶר שִׁמְשׁ֜וֹן אֶל־ הַנַּ֨עַר הַמַּחֲזִ֣יק בְּיָדוֹ֮ הַנִּ֣יחָה אוֹתִי֒ (והימשני) וַהֲמִשֵׁ֙נִי֙ אֶת־ הָֽעַמֻּדִ֔ים אֲשֶׁ֥ר הַבַּ֖יִת נָכ֣וֹן עֲלֵיהֶ֑ם וְאֶשָּׁעֵ֖ן עֲלֵיהֶֽם׃ (כה) וְהַבַּ֗יִת מָלֵ֤א הָֽאֲנָשִׁים֙ וְהַנָּשִׁ֔ים וְשָׁ֕מָּה כֹּ֖ל סַרְנֵ֣י פְלִשְׁתִּ֑ים וְעַל־ הַגָּ֗ג כִּשְׁלֹ֤שֶׁת אֲלָפִים֙ אִ֣ישׁ וְאִשָּׁ֔ה הָרֹאִ֖ים בִּשְׂח֥וֹק שִׁמְשֽׁוֹן׃ (כו) וַיִּקְרָ֥א שִׁמְשׁ֛וֹן אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר אֲדֹנָ֣י יֱהֹוִ֡ה זָכְרֵ֣נִי נָא֩ וְחַזְּקֵ֨נִי נָ֜א אַ֣ךְ הַפַּ֤עַם הַזֶּה֙ הָאֱלֹהִ֔ים וְאִנָּקְמָ֧ה נְקַם־ אַחַ֛ת מִשְּׁתֵ֥י עֵינַ֖י מִפְּלִשְׁתִּֽים׃ (כז) וַיִּלְפֹּ֨ת שִׁמְשׁ֜וֹן אֶת־ שְׁנֵ֣י ׀ עַמּוּדֵ֣י הַתָּ֗וֶךְ אֲשֶׁ֤ר הַבַּ֙יִת֙ נָכ֣וֹן עֲלֵיהֶ֔ם וַיִּסָּמֵ֖ךְ עֲלֵיהֶ֑ם אֶחָ֥ד בִּימִינ֖וֹ וְאֶחָ֥ד בִּשְׂמֹאלֽוֹ׃ (כח) וַיֹּ֣אמֶר שִׁמְשׁ֗וֹן תָּמ֣וֹת נַפְשִׁי֮ עִם־ פְּלִשְׁתִּים֒ וַיֵּ֣ט בְּכֹ֔חַ וַיִּפֹּ֤ל הַבַּ֙יִת֙ עַל־ הַסְּרָנִ֔ים וְעַל־ כָּל־ הָעָ֖ם אֲשֶׁר־ בּ֑וֹ וַיִּהְי֤וּ הַמֵּתִים֙ אֲשֶׁ֣ר הֵמִ֣ית בְּמוֹת֔וֹ רַבִּ֕ים מֵאֲשֶׁ֥ר הֵמִ֖ית בְּחַיָּֽיו׃ (כט) וַיֵּרְד֨וּ אֶחָ֜יו וְכָל־ בֵּ֣ית אָבִיהוּ֮ וַיִּשְׂא֣וּ אֹתוֹ֒ וַֽיַּעֲל֣וּ ׀ וַיִּקְבְּר֣וּ אוֹת֗וֹ בֵּ֤ין צָרְעָה֙ וּבֵ֣ין אֶשְׁתָּאֹ֔ל בְּקֶ֖בֶר מָנ֣וֹחַ אָבִ֑יו וְה֛וּא שָׁפַ֥ט אֶת־ יִשְׂרָאֵ֖ל עֶשְׂרִ֥ים שָׁנָֽה׃ תחילת פרק יז (ל) וַֽיְהִי־ אִ֥ישׁ מֵֽהַר־ אֶפְרָ֖יִם וּשְׁמ֥וֹ מִיכָֽיְהוּ׃ (לא) וַיֹּ֣אמֶר לְאִמּ֡וֹ אֶלֶף֩ וּמֵאָ֨ה הַכֶּ֜סֶף אֲשֶׁ֣ר לֻֽקַּֽח־ לָ֗ךְ (ואתי) וְאַ֤תְּ אָלִית֙ וְגַם֙ אָמַ֣רְתְּ בְּאָזְנַ֔י הִנֵּֽה־ הַכֶּ֥סֶף אִתִּ֖י אֲנִ֣י לְקַחְתִּ֑יו וַתֹּ֣אמֶר אִמּ֔וֹ בָּר֥וּךְ בְּנִ֖י לַיהוָֽה׃ (לב) וַיָּ֛שֶׁב אֶת־ אֶֽלֶף־ וּמֵאָ֥ה הַכֶּ֖סֶף לְאִמּ֑וֹ וַתֹּ֣אמֶר אִמּ֡וֹ הַקְדֵּ֣שׁ הִקְדַּ֣שְׁתִּי אֶת־ הַכֶּסֶף֩ לַיהוָ֨ה מִיָּדִ֜י לִבְנִ֗י לַֽעֲשׂוֹת֙ פֶּ֣סֶל וּמַסֵּכָ֔ה וְעַתָּ֖ה אֲשִׁיבֶ֥נּוּ לָֽךְ׃ (לג) וַיָּ֥שֶׁב אֶת־ הַכֶּ֖סֶף לְאִמּ֑וֹ וַתִּקַּ֣ח אִמּוֹ֩ מָאתַ֨יִם כֶּ֜סֶף וַתִּתְּנֵ֣הוּ לַצּוֹרֵ֗ף וַֽיַּעֲשֵׂ֙הוּ֙ פֶּ֣סֶל וּמַסֵּכָ֔ה וַיְהִ֖י בְּבֵ֥ית מִיכָֽיְהוּ׃ (לד) וְהָאִ֣ישׁ מִיכָ֔ה ל֖וֹ בֵּ֣ית אֱלֹהִ֑ים וַיַּ֤עַשׂ אֵפוֹד֙ וּתְרָפִ֔ים וַיְמַלֵּ֗א אֶת־ יַ֤ד אַחַד֙ מִבָּנָ֔יו וַיְהִי־ ל֖וֹ לְכֹהֵֽן׃ (לה) בַּיָּמִ֣ים הָהֵ֔ם אֵ֥ין מֶ֖לֶךְ בְּיִשְׂרָאֵ֑ל אִ֛ישׁ הַיָּשָׁ֥ר בְּעֵינָ֖יו יַעֲשֶֽׂה׃ (לו) וַיְהִי־ נַ֗עַר מִבֵּ֥ית לֶ֙חֶם֙ יְהוּדָ֔ה מִמִּשְׁפַּ֖חַת יְהוּדָ֑ה וְה֥וּא לֵוִ֖י וְה֥וּא גָֽר־ שָֽׁם׃ (לז) וַיֵּ֨לֶךְ הָאִ֜ישׁ מֵהָעִ֗יר מִבֵּ֥ית לֶ֙חֶם֙ יְהוּדָ֔ה לָג֖וּר בַּאֲשֶׁ֣ר יִמְצָ֑א וַיָּבֹ֧א הַר־ אֶפְרַ֛יִם עַד־ בֵּ֥ית מִיכָ֖ה לַעֲשׂ֥וֹת דַּרְכּֽוֹ׃ (לח) וַיֹּאמֶר־ ל֥וֹ מִיכָ֖ה מֵאַ֣יִן תָּב֑וֹא וַיֹּ֨אמֶר אֵלָ֜יו לֵוִ֣י אָנֹ֗כִי מִבֵּ֥ית לֶ֙חֶם֙ יְהוּדָ֔ה וְאָנֹכִ֣י הֹלֵ֔ךְ לָג֖וּר בַּאֲשֶׁ֥ר אֶמְצָֽא׃ (לט) וַיֹּאמֶר֩ ל֨וֹ מִיכָ֜ה שְׁבָ֣ה עִמָּדִ֗י וֶֽהְיֵה־ לִי֮ לְאָ֣ב וּלְכֹהֵן֒ וְאָנֹכִ֨י אֶֽתֶּן־ לְךָ֜ עֲשֶׂ֤רֶת כֶּ֙סֶף֙ לַיָּמִ֔ים וְעֵ֥רֶךְ בְּגָדִ֖ים וּמִחְיָתֶ֑ךָ וַיֵּ֖לֶךְ הַלֵּוִֽי׃ (מ) וַיּ֥וֹאֶל הַלֵּוִ֖י לָשֶׁ֣בֶת אֶת־ הָאִ֑ישׁ וַיְהִ֤י הַנַּ֙עַר֙ ל֔וֹ כְּאַחַ֖ד מִבָּנָֽיו׃ (מא) וַיְמַלֵּ֤א מִיכָה֙ אֶת־ יַ֣ד הַלֵּוִ֔י וַיְהִי־ ל֥וֹ הַנַּ֖עַר לְכֹהֵ֑ן וַיְהִ֖י בְּבֵ֥ית מִיכָֽה׃ (מב) וַיֹּ֣אמֶר מִיכָ֔ה עַתָּ֣ה יָדַ֔עְתִּי כִּֽי־ יֵיטִ֥יב יְהוָ֖ה לִ֑י כִּ֧י הָיָה־ לִ֛י הַלֵּוִ֖י לְכֹהֵֽן׃ סוף פרק יז (מג) בַּיָּמִ֣ים הָהֵ֔ם אֵ֥ין מֶ֖לֶךְ בְּיִשְׂרָאֵ֑ל וּבַיָּמִ֣ים הָהֵ֗ם שֵׁ֣בֶט הַדָּנִ֞י מְבַקֶּשׁ־ ל֤וֹ נַֽחֲלָה֙ לָשֶׁ֔בֶת כִּי֩ לֹֽא־ נָ֨פְלָה לּ֜וֹ עַד־ הַיּ֥וֹם הַה֛וּא בְּתוֹךְ־ שִׁבְטֵ֥י יִשְׂרָאֵ֖ל בְּנַחֲלָֽה׃ (מד) וַיִּשְׁלְח֣וּ בְנֵי־ דָ֣ן ׀ מִֽמִּשְׁפַּחְתָּ֡ם חֲמִשָּׁ֣ה אֲנָשִׁ֣ים מִקְצוֹתָם֩ אֲנָשִׁ֨ים בְּנֵי־ חַ֜יִל מִצָּרְעָ֣ה וּמֵֽאֶשְׁתָּאֹ֗ל לְרַגֵּ֤ל אֶת־ הָאָ֙רֶץ֙ וּלְחָקְרָ֔הּ וַיֹּאמְר֣וּ אֲלֵהֶ֔ם לְכ֖וּ חִקְר֣וּ אֶת־ הָאָ֑רֶץ וַיָּבֹ֤אוּ הַר־ אֶפְרַ֙יִם֙ עַד־ בֵּ֣ית מִיכָ֔ה וַיָּלִ֖ינוּ שָֽׁם׃ (מה) הֵ֚מָּה עִם־ בֵּ֣ית מִיכָ֔ה וְהֵ֣מָּה הִכִּ֔ירוּ אֶת־ ק֥וֹל הַנַּ֖עַר הַלֵּוִ֑י וַיָּס֣וּרוּ שָׁ֗ם וַיֹּ֤אמְרוּ לוֹ֙ מִֽי־ הֱבִיאֲךָ֣ הֲלֹ֔ם וּמָֽה־ אַתָּ֥ה עֹשֶׂ֛ה בָּזֶ֖ה וּמַה־ לְּךָ֥ פֹֽה׃ (מו) וַיֹּ֣אמֶר אֲלֵהֶ֔ם כָּזֹ֣ה וְכָזֶ֔ה עָ֥שָׂה לִ֖י מִיכָ֑ה וַיִּשְׂכְּרֵ֕נִי וָאֱהִי־ ל֖וֹ לְכֹהֵֽן׃ (מז) וַיֹּ֥אמְרוּ ל֖וֹ שְׁאַל־ נָ֣א בֵאלֹהִ֑ים וְנֵ֣דְעָ֔ה הֲתַצְלִ֣יחַ דַּרְכֵּ֔נוּ אֲשֶׁ֥ר אֲנַ֖חְנוּ הֹלְכִ֥ים עָלֶֽיהָ׃
