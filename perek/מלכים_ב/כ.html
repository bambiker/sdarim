מלכים ב פרק כ לפי החלוקה לסדרים

פרק כ בספר מלכים ב (21 פסוקים) נמצא במלכים סדר לא (פסוקים כ–כו) ובמלכים סדר לב (פסוקים א–יד).

מלכים סדר לא

(א) וְעַתָּה֙ יְהוָ֣ה אֱלֹהֵ֔ינוּ הוֹשִׁיעֵ֥נוּ נָ֖א מִיָּד֑וֹ וְיֵֽדְעוּ֙ כָּל־ מַמְלְכ֣וֹת הָאָ֔רֶץ כִּ֥י אַתָּ֛ה יְהוָ֥ה אֱלֹהִ֖ים לְבַדֶּֽךָ׃ (ב) וַיִּשְׁלַח֙ יְשַֽׁעְיָ֣הוּ בֶן־ אָמ֔וֹץ אֶל־ חִזְקִיָּ֖הוּ לֵאמֹ֑ר כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֨ר הִתְפַּלַּ֧לְתָּ אֵלַ֛י אֶל־ סַנְחֵרִ֥ב מֶֽלֶךְ־ אַשּׁ֖וּר שָׁמָֽעְתִּי׃ (ג) זֶ֣ה הַדָּבָ֔ר אֲשֶׁר־ דִּבֶּ֥ר יְהוָ֖ה עָלָ֑יו בָּזָ֨ה לְךָ֜ לָעֲגָ֣ה לְךָ֗ בְּתוּלַת֙ בַּת־ צִיּ֔וֹן אַחֲרֶ֙יךָ֙ רֹ֣אשׁ הֵנִ֔יעָה בַּ֖ת יְרוּשָׁלִָֽם׃ (ד) אֶת־ מִ֤י חֵרַ֙פְתָּ֙ וְגִדַּ֔פְתָּ וְעַל־ מִ֖י הֲרִימ֣וֹתָ קּ֑וֹל וַתִּשָּׂ֥א מָר֛וֹם עֵינֶ֖יךָ עַל־ קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (ה) בְּיַ֣ד מַלְאָכֶיךָ֮ חֵרַ֣פְתָּ ׀ אֲדֹנָי֒ וַתֹּ֗אמֶר (ברכב) בְּרֹ֥ב רִכְבִּ֛י אֲנִ֥י עָלִ֛יתִי מְר֥וֹם הָרִ֖ים יַרְכְּתֵ֣י לְבָנ֑וֹן וְאֶכְרֹ֞ת קוֹמַ֤ת אֲרָזָיו֙ מִבְח֣וֹר בְּרֹשָׁ֔יו וְאָב֙וֹאָה֙ מְל֣וֹן קִצֹּ֔ה יַ֖עַר כַּרְמִלּֽוֹ׃ (ו) אֲנִ֣י קַ֔רְתִּי וְשָׁתִ֖יתִי מַ֣יִם זָרִ֑ים וְאַחְרִב֙ בְּכַף־ פְּעָמַ֔י כֹּ֖ל יְאֹרֵ֥י מָצֽוֹר׃ (ז) הֲלֹֽא־ שָׁמַ֤עְתָּ לְמֵֽרָחוֹק֙ אֹתָ֣הּ עָשִׂ֔יתִי לְמִ֥ימֵי קֶ֖דֶם וִֽיצַרְתִּ֑יהָ עַתָּ֣ה הֲבֵיאתִ֗יהָ וּתְהִ֗י לַהְשׁ֛וֹת גַּלִּ֥ים נִצִּ֖ים עָרִ֥ים בְּצֻרֽוֹת׃ (ח) וְיֹֽשְׁבֵיהֶן֙ קִצְרֵי־ יָ֔ד חַ֖תּוּ וַיֵּבֹ֑שׁוּ הָי֞וּ עֵ֤שֶׂב שָׂדֶה֙ וִ֣ירַק דֶּ֔שֶׁא חֲצִ֣יר גַּגּ֔וֹת וּשְׁדֵפָ֖ה לִפְנֵ֥י קָמָֽה׃ (ט) וְשִׁבְתְּךָ֛ וְצֵאתְךָ֥ וּבֹאֲךָ֖ יָדָ֑עְתִּי וְאֵ֖ת הִֽתְרַגֶּזְךָ֥ אֵלָֽי׃ (י) יַ֚עַן הִתְרַגֶּזְךָ֣ אֵלַ֔י וְשַׁאֲנַנְךָ֖ עָלָ֣ה בְאָזְנָ֑י וְשַׂמְתִּ֨י חַחִ֜י בְּאַפֶּ֗ךָ וּמִתְגִּי֙ בִּשְׂפָתֶ֔יךָ וַהֲשִׁ֣בֹתִ֔יךָ בַּדֶּ֖רֶךְ אֲשֶׁר־ בָּ֥אתָ בָּֽהּ׃ (יא) וְזֶה־ לְּךָ֣ הָא֔וֹת אָכ֤וֹל הַשָּׁנָה֙ סָפִ֔יחַ וּבַשָּׁנָ֥ה הַשֵּׁנִ֖ית סָחִ֑ישׁ וּבַשָּׁנָ֣ה הַשְּׁלִישִׁ֗ית זִרְע֧וּ וְקִצְר֛וּ וְנִטְע֥וּ כְרָמִ֖ים וְאִכְל֥וּ פִרְיָֽם׃ (יב) וְיָ֨סְפָ֜ה פְּלֵיטַ֧ת בֵּית־ יְהוּדָ֛ה הַנִּשְׁאָרָ֖ה שֹׁ֣רֶשׁ לְמָ֑טָּה וְעָשָׂ֥ה פְרִ֖י לְמָֽעְלָה׃ (יג) כִּ֤י מִירוּשָׁלִַ֙ם֙ תֵּצֵ֣א שְׁאֵרִ֔ית וּפְלֵיטָ֖ה מֵהַ֣ר צִיּ֑וֹן קִנְאַ֛ת יְהוָ֥ה צְבָא֖וֹת תַּֽעֲשֶׂה־ זֹּֽאת׃ (יד) לָכֵ֗ן כֹּֽה־ אָמַ֤ר יְהוָה֙ אֶל־ מֶ֣לֶךְ אַשּׁ֔וּר לֹ֤א יָבֹא֙ אֶל־ הָעִ֣יר הַזֹּ֔את וְלֹֽא־ יוֹרֶ֥ה שָׁ֖ם חֵ֑ץ וְלֹֽא־ יְקַדְּמֶ֣נָּה מָגֵ֔ן וְלֹֽא־ יִשְׁפֹּ֥ךְ עָלֶ֖יהָ סֹלְלָֽה׃ (טו) בַּדֶּ֥רֶךְ אֲשֶׁר־ יָבֹ֖א בָּ֣הּ יָשׁ֑וּב וְאֶל־ הָעִ֥יר הַזֹּ֛את לֹ֥א יָבֹ֖א נְאֻם־ יְהוָֽה׃ (טז) וְגַנּוֹתִ֛י אֶל־ הָעִ֥יר הַזֹּ֖את לְהֽוֹשִׁיעָ֑הּ לְמַֽעֲנִ֔י וּלְמַ֖עַן דָּוִ֥ד עַבְדִּֽי׃ (יז) וַיְהִי֮ בַּלַּ֣יְלָה הַהוּא֒ וַיֵּצֵ֣א ׀ מַלְאַ֣ךְ יְהוָ֗ה וַיַּךְ֙ בְּמַחֲנֵ֣ה אַשּׁ֔וּר מֵאָ֛ה שְׁמוֹנִ֥ים וַחֲמִשָּׁ֖ה אָ֑לֶף וַיַּשְׁכִּ֣ימוּ בַבֹּ֔קֶר וְהִנֵּ֥ה כֻלָּ֖ם פְּגָרִ֥ים מֵתִֽים׃ (יח) וַיִּסַּ֣ע וַיֵּ֔לֶךְ וַיָּ֖שָׁב סַנְחֵרִ֣יב מֶֽלֶךְ־ אַשּׁ֑וּר וַיֵּ֖שֶׁב בְּנִֽינְוֵֽה׃ (יט) וַיְהִי֩ ה֨וּא מִֽשְׁתַּחֲוֶ֜ה בֵּ֣ית ׀ נִסְרֹ֣ךְ אֱלֹהָ֗יו וְֽאַדְרַמֶּ֨לֶךְ וְשַׂרְאֶ֤צֶר בָּנָיו֙ הִכֻּ֣הוּ בַחֶ֔רֶב וְהֵ֥מָּה נִמְלְט֖וּ אֶ֣רֶץ אֲרָרָ֑ט וַיִּמְלֹ֛ךְ אֵֽסַר־ חַדֹּ֥ן בְּנ֖וֹ תַּחְתָּֽיו׃ תחילת פרק כ (כ) בַּיָּמִ֣ים הָהֵ֔ם חָלָ֥ה חִזְקִיָּ֖הוּ לָמ֑וּת וַיָּבֹ֣א אֵ֠לָיו יְשַׁעְיָ֨הוּ בֶן־ אָמ֜וֹץ הַנָּבִ֗יא וַיֹּ֨אמֶר אֵלָ֜יו כֹּֽה־ אָמַ֤ר יְהוָה֙ צַ֣ו לְבֵיתֶ֔ךָ כִּ֛י מֵ֥ת אַתָּ֖ה וְלֹ֥א תִֽחְיֶֽה׃ (כא) וַיַּסֵּ֥ב אֶת־ פָּנָ֖יו אֶל־ הַקִּ֑יר וַיִּ֨תְפַּלֵּ֔ל אֶל־ יְהוָ֖ה לֵאמֹֽר׃ (כב) אָנָּ֣ה יְהוָ֗ה זְכָר־ נָ֞א אֵ֣ת אֲשֶׁ֧ר הִתְהַלַּ֣כְתִּי לְפָנֶ֗יךָ בֶּֽאֱמֶת֙ וּבְלֵבָ֣ב שָׁלֵ֔ם וְהַטּ֥וֹב בְּעֵינֶ֖יךָ עָשִׂ֑יתִי וַיֵּ֥בְךְּ חִזְקִיָּ֖הוּ בְּכִ֥י גָדֽוֹל׃ (כג) וַיְהִ֣י יְשַׁעְיָ֔הוּ לֹ֣א יָצָ֔א (העיר) חָצֵ֖ר הַתִּֽיכֹנָ֑ה וּדְבַר־ יְהוָ֔ה הָיָ֥ה אֵלָ֖יו לֵאמֹֽר׃ (כד) שׁ֣וּב וְאָמַרְתָּ֞ אֶל־ חִזְקִיָּ֣הוּ נְגִיד־ עַמִּ֗י כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵי֙ דָּוִ֣ד אָבִ֔יךָ שָׁמַ֙עְתִּי֙ אֶת־ תְּפִלָּתֶ֔ךָ רָאִ֖יתִי אֶת־ דִּמְעָתֶ֑ךָ הִנְנִי֙ רֹ֣פֶא לָ֔ךְ בַּיּוֹם֙ הַשְּׁלִישִׁ֔י תַּעֲלֶ֖ה בֵּ֥ית יְהוָֽה׃ (כה) וְהֹסַפְתִּ֣י עַל־ יָמֶ֗יךָ חֲמֵ֤שׁ עֶשְׂרֵה֙ שָׁנָ֔ה וּמִכַּ֤ף מֶֽלֶךְ־ אַשּׁוּר֙ אַצִּ֣ילְךָ֔ וְאֵ֖ת הָעִ֣יר הַזֹּ֑את וְגַנּוֹתִי֙ עַל־ הָעִ֣יר הַזֹּ֔את לְמַֽעֲנִ֔י וּלְמַ֖עַן דָּוִ֥ד עַבְדִּֽי׃ (כו) וַיֹּ֣אמֶר יְשַֽׁעְיָ֔הוּ קְח֖וּ דְּבֶ֣לֶת תְּאֵנִ֑ים וַיִּקְח֛וּ וַיָּשִׂ֥ימוּ עַֽל־ הַשְּׁחִ֖ין וַיֶּֽחִי׃

מלכים סדר לב

(א) וַיֹּ֤אמֶר חִזְקִיָּ֙הוּ֙ אֶֽל־ יְשַׁעְיָ֔הוּ מָ֣ה א֔וֹת כִּֽי־ יִרְפָּ֥א יְהוָ֖ה לִ֑י וְעָלִ֛יתִי בַּיּ֥וֹם הַשְּׁלִישִׁ֖י בֵּ֥ית יְהוָֽה׃ (ב) וַיֹּ֣אמֶר יְשַׁעְיָ֗הוּ זֶה־ לְּךָ֤ הָאוֹת֙ מֵאֵ֣ת יְהוָ֔ה כִּ֚י יַעֲשֶׂ֣ה יְהוָ֔ה אֶת־ הַדָּבָ֖ר אֲשֶׁ֣ר דִּבֵּ֑ר הָלַ֤ךְ הַצֵּל֙ עֶ֣שֶׂר מַֽעֲל֔וֹת אִם־ יָשׁ֖וּב עֶ֥שֶׂר מַעֲלֽוֹת׃ (ג) וַיֹּ֙אמֶר֙ יְחִזְקִיָּ֔הוּ נָקֵ֣ל לַצֵּ֔ל לִנְט֖וֹת עֶ֣שֶׂר מַעֲל֑וֹת לֹ֣א כִ֔י יָשׁ֥וּב הַצֵּ֛ל אֲחֹרַנִּ֖ית עֶ֥שֶׂר מַעֲלֽוֹת׃ (ד) וַיִּקְרָ֛א יְשַׁעְיָ֥הוּ הַנָּבִ֖יא אֶל־ יְהוָ֑ה וַיָּ֣שֶׁב אֶת־ הַצֵּ֗ל בַּֽ֠מַּעֲלוֹת אֲשֶׁ֨ר יָרְדָ֜ה בְּמַעֲל֥וֹת אָחָ֛ז אֲחֹֽרַנִּ֖ית עֶ֥שֶׂר מַעֲלֽוֹת׃ (ה) בָּעֵ֣ת הַהִ֡יא שָׁלַ֡ח בְּרֹאדַ֣ךְ בַּ֠לְאֲדָן בֶּֽן־ בַּלְאֲדָ֧ן מֶֽלֶךְ־ בָּבֶ֛ל סְפָרִ֥ים וּמִנְחָ֖ה אֶל־ חִזְקִיָּ֑הוּ כִּ֣י שָׁמַ֔ע כִּ֥י חָלָ֖ה חִזְקִיָּֽהוּ׃ (ו) וַיִּשְׁמַ֣ע עֲלֵיהֶם֮ חִזְקִיָּהוּ֒ וַיַּרְאֵ֣ם אֶת־ כָּל־ בֵּ֣ית נְכֹתֹ֡ה אֶת־ הַכֶּסֶף֩ וְאֶת־ הַזָּהָ֨ב וְאֶת־ הַבְּשָׂמִ֜ים וְאֵ֣ת ׀ שֶׁ֣מֶן הַטּ֗וֹב וְאֵת֙ בֵּ֣ית כֵּלָ֔יו וְאֵ֛ת כָּל־ אֲשֶׁ֥ר נִמְצָ֖א בְּאֽוֹצְרֹתָ֑יו לֹֽא־ הָיָ֣ה דָבָ֗ר אֲ֠שֶׁר לֹֽא־ הֶרְאָ֧ם חִזְקִיָּ֛הוּ בְּבֵית֖וֹ וּבְכָל־ מֶמְשַׁלְתּֽוֹ׃ (ז) וַיָּבֹא֙ יְשַֽׁעְיָ֣הוּ הַנָּבִ֔יא אֶל־ הַמֶּ֖לֶךְ חִזְקִיָּ֑הוּ וַיֹּ֨אמֶר אֵלָ֜יו מָ֥ה אָמְר֣וּ ׀ הָאֲנָשִׁ֣ים הָאֵ֗לֶּה וּמֵאַ֙יִן֙ יָבֹ֣אוּ אֵלֶ֔יךָ וַיֹּ֙אמֶר֙ חִזְקִיָּ֔הוּ מֵאֶ֧רֶץ רְחוֹקָ֛ה בָּ֖אוּ מִבָּבֶֽל׃ (ח) וַיֹּ֕אמֶר מָ֥ה רָא֖וּ בְּבֵיתֶ֑ךָ וַיֹּ֣אמֶר חִזְקִיָּ֗הוּ אֵ֣ת כָּל־ אֲשֶׁ֤ר בְּבֵיתִי֙ רָא֔וּ לֹא־ הָיָ֥ה דָבָ֛ר אֲשֶׁ֥ר לֹֽא־ הִרְאִיתִ֖ם בְּאֹצְרֹתָֽי׃ (ט) וַיֹּ֥אמֶר יְשַׁעְיָ֖הוּ אֶל־ חִזְקִיָּ֑הוּ שְׁמַ֖ע דְּבַר־ יְהוָֽה׃ (י) הִנֵּה֮ יָמִ֣ים בָּאִים֒ וְנִשָּׂ֣א ׀ כָּל־ אֲשֶׁ֣ר בְּבֵיתֶ֗ךָ וַאֲשֶׁ֨ר אָצְר֧וּ אֲבֹתֶ֛יךָ עַד־ הַיּ֥וֹם הַזֶּ֖ה בָּבֶ֑לָה לֹֽא־ יִוָּתֵ֥ר דָּבָ֖ר אָמַ֥ר יְהוָֽה׃ (יא) וּמִבָּנֶ֜יךָ אֲשֶׁ֨ר יֵצְא֧וּ מִמְּךָ֛ אֲשֶׁ֥ר תּוֹלִ֖יד (יקח) יִקָּ֑חוּ וְהָיוּ֙ סָֽרִיסִ֔ים בְּהֵיכַ֖ל מֶ֥לֶךְ בָּבֶֽל׃ (יב) וַיֹּ֤אמֶר חִזְקִיָּ֙הוּ֙ אֶֽל־ יְשַֽׁעְיָ֔הוּ ט֥וֹב דְּבַר־ יְהוָ֖ה אֲשֶׁ֣ר דִּבַּ֑רְתָּ וַיֹּ֕אמֶר הֲל֛וֹא אִם־ שָׁל֥וֹם וֶאֱמֶ֖ת יִהְיֶ֥ה בְיָמָֽי׃ (יג) וְיֶ֨תֶר דִּבְרֵ֤י חִזְקִיָּ֙הוּ֙ וְכָל־ גְּב֣וּרָת֔וֹ וַאֲשֶׁ֣ר עָשָׂ֗ה אֶת־ הַבְּרֵכָה֙ וְאֶת־ הַתְּעָלָ֔ה וַיָּבֵ֥א אֶת־ הַמַּ֖יִם הָעִ֑ירָה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (יד) וַיִּשְׁכַּ֥ב חִזְקִיָּ֖הוּ עִם־ אֲבֹתָ֑יו וַיִּמְלֹ֛ךְ מְנַשֶּׁ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ סוף פרק כ (טו) בֶּן־ שְׁתֵּ֨ים עֶשְׂרֵ֤ה שָׁנָה֙ מְנַשֶּׁ֣ה בְמָלְכ֔וֹ וַחֲמִשִּׁ֤ים וְחָמֵשׁ֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֥ם אִמּ֖וֹ חֶפְצִי־ בָֽהּ׃ (טז) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּתֽוֹעֲבֹת֙ הַגּוֹיִ֔ם אֲשֶׁר֙ הוֹרִ֣ישׁ יְהוָ֔ה מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (יז) וַיָּ֗שָׁב וַיִּ֙בֶן֙ אֶת־ הַבָּמ֔וֹת אֲשֶׁ֥ר אִבַּ֖ד חִזְקִיָּ֣הוּ אָבִ֑יו וַיָּ֨קֶם מִזְבְּחֹ֜ת לַבַּ֗עַל וַיַּ֤עַשׂ אֲשֵׁרָה֙ כַּאֲשֶׁ֣ר עָשָׂ֗ה אַחְאָב֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיִּשְׁתַּ֙חוּ֙ לְכָל־ צְבָ֣א הַשָּׁמַ֔יִם וַֽיַּעֲבֹ֖ד אֹתָֽם׃ (יח) וּבָנָ֥ה מִזְבְּחֹ֖ת בְּבֵ֣ית יְהוָ֑ה אֲשֶׁר֙ אָמַ֣ר יְהוָ֔ה בִּירוּשָׁלִַ֖ם אָשִׂ֥ים אֶת־ שְׁמִֽי׃ (יט) וַיִּ֥בֶן מִזְבְּח֖וֹת לְכָל־ צְבָ֣א הַשָּׁמָ֑יִם בִּשְׁתֵּ֖י חַצְר֥וֹת בֵּית־ יְהוָֽה׃ (כ) וְהֶעֱבִ֤יר אֶת־ בְּנוֹ֙ בָּאֵ֔שׁ וְעוֹנֵ֣ן וְנִחֵ֔שׁ וְעָ֥שָׂה א֖וֹב וְיִדְּעֹנִ֑ים הִרְבָּ֗ה לַעֲשׂ֥וֹת הָרַ֛ע בְּעֵינֵ֥י יְהוָ֖ה לְהַכְעִֽיס׃ (כא) וַיָּ֕שֶׂם אֶת־ פֶּ֥סֶל הָאֲשֵׁרָ֖ה אֲשֶׁ֣ר עָשָׂ֑ה בַּבַּ֗יִת אֲשֶׁ֨ר אָמַ֤ר יְהוָה֙ אֶל־ דָּוִד֙ וְאֶל־ שְׁלֹמֹ֣ה בְנ֔וֹ בַּבַּ֨יִת הַזֶּ֜ה וּבִירוּשָׁלִַ֗ם אֲשֶׁ֤ר בָּחַ֙רְתִּי֙ מִכֹּל֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל אָשִׂ֥ים אֶת־ שְׁמִ֖י לְעוֹלָֽם׃ (כב) וְלֹ֣א אֹסִ֗יף לְהָנִיד֙ רֶ֣גֶל יִשְׂרָאֵ֔ל מִן־ הָ֣אֲדָמָ֔ה אֲשֶׁ֥ר נָתַ֖תִּי לַֽאֲבוֹתָ֑ם רַ֣ק ׀ אִם־ יִשְׁמְר֣וּ לַעֲשׂ֗וֹת כְּכֹל֙ אֲשֶׁ֣ר צִוִּיתִ֔ים וּלְכָל־ הַ֨תּוֹרָ֔ה אֲשֶׁר־ צִוָּ֥ה אֹתָ֖ם עַבְדִּ֥י מֹשֶֽׁה׃ (כג) וְלֹ֖א שָׁמֵ֑עוּ וַיַּתְעֵ֤ם מְנַשֶּׁה֙ לַעֲשׂ֣וֹת אֶת־ הָרָ֔ע מִן־ הַ֨גּוֹיִ֔ם אֲשֶׁר֙ הִשְׁמִ֣יד יְהוָ֔ה מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (כד) וַיְדַבֵּ֧ר יְהוָ֛ה בְּיַד־ עֲבָדָ֥יו הַנְּבִיאִ֖ים לֵאמֹֽר׃ (כה) יַעַן֩ אֲשֶׁ֨ר עָשָׂ֜ה מְנַשֶּׁ֤ה מֶֽלֶךְ־ יְהוּדָה֙ הַתֹּעֵב֣וֹת הָאֵ֔לֶּה הֵרַ֕ע מִכֹּ֛ל אֲשֶׁר־ עָשׂ֥וּ הָאֱמֹרִ֖י אֲשֶׁ֣ר לְפָנָ֑יו וַיַּחֲטִ֥א גַֽם־ אֶת־ יְהוּדָ֖ה בְּגִלּוּלָֽיו׃ (כו) לָכֵ֗ן כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִי֙ מֵבִ֣יא רָעָ֔ה עַל־ יְרוּשָׁלִַ֖ם וִֽיהוּדָ֑ה אֲשֶׁר֙ כָּל־ (שמעיו) שֹׁ֣מְעָ֔הּ תִּצַּ֖לְנָה שְׁתֵּ֥י אָזְנָֽיו׃ (כז) וְנָטִ֣יתִי עַל־ יְרוּשָׁלִַ֗ם אֵ֚ת קָ֣ו שֹֽׁמְר֔וֹן וְאֶת־ מִשְׁקֹ֖לֶת בֵּ֣ית אַחְאָ֑ב וּמָחִ֨יתִי אֶת־ יְרוּשָׁלִַ֜ם כַּֽאֲשֶׁר־ יִמְחֶ֤ה אֶת־ הַצַּלַּ֙חַת֙ מָחָ֔ה וְהָפַ֖ךְ עַל־ פָּנֶֽיהָ׃ (כח) וְנָטַשְׁתִּ֗י אֵ֚ת שְׁאֵרִ֣ית נַחֲלָתִ֔י וּנְתַתִּ֖ים בְּיַ֣ד אֹֽיְבֵיהֶ֑ם וְהָי֥וּ לְבַ֛ז וְלִמְשִׁסָּ֖ה לְכָל־ אֹיְבֵיהֶֽם׃ (כט) יַ֗עַן אֲשֶׁ֨ר עָשׂ֤וּ אֶת־ הָרַע֙ בְּעֵינַ֔י וַיִּהְי֥וּ מַכְעִסִ֖ים אֹתִ֑י מִן־ הַיּ֗וֹם אֲשֶׁ֨ר יָצְא֤וּ אֲבוֹתָם֙ מִמִּצְרַ֔יִם וְעַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (ל) וְגַם֩ דָּ֨ם נָקִ֜י שָׁפַ֤ךְ מְנַשֶּׁה֙ הַרְבֵּ֣ה מְאֹ֔ד עַ֛ד אֲשֶׁר־ מִלֵּ֥א אֶת־ יְרוּשָׁלִַ֖ם פֶּ֣ה לָפֶ֑ה לְבַ֤ד מֵֽחַטָּאתוֹ֙ אֲשֶׁ֣ר הֶחֱטִ֣יא אֶת־ יְהוּדָ֔ה לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֥י יְהוָֽה׃ (לא) וְיֶ֨תֶר דִּבְרֵ֤י מְנַשֶּׁה֙ וְכָל־ אֲשֶׁ֣ר עָשָׂ֔ה וְחַטָּאת֖וֹ אֲשֶׁ֣ר חָטָ֑א הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (לב) וַיִּשְׁכַּ֤ב מְנַשֶּׁה֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֥ר בְּגַן־ בֵּית֖וֹ בְּגַן־ עֻזָּ֑א וַיִּמְלֹ֛ךְ אָמ֥וֹן בְּנ֖וֹ תַּחְתָּֽיו׃ (לג) בֶּן־ עֶשְׂרִ֨ים וּשְׁתַּ֤יִם שָׁנָה֙ אָמ֣וֹן בְּמָלְכ֔וֹ וּשְׁתַּ֣יִם שָׁנִ֔ים מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ מְשֻׁלֶּ֥מֶת בַּת־ חָר֖וּץ מִן־ יָטְבָֽה׃ (לד) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כַּאֲשֶׁ֥ר עָשָׂ֖ה מְנַשֶּׁ֥ה אָבִֽיו׃ (לה) וַיֵּ֕לֶךְ בְּכָל־ הַדֶּ֖רֶךְ אֲשֶׁר־ הָלַ֣ךְ אָבִ֑יו וַֽיַּעֲבֹ֗ד אֶת־ הַגִּלֻּלִים֙ אֲשֶׁ֣ר עָבַ֣ד אָבִ֔יו וַיִּשְׁתַּ֖חוּ לָהֶֽם׃ (לו) וַיַּעֲזֹ֕ב אֶת־ יְהוָ֖ה אֱלֹהֵ֣י אֲבֹתָ֑יו וְלֹ֥א הָלַ֖ךְ בְּדֶ֥רֶךְ יְהוָֽה׃ (לז) וַיִּקְשְׁר֥וּ עַבְדֵֽי־ אָמ֖וֹן עָלָ֑יו וַיָּמִ֥יתוּ אֶת־ הַמֶּ֖לֶךְ בְּבֵיתֽוֹ׃ (לח) וַיַּךְ֙ עַם־ הָאָ֔רֶץ אֵ֥ת כָּל־ הַקֹּשְׁרִ֖ים עַל־ הַמֶּ֣לֶךְ אָמ֑וֹן וַיַּמְלִ֧יכוּ עַם־ הָאָ֛רֶץ אֶת־ יֹאשִׁיָּ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ (לט) וְיֶ֛תֶר דִּבְרֵ֥י אָמ֖וֹן אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (מ) וַיִּקְבְֹּ֥ר אֹת֛וֹ בִּקְבֻרָת֖וֹ בְּגַן־ עֻזָּ֑א וַיִּמְלֹ֛ךְ יֹאשִׁיָּ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ (מא) בֶּן־ שְׁמֹנֶ֤ה שָׁנָה֙ יֹאשִׁיָּ֣הוּ בְמָלְכ֔וֹ וּשְׁלֹשִׁ֤ים וְאַחַת֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ יְדִידָ֥ה בַת־ עֲדָ֖יָה מִבָּצְקַֽת׃
