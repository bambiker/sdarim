מלכים ב פרק כד לפי החלוקה לסדרים

פרק כד בספר מלכים ב (20 פסוקים) נמצא במלכים סדר לד (פסוקים יד–ל) ובמלכים סדר לה (פסוקים א–ג).

מלכים סדר לד

(א) וְכָמֹהוּ֩ לֹֽא־ הָיָ֨ה לְפָנָ֜יו מֶ֗לֶךְ אֲשֶׁר־ שָׁ֤ב אֶל־ יְהוָה֙ בְּכָל־ לְבָב֤וֹ וּבְכָל־ נַפְשׁוֹ֙ וּבְכָל־ מְאֹד֔וֹ כְּכֹ֖ל תּוֹרַ֣ת מֹשֶׁ֑ה וְאַחֲרָ֖יו לֹֽא־ קָ֥ם כָּמֹֽהוּ׃ (ב) אַ֣ךְ ׀ לֹֽא־ שָׁ֣ב יְהוָ֗ה מֵחֲר֤וֹן אַפּוֹ֙ הַגָּד֔וֹל אֲשֶׁר־ חָרָ֥ה אַפּ֖וֹ בִּֽיהוּדָ֑ה עַ֚ל כָּל־ הַכְּעָסִ֔ים אֲשֶׁ֥ר הִכְעִיס֖וֹ מְנַשֶּֽׁה׃ (ג) וַיֹּ֣אמֶר יְהוָ֗ה גַּ֤ם אֶת־ יְהוּדָה֙ אָסִיר֙ מֵעַ֣ל פָּנַ֔י כַּאֲשֶׁ֥ר הֲסִרֹ֖תִי אֶת־ יִשְׂרָאֵ֑ל וּ֠מָאַסְתִּי אֶת־ הָעִ֨יר הַזֹּ֤את אֲשֶׁר־ בָּחַ֙רְתִּי֙ אֶת־ יְר֣וּשָׁלִַ֔ם וְאֶת־ הַבַּ֔יִת אֲשֶׁ֣ר אָמַ֔רְתִּי יִהְיֶ֥ה שְׁמִ֖י שָֽׁם׃ (ד) וְיֶ֛תֶר דִּבְרֵ֥י יֹאשִׁיָּ֖הוּ וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (ה) בְּיָמָ֡יו עָלָה֩ פַרְעֹ֨ה נְכֹ֧ה מֶֽלֶךְ־ מִצְרַ֛יִם עַל־ מֶ֥לֶךְ אַשּׁ֖וּר עַל־ נְהַר־ פְּרָ֑ת וַיֵּ֨לֶךְ הַמֶּ֤לֶךְ יֹאשִׁיָּ֙הוּ֙ לִקְרָאת֔וֹ וַיְמִיתֵ֙הוּ֙ בִּמְגִדּ֔וֹ כִּרְאֹת֖וֹ אֹתֽוֹ׃ (ו) וַיַּרְכִּבֻ֨הוּ עֲבָדָ֥יו מֵת֙ מִמְּגִדּ֔וֹ וַיְבִאֻ֙הוּ֙ יְר֣וּשָׁלִַ֔ם וַֽיִּקְבְּרֻ֖הוּ בִּקְבֻֽרָת֑וֹ וַיִּקַּ֣ח עַם־ הָאָ֗רֶץ אֶת־ יְהֽוֹאָחָז֙ בֶּן־ יֹ֣אשִׁיָּ֔הוּ וַיִּמְשְׁח֥וּ אֹת֛וֹ וַיַּמְלִ֥יכוּ אֹת֖וֹ תַּ֥חַת אָבִֽיו׃ (ז) בֶּן־ עֶשְׂרִ֨ים וְשָׁלֹ֤שׁ שָׁנָה֙ יְהוֹאָחָ֣ז בְּמָלְכ֔וֹ וּשְׁלֹשָׁ֣ה חֳדָשִׁ֔ים מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ חֲמוּטַ֥ל בַּֽת־ יִרְמְיָ֖הוּ מִלִּבְנָֽה׃ (ח) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשׂ֖וּ אֲבֹתָֽיו׃ (ט) וַיַּאַסְרֵהוּ֩ פַרְעֹ֨ה נְכֹ֤ה בְרִבְלָה֙ בְּאֶ֣רֶץ חֲמָ֔ת (במלך) מִמְּלֹ֖ךְ בִּירוּשָׁלִָ֑ם וַיִּתֶּן־ עֹ֙נֶשׁ֙ עַל־ הָאָ֔רֶץ מֵאָ֥ה כִכַּר־ כֶּ֖סֶף וְכִכַּ֥ר זָהָֽב׃ (י) וַיַּמְלֵךְ֩ פַּרְעֹ֨ה נְכֹ֜ה אֶת־ אֶלְיָקִ֣ים בֶּן־ יֹאשִׁיָּ֗הוּ תַּ֚חַת יֹאשִׁיָּ֣הוּ אָבִ֔יו וַיַּסֵּ֥ב אֶת־ שְׁמ֖וֹ יְהוֹיָקִ֑ים וְאֶת־ יְהוֹאָחָ֣ז לָקָ֔ח וַיָּבֹ֥א מִצְרַ֖יִם וַיָּ֥מָת שָֽׁם׃ (יא) וְהַכֶּ֣סֶף וְהַזָּהָ֗ב נָתַ֤ן יְהוֹיָקִים֙ לְפַרְעֹ֔ה אַ֚ךְ הֶעֱרִ֣יךְ אֶת־ הָאָ֔רֶץ לָתֵ֥ת אֶת־ הַכֶּ֖סֶף עַל־ פִּ֣י פַרְעֹ֑ה אִ֣ישׁ כְּעֶרְכּ֗וֹ נָגַ֞שׂ אֶת־ הַכֶּ֤סֶף וְאֶת־ הַזָּהָב֙ אֶת־ עַ֣ם הָאָ֔רֶץ לָתֵ֖ת לְפַרְעֹ֥ה נְכֹֽה׃ (יב) בֶּן־ עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָ֨ה יְהוֹיָקִ֣ים בְּמָלְכ֔וֹ וְאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ (זבידה) זְבוּדָּ֥ה בַת־ פְּדָיָ֖ה מִן־ רוּמָֽה׃ (יג) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשׂ֖וּ אֲבֹתָֽיו׃ תחילת פרק כד (יד) בְּיָמָ֣יו עָלָ֔ה נְבֻכַדְנֶאצַּ֖ר מֶ֣לֶךְ בָּבֶ֑ל וַיְהִי־ ל֨וֹ יְהוֹיָקִ֥ים עֶ֙בֶד֙ שָׁלֹ֣שׁ שָׁנִ֔ים וַיָּ֖שָׁב וַיִּמְרָד־ בּֽוֹ׃ (טו) וַיְשַׁלַּ֣ח יְהוָ֣ה ׀ בּ֡וֹ אֶת־ גְּדוּדֵ֣י כַשְׂדִּים֩ וְאֶת־ גְּדוּדֵ֨י אֲרָ֜ם וְאֵ֣ת ׀ גְּדוּדֵ֣י מוֹאָ֗ב וְאֵת֙ גְּדוּדֵ֣י בְנֵֽי־ עַמּ֔וֹן וַיְשַׁלְּחֵ֥ם בִּֽיהוּדָ֖ה לְהַֽאֲבִיד֑וֹ כִּדְבַ֣ר יְהוָ֔ה אֲשֶׁ֣ר דִּבֶּ֔ר בְּיַ֖ד עֲבָדָ֥יו הַנְּבִיאִֽים׃ (טז) אַ֣ךְ ׀ עַל־ פִּ֣י יְהוָ֗ה הָֽיְתָה֙ בִּֽיהוּדָ֔ה לְהָסִ֖יר מֵעַ֣ל פָּנָ֑יו בְּחַטֹּ֣את מְנַשֶּׁ֔ה כְּכֹ֖ל אֲשֶׁ֥ר עָשָֽׂה׃ (יז) וְגַ֤ם דַּֽם־ הַנָּקִי֙ אֲשֶׁ֣ר שָׁפָ֔ךְ וַיְמַלֵּ֥א אֶת־ יְרוּשָׁלִַ֖ם דָּ֣ם נָקִ֑י וְלֹֽא־ אָבָ֥ה יְהוָ֖ה לִסְלֹֽחַ׃ (יח) וְיֶ֛תֶר דִּבְרֵ֥י יְהוֹיָקִ֖ים וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (יט) וַיִּשְׁכַּ֥ב יְהוֹיָקִ֖ים עִם־ אֲבֹתָ֑יו וַיִּמְלֹ֛ךְ יְהוֹיָכִ֥ין בְּנ֖וֹ תַּחְתָּֽיו׃ (כ) וְלֹֽא־ הֹסִ֥יף עוֹד֙ מֶ֣לֶךְ מִצְרַ֔יִם לָצֵ֖את מֵֽאַרְצ֑וֹ כִּֽי־ לָקַ֞ח מֶ֣לֶךְ בָּבֶ֗ל מִנַּ֤חַל מִצְרַ֙יִם֙ עַד־ נְהַר־ פְּרָ֔ת כֹּ֛ל אֲשֶׁ֥ר הָיְתָ֖ה לְמֶ֥לֶךְ מִצְרָֽיִם׃ (כא) בֶּן־ שְׁמֹנֶ֨ה עֶשְׂרֵ֤ה שָׁנָה֙ יְהוֹיָכִ֣ין בְּמָלְכ֔וֹ וּשְׁלֹשָׁ֣ה חֳדָשִׁ֔ים מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ נְחֻשְׁתָּ֥א בַת־ אֶלְנָתָ֖ן מִירוּשָׁלִָֽם׃ (כב) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה אָבִֽיו׃ (כג) בָּעֵ֣ת הַהִ֔יא (עלה) עָל֗וּ עַבְדֵ֛י נְבֻכַדְנֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל יְרוּשָׁלִָ֑ם וַתָּבֹ֥א הָעִ֖יר בַּמָּצֽוֹר׃ (כד) וַיָּבֹ֛א נְבוּכַדְנֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל עַל־ הָעִ֑יר וַעֲבָדָ֖יו צָרִ֥ים עָלֶֽיהָ׃ (כה) וַיֵּצֵ֞א יְהוֹיָכִ֤ין מֶֽלֶךְ־ יְהוּדָה֙ עַל־ מֶ֣לֶךְ בָּבֶ֔ל ה֣וּא וְאִמּ֔וֹ וַעֲבָדָ֖יו וְשָׂרָ֣יו וְסָֽרִיסָ֑יו וַיִּקַּ֤ח אֹתוֹ֙ מֶ֣לֶךְ בָּבֶ֔ל בִּשְׁנַ֥ת שְׁמֹנֶ֖ה לְמָלְכֽוֹ׃ (כו) וַיּוֹצֵ֣א מִשָּׁ֗ם אֶת־ כָּל־ אוֹצְרוֹת֙ בֵּ֣ית יְהוָ֔ה וְאֽוֹצְר֖וֹת בֵּ֣ית הַמֶּ֑לֶךְ וַיְקַצֵּ֞ץ אֶת־ כָּל־ כְּלֵ֣י הַזָּהָ֗ב אֲשֶׁ֨ר עָשָׂ֜ה שְׁלֹמֹ֤ה מֶֽלֶךְ־ יִשְׂרָאֵל֙ בְּהֵיכַ֣ל יְהוָ֔ה כַּֽאֲשֶׁ֖ר דִּבֶּ֥ר יְהוָֽה׃ (כז) וְהִגְלָ֣ה אֶת־ כָּל־ יְ֠רוּשָׁלִַם וְֽאֶת־ כָּל־ הַשָּׂרִ֞ים וְאֵ֣ת ׀ כָּל־ גִּבּוֹרֵ֣י הַחַ֗יִל (עשרה) עֲשֶׂ֤רֶת אֲלָפִים֙ גּוֹלֶ֔ה וְכָל־ הֶחָרָ֖שׁ וְהַמַּסְגֵּ֑ר לֹ֣א נִשְׁאַ֔ר זוּלַ֖ת דַּלַּ֥ת עַם־ הָאָֽרֶץ׃ (כח) וַיֶּ֥גֶל אֶת־ יְהוֹיָכִ֖ין בָּבֶ֑לָה וְאֶת־ אֵ֣ם הַ֠מֶּלֶךְ וְאֶת־ נְשֵׁ֨י הַמֶּ֜לֶךְ וְאֶת־ סָרִיסָ֗יו וְאֵת֙ (אולי) אֵילֵ֣י הָאָ֔רֶץ הוֹלִ֛יךְ גּוֹלָ֥ה מִירוּשָׁלִַ֖ם בָּבֶֽלָה׃ (כט) וְאֵת֩ כָּל־ אַנְשֵׁ֨י הַחַ֜יִל שִׁבְעַ֣ת אֲלָפִ֗ים וְהֶחָרָ֤שׁ וְהַמַּסְגֵּר֙ אֶ֔לֶף הַכֹּ֕ל גִּבּוֹרִ֖ים עֹשֵׂ֣י מִלְחָמָ֑ה וַיְבִיאֵ֧ם מֶֽלֶךְ־ בָּבֶ֛ל גּוֹלָ֖ה בָּבֶֽלָה׃ (ל) וַיַּמְלֵ֧ךְ מֶֽלֶךְ־ בָּבֶ֛ל אֶת־ מַתַּנְיָ֥ה דֹד֖וֹ תַּחְתָּ֑יו וַיַּסֵּ֥ב אֶת־ שְׁמ֖וֹ צִדְקִיָּֽהוּ׃

מלכים סדר לה

(א) בֶּן־ עֶשְׂרִ֨ים וְאַחַ֤ת שָׁנָה֙ צִדְקִיָּ֣הוּ בְמָלְכ֔וֹ וְאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ (חמיטל) חֲמוּטַ֥ל בַּֽת־ יִרְמְיָ֖הוּ מִלִּבְנָֽה׃ (ב) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה יְהוֹיָקִֽים׃ (ג) כִּ֣י ׀ עַל־ אַ֣ף יְהוָ֗ה הָיְתָ֤ה בִירוּשָׁלִַ֙ם֙ וּבִ֣יהוּדָ֔ה עַד־ הִשְׁלִכ֥וֹ אֹתָ֖ם מֵעַ֣ל פָּנָ֑יו וַיִּמְרֹ֥ד צִדְקִיָּ֖הוּ בְּמֶ֥לֶךְ בָּבֶֽל׃ סוף פרק כד (ד) וַיְהִי֩ בִשְׁנַ֨ת הַתְּשִׁיעִ֜ית לְמָלְכ֗וֹ בַּחֹ֣דֶשׁ הָעֲשִׂירִי֮ בֶּעָשׂ֣וֹר לַחֹדֶשׁ֒ בָּ֠א נְבֻכַדְנֶאצַּ֨ר מֶֽלֶךְ־ בָּבֶ֜ל ה֧וּא וְכָל־ חֵיל֛וֹ עַל־ יְרוּשָׁלִַ֖ם וַיִּ֣חַן עָלֶ֑יהָ וַיִּבְנ֥וּ עָלֶ֖יהָ דָּיֵ֥ק סָבִֽיב׃ (ה) וַתָּבֹ֥א הָעִ֖יר בַּמָּצ֑וֹר עַ֚ד עַשְׁתֵּ֣י עֶשְׂרֵ֣ה שָׁנָ֔ה לַמֶּ֖לֶךְ צִדְקִיָּֽהוּ׃ (ו) בְּתִשְׁעָ֣ה לַחֹ֔דֶשׁ וַיֶּחֱזַ֥ק הָרָעָ֖ב בָּעִ֑יר וְלֹא־ הָ֥יָה לֶ֖חֶם לְעַ֥ם הָאָֽרֶץ׃ (ז) וַתִּבָּקַ֣ע הָעִ֗יר וְכָל־ אַנְשֵׁ֨י הַמִּלְחָמָ֤ה ׀ הַלַּ֙יְלָה֙ דֶּ֜רֶךְ שַׁ֣עַר ׀ בֵּ֣ין הַחֹמֹתַ֗יִם אֲשֶׁר֙ עַל־ גַּ֣ן הַמֶּ֔לֶךְ וְכַשְׂדִּ֥ים עַל־ הָעִ֖יר סָבִ֑יב וַיֵּ֖לֶךְ דֶּ֥רֶךְ הָעֲרָבָֽה׃ (ח) וַיִּרְדְּפ֤וּ חֵיל־ כַּשְׂדִּים֙ אַחַ֣ר הַמֶּ֔לֶךְ וַיַּשִּׂ֥גוּ אֹת֖וֹ בְּעַרְב֣וֹת יְרֵח֑וֹ וְכָל־ חֵיל֔וֹ נָפֹ֖צוּ מֵעָלָֽיו׃ (ט) וַֽיִּתְפְּשׂוּ֙ אֶת־ הַמֶּ֔לֶךְ וַיַּעֲל֥וּ אֹת֛וֹ אֶל־ מֶ֥לֶךְ בָּבֶ֖ל רִבְלָ֑תָה וַיְדַבְּר֥וּ אִתּ֖וֹ מִשְׁפָּֽט׃ (י) וְאֶת־ בְּנֵי֙ צִדְקִיָּ֔הוּ שָׁחֲט֖וּ לְעֵינָ֑יו וְאֶת־ עֵינֵ֤י צִדְקִיָּ֙הוּ֙ עִוֵּ֔ר וַיַּאַסְרֵ֙הוּ֙ בַֽנְחֻשְׁתַּ֔יִם וַיְבִאֵ֖הוּ בָּבֶֽל׃ (יא) וּבַחֹ֤דֶשׁ הַֽחֲמִישִׁי֙ בְּשִׁבְעָ֣ה לַחֹ֔דֶשׁ הִ֗יא שְׁנַת֙ תְּשַֽׁע־ עֶשְׂרֵ֣ה שָׁנָ֔ה לַמֶּ֖לֶךְ נְבֻכַדְנֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֑ל בָּ֞א נְבוּזַרְאֲדָ֧ן רַב־ טַבָּחִ֛ים עֶ֥בֶד מֶֽלֶךְ־ בָּבֶ֖ל יְרוּשָׁלִָֽם׃ (יב) וַיִּשְׂרֹ֥ף אֶת־ בֵּית־ יְהוָ֖ה וְאֶת־ בֵּ֣ית הַמֶּ֑לֶךְ וְאֵ֨ת כָּל־ בָּתֵּ֧י יְרוּשָׁלִַ֛ם וְאֶת־ כָּל־ בֵּ֥ית גָּד֖וֹל שָׂרַ֥ף בָּאֵֽשׁ׃ (יג) וְאֶת־ חוֹמֹ֥ת יְרוּשָׁלִַ֖ם סָבִ֑יב נָֽתְצוּ֙ כָּל־ חֵ֣יל כַּשְׂדִּ֔ים אֲשֶׁ֖ר רַב־ טַבָּחִֽים׃ (יד) וְאֵת֩ יֶ֨תֶר הָעָ֜ם הַנִּשְׁאָרִ֣ים בָּעִ֗יר וְאֶת־ הַנֹּֽפְלִים֙ אֲשֶׁ֤ר נָפְלוּ֙ עַל־ הַמֶּ֣לֶךְ בָּבֶ֔ל וְאֵ֖ת יֶ֣תֶר הֶהָמ֑וֹן הֶגְלָ֕ה נְבוּזַרְאֲדָ֖ן רַב־ טַבָּחִֽים׃ (טו) וּמִדַּלַּ֣ת הָאָ֔רֶץ הִשְׁאִ֖יר רַב־ טַבָּחִ֑ים לְכֹֽרְמִ֖ים וּלְיֹגְבִֽים׃ (טז) וְאֶת־ עַמּוּדֵ֨י הַנְּחֹ֜שֶׁת אֲשֶׁ֣ר בֵּית־ יְהוָ֗ה וְֽאֶת־ הַמְּכֹנ֞וֹת וְאֶת־ יָ֧ם הַנְּחֹ֛שֶׁת אֲשֶׁ֥ר בְּבֵית־ יְהוָ֖ה שִׁבְּר֣וּ כַשְׂדִּ֑ים וַיִּשְׂא֥וּ אֶת־ נְחֻשְׁתָּ֖ם בָּבֶֽלָה׃ (יז) וְאֶת־ הַסִּירֹ֨ת וְאֶת־ הַיָּעִ֜ים וְאֶת־ הַֽמְזַמְּר֣וֹת וְאֶת־ הַכַּפּ֗וֹת וְאֵ֨ת כָּל־ כְּלֵ֧י הַנְּחֹ֛שֶׁת אֲשֶׁ֥ר יְשָֽׁרְתוּ־ בָ֖ם לָקָֽחוּ׃ (יח) וְאֶת־ הַמַּחְתּוֹת֙ וְאֶת־ הַמִּזְרָק֗וֹת אֲשֶׁ֤ר זָהָב֙ זָהָ֔ב וַאֲשֶׁר־ כֶּ֖סֶף כָּ֑סֶף לָקַ֖ח רַב־ טַבָּחִֽים׃ (יט) הָעַמּוּדִ֣ים ׀ שְׁנַ֗יִם הַיָּ֤ם הָֽאֶחָד֙ וְהַמְּכֹנ֔וֹת אֲשֶׁר־ עָשָׂ֥ה שְׁלֹמֹ֖ה לְבֵ֣ית יְהוָ֑ה לֹא־ הָיָ֣ה מִשְׁקָ֔ל לִנְחֹ֖שֶׁת כָּל־ הַכֵּלִ֥ים הָאֵֽלֶּה׃ (כ) שְׁמֹנֶה֩ עֶשְׂרֵ֨ה אַמָּ֜ה קוֹמַ֣ת ׀ הָעַמּ֣וּד הָאֶחָ֗ד וְכֹתֶ֨רֶת עָלָ֥יו ׀ נְחֹשֶׁת֮ וְקוֹמַ֣ת הַכֹּתֶרֶת֮ שָׁלֹ֣שׁ (אמה) אַמּוֹת֒ וּשְׂבָכָ֨ה וְרִמֹּנִ֧ים עַֽל־ הַכֹּתֶ֛רֶת סָבִ֖יב הַכֹּ֣ל נְחֹ֑שֶׁת וְכָאֵ֛לֶּה לַֽעַמּ֥וּד הַשֵּׁנִ֖י עַל־ הַשְּׂבָכָֽה׃ (כא) וַיִּקַּ֣ח רַב־ טַבָּחִ֗ים אֶת־ שְׂרָיָה֙ כֹּהֵ֣ן הָרֹ֔אשׁ וְאֶת־ צְפַנְיָ֖הוּ כֹּהֵ֣ן מִשְׁנֶ֑ה וְאֶת־ שְׁלֹ֖שֶׁת שֹׁמְרֵ֥י הַסַּֽף׃ (כב) וּמִן־ הָעִ֡יר לָקַח֩ סָרִ֨יס אֶחָ֜ד אֲ‍ֽשֶׁר־ ה֥וּא פָקִ֣יד ׀ עַל־ אַנְשֵׁ֣י הַמִּלְחָמָ֗ה וַחֲמִשָּׁ֨ה אֲנָשִׁ֜ים מֵרֹאֵ֤י פְנֵֽי־ הַמֶּ֙לֶךְ֙ אֲשֶׁ֣ר נִמְצְא֣וּ בָעִ֔יר וְאֵ֗ת הַסֹּפֵר֙ שַׂ֣ר הַצָּבָ֔א הַמַּצְבִּ֖א אֶת־ עַ֣ם הָאָ֑רֶץ וְשִׁשִּׁ֥ים אִישׁ֙ מֵעַ֣ם הָאָ֔רֶץ הַֽנִּמְצְאִ֖ים בָּעִֽיר׃ (כג) וַיִּקַּ֣ח אֹתָ֔ם נְבוּזַרְאֲדָ֖ן רַב־ טַבָּחִ֑ים וַיֹּ֧לֶךְ אֹתָ֛ם עַל־ מֶ֥לֶךְ בָּבֶ֖ל רִבְלָֽתָה׃ (כד) וַיַּ֣ךְ אֹתָם֩ מֶ֨לֶךְ בָּבֶ֧ל וַיְמִיתֵ֛ם בְּרִבְלָ֖ה בְּאֶ֣רֶץ חֲמָ֑ת וַיִּ֥גֶל יְהוּדָ֖ה מֵעַ֥ל אַדְמָתֽוֹ׃ (כה) וְהָעָ֗ם הַנִּשְׁאָר֙ בְּאֶ֣רֶץ יְהוּדָ֔ה אֲשֶׁ֣ר הִשְׁאִ֔יר נְבֽוּכַדְנֶאצַּ֖ר מֶ֣לֶךְ בָּבֶ֑ל וַיַּפְקֵ֣ד עֲלֵיהֶ֔ם אֶת־ גְּדַלְיָ֖הוּ בֶּן־ אֲחִיקָ֥ם בֶּן־ שָׁפָֽן׃ (כו) וַיִּשְׁמְעוּ֩ כָל־ שָׂרֵ֨י הַחֲיָלִ֜ים הֵ֣מָּה וְהָאֲנָשִׁ֗ים כִּֽי־ הִפְקִ֤יד מֶֽלֶךְ־ בָּבֶל֙ אֶת־ גְּדַלְיָ֔הוּ וַיָּבֹ֥אוּ אֶל־ גְּדַלְיָ֖הוּ הַמִּצְפָּ֑ה וְיִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֡ה וְיוֹחָנָ֣ן בֶּן־ קָ֠רֵחַ וּשְׂרָיָ֨ה בֶן־ תַּנְחֻ֜מֶת הַנְּטֹפָתִ֗י וְיַֽאֲזַנְיָ֙הוּ֙ בֶּן־ הַמַּ֣עֲכָתִ֔י הֵ֖מָּה וְאַנְשֵׁיהֶֽם׃ (כז) וַיִּשָּׁבַ֨ע לָהֶ֤ם גְּדַלְיָ֙הוּ֙ וּלְאַנְשֵׁיהֶ֔ם וַיֹּ֣אמֶר לָהֶ֔ם אַל־ תִּֽירְא֖וּ מֵעַבְדֵ֣י הַכַּשְׂדִּ֑ים שְׁב֣וּ בָאָ֗רֶץ וְעִבְד֛וּ אֶת־ מֶ֥לֶךְ בָּבֶ֖ל וְיִטַ֥ב לָכֶֽם׃ (כח) וַיְהִ֣י ׀ בַּחֹ֣דֶשׁ הַשְּׁבִיעִ֗י בָּ֣א יִשְׁמָעֵ֣אל בֶּן־ נְ֠תַנְיָה בֶּן־ אֱלִ֨ישָׁמָ֜ע מִזֶּ֣רַע הַמְּלוּכָ֗ה וַעֲשָׂרָ֤ה אֲנָשִׁים֙ אִתּ֔וֹ וַיַּכּ֥וּ אֶת־ גְּדַלְיָ֖הוּ וַיָּמֹ֑ת וְאֶת־ הַיְּהוּדִים֙ וְאֶת־ הַכַּשְׂדִּ֔ים אֲשֶׁר־ הָי֥וּ אִתּ֖וֹ בַּמִּצְפָּֽה׃ (כט) וַיָּקֻ֨מוּ כָל־ הָעָ֜ם מִקָּטֹ֤ן וְעַד־ גָּדוֹל֙ וְשָׂרֵ֣י הַחֲיָלִ֔ים וַיָּבֹ֖אוּ מִצְרָ֑יִם כִּ֥י יָרְא֖וּ מִפְּנֵ֥י כַשְׂדִּֽים׃ (ל) וַיְהִי֩ בִשְׁלֹשִׁ֨ים וָשֶׁ֜בַע שָׁנָ֗ה לְגָלוּת֙ יְהוֹיָכִ֣ין מֶֽלֶךְ־ יְהוּדָ֔ה בִּשְׁנֵ֤ים עָשָׂר֙ חֹ֔דֶשׁ בְּעֶשְׂרִ֥ים וְשִׁבְעָ֖ה לַחֹ֑דֶשׁ נָשָׂ֡א אֱוִ֣יל מְרֹדַךְ֩ מֶ֨לֶךְ בָּבֶ֜ל בִּשְׁנַ֣ת מָלְכ֗וֹ אֶת־ רֹ֛אשׁ יְהוֹיָכִ֥ין מֶֽלֶךְ־ יְהוּדָ֖ה מִבֵּ֥ית כֶּֽלֶא׃ (לא) וַיְדַבֵּ֥ר אִתּ֖וֹ טֹב֑וֹת וַיִּתֵּן֙ אֶת־ כִּסְא֔וֹ מֵעַ֗ל כִּסֵּ֧א הַמְּלָכִ֛ים אֲשֶׁ֥ר אִתּ֖וֹ בְּבָבֶֽל׃ (לב) וְשִׁנָּ֕א אֵ֖ת בִּגְדֵ֣י כִלְא֑וֹ וְאָכַ֨ל לֶ֧חֶם תָּמִ֛יד לְפָנָ֖יו כָּל־ יְמֵ֥י חַיָּֽיו׃ (לג) וַאֲרֻחָת֗וֹ אֲרֻחַ֨ת תָּמִ֧יד נִתְּנָה־ לּ֛וֹ מֵאֵ֥ת הַמֶּ֖לֶךְ דְּבַר־ י֣וֹם בְּיוֹמ֑וֹ כֹּ֖ל יְמֵ֥י חַיָּֽו׃
