מלכים ב פרק י לפי החלוקה לסדרים

פרק י בספר מלכים ב (36 פסוקים) נמצא במלכים סדר כד (פסוקים כו–לט) ובמלכים סדר כה (פסוקים א–כב).

מלכים סדר כד

(א) וַֽיְמַהֲר֗וּ וַיִּקְחוּ֙ אִ֣ישׁ בִּגְד֔וֹ וַיָּשִׂ֥ימוּ תַחְתָּ֖יו אֶל־ גֶּ֣רֶם הַֽמַּעֲל֑וֹת וַֽיִּתְקְעוּ֙ בַּשּׁוֹפָ֔ר וַיֹּאמְר֖וּ מָלַ֥ךְ יֵהֽוּא׃ (ב) וַיִּתְקַשֵּׁ֗ר יֵה֛וּא בֶּן־ יְהוֹשָׁפָ֥ט בֶּן־ נִמְשִׁ֖י אֶל־ יוֹרָ֑ם וְיוֹרָם֩ הָיָ֨ה שֹׁמֵ֜ר בְּרָמֹ֣ת גִּלְעָ֗ד ה֚וּא וְכָל־ יִשְׂרָאֵ֔ל מִפְּנֵ֥י חֲזָאֵ֖ל מֶֽלֶךְ־ אֲרָֽם׃ (ג) וַיָּשָׁב֩ יְהוֹרָ֨ם הַמֶּ֜לֶךְ לְהִתְרַפֵּ֣א בִיְזְרְעֶ֗אל מִן־ הַמַּכִּים֙ אֲשֶׁ֣ר יַכֻּ֣הוּ אֲרַמִּ֔ים בְּהִלָּ֣חֲמ֔וֹ אֶת־ חֲזָאֵ֖ל מֶ֣לֶךְ אֲרָ֑ם וַיֹּ֤אמֶר יֵהוּא֙ אִם־ יֵ֣שׁ נַפְשְׁכֶ֔ם אַל־ יֵצֵ֤א פָלִיט֙ מִן־ הָעִ֔יר לָלֶ֖כֶת (לגיד) לְהַגִּ֥יד בְּיִזְרְעֶֽאל׃ (ד) וַיִּרְכַּ֤ב יֵהוּא֙ וַיֵּ֣לֶךְ יִזְרְעֶ֔אלָה כִּ֥י יוֹרָ֖ם שֹׁכֵ֣ב שָׁ֑מָּה וַֽאֲחַזְיָה֙ מֶ֣לֶךְ יְהוּדָ֔ה יָרַ֖ד לִרְא֥וֹת אֶת־ יוֹרָֽם׃ (ה) וְהַצֹּפֶה֩ עֹמֵ֨ד עַֽל־ הַמִּגְדָּ֜ל בְּיִזְרְעֶ֗אל וַיַּ֞רְא אֶת־ שִׁפְעַ֤ת יֵהוּא֙ בְּבֹא֔וֹ וַיֹּ֕אמֶר שִׁפְעַ֖ת אֲנִ֣י רֹאֶ֑ה וַיֹּ֣אמֶר יְהוֹרָ֗ם קַ֥ח רַכָּ֛ב וּֽשְׁלַ֥ח לִקְרָאתָ֖ם וְיֹאמַ֥ר הֲשָׁלֽוֹם׃ (ו) וַיֵּלֶךְ֩ רֹכֵ֨ב הַסּ֜וּס לִקְרָאת֗וֹ וַיֹּ֙אמֶר֙ כֹּֽה־ אָמַ֤ר הַמֶּ֙לֶךְ֙ הֲשָׁל֔וֹם וַיֹּ֧אמֶר יֵה֛וּא מַה־ לְּךָ֥ וּלְשָׁל֖וֹם סֹ֣ב אֶֽל־ אַחֲרָ֑י וַיַּגֵּ֤ד הַצֹּפֶה֙ לֵאמֹ֔ר בָּֽא־ הַמַּלְאָ֥ךְ עַד־ הֵ֖ם וְלֹֽא־ שָֽׁב׃ (ז) וַיִּשְׁלַ֗ח רֹכֵ֣ב סוּס֮ שֵׁנִי֒ וַיָּבֹ֣א אֲלֵהֶ֔ם וַיֹּ֛אמֶר כֹּֽה־ אָמַ֥ר הַמֶּ֖לֶךְ שָׁל֑וֹם וַיֹּ֧אמֶר יֵה֛וּא מַה־ לְּךָ֥ וּלְשָׁל֖וֹם סֹ֥ב אֶֽל־ אַחֲרָֽי׃ (ח) וַיַּגֵּ֤ד הַצֹּפֶה֙ לֵאמֹ֔ר בָּ֥א עַד־ אֲלֵיהֶ֖ם וְלֹֽא־ שָׁ֑ב וְהַמִּנְהָ֗ג כְּמִנְהַג֙ יֵה֣וּא בֶן־ נִמְשִׁ֔י כִּ֥י בְשִׁגָּע֖וֹן יִנְהָֽג׃ (ט) וַיֹּ֤אמֶר יְהוֹרָם֙ אֱסֹ֔ר וַיֶּאְסֹ֖ר רִכְבּ֑וֹ וַיֵּצֵ֣א יְהוֹרָ֣ם מֶֽלֶךְ־ יִ֠שְׂרָאֵל וַאֲחַזְיָ֨הוּ מֶֽלֶךְ־ יְהוּדָ֜ה אִ֣ישׁ בְּרִכְבּ֗וֹ וַיֵּֽצְאוּ֙ לִקְרַ֣את יֵה֔וּא וַיִּמְצָאֻ֔הוּ בְּחֶלְקַ֖ת נָב֥וֹת הַיִּזְרְעֵאלִֽי׃ (י) וַיְהִ֗י כִּרְא֤וֹת יְהוֹרָם֙ אֶת־ יֵה֔וּא וַיֹּ֖אמֶר הֲשָׁל֣וֹם יֵה֑וּא וַיֹּ֙אמֶר֙ מָ֣ה הַשָּׁל֔וֹם עַד־ זְנוּנֵ֞י אִיזֶ֧בֶל אִמְּךָ֛ וּכְשָׁפֶ֖יהָ הָרַבִּֽים׃ (יא) וַיַּהֲפֹ֧ךְ יְהוֹרָ֛ם יָדָ֖יו וַיָּנֹ֑ס וַיֹּ֥אמֶר אֶל־ אֲחַזְיָ֖הוּ מִרְמָ֥ה אֲחַזְיָֽה׃ (יב) וְיֵה֞וּא מִלֵּ֧א יָד֣וֹ בַקֶּ֗שֶׁת וַיַּ֤ךְ אֶת־ יְהוֹרָם֙ בֵּ֣ין זְרֹעָ֔יו וַיֵּצֵ֥א הַחֵ֖צִי מִלִּבּ֑וֹ וַיִּכְרַ֖ע בְּרִכְבּֽוֹ׃ (יג) וַיֹּ֗אמֶר אֶל־ בִּדְקַר֙ (שלשה) שָֽׁלִשׁ֔וֹ שָׂ֚א הַשְׁלִכֵ֔הוּ בְּחֶלְקַ֕ת שְׂדֵ֖ה נָב֣וֹת הַיִּזְרְעֵאלִ֑י כִּֽי־ זְכֹ֞ר אֲנִ֣י וָאַ֗תָּה אֵ֣ת רֹכְבִ֤ים צְמָדִים֙ אַֽחֲרֵי֙ אַחְאָ֣ב אָבִ֔יו וַֽיהוָה֙ נָשָׂ֣א עָלָ֔יו אֶת־ הַמַּשָּׂ֖א הַזֶּֽה׃ (יד) אִם־ לֹ֡א אֶת־ דְּמֵ֣י נָבוֹת֩ וְאֶת־ דְּמֵ֨י בָנָ֜יו רָאִ֤יתִי אֶ֙מֶשׁ֙ נְאֻם־ יְהוָ֔ה וְשִׁלַּמְתִּ֥י לְךָ֛ בַּחֶלְקָ֥ה הַזֹּ֖את נְאֻם־ יְהוָ֑ה וְעַתָּ֗ה שָׂ֧א הַשְׁלִכֵ֛הוּ בַּחֶלְקָ֖ה כִּדְבַ֥ר יְהוָֽה׃ (טו) וַאֲחַזְיָ֤ה מֶֽלֶךְ־ יְהוּדָה֙ רָאָ֔ה וַיָּ֕נָס דֶּ֖רֶךְ בֵּ֣ית הַגָּ֑ן וַיִּרְדֹּ֨ף אַחֲרָ֜יו יֵה֗וּא וַ֠יֹּאמֶר גַּם־ אֹת֞וֹ הַכֻּ֣הוּ אֶל־ הַמֶּרְכָּבָ֗ה בְּמַֽעֲלֵה־ גוּר֙ אֲשֶׁ֣ר אֶֽת־ יִבְלְעָ֔ם וַיָּ֥נָס מְגִדּ֖וֹ וַיָּ֥מָת שָֽׁם׃ (טז) וַיַּרְכִּ֧בוּ אֹת֛וֹ עֲבָדָ֖יו יְרוּשָׁלְָ֑מָה וַיִּקְבְּר֨וּ אֹת֧וֹ בִקְבֻרָת֛וֹ עִם־ אֲבֹתָ֖יו בְּעִ֥יר דָּוִֽד׃ (יז) וּבִשְׁנַת֙ אַחַ֣ת עֶשְׂרֵ֣ה שָׁנָ֔ה לְיוֹרָ֖ם בֶּן־ אַחְאָ֑ב מָלַ֥ךְ אֲחַזְיָ֖ה עַל־ יְהוּדָֽה׃ (יח) וַיָּב֥וֹא יֵה֖וּא יִזְרְעֶ֑אלָה וְאִיזֶ֣בֶל שָׁמְעָ֗ה וַתָּ֨שֶׂם בַּפּ֤וּךְ עֵינֶ֙יהָ֙ וַתֵּ֣יטֶב אֶת־ רֹאשָׁ֔הּ וַתַּשְׁקֵ֖ף בְּעַ֥ד הַחַלּֽוֹן׃ (יט) וְיֵה֖וּא בָּ֣א בַשָּׁ֑עַר וַתֹּ֣אמֶר הֲשָׁל֔וֹם זִמְרִ֖י הֹרֵ֥ג אֲדֹנָֽיו׃ (כ) וַיִּשָּׂ֤א פָנָיו֙ אֶל־ הַ֣חַלּ֔וֹן וַיֹּ֕אמֶר מִ֥י אִתִּ֖י מִ֑י וַיַּשְׁקִ֣יפוּ אֵלָ֔יו שְׁנַ֥יִם שְׁלֹשָׁ֖ה סָרִיסִֽים׃ (כא) וַיֹּ֥אמֶר (שמטהו) שִׁמְט֖וּהָ וַֽיִּשְׁמְט֑וּהָ וַיִּ֨ז מִדָּמָ֧הּ אֶל־ הַקִּ֛יר וְאֶל־ הַסּוּסִ֖ים וַֽיִּרְמְסֶֽנָּה׃ (כב) וַיָּבֹ֖א וַיֹּ֣אכַל וַיֵּ֑שְׁתְּ וַיֹּ֗אמֶר פִּקְדוּ־ נָ֞א אֶת־ הָאֲרוּרָ֤ה הַזֹּאת֙ וְקִבְר֔וּהָ כִּ֥י בַת־ מֶ֖לֶךְ הִֽיא׃ (כג) וַיֵּלְכ֖וּ לְקָבְרָ֑הּ וְלֹא־ מָ֣צְאוּ בָ֗הּ כִּ֧י אִם־ הַגֻּלְגֹּ֛לֶת וְהָרַגְלַ֖יִם וְכַפּ֥וֹת הַיָּדָֽיִם׃ (כד) וַיָּשֻׁבוּ֮ וַיַּגִּ֣ידוּ לוֹ֒ וַיֹּ֙אמֶר֙ דְּבַר־ יְהוָ֣ה ה֔וּא אֲשֶׁ֣ר דִּבֶּ֗ר בְּיַד־ עַבְדּ֛וֹ אֵלִיָּ֥הוּ הַתִּשְׁבִּ֖י לֵאמֹ֑ר בְּחֵ֣לֶק יִזְרְעֶ֔אל יֹאכְל֥וּ הַכְּלָבִ֖ים אֶת־ בְּשַׂ֥ר אִיזָֽבֶל׃ (כה) (והית) וְֽהָיְתָ֞ה נִבְלַ֣ת אִיזֶ֗בֶל כְּדֹ֛מֶן עַל־ פְּנֵ֥י הַשָּׂדֶ֖ה בְּחֵ֣לֶק יִזְרְעֶ֑אל אֲשֶׁ֥ר לֹֽא־ יֹאמְר֖וּ זֹ֥את אִיזָֽבֶל׃ תחילת פרק י (כו) וּלְאַחְאָ֛ב שִׁבְעִ֥ים בָּנִ֖ים בְּשֹׁמְר֑וֹן וַיִּכְתֹּב֩ יֵה֨וּא סְפָרִ֜ים וַיִּשְׁלַ֣ח שֹׁמְר֗וֹן אֶל־ שָׂרֵ֤י יִזְרְעֶאל֙ הַזְּקֵנִ֔ים וְאֶל־ הָאֹמְנִ֥ים אַחְאָ֖ב לֵאמֹֽר׃ (כז) וְעַתָּ֗ה כְּבֹ֨א הַסֵּ֤פֶר הַזֶּה֙ אֲלֵיכֶ֔ם וְאִתְּכֶ֖ם בְּנֵ֣י אֲדֹנֵיכֶ֑ם וְאִתְּכֶם֙ הָרֶ֣כֶב וְהַסּוּסִ֔ים וְעִ֥יר מִבְצָ֖ר וְהַנָּֽשֶׁק׃ (כח) וּרְאִיתֶ֞ם הַטּ֤וֹב וְהַיָּשָׁר֙ מִבְּנֵ֣י אֲדֹנֵיכֶ֔ם וְשַׂמְתֶּ֖ם עַל־ כִּסֵּ֣א אָבִ֑יו וְהִֽלָּחֲמ֖וּ עַל־ בֵּ֥ית אֲדֹנֵיכֶֽם׃ (כט) וַיִּֽרְאוּ֙ מְאֹ֣ד מְאֹ֔ד וַיֹּ֣אמְר֔וּ הִנֵּה֙ שְׁנֵ֣י הַמְּלָכִ֔ים לֹ֥א עָמְד֖וּ לְפָנָ֑יו וְאֵ֖יךְ נַעֲמֹ֥ד אֲנָֽחְנוּ׃ (ל) וַיִּשְׁלַ֣ח אֲשֶׁר־ עַל־ הַבַּ֣יִת וַאֲשֶׁ֪ר עַל־ הָעִ֟יר וְהַזְּקֵנִים֩ וְהָאֹמְנִ֨ים אֶל־ יֵה֤וּא ׀ לֵאמֹר֙ עֲבָדֶ֣יךָ אֲנַ֔חְנוּ וְכֹ֛ל אֲשֶׁר־ תֹּאמַ֥ר אֵלֵ֖ינוּ נַעֲשֶׂ֑ה לֹֽא־ נַמְלִ֣יךְ אִ֔ישׁ הַטּ֥וֹב בְּעֵינֶ֖יךָ עֲשֵֽׂה׃ (לא) וַיִּכְתֹּ֣ב אֲלֵיהֶם֩ סֵ֨פֶר ׀ שֵׁנִ֜ית לֵאמֹ֗ר אִם־ לִ֨י אַתֶּ֜ם וּלְקֹלִ֣י ׀ אַתֶּ֣ם שֹׁמְעִ֗ים קְחוּ֙ אֶת־ רָאשֵׁי֙ אַנְשֵׁ֣י בְנֵֽי־ אֲדֹנֵיכֶ֔ם וּבֹ֧אוּ אֵלַ֛י כָּעֵ֥ת מָחָ֖ר יִזְרְעֶ֑אלָה וּבְנֵ֤י הַמֶּ֙לֶךְ֙ שִׁבְעִ֣ים אִ֔ישׁ אֶת־ גְּדֹלֵ֥י הָעִ֖יר מְגַדְּלִ֥ים אוֹתָֽם׃ (לב) וַיְהִ֗י כְּבֹ֤א הַסֵּ֙פֶר֙ אֲלֵיהֶ֔ם וַיִּקְחוּ֙ אֶת־ בְּנֵ֣י הַמֶּ֔לֶךְ וַֽיִּשְׁחֲט֖וּ שִׁבְעִ֣ים אִ֑ישׁ וַיָּשִׂ֤ימוּ אֶת־ רָֽאשֵׁיהֶם֙ בַּדּוּדִ֔ים וַיִּשְׁלְח֥וּ אֵלָ֖יו יִזְרְעֶֽאלָה׃ (לג) וַיָּבֹ֤א הַמַּלְאָךְ֙ וַיַּגֶּד־ ל֣וֹ לֵאמֹ֔ר הֵבִ֖יאוּ רָאשֵׁ֣י בְנֵֽי־ הַמֶּ֑לֶךְ וַיֹּ֗אמֶר שִׂ֣ימוּ אֹתָ֞ם שְׁנֵ֧י צִבֻּרִ֛ים פֶּ֥תַח הַשַּׁ֖עַר עַד־ הַבֹּֽקֶר׃ (לד) וַיְהִ֤י בַבֹּ֙קֶר֙ וַיֵּצֵ֣א וַֽיַּעֲמֹ֔ד אַ֔רְצָה אֲשֶׁר־ דִּבֶּ֥ר יְהוָ֖ה עַל־ בֵּ֣ית אַחְאָ֑ב וַיהוָ֣ה עָשָׂ֔ה אֵ֚ת אֲשֶׁ֣ר דִּבֶּ֔ר בְּיַ֖ד עַבְדּ֥וֹ אֵלִיָּֽהוּ׃ (לו) וַיַּ֣ךְ יֵה֗וּא אֵ֣ת כָּל־ הַנִּשְׁאָרִ֤ים לְבֵית־ אַחְאָב֙ בְּיִזְרְעֶ֔אל וְכָל־ גְּדֹלָ֖יו וּמְיֻדָּעָ֣יו וְכֹהֲנָ֑יו עַד־ בִּלְתִּ֥י הִשְׁאִֽיר־ ל֖וֹ שָׂרִֽיד׃ (לז) וַיָּ֙קָם֙ וַיָּבֹ֔א וַיֵּ֖לֶךְ שֹׁמְר֑וֹן ה֛וּא בֵּֽית־ עֵ֥קֶד הָרֹעִ֖ים בַּדָּֽרֶךְ׃ (לח) יְהוֹנָדָ֥ב יֵ֛שׁ וָיֵ֖שׁ תְּנָ֣ה אֶת־ יָדֶ֑ךָ וַיִּתֵּ֣ן יָד֔וֹ וַיַּעֲלֵ֥הוּ אֵלָ֖יו אֶל־ הַמֶּרְכָּבָֽה׃ (ב) וַיֹּ֙אמֶר֙ לְכָ֣ה אִתִּ֔י וּרְאֵ֖ה בְּקִנְאָתִ֣י לַיהוָ֑ה וַיַּרְכִּ֥בוּ אֹת֖וֹ בְּרִכְבּֽוֹ׃ (ג) וַיָּבֹא֙ שֹֽׁמְר֔וֹן וַ֠יַּךְ אֶת־ כָּל־ הַנִּשְׁאָרִ֧ים לְאַחְאָ֛ב בְּשֹׁמְר֖וֹן עַד־ הִשְׁמִיד֑וֹ כִּדְבַ֣ר יְהוָ֔ה אֲשֶׁ֥ר דִּבֶּ֖ר אֶל־ אֵלִיָּֽהוּ׃ (ד) וַיִּקְבֹּ֤ץ יֵהוּא֙ אֶת־ כָּל־ הָעָ֔ם וַיֹּ֣אמֶר אֲלֵהֶ֔ם אַחְאָ֕ב עָבַ֥ד אֶת־ הַבַּ֖עַל מְעָ֑ט יֵה֖וּא יַעַבְדֶ֥נּוּ הַרְבֵּֽה׃ (ה) וְעַתָּ֣ה כָל־ נְבִיאֵ֣י הַבַּ֡עַל כָּל־ עֹבְדָ֣יו וְכָל־ כֹּהֲנָיו֩ קִרְא֨וּ אֵלַ֜י אִ֣ישׁ אַל־ יִפָּקֵ֗ד כִּי֩ זֶ֨בַח גָּד֥וֹל לִי֙ לַבַּ֔עַל כֹּ֥ל אֲשֶׁר־ יִפָּקֵ֖ד לֹ֣א יִֽחְיֶ֑ה וְיֵהוּא֙ עָשָׂ֣ה בְעָקְבָּ֔ה לְמַ֥עַן הַאֲבִ֖יד אֶת־ עֹבְדֵ֥י הַבָּֽעַל׃ (ו) וַיֹּ֣אמֶר יֵה֗וּא קַדְּשׁ֧וּ עֲצָרָ֛ה לַבַּ֖עַל וַיִּקְרָֽאוּ׃ (ז) וַיִּשְׁלַ֤ח יֵהוּא֙ בְּכָל־ יִשְׂרָאֵ֔ל וַיָּבֹ֙אוּ֙ כָּל־ עֹבְדֵ֣י הַבַּ֔עַל וְלֹֽא־ נִשְׁאַ֥ר אִ֖ישׁ אֲשֶׁ֣ר לֹֽא־ בָ֑א וַיָּבֹ֙אוּ֙ בֵּ֣ית הַבַּ֔עַל וַיִּמָּלֵ֥א בֵית־ הַבַּ֖עַל פֶּ֥ה לָפֶֽה׃ (ח) וַיֹּ֗אמֶר לַֽאֲשֶׁר֙ עַל־ הַמֶּלְתָּחָ֔ה הוֹצֵ֣א לְב֔וּשׁ לְכֹ֖ל עֹבְדֵ֣י הַבָּ֑עַל וַיֹּצֵ֥א לָהֶ֖ם הַמַּלְבּֽוּשׁ׃ (ט) וַיָּבֹ֥א יֵה֛וּא וִיהוֹנָדָ֥ב בֶּן־ רֵכָ֖ב בֵּ֣ית הַבָּ֑עַל וַיֹּ֜אמֶר לְעֹבְדֵ֣י הַבַּ֗עַל חַפְּשׂ֤וּ וּרְאוּ֙ פֶּן־ יֶשׁ־ פֹּ֤ה עִמָּכֶם֙ מֵעַבְדֵ֣י יְהוָ֔ה כִּ֛י אִם־ עֹבְדֵ֥י הַבַּ֖עַל לְבַדָּֽם׃ (י) וַיָּבֹ֕אוּ (יא) וַיְהִ֞י כְּכַלֹּת֣וֹ ׀ לַעֲשׂ֣וֹת הָעֹלָ֗ה וַיֹּ֣אמֶר יֵ֠הוּא לָרָצִ֨ים וְלַשָּׁלִשִׁ֜ים בֹּ֤אוּ הַכּוּם֙ אִ֣ישׁ אַל־ יֵצֵ֔א וַיַּכּ֖וּם לְפִי־ חָ֑רֶב וַיַּשְׁלִ֗כוּ הָֽרָצִים֙ וְהַשָּׁ֣לִשִׁ֔ים וַיֵּלְכ֖וּ עַד־ עִ֥יר בֵּית־ הַבָּֽעַל׃ (יב) וַיֹּצִ֛אוּ אֶת־ מַצְּב֥וֹת בֵּית־ הַבַּ֖עַל וַֽיִּשְׂרְפֽוּהָ׃ (יג) וַֽיִּתְּצ֔וּ אֵ֖ת מַצְּבַ֣ת הַבָּ֑עַל וַֽיִּתְּצוּ֙ אֶת־ בֵּ֣ית הַבַּ֔עַל יֵה֖וּא מֵאַֽחֲרֵיהֶ֑ם עֶגְלֵי֙ הַזָּהָ֔ב אֲשֶׁ֥ר בֵּֽית־ אֵ֖ל וַאֲשֶׁ֥ר בְּדָֽן׃ (טז) וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ יֵה֗וּא יַ֤עַן אֲשֶׁר־ הֱטִיבֹ֙תָ֙ לַעֲשׂ֤וֹת הַיָּשָׁר֙ בְּעֵינַ֔י כְּכֹל֙ אֲשֶׁ֣ר בִּלְבָבִ֔י עָשִׂ֖יתָ לְבֵ֣ית אַחְאָ֑ב בְּנֵ֣י רְבִעִ֔ים יֵשְׁב֥וּ לְךָ֖ עַל־ כִּסֵּ֥א יִשְׂרָאֵֽל׃ (יז) וְיֵה֗וּא לֹ֥א שָׁמַ֛ר לָלֶ֛כֶת בְּתֽוֹרַת־ יְהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בְּכָל־ לְבָב֑וֹ לֹ֣א סָ֗ר מֵעַל֙ חַטֹּ֣אות יָֽרָבְעָ֔ם אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (יח) בַּיָּמִ֣ים הָהֵ֔ם הֵחֵ֣ל יְהוָ֔ה לְקַצּ֖וֹת בְּיִשְׂרָאֵ֑ל וַיַּכֵּ֥ם חֲזָאֵ֖ל בְּכָל־ גְּב֥וּל יִשְׂרָאֵֽל׃ (יט) מִן־ הַיַּרְדֵּן֙ מִזְרַ֣ח הַשֶּׁ֔מֶשׁ אֵ֚ת כָּל־ אֶ֣רֶץ הַגִּלְעָ֔ד הַגָּדִ֥י וְהָרֻאובֵנִ֖י וְהַֽמְנַשִּׁ֑י מֵעֲרֹעֵר֙ אֲשֶׁ֣ר עַל־ נַ֣חַל אַרְנֹ֔ן וְהַגִּלְעָ֖ד וְהַבָּשָֽׁן׃ (כ) וְיֶ֨תֶר דִּבְרֵ֥י יֵה֛וּא וְכָל־ אֲשֶׁ֥ר עָשָׂ֖ה וְכָל־ גְּבוּרָת֑וֹ הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כא) וַיִּשְׁכַּ֤ב יֵהוּא֙ עִם־ אֲבֹתָ֔יו וַיִּקְבְּר֥וּ אֹת֖וֹ אֲחַזְיָ֔הוּ (וראתה) רָאֲתָ֖ה כִּ֣י מֵ֣ת בְּנָ֑הּ וַתָּ֙קָם֙ וַתְּאַבֵּ֔ד אֵ֖ת כָּל־ זֶ֥רַע הַמַּמְלָכָֽה׃ (כד) וַתִּקַּ֣ח יְהוֹשֶׁ֣בַע בַּת־ הַמֶּֽלֶךְ־ י֠וֹרָם אֲח֨וֹת אֲחַזְיָ֜הוּ אֶת־ יוֹאָ֣שׁ הוּמָֽת׃ (כה) וַיְהִ֤י אִתָּהּ֙ בֵּ֣ית יְהוָ֔ה מִתְחַבֵּ֖א שֵׁ֣שׁ שָׁנִ֑ים וַעֲתַלְיָ֖ה מֹלֶ֥כֶת עַל־ הָאָֽרֶץ׃ (כו) וּבַשָּׁנָ֣ה הַ֠שְּׁבִיעִית שָׁלַ֨ח יְהוֹיָדָ֜ע וַיִּקַּ֣ח ׀ אֶת־ שָׂרֵ֣י (המאיות) הַמֵּא֗וֹת לַכָּרִי֙ וְלָ֣רָצִ֔ים וַיָּבֵ֥א אֹתָ֛ם אֵלָ֖יו בֵּ֣ית יְהוָ֑ה וַיִּכְרֹת֩ לָהֶ֨ם בְּרִ֜ית וַיַּשְׁבַּ֤ע אֹתָם֙ בְּבֵ֣ית יְהוָ֔ה וַיַּ֥רְא אֹתָ֖ם אֶת־ בֶּן־ הַמֶּֽלֶךְ׃ (כז) וַיְצַוֵּ֣ם לֵאמֹ֔ר זֶ֥ה הַדָּבָ֖ר אֲשֶׁ֣ר תַּעֲשׂ֑וּן הַשְּׁלִשִׁ֤ית מִכֶּם֙ בָּאֵ֣י הַשַּׁבָּ֔ת וְשֹׁ֣מְרֵ֔י מִשְׁמֶ֖רֶת בֵּ֥ית הַמֶּֽלֶךְ׃ (כח) וְהַשְּׁלִשִׁית֙ בְּשַׁ֣עַר ס֔וּר וְהַשְּׁלִשִׁ֥ית בַּשַּׁ֖עַר אַחַ֣ר הָרָצִ֑ים וּשְׁמַרְתֶּ֛ם אֶת־ מִשְׁמֶ֥רֶת הַבַּ֖יִת מַסָּֽח׃ (כט) וּשְׁתֵּ֤י הַיָּדוֹת֙ בָּכֶ֔ם כֹּ֖ל יֹצְאֵ֣י הַשַּׁבָּ֑ת וְשָֽׁמְר֛וּ אֶת־ מִשְׁמֶ֥רֶת בֵּית־ יְהוָ֖ה אֶל־ הַמֶּֽלֶךְ׃ (ל) וְהִקַּפְתֶּ֨ם עַל־ הַמֶּ֜לֶךְ סָבִ֗יב אִ֚ישׁ וְכֵלָ֣יו בְּיָד֔וֹ וְהַבָּ֥א אֶל־ הַשְּׂדֵר֖וֹת יוּמָ֑ת וִהְי֥וּ אֶת־ הַמֶּ֖לֶךְ בְּצֵאת֥וֹ וּבְבֹאֽוֹ׃ (לא) וַֽיַּעֲשׂ֞וּ שָׂרֵ֣י (המאיות) הַמֵּא֗וֹת כְּכֹ֣ל אֲשֶׁר־ צִוָּה֮ יְהוֹיָדָ֣ע הַכֹּהֵן֒ וַיִּקְחוּ֙ אִ֣ישׁ אֶת־ אֲנָשָׁ֔יו בָּאֵ֣י הַשַּׁבָּ֔ת עִ֖ם יֹצְאֵ֣י הַשַּׁבָּ֑ת וַיָּבֹ֖אוּ אֶל־ יְהוֹיָדָ֥ע הַכֹּהֵֽן׃ (לב) וַיִּתֵּ֨ן הַכֹּהֵ֜ן לְשָׂרֵ֣י (המאיות) הַמֵּא֗וֹת אֶֽת־ הַחֲנִית֙ וְאֶת־ הַשְּׁלָטִ֔ים אֲשֶׁ֖ר לַמֶּ֣לֶךְ דָּוִ֑ד אֲשֶׁ֖ר בְּבֵ֥ית יְהוָֽה׃ (לג) וַיַּעַמְד֨וּ הָרָצִ֜ים אִ֣ישׁ ׀ וְכֵלָ֣יו בְּיָד֗וֹ מִכֶּ֨תֶף הַבַּ֤יִת הַיְמָנִית֙ עַד־ כֶּ֤תֶף הַבַּ֙יִת֙ הַשְּׂמָאלִ֔ית לַמִּזְבֵּ֖חַ וְלַבָּ֑יִת עַל־ הַמֶּ֖לֶךְ סָבִֽיב׃ (לד) וַיּוֹצִ֣א אֶת־ בֶּן־ הַמֶּ֗לֶךְ וַיִּתֵּ֤ן עָלָיו֙ אֶת־ הַנֵּ֙זֶר֙ וְאֶת־ הָ֣עֵד֔וּת וַיַּמְלִ֥כוּ אֹת֖וֹ וַיִּמְשָׁחֻ֑הוּ וַיַּכּוּ־ כָ֔ף וַיֹּאמְר֖וּ יְחִ֥י הַמֶּֽלֶךְ׃ (לה) וַתִּשְׁמַ֣ע עֲתַלְיָ֔ה אֶת־ ק֥וֹל הָֽרָצִ֖ין הָעָ֑ם וַתָּבֹ֥א אֶל־ הָעָ֖ם בֵּ֥ית יְהוָֽה׃ (לו) וַתֵּ֡רֶא וְהִנֵּ֣ה הַמֶּלֶךְ֩ עֹמֵ֨ד עַֽל־ הָעַמּ֜וּד כַּמִּשְׁפָּ֗ט וְהַשָּׂרִ֤ים וְהַחֲצֹֽצְרוֹת֙ אֶל־ הַמֶּ֔לֶךְ וְכָל־ עַ֤ם הָאָ֙רֶץ֙ שָׂמֵ֔חַ וְתֹקֵ֖עַ בַּחֲצֹֽצְר֑וֹת וַתִּקְרַ֤ע עֲתַלְיָה֙ אֶת־ בְּגָדֶ֔יהָ וַתִּקְרָ֖א קֶ֥שֶׁר קָֽשֶׁר׃ (לז) וַיְצַו֩ יְהוֹיָדָ֨ע הַכֹּהֵ֜ן אֶת־ שָׂרֵ֥י (המיאות) הַמֵּא֣וֹת ׀ פְּקֻדֵ֣י הַחַ֗יִל וַיֹּ֤אמֶר אֲלֵיהֶם֙ הוֹצִ֤יאוּ אֹתָהּ֙ אֶל־ מִבֵּ֣ית לַשְּׂדֵרֹ֔ת וְהַבָּ֥א אַחֲרֶ֖יהָ הָמֵ֣ת בֶּחָ֑רֶב כִּ֚י אָמַ֣ר הַכֹּהֵ֔ן אַל־ תּוּמַ֖ת בֵּ֥ית יְהוָֽה׃ (לח) וַיָּשִׂ֤מוּ לָהּ֙ יָדַ֔יִם וַתָּב֛וֹא דֶּֽרֶךְ־ מְב֥וֹא הַסּוּסִ֖ים בֵּ֣ית הַמֶּ֑לֶךְ וַתּוּמַ֖ת שָֽׁם׃ (לט) וַיִּכְרֹ֨ת יְהוֹיָדָ֜ע אֶֽת־ הַבְּרִ֗ית בֵּ֤ין יְהוָה֙ וּבֵ֤ין הַמֶּ֙לֶךְ֙ וּבֵ֣ין הָעָ֔ם לִהְי֥וֹת לְעָ֖ם לַֽיהוָ֑ה וּבֵ֥ין הַמֶּ֖לֶךְ וּבֵ֥ין הָעָֽם׃ (מ) וַיָּבֹ֣אוּ כָל־ עַם֩ הָאָ֨רֶץ בֵּית־ הַבַּ֜עַל וַֽיִּתְּצֻ֗הוּ אֶת־ (מזבחתו) מִזְבְּחֹתָ֤יו וְאֶת־ צְלָמָיו֙ שִׁבְּר֣וּ הֵיטֵ֔ב וְאֵ֗ת מַתָּן֙ כֹּהֵ֣ן הַבַּ֔עַל הָרְג֖וּ לִפְנֵ֣י הַֽמִּזְבְּח֑וֹת וַיָּ֧שֶׂם הַכֹּהֵ֛ן פְּקֻדּ֖וֹת עַל־ בֵּ֥ית יְהוָֽה׃ (מא) וַיִּקַּ֣ח אֶת־ שָׂרֵ֣י הַ֠מֵּאוֹת וְאֶת־ הַכָּרִ֨י וְאֶת־ הָרָצִ֜ים וְאֵ֣ת ׀ כָּל־ עַ֣ם הָאָ֗רֶץ וַיֹּרִ֤ידוּ אֶת־ הַמֶּ֙לֶךְ֙ מִבֵּ֣ית יְהוָ֔ה וַיָּב֛וֹאוּ דֶּֽרֶך־ שַׁ֥עַר הָרָצִ֖ים בֵּ֣ית הַמֶּ֑לֶךְ וַיֵּ֖שֶׁב עַל־ כִּסֵּ֥א הַמְּלָכִֽים׃ (מב) וַיִּשְׂמַ֥ח כָּל־ עַם־ הָאָ֖רֶץ וְהָעִ֣יר שָׁקָ֑טָה וְאֶת־ עֲתַלְיָ֛הוּ הֵמִ֥יתוּ בַחֶ֖רֶב בֵּ֥ית (מלך) הַמֶּֽלֶךְ׃ (מג) בֶּן־ שֶׁ֥בַע שָׁנִ֖ים יְהוֹאָ֥שׁ בְּמָלְכֽוֹ׃ (מד) בִּשְׁנַת־ שֶׁ֤בַע לְיֵהוּא֙ מָלַ֣ךְ יְהוֹאָ֔שׁ וְאַרְבָּעִ֣ים שָׁנָ֔ה מָלַ֖ךְ בִּירֽוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ צִבְיָ֖ה מִבְּאֵ֥ר שָֽׁבַע׃
