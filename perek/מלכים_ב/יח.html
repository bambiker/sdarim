מלכים ב פרק יח לפי החלוקה לסדרים

פרק יח בספר מלכים ב (37 פסוקים) נמצא במלכים סדר כט (פסוקים מג–מז) ובמלכים סדר ל (פסוקים א–לב).

מלכים סדר כט

(א) וַיִּשְׁכַּ֤ב אָחָז֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֥ר עִם־ אֲבֹתָ֖יו בְּעִ֣יר דָּוִ֑ד וַיִּמְלֹ֛ךְ חִזְקִיָּ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ (ב) בִּשְׁנַת֙ שְׁתֵּ֣ים עֶשְׂרֵ֔ה לְאָחָ֖ז מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ הוֹשֵׁ֨עַ בֶּן־ אֵלָ֧ה בְשֹׁמְר֛וֹן עַל־ יִשְׂרָאֵ֖ל תֵּ֥שַׁע שָׁנִֽים׃ (ג) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה רַ֗ק לֹ֚א כְּמַלְכֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֥ר הָי֖וּ לְפָנָֽיו׃ (ד) עָלָ֣יו עָלָ֔ה שַׁלְמַנְאֶ֖סֶר מֶ֣לֶךְ אַשּׁ֑וּר וַֽיְהִי־ ל֤וֹ הוֹשֵׁ֙עַ֙ עֶ֔בֶד וַיָּ֥שֶׁב ל֖וֹ מִנְחָֽה׃ (ה) וַיִּמְצָא֩ מֶֽלֶךְ־ אַשּׁ֨וּר בְּהוֹשֵׁ֜עַ קֶ֗שֶׁר אֲשֶׁ֨ר שָׁלַ֤ח מַלְאָכִים֙ אֶל־ ס֣וֹא מֶֽלֶךְ־ מִצְרַ֔יִם וְלֹא־ הֶעֱלָ֥ה מִנְחָ֛ה לְמֶ֥לֶךְ אַשּׁ֖וּר כְּשָׁנָ֣ה בְשָׁנָ֑ה וַֽיַּעַצְרֵ֙הוּ֙ מֶ֣לֶךְ אַשּׁ֔וּר וַיַּאַסְרֵ֖הוּ בֵּ֥ית כֶּֽלֶא׃ (ו) וַיַּ֥עַל מֶֽלֶךְ־ אַשּׁ֖וּר בְּכָל־ הָאָ֑רֶץ וַיַּ֙עַל֙ שֹׁמְר֔וֹן וַיָּ֥צַר עָלֶ֖יהָ שָׁלֹ֥שׁ שָׁנִֽים׃ (ז) בִּשְׁנַ֨ת הַתְּשִׁיעִ֜ית לְהוֹשֵׁ֗עַ לָכַ֤ד מֶֽלֶךְ־ אַשּׁוּר֙ אֶת־ שֹׁ֣מְר֔וֹן וַיֶּ֥גֶל אֶת־ יִשְׂרָאֵ֖ל אַשּׁ֑וּרָה וַיֹּ֨שֶׁב אֹתָ֜ם בַּחְלַ֧ח וּבְחָב֛וֹר נְהַ֥ר גּוֹזָ֖ן וְעָרֵ֥י מָדָֽי׃ (ח) וַיְהִ֗י כִּֽי־ חָטְא֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ לַיהוָ֣ה אֱלֹהֵיהֶ֔ם הַמַּעֲלֶ֤ה אֹתָם֙ מֵאֶ֣רֶץ מִצְרַ֔יִם מִתַּ֕חַת יַ֖ד פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרָ֑יִם וַיִּֽירְא֖וּ אֱלֹהִ֥ים אֲחֵרִֽים׃ (ט) וַיֵּֽלְכוּ֙ בְּחֻקּ֣וֹת הַגּוֹיִ֔ם אֲשֶׁר֙ הוֹרִ֣ישׁ יְהוָ֔ה מִפְּנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל וּמַלְכֵ֥י יִשְׂרָאֵ֖ל אֲשֶׁ֥ר עָשֽׂוּ׃ (י) וַיְחַפְּא֣וּ בְנֵֽי־ יִשְׂרָאֵ֗ל דְּבָרִים֙ אֲשֶׁ֣ר לֹא־ כֵ֔ן עַל־ יְהוָ֖ה אֱלֹהֵיהֶ֑ם וַיִּבְנ֨וּ לָהֶ֤ם בָּמוֹת֙ בְּכָל־ עָ֣רֵיהֶ֔ם מִמִּגְדַּ֥ל נוֹצְרִ֖ים עַד־ עִ֥יר מִבְצָֽר׃ (יא) וַיַּצִּ֧בוּ לָהֶ֛ם מַצֵּב֖וֹת וַאֲשֵׁרִ֑ים עַ֚ל כָּל־ גִּבְעָ֣ה גְבֹהָ֔ה וְתַ֖חַת כָּל־ עֵ֥ץ רַעֲנָֽן׃ (יב) וַיְקַטְּרוּ־ שָׁם֙ בְּכָל־ בָּמ֔וֹת כַּגּוֹיִ֕ם אֲשֶׁר־ הֶגְלָ֥ה יְהוָ֖ה מִפְּנֵיהֶ֑ם וַֽיַּעֲשׂוּ֙ דְּבָרִ֣ים רָעִ֔ים לְהַכְעִ֖יס אֶת־ יְהוָֽה׃ (יג) וַיַּֽעַבְד֖וּ הַגִּלֻּלִ֑ים אֲשֶׁ֨ר אָמַ֤ר יְהוָה֙ לָהֶ֔ם לֹ֥א תַעֲשׂ֖וּ אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (יד) וַיָּ֣עַד יְהוָ֡ה בְּיִשְׂרָאֵ֣ל וּבִיהוּדָ֡ה בְּיַד֩ כָּל־ (נביאו) נְבִיאֵ֨י כָל־ חֹזֶ֜ה לֵאמֹ֗ר שֻׁ֝֠בוּ מִדַּרְכֵיכֶ֤ם הָֽרָעִים֙ וְשִׁמְרוּ֙ מִצְוֺתַ֣י חֻקּוֹתַ֔י כְּכָ֨ל־ הַתּוֹרָ֔ה אֲשֶׁ֥ר צִוִּ֖יתִי אֶת־ אֲבֹֽתֵיכֶ֑ם וַֽאֲשֶׁר֙ שָׁלַ֣חְתִּי אֲלֵיכֶ֔ם בְּיַ֖ד עֲבָדַ֥י הַנְּבִיאִֽים׃ (טו) וְלֹ֖א שָׁמֵ֑עוּ וַיַּקְשׁ֤וּ אֶת־ עָרְפָּם֙ כְּעֹ֣רֶף אֲבוֹתָ֔ם אֲשֶׁר֙ לֹ֣א הֶאֱמִ֔ינוּ בַּֽיהוָ֖ה אֱלֹהֵיהֶֽם׃ (טז) וַיִּמְאֲס֣וּ אֶת־ חֻקָּ֗יו וְאֶת־ בְּרִיתוֹ֙ אֲשֶׁ֣ר כָּרַ֣ת אֶת־ אֲבוֹתָ֔ם וְאֵת֙ עֵֽדְוֺתָ֔יו אֲשֶׁ֥ר הֵעִ֖יד בָּ֑ם וַיֵּ֨לְכ֜וּ אַחֲרֵ֤י הַהֶ֙בֶל֙ וַיֶּהְבָּ֔לוּ וְאַחֲרֵ֤י הַגּוֹיִם֙ אֲשֶׁ֣ר סְבִֽיבֹתָ֔ם אֲשֶׁ֨ר צִוָּ֤ה יְהוָה֙ אֹתָ֔ם לְבִלְתִּ֖י עֲשׂ֥וֹת כָּהֶֽם׃ (יז) וַיַּעַזְב֗וּ אֶת־ כָּל־ מִצְוֺת֙ יְהוָ֣ה אֱלֹהֵיהֶ֔ם וַיַּעֲשׂ֥וּ לָהֶ֛ם מַסֵּכָ֖ה (שנים) שְׁנֵ֣י עֲגָלִ֑ים וַיַּעֲשׂ֣וּ אֲשֵׁירָ֗ה וַיִּֽשְׁתַּחֲווּ֙ לְכָל־ צְבָ֣א הַשָּׁמַ֔יִם וַיַּעַבְד֖וּ אֶת־ הַבָּֽעַל׃ (יח) וַֽ֠יַּעֲבִירוּ אֶת־ בְּנֵיהֶ֤ם וְאֶת־ בְּנֽוֹתֵיהֶם֙ בָּאֵ֔שׁ וַיִּקְסְמ֥וּ קְסָמִ֖ים וַיְנַחֵ֑שׁוּ וַיִּֽתְמַכְּר֗וּ לַעֲשׂ֥וֹת הָרַ֛ע בְּעֵינֵ֥י יְהוָ֖ה לְהַכְעִיסֽוֹ׃ (יט) וַיִּתְאַנַּ֨ף יְהוָ֤ה מְאֹד֙ בְּיִשְׂרָאֵ֔ל וַיְסִרֵ֖ם מֵעַ֣ל פָּנָ֑יו לֹ֣א נִשְׁאַ֔ר רַ֛ק שֵׁ֥בֶט יְהוּדָ֖ה לְבַדּֽוֹ׃ (כ) גַּם־ יְהוּדָ֕ה לֹ֣א שָׁמַ֔ר אֶת־ מִצְוֺ֖ת יְהוָ֣ה אֱלֹהֵיהֶ֑ם וַיֵּ֣לְכ֔וּ בְּחֻקּ֥וֹת יִשְׂרָאֵ֖ל אֲשֶׁ֥ר עָשֽׂוּ׃ (כא) וַיִּמְאַ֨ס יְהוָ֜ה בְּכָל־ זֶ֤רַע יִשְׂרָאֵל֙ וַיְעַנֵּ֔ם וַֽיִּתְּנֵ֖ם בְּיַד־ שֹׁסִ֑ים עַ֛ד אֲשֶׁ֥ר הִשְׁלִיכָ֖ם מִפָּנָֽיו׃ (כב) כִּֽי־ קָרַ֣ע יִשְׂרָאֵ֗ל מֵעַל֙ בֵּ֣ית דָּוִ֔ד וַיַּמְלִ֖יכוּ אֶת־ יָרָבְעָ֣ם בֶּן־ נְבָ֑ט (וידא) וַיַּדַּ֨ח יָרָבְעָ֤ם אֶת־ יִשְׂרָאֵל֙ מֵאַחֲרֵ֣י יְהוָ֔ה וְהֶחֱטֵיאָ֖ם חֲטָאָ֥ה גְדוֹלָֽה׃ (כג) וַיֵּֽלְכוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל בְּכָל־ חַטֹּ֥אות יָרָבְעָ֖ם אֲשֶׁ֣ר עָשָׂ֑ה לֹא־ סָ֖רוּ מִמֶּֽנָּה׃ (כד) עַ֠ד אֲשֶׁר־ הֵסִ֨יר יְהוָ֤ה אֶת־ יִשְׂרָאֵל֙ מֵעַ֣ל פָּנָ֔יו כַּאֲשֶׁ֣ר דִּבֶּ֔ר בְּיַ֖ד כָּל־ עֲבָדָ֣יו הַנְּבִיאִ֑ים וַיִּ֨גֶל יִשְׂרָאֵ֜ל מֵעַ֤ל אַדְמָתוֹ֙ אַשּׁ֔וּרָה עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כה) וַיָּבֵ֣א מֶֽלֶךְ־ אַשּׁ֡וּר מִבָּבֶ֡ל וּ֠מִכּוּ֠תָה וּמֵעַוָּ֤א וּמֵֽחֲמָת֙ וּסְפַרְוַ֔יִם וַיֹּ֙שֶׁב֙ בְּעָרֵ֣י שֹֽׁמְר֔וֹן תַּ֖חַת בְּנֵ֣י יִשְׂרָאֵ֑ל וַיִּֽרְשׁוּ֙ אֶת־ שֹׁ֣מְר֔וֹן וַיֵּֽשְׁב֖וּ בְּעָרֶֽיהָ׃ (כו) וַיְהִ֗י בִּתְחִלַּת֙ שִׁבְתָּ֣ם שָׁ֔ם לֹ֥א יָרְא֖וּ אֶת־ יְהוָ֑ה וַיְשַׁלַּ֨ח יְהוָ֤ה בָּהֶם֙ אֶת־ הָ֣אֲרָי֔וֹת וַיִּֽהְי֥וּ הֹרְגִ֖ים בָּהֶֽם׃ (כז) וַיֹּאמְר֗וּ לְמֶ֣לֶךְ אַשּׁוּר֮ לֵאמֹר֒ הַגּוֹיִ֗ם אֲשֶׁ֤ר הִגְלִ֙יתָ֙ וַתּ֙וֹשֶׁב֙ בְּעָרֵ֣י שֹׁמְר֔וֹן לֹ֣א יָֽדְע֔וּ אֶת־ מִשְׁפַּ֖ט אֱלֹהֵ֣י הָאָ֑רֶץ וַיְשַׁלַּח־ בָּ֣ם אֶת־ הָאֲרָי֗וֹת וְהִנָּם֙ מְמִיתִ֣ים אוֹתָ֔ם כַּאֲשֶׁר֙ אֵינָ֣ם יֹדְעִ֔ים אֶת־ מִשְׁפַּ֖ט אֱלֹהֵ֥י הָאָֽרֶץ׃ (כח) וַיְצַ֨ו מֶֽלֶךְ־ אַשּׁ֜וּר לֵאמֹ֗ר הֹלִ֤יכוּ שָׁ֙מָּה֙ אֶחָ֤ד מֵהַכֹּֽהֲנִים֙ אֲשֶׁ֣ר הִגְלִיתֶ֣ם מִשָּׁ֔ם וְיֵלְכ֖וּ וְיֵ֣שְׁבוּ שָׁ֑ם וְיֹרֵ֕ם אֶת־ מִשְׁפַּ֖ט אֱלֹהֵ֥י הָאָֽרֶץ׃ (כט) וַיָּבֹ֞א אֶחָ֣ד מֵהַכֹּהֲנִ֗ים אֲשֶׁ֤ר הִגְלוּ֙ מִשֹּׁ֣מְר֔וֹן וַיֵּ֖שֶׁב בְּבֵֽית־ אֵ֑ל וַֽיְהִי֙ מוֹרֶ֣ה אֹתָ֔ם אֵ֖יךְ יִֽירְא֥וּ אֶת־ יְהוָֽה׃ (ל) וַיִּהְי֣וּ עֹשִׂ֔ים גּ֥וֹי גּ֖וֹי אֱלֹהָ֑יו וַיַּנִּ֣יחוּ ׀ בְּבֵ֣ית הַבָּמ֗וֹת אֲשֶׁ֤ר עָשׂוּ֙ הַשֹּׁ֣מְרֹנִ֔ים גּ֥וֹי גּוֹי֙ בְּעָ֣רֵיהֶ֔ם אֲשֶׁ֛ר הֵ֥ם יֹשְׁבִ֖ים שָֽׁם׃ (לא) וְאַנְשֵׁ֣י בָבֶ֗ל עָשׂוּ֙ אֶת־ סֻכּ֣וֹת בְּנ֔וֹת וְאַנְשֵׁי־ כ֔וּת עָשׂ֖וּ אֶת־ נֵֽרְגַ֑ל וְאַנְשֵׁ֥י חֲמָ֖ת עָשׂ֥וּ אֶת־ אֲשִׁימָֽא׃ (לב) וְהָעַוִּ֛ים עָשׂ֥וּ נִבְחַ֖ז וְאֶת־ תַּרְתָּ֑ק וְהַסְפַרְוִ֗ים שֹׂרְפִ֤ים אֶת־ בְּנֵיהֶם֙ בָּאֵ֔שׁ לְאַדְרַמֶּ֥לֶךְ וַֽעֲנַמֶּ֖לֶךְ (אלה) אֱלֹהֵ֥י (ספרים) סְפַרְוָֽיִם׃ (לג) וַיִּהְי֥וּ יְרֵאִ֖ים אֶת־ יְהוָ֑ה וַיַּעֲשׂ֨וּ לָהֶ֤ם מִקְצוֹתָם֙ כֹּהֲנֵ֣י בָמ֔וֹת וַיִּהְי֛וּ עֹשִׂ֥ים לָהֶ֖ם בְּבֵ֥ית הַבָּמֽוֹת׃ (לד) אֶת־ יְהוָ֖ה הָי֣וּ יְרֵאִ֑ים וְאֶת־ אֱלֹֽהֵיהֶם֙ הָי֣וּ עֹֽבְדִ֔ים כְּמִשְׁפַּט֙ הַגּוֹיִ֔ם אֲשֶׁר־ הִגְל֥וּ אֹתָ֖ם מִשָּֽׁם׃ (לה) עַ֣ד הַיּ֤וֹם הַזֶּה֙ הֵ֣ם עֹשִׂ֔ים וּלְפָרָשִֽׁים׃ (כ) עַתָּה֙ הֲמִבַּלְעֲדֵ֣י יְהוָ֔ה עָלִ֛יתִי עַל־ הַמָּק֥וֹם הַזֶּ֖ה לְהַשְׁחִת֑וֹ יְהוָה֙ אָמַ֣ר אֵלַ֔י עֲלֵ֛ה עַל־ הָאָ֥רֶץ הַזֹּ֖את וְהַשְׁחִיתָֽהּ׃ (כא) וַיֹּ֣אמֶר אֶלְיָקִ֣ים בֶּן־ חִ֠לְקִיָּהוּ וְשֶׁבְנָ֨ה וְיוֹאָ֜ח אֶל־ רַב־ שָׁקֵ֗ה דַּבֶּר־ נָ֤א אֶל־ עֲבָדֶ֙יךָ֙ אֲרָמִ֔ית כִּ֥י שֹׁמְעִ֖ים אֲנָ֑חְנוּ וְאַל־ תְּדַבֵּ֤ר עִמָּ֙נוּ֙ יְהוּדִ֔ית בְּאָזְנֵ֣י הָעָ֔ם אֲשֶׁ֖ר עַל־ הַחֹמָֽה׃ (כב) וַיֹּ֨אמֶר אֲלֵיהֶ֜ם רַב־ שָׁקֵ֗ה הַעַ֨ל אֲדֹנֶ֤יךָ וְאֵלֶ֙יךָ֙ שְׁלָחַ֣נִי אֲדֹנִ֔י לְדַבֵּ֖ר אֶת־ הַדְּבָרִ֣ים הָאֵ֑לֶּה הֲלֹ֣א עַל־ הָאֲנָשִׁ֗ים הַיֹּֽשְׁבִים֙ עַל־ הַ֣חֹמָ֔ה לֶאֱכֹ֣ל אֶת (חריהם) צוֹאָתָ֗ם וְלִשְׁתּ֛וֹת אֶת־ (שיניהם) מימֵ֥י רַגְלֵיהֶ֖ם עִמָּכֶֽם׃ (כג) וַֽיַּעֲמֹד֙ רַב־ שָׁקֵ֔ה וַיִּקְרָ֥א בְקוֹל־ גָּד֖וֹל יְהוּדִ֑ית וַיְדַבֵּ֣ר וַיֹּ֔אמֶר שִׁמְע֛וּ דְּבַר־ הַמֶּ֥לֶךְ הַגָּד֖וֹל מֶ֥לֶךְ אַשּֽׁוּר׃ (כד) כֹּ֚ה אָמַ֣ר הַמֶּ֔לֶךְ אַל־ יַשִּׁ֥יא לָכֶ֖ם חִזְקִיָּ֑הוּ כִּי־ לֹ֣א יוּכַ֔ל לְהַצִּ֥יל אֶתְכֶ֖ם מִיָּדֽוֹ׃ (כה) וְאַל־ יַבְטַ֨ח אֶתְכֶ֤ם חִזְקִיָּ֙הוּ֙ אֶל־ יְהוָ֣ה לֵאמֹ֔ר הַצֵּ֥ל יַצִּילֵ֖נוּ יְהוָ֑ה וְלֹ֤א תִנָּתֵן֙ אֶת־ הָעִ֣יר הַזֹּ֔את בְּיַ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (כו) אַֽל־ תִּשְׁמְע֖וּ אֶל־ חִזְקִיָּ֑הוּ כִּי֩ כֹ֨ה אָמַ֜ר מֶ֣לֶךְ אַשּׁ֗וּר עֲשֽׂוּ־ אִתִּ֤י בְרָכָה֙ וּצְא֣וּ אֵלַ֔י וְאִכְל֤וּ אִישׁ־ גַּפְנוֹ֙ וְאִ֣ישׁ תְּאֵֽנָת֔וֹ וּשְׁת֖וּ אִ֥ישׁ מֵֽי־ בוֹרֽוֹ׃ (כז) עַד־ בֹּאִי֩ וְלָקַחְתִּ֨י אֶתְכֶ֜ם אֶל־ אֶ֣רֶץ כְּאַרְצְכֶ֗ם אֶרֶץ֩ דָּגָ֨ן וְתִיר֜וֹשׁ אֶ֧רֶץ לֶ֣חֶם וּכְרָמִ֗ים אֶ֣רֶץ זֵ֤ית יִצְהָר֙ וּדְבַ֔שׁ וִֽחְי֖וּ וְלֹ֣א תָמֻ֑תוּ וְאַֽל־ תִּשְׁמְעוּ֙ אֶל־ חִזְקִיָּ֔הוּ כִּֽי־ יַסִּ֤ית אֶתְכֶם֙ לֵאמֹ֔ר יְהוָ֖ה יַצִּילֵֽנוּ׃ (כח) הַהַצֵּ֥ל הִצִּ֛ילוּ אֱלֹהֵ֥י הַגּוֹיִ֖ם אִ֣ישׁ אֶת־ אַרְצ֑וֹ מִיַּ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (כט) אַיֵּה֩ אֱלֹהֵ֨י חֲמָ֜ת וְאַרְפָּ֗ד אַיֵּ֛ה אֱלֹהֵ֥י סְפַרְוַ֖יִם הֵנַ֣ע וְעִוָּ֑ה כִּֽי־ הִצִּ֥ילוּ אֶת־ שֹׁמְר֖וֹן מִיָּדִֽי׃ (ל) מִ֚י בְּכָל־ אֱלֹהֵ֣י הָֽאֲרָצ֔וֹת אֲשֶׁר־ הִצִּ֥ילוּ אֶת־ אַרְצָ֖ם מִיָּדִ֑י כִּי־ יַצִּ֧יל יְהוָ֛ה אֶת־ יְרוּשָׁלִַ֖ם מִיָּדִֽי׃ (לא) וְהֶחֱרִ֣ישׁוּ הָעָ֔ם וְלֹֽא־ עָנ֥וּ אֹת֖וֹ דָּבָ֑ר כִּי־ מִצְוַ֨ת הַמֶּ֥לֶךְ הִ֛יא לֵאמֹ֖ר לֹ֥א תַעֲנֻֽהוּ׃ (לב) וַיָּבֹ֣א אֶלְיָקִ֣ים בֶּן־ חִלְקִיָּ֣ה אֲשֶׁר־ עַל־ הַ֠בַּיִת וְשֶׁבְנָ֨א הַסֹּפֵ֜ר וְיוֹאָ֨ח בֶּן־ אָסָ֧ף הַמַּזְכִּ֛יר אֶל־ חִזְקִיָּ֖הוּ קְרוּעֵ֣י בְגָדִ֑ים וַיַּגִּ֣דוּ ל֔וֹ דִּבְרֵ֖י רַב־ שָׁקֵֽה׃ סוף פרק יח (לג) וַיְהִ֗י כִּשְׁמֹ֙עַ֙ הַמֶּ֣לֶךְ חִזְקִיָּ֔הוּ וַיִּקְרַ֖ע אֶת־ בְּגָדָ֑יו וַיִּתְכַּ֣ס בַּשָּׂ֔ק וַיָּבֹ֖א בֵּ֥ית יְהוָֽה׃ (לד) וַ֠יִּשְׁלַח אֶת־ אֶלְיָקִ֨ים אֲשֶׁר־ עַל־ הַבַּ֜יִת וְשֶׁבְנָ֣א הַסֹּפֵ֗ר וְאֵת֙ זִקְנֵ֣י הַכֹּֽהֲנִ֔ים מִתְכַּסִּ֖ים בַּשַּׂקִּ֑ים אֶל־ יְשַֽׁעְיָ֥הוּ הַנָּבִ֖יא בֶּן־ אָמֽוֹץ׃ (לה) וַיֹּאמְר֣וּ אֵלָ֗יו כֹּ֚ה אָמַ֣ר חִזְקִיָּ֔הוּ יוֹם־ צָרָ֧ה וְתוֹכֵחָ֛ה וּנְאָצָ֖ה הַיּ֣וֹם הַזֶּ֑ה כִּ֣י בָ֤אוּ בָנִים֙ עַד־ מַשְׁבֵּ֔ר וְכֹ֥חַ אַ֖יִן לְלֵדָֽה׃ (לו) אוּלַ֡י יִשְׁמַע֩ יְהוָ֨ה אֱלֹהֶ֜יךָ אֵ֣ת ׀ כָּל־ דִּבְרֵ֣י רַב־ שָׁקֵ֗ה אֲשֶׁר֩ שְׁלָח֨וֹ מֶֽלֶךְ־ אַשּׁ֤וּר ׀ אֲדֹנָיו֙ לְחָרֵף֙ אֱלֹהִ֣ים חַ֔י וְהוֹכִ֙יחַ֙ בַּדְּבָרִ֔ים אֲשֶׁ֥ר שָׁמַ֖ע יְהוָ֣ה אֱלֹהֶ֑יךָ וְנָשָׂ֣אתָ תְפִלָּ֔ה בְּעַ֥ד הַשְּׁאֵרִ֖ית הַנִּמְצָאָֽה׃ (לז) וַיָּבֹ֗אוּ עַבְדֵ֛י הַמֶּ֥לֶךְ חִזְקִיָּ֖הוּ אֶל־ יְשַֽׁעַיָֽהוּ׃ (לח) וַיֹּ֤אמֶר לָהֶם֙ יְשַֽׁעְיָ֔הוּ כֹּ֥ה תֹאמְר֖וּן אֶל־ אֲדֹֽנֵיכֶ֑ם כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ תִּירָא֙ מִפְּנֵ֤י הַדְּבָרִים֙ אֲשֶׁ֣ר שָׁמַ֗עְתָּ אֲשֶׁ֧ר גִּדְּפ֛וּ נַעֲרֵ֥י מֶֽלֶךְ־ אַשּׁ֖וּר אֹתִֽי׃ (לט) הִנְנִ֨י נֹתֵ֥ן בּוֹ֙ ר֔וּחַ וְשָׁמַ֥ע שְׁמוּעָ֖ה וְשָׁ֣ב לְאַרְצ֑וֹ וְהִפַּלְתִּ֥יו בַּחֶ֖רֶב בְּאַרְצֽוֹ׃ (מ) וַיָּ֙שָׁב֙ רַב־ שָׁקֵ֔ה וַיִּמְצָא֙ אֶת־ מֶ֣לֶךְ אַשּׁ֔וּר נִלְחָ֖ם עַל־ לִבְנָ֑ה כִּ֣י שָׁמַ֔ע כִּ֥י נָסַ֖ע מִלָּכִֽישׁ׃ (מא) וַיִּשְׁמַ֗ע אֶל־ תִּרְהָ֤קָה מֶֽלֶך־ כּוּשׁ֙ לֵאמֹ֔ר הִנֵּ֥ה יָצָ֖א לְהִלָּחֵ֣ם אִתָּ֑ךְ וַיָּ֙שָׁב֙ וַיִּשְׁלַ֣ח מַלְאָכִ֔ים אֶל־ חִזְקִיָּ֖הוּ לֵאמֹֽר׃ (מב) כֹּ֣ה תֹאמְר֗וּן אֶל־ חִזְקִיָּ֤הוּ מֶֽלֶךְ־ יְהוּדָה֙ לֵאמֹ֔ר אַל־ יַשִּׁאֲךָ֣ אֱלֹהֶ֔יךָ אֲשֶׁ֥ר אַתָּ֛ה בֹּטֵ֥חַ בּ֖וֹ לֵאמֹ֑ר לֹ֤א תִנָּתֵן֙ יְר֣וּשָׁלִַ֔ם בְּיַ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (מג) הִנֵּ֣ה ׀ אַתָּ֣ה שָׁמַ֗עְתָּ אֵת֩ אֲשֶׁ֨ר עָשׂ֜וּ מַלְכֵ֥י אַשּׁ֛וּר לְכָל־ הָאֲרָצ֖וֹת לְהַֽחֲרִימָ֑ם וְאַתָּ֖ה תִּנָּצֵֽל׃ (מד) הַהִצִּ֨ילוּ אֹתָ֜ם אֱלֹהֵ֤י הַגּוֹיִם֙ אֲשֶׁ֣ר שִׁחֲת֣וּ אֲבוֹתַ֔י אֶת־ גּוֹזָ֖ן וְאֶת־ חָרָ֑ן וְרֶ֥צֶף וּבְנֵי־ עֶ֖דֶן אֲשֶׁ֥ר בִּתְלַאשָּֽׂר׃ (מה) אַיּ֤וֹ מֶֽלֶךְ־ חֲמָת֙ וּמֶ֣לֶךְ אַרְפָּ֔ד וּמֶ֖לֶךְ לָעִ֣יר סְפַרְוָ֑יִם הֵנַ֖ע וְעִוָּֽה׃ (מו) וַיִּקַּ֨ח חִזְקִיָּ֧הוּ אֶת־ הַסְּפָרִ֛ים מִיַּ֥ד הַמַּלְאָכִ֖ים וַיִּקְרָאֵ֑ם וַיַּ֙עַל֙ בֵּ֣ית יְהוָ֔ה וַיִּפְרְשֵׂ֥הוּ חִזְקִיָּ֖הוּ לִפְנֵ֥י יְהוָֽה׃ (מז) וַיִּתְפַּלֵּ֨ל חִזְקִיָּ֜הוּ לִפְנֵ֣י יְהוָה֮ וַיֹּאמַר֒ יְהוָ֞ה אֱלֹהֵ֤י יִשְׂרָאֵל֙ יֹשֵׁ֣ב הַכְּרֻבִ֔ים אַתָּה־ ה֤וּא הָֽאֱלֹהִים֙ לְבַדְּךָ֔ לְכֹ֖ל מַמְלְכ֣וֹת הָאָ֑רֶץ אַתָּ֣ה עָשִׂ֔יתָ אֶת־ הַשָּׁמַ֖יִם וְאֶת־ הָאָֽרֶץ׃ (מח) הַטֵּ֨ה יְהוָ֤ה ׀ אָזְנְךָ֙ וּֽשֲׁמָ֔ע פְּקַ֧ח יְהוָ֛ה עֵינֶ֖יךָ וּרְאֵ֑ה וּשְׁמַ֗ע אֵ֚ת דִּבְרֵ֣י סַנְחֵרִ֔יב אֲשֶׁ֣ר שְׁלָח֔וֹ לְחָרֵ֖ף אֱלֹהִ֥ים חָֽי׃ (מט) אָמְנָ֖ם יְהוָ֑ה הֶחֱרִ֜יבוּ מַלְכֵ֥י אַשּׁ֛וּר אֶת־ הַגּוֹיִ֖ם וְאֶת־ אַרְצָֽם׃ (נ) וְנָתְנ֥וּ אֶת־ אֱלֹהֵיהֶ֖ם בָּאֵ֑שׁ כִּי֩ לֹ֨א אֱלֹהִ֜ים הֵ֗מָּה כִּ֣י אִם־ מַעֲשֵׂ֧ה יְדֵֽי־ אָדָ֛ם עֵ֥ץ וָאֶ֖בֶן וַֽיְאַבְּדֽוּם׃
