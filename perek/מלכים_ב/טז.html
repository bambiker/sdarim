מלכים ב פרק טז לפי החלוקה לסדרים

פרק טז בספר מלכים ב (20 פסוקים) נמצא במלכים סדר כח (פסוקים לג–נא) ובמלכים סדר כט (פסוקים א–א).

מלכים סדר כח

(א) וַיִּשְׁכַּ֤ב עֲזַרְיָה֙ עִם־ אֲבֹתָ֔יו וַיִּקְבְּר֥וּ אֹת֛וֹ עִם־ אֲבֹתָ֖יו בְּעִ֣יר דָּוִ֑ד וַיִּמְלֹ֛ךְ יוֹתָ֥ם בְּנ֖וֹ תַּחְתָּֽיו׃ (ב) בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וּשְׁמֹנֶה֙ שָׁנָ֔ה לַעֲזַרְיָ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ זְכַרְיָ֨הוּ בֶן־ יָרָבְעָ֧ם עַל־ יִשְׂרָאֵ֛ל בְּשֹׁמְר֖וֹן שִׁשָּׁ֥ה חֳדָשִֽׁים׃ (ג) וַיַּ֤עַשׂ הָרַע֙ בְּעֵינֵ֣י יְהוָ֔ה כַּאֲשֶׁ֥ר עָשׂ֖וּ אֲבֹתָ֑יו לֹ֣א סָ֗ר מֵֽחַטֹּאות֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (ד) וַיִּקְשֹׁ֤ר עָלָיו֙ שַׁלֻּ֣ם בֶּן־ יָבֵ֔שׁ וַיַּכֵּ֥הוּ קָֽבָלְ־ עָ֖ם וַיְמִיתֵ֑הוּ וַיִּמְלֹ֖ךְ תַּחְתָּֽיו׃ (ה) וְיֶ֖תֶר דִּבְרֵ֣י זְכַרְיָ֑ה הִנָּ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (ו) ה֣וּא דְבַר־ יְהוָ֗ה אֲשֶׁ֨ר דִּבֶּ֤ר אֶל־ יֵהוּא֙ לֵאמֹ֔ר בְּנֵ֣י רְבִיעִ֔ים יֵשְׁב֥וּ לְךָ֖ עַל־ כִּסֵּ֣א יִשְׂרָאֵ֑ל וַֽיְהִי־ כֵֽן׃ (ז) שַׁלּ֤וּם בֶּן־ יָבֵישׁ֙ מָלַ֔ךְ בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָתֵ֙שַׁע֙ שָׁנָ֔ה לְעֻזִיָּ֖ה מֶ֣לֶךְ יְהוּדָ֑ה וַיִּמְלֹ֥ךְ יֶֽרַח־ יָמִ֖ים בְּשֹׁמְרֽוֹן׃ (ח) וַיַּעַל֩ מְנַחֵ֨ם בֶּן־ גָּדִ֜י מִתִּרְצָ֗ה וַיָּבֹא֙ שֹׁמְר֔וֹן וַיַּ֛ךְ אֶת־ שַׁלּ֥וּם בֶּן־ יָבֵ֖ישׁ בְּשֹׁמְר֑וֹן וַיְמִיתֵ֖הוּ וַיִּמְלֹ֥ךְ תַּחְתָּֽיו׃ (ט) וְיֶ֙תֶר֙ דִּבְרֵ֣י שַׁלּ֔וּם וְקִשְׁר֖וֹ אֲשֶׁ֣ר קָשָׁ֑ר הִנָּ֣ם כְּתֻבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (י) אָ֣ז יַכֶּֽה־ מְ֠נַחֵם אֶת־ תִּפְסַ֨ח וְאֶת־ כָּל־ אֲשֶׁר־ בָּ֤הּ וְאֶת־ גְּבוּלֶ֙יהָ֙ מִתִּרְצָ֔ה כִּ֛י לֹ֥א פָתַ֖ח וַיַּ֑ךְ אֵ֛ת כָּל־ הֶהָ֥רוֹתֶ֖יהָ בִּקֵּֽעַ׃ (יא) בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָתֵ֙שַׁע֙ שָׁנָ֔ה לַעֲזַרְיָ֖ה מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ מְנַחֵ֨ם בֶּן־ גָּדִ֧י עַל־ יִשְׂרָאֵ֛ל עֶ֥שֶׂר שָׁנִ֖ים בְּשֹׁמְרֽוֹן׃ (יב) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לֹ֣א סָ֠ר מֵעַ֨ל חַטֹּ֜אות יָרָבְעָ֧ם בֶּן־ נְבָ֛ט אֲשֶׁר־ הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל כָּל־ יָמָֽיו׃ (יג) בָּ֣א פ֤וּל מֶֽלֶךְ־ אַשּׁוּר֙ עַל־ הָאָ֔רֶץ וַיִּתֵּ֤ן מְנַחֵם֙ לְפ֔וּל אֶ֖לֶף כִּכַּר־ כָּ֑סֶף לִהְי֤וֹת יָדָיו֙ אִתּ֔וֹ לְהַחֲזִ֥יק הַמַּמְלָכָ֖ה בְּיָדֽוֹ׃ (יד) וַיֹּצֵא֩ מְנַחֵ֨ם אֶת־ הַכֶּ֜סֶף עַל־ יִשְׂרָאֵ֗ל עַ֚ל כָּל־ גִּבּוֹרֵ֣י הַחַ֔יִל לָתֵת֙ לְמֶ֣לֶךְ אַשּׁ֔וּר חֲמִשִּׁ֧ים שְׁקָלִ֛ים כֶּ֖סֶף לְאִ֣ישׁ אֶחָ֑ד וַיָּ֙שָׁב֙ מֶ֣לֶךְ אַשּׁ֔וּר וְלֹא־ עָ֥מַד שָׁ֖ם בָּאָֽרֶץ׃ (טו) וְיֶ֛תֶר דִּבְרֵ֥י מְנַחֵ֖ם וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (טז) וַיִּשְׁכַּ֥ב מְנַחֵ֖ם עִם־ אֲבֹתָ֑יו וַיִּמְלֹ֛ךְ פְּקַחְיָ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ (יז) בִּשְׁנַת֙ חֲמִשִּׁ֣ים שָׁנָ֔ה לַעֲזַרְיָ֖ה מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ פְּקַֽחְיָ֨ה בֶן־ מְנַחֵ֧ם עַל־ יִשְׂרָאֵ֛ל בְּשֹׁמְר֖וֹן שְׁנָתָֽיִם׃ (יח) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לֹ֣א סָ֗ר מֵֽחַטֹּאות֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (יט) וַיִּקְשֹׁ֣ר עָלָיו֩ פֶּ֨קַח בֶּן־ רְמַלְיָ֜הוּ שָׁלִישׁ֗וֹ וַיַּכֵּ֨הוּ בְשֹׁמְר֜וֹן בְּאַרְמ֤וֹן בֵּית־ (מלך) הַמֶּ֙לֶךְ֙ אֶת־ אַרְגֹּ֣ב וְאֶת־ הָאַרְיֵ֔ה וְעִמּ֛וֹ חֲמִשִּׁ֥ים אִ֖ישׁ מִבְּנֵ֣י גִלְעָדִ֑ים וַיְמִיתֵ֖הוּ וַיִּמְלֹ֥ךְ תַּחְתָּֽיו׃ (כ) וְיֶ֛תֶר דִּבְרֵ֥י פְקַחְיָ֖ה וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הִנָּ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כא) בִּשְׁנַ֨ת חֲמִשִּׁ֤ים וּשְׁתַּ֙יִם֙ שָׁנָ֔ה לַעֲזַרְיָ֖ה מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ פֶּ֣קַח בֶּן־ רְמַלְיָ֧הוּ עַל־ יִשְׂרָאֵ֛ל בְּשֹׁמְר֖וֹן עֶשְׂרִ֥ים שָׁנָֽה׃ (כב) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לֹ֣א סָ֗ר מִן־ חַטֹּאות֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (כג) בִּימֵ֞י פֶּ֣קַח מֶֽלֶךְ־ יִשְׂרָאֵ֗ל בָּא֮ תִּגְלַ֣ת פִּלְאֶסֶר֮ מֶ֣לֶךְ אַשּׁוּר֒ וַיִּקַּ֣ח אֶת־ עִיּ֡וֹן וְאֶת־ אָבֵ֣ל בֵּֽית־ מַעֲכָ֡ה וְאֶת־ יָ֠נוֹחַ וְאֶת־ קֶ֨דֶשׁ וְאֶת־ חָצ֤וֹר וְאֶת־ הַגִּלְעָד֙ וְאֶת־ הַגָּלִ֔ילָה כֹּ֖ל אֶ֣רֶץ נַפְתָּלִ֑י וַיַּגְלֵ֖ם אַשּֽׁוּרָה׃ (כד) וַיִּקְשָׁר־ קֶ֜שֶׁר הוֹשֵׁ֣עַ בֶּן־ אֵלָ֗ה עַל־ פֶּ֙קַח֙ בֶּן־ רְמַלְיָ֔הוּ וַיַּכֵּ֙הוּ֙ וַיְמִיתֵ֔הוּ וַיִּמְלֹ֖ךְ תַּחְתָּ֑יו בִּשְׁנַ֣ת עֶשְׂרִ֔ים לְיוֹתָ֖ם בֶּן־ עֻזִיָּֽה׃ (כה) וְיֶ֥תֶר דִּבְרֵי־ פֶ֖קַח וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הִנָּ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כו) בִּשְׁנַ֣ת שְׁתַּ֔יִם לְפֶ֥קַח בֶּן־ רְמַלְיָ֖הוּ מֶ֣לֶךְ יִשְׂרָאֵ֑ל מָלַ֛ךְ יוֹתָ֥ם בֶּן־ עֻזִיָּ֖הוּ מֶ֥לֶךְ יְהוּדָֽה׃ (כז) בֶּן־ עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָה֙ הָיָ֣ה בְמָלְכ֔וֹ וְשֵׁשׁ־ עֶשְׂרֵ֣ה שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ יְרוּשָׁ֖א בַּת־ צָדֽוֹק׃ (כח) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֧ל אֲשֶׁר־ עָשָׂ֛ה עֻזִיָּ֥הוּ אָבִ֖יו עָשָֽׂה׃ (כט) רַ֤ק הַבָּמוֹת֙ לֹ֣א סָ֔רוּ ע֗וֹד הָעָ֛ם מְזַבְּחִ֥ים וּֽמְקַטְּרִ֖ים בַּבָּמ֑וֹת ה֗וּא בָּנָ֛ה אֶת־ שַׁ֥עַר בֵּית־ יְהוָ֖ה הָעֶלְיֽוֹן׃ (ל) וְיֶ֛תֶר דִּבְרֵ֥י יוֹתָ֖ם אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (לא) בַּיָּמִ֣ים הָהֵ֔ם הֵחֵ֣ל יְהוָ֗ה לְהַשְׁלִ֙יחַ֙ בִּֽיהוּדָ֔ה רְצִ֖ין מֶ֣לֶךְ אֲרָ֑ם וְאֵ֖ת פֶּ֥קַח בֶּן־ רְמַלְיָֽהוּ׃ (לב) וַיִּשְׁכַּ֤ב יוֹתָם֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵר֙ עִם־ אֲבֹתָ֔יו בְּעִ֖יר דָּוִ֣ד אָבִ֑יו וַיִּמְלֹ֛ךְ אָחָ֥ז בְּנ֖וֹ תַּחְתָּֽיו׃ תחילת פרק טז (לג) בִּשְׁנַת֙ שְׁבַֽע־ עֶשְׂרֵ֣ה שָׁנָ֔ה לְפֶ֖קַח בֶּן־ רְמַלְיָ֑הוּ מָלַ֛ךְ אָחָ֥ז בֶּן־ יוֹתָ֖ם מֶ֥לֶךְ יְהוּדָֽה׃ (לד) בֶּן־ עֶשְׂרִ֤ים שָׁנָה֙ אָחָ֣ז בְּמָלְכ֔וֹ וְשֵׁשׁ־ עֶשְׂרֵ֣ה שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְלֹא־ עָשָׂ֣ה הַיָּשָׁ֗ר בְּעֵינֵ֛י יְהוָ֥ה אֱלֹהָ֖יו כְּדָוִ֥ד אָבִֽיו׃ (לה) וַיֵּ֕לֶךְ בְּדֶ֖רֶךְ מַלְכֵ֣י יִשְׂרָאֵ֑ל וְגַ֤ם אֶת־ בְּנוֹ֙ הֶעֱבִ֣יר בָּאֵ֔שׁ כְּתֹֽעֲבוֹת֙ הַגּוֹיִ֔ם אֲשֶׁ֨ר הוֹרִ֤ישׁ יְהוָה֙ אֹתָ֔ם מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (לו) וַיְזַבֵּ֧חַ וַיְקַטֵּ֛ר בַּבָּמ֖וֹת וְעַל־ הַגְּבָע֑וֹת וְתַ֖חַת כָּל־ עֵ֥ץ רַעֲנָֽן׃ (לז) אָ֣ז יַעֲלֶ֣ה רְצִ֣ין מֶֽלֶךְ־ אֲ֠רָם וּפֶ֨קַח בֶּן־ רְמַלְיָ֧הוּ מֶֽלֶךְ־ יִשְׂרָאֵ֛ל יְרוּשָׁלִַ֖ם וַֽאֲדוֹמִים֙ בָּ֣אוּ אֵילַ֔ת וַיֵּ֣שְׁבוּ שָׁ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (לט) וַיִּשְׁלַ֨ח אָחָ֜ז מַלְאָכִ֗ים אֶל־ תִּ֠גְלַת פְּלֶ֤סֶר מֶֽלֶךְ־ אַשּׁוּר֙ לֵאמֹ֔ר עַבְדְּךָ֥ וּבִנְךָ֖ אָ֑נִי עֲלֵ֨ה וְהוֹשִׁעֵ֜נִי מִכַּ֣ף מֶֽלֶךְ־ אֲרָ֗ם וּמִכַּף֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל הַקּוֹמִ֖ים עָלָֽי׃ (מ) וַיִּקַּ֨ח אָחָ֜ז אֶת־ הַכֶּ֣סֶף וְאֶת־ הַזָּהָ֗ב הַנִּמְצָא֙ בֵּ֣ית יְהוָ֔ה וּבְאֹֽצְר֖וֹת בֵּ֣ית הַמֶּ֑לֶךְ וַיִּשְׁלַ֥ח לְמֶֽלֶךְ־ אַשּׁ֖וּר שֹֽׁחַד׃ (מא) וַיִּשְׁמַ֤ע אֵלָיו֙ מֶ֣לֶךְ אַשּׁ֔וּר וַיַּעַל֩ מֶ֨לֶךְ אַשּׁ֤וּר אֶל־ דַּמֶּ֙שֶׂק֙ וַֽיִּתְפְּשֶׂ֔הָ וַיַּגְלֶ֖הָ קִ֑ירָה וְאֶת־ רְצִ֖ין הֵמִֽית׃ (מב) וַיֵּ֣לֶךְ הַמֶּ֣לֶךְ אָחָ֡ז לִ֠קְרַאת תִּגְלַ֨ת פִּלְאֶ֤סֶר מֶֽלֶךְ־ אַשּׁוּר֙ דּוּמֶּ֔שֶׂק וַיַּ֥רְא אֶת־ הַמִּזְבֵּ֖חַ אֲשֶׁ֣ר בְּדַמָּ֑שֶׂק וַיִּשְׁלַח֩ הַמֶּ֨לֶךְ אָחָ֜ז אֶל־ אוּרִיָּ֣ה הַכֹּהֵ֗ן אֶת־ דְּמ֧וּת הַמִּזְבֵּ֛חַ וְאֶת־ תַּבְנִית֖וֹ לְכָֽל־ מַעֲשֵֽׂהוּ׃ (מג) וַיִּ֛בֶן אוּרִיָּ֥ה הַכֹּהֵ֖ן אֶת־ הַמִּזְבֵּ֑חַ כְּכֹ֣ל אֲשֶׁר־ שָׁלַח֩ הַמֶּ֨לֶךְ אָחָ֜ז מִדַּמֶּ֗שֶׂק הַמִּזְבֵּ֖חַ וַיַּ֥עַל עָלָֽיו׃ (מה) וַיַּקְטֵ֤ר אֶת־ עֹֽלָתוֹ֙ וְאֶת־ מִנְחָת֔וֹ וַיַּסֵּ֖ךְ אֶת־ נִסְכּ֑וֹ וַיִּזְרֹ֛ק אֶת־ דַּֽם־ הַשְּׁלָמִ֥ים אֲשֶׁר־ ל֖וֹ עַל־ הַמִּזְבֵּֽחַ׃ (מו) וְאֵ֨ת הַמִּזְבַּ֣ח הַנְּחֹשֶׁת֮ אֲשֶׁ֣ר לִפְנֵ֣י יְהוָה֒ וַיַּקְרֵ֗ב מֵאֵת֙ פְּנֵ֣י הַבַּ֔יִת מִבֵּין֙ הַמִּזְבֵּ֔חַ וּמִבֵּ֖ין בֵּ֣ית יְהוָ֑ה וַיִּתֵּ֥ן אֹת֛וֹ עַל־ יֶ֥רֶךְ הַמִּזְבֵּ֖חַ צָפֽוֹנָה׃ (מז) (ויצוהו) וַיְצַוֶּ֣ה הַמֶּֽלֶךְ־ אָ֠חָז אֶת־ אוּרִיָּ֨ה הַכֹּהֵ֜ן לֵאמֹ֗ר עַ֣ל הַמִּזְבֵּ֣חַ הַגָּד֡וֹל הַקְטֵ֣ר אֶת־ עֹֽלַת־ הַבֹּקֶר֩ וְאֶת־ מִנְחַ֨ת הָעֶ֜רֶב וְֽאֶת־ עֹלַ֧ת הַמֶּ֣לֶךְ וְאֶת־ מִנְחָת֗וֹ וְ֠אֵת עֹלַ֞ת כָּל־ עַ֤ם הָאָ֙רֶץ֙ וּמִנְחָתָ֣ם וְנִסְכֵּיהֶ֔ם וְכָל־ דַּ֥ם עֹלָ֛ה וְכָל־ דַּם־ זֶ֖בַח עָלָ֣יו תִּזְרֹ֑ק וּמִזְבַּ֧ח הַנְּחֹ֛שֶׁת יִֽהְיֶה־ לִּ֖י לְבַקֵּֽר׃ (מח) וַיַּ֖עַשׂ אוּרִיָּ֣ה הַכֹּהֵ֑ן כְּכֹ֥ל אֲשֶׁר־ צִוָּ֖ה הַמֶּ֥לֶךְ אָחָֽז׃ (מט) וַיְקַצֵּץ֩ הַמֶּ֨לֶךְ אָחָ֜ז אֶת־ הַמִּסְגְּר֣וֹת הַמְּכֹנ֗וֹת וַיָּ֤סַר מֵֽעֲלֵיהֶם֙ (ואת)־ אֶת־ הַכִּיֹּ֔ר וְאֶת־ הַיָּ֣ם הוֹרִ֔ד מֵעַ֛ל הַבָּקָ֥ר הַנְּחֹ֖שֶׁת אֲשֶׁ֣ר תַּחְתֶּ֑יהָ וַיִּתֵּ֣ן אֹת֔וֹ עַ֖ל מַרְצֶ֥פֶת אֲבָנִֽים׃ (נ) וְאֶת־ (מיסך) מוּסַ֨ךְ הַשַּׁבָּ֜ת אֲשֶׁר־ בָּנ֣וּ בַבַּ֗יִת וְאֶת־ מְב֤וֹא הַמֶּ֙לֶךְ֙ (א) וַיִּשְׁכַּ֤ב אָחָז֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֥ר עִם־ אֲבֹתָ֖יו בְּעִ֣יר דָּוִ֑ד וַיִּמְלֹ֛ךְ חִזְקִיָּ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ סוף פרק טז (ב) בִּשְׁנַת֙ שְׁתֵּ֣ים עֶשְׂרֵ֔ה לְאָחָ֖ז מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ הוֹשֵׁ֨עַ בֶּן־ אֵלָ֧ה בְשֹׁמְר֛וֹן עַל־ יִשְׂרָאֵ֖ל תֵּ֥שַׁע שָׁנִֽים׃ (ג) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה רַ֗ק לֹ֚א כְּמַלְכֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֥ר הָי֖וּ לְפָנָֽיו׃ (ד) ס֣וֹא מֶֽלֶךְ־ מִצְרַ֔יִם וְלֹא־ הֶעֱלָ֥ה מִנְחָ֛ה לְמֶ֥לֶךְ אַשּׁ֖וּר כְּשָׁנָ֣ה בְשָׁנָ֑ה וַֽיַּעַצְרֵ֙הוּ֙ מֶ֣לֶךְ אַשּׁ֔וּר וַיַּאַסְרֵ֖הוּ בֵּ֥ית כֶּֽלֶא׃ (ו) וַיַּ֥עַל מֶֽלֶךְ־ אַשּׁ֖וּר בְּכָל־ הָאָ֑רֶץ וַיַּ֙עַל֙ שֹׁמְר֔וֹן וַיָּ֥צַר עָלֶ֖יהָ שָׁלֹ֥שׁ שָׁנִֽים׃ (ז) בִּשְׁנַ֨ת הַתְּשִׁיעִ֜ית לְהוֹשֵׁ֗עַ לָכַ֤ד מֶֽלֶךְ־ אַשּׁוּר֙ אֶת־ שֹׁ֣מְר֔וֹן וַיֶּ֥גֶל אֶת־ יִשְׂרָאֵ֖ל אַשּׁ֑וּרָה וַיֹּ֨שֶׁב אֹתָ֜ם בַּחְלַ֧ח וּבְחָב֛וֹר נְהַ֥ר גּוֹזָ֖ן וְעָרֵ֥י מָדָֽי׃ (ח) וַיְהִ֗י כִּֽי־ חָטְא֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ לַיהוָ֣ה אֱלֹהֵיהֶ֔ם הַמַּעֲלֶ֤ה אֹתָם֙ מֵאֶ֣רֶץ מִצְרַ֔יִם מִתַּ֕חַת יַ֖ד וַֽיַּעֲשׂוּ֙ דְּבָרִ֣ים רָעִ֔ים לְהַכְעִ֖יס אֶת־ יְהוָֽה׃ (יג) וַיַּֽעַבְד֖וּ הַגִּלֻּלִ֑ים אֲשֶׁ֨ר אָמַ֤ר יְהוָה֙ לָהֶ֔ם לֹ֥א תַעֲשׂ֖וּ אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (יד) וַיָּ֣עַד יְהוָ֡ה בְּיִשְׂרָאֵ֣ל וַֽאֲשֶׁר֙ שָׁלַ֣חְתִּי אֲלֵיכֶ֔ם בְּיַ֖ד עֲבָדַ֥י הַנְּבִיאִֽים׃ (טו) וְלֹ֖א שָׁמֵ֑עוּ וַיַּקְשׁ֤וּ אֶת־ עָרְפָּם֙ כְּעֹ֣רֶף אֲבוֹתָ֔ם אֲשֶׁר֙ לֹ֣א הֶאֱמִ֔ינוּ בַּֽיהוָ֖ה אֱלֹהֵיהֶֽם׃ (טז) וַיִּמְאֲס֣וּ אֶת־ חֻקָּ֗יו וְאֶת־ בְּרִיתוֹ֙ אֲשֶׁ֣ר כָּרַ֣ת אֶת־ אֲבוֹתָ֔ם וְאֵת֙ עֵֽדְוֺתָ֔יו אֲשֶׁ֥ר הֵעִ֖יד בָּ֑ם וַיֵּ֨לְכ֜וּ אַחֲרֵ֤י הַהֶ֙בֶל֙ וַיֶּהְבָּ֔לוּ וְאַחֲרֵ֤י הַגּוֹיִם֙ אֲשֶׁ֣ר סְבִֽיבֹתָ֔ם אֲשֶׁ֨ר צִוָּ֤ה יְהוָה֙ אֹתָ֔ם לְבִלְתִּ֖י עֲשׂ֥וֹת כָּהֶֽם׃ (יז) וַיַּעַזְב֗וּ אֶת־ כָּל־ מִצְוֺת֙ יְהוָ֣ה אֱלֹהֵיהֶ֔ם וַיַּעֲשׂ֥וּ לָהֶ֛ם מַסֵּכָ֖ה (שנים) שְׁנֵ֣י עֲגָלִ֑ים וַיַּעֲשׂ֣וּ אֲשֵׁירָ֗ה וַיִּֽשְׁתַּחֲווּ֙ לְכָל־ צְבָ֣א הַשָּׁמַ֔יִם וַיַּעַבְד֖וּ אֶת־ הַבָּֽעַל׃ (יח) וַֽ֠יַּעֲבִירוּ אֶת־ בְּנֵיהֶ֤ם וְאֶת־ בְּנֽוֹתֵיהֶם֙ בָּאֵ֔שׁ וַיִּקְסְמ֥וּ קְסָמִ֖ים וַיְנַחֵ֑שׁוּ וַיִּֽתְמַכְּר֗וּ לַעֲשׂ֥וֹת הָרַ֛ע בְּעֵינֵ֥י יְהוָ֖ה לְהַכְעִיסֽוֹ׃ (יט) וַיִּתְאַנַּ֨ף יְהוָ֤ה מְאֹד֙ בְּיִשְׂרָאֵ֔ל וַיְסִרֵ֖ם מֵעַ֣ל פָּנָ֑יו לֹ֣א נִשְׁאַ֔ר רַ֛ק שֵׁ֥בֶט יְהוּדָ֖ה לְבַדּֽוֹ׃ (כ) גַּם־ יְהוּדָ֕ה לֹ֣א שָׁמַ֔ר אֶת־ מִצְוֺ֖ת יְהוָ֣ה אֱלֹהֵיהֶ֑ם וַיֵּ֣לְכ֔וּ בְּחֻקּ֥וֹת יִשְׂרָאֵ֖ל אֲשֶׁ֥ר עָשֽׂוּ׃ (כא) וַיִּמְאַ֨ס יְהוָ֜ה בְּכָל־ זֶ֤רַע יִשְׂרָאֵל֙ וַיְעַנֵּ֔ם וַֽיִּתְּנֵ֖ם בְּיַד־ שֹׁסִ֑ים עַ֛ד אֲשֶׁ֥ר הִשְׁלִיכָ֖ם מִפָּנָֽיו׃ (כב) כִּֽי־ קָרַ֣ע יִשְׂרָאֵ֗ל מֵעַל֙ בֵּ֣ית דָּוִ֔ד וַיַּמְלִ֖יכוּ אֶת־ יָרָבְעָ֣ם בֶּן־ נְבָ֑ט (וידא) וַיַּדַּ֨ח יָרָבְעָ֤ם אֶת־ יִשְׂרָאֵל֙ מֵאַחֲרֵ֣י יְהוָ֔ה וְהֶחֱטֵיאָ֖ם חֲטָאָ֥ה גְדוֹלָֽה׃ (כג) וַיֵּֽלְכוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל בְּכָל־ חַטֹּ֥אות יָרָבְעָ֖ם אֲשֶׁ֣ר עָשָׂ֑ה לֹא־ סָ֖רוּ מִמֶּֽנָּה׃ (כד) עַ֠ד אֲשֶׁר־ הֵסִ֨יר יְהוָ֤ה אֶת־ יִשְׂרָאֵל֙ מֵעַ֣ל פָּנָ֔יו כַּאֲשֶׁ֣ר דִּבֶּ֔ר בְּיַ֖ד כָּל־ עֲבָדָ֣יו הַנְּבִיאִ֑ים וַיִּ֨גֶל יִשְׂרָאֵ֜ל מֵעַ֤ל אַדְמָתוֹ֙ אַשּׁ֔וּרָה עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כה) וַיָּבֵ֣א מֶֽלֶךְ־ אַשּׁ֡וּר מִבָּבֶ֡ל וּ֠מִכּוּ֠תָה וּמֵעַוָּ֤א וּמֵֽחֲמָת֙ וּסְפַרְוַ֔יִם וַיֹּ֙שֶׁב֙ בְּעָרֵ֣י שֹֽׁמְר֔וֹן תַּ֖חַת בְּנֵ֣י יִשְׂרָאֵ֑ל וַיִּֽרְשׁוּ֙ אֶת־ שֹׁ֣מְר֔וֹן וַיֵּֽשְׁב֖וּ בְּעָרֶֽיהָ׃ (כו) וַיְהִ֗י בִּתְחִלַּת֙ שִׁבְתָּ֣ם שָׁ֔ם לֹ֥א יָרְא֖וּ אֶת־ יְהוָ֑ה וַיְשַׁלַּ֨ח יְהוָ֤ה בָּהֶם֙ אֶת־ הָ֣אֲרָי֔וֹת וַיִּֽהְי֥וּ הֹרְגִ֖ים בָּהֶֽם׃ (כז) וַיֹּאמְר֗וּ לְמֶ֣לֶךְ אַשּׁוּר֮ לֵאמֹר֒ הַגּוֹיִ֗ם אֲשֶׁ֤ר הִגְלִ֙יתָ֙ וַתּ֙וֹשֶׁב֙ בְּעָרֵ֣י שֹׁמְר֔וֹן לֹ֣א יָֽדְע֔וּ אֶת־ מִשְׁפַּ֖ט אֱלֹהֵ֣י הָאָ֑רֶץ וַיְשַׁלַּח־ בָּ֣ם אֶת־ הָאֲרָי֗וֹת וְהִנָּם֙ מְמִיתִ֣ים אוֹתָ֔ם כַּאֲשֶׁר֙ אֵינָ֣ם יֹדְעִ֔ים אֶת־ מִשְׁפַּ֖ט אֱלֹהֵ֥י הָאָֽרֶץ׃ (כח) וַיְצַ֨ו מֶֽלֶךְ־ אַשּׁ֜וּר לֵאמֹ֗ר הֹלִ֤יכוּ שָׁ֙מָּה֙ אֶחָ֤ד מֵהַכֹּֽהֲנִים֙ אֲשֶׁ֣ר הִגְלִיתֶ֣ם מִשָּׁ֔ם וְיֵלְכ֖וּ וְיֵ֣שְׁבוּ שָׁ֑ם וְיֹרֵ֕ם אֶת־ מִשְׁפַּ֖ט אֱלֹהֵ֥י הָאָֽרֶץ׃ (כט) וַיָּבֹ֞א אֶחָ֣ד מֵהַכֹּהֲנִ֗ים אֲשֶׁ֤ר הִגְלוּ֙ מִשֹּׁ֣מְר֔וֹן וַיֵּ֖שֶׁב בְּבֵֽית־ אֵ֑ל וַֽיְהִי֙ מוֹרֶ֣ה אֹתָ֔ם אֵ֖יךְ יִֽירְא֥וּ אֶת־ יְהוָֽה׃ (ל) וַיִּהְי֣וּ עֹשִׂ֔ים גּ֥וֹי גּ֖וֹי אֱלֹהָ֑יו וַיַּנִּ֣יחוּ ׀ בְּבֵ֣ית הַבָּמ֗וֹת אֲשֶׁ֤ר עָשׂוּ֙ הַשֹּׁ֣מְרֹנִ֔ים גּ֥וֹי גּוֹי֙ בְּעָ֣רֵיהֶ֔ם אֲשֶׁ֛ר הֵ֥ם יֹשְׁבִ֖ים שָֽׁם׃ (לא) וְאַנְשֵׁ֣י בָבֶ֗ל עָשׂוּ֙ אֶת־ סֻכּ֣וֹת בְּנ֔וֹת וְאַנְשֵׁי־ כ֔וּת עָשׂ֖וּ אֶת־ נֵֽרְגַ֑ל וְאַנְשֵׁ֥י חֲמָ֖ת עָשׂ֥וּ אֶת־ אֲשִׁימָֽא׃ (לב) וְהָעַוִּ֛ים עָשׂ֥וּ נִבְחַ֖ז וְאֶת־ תַּרְתָּ֑ק וְהַסְפַרְוִ֗ים שֹׂרְפִ֤ים אֶת־ בְּנֵיהֶם֙ בָּאֵ֔שׁ לְאַדְרַמֶּ֥לֶךְ וַֽעֲנַמֶּ֖לֶךְ (אלה) אֱלֹהֵ֥י (ספרים) סְפַרְוָֽיִם׃ (לג) וַיִּהְי֥וּ יְרֵאִ֖ים אֶת־ יְהוָ֑ה וַיַּעֲשׂ֨וּ לָהֶ֤ם מִקְצוֹתָם֙ כֹּהֲנֵ֣י בָמ֔וֹת וַיִּהְי֛וּ עֹשִׂ֥ים לָהֶ֖ם בְּבֵ֥ית הַבָּמֽוֹת׃ (לד) אֶת־ יְהוָ֖ה הָי֣וּ יְרֵאִ֑ים וְאֶת־ אֱלֹֽהֵיהֶם֙ הָי֣וּ עֹֽבְדִ֔ים כְּמִשְׁפַּט֙ הַגּוֹיִ֔ם אֲשֶׁר־ הִגְל֥וּ אֹתָ֖ם מִשָּֽׁם׃ (לה) עַ֣ד הַיּ֤וֹם הַזֶּה֙ הֵ֣ם עֹשִׂ֔ים כַּמִּשְׁפָּטִ֖ים הָרִֽאשֹׁנִ֑ים אֵינָ֤ם יְרֵאִים֙ אֶת־ יְהוָ֔ה וְאֵינָ֣ם עֹשִׂ֗ים כְּחֻקֹּתָם֙ וּכְמִשְׁפָּטָ֔ם וְכַתּוֹרָ֣ה וְכַמִּצְוָ֗ה אֲשֶׁ֨ר צִוָּ֤ה יְהוָה֙ אֶת־ בְּנֵ֣י יַעֲקֹ֔ב אֲשֶׁר־ שָׂ֥ם שְׁמ֖וֹ יִשְׂרָאֵֽל׃ (לו) וַיִּכְרֹ֨ת יְהוָ֤ה אִתָּם֙ בְּרִ֔ית וַיְצַוֵּ֣ם לֵאמֹ֔ר לֹ֥א תִֽירְא֖וּ אֱלֹהִ֣ים אֲחֵרִ֑ים וְלֹא־ תִשְׁתַּחֲו֣וּ לָהֶ֔ם וְלֹ֣א תַעַבְד֔וּם וְלֹ֥א תִזְבְּח֖וּ לָהֶֽם׃ (לז) כִּ֣י אִֽם־ אֶת־ יְהוָ֗ה אֲשֶׁר֩ הֶעֱלָ֨ה אֶתְכֶ֜ם מֵאֶ֧רֶץ מִצְרַ֛יִם בְּכֹ֧חַ גָּד֛וֹל וּבִזְר֥וֹעַ נְטוּיָ֖ה אֹת֣וֹ תִירָ֑אוּ וְל֥וֹ תִֽשְׁתַּחֲו֖וּ וְל֥וֹ תִזְבָּֽחוּ׃ (לח) וְאֶת־ הַחֻקִּ֨ים וְאֶת־ הַמִּשְׁפָּטִ֜ים וְהַתּוֹרָ֤ה וְהַמִּצְוָה֙ אֲשֶׁ֣ר כָּתַ֣ב לָכֶ֔ם תִּשְׁמְר֥וּן לַעֲשׂ֖וֹת כָּל־ הַיָּמִ֑ים וְלֹ֥א תִֽירְא֖וּ אֱלֹהִ֥ים אֲחֵרִֽים׃ (לט) וְהַבְּרִ֛ית אֲשֶׁר־ כָּרַ֥תִּי אִתְּכֶ֖ם לֹ֣א תִשְׁכָּ֑חוּ וְלֹ֥א תִֽירְא֖וּ אֱלֹהִ֥ים אֲחֵרִֽים׃ (מ) כִּ֛י אִֽם־ אֶת־ יְהוָ֥ה אֱלֹהֵיכֶ֖ם תִּירָ֑אוּ וְהוּא֙ יַצִּ֣יל אֶתְכֶ֔ם מִיַּ֖ד כָּל־ אֹיְבֵיכֶֽם׃ (מא) וְלֹ֖א שָׁמֵ֑עוּ כִּ֛י אִֽם־ כְּמִשְׁפָּטָ֥ם הָֽרִאשׁ֖וֹן הֵ֥ם עֹשִֽׂים׃ (מב) וַיִּהְי֣וּ ׀ הַגּוֹיִ֣ם הָאֵ֗לֶּה יְרֵאִים֙ אֶת־ יְהוָ֔ה וְאֶת־ פְּסִֽילֵיהֶ֖ם הָי֣וּ עֹֽבְדִ֑ים גַּם־ בְּנֵיהֶ֣ם ׀ וּבְנֵ֣י בְנֵיהֶ֗ם כַּאֲשֶׁ֨ר עָשׂ֤וּ אֲבֹתָם֙ הֵ֣ם עֹשִׂ֔ים עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (מג) וַֽיְהִי֙ בִּשְׁנַ֣ת שָׁלֹ֔שׁ לְהוֹשֵׁ֥עַ בֶּן־ אֵלָ֖ה מֶ֣לֶךְ יִשְׂרָאֵ֑ל מָלַ֛ךְ חִזְקִיָּ֥ה בֶן־ אָחָ֖ז מֶ֥לֶךְ יְהוּדָֽה׃ (מד) בֶּן־ עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָה֙ הָיָ֣ה בְמָלְכ֔וֹ וְעֶשְׂרִ֤ים וָתֵ֙שַׁע֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ אֲבִ֖י בַּת־ זְכַרְיָֽה׃ (מה) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה דָּוִ֥ד אָבִֽיו׃ (מו) ה֣וּא ׀ הֵסִ֣יר אֶת־ הַבָּמ֗וֹת וְשִׁבַּר֙ אֶת־ הַמַּצֵּבֹ֔ת וְכָרַ֖ת אֶת־ הָֽאֲשֵׁרָ֑ה וְכִתַּת֩ נְחַ֨שׁ הַנְּחֹ֜שֶׁת אֲשֶׁר־ עָשָׂ֣ה מֹשֶׁ֗ה כִּ֣י עַד־ הַיָּמִ֤ים הָהֵ֙מָּה֙ הָי֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ מְקַטְּרִ֣ים ל֔וֹ וַיִּקְרָא־ ל֖וֹ נְחֻשְׁתָּֽן׃ (מז) בַּיהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בָּטָ֑ח וְאַחֲרָ֞יו לֹא־ הָיָ֣ה כָמֹ֗הוּ בְּכֹל֙ מַלְכֵ֣י יְהוּדָ֔ה וַאֲשֶׁ֥ר הָי֖וּ לְפָנָֽיו׃
