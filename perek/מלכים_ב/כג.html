מלכים ב פרק כג לפי החלוקה לסדרים

פרק כג בספר מלכים ב (37 פסוקים) נמצא במלכים סדר לג (פסוקים כ–מג) ובמלכים סדר לד (פסוקים א–יג).

מלכים סדר לג

(א) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה וַיֵּ֗לֶךְ בְּכָל־ דֶּ֙רֶךְ֙ דָּוִ֣ד אָבִ֔יו וְלֹא־ סָ֖ר יָמִ֥ין וּשְׂמֹֽאול׃ (ב) וַיְהִ֗י בִּשְׁמֹנֶ֤ה עֶשְׂרֵה֙ שָׁנָ֔ה לַמֶּ֖לֶךְ יֹאשִׁיָּ֑הוּ שָׁלַ֣ח הַ֠מֶּלֶךְ אֶת־ שָׁפָ֨ן בֶּן־ אֲצַלְיָ֤הוּ בֶן־ מְשֻׁלָּם֙ הַסֹּפֵ֔ר בֵּ֥ית יְהוָ֖ה לֵאמֹֽר׃ (ג) עֲלֵ֗ה אֶל־ חִלְקִיָּ֙הוּ֙ הַכֹּהֵ֣ן הַגָּד֔וֹל וְיַתֵּ֣ם אֶת־ הַכֶּ֔סֶף הַמּוּבָ֖א בֵּ֣ית יְהוָ֑ה אֲשֶׁ֥ר אָסְפ֛וּ שֹׁמְרֵ֥י הַסַּ֖ף מֵאֵ֥ת הָעָֽם׃ (ד) (ויתנה) וְיִתְּנֻ֗הוּ עַל־ יַד֙ עֹשֵׂ֣י הַמְּלָאכָ֔ה הַמֻּפְקָדִ֖ים (בבית) בֵּ֣ית יְהוָ֑ה וְיִתְּנ֣וּ אֹת֗וֹ לְעֹשֵׂ֤י הַמְּלָאכָה֙ אֲשֶׁר֙ בְּבֵ֣ית יְהוָ֔ה לְחַזֵּ֖ק בֶּ֥דֶק הַבָּֽיִת׃ (ה) לֶחָ֣רָשִׁ֔ים וְלַבֹּנִ֖ים וְלַגֹּֽדְרִ֑ים וְלִקְנ֤וֹת עֵצִים֙ וְאַבְנֵ֣י מַחְצֵ֔ב לְחַזֵּ֖ק אֶת־ הַבָּֽיִת׃ (ו) אַ֚ךְ לֹא־ יֵחָשֵׁ֣ב אִתָּ֔ם הַכֶּ֖סֶף הַנִּתָּ֣ן עַל־ יָדָ֑ם כִּ֥י בֶאֱמוּנָ֖ה הֵ֥ם עֹשִֽׂים׃ (ז) וַ֠יֹּאמֶר חִלְקִיָּ֜הוּ הַכֹּהֵ֤ן הַגָּדוֹל֙ עַל־ שָׁפָ֣ן הַסֹּפֵ֔ר סֵ֧פֶר הַתּוֹרָ֛ה מָצָ֖אתִי בְּבֵ֣ית יְהוָ֑ה וַיִּתֵּ֨ן חִלְקִיָּ֧ה אֶת־ הַסֵּ֛פֶר אֶל־ שָׁפָ֖ן וַיִּקְרָאֵֽהוּ׃ (ח) וַיָּבֹ֞א שָׁפָ֤ן הַסֹּפֵר֙ אֶל־ הַמֶּ֔לֶךְ וַיָּ֥שֶׁב אֶת־ הַמֶּ֖לֶךְ דָּבָ֑ר וַיֹּ֗אמֶר הִתִּ֤יכוּ עֲבָדֶ֙יךָ֙ אֶת־ הַכֶּ֙סֶף֙ הַנִּמְצָ֣א בַבַּ֔יִת וַֽיִּתְּנֻ֗הוּ עַל־ יַד֙ עֹשֵׂ֣י הַמְּלָאכָ֔ה הַמֻּפְקָדִ֖ים בֵּ֥ית יְהוָֽה׃ (ט) וַיַּגֵּ֞ד שָׁפָ֤ן הַסֹּפֵר֙ לַמֶּ֣לֶךְ לֵאמֹ֔ר סֵ֚פֶר נָ֣תַן לִ֔י חִלְקִיָּ֖ה הַכֹּהֵ֑ן וַיִּקְרָאֵ֥הוּ שָׁפָ֖ן לִפְנֵ֥י הַמֶּֽלֶךְ׃ (י) וַֽיְהִי֙ כִּשְׁמֹ֣עַ הַמֶּ֔לֶךְ אֶת־ דִּבְרֵ֖י סֵ֣פֶר הַתּוֹרָ֑ה וַיִּקְרַ֖ע אֶת־ בְּגָדָֽיו׃ (יא) וַיְצַ֣ו הַמֶּ֡לֶךְ אֶת־ חִלְקִיָּ֣ה הַכֹּהֵ֡ן וְאֶת־ אֲחִיקָ֣ם בֶּן־ שָׁ֠פָן וְאֶת־ עַכְבּ֨וֹר בֶּן־ מִיכָיָ֜ה וְאֵ֣ת ׀ שָׁפָ֣ן הַסֹּפֵ֗ר וְאֵ֛ת עֲשָׂיָ֥ה עֶֽבֶד־ הַמֶּ֖לֶךְ לֵאמֹֽר׃ (יב) לְכוּ֩ דִרְשׁ֨וּ אֶת־ יְהוָ֜ה בַּעֲדִ֣י וּבְעַד־ הָעָ֗ם וּבְעַד֙ כָּל־ יְהוּדָ֔ה עַל־ דִּבְרֵ֛י הַסֵּ֥פֶר הַנִּמְצָ֖א הַזֶּ֑ה כִּֽי־ גְדוֹלָ֞ה חֲמַ֣ת יְהוָ֗ה אֲשֶׁר־ הִיא֙ נִצְּתָ֣ה בָ֔נוּ עַל֩ אֲשֶׁ֨ר לֹֽא־ שָׁמְע֜וּ אֲבֹתֵ֗ינוּ עַל־ דִּבְרֵי֙ הַסֵּ֣פֶר הַזֶּ֔ה לַעֲשׂ֖וֹת כְּכָל־ הַכָּת֥וּב עָלֵֽינוּ׃ (יג) וַיֵּ֣לֶךְ חִלְקִיָּ֣הוּ הַ֠כֹּהֵן וַאֲחִיקָ֨ם וְעַכְבּ֜וֹר וְשָׁפָ֣ן וַעֲשָׂיָ֗ה אֶל־ חֻלְדָּ֨ה הַנְּבִיאָ֜ה אֵ֣שֶׁת ׀ שַׁלֻּ֣ם בֶּן־ תִּקְוָ֗ה בֶּן־ חַרְחַס֙ שֹׁמֵ֣ר הַבְּגָדִ֔ים וְהִ֛יא יֹשֶׁ֥בֶת בִּירוּשָׁלִַ֖ם בַּמִּשְׁנֶ֑ה וַֽיְדַבְּר֖וּ אֵלֶֽיהָ׃ (יד) וַתֹּ֣אמֶר אֲלֵיהֶ֔ם כֹּֽה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל אִמְר֣וּ לָאִ֔ישׁ אֲשֶׁר־ שָׁלַ֥ח אֶתְכֶ֖ם אֵלָֽי׃ (טו) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֨י מֵבִ֥יא רָעָ֛ה אֶל־ הַמָּק֥וֹם הַזֶּ֖ה וְעַל־ יֹֽשְׁבָ֑יו אֵ֚ת כָּל־ דִּבְרֵ֣י הַסֵּ֔פֶר אֲשֶׁ֥ר קָרָ֖א מֶ֥לֶךְ יְהוּדָֽה׃ (טז) תַּ֣חַת ׀ אֲשֶׁ֣ר עֲזָב֗וּנִי וַֽיְקַטְּרוּ֙ לֵאלֹהִ֣ים אֲחֵרִ֔ים לְמַ֙עַן֙ הַכְעִיסֵ֔נִי בְּכֹ֖ל מַעֲשֵׂ֣ה יְדֵיהֶ֑ם וְנִצְּתָ֧ה חֲמָתִ֛י בַּמָּק֥וֹם הַזֶּ֖ה וְלֹ֥א תִכְבֶּֽה׃ (יז) וְאֶל־ מֶ֣לֶךְ יְהוּדָ֗ה הַשֹּׁלֵ֤חַ אֶתְכֶם֙ לִדְרֹ֣שׁ אֶת־ יְהוָ֔ה כֹּ֥ה תֹאמְר֖וּ אֵלָ֑יו כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הַדְּבָרִ֖ים אֲשֶׁ֥ר שָׁמָֽעְתָּ׃ (יח) יַ֠עַן רַךְ־ לְבָ֨בְךָ֜ וַתִּכָּנַ֣ע ׀ מִפְּנֵ֣י יְהוָ֗ה בְּֽשָׁמְעֲךָ֡ אֲשֶׁ֣ר דִּבַּרְתִּי֩ עַל־ הַמָּק֨וֹם הַזֶּ֜ה וְעַל־ יֹשְׁבָ֗יו לִהְי֤וֹת לְשַׁמָּה֙ וְלִקְלָלָ֔ה וַתִּקְרַע֙ אֶת־ בְּגָדֶ֔יךָ וַתִּבְכֶּ֖ה לְפָנָ֑י וְגַ֧ם אָנֹכִ֛י שָׁמַ֖עְתִּי נְאֻם־ יְהוָֽה׃ (יט) לָכֵן֩ הִנְנִ֨י אֹֽסִפְךָ֜ עַל־ אֲבֹתֶ֗יךָ וְנֶאֱסַפְתָּ֣ אֶל־ קִבְרֹתֶיךָ֮ בְּשָׁלוֹם֒ וְלֹא־ תִרְאֶ֣ינָה עֵינֶ֔יךָ בְּכֹל֙ הָֽרָעָ֔ה אֲשֶׁר־ אֲנִ֥י מֵבִ֖יא עַל־ הַמָּק֣וֹם הַזֶּ֑ה וַיָּשִׁ֥יבוּ אֶת־ הַמֶּ֖לֶךְ דָּבָֽר׃ תחילת פרק כג (כ) וַיִּשְׁלַ֖ח הַמֶּ֑לֶךְ וַיַּאַסְפ֣וּ אֵלָ֔יו כָּל־ זִקְנֵ֥י יְהוּדָ֖ה וִירוּשָׁלִָֽם׃ (כא) וַיַּ֣עַל הַמֶּ֣לֶךְ בֵּית־ יְהוָ֡ה וְכָל־ אִ֣ישׁ יְהוּדָה֩ וְכָל־ יֹשְׁבֵ֨י יְרוּשָׁלִַ֜ם אִתּ֗וֹ וְהַכֹּֽהֲנִים֙ וְהַנְּבִיאִ֔ים וְכָל־ הָעָ֖ם לְמִקָּטֹ֣ן וְעַד־ גָּד֑וֹל וַיִּקְרָ֣א בְאָזְנֵיהֶ֗ם אֶת־ כָּל־ דִּבְרֵי֙ סֵ֣פֶר הַבְּרִ֔ית הַנִּמְצָ֖א בְּבֵ֥ית יְהוָֽה׃ (כב) וַיַּעֲמֹ֣ד הַ֠מֶּלֶךְ עַֽל־ הָ֨עַמּ֜וּד וַיִּכְרֹ֥ת אֶֽת־ הַבְּרִ֣ית ׀ לִפְנֵ֣י יְהוָ֗ה לָלֶ֜כֶת אַחַ֤ר יְהוָה֙ וְלִשְׁמֹ֨ר מִצְוֺתָ֜יו וְאֶת־ עֵדְוֺתָ֤יו וְאֶת־ חֻקֹּתָיו֙ בְּכָל־ לֵ֣ב וּבְכָל־ נֶ֔פֶשׁ לְהָקִ֗ים אֶת־ דִּבְרֵי֙ הַבְּרִ֣ית הַזֹּ֔את הַכְּתֻבִ֖ים עַל־ הַסֵּ֣פֶר הַזֶּ֑ה וַיַּעֲמֹ֥ד כָּל־ הָעָ֖ם בַּבְּרִֽית׃ (כג) וַיְצַ֣ו הַמֶּ֡לֶךְ אֶת־ חִלְקִיָּהוּ֩ הַכֹּהֵ֨ן הַגָּד֜וֹל וְאֶת־ כֹּהֲנֵ֣י הַמִּשְׁנֶה֮ וְאֶת־ שֹׁמְרֵ֣י הַסַּף֒ לְהוֹצִיא֙ מֵהֵיכַ֣ל יְהוָ֔ה אֵ֣ת כָּל־ הַכֵּלִ֗ים הָֽעֲשׂוּיִם֙ לַבַּ֣עַל וְלָֽאֲשֵׁרָ֔ה וּלְכֹ֖ל צְבָ֣א הַשָּׁמָ֑יִם וַֽיִּשְׂרְפֵ֞ם מִח֤וּץ לִירוּשָׁלִַ֙ם֙ בְּשַׁדְמ֣וֹת קִדְר֔וֹן וְנָשָׂ֥א אֶת־ עֲפָרָ֖ם בֵּֽית־ אֵֽל׃ (כד) וְהִשְׁבִּ֣ית אֶת־ הַכְּמָרִ֗ים אֲשֶׁ֤ר נָֽתְנוּ֙ מַלְכֵ֣י יְהוּדָ֔ה וַיְקַטֵּ֤ר בַּבָּמוֹת֙ בְּעָרֵ֣י יְהוּדָ֔ה וּמְסִבֵּ֖י יְרוּשָׁלִָ֑ם וְאֶת־ הַֽמְקַטְּרִ֣ים לַבַּ֗עַל לַשֶּׁ֤מֶשׁ וְלַיָּרֵ֙חַ֙ וְלַמַּזָּל֔וֹת וּלְכֹ֖ל צְבָ֥א הַשָּׁמָֽיִם׃ (כה) וַיֹּצֵ֣א אֶת־ הָאֲשֵׁרָה֩ מִבֵּ֨ית יְהוָ֜ה מִח֤וּץ לִירוּשָׁלִַ֙ם֙ אֶל־ נַ֣חַל קִדְר֔וֹן וַיִּשְׂרֹ֥ף אֹתָ֛הּ בְּנַ֥חַל קִדְר֖וֹן וַיָּ֣דֶק לְעָפָ֑ר וַיַּשְׁלֵךְ֙ אֶת־ עֲפָרָ֔הּ עַל־ קֶ֖בֶר בְּנֵ֥י הָעָֽם׃ (כו) וַיִּתֹּץ֙ אֶת־ בָּתֵּ֣י הַקְּדֵשִׁ֔ים אֲשֶׁ֖ר בְּבֵ֣ית יְהוָ֑ה אֲשֶׁ֣ר הַנָּשִׁ֗ים אֹרְג֥וֹת שָׁ֛ם בָּתִּ֖ים לָאֲשֵׁרָֽה׃ (כז) וַיָּבֵ֤א אֶת־ כָּל־ הַכֹּֽהֲנִים֙ מֵעָרֵ֣י יְהוּדָ֔ה וַיְטַמֵּ֣א אֶת־ הַבָּמ֗וֹת אֲשֶׁ֤ר קִטְּרוּ־ שָׁ֙מָּה֙ הַכֹּ֣הֲנִ֔ים מִגֶּ֖בַע עַד־ בְּאֵ֣ר שָׁ֑בַע וְנָתַ֞ץ אֶת־ בָּמ֣וֹת הַשְּׁעָרִ֗ים אֲשֶׁר־ פֶּ֜תַח שַׁ֤עַר יְהוֹשֻׁ֙עַ֙ שַׂר־ הָעִ֔יר אֲשֶֽׁר־ עַל־ שְׂמֹ֥אול אִ֖ישׁ בְּשַׁ֥עַר הָעִֽיר׃ (כח) אַ֗ךְ לֹ֤א יַֽעֲלוּ֙ כֹּהֲנֵ֣י הַבָּמ֔וֹת אֶל־ מִזְבַּ֥ח יְהוָ֖ה בִּירוּשָׁלִָ֑ם כִּ֛י אִם־ אָכְל֥וּ מַצּ֖וֹת בְּת֥וֹךְ אֲחֵיהֶֽם׃ (כט) וְטִמֵּ֣א אֶת־ הַתֹּ֔פֶת אֲשֶׁ֖ר בְּגֵ֣י (בני)־ בֶן־ הִנֹּ֑ם לְבִלְתִּ֗י לְהַעֲבִ֨יר אִ֜ישׁ אֶת־ בְּנ֧וֹ וְאֶת־ בִּתּ֛וֹ בָּאֵ֖שׁ לַמֹּֽלֶךְ׃ (ל) וַיַּשְׁבֵּ֣ת אֶת־ הַסּוּסִ֗ים אֲשֶׁ֣ר נָתְנוּ֩ מַלְכֵ֨י יְהוּדָ֤ה לַשֶּׁ֙מֶשׁ֙ מִבֹּ֣א בֵית־ יְהוָ֔ה אֶל־ לִשְׁכַּת֙ נְתַן־ מֶ֣לֶךְ הַסָּרִ֔יס אֲשֶׁ֖ר בַּפַּרְוָרִ֑ים וְאֶת־ מַרְכְּב֥וֹת הַשֶּׁ֖מֶשׁ שָׂרַ֥ף בָּאֵֽשׁ׃ (לא) וְאֶֽת־ הַֽמִּזְבְּח֡וֹת אֲשֶׁ֣ר עַל־ הַגָּג֩ עֲלִיַּ֨ת אָחָ֜ז אֲשֶׁר־ עָשׂ֣וּ ׀ מַלְכֵ֣י יְהוּדָ֗ה וְאֶת־ הַֽמִּזְבְּחוֹת֙ אֲשֶׁר־ עָשָׂ֣ה מְנַשֶּׁ֔ה בִּשְׁתֵּ֛י חַצְר֥וֹת בֵּית־ יְהוָ֖ה נָתַ֣ץ הַמֶּ֑לֶךְ וַיָּ֣רָץ מִשָּׁ֔ם וְהִשְׁלִ֥יךְ אֶת־ עֲפָרָ֖ם אֶל־ נַ֥חַל קִדְרֽוֹן׃ (לב) וְֽאֶת־ הַבָּמ֞וֹת אֲשֶׁ֣ר ׀ עַל־ פְּנֵ֣י יְרוּשָׁלִַ֗ם אֲשֶׁר֮ מִימִ֣ין לְהַר־ הַמַּשְׁחִית֒ אֲשֶׁ֣ר בָּ֠נָה שְׁלֹמֹ֨ה מֶֽלֶךְ־ יִשְׂרָאֵ֜ל לְעַשְׁתֹּ֣רֶת ׀ שִׁקֻּ֣ץ צִידֹנִ֗ים וְלִכְמוֹשׁ֙ שִׁקֻּ֣ץ מוֹאָ֔ב וּלְמִלְכֹּ֖ם תּוֹעֲבַ֣ת בְּנֵֽי־ עַמּ֑וֹן טִמֵּ֖א הַמֶּֽלֶךְ׃ (לג) וְשִׁבַּר֙ אֶת־ הַמַּצֵּב֔וֹת וַיִּכְרֹ֖ת אֶת־ הָאֲשֵׁרִ֑ים וַיְמַלֵּ֥א אֶת־ מְקוֹמָ֖ם עַצְמ֥וֹת אָדָֽם׃ (לד) וְגַ֨ם אֶת־ הַמִּזְבֵּ֜חַ אֲשֶׁ֣ר בְּבֵֽית־ אֵ֗ל הַבָּמָה֙ אֲשֶׁ֨ר עָשָׂ֜ה יָרָבְעָ֤ם בֶּן־ נְבָט֙ אֲשֶׁ֣ר הֶחֱטִ֣יא אֶת־ יִשְׂרָאֵ֔ל גַּ֣ם אֶת־ הַמִּזְבֵּ֧חַ הַה֛וּא וְאֶת־ הַבָּמָ֖ה נָתָ֑ץ וַיִּשְׂרֹ֧ף אֶת־ הַבָּמָ֛ה הֵדַ֥ק לְעָפָ֖ר וְשָׂרַ֥ף אֲשֵׁרָֽה׃ (לה) וַיִּ֣פֶן יֹאשִׁיָּ֗הוּ וַיַּ֨רְא אֶת־ הַקְּבָרִ֤ים אֲשֶׁר־ שָׁם֙ בָּהָ֔ר וַיִּשְׁלַ֗ח וַיִּקַּ֤ח אֶת־ הָֽעֲצָמוֹת֙ מִן־ הַקְּבָרִ֔ים וַיִּשְׂרֹ֥ף עַל־ הַמִּזְבֵּ֖חַ וַֽיְטַמְּאֵ֑הוּ כִּדְבַ֣ר יְהוָ֗ה אֲשֶׁ֤ר קָרָא֙ אִ֣ישׁ הָאֱלֹהִ֔ים אֲשֶׁ֣ר קָרָ֔א אֶת־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (לו) וַיֹּ֕אמֶר מָ֚ה הַצִּיּ֣וּן הַלָּ֔ז אֲשֶׁ֖ר אֲנִ֣י רֹאֶ֑ה וַיֹּאמְר֨וּ אֵלָ֜יו אַנְשֵׁ֣י הָעִ֗יר הַקֶּ֤בֶר אִישׁ־ הָֽאֱלֹהִים֙ אֲשֶׁר־ בָּ֣א מִֽיהוּדָ֔ה וַיִּקְרָ֗א אֶת־ הַדְּבָרִ֤ים הָאֵ֙לֶּה֙ אֲשֶׁ֣ר עָשִׂ֔יתָ עַ֖ל הַמִּזְבַּ֥ח בֵּֽית־ אֵֽל׃ (לז) וַיֹּ֙אמֶר֙ הַנִּ֣יחוּ ל֔וֹ אִ֖ישׁ אַל־ יָנַ֣ע עַצְמֹתָ֑יו וַֽיְמַלְּטוּ֙ עַצְמֹתָ֔יו אֵ֚ת עַצְמ֣וֹת הַנָּבִ֔יא אֲשֶׁר־ בָּ֖א מִשֹּׁמְרֽוֹן׃ (לח) וְגַם֩ אֶת־ כָּל־ בָּתֵּ֨י הַבָּמ֜וֹת אֲשֶׁ֣ר ׀ בְּעָרֵ֣י שֹׁמְר֗וֹן אֲשֶׁ֨ר עָשׂ֜וּ מַלְכֵ֤י יִשְׂרָאֵל֙ לְהַכְעִ֔יס הֵסִ֖יר יֹֽאשִׁיָּ֑הוּ וַיַּ֣עַשׂ לָהֶ֔ם כְּכָל־ הַֽמַּעֲשִׂ֔ים אֲשֶׁ֥ר עָשָׂ֖ה בְּבֵֽית־ אֵֽל׃ (לט) וַ֠יִּזְבַּח אֶת־ כָּל־ כֹּהֲנֵ֨י הַבָּמ֤וֹת אֲשֶׁר־ שָׁם֙ עַל־ הַֽמִּזְבְּח֔וֹת וַיִּשְׂרֹ֛ף אֶת־ עַצְמ֥וֹת אָדָ֖ם עֲלֵיהֶ֑ם וַיָּ֖שָׁב יְרוּשָׁלִָֽם׃ (מ) וַיְצַ֤ו הַמֶּ֙לֶךְ֙ אֶת־ כָּל־ הָעָ֣ם לֵאמֹ֔ר עֲשׂ֣וּ פֶ֔סַח לַֽיהוָ֖ה אֱלֹֽהֵיכֶ֑ם כַּכָּת֕וּב עַ֛ל סֵ֥פֶר הַבְּרִ֖ית הַזֶּֽה׃ (מא) כִּ֣י לֹ֤א נַֽעֲשָׂה֙ כַּפֶּ֣סַח הַזֶּ֔ה מִימֵי֙ הַשֹּׁ֣פְטִ֔ים אֲשֶׁ֥ר שָׁפְט֖וּ אֶת־ יִשְׂרָאֵ֑ל וְכֹ֗ל יְמֵ֛י מַלְכֵ֥י יִשְׂרָאֵ֖ל וּמַלְכֵ֥י יְהוּדָֽה׃ (מב) כִּ֗י אִם־ בִּשְׁמֹנֶ֤ה עֶשְׂרֵה֙ שָׁנָ֔ה לַמֶּ֖לֶךְ יֹֽאשִׁיָּ֑הוּ נַעֲשָׂ֞ה הַפֶּ֧סַח הַזֶּ֛ה לַיהוָ֖ה בִּירוּשָׁלִָֽם׃ (מג) וְגַ֣ם אֶת־ הָאֹב֣וֹת וְאֶת־ הַ֠יִּדְּעֹנִים וְאֶת־ הַתְּרָפִ֨ים וְאֶת־ הַגִּלֻּלִ֜ים וְאֵ֣ת כָּל־ הַשִּׁקֻּצִ֗ים אֲשֶׁ֤ר נִרְאוּ֙ בְּאֶ֤רֶץ יְהוּדָה֙ וּבִיר֣וּשָׁלִַ֔ם בִּעֵ֖ר יֹֽאשִׁיָּ֑הוּ לְ֠מַעַן הָקִ֞ים אֶת־ דִּבְרֵ֤י הַתּוֹרָה֙ הַכְּתֻבִ֣ים עַל־ הַסֵּ֔פֶר אֲשֶׁ֥ר מָצָ֛א חִלְקִיָּ֥הוּ הַכֹּהֵ֖ן בֵּ֥ית יְהוָֽה׃

(א) וְכָמֹהוּ֩ לֹֽא־ הָיָ֨ה לְפָנָ֜יו מֶ֗לֶךְ אֲשֶׁר־ שָׁ֤ב אֶל־ יְהוָה֙ בְּכָל־ לְבָב֤וֹ וּבְכָל־ נַפְשׁוֹ֙ וּבְכָל־ מְאֹד֔וֹ כְּכֹ֖ל תּוֹרַ֣ת מֹשֶׁ֑ה וְאַחֲרָ֖יו לֹֽא־ קָ֥ם כָּמֹֽהוּ׃ (ב) אַ֣ךְ ׀ לֹֽא־ שָׁ֣ב יְהוָ֗ה מֵחֲר֤וֹן אַפּוֹ֙ הַגָּד֔וֹל אֲשֶׁר־ חָרָ֥ה אַפּ֖וֹ בִּֽיהוּדָ֑ה עַ֚ל כָּל־ הַכְּעָסִ֔ים אֲשֶׁ֥ר הִכְעִיס֖וֹ מְנַשֶּֽׁה׃ (ג) וַיֹּ֣אמֶר יְהוָ֗ה גַּ֤ם אֶת־ יְהוּדָה֙ אָסִיר֙ מֵעַ֣ל פָּנַ֔י כַּאֲשֶׁ֥ר הֲסִרֹ֖תִי אֶת־ יִשְׂרָאֵ֑ל וּ֠מָאַסְתִּי אֶת־ הָעִ֨יר הַזֹּ֤את אֲשֶׁר־ בָּחַ֙רְתִּי֙ אֶת־ יְר֣וּשָׁלִַ֔ם וְאֶת־ הַבַּ֔יִת אֲשֶׁ֣ר אָמַ֔רְתִּי יִהְיֶ֥ה שְׁמִ֖י שָֽׁם׃ (ד) וְיֶ֛תֶר דִּבְרֵ֥י יֹאשִׁיָּ֖הוּ וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (ה) בְּיָמָ֡יו עָלָה֩ פַרְעֹ֨ה נְכֹ֧ה מֶֽלֶךְ־ מִצְרַ֛יִם עַל־ מֶ֥לֶךְ אַשּׁ֖וּר עַל־ נְהַר־ פְּרָ֑ת וַיֵּ֨לֶךְ הַמֶּ֤לֶךְ יֹאשִׁיָּ֙הוּ֙ לִקְרָאת֔וֹ וַיְמִיתֵ֙הוּ֙ בִּמְגִדּ֔וֹ כִּרְאֹת֖וֹ אֹתֽוֹ׃ (ו) וַיַּרְכִּבֻ֨הוּ עֲבָדָ֥יו מֵת֙ מִמְּגִדּ֔וֹ וַיְבִאֻ֙הוּ֙ יְר֣וּשָׁלִַ֔ם וַֽיִּקְבְּרֻ֖הוּ בִּקְבֻֽרָת֑וֹ וַיִּקַּ֣ח עַם־ הָאָ֗רֶץ אֶת־ יְהֽוֹאָחָז֙ בֶּן־ יֹ֣אשִׁיָּ֔הוּ וַיִּמְשְׁח֥וּ אֹת֛וֹ וַיַּמְלִ֥יכוּ אֹת֖וֹ תַּ֥חַת אָבִֽיו׃ (ז) בֶּן־ עֶשְׂרִ֨ים וְשָׁלֹ֤שׁ שָׁנָה֙ יְהוֹאָחָ֣ז בְּמָלְכ֔וֹ וּשְׁלֹשָׁ֣ה חֳדָשִׁ֔ים מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ חֲמוּטַ֥ל בַּֽת־ יִרְמְיָ֖הוּ מִלִּבְנָֽה׃ (ח) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשׂ֖וּ אֲבֹתָֽיו׃ (ט) וַיַּאַסְרֵהוּ֩ פַרְעֹ֨ה נְכֹ֤ה בְרִבְלָה֙ בְּאֶ֣רֶץ חֲמָ֔ת (במלך) מִמְּלֹ֖ךְ בִּירוּשָׁלִָ֑ם וַיִּתֶּן־ עֹ֙נֶשׁ֙ עַל־ הָאָ֔רֶץ מֵאָ֥ה כִכַּר־ כֶּ֖סֶף וְכִכַּ֥ר זָהָֽב׃ (י) וַיַּמְלֵךְ֩ פַּרְעֹ֨ה נְכֹ֜ה אֶת־ אֶלְיָקִ֣ים בֶּן־ יֹאשִׁיָּ֗הוּ תַּ֚חַת יֹאשִׁיָּ֣הוּ אָבִ֔יו וַיַּסֵּ֥ב אֶת־ שְׁמ֖וֹ יְהוֹיָקִ֑ים וְאֶת־ יְהוֹאָחָ֣ז לָקָ֔ח וַיָּבֹ֥א מִצְרַ֖יִם וַיָּ֥מָת שָֽׁם׃ (יא) וְהַכֶּ֣סֶף וְהַזָּהָ֗ב נָתַ֤ן יְהוֹיָקִים֙ לְפַרְעֹ֔ה אַ֚ךְ הֶעֱרִ֣יךְ אֶת־ הָאָ֔רֶץ לָתֵ֥ת אֶת־ הַכֶּ֖סֶף עַל־ פִּ֣י פַרְעֹ֑ה אִ֣ישׁ כְּעֶרְכּ֗וֹ נָגַ֞שׂ אֶת־ הַכֶּ֤סֶף וְאֶת־ הַזָּהָב֙ אֶת־ עַ֣ם הָאָ֔רֶץ לָתֵ֖ת לְפַרְעֹ֥ה נְכֹֽה׃ (יב) בֶּן־ עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָ֨ה יְהוֹיָקִ֣ים בְּמָלְכ֔וֹ וְאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ (זבידה) זְבוּדָּ֥ה בַת־ פְּדָיָ֖ה מִן־ רוּמָֽה׃ (יג) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשׂ֖וּ אֲבֹתָֽיו׃ סוף פרק כג (יד) בְּיָמָ֣יו עָלָ֔ה נְבֻכַדְנֶאצַּ֖ר מֶ֣לֶךְ בָּבֶ֑ל וַיְהִי־ ל֨וֹ יְהוֹיָקִ֥ים עֶ֙בֶד֙ שָׁלֹ֣שׁ שָׁנִ֔ים וַיָּ֖שָׁב וַיִּמְרָד־ בּֽוֹ׃ (טו) וַיְשַׁלַּ֣ח יְהוָ֣ה ׀ בּ֡וֹ אֶת־ גְּדוּדֵ֣י כַשְׂדִּים֩ וְאֶת־ גְּדוּדֵ֨י אֲרָ֜ם וְאֵ֣ת ׀ גְּדוּדֵ֣י מוֹאָ֗ב וְאֵת֙ גְּדוּדֵ֣י בְנֵֽי־ עַמּ֔וֹן וַיְשַׁלְּחֵ֥ם בִּֽיהוּדָ֖ה לְהַֽאֲבִיד֑וֹ כִּדְבַ֣ר יְהוָ֔ה אֲשֶׁ֣ר דִּבֶּ֔ר בְּיַ֖ד עֲבָדָ֥יו הַנְּבִיאִֽים׃ (טז) אַ֣ךְ ׀ עַל־ פִּ֣י יְהוָ֗ה הָֽיְתָה֙ בִּֽיהוּדָ֔ה לְהָסִ֖יר מֵעַ֣ל פָּנָ֑יו בְּחַטֹּ֣את מְנַשֶּׁ֔ה כְּכֹ֖ל אֲשֶׁ֥ר עָשָֽׂה׃ (יז) וְגַ֤ם דַּֽם־ הַנָּקִי֙ אֲשֶׁ֣ר שָׁפָ֔ךְ וַיְמַלֵּ֥א אֶת־ יְרוּשָׁלִַ֖ם דָּ֣ם נָקִ֑י וְלֹֽא־ אָבָ֥ה יְהוָ֖ה לִסְלֹֽחַ׃ (יח) וְיֶ֛תֶר דִּבְרֵ֥י יְהוֹיָקִ֖ים וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (יט) וַיִּשְׁכַּ֥ב יְהוֹיָקִ֖ים עִם־ אֲבֹתָ֑יו וַיִּמְלֹ֛ךְ יְהוֹיָכִ֥ין בְּנ֖וֹ תַּחְתָּֽיו׃ (כ) וְלֹֽא־ הֹסִ֥יף עוֹד֙ מֶ֣לֶךְ מִצְרַ֔יִם לָצֵ֖את מֵֽאַרְצ֑וֹ כִּֽי־ לָקַ֞ח מֶ֣לֶךְ בָּבֶ֗ל מִנַּ֤חַל מִצְרַ֙יִם֙ עַד־ נְהַר־ פְּרָ֔ת כֹּ֛ל אֲשֶׁ֥ר הָיְתָ֖ה לְמֶ֥לֶךְ מִצְרָֽיִם׃ (כא) בֶּן־ שְׁמֹנֶ֨ה עֶשְׂרֵ֤ה שָׁנָה֙ יְהוֹיָכִ֣ין בְּמָלְכ֔וֹ וּשְׁלֹשָׁ֣ה חֳדָשִׁ֔ים מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ נְחֻשְׁתָּ֥א בַת־ אֶלְנָתָ֖ן מִירוּשָׁלִָֽם׃ (כב) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה אָבִֽיו׃ (כג) בָּעֵ֣ת הַהִ֔יא (עלה) עָל֗וּ עַבְדֵ֛י נְבֻכַדְנֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל יְרוּשָׁלִָ֑ם וַתָּבֹ֥א הָעִ֖יר בַּמָּצֽוֹר׃ (כד) וַיָּבֹ֛א נְבוּכַדְנֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל עַל־ הָעִ֑יר וַעֲבָדָ֖יו צָרִ֥ים עָלֶֽיהָ׃ (כה) וַיֵּצֵ֞א יְהוֹיָכִ֤ין מֶֽלֶךְ־ יְהוּדָה֙ עַל־ מֶ֣לֶךְ בָּבֶ֔ל ה֣וּא וְאִמּ֔וֹ וַעֲבָדָ֖יו וְשָׂרָ֣יו וְסָֽרִיסָ֑יו וַיִּקַּ֤ח אֹתוֹ֙ מֶ֣לֶךְ בָּבֶ֔ל בִּשְׁנַ֥ת שְׁמֹנֶ֖ה לְמָלְכֽוֹ׃ (כו) וַיּוֹצֵ֣א מִשָּׁ֗ם אֶת־ כָּל־ אוֹצְרוֹת֙ בֵּ֣ית יְהוָ֔ה וְאֽוֹצְר֖וֹת בֵּ֣ית הַמֶּ֑לֶךְ וַיְקַצֵּ֞ץ אֶת־ כָּל־ כְּלֵ֣י הַזָּהָ֗ב אֲשֶׁ֨ר עָשָׂ֜ה שְׁלֹמֹ֤ה מֶֽלֶךְ־ יִשְׂרָאֵל֙ בְּהֵיכַ֣ל יְהוָ֔ה כַּֽאֲשֶׁ֖ר דִּבֶּ֥ר יְהוָֽה׃ (כז) וְהִגְלָ֣ה אֶת־ כָּל־ יְ֠רוּשָׁלִַם וְֽאֶת־ כָּל־ הַשָּׂרִ֞ים וְאֵ֣ת ׀ כָּל־ גִּבּוֹרֵ֣י הַחַ֗יִל (עשרה) עֲשֶׂ֤רֶת אֲלָפִים֙ גּוֹלֶ֔ה וְכָל־ הֶחָרָ֖שׁ וְהַמַּסְגֵּ֑ר לֹ֣א נִשְׁאַ֔ר זוּלַ֖ת דַּלַּ֥ת עַם־ הָאָֽרֶץ׃ (כח) וַיֶּ֥גֶל אֶת־ יְהוֹיָכִ֖ין בָּבֶ֑לָה וְאֶת־ אֵ֣ם הַ֠מֶּלֶךְ וְאֶת־ נְשֵׁ֨י הַמֶּ֜לֶךְ וְאֶת־ סָרִיסָ֗יו וְאֵת֙ (אולי) אֵילֵ֣י הָאָ֔רֶץ הוֹלִ֛יךְ גּוֹלָ֥ה מִירוּשָׁלִַ֖ם בָּבֶֽלָה׃ (כט) וְאֵת֩ כָּל־ אַנְשֵׁ֨י הַחַ֜יִל שִׁבְעַ֣ת אֲלָפִ֗ים וְהֶחָרָ֤שׁ וְהַמַּסְגֵּר֙ אֶ֔לֶף הַכֹּ֕ל גִּבּוֹרִ֖ים עֹשֵׂ֣י מִלְחָמָ֑ה וַיְבִיאֵ֧ם מֶֽלֶךְ־ בָּבֶ֛ל גּוֹלָ֖ה בָּבֶֽלָה׃ (ל) וַיַּמְלֵ֧ךְ מֶֽלֶךְ־ בָּבֶ֛ל אֶת־ מַתַּנְיָ֥ה דֹד֖וֹ תַּחְתָּ֑יו וַיַּסֵּ֥ב אֶת־ שְׁמ֖וֹ צִדְקִיָּֽהוּ׃
