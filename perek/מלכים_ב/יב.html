מלכים ב פרק יב לפי החלוקה לסדרים

פרק יב בספר מלכים ב (22 פסוקים) נמצא במלכים סדר כה (פסוקים מג–מד) ובמלכים סדר כו (פסוקים א–כ).

מלכים סדר כה

(א) וַיֵּ֣לֶךְ מִשָּׁ֡ם וַיִּמְצָ֣א אֶת־ יְהוֹנָדָב֩ בֶּן־ רֵכָ֨ב לִקְרָאת֜וֹ וַֽיְבָרְכֵ֗הוּ וַיֹּ֨אמֶר אֵלָ֜יו הֲיֵ֧שׁ אֶת־ לְבָבְךָ֣ יָשָׁ֗ר כַּאֲשֶׁ֤ר לְבָבִי֙ עִם־ לְבָבֶ֔ךָ וַיֹּ֨אמֶר יְהוֹנָדָ֥ב יֵ֛שׁ וָיֵ֖שׁ תְּנָ֣ה אֶת־ יָדֶ֑ךָ וַיִּתֵּ֣ן יָד֔וֹ וַיַּעֲלֵ֥הוּ אֵלָ֖יו אֶל־ הַמֶּרְכָּבָֽה׃ (ב) וַיֹּ֙אמֶר֙ לְכָ֣ה אִתִּ֔י וּרְאֵ֖ה בְּקִנְאָתִ֣י לַיהוָ֑ה וַיַּרְכִּ֥בוּ אֹת֖וֹ בְּרִכְבּֽוֹ׃ (ג) וַיָּבֹא֙ שֹֽׁמְר֔וֹן וַ֠יַּךְ אֶת־ כָּל־ הַנִּשְׁאָרִ֧ים לְאַחְאָ֛ב בְּשֹׁמְר֖וֹן עַד־ הִשְׁמִיד֑וֹ כִּדְבַ֣ר יְהוָ֔ה אֲשֶׁ֥ר דִּבֶּ֖ר אֶל־ אֵלִיָּֽהוּ׃ (ד) וַיִּקְבֹּ֤ץ יֵהוּא֙ אֶת־ כָּל־ הָעָ֔ם וַיֹּ֣אמֶר אֲלֵהֶ֔ם אַחְאָ֕ב עָבַ֥ד אֶת־ הַבַּ֖עַל מְעָ֑ט יֵה֖וּא יַעַבְדֶ֥נּוּ הַרְבֵּֽה׃ (ה) וְעַתָּ֣ה כָל־ נְבִיאֵ֣י הַבַּ֡עַל כָּל־ עֹבְדָ֣יו וְכָל־ כֹּהֲנָיו֩ קִרְא֨וּ אֵלַ֜י אִ֣ישׁ אַל־ יִפָּקֵ֗ד כִּי֩ זֶ֨בַח גָּד֥וֹל לִי֙ לַבַּ֔עַל כֹּ֥ל אֲשֶׁר־ יִפָּקֵ֖ד לֹ֣א יִֽחְיֶ֑ה וְיֵהוּא֙ עָשָׂ֣ה בְעָקְבָּ֔ה לְמַ֥עַן הַאֲבִ֖יד אֶת־ עֹבְדֵ֥י הַבָּֽעַל׃ (ו) וַיֹּ֣אמֶר יֵה֗וּא קַדְּשׁ֧וּ עֲצָרָ֛ה לַבַּ֖עַל וַיִּקְרָֽאוּ׃ (ז) וַיִּשְׁלַ֤ח יֵהוּא֙ בְּכָל־ יִשְׂרָאֵ֔ל וַיָּבֹ֙אוּ֙ כָּל־ עֹבְדֵ֣י הַבַּ֔עַל וְלֹֽא־ נִשְׁאַ֥ר אִ֖ישׁ אֲשֶׁ֣ר לֹֽא־ בָ֑א וַיָּבֹ֙אוּ֙ בֵּ֣ית הַבַּ֔עַל וַיִּמָּלֵ֥א בֵית־ הַבַּ֖עַל פֶּ֥ה לָפֶֽה׃ (ח) וַיֹּ֗אמֶר לַֽאֲשֶׁר֙ עַל־ הַמֶּלְתָּחָ֔ה הוֹצֵ֣א לְב֔וּשׁ לְכֹ֖ל עֹבְדֵ֣י הַבָּ֑עַל וַיֹּצֵ֥א לָהֶ֖ם הַמַּלְבּֽוּשׁ׃ (ט) וַיָּבֹ֥א יֵה֛וּא וִיהוֹנָדָ֥ב בֶּן־ רֵכָ֖ב בֵּ֣ית הַבָּ֑עַל וַיֹּ֜אמֶר לְעֹבְדֵ֣י הַבַּ֗עַל חַפְּשׂ֤וּ וּרְאוּ֙ פֶּן־ יֶשׁ־ פֹּ֤ה עִמָּכֶם֙ מֵעַבְדֵ֣י יְהוָ֔ה כִּ֛י אִם־ עֹבְדֵ֥י הַבַּ֖עַל לְבַדָּֽם׃ (י) וַיָּבֹ֕אוּ לַעֲשׂ֖וֹת זְבָחִ֣ים וְעֹל֑וֹת וְיֵה֞וּא שָׂם־ ל֤וֹ בַחוּץ֙ שְׁמֹנִ֣ים אִ֔ישׁ וַיֹּ֗אמֶר הָאִ֤ישׁ אֲשֶׁר־ יִמָּלֵט֙ מִן־ הָאֲנָשִׁ֗ים אֲשֶׁ֤ר אֲנִי֙ מֵבִ֣יא עַל־ יְדֵיכֶ֔ם נַפְשׁ֖וֹ תַּ֥חַת נַפְשֽׁוֹ׃ (יא) וַיְהִ֞י כְּכַלֹּת֣וֹ ׀ לַעֲשׂ֣וֹת הָעֹלָ֗ה וַיֹּ֣אמֶר יֵ֠הוּא לָרָצִ֨ים וְלַשָּׁלִשִׁ֜ים בֹּ֤אוּ הַכּוּם֙ אִ֣ישׁ אַל־ יֵצֵ֔א וַיַּכּ֖וּם לְפִי־ חָ֑רֶב וַיַּשְׁלִ֗כוּ הָֽרָצִים֙ וְהַשָּׁ֣לִשִׁ֔ים וַיֵּלְכ֖וּ עַד־ עִ֥יר בֵּית־ הַבָּֽעַל׃ (יב) וַיֹּצִ֛אוּ אֶת־ מַצְּב֥וֹת בֵּית־ הַבַּ֖עַל וַֽיִּשְׂרְפֽוּהָ׃ (יג) וַֽיִּתְּצ֔וּ אֵ֖ת מַצְּבַ֣ת הַבָּ֑עַל וַֽיִּתְּצוּ֙ אֶת־ בֵּ֣ית הַבַּ֔עַל וַיְשִׂמֻ֥הוּ (למחראות) לְמֽוֹצָא֖וֹת עַד־ הַיּֽוֹם׃ (יד) וַיַּשְׁמֵ֥ד יֵה֛וּא אֶת־ הַבַּ֖עַל מִיִּשְׂרָאֵֽל׃ (טו) רַ֠ק חֲטָאֵ֞י יָרָבְעָ֤ם בֶּן־ נְבָט֙ אֲשֶׁ֣ר הֶחֱטִ֣יא אֶת־ יִשְׂרָאֵ֔ל לֹֽא־ סָ֥ר יֵה֖וּא מֵאַֽחֲרֵיהֶ֑ם עֶגְלֵי֙ הַזָּהָ֔ב אֲשֶׁ֥ר בֵּֽית־ אֵ֖ל וַאֲשֶׁ֥ר בְּדָֽן׃ (טז) וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ יֵה֗וּא יַ֤עַן אֲשֶׁר־ הֱטִיבֹ֙תָ֙ לַעֲשׂ֤וֹת הַיָּשָׁר֙ בְּעֵינַ֔י כְּכֹל֙ אֲשֶׁ֣ר בִּלְבָבִ֔י עָשִׂ֖יתָ לְבֵ֣ית אַחְאָ֑ב בְּנֵ֣י רְבִעִ֔ים יֵשְׁב֥וּ לְךָ֖ עַל־ כִּסֵּ֥א יִשְׂרָאֵֽל׃ (יז) וְיֵה֗וּא לֹ֥א שָׁמַ֛ר לָלֶ֛כֶת בְּתֽוֹרַת־ יְהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בְּכָל־ לְבָב֑וֹ לֹ֣א סָ֗ר מֵעַל֙ חַטֹּ֣אות יָֽרָבְעָ֔ם אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (יח) בַּיָּמִ֣ים הָהֵ֔ם הֵחֵ֣ל יְהוָ֔ה לְקַצּ֖וֹת בְּיִשְׂרָאֵ֑ל וַיַּכֵּ֥ם חֲזָאֵ֖ל בְּכָל־ גְּב֥וּל יִשְׂרָאֵֽל׃ (יט) מִן־ הַיַּרְדֵּן֙ מִזְרַ֣ח הַשֶּׁ֔מֶשׁ אֵ֚ת כָּל־ אֶ֣רֶץ הַגִּלְעָ֔ד הַגָּדִ֥י וְהָרֻאובֵנִ֖י וְהַֽמְנַשִּׁ֑י מֵעֲרֹעֵר֙ אֲשֶׁ֣ר עַל־ נַ֣חַל אַרְנֹ֔ן וְהַגִּלְעָ֖ד וְהַבָּשָֽׁן׃ (כ) וְיֶ֨תֶר דִּבְרֵ֥י יֵה֛וּא וְכָל־ אֲשֶׁ֥ר עָשָׂ֖ה וְכָל־ גְּבוּרָת֑וֹ הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כא) וַיִּשְׁכַּ֤ב יֵהוּא֙ עִם־ אֲבֹתָ֔יו וַיִּקְבְּר֥וּ אֹת֖וֹ בְּשֹׁמְר֑וֹן וַיִּמְלֹ֛ךְ יְהוֹאָחָ֥ז בְּנ֖וֹ תַּחְתָּֽיו׃ (כב) וְהַיָּמִ֗ים אֲשֶׁ֨ר מָלַ֤ךְ יֵהוּא֙ עַל־ יִשְׂרָאֵ֔ל עֶשְׂרִ֥ים וּשְׁמֹנֶֽה־ שָׁנָ֖ה בְּשֹׁמְרֽוֹן׃ (כג) וַֽעֲתַלְיָה֙ אֵ֣ם אֲחַזְיָ֔הוּ (וראתה) רָאֲתָ֖ה כִּ֣י מֵ֣ת בְּנָ֑הּ וַתָּ֙קָם֙ וַתְּאַבֵּ֔ד אֵ֖ת כָּל־ זֶ֥רַע הַמַּמְלָכָֽה׃ (כד) וַתִּקַּ֣ח יְהוֹשֶׁ֣בַע בַּת־ הַמֶּֽלֶךְ־ י֠וֹרָם אֲח֨וֹת אֲחַזְיָ֜הוּ אֶת־ יוֹאָ֣שׁ בֶּן־ אֲחַזְיָ֗ה וַתִּגְנֹ֤ב אֹתוֹ֙ מִתּ֤וֹךְ בְּנֵֽי־ הַמֶּ֙לֶךְ֙ (הממותתים) הַמּ֣וּמָתִ֔ים אֹת֥וֹ וְאֶת־ מֵינִקְתּ֖וֹ בַּחֲדַ֣ר הַמִּטּ֑וֹת וַיַּסְתִּ֧רוּ אֹת֛וֹ מִפְּנֵ֥י עֲתַלְיָ֖הוּ וְלֹ֥א הוּמָֽת׃ (כה) וַיְהִ֤י אִתָּהּ֙ בֵּ֣ית יְהוָ֔ה מִתְחַבֵּ֖א שֵׁ֣שׁ שָׁנִ֑ים וַעֲתַלְיָ֖ה מֹלֶ֥כֶת עַל־ הָאָֽרֶץ׃ (כו) וּבַשָּׁנָ֣ה הַ֠שְּׁבִיעִית שָׁלַ֨ח יְהוֹיָדָ֜ע וַיִּקַּ֣ח ׀ אֶת־ שָׂרֵ֣י (המאיות) הַמֵּא֗וֹת לַכָּרִי֙ וְלָ֣רָצִ֔ים וַיָּבֵ֥א אֹתָ֛ם אֵלָ֖יו בֵּ֣ית יְהוָ֑ה וַיִּכְרֹת֩ לָהֶ֨ם בְּרִ֜ית וַיַּשְׁבַּ֤ע אֹתָם֙ בְּבֵ֣ית יְהוָ֔ה וַיַּ֥רְא אֹתָ֖ם אֶת־ בֶּן־ הַמֶּֽלֶךְ׃ (כז) וַיְצַוֵּ֣ם לֵאמֹ֔ר זֶ֥ה הַדָּבָ֖ר אֲשֶׁ֣ר תַּעֲשׂ֑וּן הַשְּׁלִשִׁ֤ית מִכֶּם֙ בָּאֵ֣י הַשַּׁבָּ֔ת וְשֹׁ֣מְרֵ֔י מִשְׁמֶ֖רֶת בֵּ֥ית הַמֶּֽלֶךְ׃ (כח) וְהַשְּׁלִשִׁית֙ בְּשַׁ֣עַר ס֔וּר וְהַשְּׁלִשִׁ֥ית בַּשַּׁ֖עַר אַחַ֣ר הָרָצִ֑ים וּשְׁמַרְתֶּ֛ם אֶת־ מִשְׁמֶ֥רֶת הַבַּ֖יִת מַסָּֽח׃ (כט) וּשְׁתֵּ֤י הַיָּדוֹת֙ בָּכֶ֔ם כֹּ֖ל יֹצְאֵ֣י הַשַּׁבָּ֑ת וְשָֽׁמְר֛וּ אֶת־ מִשְׁמֶ֥רֶת בֵּית־ יְהוָ֖ה אֶל־ הַמֶּֽלֶךְ׃ (ל) וְהִקַּפְתֶּ֨ם עַל־ הַמֶּ֜לֶךְ סָבִ֗יב אִ֚ישׁ וְכֵלָ֣יו בְּיָד֔וֹ וְהַבָּ֥א אֶל־ הַשְּׂדֵר֖וֹת יוּמָ֑ת וִהְי֥וּ אֶת־ הַמֶּ֖לֶךְ בְּצֵאת֥וֹ וּבְבֹאֽוֹ׃ (לא) וַֽיַּעֲשׂ֞וּ שָׂרֵ֣י (המאיות) הַמֵּא֗וֹת כְּכֹ֣ל אֲשֶׁר־ צִוָּה֮ יְהוֹיָדָ֣ע הַכֹּהֵן֒ וַיִּקְחוּ֙ אִ֣ישׁ אֶת־ אֲנָשָׁ֔יו בָּאֵ֣י הַשַּׁבָּ֔ת עִ֖ם יֹצְאֵ֣י הַשַּׁבָּ֑ת וַיָּבֹ֖אוּ אֶל־ יְהוֹיָדָ֥ע הַכֹּהֵֽן׃ (לב) וַיִּתֵּ֨ן הַכֹּהֵ֜ן לְשָׂרֵ֣י (המאיות) הַמֵּא֗וֹת אֶֽת־ הַחֲנִית֙ וְאֶת־ הַשְּׁלָטִ֔ים אֲשֶׁ֖ר לַמֶּ֣לֶךְ דָּוִ֑ד אֲשֶׁ֖ר בְּבֵ֥ית יְהוָֽה׃ (לג) וַיַּעַמְד֨וּ הָרָצִ֜ים אִ֣ישׁ ׀ וְכֵלָ֣יו בְּיָד֗וֹ מִכֶּ֨תֶף הַבַּ֤יִת הַיְמָנִית֙ עַד־ כֶּ֤תֶף הַבַּ֙יִת֙ הַשְּׂמָאלִ֔ית לַמִּזְבֵּ֖חַ וְלַבָּ֑יִת עַל־ הַמֶּ֖לֶךְ סָבִֽיב׃ (לד) וַיּוֹצִ֣א אֶת־ בֶּן־ הַמֶּ֗לֶךְ וַיִּתֵּ֤ן עָלָיו֙ אֶת־ הַנֵּ֙זֶר֙ וְאֶת־ הָ֣עֵד֔וּת וַיַּמְלִ֥כוּ אֹת֖וֹ וַיִּמְשָׁחֻ֑הוּ וַיַּכּוּ־ כָ֔ף וַיֹּאמְר֖וּ יְחִ֥י הַמֶּֽלֶךְ׃ (לה) וַתִּשְׁמַ֣ע עֲתַלְיָ֔ה אֶת־ ק֥וֹל הָֽרָצִ֖ין הָעָ֑ם וַתָּבֹ֥א אֶל־ הָעָ֖ם בֵּ֥ית יְהוָֽה׃ (לו) וַתֵּ֡רֶא יָרָבְעָ֧ם בֶּן־ נְבָ֛ט אֲשֶׁר־ הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל לֹא־ סָ֥ר מִמֶּֽנָּה׃ (כג) וַיִּֽחַר־ אַ֥ף יְהוָ֖ה בְּיִשְׂרָאֵ֑ל וַֽיִּתְּנֵ֞ם בְּיַ֣ד ׀ חֲזָאֵ֣ל מֶֽלֶךְ־ אֲרָ֗ם וּבְיַ֛ד בֶּן־ הֲדַ֥ד בֶּן־ חֲזָאֵ֖ל כָּל־ הַיָּמִֽים׃ (כד) וַיְחַ֥ל יְהוֹאָחָ֖ז אֶת־ פְּנֵ֣י יְהוָ֑ה וַיִּשְׁמַ֤ע אֵלָיו֙ יְהוָ֔ה כִּ֤י רָאָה֙ אֶת־ לַ֣חַץ יִשְׂרָאֵ֔ל כִּֽי־ לָחַ֥ץ אֹתָ֖ם מֶ֥לֶךְ אֲרָֽם׃ (כה) וַיִּתֵּ֨ן יְהוָ֤ה לְיִשְׂרָאֵל֙ מוֹשִׁ֔יעַ וַיֵּ֣צְא֔וּ מִתַּ֖חַת יַד־ אֲרָ֑ם וַיֵּשְׁב֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל בְּאָהֳלֵיהֶ֖ם כִּתְמ֥וֹל שִׁלְשֽׁוֹם׃ (כו) אַ֠ךְ לֹֽא־ סָ֜רוּ מֵחַטֹּ֧אות בֵּית־ יָרָבְעָ֛ם אֲשֶׁר־ (החטי) הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל בָּ֣הּ הָלָ֑ךְ וְגַם֙ הָאֲשֵׁרָ֔ה עָמְדָ֖ה בְּשֹׁמְרֽוֹן׃ (כז) כִּ֣י לֹא֩ הִשְׁאִ֨יר לִיהוֹאָחָ֜ז עָ֗ם כִּ֣י אִם־ חֲמִשִּׁ֤ים פָּֽרָשִׁים֙ וַעֲשָׂ֣רָה רֶ֔כֶב וַעֲשֶׂ֥רֶת אֲלָפִ֖ים רַגְלִ֑י כִּ֤י אִבְּדָם֙ מֶ֣לֶךְ אֲרָ֔ם וַיְשִׂמֵ֥ם כֶּֽעָפָ֖ר לָדֻֽשׁ׃ (כח) וְיֶ֨תֶר דִּבְרֵ֧י יְהוֹאָחָ֛ז וְכָל־ אֲשֶׁ֥ר עָשָׂ֖ה וּגְבוּרָת֑וֹ הֲלוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כט) וַיִּשְׁכַּ֤ב יְהֽוֹאָחָז֙ עִם־ אֲבֹתָ֔יו וַֽיִּקְבְּרֻ֖הוּ בְּשֹׁמְר֑וֹן וַיִּמְלֹ֛ךְ יוֹאָ֥שׁ בְּנ֖וֹ תַּחְתָּֽיו׃ (ל) בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָשֶׁ֙בַע֙ שָׁנָ֔ה לְיוֹאָ֖שׁ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ יְהוֹאָ֨שׁ בֶּן־ יְהוֹאָחָ֤ז עַל־ יִשְׂרָאֵל֙ בְּשֹׁ֣מְר֔וֹן שֵׁ֥שׁ עֶשְׂרֵ֖ה שָׁנָֽה׃ (לא) וַיַּֽעֲשֶׂ֥ה הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לֹ֣א סָ֗ר מִכָּל־ חַטֹּ֞אות יָרָבְעָ֧ם בֶּן־ נְבָ֛ט אֲשֶׁר־ הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל בָּ֥הּ הָלָֽךְ׃ (לב) וְיֶ֨תֶר דִּבְרֵ֤י יוֹאָשׁ֙ וְכָל־ אֲשֶׁ֣ר עָשָׂ֔ה וּגְב֣וּרָת֔וֹ אֲשֶׁ֣ר נִלְחַ֔ם עִ֖ם אֲמַצְיָ֣ה מֶֽלֶךְ־ יְהוּדָ֑ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (לג) וַיִּשְׁכַּ֤ב יוֹאָשׁ֙ עִם־ אֲבֹתָ֔יו וְיָרָבְעָ֖ם יָשַׁ֣ב עַל־ כִּסְא֑וֹ וַיִּקָּבֵ֤ר יוֹאָשׁ֙ בְּשֹׁ֣מְר֔וֹן עִ֖ם מַלְכֵ֥י יִשְׂרָאֵֽל׃ (לד) וֶֽאֱלִישָׁע֙ חָלָ֣ה אֶת־ חָלְי֔וֹ אֲשֶׁ֥ר יָמ֖וּת בּ֑וֹ וַיֵּ֨רֶד אֵלָ֜יו יוֹאָ֣שׁ מֶֽלֶךְ־ יִשְׂרָאֵ֗ל וַיֵּ֤בְךְּ עַל־ פָּנָיו֙ וַיֹּאמַ֔ר אָבִ֣י ׀ אָבִ֔י רֶ֥כֶב יִשְׂרָאֵ֖ל וּפָרָשָֽׁיו׃ (לה) וַיֹּ֤אמֶר לוֹ֙ אֱלִישָׁ֔ע קַ֖ח קֶ֣שֶׁת וְחִצִּ֑ים וַיִּקַּ֥ח אֵלָ֖יו קֶ֥שֶׁת וְחִצִּֽים׃ (לו) וַיֹּ֣אמֶר ׀ לְמֶ֣לֶךְ יִשְׂרָאֵ֗ל הַרְכֵּ֤ב יָֽדְךָ֙ עַל־ הַקֶּ֔שֶׁת וַיַּרְכֵּ֖ב יָד֑וֹ וַיָּ֧שֶׂם אֱלִישָׁ֛ע יָדָ֖יו עַל־ יְדֵ֥י הַמֶּֽלֶךְ׃ (לז) וַיֹּ֗אמֶר פְּתַ֧ח הַחַלּ֛וֹן קֵ֖דְמָה וַיִּפְתָּ֑ח וַיֹּ֤אמֶר אֱלִישָׁ֤ע יְרֵה֙ וַיּ֔וֹר וַיֹּ֗אמֶר חֵץ־ תְּשׁוּעָ֤ה לַֽיהוָה֙ וְחֵ֣ץ תְּשׁוּעָ֣ה בַֽאֲרָ֔ם וְהִכִּיתָ֧ אֶת־ אֲרָ֛ם בַּאֲפֵ֖ק עַד־ כַּלֵּֽה׃ (לח) וַיֹּ֛אמֶר קַ֥ח הַחִצִּ֖ים וַיִּקָּ֑ח וַיֹּ֤אמֶר לְמֶֽלֶךְ־ יִשְׂרָאֵל֙ הַךְ־ אַ֔רְצָה וַיַּ֥ךְ שָֽׁלֹשׁ־ פְּעָמִ֖ים וַֽיַּעֲמֹֽד׃ (לט) וַיִּקְצֹ֨ף עָלָ֜יו אִ֣ישׁ הָאֱלֹהִ֗ים וַיֹּ֙אמֶר֙ לְהַכּ֨וֹת חָמֵ֤שׁ אוֹ־ שֵׁשׁ֙ פְּעָמִ֔ים אָ֛ז הִכִּ֥יתָ אֶת־ אֲרָ֖ם עַד־ כַּלֵּ֑ה וְעַתָּ֕ה שָׁלֹ֥שׁ פְּעָמִ֖ים תַּכֶּ֥ה אֶת־ אֲרָֽם׃ (מ) וַיָּ֥מָת אֱלִישָׁ֖ע וַֽיִּקְבְּרֻ֑הוּ וּגְדוּדֵ֥י מוֹאָ֛ב יָבֹ֥אוּ בָאָ֖רֶץ בָּ֥א שָׁנָֽה׃ (מא) וַיְהִ֞י הֵ֣ם ׀ קֹבְרִ֣ים אִ֗ישׁ וְהִנֵּה֙ רָא֣וּ אֶֽת־ הַגְּד֔וּד וַיַּשְׁלִ֥יכוּ אֶת־ הָאִ֖ישׁ בְּקֶ֣בֶר אֱלִישָׁ֑ע וַיֵּ֜לֶךְ וַיִּגַּ֤ע הָאִישׁ֙ בְּעַצְמ֣וֹת אֱלִישָׁ֔ע וַיְחִ֖י וַיָּ֥קָם עַל־ רַגְלָֽיו׃ (מב) וַֽחֲזָאֵל֙ מֶ֣לֶךְ אֲרָ֔ם לָחַ֖ץ אֶת־ יִשְׂרָאֵ֑ל כֹּ֖ל יְמֵ֥י יְהוֹאָחָֽז׃
