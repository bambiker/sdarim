מלכים ב פרק ז לפי החלוקה לסדרים

פרק ז בספר מלכים ב (20 פסוקים) נמצא במלכים סדר כב (פסוקים יב–כו) ובמלכים סדר כג (פסוקים א–ה).

מלכים סדר כב

(א) וַיִּכְרֶ֨ה לָהֶ֜ם כֵּרָ֣ה גְדוֹלָ֗ה וַיֹּֽאכְלוּ֙ וַיִּשְׁתּ֔וּ וַֽיְשַׁלְּחֵ֔ם וַיֵּלְכ֖וּ אֶל־ אֲדֹֽנֵיהֶ֑ם וְלֹֽא־ יָ֤סְפוּ עוֹד֙ גְּדוּדֵ֣י אֲרָ֔ם לָב֖וֹא בְּאֶ֥רֶץ יִשְׂרָאֵֽל׃ (ב) וַֽיְהִי֙ אַחֲרֵי־ כֵ֔ן וַיִּקְבֹּ֛ץ בֶּן־ הֲדַ֥ד מֶֽלֶךְ־ אֲרָ֖ם אֶת־ כָּל־ מַחֲנֵ֑הוּ וַיַּ֕עַל וַיָּ֖צַר עַל־ שֹׁמְרֽוֹן׃ (ג) וַיְהִ֨י רָעָ֤ב גָּדוֹל֙ בְּשֹׁ֣מְר֔וֹן וְהִנֵּ֖ה צָרִ֣ים עָלֶ֑יהָ עַ֣ד הֱי֤וֹת רֹאשׁ־ חֲמוֹר֙ בִּשְׁמֹנִ֣ים כֶּ֔סֶף וְרֹ֛בַע הַקַּ֥ב (חרייונים) דִּבְיוֹנִ֖ים בַּחֲמִשָּׁה־ כָֽסֶף׃ (ד) וַֽיְהִי֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל עֹבֵ֖ר עַל־ הַחֹמָ֑ה וְאִשָּׁ֗ה צָעֲקָ֤ה אֵלָיו֙ לֵאמֹ֔ר הוֹשִׁ֖יעָה אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (ה) וַיֹּ֙אמֶר֙ אַל־ יוֹשִׁעֵ֣ךְ יְהוָ֔ה מֵאַ֖יִן אֽוֹשִׁיעֵ֑ךְ הֲמִן־ הַגֹּ֖רֶן א֥וֹ מִן־ הַיָּֽקֶב׃ (ו) וַיֹּֽאמֶר־ לָ֥הּ הַמֶּ֖לֶךְ מַה־ לָּ֑ךְ וַתֹּ֗אמֶר הָאִשָּׁ֨ה הַזֹּ֜את אָמְרָ֣ה אֵלַ֗י תְּנִ֤י אֶת־ בְּנֵךְ֙ וְנֹאכְלֶ֣נּוּ הַיּ֔וֹם וְאֶת־ בְּנִ֖י נֹאכַ֥ל מָחָֽר׃ (ז) וַנְּבַשֵּׁ֥ל אֶת־ בְּנִ֖י וַנֹּֽאכְלֵ֑הוּ וָאֹמַ֨ר אֵלֶ֜יהָ בַּיּ֣וֹם הָאַחֵ֗ר תְּנִ֤י אֶת־ בְּנֵךְ֙ וְנֹ֣אכְלֶ֔נּוּ וַתַּחְבִּ֖א אֶת־ בְּנָֽהּ׃ (ח) וַיְהִי֩ כִשְׁמֹ֨עַ הַמֶּ֜לֶךְ אֶת־ דִּבְרֵ֤י הָֽאִשָּׁה֙ וַיִּקְרַ֣ע אֶת־ בְּגָדָ֔יו וְה֖וּא עֹבֵ֣ר עַל־ הַחֹמָ֑ה וַיַּ֣רְא הָעָ֔ם וְהִנֵּ֥ה הַשַּׂ֛ק עַל־ בְּשָׂר֖וֹ מִבָּֽיִת׃ (ט) וַיֹּ֕אמֶר כֹּֽה־ יַעֲשֶׂה־ לִּ֥י אֱלֹהִ֖ים וְכֹ֣ה יוֹסִ֑ף אִֽם־ יַעֲמֹ֞ד רֹ֣אשׁ אֱלִישָׁ֧ע בֶּן־ שָׁפָ֛ט עָלָ֖יו הַיּֽוֹם׃ (י) וֶאֱלִישָׁע֙ יֹשֵׁ֣ב בְּבֵית֔וֹ וְהַזְּקֵנִ֖ים יֹשְׁבִ֣ים אִתּ֑וֹ וַיִּשְׁלַ֨ח אִ֜ישׁ מִלְּפָנָ֗יו בְּטֶ֣רֶם יָבֹא֩ הַמַּלְאָ֨ךְ אֵלָ֜יו וְה֣וּא ׀ אָמַ֣ר אֶל־ הַזְּקֵנִ֗ים הַרְּאִיתֶם֙ כִּֽי־ שָׁלַ֞ח בֶּן־ הַֽמְרַצֵּ֤חַ הַזֶּה֙ לְהָסִ֣יר אֶת־ רֹאשִׁ֔י רְא֣וּ ׀ כְּבֹ֣א הַמַּלְאָ֗ךְ סִגְר֤וּ הַדֶּ֙לֶת֙ וּלְחַצְתֶּ֤ם אֹתוֹ֙ בַּדֶּ֔לֶת הֲל֗וֹא ק֛וֹל רַגְלֵ֥י אֲדֹנָ֖יו אַחֲרָֽיו׃ (יא) עוֹדֶ֙נּוּ֙ מְדַבֵּ֣ר עִמָּ֔ם וְהִנֵּ֥ה הַמַּלְאָ֖ךְ יֹרֵ֣ד אֵלָ֑יו וַיֹּ֗אמֶר הִנֵּֽה־ זֹ֤את הָֽרָעָה֙ מֵאֵ֣ת יְהוָ֔ה מָֽה־ אוֹחִ֥יל לַיהוָ֖ה עֽוֹד׃ תחילת פרק ז (יב) וַיֹּ֣אמֶר אֱלִישָׁ֔ע שִׁמְע֖וּ דְּבַר־ יְהוָ֑ה כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה כָּעֵ֤ת ׀ מָחָר֙ סְאָֽה־ סֹ֣לֶת בְּשֶׁ֗קֶל וְסָאתַ֧יִם שְׂעֹרִ֛ים בְּשֶׁ֖קֶל בְּשַׁ֥עַר שֹׁמְרֽוֹן׃ (יג) וַיַּ֣עַן הַשָּׁלִ֡ישׁ אֲשֶׁר־ לַמֶּלֶךְ֩ נִשְׁעָ֨ן עַל־ יָד֜וֹ אֶת־ אִ֣ישׁ הָאֱלֹהִים֮ וַיֹּאמַר֒ הִנֵּ֣ה יְהוָ֗ה עֹשֶׂ֤ה אֲרֻבּוֹת֙ בַּשָּׁמַ֔יִם הֲיִהְיֶ֖ה הַדָּבָ֣ר הַזֶּ֑ה וַיֹּ֗אמֶר הִנְּכָ֤ה רֹאֶה֙ בְּעֵינֶ֔יךָ וּמִשָּׁ֖ם לֹ֥א תֹאכֵֽל׃ (יד) וְאַרְבָּעָ֧ה אֲנָשִׁ֛ים הָי֥וּ מְצֹרָעִ֖ים פֶּ֣תַח הַשָּׁ֑עַר וַיֹּֽאמְרוּ֙ אִ֣ישׁ אֶל־ רֵעֵ֔הוּ מָ֗ה אֲנַ֛חְנוּ יֹשְׁבִ֥ים פֹּ֖ה עַד־ מָֽתְנוּ׃ (טו) אִם־ אָמַרְנוּ֩ נָב֨וֹא הָעִ֜יר וְהָרָעָ֤ב בָּעִיר֙ וָמַ֣תְנוּ שָׁ֔ם וְאִם־ יָשַׁ֥בְנוּ פֹ֖ה וָמָ֑תְנוּ וְעַתָּ֗ה לְכוּ֙ וְנִפְּלָה֙ אֶל־ מַחֲנֵ֣ה אֲרָ֔ם אִם־ יְחַיֻּ֣נוּ נִֽחְיֶ֔ה וְאִם־ יְמִיתֻ֖נוּ וָמָֽתְנוּ׃ (טז) וַיָּק֣וּמוּ בַנֶּ֔שֶׁף לָב֖וֹא אֶל־ מַחֲנֵ֣ה אֲרָ֑ם וַיָּבֹ֗אוּ עַד־ קְצֵה֙ מַחֲנֵ֣ה אֲרָ֔ם וְהִנֵּ֥ה אֵֽין־ שָׁ֖ם אִֽישׁ׃ (יז) וַאדֹנָ֞י הִשְׁמִ֣יעַ ׀ אֶת־ מַחֲנֵ֣ה אֲרָ֗ם ק֥וֹל רֶ֙כֶב֙ ק֣וֹל ס֔וּס ק֖וֹל חַ֣יִל גָּד֑וֹל וַיֹּאמְר֞וּ אִ֣ישׁ אֶל־ אָחִ֗יו הִנֵּ֣ה שָֽׂכַר־ עָלֵינוּ֩ מֶ֨לֶךְ יִשְׂרָאֵ֜ל אֶת־ מַלְכֵ֧י הַחִתִּ֛ים וְאֶת־ מַלְכֵ֥י מִצְרַ֖יִם לָב֥וֹא עָלֵֽינוּ׃ (יח) וַיָּקוּמוּ֮ וַיָּנ֣וּסוּ בַנֶּשֶׁף֒ וַיַּעַזְב֣וּ אֶת־ אָהֳלֵיהֶ֗ם וְאֶת־ סֽוּסֵיהֶם֙ וְאֶת־ חֲמֹ֣רֵיהֶ֔ם הַֽמַּחֲנֶ֖ה כַּאֲשֶׁר־ הִ֑יא וַיָּנֻ֖סוּ אֶל־ נַפְשָֽׁם׃ (יט) וַיָּבֹאוּ֩ הַֽמְצֹרָעִ֨ים הָאֵ֜לֶּה עַד־ קְצֵ֣ה הַֽמַּחֲנֶ֗ה וַיָּבֹ֜אוּ אֶל־ אֹ֤הֶל אֶחָד֙ וַיֹּאכְל֣וּ וַיִּשְׁתּ֔וּ וַיִּשְׂא֣וּ מִשָּׁ֗ם כֶּ֤סֶף וְזָהָב֙ וּבְגָדִ֔ים וַיֵּלְכ֖וּ וַיַּטְמִ֑נוּ וַיָּשֻׁ֗בוּ וַיָּבֹ֙אוּ֙ אֶל־ אֹ֣הֶל אַחֵ֔ר וַיִּשְׂא֣וּ מִשָּׁ֔ם וַיֵּלְכ֖וּ וַיַּטְמִֽנוּ׃ (כ) וַיֹּאמְרוּ֩ אִ֨ישׁ אֶל־ רֵעֵ֜הוּ לֹֽא־ כֵ֣ן ׀ אֲנַ֣חְנוּ עֹשִׂ֗ים הַיּ֤וֹם הַזֶּה֙ יוֹם־ בְּשֹׂרָ֣ה ה֔וּא וַאֲנַ֣חְנוּ מַחְשִׁ֗ים וְחִכִּ֛ינוּ עַד־ א֥וֹר הַבֹּ֖קֶר וּמְצָאָ֣נוּ עָו֑וֹן וְעַתָּה֙ לְכ֣וּ וְנָבֹ֔אָה וְנַגִּ֖ידָה בֵּ֥ית הַמֶּֽלֶךְ׃ (כא) וַיָּבֹ֗אוּ וַֽיִּקְרְאוּ֮ אֶל־ שֹׁעֵ֣ר הָעִיר֒ וַיַּגִּ֤ידוּ לָהֶם֙ לֵאמֹ֔ר בָּ֚אנוּ אֶל־ מַחֲנֵ֣ה אֲרָ֔ם וְהִנֵּ֥ה אֵֽין־ שָׁ֛ם אִ֖ישׁ וְק֣וֹל אָדָ֑ם כִּ֣י אִם־ הַסּ֤וּס אָסוּר֙ וְהַחֲמ֣וֹר אָס֔וּר וְאֹהָלִ֖ים כַּאֲשֶׁר־ הֵֽמָּה׃ (כב) וַיִּקְרָ֖א הַשֹּֽׁעֲרִ֑ים וַיַּגִּ֕ידוּ בֵּ֥ית הַמֶּ֖לֶךְ פְּנִֽימָה׃ (כג) וַיָּ֨קָם הַמֶּ֜לֶךְ לַ֗יְלָה וַיֹּ֙אמֶר֙ אֶל־ עֲבָדָ֔יו אַגִּֽידָה־ נָּ֣א לָכֶ֔ם אֵ֛ת אֲשֶׁר־ עָ֥שׂוּ לָ֖נוּ אֲרָ֑ם יָדְע֞וּ כִּי־ רְעֵבִ֣ים אֲנַ֗חְנוּ וַיֵּצְא֤וּ מִן־ הַֽמַּחֲנֶה֙ לְהֵחָבֵ֤ה (בהשדה) בַשָּׂדֶה֙ לֵאמֹ֔ר כִּֽי־ יֵצְא֤וּ מִן־ הָעִיר֙ וְנִתְפְּשֵׂ֣ם חַיִּ֔ים וְאֶל־ הָעִ֖יר נָבֹֽא׃ (כד) וַיַּעַן֩ אֶחָ֨ד מֵעֲבָדָ֜יו וַיֹּ֗אמֶר וְיִקְחוּ־ נָ֞א חֲמִשָּׁ֣ה מִן־ הַסּוּסִים֮ הַֽנִּשְׁאָרִים֮ אֲשֶׁ֣ר נִשְׁאֲרוּ־ בָהּ֒ הִנָּ֗ם כְּכָל־ (ההמון) הֲמ֤וֹן יִשְׂרָאֵל֙ אֲשֶׁ֣ר נִשְׁאֲרוּ־ בָ֔הּ הִנָּ֕ם כְּכָל־ הֲמ֥וֹן יִשְׂרָאֵ֖ל אֲשֶׁר־ תָּ֑מּוּ וְנִשְׁלְחָ֖ה וְנִרְאֶֽה׃ (כה) וַיִּקְח֕וּ שְׁנֵ֖י רֶ֣כֶב סוּסִ֑ים וַיִּשְׁלַ֨ח הַמֶּ֜לֶךְ אַחֲרֵ֧י מַחֲנֵֽה־ אֲרָ֛ם לֵאמֹ֖ר לְכ֥וּ וּרְאֽוּ׃ (כו) וַיֵּלְכ֣וּ אַחֲרֵיהֶם֮ עַד־ הַיַּרְדֵּן֒ וְהִנֵּ֣ה כָל־ הַדֶּ֗רֶךְ מְלֵאָ֤ה בְגָדִים֙ וְכֵלִ֔ים אֲשֶׁר־ הִשְׁלִ֥יכוּ אֲרָ֖ם (בהחפזם) בְּחָפְזָ֑ם וַיָּשֻׁ֙בוּ֙ הַמַּלְאָכִ֔ים וַיַּגִּ֖דוּ לַמֶּֽלֶךְ׃

מלכים סדר כג

(א) וַיֵּצֵ֣א הָעָ֔ם וַיָּבֹ֕זּוּ אֵ֖ת מַחֲנֵ֣ה אֲרָ֑ם וַיְהִ֨י סְאָה־ סֹ֜לֶת בְּשֶׁ֗קֶל וְסָאתַ֧יִם שְׂעֹרִ֛ים בְּשֶׁ֖קֶל כִּדְבַ֥ר יְהוָֽה׃ (ב) וְהַמֶּלֶךְ֩ הִפְקִ֨יד אֶת־ הַשָּׁלִ֜ישׁ אֲשֶׁר־ נִשְׁעָ֤ן עַל־ יָדוֹ֙ עַל־ הַשַּׁ֔עַר וַיִּרְמְסֻ֧הוּ הָעָ֛ם בַּשַּׁ֖עַר וַיָּמֹ֑ת כַּאֲשֶׁ֤ר דִּבֶּר֙ אִ֣ישׁ הָאֱלֹהִ֔ים אֲשֶׁ֣ר דִּבֶּ֔ר בְּרֶ֥דֶת הַמֶּ֖לֶךְ אֵלָֽיו׃ (ג) וַיְהִ֗י כְּדַבֵּר֙ אִ֣ישׁ הָאֱלֹהִ֔ים אֶל־ הַמֶּ֖לֶךְ לֵאמֹ֑ר סָאתַ֨יִם שְׂעֹרִ֜ים בְּשֶׁ֗קֶל וּֽסְאָה־ סֹ֙לֶת֙ בְּשֶׁ֔קֶל יִהְיֶה֙ כָּעֵ֣ת מָחָ֔ר בְּשַׁ֖עַר שֹׁמְרֽוֹן׃ (ד) וַיַּ֨עַן הַשָּׁלִ֜ישׁ אֶת־ אִ֣ישׁ הָאֱלֹהִים֮ אֹת֥וֹ הָעָ֛ם בַּשַּׁ֖עַר וַיָּמֹֽת׃ סוף פרק ז (ו) וֶאֱלִישָׁ֡ע דִּבֶּ֣ר אֶל־ הָאִשָּׁה֩ אֲשֶׁר־ הֶחֱיָ֨ה אֶת־ בְּנָ֜הּ לֵאמֹ֗ר ק֤וּמִי וּלְכִי֙ (אתי) אַ֣תְּ וּבֵיתֵ֔ךְ וְג֖וּרִי בַּאֲשֶׁ֣ר תָּג֑וּרִי כִּֽי־ קָרָ֤א יְהוָה֙ לָֽרָעָ֔ב וְגַם־ בָּ֥א אֶל־ הָאָ֖רֶץ שֶׁ֥בַע שָׁנִֽים׃ (ז) וַתָּ֙קָם֙ הָֽאִשָּׁ֔ה וַתַּ֕עַשׂ כִּדְבַ֖ר אִ֣ישׁ הָאֱלֹהִ֑ים וַתֵּ֤לֶךְ הִיא֙ וּבֵיתָ֔הּ וַתָּ֥גָר בְּאֶֽרֶץ־ פְּלִשְׁתִּ֖ים שֶׁ֥בַע שָׁנִֽים׃ (ח) וַיְהִ֗י מִקְצֵה֙ שֶׁ֣בַע שָׁנִ֔ים וַתָּ֥שָׁב הָאִשָּׁ֖ה מֵאֶ֣רֶץ פְּלִשְׁתִּ֑ים וַתֵּצֵא֙ לִצְעֹ֣ק אֶל־ הַמֶּ֔לֶךְ אֶל־ בֵּיתָ֖הּ וְאֶל־ שָׂדָֽהּ׃ (ט) אֲרַמִּים֙ בָּֽרָמָ֔ה בְּהִלָּ֣חֲמ֔וֹ אֶת־ חֲזָהאֵ֖ל מֶ֣לֶךְ אֲרָ֑ם וַאֲחַזְיָ֨הוּ בֶן־ יְהוֹרָ֜ם מֶ֣לֶךְ יְהוּדָ֗ה יָרַ֡ד לִרְא֞וֹת אֶת־ יוֹרָ֧ם בֶּן־ אַחְאָ֛ב בְּיִזְרְעֶ֖אל כִּֽי־ חֹלֶ֥ה הֽוּא׃ (לה) וֶאֱלִישָׁע֙ הַנָּבִ֔יא קָרָ֕א לְאַחַ֖ד מִבְּנֵ֣י הַנְּבִיאִ֑ים וַיֹּ֨אמֶר ל֜וֹ חֲגֹ֣ר מָתְנֶ֗יךָ וְ֠קַח פַּ֣ךְ הַשֶּׁ֤מֶן הַזֶּה֙ בְּיָדֶ֔ךָ וְלֵ֖ךְ רָמֹ֥ת גִּלְעָֽד׃ (לו) וּבָ֖אתָ שָׁ֑מָּה וּרְאֵֽה־ שָׁ֠ם יֵה֨וּא בֶן־ יְהוֹשָׁפָ֜ט בֶּן־ נִמְשִׁ֗י וּבָ֙אתָ֙ וַהֲקֵֽמֹתוֹ֙ מִתּ֣וֹך אֶחָ֔יו וְהֵבֵיאתָ֥ אֹת֖וֹ חֶ֥דֶר בְּחָֽדֶר׃ (לז) וְלָקַחְתָּ֤ פַךְ־ הַשֶּׁ֙מֶן֙ וְיָצַקְתָּ֣ עַל־ רֹאשׁ֔וֹ וְאָֽמַרְתָּ֙ כֹּֽה־ אָמַ֣ר יְהוָ֔ה מְשַׁחְתִּ֥יךָֽ לְמֶ֖לֶךְ אֶל־ יִשְׂרָאֵ֑ל וּפָתַחְתָּ֥ הַדֶּ֛לֶת וְנַ֖סְתָּה וְלֹ֥א תְחַכֶּֽה׃ (לח) וַיֵּ֧לֶךְ הַנַּ֛עַר הַנַּ֥עַר הַנָּבִ֖יא רָמֹ֥ת גִּלְעָֽד׃ (לט) וַיָּבֹ֗א וְהִנֵּ֨ה שָׂרֵ֤י הַחַ֙יִל֙ יֹֽשְׁבִ֔ים וַיֹּ֕אמֶר דָּבָ֥ר לִ֛י אֵלֶ֖יךָ הַשָּׂ֑ר וַיֹּ֤אמֶר יֵהוּא֙ אֶל־ מִ֣י מִכֻּלָּ֔נוּ וַיֹּ֖אמֶר אֵלֶ֥יךָ הַשָּֽׂר׃ (מ) וַיָּ֙קָם֙ וַיָּבֹ֣א הַבַּ֔יְתָה וַיִּצֹ֥ק הַשֶּׁ֖מֶן אֶל־ רֹאשׁ֑וֹ וַיֹּ֣אמֶר ל֗וֹ כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל מְשַׁחְתִּ֧יךָֽ לְמֶ֛לֶךְ אֶל־ עַ֥ם יְהוָ֖ה אֶל־ יִשְׂרָאֵֽל׃ (מא) וְהִ֨כִּיתָ֔ה אֶת־ בֵּ֥ית אַחְאָ֖ב אֲדֹנֶ֑יךָ וְנִקַּמְתִּ֞י דְּמֵ֣י ׀ עֲבָדַ֣י הַנְּבִיאִ֗ים וּדְמֵ֛י כָּל־ עַבְדֵ֥י יְהוָ֖ה מִיַּ֥ד אִיזָֽבֶל׃ (מב) וְאָבַ֖ד כָּל־ בֵּ֣ית אַחְאָ֑ב וְהִכְרַתִּ֤י לְאַחְאָב֙ מַשְׁתִּ֣ין בְּקִ֔יר וְעָצ֥וּר וְעָז֖וּב בְּיִשְׂרָאֵֽל׃ (מג) וְנָֽתַתִּי֙ אֶת־ בֵּ֣ית אַחְאָ֔ב כְּבֵ֖ית יָרָבְעָ֣ם בֶּן־ נְבָ֑ט וּכְבֵ֖ית בַּעְשָׁ֥א בֶן־ אֲחִיָּֽה׃ (מד) וְאֶת־ אִיזֶ֜בֶל יֹאכְל֧וּ הַכְּלָבִ֛ים בְּחֵ֥לֶק יִזְרְעֶ֖אל וְאֵ֣ין קֹבֵ֑ר וַיִּפְתַּ֥ח הַדֶּ֖לֶת וַיָּנֹֽס׃ (מה) וְיֵה֗וּא יָצָא֙ אֶל־ עַבְדֵ֣י אֲדֹנָ֔יו וַיֹּ֤אמֶר לוֹ֙ הֲשָׁל֔וֹם מַדּ֛וּעַ בָּֽא־ הַמְשֻׁגָּ֥ע הַזֶּ֖ה אֵלֶ֑יךָ וַיֹּ֣אמֶר אֲלֵיהֶ֔ם אַתֶּ֛ם יְדַעְתֶּ֥ם אֶת־ הָאִ֖ישׁ וְאֶת־ שִׂיחֽוֹ׃ (מו) וַיֹּאמְר֣וּ שֶׁ֔קֶר הַגֶּד־ נָ֖א לָ֑נוּ וַיֹּ֗אמֶר כָּזֹ֤את וְכָזֹאת֙ אָמַ֤ר אֵלַי֙ לֵאמֹ֔ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה מְשַׁחְתִּ֥יךָֽ לְמֶ֖לֶךְ אֶל־ יִשְׂרָאֵֽל׃
