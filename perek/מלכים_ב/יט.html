מלכים ב פרק יט לפי החלוקה לסדרים

פרק יט בספר מלכים ב (37 פסוקים) נמצא במלכים סדר ל (פסוקים לג–נ) ובמלכים סדר לא (פסוקים א–יט).

מלכים סדר ל

(א) וַיִּדְבַּק֙ בַּֽיהוָ֔ה לֹא־ סָ֖ר מֵאַֽחֲרָ֑יו וַיִּשְׁמֹר֙ מִצְוֺתָ֔יו אֲשֶׁר־ צִוָּ֥ה יְהוָ֖ה אֶת־ מֹשֶֽׁה׃ (ב) וְהָיָ֤ה יְהוָה֙ עִמּ֔וֹ בְּכֹ֥ל אֲשֶׁר־ יֵצֵ֖א יַשְׂכִּ֑יל וַיִּמְרֹ֥ד בְּמֶֽלֶךְ־ אַשּׁ֖וּר וְלֹ֥א עֲבָדֽוֹ׃ (ג) הֽוּא־ הִכָּ֧ה אֶת־ פְּלִשְׁתִּ֛ים עַד־ עַזָּ֖ה וְאֶת־ גְּבוּלֶ֑יהָ מִמִּגְדַּ֥ל נוֹצְרִ֖ים עַד־ עִ֥יר מִבְצָֽר׃ (ד) וַֽיְהִ֞י בַּשָּׁנָ֤ה הָֽרְבִיעִית֙ לַמֶּ֣לֶךְ חִזְקִיָּ֔הוּ הִ֚יא הַשָּׁנָ֣ה הַשְּׁבִיעִ֔ית לְהוֹשֵׁ֥עַ בֶּן־ אֵלָ֖ה מֶ֣לֶךְ יִשְׂרָאֵ֑ל עָלָ֞ה שַׁלְמַנְאֶ֧סֶר מֶֽלֶךְ־ אַשּׁ֛וּר עַל־ שֹׁמְר֖וֹן וַיָּ֥צַר עָלֶֽיהָ׃ (ה) וַֽיִּלְכְּדֻ֗הָ מִקְצֵה֙ שָׁלֹ֣שׁ שָׁנִ֔ים בִּשְׁנַת־ שֵׁ֖שׁ לְחִזְקִיָּ֑ה הִ֣יא שְׁנַת־ תֵּ֗שַׁע לְהוֹשֵׁ֙עַ֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל נִלְכְּדָ֖ה שֹׁמְרֽוֹן׃ (ו) וַיֶּ֧גֶל מֶֽלֶךְ־ אַשּׁ֛וּר אֶת־ יִשְׂרָאֵ֖ל אַשּׁ֑וּרָה וַיַּנְחֵ֞ם בַּחְלַ֧ח וּבְחָב֛וֹר נְהַ֥ר גּוֹזָ֖ן וְעָרֵ֥י מָדָֽי׃ (ז) עַ֣ל ׀ אֲשֶׁ֣ר לֹֽא־ שָׁמְע֗וּ בְּקוֹל֙ יְהוָ֣ה אֱלֹהֵיהֶ֔ם וַיַּעַבְרוּ֙ אֶת־ בְּרִית֔וֹ אֵ֚ת כָּל־ אֲשֶׁ֣ר צִוָּ֔ה מֹשֶׁ֖ה עֶ֣בֶד יְהוָ֑ה וְלֹ֥א שָׁמְע֖וּ וְלֹ֥א עָשֽׂוּ׃ (ח) וּבְאַרְבַּע֩ עֶשְׂרֵ֨ה שָׁנָ֜ה לַמֶּ֣לֶךְ חִזְקִיָּ֗ה עָלָ֞ה סַנְחֵרִ֤יב מֶֽלֶךְ־ אַשּׁוּר֙ עַ֣ל כָּל־ עָרֵ֧י יְהוּדָ֛ה הַבְּצֻר֖וֹת וַֽיִּתְפְּשֵֽׂם׃ (ט) וַיִּשְׁלַ֣ח חִזְקִיָּ֣ה מֶֽלֶךְ־ יְהוּדָ֣ה אֶל־ מֶֽלֶךְ־ אַשּׁוּר֩ ׀ לָכִ֨ישָׁה ׀ לֵאמֹ֤ר ׀ חָטָ֙אתִי֙ שׁ֣וּב מֵֽעָלַ֔י אֵ֛ת אֲשֶׁר־ תִּתֵּ֥ן עָלַ֖י אֶשָּׂ֑א וַיָּ֨שֶׂם מֶֽלֶךְ־ אַשּׁ֜וּר עַל־ חִזְקִיָּ֣ה מֶֽלֶךְ־ יְהוּדָ֗ה שְׁלֹ֤שׁ מֵאוֹת֙ כִּכַּר־ כֶּ֔סֶף וּשְׁלֹשִׁ֖ים כִּכַּ֥ר זָהָֽב׃ (י) וַיִּתֵּן֙ חִזְקִיָּ֔ה אֶת־ כָּל־ הַכֶּ֖סֶף הַנִּמְצָ֣א בֵית־ יְהוָ֑ה וּבְאֹצְר֖וֹת בֵּ֥ית הַמֶּֽלֶךְ׃ (יא) בָּעֵ֣ת הַהִ֗יא קִצַּ֨ץ חִזְקִיָּ֜ה אֶת־ דַּלְת֨וֹת הֵיכַ֤ל יְהוָה֙ וְאֶת־ הָאֹ֣מְנ֔וֹת אֲשֶׁ֣ר צִפָּ֔ה חִזְקִיָּ֖ה מֶ֣לֶךְ יְהוּדָ֑ה וַֽיִּתְּנֵ֖ם לְמֶ֥לֶךְ אַשּֽׁוּר׃ (יב) וַיִּשְׁלַ֣ח מֶֽלֶךְ־ אַשּׁ֡וּר אֶת־ תַּרְתָּ֥ן וְאֶת־ רַב־ סָרִ֣יס ׀ וְאֶת־ רַב־ שָׁקֵ֨ה מִן־ לָכִ֜ישׁ אֶל־ הַמֶּ֧לֶךְ חִזְקִיָּ֛הוּ בְּחֵ֥יל כָּבֵ֖ד יְרוּשָׁלִָ֑ם וַֽיַּעֲלוּ֙ וַיָּבֹ֣אוּ יְרוּשָׁלִַ֔ם וַיַּעֲל֣וּ וַיָּבֹ֗אוּ וַיַּֽעַמְדוּ֙ בִּתְעָלַת֙ הַבְּרֵכָ֣ה הָֽעֶלְיוֹנָ֔ה אֲשֶׁ֕ר בִּמְסִלַּ֖ת שְׂדֵ֥ה כוֹבֵֽס׃ (יג) וַֽיִּקְרְאוּ֙ אֶל־ הַמֶּ֔לֶךְ וַיֵּצֵ֧א אֲלֵהֶ֛ם אֶלְיָקִ֥ים בֶּן־ חִלְקִיָּ֖הוּ אֲשֶׁ֣ר עַל־ הַבָּ֑יִת וְשֶׁבְנָה֙ הַסֹּפֵ֔ר וְיוֹאָ֥ח בֶּן־ אָסָ֖ף הַמַּזְכִּֽיר׃ (יד) וַיֹּ֤אמֶר אֲלֵהֶם֙ רַב־ שָׁקֵ֔ה אִמְרוּ־ נָ֖א אֶל־ חִזְקִיָּ֑הוּ כֹּֽה־ אָמַ֞ר הַמֶּ֤לֶךְ הַגָּדוֹל֙ מֶ֣לֶךְ אַשּׁ֔וּר מָ֧ה הַבִּטָּח֛וֹן הַזֶּ֖ה אֲשֶׁ֥ר בָּטָֽחְתָּ׃ (טו) אָמַ֙רְתָּ֙ אַךְ־ דְּבַר־ שְׂפָתַ֔יִם עֵצָ֥ה וּגְבוּרָ֖ה לַמִּלְחָמָ֑ה עַתָּה֙ עַל־ מִ֣י בָטַ֔חְתָּ כִּ֥י מָרַ֖דְתָּ בִּֽי׃ (טז) עַתָּ֡ה הִנֵּ֣ה בָטַ֣חְתָּ לְּךָ֡ עַל־ מִשְׁעֶנֶת֩ הַקָּנֶ֨ה הָרָצ֤וּץ הַזֶּה֙ עַל־ מִצְרַ֔יִם אֲשֶׁ֨ר יִסָּמֵ֥ךְ אִישׁ֙ עָלָ֔יו וּבָ֥א בְכַפּ֖וֹ וּנְקָבָ֑הּ כֵּ֚ן פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם לְכָֽל־ הַבֹּטְחִ֖ים עָלָֽיו׃ (יז) וְכִי־ תֹאמְר֣וּן אֵלַ֔י אֶל־ יְהוָ֥ה אֱלֹהֵ֖ינוּ בָּטָ֑חְנוּ הֲלוֹא־ ה֗וּא אֲשֶׁ֨ר הֵסִ֤יר חִזְקִיָּ֙הוּ֙ אֶת־ בָּמֹתָ֣יו וְאֶת־ מִזְבְּחֹתָ֔יו וַיֹּ֤אמֶר לִֽיהוּדָה֙ וְלִיר֣וּשָׁלִַ֔ם לִפְנֵי֙ הַמִּזְבֵּ֣חַ הַזֶּ֔ה תִּֽשְׁתַּחֲו֖וּ בִּירוּשָׁלִָֽם׃ (יח) וְעַתָּה֙ הִתְעָ֣רֶב נָ֔א אֶת־ אֲדֹנִ֖י אֶת־ מֶ֣לֶךְ אַשּׁ֑וּר וְאֶתְּנָ֤ה לְךָ֙ אַלְפַּ֣יִם סוּסִ֔ים אִם־ תּוּכַ֕ל לָ֥תֶת לְךָ֖ רֹכְבִ֥ים עֲלֵיהֶֽם׃ (יט) וְאֵ֣יךְ תָּשִׁ֗יב אֵ֠ת פְּנֵ֨י פַחַ֥ת אַחַ֛ד עַבְדֵ֥י אֲדֹנִ֖י הַקְּטַנִּ֑ים וַתִּבְטַ֤ח לְךָ֙ עַל־ מִצְרַ֔יִם לְרֶ֖כֶב וּלְפָרָשִֽׁים׃ (כ) עַתָּה֙ הֲמִבַּלְעֲדֵ֣י יְהוָ֔ה עָלִ֛יתִי עַל־ הַמָּק֥וֹם הַזֶּ֖ה לְהַשְׁחִת֑וֹ יְהוָה֙ אָמַ֣ר אֵלַ֔י עֲלֵ֛ה עַל־ הָאָ֥רֶץ הַזֹּ֖את וְהַשְׁחִיתָֽהּ׃ (כא) וַיֹּ֣אמֶר אֶלְיָקִ֣ים בֶּן־ חִ֠לְקִיָּהוּ וְשֶׁבְנָ֨ה וְיוֹאָ֜ח אֶל־ רַב־ שָׁקֵ֗ה דַּבֶּר־ נָ֤א אֶל־ עֲבָדֶ֙יךָ֙ אֲרָמִ֔ית כִּ֥י שֹׁמְעִ֖ים אֲנָ֑חְנוּ וְאַל־ תְּדַבֵּ֤ר עִמָּ֙נוּ֙ יְהוּדִ֔ית בְּאָזְנֵ֣י הָעָ֔ם אֲשֶׁ֖ר עַל־ הַחֹמָֽה׃ (כב) וַיֹּ֨אמֶר אֲלֵיהֶ֜ם רַב־ שָׁקֵ֗ה הַעַ֨ל אֲדֹנֶ֤יךָ וְאֵלֶ֙יךָ֙ שְׁלָחַ֣נִי אֲדֹנִ֔י לְדַבֵּ֖ר אֶת־ הַדְּבָרִ֣ים הָאֵ֑לֶּה הֲלֹ֣א עַל־ הָאֲנָשִׁ֗ים הַיֹּֽשְׁבִים֙ עַל־ הַ֣חֹמָ֔ה לֶאֱכֹ֣ל אֶת (חריהם) צוֹאָתָ֗ם וְלִשְׁתּ֛וֹת אֶת־ (שיניהם) מימֵ֥י רַגְלֵיהֶ֖ם עִמָּכֶֽם׃ (כג) וַֽיַּעֲמֹד֙ רַב־ שָׁקֵ֔ה וַיִּקְרָ֥א בְקוֹל־ גָּד֖וֹל יְהוּדִ֑ית וַיְדַבֵּ֣ר וַיֹּ֔אמֶר שִׁמְע֛וּ דְּבַר־ הַמֶּ֥לֶךְ הַגָּד֖וֹל מֶ֥לֶךְ אַשּֽׁוּר׃ (כד) כֹּ֚ה אָמַ֣ר הַמֶּ֔לֶךְ אַל־ יַשִּׁ֥יא לָכֶ֖ם חִזְקִיָּ֑הוּ כִּי־ לֹ֣א יוּכַ֔ל לְהַצִּ֥יל אֶתְכֶ֖ם מִיָּדֽוֹ׃ (כה) וְאַל־ יַבְטַ֨ח אֶתְכֶ֤ם חִזְקִיָּ֙הוּ֙ אֶל־ יְהוָ֣ה לֵאמֹ֔ר הַצֵּ֥ל יַצִּילֵ֖נוּ יְהוָ֑ה וְלֹ֤א תִנָּתֵן֙ אֶת־ הָעִ֣יר הַזֹּ֔את בְּיַ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (כו) אַֽל־ תִּשְׁמְע֖וּ אֶל־ חִזְקִיָּ֑הוּ כִּי֩ כֹ֨ה אָמַ֜ר מֶ֣לֶךְ אַשּׁ֗וּר עֲשֽׂוּ־ אִתִּ֤י בְרָכָה֙ וּצְא֣וּ אֵלַ֔י וְאִכְל֤וּ אִישׁ־ גַּפְנוֹ֙ וְאִ֣ישׁ תְּאֵֽנָת֔וֹ וּשְׁת֖וּ אִ֥ישׁ מֵֽי־ בוֹרֽוֹ׃ (כז) עַד־ בֹּאִי֩ וְלָקַחְתִּ֨י אֶתְכֶ֜ם אֶל־ אֶ֣רֶץ כְּאַרְצְכֶ֗ם אֶרֶץ֩ דָּגָ֨ן וְתִיר֜וֹשׁ אֶ֧רֶץ לֶ֣חֶם וּכְרָמִ֗ים אֶ֣רֶץ זֵ֤ית יִצְהָר֙ וּדְבַ֔שׁ וִֽחְי֖וּ וְלֹ֣א תָמֻ֑תוּ וְאַֽל־ תִּשְׁמְעוּ֙ אֶל־ חִזְקִיָּ֔הוּ כִּֽי־ יַסִּ֤ית אֶתְכֶם֙ לֵאמֹ֔ר יְהוָ֖ה יַצִּילֵֽנוּ׃ (כח) הַהַצֵּ֥ל הִצִּ֛ילוּ אֱלֹהֵ֥י הַגּוֹיִ֖ם אִ֣ישׁ אֶת־ אַרְצ֑וֹ מִיַּ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (כט) אַיֵּה֩ אֱלֹהֵ֨י חֲמָ֜ת וְאַרְפָּ֗ד אַיֵּ֛ה אֱלֹהֵ֥י סְפַרְוַ֖יִם הֵנַ֣ע וְעִוָּ֑ה כִּֽי־ הִצִּ֥ילוּ אֶת־ שֹׁמְר֖וֹן מִיָּדִֽי׃ (ל) מִ֚י בְּכָל־ אֱלֹהֵ֣י הָֽאֲרָצ֔וֹת אֲשֶׁר־ הִצִּ֥ילוּ אֶת־ אַרְצָ֖ם מִיָּדִ֑י כִּי־ יַצִּ֧יל יְהוָ֛ה אֶת־ יְרוּשָׁלִַ֖ם מִיָּדִֽי׃ (לא) וְהֶחֱרִ֣ישׁוּ הָעָ֔ם וְלֹֽא־ עָנ֥וּ אֹת֖וֹ דָּבָ֑ר כִּי־ מִצְוַ֨ת הַמֶּ֥לֶךְ הִ֛יא לֵאמֹ֖ר לֹ֥א תַעֲנֻֽהוּ׃ (לב) וַיָּבֹ֣א אֶלְיָקִ֣ים בֶּן־ חִלְקִיָּ֣ה אֲשֶׁר־ עַל־ הַ֠בַּיִת וְשֶׁבְנָ֨א הַסֹּפֵ֜ר וְיוֹאָ֨ח בֶּן־ אָסָ֧ף הַמַּזְכִּ֛יר אֶל־ חִזְקִיָּ֖הוּ קְרוּעֵ֣י בְגָדִ֑ים וַיַּגִּ֣דוּ ל֔וֹ דִּבְרֵ֖י רַב־ שָׁקֵֽה׃ תחילת פרק יט (לג) וַיְהִ֗י כִּשְׁמֹ֙עַ֙ הַמֶּ֣לֶךְ חִזְקִיָּ֔הוּ וַיִּקְרַ֖ע אֶת־ בְּגָדָ֑יו וַיִּתְכַּ֣ס בַּשָּׂ֔ק וַיָּבֹ֖א בֵּ֥ית יְהוָֽה׃ (לד) וַ֠יִּשְׁלַח אֶת־ אֶלְיָקִ֨ים אֲשֶׁר־ עַל־ הַבַּ֜יִת וְשֶׁבְנָ֣א הַסֹּפֵ֗ר וְאֵת֙ זִקְנֵ֣י הַכֹּֽהֲנִ֔ים מִתְכַּסִּ֖ים בַּשַּׂקִּ֑ים אֶל־ יְשַֽׁעְיָ֥הוּ הַנָּבִ֖יא בֶּן־ אָמֽוֹץ׃ (לה) וַיֹּאמְר֣וּ אֵלָ֗יו כֹּ֚ה אָמַ֣ר חִזְקִיָּ֔הוּ יוֹם־ צָרָ֧ה וְתוֹכֵחָ֛ה וּנְאָצָ֖ה הַיּ֣וֹם הַזֶּ֑ה כִּ֣י בָ֤אוּ בָנִים֙ עַד־ מַשְׁבֵּ֔ר וְכֹ֥חַ אַ֖יִן לְלֵדָֽה׃ (לו) אוּלַ֡י יִשְׁמַע֩ יְהוָ֨ה אֱלֹהֶ֜יךָ אֵ֣ת ׀ כָּל־ דִּבְרֵ֣י רַב־ שָׁקֵ֗ה אֲשֶׁר֩ שְׁלָח֨וֹ מֶֽלֶךְ־ אַשּׁ֤וּר ׀ אֲדֹנָיו֙ לְחָרֵף֙ אֱלֹהִ֣ים חַ֔י וְהוֹכִ֙יחַ֙ בַּדְּבָרִ֔ים אֲשֶׁ֥ר שָׁמַ֖ע יְהוָ֣ה אֱלֹהֶ֑יךָ וְנָשָׂ֣אתָ תְפִלָּ֔ה בְּעַ֥ד הַשְּׁאֵרִ֖ית הַנִּמְצָאָֽה׃ (לז) וַיָּבֹ֗אוּ עַבְדֵ֛י הַמֶּ֥לֶךְ חִזְקִיָּ֖הוּ אֶל־ יְשַֽׁעַיָֽהוּ׃ (לח) וַיֹּ֤אמֶר לָהֶם֙ יְשַֽׁעְיָ֔הוּ כֹּ֥ה תֹאמְר֖וּן אֶל־ אֲדֹֽנֵיכֶ֑ם כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ תִּירָא֙ מִפְּנֵ֤י הַדְּבָרִים֙ אֲשֶׁ֣ר שָׁמַ֗עְתָּ אֲשֶׁ֧ר גִּדְּפ֛וּ נַעֲרֵ֥י מֶֽלֶךְ־ אַשּׁ֖וּר אֹתִֽי׃ (לט) הִנְנִ֨י נֹתֵ֥ן בּוֹ֙ ר֔וּחַ וְשָׁמַ֥ע שְׁמוּעָ֖ה וְשָׁ֣ב לְאַרְצ֑וֹ וְהִפַּלְתִּ֥יו בַּחֶ֖רֶב בְּאַרְצֽוֹ׃ (מ) וַיָּ֙שָׁב֙ רַב־ שָׁקֵ֔ה וַיִּמְצָא֙ אֶת־ מֶ֣לֶךְ אַשּׁ֔וּר נִלְחָ֖ם עַל־ לִבְנָ֑ה כִּ֣י שָׁמַ֔ע כִּ֥י נָסַ֖ע מִלָּכִֽישׁ׃ (מא) וַיִּשְׁמַ֗ע אֶל־ תִּרְהָ֤קָה מֶֽלֶך־ כּוּשׁ֙ לֵאמֹ֔ר הִנֵּ֥ה יָצָ֖א לְהִלָּחֵ֣ם אִתָּ֑ךְ וַיָּ֙שָׁב֙ וַיִּשְׁלַ֣ח מַלְאָכִ֔ים אֶל־ חִזְקִיָּ֖הוּ לֵאמֹֽר׃ (מב) כֹּ֣ה תֹאמְר֗וּן אֶל־ חִזְקִיָּ֤הוּ מֶֽלֶךְ־ יְהוּדָה֙ לֵאמֹ֔ר אַל־ יַשִּׁאֲךָ֣ אֱלֹהֶ֔יךָ אֲשֶׁ֥ר אַתָּ֛ה בֹּטֵ֥חַ בּ֖וֹ לֵאמֹ֑ר לֹ֤א תִנָּתֵן֙ יְר֣וּשָׁלִַ֔ם בְּיַ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (מג) הִנֵּ֣ה ׀ אַתָּ֣ה שָׁמַ֗עְתָּ אֵת֩ אֲשֶׁ֨ר עָשׂ֜וּ מַלְכֵ֥י אַשּׁ֛וּר לְכָל־ הָאֲרָצ֖וֹת לְהַֽחֲרִימָ֑ם וְאַתָּ֖ה תִּנָּצֵֽל׃ (מד) הַהִצִּ֨ילוּ אֹתָ֜ם אֱלֹהֵ֤י הַגּוֹיִם֙ אֲשֶׁ֣ר שִׁחֲת֣וּ אֲבוֹתַ֔י אֶת־ גּוֹזָ֖ן וְאֶת־ חָרָ֑ן וְרֶ֥צֶף וּבְנֵי־ עֶ֖דֶן אֲשֶׁ֥ר בִּתְלַאשָּֽׂר׃ (מה) אַיּ֤וֹ מֶֽלֶךְ־ חֲמָת֙ וּמֶ֣לֶךְ אַרְפָּ֔ד וּמֶ֖לֶךְ לָעִ֣יר סְפַרְוָ֑יִם הֵנַ֖ע וְעִוָּֽה׃ (מו) וַיִּקַּ֨ח חִזְקִיָּ֧הוּ אֶת־ הַסְּפָרִ֛ים מִיַּ֥ד הַמַּלְאָכִ֖ים וַיִּקְרָאֵ֑ם וַיַּ֙עַל֙ בֵּ֣ית יְהוָ֔ה וַיִּפְרְשֵׂ֥הוּ חִזְקִיָּ֖הוּ לִפְנֵ֥י יְהוָֽה׃ (מז) וַיִּתְפַּלֵּ֨ל חִזְקִיָּ֜הוּ לִפְנֵ֣י יְהוָה֮ וַיֹּאמַר֒ יְהוָ֞ה אֱלֹהֵ֤י יִשְׂרָאֵל֙ יֹשֵׁ֣ב הַכְּרֻבִ֔ים אַתָּה־ ה֤וּא הָֽאֱלֹהִים֙ לְבַדְּךָ֔ לְכֹ֖ל מַמְלְכ֣וֹת הָאָ֑רֶץ אַתָּ֣ה עָשִׂ֔יתָ אֶת־ הַשָּׁמַ֖יִם וְאֶת־ הָאָֽרֶץ׃ (מח) הַטֵּ֨ה יְהוָ֤ה ׀ אָזְנְךָ֙ וּֽשֲׁמָ֔ע פְּקַ֧ח יְהוָ֛ה עֵינֶ֖יךָ וּרְאֵ֑ה וּשְׁמַ֗ע אֵ֚ת דִּבְרֵ֣י סַנְחֵרִ֔יב אֲשֶׁ֣ר שְׁלָח֔וֹ לְחָרֵ֖ף אֱלֹהִ֥ים חָֽי׃ (מט) אָמְנָ֖ם יְהוָ֑ה הֶחֱרִ֜יבוּ מַלְכֵ֥י אַשּׁ֛וּר אֶת־ הַגּוֹיִ֖ם וְאֶת־ אַרְצָֽם׃ (נ) וְנָתְנ֥וּ אֶת־ אֱלֹהֵיהֶ֖ם בָּאֵ֑שׁ כִּי֩ לֹ֨א אֱלֹהִ֜ים הֵ֗מָּה כִּ֣י אִם־ מַעֲשֵׂ֧ה יְדֵֽי־ אָדָ֛ם עֵ֥ץ וָאֶ֖בֶן וַֽיְאַבְּדֽוּם׃

(א) וְעַתָּה֙ יְהוָ֣ה אֱלֹהֵ֔ינוּ הוֹשִׁיעֵ֥נוּ נָ֖א מִיָּד֑וֹ וְיֵֽדְעוּ֙ כָּל־ מַמְלְכ֣וֹת הָאָ֔רֶץ כִּ֥י אַתָּ֛ה יְהוָ֥ה אֱלֹהִ֖ים לְבַדֶּֽךָ׃ (ב) וַיִּשְׁלַח֙ יְשַֽׁעְיָ֣הוּ בֶן־ אָמ֔וֹץ אֶל־ חִזְקִיָּ֖הוּ לֵאמֹ֑ר כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֨ר הִתְפַּלַּ֧לְתָּ אֵלַ֛י אֶל־ סַנְחֵרִ֥ב מֶֽלֶךְ־ אַשּׁ֖וּר שָׁמָֽעְתִּי׃ (ג) זֶ֣ה הַדָּבָ֔ר אֲשֶׁר־ דִּבֶּ֥ר יְהוָ֖ה עָלָ֑יו עֵינֶ֖יךָ עַל־ קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (ה) בְּיַ֣ד מַלְאָכֶיךָ֮ חֵרַ֣פְתָּ ׀ אֲדֹנָי֒ וַתֹּ֗אמֶר (ברכב) בְּרֹ֥ב רִכְבִּ֛י אֲנִ֥י עָלִ֛יתִי מְר֥וֹם הָרִ֖ים יַרְכְּתֵ֣י לְבָנ֑וֹן וְאֶכְרֹ֞ת קוֹמַ֤ת אֲרָזָיו֙ מִבְח֣וֹר בְּרֹשָׁ֔יו וְאָב֙וֹאָה֙ מְל֣וֹן קִצֹּ֔ה יַ֖עַר כַּרְמִלּֽוֹ׃ (ו) אֲנִ֣י קַ֔רְתִּי וְשָׁתִ֖יתִי מַ֣יִם זָרִ֑ים וְאַחְרִב֙ בְּכַף־ פְּעָמַ֔י כֹּ֖ל יְאֹרֵ֥י מָצֽוֹר׃ (ז) הֲלֹֽא־ שָׁמַ֤עְתָּ לְמֵֽרָחוֹק֙ אֹתָ֣הּ עָשִׂ֔יתִי לְמִ֥ימֵי קֶ֖דֶם וִֽיצַרְתִּ֑יהָ עַתָּ֣ה הֲבֵיאתִ֗יהָ וּתְהִ֗י לַהְשׁ֛וֹת גַּלִּ֥ים נִצִּ֖ים עָרִ֥ים בְּצֻרֽוֹת׃ (ח) וְיֹֽשְׁבֵיהֶן֙ קִצְרֵי־ יָ֔ד חַ֖תּוּ וַיֵּבֹ֑שׁוּ הָי֞וּ עֵ֤שֶׂב שָׂדֶה֙ וִ֣ירַק דֶּ֔שֶׁא חֲצִ֣יר גַּגּ֔וֹת וּשְׁדֵפָ֖ה לִפְנֵ֥י קָמָֽה׃ (ט) וְשִׁבְתְּךָ֛ וְצֵאתְךָ֥ וּבֹאֲךָ֖ יָדָ֑עְתִּי וְאֵ֖ת הִֽתְרַגֶּזְךָ֥ אֵלָֽי׃ (י) יַ֚עַן הִתְרַגֶּזְךָ֣ אֵלַ֔י וְשַׁאֲנַנְךָ֖ עָלָ֣ה בְאָזְנָ֑י וְשַׂמְתִּ֨י חַחִ֜י בְּאַפֶּ֗ךָ וּמִתְגִּי֙ בִּשְׂפָתֶ֔יךָ וַהֲשִׁ֣בֹתִ֔יךָ בַּדֶּ֖רֶךְ אֲשֶׁר־ בָּ֥אתָ בָּֽהּ׃ (יא) וְזֶה־ לְּךָ֣ הָא֔וֹת אָכ֤וֹל הַשָּׁנָה֙ סָפִ֔יחַ וּבַשָּׁנָ֥ה הַשֵּׁנִ֖ית סָחִ֑ישׁ וּבַשָּׁנָ֣ה הַשְּׁלִישִׁ֗ית זִרְע֧וּ וְקִצְר֛וּ וְנִטְע֥וּ כְרָמִ֖ים וְאִכְל֥וּ פִרְיָֽם׃ (יב) וְיָ֨סְפָ֜ה פְּלֵיטַ֧ת בֵּית־ יְהוּדָ֛ה כֹּֽה־ אָמַ֤ר יְהוָה֙ אֶל־ מֶ֣לֶךְ אַשּׁ֔וּר לֹ֤א יָבֹא֙ אֶל־ הָעִ֣יר הַזֹּ֔את וְלֹֽא־ יוֹרֶ֥ה שָׁ֖ם חֵ֑ץ וְלֹֽא־ יְקַדְּמֶ֣נָּה מָגֵ֔ן וְלֹֽא־ יִשְׁפֹּ֥ךְ עָלֶ֖יהָ סֹלְלָֽה׃ (טו) בַּדֶּ֥רֶךְ אֲשֶׁר־ יָבֹ֖א בָּ֣הּ יָשׁ֑וּב וְאֶל־ הָעִ֥יר הַזֹּ֛את לֹ֥א יָבֹ֖א נְאֻם־ יְהוָֽה׃ (טז) וְגַנּוֹתִ֛י אֶל־ הָעִ֥יר הַזֹּ֖את לְהֽוֹשִׁיעָ֑הּ לְמַֽעֲנִ֔י וּלְמַ֖עַן דָּוִ֥ד עַבְדִּֽי׃ (יז) וַיְהִי֮ בַּלַּ֣יְלָה הַהוּא֒ וַיֵּצֵ֣א ׀ מַלְאַ֣ךְ יְהוָ֗ה וַיַּךְ֙ בְּמַחֲנֵ֣ה אַשּׁ֔וּר מֵאָ֛ה שְׁמוֹנִ֥ים וַחֲמִשָּׁ֖ה אָ֑לֶף וַיַּשְׁכִּ֣ימוּ בַבֹּ֔קֶר וְהִנֵּ֥ה כֻלָּ֖ם פְּגָרִ֥ים מֵתִֽים׃ (יח) וַיִּסַּ֣ע וַיֵּ֔לֶךְ וַיָּ֖שָׁב סַנְחֵרִ֣יב מֶֽלֶךְ־ אַשּׁ֑וּר וַיֵּ֖שֶׁב בְּנִֽינְוֵֽה׃ (יט) וַיְהִי֩ ה֨וּא מִֽשְׁתַּחֲוֶ֜ה בֵּ֣ית ׀ נִסְרֹ֣ךְ אֱלֹהָ֗יו וְֽאַדְרַמֶּ֨לֶךְ וְשַׂרְאֶ֤צֶר בָּנָיו֙ הִכֻּ֣הוּ בַחֶ֔רֶב וְהֵ֥מָּה נִמְלְט֖וּ אֶ֣רֶץ אֲרָרָ֑ט וַיִּמְלֹ֛ךְ אֵֽסַר־ חַדֹּ֥ן בְּנ֖וֹ תַּחְתָּֽיו׃ סוף פרק יט (כ) בַּיָּמִ֣ים הָהֵ֔ם חָלָ֥ה חִזְקִיָּ֖הוּ לָמ֑וּת וַיָּבֹ֣א אֵ֠לָיו יְשַׁעְיָ֨הוּ בֶן־ אָמ֜וֹץ הַנָּבִ֗יא וַיֹּ֨אמֶר אֵלָ֜יו כֹּֽה־ אָמַ֤ר יְהוָה֙ צַ֣ו לְבֵיתֶ֔ךָ כִּ֛י מֵ֥ת אַתָּ֖ה וְלֹ֥א תִֽחְיֶֽה׃ (כא) וַיַּסֵּ֥ב אֶת־ פָּנָ֖יו אֶל־ הַקִּ֑יר וַיִּ֨תְפַּלֵּ֔ל אֶל־ יְהוָ֖ה לֵאמֹֽר׃ (כב) אָנָּ֣ה יְהוָ֗ה זְכָר־ נָ֞א אֵ֣ת אֲשֶׁ֧ר הִתְהַלַּ֣כְתִּי לְפָנֶ֗יךָ בֶּֽאֱמֶת֙ וּבְלֵבָ֣ב שָׁלֵ֔ם וְהַטּ֥וֹב בְּעֵינֶ֖יךָ עָשִׂ֑יתִי וַיֵּ֥בְךְּ חִזְקִיָּ֖הוּ בְּכִ֥י גָדֽוֹל׃ (כג) וַיְהִ֣י יְשַׁעְיָ֔הוּ לֹ֣א יָצָ֔א (העיר) חָצֵ֖ר הַתִּֽיכֹנָ֑ה וּדְבַר־ יְהוָ֔ה הָיָ֥ה אֵלָ֖יו לֵאמֹֽר׃ (כד) שׁ֣וּב וְאָמַרְתָּ֞ אֶל־ חִזְקִיָּ֣הוּ נְגִיד־ עַמִּ֗י כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵי֙ דָּוִ֣ד אָבִ֔יךָ שָׁמַ֙עְתִּי֙ אֶת־ תְּפִלָּתֶ֔ךָ רָאִ֖יתִי אֶת־ דִּמְעָתֶ֑ךָ הִנְנִי֙ רֹ֣פֶא לָ֔ךְ בַּיּוֹם֙ הַשְּׁלִישִׁ֔י תַּעֲלֶ֖ה בֵּ֥ית יְהוָֽה׃ (כה) וְהֹסַפְתִּ֣י עַל־ יָמֶ֗יךָ חֲמֵ֤שׁ עֶשְׂרֵה֙ שָׁנָ֔ה וּמִכַּ֤ף מֶֽלֶךְ־ אַשּׁוּר֙ אַצִּ֣ילְךָ֔ וְאֵ֖ת הָעִ֣יר הַזֹּ֑את וְגַנּוֹתִי֙ עַל־ הָעִ֣יר הַזֹּ֔את לְמַֽעֲנִ֔י וּלְמַ֖עַן דָּוִ֥ד עַבְדִּֽי׃ (כו) וַיֹּ֣אמֶר יְשַֽׁעְיָ֔הוּ קְח֖וּ דְּבֶ֣לֶת תְּאֵנִ֑ים וַיִּקְח֛וּ וַיָּשִׂ֥ימוּ עַֽל־ הַשְּׁחִ֖ין וַיֶּֽחִי׃
