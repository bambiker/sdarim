מלכים ב פרק יג לפי החלוקה לסדרים

פרק יג בספר מלכים ב (25 פסוקים) נמצא במלכים סדר כו (פסוקים כא–מב) ובמלכים סדר כז (פסוקים א–ג).

מלכים סדר כו

(א) וַיַּ֨עַשׂ יְהוֹאָ֧שׁ הַיָּשָׁ֛ר בְּעֵינֵ֥י יְהוָ֖ה כָּל־ יָמָ֑יו אֲשֶׁ֣ר הוֹרָ֔הוּ יְהוֹיָדָ֖ע הַכֹּהֵֽן׃ (ב) רַ֥ק הַבָּמ֖וֹת לֹא־ סָ֑רוּ ע֥וֹד הָעָ֛ם מְזַבְּחִ֥ים וּֽמְקַטְּרִ֖ים בַּבָּמֽוֹת׃ (ג) וַיֹּ֨אמֶר יְהוֹאָ֜שׁ אֶל־ הַכֹּהֲנִ֗ים כֹּל֩ כֶּ֨סֶף הַקֳּדָשִׁ֜ים אֲשֶׁר־ יוּבָ֤א בֵית־ יְהוָה֙ כֶּ֣סֶף עוֹבֵ֔ר אִ֕ישׁ כֶּ֥סֶף נַפְשׁ֖וֹת עֶרְכּ֑וֹ כָּל־ כֶּ֗סֶף אֲשֶׁ֤ר יַֽעֲלֶה֙ עַ֣ל לֶב־ אִ֔ישׁ לְהָבִ֖יא בֵּ֥ית יְהוָֽה׃ (ד) יִקְח֤וּ לָהֶם֙ הַכֹּ֣הֲנִ֔ים אִ֖ישׁ מֵאֵ֣ת מַכָּר֑וֹ וְהֵ֗ם יְחַזְּקוּ֙ אֶת־ בֶּ֣דֶק הַבַּ֔יִת לְכֹ֛ל אֲשֶׁר־ יִמָּצֵ֥א שָׁ֖ם בָּֽדֶק׃ (ה) וַיְהִ֗י בִּשְׁנַ֨ת עֶשְׂרִ֧ים וְשָׁלֹ֛שׁ שָׁנָ֖ה לַמֶּ֣לֶךְ יְהוֹאָ֑שׁ לֹֽא־ חִזְּק֥וּ הַכֹּהֲנִ֖ים אֶת־ בֶּ֥דֶק הַבָּֽיִת׃ (ו) וַיִּקְרָא֩ הַמֶּ֨לֶךְ יְהוֹאָ֜שׁ לִיהוֹיָדָ֤ע הַכֹּהֵן֙ וְלַכֹּ֣הֲנִ֔ים וַיֹּ֣אמֶר אֲלֵהֶ֔ם מַדּ֛וּעַ אֵינְכֶ֥ם מְחַזְּקִ֖ים אֶת־ בֶּ֣דֶק הַבָּ֑יִת וְעַתָּ֗ה אַל־ תִּקְחוּ־ כֶ֙סֶף֙ מֵאֵ֣ת מַכָּֽרֵיכֶ֔ם כִּֽי־ לְבֶ֥דֶק הַבַּ֖יִת תִּתְּנֻֽהוּ׃ (ז) וַיֵּאֹ֖תוּ הַכֹּֽהֲנִ֑ים לְבִלְתִּ֤י קְחַת־ כֶּ֙סֶף֙ מֵאֵ֣ת הָעָ֔ם וּלְבִלְתִּ֥י חַזֵּ֖ק אֶת־ בֶּ֥דֶק הַבָּֽיִת׃ (ח) וַיִּקַּ֞ח יְהוֹיָדָ֤ע הַכֹּהֵן֙ אֲר֣וֹן אֶחָ֔ד וַיִּקֹּ֥ב חֹ֖ר בְּדַלְתּ֑וֹ וַיִּתֵּ֣ן אֹתוֹ֩ אֵ֨צֶל הַמִּזְבֵּ֜חַ (בימין) מִיָּמִ֗ין בְּבֽוֹא־ אִישׁ֙ בֵּ֣ית יְהוָ֔ה וְנָֽתְנוּ־ שָׁ֤מָּה הַכֹּֽהֲנִים֙ שֹׁמְרֵ֣י הַסַּ֔ף אֶת־ כָּל־ הַכֶּ֖סֶף הַמּוּבָ֥א בֵית־ יְהוָֽה׃ (ט) וַֽיְהִי֙ כִּרְאוֹתָ֔ם כִּֽי־ רַ֥ב הַכֶּ֖סֶף בָּֽאָר֑וֹן וַיַּ֨עַל סֹפֵ֤ר הַמֶּ֙לֶךְ֙ וְהַכֹּהֵ֣ן הַגָּד֔וֹל וַיָּצֻ֙רוּ֙ וַיִּמְנ֔וּ אֶת־ הַכֶּ֖סֶף הַנִּמְצָ֥א בֵית־ יְהוָֽה׃ (י) וְנָתְנוּ֙ אֶת־ הַכֶּ֣סֶף הַֽמְתֻכָּ֔ן עַל־ (יד) יְדֵי֙ עֹשֵׂ֣י הַמְּלָאכָ֔ה (הפקדים) הַמֻּפְקָדִ֖ים בֵּ֣ית יְהוָ֑ה וַיּוֹצִיאֻ֜הוּ לְחָרָשֵׁ֤י הָעֵץ֙ וְלַבֹּנִ֔ים הָעֹשִׂ֖ים בֵּ֥ית יְהוָֽה׃ (יא) וְלַגֹּֽדְרִים֙ וּלְחֹצְבֵ֣י הָאֶ֔בֶן וְלִקְנ֤וֹת עֵצִים֙ וְאַבְנֵ֣י מַחְצֵ֔ב לְחַזֵּ֖ק אֶת־ בֶּ֣דֶק בֵּית־ יְהוָ֑ה וּלְכֹ֛ל אֲשֶׁר־ יֵצֵ֥א עַל־ הַבַּ֖יִת לְחָזְקָֽה׃ (יב) אַךְ֩ לֹ֨א יֵעָשֶׂ֜ה בֵּ֣ית יְהוָ֗ה סִפּ֥וֹת כֶּ֙סֶף֙ מְזַמְּר֤וֹת מִזְרָקוֹת֙ חֲצֹ֣צְר֔וֹת כָּל־ כְּלִ֥י זָהָ֖ב וּכְלִי־ כָ֑סֶף מִן־ הַכֶּ֖סֶף הַמּוּבָ֥א בֵית־ יְהוָֽה׃ (יג) כִּֽי־ לְעֹשֵׂ֥י הַמְּלָאכָ֖ה יִתְּנֻ֑הוּ וְחִזְּקוּ־ ב֖וֹ אֶת־ בֵּ֥ית יְהוָֽה׃ (יד) וְלֹ֧א יְחַשְּׁב֣וּ אֶת־ הָאֲנָשִׁ֗ים אֲשֶׁ֨ר יִתְּנ֤וּ אֶת־ הַכֶּ֙סֶף֙ עַל־ יָדָ֔ם לָתֵ֖ת לְעֹשֵׂ֣י הַמְּלָאכָ֑ה כִּ֥י בֶאֱמֻנָ֖ה הֵ֥ם עֹשִֽׂים׃ (טו) כֶּ֤סֶף אָשָׁם֙ וְכֶ֣סֶף חַטָּא֔וֹת לֹ֥א יוּבָ֖א בֵּ֣ית יְהוָ֑ה לַכֹּהֲנִ֖ים יִהְיֽוּ׃ (טז) אָ֣ז יַעֲלֶ֗ה חֲזָאֵל֙ מֶ֣לֶךְ אֲרָ֔ם וַיִּלָּ֥חֶם עַל־ גַּ֖ת וַֽיִּלְכְּדָ֑הּ וַיָּ֤שֶׂם חֲזָאֵל֙ פָּנָ֔יו לַעֲל֖וֹת עַל־ יְרוּשָׁלִָֽם׃ (יז) וַיִּקַּ֞ח יְהוֹאָ֣שׁ מֶֽלֶךְ־ יְהוּדָ֗ה אֵ֣ת כָּל־ הַקֳּדָשִׁ֡ים אֲשֶׁר־ הִקְדִּ֣ישׁוּ יְהוֹשָׁפָ֣ט וִיהוֹרָם֩ וַאֲחַזְיָ֨הוּ אֲבֹתָ֜יו מַלְכֵ֤י יְהוּדָה֙ וְאֶת־ קֳדָשָׁ֔יו וְאֵ֣ת כָּל־ הַזָּהָ֗ב הַנִּמְצָ֛א בְּאֹצְר֥וֹת בֵּית־ יְהוָ֖ה וּבֵ֣ית הַמֶּ֑לֶךְ וַיִּשְׁלַ֗ח לַֽחֲזָאֵל֙ מֶ֣לֶךְ אֲרָ֔ם וַיַּ֖עַל מֵעַ֥ל יְרוּשָׁלִָֽם׃ (יח) וְיֶ֛תֶר דִּבְרֵ֥י יוֹאָ֖שׁ וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (יט) וַיָּקֻ֥מוּ עֲבָדָ֖יו וַיִּקְשְׁרֽוּ־ קָ֑שֶׁר וַיַּכּוּ֙ אֶת־ יוֹאָ֔שׁ בֵּ֥ית מִלֹּ֖א הַיּוֹרֵ֥ד סִלָּֽא׃ (כ) וְיוֹזָבָ֣ד בֶּן־ שִׁ֠מְעָת וִיהוֹזָבָ֨ד בֶּן־ שֹׁמֵ֤ר ׀ עֲבָדָיו֙ הִכֻּ֣הוּ וַיָּמֹ֔ת וַיִּקְבְּר֥וּ אֹת֛וֹ עִם־ אֲבֹתָ֖יו בְּעִ֣יר דָּוִ֑ד וַיִּמְלֹ֛ךְ אֲמַצְיָ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ תחילת פרק יג (כא) בִּשְׁנַ֨ת עֶשְׂרִ֤ים וְשָׁלֹשׁ֙ שָׁנָ֔ה לְיוֹאָ֥שׁ בֶּן־ אֲחַזְיָ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ יְהוֹאָחָ֨ז בֶּן־ יֵה֤וּא עַל־ יִשְׂרָאֵל֙ בְּשֹׁ֣מְר֔וֹן שְׁבַ֥ע עֶשְׂרֵ֖ה שָׁנָֽה׃ (כב) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַ֠יֵּלֶךְ אַחַ֨ר חַטֹּ֜את יָרָבְעָ֧ם בֶּן־ נְבָ֛ט אֲשֶׁר־ הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל לֹא־ סָ֥ר מִמֶּֽנָּה׃ (כג) וַיִּֽחַר־ אַ֥ף יְהוָ֖ה בְּיִשְׂרָאֵ֑ל וַֽיִּתְּנֵ֞ם בְּיַ֣ד ׀ חֲזָאֵ֣ל מֶֽלֶךְ־ אֲרָ֗ם וּבְיַ֛ד בֶּן־ הֲדַ֥ד בֶּן־ חֲזָאֵ֖ל כָּל־ הַיָּמִֽים׃ (כד) וַיְחַ֥ל יְהוֹאָחָ֖ז אֶת־ פְּנֵ֣י יְהוָ֑ה וַיִּשְׁמַ֤ע אֵלָיו֙ יְהוָ֔ה כִּ֤י רָאָה֙ אֶת־ לַ֣חַץ יִשְׂרָאֵ֔ל כִּֽי־ לָחַ֥ץ אֹתָ֖ם מֶ֥לֶךְ אֲרָֽם׃ (כה) וַיִּתֵּ֨ן יְהוָ֤ה לְיִשְׂרָאֵל֙ מוֹשִׁ֔יעַ וַיֵּ֣צְא֔וּ מִתַּ֖חַת יַד־ אֲרָ֑ם וַיֵּשְׁב֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל בְּאָהֳלֵיהֶ֖ם כִּתְמ֥וֹל שִׁלְשֽׁוֹם׃ (כו) אַ֠ךְ לֹֽא־ סָ֜רוּ מֵחַטֹּ֧אות בֵּית־ יָרָבְעָ֛ם אֲשֶׁר־ (החטי) הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל בָּ֣הּ הָלָ֑ךְ וְגַם֙ הָאֲשֵׁרָ֔ה עָמְדָ֖ה בְּשֹׁמְרֽוֹן׃ (כז) כִּ֣י לֹא֩ הִשְׁאִ֨יר לִיהוֹאָחָ֜ז עָ֗ם כִּ֣י אִם־ חֲמִשִּׁ֤ים פָּֽרָשִׁים֙ וַעֲשָׂ֣רָה רֶ֔כֶב וַעֲשֶׂ֥רֶת אֲלָפִ֖ים רַגְלִ֑י כִּ֤י אִבְּדָם֙ מֶ֣לֶךְ אֲרָ֔ם וַיְשִׂמֵ֥ם כֶּֽעָפָ֖ר לָדֻֽשׁ׃ (כח) וְיֶ֨תֶר דִּבְרֵ֧י יְהוֹאָחָ֛ז וְכָל־ אֲשֶׁ֥ר עָשָׂ֖ה וּגְבוּרָת֑וֹ הֲלוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כט) וַיִּשְׁכַּ֤ב יְהֽוֹאָחָז֙ עִם־ אֲבֹתָ֔יו וַֽיִּקְבְּרֻ֖הוּ בְּשֹׁמְר֑וֹן וַיִּמְלֹ֛ךְ יוֹאָ֥שׁ בְּנ֖וֹ תַּחְתָּֽיו׃ (ל) בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָשֶׁ֙בַע֙ שָׁנָ֔ה לְיוֹאָ֖שׁ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ יְהוֹאָ֨שׁ בֶּן־ יְהוֹאָחָ֤ז עַל־ יִשְׂרָאֵל֙ בְּשֹׁ֣מְר֔וֹן שֵׁ֥שׁ עֶשְׂרֵ֖ה שָׁנָֽה׃ (לא) וַיַּֽעֲשֶׂ֥ה הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לֹ֣א סָ֗ר מִכָּל־ חַטֹּ֞אות יָרָבְעָ֧ם בֶּן־ נְבָ֛ט אֲשֶׁר־ הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל בָּ֥הּ הָלָֽךְ׃ (לב) וְיֶ֨תֶר דִּבְרֵ֤י יוֹאָשׁ֙ וְכָל־ אֲשֶׁ֣ר עָשָׂ֔ה וּגְב֣וּרָת֔וֹ אֲשֶׁ֣ר נִלְחַ֔ם עִ֖ם אֲמַצְיָ֣ה מֶֽלֶךְ־ יְהוּדָ֑ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (לג) וַיִּשְׁכַּ֤ב יוֹאָשׁ֙ עִם־ אֲבֹתָ֔יו וְיָרָבְעָ֖ם יָשַׁ֣ב עַל־ כִּסְא֑וֹ וַיִּקָּבֵ֤ר יוֹאָשׁ֙ בְּשֹׁ֣מְר֔וֹן עִ֖ם מַלְכֵ֥י יִשְׂרָאֵֽל׃ (לד) וֶֽאֱלִישָׁע֙ חָלָ֣ה אֶת־ חָלְי֔וֹ אֲשֶׁ֥ר יָמ֖וּת בּ֑וֹ וַיֵּ֨רֶד אֵלָ֜יו יוֹאָ֣שׁ מֶֽלֶךְ־ יִשְׂרָאֵ֗ל וַיֵּ֤בְךְּ עַל־ פָּנָיו֙ וַיֹּאמַ֔ר אָבִ֣י ׀ אָבִ֔י רֶ֥כֶב יִשְׂרָאֵ֖ל וּפָרָשָֽׁיו׃ (לה) וַיֹּ֤אמֶר לוֹ֙ אֱלִישָׁ֔ע קַ֖ח קֶ֣שֶׁת וְחִצִּ֑ים וַיִּקַּ֥ח אֵלָ֖יו קֶ֥שֶׁת וְחִצִּֽים׃ (לו) וַיֹּ֣אמֶר ׀ לְמֶ֣לֶךְ יִשְׂרָאֵ֗ל הַרְכֵּ֤ב יָֽדְךָ֙ עַל־ הַקֶּ֔שֶׁת וַיַּרְכֵּ֖ב יָד֑וֹ וַיָּ֧שֶׂם אֱלִישָׁ֛ע יָדָ֖יו עַל־ יְדֵ֥י הַמֶּֽלֶךְ׃ (לז) וַיֹּ֗אמֶר פְּתַ֧ח הַחַלּ֛וֹן קֵ֖דְמָה וַיִּפְתָּ֑ח וַיֹּ֤אמֶר אֱלִישָׁ֤ע יְרֵה֙ וַיּ֔וֹר וַיֹּ֗אמֶר חֵץ־ תְּשׁוּעָ֤ה לַֽיהוָה֙ וְחֵ֣ץ תְּשׁוּעָ֣ה בַֽאֲרָ֔ם וְהִכִּיתָ֧ אֶת־ אֲרָ֛ם בַּאֲפֵ֖ק עַד־ כַּלֵּֽה׃ (לח) וַיֹּ֛אמֶר קַ֥ח הַחִצִּ֖ים וַיִּקָּ֑ח וַיֹּ֤אמֶר לְמֶֽלֶךְ־ יִשְׂרָאֵל֙ הַךְ־ אַ֔רְצָה וַיַּ֥ךְ שָֽׁלֹשׁ־ פְּעָמִ֖ים וַֽיַּעֲמֹֽד׃ (לט) וַיִּקְצֹ֨ף עָלָ֜יו אִ֣ישׁ הָאֱלֹהִ֗ים וַיֹּ֙אמֶר֙ לְהַכּ֨וֹת חָמֵ֤שׁ אוֹ־ שֵׁשׁ֙ פְּעָמִ֔ים אָ֛ז הִכִּ֥יתָ אֶת־ אֲרָ֖ם עַד־ כַּלֵּ֑ה וְעַתָּ֕ה שָׁלֹ֥שׁ פְּעָמִ֖ים תַּכֶּ֥ה אֶת־ אֲרָֽם׃ (מ) וַיָּ֥מָת אֱלִישָׁ֖ע וַֽיִּקְבְּרֻ֑הוּ וּגְדוּדֵ֥י מוֹאָ֛ב יָבֹ֥אוּ בָאָ֖רֶץ בָּ֥א שָׁנָֽה׃ (מא) וַיְהִ֞י הֵ֣ם ׀ קֹבְרִ֣ים אִ֗ישׁ וְהִנֵּה֙ רָא֣וּ אֶֽת־ הַגְּד֔וּד וַיַּשְׁלִ֥יכוּ אֶת־ הָאִ֖ישׁ בְּקֶ֣בֶר אֱלִישָׁ֑ע וַיֵּ֜לֶךְ וַיִּגַּ֤ע הָאִישׁ֙ בְּעַצְמ֣וֹת אֱלִישָׁ֔ע וַיְחִ֖י וַיָּ֥קָם עַל־ רַגְלָֽיו׃ (מב) וַֽחֲזָאֵל֙ מֶ֣לֶךְ אֲרָ֔ם לָחַ֖ץ אֶת־ יִשְׂרָאֵ֑ל כֹּ֖ל יְמֵ֥י יְהוֹאָחָֽז׃

מלכים סדר כז

(א) וַיָּחָן֩ יְהוָ֨ה אֹתָ֤ם וַֽיְרַחֲמֵם֙ וַיִּ֣פֶן אֲלֵיהֶ֔ם לְמַ֣עַן בְּרִית֔וֹ אֶת־ אַבְרָהָ֖ם יִצְחָ֣ק וְיַֽעֲקֹ֑ב וְלֹ֤א אָבָה֙ הַשְׁחִיתָ֔ם וְלֹֽא־ הִשְׁלִיכָ֥ם מֵֽעַל־ פָּנָ֖יו עַד־ עָֽתָּה׃ (ב) וַיָּ֖מָת חֲזָאֵ֣ל מֶֽלֶךְ־ אֲרָ֑ם וַיִּמְלֹ֛ךְ בֶּן־ הֲדַ֥ד בְּנ֖וֹ תַּחְתָּֽיו׃ (ג) וַיָּ֜שָׁב יְהוֹאָ֣שׁ בֶּן־ יְהוֹאָחָ֗ז וַיִּקַּ֤ח אֶת־ הֶֽעָרִים֙ מִיַּד֙ בֶּן־ הֲדַ֣ד בֶּן־ חֲזָאֵ֔ל אֲשֶׁ֣ר לָקַ֗ח מִיַּ֛ד יְהוֹאָחָ֥ז אָבִ֖יו בַּמִּלְחָמָ֑ה שָׁלֹ֤שׁ פְּעָמִים֙ הִכָּ֣הוּ יוֹאָ֔שׁ וַיָּ֖שֶׁב אֶת־ עָרֵ֥י יִשְׂרָאֵֽל׃ סוף פרק יג (ד) בִּשְׁנַ֣ת שְׁתַּ֔יִם לְיוֹאָ֥שׁ בֶּן־ יוֹאָחָ֖ז מֶ֣לֶךְ יִשְׂרָאֵ֑ל מָלַ֛ךְ אֲמַצְיָ֥הוּ בֶן־ יוֹאָ֖שׁ מֶ֥לֶךְ יְהוּדָֽה׃ (ה) בֶּן־ עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָה֙ הָיָ֣ה בְמָלְכ֔וֹ וְעֶשְׂרִ֤ים וָתֵ֙שַׁע֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ (יהועדין) יְהֽוֹעַדָּ֖ן מִן־ יְרוּשָׁלִָֽם׃ (ו) וַיַּ֤עַשׂ הַיָּשָׁר֙ בְּעֵינֵ֣י יְהוָ֔ה רַ֕ק לֹ֖א כְּדָוִ֣ד אָבִ֑יו כְּכֹ֧ל אֲשֶׁר־ עָשָׂ֛ה יוֹאָ֥שׁ אָבִ֖יו עָשָֽׂה׃ (ז) רַ֥ק הַבָּמ֖וֹת לֹא־ סָ֑רוּ ע֥וֹד הָעָ֛ם מְזַבְּחִ֥ים וּֽמְקַטְּרִ֖ים בַּבָּמֽוֹת׃ (ח) וַיְהִ֕י כַּאֲשֶׁ֛ר חָזְקָ֥ה הַמַּמְלָכָ֖ה בְּיָד֑וֹ וַיַּךְ֙ אֶת־ עֲבָדָ֔יו הַמַּכִּ֖ים אֶת־ הַמֶּ֥לֶךְ אָבִֽיו׃ (ט) וְאֶת־ בְּנֵ֥י הַמַּכִּ֖ים לֹ֣א הֵמִ֑ית כַּכָּת֣וּב בְּסֵ֣פֶר תּֽוֹרַת־ מֹ֠שֶׁה אֲשֶׁר־ צִוָּ֨ה יְהוָ֜ה לֵאמֹ֗ר לֹא־ יוּמְת֨וּ אָב֤וֹת עַל־ בָּנִים֙ וּבָנִים֙ לֹא־ יוּמְת֣וּ עַל־ אָב֔וֹת כִּ֛י אִם־ אִ֥ישׁ בְּחֶטְא֖וֹ (ימות) יוּמָֽת׃ (י) הוּא־ הִכָּ֨ה אֶת־ אֱד֤וֹם בְּגֵיא־ (המלח) מֶ֙לַח֙ עֲשֶׂ֣רֶת אֲלָפִ֔ים וְתָפַ֥שׂ אֶת־ הַסֶּ֖לַע בַּמִּלְחָמָ֑ה וַיִּקְרָ֤א אֶת־ שְׁמָהּ֙ יָקְתְאֵ֔ל עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (יא) אָ֣ז שָׁלַ֤ח אֲמַצְיָה֙ מַלְאָכִ֔ים אֶל־ יְהוֹאָ֨שׁ בֶּן־ יְהוֹאָחָ֧ז בֶּן־ יֵה֛וּא מֶ֥לֶךְ יִשְׂרָאֵ֖ל לֵאמֹ֑ר לְכָ֖ה נִתְרָאֶ֥ה פָנִֽים׃ (יב) וַיִּשְׁלַ֞ח יְהוֹאָ֣שׁ מֶֽלֶךְ־ יִשְׂרָאֵ֗ל אֶל־ אֲמַצְיָ֣הוּ מֶֽלֶךְ־ יְהוּדָה֮ לֵאמֹר֒ הַח֜וֹחַ אֲשֶׁ֣ר בַּלְּבָנ֗וֹן שָׁ֠לַח אֶל־ הָאֶ֜רֶז אֲשֶׁ֤ר בַּלְּבָנוֹן֙ לֵאמֹ֔ר תְּנָֽה־ אֶת־ בִּתְּךָ֥ לִבְנִ֖י לְאִשָּׁ֑ה וַֽתַּעֲבֹ֞ר חַיַּ֤ת הַשָּׂדֶה֙ אֲשֶׁ֣ר בַּלְּבָנ֔וֹן וַתִּרְמֹ֖ס אֶת־ הַחֽוֹחַ׃ (יג) הַכֵּ֤ה הִכִּ֙יתָ֙ אֶת־ אֱד֔וֹם וּֽנְשָׂאֲךָ֖ לִבֶּ֑ךָ הִכָּבֵד֙ וְשֵׁ֣ב בְּבֵיתֶ֔ךָ וְלָ֤מָּה תִתְגָּרֶה֙ בְּרָעָ֔ה וְנָ֣פַלְתָּ֔ה אַתָּ֖ה וִיהוּדָ֥ה עִמָּֽךְ׃ (יד) וְלֹא־ שָׁמַ֣ע אֲמַצְיָ֔הוּ וַיַּ֨עַל יְהוֹאָ֤שׁ מֶֽלֶךְ־ יִשְׂרָאֵל֙ וַיִּתְרָא֣וּ פָנִ֔ים ה֖וּא וַאֲמַצְיָ֣הוּ מֶֽלֶךְ־ יְהוּדָ֑ה בְּבֵ֥ית שֶׁ֖מֶשׁ אֲשֶׁ֥ר לִיהוּדָֽה׃ (טו) וַיִּנָּ֥גֶף יְהוּדָ֖ה לִפְנֵ֣י יִשְׂרָאֵ֑ל וַיָּנֻ֖סוּ אִ֥ישׁ (לאהלו) לְאֹהָלָֽיו׃ (טז) וְאֵת֩ אֲמַצְיָ֨הוּ מֶֽלֶךְ־ יְהוּדָ֜ה בֶּן־ יְהוֹאָ֣שׁ בֶּן־ אֲחַזְיָ֗הוּ תָּפַ֛שׂ יְהוֹאָ֥שׁ מֶֽלֶךְ־ יִשְׂרָאֵ֖ל בְּבֵ֣ית שָׁ֑מֶשׁ (ויבאו) וַיָּבֹא֙ יְר֣וּשָׁלִַ֔ם וַיִּפְרֹץ֩ בְּחוֹמַ֨ת יְרוּשָׁלִַ֜ם בְּשַׁ֤עַר אֶפְרַ֙יִם֙ עַד־ שַׁ֣עַר הַפִּנָּ֔ה אַרְבַּ֥ע מֵא֖וֹת אַמָּֽה׃ (יז) וְלָקַ֣ח אֶת־ כָּל־ הַזָּהָב־ וְ֠הַכֶּסֶף וְאֵ֨ת כָּל־ הַכֵּלִ֜ים הַנִּמְצְאִ֣ים בֵּית־ יְהוָ֗ה וּבְאֹֽצְרוֹת֙ בֵּ֣ית הַמֶּ֔לֶךְ וְאֵ֖ת בְּנֵ֣י הַתַּֽעֲרֻב֑וֹת וַיָּ֖שָׁב שֹׁמְרֽוֹנָה׃ (יח) וְיֶתֶר֩ דִּבְרֵ֨י יְהוֹאָ֜שׁ אֲשֶׁ֤ר עָשָׂה֙ וּגְב֣וּרָת֔וֹ וַאֲשֶׁ֣ר נִלְחַ֔ם עִ֖ם אֲמַצְיָ֣הוּ מֶֽלֶךְ־ יְהוּדָ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (יט) וַיִּשְׁכַּ֤ב יְהוֹאָשׁ֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵר֙ בְּשֹׁ֣מְר֔וֹן עִ֖ם מַלְכֵ֣י יִשְׂרָאֵ֑ל וַיִּמְלֹ֛ךְ יָרָבְעָ֥ם בְּנ֖וֹ תַּחְתָּֽיו׃ (כ) וַיְחִ֨י אֲמַצְיָ֤הוּ בֶן־ יוֹאָשׁ֙ מֶ֣לֶךְ יְהוּדָ֔ה אַֽחֲרֵ֣י מ֔וֹת יְהוֹאָ֥שׁ בֶּן־ יְהֽוֹאָחָ֖ז מֶ֣לֶךְ יִשְׂרָאֵ֑ל חֲמֵ֥שׁ עֶשְׂרֵ֖ה שָׁנָֽה׃ (כא) וְיֶ֖תֶר דִּבְרֵ֣י אֲמַצְיָ֑הוּ הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (כב) וַיִּקְשְׁר֨וּ עָלָ֥יו קֶ֛שֶׁר בִּירוּשָׁלִַ֖ם וַיָּ֣נָס לָכִ֑ישָׁה וַיִּשְׁלְח֤וּ אַֽחֲרָיו֙ לָכִ֔ישָׁה וַיְמִתֻ֖הוּ שָֽׁם׃ (כג) וַיִּשְׂא֥וּ אֹת֖וֹ עַל־ הַסּוּסִ֑ים וַיִּקָּבֵ֧ר בִּירוּשָׁלִַ֛ם עִם־ אֲבֹתָ֖יו בְּעִ֥יר דָּוִֽד׃ (כד) וַיִּקְח֞וּ כָּל־ עַ֤ם יְהוּדָה֙ אֶת־ עֲזַרְיָ֔ה וְה֕וּא בֶּן־ שֵׁ֥שׁ עֶשְׂרֵ֖ה שָׁנָ֑ה וַיַּמְלִ֣כוּ אֹת֔וֹ תַּ֖חַת אָבִ֥יו אֲמַצְיָֽהוּ׃ (כה) ה֚וּא בָּנָ֣ה אֶת־ אֵילַ֔ת וַיְשִׁבֶ֖הָ לִֽיהוּדָ֑ה אַחֲרֵ֥י שְׁכַֽב־ הַמֶּ֖לֶךְ עִם־ אֲבֹתָֽיו׃ (כו) בִּשְׁנַת֙ חֲמֵשׁ־ עֶשְׂרֵ֣ה שָׁנָ֔ה לַאֲמַצְיָ֥הוּ בֶן־ יוֹאָ֖שׁ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ יָרָבְעָ֨ם בֶּן־ יוֹאָ֤שׁ מֶֽלֶךְ־ יִשְׂרָאֵל֙ בְּשֹׁ֣מְר֔וֹן אַרְבָּעִ֥ים וְאַחַ֖ת שָׁנָֽה׃ (כז) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לֹ֣א סָ֗ר מִכָּל־ חַטֹּאות֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (כח) ה֗וּא הֵשִׁיב֙ אֶת־ גְּב֣וּל יִשְׂרָאֵ֔ל מִלְּב֥וֹא חֲמָ֖ת עַד־ יָ֣ם הָעֲרָבָ֑ה כִּדְבַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֣ר דִּבֶּ֗ר בְּיַד־ עַבְדּ֞וֹ יוֹנָ֤ה בֶן־ אֲמִתַּי֙ הַנָּבִ֔יא אֲשֶׁ֖ר מִגַּ֥ת הַחֵֽפֶר׃ (כט) כִּי־ רָאָ֧ה יְהוָ֛ה אֶת־ עֳנִ֥י יִשְׂרָאֵ֖ל מֹרֶ֣ה מְאֹ֑ד וְאֶ֤פֶס עָצוּר֙ וְאֶ֣פֶס עָז֔וּב וְאֵ֥ין עֹזֵ֖ר לְיִשְׂרָאֵֽל׃ (ל) וְלֹא־ דִבֶּ֣ר יְהוָ֔ה לִמְחוֹת֙ אֶת־ שֵׁ֣ם יִשְׂרָאֵ֔ל מִתַּ֖חַת הַשָּׁמָ֑יִם וַיּ֣וֹשִׁיעֵ֔ם בְּיַ֖ד יָרָבְעָ֥ם בֶּן־ יוֹאָֽשׁ׃ (לא) וְיֶתֶר֩ דִּבְרֵ֨י יָרָבְעָ֜ם וְכָל־ אֲשֶׁ֤ר עָשָׂה֙ וּגְבוּרָת֣וֹ אֲשֶׁר־ נִלְחָ֔ם וַאֲשֶׁ֨ר הֵשִׁ֜יב אֶת־ דַּמֶּ֧שֶׂק וְאֶת־ חֲמָ֛ת לִיהוּדָ֖ה בְּיִשְׂרָאֵ֑ל הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (לב) וַיִּשְׁכַּ֤ב יָֽרָבְעָם֙ עִם־ אֲבֹתָ֔יו עִ֖ם מַלְכֵ֣י יִשְׂרָאֵ֑ל וַיִּמְלֹ֛ךְ זְכַרְיָ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ (לג) בִּשְׁנַ֨ת עֶשְׂרִ֤ים וָשֶׁ֙בַע֙ שָׁנָ֔ה לְיָרָבְעָ֖ם מֶ֣לֶךְ יִשְׂרָאֵ֑ל מָלַ֛ךְ עֲזַרְיָ֥ה בֶן־ אֲמַצְיָ֖ה מֶ֥לֶךְ יְהוּדָֽה׃ (לד) בֶּן־ שֵׁ֨שׁ עֶשְׂרֵ֤ה שָׁנָה֙ הָיָ֣ה בְמָלְכ֔וֹ וַחֲמִשִּׁ֤ים וּשְׁתַּ֙יִם֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ יְכָלְיָ֖הוּ מִירוּשָׁלִָֽם׃ (לה) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה אֲמַצְיָ֥הוּ אָבִֽיו׃ (לו) רַ֥ק הַבָּמ֖וֹת לֹא־ סָ֑רוּ ע֥וֹד הָעָ֛ם מְזַבְּחִ֥ים וּֽמְקַטְּרִ֖ים בַּבָּמֽוֹת׃ (לז) וַיְנַגַּ֨ע יְהוָ֜ה אֶת־ הַמֶּ֗לֶךְ וַיְהִ֤י מְצֹרָע֙ עַד־ י֣וֹם מֹת֔וֹ וַיֵּ֖שֶׁב בְּבֵ֣ית הַחָפְשִׁ֑ית וְיוֹתָ֤ם בֶּן־ הַמֶּ֙לֶךְ֙ עַל־ הַבַּ֔יִת שֹׁפֵ֖ט אֶת־ עַ֥ם הָאָֽרֶץ׃ (לח) וְיֶ֛תֶר דִּבְרֵ֥י עֲזַרְיָ֖הוּ וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃
