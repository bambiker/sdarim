דברי הימים א פרק טז לפי החלוקה לסדרים

פרק טז בספר דברי הימים א (43 פסוקים) נמצא בדברי הימים סדר ו (פסוקים סב–צו) ובדברי הימים סדר ז (פסוקים א–ח).

דברי הימים סדר ו

(א) וְגַ֣ם הַקְּרֽוֹבִים־ אֲ֠לֵיהֶם עַד־ יִשָׂשכָ֨ר וּזְבֻל֜וּן וְנַפְתָּלִ֗י מְבִיאִ֣ים לֶ֡חֶם בַּחֲמוֹרִ֣ים וּבַגְּמַלִּ֣ים וּבַפְּרָדִ֣ים ׀ וּֽבַבָּקָ֡ר מַאֲכָ֡ל קֶ֠מַח דְּבֵלִ֨ים וְצִמּוּקִ֧ים וְיַֽיִן־ וְשֶׁ֛מֶן וּבָקָ֥ר וְצֹ֖אן לָרֹ֑ב כִּ֥י שִׂמְחָ֖ה בְּיִשְׂרָאֵֽל׃ (ב) וַיִּוָּעַ֣ץ דָּוִ֗יד עִם־ שָׂרֵ֧י הָאֲלָפִ֛ים וְהַמֵּא֖וֹת לְכָל־ נָגִֽיד׃ (ג) וַיֹּ֨אמֶר דָּוִ֜יד לְכֹ֣ל ׀ קְהַ֣ל יִשְׂרָאֵ֗ל אִם־ עֲלֵיכֶ֨ם ט֜וֹב וּמִן־ יְהוָ֣ה אֱלֹהֵ֗ינוּ נִפְרְצָה֙ נִשְׁלְחָ֞ה עַל־ אַחֵ֣ינוּ הַנִּשְׁאָרִ֗ים בְּכֹל֙ אַרְצ֣וֹת יִשְׂרָאֵ֔ל וְעִמָּהֶ֛ם הַכֹּהֲנִ֥ים וְהַלְוִיִּ֖ם בְּעָרֵ֣י מִגְרְשֵׁיהֶ֑ם וְיִקָּבְצ֖וּ אֵלֵֽינוּ׃ (ד) וְנָסֵ֛בָּה אֶת־ אֲר֥וֹן אֱלֹהֵ֖ינוּ אֵלֵ֑ינוּ כִּי־ לֹ֥א דְרַשְׁנֻ֖הוּ בִּימֵ֥י שָׁאֽוּל׃ (ה) וַיֹּאמְר֥וּ כָֽל־ הַקָּהָ֖ל לַעֲשׂ֣וֹת כֵּ֑ן כִּֽי־ יָשַׁ֥ר הַדָּבָ֖ר בְּעֵינֵ֥י כָל־ הָעָֽם׃ (ו) וַיַּקְהֵ֤ל דָּוִיד֙ אֶת־ כָּל־ יִשְׂרָאֵ֔ל מִן־ שִׁיח֥וֹר מִצְרַ֖יִם וְעַד־ לְב֣וֹא חֲמָ֑ת לְהָבִיא֙ אֶת־ אֲר֣וֹן הָאֱלֹהִ֔ים מִקִּרְיַ֖ת יְעָרִֽים׃ (ז) וַיַּ֨עַל דָּוִ֤יד וְכָל־ יִשְׂרָאֵל֙ בַּעֲלָ֔תָה אֶל־ קִרְיַ֥ת יְעָרִ֖ים אֲשֶׁ֣ר לִיהוּדָ֑ה לְהַעֲל֣וֹת מִשָּׁ֗ם אֵת֩ אֲר֨וֹן הָאֱלֹהִ֧ים ׀ יְהוָ֛ה יוֹשֵׁ֥ב הַכְּרוּבִ֖ים אֲשֶׁר־ נִקְרָ֥א שֵֽׁם׃ (ח) וַיַּרְכִּ֜יבוּ אֶת־ אֲר֤וֹן הָאֱלֹהִים֙ עַל־ עֲגָלָ֣ה חֲדָשָׁ֔ה מִבֵּ֖ית אֲבִינָדָ֑ב וְעֻזָּ֣א וְאַחְי֔וֹ נֹהֲגִ֖ים בָּעֲגָלָֽה׃ (ט) וְדָוִ֣יד וְכָל־ יִשְׂרָאֵ֗ל מְשַׂחֲקִ֛ים לִפְנֵ֥י הָאֱלֹהִ֖ים בְּכָל־ עֹ֑ז וּבְשִׁירִ֤ים וּבְכִנֹּרוֹת֙ וּבִנְבָלִ֣ים וּבְתֻפִּ֔ים וּבִמְצִלְתַּ֖יִם וּבַחֲצֹצְרֽוֹת׃ (י) וַיָּבֹ֖אוּ עַד־ גֹּ֣רֶן כִּידֹ֑ן וַיִּשְׁלַ֨ח עֻזָּ֜א אֶת־ יָד֗וֹ לֶאֱחֹז֙ אֶת־ הָ֣אָר֔וֹן כִּ֥י שָֽׁמְט֖וּ הַבָּקָֽר׃ (יא) וַיִּֽחַר־ אַ֤ף יְהוָה֙ בְּעֻזָּ֔א וַיַּכֵּ֕הוּ עַ֛ל אֲשֶׁר־ שָׁלַ֥ח יָד֖וֹ עַל־ הָאָר֑וֹן וַיָּ֥מָת שָׁ֖ם לִפְנֵ֥י אֱלֹהִֽים׃ (יב) וַיִּ֣חַר לְדָוִ֔יד כִּֽי־ פָרַ֧ץ יְהוָ֛ה פֶּ֖רֶץ בְּעֻזָּ֑א וַיִּקְרָ֞א לַמָּק֤וֹם הַהוּא֙ פֶּ֣רֶץ עֻזָּ֔א עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (יג) וַיִּירָ֤א דָוִיד֙ אֶת־ הָ֣אֱלֹהִ֔ים בַּיּ֥וֹם הַה֖וּא לֵאמֹ֑ר הֵ֚יךְ אָבִ֣יא אֵלַ֔י אֵ֖ת אֲר֥וֹן הָאֱלֹהִֽים׃ (יד) וְלֹֽא־ הֵסִ֨יר דָּוִ֧יד אֶת־ הָאָר֛וֹן אֵלָ֖יו אֶל־ עִ֣יר דָּוִ֑יד וַיַּטֵּ֕הוּ אֶל־ בֵּ֥ית עֹבֵֽד־ אֱדֹ֖ם הַגִּתִּֽי׃ (טו) וַיֵּשֶׁב֩ אֲר֨וֹן הָאֱלֹהִ֜ים עִם־ בֵּ֨ית עֹבֵ֥ד אֱדֹ֛ם בְּבֵית֖וֹ שְׁלֹשָׁ֣ה חֳדָשִׁ֑ים וַיְבָ֧רֶךְ יְהוָ֛ה אֶת־ בֵּ֥ית עֹבֵֽד־ אֱדֹ֖ם וְאֶת־ כָּל־ אֲשֶׁר־ לֽוֹ׃ (טז) וַ֠יִּשְׁלַח (חירם) חוּרָ֨ם מֶֽלֶךְ־ צֹ֥ר מַלְאָכִים֮ אֶל־ דָּוִיד֒ וַעֲצֵ֣י אֲרָזִ֔ים וְחָרָשֵׁ֣י קִ֔יר וְחָרָשֵׁ֖י עֵצִ֑ים לִבְנ֥וֹת ל֖וֹ בָּֽיִת׃ (יז) וַיֵּ֣דַע דָּוִ֔יד כִּֽי־ הֱכִינ֧וֹ יְהוָ֛ה לְמֶ֖לֶךְ עַל־ יִשְׂרָאֵ֑ל כִּֽי־ נִשֵּׂ֤את לְמַ֙עְלָה֙ מַלְכוּת֔וֹ בַּעֲב֖וּר עַמּ֥וֹ יִשְׂרָאֵֽל׃ (יח) וַיִּקַּ֨ח דָּוִ֥יד ע֛וֹד נָשִׁ֖ים בִּירוּשָׁלִָ֑ם וַיּ֧וֹלֶד דָּוִ֛יד ע֖וֹד בָּנִ֥ים וּבָנֽוֹת׃ (יט) וְאֵ֙לֶּה֙ שְׁמ֣וֹת הַיְלוּדִ֔ים אֲשֶׁ֥ר הָיוּ־ ל֖וֹ בִּירוּשָׁלִָ֑ם שַׁמּ֣וּעַ וְשׁוֹבָ֔ב נָתָ֖ן וּשְׁלֹמֹֽה׃ (כ) וְיִבְחָ֥ר וֶאֱלִישׁ֖וּעַ וְאֶלְפָּֽלֶט׃ (כא) וְנֹ֥גַהּ וְנֶ֖פֶג וְיָפִֽיעַ׃ (כב) וֶאֱלִישָׁמָ֥ע וּבְעֶלְיָדָ֖ע וֶאֱלִיפָֽלֶט׃ (כג) וַיִּשְׁמְע֣וּ פְלִשְׁתִּ֗ים כִּי־ נִמְשַׁ֨ח דָּוִ֤יד לְמֶ֙לֶךְ֙ עַל־ כָּל־ יִשְׂרָאֵ֔ל וַיַּעֲל֥וּ כָל־ פְּלִשְׁתִּ֖ים לְבַקֵּ֣שׁ אֶת־ דָּוִ֑יד וַיִּשְׁמַ֣ע דָּוִ֔יד וַיֵּצֵ֖א לִפְנֵיהֶֽם׃ (כד) וּפְלִשְׁתִּ֖ים בָּ֑אוּ וַֽיִּפְשְׁט֖וּ בְּעֵ֥מֶק רְפָאִֽים׃ (כה) וַיִּשְׁאַ֨ל דָּוִ֤יד בֵּאלֹהִים֙ לֵאמֹ֔ר הַאֶֽעֱלֶה֙ עַל־ (פלשתיים) פְּלִשְׁתִּ֔ים וּנְתַתָּ֖ם בְּיָדִ֑י וַיֹּ֨אמֶר ל֤וֹ יְהוָה֙ עֲלֵ֔ה וּנְתַתִּ֖ים בְּיָדֶֽךָ׃ (כו) וַיַּעֲל֥וּ בְּבַֽעַל־ פְּרָצִים֮ וַיַּכֵּ֣ם שָׁ֣ם דָּוִיד֒ וַיֹּ֣אמֶר דָּוִ֔יד פָּרַ֨ץ הָֽאֱלֹהִ֧ים אֶת־ אוֹיְבַ֛י בְּיָדִ֖י כְּפֶ֣רֶץ מָ֑יִם עַל־ כֵּ֗ן קָֽרְא֛וּ שֵֽׁם־ הַמָּק֥וֹם הַה֖וּא בַּ֥עַל פְּרָצִֽים׃ (כז) וַיַּעַזְבוּ־ שָׁ֖ם אֶת־ אֱלֹֽהֵיהֶ֑ם וַיֹּ֣אמֶר דָּוִ֔יד וַיִּשָּׂרְפ֖וּ בָּאֵֽשׁ׃ (כח) וַיֹּסִ֤יפוּ עוֹד֙ פְּלִשְׁתִּ֔ים וַֽיִּפְשְׁט֖וּ בָּעֵֽמֶק׃ (כט) וַיִּשְׁאַ֨ל ע֤וֹד דָּוִיד֙ בֵּֽאלֹהִ֔ים וַיֹּ֤אמֶר לוֹ֙ הָֽאֱלֹהִ֔ים לֹ֥א תַֽעֲלֶ֖ה אַֽחֲרֵיהֶ֑ם הָסֵב֙ מֵֽעֲלֵיהֶ֔ם וּבָ֥אתָ לָהֶ֖ם מִמּ֥וּל הַבְּכָאִֽים׃ (ל) וִ֠יהִי כְּֽשָׁמְעֲךָ֞ אֶת־ ק֤וֹל הַצְּעָדָה֙ בְּרָאשֵׁ֣י הַבְּכָאִ֔ים אָ֖ז תֵּצֵ֣א בַמִּלְחָמָ֑ה כִּֽי־ יָצָ֤א הָֽאֱלֹהִים֙ לְפָנֶ֔יךָ לְהַכּ֖וֹת אֶת־ מַחֲנֵ֥ה פְלִשְׁתִּֽים׃ (לא) וַיַּ֣עַשׂ דָּוִ֔יד כַּֽאֲשֶׁ֥ר צִוָּ֖הוּ הָֽאֱלֹהִ֑ים וַיַּכּוּ֙ אֶת־ מַחֲנֵ֣ה פְלִשְׁתִּ֔ים מִגִּבְע֖וֹן וְעַד־ גָּֽזְרָה׃ (לב) וַיֵּצֵ֥א שֵׁם־ דָּוִ֖יד בְּכָל־ הָֽאֲרָצ֑וֹת וַֽיהוָ֛ה נָתַ֥ן אֶת־ פַּחְדּ֖וֹ עַל־ כָּל־ הַגּוֹיִֽם׃ (לג) וַיַּֽעַשׂ־ ל֥וֹ בָתִּ֖ים בְּעִ֣יר דָּוִ֑יד וַיָּ֤כֶן מָקוֹם֙ לַֽאֲר֣וֹן הָֽאֱלֹהִ֔ים וַיֶּט־ ל֖וֹ אֹֽהֶל׃ (לד) אָ֚ז אָמַ֣ר דָּוִ֔יד לֹ֤א לָשֵׂאת֙ אֶת־ אֲר֣וֹן הָֽאֱלֹהִ֔ים כִּ֖י אִם־ הַלְוִיִּ֑ם כִּי־ בָ֣ם ׀ בָּחַ֣ר יְהוָ֗ה לָשֵׂ֞את אֶת־ אֲר֧וֹן יְהוָ֛ה וּֽלְשָׁרְת֖וֹ עַד־ עוֹלָֽם׃ (לה) וַיַּקְהֵ֥ל דָּוִ֛יד אֶת־ כָּל־ יִשְׂרָאֵ֖ל אֶל־ יְרֽוּשָׁלִָ֑ם לְהַעֲלוֹת֙ אֶת־ אֲר֣וֹן יְהוָ֔ה אֶל־ מְקוֹמ֖וֹ אֲשֶׁר־ הֵכִ֥ין לֽוֹ׃ (לו) וַיֶּאֱסֹ֥ף דָּוִ֛יד אֶת־ בְּנֵ֥י אַֽהֲרֹ֖ן וְאֶת־ הַלְוִיִּֽם׃ (לז) לִבְנֵ֖י קְהָ֑ת אוּרִיאֵ֣ל הַשָּׂ֔ר וְאֶחָ֖יו מֵאָ֥ה וְעֶשְׂרִֽים׃ (לח) לִבְנֵ֖י מְרָרִ֑י עֲשָׂיָ֣ה הַשָּׂ֔ר וְאֶחָ֖יו מָאתַ֥יִם וְעֶשְׂרִֽים׃ (לט) לִבְנֵ֖י גֵּרְשׁ֑וֹם יוֹאֵ֣ל הַשָּׂ֔ר וְאֶחָ֖יו מֵאָ֥ה וּשְׁלֹשִֽׁים׃ (מ) לִבְנֵ֖י אֱלִֽיצָפָ֑ן שְׁמַֽעְיָ֥ה הַשָּׂ֖ר וְאֶחָ֥יו מָאתָֽיִם׃ (מא) לִבְנֵ֖י חֶבְר֑וֹן אֱלִיאֵ֥ל הַשָּׂ֖ר וְאֶחָ֥יו שְׁמוֹנִֽים׃ (מב) לִבְנֵ֖י עֻזִּיאֵ֑ל עַמִּינָדָ֣ב הַשָּׂ֔ר וְאֶחָ֕יו מֵאָ֖ה וּשְׁנֵ֥ים עָשָֽׂר׃ (מג) וַיִּקְרָ֣א דָוִ֔יד לְצָד֥וֹק וּלְאֶבְיָתָ֖ר הַכֹּֽהֲנִ֑ים וּמְבֹרָ֖ךְ לְעוֹלָֽם׃ (לו) וַיְהִי֙ אַחֲרֵי־ כֵ֔ן וַיַּ֥ךְ דָּוִ֛יד אֶת־ פְּלִשְׁתִּ֖ים וַיַּכְנִיעֵ֑ם וַיִּקַּ֛ח אֶת־ גַּ֥ת וּבְנֹתֶ֖יהָ מִיַּ֥ד פְּלִשְׁתִּֽים׃ (לז) וַיַּ֖ךְ אֶת־ מוֹאָ֑ב וַיִּהְי֤וּ מוֹאָב֙ עֲבָדִ֣ים לְדָוִ֔יד נֹשְׂאֵ֖י מִנְחָֽה׃ (לח) וַיַּ֥ךְ דָּוִ֛יד אֶת־ הֲדַדְעֶ֥זֶר מֶֽלֶךְ־ צוֹבָ֖ה חֲמָ֑תָה בְּלֶכְתּ֕וֹ לְהַצִּ֥יב יָד֖וֹ בִּֽנְהַר־ פְּרָֽת׃ (לט) וַיִּלְכֹּד֩ דָּוִ֨יד מִמֶּ֜נּוּ אֶ֣לֶף רֶ֗כֶב וְשִׁבְעַ֤ת אֲלָפִים֙ פָּֽרָשִׁ֔ים וְעֶשְׂרִ֥ים אֶ֖לֶף אִ֣ישׁ רַגְלִ֑י וַיְעַקֵּ֤ר דָּוִיד֙ אֶת־ כָּל־ הָרֶ֔כֶב וַיּוֹתֵ֥ר מִמֶּ֖נּוּ מֵ֥אָה רָֽכֶב׃ (מ) וַיָּבֹא֙ אֲרַ֣ם דַּרְמֶ֔שֶׂק לַעְז֕וֹר לַהֲדַדְעֶ֖זֶר מֶ֣לֶךְ צוֹבָ֑ה וַיַּ֤ךְ דָּוִיד֙ בַּאֲרָ֔ם עֶשְׂרִֽים־ וּשְׁנַ֥יִם אֶ֖לֶף אִֽישׁ׃ (מא) וַיָּ֤שֶׂם דָּוִיד֙ בַּאֲרַ֣ם דַּרְמֶ֔שֶׂק וַיְהִ֤י אֲרָם֙ לְדָוִ֔יד עֲבָדִ֖ים נֹשְׂאֵ֣י מִנְחָ֑ה וַיּ֤וֹשַׁע יְהוָה֙ לְדָוִ֔יד בְּכֹ֖ל אֲשֶׁ֥ר הָלָֽךְ׃ (מב) וַיִּקַּ֣ח דָּוִ֗יד אֵ֚ת שִׁלְטֵ֣י הַזָּהָ֔ב אֲשֶׁ֣ר הָי֔וּ עַ֖ל עַבְדֵ֣י הֲדַדְעָ֑זֶר וַיְבִיאֵ֖ם יְרוּשָׁלִָֽם׃ (מג) וּמִטִּבְחַ֤ת וּמִכּוּן֙ עָרֵ֣י הֲדַדְעֶ֔זֶר לָקַ֥ח דָּוִ֛יד נְחֹ֖שֶׁת רַבָּ֣ה מְאֹ֑ד בָּ֣הּ ׀ עָשָׂ֣ה שְׁלֹמֹ֗ה אֶת־ יָ֤ם הַנְּחֹ֙שֶׁת֙ וְאֶת־ הָֽעַמּוּדִ֔ים וְאֵ֖ת כְּלֵ֥י הַנְּחֹֽשֶׁת׃ (מד) וַיִּשְׁמַ֕ע תֹּ֖עוּ מֶ֣לֶךְ חֲמָ֑ת כִּ֚י הִכָּ֣ה דָוִ֔יד אֶת־ כָּל־ חֵ֖יל הֲדַדְעֶ֥זֶר מֶֽלֶךְ־ צוֹבָֽה׃ (מה) וַיִּשְׁלַ֣ח אֶת־ הֲדֽוֹרָם־ בְּנ֣וֹ אֶל־ הַמֶּֽלֶךְ־ דָּ֠וִיד (לשאול)־ לִשְׁאָל־ ל֨וֹ לְשָׁל֜וֹם וּֽלְבָרֲכ֗וֹ עַל֩ אֲשֶׁ֨ר נִלְחַ֤ם בַּהֲדַדְעֶ֙זֶר֙ וַיַּכֵּ֔הוּ כִּי־ אִ֛ישׁ מִלְחֲמ֥וֹת תֹּ֖עוּ הָיָ֣ה הֲדַדְעָ֑זֶר וְכֹ֗ל כְּלֵ֛י זָהָ֥ב וָכֶ֖סֶף וּנְחֹֽשֶׁת׃ (מו) גַּם־ אֹתָ֗ם הִקְדִּ֞ישׁ הַמֶּ֤לֶךְ דָּוִיד֙ לַיהוָ֔ה עִם־ הַכֶּ֙סֶף֙ וְהַזָּהָ֔ב אֲשֶׁ֥ר נָשָׂ֖א מִכָּל־ הַגּוֹיִ֑ם מֵֽאֱד֤וֹם וּמִמּוֹאָב֙ וּמִבְּנֵ֣י עַמּ֔וֹן וּמִפְּלִשְׁתִּ֖ים וּמֵֽעֲמָלֵֽק׃ (מז) וְאַבְשַׁ֣י בֶּן־ צְרוּיָ֗ה הִכָּ֤ה אֶת־ אֱדוֹם֙ בְּגֵ֣יא הַמֶּ֔לַח שְׁמוֹנָ֥ה עָשָׂ֖ר אָֽלֶף׃ (מח) וַיָּ֤שֶׂם בֶּֽאֱדוֹם֙ נְצִיבִ֔ים וַיִּהְי֥וּ כָל־ אֱד֖וֹם עֲבָדִ֣ים לְדָוִ֑יד וַיּ֤וֹשַׁע יְהוָה֙ אֶת־ דָּוִ֔יד בְּכֹ֖ל אֲשֶׁ֥ר הָלָֽךְ׃ (מט) וַיִּמְלֹ֥ךְ דָּוִ֖יד עַל־ כָּל־ יִשְׂרָאֵ֑ל וַיְהִ֗י עֹשֶׂ֛ה מִשְׁפָּ֥ט וּצְדָקָ֖ה לְכָל־ עַמּֽוֹ׃ (נ) וְיוֹאָ֥ב בֶּן־ צְרוּיָ֖ה עַל־ הַצָּבָ֑א וִיהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד מַזְכִּֽיר׃ (נא) וְצָד֧וֹק בֶּן־ אֲחִיט֛וּב וַאֲבִימֶ֥לֶךְ בֶּן־ אֶבְיָתָ֖ר כֹּהֲנִ֑ים וְשַׁוְשָׁ֖א סוֹפֵֽר׃ (נב) וּבְנָיָ֙הוּ֙ בֶּן־ יְה֣וֹיָדָ֔ע עַל־ הַכְּרֵתִ֖י וְהַפְּלֵתִ֑י וּבְנֵי־ דָוִ֥יד הָרִאשֹׁנִ֖ים לְיַ֥ד הַמֶּֽלֶךְ׃ (נג) וַיְהִי֙ אַחֲרֵי־ כֵ֔ן וַיָּ֕מָת נָחָ֖שׁ מֶ֣לֶךְ בְּנֵי־ עַמּ֑וֹן וַיִּמְלֹ֥ךְ בְּנ֖וֹ תַּחְתָּֽיו׃ (נד) וַיֹּ֨אמֶר דָּוִ֜יד אֶֽעֱשֶׂה־ חֶ֣סֶד ׀ עִם־ חָנ֣וּן בֶּן־ נָחָ֗שׁ כִּֽי־ עָשָׂ֨ה אָבִ֤יו עִמִּי֙ חֶ֔סֶד וַיִּשְׁלַ֥ח דָּוִ֛יד מַלְאָכִ֖ים לְנַחֲמ֣וֹ עַל־ אָבִ֑יו וַיָּבֹאוּ֩ עַבְדֵ֨י דָוִ֜יד אֶל־ אֶ֧רֶץ בְּנֵי־ עַמּ֛וֹן אֶל־ חָנ֖וּן לְנַחֲמֽוֹ׃ (נה) וַיֹּאמְרוּ֩ שָׂרֵ֨י בְנֵי־ עַמּ֜וֹן לְחָנ֗וּן הַֽמְכַבֵּ֨ד דָּוִ֤יד אֶת־ אָבִ֙יךָ֙ בְּעֵינֶ֔יךָ כִּֽי־ שָׁלַ֥ח לְךָ֖ מְנַחֲמִ֑ים הֲלֹ֡א בַּ֠עֲבוּר לַחְקֹ֨ר וְלַהֲפֹ֤ךְ וּלְרַגֵּל֙ הָאָ֔רֶץ בָּ֥אוּ עֲבָדָ֖יו אֵלֶֽיךָ׃ (נו) וַיִּקַּ֨ח חָנ֜וּן אֶת־ עַבְדֵ֤י דָוִיד֙ וַֽיְגַלְּחֵ֔ם וַיִּכְרֹ֧ת אֶת־ מַדְוֵיהֶ֛ם בַּחֵ֖צִי עַד־ הַמִּפְשָׂעָ֑ה וַֽיְשַׁלְּחֵֽם׃ (נז) וַיֵּלְכוּ֩ וַיַּגִּ֨ידוּ לְדָוִ֤יד עַל־ הָֽאֲנָשִׁים֙ וַיִּשְׁלַ֣ח לִקְרָאתָ֔ם כִּי־ הָי֥וּ הָאֲנָשִׁ֖ים נִכְלָמִ֣ים מְאֹ֑ד וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ שְׁב֣וּ בִֽירֵח֔וֹ עַ֛ד אֲשֶׁר־ יְצַמַּ֥ח זְקַנְכֶ֖ם וְשַׁבְתֶּֽם׃ (נח) וַיִּרְאוּ֙ בְּנֵ֣י עַמּ֔וֹן כִּ֥י הִֽתְבָּאֲשׁ֖וּ עִם־ דָּוִ֑יד וַיִּשְׁלַ֣ח חָ֠נוּן וּבְנֵ֨י עַמּ֜וֹן אֶ֣לֶף כִּכַּר־ כֶּ֗סֶף לִשְׂכֹּ֣ר לָ֠הֶם מִן־ אֲרַ֨ם נַהֲרַ֜יִם וּמִן־ אֲרַ֤ם מַעֲכָה֙ וּמִצּוֹבָ֔ה רֶ֖כֶב וּפָרָשִֽׁים׃ (נט) וַיִּשְׂכְּר֣וּ לָהֶ֡ם שְׁנַיִם֩ וּשְׁלֹשִׁ֨ים אֶ֜לֶף רֶ֗כֶב וְאֶת־ מֶ֤לֶךְ מַעֲכָה֙ וְאֶת־ עַמּ֔וֹ וַיָּבֹ֕אוּ וַֽיַּחֲנ֖וּ לִפְנֵ֣י מֵידְבָ֑א וּבְנֵ֣י עַמּ֗וֹן נֶאֶסְפוּ֙ מֵעָ֣רֵיהֶ֔ם וַיָּבֹ֖אוּ לַמִּלְחָמָֽה׃ (ס) וַיִּשְׁמַ֖ע דָּוִ֑יד וַיִּשְׁלַח֙ אֶת־ יוֹאָ֔ב וְאֵ֥ת כָּל־ צָבָ֖א הַגִּבּוֹרִֽים׃ (סא) וַיֵּצְאוּ֙ בְּנֵ֣י עַמּ֔וֹן וַיַּֽעַרְכ֥וּ מִלְחָמָ֖ה פֶּ֣תַח הָעִ֑יר וְהַמְּלָכִ֣ים אֲשֶׁר־ בָּ֔אוּ לְבַדָּ֖ם בַּשָּׂדֶֽה׃ (סב) וַיַּ֣רְא יוֹאָ֗ב כִּֽי־ הָיְתָ֧ה פְנֵי־ הַמִּלְחָמָ֛ה אֵלָ֖יו פָּנִ֣ים וְאָח֑וֹר וַיִּבְחַ֗ר מִכָּל־ בָּחוּר֙ בְּיִשְׂרָאֵ֔ל וַֽיַּעֲרֹ֖ךְ לִקְרַ֥את אֲרָֽם׃ (סג) וְאֵת֙ יֶ֣תֶר הָעָ֔ם נָתַ֕ן בְּיַ֖ד אַבְשַׁ֣י אָחִ֑יו וַיַּ֣עַרְכ֔וּ לִקְרַ֖את בְּנֵ֥י עַמּֽוֹן׃ (סד) וַיֹּ֗אמֶר אִם־ תֶּחֱזַ֤ק מִמֶּ֙נִּי֙ אֲרָ֔ם וְהָיִ֥יתָ לִּ֖י לִתְשׁוּעָ֑ה וְאִם־ בְּנֵ֥י עַמּ֛וֹן יֶֽחֶזְק֥וּ מִמְּךָ֖ וְהוֹשַׁעְתִּֽיךָ׃
