פרק יא בספר דברי הימים א (47 פסוקים) נמצא בדברי הימים סדר ד (פסוקים ס–סז) ובדברי הימים סדר ה (פסוקים א–לט).

דברי הימים סדר ד

(א) וַֽיִּהְי֣וּ בְנֵי־ א֠וּלָם אֲנָשִׁ֨ים גִּבֹּרֵי־ חַ֜יִל דֹּ֣רְכֵי קֶ֗שֶׁת וּמַרְבִּ֤ים בָּנִים֙ וּבְנֵ֣י בָנִ֔ים מֵאָ֖ה וַחֲמִשִּׁ֑ים כָּל־ אֵ֖לֶּה מִבְּנֵ֥י בִנְיָמִֽן׃ (ב) וְכָל־ יִשְׂרָאֵל֙ הִתְיַחְשׂ֔וּ וְהִנָּ֣ם כְּתוּבִ֔ים עַל־ סֵ֖פֶר מַלְכֵ֣י יִשְׂרָאֵ֑ל וִיהוּדָ֛ה הָגְל֥וּ לְבָבֶ֖ל בְּמַעֲלָֽם׃ (ג) וְהַיּוֹשְׁבִים֙ הָרִ֣אשֹׁנִ֔ים אֲשֶׁ֥ר בַּאֲחֻזָּתָ֖ם בְּעָרֵיהֶ֑ם יִשְׂרָאֵל֙ הַכֹּ֣הֲנִ֔ים הַלְוִיִּ֖ם וְהַנְּתִינִֽים׃ (ד) וּבִירוּשָׁלִַ֙ם֙ יָשְׁב֔וּ מִן־ בְּנֵ֥י יְהוּדָ֖ה וּמִן־ בְּנֵ֣י בִנְיָמִ֑ן וּמִן־ בְּנֵ֥י אֶפְרַ֖יִם וּמְנַשֶּֽׁה׃ (ה) עוּתַ֨י בֶּן־ עַמִּיה֤וּד בֶּן־ עָמְרִי֙ בֶּן־ אִמְרִ֣י בֶן־ (בנימן)־ בָּנִ֔י מִן־ בְּנֵי־ פֶ֖רֶץ בֶּן־ יְהוּדָֽה׃ (ו) וּמִן־ הַשִּׁ֣ילוֹנִ֔י עֲשָׂיָ֥ה הַבְּכ֖וֹר וּבָנָֽיו׃ (ז) וּמִן־ בְּנֵי־ זֶ֖רַח יְעוּאֵ֑ל וַאֲחֵיהֶ֖ם שֵׁשׁ־ מֵא֥וֹת וְתִשְׁעִֽים׃ (ח) וּמִן־ בְּנֵ֖י בִּנְיָמִ֑ן סַלּוּא֙ בֶּן־ מְשֻׁלָּ֔ם בֶּן־ הוֹדַוְיָ֖ה בֶּן־ הַסְּנֻאָֽה׃ (ט) וְיִבְנְיָה֙ בֶּן־ יְרֹחָ֔ם וְאֵלָ֥ה בֶן־ עֻזִּ֖י בֶּן־ מִכְרִ֑י וּמְשֻׁלָּם֙ בֶּן־ שְׁפַטְיָ֔ה בֶּן־ רְעוּאֵ֖ל בֶּן־ יִבְנִיָּֽה׃ (י) וַאֲחֵיהֶם֙ לְתֹ֣לְדוֹתָ֔ם תְּשַׁ֥ע מֵא֖וֹת וַחֲמִשִּׁ֣ים וְשִׁשָּׁ֑ה כָּל־ אֵ֣לֶּה אֲנָשִׁ֔ים רָאשֵׁ֥י אָב֖וֹת לְבֵ֥ית אֲבֹתֵיהֶֽם׃ (יא) וּמִן־ הַֽכֹּהֲנִ֑ים יְדַֽעְיָ֥ה וִיהוֹיָרִ֖יב וְיָכִֽין׃ (יב) וַעֲזַרְיָ֨ה בֶן־ חִלְקִיָּ֜ה בֶּן־ מְשֻׁלָּ֣ם בֶּן־ צָד֗וֹק בֶּן־ מְרָיוֹת֙ בֶּן־ אֲחִיט֔וּב נְגִ֖יד בֵּ֥ית הָאֱלֹהִֽים׃ (יג) וַעֲדָיָה֙ בֶּן־ יְרֹחָ֔ם בֶּן־ פַּשְׁח֖וּר בֶּן־ מַלְכִּיָּ֑ה וּמַעְשַׂ֨י בֶּן־ עֲדִיאֵ֧ל בֶּן־ יַחְזֵ֛רָה בֶּן־ מְשֻׁלָּ֥ם בֶּן־ מְשִׁלֵּמִ֖ית בֶּן־ אִמֵּֽר׃ (יד) וַאֲחֵיהֶ֗ם רָאשִׁים֙ לְבֵ֣ית אֲבוֹתָ֔ם אֶ֕לֶף וּשְׁבַ֥ע מֵא֖וֹת וְשִׁשִּׁ֑ים גִּבּ֣וֹרֵי חֵ֔יל מְלֶ֖אכֶת עֲבוֹדַ֥ת בֵּית־ הָאֱלֹהִֽים׃ (טו) וּמִֽן־ הַלְוִיִּ֑ם שְׁמַֽעְיָ֧ה בֶן־ חַשּׁ֛וּב בֶּן־ עַזְרִיקָ֥ם בֶּן־ חֲשַׁבְיָ֖ה מִן־ בְּנֵ֥י מְרָרִֽי׃ (טז) וּבַקְבַּקַּ֥ר חֶ֖רֶשׁ וְגָלָ֑ל וּמַתַּנְיָה֙ בֶּן־ מִיכָ֔א בֶּן־ זִכְרִ֖י בֶּן־ אָסָֽף׃ (יז) וְעֹבַדְיָה֙ בֶּֽן־ שְׁמַֽעְיָ֔ה בֶּן־ גָּלָ֖ל בֶּן־ יְדוּת֑וּן וּבֶרֶכְיָ֤ה בֶן־ אָסָא֙ בֶּן־ אֶלְקָנָ֔ה הַיּוֹשֵׁ֖ב בְּחַצְרֵ֥י נְטוֹפָתִֽי׃ (יח) וְהַשֹּׁעֲרִים֙ שַׁלּ֣וּם וְעַקּ֔וּב וְטַלְמֹ֖ן וַאֲחִימָ֑ן וַאֲחִיהֶ֥ם שַׁלּ֖וּם הָרֹֽאשׁ׃ (יט) וְֽעַד־ הֵ֔נָּה בְּשַׁ֥עַר הַמֶּ֖לֶךְ מִזְרָ֑חָה הֵ֚מָּה הַשֹּׁ֣עֲרִ֔ים לְמַחֲנ֖וֹת בְּנֵ֥י לֵוִֽי׃ (כ) וְשַׁלּ֣וּם בֶּן־ ק֠וֹרֵא בֶּן־ אֶבְיָסָ֨ף בֶּן־ קֹ֜רַח וְֽאֶחָ֧יו לְבֵית־ אָבִ֣יו הַקָּרְחִ֗ים עַ֚ל מְלֶ֣אכֶת הָעֲבוֹדָ֔ה שֹׁמְרֵ֥י הַסִּפִּ֖ים לָאֹ֑הֶל וַאֲבֹֽתֵיהֶם֙ עַל־ מַחֲנֵ֣ה יְהוָ֔ה שֹׁמְרֵ֖י הַמָּבֽוֹא׃ (כא) וּפִֽינְחָ֣ס בֶּן־ אֶלְעָזָ֗ר נָגִ֨יד הָיָ֧ה עֲלֵיהֶ֛ם לְפָנִ֖ים יְהוָ֥ה ׀ עִמּֽוֹ׃ (כב) זְכַרְיָה֙ בֶּ֣ן מְשֶֽׁלֶמְיָ֔ה שֹׁעֵ֥ר פֶּ֖תַח לְאֹ֥הֶל מוֹעֵֽד׃ (כג) כֻּלָּ֤ם הַבְּרוּרִים֙ לְשֹׁעֲרִ֣ים בַּסִּפִּ֔ים מָאתַ֖יִם וּשְׁנֵ֣ים עָשָׂ֑ר הֵ֤מָּה בְחַצְרֵיהֶם֙ הִתְיַחְשָׂ֔ם הֵ֣מָּה יִסַּ֥ד דָּוִ֛יד וּשְׁמוּאֵ֥ל הָרֹאֶ֖ה בֶּאֱמוּנָתָֽם׃ (כד) וְהֵ֨ם וּבְנֵיהֶ֜ם עַל־ הַשְּׁעָרִ֧ים לְבֵית־ יְהוָ֛ה לְבֵ֥ית־ הָאֹ֖הֶל לְמִשְׁמָרֽוֹת׃ (כה) לְאַרְבַּ֣ע רוּח֔וֹת יִהְי֖וּ הַשֹּׁעֲרִ֑ים מִזְרָ֥ח יָ֖מָּה צָפ֥וֹנָה וָנֶֽגְבָּה׃ (כו) וַאֲחֵיהֶ֨ם בְּחַצְרֵיהֶ֜ם לָב֨וֹא לְשִׁבְעַ֧ת הַיָּמִ֛ים מֵעֵ֥ת אֶל־ עֵ֖ת עִם־ אֵֽלֶּה׃ (כז) כִּ֣י בֶאֱמוּנָ֞ה הֵ֗מָּה אַרְבַּ֙עַת֙ גִּבֹּרֵ֣י הַשֹּׁעֲרִ֔ים הֵ֖ם הַלְוִיִּ֑ם וְהָיוּ֙ עַל־ הַלְּשָׁכ֔וֹת וְעַ֥ל הָאֹצְר֖וֹת בֵּ֥ית הָאֱלֹהִֽים׃ (כח) וּסְבִיב֥וֹת בֵּית־ הָאֱלֹהִ֖ים יָלִ֑ינוּ כִּֽי־ עֲלֵיהֶ֣ם מִשְׁמֶ֔רֶת וְהֵ֥ם עַל־ הַמַּפְתֵּ֖חַ וְלַבֹּ֥קֶר לַבֹּֽקֶר׃ (כט) וּמֵהֶ֖ם עַל־ כְּלֵ֣י הָעֲבוֹדָ֑ה כִּֽי־ בְמִסְפָּ֣ר יְבִיא֔וּם וּבְמִסְפָּ֖ר יוֹצִיאֽוּם׃ (ל) וּמֵהֶ֗ם מְמֻנִּים֙ עַל־ הַכֵּלִ֔ים וְעַ֖ל כָּל־ כְּלֵ֣י הַקֹּ֑דֶשׁ וְעַל־ הַסֹּ֙לֶת֙ וְהַיַּ֣יִן וְהַשֶּׁ֔מֶן וְהַלְּבוֹנָ֖ה וְהַבְּשָׂמִֽים׃ (לא) וּמִן־ בְּנֵי֙ הַכֹּ֣הֲנִ֔ים רֹקְחֵ֥י הַמִּרְקַ֖חַת לַבְּשָׂמִֽים׃ (לב) וּמַתִּתְיָה֙ מִן־ הַלְוִיִּ֔ם ה֥וּא הַבְּכ֖וֹר לְשַׁלֻּ֣ם הַקָּרְחִ֑י בֶּאֱמוּנָ֕ה עַ֖ל מַעֲשֵׂ֥ה הַחֲבִתִּֽים׃ (לג) וּמִן־ בְּנֵ֧י הַקְּהָתִ֛י מִן־ אֲחֵיהֶ֖ם עַל־ לֶ֣חֶם הַֽמַּעֲרָ֑כֶת לְהָכִ֖ין שַׁבַּ֥ת שַׁבָּֽת׃ (לד) וְאֵ֣לֶּה הַ֠מְשֹׁרְרִים רָאשֵׁ֨י אָב֧וֹת לַלְוִיִּ֛ם בַּלְּשָׁכֹ֖ת (פטירים) פְּטוּרִ֑ים כִּֽי־ יוֹמָ֥ם וָלַ֛יְלָה עֲלֵיהֶ֖ם בַּמְּלָאכָֽה׃ (לה) אֵלֶּה֩ רָאשֵׁ֨י הָאָב֧וֹת לַלְוִיִּ֛ם לְתֹלְדוֹתָ֖ם רָאשִׁ֑ים אֵ֖לֶּה יָשְׁב֥וּ בִירוּשָׁלִָֽם׃ (לו) וּבְגִבְע֛וֹן יָשְׁב֥וּ אֲבִֽי־ גִבְע֖וֹן (יעואל) יְעִיאֵ֑ל וְשֵׁ֥ם אִשְׁתּ֖וֹ מַעֲכָֽה׃ (לז) וּבְנ֥וֹ הַבְּכ֖וֹר עַבְדּ֑וֹן וְצ֣וּר וְקִ֔ישׁ וּבַ֥עַל וְנֵ֖ר וְנָדָֽב׃ (לח) וּגְד֣וֹר וְאַחְי֔וֹ וּזְכַרְיָ֖ה וּמִקְלֽוֹת׃ (לט) וּמִקְל֖וֹת הוֹלִ֣יד אֶת־ שִׁמְאָ֑ם וְאַף־ הֵ֗ם נֶ֧גֶד אֲחֵיהֶ֛ם יָשְׁב֥וּ בִירֽוּשָׁלִַ֖ם עִם־ אֲחֵיהֶֽם׃ (מ) וְנֵר֙ הוֹלִ֣יד אֶת־ קִ֔ישׁ וְקִ֖ישׁ הוֹלִ֣יד אֶת־ שָׁא֑וּל וְשָׁא֗וּל הוֹלִ֤יד אֶת־ יְהֽוֹנָתָן֙ וְאֶת־ מַלְכִּי־ שׁ֔וּעַ וְאֶת־ אֲבִינָדָ֖ב וְאֶת־ אֶשְׁבָּֽעַל׃ (מא) וּבֶן־ יְהוֹנָתָ֖ן מְרִ֣יב בָּ֑עַל וּמְרִי־ בַ֖עַל הוֹלִ֥יד אֶת־ מִיכָֽה׃ (מב) וּבְנֵ֖י מִיכָ֑ה פִּית֥וֹן וָמֶ֖לֶךְ וְתַחְרֵֽעַ׃ (מג) וְאָחָז֙ הוֹלִ֣יד אֶת־ יַעְרָ֔ה וְיַעְרָ֗ה הוֹלִ֛יד אֶת־ עָלֶ֥מֶת וְאֶת־ עַזְמָ֖וֶת וְאֶת־ זִמְרִ֑י וְזִמְרִ֖י הוֹלִ֥יד אֶת־ מוֹצָֽא׃ (מד) וּמוֹצָ֖א הוֹלִ֣יד אֶת־ בִּנְעָ֑א וּרְפָיָ֥ה בְנ֛וֹ אֶלְעָשָׂ֥ה בְנ֖וֹ אָצֵ֥ל בְּנֽוֹ׃ (מה) וּלְאָצֵל֮ שִׁשָּׁ֣ה בָנִים֒ וְאֵ֣לֶּה שְׁמוֹתָ֗ם עַזְרִיקָ֥ם ׀ בֹּ֙כְרוּ֙ וְיִשְׁמָעֵ֣אל וּשְׁעַרְיָ֔ה וְעֹבַדְיָ֖ה וְחָנָ֑ן אֵ֖לֶּה בְּנֵ֥י אָצַֽל׃ (מו) וּפְלִשְׁתִּ֖ים נִלְחֲמ֣וּ בְיִשְׂרָאֵ֑ל וַיָּ֑נָס אִֽישׁ־ יִשְׂרָאֵל֙ מִפְּנֵ֣י פְלִשְׁתִּ֔ים וַיִּפְּל֥וּ חֲלָלִ֖ים בְּהַ֥ר גִּלְבֹּֽעַ׃ (מז) וַיַּדְבְּק֣וּ אָ֑לֶף וְעִמָּהֶם֙ בְּצִנָּ֣ה וַחֲנִ֔ית שְׁלֹשִׁ֥ים וְשִׁבְעָ֖ה אָֽלֶף׃ (עה) וּמִן־ הַדָּנִי֙ עֹרְכֵ֣י מִלְחָמָ֔ה עֶשְׂרִֽים־ וּשְׁמוֹנָ֥ה אֶ֖לֶף וְשֵׁ֥שׁ מֵאֽוֹת׃ (עו) וּמֵאָשֵׁ֗ר יוֹצְאֵ֥י צָבָ֛א לַעֲרֹ֥ךְ מִלְחָמָ֖ה אַרְבָּעִ֥ים אָֽלֶף׃ (עז) וּמֵעֵ֣בֶר לַ֠יַּרְדֵּן מִן־ הָראוּבֵנִ֨י וְהַגָּדִ֜י וַחֲצִ֣י ׀ שֵׁ֣בֶט מְנַשֶּׁ֗ה בְּכֹל֙ כְּלֵי֙ צְבָ֣א מִלְחָמָ֔ה מֵאָ֥ה וְעֶשְׂרִ֖ים אָֽלֶף׃ (עח) כָּל־ אֵ֜לֶּה אַנְשֵׁ֣י מִלְחָמָה֮ עֹדְרֵ֣י מַעֲרָכָה֒ בְּלֵבָ֤ב שָׁלֵם֙ בָּ֣אוּ חֶבְר֔וֹנָה לְהַמְלִ֥יךְ אֶת־ דָּוִ֖יד עַל־ כָּל־ יִשְׂרָאֵ֑ל וְ֠גַם כָּל־ שֵׁרִ֧ית יִשְׂרָאֵ֛ל לֵ֥ב אֶחָ֖ד לְהַמְלִ֥יךְ אֶת־ דָּוִֽיד׃ (עט) וַיִּהְיוּ־ שָׁ֤ם עִם־ דָּוִיד֙ יָמִ֣ים שְׁלוֹשָׁ֔ה אֹכְלִ֖ים וְשׁוֹתִ֑ים כִּֽי־ הֵכִ֥ינוּ לָהֶ֖ם אֲחֵיהֶֽם׃
