דברי הימים סדר ב

(א) וַיִּקְרָ֣א יַ֠עְבֵּץ לֵאלֹהֵ֨י יִשְׂרָאֵ֜ל לֵאמֹ֗ר אִם־ בָּרֵ֨ךְ תְּבָרֲכֵ֜נִי וְהִרְבִּ֤יתָ אֶת־ גְּבוּלִי֙ וְהָיְתָ֤ה יָדְךָ֙ עִמִּ֔י וְעָשִׂ֥יתָ מֵּרָעָ֖ה לְבִלְתִּ֣י עָצְבִּ֑י וַיָּבֵ֥א אֱלֹהִ֖ים אֵ֥ת אֲשֶׁר־ שָׁאָֽל׃ (ב) וּכְל֥וּב אֲחִֽי־ שׁוּחָ֖ה הוֹלִ֣יד אֶת־ מְחִ֑יר ה֖וּא אֲבִ֥י אֶשְׁתּֽוֹן׃ (ג) וְאֶשְׁתּ֗וֹן הוֹלִ֞יד אֶת־ בֵּ֤ית רָפָא֙ וְאֶת־ פָּסֵ֔חַ וְאֶת־ תְּחִנָּ֖ה אֲבִ֣י עִ֣יר נָחָ֑שׁ אֵ֖לֶּה אַנְשֵׁ֥י רֵכָֽה׃ (ד) וּבְנֵ֣י קְנַ֔ז עָתְנִיאֵ֖ל וּשְׂרָיָ֑ה וּבְנֵ֥י עָתְנִיאֵ֖ל חֲתַֽת׃ (ה) וּמְעוֹנֹתַ֖י הוֹלִ֣יד אֶת־ עָפְרָ֑ה וּשְׂרָיָ֗ה הוֹלִ֤יד אֶת־ יוֹאָב֙ אֲבִי֙ גֵּ֣יא חֲרָשִׁ֔ים כִּ֥י חֲרָשִׁ֖ים הָיֽוּ׃ (ו) וּבְנֵי֙ כָּלֵ֣ב בֶּן־ יְפֻנֶּ֔ה עִ֥ירוּ אֵלָ֖ה וָנָ֑עַם וּבְנֵ֥י אֵלָ֖ה וּקְנַֽז׃ (ז) וּבְנֵ֖י יְהַלֶּלְאֵ֑ל זִ֣יף וְזִיפָ֔ה תִּירְיָ֖א וַאֲשַׂרְאֵֽל׃ (ח) וּבֶן־ עֶזְרָ֔ה יֶ֥תֶר וּמֶ֖רֶד וְעֵ֣פֶר וְיָל֑וֹן וַתַּ֙הַר֙ אֶת־ מִרְיָ֣ם וְאֶת־ שַׁמַּ֔י וְאֶת־ יִשְׁבָּ֖ח אֲבִ֥י אֶשְׁתְּמֹֽעַ׃ (ט) וְאִשְׁתּ֣וֹ הַיְהֻדִיָּ֗ה יָלְדָ֞ה אֶת־ יֶ֨רֶד אֲבִ֤י גְדוֹר֙ וְאֶת־ חֶ֙בֶר֙ אֲבִ֣י שׂוֹכ֔וֹ וְאֶת־ יְקֽוּתִיאֵ֖ל אֲבִ֣י זָנ֑וֹחַ וְאֵ֗לֶּה בְּנֵי֙ בִּתְיָ֣ה בַת־ פַּרְעֹ֔ה אֲשֶׁ֥ר לָקַ֖ח מָֽרֶד׃ (י) וּבְנֵי֙ אֵ֣שֶׁת הֽוֹדִיָּ֔ה אֲח֣וֹת נַ֔חַם אֲבִ֥י קְעִילָ֖ה הַגַּרְמִ֑י וְאֶשְׁתְּמֹ֖עַ הַמַּעֲכָתִֽי׃ (יא) וּבְנֵ֣י שִׁימ֔וֹן אַמְנ֣וֹן וְרִנָּ֔ה בֶּן־ חָנָ֖ן (ותולון) וְתִיל֑וֹן וּבְנֵ֣י יִשְׁעִ֔י זוֹחֵ֖ת וּבֶן־ זוֹחֵֽת׃ (יב) בְּנֵי֙ שֵׁלָ֣ה בֶן־ יְהוּדָ֔ה עֵ֚ר אֲבִ֣י לֵכָ֔ה וְלַעְדָּ֖ה אֲבִ֣י מָרֵשָׁ֑ה וּמִשְׁפְּח֛וֹת בֵּית־ עֲבֹדַ֥ת הַבֻּ֖ץ לְבֵ֥ית אַשְׁבֵּֽעַ׃ (יג) וְיוֹקִ֞ים וְאַנְשֵׁ֣י כֹזֵבָ֗א וְיוֹאָ֧שׁ וְשָׂרָ֛ף אֲשֶׁר־ בָּעֲל֥וּ לְמוֹאָ֖ב וְיָשֻׁ֣בִי לָ֑חֶם וְהַדְּבָרִ֖ים עַתִּיקִֽים׃ (יד) הֵ֚מָּה הַיּ֣וֹצְרִ֔ים וְיֹשְׁבֵ֥י נְטָעִ֖ים וּגְדֵרָ֑ה עִם־ הַמֶּ֥לֶךְ בִּמְלַאכְתּ֖וֹ יָ֥שְׁבוּ שָֽׁם׃ (טו) בְּנֵ֖י שִׁמְע֑וֹן נְמוּאֵ֣ל וְיָמִ֔ין יָרִ֖יב זֶ֥רַח שָׁאֽוּל׃ (טז) שַׁלֻּ֥ם בְּנ֛וֹ מִבְשָׂ֥ם בְּנ֖וֹ מִשְׁמָ֥ע בְּנֽוֹ׃ (יז) וּבְנֵ֖י מִשְׁמָ֑ע חַמּוּאֵ֥ל בְּנ֛וֹ זַכּ֥וּר בְּנ֖וֹ שִׁמְעִ֥י בְנֽוֹ׃ (יח) וּלְשִׁמְעִ֞י בָּנִ֨ים שִׁשָּׁ֤ה עָשָׂר֙ וּבָנ֣וֹת שֵׁ֔שׁ וּלְאֶחָ֕יו אֵ֖ין בָּנִ֣ים רַבִּ֑ים וְכֹל֙ מִשְׁפַּחְתָּ֔ם לֹ֥א הִרְבּ֖וּ עַד־ בְּנֵ֥י יְהוּדָֽה׃ (יט) וַיֵּֽשְׁב֛וּ בִּבְאֵֽר־ שֶׁ֥בַע וּמוֹלָדָ֖ה וַחֲצַ֥ר שׁוּעָֽל׃ (כ) וּבְבִלְהָ֥ה וּבְעֶ֖צֶם וּבְתוֹלָֽד׃ (כא) וּבִבְתוּאֵ֥ל וּבְחָרְמָ֖ה וּבְצִֽיקְלָֽג׃ (כב) וּבְבֵ֤ית מַרְכָּבוֹת֙ וּבַחֲצַ֣ר סוּסִ֔ים וּבְבֵ֥ית בִּרְאִ֖י וּֽבְשַׁעֲרָ֑יִם אֵ֥לֶּה עָרֵיהֶ֖ם עַד־ מְלֹ֥ךְ דָּוִֽיד׃ (כג) וְחַצְרֵיהֶם֙ עֵיטָ֣ם וָעַ֔יִן רִמּ֥וֹן וְתֹ֖כֶן וְעָשָׁ֑ן עָרִ֖ים חָמֵֽשׁ׃ (כד) וְכָל־ חַצְרֵיהֶ֗ם אֲשֶׁ֧ר סְבִיב֛וֹת הֶעָרִ֥ים הָאֵ֖לֶּה עַד־ בָּ֑עַל זֹ֚את מוֹשְׁבֹתָ֔ם וְהִתְיַחְשָׂ֖ם לָהֶֽם׃ (כה) וּמְשׁוֹבָ֣ב וְיַמְלֵ֔ךְ וְיוֹשָׁ֖ה בֶּן־ אֲמַצְיָֽה׃ (כו) וְיוֹאֵ֑ל וְיֵהוּא֙ בֶּן־ י֣וֹשִׁבְיָ֔ה בֶּן־ שְׂרָיָ֖ה בֶּן־ עֲשִׂיאֵֽל׃ (כז) וְאֶלְיוֹעֵינַ֡י וְֽיַעֲקֹ֡בָה וִ֠ישׁוֹחָיָה וַעֲשָׂיָ֧ה וַעֲדִיאֵ֛ל וִישִׂימִאֵ֖ל וּבְנָיָֽה׃ (כח) וְזִיזָ֨א בֶן־ שִׁפְעִ֧י בֶן־ אַלּ֛וֹן בֶּן־ יְדָיָ֥ה בֶן־ שִׁמְרִ֖י בֶּן־ שְׁמַֽעְיָֽה׃ (כט) אֵ֚לֶּה הַבָּאִ֣ים בְּשֵׁמ֔וֹת נְשִׂיאִ֖ים בְּמִשְׁפְּחוֹתָ֑ם וּבֵית֙ אֲב֣וֹתֵיהֶ֔ם פָּרְצ֖וּ לָרֽוֹב׃ (ל) וַיֵּלְכוּ֙ לִמְב֣וֹא גְדֹ֔ר עַ֖ד לְמִזְרַ֣ח הַגָּ֑יְא לְבַקֵּ֥שׁ מִרְעֶ֖ה לְצֹאנָֽם׃ (לא) וַֽיִּמְצְא֤וּ מִרְעֶה֙ שָׁמֵ֣ן וָט֔וֹב וְהָאָ֙רֶץ֙ רַחֲבַ֣ת יָדַ֔יִם וְשֹׁקֶ֖טֶת וּשְׁלֵוָ֑ה כִּ֣י מִן־ חָ֔ם הַיֹּשְׁבִ֥ים שָׁ֖ם לְפָנִֽים׃ (לב) וַיָּבֹ֡אוּ אֵלֶּה֩ הַכְּתוּבִ֨ים בְּשֵׁמ֜וֹת בִּימֵ֣י ׀ יְחִזְקִיָּ֣הוּ מֶֽלֶךְ־ יְהוּדָ֗ה וַיַּכּ֨וּ אֶת־ אָהֳלֵיהֶ֜ם וְאֶת־ (המעינים) הַמְּעוּנִ֨ים אֲשֶׁ֤ר נִמְצְאוּ־ שָׁ֙מָּה֙ וַיַּחֲרִימֻם֙ עַד־ הַיּ֣וֹם הַזֶּ֔ה וַיֵּשְׁב֖וּ תַּחְתֵּיהֶ֑ם כִּֽי־ מִרְעֶ֥ה לְצֹאנָ֖ם שָֽׁם׃ (לג) וּמֵהֶ֣ם ׀ מִן־ בְּנֵ֣י שִׁמְע֗וֹן הָלְכוּ֙ לְהַ֣ר שֵׂעִ֔יר אֲנָשִׁ֖ים חֲמֵ֣שׁ מֵא֑וֹת וּפְלַטְיָ֡ה וּ֠נְעַרְיָה וּרְפָיָ֧ה וְעֻזִּיאֵ֛ל בְּנֵ֥י יִשְׁעִ֖י בְּרֹאשָֽׁם׃ (לד) וַיַּכּ֕וּ אֶת־ שְׁאֵרִ֥ית הַפְּלֵטָ֖ה לַעֲמָלֵ֑ק וַיֵּ֣שְׁבוּ שָׁ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ תחילת פרק ה (לה) וּבְנֵ֨י רְאוּבֵ֥ן בְּכֽוֹר־ יִשְׂרָאֵל֮ כִּ֣י ה֣וּא הַבְּכוֹר֒ וּֽבְחַלְּלוֹ֙ יְצוּעֵ֣י אָבִ֔יו נִתְּנָה֙ בְּכֹ֣רָת֔וֹ לִבְנֵ֥י יוֹסֵ֖ף בֶּן־ יִשְׂרָאֵ֑ל וְלֹ֥א לְהִתְיַחֵ֖שׂ לַבְּכֹרָֽה׃ (לו) כִּ֤י יְהוּדָה֙ גָּבַ֣ר בְּאֶחָ֔יו וּלְנָגִ֖יד מִמֶּ֑נּוּ וְהַבְּכֹרָ֖ה לְיוֹסֵֽף׃ (לז) בְּנֵ֥י רְאוּבֵ֖ן בְּכ֣וֹר יִשְׂרָאֵ֑ל חֲנ֥וֹךְ וּפַלּ֖וּא חֶצְר֥וֹן וְכַרְמִֽי׃ (לח) בְּנֵ֖י יוֹאֵ֑ל שְׁמַֽעְיָ֥ה בְנ֛וֹ גּ֥וֹג בְּנ֖וֹ שִׁמְעִ֥י בְנֽוֹ׃ (לט) מִיכָ֥ה בְנ֛וֹ רְאָיָ֥ה בְנ֖וֹ בַּ֥עַל בְּנֽוֹ׃ (מ) בְּאֵרָ֣ה בְנ֔וֹ אֲשֶׁ֣ר הֶגְלָ֔ה תִּלְּגַ֥ת פִּלְנְאֶ֖סֶר מֶ֣לֶךְ אַשֻּׁ֑ר ה֥וּא נָשִׂ֖יא לָרֽאוּבֵנִֽי׃ (מא) וְאֶחָיו֙ לְמִשְׁפְּחֹתָ֔יו בְּהִתְיַחֵ֖שׂ לְתֹלְדוֹתָ֑ם הָרֹ֥אשׁ יְעִיאֵ֖ל וּזְכַרְיָֽהוּ׃ (מב) וּבֶ֙לַע֙ בֶּן־ עָזָ֔ז בֶּן־ שֶׁ֖מַע בֶּן־ יוֹאֵ֑ל ה֚וּא יוֹשֵׁ֣ב בַּעֲרֹעֵ֔ר וְעַד־ נְב֖וֹ וּבַ֥עַל מְעֽוֹן׃ (מג) וְלַמִּזְרָ֗ח יָשַׁב֙ עַד־ לְב֣וֹא מִדְבָּ֔רָה לְמִן־ הַנָּהָ֖ר פְּרָ֑ת כִּ֧י מִקְנֵיהֶ֛ם רָב֖וּ בְּאֶ֥רֶץ גִּלְעָֽד׃ (מד) וּבִימֵ֣י שָׁא֗וּל עָשׂ֤וּ מִלְחָמָה֙ עִם־ הַֽהַגְרִאִ֔ים וַֽיִּפְּל֖וּ בְּיָדָ֑ם וַיֵּשְׁבוּ֙ בְּאָ֣הֳלֵיהֶ֔ם עַֽל־ כָּל־ פְּנֵ֖י מִזְרָ֥ח לַגִּלְעָֽד׃ (מה) וּבְנֵי־ גָ֣ד לְנֶגְדָּ֗ם יָֽשְׁב֛וּ בְּאֶ֥רֶץ הַבָּשָׁ֖ן עַד־ סַלְכָֽה׃ (מו) יוֹאֵ֣ל הָרֹ֔אשׁ וְשָׁפָ֖ם הַמִּשְׁנֶ֑ה וְיַעְנַ֥י וְשָׁפָ֖ט בַּבָּשָֽׁן׃ (מז) וַאֲחֵיהֶ֞ם לְבֵ֣ית אֲבוֹתֵיהֶ֗ם מִֽיכָאֵ֡ל וּמְשֻׁלָּ֡ם וְ֠שֶׁבַע וְיוֹרַ֧י וְיַעְכָּ֛ן וְזִ֥יעַ וָעֵ֖בֶר שִׁבְעָֽה׃ (מח) אֵ֣לֶּה ׀ בְּנֵ֣י אֲבִיחַ֗יִל בֶּן־ חוּרִ֡י בֶּן־ יָ֠רוֹחַ בֶּן־ גִּלְעָ֧ד בֶּן־ מִיכָאֵ֛ל בֶּן־ יְשִׁישַׁ֥י בֶּן־ יַחְדּ֖וֹ בֶּן־ בּֽוּז׃ (מט) אֲחִי֙ בֶּן־ עַבְדִּיאֵ֣ל בֶּן־ גּוּנִ֔י רֹ֖אשׁ לְבֵ֥ית אֲבוֹתָֽם׃ (נ) וַיֵּֽשְׁב֛וּ בַּגִּלְעָ֥ד בַּבָּשָׁ֖ן וּבִבְנֹתֶ֑יהָ וּבְכָֽל־ מִגְרְשֵׁ֥י שָׁר֖וֹן עַל־ תּוֹצְאוֹתָֽם׃ (נא) כֻּלָּם֙ הִתְיַחְשׂ֔וּ בִּימֵ֖י יוֹתָ֣ם מֶֽלֶךְ־ יְהוּדָ֑ה וּבִימֵ֖י יָרָבְעָ֥ם מֶֽלֶךְ־ יִשְׂרָאֵֽל׃ (נב) בְּנֵֽי־ רְאוּבֵ֨ן וְגָדִ֜י וַחֲצִ֥י שֵֽׁבֶט־ מְנַשֶּׁה֮ מִן־ בְּנֵי־ חַיִל֒ אֲ֠נָשִׁים נֹשְׂאֵ֨י מָגֵ֤ן וְחֶ֙רֶב֙ וְדֹ֣רְכֵי קֶ֔שֶׁת וּלְמוּדֵ֖י מִלְחָמָ֑ה אַרְבָּעִ֨ים וְאַרְבָּעָ֥ה אֶ֛לֶף וּשְׁבַע־ מֵא֥וֹת וְשִׁשִּׁ֖ים יֹצְאֵ֥י צָבָֽא׃ (נג) וַיַּעֲשׂ֥וּ מִלְחָמָ֖ה עִם־ הַֽהַגְרִיאִ֑ים וִיט֥וּר וְנָפִ֖ישׁ וְנוֹדָֽב׃ (נד) וַיֵּעָזְר֣וּ עֲלֵיהֶ֔ם וַיִּנָּתְנ֤וּ בְיָדָם֙ הַֽהַגְרִיאִ֔ים וְכֹ֖ל שֶׁ֣עִמָּהֶ֑ם כִּ֠י לֵאלֹהִ֤ים זָעֲקוּ֙ בַּמִּלְחָמָ֔ה וְנַעְתּ֥וֹר לָהֶ֖ם כִּי־ בָ֥טְחוּ בֽוֹ׃ (נה) וַיִּשְׁבּ֣וּ מִקְנֵיהֶ֗ם גְּֽמַלֵּיהֶ֞ם חֲמִשִּׁ֥ים אֶ֙לֶף֙ וְצֹ֗אן מָאתַ֤יִם וַחֲמִשִּׁים֙ אֶ֔לֶף וַחֲמוֹרִ֖ים אַלְפָּ֑יִם וְנֶ֥פֶשׁ אָדָ֖ם מֵ֥אָה אָֽלֶף׃ (נו) כִּֽי־ חֲלָלִ֤ים רַבִּים֙ נָפָ֔לוּ כִּ֥י מֵהָאֱלֹהִ֖ים הַמִּלְחָמָ֑ה וַיֵּשְׁב֥וּ תַחְתֵּיהֶ֖ם עַד־ הַגֹּלָֽה׃ (נז) וּבְנֵ֗י חֲצִי֙ שֵׁ֣בֶט מְנַשֶּׁ֔ה יָשְׁב֖וּ בָּאָ֑רֶץ מִבָּשָׁ֞ן עַד־ בַּ֧עַל חֶרְמ֛וֹן וּשְׂנִ֥יר וְהַר־ חֶרְמ֖וֹן הֵ֥מָּה רָבֽוּ׃ (נח) וְאֵ֖לֶּה רָאשֵׁ֣י בֵית־ אֲבוֹתָ֑ם וְעֵ֡פֶר וְיִשְׁעִ֡י וֶאֱלִיאֵ֡ל וְ֠עַזְרִיאֵל וְיִרְמְיָ֨ה וְהוֹדַוְיָ֜ה וְיַחְדִּיאֵ֗ל אֲנָשִׁים֙ גִּבּ֣וֹרֵי חַ֔יִל אַנְשֵׁ֣י שֵׁמ֔וֹת רָאשִׁ֖ים לְבֵ֥ית אֲבוֹתָֽם׃ (נט) וַיִּֽמְעֲל֔וּ בֵּאלֹהֵ֖י אֲבוֹתֵיהֶ֑ם וַיִּזְנ֗וּ אַחֲרֵי֙ אֱלֹהֵ֣י עַמֵּי־ הָאָ֔רֶץ אֲשֶׁר־ הִשְׁמִ֥יד אֱלֹהִ֖ים מִפְּנֵיהֶֽם׃ (ס) וַיָּעַר֩ אֱלֹהֵ֨י יִשְׂרָאֵ֜ל אֶת־ ר֣וּחַ ׀ פּ֣וּל מֶֽלֶךְ־ אַשּׁ֗וּר וְאֶת־ ר֙וּחַ֙ תִּלְּגַ֤ת פִּלְנֶ֙סֶר֙ מֶ֣לֶךְ אַשּׁ֔וּר וַיַּגְלֵם֙ לָראוּבֵנִ֣י וְלַגָּדִ֔י וְלַחֲצִ֖י שֵׁ֣בֶט מְנַשֶּׁ֑ה וַ֠יְבִיאֵם לַחְלַ֨ח וְחָב֤וֹר וְהָרָא֙ וּנְהַ֣ר גּוֹזָ֔ן עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (סא) בְּנֵ֖י לֵוִ֑י גֵּרְשׁ֕וֹן קְהָ֖ת וּמְרָרִֽי׃ (סב) וּבְנֵ֖י קְהָ֑ת עַמְרָ֣ם יִצְהָ֔ר וְחֶבְר֖וֹן וְעֻזִּיאֵֽל׃ (סג) וּבְנֵ֣י עַמְרָ֔ם אַהֲרֹ֥ן וּמֹשֶׁ֖ה וּמִרְיָ֑ם וּבְנֵ֣י אַהֲרֹ֔ן נָדָב֙ וַאֲבִיה֔וּא אֶלְעָזָ֖ר וְאִיתָמָֽר׃ (סד) אֶלְעָזָר֙ הוֹלִ֣יד אֶת־ פִּֽינְחָ֔ס פִּֽינְחָ֖ס הֹלִ֥יד אֶת־ אֲבִישֽׁוּעַ׃ (סה) וַאֲבִישׁ֙וּעַ֙ הוֹלִ֣יד אֶת־ בֻּקִּ֔י וּבֻקִּ֖י הוֹלִ֥יד אֶת־ עֻזִּֽי׃ (סו) וְעֻזִּי֙ הוֹלִ֣יד אֶת־ זְרַֽחְיָ֔ה וּֽזְרַֽחְיָ֖ה הוֹלִ֥יד אֶת־ מְרָיֽוֹת׃ (סז) מְרָיוֹת֙ הוֹלִ֣יד אֶת־ אֲמַרְיָ֔ה וַאֲמַרְיָ֖ה הוֹלִ֥יד אֶת־ אֲחִיטֽוּב׃ (סח) וַאֲחִיטוּב֙ הוֹלִ֣יד אֶת־ צָד֔וֹק וְצָד֖וֹק הוֹלִ֥יד אֶת־ אֲחִימָֽעַץ׃ (סט) וַאֲחִימַ֙עַץ֙ הוֹלִ֣יד אֶת־ עֲזַרְיָ֔ה וַעֲזַרְיָ֖ה הוֹלִ֥יד אֶת־ יוֹחָנָֽן׃ (ע) וְיוֹחָנָ֖ן הוֹלִ֣יד אֶת־ עֲזַרְיָ֑ה ה֚וּא אֲשֶׁ֣ר כִּהֵ֔ן בַּבַּ֕יִת אֲשֶׁר־ בָּנָ֥ה שְׁלֹמֹ֖ה בִּירוּשָׁלִָֽם׃ (עא) וַיּ֥וֹלֶד עֲזַרְיָ֖ה אֶת־ אֲמַרְיָ֑ה וַאֲמַרְיָ֖ה הוֹלִ֥יד אֶת־ אֲחִיטֽוּב׃ (עב) וַאֲחִיטוּב֙ הוֹלִ֣יד אֶת־ צָד֔וֹק וְצָד֖וֹק הוֹלִ֥יד אֶת־ שַׁלּֽוּם׃ (עג) וְשַׁלּוּם֙ הוֹלִ֣יד אֶת־ חִלְקִיָּ֔ה וְחִלְקִיָּ֖ה הוֹלִ֥יד אֶת־ עֲזַרְיָֽה׃ (עד) וַעֲזַרְיָה֙ הוֹלִ֣יד אֶת־ שְׂרָיָ֔ה וּשְׂרָיָ֖ה הוֹלִ֥יד אֶת־ יְהוֹצָדָֽק׃ (עה) וִיהוֹצָדָ֣ק הָלַ֔ךְ בְּהַגְל֣וֹת יְהוָ֔ה אֶת־ יְהוּדָ֖ה וִירוּשָׁלִָ֑ם בְּיַ֖ד נְבֻכַדְנֶאצַּֽר׃ סוף פרק ה (עו) בְּנֵ֖י לֵוִ֑י גֵּרְשֹׁ֕ם קְהָ֖ת וּמְרָרִֽי׃ (עז) וְאֵ֛לֶּה שְׁמ֥וֹת בְּֽנֵי־ גֵרְשׁ֖וֹם לִבְנִ֥י וְשִׁמְעִֽי׃ (עח) וּבְנֵ֖י קְהָ֑ת עַמְרָ֣ם וְיִצְהָ֔ר וְחֶבְר֖וֹן וְעֻזִּיאֵֽל׃ (עט) בְּנֵ֥י מְרָרִ֖י מַחְלִ֣י וּמֻשִׁ֑י וְאֵ֛לֶּה מִשְׁפְּח֥וֹת הַלֵּוִ֖י לַאֲבוֹתֵיהֶֽם׃ (פ) לְֽגֵרְשׁ֑וֹם לִבְנִ֥י בְנ֛וֹ יַ֥חַת בְּנ֖וֹ זִמָּ֥ה בְנֽוֹ׃ (פא) יוֹאָ֤ח בְּנוֹ֙ עִדּ֣וֹ בְנ֔וֹ זֶ֥רַח בְּנ֖וֹ יְאָתְרַ֥י בְּנֽוֹ׃ (פב) בְּנֵ֖י קְהָ֑ת עַמִּינָדָ֣ב בְּנ֔וֹ קֹ֥רַח בְּנ֖וֹ אַסִּ֥יר בְּנֽוֹ׃ (פג) אֶלְקָנָ֥ה בְנ֛וֹ וְאֶבְיָסָ֥ף בְּנ֖וֹ וְאַסִּ֥יר בְּנֽוֹ׃ (פד) תַּ֤חַת בְּנוֹ֙ אוּרִיאֵ֣ל בְּנ֔וֹ עֻזִּיָּ֥ה בְנ֖וֹ וְשָׁא֥וּל בְּנֽוֹ׃ (פה) וּבְנֵי֙ אֶלְקָנָ֔ה עֲמָשַׂ֖י וַאֲחִימֽוֹת׃ (פו) אֶלְקָנָ֑ה (בנו) בְּנֵי֙ אֶלְקָנָ֔ה צוֹפַ֥י בְּנ֖וֹ וְנַ֥חַת בְּנֽוֹ׃ (פז) אֱלִיאָ֥ב בְּנ֛וֹ יְרֹחָ֥ם בְּנ֖וֹ אֶלְקָנָ֥ה בְנֽוֹ׃ (פח) וּבְנֵ֧י שְׁמוּאֵ֛ל הַבְּכֹ֥ר וַשְׁנִ֖י וַאֲבִיָּֽה׃ (פט) בְּנֵ֥י מְרָרִ֖י מַחְלִ֑י לִבְנִ֥י בְנ֛וֹ שִׁמְעִ֥י בְנ֖וֹ עֻזָּ֥ה בְנֽוֹ׃ (צ) שִׁמְעָ֥א בְנ֛וֹ חַגִּיָּ֥ה בְנ֖וֹ עֲשָׂיָ֥ה בְנֽוֹ׃ (צא) וְאֵ֗לֶּה אֲשֶׁ֨ר הֶעֱמִ֥יד דָּוִ֛יד עַל־ יְדֵי־ שִׁ֖יר בֵּ֣ית יְהוָ֑ה מִמְּנ֖וֹחַ הָאָרֽוֹן׃ (צב) וַיִּהְי֨וּ מְשָׁרְתִ֜ים לִפְנֵ֨י מִשְׁכַּ֤ן אֹֽהֶל־ מוֹעֵד֙ בַּשִּׁ֔יר עַד־ בְּנ֧וֹת שְׁלֹמֹ֛ה אֶת־ בֵּ֥ית יְהוָ֖ה בִּירוּשָׁלִָ֑ם וַיַּעַמְד֥וּ כְמִשְׁפָּטָ֖ם עַל־ עֲבוֹדָתָֽם׃ (צג) וְאֵ֥לֶּה הָעֹמְדִ֖ים וּבְנֵיהֶ֑ם מִבְּנֵי֙ הַקְּהָתִ֔י הֵימָן֙ הַמְשׁוֹרֵ֔ר בֶּן־ יוֹאֵ֖ל בֶּן־ שְׁמוּאֵֽל׃ (צד) בֶּן־ אֶלְקָנָה֙ בֶּן־ יְרֹחָ֔ם בֶּן־ אֱלִיאֵ֖ל בֶּן־ תּֽוֹחַ׃ (צה) בֶּן־ (ציף) צוּף֙ בֶּן־ אֶלְקָנָ֔ה בֶּן־ מַ֖חַת בֶּן־ עֲמָשָֽׂי׃ (צו) בֶּן־ אֶלְקָנָה֙ בֶּן־ יוֹאֵ֔ל בֶּן־ עֲזַרְיָ֖ה בֶּן־ צְפַנְיָֽה׃ (צז) בֶּן־ תַּ֙חַת֙ בֶּן־ אַסִּ֔יר בֶּן־ אֶבְיָסָ֖ף בֶּן־ קֹֽרַח׃ (צח) בֶּן־ יִצְהָ֣ר בֶּן־ קְהָ֔ת בֶּן־ לֵוִ֖י בֶּן־ יִשְׂרָאֵֽל׃ (צט) וְאָחִ֣יו אָסָ֔ף הָעֹמֵ֖ד עַל־ יְמִינ֑וֹ אָסָ֥ף בֶּן־ בֶּרֶכְיָ֖הוּ בֶּן־ שִׁמְעָֽא׃ (ק) בֶּן־ מִיכָאֵ֥ל בֶּן־ בַּעֲשֵׂיָ֖ה בֶּן־ מַלְכִּיָּֽה׃ (קא) בֶּן־ אֶתְנִ֥י בֶן־ זֶ֖רַח בֶּן־ עֲדָיָֽה׃ (קב) בֶּן־ אֵיתָ֥ן בֶּן־ זִמָּ֖ה בֶּן־ שִׁמְעִֽי׃ (קג) בֶּן־ יַ֥חַת בֶּן־ גֵּרְשֹׁ֖ם בֶּן־ לֵוִֽי׃ (קד) וּבְנֵ֧י מְרָרִ֛י אֲחֵיהֶ֖ם עַֽל־ הַשְּׂמֹ֑אול אֵיתָן֙ בֶּן־ קִישִׁ֔י בֶּן־ עַבְדִּ֖י בֶּן־ מַלּֽוּךְ׃ (קה) בֶּן־ חֲשַׁבְיָ֥ה בֶן־
