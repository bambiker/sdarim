דברי הימים סדר ח

(א) חֲזַ֤ק וְנִֽתְחַזְּקָה֙ בְּעַד־ עַמֵּ֔נוּ וּבְעַ֖ד עָרֵ֣י אֱלֹהֵ֑ינוּ וַֽיהוָ֔ה הַטּ֥וֹב בְּעֵינָ֖יו יַעֲשֶֽׂה׃ (ב) וַיִּגַּ֨שׁ יוֹאָ֜ב וְהָעָ֧ם אֲשֶׁר־ עִמּ֛וֹ לִפְנֵ֥י אֲרָ֖ם לַמִּלְחָמָ֑ה וַיָּנ֖וּסוּ מִפָּנָֽיו׃ (ג) וּבְנֵ֨י עַמּ֤וֹן רָאוּ֙ כִּי־ נָ֣ס אֲרָ֔ם וַיָּנ֣וּסוּ גַם־ הֵ֗ם מִפְּנֵי֙ אַבְשַׁ֣י אָחִ֔יו וַיָּבֹ֖אוּ הָעִ֑ירָה וַיָּבֹ֥א יוֹאָ֖ב יְרוּשָׁלִָֽם׃ (ד) וַיַּ֣רְא אֲרָ֗ם כִּ֣י נִגְּפוּ֮ לִפְנֵ֣י יִשְׂרָאֵל֒ וַֽיִּשְׁלְחוּ֙ מַלְאָכִ֔ים וַיּוֹצִ֣יאוּ אֶת־ אֲרָ֔ם אֲשֶׁ֖ר מֵעֵ֣בֶר הַנָּהָ֑ר וְשׁוֹפַ֛ךְ שַׂר־ צְבָ֥א הֲדַדְעֶ֖זֶר לִפְנֵיהֶֽם׃ (ה) וַיֻּגַּ֣ד לְדָוִ֗יד וַיֶּאֱסֹ֤ף אֶת־ כָּל־ יִשְׂרָאֵל֙ וַיַּעֲבֹ֣ר הַיַּרְדֵּ֔ן וַיָּבֹ֣א אֲלֵהֶ֔ם וַֽיַּעֲרֹ֖ךְ אֲלֵהֶ֑ם וַיַּעֲרֹ֨ךְ דָּוִ֜יד לִקְרַ֤את אֲרָם֙ מִלְחָמָ֔ה וַיִּֽלָּחֲמ֖וּ עִמּֽוֹ׃ (ו) וַיָּ֣נָס אֲרָם֮ מִלִּפְנֵ֣י יִשְׂרָאֵל֒ וַיַּהֲרֹ֨ג דָּוִ֜יד מֵאֲרָ֗ם שִׁבְעַ֤ת אֲלָפִים֙ רֶ֔כֶב וְאַרְבָּעִ֥ים אֶ֖לֶף אִ֣ישׁ רַגְלִ֑י וְאֵ֛ת שׁוֹפַ֥ךְ שַֽׂר־ הַצָּבָ֖א הֵמִֽית׃ (ז) וַיִּרְא֞וּ עַבְדֵ֣י הֲדַדְעֶ֗זֶר כִּ֤י נִגְּפוּ֙ לִפְנֵ֣י יִשְׂרָאֵ֔ל וַיַּשְׁלִ֥ימוּ עִם־ דָּוִ֖יד וַיַּֽעַבְדֻ֑הוּ וְלֹא־ אָבָ֣ה אֲרָ֔ם לְהוֹשִׁ֥יעַ אֶת־ בְּנֵי־ עַמּ֖וֹן עֽוֹד׃ (ח) וַיְהִ֡י לְעֵת֩ תְּשׁוּבַ֨ת הַשָּׁנָ֜ה לְעֵ֣ת ׀ צֵ֣את הַמְּלָכִ֗ים וַיִּנְהַ֣ג יוֹאָב֩ אֶת־ חֵ֨יל הַצָּבָ֜א וַיַּשְׁחֵ֣ת ׀ אֶת־ אֶ֣רֶץ בְּנֵֽי־ עַמּ֗וֹן וַיָּבֹא֙ וַיָּ֣צַר אֶת־ רַבָּ֔ה וְדָוִ֖יד יֹשֵׁ֣ב בִּירֽוּשָׁלִָ֑ם וַיַּ֥ךְ יוֹאָ֛ב אֶת־ רַבָּ֖ה וַיֶּֽהֶרְסֶֽהָ׃ (ט) וַיִּקַּ֣ח דָּוִ֣יד אֶת־ עֲטֶֽרֶת־ מַלְכָּם֩ מֵעַ֨ל רֹאשׁ֜וֹ וַֽיִּמְצָאָ֣הּ ׀ מִשְׁקַ֣ל כִּכַּר־ זָהָ֗ב וּבָהּ֙ אֶ֣בֶן יְקָרָ֔ה וַתְּהִ֖י עַל־ רֹ֣אשׁ דָּוִ֑יד וּשְׁלַ֥ל הָעִ֛יר הוֹצִ֖יא הַרְבֵּ֥ה מְאֹֽד׃ (י) וְאֶת־ הָעָ֨ם אֲשֶׁר־ בָּ֜הּ הוֹצִ֗יא וַיָּ֨שַׂר בַּמְּגֵרָ֜ה וּבַחֲרִיצֵ֤י הַבַּרְזֶל֙ וּבַמְּגֵר֔וֹת וְכֵן֙ יַעֲשֶׂ֣ה דָוִ֔יד לְכֹ֖ל עָרֵ֣י בְנֵי־ עַמּ֑וֹן וַיָּ֧שָׁב דָּוִ֛יד וְכָל־ הָעָ֖ם יְרוּשָׁלִָֽם׃ (יא) וַיְהִי֙ אַחֲרֵיכֵ֔ן וַתַּעֲמֹ֧ד מִלְחָמָ֛ה בְּגֶ֖זֶר עִם־ פְּלִשְׁתִּ֑ים אָ֣ז הִכָּ֞ה סִבְּכַ֣י הַחֻֽשָׁתִ֗י אֶת־ סִפַּ֛י מִילִדֵ֥י הָרְפָאִ֖ים וַיִּכָּנֵֽעוּ׃ (יב) וַתְּהִי־ ע֥וֹד מִלְחָמָ֖ה אֶת־ פְּלִשְׁתִּ֑ים וַיַּ֞ךְ אֶלְחָנָ֣ן בֶּן־ (יעור) יָעִ֗יר אֶת־ לַחְמִי֙ אֲחִי֙ גָּלְיָ֣ת הַגִּתִּ֔י וְעֵ֣ץ חֲנִית֔וֹ כִּמְנ֖וֹר אֹרְגִֽים׃ (יג) וַתְּהִי־ ע֥וֹד מִלְחָמָ֖ה בְּגַ֑ת וַיְהִ֣י ׀ אִ֣ישׁ מִדָּ֗ה וְאֶצְבְּעֹתָ֤יו שֵׁשׁ־ וָשֵׁשׁ֙ עֶשְׂרִ֣ים וְאַרְבַּ֔ע וְגַם־ ה֖וּא נוֹלַ֥ד לְהָרָפָֽא׃ (יד) וַיְחָרֵ֖ף אֶת־ יִשְׂרָאֵ֑ל וַיַּכֵּ֙הוּ֙ יְה֣וֹנָתָ֔ן בֶּן־ שִׁמְעָ֖א אֲחִ֥י דָוִֽיד׃ (טו) אֵ֛ל נוּלְּד֥וּ לְהָרָפָ֖א בְּגַ֑ת וַיִּפְּל֥וּ בְיַד־ דָּוִ֖יד וּבְיַד־ עֲבָדָֽיו׃ (טז) וַיַּֽעֲמֹ֥ד שָׂטָ֖ן עַל־ יִשְׂרָאֵ֑ל וַיָּ֙סֶת֙ אֶת־ דָּוִ֔יד לִמְנ֖וֹת אֶת־ יִשְׂרָאֵֽל׃ (יז) וַיֹּ֨אמֶר דָּוִ֤יד אֶל־ יוֹאָב֙ וְאֶל־ שָׂרֵ֣י הָעָ֔ם לְכ֗וּ סִפְרוּ֙ אֶת־ יִשְׂרָאֵ֔ל מִבְּאֵ֥ר שֶׁ֖בַע וְעַד־ דָּ֑ן וְהָבִ֣יאוּ אֵלַ֔י וְאֵדְעָ֖ה אֶת־ מִסְפָּרָֽם׃ (יח) וַיֹּ֣אמֶר יוֹאָ֗ב יוֹסֵף֩ יְהוָ֨ה עַל־ עַמּ֤וֹ ׀ כָּהֵם֙ מֵאָ֣ה פְעָמִ֔ים הֲלֹא֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ כֻּלָּ֥ם לַאדֹנִ֖י לַעֲבָדִ֑ים לָ֣מָּה יְבַקֵּ֥שׁ זֹאת֙ אֲדֹנִ֔י לָ֛מָּה יִהְיֶ֥ה לְאַשְׁמָ֖ה לְיִשְׂרָאֵֽל׃ (יט) וּדְבַר־ הַמֶּ֖לֶךְ חָזַ֣ק עַל־ יוֹאָ֑ב וַיֵּצֵ֣א יוֹאָ֗ב וַיִּתְהַלֵּךְ֙ בְּכָל־ יִשְׂרָאֵ֔ל וַיָּבֹ֖א יְרוּשָׁלִָֽם׃ (כ) וַיִּתֵּ֥ן יוֹאָ֛ב אֶת־ מִסְפַּ֥ר מִפְקַד־ הָעָ֖ם אֶל־ דָּוִ֑יד וַיְהִ֣י כָֽל־ יִשְׂרָאֵ֡ל אֶ֣לֶף אֲלָפִים֩ וּמֵאָ֨ה אֶ֤לֶף אִישׁ֙ שֹׁ֣לֵֽף חֶ֔רֶב וִֽיהוּדָ֕ה אַרְבַּע֩ מֵא֨וֹת וְשִׁבְעִ֥ים אֶ֛לֶף אִ֖ישׁ שֹׁ֥לֵֽף חָֽרֶב׃ (כא) וְלֵוִי֙ וּבִנְיָמִ֔ן לֹ֥א פָקַ֖ד בְּתוֹכָ֑ם כִּֽי־ נִתְעַ֥ב דְּבַר־ הַמֶּ֖לֶךְ אֶת־ יוֹאָֽב׃ (כב) וַיֵּ֙רַע֙ בְּעֵינֵ֣י הָאֱלֹהִ֔ים עַל־ הַדָּבָ֖ר הַזֶּ֑ה וַיַּ֖ךְ אֶת־ יִשְׂרָאֵֽל׃ (כג) וַיֹּ֤אמֶר דָּוִיד֙ אֶל־ הָ֣אֱלֹהִ֔ים חָטָ֣אתִֽי מְאֹ֔ד אֲשֶׁ֥ר עָשִׂ֖יתִי אֶת־ הַדָּבָ֣ר הַזֶּ֑ה וְעַתָּ֗ה הַֽעֲבֶר־ נָא֙ אֶת־ עֲו֣וֹן עַבְדְּךָ֔ כִּ֥י נִסְכַּ֖לְתִּי מְאֹֽד׃ (כד) וַיְדַבֵּ֤ר יְהוָה֙ אֶל־ גָּ֔ד חֹזֵ֥ה דָוִ֖יד לֵאמֹֽר׃ (כה) לֵךְ֩ וְדִבַּרְתָּ֨ אֶל־ דָּוִ֜יד לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה שָׁל֕וֹשׁ אֲנִ֖י נֹטֶ֣ה עָלֶ֑יךָ בְּחַר־ לְךָ֛ אַחַ֥ת מֵהֵ֖נָּה וְאֶֽעֱשֶׂה־ לָּֽךְ׃ (כו) וַיָּ֥בֹא גָ֖ד אֶל־ דָּוִ֑יד וַיֹּ֥אמֶר ל֛וֹ כֹּֽה־ אָמַ֥ר יְהוָ֖ה קַבֶּל־ לָֽךְ׃ (כז) אִם־ שָׁל֨וֹשׁ שָׁנִ֜ים רָעָ֗ב וְאִם־ שְׁלֹשָׁ֨ה חֳדָשִׁ֜ים נִסְפֶּ֥ה מִפְּנֵי־ צָרֶיךָ֮ וְחֶ֣רֶב אוֹיְבֶ֣ךָ ׀ לְמַשֶּׂגֶת֒ וְאִם־ שְׁלֹ֣שֶׁת יָ֠מִים חֶ֣רֶב יְהוָ֤ה וְדֶ֙בֶר֙ בָּאָ֔רֶץ וּמַלְאַ֣ךְ יְהוָ֔ה מַשְׁחִ֖ית בְּכָל־ גְּב֣וּל יִשְׂרָאֵ֑ל וְעַתָּ֣ה רְאֵ֔ה מָֽה־ אָשִׁ֥יב אֶת־ שֹׁלְחִ֖י דָּבָֽר׃ (כח) וַיֹּ֧אמֶר דָּוִ֛יד אֶל־ גָּ֖ד צַר־ לִ֣י מְאֹ֑ד אֶפְּלָה־ נָּ֣א בְיַד־ יְהוָ֗ה כִּֽי־ רַבִּ֤ים רַחֲמָיו֙ מְאֹ֔ד וּבְיַד־ אָדָ֖ם אַל־ אֶפֹּֽל׃ (כט) וַיִּתֵּ֧ן יְהוָ֛ה דֶּ֖בֶר בְּיִשְׂרָאֵ֑ל וַיִּפֹּל֙ מִיִּשְׂרָאֵ֔ל שִׁבְעִ֥ים אֶ֖לֶף אִֽישׁ׃ (ל) וַיִּשְׁלַח֩ הָאֱלֹהִ֨ים ׀ מַלְאָ֥ךְ ׀ לִֽירוּשָׁלִַם֮ לְהַשְׁחִיתָהּ֒ וּכְהַשְׁחִ֗ית רָאָ֤ה יְהוָה֙ וַיִּנָּ֣חֶם עַל־ הָֽרָעָ֔ה וַיֹּ֨אמֶר לַמַּלְאָ֤ךְ הַמַּשְׁחִית֙ רַ֔ב עַתָּ֖ה הֶ֣רֶף יָדֶ֑ךָ וּמַלְאַ֤ךְ יְהוָה֙ עֹמֵ֔ד עִם־ גֹּ֖רֶן אָרְנָ֥ן הַיְבוּסִֽי׃ (לא) וַיִּשָּׂ֨א דָוִ֜יד אֶת־ עֵינָ֗יו וַיַּ֞רְא אֶת־ מַלְאַ֤ךְ יְהוָה֙ עֹמֵ֗ד בֵּ֤ין הָאָ֙רֶץ֙ וּבֵ֣ין הַשָּׁמַ֔יִם וְחַרְבּ֤וֹ שְׁלוּפָה֙ בְּיָד֔וֹ נְטוּיָ֖ה עַל־ יְרוּשָׁלִָ֑ם וַיִּפֹּ֨ל דָּוִ֧יד וְהַזְּקֵנִ֛ים מְכֻסִּ֥ים בַּשַּׂקִּ֖ים עַל־ פְּנֵיהֶֽם׃ (לב) וַיֹּ֣אמֶר דָּוִ֣יד אֶֽל־ הָאֱלֹהִ֡ים הֲלֹא֩ אֲנִ֨י לַכֹּהֲנִ֖ים וְלַלְוִיִּ֑ם אָב֣וֹת הָרֹ֔אשׁ לְעֻמַּ֖ת אָחִ֥יו הַקָּטָֽן׃ (סה) וַיַּבְדֵּ֣ל דָּוִיד֩ וְשָׂרֵ֨י הַצָּבָ֜א לַעֲבֹדָ֗ה לִבְנֵ֤י אָסָף֙ וְהֵימָ֣ן וִֽידוּת֔וּן (הנביאים) הַֽנִּבְּאִ֛ים בְּכִנֹּר֥וֹת בִּנְבָלִ֖ים וּבִמְצִלְתָּ֑יִם וַֽיְהִי֙ מִסְפָּרָ֔ם אַנְשֵׁ֥י מְלָאכָ֖ה לַעֲבֹדָתָֽם׃ (סו) לִבְנֵ֣י אָסָ֗ף זַכּ֧וּר וְיוֹסֵ֛ף וּנְתַנְיָ֥ה וַאֲשַׂרְאֵ֖לָה בְּנֵ֣י אָסָ֑ף עַ֚ל יַד־ אָסָ֔ף הַנִּבָּ֖א עַל־ יְדֵ֥י הַמֶּֽלֶךְ׃ (סז) לִידוּת֑וּן בְּנֵ֣י יְדוּת֡וּן גְּדַלְיָ֡הוּ וּצְרִ֡י וִֽ֠ישַׁעְיָהוּ חֲשַׁבְיָ֨הוּ וּמַתִּתְיָ֜הוּ שִׁשָּׁ֗ה עַל֩ יְדֵ֨י אֲבִיהֶ֤ם יְדוּתוּן֙ בַּכִּנּ֔וֹר הַנִּבָּ֕א עַל־ הֹד֥וֹת וְהַלֵּ֖ל לַיהוָֽה׃ (סח) לְהֵימָ֑ן בְּנֵ֣י הֵימָ֡ן בֻּקִּיָּ֡הוּ מַתַּנְיָ֡הוּ עֻ֠זִּיאֵל שְׁבוּאֵ֨ל וִֽירִימ֜וֹת חֲנַנְיָ֣ה חֲנָ֗נִי אֱלִיאָ֤תָה גִדַּ֙לְתִּי֙ וְרֹמַ֣מְתִּי עֶ֔זֶר יָשְׁבְּקָ֣שָׁה מַלּ֔וֹתִי הוֹתִ֖יר מַחֲזִיאֽוֹת׃ (סט) כָּל־ אֵ֨לֶּה בָנִ֜ים לְהֵימָ֗ן חֹזֵ֥ה הַמֶּ֛לֶךְ בְּדִבְרֵ֥י הָאֱלֹהִ֖ים לְהָרִ֣ים קָ֑רֶן וַיִּתֵּ֨ן הָאֱלֹהִ֜ים לְהֵימָ֗ן בָּנִ֛ים אַרְבָּעָ֥ה עָשָׂ֖ר וּבָנ֥וֹת שָׁלֽוֹשׁ׃ (ע) כָּל־ אֵ֣לֶּה עַל־ יְדֵי֩ אֲבִיהֶ֨ם בַּשִּׁ֜יר בֵּ֣ית יְהוָ֗ה בִּמְצִלְתַּ֙יִם֙ נְבָלִ֣ים וְכִנֹּר֔וֹת לַעֲבֹדַ֖ת בֵּ֣ית הָאֱלֹהִ֑ים עַ֚ל יְדֵ֣י הַמֶּ֔לֶךְ אָסָ֥ף וִידוּת֖וּן וְהֵימָֽן׃ (עא) וַיְהִ֤י מִסְפָּרָם֙ עִם־ אֲחֵיהֶ֔ם מְלֻמְּדֵי־ שִׁ֖יר לַיהוָ֑ה כָּל־ הַ֨מֵּבִ֔ין מָאתַ֖יִם שְׁמוֹנִ֥ים וּשְׁמוֹנָֽה׃ (עב) וַיַּפִּ֜ילוּ גּוֹרָל֣וֹת מִשְׁמֶ֗רֶת לְעֻמַּת֙ כַּקָּטֹ֣ן כַּגָּד֔וֹל מֵבִ֖ין עִם־ תַּלְמִֽיד׃ (עג) וַיֵּצֵ֞א הַגּוֹרָ֧ל הָרִאשׁ֛וֹן לְאָסָ֖ף לְיוֹסֵ֑ף גְּדַלְיָ֙הוּ֙ הַשֵּׁנִ֔י הֽוּא־ וְאֶחָ֥יו וּבָנָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עד) הַשְּׁלִשִׁ֣י זַכּ֔וּר בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עה) הָרְבִיעִי֙ לַיִּצְרִ֔י בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עו) הַחֲמִישִׁ֣י נְתַנְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עז) הַשִּׁשִּׁ֣י בֻקִּיָּ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עח) הַשְּׁבִעִ֣י יְשַׂרְאֵ֔לָה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עט) הַשְּׁמִינִ֣י יְשַֽׁעְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פ) הַתְּשִׁיעִ֣י מַתַּנְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פא) הָעֲשִׂירִ֣י שִׁמְעִ֔י בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פב) עַשְׁתֵּֽי־ עָשָׂ֣ר עֲזַרְאֵ֔ל בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פג) הַשְּׁנֵ֤ים עָשָׂר֙ לַחֲשַׁבְיָ֔ה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פד) לִשְׁלֹשָׁ֤ה עָשָׂר֙ שֽׁוּבָאֵ֔ל בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פה) לְאַרְבָּעָ֤ה עָשָׂר֙ מַתִּתְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פו) לַחֲמִשָּׁ֤ה עָשָׂר֙ לִֽירֵמ֔וֹת בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פז) לְשִׁשָּׁ֤ה עָשָׂר֙ לַחֲנַנְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פח) לְשִׁבְעָ֤ה עָשָׂר֙ לְיָשְׁבְּקָ֔שָׁה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פט) לִשְׁמוֹנָ֤ה עָשָׂר֙ לַחֲנָ֔נִי בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צ) לְתִשְׁעָ֤ה עָשָׂר֙ לְמַלּ֔וֹתִי בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צא) לְעֶשְׂרִים֙ לֶֽאֱלִיָּ֔תָה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צב) לְאֶחָ֤ד וְעֶשְׂרִים֙ לְהוֹתִ֔יר בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צג) לִשְׁנַ֤יִם וְעֶשְׂרִים֙ לְגִדַּ֔לְתִּי בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צד) לִשְׁלֹשָׁ֤ה וְעֶשְׂרִים֙ לְמַ֣חֲזִיא֔וֹת בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צה) לְאַרְבָּעָ֤ה וְעֶשְׂרִים֙ לְרוֹמַ֣מְתִּי עָ֔זֶר בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צו) לְמַחְלְק֖וֹת לְשֹׁעֲרִ֑ים לַקָּרְחִ֕ים מְשֶֽׁלֶמְיָ֥הוּ בֶן־ קֹרֵ֖א מִן־ בְּנֵ֥י אָסָֽף׃ (צז) וְלִמְשֶֽׁלֶמְיָ֖הוּ בָּנִ֑ים זְכַרְיָ֤הוּ הַבְּכוֹר֙ יְדִֽיעֲאֵ֣ל הַשֵּׁנִ֔י זְבַדְיָ֙הוּ֙ הַשְּׁלִישִׁ֔י יַתְנִיאֵ֖ל הָרְבִיעִֽי׃ (צח) עֵילָ֤ם הַחֲמִישִׁי֙ יְהוֹחָנָ֣ן הַשִּׁשִּׁ֔י אֶלְיְהוֹעֵינַ֖י הַשְּׁבִיעִֽי׃ (צט) וּלְעֹבֵ֥ד אֱדֹ֖ם בָּנִ֑ים
