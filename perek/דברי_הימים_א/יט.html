דברי הימים סדר ז

(א) בָּר֤וּךְ יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל מִן־ הָעוֹלָ֖ם וְעַ֣ד הָעֹלָ֑ם וַיֹּאמְר֤וּ כָל־ הָעָם֙ אָמֵ֔ן וְהַלֵּ֖ל לַֽיהוָֽה׃ (ב) וַיַּֽעֲזָב־ שָׁ֗ם לִפְנֵי֙ אֲר֣וֹן בְּרִית־ יְהוָ֔ה לְאָסָ֖ף וּלְאֶחָ֑יו לְשָׁרֵ֞ת לִפְנֵ֧י הָאָר֛וֹן תָּמִ֖יד לִדְבַר־ י֥וֹם בְּיוֹמֽוֹ׃ (ג) וְעֹבֵ֥ד אֱדֹ֛ם וַאֲחֵיהֶ֖ם שִׁשִּׁ֣ים וּשְׁמוֹנָ֑ה וְעֹבֵ֨ד אֱדֹ֧ם בֶּן־ יְדִית֛וּן וְחֹסָ֖ה לְשֹׁעֲרִֽים׃ (ד) וְאֵ֣ת ׀ צָד֣וֹק הַכֹּהֵ֗ן וְאֶחָיו֙ הַכֹּ֣הֲנִ֔ים לִפְנֵ֖י מִשְׁכַּ֣ן יְהוָ֑ה בַּבָּמָ֖ה אֲשֶׁ֥ר בְּגִבְעֽוֹן׃ (ה) לְֽהַעֲלוֹת֩ עֹל֨וֹת לַיהוָ֜ה עַל־ מִזְבַּ֧ח הָעֹלָ֛ה תָּמִ֖יד לַבֹּ֣קֶר וְלָעָ֑רֶב וּלְכָל־ הַכָּתוּב֙ בְּתוֹרַ֣ת יְהוָ֔ה אֲשֶׁ֥ר צִוָּ֖ה עַל־ יִשְׂרָאֵֽל׃ (ו) וְעִמָּהֶם֙ הֵימָ֣ן וִֽידוּת֔וּן וּשְׁאָר֙ הַבְּרוּרִ֔ים אֲשֶׁ֥ר נִקְּב֖וּ בְּשֵׁמ֑וֹת לְהֹדוֹת֙ לַֽיהוָ֔ה כִּ֥י לְעוֹלָ֖ם חַסְדּֽוֹ׃ (ז) וְעִמָּהֶם֩ הֵימָ֨ן וִֽידוּת֜וּן חֲצֹצְר֤וֹת וּמְצִלְתַּ֙יִם֙ לְמַשְׁמִיעִ֔ים וּכְלֵ֖י שִׁ֣יר הָאֱלֹהִ֑ים וּבְנֵ֥י יְדוּת֖וּן לַשָּֽׁעַר׃ (ח) וַיֵּלְכ֥וּ כָל־ הָעָ֖ם אִ֣ישׁ לְבֵית֑וֹ וַיִּסֹּ֥ב דָּוִ֖יד לְבָרֵ֥ךְ אֶת־ בֵּיתֽוֹ׃ (ט) וַיְהִ֕י כַּאֲשֶׁ֛ר יָשַׁ֥ב דָּוִ֖יד בְּבֵית֑וֹ וַיֹּ֨אמֶר דָּוִ֜יד אֶל־ נָתָ֣ן הַנָּבִ֗יא הִנֵּ֨ה אָנֹכִ֤י יוֹשֵׁב֙ בְּבֵ֣ית הָֽאֲרָזִ֔ים וַאֲר֥וֹן בְּרִית־ יְהוָ֖ה תַּ֥חַת יְרִיעֽוֹת׃ (י) וַיֹּ֤אמֶר נָתָן֙ אֶל־ דָּוִ֔יד כֹּ֛ל אֲשֶׁ֥ר בִּֽלְבָבְךָ֖ עֲשֵׂ֑ה כִּ֥י הָאֱלֹהִ֖ים עִמָּֽךְ׃ (יא) וַֽיְהִ֖י בַּלַּ֣יְלָה הַה֑וּא וַיְהִי֙ דְּבַר־ אֱלֹהִ֔ים אֶל־ נָתָ֖ן לֵאמֹֽר׃ (יב) לֵ֤ךְ וְאָמַרְתָּ֙ אֶל־ דָּוִ֣יד עַבְדִּ֔י כֹּ֖ה אָמַ֣ר יְהוָ֑ה לֹ֥א אַתָּ֛ה תִּבְנֶה־ לִּ֥י הַבַּ֖יִת לָשָֽׁבֶת׃ (יג) כִּ֣י לֹ֤א יָשַׁ֙בְתִּי֙ בְּבַ֔יִת מִן־ הַיּ֗וֹם אֲשֶׁ֤ר הֶעֱלֵ֙יתִי֙ אֶת־ יִשְׂרָאֵ֔ל עַ֖ד הַיּ֣וֹם הַזֶּ֑ה וָֽאֶהְיֶ֛ה מֵאֹ֥הֶל אֶל־ אֹ֖הֶל וּמִמִּשְׁכָּֽן׃ (יד) בְּכֹ֥ל אֲשֶֽׁר־ הִתְהַלַּכְתִּי֮ בְּכָל־ יִשְׂרָאֵל֒ הֲדָבָ֣ר דִּבַּ֗רְתִּי אֶת־ אַחַד֙ שֹׁפְטֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֥ר צִוִּ֛יתִי לִרְע֥וֹת אֶת־ עַמִּ֖י לֵאמֹ֑ר לָ֛מָּה לֹא־ בְנִיתֶ֥ם לִ֖י בֵּ֥ית אֲרָזִֽים׃ (טו) וְ֠עַתָּה כֹּֽה־ תֹאמַ֞ר לְעַבְדִּ֣י לְדָוִ֗יד כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת אֲנִ֤י לְקַחְתִּ֙יךָ֙ מִן־ הַנָּוֶ֔ה מִֽן־ אַחֲרֵ֖י הַצֹּ֑אן לִהְי֣וֹת נָגִ֔יד עַ֖ל עַמִּ֥י יִשְׂרָאֵֽל׃ (טז) וָֽאֶהְיֶ֣ה עִמְּךָ֗ בְּכֹל֙ אֲשֶׁ֣ר הָלַ֔כְתָּ וָאַכְרִ֥ית אֶת־ כָּל־ אוֹיְבֶ֖יךָ מִפָּנֶ֑יךָ וְעָשִׂ֤יתִֽי לְךָ֙ שֵׁ֔ם כְּשֵׁ֥ם הַגְּדוֹלִ֖ים אֲשֶׁ֥ר בָּאָֽרֶץ׃ (יז) וְשַׂמְתִּ֣י מָ֠קוֹם לְעַמִּ֨י יִשְׂרָאֵ֤ל וּנְטַעְתִּ֙יהוּ֙ וְשָׁכַ֣ן תַּחְתָּ֔יו וְלֹ֥א יִרְגַּ֖ז ע֑וֹד וְלֹא־ יוֹסִ֤יפוּ בְנֵי־ עַוְלָה֙ לְבַלֹּת֔וֹ כַּאֲשֶׁ֖ר בָּרִאשׁוֹנָֽה׃ (יח) וּלְמִיָּמִ֗ים אֲשֶׁ֨ר צִוִּ֤יתִי שֹֽׁפְטִים֙ עַל־ עַמִּ֣י יִשְׂרָאֵ֔ל וְהִכְנַ֖עְתִּי אֶת־ כָּל־ אוֹיְבֶ֑יךָ וָאַגִּ֣ד לָ֔ךְ וּבַ֖יִת יִֽבְנֶה־ לְּךָ֥ יְהוָֽה׃ (יט) וְהָיָ֗ה כִּֽי־ מָלְא֤וּ יָמֶ֙יךָ֙ לָלֶ֣כֶת עִם־ אֲבֹתֶ֔יךָ וַהֲקִֽימוֹתִ֤י אֶֽת־ זַרְעֲךָ֙ אַחֲרֶ֔יךָ אֲשֶׁ֥ר יִהְיֶ֖ה מִבָּנֶ֑יךָ וַהֲכִינוֹתִ֖י אֶת־ מַלְכוּתֽוֹ׃ (כ) ה֥וּא יִבְנֶה־ לִּ֖י בָּ֑יִת וְכֹנַנְתִּ֥י אֶת־ כִּסְא֖וֹ עַד־ עוֹלָֽם׃ (כא) אֲנִי֙ אֶֽהְיֶה־ לּ֣וֹ לְאָ֔ב וְה֖וּא יִֽהְיֶה־ לִּ֣י לְבֵ֑ן וְחַסְדִּי֙ לֹא־ אָסִ֣יר מֵֽעִמּ֔וֹ כַּאֲשֶׁ֣ר הֲסִיר֔וֹתִי מֵאֲשֶׁ֥ר הָיָ֖ה לְפָנֶֽיךָ׃ (כב) וְהַֽעֲמַדְתִּ֛יהוּ בְּבֵיתִ֥י וּבְמַלְכוּתִ֖י עַד־ הָעוֹלָ֑ם וְכִסְא֕וֹ יִהְיֶ֥ה נָכ֖וֹן עַד־ עוֹלָֽם׃ (כג) כְּכֹל֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה וּכְכֹ֖ל הֶחָז֣וֹן הַזֶּ֑ה כֵּ֛ן דִּבֶּ֥ר נָתָ֖ן אֶל־ דָּוִֽיד׃ (כד) וַיָּבֹא֙ הַמֶּ֣לֶךְ דָּוִ֔יד וַיֵּ֖שֶׁב לִפְנֵ֣י יְהוָ֑ה וַיֹּ֗אמֶר מִֽי־ אֲנִ֞י יְהוָ֤ה אֱלֹהִים֙ וּמִ֣י בֵיתִ֔י כִּ֥י הֲבִיאֹתַ֖נִי עַד־ הֲלֹֽם׃ (כה) וַתִּקְטַ֨ן זֹ֤את בְּעֵינֶ֙יךָ֙ אֱלֹהִ֔ים וַתְּדַבֵּ֥ר עַל־ בֵּֽית־ עַבְדְּךָ֖ לְמֵרָח֑וֹק וּרְאִיתַ֗נִי כְּת֧וֹר הָאָדָ֛ם הַֽמַּעֲלָ֖ה יְהוָ֥ה אֱלֹהִֽים׃ (כו) מַה־ יּוֹסִ֨יף ע֥וֹד דָּוִ֛יד אֵלֶ֖יךָ לְכָב֣וֹד אֶת־ עַבְדֶּ֑ךָ וְאַתָּ֖ה אֶֽת־ עַבְדְּךָ֥ יָדָֽעְתָּ׃ (כז) יְהוָ֕ה בַּעֲב֤וּר עַבְדְּךָ֙ וּֽכְלִבְּךָ֔ עָשִׂ֕יתָ אֵ֥ת כָּל־ הַגְּדוּלָּ֖ה הַזֹּ֑את לְהֹדִ֖יעַ אֶת־ כָּל־ הַגְּדֻלּֽוֹת׃ (כח) יְהוָה֙ אֵ֣ין כָּמ֔וֹךָ וְאֵ֥ין אֱלֹהִ֖ים זוּלָתֶ֑ךָ בְּכֹ֥ל אֲשֶׁר־ שָׁמַ֖עְנוּ בְּאָזְנֵֽינוּ׃ (כט) וּמִי֙ כְּעַמְּךָ֣ יִשְׂרָאֵ֔ל גּ֥וֹי אֶחָ֖ד בָּאָ֑רֶץ אֲשֶׁר֩ הָלַ֨ךְ הָאֱלֹהִ֜ים לִפְדּ֧וֹת ל֣וֹ עָ֗ם לָשׂ֤וּם לְךָ֙ שֵׁ֚ם גְּדֻלּ֣וֹת וְנֹרָא֔וֹת לְגָרֵ֗שׁ מִפְּנֵ֧י עַמְּךָ֛ אֲשֶׁר־ פָּדִ֥יתָ מִמִּצְרַ֖יִם גּוֹיִֽם׃ (ל) וַ֠תִּתֵּן אֶת־ עַמְּךָ֨ יִשְׂרָאֵ֧ל ׀ לְךָ֛ לְעָ֖ם עַד־ עוֹלָ֑ם וְאַתָּ֣ה יְהוָ֔ה הָיִ֥יתָ לָהֶ֖ם לֵאלֹהִֽים׃ (לא) וְעַתָּ֣ה יְהוָ֔ה הַדָּבָ֗ר אֲשֶׁ֨ר דִּבַּ֤רְתָּ עַֽל־ עַבְדְּךָ֙ וְעַל־ בֵּית֔וֹ יֵאָמֵ֖ן עַד־ עוֹלָ֑ם וַעֲשֵׂ֖ה כַּאֲשֶׁ֥ר דִּבַּֽרְתָּ׃ (לב) וְ֠יֵֽאָמֵן וְיִגְדַּ֨ל שִׁמְךָ֤ עַד־ עוֹלָם֙ לֵאמֹ֔ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֱלֹהִ֖ים לְיִשְׂרָאֵ֑ל וּבֵית־ דָּוִ֥יד עַבְדְּךָ֖ נָכ֥וֹן לְפָנֶֽיךָ׃ (לג) כִּ֣י ׀ אַתָּ֣ה אֱלֹהַ֗י גָּלִ֙יתָ֙ אֶת־ אֹ֣זֶן עַבְדְּךָ֔ לִבְנ֥וֹת ל֖וֹ בָּ֑יִת עַל־ כֵּן֙ מָצָ֣א עַבְדְּךָ֔ לְהִתְפַּלֵּ֖ל לְפָנֶֽיךָ׃ (לד) וְעַתָּ֣ה יְהוָ֔ה אַתָּה־ ה֖וּא הָאֱלֹהִ֑ים וַתְּדַבֵּר֙ עַֽל־ עַבְדְּךָ֔ הַטּוֹבָ֖ה הַזֹּֽאת׃ (לה) וְעַתָּ֗ה הוֹאַ֙לְתָּ֙ לְבָרֵךְ֙ אֶת־ בֵּ֣ית עַבְדְּךָ֔ לִהְי֥וֹת לְעוֹלָ֖ם לְפָנֶ֑יךָ כִּֽי־ אַתָּ֤ה יְהוָה֙ בֵּרַ֔כְתָּ וּמְבֹרָ֖ךְ לְעוֹלָֽם׃ (לו) וַיְהִי֙ אַחֲרֵי־ כֵ֔ן וַיַּ֥ךְ דָּוִ֛יד אֶת־ פְּלִשְׁתִּ֖ים וַיַּכְנִיעֵ֑ם וַיִּקַּ֛ח אֶת־ גַּ֥ת וּבְנֹתֶ֖יהָ מִיַּ֥ד פְּלִשְׁתִּֽים׃ (לז) וַיַּ֖ךְ אֶת־ מוֹאָ֑ב וַיִּהְי֤וּ מוֹאָב֙ עֲבָדִ֣ים לְדָוִ֔יד נֹשְׂאֵ֖י מִנְחָֽה׃ (לח) וַיַּ֥ךְ דָּוִ֛יד אֶת־ הֲדַדְעֶ֥זֶר מֶֽלֶךְ־ צוֹבָ֖ה חֲמָ֑תָה בְּלֶכְתּ֕וֹ לְהַצִּ֥יב יָד֖וֹ בִּֽנְהַר־ פְּרָֽת׃ (לט) וַיִּלְכֹּד֩ אֲשֶׁר־ עָשָׂ֨ה מֹשֶׁ֧ה בַמִּדְבָּ֛ר וּמִזְבַּ֥ח הָעוֹלָ֖ה בָּעֵ֣ת הַהִ֑יא בַּבָּמָ֖ה בְּגִבְעֽוֹן׃ (מה) וְלֹא־ יָכֹ֥ל דָּוִ֛יד לָלֶ֥כֶת לְפָנָ֖יו לִדְרֹ֣שׁ אֱלֹהִ֑ים כִּ֣י נִבְעַ֔ת מִפְּנֵ֕י חֶ֖רֶב מַלְאַ֥ךְ יְהוָֽה׃ (מו) וַיֹּ֣אמֶר דָּוִ֔יד זֶ֣ה ה֔וּא בֵּ֖ית יְהוָ֣ה הָאֱלֹהִ֑ים וְזֶה־ מִּזְבֵּ֥חַ לְעֹלָ֖ה לְיִשְׂרָאֵֽל׃ (מז) וַיֹּ֣אמֶר דָּוִ֔יד לִכְנוֹס֙ אֶת־ הַגֵּרִ֔ים אֲשֶׁ֖ר בְּאֶ֣רֶץ יִשְׂרָאֵ֑ל וַיַּעֲמֵ֣ד חֹֽצְבִ֗ים לַחְצוֹב֙ אַבְנֵ֣י גָזִ֔ית לִבְנ֖וֹת בֵּ֥ית הָאֱלֹהִֽים׃ (מח) וּבַרְזֶ֣ל ׀ לָ֠רֹב לַֽמִּסְמְרִ֞ים לְדַלְת֧וֹת הַשְּׁעָרִ֛ים וְלַֽמְחַבְּר֖וֹת הֵכִ֣ין דָּוִ֑יד וּנְחֹ֥שֶׁת לָרֹ֖ב אֵ֥ין מִשְׁקָֽל׃ (מט) וַעֲצֵ֥י אֲרָזִ֖ים לְאֵ֣ין מִסְפָּ֑ר כִּֽי הֵ֠בִיאוּ הַצִּֽידֹנִ֨ים וְהַצֹּרִ֜ים עֲצֵ֧י אֲרָזִ֛ים לָרֹ֖ב לְדָוִֽיד׃ (נ) וַיֹּ֣אמֶר דָּוִ֗יד שְׁלֹמֹ֣ה בְנִי֮ נַ֣עַר וָרָךְ֒ וְהַבַּ֜יִת לִבְנ֣וֹת לַיהוָ֗ה לְהַגְדִּ֨יל ׀ לְמַ֜עְלָה לְשֵׁ֤ם וּלְתִפְאֶ֙רֶת֙ לְכָל־ הָ֣אֲרָצ֔וֹת אָכִ֥ינָה נָּ֖א ל֑וֹ וַיָּ֧כֶן דָּוִ֛יד לָרֹ֖ב לִפְנֵ֥י מוֹתֽוֹ׃ (נא) וַיִּקְרָ֖א לִשְׁלֹמֹ֣ה בְנ֑וֹ וַיְצַוֵּ֙הוּ֙ לִבְנ֣וֹת בַּ֔יִת לַיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (נב) וַיֹּ֥אמֶר דָּוִ֖יד לִשְׁלֹמֹ֑ה (בנו) בְּנִ֕י אֲנִי֙ הָיָ֣ה עִם־ לְבָבִ֔י לִבְנ֣וֹת בַּ֔יִת לְשֵׁ֖ם יְהוָ֥ה אֱלֹהָֽי׃ (נג) וַיְהִ֨י עָלַ֤י דְּבַר־ יְהוָה֙ לֵאמֹ֔ר דָּ֤ם לָרֹב֙ שָׁפַ֔כְתָּ וּמִלְחָמ֥וֹת גְּדֹל֖וֹת עָשִׂ֑יתָ לֹֽא־ תִבְנֶ֥ה בַ֙יִת֙ לִשְׁמִ֔י כִּ֚י דָּמִ֣ים רַבִּ֔ים שָׁפַ֥כְתָּ אַ֖רְצָה לְפָנָֽי׃ (נד) הִנֵּה־ בֵ֞ן נוֹלָ֣ד לָ֗ךְ ה֤וּא יִהְיֶה֙ אִ֣ישׁ מְנוּחָ֔ה וַהֲנִח֥וֹתִי ל֛וֹ מִכָּל־ אוֹיְבָ֖יו מִסָּבִ֑יב כִּ֤י שְׁלֹמֹה֙ יִהְיֶ֣ה שְׁמ֔וֹ וְשָׁל֥וֹם וָשֶׁ֛קֶט אֶתֵּ֥ן עַל־ יִשְׂרָאֵ֖ל בְּיָמָֽיו׃ (נה) הֽוּא־ יִבְנֶ֥ה בַ֙יִת֙ לִשְׁמִ֔י וְהוּא֙ יִהְיֶה־ לִּ֣י לְבֵ֔ן וַאֲנִי־ ל֖וֹ לְאָ֑ב וַהֲכִ֨ינוֹתִ֜י כִּסֵּ֧א מַלְכוּת֛וֹ עַל־ יִשְׂרָאֵ֖ל עַד־ עוֹלָֽם׃ (נו) עַתָּ֣ה בְנִ֔י יְהִ֥י יְהוָ֖ה עִמָּ֑ךְ וְהִצְלַחְתָּ֗ וּבָנִ֙יתָ֙ בֵּ֚ית יְהוָ֣ה אֱלֹהֶ֔יךָ כַּאֲשֶׁ֖ר דִּבֶּ֥ר עָלֶֽיךָ׃ (נז) אַ֣ךְ יִֽתֶּן־ לְּךָ֤ יְהוָה֙ שֵׂ֣כֶל וּבִינָ֔ה וִֽיצַוְּךָ֖ עַל־ יִשְׂרָאֵ֑ל וְלִשְׁמ֕וֹר אֶת־ תּוֹרַ֖ת יְהוָ֥ה אֱלֹהֶֽיךָ׃ (נח) אָ֣ז תַּצְלִ֔יחַ אִם־ תִּשְׁמ֗וֹר לַעֲשׂוֹת֙ אֶת־ הַֽחֻקִּ֣ים וְאֶת־ הַמִּשְׁפָּטִ֔ים אֲשֶׁ֨ר צִוָּ֧ה יְהוָ֛ה אֶת־ מֹשֶׁ֖ה עַל־ יִשְׂרָאֵ֑ל חֲזַ֣ק וֶאֱמָ֔ץ אַל־ תִּירָ֖א וְאַל־ תֵּחָֽת׃ (נט) וְהִנֵּ֨ה בְעָנְיִ֜י הֲכִינ֣וֹתִי לְבֵית־ יְהוָ֗ה זָהָ֞ב כִּכָּרִ֤ים מֵֽאָה־ אֶ֙לֶף֙ וְכֶ֗סֶף אֶ֤לֶף אֲלָפִים֙ כִּכָּרִ֔ים וְלַנְּחֹ֤שֶׁת וְלַבַּרְזֶל֙ אֵ֣ין מִשְׁקָ֔ל כִּ֥י לָרֹ֖ב הָיָ֑ה וְעֵצִ֤ים וַאֲבָנִים֙ הֲכִינ֔וֹתִי וַעֲלֵיהֶ֖ם תּוֹסִֽיף׃ (ס) וְעִמְּךָ֤ לָרֹב֙ עֹשֵׂ֣י מְלָאכָ֔ה חֹצְבִ֕ים וְחָרָשֵׁ֥י אֶ֖בֶן וָעֵ֑ץ וְכָל־ חָכָ֖ם בְּכָל־ מְלָאכָֽה׃ (סא) לַזָּהָ֥ב לַכֶּ֛סֶף וְלַנְּחֹ֥שֶׁת וְלַבַּרְזֶ֖ל אֵ֣ין מִסְפָּ֑ר ק֣וּם וַעֲשֵׂ֔ה וִיהִ֥י יְהוָ֖ה עִמָּֽךְ׃ (סב) וַיְצַ֤ו דָּוִיד֙ לְכָל־ שָׂרֵ֣י יִשְׂרָאֵ֔ל לַעְזֹ֖ר לִשְׁלֹמֹ֥ה בְנֽוֹ׃ (סג) הֲלֹ֨א יְהוָ֤ה אֱלֹֽהֵיכֶם֙ עִמָּכֶ֔ם וְהֵנִ֥יחַ לָכֶ֖ם מִסָּבִ֑יב כִּ֣י ׀ נָתַ֣ן בְּיָדִ֗י אֵ֚ת יֹשְׁבֵ֣י הָאָ֔רֶץ וְנִכְבְּשָׁ֥ה הָאָ֛רֶץ לִפְנֵ֥י יְהוָ֖ה וְלִפְנֵ֥י עַמּֽוֹ׃
