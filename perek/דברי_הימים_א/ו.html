דברי הימים סדר ב

(א) וַיִּקְרָ֣א יַ֠עְבֵּץ לֵאלֹהֵ֨י יִשְׂרָאֵ֜ל לֵאמֹ֗ר אִם־ בָּרֵ֨ךְ תְּבָרֲכֵ֜נִי וְהִרְבִּ֤יתָ אֶת־ גְּבוּלִי֙ וְהָיְתָ֤ה יָדְךָ֙ עִמִּ֔י וְעָשִׂ֥יתָ מֵּרָעָ֖ה לְבִלְתִּ֣י עָצְבִּ֑י וַיָּבֵ֥א אֱלֹהִ֖ים אֵ֥ת אֲשֶׁר־ שָׁאָֽל׃ (ב) וּכְל֥וּב אֲחִֽי־ שׁוּחָ֖ה הוֹלִ֣יד אֶת־ מְחִ֑יר ה֖וּא אֲבִ֥י אֶשְׁתּֽוֹן׃ (ג) וְאֶשְׁתּ֗וֹן הוֹלִ֞יד אֶת־ בֵּ֤ית רָפָא֙ וְאֶת־ פָּסֵ֔חַ וְאֶת־ תְּחִנָּ֖ה אֲבִ֣י עִ֣יר נָחָ֑שׁ אֵ֖לֶּה אַנְשֵׁ֥י רֵכָֽה׃ (ד) וּבְנֵ֣י קְנַ֔ז עָתְנִיאֵ֖ל וּשְׂרָיָ֑ה וּבְנֵ֥י עָתְנִיאֵ֖ל חֲתַֽת׃ (ה) וּמְעוֹנֹתַ֖י הוֹלִ֣יד אֶת־ עָפְרָ֑ה וּשְׂרָיָ֗ה הוֹלִ֤יד אֶת־ יוֹאָב֙ אֲבִי֙ גֵּ֣יא חֲרָשִׁ֔ים כִּ֥י חֲרָשִׁ֖ים הָיֽוּ׃ (ו) וּבְנֵי֙ כָּלֵ֣ב בֶּן־ יְפֻנֶּ֔ה עִ֥ירוּ אֵלָ֖ה וָנָ֑עַם וּבְנֵ֥י אֵלָ֖ה וּקְנַֽז׃ (ז) וּבְנֵ֖י יְהַלֶּלְאֵ֑ל זִ֣יף וְזִיפָ֔ה תִּירְיָ֖א וַאֲשַׂרְאֵֽל׃ (ח) וּבֶן־ עֶזְרָ֔ה יֶ֥תֶר וּמֶ֖רֶד וְעֵ֣פֶר וְיָל֑וֹן וַתַּ֙הַר֙ אֶת־ מִרְיָ֣ם וְאֶת־ שַׁמַּ֔י וְאֶת־ יִשְׁבָּ֖ח אֲבִ֥י אֶשְׁתְּמֹֽעַ׃ (ט) וְאִשְׁתּ֣וֹ הַיְהֻדִיָּ֗ה יָלְדָ֞ה אֶת־ יֶ֨רֶד אֲבִ֤י גְדוֹר֙ וְאֶת־ חֶ֙בֶר֙ אֲבִ֣י שׂוֹכ֔וֹ וְאֶת־ יְקֽוּתִיאֵ֖ל אֲבִ֣י זָנ֑וֹחַ וְאֵ֗לֶּה בְּנֵי֙ בִּתְיָ֣ה בַת־ פַּרְעֹ֔ה אֲשֶׁ֥ר לָקַ֖ח מָֽרֶד׃ (י) וּבְנֵי֙ אֵ֣שֶׁת הֽוֹדִיָּ֔ה אֲח֣וֹת נַ֔חַם אֲבִ֥י קְעִילָ֖ה הַגַּרְמִ֑י וְאֶשְׁתְּמֹ֖עַ הַמַּעֲכָתִֽי׃ (יא) וּבְנֵ֣י שִׁימ֔וֹן אַמְנ֣וֹן וְרִנָּ֔ה בֶּן־ חָנָ֖ן (ותולון) וְתִיל֑וֹן וּבְנֵ֣י יִשְׁעִ֔י זוֹחֵ֖ת וּבֶן־ זוֹחֵֽת׃ (יב) בְּנֵי֙ שֵׁלָ֣ה בֶן־ יְהוּדָ֔ה עֵ֚ר אֲבִ֣י לֵכָ֔ה וְלַעְדָּ֖ה אֲבִ֣י מָרֵשָׁ֑ה וּמִשְׁפְּח֛וֹת בֵּית־ עֲבֹדַ֥ת הַבֻּ֖ץ לְבֵ֥ית אַשְׁבֵּֽעַ׃ (יג) וְיוֹקִ֞ים וְאַנְשֵׁ֣י כֹזֵבָ֗א וְיוֹאָ֧שׁ וְשָׂרָ֛ף אֲשֶׁר־ בָּעֲל֥וּ לְמוֹאָ֖ב וְיָשֻׁ֣בִי לָ֑חֶם וְהַדְּבָרִ֖ים עַתִּיקִֽים׃ (יד) הֵ֚מָּה הַיּ֣וֹצְרִ֔ים וְיֹשְׁבֵ֥י נְטָעִ֖ים וּגְדֵרָ֑ה עִם־ הַמֶּ֥לֶךְ בִּמְלַאכְתּ֖וֹ יָ֥שְׁבוּ שָֽׁם׃ (טו) בְּנֵ֖י שִׁמְע֑וֹן נְמוּאֵ֣ל וְיָמִ֔ין יָרִ֖יב זֶ֥רַח שָׁאֽוּל׃ (טז) שַׁלֻּ֥ם בְּנ֛וֹ מִבְשָׂ֥ם בְּנ֖וֹ מִשְׁמָ֥ע בְּנֽוֹ׃ (יז) וּבְנֵ֖י מִשְׁמָ֑ע חַמּוּאֵ֥ל בְּנ֛וֹ זַכּ֥וּר בְּנ֖וֹ שִׁמְעִ֥י בְנֽוֹ׃ (יח) וּלְשִׁמְעִ֞י בָּנִ֨ים שִׁשָּׁ֤ה עָשָׂר֙ וּבָנ֣וֹת שֵׁ֔שׁ וּלְאֶחָ֕יו אֵ֖ין בָּנִ֣ים רַבִּ֑ים וְכֹל֙ מִשְׁפַּחְתָּ֔ם לֹ֥א הִרְבּ֖וּ עַד־ בְּנֵ֥י יְהוּדָֽה׃ (יט) וַיֵּֽשְׁב֛וּ בִּבְאֵֽר־ שֶׁ֥בַע וּמוֹלָדָ֖ה וַחֲצַ֥ר שׁוּעָֽל׃ (כ) וּבְבִלְהָ֥ה וּבְעֶ֖צֶם וּבְתוֹלָֽד׃ (כא) וּבִבְתוּאֵ֥ל וּבְחָרְמָ֖ה וּבְצִֽיקְלָֽג׃ (כב) וּבְבֵ֤ית מַרְכָּבוֹת֙ וּבַחֲצַ֣ר סוּסִ֔ים וּבְבֵ֥ית בִּרְאִ֖י וּֽבְשַׁעֲרָ֑יִם אֵ֥לֶּה עָרֵיהֶ֖ם עַד־ מְלֹ֥ךְ דָּוִֽיד׃ (כג) וְחַצְרֵיהֶם֙ עֵיטָ֣ם וָעַ֔יִן רִמּ֥וֹן וְתֹ֖כֶן וְעָשָׁ֑ן עָרִ֖ים חָמֵֽשׁ׃ (כד) וְכָל־ חַצְרֵיהֶ֗ם אֲשֶׁ֧ר סְבִיב֛וֹת הֶעָרִ֥ים הָאֵ֖לֶּה עַד־ בָּ֑עַל זֹ֚את מוֹשְׁבֹתָ֔ם וְהִתְיַחְשָׂ֖ם לָהֶֽם׃ (כה) וּמְשׁוֹבָ֣ב וְיַמְלֵ֔ךְ וְיוֹשָׁ֖ה בֶּן־ אֲמַצְיָֽה׃ (כו) וְיוֹאֵ֑ל וְיֵהוּא֙ בֶּן־ י֣וֹשִׁבְיָ֔ה בֶּן־ שְׂרָיָ֖ה בֶּן־ עֲשִׂיאֵֽל׃ (כז) וְאֶלְיוֹעֵינַ֡י וְֽיַעֲקֹ֡בָה וִ֠ישׁוֹחָיָה וַעֲשָׂיָ֧ה וַעֲדִיאֵ֛ל וִישִׂימִאֵ֖ל וּבְנָיָֽה׃ (כח) וְזִיזָ֨א בֶן־ שִׁפְעִ֧י בֶן־ אַלּ֛וֹן בֶּן־ יְדָיָ֥ה בֶן־ שִׁמְרִ֖י בֶּן־ שְׁמַֽעְיָֽה׃ (כט) אֵ֚לֶּה הַבָּאִ֣ים בְּשֵׁמ֔וֹת נְשִׂיאִ֖ים בְּמִשְׁפְּחוֹתָ֑ם וּבֵית֙ אֲב֣וֹתֵיהֶ֔ם פָּרְצ֖וּ לָרֽוֹב׃ (ל) וַיֵּלְכוּ֙ לִמְב֣וֹא גְדֹ֔ר עַ֖ד לְמִזְרַ֣ח הַגָּ֑יְא לְבַקֵּ֥שׁ מִרְעֶ֖ה לְצֹאנָֽם׃ (לא) וַֽיִּמְצְא֤וּ מִרְעֶה֙ שָׁמֵ֣ן וָט֔וֹב וְהָאָ֙רֶץ֙ רַחֲבַ֣ת יָדַ֔יִם וְשֹׁקֶ֖טֶת וּשְׁלֵוָ֑ה כִּ֣י מִן־ חָ֔ם הַיֹּשְׁבִ֥ים שָׁ֖ם לְפָנִֽים׃ (לב) וַיָּבֹ֡אוּ אֵלֶּה֩ הַכְּתוּבִ֨ים בְּשֵׁמ֜וֹת בִּימֵ֣י ׀ יְחִזְקִיָּ֣הוּ מֶֽלֶךְ־ יְהוּדָ֗ה וַיַּכּ֨וּ אֶת־ אָהֳלֵיהֶ֜ם וְאֶת־ (המעינים) הַמְּעוּנִ֨ים אֲשֶׁ֤ר נִמְצְאוּ־ שָׁ֙מָּה֙ וַיַּחֲרִימֻם֙ עַד־ הַיּ֣וֹם הַזֶּ֔ה וַיֵּשְׁב֖וּ תַּחְתֵּיהֶ֑ם כִּֽי־ מִרְעֶ֥ה לְצֹאנָ֖ם שָֽׁם׃ (לג) וּמֵהֶ֣ם ׀ מִן־ בְּנֵ֣י שִׁמְע֗וֹן הָלְכוּ֙ לְהַ֣ר שֵׂעִ֔יר אֲנָשִׁ֖ים חֲמֵ֣שׁ מֵא֑וֹת וּפְלַטְיָ֡ה וּ֠נְעַרְיָה וּרְפָיָ֧ה וְעֻזִּיאֵ֛ל בְּנֵ֥י יִשְׁעִ֖י בְּרֹאשָֽׁם׃ (לד) וַיַּכּ֕וּ אֶת־ שְׁאֵרִ֥ית הַפְּלֵטָ֖ה לַעֲמָלֵ֑ק וַיֵּ֣שְׁבוּ שָׁ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (לה) וּבְנֵ֨י רְאוּבֵ֥ן בְּכֽוֹר־ יִשְׂרָאֵל֮ כִּ֣י ה֣וּא הַבְּכוֹר֒ וּֽבְחַלְּלוֹ֙ יְצוּעֵ֣י אָבִ֔יו נִתְּנָה֙ בְּכֹ֣רָת֔וֹ לִבְנֵ֥י יוֹסֵ֖ף בֶּן־ יִשְׂרָאֵ֑ל וְלֹ֥א לְהִתְיַחֵ֖שׂ לַבְּכֹרָֽה׃ (לו) כִּ֤י יְהוּדָה֙ גָּבַ֣ר בְּאֶחָ֔יו וּלְנָגִ֖יד מִמֶּ֑נּוּ וְהַבְּכֹרָ֖ה לְיוֹסֵֽף׃ (לז) בְּנֵ֥י רְאוּבֵ֖ן בְּכ֣וֹר יִשְׂרָאֵ֑ל חֲנ֥וֹךְ וּפַלּ֖וּא חֶצְר֥וֹן וְכַרְמִֽי׃ (לח) בְּנֵ֖י יוֹאֵ֑ל שְׁמַֽעְיָ֥ה בְנ֛וֹ גּ֥וֹג בְּנ֖וֹ שִׁמְעִ֥י בְנֽוֹ׃ (לט) מִיכָ֥ה בְנ֛וֹ רְאָיָ֥ה בְנ֖וֹ בַּ֥עַל בְּנֽוֹ׃ (מ) בְּאֵרָ֣ה בְנ֔וֹ אֲשֶׁ֣ר הֶגְלָ֔ה תִּלְּגַ֥ת פִּלְנְאֶ֖סֶר מֶ֣לֶךְ אַשֻּׁ֑ר ה֥וּא נָשִׂ֖יא לָרֽאוּבֵנִֽי׃ (מא) וְאֶחָיו֙ לְמִשְׁפְּחֹתָ֔יו בְּהִתְיַחֵ֖שׂ לְתֹלְדוֹתָ֑ם הָרֹ֥אשׁ יְעִיאֵ֖ל וּזְכַרְיָֽהוּ׃ (מב) וּבֶ֙לַע֙ בֶּן־ עָזָ֔ז בֶּן־ שֶׁ֖מַע בֶּן־ יוֹאֵ֑ל ה֚וּא יוֹשֵׁ֣ב בַּעֲרֹעֵ֔ר וְעַד־ נְב֖וֹ וּבַ֥עַל מְעֽוֹן׃ (מג) וְלַמִּזְרָ֗ח יָשַׁב֙ עַד־ לְב֣וֹא מִדְבָּ֔רָה לְמִן־ הַנָּהָ֖ר פְּרָ֑ת כִּ֧י מִקְנֵיהֶ֛ם רָב֖וּ בְּאֶ֥רֶץ גִּלְעָֽד׃ (מד) וּבִימֵ֣י שָׁא֗וּל עָשׂ֤וּ מִלְחָמָה֙ עִם־ הַֽהַגְרִאִ֔ים וַֽיִּפְּל֖וּ בְּיָדָ֑ם וַיֵּשְׁבוּ֙ בְּאָ֣הֳלֵיהֶ֔ם עַֽל־ כָּל־ פְּנֵ֖י מִזְרָ֥ח לַגִּלְעָֽד׃ (מה) וּבְנֵי־ גָ֣ד לְנֶגְדָּ֗ם יָֽשְׁב֛וּ בְּאֶ֥רֶץ הַבָּשָׁ֖ן עַד־ סַלְכָֽה׃ (מו) יוֹאֵ֣ל הָרֹ֔אשׁ וְשָׁפָ֖ם הַמִּשְׁנֶ֑ה וְיַעְנַ֥י וְשָׁפָ֖ט בַּבָּשָֽׁן׃ (מז) וַאֲחֵיהֶ֞ם לְבֵ֣ית אֲבוֹתֵיהֶ֗ם מִֽיכָאֵ֡ל וּמְשֻׁלָּ֡ם וְ֠שֶׁבַע וְיוֹרַ֧י וְיַעְכָּ֛ן וְזִ֥יעַ וָעֵ֖בֶר שִׁבְעָֽה׃ (מח) אֵ֣לֶּה ׀ בְּנֵ֣י אֲבִיחַ֗יִל בֶּן־ חוּרִ֡י בֶּן־ יָ֠רוֹחַ בֶּן־ גִּלְעָ֧ד בֶּן־ מִיכָאֵ֛ל בֶּן־ עֵ֣שֶׁק אָחִ֑יו אוּלָ֣ם בְּכֹר֔וֹ יְעוּשׁ֙ הַשֵּׁנִ֔י וֶֽאֱלִיפֶ֖לֶט הַשְּׁלִשִֽׁי׃
