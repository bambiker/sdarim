פרק יב בספר דברי הימים א (41 פסוקים) נמצא בדברי הימים סדר ה (פסוקים מ–עט) ובדברי הימים סדר ו (פסוקים א–א).

דברי הימים סדר ה

(א) וַיֵּ֥לֶךְ דָּוִ֖יד הָל֣וֹךְ וְגָד֑וֹל וַיהוָ֥ה צְבָא֖וֹת עִמּֽוֹ׃ (ב) וְאֵ֨לֶּה רָאשֵׁ֤י הַגִּבּוֹרִים֙ אֲשֶׁ֣ר לְדָוִ֔יד הַמִּתְחַזְּקִ֨ים עִמּ֧וֹ בְמַלְכוּת֛וֹ עִם־ כָּל־ יִשְׂרָאֵ֖ל לְהַמְלִיכ֑וֹ כִּדְבַ֥ר יְהוָ֖ה עַל־ יִשְׂרָאֵֽל׃ (ג) וְאֵ֛לֶּה מִסְפַּ֥ר הַגִּבֹּרִ֖ים אֲשֶׁ֣ר לְדָוִ֑יד יָשָׁבְעָ֣ם בֶּן־ חַכְמוֹנִ֗י רֹ֚אשׁ (השלושים) הַשָּׁ֣לִישִׁ֔ים הֽוּא־ עוֹרֵ֧ר אֶת־ חֲנִית֛וֹ עַל־ שְׁלֹשׁ־ מֵא֥וֹת חָלָ֖ל בְּפַ֥עַם אֶחָֽת׃ (ד) וְאַחֲרָ֛יו אֶלְעָזָ֥ר בֶּן־ דּוֹד֖וֹ הָאֲחוֹחִ֑י ה֖וּא בִּשְׁלוֹשָׁ֥ה הַגִּבֹּרִֽים׃ (ה) הֽוּא־ הָיָ֨ה עִם־ דָּוִ֜יד בַּפַּ֣ס דַּמִּ֗ים וְהַפְּלִשְׁתִּים֙ נֶאֱסְפוּ־ שָׁ֣ם לַמִּלְחָמָ֔ה וַתְּהִ֛י חֶלְקַ֥ת הַשָּׂדֶ֖ה מְלֵאָ֣ה שְׂעוֹרִ֑ים וְהָעָ֥ם נָ֖סוּ מִפְּנֵ֥י פְלִשְׁתִּֽים׃ (ו) וַיִּֽתְיַצְּב֤וּ בְתוֹךְ־ הַחֶלְקָה֙ וַיַּצִּיל֔וּהָ וַיַּכּ֖וּ אֶת־ פְּלִשְׁתִּ֑ים וַיּ֥וֹשַׁע יְהוָ֖ה תְּשׁוּעָ֥ה גְדוֹלָֽה׃ (ז) וַיֵּרְד֡וּ שְֽׁלוֹשָׁה֩ מִן־ הַשְּׁלוֹשִׁ֨ים רֹ֤אשׁ עַל־ הַצֻּר֙ אֶל־ דָּוִ֔יד אֶל־ מְעָרַ֖ת עֲדֻלָּ֑ם וּמַחֲנֵ֣ה פְלִשְׁתִּ֔ים חֹנָ֖ה בְּעֵ֥מֶק רְפָאִֽים׃ (ח) וְדָוִ֖יד אָ֣ז בַּמְּצוּדָ֑ה וּנְצִ֣יב פְּלִשְׁתִּ֔ים אָ֖ז בְּבֵ֥ית לָֽחֶם׃ (ט) (ויתאו) וַיִּתְאָ֥יו דָּוִ֖יד וַיֹּאמַ֑ר מִ֚י יַשְׁקֵ֣נִי מַ֔יִם מִבּ֥וֹר בֵּֽית־ לֶ֖חֶם אֲשֶׁ֥ר בַּשָּֽׁעַר׃ (י) וַיִּבְקְע֨וּ הַשְּׁלֹשָׁ֜ה בְּמַחֲנֵ֣ה פְלִשְׁתִּ֗ים וַיִּֽשְׁאֲבוּ־ מַ֙יִם֙ מִבּ֤וֹר בֵּֽית־ לֶ֙חֶם֙ אֲשֶׁ֣ר בַּשַּׁ֔עַר וַיִּשְׂא֖וּ וַיָּבִ֣אוּ אֶל־ דָּוִ֑יד וְלֹֽא־ אָבָ֤ה דָוִיד֙ לִשְׁתּוֹתָ֔ם וַיְנַסֵּ֥ךְ אֹתָ֖ם לַיהוָֽה׃ (יא) וַיֹּ֡אמֶר חָלִילָה֩ לִּ֨י מֵאֱלֹהַ֜י מֵעֲשׂ֣וֹת זֹ֗את הֲדַ֣ם הָאֲנָשִׁים֩ הָאֵ֨לֶּה אֶשְׁתֶּ֤ה בְנַפְשׁוֹתָם֙ כִּ֣י בְנַפְשׁוֹתָ֣ם הֱבִיא֔וּם וְלֹ֥א אָבָ֖ה לִשְׁתּוֹתָ֑ם אֵ֣לֶּה עָשׂ֔וּ שְׁלֹ֖שֶׁת הַגִּבּוֹרִֽים׃ (יב) וְאַבְשַׁ֣י אֲחִֽי־ יוֹאָ֗ב ה֤וּא הָיָה֙ רֹ֣אשׁ הַשְּׁלוֹשָׁ֔ה וְהוּא֙ עוֹרֵ֣ר אֶת־ חֲנִית֔וֹ עַל־ שְׁלֹ֥שׁ מֵא֖וֹת חָלָ֑ל (ולא)־ וְלוֹ־ שֵׁ֖ם בַּשְּׁלוֹשָֽׁה׃ (יג) מִן־ הַשְּׁלוֹשָׁ֤ה בַשְּׁנַ֙יִם֙ נִכְבָּ֔ד וַיְהִ֥י לָהֶ֖ם לְשָׂ֑ר וְעַד־ הַשְּׁלוֹשָׁ֖ה לֹֽא־ בָֽא׃ (יד) בְּנָיָ֨ה בֶן־ יְהוֹיָדָ֧ע בֶּן־ אִֽישׁ־ חַ֛יִל רַב־ פְּעָלִ֖ים מִֽן־ קַבְצְאֵ֑ל ה֣וּא הִכָּ֗ה אֵ֣ת שְׁנֵ֤י אֲרִיאֵל֙ מוֹאָ֔ב וְ֠הוּא יָרַ֞ד וְהִכָּ֧ה אֶֽת־ הָאֲרִ֛י בְּת֥וֹךְ הַבּ֖וֹר בְּי֥וֹם הַשָּֽׁלֶג׃ (טו) וְהֽוּא־ הִכָּה֩ אֶת־ הָאִ֨ישׁ הַמִּצְרִ֜י אִ֥ישׁ מִדָּ֣ה ׀ חָמֵ֣שׁ בָּאַמָּ֗ה וּבְיַ֨ד הַמִּצְרִ֤י חֲנִית֙ כִּמְנ֣וֹר אֹרְגִ֔ים וַיֵּ֥רֶד אֵלָ֖יו בַּשָּׁ֑בֶט וַיִּגְזֹ֤ל אֶֽת־ הַחֲנִית֙ מִיַּ֣ד הַמִּצְרִ֔י וַיַּהַרְגֵ֖הוּ בַּחֲנִיתֽוֹ׃ (טז) אֵ֣לֶּה עָשָׂ֔ה בְּנָיָ֖הוּ בֶּן־ יְהוֹיָדָ֑ע וְלוֹ־ שֵׁ֖ם בִּשְׁלוֹשָׁ֥ה הַגִּבֹּרִֽים׃ (יז) מִן־ הַשְּׁלוֹשִׁ֗ים הִנּ֤וֹ נִכְבָּד֙ ה֔וּא וְאֶל־ הַשְּׁלוֹשָׁ֖ה לֹא־ בָ֑א וַיְשִׂימֵ֥הוּ דָוִ֖יד עַל־ מִשְׁמַעְתּֽוֹ׃ (יח) וְגִבּוֹרֵ֖י הַחֲיָלִ֑ים עֲשָׂה־ אֵל֙ אֲחִ֣י יוֹאָ֔ב אֶלְחָנָ֥ן בֶּן־ דּוֹד֖וֹ מִבֵּ֥ית לָֽחֶם׃ (יט) שַׁמּוֹת֙ הַהֲרוֹרִ֔י חֶ֖לֶץ הַפְּלוֹנִֽי׃ (כ) עִירָ֤א בֶן־ עִקֵּשׁ֙ הַתְּקוֹעִ֔י אֲבִיעֶ֖זֶר הָעֲנְּתוֹתִֽי׃ (כא) סִבְּכַי֙ הַחֻ֣שָׁתִ֔י עִילַ֖י הָאֲחוֹחִֽי׃ (כב) מַהְרַי֙ הַנְּטֹ֣פָתִ֔י חֵ֥לֶד בֶּֽן־ בַּֽעֲנָ֖ה הַנְּטוֹפָתִֽי׃ (כג) אִיתַ֣י בֶּן־ רִיבַ֗י מִגִּבְעַת֙ בְּנֵ֣י בִנְיָמִ֔ן בְּנָיָ֖ה הַפִּרְעָתֹנִֽי׃ (כד) חוּרַי֙ מִנַּ֣חֲלֵי גָ֔עַשׁ אֲבִיאֵ֖ל הָעַרְבָתִֽי׃ (כה) עַזְמָ֙וֶת֙ הַבַּ֣חֲרוּמִ֔י אֶלְיַחְבָּ֖א הַשַּׁעַלְבֹנִֽי׃ (כו) בְּנֵ֗י הָשֵׁם֙ הַגִּ֣זוֹנִ֔י יוֹנָתָ֥ן בֶּן־ שָׁגֵ֖ה הַהֲרָרִֽי׃ (כז) אֲחִיאָ֧ם בֶּן־ שָׂכָ֛ר הַהֲרָרִ֖י אֱלִיפַ֥ל בֶּן־ אֽוּר׃ (כח) חֵ֚פֶר הַמְּכֵ֣רָתִ֔י אֲחִיָּ֖ה הַפְּלֹנִֽי׃ (כט) חֶצְרוֹ֙ הַֽכַּרְמְלִ֔י נַעֲרַ֖י בֶּן־ אֶזְבָּֽי׃ (ל) יוֹאֵל֙ אֲחִ֣י נָתָ֔ן מִבְחָ֖ר בֶּן־ הַגְרִֽי׃ (לא) צֶ֖לֶק הָעַמּוֹנִ֑י נַחְרַי֙ הַבֵּ֣רֹתִ֔י נֹשֵׂ֕א כְּלֵ֖י יוֹאָ֥ב בֶּן־ צְרוּיָֽה׃ (לב) עִירָא֙ הַיִּתְרִ֔י גָּרֵ֖ב הַיִּתְרִֽי׃ (לג) אֽוּרִיָּה֙ הַחִתִּ֔י זָבָ֖ד בֶּן־ אַחְלָֽי׃ (לד) עֲדִינָ֨א בֶן־ שִׁיזָ֜א הָרֽאוּבֵנִ֗י רֹ֛אשׁ לָרֽאוּבֵנִ֖י וְעָלָ֥יו שְׁלוֹשִֽׁים׃ (לה) חָנָן֙ בֶּֽן־ מַעֲכָ֔ה וְיוֹשָׁפָ֖ט הַמִּתְנִֽי׃ (לו) עֻזִיָּ֖א הָעֲשְׁתְּרָתִ֑י שָׁמָע֙ (ויעואל) וִֽיעִיאֵ֔ל בְּנֵ֖י חוֹתָ֥ם הָעֲרֹעֵרִֽי׃ (לז) יְדִֽיעֲאֵל֙ בֶּן־ שִׁמְרִ֔י וְיֹחָ֥א אָחִ֖יו הַתִּיצִֽי׃ (לח) אֱלִיאֵל֙ הַֽמַּחֲוִ֔ים וִירִיבַ֥י וְיוֹשַׁוְיָ֖ה בְּנֵ֣י אֶלְנָ֑עַם וְיִתְמָ֖ה הַמּוֹאָבִֽי׃ (לט) אֱלִיאֵ֣ל וְעוֹבֵ֔ד וְיַעֲשִׂיאֵ֖ל הַמְּצֹבָיָֽה׃ תחילת פרק יב (מ) וְאֵ֗לֶּה הַבָּאִ֤ים אֶל־ דָּוִיד֙ לְצִ֣יקְלַ֔ג ע֣וֹד עָצ֔וּר מִפְּנֵ֖י שָׁא֣וּל בֶּן־ קִ֑ישׁ וְהֵ֙מָּה֙ בַּגִּבּוֹרִ֔ים עֹזְרֵ֖י הַמִּלְחָמָֽה׃ (מא) נֹ֣שְׁקֵי קֶ֗שֶׁת מַיְמִינִ֤ים וּמַשְׂמִאלִים֙ בָּֽאֲבָנִ֔ים וּבַחִצִּ֖ים בַּקָּ֑שֶׁת מֵאֲחֵ֥י שָׁא֖וּל מִבִּנְיָמִֽן׃ (מב) הָרֹ֨אשׁ אֲחִיעֶ֜זֶר וְיוֹאָ֗שׁ בְּנֵי֙ הַשְּׁמָעָ֣ה הַגִּבְעָתִ֔י (ויזואל) וִיזִיאֵ֥ל וָפֶ֖לֶט בְּנֵ֣י עַזְמָ֑וֶת וּבְרָכָ֕ה וְיֵה֖וּא הָעֲנְּתֹתִֽי׃ (מג) וְיִֽשְׁמַֽעְיָ֧ה הַגִּבְעוֹנִ֛י גִּבּ֥וֹר בַּשְּׁלֹשִׁ֖ים וְעַל־ הַשְּׁלֹשִֽׁים׃ (מד) וְיִרְמְיָ֤ה וְיַחֲזִיאֵל֙ וְי֣וֹחָנָ֔ן וְיוֹזָבָ֖ד הַגְּדֵרָתִֽי׃ (מה) אֶלְעוּזַ֤י וִירִימוֹת֙ וּבְעַלְיָ֣ה וּשְׁמַרְיָ֔הוּ וּשְׁפַטְיָ֖הוּ (החריפי) הַחֲרוּפִֽי׃ (מו) אֶלְקָנָ֡ה וְ֠יִשִּׁיָּהוּ וַעֲזַרְאֵ֧ל וְיוֹעֶ֛זֶר וְיָשָׁבְעָ֖ם הַקָּרְחִֽים׃ (מז) וְיוֹעֵאלָ֧ה וּזְבַדְיָ֛ה בְּנֵ֥י יְרֹחָ֖ם מִן־ הַגְּדֽוֹר׃ (מח) וּמִן־ הַגָּדִ֡י נִבְדְּל֣וּ אֶל־ דָּוִיד֩ לַמְצַ֨ד מִדְבָּ֜רָה גִּבֹּרֵ֣י הַחַ֗יִל אַנְשֵׁ֤י צָבָא֙ לַמִּלְחָמָ֔ה עֹרְכֵ֥י צִנָּ֖ה וָרֹ֑מַח וּפְנֵ֤י אַרְיֵה֙ פְּנֵיהֶ֔ם וְכִצְבָאיִ֥ם עַל־ הֶהָרִ֖ים לְמַהֵֽר׃ (מט) עֵ֖זֶר הָרֹ֑אשׁ עֹבַדְיָה֙ הַשֵּׁנִ֔י אֱלִיאָ֖ב הַשְּׁלִשִֽׁי׃ (נ) מִשְׁמַנָּה֙ הָרְבִיעִ֔י יִרְמְיָ֖ה הַחֲמִשִֽׁי׃ (נא) עַתַּי֙ הַשִּׁשִּׁ֔י אֱלִיאֵ֖ל הַשְּׁבִעִֽי׃ (נב) בּ֗וּץ וְכָל־ הַלְוִיִּם֙ הַנֹּשְׂאִ֣ים אֶת־ הָאָר֔וֹן וְהַמְשֹׁ֣רְרִ֔ים וּכְנַנְיָ֛ה הַשַּׂ֥ר הַמַּשָּׂ֖א הַמְשֹֽׁרְרִ֑ים וְעַל־ דָּוִ֖יד אֵפ֥וֹד בָּֽד׃ (ס) וְכָל־ יִשְׂרָאֵ֗ל מַעֲלִים֙ אֶת־ אֲר֣וֹן בְּרִית־ יְהוָ֔ה בִּתְרוּעָה֙ וּבְק֣וֹל שׁוֹפָ֔ר וּבַחֲצֹצְר֖וֹת וּבִמְצִלְתָּ֑יִם מַשְׁמִעִ֕ים בִּנְבָלִ֖ים וְכִנֹּרֽוֹת׃ (סא) וַיְהִ֗י אֲרוֹן֙ בְּרִ֣ית יְהוָ֔ה בָּ֖א עַד־ עִ֣יר דָּוִ֑יד וּמִיכַ֨ל בַּת־ שָׁא֜וּל נִשְׁקְפָ֣ה ׀ בְּעַ֣ד הַחַלּ֗וֹן וַתֵּ֨רֶא אֶת־ הַמֶּ֤לֶךְ דָּוִיד֙ מְרַקֵּ֣ד וּמְשַׂחֵ֔ק וַתִּ֥בֶז ל֖וֹ בְּלִבָּֽהּ׃ (סב) וַיָּבִ֙יאוּ֙ אֶת־ אֲר֣וֹן הָֽאֱלֹהִ֔ים וַיַּצִּ֣יגוּ אֹת֔וֹ בְּת֣וֹךְ הָאֹ֔הֶל אֲשֶׁ֥ר נָֽטָה־ ל֖וֹ דָּוִ֑יד וַיַּקְרִ֛יבוּ עֹל֥וֹת וּשְׁלָמִ֖ים לִפְנֵ֥י הָאֱלֹהִֽים׃ (סג) וַיְכַ֣ל דָּוִ֔יד מֵהַעֲל֥וֹת הָעֹלָ֖ה וְהַשְּׁלָמִ֑ים וַיְבָ֥רֶךְ אֶת־ הָעָ֖ם בְּשֵׁ֥ם יְהוָֽה׃ (סד) וַיְחַלֵּק֙ לְכָל־ אִ֣ישׁ יִשְׂרָאֵ֔ל מֵאִ֖ישׁ וְעַד־ אִשָּׁ֑ה לְאִישׁ֙ כִּכַּר־ לֶ֔חֶם וְאֶשְׁפָּ֖ר וַאֲשִׁישָֽׁה׃ (סה) וַיִּתֵּ֞ן לִפְנֵ֨י אֲר֧וֹן יְהוָ֛ה מִן־ הַלְוִיִּ֖ם מְשָׁרְתִ֑ים וּלְהַזְכִּיר֙ וּלְהוֹד֣וֹת וּלְהַלֵּ֔ל לַיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (סו) אָסָ֥ף הָרֹ֖אשׁ וּמִשְׁנֵ֣הוּ זְכַרְיָ֑ה יְעִיאֵ֡ל וּשְׁמִֽירָמ֡וֹת וִֽיחִיאֵ֡ל וּמַתִּתְיָ֡ה וֶאֱלִיאָ֡ב וּבְנָיָהוּ֩ וְעֹבֵ֨ד אֱדֹ֜ם וִֽיעִיאֵ֗ל בִּכְלֵ֤י נְבָלִים֙ וּבְכִנֹּר֔וֹת וְאָסָ֖ף בַּֽמְצִלְתַּ֥יִם מַשְׁמִֽיעַ׃ (סז) וּבְנָיָ֥הוּ וְיַחֲזִיאֵ֖ל הַכֹּהֲנִ֑ים בַּחֲצֹצְר֣וֹת תָּמִ֔יד לִפְנֵ֖י אֲר֥וֹן בְּרִית־ הָאֱלֹהִֽים׃ (סח) בַּיּ֣וֹם הַה֗וּא אָ֣ז נָתַ֤ן דָּוִיד֙ בָּרֹ֔אשׁ לְהֹד֖וֹת לַיהוָ֑ה בְּיַד־ אָסָ֖ף וְאֶחָֽיו׃ (סט) הוֹד֤וּ לַֽיהוָה֙ קִרְא֣וּ בִשְׁמ֔וֹ הוֹדִ֥יעוּ בָעַמִּ֖ים עֲלִילֹתָֽיו׃ (ע) שִׁ֤ירוּ לוֹ֙ זַמְּרוּ־ ל֔וֹ שִׂ֖יחוּ בְּכָל־ נִפְלְאֹתָֽיו׃ (עא) הִֽתְהַֽלְלוּ֙ בְּשֵׁ֣ם קָדְשׁ֔וֹ יִשְׂמַ֕ח לֵ֖ב מְבַקְשֵׁ֥י יְהוָֽה׃ (עב) דִּרְשׁ֤וּ יְהוָה֙ וְעֻזּ֔וֹ בַּקְּשׁ֥וּ פָנָ֖יו תָּמִֽיד׃ (עג) זִכְר֗וּ נִפְלְאֹתָיו֙ אֲשֶׁ֣ר עָשָׂ֔ה מֹפְתָ֖יו וּמִשְׁפְּטֵי־ פִֽיהוּ׃ (עד) זֶ֚רַע יִשְׂרָאֵ֣ל עַבְדּ֔וֹ בְּנֵ֥י יַעֲקֹ֖ב בְּחִירָֽיו׃ (עה) ה֚וּא יְהוָ֣ה אֱלֹהֵ֔ינוּ בְּכָל־ הָאָ֖רֶץ מִשְׁפָּטָֽיו׃ (עו) זִכְר֤וּ לְעוֹלָם֙ בְּרִית֔וֹ דָּבָ֥ר צִוָּ֖ה לְאֶ֥לֶף דּֽוֹר׃ (עז) אֲשֶׁ֤ר כָּרַת֙ אֶת־ אַבְרָהָ֔ם וּשְׁבוּעָת֖וֹ לְיִצְחָֽק׃ (עח) וַיַּעֲמִידֶ֤הָ לְיַעֲקֹב֙ לְחֹ֔ק לְיִשְׂרָאֵ֖ל בְּרִ֥ית עוֹלָֽם׃ (עט) לֵאמֹ֗ר לְךָ֙ אֶתֵּ֣ן אֶֽרֶץ־ כְּנָ֔עַן חֶ֖בֶל נַחֲלַתְכֶֽם׃ (פ) בִּֽהְיֽוֹתְכֶם֙ מְתֵ֣י מִסְפָּ֔ר כִּמְעַ֖ט וְגָרִ֥ים בָּֽהּ׃ (פא) וַיִּֽתְהַלְּכוּ֙ מִגּ֣וֹי אֶל־ גּ֔וֹי וּמִמַּמְלָכָ֖ה אֶל־ עַ֥ם אַחֵֽר׃ (פב) לֹא־ הִנִּ֤יחַ לְאִישׁ֙ לְעָשְׁקָ֔ם וַיּ֥וֹכַח עֲלֵיהֶ֖ם מְלָכִֽים׃ (פג) אַֽל־ תִּגְּעוּ֙ בִּמְשִׁיחָ֔י וּבִנְבִיאַ֖י אַל־ תָּרֵֽעוּ׃ (פד) שִׁ֤ירוּ לַֽיהוָה֙ כָּל־ הָאָ֔רֶץ בַּשְּׂר֥וּ מִיּֽוֹם־ אֶל־ י֖וֹם יְשׁוּעָתֽוֹ׃ (פה) סַפְּר֤וּ בַגּוֹיִם֙ אֶת־ כְּבוֹד֔וֹ בְּכָל־ הָעַמִּ֖ים נִפְלְאֹתָֽיו׃ (פו) כִּי֩ גָד֨וֹל יְהוָ֤ה וּמְהֻלָּל֙ מְאֹ֔ד וְנוֹרָ֥א ה֖וּא עַל־ כָּל־ אֱלֹהִֽים׃ (פז) כִּ֠י כָּל־ אֱלֹהֵ֤י הָעַמִּים֙ אֱלִילִ֔ים וַיהוָ֖ה שָׁמַ֥יִם עָשָֽׂה׃ (פח) ה֤וֹד וְהָדָר֙ לְפָנָ֔יו עֹ֥ז וְחֶדְוָ֖ה בִּמְקֹמֽוֹ׃ (פט) הָב֤וּ לַֽיהוָה֙ מִשְׁפְּח֣וֹת עַמִּ֔ים הָב֥וּ לַיהוָ֖ה כָּב֥וֹד וָעֹֽז׃ (צ) הָב֥וּ לַיהוָ֖ה כְּב֣וֹד שְׁמ֑וֹ שְׂא֤וּ מִנְחָה֙ וּבֹ֣אוּ לְפָנָ֔יו הִשְׁתַּחֲו֥וּ לַיהוָ֖ה בְּהַדְרַת־ קֹֽדֶשׁ׃ (צא) חִ֤ילוּ מִלְּפָנָיו֙ כָּל־ הָאָ֔רֶץ אַף־ תִּכּ֥וֹן תֵּבֵ֖ל בַּל־ תִּמּֽוֹט׃ (צב) יִשְׂמְח֤וּ הַשָּׁמַ֙יִם֙ וְתָגֵ֣ל הָאָ֔רֶץ וְיֹאמְר֥וּ בַגּוֹיִ֖ם יְהוָ֥ה מָלָֽךְ׃ (צג) יִרְעַ֤ם הַיָּם֙ וּמְלוֹא֔וֹ יַעֲלֹ֥ץ הַשָּׂדֶ֖ה וְכָל־ אֲשֶׁר־ בּֽוֹ׃ (צד) אָ֥ז יְרַנְּנ֖וּ עֲצֵ֣י הַיָּ֑עַר מִלִּפְנֵ֣י יְהוָ֔ה כִּי־ בָ֖א לִשְׁפּ֥וֹט אֶת־ הָאָֽרֶץ׃ (צה) הוֹד֤וּ לַיהוָה֙ כִּ֣י ט֔וֹב כִּ֥י לְעוֹלָ֖ם חַסְדּֽוֹ׃ (צו) וְאִמְר֕וּ הוֹשִׁיעֵ֙נוּ֙ אֱלֹהֵ֣י יִשְׁעֵ֔נוּ וְקַבְּצֵ֥נוּ וְהַצִּילֵ֖נוּ מִן־ הַגּוֹיִ֑ם לְהֹדוֹת֙ לְשֵׁ֣ם קָדְשֶׁ֔ךָ לְהִשְׁתַּבֵּ֖חַ בִּתְהִלָּתֶֽךָ׃
