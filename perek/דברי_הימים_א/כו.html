דברי הימים סדר ט

(א) עַתָּ֗ה תְּנ֤וּ לְבַבְכֶם֙ וְנַפְשְׁכֶ֔ם לִדְר֖וֹשׁ לַיהוָ֣ה אֱלֹהֵיכֶ֑ם וְק֗וּמוּ וּבְנוּ֙ אֶת־ מִקְדַּשׁ֙ יְהוָ֣ה הָֽאֱלֹהִ֔ים לְהָבִ֞יא אֶת־ אֲר֣וֹן בְּרִית־ יְהוָ֗ה וּכְלֵי֙ קֹ֣דֶשׁ הָֽאֱלֹהִ֔ים לַבַּ֖יִת הַנִּבְנֶ֥ה לְשֵׁם־ יְהוָֽה׃ (ב) וְדָוִ֥יד זָקֵ֖ן וְשָׂבַ֣ע יָמִ֑ים וַיַּמְלֵ֛ךְ אֶת־ שְׁלֹמֹ֥ה בְנ֖וֹ עַל־ יִשְׂרָאֵֽל׃ (ג) וַיֶּאֱסֹף֙ אֶת־ כָּל־ שָׂרֵ֣י יִשְׂרָאֵ֔ל וְהַכֹּהֲנִ֖ים וְהַלְוִיִּֽם׃ (ד) וַיִּסָּֽפְרוּ֙ הַלְוִיִּ֔ם מִבֶּ֛ן שְׁלֹשִׁ֥ים שָׁנָ֖ה וָמָ֑עְלָה וַיְהִ֨י מִסְפָּרָ֤ם לְגֻלְגְּלֹתָם֙ לִגְבָרִ֔ים שְׁלֹשִׁ֥ים וּשְׁמוֹנָ֖ה אָֽלֶף׃ (ה) מֵאֵ֗לֶּה לְנַצֵּ֙חַ֙ עַל־ מְלֶ֣אכֶת בֵּית־ יְהוָ֔ה עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָ֑לֶף וְשֹׁטְרִ֥ים וְשֹׁפְטִ֖ים שֵׁ֥שֶׁת אֲלָפִֽים׃ (ו) וְאַרְבַּ֥עַת אֲלָפִ֖ים שֹׁעֲרִ֑ים וְאַרְבַּ֤עַת אֲלָפִים֙ מְהַֽלְלִ֣ים לַיהוָ֔ה בַּכֵּלִ֕ים אֲשֶׁ֥ר עָשִׂ֖יתִי לְהַלֵּֽל׃ (ז) וַיֶּֽחָלְקֵ֥ם דָּוִ֖יד מַחְלְק֑וֹת לִבְנֵ֣י לֵוִ֔י לְגֵרְשׁ֖וֹן קְהָ֥ת וּמְרָרִֽי׃ (ח) לַגֵּרְשֻׁנִּ֖י לַעְדָּ֥ן וְשִׁמְעִֽי׃ (ט) בְּנֵ֣י לַעְדָּ֗ן הָרֹ֧אשׁ יְחִיאֵ֛ל וְזֵתָ֥ם וְיוֹאֵ֖ל שְׁלֹשָֽׁה׃ (י) בְּנֵ֣י שִׁמְעִ֗י (שלמות) שְׁלוֹמִ֧ית וַחֲזִיאֵ֛ל וְהָרָ֖ן שְׁלֹשָׁ֑ה אֵ֛לֶּה רָאשֵׁ֥י הָאָב֖וֹת לְלַעְדָּֽן׃ (יא) וּבְנֵ֣י שִׁמְעִ֔י יַ֣חַת זִינָ֔א וִיע֖וּשׁ וּבְרִיעָ֑ה אֵ֥לֶּה בְנֵי־ שִׁמְעִ֖י אַרְבָּעָֽה׃ (יב) וַֽיְהִי־ יַ֣חַת הָרֹ֔אשׁ וְזִיזָ֖ה הַשֵּׁנִ֑י וִיע֤וּשׁ וּבְרִיעָה֙ לֹֽא־ הִרְבּ֣וּ בָנִ֔ים וַיִּֽהְיוּ֙ לְבֵ֣ית אָ֔ב לִפְקֻדָּ֖ה אֶחָֽת׃ (יג) בְּנֵ֣י קְהָ֗ת עַמְרָ֥ם יִצְהָ֛ר חֶבְר֥וֹן וְעֻזִּיאֵ֖ל אַרְבָּעָֽה׃ (יד) בְּנֵ֥י עַמְרָ֖ם אַהֲרֹ֣ן וּמֹשֶׁ֑ה וַיִּבָּדֵ֣ל אַהֲרֹ֡ן לְֽהַקְדִּישׁוֹ֩ קֹ֨דֶשׁ קָֽדָשִׁ֤ים הֽוּא־ וּבָנָיו֙ עַד־ עוֹלָ֔ם לְהַקְטִיר֩ לִפְנֵ֨י יְהוָ֧ה לְשָׁרְת֛וֹ וּלְבָרֵ֥ךְ בִּשְׁמ֖וֹ עַד־ עוֹלָֽם׃ (טו) וּמֹשֶׁ֖ה אִ֣ישׁ הָאֱלֹהִ֑ים בָּנָ֕יו יִקָּרְא֖וּ עַל־ שֵׁ֥בֶט הַלֵּוִֽי׃ (טז) בְּנֵ֣י מֹשֶׁ֔ה גֵּרְשֹׁ֖ם וֶאֱלִיעֶֽזֶר׃ (יז) בְּנֵ֥י גֵרְשׁ֖וֹם שְׁבוּאֵ֥ל הָרֹֽאשׁ׃ (יח) וַיִּֽהְי֥וּ בְנֵי־ אֱלִיעֶ֖זֶר רְחַבְיָ֣ה הָרֹ֑אשׁ וְלֹא־ הָיָ֤ה לֶאֱלִיעֶ֙זֶר֙ בָּנִ֣ים אֲחֵרִ֔ים וּבְנֵ֥י רְחַבְיָ֖ה רָב֥וּ לְמָֽעְלָה׃ (יט) בְּנֵ֥י יִצְהָ֖ר שְׁלֹמִ֥ית הָרֹֽאשׁ׃ (כ) בְּנֵ֖י חֶבְר֑וֹן יְרִיָּ֤הוּ הָרֹאשׁ֙ אֲמַרְיָ֣ה הַשֵּׁנִ֔י יַחֲזִיאֵל֙ הַשְּׁלִישִׁ֔י וִֽיקַמְעָ֖ם הָרְבִיעִֽי׃ (כא) בְּנֵ֖י עֻזִּיאֵ֑ל מִיכָ֣ה הָרֹ֔אשׁ וְיִשִּׁיָּ֖ה הַשֵּׁנִֽי׃ (כב) בְּנֵ֤י מְרָרִי֙ מַחְלִ֣י וּמוּשִׁ֔י בְּנֵ֥י מַחְלִ֖י אֶלְעָזָ֥ר וְקִֽישׁ׃ (כג) וַיָּ֙מָת֙ אֶלְעָזָ֔ר וְלֹא־ הָ֥יוּ ל֛וֹ בָּנִ֖ים כִּ֣י אִם־ בָּנ֑וֹת וַיִּשָּׂא֥וּם בְּנֵי־ קִ֖ישׁ אֲחֵיהֶֽם׃ (כד) בְּנֵ֣י מוּשִׁ֗י מַחְלִ֥י וְעֵ֛דֶר וִירֵמ֖וֹת שְׁלֹשָֽׁה׃ (כה) אֵ֣לֶּה בְנֵֽי־ לֵוִי֩ לְבֵ֨ית אֲבֹתֵיהֶ֜ם רָאשֵׁ֧י הָאָב֣וֹת לִפְקוּדֵיהֶ֗ם בְּמִסְפַּ֤ר שֵׁמוֹת֙ לְגֻלְגְּלֹתָ֔ם עֹשֵׂה֙ הַמְּלָאכָ֔ה לַעֲבֹדַ֖ת בֵּ֣ית יְהוָ֑ה מִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וָמָֽעְלָה׃ (כו) כִּ֚י אָמַ֣ר דָּוִ֔יד הֵנִ֛יחַ יְהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל לְעַמּ֑וֹ וַיִּשְׁכֹּ֥ן בִּירוּשָׁלִַ֖ם עַד־ לְעוֹלָֽם׃ (כז) וְגַ֖ם לַלְוִיִּ֑ם אֵין־ לָשֵׂ֧את אֶת־ הַמִּשְׁכָּ֛ן וְאֶת־ כָּל־ כֵּלָ֖יו לַעֲבֹדָתֽוֹ׃ (כח) כִּ֣י בְדִבְרֵ֤י דָוִיד֙ הָאַ֣חֲרֹנִ֔ים הֵ֖מָּה מִסְפַּ֣ר בְּנֵי־ לֵוִ֑י מִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וּלְמָֽעְלָה׃ (כט) כִּ֣י מַעֲמָדָ֞ם לְיַד־ בְּנֵ֣י אַהֲרֹ֗ן לַעֲבֹדַת֙ בֵּ֣ית יְהוָ֔ה עַל־ הַחֲצֵרוֹת֙ וְעַל־ הַלְּשָׁכ֔וֹת וְעַֽל־ טָהֳרַ֖ת לְכָל־ קֹ֑דֶשׁ וּמַֽעֲשֵׂ֔ה עֲבֹדַ֖ת בֵּ֥ית הָאֱלֹהִֽים׃ (ל) וּלְלֶ֨חֶם הַֽמַּעֲרֶ֜כֶת וּלְסֹ֤לֶת לְמִנְחָה֙ וְלִרְקִיקֵ֣י הַמַּצּ֔וֹת וְלַֽמַּחֲבַ֖ת וְלַמֻּרְבָּ֑כֶת וּלְכָל־ מְשׂוּרָ֖ה וּמִדָּֽה׃ (לא) וְלַעֲמֹד֙ בַּבֹּ֣קֶר בַּבֹּ֔קֶר לְהֹד֥וֹת וּלְהַלֵּ֖ל לַיהוָ֑ה וְכֵ֖ן לָעָֽרֶב׃ (לב) וּלְכֹ֨ל הַעֲל֤וֹת עֹלוֹת֙ לַיהוָ֔ה לַשַּׁבָּת֔וֹת לֶחֳדָשִׁ֖ים וְלַמֹּעֲדִ֑ים בְּמִסְפָּ֨ר כְּמִשְׁפָּ֧ט עֲלֵיהֶ֛ם תָּמִ֖יד לִפְנֵ֥י יְהוָֽה׃ (לג) וְשָׁמְר֞וּ אֶת־ מִשְׁמֶ֣רֶת אֹֽהֶל־ מוֹעֵ֗ד וְאֵת֙ מִשְׁמֶ֣רֶת הַקֹּ֔דֶשׁ וּמִשְׁמֶ֕רֶת בְּנֵ֥י אַהֲרֹ֖ן אֲחֵיהֶ֑ם לַעֲבֹדַ֖ת בֵּ֥ית יְהוָֽה׃ (לד) וְלִבְנֵ֥י אַהֲרֹ֖ן מַחְלְקוֹתָ֑ם בְּנֵ֣י אַהֲרֹ֔ן נָדָב֙ וַאֲבִיה֔וּא אֶלְעָזָ֖ר וְאִיתָמָֽר׃ (לה) וַיָּ֨מָת נָדָ֤ב וַאֲבִיהוּא֙ לִפְנֵ֣י אֲבִיהֶ֔ם וּבָנִ֖ים לֹא־ הָי֣וּ לָהֶ֑ם וַֽיְכַהֲנ֔וּ אֶלְעָזָ֖ר וְאִיתָמָֽר׃ (לו) וַיֶּֽחָלְקֵ֣ם דָּוִ֔יד וְצָדוֹק֙ מִן־ בְּנֵ֣י אֶלְעָזָ֔ר וַאֲחִימֶ֖לֶךְ מִן־ בְּנֵ֣י אִיתָמָ֑ר לִפְקֻדָּתָ֖ם בַּעֲבֹדָתָֽם׃ (לז) וַיִּמָּצְא֣וּ בְֽנֵי־ אֶלְעָזָ֡ר רַבִּ֞ים לְרָאשֵׁ֧י הַגְּבָרִ֛ים מִן־ בְּנֵ֥י אִיתָמָ֖ר וַֽיַּחְלְק֑וּם לִבְנֵ֨י אֶלְעָזָ֜ר רָאשִׁ֤ים לְבֵית־ אָבוֹת֙ שִׁשָּׁ֣ה עָשָׂ֔ר וְלִבְנֵ֧י אִיתָמָ֛ר לְבֵ֥ית אֲבוֹתָ֖ם שְׁמוֹנָֽה׃ (לח) וַיַּחְלְק֥וּם הַלֵּוִ֗י לִפְנֵ֨י הַמֶּ֤לֶךְ וְהַשָּׂרִים֙ וְצָד֣וֹק הַכֹּהֵ֗ן וַאֲחִימֶ֙לֶךְ֙ בֶּן־ אֶבְיָתָ֔ר וְרָאשֵׁי֙ הָֽאָב֔וֹת לַכֹּהֲנִ֖ים וְלַלְוִיִּ֑ם בֵּֽית־ אָ֣ב אֶחָ֗ד אָחֻז֙ לְאֶלְעָזָ֔ר וְאָחֻ֥ז ׀ אֶפְרַ֔יִם הוֹשֵׁ֖עַ בֶּן־ עֲזַזְיָ֑הוּ לַחֲצִי֙ שֵׁ֣בֶט מְנַשֶּׁ֔ה יוֹאֵ֖ל בֶּן־ פְּדָיָֽהוּ׃ (מט) לַחֲצִ֤י הַֽמְנַשֶּׁה֙ גִּלְעָ֔דָה יִדּ֖וֹ בֶּן־ זְכַרְיָ֑הוּ לְבִנְיָמִ֔ן יַעֲשִׂיאֵ֖ל בֶּן־ אַבְנֵֽר׃ (נ) לְדָ֕ן עֲזַרְאֵ֖ל בֶּן־ יְרֹחָ֑ם אֵ֕לֶּה שָׂרֵ֖י שִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (נא) וְלֹא־ נָשָׂ֤א דָוִיד֙ מִסְפָּרָ֔ם לְמִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וּלְמָ֑טָּה כִּ֚י אָמַ֣ר יְהוָ֔ה לְהַרְבּ֥וֹת אֶת־ יִשְׂרָאֵ֖ל כְּכוֹכְבֵ֥י הַשָּׁמָֽיִם׃ (נב) יוֹאָ֨ב בֶּן־ צְרוּיָ֜ה הֵחֵ֤ל לִמְנוֹת֙ וְלֹ֣א כִלָּ֔ה וַיְהִ֥י בָזֹ֛את קֶ֖צֶף עַל־ יִשְׂרָאֵ֑ל וְלֹ֤א עָלָה֙ הַמִּסְפָּ֔ר בְּמִסְפַּ֥ר דִּבְרֵֽי־ הַיָּמִ֖ים לַמֶּ֥לֶךְ דָּוִֽיד׃ (נג) וְעַל֙ אֹצְר֣וֹת הַמֶּ֔לֶךְ עַזְמָ֖וֶת בֶּן־ עֲדִיאֵ֑ל וְעַ֣ל הָֽאֹצָר֡וֹת בַּשָּׂדֶ֞ה בֶּעָרִ֤ים וּבַכְּפָרִים֙ וּבַמִּגְדָּל֔וֹת יְהוֹנָתָ֖ן בֶּן־ עֻזִּיָּֽהוּ׃ (נד) וְעַ֗ל עֹשֵׂי֙ מְלֶ֣אכֶת הַשָּׂדֶ֔ה לַעֲבֹדַ֖ת הָאֲדָמָ֑ה עֶזְרִ֖י בֶּן־ כְּלֽוּב׃ (נה) וְעַל־ הַ֨כְּרָמִ֔ים שִׁמְעִ֖י הָרָֽמָתִ֑י וְעַ֤ל שֶׁבַּכְּרָמִים֙ לְאֹצְר֣וֹת הַיַּ֔יִן זַבְדִּ֖י הַשִּׁפְמִֽי׃ (נו) וְעַל־ הַזֵּיתִ֤ים וְהַשִּׁקְמִים֙ אֲשֶׁ֣ר בַּשְּׁפֵלָ֔ה בַּ֥עַל חָנָ֖ן הַגְּדֵרִ֑י וְעַל־ אֹצְר֥וֹת הַשֶּׁ֖מֶן יוֹעָֽשׁ׃ (נז) וְעַל־ הַבָּקָר֙ הָרֹעִ֣ים בַּשָּׁר֔וֹן (שטרי) שִׁרְטַ֖י הַשָּׁרוֹנִ֑י וְעַל־ הַבָּקָר֙ בָּֽעֲמָקִ֔ים שָׁפָ֖ט בֶּן־ עַדְלָֽי׃ (נח) וְעַל־ הַ֨גְּמַלִּ֔ים אוֹבִ֖יל הַיִּשְׁמְעֵלִ֑י וְעַל־ הָ֣אֲתֹנ֔וֹת יֶחְדְּיָ֖הוּ הַמֵּרֹנֹתִֽי׃ (נט) וְעַל־ הַצֹּ֖אן יָזִ֣יז הַֽהַגְרִ֑י כָּל־ אֵ֙לֶּה֙ שָׂרֵ֣י הָרְכ֔וּשׁ אֲשֶׁ֖ר לַמֶּ֥לֶךְ דָּוִֽיד׃ (ס) וִֽיהוֹנָתָ֤ן דּוֹד־ דָּוִיד֙ יוֹעֵ֔ץ אִישׁ־ מֵבִ֥ין וְסוֹפֵ֖ר ה֑וּא וִֽיחִיאֵ֥ל בֶּן־ חַכְמוֹנִ֖י עִם־ בְּנֵ֥י הַמֶּֽלֶךְ׃ (סא) וַאֲחִיתֹ֖פֶל יוֹעֵ֣ץ לַמֶּ֑לֶךְ וְחוּשַׁ֥י הָאַרְכִּ֖י רֵ֥עַ הַמֶּֽלֶךְ׃ (סב) וְאַחֲרֵ֣י אֲחִיתֹ֗פֶל יְהוֹיָדָ֤ע בֶּן־ בְּנָיָ֙הוּ֙ וְאֶבְיָתָ֔ר וְשַׂר־ צָבָ֥א לַמֶּ֖לֶךְ יוֹאָֽב׃ (סג) וַיַּקְהֵ֣ל דָּוִ֣יד אֶת־ כָּל־ שָׂרֵ֣י יִשְׂרָאֵ֡ל שָׂרֵ֣י הַשְּׁבָטִ֣ים וְשָׂרֵ֣י הַמַּחְלְק֣וֹת הַמְשָׁרְתִ֪ים אֶת־ הַמֶּ֟לֶךְ וְשָׂרֵ֣י הָאֲלָפִ֣ים וְשָׂרֵ֣י הַמֵּא֡וֹת וְשָׂרֵ֣י כָל־ רְכוּשׁ־ וּמִקְנֶה֩ ׀ לַמֶּ֨לֶךְ וּלְבָנָ֜יו עִם־ הַסָּרִיסִ֧ים וְהַגִּבּוֹרִ֛ים וּֽלְכָל־ גִּבּ֥וֹר חָ֖יִל אֶל־ יְרוּשָׁלִָֽם׃ (סד) וַיָּ֨קָם דָּוִ֤יד הַמֶּ֙לֶךְ֙ עַל־ רַגְלָ֔יו וַיֹּ֕אמֶר שְׁמָע֖וּנִי אַחַ֣י וְעַמִּ֑י אֲנִ֣י עִם־ לְבָבִ֡י לִבְנוֹת֩ בֵּ֨ית מְנוּחָ֜ה לַאֲר֣וֹן בְּרִית־ יְהוָ֗ה וְלַהֲדֹם֙ רַגְלֵ֣י אֱלֹהֵ֔ינוּ וַהֲכִינ֖וֹתִי לִבְנֽוֹת׃ (סה) וְהָאֱלֹהִים֙ אָ֣מַר לִ֔י לֹא־ תִבְנֶ֥ה בַ֖יִת לִשְׁמִ֑י כִּ֣י אִ֧ישׁ מִלְחָמ֛וֹת אַ֖תָּה וְדָמִ֥ים שָׁפָֽכְתָּ׃ (סו) וַיִּבְחַ֡ר יְהוָ֣ה אֱלֹהֵי֩ יִשְׂרָאֵ֨ל בִּ֜י מִכֹּ֣ל בֵּית־ אָבִ֗י לִהְי֨וֹת לְמֶ֤לֶךְ עַל־ יִשְׂרָאֵל֙ לְעוֹלָ֔ם כִּ֤י בִֽיהוּדָה֙ בָּחַ֣ר לְנָגִ֔יד וּבְבֵ֥ית יְהוּדָ֖ה בֵּ֣ית אָבִ֑י וּבִבְנֵ֣י אָבִ֔י בִּ֣י רָצָ֔ה לְהַמְלִ֖יךְ עַל־ כָּל־ יִשְׂרָאֵֽל׃ (סז) וּמִכָּ֨ל־ בָּנַ֔י כִּ֚י רַבִּ֣ים בָּנִ֔ים נָ֥תַן לִ֖י יְהוָ֑ה וַיִּבְחַר֙ בִּשְׁלֹמֹ֣ה בְנִ֔י לָשֶׁ֗בֶת עַל־ כִּסֵּ֛א מַלְכ֥וּת יְהוָ֖ה עַל־ יִשְׂרָאֵֽל׃ (סח) וַיֹּ֣אמֶר לִ֔י שְׁלֹמֹ֣ה בִנְךָ֔ הֽוּא־ יִבְנֶ֥ה בֵיתִ֖י וַחֲצֵרוֹתָ֑י כִּי־ בָחַ֨רְתִּי ב֥וֹ לִי֙ לְבֵ֔ן וַאֲנִ֖י אֶֽהְיֶה־ לּ֥וֹ לְאָֽב׃ (סט) וַהֲכִינוֹתִ֥י אֶת־ מַלְכוּת֖וֹ עַד־ לְעוֹלָ֑ם אִם־ יֶחֱזַ֗ק לַעֲשׂ֛וֹת מִצְוֺתַ֥י וּמִשְׁפָּטַ֖י כַּיּ֥וֹם הַזֶּֽה׃ (ע) וְ֠עַתָּה לְעֵינֵ֨י כָל־ יִשְׂרָאֵ֤ל קְהַל־ יְהוָה֙ וּבְאָזְנֵ֣י אֱלֹהֵ֔ינוּ שִׁמְר֣וּ וְדִרְשׁ֔וּ כָּל־ מִצְוֺ֖ת יְהוָ֣ה אֱלֹהֵיכֶ֑ם לְמַ֤עַן תִּֽירְשׁוּ֙ אֶת־ הָאָ֣רֶץ הַטּוֹבָ֔ה וְהִנְחַלְתֶּ֛ם לִבְנֵיכֶ֥ם אַחֲרֵיכֶ֖ם עַד־ עוֹלָֽם׃ (עא) וְאַתָּ֣ה שְׁלֹמֹֽה־ בְנִ֡י דַּע֩ אֶת־ אֱלֹהֵ֨י אָבִ֜יךָ וְעָבְדֵ֗הוּ בְּלֵ֤ב שָׁלֵם֙ וּבְנֶ֣פֶשׁ חֲפֵצָ֔ה כִּ֤י כָל־ לְבָבוֹת֙ דּוֹרֵ֣שׁ יְהוָ֔ה וְכָל־ יֵ֥צֶר מַחֲשָׁב֖וֹת מֵבִ֑ין אִֽם־ תִּדְרְשֶׁ֙נּוּ֙ יִמָּ֣צֵא לָ֔ךְ וְאִם־ תַּֽעַזְבֶ֖נּוּ יַזְנִיחֲךָ֥ לָעַֽד׃
