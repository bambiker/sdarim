פרק ח בספר עזרא (36 פסוקים) נמצא בעזרא ונחמיה סדר ג (פסוקים לד–סז) ובעזרא ונחמיה סדר ד (פסוקים א–ב).

עזרא ונחמיה סדר ג

(א) וַהֲקִ֨ימוּ כָהֲנַיָּ֜א בִּפְלֻגָּתְה֗וֹן וְלֵוָיֵא֙ בְּמַחְלְקָ֣תְה֔וֹן עַל־ עֲבִידַ֥ת אֱלָהָ֖א דִּ֣י בִירוּשְׁלֶ֑ם כִּכְתָ֖ב סְפַ֥ר מֹשֶֽׁה׃ (ב) וַיַּעֲשׂ֥וּ בְנֵי־ הַגּוֹלָ֖ה אֶת־ הַפָּ֑סַח בְּאַרְבָּעָ֥ה עָשָׂ֖ר לַחֹ֥דֶשׁ הָרִאשֽׁוֹן׃ (ג) כִּ֣י הִֽטַּהֲר֞וּ הַכֹּהֲנִ֧ים וְהַלְוִיִּ֛ם כְּאֶחָ֖ד כֻּלָּ֣ם טְהוֹרִ֑ים וַיִּשְׁחֲט֤וּ הַפֶּ֙סַח֙ לְכָל־ בְּנֵ֣י הַגּוֹלָ֔ה וְלַאֲחֵיהֶ֥ם הַכֹּהֲנִ֖ים וְלָהֶֽם׃ (ד) וַיֹּאכְל֣וּ בְנֵֽי־ יִשְׂרָאֵ֗ל הַשָּׁבִים֙ מֵֽהַגּוֹלָ֔ה וְכֹ֗ל הַנִּבְדָּ֛ל מִטֻּמְאַ֥ת גּוֹיֵֽ־ הָאָ֖רֶץ אֲלֵהֶ֑ם לִדְרֹ֕שׁ לַֽיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (ה) וַיַּֽעֲשׂ֧וּ חַג־ מַצּ֛וֹת שִׁבְעַ֥ת יָמִ֖ים בְּשִׂמְחָ֑ה כִּ֣י ׀ שִׂמְּחָ֣ם יְהוָ֗ה וְֽהֵסֵ֞ב לֵ֤ב מֶֽלֶךְ־ אַשּׁוּר֙ עֲלֵיהֶ֔ם לְחַזֵּ֣ק יְדֵיהֶ֔ם בִּמְלֶ֥אכֶת בֵּית־ הָאֱלֹהִ֖ים אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (ו) וְאַחַר֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה בְּמַלְכ֖וּת אַרְתַּחְשַׁ֣סְתְּא מֶֽלֶךְ־ פָּרָ֑ס עֶזְרָא֙ בֶּן־ שְׂרָיָ֔ה בֶּן־ עֲזַרְיָ֖ה בֶּן־ חִלְקִיָּֽה׃ (ז) בֶּן־ שַׁלּ֥וּם בֶּן־ צָד֖וֹק בֶּן־ אֲחִיטֽוּב׃ (ח) בֶּן־ אֲמַרְיָ֥ה בֶן־ עֲזַרְיָ֖ה בֶּן־ מְרָיֽוֹת׃ (ט) בֶּן־ זְרַֽחְיָ֥ה בֶן־ עֻזִּ֖י בֶּן־ בֻּקִּֽי׃ (י) בֶּן־ אֲבִישׁ֗וּעַ בֶּן־ פִּֽינְחָס֙ בֶּן־ אֶלְעָזָ֔ר בֶּן־ אַהֲרֹ֥ן הַכֹּהֵ֖ן הָרֹֽאשׁ׃ (יא) ה֤וּא עֶזְרָא֙ עָלָ֣ה מִבָּבֶ֔ל וְהֽוּא־ סֹפֵ֤ר מָהִיר֙ בְּתוֹרַ֣ת מֹשֶׁ֔ה אֲשֶׁר־ נָתַ֥ן יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וַיִּתֶּן־ ל֣וֹ הַמֶּ֗לֶךְ כְּיַד־ יְהוָ֤ה אֱלֹהָיו֙ עָלָ֔יו כֹּ֖ל בַּקָּשָׁתֽוֹ׃ (יב) וַיַּֽעֲל֣וּ מִבְּנֵֽי־ יִ֠שְׂרָאֵל וּמִן־ הַכֹּהֲנִ֨ים וְהַלְוִיִּ֜ם וְהַמְשֹׁרְרִ֧ים וְהַשֹּׁעֲרִ֛ים וְהַנְּתִינִ֖ים אֶל־ יְרוּשָׁלִָ֑ם בִּשְׁנַת־ שֶׁ֖בַע לְאַרְתַּחְשַׁ֥סְתְּא הַמֶּֽלֶךְ׃ (יג) וַיָּבֹ֥א יְרוּשָׁלִַ֖ם בַּחֹ֣דֶשׁ הַחֲמִישִׁ֑י הִ֛יא שְׁנַ֥ת הַשְּׁבִיעִ֖ית לַמֶּֽלֶךְ׃ (יד) כִּ֗י בְּאֶחָד֙ לַחֹ֣דֶשׁ הָרִאשׁ֔וֹן ה֣וּא יְסֻ֔ד הַֽמַּעֲלָ֖ה מִבָּבֶ֑ל וּבְאֶחָ֞ד לַחֹ֣דֶשׁ הַחֲמִישִׁ֗י בָּ֚א אֶל־ יְר֣וּשָׁלִַ֔ם כְּיַד־ אֱלֹהָ֖יו הַטּוֹבָ֥ה עָלָֽיו׃ (טו) כִּ֤י עֶזְרָא֙ הֵכִ֣ין לְבָב֔וֹ לִדְר֛וֹשׁ אֶת־ תּוֹרַ֥ת יְהוָ֖ה וְלַעֲשֹׂ֑ת וּלְלַמֵּ֥ד בְּיִשְׂרָאֵ֖ל חֹ֥ק וּמִשְׁפָּֽט׃ (טז) וְזֶ֣ה ׀ פַּרְשֶׁ֣גֶן הַֽנִּשְׁתְּוָ֗ן אֲשֶׁ֤ר נָתַן֙ הַמֶּ֣לֶךְ אַרְתַּחְשַׁ֔סְתְּא לְעֶזְרָ֥א הַכֹּהֵ֖ן הַסֹּפֵ֑ר סֹפֵ֞ר דִּבְרֵ֧י מִצְוֺת־ יְהוָ֛ה וְחֻקָּ֖יו עַל־ יִשְׂרָאֵֽל׃ (יז) אַ֨רְתַּחְשַׁ֔סְתְּא מֶ֖לֶךְ מַלְכַיָּ֑א לְעֶזְרָ֣א כָ֠הֲנָא סָפַ֨ר דָּתָ֜א דִּֽי־ אֱלָ֧הּ שְׁמַיָּ֛א גְּמִ֖יר וּכְעֶֽנֶת׃ (יח) מִנִּי֮ שִׂ֣ים טְעֵם֒ דִּ֣י כָל־ מִתְנַדַּ֣ב בְּמַלְכוּתִי֩ מִן־ עַמָּ֨ה יִשְׂרָאֵ֜ל וְכָהֲנ֣וֹהִי וְלֵוָיֵ֗א לִמְהָ֧ךְ לִֽירוּשְׁלֶ֛ם עִמָּ֖ךְ יְהָֽךְ׃ (יט) כָּל־ קֳבֵ֗ל דִּי֩ מִן־ קֳדָ֨ם מַלְכָּ֜א וְשִׁבְעַ֤ת יָעֲטֹ֙הִי֙ שְׁלִ֔יחַ לְבַקָּרָ֥א עַל־ יְה֖וּד וְלִֽירוּשְׁלֶ֑ם בְּדָ֥ת אֱלָהָ֖ךְ דִּ֥י בִידָֽךְ׃ (כ) וּלְהֵיבָלָ֖ה כְּסַ֣ף וּדְהַ֑ב דִּֽי־ מַלְכָּ֣א וְיָעֲט֗וֹהִי הִתְנַדַּ֙בוּ֙ לֶאֱלָ֣הּ יִשְׂרָאֵ֔ל דִּ֥י בִֽירוּשְׁלֶ֖ם מִשְׁכְּנֵֽהּ׃ (כא) וְכֹל֙ כְּסַ֣ף וּדְהַ֔ב דִּ֣י תְהַשְׁכַּ֔ח בְּכֹ֖ל מְדִינַ֣ת בָּבֶ֑ל עִם֩ הִתְנַדָּב֨וּת עַמָּ֤א וְכָֽהֲנַיָּא֙ מִֽתְנַדְּבִ֔ין לְבֵ֥ית אֱלָהֲהֹ֖ם דִּ֥י בִירוּשְׁלֶֽם׃ (כב) כָּל־ קֳבֵ֣ל דְּנָה֩ אָסְפַּ֨רְנָא תִקְנֵ֜א בְּכַסְפָּ֣א דְנָ֗ה תּוֹרִ֤ין ׀ דִּכְרִין֙ אִמְּרִ֔ין וּמִנְחָתְה֖וֹן וְנִסְכֵּיה֑וֹן וּתְקָרֵ֣ב הִמּ֔וֹ עַֽל־ מַדְבְּחָ֔ה דִּ֛י בֵּ֥ית אֱלָהֲכֹ֖ם דִּ֥י בִירוּשְׁלֶֽם׃ (כג) וּמָ֣ה דִי֩ (עליך) עֲלָ֨ךְ וְעַל־ (אחיך) אֶחָ֜ךְ יֵיטַ֗ב בִּשְׁאָ֛ר כַּסְפָּ֥א וְדַהֲבָ֖ה לְמֶעְבַּ֑ד כִּרְע֥וּת אֱלָהֲכֹ֖ם תַּעַבְדֽוּן׃ (כד) וּמָֽאנַיָּא֙ דִּֽי־ מִתְיַהֲבִ֣ין לָ֔ךְ לְפָלְחָ֖ן בֵּ֣ית אֱלָהָ֑ךְ הַשְׁלֵ֕ם קֳדָ֖ם אֱלָ֥הּ יְרוּשְׁלֶֽם׃ (כה) וּשְׁאָ֗ר חַשְׁחוּת֙ בֵּ֣ית אֱלָהָ֔ךְ דִּ֥י יִפֶּל־ לָ֖ךְ לְמִנְתַּ֑ן תִּנְתֵּ֕ן מִן־ בֵּ֖ית גִּנְזֵ֥י מַלְכָּֽא׃ (כו) וּ֠מִנִּי אֲנָ֞ה אַרְתַּחְשַׁ֤סְתְּא מַלְכָּא֙ שִׂ֣ים טְעֵ֔ם לְכֹל֙ גִּזַּֽבְרַיָּ֔א דִּ֖י בַּעֲבַ֣ר נַהֲרָ֑ה דִּ֣י כָל־ דִּ֣י יִ֠שְׁאֲלֶנְכוֹן עֶזְרָ֨א כָהֲנָ֜ה סָפַ֤ר דָּתָא֙ דִּֽי־ אֱלָ֣הּ שְׁמַיָּ֔א אָסְפַּ֖רְנָא יִתְעֲבִֽד׃ (כז) עַד־ כְּסַף֮ כַּכְּרִ֣ין מְאָה֒ וְעַד־ חִנְטִין֙ כֹּרִ֣ין מְאָ֔ה וְעַד־ חֲמַר֙ בַּתִּ֣ין מְאָ֔ה וְעַד־ בַּתִּ֥ין מְשַׁ֖ח מְאָ֑ה וּמְלַ֖ח דִּי־ לָ֥א כְתָֽב׃ (כח) כָּל־ דִּ֗י מִן־ טַ֙עַם֙ אֱלָ֣הּ שְׁמַיָּ֔א יִתְעֲבֵד֙ אַדְרַזְדָּ֔א לְבֵ֖ית אֱלָ֣הּ שְׁמַיָּ֑א דִּֽי־ לְמָ֤ה לֶֽהֱוֵא֙ קְצַ֔ף עַל־ מַלְכ֥וּת מַלְכָּ֖א וּבְנֽוֹהִי׃ (כט) וּלְכֹ֣ם מְהוֹדְעִ֗ין דִּ֣י כָל־ כָּהֲנַיָּ֣א וְ֠לֵוָיֵא זַמָּ֨רַיָּ֤א תָרָֽעַיָּא֙ נְתִ֣ינַיָּ֔א וּפָ֣לְחֵ֔י בֵּ֖ית אֱלָהָ֣א דְנָ֑ה מִנְדָּ֤ה בְלוֹ֙ וַהֲלָ֔ךְ לָ֥א שַׁלִּ֖יט לְמִרְמֵ֥א עֲלֵיהֹֽם׃ (ל) וְאַ֣נְתְּ עֶזְרָ֗א כְּחָכְמַ֨ת אֱלָהָ֤ךְ דִּֽי־ בִידָךְ֙ מֶ֣נִּי שָׁפְטִ֞ין וְדַיָּנִ֗ין דִּי־ לֶהֱוֺ֤ן (דאנין) דָּאיְנִין֙ לְכָל־ עַמָּה֙ דִּ֚י בַּעֲבַ֣ר נַהֲרָ֔ה לְכָל־ יָדְעֵ֖י דָּתֵ֣י אֱלָהָ֑ךְ וְדִ֧י לָ֦א יָדַ֖ע תְּהוֹדְעֽוּן׃ (לא) וְכָל־ דִּי־ לָא֩ לֶהֱוֵ֨א עָבֵ֜ד דָּתָ֣א דִֽי־ אֱלָהָ֗ךְ וְדָתָא֙ דִּ֣י מַלְכָּ֔א אָסְפַּ֕רְנָא דִּינָ֕ה לֶהֱוֵ֥א מִתְעֲבֵ֖ד מִנֵּ֑הּ הֵ֤ן לְמוֹת֙ הֵ֣ן (לשרשו) לִשְׁרֹשִׁ֔י הֵן־ לַעֲנָ֥שׁ נִכְסִ֖ין וְלֶאֱסוּרִֽין׃ (לב) בָּר֥וּךְ יְהוָ֖ה אֱלֹהֵ֣י אֲבוֹתֵ֑ינוּ אֲשֶׁ֨ר נָתַ֤ן כָּזֹאת֙ בְּלֵ֣ב הַמֶּ֔לֶךְ לְפָאֵ֕ר אֶת־ בֵּ֥ית יְהוָ֖ה אֲשֶׁ֥ר בִּירוּשָׁלִָֽם׃ (לג) וְעָלַ֣י הִטָּה־ חֶ֗סֶד לִפְנֵ֤י הַמֶּ֙לֶךְ֙ וְיֽוֹעֲצָ֔יו וּלְכָל־ שָׂרֵ֥י הַמֶּ֖לֶךְ הַגִּבֹּרִ֑ים וַאֲנִ֣י הִתְחַזַּ֗קְתִּי כְּיַד־ יְהוָ֤ה אֱלֹהַי֙ עָלַ֔י וָאֶקְבְּצָ֧ה מִיִּשְׂרָאֵ֛ל רָאשִׁ֖ים לַעֲל֥וֹת עִמִּֽי׃ תחילת פרק ח (לד) מַֽעֲשֵׂיָה֙ וֶֽאֱלִיעֶ֔זֶר וְיָרִ֖יב וּגְדַלְיָֽה׃ (לו) וַיִּתְּנ֥וּ יָדָ֖ם לְהוֹצִ֣יא נְשֵׁיהֶ֑ם וַאֲשֵׁמִ֥ים אֵֽיל־ צֹ֖אן עַל־ אַשְׁמָתָֽם׃ (לז) וּמִבְּנֵ֣י אִמֵּ֔ר חֲנָ֖נִי וּזְבַדְיָֽה׃ (לח) וּמִבְּנֵ֖י חָרִ֑ם מַעֲשֵׂיָ֤ה וְאֵֽלִיָּה֙ וּֽשְׁמַֽעְיָ֔ה וִיחִיאֵ֖ל וְעֻזִיָּֽה׃ (לט) וּמִבְּנֵ֖י פַּשְׁח֑וּר אֶלְיוֹעֵינַ֤י מַֽעֲשֵׂיָה֙ יִשְׁמָעֵ֣אל נְתַנְאֵ֔ל יוֹזָבָ֖ד וְאֶלְעָשָֽׂה׃ (מ) וּמִֽן־ הַלְוִיִּ֑ם יוֹזָבָ֣ד וְשִׁמְעִ֗י וְקֵֽלָיָה֙ ה֣וּא קְלִיטָ֔א פְּתַֽחְיָ֥ה יְהוּדָ֖ה וֶאֱלִיעֶֽזֶר׃ (מא) וּמִן־ הַמְשֹׁרְרִ֖ים אֶלְיָשִׁ֑יב וּמִן־ הַשֹּׁ֣עֲרִ֔ים שַׁלֻּ֥ם וָטֶ֖לֶם וְאוּרִֽי׃ (מב) וּמִֽיִּשְׂרָאֵ֑ל מִבְּנֵ֣י פַרְעֹ֡שׁ רַ֠מְיָה וְיִזִּיָּ֤ה וּמַלְכִּיָּה֙ וּמִיָּמִ֣ן וְאֶלְעָזָ֔ר וּמַלְכִּיָּ֖ה וּבְנָיָֽה׃ (מג) וּמִבְּנֵ֖י עֵילָ֑ם מַתַּנְיָ֤ה זְכַרְיָה֙ וִיחִיאֵ֣ל וְעַבְדִּ֔י וִירֵמ֖וֹת וְאֵלִיָּֽה׃ (מד) וּמִבְּנֵ֖י זַתּ֑וּא אֶלְיוֹעֵנַ֤י אֶלְיָשִׁיב֙ מַתַּנְיָ֣ה וִֽירֵמ֔וֹת וְזָבָ֖ד וַעֲזִיזָֽא׃ (מה) וּמִבְּנֵ֖י בֵּבָ֑י יְהוֹחָנָ֥ן חֲנַנְיָ֖ה זַבַּ֥י עַתְלָֽי׃ (מו) וּמִבְּנֵ֖י בָּנִ֑י מְשֻׁלָּ֤ם מַלּוּךְ֙ וַעֲדָיָ֔ה יָשׁ֖וּב וּשְׁאָ֥ל (ירמות) וְרָמֽוֹת׃ (מז) וּמִבְּנֵ֛י פַּחַ֥ת מוֹאָ֖ב עַדְנָ֣א וּכְלָ֑ל בְּנָיָ֤ה מַעֲשֵׂיָה֙ מַתַּנְיָ֣ה בְצַלְאֵ֔ל וּבִנּ֖וּי וּמְנַשֶּֽׁה׃ (מח) וּבְנֵ֖י חָרִ֑ם אֱלִיעֶ֧זֶר יִשִּׁיָּ֛ה מַלְכִּיָּ֖ה שְׁמַֽעְיָ֥ה שִׁמְעֽוֹן׃ (מט) בְּנְיָמִ֥ן מַלּ֖וּךְ שְׁמַרְיָֽה׃ (נ) מִבְּנֵ֖י חָשֻׁ֑ם מַתְּנַ֤י מַתַּתָּה֙ זָבָ֣ד אֱלִיפֶ֔לֶט יְרֵמַ֥י מְנַשֶּׁ֖ה שִׁמְעִֽי׃ (נא) מִבְּנֵ֣י בָנִ֔י מַעֲדַ֥י עַמְרָ֖ם וְאוּאֵֽל׃ (נב) בְּנָיָ֥ה בֵדְיָ֖ה (כלהי) כְּלֽוּהוּ׃ (נג) וַנְיָ֥ה מְרֵמ֖וֹת אֶלְיָשִֽׁיב׃ (נד) מַתַּנְיָ֥ה מַתְּנַ֖י (ויעשו) וְיַעֲשָֽׂי׃ (נה) וּבָנִ֥י וּבִנּ֖וּי שִׁמְעִֽי׃ (נו) וְשֶֽׁלֶמְיָ֥ה וְנָתָ֖ן וַעֲדָיָֽה׃ (נז) מַכְנַדְבַ֥י שָׁשַׁ֖י שָׁרָֽי׃ (נח) עֲזַרְאֵ֥ל וְשֶׁלֶמְיָ֖הוּ שְׁמַרְיָֽה׃ (נט) שַׁלּ֥וּם אֲמַרְיָ֖ה יוֹסֵֽף׃ (ס) מִבְּנֵ֖י נְב֑וֹ יְעִיאֵ֤ל מַתִּתְיָה֙ זָבָ֣ד זְבִינָ֔א (ידו) יַדַּ֥י וְיוֹאֵ֖ל בְּנָיָֽה׃ (סא) כָּל־ אֵ֕לֶּה (נשאי) נָשְׂא֖וּ נָשִׁ֣ים נָכְרִיּ֑וֹת וְיֵ֣שׁ מֵהֶ֣ם נָשִׁ֔ים וַיָּשִׂ֖ימוּ בָּנִֽים׃ (סב) דִּבְרֵ֥י נְחֶמְיָ֖ה בֶּן־ חֲכַלְיָ֑ה וַיְהִ֤י בְחֹֽדֶשׁ־ (כסלו) כִּסְלֵיו֙ שְׁנַ֣ת עֶשְׂרִ֔ים וַאֲנִ֥י הָיִ֖יתִי בְּשׁוּשַׁ֥ן הַבִּירָֽה׃ (סג) וַיָּבֹ֨א חֲנָ֜נִי אֶחָ֧ד מֵאַחַ֛י ה֥וּא וַאֲנָשִׁ֖ים מִֽיהוּדָ֑ה וָאֶשְׁאָלֵ֞ם עַל־ הַיְּהוּדִ֧ים הַפְּלֵיטָ֛ה אֲשֶֽׁר־ נִשְׁאֲר֥וּ מִן־ הַשֶּׁ֖בִי וְעַל־ יְרוּשָׁלִָֽם׃ (סד) וַיֹּאמְרוּ֮ לִי֒ הַֽנִּשְׁאָרִ֞ים אֲשֶֽׁר־ נִשְׁאֲר֤וּ מִן־ הַשְּׁבִי֙ שָׁ֣ם בַּמְּדִינָ֔ה בְּרָעָ֥ה גְדֹלָ֖ה וּבְחֶרְפָּ֑ה וְחוֹמַ֤ת יְרוּשָׁלִַ֙ם֙ מְפֹרָ֔צֶת וּשְׁעָרֶ֖יהָ נִצְּת֥וּ בָאֵֽשׁ׃ (סה) וַיְהִ֞י כְּשָׁמְעִ֣י ׀ אֶת־ הַדְּבָרִ֣ים הָאֵ֗לֶּה יָשַׁ֙בְתִּי֙ וָֽאֶבְכֶּ֔ה וָאֶתְאַבְּלָ֖ה יָמִ֑ים וָֽאֱהִ֥י צָם֙ וּמִתְפַּלֵּ֔ל לִפְנֵ֖י אֱלֹהֵ֥י הַשָּׁמָֽיִם׃ (סו) וָאֹמַ֗ר אָֽנָּ֤א יְהוָה֙ אֱלֹהֵ֣י הַשָּׁמַ֔יִם הָאֵ֥ל הַגָּד֖וֹל וְהַנּוֹרָ֑א שֹׁמֵ֤ר הַבְּרִית֙ וָחֶ֔סֶד לְאֹהֲבָ֖יו וּלְשֹׁמְרֵ֥י מִצְוֺתָֽיו׃ (סז) תְּהִ֣י נָ֣א אָזְנְךָֽ־ קַשֶּׁ֣בֶת וְֽעֵינֶ֪יךָ פְתֻוּח֟וֹת לִשְׁמֹ֣עַ אֶל־ תְּפִלַּ֣ת עַבְדְּךָ֡ אֲשֶׁ֣ר אָנֹכִי֩ מִתְפַּלֵּ֨ל לְפָנֶ֤יךָ הַיּוֹם֙ יוֹמָ֣ם וָלַ֔יְלָה עַל־ בְּנֵ֥י יִשְׂרָאֵ֖ל עֲבָדֶ֑יךָ וּמִתְוַדֶּ֗ה עַל־ חַטֹּ֤אות בְּנֵֽי־ יִשְׂרָאֵל֙ אֲשֶׁ֣ר חָטָ֣אנוּ לָ֔ךְ וַאֲנִ֥י וּבֵית־ אָבִ֖י חָטָֽאנוּ׃ (סח) חֲבֹ֖ל חָבַ֣לְנוּ לָ֑ךְ וְלֹא־ שָׁמַ֣רְנוּ אֶת־ הַמִּצְוֺ֗ת וְאֶת־ הַֽחֻקִּים֙ וְאֶת־ הַמִּשְׁפָּטִ֔ים אֲשֶׁ֥ר צִוִּ֖יתָ אֶת־ מֹשֶׁ֥ה עַבְדֶּֽךָ׃ (סט) זְכָר־ נָא֙ אֶת־ הַדָּבָ֔ר אֲשֶׁ֥ר צִוִּ֛יתָ אֶת־ מֹשֶׁ֥ה עַבְדְּךָ֖ לֵאמֹ֑ר אַתֶּ֣ם תִּמְעָ֔לוּ אֲנִ֕י אָפִ֥יץ אֶתְכֶ֖ם בָּעַמִּֽים׃ (ע) וְשַׁבְתֶּ֣ם אֵלַ֔י וּשְׁמַרְתֶּם֙ מִצְוֺתַ֔י וַעֲשִׂיתֶ֖ם אֹתָ֑ם אִם־ יִהְיֶ֨ה נִֽדַּחֲכֶ֜ם בִּקְצֵ֤ה הַשָּׁמַ֙יִם֙ מִשָּׁ֣ם אֲקַבְּצֵ֔ם (והבואתים) וַהֲבִֽיאוֹתִים֙ אֶל־ הַמָּק֔וֹם אֲשֶׁ֣ר בָּחַ֔רְתִּי לְשַׁכֵּ֥ן אֶת־ שְׁמִ֖י שָֽׁם׃ (עא) וְהֵ֥ם עֲבָדֶ֖יךָ וְעַמֶּ֑ךָ אֲשֶׁ֤ר פָּדִ֙יתָ֙ בְּכֹחֲךָ֣
