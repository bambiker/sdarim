עמוס פרק ז לפי החלוקה לסדרים

פרק ז בספר עמוס (17 פסוקים) נמצא בתרי עשר סדר ז (פסוקים כט–מב) ובתרי עשר סדר ח (פסוקים א–ג).

תרי עשר סדר ז

(א) דִּרְשׁוּ־ ט֥וֹב וְאַל־ רָ֖ע לְמַ֣עַן תִּֽחְי֑וּ וִיהִי־ כֵ֞ן יְהוָ֧ה אֱלֹהֵֽי־ צְבָא֛וֹת אִתְּכֶ֖ם כַּאֲשֶׁ֥ר אֲמַרְתֶּֽם׃ (ב) שִׂנְאוּ־ רָע֙ וְאֶ֣הֱבוּ ט֔וֹב וְהַצִּ֥יגוּ בַשַּׁ֖עַר מִשְׁפָּ֑ט אוּלַ֗י יֶֽחֱנַ֛ן יְהוָ֥ה אֱלֹהֵֽי־ צְבָא֖וֹת שְׁאֵרִ֥ית יוֹסֵֽף׃ (ג) לָ֠כֵן כֹּֽה־ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֤י צְבָאוֹת֙ אֲדֹנָ֔י בְּכָל־ רְחֹב֣וֹת מִסְפֵּ֔ד וּבְכָל־ חוּצ֖וֹת יֹאמְר֣וּ הוֹ־ ה֑וֹ וְקָרְא֤וּ אִכָּר֙ אֶל־ אֵ֔בֶל וּמִסְפֵּ֖ד אֶל־ י֥וֹדְעֵי נֶֽהִי׃ (ד) וּבְכָל־ כְּרָמִ֖ים מִסְפֵּ֑ד כִּֽי־ אֶעֱבֹ֥ר בְּקִרְבְּךָ֖ אָמַ֥ר יְהוָֽה׃ (ה) ה֥וֹי הַמִּתְאַוִּ֖ים אֶת־ י֣וֹם יְהוָ֑ה לָמָּה־ זֶּ֥ה לָכֶ֛ם י֥וֹם יְהוָ֖ה הוּא־ חֹ֥שֶׁךְ וְלֹא־ אֽוֹר׃ (ו) כַּאֲשֶׁ֨ר יָנ֥וּס אִישׁ֙ מִפְּנֵ֣י הָאֲרִ֔י וּפְגָע֖וֹ הַדֹּ֑ב וּבָ֣א הַבַּ֔יִת וְסָמַ֤ךְ יָדוֹ֙ עַל־ הַקִּ֔יר וּנְשָׁכ֖וֹ הַנָּחָֽשׁ׃ (ז) הֲלֹא־ חֹ֛שֶׁךְ י֥וֹם יְהוָ֖ה וְלֹא־ א֑וֹר וְאָפֵ֖ל וְלֹא־ נֹ֥גַֽהּ לֽוֹ׃ (ח) שָׂנֵ֥אתִי מָאַ֖סְתִּי חַגֵּיכֶ֑ם וְלֹ֥א אָרִ֖יחַ בְּעַצְּרֹֽתֵיכֶֽם׃ (ט) כִּ֣י אִם־ תַּעֲלוּ־ לִ֥י עֹל֛וֹת וּמִנְחֹתֵיכֶ֖ם לֹ֣א אֶרְצֶ֑ה וְשֶׁ֥לֶם מְרִיאֵיכֶ֖ם לֹ֥א אַבִּֽיט׃ (י) הָסֵ֥ר מֵעָלַ֖י הֲמ֣וֹן שִׁרֶ֑יךָ וְזִמְרַ֥ת נְבָלֶ֖יךָ לֹ֥א אֶשְׁמָֽע׃ (יא) וְיִגַּ֥ל כַּמַּ֖יִם מִשְׁפָּ֑ט וּצְדָקָ֖ה כְּנַ֥חַל אֵיתָֽן׃ (יב) הַזְּבָחִ֨ים וּמִנְחָ֜ה הִֽגַּשְׁתֶּם־ לִ֧י בַמִּדְבָּ֛ר אַרְבָּעִ֥ים שָׁנָ֖ה בֵּ֥ית יִשְׂרָאֵֽל׃ (יג) וּנְשָׂאתֶ֗ם אֵ֚ת סִכּ֣וּת מַלְכְּכֶ֔ם וְאֵ֖ת כִּיּ֣וּן צַלְמֵיכֶ֑ם כּוֹכַב֙ אֱלֹ֣הֵיכֶ֔ם אֲשֶׁ֥ר עֲשִׂיתֶ֖ם לָכֶֽם׃ (יד) וְהִגְלֵיתִ֥י אֶתְכֶ֖ם מֵהָ֣לְאָה לְדַמָּ֑שֶׂק אָמַ֛ר יְהוָ֥ה אֱלֹהֵֽי־ צְבָא֖וֹת שְׁמֽוֹ׃ (טו) ה֚וֹי הַשַּׁאֲנַנִּ֣ים בְּצִיּ֔וֹן וְהַבֹּטְחִ֖ים בְּהַ֣ר שֹׁמְר֑וֹן נְקֻבֵי֙ רֵאשִׁ֣ית הַגּוֹיִ֔ם וּבָ֥אוּ לָהֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל׃ (טז) עִבְר֤וּ כַֽלְנֵה֙ וּרְא֔וּ וּלְכ֥וּ מִשָּׁ֖ם חֲמַ֣ת רַבָּ֑ה וּרְד֣וּ גַת־ פְּלִשְׁתּ֗ים הֲטוֹבִים֙ מִן־ הַמַּמְלָכ֣וֹת הָאֵ֔לֶּה אִם־ רַ֥ב גְּבוּלָ֖ם מִגְּבֻלְכֶֽם׃ (יז) הַֽמְנַדִּ֖ים לְי֣וֹם רָ֑ע וַתַּגִּישׁ֖וּן שֶׁ֥בֶת חָמָֽס׃ (יח) הַשֹּֽׁכְבִים֙ עַל־ מִטּ֣וֹת שֵׁ֔ן וּסְרֻחִ֖ים עַל־ עַרְשׂוֹתָ֑ם וְאֹכְלִ֤ים כָּרִים֙ מִצֹּ֔אן וַעֲגָלִ֖ים מִתּ֥וֹךְ מַרְבֵּֽק׃ (יט) הַפֹּרְטִ֖ים עַל־ פִּ֣י הַנָּ֑בֶל כְּדָוִ֕יד חָשְׁב֥וּ לָהֶ֖ם כְּלֵי־ שִֽׁיר׃ (כ) הַשֹּׁתִ֤ים בְּמִזְרְקֵי֙ יַ֔יִן וְרֵאשִׁ֥ית שְׁמָנִ֖ים יִמְשָׁ֑חוּ וְלֹ֥א נֶחְל֖וּ עַל־ שֵׁ֥בֶר יוֹסֵֽף (כא) לָכֵ֛ן עַתָּ֥ה יִגְל֖וּ בְּרֹ֣אשׁ גֹּלִ֑ים וְסָ֖ר מִרְזַ֥ח סְרוּחִֽים׃ (כב) נִשְׁבַּע֩ אֲדֹנָ֨י יְהוִ֜ה בְּנַפְשׁ֗וֹ נְאֻם־ יְהוָה֙ אֱלֹהֵ֣י צְבָא֔וֹת מְתָאֵ֤ב אָֽנֹכִי֙ אֶת־ גְּא֣וֹן יַֽעֲקֹ֔ב וְאַרְמְנֹתָ֖יו שָׂנֵ֑אתִי וְהִסְגַּרְתִּ֖י עִ֥יר וּמְלֹאָֽהּ׃ (כג) וְהָיָ֗ה אִם־ יִוָּ֨תְר֜וּ עֲשָׂרָ֧ה אֲנָשִׁ֛ים בְּבַ֥יִת אֶחָ֖ד וָמֵֽתוּ׃ (כד) וּנְשָׂא֞וֹ דּוֹד֣וֹ וּמְסָרְפ֗וֹ לְהוֹצִ֣יא עֲצָמִים֮ מִן־ הַבַּיִת֒ וְאָמַ֞ר לַאֲשֶׁ֨ר בְּיַרְכְּתֵ֥י הַבַּ֛יִת הַע֥וֹד עִמָּ֖ךְ וְאָמַ֣ר אָ֑פֶס וְאָמַ֣ר הָ֔ס כִּ֛י לֹ֥א לְהַזְכִּ֖יר בְּשֵׁ֥ם יְהוָֽה׃ (כה) כִּֽי־ הִנֵּ֤ה יְהוָה֙ מְצַוֶּ֔ה וְהִכָּ֛ה הַבַּ֥יִת הַגָּד֖וֹל רְסִיסִ֑ים וְהַבַּ֥יִת הַקָּטֹ֖ן בְּקִעִֽים׃ (כו) הַיְרֻצ֤וּן בַּסֶּ֙לַע֙ סוּסִ֔ים אִֽם־ יַחֲר֖וֹשׁ בַּבְּקָרִ֑ים כִּֽי־ הֲפַכְתֶּ֤ם לְרֹאשׁ֙ מִשְׁפָּ֔ט וּפְרִ֥י צְדָקָ֖ה לְלַעֲנָֽה׃ (כז) הַשְּׂמֵחִ֖ים לְלֹ֣א דָבָ֑ר הָאֹ֣מְרִ֔ים הֲל֣וֹא בְחָזְקֵ֔נוּ לָקַ֥חְנוּ לָ֖נוּ קַרְנָֽיִם׃ (כח) כִּ֡י הִנְנִי֩ מֵקִ֨ים עֲלֵיכֶ֜ם בֵּ֣ית יִשְׂרָאֵ֗ל נְאֻם־ יְהוָ֛ה אֱלֹהֵ֥י הַצְּבָא֖וֹת גּ֑וֹי וְלָחֲצ֥וּ אֶתְכֶ֛ם מִלְּב֥וֹא חֲמָ֖ת עַד־ נַ֥חַל הָעֲרָבָֽה׃ תחילת פרק ז (כט) כֹּ֤ה הִרְאַ֙נִי֙ אֲדֹנָ֣י יְהוִ֔ה וְהִנֵּה֙ יוֹצֵ֣ר גֹּבַ֔י בִּתְחִלַּ֖ת עֲל֣וֹת הַלָּ֑קֶשׁ וְהִ֨נֵּה־ לֶ֔קֶשׁ אַחַ֖ר גִּזֵּ֥י הַמֶּֽלֶךְ׃ (ל) וְהָיָ֗ה אִם־ כִּלָּה֙ לֶֽאֱכוֹל֙ אֶת־ עֵ֣שֶׂב הָאָ֔רֶץ וָאֹמַ֗ר אֲדֹנָ֤י יְהוִה֙ סְֽלַֽח־ נָ֔א מִ֥י יָק֖וּם יַֽעֲקֹ֑ב כִּ֥י קָטֹ֖ן הֽוּא׃ (לא) נִחַ֥ם יְהוָ֖ה עַל־ זֹ֑את לֹ֥א תִהְיֶ֖ה אָמַ֥ר יְהוָֽה׃ (לב) כֹּ֤ה הִרְאַ֙נִי֙ אֲדֹנָ֣י יְהוִ֔ה וְהִנֵּ֥ה קֹרֵ֛א לָרִ֥ב בָּאֵ֖שׁ אֲדֹנָ֣י יְהוִ֑ה וַתֹּ֙אכַל֙ אֶת־ תְּה֣וֹם רַבָּ֔ה וְאָכְלָ֖ה אֶת־ הַחֵֽלֶק׃ (לג) וָאֹמַ֗ר אֲדֹנָ֤י יְהוִה֙ חֲדַל־ נָ֔א מִ֥י יָק֖וּם יַעֲקֹ֑ב כִּ֥י קָטֹ֖ן הֽוּא׃ (לד) נִחַ֥ם יְהוָ֖ה עַל־ זֹ֑את גַּם־ הִיא֙ לֹ֣א תִֽהְיֶ֔ה אָמַ֖ר אֲדֹנָ֥י יְהוִֽה׃ (לה) כֹּ֣ה הִרְאַ֔נִי וְהִנֵּ֧ה אֲדֹנָ֛י נִצָּ֖ב עַל־ חוֹמַ֣ת אֲנָ֑ךְ וּבְיָד֖וֹ אֲנָֽךְ׃ (לו) וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י מָֽה־ אַתָּ֤ה רֹאֶה֙ עָמ֔וֹס וָאֹמַ֖ר אֲנָ֑ךְ וַיֹּ֣אמֶר אֲדֹנָ֗י הִנְנִ֨י שָׂ֤ם אֲנָךְ֙ בְּקֶ֙רֶב֙ עַמִּ֣י יִשְׂרָאֵ֔ל לֹֽא־ אוֹסִ֥יף ע֖וֹד עֲב֥וֹר לֽוֹ׃ (לז) וְנָשַׁ֙מּוּ֙ בָּמ֣וֹת יִשְׂחָ֔ק וּמִקְדְּשֵׁ֥י יִשְׂרָאֵ֖ל יֶחֱרָ֑בוּ וְקַמְתִּ֛י עַל־ בֵּ֥ית יָרָבְעָ֖ם בֶּחָֽרֶב׃ (לח) וַיִּשְׁלַ֗ח אֲמַצְיָה֙ כֹּהֵ֣ן בֵּֽית־ אֵ֔ל אֶל־ יָרָבְעָ֥ם מֶֽלֶךְ־ יִשְׂרָאֵ֖ל לֵאמֹ֑ר קָשַׁ֨ר עָלֶ֜יךָ עָמ֗וֹס בְּקֶ֙רֶב֙ בֵּ֣ית יִשְׂרָאֵ֔ל לֹא־ תוּכַ֣ל הָאָ֔רֶץ לְהָכִ֖יל אֶת־ כָּל־ דְּבָרָֽיו׃ (לט) כִּי־ כֹה֙ אָמַ֣ר עָמ֔וֹס בַּחֶ֖רֶב יָמ֣וּת יָרָבְעָ֑ם וְיִ֨שְׂרָאֵ֔ל גָּלֹ֥ה יִגְלֶ֖ה מֵעַ֥ל אַדְמָתֽוֹ׃ (מ) וַיֹּ֤אמֶר אֲמַצְיָה֙ אֶל־ עָמ֔וֹס חֹזֶ֕ה לֵ֥ךְ בְּרַח־ לְךָ֖ אֶל־ אֶ֣רֶץ יְהוּדָ֑ה וֶאֱכָל־ שָׁ֣ם לֶ֔חֶם וְשָׁ֖ם תִּנָּבֵֽא׃ (מא) וּבֵֽית־ אֵ֔ל לֹֽא־ תוֹסִ֥יף ע֖וֹד לְהִנָּבֵ֑א כִּ֤י מִקְדַּשׁ־ מֶ֙לֶךְ֙ ה֔וּא וּבֵ֥ית מַמְלָכָ֖ה הֽוּא׃ (מב) וַיַּ֤עַן עָמוֹס֙ וַיֹּ֣אמֶר אֶל־ אֲמַצְיָ֔ה לֹא־ נָבִ֣יא אָנֹ֔כִי וְלֹ֥א בֶן־ נָבִ֖יא אָנֹ֑כִי כִּֽי־ בוֹקֵ֥ר אָנֹ֖כִי וּבוֹלֵ֥ס שִׁקְמִֽים׃

תרי עשר סדר ח

(א) וַיִּקָּחֵ֣נִי יְהוָ֔ה מֵאַחֲרֵ֖י הַצֹּ֑אן וַיֹּ֤אמֶר אֵלַי֙ יְהוָ֔ה לֵ֥ךְ הִנָּבֵ֖א אֶל־ עַמִּ֥י יִשְׂרָאֵֽל׃ (ב) וְעַתָּ֖ה שְׁמַ֣ע דְּבַר־ יְהוָ֑ה אַתָּ֣ה אֹמֵ֗ר לֹ֤א תִנָּבֵא֙ עַל־ יִשְׂרָאֵ֔ל וְלֹ֥א תַטִּ֖יף עַל־ בֵּ֥ית יִשְׂחָֽק׃ (ג) לָכֵ֞ן כֹּה־ אָמַ֣ר יְהוָ֗ה אִשְׁתְּךָ֞ בָּעִ֤יר תִּזְנֶה֙ וּבָנֶ֤יךָ וּבְנֹתֶ֙יךָ֙ בַּחֶ֣רֶב יִפֹּ֔לוּ וְאַדְמָתְךָ֖ בַּחֶ֣בֶל תְּחֻלָּ֑ק וְאַתָּ֗ה עַל־ אֲדָמָ֤ה טְמֵאָה֙ תָּמ֔וּת וְיִ֨שְׂרָאֵ֔ל גָּלֹ֥ה יִגְלֶ֖ה מֵעַ֥ל אַדְמָתֽוֹ׃ סוף פרק ז (ד) כֹּ֥ה הִרְאַ֖נִי אֲדֹנָ֣י יְהוִ֑ה וְהִנֵּ֖ה כְּל֥וּב קָֽיִץ׃ (ה) וַיֹּ֗אמֶר מָֽה־ אַתָּ֤ה רֹאֶה֙ עָמ֔וֹס עֵשָׂ֔ו נִבְע֖וּ מַצְפֻּנָֽיו׃ (לט) עַֽד־ הַגְּב֣וּל שִׁלְּח֗וּךָ כֹּ֚ל אַנְשֵׁ֣י בְרִיתֶ֔ךָ הִשִּׁיא֛וּךָ יָכְל֥וּ לְךָ֖ אַנְשֵׁ֣י שְׁלֹמֶ֑ךָ לַחְמְךָ֗ יָשִׂ֤ימוּ מָזוֹר֙ תַּחְתֶּ֔יךָ אֵ֥ין תְּבוּנָ֖ה בּֽוֹ׃ (מ) הֲל֛וֹא בַּיּ֥וֹם הַה֖וּא נְאֻם־ יְהוָ֑ה וְהַאֲבַדְתִּ֤י חֲכָמִים֙ מֵֽאֱד֔וֹם וּתְבוּנָ֖ה מֵהַ֥ר עֵשָֽׂו׃ (מא) וְחַתּ֥וּ גִבּוֹרֶ֖יךָ תֵּימָ֑ן לְמַ֧עַן יִכָּֽרֶת־ אִ֛ישׁ מֵהַ֥ר עֵשָׂ֖ו מִקָּֽטֶל׃ (מב) מֵחֲמַ֛ס אָחִ֥יךָ יַעֲקֹ֖ב תְּכַסְּךָ֣ בוּשָׁ֑ה וְנִכְרַ֖תָּ לְעוֹלָֽם׃ (מג) בְּיוֹם֙ עֲמָֽדְךָ֣ מִנֶּ֔גֶד בְּי֛וֹם שְׁב֥וֹת זָרִ֖ים חֵיל֑וֹ וְנָכְרִ֞ים בָּ֣אוּ (שערו) שְׁעָרָ֗יו וְעַל־ יְרוּשָׁלִַ֙ם֙ יַדּ֣וּ גוֹרָ֔ל גַּם־ אַתָּ֖ה כְּאַחַ֥ד מֵהֶֽם׃ (מד) וְאַל־ תֵּ֤רֶא בְיוֹם־ אָחִ֙יךָ֙ בְּי֣וֹם נָכְר֔וֹ וְאַל־ תִּשְׂמַ֥ח לִבְנֵֽי־ יְהוּדָ֖ה בְּי֣וֹם אָבְדָ֑ם וְאַל־ תַּגְדֵּ֥ל פִּ֖יךָ בְּי֥וֹם צָרָֽה׃ (מה) אַל־ תָּב֤וֹא בְשַֽׁעַר־ עַמִּי֙ בְּי֣וֹם אֵידָ֔ם אַל־ תֵּ֧רֶא גַם־ אַתָּ֛ה בְּרָעָת֖וֹ בְּי֣וֹם אֵיד֑וֹ וְאַל־ תִּשְׁלַ֥חְנָה בְחֵיל֖וֹ בְּי֥וֹם אֵידֽוֹ׃ (מו) וְאַֽל־ תַּעֲמֹד֙ עַל־ הַפֶּ֔רֶק לְהַכְרִ֖ית אֶת־ פְּלִיטָ֑יו וְאַל־ תַּסְגֵּ֥ר שְׂרִידָ֖יו בְּי֥וֹם צָרָֽה׃ (מז) כִּֽי־ קָר֥וֹב יוֹם־ יְהוָ֖ה עַל־ כָּל־ הַגּוֹיִ֑ם כַּאֲשֶׁ֤ר עָשִׂ֙יתָ֙ יֵעָ֣שֶׂה לָּ֔ךְ גְּמֻלְךָ֖ יָשׁ֥וּב בְּרֹאשֶֽׁךָ׃ (מח) כִּ֗י כַּֽאֲשֶׁ֤ר שְׁתִיתֶם֙ עַל־ הַ֣ר קָדְשִׁ֔י יִשְׁתּ֥וּ כָֽל־ הַגּוֹיִ֖ם תָּמִ֑יד וְשָׁת֣וּ וְלָע֔וּ וְהָי֖וּ כְּל֥וֹא הָיֽוּ׃ (מט) וּבְהַ֥ר צִיּ֛וֹן תִּהְיֶ֥ה פְלֵיטָ֖ה וְהָ֣יָה קֹ֑דֶשׁ וְיָֽרְשׁוּ֙ בֵּ֣ית יַֽעֲקֹ֔ב אֵ֖ת מוֹרָֽשֵׁיהֶם׃ (נ) וְהָיָה֩ בֵית־ יַעֲקֹ֨ב אֵ֜שׁ וּבֵ֧ית יוֹסֵ֣ף לֶהָבָ֗ה וּבֵ֤ית עֵשָׂו֙ לְקַ֔שׁ וְדָלְק֥וּ בָהֶ֖ם וַאֲכָל֑וּם וְלֹֽא־ יִֽהְיֶ֤ה שָׂרִיד֙ לְבֵ֣ית עֵשָׂ֔ו כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (נא) וְיָרְשׁ֨וּ הַנֶּ֜גֶב אֶת־ הַ֣ר עֵשָׂ֗ו וְהַשְּׁפֵלָה֙ אֶת־ פְּלִשְׁתִּ֔ים וְיָרְשׁוּ֙ אֶת־ שְׂדֵ֣ה אֶפְרַ֔יִם וְאֵ֖ת שְׂדֵ֣ה שֹׁמְר֑וֹן וּבִנְיָמִ֖ן אֶת־ הַגִּלְעָֽד׃ (נב) וְגָלֻ֣ת הַֽחֵל־ הַ֠זֶּה לִבְנֵ֨י יִשְׂרָאֵ֤ל אֲשֶֽׁר־ כְּנַעֲנִים֙ עַד־ צָ֣רְפַ֔ת וְגָלֻ֥ת יְרוּשָׁלִַ֖ם אֲשֶׁ֣ר בִּסְפָרַ֑ד יִֽרְשׁ֕וּ אֵ֖ת עָרֵ֥י הַנֶּֽגֶב׃
