דברים פרק כב לפי החלוקה לסדרים

פרק כב בספר דברים (29 פסוקים) נמצא בדברים סדר טז (פסוקים לה–לט) ובדברים סדר יז (פסוקים א–כד).

דברים סדר טז

(א) כִּֽי־ תִקְרַ֣ב אֶל־ עִ֔יר לְהִלָּחֵ֖ם עָלֶ֑יהָ וְקָרָ֥אתָ אֵלֶ֖יהָ לְשָׁלֽוֹם׃ (ב) וְהָיָה֙ אִם־ שָׁל֣וֹם תַּֽעַנְךָ֔ וּפָתְחָ֖ה לָ֑ךְ וְהָיָ֞ה כָּל־ הָעָ֣ם הַנִּמְצָא־ בָ֗הּ יִהְי֥וּ לְךָ֛ לָמַ֖ס וַעֲבָדֽוּךָ׃ (ג) וְאִם־ לֹ֤א תַשְׁלִים֙ עִמָּ֔ךְ וְעָשְׂתָ֥ה עִמְּךָ֖ מִלְחָמָ֑ה וְצַרְתָּ֖ עָלֶֽיהָּ׃ (ד) וּנְתָנָ֛הּ יְהוָ֥ה אֱלֹהֶ֖יךָ בְּיָדֶ֑ךָ וְהִכִּיתָ֥ אֶת־ כָּל־ זְכוּרָ֖הּ לְפִי־ חָֽרֶב׃ (ה) רַ֣ק הַ֠נָּשִׁים וְהַטַּ֨ף וְהַבְּהֵמָ֜ה וְכֹל֩ אֲשֶׁ֨ר יִהְיֶ֥ה בָעִ֛יר כָּל־ שְׁלָלָ֖הּ תָּבֹ֣ז לָ֑ךְ וְאָֽכַלְתָּ֙ אֶת־ שְׁלַ֣ל אֹיְבֶ֔יךָ אֲשֶׁ֥ר נָתַ֛ן יְהוָ֥ה אֱלֹהֶ֖יךָ לָֽךְ׃ (ו) כֵּ֤ן תַּעֲשֶׂה֙ לְכָל־ הֶ֣עָרִ֔ים הָרְחֹקֹ֥ת מִמְּךָ֖ מְאֹ֑ד אֲשֶׁ֛ר לֹא־ מֵעָרֵ֥י הַגּֽוֹיִם־ הָאֵ֖לֶּה הֵֽנָּה׃ (ז) רַ֗ק מֵעָרֵ֤י הָֽעַמִּים֙ הָאֵ֔לֶּה אֲשֶׁר֙ יְהוָ֣ה אֱלֹהֶ֔יךָ נֹתֵ֥ן לְךָ֖ נַחֲלָ֑ה לֹ֥א תְחַיֶּ֖ה כָּל־ נְשָׁמָֽה׃ (ח) כִּֽי־ הַחֲרֵ֣ם תַּחֲרִימֵ֗ם הַחִתִּ֤י וְהָאֱמֹרִי֙ הַכְּנַעֲנִ֣י וְהַפְּרִזִּ֔י הַחִוִּ֖י וְהַיְבוּסִ֑י כַּאֲשֶׁ֥ר צִוְּךָ֖ יְהוָ֥ה אֱלֹהֶֽיךָ׃ (ט) לְמַ֗עַן אֲשֶׁ֨ר לֹֽא־ יְלַמְּד֤וּ אֶתְכֶם֙ לַעֲשׂ֔וֹת כְּכֹל֙ תּֽוֹעֲבֹתָ֔ם אֲשֶׁ֥ר עָשׂ֖וּ לֵֽאלֹהֵיהֶ֑ם וַחֲטָאתֶ֖ם לַיהוָ֥ה אֱלֹהֵיכֶֽם׃ (י) כִּֽי־ תָצ֣וּר אֶל־ עִיר֩ יָמִ֨ים רַבִּ֜ים לְֽהִלָּחֵ֧ם עָלֶ֣יהָ לְתָפְשָׂ֗הּ לֹֽא־ תַשְׁחִ֤ית אֶת־ עֵצָהּ֙ לִנְדֹּ֤חַ עָלָיו֙ גַּרְזֶ֔ן כִּ֚י מִמֶּ֣נּוּ תֹאכֵ֔ל וְאֹת֖וֹ לֹ֣א תִכְרֹ֑ת כִּ֤י הָֽאָדָם֙ עֵ֣ץ הַשָּׂדֶ֔ה לָבֹ֥א מִפָּנֶ֖יךָ בַּמָּצֽוֹר׃ (יא) רַ֞ק עֵ֣ץ אֲשֶׁר־ תֵּדַ֗ע כִּֽי־ לֹא־ עֵ֤ץ מַאֲכָל֙ ה֔וּא אֹת֥וֹ תַשְׁחִ֖ית וְכָרָ֑תָּ וּבָנִ֣יתָ מָצ֗וֹר עַל־ הָעִיר֙ אֲשֶׁר־ הִ֨וא עֹשָׂ֧ה עִמְּךָ֛ מִלְחָמָ֖ה עַ֥ד רִדְתָּֽהּ׃ (יב) כִּי־ יִמָּצֵ֣א חָלָ֗ל בָּאֲדָמָה֙ אֲשֶׁר֩ יְהוָ֨ה אֱלֹהֶ֜יךָ נֹתֵ֤ן לְךָ֙ לְרִשְׁתָּ֔הּ נֹפֵ֖ל בַּשָּׂדֶ֑ה לֹ֥א נוֹדַ֖ע מִ֥י הִכָּֽהוּ׃ (יג) וְיָצְא֥וּ זְקֵנֶ֖יךָ וְשֹׁפְטֶ֑יךָ וּמָדְדוּ֙ אֶל־ הֶ֣עָרִ֔ים אֲשֶׁ֖ר סְבִיבֹ֥ת הֶחָלָֽל׃ (יד) וְהָיָ֣ה הָעִ֔יר הַקְּרֹבָ֖ה אֶל־ הֶחָלָ֑ל וְלָֽקְח֡וּ זִקְנֵי֩ הָעִ֨יר הַהִ֜וא עֶגְלַ֣ת בָּקָ֗ר אֲשֶׁ֤ר לֹֽא־ עֻבַּד֙ בָּ֔הּ אֲשֶׁ֥ר לֹא־ מָשְׁכָ֖ה בְּעֹֽל׃ (טו) וְהוֹרִ֡דוּ זִקְנֵי֩ הָעִ֨יר הַהִ֤וא אֶת־ הָֽעֶגְלָה֙ אֶל־ נַ֣חַל אֵיתָ֔ן אֲשֶׁ֛ר לֹא־ יֵעָבֵ֥ד בּ֖וֹ וְלֹ֣א יִזָּרֵ֑עַ וְעָֽרְפוּ־ שָׁ֥ם אֶת־ הָעֶגְלָ֖ה בַּנָּֽחַל׃ (טז) וְנִגְּשׁ֣וּ הַכֹּהֲנִים֮ בְּנֵ֣י לֵוִי֒ כִּ֣י בָ֗ם בָּחַ֞ר יְהוָ֤ה אֱלֹהֶ֙יךָ֙ לְשָׁ֣רְת֔וֹ וּלְבָרֵ֖ךְ בְּשֵׁ֣ם יְהוָ֑ה וְעַל־ פִּיהֶ֥ם יִהְיֶ֖ה כָּל־ רִ֥יב וְכָל־ נָֽגַע׃ (יז) וְכֹ֗ל זִקְנֵי֙ הָעִ֣יר הַהִ֔וא הַקְּרֹבִ֖ים אֶל־ הֶחָלָ֑ל יִרְחֲצוּ֙ אֶת־ יְדֵיהֶ֔ם עַל־ הָעֶגְלָ֖ה הָעֲרוּפָ֥ה בַנָּֽחַל׃ (יח) וְעָנ֖וּ וְאָמְר֑וּ יָדֵ֗ינוּ לֹ֤א (שפכה) שָֽׁפְכוּ֙ אֶת־ הַדָּ֣ם הַזֶּ֔ה וְעֵינֵ֖ינוּ לֹ֥א רָאֽוּ׃ (יט) כַּפֵּר֩ לְעַמְּךָ֨ יִשְׂרָאֵ֤ל אֲשֶׁר־ פָּדִ֙יתָ֙ יְהוָ֔ה וְאַל־ תִּתֵּן֙ דָּ֣ם נָקִ֔י בְּקֶ֖רֶב עַמְּךָ֣ יִשְׂרָאֵ֑ל וְנִכַּפֵּ֥ר לָהֶ֖ם הַדָּֽם׃ (כ) וְאַתָּ֗ה תְּבַעֵ֛ר הַדָּ֥ם הַנָּקִ֖י מִקִּרְבֶּ֑ךָ כִּֽי־ תַעֲשֶׂ֥ה הַיָּשָׁ֖ר בְּעֵינֵ֥י יְהוָֽה׃ (כא) כִּֽי־ תֵצֵ֥א לַמִּלְחָמָ֖ה עַל־ אֹיְבֶ֑יךָ וּנְתָנ֞וֹ יְהוָ֧ה אֱלֹהֶ֛יךָ בְּיָדֶ֖ךָ וְשָׁבִ֥יתָ שִׁבְיֽוֹ׃ (כב) וְרָאִיתָ֙ בַּשִּׁבְיָ֔ה אֵ֖שֶׁת יְפַת־ תֹּ֑אַר וְחָשַׁקְתָּ֣ בָ֔הּ וְלָקַחְתָּ֥ לְךָ֖ לְאִשָּֽׁה׃ (כג) וַהֲבֵאתָ֖הּ אֶל־ תּ֣וֹךְ בֵּיתֶ֑ךָ וְגִלְּחָה֙ אֶת־ רֹאשָׁ֔הּ וְעָשְׂתָ֖ה אֶת־ צִפָּרְנֶֽיהָ׃ (כד) וְהֵסִ֩ירָה֩ אֶת־ שִׂמְלַ֨ת שִׁבְיָ֜הּ מֵעָלֶ֗יהָ וְיָֽשְׁבָה֙ בְּבֵיתֶ֔ךָ וּבָֽכְתָ֛ה אֶת־ אָבִ֥יהָ וְאֶת־ אִמָּ֖הּ יֶ֣רַח יָמִ֑ים וְאַ֨חַר כֵּ֜ן תָּב֤וֹא אֵלֶ֙יהָ֙ וּבְעַלְתָּ֔הּ וְהָיְתָ֥ה לְךָ֖ לְאִשָּֽׁה׃ (כה) וְהָיָ֞ה אִם־ לֹ֧א חָפַ֣צְתָּ בָּ֗הּ וְשִׁלַּחְתָּהּ֙ לְנַפְשָׁ֔הּ וּמָכֹ֥ר לֹא־ תִמְכְּרֶ֖נָּה בַּכָּ֑סֶף לֹא־ תִתְעַמֵּ֣ר בָּ֔הּ תַּ֖חַת אֲשֶׁ֥ר עִנִּיתָֽהּ׃ (כו) כִּֽי־ תִהְיֶ֨יןָ לְאִ֜ישׁ שְׁתֵּ֣י נָשִׁ֗ים הָאַחַ֤ת אֲהוּבָה֙ וְהָאַחַ֣ת שְׂנוּאָ֔ה וְיָֽלְדוּ־ ל֣וֹ בָנִ֔ים הָאֲהוּבָ֖ה וְהַשְּׂנוּאָ֑ה וְהָיָ֛ה הַבֵּ֥ן הַבְּכ֖וֹר לַשְּׂנִיאָֽה׃ (כז) וְהָיָ֗ה בְּיוֹם֙ הַנְחִיל֣וֹ אֶת־ בָּנָ֔יו אֵ֥ת אֲשֶׁר־ יִהְיֶ֖ה ל֑וֹ לֹ֣א יוּכַ֗ל לְבַכֵּר֙ אֶת־ בֶּן־ הָ֣אֲהוּבָ֔ה עַל־ פְּנֵ֥י בֶן־ הַשְּׂנוּאָ֖ה הַבְּכֹֽר׃ (כח) כִּי֩ אֶת־ הַבְּכֹ֨ר בֶּן־ הַשְּׂנוּאָ֜ה יַכִּ֗יר לָ֤תֶת לוֹ֙ פִּ֣י שְׁנַ֔יִם בְּכֹ֥ל אֲשֶׁר־ יִמָּצֵ֖א ל֑וֹ כִּי־ הוּא֙ רֵאשִׁ֣ית אֹנ֔וֹ ל֖וֹ מִשְׁפַּ֥ט הַבְּכֹרָֽה׃ (כט) כִּֽי־ יִהְיֶ֣ה לְאִ֗ישׁ בֵּ֚ן סוֹרֵ֣ר וּמוֹרֶ֔ה אֵינֶ֣נּוּ שֹׁמֵ֔עַ בְּק֥וֹל אָבִ֖יו וּבְק֣וֹל אִמּ֑וֹ וְיסְּר֣וּ אֹת֔וֹ וְלֹ֥א יִשְׁמַ֖ע אֲלֵיהֶֽם׃ (ל) וְתָ֥פְשׂוּ ב֖וֹ אָבִ֣יו וְאִמּ֑וֹ וְהוֹצִ֧יאוּ אֹת֛וֹ אֶל־ זִקְנֵ֥י עִיר֖וֹ וְאֶל־ שַׁ֥עַר מְקֹמֽוֹ׃ (לא) וְאָמְר֞וּ אֶל־ זִקְנֵ֣י עִיר֗וֹ בְּנֵ֤נוּ זֶה֙ סוֹרֵ֣ר וּמֹרֶ֔ה אֵינֶ֥נּוּ שֹׁמֵ֖עַ בְּקֹלֵ֑נוּ זוֹלֵ֖ל וְסֹבֵֽא׃ (לב) וּ֠רְגָמֻהוּ כָּל־ אַנְשֵׁ֨י עִיר֤וֹ בָֽאֲבָנִים֙ וָמֵ֔ת וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּ֑ךָ וְכָל־ יִשְׂרָאֵ֖ל יִשְׁמְע֥וּ וְיִרָֽאוּ׃ (לג) וְכִֽי־ יִהְיֶ֣ה בְאִ֗ישׁ חֵ֛טְא מִשְׁפַּט־ מָ֖וֶת וְהוּמָ֑ת וְתָלִ֥יתָ אֹת֖וֹ עַל־ עֵֽץ׃ (לד) לֹא־ תָלִ֨ין נִבְלָת֜וֹ עַל־ הָעֵ֗ץ כִּֽי־ קָב֤וֹר תִּקְבְּרֶ֙נּוּ֙ בַּיּ֣וֹם הַה֔וּא כִּֽי־ קִלְלַ֥ת אֱלֹהִ֖ים תָּל֑וּי וְלֹ֤א תְטַמֵּא֙ אֶת־ אַדְמָ֣תְךָ֔ אֲשֶׁר֙ יְהוָ֣ה אֱלֹהֶ֔יךָ נֹתֵ֥ן לְךָ֖ נַחֲלָֽה׃ תחילת פרק כב (לה) לֹֽא־ תִרְאֶה֩ אֶת־ שׁ֨וֹר אָחִ֜יךָ א֤וֹ אֶת־ שֵׂיוֹ֙ נִדָּחִ֔ים וְהִתְעַלַּמְתָּ֖ מֵהֶ֑ם הָשֵׁ֥ב תְּשִׁיבֵ֖ם לְאָחִֽיךָ׃ (לו) וְאִם־ לֹ֨א קָר֥וֹב אָחִ֛יךָ אֵלֶ֖יךָ וְלֹ֣א יְדַעְתּ֑וֹ וַאֲסַפְתּוֹ֙ אֶל־ תּ֣וֹךְ בֵּיתֶ֔ךָ וְהָיָ֣ה עִמְּךָ֗ עַ֣ד דְּרֹ֤שׁ אָחִ֙יךָ֙ אֹת֔וֹ וַהֲשֵׁבֹת֖וֹ לֽוֹ׃ (לז) וְכֵ֧ן תַּעֲשֶׂ֣ה לַחֲמֹר֗וֹ וְכֵ֣ן תַּעֲשֶׂה֮ לְשִׂמְלָתוֹ֒ וְכֵ֣ן תַּעֲשֶׂ֜ה לְכָל־ אֲבֵדַ֥ת אָחִ֛יךָ אֲשֶׁר־ תֹּאבַ֥ד מִמֶּ֖נּוּ וּמְצָאתָ֑הּ לֹ֥א תוּכַ֖ל לְהִתְעַלֵּֽם׃ (לח) לֹא־ תִרְאֶה֩ אֶת־ חֲמ֨וֹר אָחִ֜יךָ א֤וֹ שׁוֹרוֹ֙ נֹפְלִ֣ים בַּדֶּ֔רֶךְ וְהִתְעַלַּמְתָּ֖ מֵהֶ֑ם הָקֵ֥ם תָּקִ֖ים עִמּֽוֹ׃ (לט) לֹא־ יִהְיֶ֤ה כְלִי־ גֶ֙בֶר֙ עַל־ אִשָּׁ֔ה וְלֹא־ יִלְבַּ֥שׁ גֶּ֖בֶר שִׂמְלַ֣ת אִשָּׁ֑ה כִּ֧י תוֹעֲבַ֛ת יְהוָ֥ה אֱלֹהֶ֖יךָ כָּל־ עֹ֥שֵׂה אֵֽלֶּה׃

דברים סדר יז

(א) כִּ֣י יִקָּרֵ֣א קַן־ צִפּ֣וֹר ׀ לְפָנֶ֡יךָ בַּדֶּ֜רֶךְ בְּכָל־ עֵ֣ץ ׀ א֣וֹ עַל־ הָאָ֗רֶץ אֶפְרֹחִים֙ א֣וֹ בֵיצִ֔ים וְהָאֵ֤ם רֹבֶ֙צֶת֙ עַל־ הָֽאֶפְרֹחִ֔ים א֖וֹ עַל־ הַבֵּיצִ֑ים לֹא־ תִקַּ֥ח הָאֵ֖ם עַל־ הַבָּנִֽים׃ (ב) שַׁלֵּ֤חַ תְּשַׁלַּח֙ אֶת־ הָאֵ֔ם וְאֶת־ הַבָּנִ֖ים תִּֽקַּֽח־ לָ֑ךְ לְמַ֙עַן֙ יִ֣יטַב לָ֔ךְ וְהַאֲרַכְתָּ֖ יָמִֽים׃ (ג) כִּ֤י תִבְנֶה֙ בַּ֣יִת חָדָ֔שׁ וְעָשִׂ֥יתָ מַעֲקֶ֖ה לְגַגֶּ֑ךָ וְלֹֽא־ תָשִׂ֤ים דָּמִים֙ בְּבֵיתֶ֔ךָ כִּֽי־ יִפֹּ֥ל הַנֹּפֵ֖ל מִמֶּֽנּוּ׃ (ד) לֹא־ תִזְרַ֥ע כַּרְמְךָ֖ כִּלְאָ֑יִם פֶּן־ תִּקְדַּ֗שׁ הַֽמְלֵאָ֤ה הַזֶּ֙רַע֙ אֲשֶׁ֣ר תִּזְרָ֔ע וּתְבוּאַ֖ת הַכָּֽרֶם׃ (ה) לֹֽא־ תַחֲרֹ֥שׁ בְּשׁוֹר־ וּבַחֲמֹ֖ר יַחְדָּֽו׃ (ו) לֹ֤א תִלְבַּשׁ֙ שַֽׁעַטְנֵ֔ז צֶ֥מֶר וּפִשְׁתִּ֖ים יַחְדָּֽו׃ (ז) גְּדִלִ֖ים תַּעֲשֶׂה־ לָּ֑ךְ עַל־ אַרְבַּ֛ע כַּנְפ֥וֹת כְּסוּתְךָ֖ אֲשֶׁ֥ר תְּכַסֶּה־ בָּֽהּ׃ (ח) כִּֽי־ יִקַּ֥ח אִ֖ישׁ אִשָּׁ֑ה וּבָ֥א אֵלֶ֖יהָ וּשְׂנֵאָֽהּ׃ (ט) וְשָׂ֥ם לָהּ֙ עֲלִילֹ֣ת דְּבָרִ֔ים וְהוֹצִ֥יא עָלֶ֖יהָ שֵׁ֣ם רָ֑ע וְאָמַ֗ר אֶת־ הָאִשָּׁ֤ה הַזֹּאת֙ לָקַ֔חְתִּי וָאֶקְרַ֣ב אֵלֶ֔יהָ וְלֹא־ מָצָ֥אתִי לָ֖הּ בְּתוּלִֽים׃ (י) וְלָקַ֛ח אֲבִ֥י (הנער) הַֽנַּעֲרָ֖ה וְאִמָּ֑הּ וְהוֹצִ֜יאוּ אֶת־ בְּתוּלֵ֧י (הנער) הַֽנַּעֲרָ֛ה אֶל־ זִקְנֵ֥י הָעִ֖יר הַשָּֽׁעְרָה׃ (יא) וְאָמַ֛ר אֲבִ֥י (הנער) הַֽנַּעַרָ֖ה אֶל־ הַזְּקֵנִ֑ים אֶת־ בִּתִּ֗י נָתַ֜תִּי לָאִ֥ישׁ הַזֶּ֛ה לְאִשָּׁ֖ה וַיִּשְׂנָאֶֽהָ׃ (יב) וְהִנֵּה־ ה֡וּא שָׂם֩ עֲלִילֹ֨ת דְּבָרִ֜ים לֵאמֹ֗ר לֹֽא־ מָצָ֤אתִי לְבִתְּךָ֙ בְּתוּלִ֔ים וְאֵ֖לֶּה בְּתוּלֵ֣י בִתִּ֑י וּפָֽרְשׂוּ֙ הַשִּׂמְלָ֔ה לִפְנֵ֖י זִקְנֵ֥י הָעִֽיר׃ (יג) וְלָֽקְח֛וּ זִקְנֵ֥י הָֽעִיר־ הַהִ֖וא אֶת־ הָאִ֑ישׁ וְיִסְּר֖וּ אֹתֽוֹ׃ (יד) וְעָנְשׁ֨וּ אֹת֜וֹ מֵ֣אָה כֶ֗סֶף וְנָתְנוּ֙ לַאֲבִ֣י הַֽנַּעֲרָ֔ה כִּ֤י הוֹצִיא֙ שֵׁ֣ם רָ֔ע עַ֖ל בְּתוּלַ֣ת יִשְׂרָאֵ֑ל וְלֽוֹ־ תִהְיֶ֣ה לְאִשָּׁ֔ה לֹא־ יוּכַ֥ל לְשַּׁלְּחָ֖הּ כָּל־ יָמָֽיו׃ (טו) וְאִם־ אֱמֶ֣ת הָיָ֔ה הַדָּבָ֖ר הַזֶּ֑ה לֹא־ נִמְצְא֥וּ בְתוּלִ֖ים (לנער) לַֽנַּעֲרָֽה׃ (טז) וְהוֹצִ֨יאוּ אֶת־ (הנער) הַֽנַּעֲרָ֜ה אֶל־ פֶּ֣תַח בֵּית־ אָבִ֗יהָ וּסְקָלוּהָ֩ אַנְשֵׁ֨י עִירָ֤הּ בָּאֲבָנִים֙ וָמֵ֔תָה כִּֽי־ עָשְׂתָ֤ה נְבָלָה֙ בְּיִשְׂרָאֵ֔ל לִזְנ֖וֹת בֵּ֣ית אָבִ֑יהָ וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּֽךָ׃ (יז) כִּֽי־ יִמָּצֵ֨א אִ֜ישׁ שֹׁכֵ֣ב ׀ עִם־ אִשָּׁ֣ה בְעֻֽלַת־ בַּ֗עַל וּמֵ֙תוּ֙ גַּם־ שְׁנֵיהֶ֔ם הָאִ֛ישׁ הַשֹּׁכֵ֥ב עִם־ הָאִשָּׁ֖ה וְהָאִשָּׁ֑ה וּבִֽעַרְתָּ֥ הָרָ֖ע מִיִּשְׂרָאֵֽל׃ (יח) כִּ֤י יִהְיֶה֙ (נער) נַעֲרָ֣ה בְתוּלָ֔ה מְאֹרָשָׂ֖ה לְאִ֑ישׁ וּמְצָאָ֥הּ אִ֛ישׁ בָּעִ֖יר וְשָׁכַ֥ב עִמָּֽהּ׃ (יט) וְהוֹצֵאתֶ֨ם אֶת־ שְׁנֵיהֶ֜ם אֶל־ שַׁ֣עַר ׀ הָעִ֣יר הַהִ֗וא וּסְקַלְתֶּ֨ם אֹתָ֥ם בָּאֲבָנִים֮ וָמֵתוּ֒ אֶת־ (הנער) הַֽנַּעֲרָ֗ה עַל־ דְּבַר֙ אֲשֶׁ֣ר לֹא־ צָעֲקָ֣ה בָעִ֔יר וְאֶ֨ת־ הָאִ֔ישׁ עַל־ דְּבַ֥ר אֲשֶׁר־ עִנָּ֖ה אֶת־ אֵ֣שֶׁת רֵעֵ֑הוּ וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּֽךָ׃ (כ) וְֽאִם־ בַּשָּׂדֶ֞ה יִמְצָ֣א הָאִ֗ישׁ אֶת־ (הנער) הַֽנַּעֲרָה֙ הַמְאֹ֣רָשָׂ֔ה וְהֶחֱזִֽיק־ בָּ֥הּ הָאִ֖ישׁ וְשָׁכַ֣ב עִמָּ֑הּ וּמֵ֗ת הָאִ֛ישׁ אֲשֶׁר־ שָׁכַ֥ב עִמָּ֖הּ לְבַדּֽוֹ׃ (כא) (ולנער) וְלַֽנַּעֲרָה֙ לֹא־ תַעֲשֶׂ֣ה דָבָ֔ר אֵ֥ין (לנער) לַֽנַּעֲרָ֖ה חֵ֣טְא מָ֑וֶת כִּ֡י כַּאֲשֶׁר֩ יָק֨וּם אִ֤ישׁ עַל־ רֵעֵ֙הוּ֙ וּרְצָח֣וֹ נֶ֔פֶשׁ כֵּ֖ן הַדָּבָ֥ר הַזֶּֽה׃ (כב) כִּ֥י בַשָּׂדֶ֖ה מְצָאָ֑הּ צָעֲקָ֗ה (הנער) הַֽנַּעֲרָה֙ הַמְאֹ֣רָשָׂ֔ה וְאֵ֥ין מוֹשִׁ֖יעַ לָֽהּ׃ (כג) כִּֽי־ יִמְצָ֣א אִ֗ישׁ (נער) נַעֲרָ֤ה בְתוּלָה֙ אֲשֶׁ֣ר לֹא־ אֹרָ֔שָׂה וּתְפָשָׂ֖הּ וְשָׁכַ֣ב עִמָּ֑הּ וְנִמְצָֽאוּ׃ (כד) וְ֠נָתַן הָאִ֨ישׁ הַשֹּׁכֵ֥ב עִמָּ֛הּ לַאֲבִ֥י (הנער) הַֽנַּעֲרָ֖ה חֲמִשִּׁ֣ים כָּ֑סֶף וְלֽוֹ־ תִהְיֶ֣ה לְאִשָּׁ֗ה תַּ֚חַת אֲשֶׁ֣ר עִנָּ֔הּ לֹא־ יוּכַ֥ל שַׁלְּחָ֖ה כָּל־ יָמָֽיו׃ סוף פרק כב (כה) לֹא־ יִקַּ֥ח אִ֖ישׁ אֶת־ אֵ֣שֶׁת אָבִ֑יו וְלֹ֥א יְגַלֶּ֖ה כְּנַ֥ף אָבִֽיו׃ (כו) לֹֽא־ יָבֹ֧א פְצֽוּעַ־ דַּכָּ֛א וּכְר֥וּת שָׁפְכָ֖ה בִּקְהַ֥ל יְהוָֽה׃ (כז) לֹא־ יָבֹ֥א מַמְזֵ֖ר בִּקְהַ֣ל יְהוָ֑ה גַּ֚ם דּ֣וֹר עֲשִׂירִ֔י לֹא־ יָ֥בֹא ל֖וֹ בִּקְהַ֥ל יְהוָֽה׃ (כח) לֹֽא־ יָבֹ֧א עַמּוֹנִ֛י וּמוֹאָבִ֖י בִּקְהַ֣ל יְהוָ֑ה גַּ֚ם דּ֣וֹר עֲשִׂירִ֔י לֹא־ יָבֹ֥א לָהֶ֛ם בִּקְהַ֥ל יְהוָ֖ה עַד־ עוֹלָֽם׃ (כט) עַל־ דְּבַ֞ר אֲשֶׁ֨ר לֹא־ קִדְּמ֤וּ אֶתְכֶם֙ בַּלֶּ֣חֶם וּבַמַּ֔יִם בַּדֶּ֖רֶךְ בְּצֵאתְכֶ֣ם מִמִּצְרָ֑יִם וַאֲשֶׁר֩ שָׂכַ֨ר עָלֶ֜יךָ אֶת־ בִּלְעָ֣ם בֶּן־ בְּע֗וֹר מִפְּת֛וֹר אֲרַ֥ם נַהֲרַ֖יִם לְקַֽלְלֶֽךָּ׃ (ל) וְלֹֽא־ אָבָ֞ה יְהוָ֤ה אֱלֹהֶ֙יךָ֙ לִשְׁמֹ֣עַ אֶל־ בִּלְעָ֔ם וַיַּהֲפֹךְ֩ יְהוָ֨ה אֱלֹהֶ֧יךָ לְּךָ֛ אֶת־ הַקְּלָלָ֖ה לִבְרָכָ֑ה כִּ֥י אֲהֵֽבְךָ֖ יְהוָ֥ה אֱלֹהֶֽיךָ׃ (לא) לֹא־ תִדְרֹ֥שׁ שְׁלֹמָ֖ם וְטֹבָתָ֑ם כָּל־ יָמֶ֖יךָ לְעוֹלָֽם׃ (לב) לֹֽא־ תְתַעֵ֣ב אֲדֹמִ֔י כִּ֥י אָחִ֖יךָ ה֑וּא לֹא־ תְתַעֵ֣ב מִצְרִ֔י כִּי־ גֵ֖ר הָיִ֥יתָ בְאַרְצֽוֹ׃ (לג) בָּנִ֛ים אֲשֶׁר־ יִוָּלְד֥וּ לָהֶ֖ם דּ֣וֹר שְׁלִישִׁ֑י יָבֹ֥א לָהֶ֖ם בִּקְהַ֥ל יְהוָֽה׃
