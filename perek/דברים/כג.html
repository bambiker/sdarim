דברים פרק כג לפי החלוקה לסדרים

פרק כג בספר דברים (26 פסוקים) נמצא בדברים סדר יז (פסוקים כה–לג) ובדברים סדר יח (פסוקים א–יב) ובדברים סדר יט (פסוקים א–ה).

דברים סדר יז

(א) כִּ֣י יִקָּרֵ֣א קַן־ צִפּ֣וֹר ׀ לְפָנֶ֡יךָ בַּדֶּ֜רֶךְ בְּכָל־ עֵ֣ץ ׀ א֣וֹ עַל־ הָאָ֗רֶץ אֶפְרֹחִים֙ א֣וֹ בֵיצִ֔ים וְהָאֵ֤ם רֹבֶ֙צֶת֙ עַל־ הָֽאֶפְרֹחִ֔ים א֖וֹ עַל־ הַבֵּיצִ֑ים לֹא־ תִקַּ֥ח הָאֵ֖ם עַל־ הַבָּנִֽים׃ (ב) שַׁלֵּ֤חַ תְּשַׁלַּח֙ אֶת־ הָאֵ֔ם וְאֶת־ הַבָּנִ֖ים תִּֽקַּֽח־ לָ֑ךְ לְמַ֙עַן֙ יִ֣יטַב לָ֔ךְ וְהַאֲרַכְתָּ֖ יָמִֽים׃ (ג) כִּ֤י תִבְנֶה֙ בַּ֣יִת חָדָ֔שׁ וְעָשִׂ֥יתָ מַעֲקֶ֖ה לְגַגֶּ֑ךָ וְלֹֽא־ תָשִׂ֤ים דָּמִים֙ בְּבֵיתֶ֔ךָ כִּֽי־ יִפֹּ֥ל הַנֹּפֵ֖ל מִמֶּֽנּוּ׃ (ד) לֹא־ תִזְרַ֥ע כַּרְמְךָ֖ כִּלְאָ֑יִם פֶּן־ תִּקְדַּ֗שׁ הַֽמְלֵאָ֤ה הַזֶּ֙רַע֙ אֲשֶׁ֣ר תִּזְרָ֔ע וּתְבוּאַ֖ת הַכָּֽרֶם׃ (ה) לֹֽא־ תַחֲרֹ֥שׁ בְּשׁוֹר־ וּבַחֲמֹ֖ר יַחְדָּֽו׃ (ו) לֹ֤א תִלְבַּשׁ֙ שַֽׁעַטְנֵ֔ז צֶ֥מֶר וּפִשְׁתִּ֖ים יַחְדָּֽו׃ (ז) גְּדִלִ֖ים תַּעֲשֶׂה־ לָּ֑ךְ עַל־ אַרְבַּ֛ע כַּנְפ֥וֹת כְּסוּתְךָ֖ אֲשֶׁ֥ר תְּכַסֶּה־ בָּֽהּ׃ (ח) כִּֽי־ יִקַּ֥ח אִ֖ישׁ אִשָּׁ֑ה וּבָ֥א אֵלֶ֖יהָ וּשְׂנֵאָֽהּ׃ (ט) וְשָׂ֥ם לָהּ֙ עֲלִילֹ֣ת דְּבָרִ֔ים וְהוֹצִ֥יא עָלֶ֖יהָ שֵׁ֣ם רָ֑ע וְאָמַ֗ר אֶת־ הָאִשָּׁ֤ה הַזֹּאת֙ לָקַ֔חְתִּי וָאֶקְרַ֣ב אֵלֶ֔יהָ וְלֹא־ מָצָ֥אתִי לָ֖הּ בְּתוּלִֽים׃ (י) וְלָקַ֛ח אֲבִ֥י (הנער) הַֽנַּעֲרָ֖ה וְאִמָּ֑הּ וְהוֹצִ֜יאוּ אֶת־ בְּתוּלֵ֧י (הנער) הַֽנַּעֲרָ֛ה אֶל־ זִקְנֵ֥י הָעִ֖יר הַשָּֽׁעְרָה׃ (יא) וְאָמַ֛ר אֲבִ֥י (הנער) הַֽנַּעַרָ֖ה אֶל־ הַזְּקֵנִ֑ים אֶת־ בִּתִּ֗י נָתַ֜תִּי לָאִ֥ישׁ הַזֶּ֛ה לְאִשָּׁ֖ה וַיִּשְׂנָאֶֽהָ׃ (יב) וְהִנֵּה־ ה֡וּא שָׂם֩ עֲלִילֹ֨ת דְּבָרִ֜ים לֵאמֹ֗ר לֹֽא־ מָצָ֤אתִי לְבִתְּךָ֙ בְּתוּלִ֔ים וְאֵ֖לֶּה בְּתוּלֵ֣י בִתִּ֑י וּפָֽרְשׂוּ֙ הַשִּׂמְלָ֔ה לִפְנֵ֖י זִקְנֵ֥י הָעִֽיר׃ (יג) וְלָֽקְח֛וּ זִקְנֵ֥י הָֽעִיר־ הַהִ֖וא אֶת־ הָאִ֑ישׁ וְיִסְּר֖וּ אֹתֽוֹ׃ (יד) וְעָנְשׁ֨וּ אֹת֜וֹ מֵ֣אָה כֶ֗סֶף וְנָתְנוּ֙ לַאֲבִ֣י הַֽנַּעֲרָ֔ה כִּ֤י הוֹצִיא֙ שֵׁ֣ם רָ֔ע עַ֖ל בְּתוּלַ֣ת יִשְׂרָאֵ֑ל וְלֽוֹ־ תִהְיֶ֣ה לְאִשָּׁ֔ה לֹא־ יוּכַ֥ל לְשַּׁלְּחָ֖הּ כָּל־ יָמָֽיו׃ (טו) וְאִם־ אֱמֶ֣ת הָיָ֔ה הַדָּבָ֖ר הַזֶּ֑ה לֹא־ נִמְצְא֥וּ בְתוּלִ֖ים (לנער) לַֽנַּעֲרָֽה׃ (טז) וְהוֹצִ֨יאוּ אֶת־ (הנער) הַֽנַּעֲרָ֜ה אֶל־ פֶּ֣תַח בֵּית־ אָבִ֗יהָ וּסְקָלוּהָ֩ אַנְשֵׁ֨י עִירָ֤הּ בָּאֲבָנִים֙ וָמֵ֔תָה כִּֽי־ עָשְׂתָ֤ה נְבָלָה֙ בְּיִשְׂרָאֵ֔ל לִזְנ֖וֹת בֵּ֣ית אָבִ֑יהָ וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּֽךָ׃ (יז) כִּֽי־ יִמָּצֵ֨א אִ֜ישׁ שֹׁכֵ֣ב ׀ עִם־ אִשָּׁ֣ה בְעֻֽלַת־ בַּ֗עַל וּמֵ֙תוּ֙ גַּם־ שְׁנֵיהֶ֔ם הָאִ֛ישׁ הַשֹּׁכֵ֥ב עִם־ הָאִשָּׁ֖ה וְהָאִשָּׁ֑ה וּבִֽעַרְתָּ֥ הָרָ֖ע מִיִּשְׂרָאֵֽל׃ (יח) כִּ֤י יִהְיֶה֙ (נער) נַעֲרָ֣ה בְתוּלָ֔ה מְאֹרָשָׂ֖ה לְאִ֑ישׁ וּמְצָאָ֥הּ אִ֛ישׁ בָּעִ֖יר וְשָׁכַ֥ב עִמָּֽהּ׃ (יט) וְהוֹצֵאתֶ֨ם אֶת־ שְׁנֵיהֶ֜ם אֶל־ שַׁ֣עַר ׀ הָעִ֣יר הַהִ֗וא וּסְקַלְתֶּ֨ם אֹתָ֥ם בָּאֲבָנִים֮ וָמֵתוּ֒ אֶת־ (הנער) הַֽנַּעֲרָ֗ה עַל־ דְּבַר֙ אֲשֶׁ֣ר לֹא־ צָעֲקָ֣ה בָעִ֔יר וְאֶ֨ת־ הָאִ֔ישׁ עַל־ דְּבַ֥ר אֲשֶׁר־ עִנָּ֖ה אֶת־ אֵ֣שֶׁת רֵעֵ֑הוּ וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּֽךָ׃ (כ) וְֽאִם־ בַּשָּׂדֶ֞ה יִמְצָ֣א הָאִ֗ישׁ אֶת־ (הנער) הַֽנַּעֲרָה֙ הַמְאֹ֣רָשָׂ֔ה וְהֶחֱזִֽיק־ בָּ֥הּ הָאִ֖ישׁ וְשָׁכַ֣ב עִמָּ֑הּ וּמֵ֗ת הָאִ֛ישׁ אֲשֶׁר־ שָׁכַ֥ב עִמָּ֖הּ לְבַדּֽוֹ׃ (כא) (ולנער) וְלַֽנַּעֲרָה֙ לֹא־ תַעֲשֶׂ֣ה דָבָ֔ר אֵ֥ין (לנער) לַֽנַּעֲרָ֖ה חֵ֣טְא מָ֑וֶת כִּ֡י כַּאֲשֶׁר֩ יָק֨וּם אִ֤ישׁ עַל־ רֵעֵ֙הוּ֙ וּרְצָח֣וֹ נֶ֔פֶשׁ כֵּ֖ן הַדָּבָ֥ר הַזֶּֽה׃ (כב) כִּ֥י בַשָּׂדֶ֖ה מְצָאָ֑הּ צָעֲקָ֗ה (הנער) הַֽנַּעֲרָה֙ הַמְאֹ֣רָשָׂ֔ה וְאֵ֥ין מוֹשִׁ֖יעַ לָֽהּ׃ (כג) כִּֽי־ יִמְצָ֣א אִ֗ישׁ (נער) נַעֲרָ֤ה בְתוּלָה֙ אֲשֶׁ֣ר לֹא־ אֹרָ֔שָׂה וּתְפָשָׂ֖הּ וְשָׁכַ֣ב עִמָּ֑הּ וְנִמְצָֽאוּ׃ (כד) וְ֠נָתַן הָאִ֨ישׁ הַשֹּׁכֵ֥ב עִמָּ֛הּ לַאֲבִ֥י (הנער) הַֽנַּעֲרָ֖ה חֲמִשִּׁ֣ים כָּ֑סֶף וְלֽוֹ־ תִהְיֶ֣ה לְאִשָּׁ֗ה תַּ֚חַת אֲשֶׁ֣ר עִנָּ֔הּ לֹא־ יוּכַ֥ל שַׁלְּחָ֖ה כָּל־ יָמָֽיו׃ תחילת פרק כג (כה) לֹא־ יִקַּ֥ח אִ֖ישׁ אֶת־ אֵ֣שֶׁת אָבִ֑יו וְלֹ֥א יְגַלֶּ֖ה כְּנַ֥ף אָבִֽיו׃ (כו) לֹֽא־ יָבֹ֧א פְצֽוּעַ־ דַּכָּ֛א וּכְר֥וּת שָׁפְכָ֖ה בִּקְהַ֥ל יְהוָֽה׃ (כז) לֹא־ יָבֹ֥א מַמְזֵ֖ר בִּקְהַ֣ל יְהוָ֑ה גַּ֚ם דּ֣וֹר עֲשִׂירִ֔י לֹא־ יָ֥בֹא ל֖וֹ בִּקְהַ֥ל יְהוָֽה׃ (כח) לֹֽא־ יָבֹ֧א עַמּוֹנִ֛י וּמוֹאָבִ֖י בִּקְהַ֣ל יְהוָ֑ה גַּ֚ם דּ֣וֹר עֲשִׂירִ֔י לֹא־ יָבֹ֥א לָהֶ֛ם בִּקְהַ֥ל יְהוָ֖ה עַד־ עוֹלָֽם׃ (כט) עַל־ דְּבַ֞ר אֲשֶׁ֨ר לֹא־ קִדְּמ֤וּ אֶתְכֶם֙ בַּלֶּ֣חֶם וּבַמַּ֔יִם בַּדֶּ֖רֶךְ בְּצֵאתְכֶ֣ם מִמִּצְרָ֑יִם וַאֲשֶׁר֩ שָׂכַ֨ר עָלֶ֜יךָ אֶת־ בִּלְעָ֣ם בֶּן־ בְּע֗וֹר מִפְּת֛וֹר אֲרַ֥ם נַהֲרַ֖יִם לְקַֽלְלֶֽךָּ׃ (ל) וְלֹֽא־ אָבָ֞ה יְהוָ֤ה אֱלֹהֶ֙יךָ֙ לִשְׁמֹ֣עַ אֶל־ בִּלְעָ֔ם וַיַּהֲפֹךְ֩ יְהוָ֨ה אֱלֹהֶ֧יךָ לְּךָ֛ אֶת־ הַקְּלָלָ֖ה לִבְרָכָ֑ה כִּ֥י אֲהֵֽבְךָ֖ יְהוָ֥ה אֱלֹהֶֽיךָ׃ (לא) לֹא־ תִדְרֹ֥שׁ שְׁלֹמָ֖ם וְטֹבָתָ֑ם כָּל־ יָמֶ֖יךָ לְעוֹלָֽם׃ (לב) לֹֽא־ תְתַעֵ֣ב אֲדֹמִ֔י כִּ֥י אָחִ֖יךָ ה֑וּא לֹא־ תְתַעֵ֣ב מִצְרִ֔י כִּי־ גֵ֖ר הָיִ֥יתָ בְאַרְצֽוֹ׃ (לג) בָּנִ֛ים אֲשֶׁר־ יִוָּלְד֥וּ לָהֶ֖ם דּ֣וֹר שְׁלִישִׁ֑י יָבֹ֥א לָהֶ֖ם בִּקְהַ֥ל יְהוָֽה׃

(א) כִּֽי־ תֵצֵ֥א מַחֲנֶ֖ה עַל־ אֹיְבֶ֑יךָ וְנִ֨שְׁמַרְתָּ֔ מִכֹּ֖ל דָּבָ֥ר רָֽע׃ (ב) כִּֽי־ יִהְיֶ֤ה בְךָ֙ אִ֔ישׁ אֲשֶׁ֛ר לֹא־ יִהְיֶ֥ה טָה֖וֹר מִקְּרֵה־ לָ֑יְלָה וְיָצָא֙ אֶל־ מִח֣וּץ לַֽמַּחֲנֶ֔ה לֹ֥א יָבֹ֖א אֶל־ תּ֥וֹךְ הַֽמַּחֲנֶֽה׃ (ג) וְהָיָ֥ה לִפְנֽוֹת־ עֶ֖רֶב יִרְחַ֣ץ בַּמָּ֑יִם וּכְבֹ֣א הַשֶּׁ֔מֶשׁ יָבֹ֖א אֶל־ תּ֥וֹךְ הַֽמַּחֲנֶה׃ (ד) וְיָד֙ תִּהְיֶ֣ה לְךָ֔ מִח֖וּץ לַֽמַּחֲנֶ֑ה וְיָצָ֥אתָ שָׁ֖מָּה חֽוּץ׃ (ה) וְיָתֵ֛ד תִּהְיֶ֥ה לְךָ֖ עַל־ אֲזֵנֶ֑ךָ וְהָיָה֙ בְּשִׁבְתְּךָ֣ ח֔וּץ וְחָפַרְתָּ֣ה בָ֔הּ וְשַׁבְתָּ֖ וְכִסִּ֥יתָ אֶת־ צֵאָתֶֽךָ׃ (ו) כִּי֩ יְהוָ֨ה אֱלֹהֶ֜יךָ מִתְהַלֵּ֣ךְ ׀ בְּקֶ֣רֶב מַחֲנֶ֗ךָ לְהַצִּֽילְךָ֙ וְלָתֵ֤ת אֹיְבֶ֙יךָ֙ לְפָנֶ֔יךָ וְהָיָ֥ה מַחֲנֶ֖יךָ קָד֑וֹשׁ וְלֹֽא־ יִרְאֶ֤ה בְךָ֙ עֶרְוַ֣ת דָּבָ֔ר וְשָׁ֖ב מֵאַחֲרֶֽיךָ׃ (ז) לֹא־ תַסְגִּ֥יר עֶ֖בֶד אֶל־ אֲדֹנָ֑יו אֲשֶׁר־ יִנָּצֵ֥ל אֵלֶ֖יךָ מֵעִ֥ם אֲדֹנָֽיו׃ (ח) עִמְּךָ֞ יֵשֵׁ֣ב בְּקִרְבְּךָ֗ בַּמָּק֧וֹם אֲשֶׁר־ יִבְחַ֛ר בְּאַחַ֥ד שְׁעָרֶ֖יךָ בַּטּ֣וֹב ל֑וֹ לֹ֖א תּוֹנֶֽנּוּ׃ (ט) לֹא־ תִהְיֶ֥ה קְדֵשָׁ֖ה מִבְּנ֣וֹת יִשְׂרָאֵ֑ל וְלֹֽא־ יִהְיֶ֥ה קָדֵ֖שׁ מִבְּנֵ֥י יִשְׂרָאֵל׃ (י) לֹא־ תָבִיא֩ אֶתְנַ֨ן זוֹנָ֜ה וּמְחִ֣יר כֶּ֗לֶב בֵּ֛ית יְהוָ֥ה אֱלֹהֶ֖יךָ לְכָל־ נֶ֑דֶר כִּ֧י תוֹעֲבַ֛ת יְהוָ֥ה אֱלֹהֶ֖יךָ גַּם־ שְׁנֵיהֶֽם׃ (יא) לֹא־ תַשִּׁ֣יךְ לְאָחִ֔יךָ נֶ֥שֶׁךְ כֶּ֖סֶף נֶ֣שֶׁךְ אֹ֑כֶל נֶ֕שֶׁךְ כָּל־ דָּבָ֖ר אֲשֶׁ֥ר יִשָּֽׁךְ׃ (יב) לַנָּכְרִ֣י תַשִּׁ֔יךְ וּלְאָחִ֖יךָ לֹ֣א תַשִּׁ֑יךְ לְמַ֨עַן יְבָרֶכְךָ֜ יְהוָ֣ה אֱלֹהֶ֗יךָ בְּכֹל֙ מִשְׁלַ֣ח יָדֶ֔ךָ עַל־ הָאָ֕רֶץ אֲשֶׁר־ אַתָּ֥ה בָא־ שָׁ֖מָּה לְרִשְׁתָּֽהּ׃

דברים סדר יט

(א) כִּֽי־ תִדֹּ֥ר נֶ֙דֶר֙ לַיהוָ֣ה אֱלֹהֶ֔יךָ לֹ֥א תְאַחֵ֖ר לְשַׁלְּמ֑וֹ כִּֽי־ דָּרֹ֨שׁ יִדְרְשֶׁ֜נּוּ יְהוָ֤ה אֱלֹהֶ֙יךָ֙ מֵֽעִמָּ֔ךְ וְהָיָ֥ה בְךָ֖ חֵֽטְא׃ (ב) וְכִ֥י תֶחְדַּ֖ל לִנְדֹּ֑ר לֹֽא־ יִהְיֶ֥ה בְךָ֖ חֵֽטְא׃ (ג) מוֹצָ֥א שְׂפָתֶ֖יךָ תִּשְׁמֹ֣ר וְעָשִׂ֑יתָ כַּאֲשֶׁ֨ר נָדַ֜רְתָּ לַיהוָ֤ה אֱלֹהֶ֙יךָ֙ נְדָבָ֔ה אֲשֶׁ֥ר דִּבַּ֖רְתָּ בְּפִֽיךָ׃ (ד) כִּ֤י תָבֹא֙ בְּכֶ֣רֶם רֵעֶ֔ךָ וְאָכַלְתָּ֧ עֲנָבִ֛ים כְּנַפְשְׁךָ֖ שָׂבְעֶ֑ךָ וְאֶֽל־ כֶּלְיְךָ֖ לֹ֥א תִתֵּֽן׃ (ה) כִּ֤י תָבֹא֙ בְּקָמַ֣ת רֵעֶ֔ךָ וְקָטַפְתָּ֥ מְלִילֹ֖ת בְּיָדֶ֑ךָ וְחֶרְמֵשׁ֙ לֹ֣א תָנִ֔יף עַ֖ל קָמַ֥ת רֵעֶֽךָ׃ סוף פרק כג (ו) כִּֽי־ יִקַּ֥ח אִ֛ישׁ אִשָּׁ֖ה וּבְעָלָ֑הּ וְהָיָ֞ה אִם־ לֹ֧א תִמְצָא־ חֵ֣ן בְּעֵינָ֗יו כִּי־ מָ֤צָא בָהּ֙ עֶרְוַ֣ת דָּבָ֔ר וְכָ֨תַב לָ֜הּ סֵ֤פֶר כְּרִיתֻת֙ וְנָתַ֣ן בְּיָדָ֔הּ וְשִׁלְּחָ֖הּ מִבֵּיתֽוֹ׃ (ז) וְיָצְאָ֖ה מִבֵּית֑וֹ וְהָלְכָ֖ה וְהָיְתָ֥ה לְאִישׁ־ אַחֵֽר׃ (ח) וּשְׂנֵאָהּ֮ הָאִ֣ישׁ הָאַחֲרוֹן֒ וְכָ֨תַב לָ֜הּ סֵ֤פֶר כְּרִיתֻת֙ וְנָתַ֣ן בְּיָדָ֔הּ וְשִׁלְּחָ֖הּ מִבֵּית֑וֹ א֣וֹ כִ֤י יָמוּת֙ הָאִ֣ישׁ הָאַחֲר֔וֹן אֲשֶׁר־ לְקָחָ֥הּ ל֖וֹ לְאִשָּֽׁה׃ (ט) לֹא־ יוּכַ֣ל בַּעְלָ֣הּ הָרִאשׁ֣וֹן אֲשֶֽׁר־ שִׁ֠לְּחָהּ לָשׁ֨וּב לְקַחְתָּ֜הּ לִהְי֧וֹת ל֣וֹ לְאִשָּׁ֗ה אַחֲרֵי֙ אֲשֶׁ֣ר הֻטַּמָּ֔אָה כִּֽי־ תוֹעֵבָ֥ה הִ֖וא לִפְנֵ֣י יְהוָ֑ה וְלֹ֤א תַחֲטִיא֙ אֶת־ הָאָ֔רֶץ אֲשֶׁר֙ יְהוָ֣ה אֱלֹהֶ֔יךָ נֹתֵ֥ן לְךָ֖ נַחֲלָֽה׃ (י) כִּֽי־ יִקַּ֥ח אִישׁ֙ אִשָּׁ֣ה חֲדָשָׁ֔ה לֹ֤א יֵצֵא֙ בַּצָּבָ֔א וְלֹא־ יַעֲבֹ֥ר עָלָ֖יו לְכָל־ דָּבָ֑ר נָקִ֞י יִהְיֶ֤ה לְבֵיתוֹ֙ שָׁנָ֣ה אֶחָ֔ת וְשִׂמַּ֖ח אֶת־ אִשְׁתּ֥וֹ אֲשֶׁר־ לָקָֽח׃ (יא) לֹא־ יַחֲבֹ֥ל רֵחַ֖יִם וָרָ֑כֶב כִּי־ נֶ֖פֶשׁ ה֥וּא חֹבֵֽל׃ (יב) כִּי־ יִמָּצֵ֣א אִ֗ישׁ גֹּנֵ֨ב נֶ֤פֶשׁ מֵאֶחָיו֙ מִבְּנֵ֣י יִשְׂרָאֵ֔ל וְהִתְעַמֶּר־ בּ֖וֹ וּמְכָר֑וֹ וּמֵת֙ הַגַּנָּ֣ב הַה֔וּא וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּֽךָ׃ (יג) הִשָּׁ֧מֶר בְּנֶֽגַע־ הַצָּרַ֛עַת לִשְׁמֹ֥ר מְאֹ֖ד וְלַעֲשׂ֑וֹת כְּכֹל֩ אֲשֶׁר־ יוֹר֨וּ אֶתְכֶ֜ם הַכֹּהֲנִ֧ים הַלְוִיִּ֛ם כַּאֲשֶׁ֥ר צִוִּיתִ֖ם תִּשְׁמְר֥וּ לַעֲשֽׂוֹת׃ (יד) זָכ֕וֹר אֵ֧ת אֲשֶׁר־ עָשָׂ֛ה יְהוָ֥ה אֱלֹהֶ֖יךָ לְמִרְיָ֑ם בַּדֶּ֖רֶךְ בְּצֵאתְכֶ֥ם מִמִּצְרָֽיִם׃ (טו) כִּֽי־ תַשֶּׁ֥ה בְרֵֽעֲךָ מַשַּׁ֣את מְא֑וּמָה לֹא־ תָבֹ֥א אֶל־ בֵּית֖וֹ לַעֲבֹ֥ט עֲבֹטֽוֹ׃ (טז) בַּח֖וּץ תַּעֲמֹ֑ד וְהָאִ֗ישׁ אֲשֶׁ֤ר אַתָּה֙ נֹשֶׁ֣ה ב֔וֹ יוֹצִ֥יא אֵלֶ֛יךָ אֶֽת־ הַעֲב֖וֹט הַחֽוּצָה׃ (יז) וְאִם־ אִ֥ישׁ עָנִ֖י ה֑וּא לֹ֥א תִשְׁכַּ֖ב בַּעֲבֹטֽוֹ׃ (יח) הָשֵׁב֩ תָּשִׁ֨יב ל֤וֹ אֶֽת־ הַעֲבוֹט֙ כְּבֹ֣א הַשֶּׁ֔מֶשׁ וְשָׁכַ֥ב בְּשַׂלְמָת֖וֹ וּבֵֽרֲכֶ֑ךָּ וּלְךָ֙ תִּהְיֶ֣ה צְדָקָ֔ה לִפְנֵ֖י יְהוָ֥ה אֱלֹהֶֽיךָ׃ (יט) לֹא־ תַעֲשֹׁ֥ק שָׂכִ֖יר עָנִ֣י וְאֶבְי֑וֹן מֵאַחֶ֕יךָ א֧וֹ מִגֵּרְךָ֛ אֲשֶׁ֥ר בְּאַרְצְךָ֖ בִּשְׁעָרֶֽיךָ׃ (כ) בְּיוֹמוֹ֩ תִתֵּ֨ן שְׂכָר֜וֹ וְֽלֹא־ תָב֧וֹא עָלָ֣יו הַשֶּׁ֗מֶשׁ כִּ֤י עָנִי֙ ה֔וּא וְאֵלָ֕יו ה֥וּא נֹשֵׂ֖א אֶת־ נַפְשׁ֑וֹ וְלֹֽא־ יִקְרָ֤א עָלֶ֙יךָ֙ אֶל־ יְהוָ֔ה וְהָיָ֥ה בְךָ֖ חֵֽטְא׃ (כא) לֹֽא־ יוּמְת֤וּ אָבוֹת֙ עַל־ בָּנִ֔ים וּבָנִ֖ים לֹא־ יוּמְת֣וּ עַל־ אָב֑וֹת אִ֥יש בְּחֶטְא֖וֹ יוּמָֽתוּ׃ (כב) לֹ֣א תַטֶּ֔ה מִשְׁפַּ֖ט גֵּ֣ר יָת֑וֹם וְלֹ֣א תַחֲבֹ֔ל בֶּ֖גֶד אַלְמָנָֽה׃ (כג) וְזָכַרְתָּ֗ כִּ֣י עֶ֤בֶד הָיִ֙יתָ֙ בְּמִצְרַ֔יִם וַֽיִּפְדְּךָ֛ יְהוָ֥ה אֱלֹהֶ֖יךָ מִשָּׁ֑ם עַל־ כֵּ֞ן אָנֹכִ֤י מְצַוְּךָ֙ לַעֲשׂ֔וֹת אֶת־ הַדָּבָ֖ר הַזֶּֽה׃
