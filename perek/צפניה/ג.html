צפניה פרק ג לפי החלוקה לסדרים

פרק ג בספר צפניה (20 פסוקים) נמצא בתרי עשר סדר יד (פסוקים לד–נב) ובתרי עשר סדר טו (פסוקים א–א).

תרי עשר סדר יד

(א) דְּבַר־ יְהוָ֣ה ׀ אֲשֶׁ֣ר הָיָ֗ה אֶל־ צְפַנְיָה֙ בֶּן־ כּוּשִׁ֣י בֶן־ גְּדַלְיָ֔ה בֶּן־ אֲמַרְיָ֖ה בֶּן־ חִזְקִיָּ֑ה בִּימֵ֛י יֹאשִׁיָּ֥הוּ בֶן־ אָמ֖וֹן מֶ֥לֶךְ יְהוּדָֽה׃ (ב) אָסֹ֨ף אָסֵ֜ף כֹּ֗ל מֵעַ֛ל פְּנֵ֥י הָאֲדָמָ֖ה נְאֻם־ יְהוָֽה׃ (ג) אָסֵ֨ף אָדָ֜ם וּבְהֵמָ֗ה אָסֵ֤ף עוֹף־ הַשָּׁמַ֙יִם֙ וּדְגֵ֣י הַיָּ֔ם וְהַמַּכְשֵׁל֖וֹת אֶת־ הָרְשָׁעִ֑ים וְהִכְרַתִּ֣י אֶת־ הָאָדָ֗ם מֵעַ֛ל פְּנֵ֥י הָאֲדָמָ֖ה נְאֻם־ יְהוָֽה׃ (ד) וְנָטִ֤יתִי יָדִי֙ עַל־ יְהוּדָ֔ה וְעַ֖ל כָּל־ יוֹשְׁבֵ֣י יְרוּשָׁלִָ֑ם וְהִכְרַתִּ֞י מִן־ הַמָּק֤וֹם הַזֶּה֙ אֶת־ שְׁאָ֣ר הַבַּ֔עַל אֶת־ שֵׁ֥ם הַכְּמָרִ֖ים עִם־ הַכֹּהֲנִֽים׃ (ה) וְאֶת־ הַמִּשְׁתַּחֲוִ֥ים עַל־ הַגַּגּ֖וֹת לִצְבָ֣א הַשָּׁמָ֑יִם וְאֶת־ הַמִּֽשְׁתַּחֲוִים֙ הַנִּשְׁבָּעִ֣ים לַֽיהוָ֔ה וְהַנִּשְׁבָּעִ֖ים בְּמַלְכָּֽם׃ (ו) וְאֶת־ הַנְּסוֹגִ֖ים מֵאַחֲרֵ֣י יְהוָ֑ה וַאֲשֶׁ֛ר לֹֽא־ בִקְשׁ֥וּ אֶת־ יְהוָ֖ה וְלֹ֥א דְרָשֻֽׁהוּ׃ (ז) הַ֕ס מִפְּנֵ֖י אֲדֹנָ֣י יְהוִ֑ה כִּ֤י קָרוֹב֙ י֣וֹם יְהוָ֔ה כִּֽי־ הֵכִ֧ין יְהוָ֛ה זֶ֖בַח הִקְדִּ֥ישׁ קְרֻאָֽיו׃ (ח) וְהָיָ֗ה בְּיוֹם֙ זֶ֣בַח יְהוָ֔ה וּפָקַדְתִּ֥י עַל־ הַשָּׂרִ֖ים וְעַל־ בְּנֵ֣י הַמֶּ֑לֶךְ וְעַ֥ל כָּל־ הַלֹּבְשִׁ֖ים מַלְבּ֥וּשׁ נָכְרִֽי׃ (ט) וּפָקַדְתִּ֗י עַ֧ל כָּל־ הַדּוֹלֵ֛ג עַל־ הַמִּפְתָּ֖ן בַּיּ֣וֹם הַה֑וּא הַֽמְמַלְאִ֛ים בֵּ֥ית אֲדֹנֵיהֶ֖ם חָמָ֥ס וּמִרְמָֽה׃ (י) וְהָיָה֩ בַיּ֨וֹם הַה֜וּא נְאֻם־ יְהוָ֗ה ק֤וֹל צְעָקָה֙ מִשַּׁ֣עַר הַדָּגִ֔ים וִֽילָלָ֖ה מִן־ הַמִּשְׁנֶ֑ה וְשֶׁ֥בֶר גָּד֖וֹל מֵהַגְּבָעֽוֹת׃ (יא) הֵילִ֖ילוּ יֹשְׁבֵ֣י הַמַּכְתֵּ֑שׁ כִּ֤י נִדְמָה֙ כָּל־ עַ֣ם כְּנַ֔עַן נִכְרְת֖וּ כָּל־ נְטִ֥ילֵי כָֽסֶף׃ (יב) וְהָיָה֙ בָּעֵ֣ת הַהִ֔יא אֲחַפֵּ֥שׂ אֶת־ יְרוּשָׁלִַ֖ם בַּנֵּר֑וֹת וּפָקַדְתִּ֣י עַל־ הָאֲנָשִׁ֗ים הַקֹּֽפְאִים֙ עַל־ שִׁמְרֵיהֶ֔ם הָאֹֽמְרִים֙ בִּלְבָבָ֔ם לֹֽא־ יֵיטִ֥יב יְהוָ֖ה וְלֹ֥א יָרֵֽעַ׃ (יג) וְהָיָ֤ה חֵילָם֙ לִמְשִׁסָּ֔ה וּבָתֵּיהֶ֖ם לִשְׁמָמָ֑ה וּבָנ֤וּ בָתִּים֙ וְלֹ֣א יֵשֵׁ֔בוּ וְנָטְע֣וּ כְרָמִ֔ים וְלֹ֥א יִשְׁתּ֖וּ אֶת־ יֵינָֽם׃ (יד) קָר֤וֹב יוֹם־ יְהוָה֙ הַגָּד֔וֹל קָר֖וֹב וּמַהֵ֣ר מְאֹ֑ד ק֚וֹל י֣וֹם יְהוָ֔ה מַ֥ר צֹרֵ֖חַ שָׁ֥ם גִּבּֽוֹר׃ (טו) י֥וֹם עֶבְרָ֖ה הַיּ֣וֹם הַה֑וּא י֧וֹם צָרָ֣ה וּמְצוּקָ֗ה י֤וֹם שֹׁאָה֙ וּמְשׁוֹאָ֔ה י֥וֹם חֹ֙שֶׁךְ֙ וַאֲפֵלָ֔ה י֥וֹם עָנָ֖ן וַעֲרָפֶֽל׃ (טז) י֥וֹם שׁוֹפָ֖ר וּתְרוּעָ֑ה עַ֚ל הֶעָרִ֣ים הַבְּצֻר֔וֹת וְעַ֖ל הַפִּנּ֥וֹת הַגְּבֹהֽוֹת׃ (יז) וַהֲצֵרֹ֣תִי לָאָדָ֗ם וְהָֽלְכוּ֙ כַּֽעִוְרִ֔ים כִּ֥י לַֽיהוָ֖ה חָטָ֑אוּ וְשֻׁפַּ֤ךְ דָּמָם֙ כֶּֽעָפָ֔ר וּלְחֻמָ֖ם כַּגְּלָלִֽים׃ (יח) גַּם־ כַּסְפָּ֨ם גַּם־ זְהָבָ֜ם לֹֽא־ יוּכַ֣ל לְהַצִּילָ֗ם בְּיוֹם֙ עֶבְרַ֣ת יְהוָ֔ה וּבְאֵשׁ֙ קִנְאָת֔וֹ תֵּאָכֵ֖ל כָּל־ הָאָ֑רֶץ כִּֽי־ כָלָ֤ה אַךְ־ נִבְהָלָה֙ יַֽעֲשֶׂ֔ה אֵ֥ת כָּל־ יֹשְׁבֵ֖י הָאָֽרֶץ׃ (יט) הִֽתְקוֹשְׁשׁ֖וּ וָק֑וֹשּׁוּ הַגּ֖וֹי לֹ֥א נִכְסָֽף׃ (כ) בְּטֶ֙רֶם֙ לֶ֣דֶת חֹ֔ק כְּמֹ֖ץ עָ֣בַר י֑וֹם בְּטֶ֣רֶם ׀ לֹא־ יָב֣וֹא עֲלֵיכֶ֗ם חֲרוֹן֙ אַף־ יְהוָ֔ה בְּטֶ֙רֶם֙ לֹא־ יָב֣וֹא עֲלֵיכֶ֔ם י֖וֹם אַף־ יְהוָֽה׃ (כא) בַּקְּשׁ֤וּ אֶת־ יְהוָה֙ כָּל־ עַנְוֵ֣י הָאָ֔רֶץ אֲשֶׁ֥ר מִשְׁפָּט֖וֹ פָּעָ֑לוּ בַּקְּשׁוּ־ צֶ֙דֶק֙ בַּקְּשׁ֣וּ עֲנָוָ֔ה אוּלַי֙ תִּסָּ֣תְר֔וּ בְּי֖וֹם אַף־ יְהוָֽה׃ (כב) כִּ֤י עַזָּה֙ עֲזוּבָ֣ה תִֽהְיֶ֔ה וְאַשְׁקְל֖וֹן לִשְׁמָמָ֑ה אַשְׁדּ֗וֹד בַּֽצָּהֳרַ֙יִם֙ יְגָ֣רְשׁ֔וּהָ וְעֶקְר֖וֹן תֵּעָקֵֽר׃ (כג) ה֗וֹי יֹֽשְׁבֵ֛י חֶ֥בֶל הַיָּ֖ם גּ֣וֹי כְּרֵתִ֑ים דְּבַר־ יְהוָ֣ה עֲלֵיכֶ֗ם כְּנַ֙עַן֙ אֶ֣רֶץ פְּלִשְׁתִּ֔ים וְהַאֲבַדְתִּ֖יךְ מֵאֵ֥ין יוֹשֵֽׁב׃ (כד) וְֽהָיְתָ֞ה חֶ֣בֶל הַיָּ֗ם נְוֺ֛ת כְּרֹ֥ת רֹעִ֖ים וְגִדְר֥וֹת צֹֽאן׃ (כה) וְהָ֣יָה חֶ֗בֶל לִשְׁאֵרִ֛ית בֵּ֥ית יְהוּדָ֖ה עֲלֵיהֶ֣ם יִרְע֑וּן בְּבָתֵּ֣י אַשְׁקְל֗וֹן בָּעֶ֙רֶב֙ יִרְבָּצ֔וּן כִּ֧י יִפְקְדֵ֛ם יְהוָ֥ה אֱלֹהֵיהֶ֖ם וְשָׁ֥ב (שבותם) שְׁבִיתָֽם׃ (כו) שָׁמַ֙עְתִּי֙ חֶרְפַּ֣ת מוֹאָ֔ב וְגִדּוּפֵ֖י בְּנֵ֣י עַמּ֑וֹן אֲשֶׁ֤ר חֵֽרְפוּ֙ אֶת־ עַמִּ֔י וַיַּגְדִּ֖ילוּ עַל־ גְּבוּלָֽם׃ (כז) לָכֵ֣ן חַי־ אָ֡נִי נְאֻם֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל כִּֽי־ מוֹאָ֞ב כִּסְדֹ֤ם תִּֽהְיֶה֙ וּבְנֵ֤י עַמּוֹן֙ כַּֽעֲמֹרָ֔ה מִמְשַׁ֥ק חָר֛וּל וּמִכְרֵה־ מֶ֥לַח וּשְׁמָמָ֖ה עַד־ עוֹלָ֑ם שְׁאֵרִ֤ית עַמִּי֙ יְבָזּ֔וּם וְיֶ֥תֶר (גוי) גּוֹיִ֖י יִנְחָלֽוּם׃ (כח) זֹ֥את לָהֶ֖ם תַּ֣חַת גְּאוֹנָ֑ם כִּ֤י חֵֽרְפוּ֙ וַיַּגְדִּ֔לוּ עַל־ עַ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כט) נוֹרָ֤א יְהוָה֙ עֲלֵיהֶ֔ם כִּ֣י רָזָ֔ה אֵ֖ת כָּל־ אֱלֹהֵ֣י הָאָ֑רֶץ וְיִשְׁתַּֽחֲווּ־ לוֹ֙ אִ֣ישׁ מִמְּקוֹמ֔וֹ כֹּ֖ל אִיֵּ֥י הַגּוֹיִֽם׃ (ל) גַּם־ אַתֶּ֣ם כּוּשִׁ֔ים חַֽלְלֵ֥י חַרְבִּ֖י הֵֽמָּה׃ (לא) וְיֵ֤ט יָדוֹ֙ עַל־ צָפ֔וֹן וִֽיאַבֵּ֖ד אֶת־ אַשּׁ֑וּר וְיָשֵׂ֤ם אֶת־ נִֽינְוֵה֙ לִשְׁמָמָ֔ה צִיָּ֖ה כַּמִּדְבָּֽר׃ (לב) וְרָבְצ֨וּ בְתוֹכָ֤הּ עֲדָרִים֙ כָּל־ חַיְתוֹ־ ג֔וֹי גַּם־ קָאַת֙ גַּם־ קִפֹּ֔ד בְּכַפְתֹּרֶ֖יהָ יָלִ֑ינוּ ק֠וֹל יְשׁוֹרֵ֤ר בַּֽחַלּוֹן֙ חֹ֣רֶב בַּסַּ֔ף כִּ֥י אַרְזָ֖ה עֵרָֽה׃ (לג) זֹ֞֠את הָעִ֤יר הָעַלִּיזָה֙ הַיּוֹשֶׁ֣בֶת לָבֶ֔טַח הָאֹֽמְרָה֙ בִּלְבָבָ֔הּ אֲנִ֖י וְאַפְסִ֣י ע֑וֹד אֵ֣יךְ ׀ הָיְתָ֣ה לְשַׁמָּ֗ה מַרְבֵּץ֙ לַֽחַיָּ֔ה כֹּ֚ל עוֹבֵ֣ר עָלֶ֔יהָ יִשְׁרֹ֖ק יָנִ֥יעַ יָדֽוֹ׃ תחילת פרק ג (לד) ה֥וֹי מֹרְאָ֖ה וְנִגְאָלָ֑ה הָעִ֖יר הַיּוֹנָֽה׃ (לה) לֹ֤א שָֽׁמְעָה֙ בְּק֔וֹל לֹ֥א לָקְחָ֖ה מוּסָ֑ר בַּֽיהוָה֙ לֹ֣א בָטָ֔חָה אֶל־ אֱלֹהֶ֖יהָ לֹ֥א קָרֵֽבָה׃ (לו) שָׂרֶ֣יהָ בְקִרְבָּ֔הּ אֲרָי֖וֹת שֹֽׁאֲגִ֑ים שֹׁפְטֶ֙יהָ֙ זְאֵ֣בֵי עֶ֔רֶב לֹ֥א גָרְמ֖וּ לַבֹּֽקֶר׃ (לז) נְבִיאֶ֙יהָ֙ פֹּֽחֲזִ֔ים אַנְשֵׁ֖י בֹּֽגְד֑וֹת כֹּהֲנֶ֙יהָ֙ חִלְּלוּ־ קֹ֔דֶשׁ חָמְס֖וּ תּוֹרָֽה׃ (לח) יְהוָ֤ה צַדִּיק֙ בְּקִרְבָּ֔הּ לֹ֥א יַעֲשֶׂ֖ה עַוְלָ֑ה בַּבֹּ֨קֶר בַּבֹּ֜קֶר מִשְׁפָּט֨וֹ יִתֵּ֤ן לָאוֹר֙ לֹ֣א נֶעְדָּ֔ר וְלֹֽא־ יוֹדֵ֥עַ עַוָּ֖ל בֹּֽשֶׁת׃ (לט) הִכְרַ֣תִּי גוֹיִ֗ם נָשַׁ֙מּוּ֙ פִּנּוֹתָ֔ם הֶחֱרַ֥בְתִּי חֽוּצוֹתָ֖ם מִבְּלִ֣י עוֹבֵ֑ר נִצְדּ֧וּ עָרֵיהֶ֛ם מִבְּלִי־ אִ֖ישׁ מֵאֵ֥ין יוֹשֵֽׁב׃ (מ) אָמַ֜רְתִּי אַךְ־ תִּירְאִ֤י אוֹתִי֙ תִּקְחִ֣י מוּסָ֔ר וְלֹֽא־ יִכָּרֵ֣ת מְעוֹנָ֔הּ כֹּ֥ל אֲשֶׁר־ פָּקַ֖דְתִּי עָלֶ֑יהָ פרק ג (ב) בִּשְׁנַ֤ת שְׁתַּ֙יִם֙ לְדָרְיָ֣וֶשׁ הַמֶּ֔לֶךְ בַּחֹ֙דֶשׁ֙ הַשִּׁשִּׁ֔י בְּי֥וֹם אֶחָ֖ד לַחֹ֑דֶשׁ הָיָ֨ה דְבַר־ יְהוָ֜ה בְּיַד־ חַגַּ֣י הַנָּבִ֗יא אֶל־ זְרֻבָּבֶ֤ל בֶּן־ שְׁאַלְתִּיאֵל֙ אֶֽת־ הַדָּבָ֞ר אֲשֶׁר־ כָּרַ֤תִּי אִתְּכֶם֙ בְּצֵאתְכֶ֣ם מִמִּצְרַ֔יִם וְרוּחִ֖י עֹמֶ֣דֶת בְּתוֹכְכֶ֑ם אַל־ תִּירָֽאוּ׃ (כב) כִּ֣י כֹ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת ע֥וֹד אַחַ֖ת מְעַ֣ט הִ֑יא וַאֲנִ֗י מַרְעִישׁ֙ אֶת־ הַשָּׁמַ֣יִם וְאֶת־ הָאָ֔רֶץ וְאֶת־ הַיָּ֖ם וְאֶת־ הֶחָרָבָֽה׃ (כג) וְהִרְעַשְׁתִּי֙ אֶת־ כָּל־ הַגּוֹיִ֔ם וּבָ֖אוּ חֶמְדַּ֣ת כָּל־ הַגּוֹיִ֑ם וּמִלֵּאתִ֞י אֶת־ הַבַּ֤יִת הַזֶּה֙ כָּב֔וֹד אָמַ֖ר יְהוָ֥ה צְבָאֽוֹת׃ (כד) לִ֥י הַכֶּ֖סֶף וְלִ֣י הַזָּהָ֑ב נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כה) גָּד֣וֹל יִֽהְיֶ֡ה כְּבוֹד֩ הַבַּ֨יִת הַזֶּ֤ה הָאַֽחֲרוֹן֙ מִן־ הָ֣רִאשׁ֔וֹן אָמַ֖ר יְהוָ֣ה צְבָא֑וֹת וּבַמָּק֤וֹם הַזֶּה֙ אֶתֵּ֣ן שָׁל֔וֹם נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כו) בְּעֶשְׂרִ֤ים וְאַרְבָּעָה֙ לַתְּשִׁיעִ֔י בִּשְׁנַ֥ת שְׁתַּ֖יִם לְדָרְיָ֑וֶשׁ הָיָה֙ דְּבַר־ יְהוָ֔ה אֶל־ חַגַּ֥י הַנָּבִ֖יא לֵאמֹֽר׃ (כז) כֹּ֥ה אָמַ֖ר יְהוָ֣ה צְבָא֑וֹת שְׁאַל־ נָ֧א אֶת־ הַכֹּהֲנִ֛ים תּוֹרָ֖ה לֵאמֹֽר׃ (כח) הֵ֣ן ׀ יִשָּׂא־ אִ֨ישׁ בְּשַׂר־ קֹ֜דֶשׁ בִּכְנַ֣ף בִּגְד֗וֹ וְנָגַ֣ע בִּ֠כְנָפוֹ אֶל־ הַלֶּ֨חֶם וְאֶל־ הַנָּזִ֜יד וְאֶל־ הַיַּ֧יִן וְאֶל־ שֶׁ֛מֶן וְאֶל־ כָּל־ מַאֲכָ֖ל הֲיִקְדָּ֑שׁ וַיַּעֲנ֧וּ הַכֹּהֲנִ֛ים וַיֹּאמְר֖וּ לֹֽא׃ (כט) וַיֹּ֣אמֶר חַגַּ֔י אִם־ יִגַּ֧ע טְמֵא־ נֶ֛פֶשׁ בְּכָל־ אֵ֖לֶּה הֲיִטְמָ֑א וַיַּעֲנ֧וּ הַכֹּהֲנִ֛ים וַיֹּאמְר֖וּ יִטְמָֽא׃ (ל) וַיַּ֨עַן חַגַּ֜י וַיֹּ֗אמֶר כֵּ֣ן הָֽעָם־ הַ֠זֶּה וְכֵן־ הַגּ֨וֹי הַזֶּ֤ה לְפָנַי֙ נְאֻם־ יְהוָ֔ה וְכֵ֖ן כָּל־ מַעֲשֵׂ֣ה יְדֵיהֶ֑ם וַאֲשֶׁ֥ר יַקְרִ֛יבוּ שָׁ֖ם טָמֵ֥א הֽוּא׃ (לא) וְעַתָּה֙ שִֽׂימוּ־ נָ֣א לְבַבְכֶ֔ם מִן־ הַיּ֥וֹם הַזֶּ֖ה וָמָ֑עְלָה מִטֶּ֧רֶם שֽׂוּם־ אֶ֛בֶן אֶל־ אֶ֖בֶן בְּהֵיכַ֥ל יְהוָֽה׃ (לב) מִֽהְיוֹתָ֥ם בָּא֙ אֶל־ עֲרֵמַ֣ת עֶשְׂרִ֔ים וְהָיְתָ֖ה עֲשָׂרָ֑ה בָּ֣א אֶל־ הַיֶּ֗קֶב לַחְשֹׂף֙ חֲמִשִּׁ֣ים פּוּרָ֔ה וְהָיְתָ֖ה עֶשְׂרִֽים׃ (לג) הִכֵּ֨יתִי אֶתְכֶ֜ם בַּשִּׁדָּפ֤וֹן וּבַיֵּֽרָקוֹן֙ וּבַבָּרָ֔ד אֵ֖ת כָּל־ מַעֲשֵׂ֣ה יְדֵיכֶ֑ם וְאֵין־ אֶתְכֶ֥ם אֵלַ֖י נְאֻם־ יְהוָֽה׃ (לד) שִׂימוּ־ נָ֣א לְבַבְכֶ֔ם מִן־ הַיּ֥וֹם הַזֶּ֖ה וָמָ֑עְלָה מִיּוֹם֩ עֶשְׂרִ֨ים וְאַרְבָּעָ֜ה לַתְּשִׁיעִ֗י לְמִן־ הַיּ֛וֹם אֲשֶׁר־ יֻסַּ֥ד הֵֽיכַל־ יְהוָ֖ה שִׂ֥ימוּ לְבַבְכֶֽם׃ (לה) הַע֤וֹד הַזֶּ֙רַע֙ בַּמְּגוּרָ֔ה וְעַד־ הַגֶּ֨פֶן וְהַתְּאֵנָ֧ה וְהָרִמּ֛וֹן וְעֵ֥ץ הַזַּ֖יִת לֹ֣א נָשָׂ֑א מִן־ הַיּ֥וֹם הַזֶּ֖ה אֲבָרֵֽךְ׃ (לו) וַיְהִ֨י דְבַר־ יְהוָ֤ה ׀ שֵׁנִית֙ אֶל־ חַגַּ֔י בְּעֶשְׂרִ֧ים וְאַרְבָּעָ֛ה לַחֹ֖דֶשׁ לֵאמֹֽר׃ (לז) אֱמֹ֕ר אֶל־ זְרֻבָּבֶ֥ל פַּֽחַת־ יְהוּדָ֖ה לֵאמֹ֑ר אֲנִ֣י מַרְעִ֔ישׁ אֶת־ הַשָּׁמַ֖יִם וְאֶת־ הָאָֽרֶץ׃ (לח) וְהָֽפַכְתִּי֙ כִּסֵּ֣א מַמְלָכ֔וֹת וְהִ֨שְׁמַדְתִּ֔י חֹ֖זֶק מַמְלְכ֣וֹת הַגּוֹיִ֑ם וְהָפַכְתִּ֤י מֶרְכָּבָה֙ וְרֹ֣כְבֶ֔יהָ וְיָרְד֤וּ סוּסִים֙ וְרֹ֣כְבֵיהֶ֔ם אִ֖ישׁ בְּחֶ֥רֶב אָחִֽיו׃
