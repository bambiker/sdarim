בראשית פרק לג לפי החלוקה לסדרים

פרק לג בספר בראשית (20 פסוקים) נמצא בבראשית סדר ל (פסוקים לא–מז) ובבראשית סדר לא (פסוקים א–ג).

בראשית סדר ל

(א) וַיִּשְׁלַ֨ח יַעֲקֹ֤ב מַלְאָכִים֙ לְפָנָ֔יו אֶל־ עֵשָׂ֖ו אָחִ֑יו אַ֥רְצָה שֵׂעִ֖יר שְׂדֵ֥ה אֱדֽוֹם׃ (ב) וַיְצַ֤ו אֹתָם֙ לֵאמֹ֔ר כֹּ֣ה תֹאמְר֔וּן לַֽאדֹנִ֖י לְעֵשָׂ֑ו כֹּ֤ה אָמַר֙ עַבְדְּךָ֣ יַעֲקֹ֔ב עִם־ לָבָ֣ן גַּ֔רְתִּי וָאֵחַ֖ר עַד־ עָֽתָּה׃ (ג) וַֽיְהִי־ לִי֙ שׁ֣וֹר וַחֲמ֔וֹר צֹ֖אן וְעֶ֣בֶד וְשִׁפְחָ֑ה וָֽאֶשְׁלְחָה֙ לְהַגִּ֣יד לַֽאדֹנִ֔י לִמְצֹא־ חֵ֖ן בְּעֵינֶֽיךָ׃ (ד) וַיָּשֻׁ֙בוּ֙ הַמַּלְאָכִ֔ים אֶֽל־ יַעֲקֹ֖ב לֵאמֹ֑ר בָּ֤אנוּ אֶל־ אָחִ֙יךָ֙ אֶל־ עֵשָׂ֔ו וְגַם֙ הֹלֵ֣ךְ לִקְרָֽאתְךָ֔ וְאַרְבַּע־ מֵא֥וֹת אִ֖ישׁ עִמּֽוֹ׃ (ה) וַיִּירָ֧א יַעֲקֹ֛ב מְאֹ֖ד וַיֵּ֣צֶר ל֑וֹ וַיַּ֜חַץ אֶת־ הָעָ֣ם אֲשֶׁר־ אִתּ֗וֹ וְאֶת־ הַצֹּ֧אן וְאֶת־ הַבָּקָ֛ר וְהַגְּמַלִּ֖ים לִשְׁנֵ֥י מַחֲנֽוֹת׃ (ו) וַיֹּ֕אמֶר אִם־ יָב֥וֹא עֵשָׂ֛ו אֶל־ הַמַּחֲנֶ֥ה הָאַחַ֖ת וְהִכָּ֑הוּ וְהָיָ֛ה הַמַּחֲנֶ֥ה הַנִּשְׁאָ֖ר לִפְלֵיטָֽה׃ (ז) וַיֹּאמֶר֮ יַעֲקֹב֒ אֱלֹהֵי֙ אָבִ֣י אַבְרָהָ֔ם וֵאלֹהֵ֖י אָבִ֣י יִצְחָ֑ק יְהוָ֞ה הָאֹמֵ֣ר אֵלַ֗י שׁ֧וּב לְאַרְצְךָ֛ וּלְמוֹלַדְתְּךָ֖ וְאֵיטִ֥יבָה עִמָּֽךְ׃ (ח) קָטֹ֜נְתִּי מִכֹּ֤ל הַחֲסָדִים֙ וּמִכָּל־ הָ֣אֱמֶ֔ת אֲשֶׁ֥ר עָשִׂ֖יתָ אֶת־ עַבְדֶּ֑ךָ כִּ֣י בְמַקְלִ֗י עָבַ֙רְתִּי֙ אֶת־ הַיַּרְדֵּ֣ן הַזֶּ֔ה וְעַתָּ֥ה הָיִ֖יתִי לִשְׁנֵ֥י מַחֲנֽוֹת׃ (ט) הַצִּילֵ֥נִי נָ֛א מִיַּ֥ד אָחִ֖י מִיַּ֣ד עֵשָׂ֑ו כִּֽי־ יָרֵ֤א אָנֹכִי֙ אֹת֔וֹ פֶּן־ יָב֣וֹא וְהִכַּ֔נִי אֵ֖ם עַל־ בָּנִֽים׃ (י) וְאַתָּ֣ה אָמַ֔רְתָּ הֵיטֵ֥ב אֵיטִ֖יב עִמָּ֑ךְ וְשַׂמְתִּ֤י אֶֽת־ זַרְעֲךָ֙ כְּח֣וֹל הַיָּ֔ם אֲשֶׁ֥ר לֹא־ יִסָּפֵ֖ר מֵרֹֽב׃ (יא) וַיָּ֥לֶן שָׁ֖ם בַּלַּ֣יְלָה הַה֑וּא וַיִּקַּ֞ח מִן־ הַבָּ֧א בְיָד֛וֹ מִנְחָ֖ה לְעֵשָׂ֥ו אָחִֽיו׃ (יב) עִזִּ֣ים מָאתַ֔יִם וּתְיָשִׁ֖ים עֶשְׂרִ֑ים רְחֵלִ֥ים מָאתַ֖יִם וְאֵילִ֥ים עֶשְׂרִֽים׃ (יג) גְּמַלִּ֧ים מֵינִיק֛וֹת וּבְנֵיהֶ֖ם שְׁלֹשִׁ֑ים פָּר֤וֹת אַרְבָּעִים֙ וּפָרִ֣ים עֲשָׂרָ֔ה אֲתֹנֹ֣ת עֶשְׂרִ֔ים וַעְיָרִ֖ם עֲשָׂרָֽה׃ (יד) וַיִּתֵּן֙ בְּיַד־ עֲבָדָ֔יו עֵ֥דֶר עֵ֖דֶר לְבַדּ֑וֹ וַ֤יֹּאמֶר אֶל־ עֲבָדָיו֙ עִבְר֣וּ לְפָנַ֔י וְרֶ֣וַח תָּשִׂ֔ימוּ בֵּ֥ין עֵ֖דֶר וּבֵ֥ין עֵֽדֶר׃ (טו) וַיְצַ֥ו אֶת־ הָרִאשׁ֖וֹן לֵאמֹ֑ר כִּ֣י יִֽפְגָּשְׁךָ֞ עֵשָׂ֣ו אָחִ֗י וִשְׁאֵֽלְךָ֙ לֵאמֹ֔ר לְמִי־ אַ֙תָּה֙ וְאָ֣נָה תֵלֵ֔ךְ וּלְמִ֖י אֵ֥לֶּה לְפָנֶֽיךָ׃ (טז) וְאָֽמַרְתָּ֙ לְעַבְדְּךָ֣ לְיַעֲקֹ֔ב מִנְחָ֥ה הִוא֙ שְׁלוּחָ֔ה לַֽאדֹנִ֖י לְעֵשָׂ֑ו וְהִנֵּ֥ה גַם־ ה֖וּא אַחֲרֵֽינוּ׃ (יז) וַיְצַ֞ו גַּ֣ם אֶת־ הַשֵּׁנִ֗י גַּ֚ם אֶת־ הַשְּׁלִישִׁ֔י גַּ֚ם אֶת־ כָּל־ הַהֹ֣לְכִ֔ים אַחֲרֵ֥י הָעֲדָרִ֖ים לֵאמֹ֑ר כַּדָּבָ֤ר הַזֶּה֙ תְּדַבְּר֣וּן אֶל־ עֵשָׂ֔ו בְּמֹצַאֲכֶ֖ם אֹתֽוֹ׃ (יח) וַאֲמַרְתֶּ֕ם גַּ֗ם הִנֵּ֛ה עַבְדְּךָ֥ יַעֲקֹ֖ב אַחֲרֵ֑ינוּ כִּֽי־ אָמַ֞ר אֲכַפְּרָ֣ה פָנָ֗יו בַּמִּנְחָה֙ הַהֹלֶ֣כֶת לְפָנָ֔י וְאַחֲרֵי־ כֵן֙ אֶרְאֶ֣ה פָנָ֔יו אוּלַ֖י יִשָּׂ֥א פָנָֽי׃ (יט) וַתַּעֲבֹ֥ר הַמִּנְחָ֖ה עַל־ פָּנָ֑יו וְה֛וּא לָ֥ן בַּלַּֽיְלָה־ הַה֖וּא בַּֽמַּחֲנֶֽה׃ (כ) וַיָּ֣קָם ׀ בַּלַּ֣יְלָה ה֗וּא וַיִּקַּ֞ח אֶת־ שְׁתֵּ֤י נָשָׁיו֙ וְאֶת־ שְׁתֵּ֣י שִׁפְחֹתָ֔יו וְאֶת־ אַחַ֥ד עָשָׂ֖ר יְלָדָ֑יו וַֽיַּעֲבֹ֔ר אֵ֖ת מַעֲבַ֥ר יַבֹּֽק׃ (כא) וַיִּקָּחֵ֔ם וַיַּֽעֲבִרֵ֖ם אֶת־ הַנָּ֑חַל וַֽיַּעֲבֵ֖ר אֶת־ אֲשֶׁר־ לוֹ׃ (כב) וַיִּוָּתֵ֥ר יַעֲקֹ֖ב לְבַדּ֑וֹ וַיֵּאָבֵ֥ק אִישׁ֙ עִמּ֔וֹ עַ֖ד עֲל֥וֹת הַשָּֽׁחַר׃ (כג) וַיַּ֗רְא כִּ֣י לֹ֤א יָכֹל֙ ל֔וֹ וַיִּגַּ֖ע בְּכַף־ יְרֵכ֑וֹ וַתֵּ֙קַע֙ כַּף־ יֶ֣רֶךְ יַעֲקֹ֔ב בְּהֵֽאָבְק֖וֹ עִמּֽוֹ׃ (כד) וַיֹּ֣אמֶר שַׁלְּחֵ֔נִי כִּ֥י עָלָ֖ה הַשָּׁ֑חַר וַיֹּ֙אמֶר֙ לֹ֣א אֲשַֽׁלֵּחֲךָ֔ כִּ֖י אִם־ בֵּרַכְתָּֽנִי׃ (כה) וַיֹּ֥אמֶר אֵלָ֖יו מַה־ שְּׁמֶ֑ךָ וַיֹּ֖אמֶר יַעֲקֹֽב׃ (כו) וַיֹּ֗אמֶר לֹ֤א יַעֲקֹב֙ יֵאָמֵ֥ר עוֹד֙ שִׁמְךָ֔ כִּ֖י אִם־ יִשְׂרָאֵ֑ל כִּֽי־ שָׂרִ֧יתָ עִם־ אֱלֹהִ֛ים וְעִם־ אֲנָשִׁ֖ים וַתּוּכָֽל׃ (כז) וַיִּשְׁאַ֣ל יַעֲקֹ֗ב וַיֹּ֙אמֶר֙ הַגִּֽידָה־ נָּ֣א שְׁמֶ֔ךָ וַיֹּ֕אמֶר לָ֥מָּה זֶּ֖ה תִּשְׁאַ֣ל לִשְׁמִ֑י וַיְבָ֥רֶךְ אֹת֖וֹ שָֽׁם׃ (כח) וַיִּקְרָ֧א יַעֲקֹ֛ב שֵׁ֥ם הַמָּק֖וֹם פְּנִיאֵ֑ל כִּֽי־ רָאִ֤יתִי אֱלֹהִים֙ פָּנִ֣ים אֶל־ פָּנִ֔ים וַתִּנָּצֵ֖ל נַפְשִֽׁי׃ (כט) וַיִּֽזְרַֽח־ ל֣וֹ הַשֶּׁ֔מֶשׁ כַּאֲשֶׁ֥ר עָבַ֖ר אֶת־ פְּנוּאֵ֑ל וְה֥וּא צֹלֵ֖עַ עַל־ יְרֵכֽוֹ׃ (ל) עַל־ כֵּ֡ן לֹֽא־ יֹאכְל֨וּ בְנֵֽי־ יִשְׂרָאֵ֜ל אֶת־ גִּ֣יד הַנָּשֶׁ֗ה אֲשֶׁר֙ עַל־ כַּ֣ף הַיָּרֵ֔ךְ עַ֖ד הַיּ֣וֹם הַזֶּ֑ה כִּ֤י נָגַע֙ בְּכַף־ יֶ֣רֶךְ יַעֲקֹ֔ב בְּגִ֖יד הַנָּשֶֽׁה׃ תחילת פרק לג (לא) וַיִּשָּׂ֨א יַעֲקֹ֜ב עֵינָ֗יו וַיַּרְא֙ וְהִנֵּ֣ה עֵשָׂ֣ו בָּ֔א וְעִמּ֕וֹ אַרְבַּ֥ע מֵא֖וֹת אִ֑ישׁ וַיַּ֣חַץ אֶת־ הַיְלָדִ֗ים עַל־ לֵאָה֙ וְעַל־ רָחֵ֔ל וְעַ֖ל שְׁתֵּ֥י הַשְּׁפָחֽוֹת׃ (לב) וַיָּ֧שֶׂם אֶת־ הַשְּׁפָח֛וֹת וְאֶת־ יַלְדֵיהֶ֖ן רִֽאשֹׁנָ֑ה וְאֶת־ לֵאָ֤ה וִֽילָדֶ֙יהָ֙ אַחֲרֹנִ֔ים וְאֶת־ רָחֵ֥ל וְאֶת־ יוֹסֵ֖ף אַחֲרֹנִֽים׃ (לג) וְה֖וּא עָבַ֣ר לִפְנֵיהֶ֑ם וַיִּשְׁתַּ֤חוּ אַ֙רְצָה֙ שֶׁ֣בַע פְּעָמִ֔ים עַד־ גִּשְׁתּ֖וֹ עַד־ אָחִֽיו׃ (לד) וַיָּ֨רָץ עֵשָׂ֤ו לִקְרָאתוֹ֙ וַֽיְחַבְּקֵ֔הוּ וַיִּפֹּ֥ל עַל־ צַוָּארָ֖ו וַׄיִּׄשָּׁׄקֵ֑ׄהׄוּׄ וַיִּבְכּֽוּ׃ (לה) וַיִּשָּׂ֣א אֶת־ עֵינָ֗יו וַיַּ֤רְא אֶת־ הַנָּשִׁים֙ וְאֶת־ הַיְלָדִ֔ים וַיֹּ֖אמֶר מִי־ אֵ֣לֶּה לָּ֑ךְ וַיֹּאמַ֕ר הַיְלָדִ֕ים אֲשֶׁר־ חָנַ֥ן אֱלֹהִ֖ים אֶת־ עַבְדֶּֽךָ׃ (לו) וַתִּגַּ֧שְׁןָ הַשְּׁפָח֛וֹת הֵ֥נָּה וְיַלְדֵיהֶ֖ן וַתִּֽשְׁתַּחֲוֶֽיןָ׃ (לז) וַתִּגַּ֧שׁ גַּם־ לֵאָ֛ה וִילָדֶ֖יהָ וַיִּֽשְׁתַּחֲו֑וּ וְאַחַ֗ר נִגַּ֥שׁ יוֹסֵ֛ף וְרָחֵ֖ל וַיִּֽשְׁתַּחֲוֽוּ׃ (לח) וַיֹּ֕אמֶר מִ֥י לְךָ֛ כָּל־ הַמַּחֲנֶ֥ה הַזֶּ֖ה אֲשֶׁ֣ר פָּגָ֑שְׁתִּי וַיֹּ֕אמֶר לִמְצֹא־ חֵ֖ן בְּעֵינֵ֥י אֲדֹנִֽי׃ (לט) וַיֹּ֥אמֶר עֵשָׂ֖ו יֶשׁ־ לִ֣י רָ֑ב אָחִ֕י יְהִ֥י לְךָ֖ אֲשֶׁר־ לָֽךְ׃ (מ) וַיֹּ֣אמֶר יַעֲקֹ֗ב אַל־ נָא֙ אִם־ נָ֨א מָצָ֤אתִי חֵן֙ בְּעֵינֶ֔יךָ וְלָקַחְתָּ֥ מִנְחָתִ֖י מִיָּדִ֑י כִּ֣י עַל־ כֵּ֞ן רָאִ֣יתִי פָנֶ֗יךָ כִּרְאֹ֛ת פְּנֵ֥י אֱלֹהִ֖ים וַתִּרְצֵֽנִי׃ (מא) קַח־ נָ֤א אֶת־ בִּרְכָתִי֙ אֲשֶׁ֣ר הֻבָ֣את לָ֔ךְ כִּֽי־ חַנַּ֥נִי אֱלֹהִ֖ים וְכִ֣י יֶשׁ־ לִי־ כֹ֑ל וַיִּפְצַר־ בּ֖וֹ וַיִּקָּֽח׃ (מב) וַיֹּ֖אמֶר נִסְעָ֣ה וְנֵלֵ֑כָה וְאֵלְכָ֖ה לְנֶגְדֶּֽךָ׃ (מג) וַיֹּ֣אמֶר אֵלָ֗יו אֲדֹנִ֤י יֹדֵ֙עַ֙ כִּֽי־ הַיְלָדִ֣ים רַכִּ֔ים וְהַצֹּ֥אן וְהַבָּקָ֖ר עָל֣וֹת עָלָ֑י וּדְפָקוּם֙ י֣וֹם אֶחָ֔ד וָמֵ֖תוּ כָּל־ הַצֹּֽאן׃ (מד) יַעֲבָר־ נָ֥א אֲדֹנִ֖י לִפְנֵ֣י עַבְדּ֑וֹ וַאֲנִ֞י בְּיֹשֵׁ֣ב הָאָ֔רֶץ בַּֽכְּנַעֲנִ֖י וּבַפְּרִזִּ֑י וַאֲנִי֙ מְתֵ֣י מִסְפָּ֔ר וְנֶאֶסְפ֤וּ עָלַי֙ וְהִכּ֔וּנִי וְנִשְׁמַדְתִּ֖י אֲנִ֥י וּבֵיתִֽי׃ (לד) וַיֹּאמְר֑וּ הַכְזוֹנָ֕ה יַעֲשֶׂ֖ה אֶת־ אֲחוֹתֵֽנוּ׃
(לה) וַיֹּ֤אמֶר אֱלֹהִים֙ אֶֽל־ יַעֲקֹ֔ב ק֛וּם עֲלֵ֥ה בֵֽית־ אֵ֖ל וְשֶׁב־ שָׁ֑ם וַעֲשֵׂה־ שָׁ֣ם מִזְבֵּ֔חַ לָאֵל֙ הַנִּרְאֶ֣ה אֵלֶ֔יךָ בְּבָרְחֲךָ֔ מִפְּנֵ֖י עֵשָׂ֥ו אָחִֽיךָ׃ (לו) וַיֹּ֤אמֶר יַעֲקֹב֙ אֶל־ בֵּית֔וֹ וְאֶ֖ל כָּל־ אֲשֶׁ֣ר עִמּ֑וֹ הָסִ֜רוּ אֶת־ אֱלֹהֵ֤י הַנֵּכָר֙ אֲשֶׁ֣ר בְּתֹכְכֶ֔ם וְהִֽטַּהֲר֔וּ וְהַחֲלִ֖יפוּ שִׂמְלֹתֵיכֶֽם׃ (לז) וְנָק֥וּמָה וְנַעֲלֶ֖ה בֵּֽית־ אֵ֑ל וְאֶֽעֱשֶׂה־ שָּׁ֣ם מִזְבֵּ֗חַ לָאֵ֞ל הָעֹנֶ֤ה אֹתִי֙ בְּי֣וֹם צָֽרָתִ֔י וַיְהִי֙ עִמָּדִ֔י בַּדֶּ֖רֶךְ אֲשֶׁ֥ר הָלָֽכְתִּי׃ (לח) וַיִּתְּנ֣וּ אֶֽל־ יַעֲקֹ֗ב אֵ֣ת כָּל־ אֱלֹהֵ֤י הַנֵּכָר֙ אֲשֶׁ֣ר בְּיָדָ֔ם וְאֶת־ הַנְּזָמִ֖ים אֲשֶׁ֣ר בְּאָזְנֵיהֶ֑ם וַיִּטְמֹ֤ן אֹתָם֙ יַעֲקֹ֔ב תַּ֥חַת הָאֵלָ֖ה אֲשֶׁ֥ר עִם־ שְׁכֶֽם׃ (לט) וַיִּסָּ֑עוּ וַיְהִ֣י ׀ חִתַּ֣ת אֱלֹהִ֗ים עַל־ הֶֽעָרִים֙ אֲשֶׁר֙ סְבִיבֹ֣תֵיהֶ֔ם וְלֹ֣א רָֽדְפ֔וּ אַחֲרֵ֖י בְּנֵ֥י יַעֲקֹֽב׃ (מ) וַיָּבֹ֨א יַעֲקֹ֜ב ל֗וּזָה אֲשֶׁר֙ בְּאֶ֣רֶץ כְּנַ֔עַן הִ֖וא בֵּֽית־ אֵ֑ל ה֖וּא וְכָל־ הָעָ֥ם אֲשֶׁר־ עִמּֽוֹ׃ (מא) וַיִּ֤בֶן שָׁם֙ מִזְבֵּ֔חַ וַיִּקְרָא֙ לַמָּק֔וֹם אֵ֖ל בֵּֽית־ אֵ֑ל כִּ֣י שָׁ֗ם נִגְל֤וּ אֵלָיו֙ הָֽאֱלֹהִ֔ים בְּבָרְח֖וֹ מִפְּנֵ֥י אָחִֽיו׃ (מב) וַתָּ֤מָת דְּבֹרָה֙ מֵינֶ֣קֶת רִבְקָ֔ה וַתִּקָּבֵ֛ר מִתַּ֥חַת לְבֵֽית־ אֵ֖ל תַּ֣חַת הָֽאַלּ֑וֹן וַיִּקְרָ֥א שְׁמ֖וֹ אַלּ֥וֹן בָּכֽוּת׃
