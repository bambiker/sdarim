בראשית פרק ח לפי החלוקה לסדרים

פרק ח בספר בראשית (22 פסוקים) נמצא בבראשית סדר ו (פסוקים א–יד) ובבראשית סדר ז (פסוקים א–ח).

בראשית סדר ו

תחילת פרק ח (א) וַיִּזְכֹּ֤ר אֱלֹהִים֙ אֶת־ נֹ֔חַ וְאֵ֤ת כָּל־ הַֽחַיָּה֙ וְאֶת־ כָּל־ הַבְּהֵמָ֔ה אֲשֶׁ֥ר אִתּ֖וֹ בַּתֵּבָ֑ה וַיַּעֲבֵ֨ר אֱלֹהִ֥ים ר֙וּחַ֙ עַל־ הָאָ֔רֶץ וַיָּשֹׁ֖כּוּ הַמָּֽיִם׃ (ב) וַיִּסָּֽכְרוּ֙ מַעְיְנֹ֣ת תְּה֔וֹם וַֽאֲרֻבֹּ֖ת הַשָּׁמָ֑יִם וַיִּכָּלֵ֥א הַגֶּ֖שֶׁם מִן־ הַשָּׁמָֽיִם׃ (ג) וַיָּשֻׁ֧בוּ הַמַּ֛יִם מֵעַ֥ל הָאָ֖רֶץ הָל֣וֹךְ וָשׁ֑וֹב וַיַּחְסְר֣וּ הַמַּ֔יִם מִקְצֵ֕ה חֲמִשִּׁ֥ים וּמְאַ֖ת יֽוֹם׃ (ד) וַתָּ֤נַח הַתֵּבָה֙ בַּחֹ֣דֶשׁ הַשְּׁבִיעִ֔י בְּשִׁבְעָה־ עָשָׂ֥ר י֖וֹם לַחֹ֑דֶשׁ עַ֖ל הָרֵ֥י אֲרָרָֽט׃ (ה) וְהַמַּ֗יִם הָיוּ֙ הָל֣וֹךְ וְחָס֔וֹר עַ֖ד הַחֹ֣דֶשׁ הָֽעֲשִׂירִ֑י בָּֽעֲשִׂירִי֙ בְּאֶחָ֣ד לַחֹ֔דֶשׁ נִרְא֖וּ רָאשֵׁ֥י הֶֽהָרִֽים׃ (ו) וַֽיְהִ֕י מִקֵּ֖ץ אַרְבָּעִ֣ים י֑וֹם וַיִּפְתַּ֣ח נֹ֔חַ אֶת־ חַלּ֥וֹן הַתֵּבָ֖ה אֲשֶׁ֥ר עָשָֽׂה׃ (ז) וַיְשַׁלַּ֖ח אֶת־ הָֽעֹרֵ֑ב וַיֵּצֵ֤א יָצוֹא֙ וָשׁ֔וֹב עַד־ יְבֹ֥שֶׁת הַמַּ֖יִם מֵעַ֥ל הָאָֽרֶץ׃ (ח) וַיְשַׁלַּ֥ח אֶת־ הַיּוֹנָ֖ה מֵאִתּ֑וֹ לִרְאוֹת֙ הֲקַ֣לּוּ הַמַּ֔יִם מֵעַ֖ל פְּנֵ֥י הָֽאֲדָמָֽה׃ (ט) וְלֹֽא־ מָצְאָה֩ הַיּוֹנָ֨ה מָנ֜וֹחַ לְכַף־ רַגְלָ֗הּ וַתָּ֤שָׁב אֵלָיו֙ אֶל־ הַתֵּבָ֔ה כִּי־ מַ֖יִם עַל־ פְּנֵ֣י כָל־ הָאָ֑רֶץ וַיִּשְׁלַ֤ח יָדוֹ֙ וַיִּקָּחֶ֔הָ וַיָּבֵ֥א אֹתָ֛הּ אֵלָ֖יו אֶל־ הַתֵּבָֽה׃ (י) וַיָּ֣חֶל ע֔וֹד שִׁבְעַ֥ת יָמִ֖ים אֲחֵרִ֑ים וַיֹּ֛סֶף שַׁלַּ֥ח אֶת־ הַיּוֹנָ֖ה מִן־ הַתֵּבָֽה׃ (יא) וַתָּבֹ֨א אֵלָ֤יו הַיּוֹנָה֙ לְעֵ֣ת עֶ֔רֶב וְהִנֵּ֥ה עֲלֵה־ זַ֖יִת טָרָ֣ף בְּפִ֑יהָ וַיֵּ֣דַע נֹ֔חַ כִּי־ קַ֥לּוּ הַמַּ֖יִם מֵעַ֥ל הָאָֽרֶץ׃ (יב) וַיִּיָּ֣חֶל ע֔וֹד שִׁבְעַ֥ת יָמִ֖ים אֲחֵרִ֑ים וַיְשַׁלַּח֙ אֶת־ הַיּוֹנָ֔ה וְלֹֽא־ יָסְפָ֥ה שׁוּב־ אֵלָ֖יו עֽוֹד׃ (יג) וַֽ֠יְהִי בְּאַחַ֨ת וְשֵׁשׁ־ מֵא֜וֹת שָׁנָ֗ה בָּֽרִאשׁוֹן֙ בְּאֶחָ֣ד לַחֹ֔דֶשׁ חָֽרְב֥וּ הַמַּ֖יִם מֵעַ֣ל הָאָ֑רֶץ וַיָּ֤סַר נֹ֙חַ֙ אֶת־ מִכְסֵ֣ה הַתֵּבָ֔ה וַיַּ֕רְא וְהִנֵּ֥ה חָֽרְב֖וּ פְּנֵ֥י הָֽאֲדָמָֽה׃ (יד) וּבַחֹ֙דֶשׁ֙ הַשֵּׁנִ֔י בְּשִׁבְעָ֧ה וְעֶשְׂרִ֛ים י֖וֹם לַחֹ֑דֶשׁ יָבְשָׁ֖ה הָאָֽרֶץ׃     

בראשית סדר ז

(א) וַיְדַבֵּ֥ר אֱלֹהִ֖ים אֶל־ נֹ֥חַ לֵאמֹֽר׃ (ב) צֵ֖א מִן־ הַתֵּבָ֑ה אַתָּ֕ה וְאִשְׁתְּךָ֛ וּבָנֶ֥יךָ וּנְשֵֽׁי־ בָנֶ֖יךָ אִתָּֽךְ׃ (ג) כָּל־ הַחַיָּ֨ה אֲשֶֽׁר־ אִתְּךָ֜ מִכָּל־ בָּשָׂ֗ר בָּע֧וֹף וּבַבְּהֵמָ֛ה וּבְכָל־ הָרֶ֛מֶשׂ הָרֹמֵ֥שׂ עַל־ הָאָ֖רֶץ (הוצא) הַיְצֵ֣א אִתָּ֑ךְ וְשָֽׁרְצ֣וּ בָאָ֔רֶץ וּפָר֥וּ וְרָב֖וּ עַל־ הָאָֽרֶץ׃ (ד) וַיֵּ֖צֵא־ נֹ֑חַ וּבָנָ֛יו וְאִשְׁתּ֥וֹ וּנְשֵֽׁי־ בָנָ֖יו אִתּֽוֹ׃ (ה) כָּל־ הַֽחַיָּ֗ה כָּל־ הָרֶ֙מֶשׂ֙ וְכָל־ הָע֔וֹף כֹּ֖ל רוֹמֵ֣שׂ עַל־ הָאָ֑רֶץ לְמִשְׁפְּחֹ֣תֵיהֶ֔ם יָצְא֖וּ מִן־ הַתֵּבָֽה׃ (ו) וַיִּ֥בֶן נֹ֛חַ מִזְבֵּ֖חַ לַֽיהוָ֑ה וַיִּקַּ֞ח מִכֹּ֣ל ׀ הַבְּהֵמָ֣ה הַטְּהוֹרָ֗ה וּמִכֹּל֙ הָע֣וֹף הַטָּהֹ֔ר וַיַּ֥עַל עֹלֹ֖ת בַּמִּזְבֵּֽחַ׃ (ז) וַיָּ֣רַח יְהוָה֮ אֶת־ רֵ֣יחַ הַנִּיחֹחַ֒ וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ לִבּ֗וֹ לֹֽא־ אֹ֠סִף לְקַלֵּ֨ל ע֤וֹד אֶת־ הָֽאֲדָמָה֙ בַּעֲב֣וּר הָֽאָדָ֔ם כִּ֠י יֵ֣צֶר לֵ֧ב הָאָדָ֛ם רַ֖ע מִנְּעֻרָ֑יו וְלֹֽא־ אֹסִ֥ף ע֛וֹד לְהַכּ֥וֹת אֶת־ כָּל־ חַ֖י כַּֽאֲשֶׁ֥ר עָשִֽׂיתִי׃ (ח) עֹ֖ד כָּל־ יְמֵ֣י הָאָ֑רֶץ זֶ֡רַע וְ֠קָצִיר וְקֹ֨ר וָחֹ֜ם וְקַ֧יִץ וָחֹ֛רֶף וְי֥וֹם וָלַ֖יְלָה לֹ֥א יִשְׁבֹּֽתוּ׃ סוף פרק ח (ט) וַיְבָ֣רֶךְ אֱלֹהִ֔ים אֶת־ נֹ֖חַ וְאֶת־ בָּנָ֑יו וַיֹּ֧אמֶר לָהֶ֛ם פְּר֥וּ וּרְב֖וּ וּמִלְא֥וּ אֶת־ הָאָֽרֶץ׃ (י) וּמוֹרַאֲכֶ֤ם וְחִתְּכֶם֙ יִֽהְיֶ֔ה עַ֚ל כָּל־ חַיַּ֣ת הָאָ֔רֶץ וְעַ֖ל כָּל־ ע֣וֹף הַשָּׁמָ֑יִם בְּכֹל֩ אֲשֶׁ֨ר תִּרְמֹ֧שׂ הָֽאֲדָמָ֛ה וּֽבְכָל־ דְּגֵ֥י הַיָּ֖ם בְּיֶדְכֶ֥ם נִתָּֽנוּ׃ (יא) כָּל־ רֶ֙מֶשׂ֙ אֲשֶׁ֣ר הוּא־ חַ֔י לָכֶ֥ם יִהְיֶ֖ה לְאָכְלָ֑ה כְּיֶ֣רֶק עֵ֔שֶׂב נָתַ֥תִּי לָכֶ֖ם אֶת־ כֹּֽל׃ (יב) אַךְ־ בָּשָׂ֕ר בְּנַפְשׁ֥וֹ דָמ֖וֹ לֹ֥א תֹאכֵֽלוּ׃ (יג) וְאַ֨ךְ אֶת־ דִּמְכֶ֤ם לְנַפְשֹֽׁתֵיכֶם֙ אֶדְרֹ֔שׁ מִיַּ֥ד כָּל־ חַיָּ֖ה אֶדְרְשֶׁ֑נּוּ וּמִיַּ֣ד הָֽאָדָ֗ם מִיַּד֙ אִ֣ישׁ אָחִ֔יו אֶדְרֹ֖שׁ אֶת־ נֶ֥פֶשׁ הָֽאָדָֽם׃ (יד) שֹׁפֵךְ֙ דַּ֣ם הָֽאָדָ֔ם בָּֽאָדָ֖ם דָּמ֣וֹ יִשָּׁפֵ֑ךְ כִּ֚י בְּצֶ֣לֶם אֱלֹהִ֔ים עָשָׂ֖ה אֶת־ הָאָדָֽם׃ (טו) וְאַתֶּ֖ם פְּר֣וּ וּרְב֑וּ שִׁרְצ֥וּ בָאָ֖רֶץ וּרְבוּ־ בָֽהּ׃      (טז) וַיֹּ֤אמֶר אֱלֹהִים֙ אֶל־ נֹ֔חַ וְאֶל־ בָּנָ֥יו אִתּ֖וֹ לֵאמֹֽר׃ (יז) וַאֲנִ֕י הִנְנִ֥י מֵקִ֛ים אֶת־ בְּרִיתִ֖י אִתְּכֶ֑ם וְאֶֽת־ זַרְעֲכֶ֖ם אַֽחֲרֵיכֶֽם׃ (יח) וְאֵ֨ת כָּל־ נֶ֤פֶשׁ הַֽחַיָּה֙ אֲשֶׁ֣ר אִתְּכֶ֔ם בָּע֧וֹף בַּבְּהֵמָ֛ה וּֽבְכָל־ חַיַּ֥ת הָאָ֖רֶץ אִתְּכֶ֑ם מִכֹּל֙ יֹצְאֵ֣י הַתֵּבָ֔ה לְכֹ֖ל חַיַּ֥ת הָאָֽרֶץ׃ (יט) וַהֲקִמֹתִ֤י אֶת־ בְּרִיתִי֙ אִתְּכֶ֔ם וְלֹֽא־ יִכָּרֵ֧ת כָּל־ בָּשָׂ֛ר ע֖וֹד מִמֵּ֣י הַמַּבּ֑וּל וְלֹֽא־ יִהְיֶ֥ה ע֛וֹד מַבּ֖וּל לְשַׁחֵ֥ת הָאָֽרֶץ׃ (כ) וַיֹּ֣אמֶר אֱלֹהִ֗ים זֹ֤את אֽוֹת־ הַבְּרִית֙ אֲשֶׁר־ אֲנִ֣י נֹתֵ֗ן בֵּינִי֙ וּבֵ֣ינֵיכֶ֔ם וּבֵ֛ין כָּל־ נֶ֥פֶשׁ חַיָּ֖ה אֲשֶׁ֣ר אִתְּכֶ֑ם לְדֹרֹ֖ת עוֹלָֽם׃ (כא) אֶת־ קַשְׁתִּ֕י נָתַ֖תִּי בֶּֽעָנָ֑ן וְהָֽיְתָה֙ לְא֣וֹת בְּרִ֔ית בֵּינִ֖י וּבֵ֥ין הָאָֽרֶץ׃ (כב) וְהָיָ֕ה בְּעַֽנְנִ֥י עָנָ֖ן עַל־ הָאָ֑רֶץ וְנִרְאֲתָ֥ה הַקֶּ֖שֶׁת בֶּעָנָֽן׃ (כג) וְזָכַרְתִּ֣י אֶת־ בְּרִיתִ֗י אֲשֶׁ֤ר בֵּינִי֙ וּבֵ֣ינֵיכֶ֔ם וּבֵ֛ין כָּל־ נֶ֥פֶשׁ חַיָּ֖ה בְּכָל־ בָּשָׂ֑ר וְלֹֽא־ יִֽהְיֶ֨ה ע֤וֹד הַמַּ֙יִם֙ לְמַבּ֔וּל לְשַׁחֵ֖ת כָּל־ בָּשָֽׂר׃ (כד) וְהָיְתָ֥ה הַקֶּ֖שֶׁת בֶּֽעָנָ֑ן וּרְאִיתִ֗יהָ לִזְכֹּר֙ בְּרִ֣ית עוֹלָ֔ם בֵּ֣ין אֱלֹהִ֔ים וּבֵין֙ כָּל־ נֶ֣פֶשׁ חַיָּ֔ה בְּכָל־ בָּשָׂ֖ר אֲשֶׁ֥ר עַל־ הָאָֽרֶץ׃ (כה) וַיֹּ֥אמֶר אֱלֹהִ֖ים אֶל־ נֹ֑חַ זֹ֤את אֽוֹת־ הַבְּרִית֙ אֲשֶׁ֣ר הֲקִמֹ֔תִי בֵּינִ֕י וּבֵ֥ין כָּל־ בָּשָׂ֖ר אֲשֶׁ֥ר עַל־ הָאָֽרֶץ׃
