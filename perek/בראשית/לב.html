בראשית פרק לב לפי החלוקה לסדרים

פרק לב בספר בראשית (33 פסוקים) נמצא בבראשית סדר כט (פסוקים נג–נה) ובבראשית סדר ל (פסוקים א–ל).

בראשית סדר כט

(א) וַיֹּ֤אמֶר יְהוָה֙ אֶֽל־ יַעֲקֹ֔ב שׁ֛וּב אֶל־ אֶ֥רֶץ אֲבוֹתֶ֖יךָ וּלְמוֹלַדְתֶּ֑ךָ וְאֶֽהְיֶ֖ה עִמָּֽךְ׃ (ב) וַיִּשְׁלַ֣ח יַעֲקֹ֔ב וַיִּקְרָ֖א לְרָחֵ֣ל וּלְלֵאָ֑ה הַשָּׂדֶ֖ה אֶל־ צֹאנֽוֹ׃ (ג) וַיֹּ֣אמֶר לָהֶ֗ן רֹאֶ֤ה אָנֹכִי֙ אֶת־ פְּנֵ֣י אֲבִיכֶ֔ן כִּֽי־ אֵינֶ֥נּוּ אֵלַ֖י כִּתְמֹ֣ל שִׁלְשֹׁ֑ם וֵֽאלֹהֵ֣י אָבִ֔י הָיָ֖ה עִמָּדִֽי׃ (ד) וְאַתֵּ֖נָה יְדַעְתֶּ֑ן כִּ֚י בְּכָל־ כֹּחִ֔י עָבַ֖דְתִּי אֶת־ אֲבִיכֶֽן׃ (ה) וַאֲבִיכֶן֙ הֵ֣תֶל בִּ֔י וְהֶחֱלִ֥ף אֶת־ מַשְׂכֻּרְתִּ֖י עֲשֶׂ֣רֶת מֹנִ֑ים וְלֹֽא־ נְתָנ֣וֹ אֱלֹהִ֔ים לְהָרַ֖ע עִמָּדִֽי׃ (ו) אִם־ כֹּ֣ה יֹאמַ֗ר נְקֻדִּים֙ יִהְיֶ֣ה שְׂכָרֶ֔ךָ וְיָלְד֥וּ כָל־ הַצֹּ֖אן נְקֻדִּ֑ים וְאִם־ כֹּ֣ה יֹאמַ֗ר עֲקֻדִּים֙ יִהְיֶ֣ה שְׂכָרֶ֔ךָ וְיָלְד֥וּ כָל־ הַצֹּ֖אן עֲקֻדִּֽים׃ (ז) וַיַּצֵּ֧ל אֱלֹהִ֛ים אֶת־ מִקְנֵ֥ה אֲבִיכֶ֖ם וַיִּתֶּן־ לִֽי׃ (ח) וַיְהִ֗י בְּעֵת֙ יַחֵ֣ם הַצֹּ֔אן וָאֶשָּׂ֥א עֵינַ֛י וָאֵ֖רֶא בַּחֲל֑וֹם וְהִנֵּ֤ה הָֽעַתֻּדִים֙ הָעֹלִ֣ים עַל־ הַצֹּ֔אן עֲקֻדִּ֥ים נְקֻדִּ֖ים וּבְרֻדִּֽים׃ (ט) וַיֹּ֨אמֶר אֵלַ֜י מַלְאַ֧ךְ הָאֱלֹהִ֛ים בַּחֲל֖וֹם יַֽעֲקֹ֑ב וָאֹמַ֖ר הִנֵּֽנִי׃ (י) וַיֹּ֗אמֶר שָׂא־ נָ֨א עֵינֶ֤יךָ וּרְאֵה֙ כָּל־ הָֽעַתֻּדִים֙ הָעֹלִ֣ים עַל־ הַצֹּ֔אן עֲקֻדִּ֥ים נְקֻדִּ֖ים וּבְרֻדִּ֑ים כִּ֣י רָאִ֔יתִי אֵ֛ת כָּל־ אֲשֶׁ֥ר לָבָ֖ן עֹ֥שֶׂה לָּֽךְ׃ (יא) אָנֹכִ֤י הָאֵל֙ בֵּֽית־ אֵ֔ל אֲשֶׁ֨ר מָשַׁ֤חְתָּ שָּׁם֙ מַצֵּבָ֔ה אֲשֶׁ֨ר נָדַ֥רְתָּ לִּ֛י שָׁ֖ם נֶ֑דֶר עַתָּ֗ה ק֥וּם צֵא֙ מִן־ הָאָ֣רֶץ הַזֹּ֔את וְשׁ֖וּב אֶל־ אֶ֥רֶץ מוֹלַדְתֶּֽךָ׃ (יב) וַתַּ֤עַן רָחֵל֙ וְלֵאָ֔ה וַתֹּאמַ֖רְנָה ל֑וֹ הַע֥וֹד לָ֛נוּ חֵ֥לֶק וְנַחֲלָ֖ה בְּבֵ֥ית אָבִֽינוּ׃ (יג) הֲל֧וֹא נָכְרִיּ֛וֹת נֶחְשַׁ֥בְנוּ ל֖וֹ כִּ֣י מְכָרָ֑נוּ וַיֹּ֥אכַל גַּם־ אָכ֖וֹל אֶת־ כַּסְפֵּֽנוּ׃ (יד) כִּ֣י כָל־ הָעֹ֗שֶׁר אֲשֶׁ֨ר הִצִּ֤יל אֱלֹהִים֙ מֵֽאָבִ֔ינוּ לָ֥נוּ ה֖וּא וּלְבָנֵ֑ינוּ וְעַתָּ֗ה כֹּל֩ אֲשֶׁ֨ר אָמַ֧ר אֱלֹהִ֛ים אֵלֶ֖יךָ עֲשֵֽׂה׃ (טו) וַיָּ֖קָם יַעֲקֹ֑ב וַיִּשָּׂ֛א אֶת־ בָּנָ֥יו וְאֶת־ נָשָׁ֖יו עַל־ הַגְּמַלִּֽים׃ (טז) וַיִּנְהַ֣ג אֶת־ כָּל־ מִקְנֵ֗הוּ וְאֶת־ כָּל־ רְכֻשׁוֹ֙ אֲשֶׁ֣ר רָכָ֔שׁ מִקְנֵה֙ קִנְיָנ֔וֹ אֲשֶׁ֥ר רָכַ֖שׁ בְּפַדַּ֣ן אֲרָ֑ם לָב֛וֹא אֶל־ יִצְחָ֥ק אָבִ֖יו אַ֥רְצָה כְּנָֽעַן׃ (יז) וְלָבָ֣ן הָלַ֔ךְ לִגְזֹ֖ז אֶת־ צֹאנ֑וֹ וַתִּגְנֹ֣ב רָחֵ֔ל אֶת־ הַתְּרָפִ֖ים אֲשֶׁ֥ר לְאָבִֽיהָ׃ (יח) וַיִּגְנֹ֣ב יַעֲקֹ֔ב אֶת־ לֵ֥ב לָבָ֖ן הָאֲרַמִּ֑י עַל־ בְּלִי֙ הִגִּ֣יד ל֔וֹ כִּ֥י בֹרֵ֖חַ הֽוּא׃ (יט) וַיִּבְרַ֥ח הוּא֙ וְכָל־ אֲשֶׁר־ ל֔וֹ וַיָּ֖קָם וַיַּעֲבֹ֣ר אֶת־ הַנָּהָ֑ר וַיָּ֥שֶׂם אֶת־ פָּנָ֖יו הַ֥ר הַגִּלְעָֽד׃ (כ) וַיֻּגַּ֥ד לְלָבָ֖ן בַּיּ֣וֹם הַשְּׁלִישִׁ֑י כִּ֥י בָרַ֖ח יַעֲקֹֽב׃ (כא) וַיִּקַּ֤ח אֶת־ אֶחָיו֙ עִמּ֔וֹ וַיִּרְדֹּ֣ף אַחֲרָ֔יו דֶּ֖רֶךְ שִׁבְעַ֣ת יָמִ֑ים וַיַּדְבֵּ֥ק אֹת֖וֹ בְּהַ֥ר הַגִּלְעָֽד׃ (כב) וַיָּבֹ֧א אֱלֹהִ֛ים אֶל־ לָבָ֥ן הָאֲרַמִּ֖י בַּחֲלֹ֣ם הַלָּ֑יְלָה וַיֹּ֣אמֶר ל֗וֹ הִשָּׁ֧מֶר לְךָ֛ פֶּן־ תְּדַבֵּ֥ר עִֽם־ יַעֲקֹ֖ב מִטּ֥וֹב עַד־ רָֽע׃ (כג) וַיַּשֵּׂ֥ג לָבָ֖ן אֶֽת־ יַעֲקֹ֑ב וְיַעֲקֹ֗ב תָּקַ֤ע אֶֽת־ אָהֳלוֹ֙ בָּהָ֔ר וְלָבָ֛ן תָּקַ֥ע אֶת־ אֶחָ֖יו בְּהַ֥ר הַגִּלְעָֽד׃ (כד) וַיֹּ֤אמֶר לָבָן֙ לְיַעֲקֹ֔ב מֶ֣ה עָשִׂ֔יתָ וַתִּגְנֹ֖ב אֶת־ לְבָבִ֑י וַתְּנַהֵג֙ אֶת־ בְּנֹתַ֔י כִּשְׁבֻי֖וֹת חָֽרֶב׃ (כה) לָ֤מָּה נַחְבֵּ֙אתָ֙ לִבְרֹ֔חַ וַתִּגְנֹ֖ב אֹתִ֑י וְלֹא־ הִגַּ֣דְתָּ לִּ֔י וָֽאֲשַׁלֵּחֲךָ֛ בְּשִׂמְחָ֥ה וּבְשִׁרִ֖ים בְּתֹ֥ף וּבְכִנּֽוֹר׃ (כו) וְלֹ֣א נְטַשְׁתַּ֔נִי לְנַשֵּׁ֥ק לְבָנַ֖י וְלִבְנֹתָ֑י עַתָּ֖ה הִסְכַּ֥לְתָּֽ עֲשֽׂוֹ׃ (כז) יֶשׁ־ לְאֵ֣ל יָדִ֔י לַעֲשׂ֥וֹת עִמָּכֶ֖ם רָ֑ע וֵֽאלֹהֵ֨י אֲבִיכֶ֜ם אֶ֣מֶשׁ ׀ אָמַ֧ר אֵלַ֣י לֵאמֹ֗ר הִשָּׁ֧מֶר לְךָ֛ מִדַּבֵּ֥ר עִֽם־ יַעֲקֹ֖ב מִטּ֥וֹב עַד־ רָֽע׃ (כח) וְעַתָּה֙ הָלֹ֣ךְ הָלַ֔כְתָּ כִּֽי־ נִכְסֹ֥ף נִכְסַ֖פְתָּה לְבֵ֣ית אָבִ֑יךָ לָ֥מָּה גָנַ֖בְתָּ אֶת־ אֱלֹהָֽי׃ (כט) וַיַּ֥עַן יַעֲקֹ֖ב וַיֹּ֣אמֶר לְלָבָ֑ן כִּ֣י יָרֵ֔אתִי כִּ֣י אָמַ֔רְתִּי פֶּן־ תִּגְזֹ֥ל אֶת־ בְּנוֹתֶ֖יךָ מֵעִמִּֽי׃ (ל) עִ֠ם אֲשֶׁ֨ר תִּמְצָ֣א אֶת־ אֱלֹהֶיךָ֮ לֹ֣א יִֽחְיֶה֒ נֶ֣גֶד אַחֵ֧ינוּ הַֽכֶּר־ לְךָ֛ מָ֥ה עִמָּדִ֖י וְקַֽח־ לָ֑ךְ וְלֹֽא־ יָדַ֣ע יַעֲקֹ֔ב כִּ֥י רָחֵ֖ל גְּנָבָֽתַם׃ (לא) וַיָּבֹ֨א לָבָ֜ן בְּאֹ֥הֶל יַעֲקֹ֣ב ׀ וּבְאֹ֣הֶל לֵאָ֗ה וּבְאֹ֛הֶל שְׁתֵּ֥י הָאֲמָהֹ֖ת וְלֹ֣א מָצָ֑א וַיֵּצֵא֙ מֵאֹ֣הֶל לֵאָ֔ה וַיָּבֹ֖א בְּאֹ֥הֶל רָחֵֽל׃ (לב) וְרָחֵ֞ל לָקְחָ֣ה אֶת־ הַתְּרָפִ֗ים וַתְּשִׂמֵ֛ם בְּכַ֥ר הַגָּמָ֖ל וַתֵּ֣שֶׁב עֲלֵיהֶ֑ם וַיְמַשֵּׁ֥שׁ לָבָ֛ן אֶת־ כָּל־ הָאֹ֖הֶל וְלֹ֥א מָצָֽא׃ (לג) וַתֹּ֣אמֶר אֶל־ אָבִ֗יהָ אַל־ יִ֙חַר֙ בְּעֵינֵ֣י אֲדֹנִ֔י כִּ֣י ל֤וֹא אוּכַל֙ לָק֣וּם מִפָּנֶ֔יךָ כִּי־ דֶ֥רֶךְ נָשִׁ֖ים לִ֑י וַיְחַפֵּ֕שׂ וְלֹ֥א מָצָ֖א אֶת־ הַתְּרָפִֽים׃ (לד) וַיִּ֥חַר לְיַעֲקֹ֖ב וַיָּ֣רֶב בְּלָבָ֑ן וַיַּ֤עַן יַעֲקֹב֙ וַיֹּ֣אמֶר לְלָבָ֔ן מַה־ פִּשְׁעִי֙ מַ֣ה חַטָּאתִ֔י כִּ֥י דָלַ֖קְתָּ אַחֲרָֽי׃ (לה) כִּֽי־ מִשַּׁ֣שְׁתָּ אֶת־ כָּל־ כֵּלַ֗י מַה־ מָּצָ֙אתָ֙ מִכֹּ֣ל כְּלֵי־ בֵיתֶ֔ךָ שִׂ֣ים כֹּ֔ה נֶ֥גֶד אַחַ֖י וְאַחֶ֑יךָ וְיוֹכִ֖יחוּ בֵּ֥ין שְׁנֵֽינוּ׃ (לו) זֶה֩ עֶשְׂרִ֨ים שָׁנָ֤ה אָנֹכִי֙ עִמָּ֔ךְ רְחֵלֶ֥יךָ וְעִזֶּ֖יךָ לֹ֣א שִׁכֵּ֑לוּ וְאֵילֵ֥י צֹאנְךָ֖ לֹ֥א אָכָֽלְתִּי׃ (לז) טְרֵפָה֙ לֹא־ הֵבֵ֣אתִי אֵלֶ֔יךָ אָנֹכִ֣י אֲחַטֶּ֔נָּה מִיָּדִ֖י תְּבַקְשֶׁ֑נָּה גְּנֻֽבְתִ֣י י֔וֹם וּגְנֻֽבְתִ֖י לָֽיְלָה׃ (לח) הָיִ֧יתִי בַיּ֛וֹם אֲכָלַ֥נִי חֹ֖רֶב וְקֶ֣רַח בַּלָּ֑יְלָה וַתִּדַּ֥ד שְׁנָתִ֖י מֵֽעֵינָֽי׃ (לט) זֶה־ לִּ֞י עֶשְׂרִ֣ים שָׁנָה֮ בְּבֵיתֶךָ֒ עֲבַדְתִּ֜יךָ אַרְבַּֽע־ עֶשְׂרֵ֤ה שָׁנָה֙ בִּשְׁתֵּ֣י בְנֹתֶ֔יךָ וְשֵׁ֥שׁ שָׁנִ֖ים בְּצֹאנֶ֑ךָ וַתַּחֲלֵ֥ף אֶת־ מַשְׂכֻּרְתִּ֖י עֲשֶׂ֥רֶת מֹנִֽים׃ (מ) לוּלֵ֡י אֱלֹהֵ֣י אָבִי֩ אֱלֹהֵ֨י אַבְרָהָ֜ם וּפַ֤חַד יִצְחָק֙ הָ֣יָה לִ֔י כִּ֥י עַתָּ֖ה רֵיקָ֣ם שִׁלַּחְתָּ֑נִי אֶת־ עָנְיִ֞י וְאֶת־ יְגִ֧יעַ כַּפַּ֛י רָאָ֥ה אֱלֹהִ֖ים וַיּ֥וֹכַח אָֽמֶשׁ׃ (מא) וַיַּ֨עַן לָבָ֜ן וַיֹּ֣אמֶר אֶֽל־ יַעֲקֹ֗ב הַבָּנ֨וֹת בְּנֹתַ֜י וְהַבָּנִ֤ים בָּנַי֙ וְהַצֹּ֣אן צֹאנִ֔י וְכֹ֛ל אֲשֶׁר־ אַתָּ֥ה רֹאֶ֖ה לִי־ ה֑וּא וְלִבְנֹתַ֞י מָֽה־ אֶֽעֱשֶׂ֤ה לָאֵ֙לֶּה֙ הַיּ֔וֹם א֥וֹ לִבְנֵיהֶ֖ן אֲשֶׁ֥ר יָלָֽדוּ׃ (מב) וְעַתָּ֗ה לְכָ֛ה נִכְרְתָ֥ה בְרִ֖ית אֲנִ֣י וָאָ֑תָּה וְהָיָ֥ה לְעֵ֖ד בֵּינִ֥י וּבֵינֶֽךָ׃ (מג) וַיִּקַּ֥ח יַעֲקֹ֖ב אָ֑בֶן וַיְרִימֶ֖הָ מַצֵּבָֽה׃ (מד) וַיֹּ֨אמֶר יַעֲקֹ֤ב אֹת֖וֹ שָֽׁם׃ (כח) וַיִּקְרָ֧א יַעֲקֹ֛ב שֵׁ֥ם הַמָּק֖וֹם פְּנִיאֵ֑ל כִּֽי־ רָאִ֤יתִי אֱלֹהִים֙ פָּנִ֣ים אֶל־ פָּנִ֔ים וַתִּנָּצֵ֖ל נַפְשִֽׁי׃ (כט) וַיִּֽזְרַֽח־ ל֣וֹ הַשֶּׁ֔מֶשׁ כַּאֲשֶׁ֥ר עָבַ֖ר אֶת־ פְּנוּאֵ֑ל וְה֥וּא צֹלֵ֖עַ עַל־ יְרֵכֽוֹ׃ (ל) עַל־ כֵּ֡ן לֹֽא־ יֹאכְל֨וּ בְנֵֽי־ יִשְׂרָאֵ֜ל אֶת־ גִּ֣יד הַנָּשֶׁ֗ה אֲשֶׁר֙ עַל־ כַּ֣ף הַיָּרֵ֔ךְ עַ֖ד הַיּ֣וֹם הַזֶּ֑ה כִּ֤י נָגַע֙ בְּכַף־ יֶ֣רֶךְ יַעֲקֹ֔ב בְּגִ֖יד הַנָּשֶֽׁה׃ סוף פרק לב (לא) וַיִּשָּׂ֨א יַעֲקֹ֜ב עֵינָ֗יו וַיַּרְא֙ וְהִנֵּ֣ה עֵשָׂ֣ו בָּ֔א וְעִמּ֕וֹ אַרְבַּ֥ע מֵא֖וֹת אִ֑ישׁ וַיַּ֣חַץ אֶת־ הַיְלָדִ֗ים עַל־ לֵאָה֙ וְעַל־ רָחֵ֔ל וְעַ֖ל שְׁתֵּ֥י הַשְּׁפָחֽוֹת׃ (לב) וַיָּ֧שֶׂם אֶת־ הַשְּׁפָח֛וֹת וְאֶת־ יַלְדֵיהֶ֖ן רִֽאשֹׁנָ֑ה וְאֶת־ לֵאָ֤ה וִֽילָדֶ֙יהָ֙ אַחֲרֹנִ֔ים וְאֶת־ רָחֵ֥ל וְאֶת־ יוֹסֵ֖ף אַחֲרֹנִֽים׃ (לג) וְה֖וּא עָבַ֣ר לִפְנֵיהֶ֑ם וַיִּשְׁתַּ֤חוּ אַ֙רְצָה֙ שֶׁ֣בַע פְּעָמִ֔ים עַד־ גִּשְׁתּ֖וֹ עַד־ אָחִֽיו׃ (לד) וַיָּ֨רָץ עֵשָׂ֤ו לִקְרָאתוֹ֙ וַֽיְחַבְּקֵ֔הוּ וַיִּפֹּ֥ל עַל־ צַוָּארָ֖ו וַׄיִּׄשָּׁׄקֵ֑ׄהׄוּׄ וַיִּבְכּֽוּ׃ (לה) וַיִּשָּׂ֣א אֶת־ עֵינָ֗יו וַיַּ֤רְא אֶת־ הַנָּשִׁים֙ וְאֶת־ הַיְלָדִ֔ים וַיֹּ֖אמֶר מִי־ אֵ֣לֶּה לָּ֑ךְ וַיֹּאמַ֕ר הַיְלָדִ֕ים אֲשֶׁר־ חָנַ֥ן אֱלֹהִ֖ים אֶת־ עַבְדֶּֽךָ׃ (לו) וַתִּגַּ֧שְׁןָ הַשְּׁפָח֛וֹת הֵ֥נָּה וְיַלְדֵיהֶ֖ן וַתִּֽשְׁתַּחֲוֶֽיןָ׃ (לז) וַתִּגַּ֧שׁ גַּם־ לֵאָ֛ה וִילָדֶ֖יהָ וַיִּֽשְׁתַּחֲו֑וּ וְאַחַ֗ר נִגַּ֥שׁ יוֹסֵ֛ף וְרָחֵ֖ל וַיִּֽשְׁתַּחֲוֽוּ׃ (לח) וַיֹּ֕אמֶר מִ֥י לְךָ֛ כָּל־ הַמַּחֲנֶ֥ה הַזֶּ֖ה אֲשֶׁ֣ר פָּגָ֑שְׁתִּי וַיֹּ֕אמֶר לִמְצֹא־ חֵ֖ן בְּעֵינֵ֥י אֲדֹנִֽי׃ (לט) וַיֹּ֥אמֶר עֵשָׂ֖ו יֶשׁ־ לִ֣י רָ֑ב אָחִ֕י יְהִ֥י לְךָ֖ אֲשֶׁר־ לָֽךְ׃ (מ) וַיֹּ֣אמֶר יַעֲקֹ֗ב אַל־ נָא֙ אִם־ נָ֨א מָצָ֤אתִי חֵן֙ בְּעֵינֶ֔יךָ וְלָקַחְתָּ֥ מִנְחָתִ֖י מִיָּדִ֑י כִּ֣י עַל־ כֵּ֞ן רָאִ֣יתִי פָנֶ֗יךָ כִּרְאֹ֛ת פְּנֵ֥י אֱלֹהִ֖ים וַתִּרְצֵֽנִי׃ (מא) קַח־ נָ֤א אֶת־ בִּרְכָתִי֙ אֲשֶׁ֣ר הֻבָ֣את לָ֔ךְ כִּֽי־ חַנַּ֥נִי אֱלֹהִ֖ים וְכִ֣י יֶשׁ־ לִי־ כֹ֑ל וַיִּפְצַר־ בּ֖וֹ וַיִּקָּֽח׃ (מב) וַיֹּ֖אמֶר נִסְעָ֣ה וְנֵלֵ֑כָה וְאֵלְכָ֖ה לְנֶגְדֶּֽךָ׃ (מג) וַיֹּ֣אמֶר אֵלָ֗יו אֲדֹנִ֤י יֹדֵ֙עַ֙ כִּֽי־ הַיְלָדִ֣ים רַכִּ֔ים וְהַצֹּ֥אן וְהַבָּקָ֖ר עָל֣וֹת עָלָ֑י וּדְפָקוּם֙ י֣וֹם אֶחָ֔ד וָמֵ֖תוּ כָּל־ הַצֹּֽאן׃ (מד) יַעֲבָר־ נָ֥א אֲדֹנִ֖י לִפְנֵ֣י עַבְדּ֑וֹ וַאֲנִ֞י אֶֽתְנָהֲלָ֣ה לְאִטִּ֗י לְרֶ֨גֶל הַמְּלָאכָ֤ה אֲשֶׁר־ לְפָנַי֙ וּלְרֶ֣גֶל הַיְלָדִ֔ים עַ֛ד אֲשֶׁר־ אָבֹ֥א אֶל־ אֲדֹנִ֖י שֵׂעִֽירָה׃ (מה) וַיֹּ֣אמֶר עֵשָׂ֔ו אַצִּֽיגָה־ נָּ֣א עִמְּךָ֔ מִן־ הָעָ֖ם אֲשֶׁ֣ר אִתִּ֑י וַיֹּ֙אמֶר֙ לָ֣מָּה זֶּ֔ה אֶמְצָא־ חֵ֖ן בְּעֵינֵ֥י אֲדֹנִֽי׃ (מו) וַיָּשָׁב֩ בַּיּ֨וֹם הַה֥וּא עֵשָׂ֛ו לְדַרְכּ֖וֹ שֵׂעִֽירָה׃ (מז) וְיַעֲקֹב֙ נָסַ֣ע סֻכֹּ֔תָה וַיִּ֥בֶן ל֖וֹ בָּ֑יִת וּלְמִקְנֵ֙הוּ֙ עָשָׂ֣ה סֻכֹּ֔ת עַל־ כֵּ֛ן קָרָ֥א שֵׁם־ הַמָּק֖וֹם סֻכּֽוֹת׃
