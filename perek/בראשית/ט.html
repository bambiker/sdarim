בראשית פרק ט לפי החלוקה לסדרים

פרק ט בספר בראשית (29 פסוקים) נמצא בבראשית סדר ז (פסוקים ט–כה) ובבראשית סדר ח (פסוקים א–יב).

בראשית סדר ז

(א) וַיְדַבֵּ֥ר אֱלֹהִ֖ים אֶל־ נֹ֥חַ לֵאמֹֽר׃ (ב) צֵ֖א מִן־ הַתֵּבָ֑ה אַתָּ֕ה וְאִשְׁתְּךָ֛ וּבָנֶ֥יךָ וּנְשֵֽׁי־ בָנֶ֖יךָ אִתָּֽךְ׃ (ג) כָּל־ הַחַיָּ֨ה אֲשֶֽׁר־ אִתְּךָ֜ מִכָּל־ בָּשָׂ֗ר בָּע֧וֹף וּבַבְּהֵמָ֛ה וּבְכָל־ הָרֶ֛מֶשׂ הָרֹמֵ֥שׂ עַל־ הָאָ֖רֶץ (הוצא) הַיְצֵ֣א אִתָּ֑ךְ וְשָֽׁרְצ֣וּ בָאָ֔רֶץ וּפָר֥וּ וְרָב֖וּ עַל־ הָאָֽרֶץ׃ (ד) וַיֵּ֖צֵא־ נֹ֑חַ וּבָנָ֛יו וְאִשְׁתּ֥וֹ וּנְשֵֽׁי־ בָנָ֖יו אִתּֽוֹ׃ (ה) כָּל־ הַֽחַיָּ֗ה כָּל־ הָרֶ֙מֶשׂ֙ וְכָל־ הָע֔וֹף כֹּ֖ל רוֹמֵ֣שׂ עַל־ הָאָ֑רֶץ לְמִשְׁפְּחֹ֣תֵיהֶ֔ם יָצְא֖וּ מִן־ הַתֵּבָֽה׃ (ו) וַיִּ֥בֶן נֹ֛חַ מִזְבֵּ֖חַ לַֽיהוָ֑ה וַיִּקַּ֞ח מִכֹּ֣ל ׀ הַבְּהֵמָ֣ה הַטְּהוֹרָ֗ה וּמִכֹּל֙ הָע֣וֹף הַטָּהֹ֔ר וַיַּ֥עַל עֹלֹ֖ת בַּמִּזְבֵּֽחַ׃ (ז) וַיָּ֣רַח יְהוָה֮ אֶת־ רֵ֣יחַ הַנִּיחֹחַ֒ וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ לִבּ֗וֹ לֹֽא־ אֹ֠סִף לְקַלֵּ֨ל ע֤וֹד אֶת־ הָֽאֲדָמָה֙ בַּעֲב֣וּר הָֽאָדָ֔ם כִּ֠י יֵ֣צֶר לֵ֧ב הָאָדָ֛ם רַ֖ע מִנְּעֻרָ֑יו וְלֹֽא־ אֹסִ֥ף ע֛וֹד לְהַכּ֥וֹת אֶת־ כָּל־ חַ֖י כַּֽאֲשֶׁ֥ר עָשִֽׂיתִי׃ (ח) עֹ֖ד כָּל־ יְמֵ֣י הָאָ֑רֶץ זֶ֡רַע וְ֠קָצִיר וְקֹ֨ר וָחֹ֜ם וְקַ֧יִץ וָחֹ֛רֶף וְי֥וֹם וָלַ֖יְלָה לֹ֥א יִשְׁבֹּֽתוּ׃ תחילת פרק ט (ט) וַיְבָ֣רֶךְ אֱלֹהִ֔ים אֶת־ נֹ֖חַ וְאֶת־ בָּנָ֑יו וַיֹּ֧אמֶר לָהֶ֛ם פְּר֥וּ וּרְב֖וּ וּמִלְא֥וּ אֶת־ הָאָֽרֶץ׃ (י) וּמוֹרַאֲכֶ֤ם וְחִתְּכֶם֙ יִֽהְיֶ֔ה עַ֚ל כָּל־ חַיַּ֣ת הָאָ֔רֶץ וְעַ֖ל כָּל־ ע֣וֹף הַשָּׁמָ֑יִם בְּכֹל֩ אֲשֶׁ֨ר תִּרְמֹ֧שׂ הָֽאֲדָמָ֛ה וּֽבְכָל־ דְּגֵ֥י הַיָּ֖ם בְּיֶדְכֶ֥ם נִתָּֽנוּ׃ (יא) כָּל־ רֶ֙מֶשׂ֙ אֲשֶׁ֣ר הוּא־ חַ֔י לָכֶ֥ם יִהְיֶ֖ה לְאָכְלָ֑ה כְּיֶ֣רֶק עֵ֔שֶׂב נָתַ֥תִּי לָכֶ֖ם אֶת־ כֹּֽל׃ (יב) אַךְ־ בָּשָׂ֕ר בְּנַפְשׁ֥וֹ דָמ֖וֹ לֹ֥א תֹאכֵֽלוּ׃ (יג) וְאַ֨ךְ אֶת־ דִּמְכֶ֤ם לְנַפְשֹֽׁתֵיכֶם֙ אֶדְרֹ֔שׁ מִיַּ֥ד כָּל־ חַיָּ֖ה אֶדְרְשֶׁ֑נּוּ וּמִיַּ֣ד הָֽאָדָ֗ם מִיַּד֙ אִ֣ישׁ אָחִ֔יו אֶדְרֹ֖שׁ אֶת־ נֶ֥פֶשׁ הָֽאָדָֽם׃ (יד) שֹׁפֵךְ֙ דַּ֣ם הָֽאָדָ֔ם בָּֽאָדָ֖ם דָּמ֣וֹ יִשָּׁפֵ֑ךְ כִּ֚י בְּצֶ֣לֶם אֱלֹהִ֔ים עָשָׂ֖ה אֶת־ הָאָדָֽם׃ (טו) וְאַתֶּ֖ם פְּר֣וּ וּרְב֑וּ שִׁרְצ֥וּ בָאָ֖רֶץ וּרְבוּ־ בָֽהּ׃      (טז) וַיֹּ֤אמֶר אֱלֹהִים֙ אֶל־ נֹ֔חַ וְאֶל־ בָּנָ֥יו אִתּ֖וֹ לֵאמֹֽר׃ (יז) וַאֲנִ֕י הִנְנִ֥י מֵקִ֛ים אֶת־ בְּרִיתִ֖י אִתְּכֶ֑ם וְאֶֽת־ זַרְעֲכֶ֖ם אַֽחֲרֵיכֶֽם׃ (יח) וְאֵ֨ת כָּל־ נֶ֤פֶשׁ הַֽחַיָּה֙ אֲשֶׁ֣ר אִתְּכֶ֔ם בָּע֧וֹף בַּבְּהֵמָ֛ה וּֽבְכָל־ חַיַּ֥ת הָאָ֖רֶץ אִתְּכֶ֑ם מִכֹּל֙ יֹצְאֵ֣י הַתֵּבָ֔ה לְכֹ֖ל חַיַּ֥ת הָאָֽרֶץ׃ (יט) וַהֲקִמֹתִ֤י אֶת־ בְּרִיתִי֙ אִתְּכֶ֔ם וְלֹֽא־ יִכָּרֵ֧ת כָּל־ בָּשָׂ֛ר ע֖וֹד מִמֵּ֣י הַמַּבּ֑וּל וְלֹֽא־ יִהְיֶ֥ה ע֛וֹד מַבּ֖וּל לְשַׁחֵ֥ת הָאָֽרֶץ׃ (כ) וַיֹּ֣אמֶר אֱלֹהִ֗ים זֹ֤את אֽוֹת־ הַבְּרִית֙ אֲשֶׁר־ אֲנִ֣י נֹתֵ֗ן בֵּינִי֙ וּבֵ֣ינֵיכֶ֔ם וּבֵ֛ין כָּל־ נֶ֥פֶשׁ חַיָּ֖ה אֲשֶׁ֣ר אִתְּכֶ֑ם לְדֹרֹ֖ת עוֹלָֽם׃ (כא) אֶת־ קַשְׁתִּ֕י נָתַ֖תִּי בֶּֽעָנָ֑ן וְהָֽיְתָה֙ לְא֣וֹת בְּרִ֔ית בֵּינִ֖י וּבֵ֥ין הָאָֽרֶץ׃ (כב) וְהָיָ֕ה בְּעַֽנְנִ֥י עָנָ֖ן עַל־ הָאָ֑רֶץ וְנִרְאֲתָ֥ה הַקֶּ֖שֶׁת בֶּעָנָֽן׃ (כג) וְזָכַרְתִּ֣י אֶת־ בְּרִיתִ֗י אֲשֶׁ֤ר בֵּינִי֙ וּבֵ֣ינֵיכֶ֔ם וּבֵ֛ין כָּל־ נֶ֥פֶשׁ חַיָּ֖ה בְּכָל־ בָּשָׂ֑ר וְלֹֽא־ יִֽהְיֶ֨ה ע֤וֹד הַמַּ֙יִם֙ לְמַבּ֔וּל לְשַׁחֵ֖ת כָּל־ בָּשָֽׂר׃ (כד) וְהָיְתָ֥ה הַקֶּ֖שֶׁת בֶּֽעָנָ֑ן וּרְאִיתִ֗יהָ לִזְכֹּר֙ בְּרִ֣ית עוֹלָ֔ם בֵּ֣ין אֱלֹהִ֔ים וּבֵין֙ כָּל־ נֶ֣פֶשׁ חַיָּ֔ה בְּכָל־ בָּשָׂ֖ר אֲשֶׁ֥ר עַל־ הָאָֽרֶץ׃ (כה) וַיֹּ֥אמֶר אֱלֹהִ֖ים אֶל־ נֹ֑חַ זֹ֤את אֽוֹת־ הַבְּרִית֙ אֲשֶׁ֣ר הֲקִמֹ֔תִי בֵּינִ֕י וּבֵ֥ין כָּל־ בָּשָׂ֖ר אֲשֶׁ֥ר עַל־ הָאָֽרֶץ׃

בראשית סדר ח

(א) וַיִּֽהְי֣וּ בְנֵי־ נֹ֗חַ הַיֹּֽצְאִים֙ מִן־ הַתֵּבָ֔ה שֵׁ֖ם וְחָ֣ם וָיָ֑פֶת וְחָ֕ם ה֖וּא אֲבִ֥י כְנָֽעַן׃ (ב) שְׁלֹשָׁ֥ה אֵ֖לֶּה בְּנֵי־ נֹ֑חַ וּמֵאֵ֖לֶּה נָֽפְצָ֥ה כָל־ הָאָֽרֶץ׃ (ג) וַיָּ֥חֶל נֹ֖חַ אִ֣ישׁ הָֽאֲדָמָ֑ה וַיִּטַּ֖ע כָּֽרֶם׃ (ד) וַיֵּ֥שְׁתְּ מִן־ הַיַּ֖יִן וַיִּשְׁכָּ֑ר וַיִּתְגַּ֖ל בְּת֥וֹךְ אָהֳלֹֽה׃ (ה) וַיַּ֗רְא חָ֚ם אֲבִ֣י כְנַ֔עַן אֵ֖ת עֶרְוַ֣ת אָבִ֑יו וַיַּגֵּ֥ד לִשְׁנֵֽי־ אֶחָ֖יו בַּחֽוּץ׃ (ו) וַיִּקַּח֩ שֵׁ֨ם וָיֶ֜פֶת אֶת־ הַשִּׂמְלָ֗ה וַיָּשִׂ֙ימוּ֙ עַל־ שְׁכֶ֣ם שְׁנֵיהֶ֔ם וַיֵּֽלְכוּ֙ אֲחֹ֣רַנִּ֔ית וַיְכַסּ֕וּ אֵ֖ת עֶרְוַ֣ת אֲבִיהֶ֑ם וּפְנֵיהֶם֙ אֲחֹ֣רַנִּ֔ית וְעֶרְוַ֥ת אֲבִיהֶ֖ם לֹ֥א רָאֽוּ׃ (ז) וַיִּ֥יקֶץ נֹ֖חַ מִיֵּינ֑וֹ וַיֵּ֕דַע אֵ֛ת אֲשֶׁר־ עָ֥שָׂה־ ל֖וֹ בְּנ֥וֹ הַקָּטָֽן׃ (ח) וַיֹּ֖אמֶר אָר֣וּר כְּנָ֑עַן עֶ֥בֶד עֲבָדִ֖ים יִֽהְיֶ֥ה לְאֶחָֽיו׃ (ט) וַיֹּ֕אמֶר בָּר֥וּךְ יְהֹוָ֖ה אֱלֹ֣הֵי שֵׁ֑ם וִיהִ֥י כְנַ֖עַן עֶ֥בֶד לָֽמוֹ׃ (י) יַ֤פְתְּ אֱלֹהִים֙ לְיֶ֔פֶת וְיִשְׁכֹּ֖ן בְּאָֽהֳלֵי־ שֵׁ֑ם וִיהִ֥י כְנַ֖עַן עֶ֥בֶד לָֽמוֹ׃ (יא) וַֽיְחִי־ נֹ֖חַ אַחַ֣ר הַמַּבּ֑וּל שְׁלֹ֤שׁ מֵאוֹת֙ שָׁנָ֔ה וַֽחֲמִשִּׁ֖ים שָׁנָֽה׃ (יב) וַיִּֽהְיוּ֙ כָּל־ יְמֵי־ נֹ֔חַ תְּשַׁ֤ע מֵאוֹת֙ שָׁנָ֔ה וַחֲמִשִּׁ֖ים שָׁנָ֑ה וַיָּמֹֽת׃
סוף פרק ט (יג) וְאֵ֙לֶּה֙ תּוֹלְדֹ֣ת בְּנֵי־ נֹ֔חַ שֵׁ֖ם חָ֣ם וָיָ֑פֶת וַיִּוָּלְד֥וּ לָהֶ֛ם בָּנִ֖ים אַחַ֥ר הַמַּבּֽוּל׃ (יד) בְּנֵ֣י יֶ֔פֶת גֹּ֣מֶר וּמָג֔וֹג וּמָדַ֖י וְיָוָ֣ן וְתֻבָ֑ל וּמֶ֖שֶׁךְ וְתִירָֽס׃ (טו) וּבְנֵ֖י גֹּ֑מֶר אַשְׁכֲּנַ֥ז וְרִיפַ֖ת וְתֹגַרְמָֽה׃ (טז) וּבְנֵ֥י יָוָ֖ן אֱלִישָׁ֣ה וְתַרְשִׁ֑ישׁ כִּתִּ֖ים וְדֹדָנִֽים׃ (יז) מֵ֠אֵלֶּה נִפְרְד֞וּ אִיֵּ֤י הַגּוֹיִם֙ בְּאַרְצֹתָ֔ם אִ֖ישׁ לִלְשֹׁנ֑וֹ לְמִשְׁפְּחֹתָ֖ם בְּגוֹיֵהֶֽם׃ (יח) וּבְנֵ֖י חָ֑ם כּ֥וּשׁ וּמִצְרַ֖יִם וּפ֥וּט וּכְנָֽעַן׃ (יט) וּבְנֵ֣י כ֔וּשׁ סְבָא֙ וַֽחֲוִילָ֔ה וְסַבְתָּ֥ה וְרַעְמָ֖ה וְסַבְתְּכָ֑א וּבְנֵ֥י רַעְמָ֖ה שְׁבָ֥א וּדְדָֽן׃ (כ) וְכ֖וּשׁ יָלַ֣ד אֶת־ נִמְרֹ֑ד ה֣וּא הֵחֵ֔ל לִֽהְי֥וֹת גִּבֹּ֖ר בָּאָֽרֶץ׃ (כא) הֽוּא־ הָיָ֥ה גִבֹּֽר־ צַ֖יִד לִפְנֵ֣י יְהוָ֑ה עַל־ כֵּן֙ יֵֽאָמַ֔ר כְּנִמְרֹ֛ד גִּבּ֥וֹר צַ֖יִד לִפְנֵ֥י יְהוָֽה׃ (כב) וַתְּהִ֨י רֵאשִׁ֤ית מַמְלַכְתּוֹ֙ בָּבֶ֔ל וְאֶ֖רֶךְ וְאַכַּ֣ד וְכַלְנֵ֑ה בְּאֶ֖רֶץ שִׁנְעָֽר׃ (כג) מִן־ הָאָ֥רֶץ הַהִ֖וא יָצָ֣א אַשּׁ֑וּר וַיִּ֙בֶן֙ אֶת־ נִ֣ינְוֵ֔ה וְאֶת־ רְחֹבֹ֥ת עִ֖יר וְאֶת־ כָּֽלַח׃ (כד) וְֽאֶת־ רֶ֔סֶן בֵּ֥ין נִֽינְוֵ֖ה וּבֵ֣ין כָּ֑לַח הִ֖וא הָעִ֥יר הַגְּדֹלָֽה׃ (כה) וּמִצְרַ֡יִם יָלַ֞ד אֶת־ לוּדִ֧ים וְאֶת־ עֲנָמִ֛ים וְאֶת־ לְהָבִ֖ים וְאֶת־ נַפְתֻּחִֽים׃ (כו) וְֽאֶת־ פַּתְרֻסִ֞ים וְאֶת־ כַּסְלֻחִ֗ים אֲשֶׁ֨ר יָצְא֥וּ מִשָּׁ֛ם פְּלִשְׁתִּ֖ים וְאֶת־ כַּפְתֹּרִֽים׃      (כז) וּכְנַ֗עַן יָלַ֛ד אֶת־ צִידֹ֥ן בְּכֹר֖וֹ וְאֶת־ חֵֽת׃ (כח) וְאֶת־ הַיְבוּסִי֙ וְאֶת־ הָ֣אֱמֹרִ֔י וְאֵ֖ת הַגִּרְגָּשִֽׁי׃ (כט) וְאֶת־ הַֽחִוִּ֥י וְאֶת־ הַֽעַרְקִ֖י וְאֶת־ הַסִּינִֽי׃ (ל) וְאֶת־ הָֽאַרְוָדִ֥י וְאֶת־ הַצְּמָרִ֖י וְאֶת־ הַֽחֲמָתִ֑י וְאַחַ֣ר נָפֹ֔צוּ מִשְׁפְּח֖וֹת הַֽכְּנַעֲנִֽי׃ (לא) וַֽיְהִ֞י גְּב֤וּל הַֽכְּנַעֲנִי֙ מִצִּידֹ֔ן בֹּאֲכָ֥ה גְרָ֖רָה עַד־ עַזָּ֑ה בֹּאֲכָ֞ה סְדֹ֧מָה וַעֲמֹרָ֛ה וְאַדְמָ֥ה וּצְבֹיִ֖ם עַד־ לָֽשַׁע׃ (לב) אֵ֣לֶּה בְנֵי־ חָ֔ם לְמִשְׁפְּחֹתָ֖ם לִלְשֹֽׁנֹתָ֑ם בְּאַרְצֹתָ֖ם בְּגוֹיֵהֶֽם׃      (לג) וּלְשֵׁ֥ם יֻלַּ֖ד גַּם־ ה֑וּא אֲבִי֙ כָּל־ בְּנֵי־ עֵ֔בֶר אֲחִ֖י יֶ֥פֶת הַגָּדֽוֹל׃ (לד) בְּנֵ֥י שֵׁ֖ם עֵילָ֣ם וְאַשּׁ֑וּר וְאַרְפַּכְשַׁ֖ד וְל֥וּד וַֽאֲרָֽם׃ (לה) וּבְנֵ֖י אֲרָ֑ם ע֥וּץ וְח֖וּל וְגֶ֥תֶר וָמַֽשׁ׃ (לו) וְאַרְפַּכְשַׁ֖ד יָלַ֣ד אֶת־ שָׁ֑לַח וְשֶׁ֖לַח יָלַ֥ד אֶת־ עֵֽבֶר׃ (לז) וּלְעֵ֥בֶר יֻלַּ֖ד שְׁנֵ֣י בָנִ֑ים שֵׁ֣ם הָֽאֶחָ֞ד פֶּ֗לֶג כִּ֤י בְיָמָיו֙ נִפְלְגָ֣ה הָאָ֔רֶץ וְשֵׁ֥ם אָחִ֖יו יָקְטָֽן׃ (לח) וְיָקְטָ֣ן יָלַ֔ד אֶת־ אַלְמוֹדָ֖ד וְאֶת־ שָׁ֑לֶף וְאֶת־ חֲצַרְמָ֖וֶת וְאֶת־ יָֽרַח׃ (לט) וְאֶת־ הֲדוֹרָ֥ם וְאֶת־ אוּזָ֖ל וְאֶת־ דִּקְלָֽה׃ (מ) וְאֶת־ עוֹבָ֥ל וְאֶת־ אֲבִֽימָאֵ֖ל וְאֶת־ שְׁבָֽא׃ (מא) וְאֶת־ אוֹפִ֥ר וְאֶת־ חֲוִילָ֖ה וְאֶת־ יוֹבָ֑ב כָּל־ אֵ֖לֶּה בְּנֵ֥י יָקְטָֽן׃ (מב) וַֽיְהִ֥י מוֹשָׁבָ֖ם מִמֵּשָׁ֑א בֹּאֲכָ֥ה סְפָ֖רָה הַ֥ר הַקֶּֽדֶם׃ (מג) אֵ֣לֶּה בְנֵי־ שֵׁ֔ם לְמִשְׁפְּחֹתָ֖ם לִלְשֹׁנֹתָ֑ם בְּאַרְצֹתָ֖ם לְגוֹיֵהֶֽם׃ (מד) אֵ֣לֶּה מִשְׁפְּחֹ֧ת בְּנֵי־ נֹ֛חַ לְתוֹלְדֹתָ֖ם בְּגוֹיֵהֶ֑ם וּמֵאֵ֜לֶּה נִפְרְד֧וּ הַגּוֹיִ֛ם בָּאָ֖רֶץ אַחַ֥ר הַמַּבּֽוּל׃
