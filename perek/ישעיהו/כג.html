ישעיהו פרק כג לפי החלוקה לסדרים

פרק כג בספר ישעיהו (18 פסוקים) נמצא בישעיהו סדר ט (פסוקים ד–כא).

ישעיהו סדר ט

(א) וּתְקַעְתִּ֥יו יָתֵ֖ד בְּמָק֣וֹם נֶאֱמָ֑ן וְהָיָ֛ה לְכִסֵּ֥א כָב֖וֹד לְבֵ֥ית אָבִֽיו׃ (ב) וְתָל֨וּ עָלָ֜יו כֹּ֣ל ׀ כְּב֣וֹד בֵּית־ אָבִ֗יו הַצֶּֽאֱצָאִים֙ וְהַצְּפִע֔וֹת כֹּ֖ל כְּלֵ֣י הַקָּטָ֑ן מִכְּלֵי֙ הָֽאַגָּנ֔וֹת וְעַ֖ד כָּל־ כְּלֵ֥י הַנְּבָלִֽים׃ (ג) בַּיּ֣וֹם הַה֗וּא נְאֻם֙ יְהוָ֣ה צְבָא֔וֹת תָּמוּשׁ֙ הַיָּתֵ֔ד הַתְּקוּעָ֖ה בְּמָק֣וֹם נֶאֱמָ֑ן וְנִגְדְּעָ֣ה וְנָפְלָ֗ה וְנִכְרַת֙ הַמַּשָּׂ֣א אֲשֶׁר־ עָלֶ֔יהָ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ תחילת פרק כג (ד) מַשָּׂ֖א צֹ֑ר הֵילִ֣ילוּ ׀ אֳנִיּ֣וֹת תַּרְשִׁ֗ישׁ כִּֽי־ שֻׁדַּ֤ד מִבַּ֙יִת֙ מִבּ֔וֹא מֵאֶ֥רֶץ כִּתִּ֖ים נִגְלָה־ לָֽמוֹ׃ (ה) דֹּ֖מּוּ יֹ֣שְׁבֵי אִ֑י סֹחֵ֥ר צִיד֛וֹן עֹבֵ֥ר יָ֖ם מִלְאֽוּךְ׃ (ו) וּבְמַ֤יִם רַבִּים֙ זֶ֣רַע שִׁחֹ֔ר קְצִ֥יר יְא֖וֹר תְּבֽוּאָתָ֑הּ וַתְּהִ֖י סְחַ֥ר גּוֹיִֽם׃ (ז) בּ֣וֹשִׁי צִיד֔וֹן כִּֽי־ אָמַ֣ר יָ֔ם מָע֥וֹז הַיָּ֖ם לֵאמֹ֑ר לֹֽא־ חַ֣לְתִּי וְלֹֽא־ יָלַ֗דְתִּי וְלֹ֥א גִדַּ֛לְתִּי בַּחוּרִ֖ים רוֹמַ֥מְתִּי בְתוּלֽוֹת׃ (ח) כַּֽאֲשֶׁר־ שֵׁ֖מַע לְמִצְרָ֑יִם יָחִ֖ילוּ כְּשֵׁ֥מַע צֹֽר׃ (ט) עִבְר֖וּ תַּרְשִׁ֑ישָׁה הֵילִ֖ילוּ יֹ֥שְׁבֵי אִֽי׃ (י) הֲזֹ֥את לָכֶ֖ם עַלִּיזָ֑ה מִֽימֵי־ קֶ֤דֶם קַדְמָתָהּ֙ יֹבִל֣וּהָ רַגְלֶ֔יהָ מֵֽרָח֖וֹק לָגֽוּר׃ (יא) מִ֚י יָעַ֣ץ זֹ֔את עַל־ צֹ֖ר הַמַּֽעֲטִירָ֑ה אֲשֶׁ֤ר סֹחֲרֶ֙יה֙ שָׂרִ֔ים כִּנְעָנֶ֖יהָ נִכְבַּדֵּי־ אָֽרֶץ׃ (יב) יְהוָ֥ה צְבָא֖וֹת יְעָצָ֑הּ לְחַלֵּל֙ גְּא֣וֹן כָּל־ צְבִ֔י לְהָקֵ֖ל כָּל־ נִכְבַּדֵּי־ אָֽרֶץ׃ (יג) עִבְרִ֥י אַרְצֵ֖ךְ כַּיְאֹ֑ר בַּת־ תַּרְשִׁ֕ישׁ אֵ֖ין מֵ֥זַח עֽוֹד׃ (יד) יָדוֹ֙ נָטָ֣ה עַל־ הַיָּ֔ם הִרְגִּ֖יז מַמְלָכ֑וֹת יְהוָה֙ צִוָּ֣ה אֶל־ כְּנַ֔עַן לַשְׁמִ֖ד מָעֻזְנֶֽיהָ׃ (טו) וַיֹּ֕אמֶר לֹֽא־ תוֹסִ֥יפִי ע֖וֹד לַעְל֑וֹז הַֽמְעֻשָּׁקָ֞ה בְּתוּלַ֣ת בַּת־ צִיד֗וֹן (כתיים) כִּתִּים֙ ק֣וּמִי עֲבֹ֔רִי גַּם־ שָׁ֖ם לֹא־ יָנ֥וּחַֽ לָֽךְ׃ (טז) הֵ֣ן ׀ אֶ֣רֶץ כַּשְׂדִּ֗ים זֶ֤ה הָעָם֙ לֹ֣א הָיָ֔ה אַשּׁ֖וּר יְסָדָ֣הּ לְצִיִּ֑ים הֵקִ֣ימוּ (בחיניו) בַחוּנָ֗יו עֹרְרוּ֙ אַרְמְנוֹתֶ֔יהָ שָׂמָ֖הּ לְמַפֵּלָֽה׃ (יז) הֵילִ֖ילוּ אֳנִיּ֣וֹת תַּרְשִׁ֑ישׁ כִּ֥י שֻׁדַּ֖ד מָעֻזְּכֶֽן׃ (יח) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא וְנִשְׁכַּ֤חַת צֹר֙ שִׁבְעִ֣ים שָׁנָ֔ה כִּימֵ֖י מֶ֣לֶךְ אֶחָ֑ד מִקֵּ֞ץ שִׁבְעִ֤ים שָׁנָה֙ יִהְיֶ֣ה לְצֹ֔ר כְּשִׁירַ֖ת הַזּוֹנָֽה׃ (יט) קְחִ֥י כִנּ֛וֹר סֹ֥בִּי עִ֖יר זוֹנָ֣ה נִשְׁכָּחָ֑ה הֵיטִ֤יבִי נַגֵּן֙ הַרְבִּי־ שִׁ֔יר לְמַ֖עַן תִּזָּכֵֽרִי׃ (כ) וְהָיָ֞ה מִקֵּ֣ץ ׀ שִׁבְעִ֣ים שָׁנָ֗ה יִפְקֹ֤ד יְהוָה֙ אֶת־ צֹ֔ר וְשָׁבָ֖ה לְאֶתְנַנָּ֑ה וְזָֽנְתָ֛ה אֶת־ כָּל־ מַמְלְכ֥וֹת הָאָ֖רֶץ עַל־ פְּנֵ֥י הָאֲדָמָֽה׃ (כא) וְהָיָ֨ה סַחְרָ֜הּ וְאֶתְנַנָּ֗הּ קֹ֚דֶשׁ לַֽיהוָ֔ה לֹ֥א יֵֽאָצֵ֖ר וְלֹ֣א יֵֽחָסֵ֑ן כִּ֣י לַיֹּשְׁבִ֞ים לִפְנֵ֤י יְהוָה֙ יִֽהְיֶ֣ה סַחְרָ֔הּ לֶאֱכֹ֥ל לְשָׂבְעָ֖ה וְלִמְכַסֶּ֥ה עָתִֽיק׃ סוף פרק כג (כב) הִנֵּ֧ה יְהוָ֛ה בּוֹקֵ֥ק הָאָ֖רֶץ וּבֽוֹלְקָ֑הּ וְעִוָּ֣ה פָנֶ֔יהָ וְהֵפִ֖יץ יֹשְׁבֶֽיהָ׃ (כג) וְהָיָ֤ה כָעָם֙ כַּכֹּהֵ֔ן כַּעֶ֙בֶד֙ כַּֽאדֹנָ֔יו כַּשִּׁפְחָ֖ה כַּגְּבִרְתָּ֑הּ כַּקּוֹנֶה֙ כַּמּוֹכֵ֔ר כַּמַּלְוֶה֙ כַּלֹּוֶ֔ה כַּנֹּשֶׁ֕ה כַּאֲשֶׁ֖ר נֹשֶׁ֥א בֽוֹ׃ (כד) הִבּ֧וֹק ׀ תִּבּ֛וֹק הָאָ֖רֶץ וְהִבּ֣וֹז ׀ תִּבּ֑וֹז כִּ֣י יְהוָ֔ה דִּבֶּ֖ר אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (כה) אָבְלָ֤ה נָֽבְלָה֙ הָאָ֔רֶץ אֻמְלְלָ֥ה נָבְלָ֖ה תֵּבֵ֑ל אֻמְלָ֖לוּ מְר֥וֹם עַם־ הָאָֽרֶץ׃ (כו) וְהָאָ֥רֶץ חָנְפָ֖ה תַּ֣חַת יֹשְׁבֶ֑יהָ כִּֽי־ עָבְר֤וּ תוֹרֹת֙ חָ֣לְפוּ חֹ֔ק הֵפֵ֖רוּ בְּרִ֥ית עוֹלָֽם׃ (כז) עַל־ כֵּ֗ן אָלָה֙ אָ֣כְלָה אֶ֔רֶץ וַֽיֶּאְשְׁמ֖וּ יֹ֣שְׁבֵי בָ֑הּ עַל־ כֵּ֗ן חָרוּ֙ יֹ֣שְׁבֵי אֶ֔רֶץ וְנִשְׁאַ֥ר אֱנ֖וֹשׁ מִזְעָֽר׃ (כח) אָבַ֥ל תִּיר֖וֹשׁ אֻמְלְלָה־ גָ֑פֶן נֶאֶנְח֖וּ כָּל־ שִׂמְחֵי־ לֵֽב׃ (כט) שָׁבַת֙ מְשׂ֣וֹשׂ תֻּפִּ֔ים חָדַ֖ל שְׁא֣וֹן עַלִּיזִ֑ים שָׁבַ֖ת מְשׂ֥וֹשׂ כִּנּֽוֹר׃ (ל) בַּשִּׁ֖יר לֹ֣א יִשְׁתּוּ־ יָ֑יִן יֵמַ֥ר שֵׁכָ֖ר לְשֹׁתָֽיו׃ (לא) נִשְׁבְּרָ֖ה קִרְיַת־ תֹּ֑הוּ סֻגַּ֥ר כָּל־ בַּ֖יִת מִבּֽוֹא׃ (לב) צְוָחָ֥ה עַל־ הַיַּ֖יִן בַּֽחוּצ֑וֹת עָֽרְבָה֙ כָּל־ שִׂמְחָ֔ה גָּלָ֖ה מְשׂ֥וֹשׂ הָאָֽרֶץ׃ (לג) נִשְׁאַ֥ר בָּעִ֖יר שַׁמָּ֑ה וּשְׁאִיָּ֖ה יֻכַּת־ שָֽׁעַר׃ (לד) כִּ֣י כֹ֥ה יִהְיֶ֛ה בְּקֶ֥רֶב הָאָ֖רֶץ בְּת֣וֹךְ הָֽעַמִּ֑ים כְּנֹ֣קֶף זַ֔יִת כְּעוֹלֵלֹ֖ת אִם־ כָּלָ֥ה בָצִֽיר׃ (לה) הֵ֛מָּה יִשְׂא֥וּ קוֹלָ֖ם יָרֹ֑נּוּ בִּגְא֣וֹן יְהוָ֔ה צָהֲל֖וּ מִיָּֽם׃ (לו) עַל־ כֵּ֥ן בָּאֻרִ֖ים כַּבְּד֣וּ יְהוָ֑ה בְּאִיֵּ֣י הַיָּ֔ם שֵׁ֥ם יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (לז) מִכְּנַ֨ף הָאָ֜רֶץ זְמִרֹ֤ת שָׁמַ֙עְנוּ֙ צְבִ֣י לַצַּדִּ֔יק וָאֹמַ֛ר רָזִי־ לִ֥י רָֽזִי־ לִ֖י א֣וֹי לִ֑י בֹּגְדִ֣ים בָּגָ֔דוּ וּבֶ֥גֶד בּוֹגְדִ֖ים בָּגָֽדוּ׃ (לח) פַּ֥חַד וָפַ֖חַת וָפָ֑ח עָלֶ֖יךָ יוֹשֵׁ֥ב הָאָֽרֶץ׃ (לט) וְֽ֠הָיָה הַנָּ֞ס מִקּ֤וֹל הַפַּ֙חַד֙ יִפֹּ֣ל אֶל־ הַפַּ֔חַת וְהָֽעוֹלֶה֙ מִתּ֣וֹךְ הַפַּ֔חַת יִלָּכֵ֖ד בַּפָּ֑ח כִּֽי־ אֲרֻבּ֤וֹת מִמָּרוֹם֙ נִפְתָּ֔חוּ וַֽיִּרְעֲשׁ֖וּ מ֥וֹסְדֵי אָֽרֶץ׃ (מ) רֹ֥עָה הִֽתְרֹעֲעָ֖ה הָאָ֑רֶץ פּ֤וֹר הִֽתְפּוֹרְרָה֙ אֶ֔רֶץ מ֥וֹט הִֽתְמוֹטְטָ֖ה אָֽרֶץ׃ (מא) נ֣וֹעַ תָּנ֤וּעַ אֶ֙רֶץ֙ כַּשִּׁכּ֔וֹר וְהִֽתְנוֹדְדָ֖ה כַּמְּלוּנָ֑ה וְכָבַ֤ד עָלֶ֙יהָ֙ פִּשְׁעָ֔הּ וְנָפְלָ֖ה וְלֹא־ תֹסִ֥יף קֽוּם׃ (מב) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא יִפְקֹ֧ד יְהוָ֛ה עַל־ צְבָ֥א הַמָּר֖וֹם בַּמָּר֑וֹם וְעַל־ מַלְכֵ֥י הָאֲדָמָ֖ה עַל־ הָאֲדָמָֽה׃ (מג) וְאֻסְּפ֨וּ אֲסֵפָ֤ה אַסִּיר֙ עַל־ בּ֔וֹר וְסֻגְּר֖וּ עַל־ מַסְגֵּ֑ר וּמֵרֹ֥ב יָמִ֖ים יִפָּקֵֽדוּ׃
