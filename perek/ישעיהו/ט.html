ישעיהו פרק ט לפי החלוקה לסדרים

פרק ט בספר ישעיהו (20 פסוקים) נמצא בישעיהו סדר ג (פסוקים ס–סד) ובישעיהו סדר ד (פסוקים א–טו).

ישעיהו סדר ג

(א) וְקָרָ֨א זֶ֤ה אֶל־ זֶה֙ וְאָמַ֔ר קָד֧וֹשׁ ׀ קָד֛וֹשׁ קָד֖וֹשׁ יְהוָ֣ה צְבָא֑וֹת מְלֹ֥א כָל־ הָאָ֖רֶץ כְּבוֹדֽוֹ׃ (ב) וַיָּנֻ֙עוּ֙ אַמּ֣וֹת הַסִּפִּ֔ים מִקּ֖וֹל הַקּוֹרֵ֑א וְהַבַּ֖יִת יִמָּלֵ֥א עָשָֽׁן׃ (ג) וָאֹמַ֞ר אֽוֹי־ לִ֣י כִֽי־ נִדְמֵ֗יתִי כִּ֣י אִ֤ישׁ טְמֵֽא־ שְׂפָתַ֙יִם֙ אָנֹ֔כִי וּבְתוֹךְ֙ עַם־ טְמֵ֣א שְׂפָתַ֔יִם אָנֹכִ֖י יוֹשֵׁ֑ב כִּ֗י אֶת־ הַמֶּ֛לֶךְ יְהוָ֥ה צְבָא֖וֹת רָא֥וּ עֵינָֽי׃ (ד) וַיָּ֣עָף אֵלַ֗י אֶחָד֙ מִן־ הַשְּׂרָפִ֔ים וּבְיָד֖וֹ רִצְפָּ֑ה בְּמֶ֨לְקַחַ֔יִם לָקַ֖ח מֵעַ֥ל הַמִּזְבֵּֽחַ׃ (ה) וַיַּגַּ֣ע עַל־ פִּ֔י וַיֹּ֕אמֶר הִנֵּ֛ה נָגַ֥ע זֶ֖ה עַל־ שְׂפָתֶ֑יךָ וְסָ֣ר עֲוֺנֶ֔ךָ וְחַטָּאתְךָ֖ תְּכֻפָּֽר׃ (ו) וָאֶשְׁמַ֞ע אֶת־ ק֤וֹל אֲדֹנָי֙ אֹמֵ֔ר אֶת־ מִ֥י אֶשְׁלַ֖ח וּמִ֣י יֵֽלֶךְ־ לָ֑נוּ וָאֹמַ֖ר הִנְנִ֥י שְׁלָחֵֽנִי׃ (ז) וַיֹּ֕אמֶר לֵ֥ךְ וְאָמַרְתָּ֖ לָעָ֣ם הַזֶּ֑ה שִׁמְע֤וּ שָׁמ֙וֹעַ֙ וְאַל־ תָּבִ֔ינוּ וּרְא֥וּ רָא֖וֹ וְאַל־ תֵּדָֽעוּ׃ (ח) הַשְׁמֵן֙ לֵב־ הָעָ֣ם הַזֶּ֔ה וְאָזְנָ֥יו הַכְבֵּ֖ד וְעֵינָ֣יו הָשַׁ֑ע פֶּן־ יִרְאֶ֨ה בְעֵינָ֜יו וּבְאָזְנָ֣יו יִשְׁמָ֗ע וּלְבָב֥וֹ יָבִ֛ין וָשָׁ֖ב וְרָ֥פָא לֽוֹ׃ (ט) וָאֹמַ֕ר עַד־ מָתַ֖י אֲדֹנָ֑י וַיֹּ֡אמֶר עַ֣ד אֲשֶׁר֩ אִם־ שָׁא֨וּ עָרִ֜ים מֵאֵ֣ין יוֹשֵׁ֗ב וּבָתִּים֙ מֵאֵ֣ין אָדָ֔ם וְהָאֲדָמָ֖ה תִּשָּׁאֶ֥ה שְׁמָמָֽה׃ (י) וְרִחַ֥ק יְהוָ֖ה אֶת־ הָאָדָ֑ם וְרַבָּ֥ה הָעֲזוּבָ֖ה בְּקֶ֥רֶב הָאָֽרֶץ׃ (יא) וְע֥וֹד בָּהּ֙ עֲשִׂ֣רִיָּ֔ה וְשָׁ֖בָה וְהָיְתָ֣ה לְבָעֵ֑ר כָּאֵלָ֣ה וְכָאַלּ֗וֹן אֲשֶׁ֤ר בְּשַׁלֶּ֙כֶת֙ מַצֶּ֣בֶת בָּ֔ם זֶ֥רַע קֹ֖דֶשׁ מַצַּבְתָּֽהּ׃ (יב) וַיְהִ֡י בִּימֵ֣י אָ֠חָז בֶּן־ יוֹתָ֨ם בֶּן־ עֻזִּיָּ֜הוּ מֶ֣לֶךְ יְהוּדָ֗ה עָלָ֣ה רְצִ֣ין מֶֽלֶךְ־ אֲ֠רָם וּפֶ֨קַח בֶּן־ רְמַלְיָ֤הוּ מֶֽלֶךְ־ יִשְׂרָאֵל֙ יְר֣וּשָׁלִַ֔ם לַמִּלְחָמָ֖ה עָלֶ֑יהָ וְלֹ֥א יָכֹ֖ל לְהִלָּחֵ֥ם עָלֶֽיהָ׃ (יג) וַיֻּגַּ֗ד לְבֵ֤ית דָּוִד֙ לֵאמֹ֔ר נָ֥חָֽה אֲרָ֖ם עַל־ אֶפְרָ֑יִם וַיָּ֤נַע לְבָבוֹ֙ וּלְבַ֣ב עַמּ֔וֹ כְּנ֥וֹעַ עֲצֵי־ יַ֖עַר מִפְּנֵי־ רֽוּחַ׃ (יד) וַיֹּ֣אמֶר יְהוָה֮ אֶֽל־ יְשַׁעְיָהוּ֒ צֵא־ נָא֙ לִקְרַ֣את אָחָ֔ז אַתָּ֕ה וּשְׁאָ֖ר יָשׁ֣וּב בְּנֶ֑ךָ אֶל־ קְצֵ֗ה תְּעָלַת֙ הַבְּרֵכָ֣ה הָעֶלְיוֹנָ֔ה אֶל־ מְסִלַּ֖ת שְׂדֵ֥ה כוֹבֵֽס׃ (טו) וְאָמַרְתָּ֣ אֵ֠לָיו הִשָּׁמֵ֨ר וְהַשְׁקֵ֜ט אַל־ תִּירָ֗א וּלְבָבְךָ֙ אַל־ יֵרַ֔ךְ מִשְּׁנֵ֨י זַנְב֧וֹת הָאוּדִ֛ים הָעֲשֵׁנִ֖ים הָאֵ֑לֶּה בָּחֳרִי־ אַ֛ף רְצִ֥ין וַאֲרָ֖ם וּבֶן־ רְמַלְיָֽהוּ׃ (טז) יַ֗עַן כִּֽי־ יָעַ֥ץ עָלֶ֛יךָ אֲרָ֖ם רָעָ֑ה אֶפְרַ֥יִם וּבֶן־ רְמַלְיָ֖הוּ לֵאמֹֽר׃ (יז) נַעֲלֶ֤ה בִֽיהוּדָה֙ וּנְקִיצֶ֔נָּה וְנַבְקִעֶ֖נָּה אֵלֵ֑ינוּ וְנַמְלִ֥יךְ מֶ֙לֶךְ֙ בְּתוֹכָ֔הּ אֵ֖ת בֶּן־ טָֽבְאַֽל׃ (יח) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה לֹ֥א תָק֖וּם וְלֹ֥א תִֽהְיֶֽה׃ (יט) כִּ֣י רֹ֤אשׁ אֲרָם֙ דַּמֶּ֔שֶׂק וְרֹ֥אשׁ דַּמֶּ֖שֶׂק רְצִ֑ין וּבְע֗וֹד שִׁשִּׁ֤ים וְחָמֵשׁ֙ שָׁנָ֔ה יֵחַ֥ת אֶפְרַ֖יִם מֵעָֽם׃ (כ) וְרֹ֤אשׁ אֶפְרַ֙יִם֙ שֹׁמְר֔וֹן וְרֹ֥אשׁ שֹׁמְר֖וֹן בֶּן־ רְמַלְיָ֑הוּ אִ֚ם לֹ֣א תַאֲמִ֔ינוּ כִּ֖י לֹ֥א תֵאָמֵֽנוּ׃ (כא) וַיּ֣וֹסֶף יְהוָ֔ה דַּבֵּ֥ר אֶל־ אָחָ֖ז לֵאמֹֽר׃ (כב) שְׁאַל־ לְךָ֣ א֔וֹת מֵעִ֖ם יְהוָ֣ה אֱלֹהֶ֑יךָ הַעְמֵ֣ק שְׁאָ֔לָה א֖וֹ הַגְבֵּ֥הַּ לְמָֽעְלָה׃ (כג) וַיֹּ֖אמֶר אָחָ֑ז לֹא־ אֶשְׁאַ֥ל וְלֹֽא־ אֲנַסֶּ֖ה אֶת־ יְהוָֽה׃ (כד) וַיֹּ֕אמֶר שִׁמְעוּ־ נָ֖א בֵּ֣ית דָּוִ֑ד הַמְעַ֤ט מִכֶּם֙ הַלְא֣וֹת אֲנָשִׁ֔ים כִּ֥י תַלְא֖וּ גַּ֥ם אֶת־ אֱלֹהָֽי׃ (כה) לָ֠כֵן יִתֵּ֨ן אֲדֹנָ֥י ה֛וּא לָכֶ֖ם א֑וֹת הִנֵּ֣ה הָעַלְמָ֗ה הָרָה֙ וְיֹלֶ֣דֶת בֵּ֔ן וְקָרָ֥את שְׁמ֖וֹ עִמָּ֥נוּ אֵֽל׃ (כו) חֶמְאָ֥ה וּדְבַ֖שׁ יֹאכֵ֑ל לְדַעְתּ֛וֹ מָא֥וֹס בָּרָ֖ע וּבָח֥וֹר בַּטּֽוֹב׃ (כז) כִּ֠י בְּטֶ֨רֶם יֵדַ֥ע הַנַּ֛עַר מָאֹ֥ס בָּרָ֖ע וּבָחֹ֣ר בַּטּ֑וֹב תֵּעָזֵ֤ב הָאֲדָמָה֙ אֲשֶׁ֣ר אַתָּ֣ה קָ֔ץ מִפְּנֵ֖י שְׁנֵ֥י מְלָכֶֽיהָ׃ (כח) יָבִ֨יא יְהוָ֜ה עָלֶ֗יךָ וְעַֽל־ עַמְּךָ֮ וְעַל־ בֵּ֣ית אָבִיךָ֒ יָמִים֙ אֲשֶׁ֣ר לֹא־ בָ֔אוּ לְמִיּ֥וֹם סוּר־ אֶפְרַ֖יִם מֵעַ֣ל יְהוּדָ֑ה אֵ֖ת מֶ֥לֶךְ אַשּֽׁוּר׃ (כט) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא יִשְׁרֹ֤ק יְהוָה֙ לַזְּב֔וּב אֲשֶׁ֥ר בִּקְצֵ֖ה יְאֹרֵ֣י מִצְרָ֑יִם וְלַ֨דְּבוֹרָ֔ה אֲשֶׁ֖ר בְּאֶ֥רֶץ אַשּֽׁוּר׃ (ל) וּבָ֨אוּ וְנָח֤וּ כֻלָּם֙ בְּנַחֲלֵ֣י הַבַּתּ֔וֹת וּבִנְקִיקֵ֖י הַסְּלָעִ֑ים וּבְכֹל֙ הַנַּ֣עֲצוּצִ֔ים וּבְכֹ֖ל הַנַּהֲלֹלִֽים׃ (לא) בַּיּ֣וֹם הַה֡וּא יְגַלַּ֣ח אֲדֹנָי֩ בְּתַ֨עַר הַשְּׂכִירָ֜ה בְּעֶבְרֵ֤י נָהָר֙ בְּמֶ֣לֶךְ אַשּׁ֔וּר אֶת־ הָרֹ֖אשׁ וְשַׂ֣עַר הָרַגְלָ֑יִם וְגַ֥ם אֶת־ הַזָּקָ֖ן תִּסְפֶּֽה׃ (לב) וְהָיָ֖ה בַּיּ֣וֹם הַה֑וּא יְחַיֶּה־ אִ֛ישׁ עֶגְלַ֥ת בָּקָ֖ר וּשְׁתֵּי־ צֹֽאן׃ (לג) וְהָיָ֗ה מֵרֹ֛ב עֲשׂ֥וֹת חָלָ֖ב יֹאכַ֣ל חֶמְאָ֑ה כִּֽי־ חֶמְאָ֤ה וּדְבַשׁ֙ יֹאכֵ֔ל כָּל־ הַנּוֹתָ֖ר בְּקֶ֥רֶב הָאָֽרֶץ׃ (לד) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא יִֽהְיֶ֣ה כָל־ מָק֗וֹם אֲשֶׁ֧ר יִֽהְיֶה־ שָּׁ֛ם אֶ֥לֶף גֶּ֖פֶן בְּאֶ֣לֶף כָּ֑סֶף לַשָּׁמִ֥יר וְלַשַּׁ֖יִת יִֽהְיֶֽה׃ (לה) בַּחִצִּ֥ים וּבַקֶּ֖שֶׁת יָ֣בוֹא שָׁ֑מָּה כִּי־ שָׁמִ֥יר וָשַׁ֖יִת תִּֽהְיֶ֥ה כָל־ הָאָֽרֶץ׃ (לו) וְכֹ֣ל הֶהָרִ֗ים אֲשֶׁ֤ר בַּמַּעְדֵּר֙ יֵעָ֣דֵר֔וּן לֹֽא־ תָב֣וֹא שָׁ֔מָּה יִרְאַ֖ת שָׁמִ֣יר וָשָׁ֑יִת וְהָיָה֙ לְמִשְׁלַ֣ח שׁ֔וֹר וּלְמִרְמַ֖ס שֶֽׂה׃ (לז) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י קַח־ לְךָ֖ גִּלָּי֣וֹן גָּד֑וֹל וּכְתֹ֤ב עָלָיו֙ בְּחֶ֣רֶט אֱנ֔וֹשׁ לְמַהֵ֥ר שָׁלָ֖ל חָ֥שׁ בַּֽז׃ (לח) וְאָעִ֣ידָה לִּ֔י עֵדִ֖ים נֶאֱמָנִ֑ים אֵ֚ת אוּרִיָּ֣ה הַכֹּהֵ֔ן וְאֶת־ זְכַרְיָ֖הוּ בֶּ֥ן יְבֶרֶכְיָֽהוּ׃ (לט) וָאֶקְרַב֙ אֶל־ הַנְּבִיאָ֔ה וַתַּ֖הַר וַתֵּ֣לֶד בֵּ֑ן וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י קְרָ֣א שְׁמ֔וֹ מַהֵ֥ר שָׁלָ֖ל חָ֥שׁ בַּֽז׃ (מ) כִּ֗י בְּטֶ֙רֶם֙ יֵדַ֣ע הַנַּ֔עַר קְרֹ֖א אָבִ֣י וְאִמִּ֑י יִשָּׂ֣א ׀ אֶת־ חֵ֣יל דַּמֶּ֗שֶׂק וְאֵת֙ שְׁלַ֣ל שֹׁמְר֔וֹן לִפְנֵ֖י מֶ֥לֶךְ אַשּֽׁוּר׃ (מא) וַיֹּ֣סֶף יְהוָ֔ה דַּבֵּ֥ר אֵלַ֛י ע֖וֹד לֵאמֹֽר׃ (מב) יַ֗עַן כִּ֤י מָאַס֙ הָעָ֣ם הַזֶּ֔ה אֵ֚ת מֵ֣י הַשִּׁלֹ֔חַ הַהֹלְכִ֖ים לְאַ֑ט וּמְשׂ֥וֹשׂ אֶת־ רְצִ֖ין וּבֶן־ רְמַלְיָֽהוּ׃ (מג) וְלָכֵ֡ן הִנֵּ֣ה אֲדֹנָי֩ מַעֲלֶ֨ה עֲלֵיהֶ֜ם אֶת־ מֵ֣י הַנָּהָ֗ר הָעֲצוּמִים֙ וְהָ֣רַבִּ֔ים אֶת־ מֶ֥לֶךְ אַשּׁ֖וּר וְאֶת־ כָּל־ כְּבוֹד֑וֹ וְעָלָה֙ עַל־ כָּל־ אֲפִיקָ֔יו וְהָלַ֖ךְ עַל־ כָּל־ גְּדוֹתָֽיו׃ (מד) וְחָלַ֤ף בִּֽיהוּדָה֙ שָׁטַ֣ף וְעָבַ֔ר עַד־ צַוָּ֖אר יַגִּ֑יעַ וְהָיָה֙ מֻטּ֣וֹת כְּנָפָ֔יו מְלֹ֥א רֹֽחַב־ אַרְצְךָ֖ עִמָּ֥נוּ אֵֽל׃ (מה) כֵ֣ן יְדַמֶּ֔ה וּלְבָב֖וֹ לֹא־ כֵ֣ן יַחְשֹׁ֑ב כִּ֚י לְהַשְׁמִ֣יד בִּלְבָב֔וֹ וּלְהַכְרִ֥ית גּוֹיִ֖ם לֹ֥א מְעָֽט׃ (כג) כִּ֖י יֹאמַ֑ר הֲלֹ֥א שָׂרַ֛י יַחְדָּ֖ו מְלָכִֽים׃ (כד) הֲלֹ֥א כְּכַרְכְּמִ֖ישׁ כַּלְנ֑וֹ אִם־ לֹ֤א כְאַרְפַּד֙ חֲמָ֔ת אִם־ לֹ֥א כְדַמֶּ֖שֶׂק שֹׁמְרֽוֹן׃ (כה) כַּאֲשֶׁר֙ מָצְאָ֣ה יָדִ֔י לְמַמְלְכֹ֖ת הָאֱלִ֑יל וּפְסִ֣ילֵיהֶ֔ם מִירֽוּשָׁלִַ֖ם וּמִשֹּׁמְרֽוֹן׃ (כו) הֲלֹ֗א כַּאֲשֶׁ֥ר עָשִׂ֛יתִי לְשֹׁמְר֖וֹן וְלֶאֱלִילֶ֑יהָ כֵּ֛ן אֶעֱשֶׂ֥ה לִירוּשָׁלִַ֖ם וְלַעֲצַבֶּֽיהָ׃ (כז) וְהָיָ֗ה כִּֽי־ יְבַצַּ֤ע אֲדֹנָי֙ אֶת־ כָּל־ מַֽעֲשֵׂ֔הוּ בְּהַ֥ר צִיּ֖וֹן וּבִירוּשָׁלִָ֑ם אֶפְקֹ֗ד עַל־ פְּרִי־ גֹ֙דֶל֙ לְבַ֣ב מֶֽלֶךְ־ אַשּׁ֔וּר וְעַל־ תִּפְאֶ֖רֶת ר֥וּם עֵינָֽיו׃ (כח) כִּ֣י אָמַ֗ר בְּכֹ֤חַ יָדִי֙ עָשִׂ֔יתִי וּבְחָכְמָתִ֖י כִּ֣י נְבֻנ֑וֹתִי וְאָסִ֣יר ׀ גְּבוּלֹ֣ת עַמִּ֗ים (ועתידתיהם) וַעֲתוּדֽוֹתֵיהֶם֙ שׁוֹשֵׂ֔תִי וְאוֹרִ֥יד כַּאבִּ֖יר יוֹשְׁבִֽים׃ (כט) וַתִּמְצָ֨א כַקֵּ֤ן ׀ יָדִי֙ לְחֵ֣יל הָֽעַמִּ֔ים וְכֶאֱסֹף֙ בֵּיצִ֣ים עֲזֻב֔וֹת כָּל־ הָאָ֖רֶץ אֲנִ֣י אָסָ֑פְתִּי וְלֹ֤א הָיָה֙ נֹדֵ֣ד כָּנָ֔ף וּפֹצֶ֥ה פֶ֖ה וּמְצַפְצֵֽף׃ (ל) הֲיִתְפָּאֵר֙ הַגַּרְזֶ֔ן עַ֖ל הַחֹצֵ֣ב בּ֑וֹ אִם־ יִתְגַּדֵּ֤ל הַמַּשּׂוֹר֙ עַל־ מְנִיפ֔וֹ כְּהָנִ֥יף שֵׁ֙בֶט֙ וְאֶת־ מְרִימָ֔יו כְּהָרִ֥ים מַטֶּ֖ה לֹא־ עֵֽץ׃ (לא) לָ֠כֵן יְשַׁלַּ֨ח הָאָד֜וֹן יְהוָ֧ה צְבָא֛וֹת בְּמִשְׁמַנָּ֖יו רָז֑וֹן וְתַ֧חַת כְּבֹד֛וֹ יֵקַ֥ד יְקֹ֖ד כִּיק֥וֹד אֵֽשׁ׃ (לב) וְהָיָ֤ה אֽוֹר־ יִשְׂרָאֵל֙ לְאֵ֔שׁ וּקְדוֹשׁ֖וֹ לְלֶהָבָ֑ה וּבָעֲרָ֗ה וְאָֽכְלָ֛ה שִׁית֥וֹ וּשְׁמִיר֖וֹ בְּי֥וֹם אֶחָֽד׃ (לג) וּכְב֤וֹד יַעְרוֹ֙ וְכַרְמִלּ֔וֹ מִנֶּ֥פֶשׁ וְעַד־ בָּשָׂ֖ר יְכַלֶּ֑ה וְהָיָ֖ה כִּמְסֹ֥ס נֹסֵֽס׃ (לד) וּשְׁאָ֥ר עֵ֛ץ יַעְר֖וֹ מִסְפָּ֣ר יִֽהְי֑וּ וְנַ֖עַר יִכְתְּבֵֽם׃ (לה) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא לֹֽא־ יוֹסִ֨יף ע֜וֹד שְׁאָ֤ר יִשְׂרָאֵל֙ וּפְלֵיטַ֣ת בֵּֽית־ יַעֲקֹ֔ב לְהִשָּׁעֵ֖ן עַל־ מַכֵּ֑הוּ וְנִשְׁעַ֗ן עַל־ יְהוָ֛ה קְד֥וֹשׁ יִשְׂרָאֵ֖ל בֶּאֱמֶֽת׃ (לו) שְׁאָ֥ר יָשׁ֖וּב שְׁאָ֣ר יַעֲקֹ֑ב אֶל־ אֵ֖ל גִּבּֽוֹר׃ (לז) כִּ֣י אִם־ יִהְיֶ֞ה עַמְּךָ֤ יִשְׂרָאֵל֙ כְּח֣וֹל הַיָּ֔ם שְׁאָ֖ר יָשׁ֣וּב בּ֑וֹ כִּלָּי֥וֹן חָר֖וּץ שׁוֹטֵ֥ף צְדָקָֽה׃ (לח) כִּ֥י כָלָ֖ה וְנֶחֱרָצָ֑ה אֲדֹנָ֤י יְהוִה֙ צְבָא֔וֹת עֹשֶׂ֖ה בְּקֶ֥רֶב כָּל־ הָאָֽרֶץ׃ (לט) לָכֵ֗ן כֹּֽה־ אָמַ֞ר אֲדֹנָ֤י יְהוִה֙ צְבָא֔וֹת אַל־ תִּירָ֥א עַמִּ֛י יֹשֵׁ֥ב צִיּ֖וֹן מֵֽאַשּׁ֑וּר בַּשֵּׁ֣בֶט יַכֶּ֔כָּה וּמַטֵּ֥הוּ יִשָּֽׂא־ עָלֶ֖יךָ בְּדֶ֥רֶךְ מִצְרָֽיִם׃ (מ) כִּי־ ע֖וֹד מְעַ֣ט מִזְעָ֑ר וְכָ֣לָה זַ֔עַם וְאַפִּ֖י עַל־ תַּבְלִיתָֽם׃ (מא) וְעוֹרֵ֨ר עָלָ֜יו יְהוָ֤ה צְבָאוֹת֙ שׁ֔וֹט כְּמַכַּ֥ת מִדְיָ֖ן בְּצ֣וּר עוֹרֵ֑ב וּמַטֵּ֙הוּ֙ עַל־ הַיָּ֔ם וּנְשָׂא֖וֹ בְּדֶ֥רֶךְ מִצְרָֽיִם׃ (מב) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא יָס֤וּר סֻבֳּלוֹ֙ מֵעַ֣ל שִׁכְמֶ֔ךָ וְעֻלּ֖וֹ מֵעַ֣ל צַוָּארֶ֑ךָ וְחֻבַּ֥ל עֹ֖ל מִפְּנֵי־ שָֽׁמֶן׃ (מג) בָּ֥א עַל־ עַיַּ֖ת עָבַ֣ר בְּמִגְר֑וֹן לְמִכְמָ֖שׂ יַפְקִ֥יד כֵּלָֽיו׃ (מד) עָֽבְרוּ֙ מַעְבָּרָ֔ה גֶּ֖בַע מָל֣וֹן לָ֑נוּ חָֽרְדָה֙ הָֽרָמָ֔ה גִּבְעַ֥ת שָׁא֖וּל נָֽסָה׃ (מה) צַהֲלִ֥י קוֹלֵ֖ךְ בַּת־ גַּלִּ֑ים הַקְשִׁ֥יבִי לַ֖יְשָׁה עֲנִיָּ֥ה עֲנָתֽוֹת׃ (מו) נָדְדָ֖ה מַדְמֵנָ֑ה יֹשְׁבֵ֥י הַגֵּבִ֖ים הֵעִֽיזוּ׃ (מז) ע֥וֹד הַיּ֖וֹם בְּנֹ֣ב לַֽעֲמֹ֑ד יְנֹפֵ֤ף יָדוֹ֙ הַ֣ר (בית)־ בַּת־ צִיּ֔וֹן גִּבְעַ֖ת יְרוּשָׁלִָֽם׃ (מח) הִנֵּ֤ה הָאָדוֹן֙ יְהוָ֣ה צְבָא֔וֹת מְסָעֵ֥ף פֻּארָ֖ה בְּמַעֲרָצָ֑ה וְרָמֵ֤י הַקּוֹמָה֙ גְּדוּעִ֔ים וְהַגְּבֹהִ֖ים יִשְׁפָּֽלוּ׃ (מט) וְנִקַּ֛ף סִֽבְכֵ֥י הַיַּ֖עַר בַּבַּרְזֶ֑ל וְהַלְּבָנ֖וֹן בְּאַדִּ֥יר יִפּֽוֹל׃ (נ) וְיָצָ֥א חֹ֖טֶר מִגֵּ֣זַע יִשָׁ֑י וְנֵ֖צֶר מִשָּׁרָשָׁ֥יו יִפְרֶֽה׃
