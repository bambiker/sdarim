ישעיהו פרק כז לפי החלוקה לסדרים

פרק כז בספר ישעיהו (13 פסוקים) נמצא בישעיהו סדר י (פסוקים לה–מה) ובישעיהו סדר יא (פסוקים א–ב).

ישעיהו סדר י

(א) וְחָֽפְרָה֙ הַלְּבָנָ֔ה וּבוֹשָׁ֖ה הַֽחַמָּ֑ה כִּֽי־ מָלַ֞ךְ יְהוָ֣ה צְבָא֗וֹת בְּהַ֤ר צִיּוֹן֙ וּבִיר֣וּשָׁלִַ֔ם וְנֶ֥גֶד זְקֵנָ֖יו כָּבֽוֹד׃ (ב) יְהוָ֤ה אֱלֹהַי֙ אַתָּ֔ה אֲרֽוֹמִמְךָ֙ אוֹדֶ֣ה שִׁמְךָ֔ כִּ֥י עָשִׂ֖יתָ פֶּ֑לֶא עֵצ֥וֹת מֵֽרָח֖וֹק אֱמ֥וּנָה אֹֽמֶן׃ (ג) כִּ֣י שַׂ֤מְתָּ מֵעִיר֙ לַגָּ֔ל קִרְיָ֥ה בְצוּרָ֖ה לְמַפֵּלָ֑ה אַרְמ֤וֹן זָרִים֙ מֵעִ֔יר לְעוֹלָ֖ם לֹ֥א יִבָּנֶֽה׃ (ד) עַל־ כֵּ֖ן יְכַבְּד֣וּךָ עַם־ עָ֑ז קִרְיַ֛ת גּוֹיִ֥ם עָרִיצִ֖ים יִירָאֽוּךָ׃ (ה) כִּֽי־ הָיִ֨יתָ מָע֥וֹז לַדָּ֛ל מָע֥וֹז לָאֶבְי֖וֹן בַּצַּר־ ל֑וֹ מַחְסֶ֤ה מִזֶּ֙רֶם֙ צֵ֣ל מֵחֹ֔רֶב כִּ֛י ר֥וּחַ עָרִיצִ֖ים כְּזֶ֥רֶם קִֽיר׃ (ו) כְּחֹ֣רֶב בְּצָי֔וֹן שְׁא֥וֹן זָרִ֖ים תַּכְנִ֑יעַ חֹ֚רֶב בְּצֵ֣ל עָ֔ב זְמִ֥יר עָֽרִיצִ֖ים יַעֲנֶֽה׃ (ז) וְעָשָׂה֩ יְהוָ֨ה צְבָא֜וֹת לְכָל־ הָֽעַמִּים֙ בָּהָ֣ר הַזֶּ֔ה מִשְׁתֵּ֥ה שְׁמָנִ֖ים מִשְׁתֵּ֣ה שְׁמָרִ֑ים שְׁמָנִים֙ מְמֻ֣חָיִ֔ם שְׁמָרִ֖ים מְזֻקָּקִֽים׃ (ח) וּבִלַּע֙ בָּהָ֣ר הַזֶּ֔ה פְּנֵֽי־ הַלּ֥וֹט ׀ הַלּ֖וֹט עַל־ כָּל־ הָֽעַמִּ֑ים וְהַמַּסֵּכָ֥ה הַנְּסוּכָ֖ה עַל־ כָּל־ הַגּוֹיִֽם׃ (ט) בִּלַּ֤ע הַמָּ֙וֶת֙ לָנֶ֔צַח וּמָחָ֨ה אֲדֹנָ֧י יְהוִ֛ה דִּמְעָ֖ה מֵעַ֣ל כָּל־ פָּנִ֑ים וְחֶרְפַּ֣ת עַמּ֗וֹ יָסִיר֙ מֵעַ֣ל כָּל־ הָאָ֔רֶץ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (י) וְאָמַר֙ בַּיּ֣וֹם הַה֔וּא הִנֵּ֨ה אֱלֹהֵ֥ינוּ זֶ֛ה קִוִּ֥ינוּ ל֖וֹ וְיֽוֹשִׁיעֵ֑נוּ זֶ֤ה יְהוָה֙ קִוִּ֣ינוּ ל֔וֹ נָגִ֥ילָה וְנִשְׂמְחָ֖ה בִּישׁוּעָתֽוֹ׃ (יא) כִּֽי־ תָנ֥וּחַ יַד־ יְהוָ֖ה בָּהָ֣ר הַזֶּ֑ה וְנָ֤דוֹשׁ מוֹאָב֙ תַּחְתָּ֔יו כְּהִדּ֥וּשׁ מַתְבֵּ֖ן (במי) בְּמ֥וֹ מַדְמֵנָֽה׃ (יב) וּפֵרַ֤שׂ יָדָיו֙ בְּקִרְבּ֔וֹ כַּאֲשֶׁ֛ר יְפָרֵ֥שׂ הַשֹּׂחֶ֖ה לִשְׂח֑וֹת וְהִשְׁפִּיל֙ גַּֽאֲוָת֔וֹ עִ֖ם אָרְבּ֥וֹת יָדָֽיו׃ (יג) וּמִבְצַ֞ר מִשְׂגַּ֣ב חוֹמֹתֶ֗יךָ הֵשַׁ֥ח הִשְׁפִּ֛יל הִגִּ֥יעַ לָאָ֖רֶץ עַד־ עָפָֽר׃ (יד) בַּיּ֣וֹם הַה֔וּא יוּשַׁ֥ר הַשִּׁיר־ הַזֶּ֖ה בְּאֶ֣רֶץ יְהוּדָ֑ה עִ֣יר עָז־ לָ֔נוּ יְשׁוּעָ֥ה יָשִׁ֖ית חוֹמ֥וֹת וָחֵֽל׃ (טו) פִּתְח֖וּ שְׁעָרִ֑ים וְיָבֹ֥א גוֹי־ צַדִּ֖יק שֹׁמֵ֥ר אֱמֻנִֽים׃ (טז) יֵ֣צֶר סָמ֔וּךְ תִּצֹּ֖ר שָׁל֣וֹם ׀ שָׁל֑וֹם כִּ֥י בְךָ֖ בָּטֽוּחַ׃ (יז) בִּטְח֥וּ בַֽיהוָ֖ה עֲדֵי־ עַ֑ד כִּ֚י בְּיָ֣הּ יְהוָ֔ה צ֖וּר עוֹלָמִֽים׃ (יח) כִּ֤י הֵשַׁח֙ יֹשְׁבֵ֣י מָר֔וֹם קִרְיָ֖ה נִשְׂגָּבָ֑ה יַשְׁפִּילֶ֤נָּה יַשְׁפִּילָהּ֙ עַד־ אֶ֔רֶץ יַגִּיעֶ֖נָּה עַד־ עָפָֽר׃ (יט) תִּרְמְסֶ֖נָּה רָ֑גֶל רַגְלֵ֥י עָנִ֖י פַּעֲמֵ֥י דַלִּֽים׃ (כ) אֹ֥רַח לַצַּדִּ֖יק מֵֽישָׁרִ֑ים יָשָׁ֕ר מַעְגַּ֥ל צַדִּ֖יק תְּפַלֵּֽס׃ (כא) אַ֣ף אֹ֧רַח מִשְׁפָּטֶ֛יךָ יְהוָ֖ה קִוִּינ֑וּךָ לְשִׁמְךָ֥ וּֽלְזִכְרְךָ֖ תַּאֲוַת־ נָֽפֶשׁ׃ (כב) נַפְשִׁ֤י אִוִּיתִ֙יךָ֙ בַּלַּ֔יְלָה אַף־ רוּחִ֥י בְקִרְבִּ֖י אֲשַֽׁחֲרֶ֑ךָּ כִּ֞י כַּאֲשֶׁ֤ר מִשְׁפָּטֶ֙יךָ֙ לָאָ֔רֶץ צֶ֥דֶק לָמְד֖וּ יֹשְׁבֵ֥י תֵבֵֽל׃ (כג) יֻחַ֤ן רָשָׁע֙ בַּל־ לָמַ֣ד צֶ֔דֶק בְּאֶ֥רֶץ נְכֹח֖וֹת יְעַוֵּ֑ל וּבַל־ יִרְאֶ֖ה גֵּא֥וּת יְהוָֽה׃ (כד) יְהוָ֛ה רָ֥מָה יָדְךָ֖ בַּל־ יֶחֱזָי֑וּן יֶחֱז֤וּ וְיֵבֹ֙שׁוּ֙ קִנְאַת־ עָ֔ם אַף־ אֵ֖שׁ צָרֶ֥יךָ תֹאכְלֵֽם׃ (כה) יְהוָ֕ה תִּשְׁפֹּ֥ת שָׁל֖וֹם לָ֑נוּ כִּ֛י גַּ֥ם כָּֽל־ מַעֲשֵׂ֖ינוּ פָּעַ֥לְתָּ לָּֽנוּ׃ (כו) יְהוָ֣ה אֱלֹהֵ֔ינוּ בְּעָל֥וּנוּ אֲדֹנִ֖ים זֽוּלָתֶ֑ךָ לְבַד־ בְּךָ֖ נַזְכִּ֥יר שְׁמֶֽךָ׃ (כז) מֵתִים֙ בַּל־ יִחְי֔וּ רְפָאִ֖ים בַּל־ יָקֻ֑מוּ לָכֵ֤ן פָּקַ֙דְתָּ֙ וַתַּשְׁמִידֵ֔ם וַתְּאַבֵּ֥ד כָּל־ זֵ֖כֶר לָֽמוֹ׃ (כח) יָסַ֤פְתָּ לַגּוֹי֙ יְהוָ֔ה יָסַ֥פְתָּ לַגּ֖וֹי נִכְבָּ֑דְתָּ רִחַ֖קְתָּ כָּל־ קַצְוֵי־ אָֽרֶץ׃ (כט) יְהוָ֖ה בַּצַּ֣ר פְּקָד֑וּךָ צָק֣וּן לַ֔חַשׁ מוּסָרְךָ֖ לָֽמוֹ׃ (ל) כְּמ֤וֹ הָרָה֙ תַּקְרִ֣יב לָלֶ֔דֶת תָּחִ֥יל תִּזְעַ֖ק בַּחֲבָלֶ֑יהָ כֵּ֛ן הָיִ֥ינוּ מִפָּנֶ֖יךָ יְהוָֽה׃ (לא) הָרִ֣ינוּ חַ֔לְנוּ כְּמ֖וֹ יָלַ֣דְנוּ ר֑וּחַ יְשׁוּעֹת֙ בַּל־ נַ֣עֲשֶׂה אֶ֔רֶץ וּבַֽל־ יִפְּל֖וּ יֹשְׁבֵ֥י תֵבֵֽל׃ (לב) יִֽחְי֣וּ מֵתֶ֔יךָ נְבֵלָתִ֖י יְקוּמ֑וּן הָקִ֨יצוּ וְרַנְּנ֜וּ שֹׁכְנֵ֣י עָפָ֗ר כִּ֣י טַ֤ל אוֹרֹת֙ טַלֶּ֔ךָ וָאָ֖רֶץ רְפָאִ֥ים תַּפִּֽיל׃ (לג) לֵ֤ךְ עַמִּי֙ בֹּ֣א בַחֲדָרֶ֔יךָ וּֽסְגֹ֥ר (דלתיך) דְּלָתְךָ֖ בַּעֲדֶ֑ךָ חֲבִ֥י כִמְעַט־ רֶ֖גַע עַד־ (יעבור)־ יַעֲבָר־ זָֽעַם׃ (לד) כִּֽי־ הִנֵּ֤ה יְהוָה֙ יֹצֵ֣א מִמְּקוֹמ֔וֹ לִפְקֹ֛ד עֲוֺ֥ן יֹֽשֵׁב־ הָאָ֖רֶץ עָלָ֑יו וְגִלְּתָ֤ה הָאָ֙רֶץ֙ אֶת־ דָּמֶ֔יהָ וְלֹֽא־ תְכַסֶּ֥ה ע֖וֹד עַל־ הֲרוּגֶֽיהָ׃ תחילת פרק כז (לה) בַּיּ֣וֹם הַה֡וּא יִפְקֹ֣ד יְהוָה֩ בְּחַרְב֨וֹ הַקָּשָׁ֜ה וְהַגְּדוֹלָ֣ה וְהַֽחֲזָקָ֗ה עַ֤ל לִוְיָתָן֙ נָחָ֣שׁ בָּרִ֔חַ וְעַל֙ לִוְיָתָ֔ן נָחָ֖שׁ עֲקַלָּת֑וֹן וְהָרַ֥ג אֶת־ הַתַּנִּ֖ין אֲשֶׁ֥ר בַּיָּֽם׃ (לו) בַּיּ֖וֹם הַה֑וּא כֶּ֥רֶם חֶ֖מֶד עַנּוּ־ לָֽהּ׃ (לז) אֲנִ֤י יְהוָה֙ נֹֽצְרָ֔הּ לִרְגָעִ֖ים אַשְׁקֶ֑נָּה פֶּ֚ן יִפְקֹ֣ד עָלֶ֔יהָ לַ֥יְלָה וָי֖וֹם אֶצֳּרֶֽנָּה׃ (לח) חֵמָ֖ה אֵ֣ין לִ֑י מִֽי־ יִתְּנֵ֜נִי שָׁמִ֥יר שַׁ֙יִת֙ בַּמִּלְחָמָ֔ה אֶפְשְׂעָ֥ה בָ֖הּ אֲצִיתֶ֥נָּה יָּֽחַד׃ (לט) א֚וֹ יַחֲזֵ֣ק בְּמָעוּזִּ֔י יַעֲשֶׂ֥ה שָׁל֖וֹם לִ֑י שָׁל֖וֹם יַֽעֲשֶׂה־ לִּֽי׃ (מ) הַבָּאִים֙ יַשְׁרֵ֣שׁ יַֽעֲקֹ֔ב יָצִ֥יץ וּפָרַ֖ח יִשְׂרָאֵ֑ל וּמָלְא֥וּ פְנֵי־ תֵבֵ֖ל תְּנוּבָֽה׃ (מא) הַכְּמַכַּ֥ת מַכֵּ֖הוּ הִכָּ֑הוּ אִם־ כְּהֶ֥רֶג הֲרֻגָ֖יו הֹרָֽג׃ (מב) בְּסַּאסְּאָ֖ה בְּשַׁלְחָ֣הּ תְּרִיבֶ֑נָּה הָגָ֛ה בְּרוּח֥וֹ הַקָּשָׁ֖ה בְּי֥וֹם קָדִֽים׃ (מג) לָכֵ֗ן בְּזֹאת֙ יְכֻפַּ֣ר עֲוֺֽן־ יַעֲקֹ֔ב וְזֶ֕ה כָּל־ פְּרִ֖י הָסִ֣ר חַטָּאת֑וֹ בְּשׂוּמ֣וֹ ׀ כָּל־ אַבְנֵ֣י מִזְבֵּ֗חַ כְּאַבְנֵי־ גִר֙ מְנֻפָּצ֔וֹת לֹֽא־ יָקֻ֥מוּ אֲשֵׁרִ֖ים וְחַמָּנִֽים׃ (מד) כִּ֣י עִ֤יר בְּצוּרָה֙ בָּדָ֔ד נָוֶ֕ה מְשֻׁלָּ֥ח וְנֶעֱזָ֖ב כַּמִּדְבָּ֑ר שָׁ֣ם יִרְעֶ֥ה עֵ֛גֶל וְשָׁ֥ם יִרְבָּ֖ץ וְכִלָּ֥ה סְעִפֶֽיהָ׃ (מה) בִּיבֹ֤שׁ קְצִירָהּ֙ תִּשָּׁבַ֔רְנָה נָשִׁ֕ים בָּא֖וֹת מְאִיר֣וֹת אוֹתָ֑הּ כִּ֣י לֹ֤א עַם־ בִּינוֹת֙ ה֔וּא עַל־ כֵּן֙ לֹֽא־ יְרַחֲמֶ֣נּוּ עֹשֵׂ֔הוּ וְיֹצְר֖וֹ לֹ֥א יְחֻנֶּֽנּוּ׃

ישעיהו סדר יא

(א) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא יַחְבֹּ֧ט יְהוָ֛ה מִשִּׁבֹּ֥לֶת הַנָּהָ֖ר עַד־ נַ֣חַל מִצְרָ֑יִם וְאַתֶּ֧ם תְּלֻקְּט֛וּ לְאַחַ֥ד אֶחָ֖ד בְּנֵ֥י יִשְׂרָאֵֽל׃ (ב) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא יִתָּקַע֮ בְּשׁוֹפָ֣ר גָּדוֹל֒ וּבָ֗אוּ הָאֹֽבְדִים֙ בְּאֶ֣רֶץ אַשּׁ֔וּר וְהַנִּדָּחִ֖ים בְּאֶ֣רֶץ מִצְרָ֑יִם וְהִשְׁתַּחֲו֧וּ לַיהוָ֛ה בְּהַ֥ר הַקֹּ֖דֶשׁ בִּירוּשָׁלִָֽם׃ סוף פרק כז (ג) ה֗וֹי עֲטֶ֤רֶת גֵּאוּת֙ שִׁכֹּרֵ֣י אֶפְרַ֔יִם וְצִ֥יץ נֹבֵ֖ל צְבִ֣י תִפְאַרְתּ֑וֹ אֲשֶׁ֛ר עַל־ רֹ֥אשׁ גֵּֽיא־ שְׁמָנִ֖ים הֲל֥וּמֵי יָֽיִן׃ (ד) הִנֵּ֨ה חָזָ֤ק וְאַמִּץ֙ לַֽאדֹנָ֔י כְּזֶ֥רֶם בָּרָ֖ד שַׂ֣עַר קָ֑טֶב כְּ֠זֶרֶם מַ֣יִם כַּבִּירִ֥ים שֹׁטְפִ֛ים הִנִּ֥יחַ לָאָ֖רֶץ בְּיָֽד׃ (ה) בְּרַגְלַ֖יִם תֵּֽרָמַ֑סְנָה עֲטֶ֥רֶת גֵּא֖וּת שִׁכּוֹרֵ֥י אֶפְרָֽיִם׃ (ו) וְֽהָ֨יְתָ֜ה צִיצַ֤ת נֹבֵל֙ צְבִ֣י תִפְאַרְתּ֔וֹ אֲשֶׁ֥ר עַל־ רֹ֖אשׁ גֵּ֣יא שְׁמָנִ֑ים כְּבִכּוּרָהּ֙ בְּטֶ֣רֶם קַ֔יִץ אֲשֶׁ֨ר יִרְאֶ֤ה הָֽרֹאֶה֙ אוֹתָ֔הּ בְּעוֹדָ֥הּ בְּכַפּ֖וֹ יִבְלָעֶֽנָּה׃ (ז) בַּיּ֣וֹם הַה֗וּא יִֽהְיֶה֙ יְהוָ֣ה צְבָא֔וֹת לַעֲטֶ֣רֶת צְבִ֔י וְלִצְפִירַ֖ת תִּפְאָרָ֑ה לִשְׁאָ֖ר עַמּֽוֹ׃ (ח) וּלְר֖וּחַ מִשְׁפָּ֑ט לַיּוֹשֵׁב֙ עַל־ הַמִּשְׁפָּ֔ט וְלִ֨גְבוּרָ֔ה מְשִׁיבֵ֥י מִלְחָמָ֖ה שָֽׁעְרָה׃ (ט) וְגַם־ אֵ֙לֶּה֙ בַּיַּ֣יִן שָׁג֔וּ וּבַשֵּׁכָ֖ר תָּע֑וּ כֹּהֵ֣ן וְנָבִיא֩ שָׁג֨וּ בַשֵּׁכָ֜ר נִבְלְע֣וּ מִן־ הַיַּ֗יִן תָּעוּ֙ מִן־ הַשֵּׁכָ֔ר שָׁגוּ֙ בָּֽרֹאֶ֔ה פָּק֖וּ פְּלִילִיָּֽה׃ (י) כִּ֚י כָּל־ שֻׁלְחָנ֔וֹת מָלְא֖וּ קִ֣יא צֹאָ֑ה בְּלִ֖י מָקֽוֹם׃ (יא) אֶת־ מִי֙ יוֹרֶ֣ה דֵעָ֔ה וְאֶת־ מִ֖י יָבִ֣ין שְׁמוּעָ֑ה גְּמוּלֵי֙ מֵֽחָלָ֔ב עַתִּיקֵ֖י מִשָּׁדָֽיִם׃ (יב) כִּ֣י צַ֤ו לָצָו֙ צַ֣ו לָצָ֔ו קַ֥ו לָקָ֖ו קַ֣ו לָקָ֑ו זְעֵ֥יר שָׁ֖ם זְעֵ֥יר שָֽׁם׃ (יג) כִּ֚י בְּלַעֲגֵ֣י שָׂפָ֔ה וּבְלָשׁ֖וֹן אַחֶ֑רֶת יְדַבֵּ֖ר אֶל־ הָעָ֥ם הַזֶּֽה׃ (יד) אֲשֶׁ֣ר ׀ אָמַ֣ר אֲלֵיהֶ֗ם זֹ֤את הַמְּנוּחָה֙ הָנִ֣יחוּ לֶֽעָיֵ֔ף וְזֹ֖את הַמַּרְגֵּעָ֑ה וְלֹ֥א אָב֖וּא שְׁמֽוֹעַ׃ (טו) וְהָיָ֨ה לָהֶ֜ם דְּבַר־ יְהוָ֗ה צַ֣ו לָצָ֞ו צַ֤ו לָצָו֙ קַ֤ו לָקָו֙ קַ֣ו לָקָ֔ו זְעֵ֥יר שָׁ֖ם זְעֵ֣יר שָׁ֑ם לְמַ֨עַן יֵלְכ֜וּ וְכָשְׁל֤וּ אָחוֹר֙ וְנִשְׁבָּ֔רוּ וְנוֹקְשׁ֖וּ וְנִלְכָּֽדוּ׃ (טז) לָכֵ֛ן שִׁמְע֥וּ דְבַר־ יְהוָ֖ה אַנְשֵׁ֣י לָצ֑וֹן מֹֽשְׁלֵי֙ הָעָ֣ם הַזֶּ֔ה אֲשֶׁ֖ר בִּירוּשָׁלִָֽם׃ (יז) כִּ֣י אֲמַרְתֶּ֗ם כָּרַ֤תְנֽוּ בְרִית֙ אֶת־ מָ֔וֶת וְעִם־ שְׁא֖וֹל עָשִׂ֣ינוּ חֹזֶ֑ה (שיט) שׁ֣וֹט שׁוֹטֵ֤ף כִּֽי־ (עבר) יַֽעֲבֹר֙ לֹ֣א יְבוֹאֵ֔נוּ כִּ֣י שַׂ֧מְנוּ כָזָ֛ב מַחְסֵ֖נוּ וּבַשֶּׁ֥קֶר נִסְתָּֽרְנוּ׃ (יח) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֛י יִסַּ֥ד בְּצִיּ֖וֹן אָ֑בֶן אֶ֣בֶן בֹּ֜חַן פִּנַּ֤ת יִקְרַת֙ מוּסָ֣ד מוּסָּ֔ד הַֽמַּאֲמִ֖ין לֹ֥א יָחִֽישׁ׃ (יט) וְשַׂמְתִּ֤י מִשְׁפָּט֙ לְקָ֔ו וּצְדָקָ֖ה לְמִשְׁקָ֑לֶת וְיָעָ֤ה בָרָד֙ מַחְסֵ֣ה כָזָ֔ב וְסֵ֥תֶר מַ֖יִם יִשְׁטֹֽפוּ׃ (כ) וְכֻפַּ֤ר בְּרִֽיתְכֶם֙ אֶת־ מָ֔וֶת וְחָזוּתְכֶ֥ם אֶת־ שְׁא֖וֹל לֹ֣א תָק֑וּם שׁ֤וֹט שׁוֹטֵף֙ כִּ֣י יַֽעֲבֹ֔ר וִהְיִ֥יתֶם ל֖וֹ לְמִרְמָֽס׃ (כא) מִדֵּ֤י עָבְרוֹ֙ יִקַּ֣ח אֶתְכֶ֔ם כִּֽי־ בַבֹּ֧קֶר בַּבֹּ֛קֶר יַעֲבֹ֖ר בַּיּ֣וֹם וּבַלָּ֑יְלָה וְהָיָ֥ה רַק־ זְוָעָ֖ה הָבִ֥ין שְׁמוּעָֽה׃ (כב) כִּֽי־ קָצַ֥ר הַמַּצָּ֖ע מֵֽהִשְׂתָּרֵ֑עַ וְהַמַּסֵּכָ֥ה צָ֖רָה כְּהִתְכַּנֵּֽס׃ (כג) כִּ֤י כְהַר־ פְּרָצִים֙ יָק֣וּם יְהוָ֔ה כְּעֵ֖מֶק בְּגִבְע֣וֹן יִרְגָּ֑ז לַעֲשׂ֤וֹת מַעֲשֵׂ֙הוּ֙ זָ֣ר מַעֲשֵׂ֔הוּ וְלַֽעֲבֹד֙ עֲבֹ֣דָת֔וֹ נָכְרִיָּ֖ה עֲבֹדָתֽוֹ׃ (כד) וְעַתָּה֙ אַל־ תִּתְלוֹצָ֔צוּ פֶּֽן־ יֶחְזְק֖וּ מֽוֹסְרֵיכֶ֑ם כִּֽי־ כָלָ֨ה וְנֶחֱרָצָ֜ה שָׁמַ֗עְתִּי מֵאֵ֨ת אֲדֹנָ֧י יְהוִ֛ה צְבָא֖וֹת עַל־ כָּל־ הָאָֽרֶץ׃ (כה) הַאֲזִ֥ינוּ וְשִׁמְע֖וּ קוֹלִ֑י הַקְשִׁ֥יבוּ וְשִׁמְע֖וּ אִמְרָתִֽי׃ (כו) הֲכֹ֣ל הַיּ֔וֹם יַחֲרֹ֥שׁ הַחֹרֵ֖שׁ לִזְרֹ֑עַ יְפַתַּ֥ח וִֽישַׂדֵּ֖ד אַדְמָתֽוֹ׃ (כז) הֲלוֹא֙ אִם־ שִׁוָּ֣ה פָנֶ֔יהָ וְהֵפִ֥יץ קֶ֖צַח וְכַמֹּ֣ן יִזְרֹ֑ק וְשָׂ֨ם חִטָּ֤ה שׂוֹרָה֙ וּשְׂעֹרָ֣ה נִסְמָ֔ן וְכֻסֶּ֖מֶת גְּבֻלָתֽוֹ׃ (כח) וְיִסְּר֥וֹ לַמִּשְׁפָּ֖ט אֱלֹהָ֥יו יוֹרֶֽנּוּ׃ (כט) כִּ֣י לֹ֤א בֶֽחָרוּץ֙ י֣וּדַשׁ קֶ֔צַח וְאוֹפַ֣ן עֲגָלָ֔ה עַל־ כַּמֹּ֖ן יוּסָּ֑ב כִּ֧י בַמַּטֶּ֛ה יֵחָ֥בֶט קֶ֖צַח וְכַמֹּ֥ן בַּשָּֽׁבֶט׃ (ל) לֶ֣חֶם יוּדָ֔ק כִּ֛י לֹ֥א לָנֶ֖צַח אָד֣וֹשׁ יְדוּשֶׁ֑נּוּ וְ֠הָמַם גִּלְגַּ֧ל עֶגְלָת֛וֹ וּפָרָשָׁ֖יו לֹֽא־ יְדֻקֶּֽנּוּ׃ (לא) גַּם־ זֹ֕את מֵעִ֛ם יְהוָ֥ה צְבָא֖וֹת יָצָ֑אָה הִפְלִ֣יא עֵצָ֔ה הִגְדִּ֖יל תּוּשִׁיָּֽה׃ (לב) ה֚וֹי אֲרִיאֵ֣ל אֲרִיאֵ֔ל קִרְיַ֖ת חָנָ֣ה דָוִ֑ד סְפ֥וּ שָׁנָ֛ה עַל־ שָׁנָ֖ה חַגִּ֥ים יִנְקֹֽפוּ׃ (לג) וַהֲצִיק֖וֹתִי לַֽאֲרִיאֵ֑ל וְהָיְתָ֤ה תַֽאֲנִיָּה֙ וַֽאֲנִיָּ֔ה וְהָ֥יְתָה לִּ֖י כַּאֲרִיאֵֽל׃ (לד) וְחָנִ֥יתִי כַדּ֖וּר עָלָ֑יִךְ וְצַרְתִּ֤י עָלַ֙יִךְ֙ מֻצָּ֔ב וַהֲקִֽימֹתִ֥י עָלַ֖יִךְ מְצֻרֹֽת׃ (לה) וְשָׁפַלְתְּ֙ מֵאֶ֣רֶץ תְּדַבֵּ֔רִי וּמֵֽעָפָ֖ר תִּשַּׁ֣ח אִמְרָתֵ֑ךְ וְֽ֠הָיָה כְּא֤וֹב מֵאֶ֙רֶץ֙ קוֹלֵ֔ךְ וּמֵעָפָ֖ר אִמְרָתֵ֥ךְ תְּצַפְצֵֽף׃ (לו) וְהָיָ֛ה כְּאָבָ֥ק דַּ֖ק הֲמ֣וֹן זָרָ֑יִךְ וּכְמֹ֤ץ עֹבֵר֙ הֲמ֣וֹן עָֽרִיצִ֔ים וְהָיָ֖ה לְפֶ֥תַע פִּתְאֹֽם׃ (לז) מֵעִ֨ם יְהוָ֤ה צְבָאוֹת֙ תִּפָּקֵ֔ד בְּרַ֥עַם וּבְרַ֖עַשׁ וְק֣וֹל גָּד֑וֹל סוּפָה֙ וּסְעָרָ֔ה וְלַ֖הַב אֵ֥שׁ אוֹכֵלָֽה׃ (לח) וְהָיָ֗ה כַּֽחֲלוֹם֙ חֲז֣וֹן לַ֔יְלָה הֲמוֹן֙ כָּל־ הַגּוֹיִ֔ם הַצֹּבְאִ֖ים עַל־ אֲרִיאֵ֑ל וְכָל־ צֹבֶ֙יהָ֙ וּמְצֹ֣דָתָ֔הּ וְהַמְּצִיקִ֖ים לָֽהּ׃ (לט) וְהָיָ֡ה כַּאֲשֶׁר֩ יַחֲלֹ֨ם הָרָעֵ֜ב וְהִנֵּ֣ה אוֹכֵ֗ל וְהֵקִיץ֮ וְרֵיקָ֣ה נַפְשׁוֹ֒ וְכַאֲשֶׁ֨ר יַחֲלֹ֤ם הַצָּמֵא֙ וְהִנֵּ֣ה שֹׁתֶ֔ה וְהֵקִיץ֙ וְהִנֵּ֣ה עָיֵ֔ף וְנַפְשׁ֖וֹ שׁוֹקֵקָ֑ה כֵּ֣ן יִֽהְיֶ֗ה הֲמוֹן֙ כָּל־ הַגּוֹיִ֔ם הַצֹּבְאִ֖ים עַל־ הַ֥ר צִיּֽוֹן׃ (מ) הִתְמַהְמְה֣וּ וּתְמָ֔הוּ הִשְׁתַּֽעַשְׁע֖וּ וָשֹׁ֑עוּ שָֽׁכְר֣וּ וְלֹא־ יַ֔יִן נָע֖וּ וְלֹ֥א שֵׁכָֽר׃ (מא) כִּֽי־ נָסַ֨ךְ עֲלֵיכֶ֤ם יְהוָה֙ ר֣וּחַ תַּרְדֵּמָ֔ה וַיְעַצֵּ֖ם אֶת־ עֵֽינֵיכֶ֑ם אֶת־ הַנְּבִיאִ֛ים וְאֶת־ רָאשֵׁיכֶ֥ם הַחֹזִ֖ים כִּסָּֽה׃ (מב) וַתְּהִ֨י לָכֶ֜ם חָז֣וּת הַכֹּ֗ל כְּדִבְרֵי֮ הַסֵּ֣פֶר הֶֽחָתוּם֒ אֲשֶֽׁר־ יִתְּנ֣וּ אֹת֗וֹ אֶל־ יוֹדֵ֥עַ (הספר) סֵ֛פֶר לֵאמֹ֖ר קְרָ֣א נָא־ זֶ֑ה וְאָמַר֙ לֹ֣א אוּכַ֔ל כִּ֥י חָת֖וּם הֽוּא׃ (מג) וְנִתַּ֣ן הַסֵּ֗פֶר עַל֩ אֲשֶׁ֨ר לֹֽא־ יָדַ֥ע סֵ֛פֶר לֵאמֹ֖ר קְרָ֣א נָא־ זֶ֑ה וְאָמַ֕ר לֹ֥א יָדַ֖עְתִּי סֵֽפֶר׃ (מד) וַיֹּ֣אמֶר אֲדֹנָ֗י יַ֚עַן כִּ֤י נִגַּשׁ֙ הָעָ֣ם הַזֶּ֔ה בְּפִ֤יו וּבִשְׂפָתָיו֙ כִּבְּד֔וּנִי וְלִבּ֖וֹ רִחַ֣ק מִמֶּ֑נִּי וַתְּהִ֤י יִרְאָתָם֙ אֹתִ֔י מִצְוַ֥ת אֲנָשִׁ֖ים מְלֻמָּדָֽה׃ (מה) לָכֵ֗ן הִנְנִ֥י יוֹסִ֛ף לְהַפְלִ֥יא אֶת־ הָֽעָם־ הַזֶּ֖ה הַפְלֵ֣א וָפֶ֑לֶא וְאָֽבְדָה֙ חָכְמַ֣ת חֲכָמָ֔יו וּבִינַ֥ת נְבֹנָ֖יו תִּסְתַּתָּֽר׃ (מו) ה֛וֹי הַמַּעֲמִיקִ֥ים מֵֽיהוָ֖ה לַסְתִּ֣ר עֵצָ֑ה וְהָיָ֤ה בְמַחְשָׁךְ֙ מַֽעֲשֵׂיהֶ֔ם וַיֹּ֣אמְר֔וּ מִ֥י רֹאֵ֖נוּ וּמִ֥י יוֹדְעֵֽנוּ׃ (מז) הַ֨פְכְּכֶ֔ם אִם־ כְּחֹ֥מֶר הַיֹּצֵ֖ר יֵֽחָשֵׁ֑ב כִּֽי־ יֹאמַ֨ר מַעֲשֶׂ֤ה לְעֹשֵׂ֙הוּ֙ לֹ֣א עָשָׂ֔נִי וְיֵ֛צֶר אָמַ֥ר לְיוֹצְר֖וֹ לֹ֥א הֵבִֽין׃ (מח) הֲלוֹא־ עוֹד֙ מְעַ֣ט מִזְעָ֔ר וְשָׁ֥ב לְבָנ֖וֹן לַכַּרְמֶ֑ל וְהַכַּרְמֶ֖ל לַיַּ֥עַר יֵחָשֵֽׁב׃ (מט) וְשָׁמְע֧וּ בַיּוֹם־ הַה֛וּא הַחֵרְשִׁ֖ים דִּבְרֵי־ סֵ֑פֶר וּמֵאֹ֣פֶל וּמֵחֹ֔שֶׁךְ עֵינֵ֥י עִוְרִ֖ים תִּרְאֶֽינָה׃ (נ) וְיָסְפ֧וּ עֲנָוִ֛ים בַּֽיהוָ֖ה שִׂמְחָ֑ה וְאֶבְיוֹנֵ֣י אָדָ֔ם בִּקְד֥וֹשׁ יִשְׂרָאֵ֖ל יָגִֽילוּ׃ (נא) כִּֽי־ אָפֵ֥ס עָרִ֖יץ וְכָ֣לָה לֵ֑ץ וְנִכְרְת֖וּ כָּל־ שֹׁ֥קְדֵי אָֽוֶן׃ (נב) מַחֲטִיאֵ֤י אָדָם֙ בְּדָבָ֔ר וְלַמּוֹכִ֥יחַ בַּשַּׁ֖עַר יְקֹשׁ֑וּן וַיַּטּ֥וּ בַתֹּ֖הוּ צַדִּֽיק׃ (נג) לָכֵ֗ן כֹּֽה־ אָמַ֤ר יְהוָה֙ אֶל־ בֵּ֣ית יַֽעֲקֹ֔ב אֲשֶׁ֥ר פָּדָ֖ה אֶת־ אַבְרָהָ֑ם לֹֽא־ עַתָּ֤ה יֵבוֹשׁ֙ יַֽעֲקֹ֔ב וְלֹ֥א עַתָּ֖ה פָּנָ֥יו יֶחֱוָֽרוּ׃
