ישעיהו פרק כד לפי החלוקה לסדרים

פרק כד בספר ישעיהו (23 פסוקים) נמצא בישעיהו סדר ט (פסוקים כב–מג) ובישעיהו סדר י (פסוקים א–א).

ישעיהו סדר ט

(א) וּתְקַעְתִּ֥יו יָתֵ֖ד בְּמָק֣וֹם נֶאֱמָ֑ן וְהָיָ֛ה לְכִסֵּ֥א כָב֖וֹד לְבֵ֥ית אָבִֽיו׃ (ב) וְתָל֨וּ עָלָ֜יו כֹּ֣ל ׀ כְּב֣וֹד בֵּית־ אָבִ֗יו הַצֶּֽאֱצָאִים֙ וְהַצְּפִע֔וֹת כֹּ֖ל כְּלֵ֣י הַקָּטָ֑ן מִכְּלֵי֙ הָֽאַגָּנ֔וֹת וְעַ֖ד כָּל־ כְּלֵ֥י הַנְּבָלִֽים׃ (ג) בַּיּ֣וֹם הַה֗וּא נְאֻם֙ יְהוָ֣ה צְבָא֔וֹת תָּמוּשׁ֙ הַיָּתֵ֔ד הַתְּקוּעָ֖ה בְּמָק֣וֹם נֶאֱמָ֑ן וְנִגְדְּעָ֣ה וְנָפְלָ֗ה וְנִכְרַת֙ הַמַּשָּׂ֣א אֲשֶׁר־ עָלֶ֔יהָ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (ד) מַשָּׂ֖א צֹ֑ר הֵילִ֣ילוּ ׀ אֳנִיּ֣וֹת תַּרְשִׁ֗ישׁ כִּֽי־ שֻׁדַּ֤ד מִבַּ֙יִת֙ מִבּ֔וֹא מֵאֶ֥רֶץ כִּתִּ֖ים נִגְלָה־ לָֽמוֹ׃ (ה) דֹּ֖מּוּ יֹ֣שְׁבֵי אִ֑י סֹחֵ֥ר צִיד֛וֹן עֹבֵ֥ר יָ֖ם מִלְאֽוּךְ׃ (ו) וּבְמַ֤יִם רַבִּים֙ זֶ֣רַע שִׁחֹ֔ר קְצִ֥יר יְא֖וֹר תְּבֽוּאָתָ֑הּ וַתְּהִ֖י סְחַ֥ר גּוֹיִֽם׃ (ז) בּ֣וֹשִׁי צִיד֔וֹן כִּֽי־ אָמַ֣ר יָ֔ם מָע֥וֹז הַיָּ֖ם לֵאמֹ֑ר לֹֽא־ חַ֣לְתִּי וְלֹֽא־ יָלַ֗דְתִּי וְלֹ֥א גִדַּ֛לְתִּי בַּחוּרִ֖ים רוֹמַ֥מְתִּי בְתוּלֽוֹת׃ (ח) כַּֽאֲשֶׁר־ שֵׁ֖מַע לְמִצְרָ֑יִם יָחִ֖ילוּ כְּשֵׁ֥מַע צֹֽר׃ (ט) עִבְר֖וּ תַּרְשִׁ֑ישָׁה הֵילִ֖ילוּ יֹ֥שְׁבֵי אִֽי׃ (י) הֲזֹ֥את לָכֶ֖ם עַלִּיזָ֑ה מִֽימֵי־ קֶ֤דֶם קַדְמָתָהּ֙ יֹבִל֣וּהָ רַגְלֶ֔יהָ מֵֽרָח֖וֹק לָגֽוּר׃ (יא) מִ֚י יָעַ֣ץ זֹ֔את עַל־ צֹ֖ר הַמַּֽעֲטִירָ֑ה אֲשֶׁ֤ר סֹחֲרֶ֙יה֙ שָׂרִ֔ים כִּנְעָנֶ֖יהָ נִכְבַּדֵּי־ אָֽרֶץ׃ (יב) יְהוָ֥ה צְבָא֖וֹת יְעָצָ֑הּ לְחַלֵּל֙ גְּא֣וֹן כָּל־ צְבִ֔י לְהָקֵ֖ל כָּל־ נִכְבַּדֵּי־ אָֽרֶץ׃ (יג) עִבְרִ֥י אַרְצֵ֖ךְ כַּיְאֹ֑ר בַּת־ תַּרְשִׁ֕ישׁ אֵ֖ין מֵ֥זַח עֽוֹד׃ (יד) יָדוֹ֙ נָטָ֣ה עַל־ הַיָּ֔ם הִרְגִּ֖יז מַמְלָכ֑וֹת יְהוָה֙ צִוָּ֣ה אֶל־ כְּנַ֔עַן לַשְׁמִ֖ד מָעֻזְנֶֽיהָ׃ (טו) וַיֹּ֕אמֶר לֹֽא־ תוֹסִ֥יפִי ע֖וֹד לַעְל֑וֹז הַֽמְעֻשָּׁקָ֞ה בְּתוּלַ֣ת בַּת־ צִיד֗וֹן (כתיים) כִּתִּים֙ ק֣וּמִי עֲבֹ֔רִי גַּם־ שָׁ֖ם לֹא־ יָנ֥וּחַֽ לָֽךְ׃ (טז) הֵ֣ן ׀ אֶ֣רֶץ כַּשְׂדִּ֗ים זֶ֤ה הָעָם֙ לֹ֣א הָיָ֔ה אַשּׁ֖וּר יְסָדָ֣הּ לְצִיִּ֑ים הֵקִ֣ימוּ (בחיניו) בַחוּנָ֗יו עֹרְרוּ֙ אַרְמְנוֹתֶ֔יהָ שָׂמָ֖הּ לְמַפֵּלָֽה׃ (יז) הֵילִ֖ילוּ אֳנִיּ֣וֹת תַּרְשִׁ֑ישׁ כִּ֥י שֻׁדַּ֖ד מָעֻזְּכֶֽן׃ (יח) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא וְנִשְׁכַּ֤חַת צֹר֙ שִׁבְעִ֣ים שָׁנָ֔ה כִּימֵ֖י מֶ֣לֶךְ אֶחָ֑ד מִקֵּ֞ץ שִׁבְעִ֤ים שָׁנָה֙ יִהְיֶ֣ה לְצֹ֔ר כְּשִׁירַ֖ת הַזּוֹנָֽה׃ (יט) קְחִ֥י כִנּ֛וֹר סֹ֥בִּי עִ֖יר זוֹנָ֣ה נִשְׁכָּחָ֑ה הֵיטִ֤יבִי נַגֵּן֙ הַרְבִּי־ שִׁ֔יר לְמַ֖עַן תִּזָּכֵֽרִי׃ (כ) וְהָיָ֞ה מִקֵּ֣ץ ׀ שִׁבְעִ֣ים שָׁנָ֗ה יִפְקֹ֤ד יְהוָה֙ אֶת־ צֹ֔ר וְשָׁבָ֖ה לְאֶתְנַנָּ֑ה וְזָֽנְתָ֛ה אֶת־ כָּל־ מַמְלְכ֥וֹת הָאָ֖רֶץ עַל־ פְּנֵ֥י הָאֲדָמָֽה׃ (כא) וְהָיָ֨ה סַחְרָ֜הּ וְאֶתְנַנָּ֗הּ קֹ֚דֶשׁ לַֽיהוָ֔ה לֹ֥א יֵֽאָצֵ֖ר וְלֹ֣א יֵֽחָסֵ֑ן כִּ֣י לַיֹּשְׁבִ֞ים לִפְנֵ֤י יְהוָה֙ יִֽהְיֶ֣ה סַחְרָ֔הּ לֶאֱכֹ֥ל לְשָׂבְעָ֖ה וְלִמְכַסֶּ֥ה עָתִֽיק׃ תחילת פרק כד (כב) הִנֵּ֧ה יְהוָ֛ה בּוֹקֵ֥ק הָאָ֖רֶץ וּבֽוֹלְקָ֑הּ וְעִוָּ֣ה פָנֶ֔יהָ וְהֵפִ֖יץ יֹשְׁבֶֽיהָ׃ (כג) וְהָיָ֤ה כָעָם֙ כַּכֹּהֵ֔ן כַּעֶ֙בֶד֙ כַּֽאדֹנָ֔יו כַּשִּׁפְחָ֖ה כַּגְּבִרְתָּ֑הּ כַּקּוֹנֶה֙ כַּמּוֹכֵ֔ר כַּמַּלְוֶה֙ כַּלֹּוֶ֔ה כַּנֹּשֶׁ֕ה כַּאֲשֶׁ֖ר נֹשֶׁ֥א בֽוֹ׃ (כד) הִבּ֧וֹק ׀ תִּבּ֛וֹק הָאָ֖רֶץ וְהִבּ֣וֹז ׀ תִּבּ֑וֹז כִּ֣י יְהוָ֔ה דִּבֶּ֖ר אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (כה) אָבְלָ֤ה נָֽבְלָה֙ הָאָ֔רֶץ אֻמְלְלָ֥ה נָבְלָ֖ה תֵּבֵ֑ל אֻמְלָ֖לוּ מְר֥וֹם עַם־ הָאָֽרֶץ׃ (כו) וְהָאָ֥רֶץ חָנְפָ֖ה תַּ֣חַת יֹשְׁבֶ֑יהָ כִּֽי־ עָבְר֤וּ תוֹרֹת֙ חָ֣לְפוּ חֹ֔ק הֵפֵ֖רוּ בְּרִ֥ית עוֹלָֽם׃ (כז) עַל־ כֵּ֗ן אָלָה֙ אָ֣כְלָה אֶ֔רֶץ וַֽיֶּאְשְׁמ֖וּ יֹ֣שְׁבֵי בָ֑הּ עַל־ כֵּ֗ן חָרוּ֙ יֹ֣שְׁבֵי אֶ֔רֶץ וְנִשְׁאַ֥ר אֱנ֖וֹשׁ מִזְעָֽר׃ (כח) אָבַ֥ל תִּיר֖וֹשׁ אֻמְלְלָה־ גָ֑פֶן נֶאֶנְח֖וּ כָּל־ שִׂמְחֵי־ לֵֽב׃ (כט) שָׁבַת֙ מְשׂ֣וֹשׂ תֻּפִּ֔ים חָדַ֖ל שְׁא֣וֹן עַלִּיזִ֑ים שָׁבַ֖ת מְשׂ֥וֹשׂ כִּנּֽוֹר׃ (ל) בַּשִּׁ֖יר לֹ֣א יִשְׁתּוּ־ יָ֑יִן יֵמַ֥ר שֵׁכָ֖ר לְשֹׁתָֽיו׃ (לא) נִשְׁבְּרָ֖ה קִרְיַת־ תֹּ֑הוּ סֻגַּ֥ר כָּל־ בַּ֖יִת מִבּֽוֹא׃ (לב) צְוָחָ֥ה עַל־ הַיַּ֖יִן בַּֽחוּצ֑וֹת עָֽרְבָה֙ כָּל־ שִׂמְחָ֔ה גָּלָ֖ה מְשׂ֥וֹשׂ הָאָֽרֶץ׃ (לג) נִשְׁאַ֥ר בָּעִ֖יר שַׁמָּ֑ה וּשְׁאִיָּ֖ה יֻכַּת־ שָֽׁעַר׃ (לד) כִּ֣י כֹ֥ה יִהְיֶ֛ה בְּקֶ֥רֶב הָאָ֖רֶץ בְּת֣וֹךְ הָֽעַמִּ֑ים כְּנֹ֣קֶף זַ֔יִת כְּעוֹלֵלֹ֖ת אִם־ כָּלָ֥ה בָצִֽיר׃ (לה) הֵ֛מָּה יִשְׂא֥וּ קוֹלָ֖ם יָרֹ֑נּוּ בִּגְא֣וֹן יְהוָ֔ה צָהֲל֖וּ מִיָּֽם׃ (לו) עַל־ כֵּ֥ן בָּאֻרִ֖ים כַּבְּד֣וּ יְהוָ֑ה בְּאִיֵּ֣י הַיָּ֔ם שֵׁ֥ם יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (לז) מִכְּנַ֨ף הָאָ֜רֶץ זְמִרֹ֤ת שָׁמַ֙עְנוּ֙ צְבִ֣י לַצַּדִּ֔יק וָאֹמַ֛ר רָזִי־ לִ֥י רָֽזִי־ לִ֖י א֣וֹי לִ֑י בֹּגְדִ֣ים בָּגָ֔דוּ וּבֶ֥גֶד בּוֹגְדִ֖ים בָּגָֽדוּ׃ (לח) פַּ֥חַד וָפַ֖חַת וָפָ֑ח עָלֶ֖יךָ יוֹשֵׁ֥ב הָאָֽרֶץ׃ (לט) וְֽ֠הָיָה הַנָּ֞ס מִקּ֤וֹל הַפַּ֙חַד֙ יִפֹּ֣ל אֶל־ הַפַּ֔חַת וְהָֽעוֹלֶה֙ מִתּ֣וֹךְ הַפַּ֔חַת יִלָּכֵ֖ד בַּפָּ֑ח כִּֽי־ אֲרֻבּ֤וֹת מִמָּרוֹם֙ נִפְתָּ֔חוּ וַֽיִּרְעֲשׁ֖וּ מ֥וֹסְדֵי אָֽרֶץ׃ (מ) רֹ֥עָה הִֽתְרֹעֲעָ֖ה הָאָ֑רֶץ פּ֤וֹר הִֽתְפּוֹרְרָה֙ אֶ֔רֶץ מ֥וֹט הִֽתְמוֹטְטָ֖ה אָֽרֶץ׃ (מא) נ֣וֹעַ תָּנ֤וּעַ אֶ֙רֶץ֙ כַּשִּׁכּ֔וֹר וְהִֽתְנוֹדְדָ֖ה כַּמְּלוּנָ֑ה וְכָבַ֤ד עָלֶ֙יהָ֙ פִּשְׁעָ֔הּ וְנָפְלָ֖ה וְלֹא־ תֹסִ֥יף קֽוּם׃ (מב) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא יִפְקֹ֧ד יְהוָ֛ה עַל־ צְבָ֥א הַמָּר֖וֹם בַּמָּר֑וֹם וְעַל־ מַלְכֵ֥י הָאֲדָמָ֖ה עַל־ הָאֲדָמָֽה׃ (מג) וְאֻסְּפ֨וּ אֲסֵפָ֤ה אַסִּיר֙ עַל־ בּ֔וֹר וְסֻגְּר֖וּ עַל־ מַסְגֵּ֑ר וּמֵרֹ֥ב יָמִ֖ים יִפָּקֵֽדוּ׃

ישעיהו סדר י

(א) וְחָֽפְרָה֙ הַלְּבָנָ֔ה וּבוֹשָׁ֖ה הַֽחַמָּ֑ה כִּֽי־ מָלַ֞ךְ יְהוָ֣ה צְבָא֗וֹת בְּהַ֤ר צִיּוֹן֙ וּבִיר֣וּשָׁלִַ֔ם וְנֶ֥גֶד זְקֵנָ֖יו כָּבֽוֹד׃ סוף פרק כד (ב) יְהוָ֤ה אֱלֹהַי֙ אַתָּ֔ה אֲרֽוֹמִמְךָ֙ אוֹדֶ֣ה שִׁמְךָ֔ כִּ֥י עָשִׂ֖יתָ פֶּ֑לֶא עֵצ֥וֹת מֵֽרָח֖וֹק אֱמ֥וּנָה אֹֽמֶן׃ (ג) כִּ֣י שַׂ֤מְתָּ מֵעִיר֙ לַגָּ֔ל קִרְיָ֥ה בְצוּרָ֖ה לְמַפֵּלָ֑ה אַרְמ֤וֹן זָרִים֙ מֵעִ֔יר לְעוֹלָ֖ם לֹ֥א יִבָּנֶֽה׃ (ד) עַל־ כֵּ֖ן יְכַבְּד֣וּךָ עַם־ עָ֑ז קִרְיַ֛ת גּוֹיִ֥ם עָרִיצִ֖ים יִירָאֽוּךָ׃ (ה) כִּֽי־ הָיִ֨יתָ מָע֥וֹז לַדָּ֛ל מָע֥וֹז לָאֶבְי֖וֹן בַּצַּר־ ל֑וֹ מַחְסֶ֤ה מִזֶּ֙רֶם֙ צֵ֣ל מֵחֹ֔רֶב כִּ֛י ר֥וּחַ עָרִיצִ֖ים כְּזֶ֥רֶם קִֽיר׃ (ו) כְּחֹ֣רֶב בְּצָי֔וֹן שְׁא֥וֹן זָרִ֖ים תַּכְנִ֑יעַ חֹ֚רֶב בְּצֵ֣ל עָ֔ב זְמִ֥יר עָֽרִיצִ֖ים יַעֲנֶֽה׃ (ז) וְעָשָׂה֩ יְהוָ֨ה צְבָא֜וֹת לְכָל־ הָֽעַמִּים֙ בָּהָ֣ר הַזֶּ֔ה מִשְׁתֵּ֥ה שְׁמָנִ֖ים מִשְׁתֵּ֣ה שְׁמָרִ֑ים שְׁמָנִים֙ מְמֻ֣חָיִ֔ם שְׁמָרִ֖ים מְזֻקָּקִֽים׃ (ח) וּבִלַּע֙ בָּהָ֣ר הַזֶּ֔ה פְּנֵֽי־ הַלּ֥וֹט ׀ הַלּ֖וֹט עַל־ כָּל־ הָֽעַמִּ֑ים וְהַמַּסֵּכָ֥ה הַנְּסוּכָ֖ה עַל־ כָּל־ הַגּוֹיִֽם׃ (ט) בִּלַּ֤ע הַמָּ֙וֶת֙ לָנֶ֔צַח וּמָחָ֨ה אֲדֹנָ֧י יְהוִ֛ה דִּמְעָ֖ה מֵעַ֣ל כָּל־ פָּנִ֑ים וְחֶרְפַּ֣ת עַמּ֗וֹ יָסִיר֙ מֵעַ֣ל כָּל־ הָאָ֔רֶץ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (י) וְאָמַר֙ בַּיּ֣וֹם הַה֔וּא הִנֵּ֨ה אֱלֹהֵ֥ינוּ זֶ֛ה קִוִּ֥ינוּ ל֖וֹ וְיֽוֹשִׁיעֵ֑נוּ זֶ֤ה יְהוָה֙ קִוִּ֣ינוּ ל֔וֹ נָגִ֥ילָה וְנִשְׂמְחָ֖ה בִּישׁוּעָתֽוֹ׃ (יא) כִּֽי־ תָנ֥וּחַ יַד־ יְהוָ֖ה בָּהָ֣ר הַזֶּ֑ה וְנָ֤דוֹשׁ מוֹאָב֙ תַּחְתָּ֔יו כְּהִדּ֥וּשׁ מַתְבֵּ֖ן (במי) בְּמ֥וֹ מַדְמֵנָֽה׃ (יב) וּפֵרַ֤שׂ יָדָיו֙ בְּקִרְבּ֔וֹ כַּאֲשֶׁ֛ר יְפָרֵ֥שׂ הַשֹּׂחֶ֖ה לִשְׂח֑וֹת וְהִשְׁפִּיל֙ גַּֽאֲוָת֔וֹ עִ֖ם אָרְבּ֥וֹת יָדָֽיו׃ (יג) וּמִבְצַ֞ר מִשְׂגַּ֣ב חוֹמֹתֶ֗יךָ הֵשַׁ֥ח הִשְׁפִּ֛יל הִגִּ֥יעַ לָאָ֖רֶץ עַד־ עָפָֽר׃ (יד) בַּיּ֣וֹם הַה֔וּא יוּשַׁ֥ר הַשִּׁיר־ הַזֶּ֖ה בְּאֶ֣רֶץ יְהוּדָ֑ה עִ֣יר עָז־ לָ֔נוּ יְשׁוּעָ֥ה יָשִׁ֖ית חוֹמ֥וֹת וָחֵֽל׃ (טו) פִּתְח֖וּ שְׁעָרִ֑ים וְיָבֹ֥א גוֹי־ צַדִּ֖יק שֹׁמֵ֥ר אֱמֻנִֽים׃ (טז) יֵ֣צֶר סָמ֔וּךְ תִּצֹּ֖ר שָׁל֣וֹם ׀ שָׁל֑וֹם כִּ֥י בְךָ֖ בָּטֽוּחַ׃ (יז) בִּטְח֥וּ בַֽיהוָ֖ה עֲדֵי־ עַ֑ד כִּ֚י בְּיָ֣הּ יְהוָ֔ה צ֖וּר עוֹלָמִֽים׃ (יח) כִּ֤י הֵשַׁח֙ יֹשְׁבֵ֣י מָר֔וֹם קִרְיָ֖ה נִשְׂגָּבָ֑ה יַשְׁפִּילֶ֤נָּה יַשְׁפִּילָהּ֙ עַד־ אֶ֔רֶץ יַגִּיעֶ֖נָּה עַד־ עָפָֽר׃ (יט) תִּרְמְסֶ֖נָּה רָ֑גֶל רַגְלֵ֥י עָנִ֖י פַּעֲמֵ֥י דַלִּֽים׃ (כ) אֹ֥רַח לַצַּדִּ֖יק מֵֽישָׁרִ֑ים יָשָׁ֕ר מַעְגַּ֥ל צַדִּ֖יק תְּפַלֵּֽס׃ (כא) אַ֣ף אֹ֧רַח מִשְׁפָּטֶ֛יךָ יְהוָ֖ה קִוִּינ֑וּךָ לְשִׁמְךָ֥ וּֽלְזִכְרְךָ֖ תַּאֲוַת־ נָֽפֶשׁ׃ (כב) נַפְשִׁ֤י אִוִּיתִ֙יךָ֙ בַּלַּ֔יְלָה אַף־ רוּחִ֥י בְקִרְבִּ֖י אֲשַֽׁחֲרֶ֑ךָּ כִּ֞י כַּאֲשֶׁ֤ר מִשְׁפָּטֶ֙יךָ֙ לָאָ֔רֶץ צֶ֥דֶק לָמְד֖וּ יֹשְׁבֵ֥י תֵבֵֽל׃ (כג) יֻחַ֤ן רָשָׁע֙ בַּל־ לָמַ֣ד צֶ֔דֶק בְּאֶ֥רֶץ נְכֹח֖וֹת יְעַוֵּ֑ל וּבַל־ יִרְאֶ֖ה גֵּא֥וּת יְהוָֽה׃ (כד) יְהוָ֛ה רָ֥מָה יָדְךָ֖ בַּל־ יֶחֱזָי֑וּן יֶחֱז֤וּ וְיֵבֹ֙שׁוּ֙ קִנְאַת־ עָ֔ם אַף־ אֵ֖שׁ צָרֶ֥יךָ תֹאכְלֵֽם׃ (כה) יְהוָ֕ה תִּשְׁפֹּ֥ת שָׁל֖וֹם לָ֑נוּ כִּ֛י גַּ֥ם כָּֽל־ מַעֲשֵׂ֖ינוּ פָּעַ֥לְתָּ לָּֽנוּ׃ (כו) יְהוָ֣ה אֱלֹהֵ֔ינוּ בְּעָל֥וּנוּ אֲדֹנִ֖ים זֽוּלָתֶ֑ךָ לְבַד־ בְּךָ֖ נַזְכִּ֥יר שְׁמֶֽךָ׃ (כז) מֵתִים֙ בַּל־ יִחְי֔וּ רְפָאִ֖ים בַּל־ יָקֻ֑מוּ לָכֵ֤ן פָּקַ֙דְתָּ֙ וַתַּשְׁמִידֵ֔ם וַתְּאַבֵּ֥ד כָּל־ זֵ֖כֶר לָֽמוֹ׃ (כח) יָסַ֤פְתָּ לַגּוֹי֙ יְהוָ֔ה יָסַ֥פְתָּ לַגּ֖וֹי נִכְבָּ֑דְתָּ רִחַ֖קְתָּ כָּל־ קַצְוֵי־ אָֽרֶץ׃ (כט) יְהוָ֖ה בַּצַּ֣ר פְּקָד֑וּךָ צָק֣וּן לַ֔חַשׁ מוּסָרְךָ֖ לָֽמוֹ׃ (ל) כְּמ֤וֹ הָרָה֙ תַּקְרִ֣יב לָלֶ֔דֶת תָּחִ֥יל תִּזְעַ֖ק בַּחֲבָלֶ֑יהָ כֵּ֛ן הָיִ֥ינוּ מִפָּנֶ֖יךָ יְהוָֽה׃ (לא) הָרִ֣ינוּ חַ֔לְנוּ כְּמ֖וֹ יָלַ֣דְנוּ ר֑וּחַ יְשׁוּעֹת֙ בַּל־ נַ֣עֲשֶׂה אֶ֔רֶץ וּבַֽל־ יִפְּל֖וּ יֹשְׁבֵ֥י תֵבֵֽל׃ (לב) יִֽחְי֣וּ מֵתֶ֔יךָ נְבֵלָתִ֖י יְקוּמ֑וּן הָקִ֨יצוּ וְרַנְּנ֜וּ שֹׁכְנֵ֣י עָפָ֗ר כִּ֣י טַ֤ל אוֹרֹת֙ טַלֶּ֔ךָ וָאָ֖רֶץ רְפָאִ֥ים תַּפִּֽיל׃ (לג) לֵ֤ךְ עַמִּי֙ בֹּ֣א בַחֲדָרֶ֔יךָ וּֽסְגֹ֥ר (דלתיך) דְּלָתְךָ֖ בַּעֲדֶ֑ךָ חֲבִ֥י כִמְעַט־ רֶ֖גַע עַד־ (יעבור)־ יַעֲבָר־ זָֽעַם׃ (לד) כִּֽי־ הִנֵּ֤ה יְהוָה֙ יֹצֵ֣א מִמְּקוֹמ֔וֹ לִפְקֹ֛ד עֲוֺ֥ן יֹֽשֵׁב־ הָאָ֖רֶץ עָלָ֑יו וְגִלְּתָ֤ה הָאָ֙רֶץ֙ אֶת־ דָּמֶ֔יהָ וְלֹֽא־ תְכַסֶּ֥ה ע֖וֹד עַל־ הֲרוּגֶֽיהָ׃ (לה) בַּיּ֣וֹם הַה֡וּא יִפְקֹ֣ד יְהוָה֩ בְּחַרְב֨וֹ הַקָּשָׁ֜ה וְהַגְּדוֹלָ֣ה וְהַֽחֲזָקָ֗ה עַ֤ל לִוְיָתָן֙ נָחָ֣שׁ בָּרִ֔חַ וְעַל֙ לִוְיָתָ֔ן נָחָ֖שׁ עֲקַלָּת֑וֹן וְהָרַ֥ג אֶת־ הַתַּנִּ֖ין אֲשֶׁ֥ר בַּיָּֽם׃ (לו) בַּיּ֖וֹם הַה֑וּא כֶּ֥רֶם חֶ֖מֶד עַנּוּ־ לָֽהּ׃ (לז) אֲנִ֤י יְהוָה֙ נֹֽצְרָ֔הּ לִרְגָעִ֖ים אַשְׁקֶ֑נָּה פֶּ֚ן יִפְקֹ֣ד עָלֶ֔יהָ לַ֥יְלָה וָי֖וֹם אֶצֳּרֶֽנָּה׃ (לח) חֵמָ֖ה אֵ֣ין לִ֑י מִֽי־ יִתְּנֵ֜נִי שָׁמִ֥יר שַׁ֙יִת֙ בַּמִּלְחָמָ֔ה אֶפְשְׂעָ֥ה בָ֖הּ אֲצִיתֶ֥נָּה יָּֽחַד׃ (לט) א֚וֹ יַחֲזֵ֣ק בְּמָעוּזִּ֔י יַעֲשֶׂ֥ה שָׁל֖וֹם לִ֑י שָׁל֖וֹם יַֽעֲשֶׂה־ לִּֽי׃ (מ) הַבָּאִים֙ יַשְׁרֵ֣שׁ יַֽעֲקֹ֔ב יָצִ֥יץ וּפָרַ֖ח יִשְׂרָאֵ֑ל וּמָלְא֥וּ פְנֵי־ תֵבֵ֖ל תְּנוּבָֽה׃ (מא) הַכְּמַכַּ֥ת מַכֵּ֖הוּ הִכָּ֑הוּ אִם־ כְּהֶ֥רֶג הֲרֻגָ֖יו הֹרָֽג׃ (מב) בְּסַּאסְּאָ֖ה בְּשַׁלְחָ֣הּ תְּרִיבֶ֑נָּה הָגָ֛ה בְּרוּח֥וֹ הַקָּשָׁ֖ה בְּי֥וֹם קָדִֽים׃ (מג) לָכֵ֗ן בְּזֹאת֙ יְכֻפַּ֣ר עֲוֺֽן־ יַעֲקֹ֔ב וְזֶ֕ה כָּל־ פְּרִ֖י הָסִ֣ר חַטָּאת֑וֹ בְּשׂוּמ֣וֹ ׀ כָּל־ אַבְנֵ֣י מִזְבֵּ֗חַ כְּאַבְנֵי־ גִר֙ מְנֻפָּצ֔וֹת לֹֽא־ יָקֻ֥מוּ אֲשֵׁרִ֖ים וְחַמָּנִֽים׃ (מד) כִּ֣י עִ֤יר בְּצוּרָה֙ בָּדָ֔ד נָוֶ֕ה מְשֻׁלָּ֥ח וְנֶעֱזָ֖ב כַּמִּדְבָּ֑ר שָׁ֣ם יִרְעֶ֥ה עֵ֛גֶל וְשָׁ֥ם יִרְבָּ֖ץ וְכִלָּ֥ה סְעִפֶֽיהָ׃ (מה) בִּיבֹ֤שׁ קְצִירָהּ֙ תִּשָּׁבַ֔רְנָה נָשִׁ֕ים בָּא֖וֹת מְאִיר֣וֹת אוֹתָ֑הּ כִּ֣י לֹ֤א עַם־ בִּינוֹת֙ ה֔וּא עַל־ כֵּן֙ לֹֽא־ יְרַחֲמֶ֣נּוּ עֹשֵׂ֔הוּ וְיֹצְר֖וֹ לֹ֥א יְחֻנֶּֽנּוּ׃
