ישעיהו פרק מט לפי החלוקה לסדרים

פרק מט בספר ישעיהו (26 פסוקים) נמצא בישעיהו סדר כ (פסוקים כב–מו) ובישעיהו סדר כא (פסוקים א–א).

ישעיהו סדר כ

(א) כִּֽי־ מֵעִ֤יר הַקֹּ֙דֶשׁ֙ נִקְרָ֔אוּ וְעַל־ אֱלֹהֵ֥י יִשְׂרָאֵ֖ל נִסְמָ֑כוּ יְהוָ֥ה צְבָא֖וֹת שְׁמֽוֹ׃ (ב) הָרִֽאשֹׁנוֹת֙ מֵאָ֣ז הִגַּ֔דְתִּי וּמִפִּ֥י יָצְא֖וּ וְאַשְׁמִיעֵ֑ם פִּתְאֹ֥ם עָשִׂ֖יתִי וַתָּבֹֽאנָה׃ (ג) מִדַּעְתִּ֕י כִּ֥י קָשֶׁ֖ה אָ֑תָּה וְגִ֤יד בַּרְזֶל֙ עָרְפֶּ֔ךָ וּמִצְחֲךָ֖ נְחוּשָֽׁה׃ (ד) וָאַגִּ֤יד לְךָ֙ מֵאָ֔ז בְּטֶ֥רֶם תָּב֖וֹא הִשְׁמַעְתִּ֑יךָ פֶּן־ תֹּאמַר֙ עָצְבִּ֣י עָשָׂ֔ם וּפִסְלִ֥י וְנִסְכִּ֖י צִוָּֽם׃ (ה) שָׁמַ֤עְתָּֽ חֲזֵה֙ כֻּלָּ֔הּ וְאַתֶּ֖ם הֲל֣וֹא תַגִּ֑ידוּ הִשְׁמַעְתִּ֤יךָ חֲדָשׁוֹת֙ מֵעַ֔תָּה וּנְצֻר֖וֹת וְלֹ֥א יְדַעְתָּֽם׃ (ו) עַתָּ֤ה נִבְרְאוּ֙ וְלֹ֣א מֵאָ֔ז וְלִפְנֵי־ י֖וֹם וְלֹ֣א שְׁמַעְתָּ֑ם פֶּן־ תֹּאמַ֖ר הִנֵּ֥ה יְדַעְתִּֽין׃ (ז) גַּ֣ם לֹֽא־ שָׁמַ֗עְתָּ גַּ֚ם לֹ֣א יָדַ֔עְתָּ גַּ֕ם מֵאָ֖ז לֹא־ פִתְּחָ֣ה אָזְנֶ֑ךָ כִּ֤י יָדַ֙עְתִּי֙ בָּג֣וֹד תִּבְגּ֔וֹד וּפֹשֵׁ֥עַ מִבֶּ֖טֶן קֹ֥רָא לָֽךְ׃ (ח) לְמַ֤עַן שְׁמִי֙ אַאֲרִ֣יךְ אַפִּ֔י וּתְהִלָּתִ֖י אֶחֱטָם־ לָ֑ךְ לְבִלְתִּ֖י הַכְרִיתֶֽךָ׃ (ט) הִנֵּ֥ה צְרַפְתִּ֖יךָ וְלֹ֣א בְכָ֑סֶף בְּחַרְתִּ֖יךָ בְּכ֥וּר עֹֽנִי׃ (י) לְמַעֲנִ֧י לְמַעֲנִ֛י אֶעֱשֶׂ֖ה כִּ֣י אֵ֣יךְ יֵחָ֑ל וּכְבוֹדִ֖י לְאַחֵ֥ר לֹֽא־ אֶתֵּֽן׃ (יא) שְׁמַ֤ע אֵלַי֙ יַֽעֲקֹ֔ב וְיִשְׂרָאֵ֖ל מְקֹרָאִ֑י אֲנִי־ הוּא֙ אֲנִ֣י רִאשׁ֔וֹן אַ֖ף אֲנִ֥י אַחֲרֽוֹן׃ (יב) אַף־ יָדִי֙ יָ֣סְדָה אֶ֔רֶץ וִֽימִינִ֖י טִפְּחָ֣ה שָׁמָ֑יִם קֹרֵ֥א אֲנִ֛י אֲלֵיהֶ֖ם יַעַמְד֥וּ יַחְדָּֽו׃ (יג) הִקָּבְצ֤וּ כֻלְּכֶם֙ וּֽשֲׁמָ֔עוּ מִ֥י בָהֶ֖ם הִגִּ֣יד אֶת־ אֵ֑לֶּה יְהוָ֣ה אֲהֵב֔וֹ יַעֲשֶׂ֤ה חֶפְצוֹ֙ בְּבָבֶ֔ל וּזְרֹע֖וֹ כַּשְׂדִּֽים׃ (יד) אֲנִ֥י אֲנִ֛י דִּבַּ֖רְתִּי אַף־ קְרָאתִ֑יו הֲבִיאֹתִ֖יו וְהִצְלִ֥יחַ דַּרְכּֽוֹ׃ (טו) קִרְב֧וּ אֵלַ֣י שִׁמְעוּ־ זֹ֗את לֹ֤א מֵרֹאשׁ֙ בַּסֵּ֣תֶר דִּבַּ֔רְתִּי מֵעֵ֥ת הֱיוֹתָ֖הּ שָׁ֣ם אָ֑נִי וְעַתָּ֗ה אֲדֹנָ֧י יְהוִ֛ה שְׁלָחַ֖נִי וְרוּחֽוֹ׃ (טז) כֹּֽה־ אָמַ֧ר יְהוָ֛ה גֹּאַלְךָ֖ קְד֣וֹשׁ יִשְׂרָאֵ֑ל אֲנִ֨י יְהוָ֤ה אֱלֹהֶ֙יךָ֙ מְלַמֶּדְךָ֣ לְהוֹעִ֔יל מַדְרִֽיכֲךָ֖ בְּדֶ֥רֶךְ תֵּלֵֽךְ׃ (יז) ל֥וּא הִקְשַׁ֖בְתָּ לְמִצְוֺתָ֑י וַיְהִ֤י כַנָּהָר֙ שְׁלוֹמֶ֔ךָ וְצִדְקָתְךָ֖ כְּגַלֵּ֥י הַיָּֽם׃ (יח) וַיְהִ֤י כַחוֹל֙ זַרְעֶ֔ךָ וְצֶאֱצָאֵ֥י מֵעֶ֖יךָ כִּמְעֹתָ֑יו לֹֽא־ יִכָּרֵ֧ת וְֽלֹא־ יִשָּׁמֵ֛ד שְׁמ֖וֹ מִלְּפָנָֽי׃ (יט) צְא֣וּ מִבָּבֶל֮ בִּרְח֣וּ מִכַּשְׂדִּים֒ בְּק֣וֹל רִנָּ֗ה הַגִּ֤ידוּ הַשְׁמִ֙יעוּ֙ זֹ֔את הוֹצִיא֖וּהָ עַד־ קְצֵ֣ה הָאָ֑רֶץ אִמְר֕וּ גָּאַ֥ל יְהוָ֖ה עַבְדּ֥וֹ יַעֲקֹֽב׃ (כ) וְלֹ֣א צָמְא֗וּ בָּחֳרָבוֹת֙ הֽוֹלִיכָ֔ם מַ֥יִם מִצּ֖וּר הִזִּ֣יל לָ֑מוֹ וַיִּ֨בְקַע־ צ֔וּר וַיָּזֻ֖בוּ מָֽיִם׃ (כא) אֵ֣ין שָׁל֔וֹם אָמַ֥ר יְהוָ֖ה לָרְשָׁעִֽים׃ תחילת פרק מט (כב) שִׁמְע֤וּ אִיִּים֙ אֵלַ֔י וְהַקְשִׁ֥יבוּ לְאֻמִּ֖ים מֵרָח֑וֹק יְהוָה֙ מִבֶּ֣טֶן קְרָאָ֔נִי מִמְּעֵ֥י אִמִּ֖י הִזְכִּ֥יר שְׁמִֽי׃ (כג) וַיָּ֤שֶׂם פִּי֙ כְּחֶ֣רֶב חַדָּ֔ה בְּצֵ֥ל יָד֖וֹ הֶחְבִּיאָ֑נִי וַיְשִׂימֵ֙נִי֙ לְחֵ֣ץ בָּר֔וּר בְּאַשְׁפָּת֖וֹ הִסְתִּירָֽנִי׃ (כד) וַיֹּ֥אמֶר לִ֖י עַבְדִּי־ אָ֑תָּה יִשְׂרָאֵ֕ל אֲשֶׁר־ בְּךָ֖ אֶתְפָּאָֽר׃ (כה) וַאֲנִ֤י אָמַ֙רְתִּי֙ לְרִ֣יק יָגַ֔עְתִּי לְתֹ֥הוּ וְהֶ֖בֶל כֹּחִ֣י כִלֵּ֑יתִי אָכֵן֙ מִשְׁפָּטִ֣י אֶת־ יְהוָ֔ה וּפְעֻלָּתִ֖י אֶת־ אֱלֹהָֽי׃ (כו) וְעַתָּ֣ה ׀ אָמַ֣ר יְהוָ֗ה יֹצְרִ֤י מִבֶּ֙טֶן֙ לְעֶ֣בֶד ל֔וֹ לְשׁוֹבֵ֤ב יַֽעֲקֹב֙ אֵלָ֔יו וְיִשְׂרָאֵ֖ל (לא) ל֣וֹ יֵאָסֵ֑ף וְאֶכָּבֵד֙ בְּעֵינֵ֣י יְהוָ֔ה וֵאלֹהַ֖י הָיָ֥ה עֻזִּֽי׃ (כז) וַיֹּ֗אמֶר נָקֵ֨ל מִֽהְיוֹתְךָ֥ לִי֙ עֶ֔בֶד לְהָקִים֙ אֶת־ שִׁבְטֵ֣י יַעֲקֹ֔ב (ונצירי) וּנְצוּרֵ֥י יִשְׂרָאֵ֖ל לְהָשִׁ֑יב וּנְתַתִּ֙יךָ֙ לְא֣וֹר גּוֹיִ֔ם לִֽהְי֥וֹת יְשׁוּעָתִ֖י עַד־ קְצֵ֥ה הָאָֽרֶץ׃ (כח) כֹּ֣ה אָֽמַר־ יְהוָה֩ גֹּאֵ֨ל יִשְׂרָאֵ֜ל קְדוֹשׁ֗וֹ לִבְזֹה־ נֶ֜פֶשׁ לִמְתָ֤עֵֽב גּוֹי֙ לְעֶ֣בֶד מֹשְׁלִ֔ים מְלָכִים֙ יִרְא֣וּ וָקָ֔מוּ שָׂרִ֖ים וְיִֽשְׁתַּחֲוּ֑וּ לְמַ֤עַן יְהוָה֙ אֲשֶׁ֣ר נֶאֱמָ֔ן קְדֹ֥שׁ יִשְׂרָאֵ֖ל וַיִּבְחָרֶֽךָּ׃ (כט) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה בְּעֵ֤ת רָצוֹן֙ עֲנִיתִ֔יךָ וּבְי֥וֹם יְשׁוּעָ֖ה עֲזַרְתִּ֑יךָ וְאֶצָּרְךָ֗ וְאֶתֶּנְךָ֙ לִבְרִ֣ית עָ֔ם לְהָקִ֣ים אֶ֔רֶץ לְהַנְחִ֖יל נְחָל֥וֹת שֹׁמֵמֽוֹת׃ (ל) לֵאמֹ֤ר לַֽאֲסוּרִים֙ צֵ֔אוּ לַאֲשֶׁ֥ר בַּחֹ֖שֶׁךְ הִגָּל֑וּ עַל־ דְּרָכִ֣ים יִרְע֔וּ וּבְכָל־ שְׁפָיִ֖ים מַרְעִיתָֽם׃ (לא) לֹ֤א יִרְעָ֙בוּ֙ וְלֹ֣א יִצְמָ֔אוּ וְלֹא־ יַכֵּ֥ם שָׁרָ֖ב וָשָׁ֑מֶשׁ כִּי־ מְרַחֲמָ֣ם יְנַהֲגֵ֔ם וְעַל־ מַבּ֥וּעֵי מַ֖יִם יְנַהֲלֵֽם׃ (לב) וְשַׂמְתִּ֥י כָל־ הָרַ֖י לַדָּ֑רֶךְ וּמְסִלֹּתַ֖י יְרֻמֽוּן׃ (לג) הִנֵּה־ אֵ֕לֶּה מֵרָח֖וֹק יָבֹ֑אוּ וְהִֽנֵּה־ אֵ֙לֶּה֙ מִצָּפ֣וֹן וּמִיָּ֔ם וְאֵ֖לֶּה מֵאֶ֥רֶץ סִינִֽים׃ (לד) רָנּ֤וּ שָׁמַ֙יִם֙ וְגִ֣ילִי אָ֔רֶץ (יפצחו) וּפִצְח֥וּ הָרִ֖ים רִנָּ֑ה כִּֽי־ נִחַ֤ם יְהוָה֙ עַמּ֔וֹ וַעֲנִיָּ֖ו יְרַחֵֽם׃ (לה) וַתֹּ֥אמֶר צִיּ֖וֹן עֲזָבַ֣נִי יְהוָ֑ה וַאדֹנָ֖י שְׁכֵחָֽנִי׃ (לו) הֲתִשְׁכַּ֤ח אִשָּׁה֙ עוּלָ֔הּ מֵרַחֵ֖ם בֶּן־ בִּטְנָ֑הּ גַּם־ אֵ֣לֶּה תִשְׁכַּ֔חְנָה וְאָנֹכִ֖י לֹ֥א אֶשְׁכָּחֵֽךְ׃ (לז) הֵ֥ן עַל־ כַּפַּ֖יִם חַקֹּתִ֑יךְ חוֹמֹתַ֥יִךְ נֶגְדִּ֖י תָּמִֽיד׃ (לח) מִֽהֲר֖וּ בָּנָ֑יִךְ מְהָֽרְסַ֥יִךְ וּמַחֲרִבַ֖יִךְ מִמֵּ֥ךְ יֵצֵֽאוּ׃ (לט) שְׂאִֽי־ סָבִ֤יב עֵינַ֙יִךְ֙ וּרְאִ֔י כֻּלָּ֖ם נִקְבְּצ֣וּ בָֽאוּ־ לָ֑ךְ חַי־ אָ֣נִי נְאֻם־ יְהוָ֗ה כִּ֤י כֻלָּם֙ כָּעֲדִ֣י תִלְבָּ֔שִׁי וּֽתְקַשְּׁרִ֖ים כַּכַּלָּֽה׃ (מ) כִּ֤י חָרְבֹתַ֙יִךְ֙ וְשֹׁ֣מְמֹתַ֔יִךְ וְאֶ֖רֶץ הֲרִֽסֻתֵ֑יךְ כִּ֤י עַתָּה֙ תֵּצְרִ֣י מִיּוֹשֵׁ֔ב וְרָחֲק֖וּ מְבַלְּעָֽיִךְ׃ (מא) ע֚וֹד יֹאמְר֣וּ בְאָזְנַ֔יִךְ בְּנֵ֖י שִׁכֻּלָ֑יִךְ צַר־ לִ֥י הַמָּק֖וֹם גְּשָׁה־ לִּ֥י וְאֵשֵֽׁבָה׃ (מב) וְאָמַ֣רְתְּ בִּלְבָבֵ֗ךְ מִ֤י יָֽלַד־ לִי֙ אֶת־ אֵ֔לֶּה וַאֲנִ֥י שְׁכוּלָ֖ה וְגַלְמוּדָ֑ה גֹּלָ֣ה ׀ וְסוּרָ֗ה וְאֵ֙לֶּה֙ מִ֣י גִדֵּ֔ל הֵ֤ן אֲנִי֙ נִשְׁאַ֣רְתִּי לְבַדִּ֔י אֵ֖לֶּה אֵיפֹ֥ה הֵֽם׃ (מג) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה הִנֵּ֨ה אֶשָּׂ֤א אֶל־ גּוֹיִם֙ יָדִ֔י וְאֶל־ עַמִּ֖ים אָרִ֣ים נִסִּ֑י וְהֵבִ֤יאוּ בָנַ֙יִךְ֙ בְּחֹ֔צֶן וּבְנֹתַ֖יִךְ עַל־ כָּתֵ֥ף תִּנָּשֶֽׂאנָה׃ (מד) וְהָי֨וּ מְלָכִ֜ים אֹֽמְנַ֗יִךְ וְשָׂרֽוֹתֵיהֶם֙ מֵינִ֣יקֹתַ֔יִךְ אֲנַחֲמֵֽךְ׃ (לב) בָּנַ֜יִךְ עֻלְּפ֥וּ שָׁכְב֛וּ בְּרֹ֥אשׁ כָּל־ חוּצ֖וֹת כְּת֣וֹא מִכְמָ֑ר הַֽמְלֵאִ֥ים חֲמַת־ יְהוָ֖ה גַּעֲרַ֥ת אֱלֹהָֽיִךְ׃ (לג) לָכֵ֛ן שִׁמְעִי־ נָ֥א זֹ֖את עֲנִיָּ֑ה וּשְׁכֻרַ֖ת וְלֹ֥א מִיָּֽיִן׃ (לד) כֹּֽה־ אָמַ֞ר אֲדֹנַ֣יִךְ יְהוָ֗ה וֵאלֹהַ֙יִךְ֙ יָרִ֣יב עַמּ֔וֹ הִנֵּ֥ה לָקַ֛חְתִּי מִיָּדֵ֖ךְ אֶת־ כּ֣וֹס הַתַּרְעֵלָ֑ה אֶת־ קֻבַּ֙עַת֙ כּ֣וֹס חֲמָתִ֔י לֹא־ תוֹסִ֥יפִי לִשְׁתּוֹתָ֖הּ עֽוֹד׃ (לה) וְשַׂמְתִּ֙יהָ֙ בְּיַד־ מוֹגַ֔יִךְ אֲשֶׁר־ אָמְר֥וּ לְנַפְשֵׁ֖ךְ שְׁחִ֣י וְנַעֲבֹ֑רָה וַתָּשִׂ֤ימִי כָאָ֙רֶץ֙ גֵּוֵ֔ךְ וְכַח֖וּץ לַעֹבְרִֽים׃ (לו) עוּרִ֥י עוּרִ֛י לִבְשִׁ֥י עֻזֵּ֖ךְ צִיּ֑וֹן לִבְשִׁ֣י ׀ בִּגְדֵ֣י תִפְאַרְתֵּ֗ךְ יְרוּשָׁלִַ֙ם֙ עִ֣יר הַקֹּ֔דֶשׁ כִּ֣י לֹ֥א יוֹסִ֛יף יָבֹא־ בָ֥ךְ ע֖וֹד עָרֵ֥ל וְטָמֵֽא׃ (לז) הִתְנַעֲרִ֧י מֵעָפָ֛ר ק֥וּמִי שְּׁבִ֖י יְרֽוּשָׁלִָ֑ם (התפתחו) הִֽתְפַּתְּחִי֙ מוֹסְרֵ֣י צַוָּארֵ֔ךְ שְׁבִיָּ֖ה בַּת־ צִיּֽוֹן׃ (לח) כִּֽי־ כֹה֙ אָמַ֣ר יְהוָ֔ה חִנָּ֖ם נִמְכַּרְתֶּ֑ם וְלֹ֥א בְכֶ֖סֶף תִּגָּאֵֽלוּ׃ (לט) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה מִצְרַ֛יִם יָֽרַד־ עַמִּ֥י בָרִֽאשֹׁנָ֖ה לָג֣וּר שָׁ֑ם וְאַשּׁ֖וּר בְּאֶ֥פֶס עֲשָׁקֽוֹ׃ (מ) וְעַתָּ֤ה (מי)־ (לי)־ מַה־ לִּי־ פֹה֙ נְאֻם־ יְהוָ֔ה כִּֽי־ לֻקַּ֥ח עַמִּ֖י חִנָּ֑ם (משלו) מֹשְׁלָ֤יו יְהֵילִ֙ילוּ֙ נְאֻם־ יְהוָ֔ה וְתָמִ֥יד כָּל־ הַיּ֖וֹם שְׁמִ֥י מִנֹּאָֽץ׃ (מא) לָכֵ֛ן יֵדַ֥ע עַמִּ֖י שְׁמִ֑י לָכֵן֙ בַּיּ֣וֹם הַה֔וּא כִּֽי־ אֲנִי־ ה֥וּא הַֽמְדַבֵּ֖ר הִנֵּֽנִי׃
