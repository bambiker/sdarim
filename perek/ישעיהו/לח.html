ישעיהו סדר טו

(א) וְעַתָּה֙ יְהוָ֣ה אֱלֹהֵ֔ינוּ הוֹשִׁיעֵ֖נוּ מִיָד֑וֹ וְיֵֽדְעוּ֙ כָּל־ מַמְלְכ֣וֹת הָאָ֔רֶץ כִּֽי־ אַתָּ֥ה יְהוָ֖ה לְבַדֶּֽךָ׃ (ב) וַיִּשְׁלַח֙ יְשַֽׁעְיָ֣הוּ בֶן־ אָמ֔וֹץ אֶל־ חִזְקִיָּ֖הוּ לֵאמֹ֑ר כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁר֙ הִתְפַּלַּ֣לְתָּ אֵלַ֔י אֶל־ סַנְחֵרִ֖יב מֶ֥לֶךְ אַשּֽׁוּר׃ (ג) זֶ֣ה הַדָּבָ֔ר אֲשֶׁר־ דִּבֶּ֥ר יְהוָ֖ה עָלָ֑יו בָּזָ֨ה לְךָ֜ לָעֲגָ֣ה לְךָ֗ בְּתוּלַת֙ בַּת־ צִיּ֔וֹן אַחֲרֶ֙יךָ֙ רֹ֣אשׁ הֵנִ֔יעָה בַּ֖ת יְרוּשָׁלִָֽם׃ (ד) אֶת־ מִ֤י חֵרַ֙פְתָּ֙ וְגִדַּ֔פְתָּ וְעַל־ מִ֖י הֲרִימ֣וֹתָה קּ֑וֹל וַתִּשָּׂ֥א מָר֛וֹם עֵינֶ֖יךָ אֶל־ קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (ה) בְּיַ֣ד עֲבָדֶיךָ֮ חֵרַ֣פְתָּ ׀ אֲדֹנָי֒ וַתֹּ֗אמֶר בְּרֹ֥ב רִכְבִּ֛י אֲנִ֥י עָלִ֛יתִי מְר֥וֹם הָרִ֖ים יַרְכְּתֵ֣י לְבָנ֑וֹן וְאֶכְרֹ֞ת קוֹמַ֤ת אֲרָזָיו֙ מִבְחַ֣ר בְּרֹשָׁ֔יו וְאָבוֹא֙ מְר֣וֹם קִצּ֔וֹ יַ֖עַר כַּרְמִלּֽוֹ׃ (ו) אֲנִ֥י קַ֖רְתִּי וְשָׁתִ֣יתִי מָ֑יִם וְאַחְרִב֙ בְּכַף־ פְּעָמַ֔י כֹּ֖ל יְאֹרֵ֥י מָצֽוֹר׃ (ז) הֲלֽוֹא־ שָׁמַ֤עְתָּ לְמֵֽרָחוֹק֙ אוֹתָ֣הּ עָשִׂ֔יתִי מִ֥ימֵי קֶ֖דֶם וִיצַרְתִּ֑יהָ עַתָּ֣ה הֲבֵאתִ֔יהָ וּתְהִ֗י לְהַשְׁא֛וֹת גַּלִּ֥ים נִצִּ֖ים עָרִ֥ים בְּצֻרֽוֹת׃ (ח) וְיֹֽשְׁבֵיהֶן֙ קִצְרֵי־ יָ֔ד חַ֖תּוּ וָבֹ֑שׁוּ הָי֞וּ עֵ֤שֶׂב שָׂדֶה֙ וִ֣ירַק דֶּ֔שֶׁא חֲצִ֣יר גַּגּ֔וֹת וּשְׁדֵמָ֖ה לִפְנֵ֥י קָמָֽה׃ (ט) וְשִׁבְתְּךָ֛ וְצֵאתְךָ֥ וּבוֹאֲךָ֖ יָדָ֑עְתִּי וְאֵ֖ת הִֽתְרַגֶּזְךָ֥ אֵלָֽי׃ (י) יַ֚עַן הִתְרַגֶּזְךָ֣ אֵלַ֔י וְשַׁאֲנַנְךָ֖ עָלָ֣ה בְאָזְנָ֑י וְשַׂמְתִּ֨י חַחִ֜י בְּאַפֶּ֗ךָ וּמִתְגִּי֙ בִּשְׂפָתֶ֔יךָ וַהֲשִׁ֣יבֹתִ֔יךָ בַּדֶּ֖רֶךְ אֲשֶׁר־ בָּ֥אתָ בָּֽהּ׃ (יא) וְזֶה־ לְּךָ֣ הָא֔וֹת אָכ֤וֹל הַשָּׁנָה֙ סָפִ֔יחַ וּבַשָּׁנָ֥ה הַשֵּׁנִ֖ית שָׁחִ֑יס וּבַשָּׁנָ֣ה הַשְּׁלִישִׁ֗ית זִרְע֧וּ וְקִצְר֛וּ וְנִטְע֥וּ כְרָמִ֖ים (ואכול) וְאִכְל֥וּ פִרְיָֽם׃ (יב) וְיָ֨סְפָ֜ה פְּלֵיטַ֧ת בֵּית־ יְהוּדָ֛ה הַנִּשְׁאָרָ֖ה שֹׁ֣רֶשׁ לְמָ֑טָּה וְעָשָׂ֥ה פְרִ֖י לְמָֽעְלָה׃ (יג) כִּ֤י מִירֽוּשָׁלִַ֙ם֙ תֵּצֵ֣א שְׁאֵרִ֔ית וּפְלֵיטָ֖ה מֵהַ֣ר צִיּ֑וֹן קִנְאַ֛ת יְהוָ֥ה צְבָא֖וֹת תַּֽעֲשֶׂה־ זֹּֽאת׃ (יד) לָכֵ֗ן כֹּֽה־ אָמַ֤ר יְהוָה֙ אֶל־ מֶ֣לֶךְ אַשּׁ֔וּר לֹ֤א יָבוֹא֙ אֶל־ הָעִ֣יר הַזֹּ֔את וְלֹֽא־ יוֹרֶ֥ה שָׁ֖ם חֵ֑ץ וְלֹֽא־ יְקַדְּמֶ֣נָּה מָגֵ֔ן וְלֹֽא־ יִשְׁפֹּ֥ךְ עֳלֶ֖יהָ סֹלְלָֽה׃ (טו) בַּדֶּ֥רֶךְ אֲשֶׁר־ בָּ֖א בָּ֣הּ יָשׁ֑וּב וְאֶל־ הָעִ֥יר הַזֹּ֛את לֹ֥א יָב֖וֹא נְאֻם־ יְהוָֽה׃ (טז) וְגַנּוֹתִ֛י עַל־ הָעִ֥יר הַזֹּ֖את לְהֽוֹשִׁיעָ֑הּ לְמַֽעֲנִ֔י וּלְמַ֖עַן דָּוִ֥ד עַבְדִּֽי׃ (יז) וַיֵּצֵ֣א ׀ מַלְאַ֣ךְ יְהוָ֗ה וַיַּכֶּה֙ בְּמַחֲנֵ֣ה אַשּׁ֔וּר מֵאָ֛ה וּשְׁמֹנִ֥ים וַחֲמִשָּׁ֖ה אָ֑לֶף וַיַּשְׁכִּ֣ימוּ בַבֹּ֔קֶר וְהִנֵּ֥ה כֻלָּ֖ם פְּגָרִ֥ים מֵתִֽים׃ (יח) וַיִּסַּ֣ע וַיֵּ֔לֶךְ וַיָּ֖שָׁב סַנְחֵרִ֣יב מֶֽלֶךְ־ אַשּׁ֑וּר וַיֵּ֖שֶׁב בְּנִֽינְוֵֽה׃ (יט) וַיְהִי֩ ה֨וּא מִֽשְׁתַּחֲוֶ֜ה בֵּ֣ית ׀ נִסְרֹ֣ךְ אֱלֹהָ֗יו וְֽאַדְרַמֶּ֨לֶךְ וְשַׂרְאֶ֤צֶר בָּנָיו֙ הִכֻּ֣הוּ בַחֶ֔רֶב וְהֵ֥מָּה נִמְלְט֖וּ אֶ֣רֶץ אֲרָרָ֑ט וַיִּמְלֹ֛ךְ אֵֽסַר־ חַדֹּ֥ן בְּנ֖וֹ תַּחְתָּֽיו׃ תחילת פרק לח (כ) בַּיָּמִ֣ים הָהֵ֔ם חָלָ֥ה חִזְקִיָּ֖הוּ לָמ֑וּת וַיָּב֣וֹא אֵ֠לָיו יְשַׁעְיָ֨הוּ בֶן־ אָמ֜וֹץ הַנָּבִ֗יא וַיֹּ֨אמֶר אֵלָ֜יו כֹּֽה־ אָמַ֤ר יְהוָה֙ צַ֣ו לְבֵיתֶ֔ךָ כִּ֛י מֵ֥ת אַתָּ֖ה וְלֹ֥א תִֽחְיֶֽה׃ (כא) וַיַּסֵּ֧ב חִזְקִיָּ֛הוּ פָּנָ֖יו אֶל־ הַקִּ֑יר וַיִּתְפַּלֵּ֖ל אֶל־ יְהוָֽה׃ (כב) וַיֹּאמַ֗ר אָנָּ֤ה יְהוָה֙ זְכָר־ נָ֞א אֵ֣ת אֲשֶׁ֧ר הִתְהַלַּ֣כְתִּי לְפָנֶ֗יךָ בֶּֽאֱמֶת֙ וּבְלֵ֣ב שָׁלֵ֔ם וְהַטּ֥וֹב בְּעֵינֶ֖יךָ עָשִׂ֑יתִי וַיֵּ֥בְךְּ חִזְקִיָּ֖הוּ בְּכִ֥י גָדֽוֹל׃ (כג) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶֽל־ יְשַׁעְיָ֖הוּ לֵאמֹֽר׃ (כד) הָל֞וֹךְ וְאָמַרְתָּ֣ אֶל־ חִזְקִיָּ֗הוּ כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵי֙ דָּוִ֣ד אָבִ֔יךָ שָׁמַ֙עְתִּי֙ אֶת־ תְּפִלָּתֶ֔ךָ רָאִ֖יתִי אֶת־ דִּמְעָתֶ֑ךָ הִנְנִי֙ יוֹסִ֣ף עַל־ יָמֶ֔יךָ חֲמֵ֥שׁ עֶשְׂרֵ֖ה שָׁנָֽה׃ (כה) וּמִכַּ֤ף מֶֽלֶךְ־ אַשּׁוּר֙ אַצִּ֣ילְךָ֔ וְאֵ֖ת הָעִ֣יר הַזֹּ֑את וְגַנּוֹתִ֖י עַל־ הָעִ֥יר הַזֹּֽאת׃ (כו) וְזֶה־ לְּךָ֥ הָא֖וֹת מֵאֵ֣ת יְהוָ֑ה אֲשֶׁר֙ יַעֲשֶׂ֣ה יְהוָ֔ה אֶת־ הַדָּבָ֥ר הַזֶּ֖ה אֲשֶׁ֥ר דִּבֵּֽר׃ (כז) הִנְנִ֣י מֵשִׁ֣יב אֶת־ צֵ֣ל הַֽמַּעֲל֡וֹת אֲשֶׁ֣ר יָרְדָה֩ בְמַעֲל֨וֹת אָחָ֥ז בַּשֶּׁ֛מֶשׁ אֲחֹרַנִּ֖ית עֶ֣שֶׂר מַעֲל֑וֹת וַתָּ֤שָׁב הַשֶּׁ֙מֶשׁ֙ עֶ֣שֶׂר מַעֲל֔וֹת בַּֽמַּעֲל֖וֹת אֲשֶׁ֥ר יָרָֽדָה׃ (כח) מִכְתָּ֖ב לְחִזְקִיָּ֣הוּ מֶֽלֶךְ־ יְהוּדָ֑ה בַּחֲלֹת֕וֹ וַיְחִ֖י מֵחָלְיֽוֹ׃ (כט) אֲנִ֣י אָמַ֗רְתִּי בִּדְמִ֥י יָמַ֛י אֵלֵ֖כָה בְּשַׁעֲרֵ֣י שְׁא֑וֹל פֻּקַּ֖דְתִּי יֶ֥תֶר שְׁנוֹתָֽי׃ (ל) אָמַ֙רְתִּי֙ לֹא־ אֶרְאֶ֣ה יָ֔הּ יָ֖הּ בְּאֶ֣רֶץ הַחַיִּ֑ים לֹא־ אַבִּ֥יט אָדָ֛ם ע֖וֹד עִם־ י֥וֹשְׁבֵי חָֽדֶל׃ (לא) דּוֹרִ֗י נִסַּ֧ע וְנִגְלָ֛ה מִנִּ֖י כְּאֹ֣הֶל רֹעִ֑י קִפַּ֨דְתִּי כָאֹרֵ֤ג חַיַּי֙ מִדַּלָּ֣ה יְבַצְּעֵ֔נִי מִיּ֥וֹם עַד־ לַ֖יְלָה תַּשְׁלִימֵֽנִי׃ (לב) שִׁוִּ֤יתִי עַד־ בֹּ֙קֶר֙ כָּֽאֲרִ֔י כֵּ֥ן יְשַׁבֵּ֖ר כָּל־ עַצְמוֹתָ֑י מִיּ֥וֹם עַד־ לַ֖יְלָה תַּשְׁלִימֵֽנִי׃ (לג) כְּס֤וּס עָגוּר֙ כֵּ֣ן אֲצַפְצֵ֔ף אֶהְגֶּ֖ה כַּיּוֹנָ֑ה דַּלּ֤וּ עֵינַי֙ לַמָּר֔וֹם אֲדֹנָ֖י עָֽשְׁקָה־ לִּ֥י עָרְבֵֽנִי׃ (לד) מָֽה־ אֲדַבֵּ֥ר וְאָֽמַר־ לִ֖י וְה֣וּא עָשָׂ֑ה אֶדַּדֶּ֥ה כָל־ שְׁנוֹתַ֖י עַל־ מַ֥ר נַפְשִֽׁי׃ (לה) אֲדֹנָ֖י עֲלֵיהֶ֣ם יִֽחְי֑וּ וּלְכָל־ בָּהֶן֙ חַיֵּ֣י רוּחִ֔י וְתַחֲלִימֵ֖נִי וְהַחֲיֵֽנִי׃ (לו) הִנֵּ֥ה לְשָׁל֖וֹם מַר־ לִ֣י מָ֑ר וְאַתָּ֞ה חָשַׁ֤קְתָּ נַפְשִׁי֙ מִשַּׁ֣חַת בְּלִ֔י כִּ֥י הִשְׁלַ֛כְתָּ אַחֲרֵ֥י גֵוְךָ֖ כָּל־ חֲטָאָֽי׃ (לז) כִּ֣י לֹ֥א שְׁא֛וֹל תּוֹדֶ֖ךָּ מָ֣וֶת יְהַלְלֶ֑ךָּ לֹֽא־ יְשַׂבְּר֥וּ יֽוֹרְדֵי־ ב֖וֹר אֶל־ אֲמִתֶּֽךָ׃ (לח) חַ֥י חַ֛י ה֥וּא יוֹדֶ֖ךָ כָּמ֣וֹנִי הַיּ֑וֹם אָ֣ב לְבָנִ֔ים יוֹדִ֖יעַ אֶל־ אֲמִתֶּֽךָ׃ (לט) יְהוָ֖ה לְהוֹשִׁיעֵ֑נִי וּנְגִנוֹתַ֧י נְנַגֵּ֛ן כָּל־ יְמֵ֥י חַיֵּ֖ינוּ עַל־ בֵּ֥ית יְהוָֽה׃ (מ) וַיֹּ֣אמֶר יְשַׁעְיָ֔הוּ יִשְׂא֖וּ דְּבֶ֣לֶת תְּאֵנִ֑ים וְיִמְרְח֥וּ עַֽל־ הַשְּׁחִ֖ין וְיֶֽחִי׃ (מא) וַיֹּ֥אמֶר חִזְקִיָּ֖הוּ מָ֣ה א֑וֹת כִּ֥י אֶעֱלֶ֖ה בֵּ֥ית יְהוָֽה׃ סוף פרק לח (מב) בָּעֵ֣ת הַהִ֡וא שָׁלַ֡ח מְרֹדַ֣ךְ בַּ֠לְאֲדָן בֶּֽן־ בַּלְאֲדָ֧ן מֶֽלֶךְ־ בָּבֶ֛ל סְפָרִ֥ים וּמִנְחָ֖ה אֶל־ חִזְקִיָּ֑הוּ וַיִּשְׁמַ֕ע כִּ֥י חָלָ֖ה וַֽיֶּחֱזָֽק׃ (מג) וַיִּשְׂמַ֣ח עֲלֵיהֶם֮ חִזְקִיָּהוּ֒ וַיַּרְאֵ֣ם אֶת־ בֵּ֣ית (נכתה) נְכֹת֡וֹ אֶת־ הַכֶּסֶף֩ וְאֶת־ הַזָּהָ֨ב וְאֶת־ הַבְּשָׂמִ֜ים וְאֵ֣ת ׀ הַשֶּׁ֣מֶן הַטּ֗וֹב וְאֵת֙ כָּל־ בֵּ֣ית כֵּלָ֔יו וְאֵ֛ת כָּל־ אֲשֶׁ֥ר נִמְצָ֖א בְּאֹֽצְרֹתָ֑יו לֹֽא־ הָיָ֣ה דָבָ֗ר אֲ֠שֶׁר לֹֽא־ הֶרְאָ֧ם חִזְקִיָּ֛הוּ בְּבֵית֖וֹ וּבְכָל־ מֶמְשַׁלְתּֽוֹ׃ (מד) וַיָּבֹא֙ יְשַׁעְיָ֣הוּ הַנָּבִ֔יא אֶל־ הַמֶּ֖לֶךְ חִזְקִיָּ֑הוּ וַיֹּ֨אמֶר אֵלָ֜יו מָ֥ה אָמְר֣וּ ׀ הָאֲנָשִׁ֣ים הָאֵ֗לֶּה וּמֵאַ֙יִן֙ יָבֹ֣אוּ אֵלֶ֔יךָ וַיֹּ֙אמֶר֙ חִזְקִיָּ֔הוּ מֵאֶ֧רֶץ רְחוֹקָ֛ה בָּ֥אוּ אֵלַ֖י מִבָּבֶֽל׃ (מה) וַיֹּ֕אמֶר מָ֥ה רָא֖וּ בְּבֵיתֶ֑ךָ וַיֹּ֣אמֶר חִזְקִיָּ֗הוּ אֵ֣ת כָּל־ אֲשֶׁ֤ר בְּבֵיתִי֙ רָא֔וּ לֹֽא־ הָיָ֥ה דָבָ֛ר אֲשֶׁ֥ר לֹֽא־ הִרְאִיתִ֖ים בְּאוֹצְרֹתָֽי׃ (מו) וַיֹּ֥אמֶר יְשַׁעְיָ֖הוּ אֶל־ חִזְקִיָּ֑הוּ שְׁמַ֖ע דְּבַר־ יְהוָ֥ה צְבָאֽוֹת׃ (מז) הִנֵּה֮ יָמִ֣ים בָּאִים֒ וְנִשָּׂ֣א ׀ כָּל־ אֲשֶׁ֣ר בְּבֵיתֶ֗ךָ וַאֲשֶׁ֨ר אָצְר֧וּ אֲבֹתֶ֛יךָ עַד־ הַיּ֥וֹם הַזֶּ֖ה בָּבֶ֑ל לֹֽא־ יִוָּתֵ֥ר דָּבָ֖ר אָמַ֥ר יְהוָֽה׃ (מח) וּמִבָּנֶ֜יךָ אֲשֶׁ֨ר יֵצְא֧וּ מִמְּךָ֛ אֲשֶׁ֥ר תּוֹלִ֖יד יִקָּ֑חוּ וְהָיוּ֙ סָרִיסִ֔ים בְּהֵיכַ֖ל מֶ֥לֶךְ בָּבֶֽל׃ (מט) וַיֹּ֤אמֶר חִזְקִיָּ֙הוּ֙ אֶֽל־ יְשַׁעְיָ֔הוּ ט֥וֹב דְּבַר־ יְהוָ֖ה אֲשֶׁ֣ר דִּבַּ֑רְתָּ וַיֹּ֕אמֶר כִּ֥י יִהְיֶ֛ה שָׁל֥וֹם וֶאֱמֶ֖ת בְּיָמָֽי׃
