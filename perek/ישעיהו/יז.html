ישעיהו פרק יז לפי החלוקה לסדרים

פרק יז בספר ישעיהו (14 פסוקים) נמצא בישעיהו סדר ז (פסוקים יא–כד).

ישעיהו סדר ז

(א) וְהוּכַ֤ן בַּחֶ֙סֶד֙ כִּסֵּ֔א וְיָשַׁ֥ב עָלָ֛יו בֶּאֱמֶ֖ת בְּאֹ֣הֶל דָּוִ֑ד שֹׁפֵ֛ט וְדֹרֵ֥שׁ מִשְׁפָּ֖ט וּמְהִ֥ר צֶֽדֶק׃ (ב) שָׁמַ֥עְנוּ גְאוֹן־ מוֹאָ֖ב גֵּ֣א מְאֹ֑ד גַּאֲוָת֧וֹ וּגְאוֹנ֛וֹ וְעֶבְרָת֖וֹ לֹא־ כֵ֥ן בַּדָּֽיו׃ (ג) לָכֵ֗ן יְיֵלִ֥יל מוֹאָ֛ב לְמוֹאָ֖ב כֻּלֹּ֣ה יְיֵלִ֑יל לַאֲשִׁישֵׁ֧י קִיר־ חֲרֶ֛שֶׂת תֶּהְגּ֖וּ אַךְ־ נְכָאִֽים׃ (ד) כִּ֣י שַׁדְמוֹת֩ חֶשְׁבּ֨וֹן אֻמְלָ֜ל גֶּ֣פֶן שִׂבְמָ֗ה בַּעֲלֵ֤י גוֹיִם֙ הָלְמ֣וּ שְׂרוּקֶּ֔יהָ עַד־ יַעְזֵ֥ר נָגָ֖עוּ תָּ֣עוּ מִדְבָּ֑ר שְׁלֻ֣חוֹתֶ֔יהָ נִטְּשׁ֖וּ עָ֥בְרוּ יָֽם׃ (ה) עַל־ כֵּ֡ן אֶבְכֶּ֞ה בִּבְכִ֤י יַעְזֵר֙ גֶּ֣פֶן שִׂבְמָ֔ה אֲרַיָּ֙וֶךְ֙ דִּמְעָתִ֔י חֶשְׁבּ֖וֹן וְאֶלְעָלֵ֑ה כִּ֧י עַל־ קֵיצֵ֛ךְ וְעַל־ קְצִירֵ֖ךְ הֵידָ֥ד נָפָֽל׃ (ו) וְנֶאֱסַ֨ף שִׂמְחָ֤ה וָגִיל֙ מִן־ הַכַּרְמֶ֔ל וּבַכְּרָמִ֥ים לֹֽא־ יְרֻנָּ֖ן לֹ֣א יְרֹעָ֑ע יַ֗יִן בַּיְקָבִ֛ים לֹֽא־ יִדְרֹ֥ךְ הַדֹּרֵ֖ךְ הֵידָ֥ד הִשְׁבַּֽתִּי׃ (ז) עַל־ כֵּן֙ מֵעַ֣י לְמוֹאָ֔ב כַּכִּנּ֖וֹר יֶֽהֱמ֑וּ וְקִרְבִּ֖י לְקִ֥יר חָֽרֶשׂ׃ (ח) וְהָיָ֧ה כִֽי־ נִרְאָ֛ה כִּֽי־ נִלְאָ֥ה מוֹאָ֖ב עַל־ הַבָּמָ֑ה וּבָ֧א אֶל־ מִקְדָּשׁ֛וֹ לְהִתְפַּלֵּ֖ל וְלֹ֥א יוּכָֽל׃ (ט) זֶ֣ה הַדָּבָ֗ר אֲשֶׁ֨ר דִּבֶּ֧ר יְהוָ֛ה אֶל־ מוֹאָ֖ב מֵאָֽז׃ (י) וְעַתָּ֗ה דִּבֶּ֣ר יְהוָה֮ לֵאמֹר֒ בְּשָׁלֹ֤שׁ שָׁנִים֙ כִּשְׁנֵ֣י שָׂכִ֔יר וְנִקְלָה֙ כְּב֣וֹד מוֹאָ֔ב בְּכֹ֖ל הֶהָמ֣וֹן הָרָ֑ב וּשְׁאָ֥ר מְעַ֛ט מִזְעָ֖ר ל֥וֹא כַבִּֽיר׃ תחילת פרק יז (יא) מַשָּׂ֖א דַּמָּ֑שֶׂק הִנֵּ֤ה דַמֶּ֙שֶׂק֙ מוּסָ֣ר מֵעִ֔יר וְהָיְתָ֖ה מְעִ֥י מַפָּלָֽה׃ (יב) עֲזֻב֖וֹת עָרֵ֣י עֲרֹעֵ֑ר לַעֲדָרִ֣ים תִּֽהְיֶ֔ינָה וְרָבְצ֖וּ וְאֵ֥ין מַחֲרִֽיד׃ (יג) וְנִשְׁבַּ֤ת מִבְצָר֙ מֵֽאֶפְרַ֔יִם וּמַמְלָכָ֥ה מִדַּמֶּ֖שֶׂק וּשְׁאָ֣ר אֲרָ֑ם כִּכְב֤וֹד בְּנֵֽי־ יִשְׂרָאֵל֙ יִֽהְי֔וּ נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (יד) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא יִדַּ֖ל כְּב֣וֹד יַעֲקֹ֑ב וּמִשְׁמַ֥ן בְּשָׂר֖וֹ יֵרָזֶֽה׃ (טו) וְהָיָ֗ה כֶּֽאֱסֹף֙ קָצִ֣יר קָמָ֔ה וּזְרֹע֖וֹ שִׁבֳּלִ֣ים יִקְצ֑וֹר וְהָיָ֛ה כִּמְלַקֵּ֥ט שִׁבֳּלִ֖ים בְּעֵ֥מֶק רְפָאִֽים׃ (טז) וְנִשְׁאַר־ בּ֤וֹ עֽוֹלֵלֹת֙ כְּנֹ֣קֶף זַ֔יִת שְׁנַ֧יִם שְׁלֹשָׁ֛ה גַּרְגְּרִ֖ים בְּרֹ֣אשׁ אָמִ֑יר אַרְבָּעָ֣ה חֲמִשָּׁ֗ה בִּסְעִפֶ֙יהָ֙ פֹּֽרִיָּ֔ה נְאֻם־ יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (יז) בַּיּ֣וֹם הַה֔וּא יִשְׁעֶ֥ה הָאָדָ֖ם עַל־ עֹשֵׂ֑הוּ וְעֵינָ֕יו אֶל־ קְד֥וֹשׁ יִשְׂרָאֵ֖ל תִּרְאֶֽינָה׃ (יח) וְלֹ֣א יִשְׁעֶ֔ה אֶל־ הַֽמִּזְבְּח֖וֹת מַעֲשֵׂ֣ה יָדָ֑יו וַאֲשֶׁ֨ר עָשׂ֤וּ אֶצְבְּעֹתָיו֙ לֹ֣א יִרְאֶ֔ה וְהָאֲשֵׁרִ֖ים וְהָחַמָּנִֽים׃ (יט) בַּיּ֨וֹם הַה֜וּא יִהְי֣וּ ׀ עָרֵ֣י מָעֻזּ֗וֹ כַּעֲזוּבַ֤ת הַחֹ֙רֶשׁ֙ וְהָ֣אָמִ֔יר אֲשֶׁ֣ר עָזְב֔וּ מִפְּנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל וְהָיְתָ֖ה שְׁמָמָֽה׃ (כ) כִּ֤י שָׁכַ֙חַתְּ֙ אֱלֹהֵ֣י יִשְׁעֵ֔ךְ וְצ֥וּר מָעֻזֵּ֖ךְ לֹ֣א זָכָ֑רְתְּ עַל־ כֵּ֗ן תִּטְּעִי֙ נִטְעֵ֣י נַעֲמָנִ֔ים וּזְמֹ֥רַת זָ֖ר תִּזְרָעֶֽנּוּ׃ (כא) בְּי֤וֹם נִטְעֵךְ֙ תְּשַׂגְשֵׂ֔גִי וּבַבֹּ֖קֶר זַרְעֵ֣ךְ תַּפְרִ֑יחִי נֵ֥ד קָצִ֛יר בְּי֥וֹם נַחֲלָ֖ה וּכְאֵ֥ב אָנֽוּשׁ׃ (כב) ה֗וֹי הֲמוֹן֙ עַמִּ֣ים רַבִּ֔ים כַּהֲמ֥וֹת יַמִּ֖ים יֶהֱמָי֑וּן וּשְׁא֣וֹן לְאֻמִּ֔ים כִּשְׁא֛וֹן מַ֥יִם כַּבִּירִ֖ים יִשָּׁאֽוּן׃ (כג) לְאֻמִּ֗ים כִּשְׁא֞וֹן מַ֤יִם רַבִּים֙ יִשָּׁא֔וּן וְגָ֥עַר בּ֖וֹ וְנָ֣ס מִמֶּרְחָ֑ק וְרֻדַּ֗ף כְּמֹ֤ץ הָרִים֙ לִפְנֵי־ ר֔וּחַ וּכְגַלְגַּ֖ל לִפְנֵ֥י סוּפָֽה׃ (כד) לְעֵ֥ת עֶ֙רֶב֙ וְהִנֵּ֣ה בַלָּהָ֔ה בְּטֶ֥רֶם בֹּ֖קֶר אֵינֶ֑נּוּ זֶ֚ה חֵ֣לֶק שׁוֹסֵ֔ינוּ וְגוֹרָ֖ל לְבֹזְזֵֽינוּ׃ סוף פרק יז (כה) ה֥וֹי אֶ֖רֶץ צִלְצַ֣ל כְּנָפָ֑יִם אֲשֶׁ֥ר מֵעֵ֖בֶר לְנַֽהֲרֵי־ כֽוּשׁ׃ (כו) הַשֹּׁלֵ֨חַ בַּיָּ֜ם צִירִ֗ים וּבִכְלֵי־ גֹמֶא֮ עַל־ פְּנֵי־ מַיִם֒ לְכ֣וּ ׀ מַלְאָכִ֣ים קַלִּ֗ים אֶל־ גּוֹי֙ מְמֻשָּׁ֣ךְ וּמוֹרָ֔ט אֶל־ עַ֥ם נוֹרָ֖א מִן־ ה֣וּא וָהָ֑לְאָה גּ֚וֹי קַו־ קָ֣ו וּמְבוּסָ֔ה אֲשֶׁר־ בָּזְא֥וּ נְהָרִ֖ים אַרְצֽוֹ׃ (כז) כָּל־ יֹשְׁבֵ֥י תֵבֵ֖ל וְשֹׁ֣כְנֵי אָ֑רֶץ כִּנְשֹׂא־ נֵ֤ס הָרִים֙ תִּרְא֔וּ וְכִתְקֹ֥עַ שׁוֹפָ֖ר תִּשְׁמָֽעוּ׃ (כח) כִּי֩ כֹ֨ה אָמַ֤ר יְהוָה֙ אֵלַ֔י (אשקוטה) אֶשְׁקֳטָ֖ה וְאַבִּ֣יטָה בִמְכוֹנִ֑י כְּחֹ֥ם צַח֙ עֲלֵי־ א֔וֹר כְּעָ֥ב טַ֖ל בְּחֹ֥ם קָצִֽיר׃ (כט) כִּֽי־ לִפְנֵ֤י קָצִיר֙ כְּתָם־ פֶּ֔רַח וּבֹ֥סֶר גֹּמֵ֖ל יִֽהְיֶ֣ה נִצָּ֑ה וְכָרַ֤ת הַזַּלְזַלִּים֙ בַּמַּזְמֵר֔וֹת וְאֶת־ הַנְּטִישׁ֖וֹת הֵסִ֥יר הֵתַֽז׃ (ל) יֵעָזְב֤וּ יַחְדָּו֙ לְעֵ֣יט הָרִ֔ים וּֽלְבֶהֱמַ֖ת הָאָ֑רֶץ וְקָ֤ץ עָלָיו֙ הָעַ֔יִט וְכָל־ בֶּהֱמַ֥ת הָאָ֖רֶץ עָלָ֥יו תֶּחֱרָֽף׃ (לא) בָּעֵת֩ הַהִ֨יא יֽוּבַל־ שַׁ֜י לַיהוָ֣ה צְבָא֗וֹת עַ֚ם מְמֻשָּׁ֣ךְ וּמוֹרָ֔ט וּמֵעַ֥ם נוֹרָ֖א מִן־ ה֣וּא וָהָ֑לְאָה גּ֣וֹי ׀ קַו־ קָ֣ו וּמְבוּסָ֗ה אֲשֶׁ֨ר בָּזְא֤וּ נְהָרִים֙ אַרְצ֔וֹ אֶל־ מְק֛וֹם שֵׁם־ יְהוָ֥ה צְבָא֖וֹת הַר־ צִיּֽוֹן׃ (לב) מַשָּׂ֖א מִצְרָ֑יִם הִנֵּ֨ה יְהוָ֜ה רֹכֵ֨ב עַל־ עָ֥ב קַל֙ וּבָ֣א מִצְרַ֔יִם וְנָע֞וּ אֱלִילֵ֤י מִצְרַ֙יִם֙ מִפָּנָ֔יו וּלְבַ֥ב מִצְרַ֖יִם יִמַּ֥ס בְּקִרְבּֽוֹ׃ (לג) וְסִכְסַכְתִּ֤י מִצְרַ֙יִם֙ בְּמִצְרַ֔יִם וְנִלְחֲמ֥וּ אִישׁ־ בְּאָחִ֖יו וְאִ֣ישׁ בְּרֵעֵ֑הוּ עִ֣יר בְּעִ֔יר מַמְלָכָ֖ה בְּמַמְלָכָֽה׃ (לד) וְנָבְקָ֤ה רֽוּחַ־ מִצְרַ֙יִם֙ בְּקִרְבּ֔וֹ וַעֲצָת֖וֹ אֲבַלֵּ֑עַ וְדָרְשׁ֤וּ אֶל־ הָֽאֱלִילִים֙ וְאֶל־ הָ֣אִטִּ֔ים וְאֶל־ הָאֹב֖וֹת וְאֶל־ הַיִּדְּעֹנִֽים׃ (לה) וְסִכַּרְתִּי֙ אֶת־ מִצְרַ֔יִם בְּיַ֖ד אֲדֹנִ֣ים קָשֶׁ֑ה וּמֶ֤לֶךְ עַז֙ יִמְשָׁל־ בָּ֔ם נְאֻ֥ם הָאָד֖וֹן יְהוָ֥ה צְבָאֽוֹת׃ (לו) וְנִשְּׁתוּ־ מַ֖יִם מֵֽהַיָּ֑ם וְנָהָ֖ר יֶחֱרַ֥ב וְיָבֵֽשׁ׃ (לז) וְהֶאֶזְנִ֣יחוּ נְהָר֔וֹת דָּלֲל֥וּ וְחָרְב֖וּ יְאֹרֵ֣י מָצ֑וֹר קָנֶ֥ה וָס֖וּף קָמֵֽלוּ׃ (לח) עָר֥וֹת עַל־ יְא֖וֹר עַל־ פִּ֣י יְא֑וֹר וְכֹל֙ מִזְרַ֣ע יְא֔וֹר יִיבַ֥שׁ נִדַּ֖ף וְאֵינֶֽנּוּ׃ (לט) וְאָנוּ֙ הַדַּיָּגִ֔ים וְאָ֣בְל֔וּ כָּל־ מַשְׁלִיכֵ֥י בַיְא֖וֹר חַכָּ֑ה וּפֹרְשֵׂ֥י מִכְמֹ֛רֶת עַל־ פְּנֵי־ מַ֖יִם אֻמְלָֽלוּ׃ (מ) וּבֹ֛שׁוּ עֹבְדֵ֥י פִשְׁתִּ֖ים שְׂרִיק֑וֹת וְאֹרְגִ֖ים חוֹרָֽי׃ (מא) וְהָי֥וּ שָׁתֹתֶ֖יהָ מְדֻכָּאִ֑ים כָּל־ עֹ֥שֵׂי שֶׂ֖כֶר אַגְמֵי־ נָֽפֶשׁ׃ (מב) אַךְ־ אֱוִלִים֙ שָׂ֣רֵי צֹ֔עַן חַכְמֵי֙ יֹעֲצֵ֣י פַרְעֹ֔ה עֵצָ֖ה נִבְעָרָ֑ה אֵ֚יךְ תֹּאמְר֣וּ אֶל־ פַּרְעֹ֔ה בֶּן־ חֲכָמִ֥ים אֲנִ֖י בֶּן־ מַלְכֵי־ קֶֽדֶם׃ (מג) אַיָּם֙ אֵפ֣וֹא חֲכָמֶ֔יךָ וְיַגִּ֥ידוּ נָ֖א לָ֑ךְ וְיֵ֣דְע֔וּ מַה־ יָּעַ֛ץ יְהוָ֥ה צְבָא֖וֹת עַל־ מִצְרָֽיִם׃ (מד) נֽוֹאֲלוּ֙ שָׂ֣רֵי צֹ֔עַן נִשְּׁא֖וּ שָׂ֣רֵי נֹ֑ף הִתְע֥וּ אֶת־ מִצְרַ֖יִם פִּנַּ֥ת שְׁבָטֶֽיהָ׃ (מה) יְהוָ֛ה מָסַ֥ךְ בְּקִרְבָּ֖הּ ר֣וּחַ עִוְעִ֑ים וְהִתְע֤וּ אֶת־ מִצְרַ֙יִם֙ בְּכָֽל־ מַעֲשֵׂ֔הוּ כְּהִתָּע֥וֹת שִׁכּ֖וֹר בְּקִיאֽוֹ׃ (מו) וְלֹֽא־ יִהְיֶ֥ה לְמִצְרַ֖יִם מַֽעֲשֶׂ֑ה אֲשֶׁ֧ר יַעֲשֶׂ֛ה רֹ֥אשׁ וְזָנָ֖ב כִּפָּ֥ה וְאַגְמֽוֹן׃ (מז) בַּיּ֣וֹם הַה֔וּא יִֽהְיֶ֥ה מִצְרַ֖יִם כַּנָּשִׁ֑ים וְחָרַ֣ד ׀ וּפָחַ֗ד מִפְּנֵי֙ תְּנוּפַת֙ יַד־ יְהוָ֣ה צְבָא֔וֹת אֲשֶׁר־ ה֖וּא מֵנִ֥יף עָלָֽיו׃ (מח) וְ֠הָיְתָה אַדְמַ֨ת יְהוּדָ֤ה לְמִצְרַ֙יִם֙ לְחָגָּ֔א כֹּל֩ אֲשֶׁ֨ר יַזְכִּ֥יר אֹתָ֛הּ אֵלָ֖יו יִפְחָ֑ד מִפְּנֵ֗י עֲצַת֙ יְהוָ֣ה צְבָא֔וֹת אֲשֶׁר־ ה֖וּא יוֹעֵ֥ץ עָלָֽיו׃ (מט) בַּיּ֣וֹם הַה֡וּא יִהְיוּ֩ חָמֵ֨שׁ עָרִ֜ים בְּאֶ֣רֶץ מִצְרַ֗יִם מְדַבְּרוֹת֙ שְׂפַ֣ת כְּנַ֔עַן וְנִשְׁבָּע֖וֹת לַיהוָ֣ה צְבָא֑וֹת עִ֣יר הַהֶ֔רֶס יֵאָמֵ֖ר לְאֶחָֽת׃ (נ) בַּיּ֣וֹם הַה֗וּא יִֽהְיֶ֤ה מִזְבֵּ֙חַ֙ לַֽיהוָ֔ה בְּת֖וֹךְ אֶ֣רֶץ מִצְרָ֑יִם וּמַצֵּבָ֥ה אֵֽצֶל־ גְּבוּלָ֖הּ לַֽיהוָֽה׃ (נא) וְהָיָ֨ה לְא֥וֹת וּלְעֵ֛ד לַֽיהוָ֥ה צְבָא֖וֹת בְּאֶ֣רֶץ מִצְרָ֑יִם כִּֽי־ יִצְעֲק֤וּ אֶל־ יְהוָה֙ מִפְּנֵ֣י לֹֽחֲצִ֔ים וְיִשְׁלַ֥ח לָהֶ֛ם מוֹשִׁ֥יעַ וָרָ֖ב וְהִצִּילָֽם׃ (נב) וְנוֹדַ֤ע יְהוָה֙ לְמִצְרַ֔יִם וְיָדְע֥וּ מִצְרַ֛יִם אֶת־ יְהוָ֖ה בַּיּ֣וֹם הַה֑וּא וְעָֽבְדוּ֙ זֶ֣בַח וּמִנְחָ֔ה וְנָדְרוּ־ נֵ֥דֶר לַֽיהוָ֖ה וְשִׁלֵּֽמוּ׃ (נג) וְנָגַ֧ף יְהוָ֛ה אֶת־ מִצְרַ֖יִם נָגֹ֣ף וְרָפ֑וֹא וְשָׁ֙בוּ֙ עַד־ יְהוָ֔ה וְנֶעְתַּ֥ר לָהֶ֖ם וּרְפָאָֽם׃ (נד) בַּיּ֣וֹם הַה֗וּא תִּהְיֶ֨ה מְסִלָּ֤ה מִמִּצְרַ֙יִם֙ אַשּׁ֔וּרָה וּבָֽא־ אַשּׁ֥וּר בְּמִצְרַ֖יִם וּמִצְרַ֣יִם בְּאַשּׁ֑וּר וְעָבְד֥וּ מִצְרַ֖יִם אֶת־ אַשּֽׁוּר׃ (נה) בַּיּ֣וֹם הַה֗וּא יִהְיֶ֤ה יִשְׂרָאֵל֙ שְׁלִ֣ישִׁיָּ֔ה לְמִצְרַ֖יִם וּלְאַשּׁ֑וּר בְּרָכָ֖ה בְּקֶ֥רֶב הָאָֽרֶץ׃
