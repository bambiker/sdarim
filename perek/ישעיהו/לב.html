ישעיהו פרק לב לפי החלוקה לסדרים

פרק לב בספר ישעיהו (20 פסוקים) נמצא בישעיהו סדר יב (פסוקים מה–סא) ובישעיהו סדר יג (פסוקים א–ג).

ישעיהו סדר יב

(א) כִּ֣י בִ֠רְאֹתוֹ יְלָדָ֞יו מַעֲשֵׂ֥ה יָדַ֛י בְּקִרְבּ֖וֹ יַקְדִּ֣ישֽׁוּ שְׁמִ֑י וְהִקְדִּ֙ישׁוּ֙ אֶת־ קְד֣וֹשׁ יַֽעֲקֹ֔ב וְאֶת־ אֱלֹהֵ֥י יִשְׂרָאֵ֖ל יַעֲרִֽיצוּ׃ (ב) וְיָדְע֥וּ תֹֽעֵי־ ר֖וּחַ בִּינָ֑ה וְרוֹגְנִ֖ים יִלְמְדוּ־ לֶֽקַח׃ (ג) ה֣וֹי בָּנִ֤ים סֽוֹרְרִים֙ נְאֻם־ יְהוָ֔ה לַעֲשׂ֤וֹת עֵצָה֙ וְלֹ֣א מִנִּ֔י וְלִנְסֹ֥ךְ מַסֵּכָ֖ה וְלֹ֣א רוּחִ֑י לְמַ֛עַן סְפ֥וֹת חַטָּ֖את עַל־ חַטָּֽאת׃ (ד) הַהֹלְכִים֙ לָרֶ֣דֶת מִצְרַ֔יִם וּפִ֖י לֹ֣א שָׁאָ֑לוּ לָעוֹז֙ בְּמָע֣וֹז פַּרְעֹ֔ה וְלַחְס֖וֹת בְּצֵ֥ל מִצְרָֽיִם׃ (ה) וְהָיָ֥ה לָכֶ֛ם מָע֥וֹז פַּרְעֹ֖ה לְבֹ֑שֶׁת וְהֶחָס֥וּת בְּצֵל־ מִצְרַ֖יִם לִכְלִמָּֽה׃ (ו) כִּֽי־ הָי֥וּ בְצֹ֖עַן שָׂרָ֑יו וּמַלְאָכָ֖יו חָנֵ֥ס יַגִּֽיעוּ׃ (ז) כֹּ֣ל (הבאיש) הֹבִ֔ישׁ עַל־ עַ֖ם לֹא־ יוֹעִ֣ילוּ לָ֑מוֹ לֹ֤א לְעֵ֙זֶר֙ וְלֹ֣א לְהוֹעִ֔יל כִּ֥י לְבֹ֖שֶׁת וְגַם־ לְחֶרְפָּֽה׃ (ח) מַשָּׂ֖א בַּהֲמ֣וֹת נֶ֑גֶב בְּאֶרֶץ֩ צָרָ֨ה וְצוּקָ֜ה לָבִ֧יא וָלַ֣יִשׁ מֵהֶ֗ם אֶפְעֶה֙ וְשָׂרָ֣ף מְעוֹפֵ֔ף יִשְׂאוּ֩ עַל־ כֶּ֨תֶף עֲיָרִ֜ים חֵֽילֵהֶ֗ם וְעַל־ דַּבֶּ֤שֶׁת גְּמַלִּים֙ אֽוֹצְרֹתָ֔ם עַל־ עַ֖ם לֹ֥א יוֹעִֽילוּ׃ (ט) וּמִצְרַ֕יִם הֶ֥בֶל וָרִ֖יק יַעְזֹ֑רוּ לָכֵן֙ קָרָ֣אתִי לָזֹ֔את רַ֥הַב הֵ֖ם שָֽׁבֶת׃ (י) עַתָּ֗ה בּ֣וֹא כָתְבָ֥הּ עַל־ ל֛וּחַ אִתָּ֖ם וְעַל־ סֵ֣פֶר חֻקָּ֑הּ וּתְהִי֙ לְי֣וֹם אַחֲר֔וֹן לָעַ֖ד עַד־ עוֹלָֽם׃ (יא) כִּ֣י עַ֤ם מְרִי֙ ה֔וּא בָּנִ֖ים כֶּחָשִׁ֑ים בָּנִ֕ים לֹֽא־ אָב֥וּ שְׁמ֖וֹעַ תּוֹרַ֥ת יְהוָֽה׃ (יב) אֲשֶׁ֨ר אָמְר֤וּ לָֽרֹאִים֙ לֹ֣א תִרְא֔וּ וְלַ֣חֹזִ֔ים לֹ֥א תֶחֱזוּ־ לָ֖נוּ נְכֹח֑וֹת דַּבְּרוּ־ לָ֣נוּ חֲלָק֔וֹת חֲז֖וּ מַהֲתַלּֽוֹת׃ (יג) ס֚וּרוּ מִנֵּי־ דֶ֔רֶךְ הַטּ֖וּ מִנֵּי־ אֹ֑רַח הַשְׁבִּ֥יתוּ מִפָּנֵ֖ינוּ אֶת־ קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (יד) לָכֵ֗ן כֹּ֤ה אָמַר֙ קְד֣וֹשׁ יִשְׂרָאֵ֔ל יַ֥עַן מָֽאָסְכֶ֖ם בַּדָּבָ֣ר הַזֶּ֑ה וַֽתִּבְטְחוּ֙ בְּעֹ֣שֶׁק וְנָל֔וֹז וַתִּֽשָּׁעֲנ֖וּ עָלָֽיו׃ (טו) לָכֵ֗ן יִֽהְיֶ֤ה לָכֶם֙ הֶעָוֺ֣ן הַזֶּ֔ה כְּפֶ֣רֶץ נֹפֵ֔ל נִבְעֶ֖ה בְּחוֹמָ֣ה נִשְׂגָּבָ֑ה אֲשֶׁר־ פִּתְאֹ֥ם לְפֶ֖תַע יָב֥וֹא שִׁבְרָֽהּ׃ (טז) וּ֠שְׁבָרָהּ כְּשֵׁ֨בֶר נֵ֧בֶל יוֹצְרִ֛ים כָּת֖וּת לֹ֣א יַחְמֹ֑ל וְלֹֽא־ יִמָּצֵ֤א בִמְכִתָּתוֹ֙ חֶ֔רֶשׂ לַחְתּ֥וֹת אֵשׁ֙ מִיָּק֔וּד וְלַחְשֹׂ֥ף מַ֖יִם מִגֶּֽבֶא׃ (יז) כִּ֣י כֹֽה־ אָמַר֩ אֲדֹנָ֨י יְהוִ֜ה קְד֣וֹשׁ יִשְׂרָאֵ֗ל בְּשׁוּבָ֤ה וָנַ֙חַת֙ תִּוָּ֣שֵׁע֔וּן בְּהַשְׁקֵט֙ וּבְבִטְחָ֔ה תִּהְיֶ֖ה גְּבֽוּרַתְכֶ֑ם וְלֹ֖א אֲבִיתֶֽם׃ (יח) וַתֹּ֨אמְר֥וּ לֹא־ כִ֛י עַל־ ס֥וּס נָנ֖וּס עַל־ כֵּ֣ן תְּנוּס֑וּן וְעַל־ קַ֣ל נִרְכָּ֔ב עַל־ כֵּ֖ן יִקַּ֥לּוּ רֹדְפֵיכֶֽם׃ (יט) אֶ֣לֶף אֶחָ֗ד מִפְּנֵי֙ גַּעֲרַ֣ת אֶחָ֔ד מִפְּנֵ֛י גַּעֲרַ֥ת חֲמִשָּׁ֖ה תָּנֻ֑סוּ עַ֣ד אִם־ נוֹתַרְתֶּ֗ם כַּתֹּ֙רֶן֙ עַל־ רֹ֣אשׁ הָהָ֔ר וְכַנֵּ֖ס עַל־ הַגִּבְעָֽה׃ (כ) וְלָכֵ֞ן יְחַכֶּ֤ה יְהוָה֙ לַֽחֲנַנְכֶ֔ם וְלָכֵ֥ן יָר֖וּם לְרַֽחֶמְכֶ֑ם כִּֽי־ אֱלֹהֵ֤י מִשְׁפָּט֙ יְהוָ֔ה אַשְׁרֵ֖י כָּל־ ח֥וֹכֵי לֽוֹ׃ (כא) כִּי־ עַ֛ם בְּצִיּ֥וֹן יֵשֵׁ֖ב בִּירֽוּשָׁלִָ֑ם בָּכ֣וֹ לֹֽא־ תִבְכֶּ֗ה חָנ֤וֹן יָחְנְךָ֙ לְק֣וֹל זַעֲקֶ֔ךָ כְּשָׁמְעָת֖וֹ עָנָֽךְ׃ (כב) וְנָתַ֨ן לָכֶ֧ם אֲדֹנָ֛י לֶ֥חֶם צָ֖ר וּמַ֣יִם לָ֑חַץ וְלֹֽא־ יִכָּנֵ֥ף עוֹד֙ מוֹרֶ֔יךָ וְהָי֥וּ עֵינֶ֖יךָ רֹא֥וֹת אֶת־ מוֹרֶֽיךָ׃ (כג) וְאָזְנֶ֙יךָ֙ תִּשְׁמַ֣עְנָה דָבָ֔ר מֵֽאַחֲרֶ֖יךָ לֵאמֹ֑ר זֶ֤ה הַדֶּ֙רֶךְ֙ לְכ֣וּ ב֔וֹ כִּ֥י תַאֲמִ֖ינוּ וְכִ֥י תַשְׂמְאִֽילוּ׃ (כד) וְטִמֵּאתֶ֗ם אֶת־ צִפּוּי֙ פְּסִילֵ֣י כַסְפֶּ֔ךָ וְאֶת־ אֲפֻדַּ֖ת מַסֵּכַ֣ת זְהָבֶ֑ךָ תִּזְרֵם֙ כְּמ֣וֹ דָוָ֔ה צֵ֖א תֹּ֥אמַר לֽוֹ׃ (כה) וְנָתַן֩ מְטַ֨ר זַרְעֲךָ֜ אֲשֶׁר־ תִּזְרַ֣ע אֶת־ הָאֲדָמָ֗ה וְלֶ֙חֶם֙ תְּבוּאַ֣ת הָֽאֲדָמָ֔ה וְהָיָ֥ה דָשֵׁ֖ן וְשָׁמֵ֑ן יִרְעֶ֥ה מִקְנֶ֛יךָ בַּיּ֥וֹם הַה֖וּא כַּ֥ר נִרְחָֽב׃ (כו) וְהָאֲלָפִ֣ים וְהָעֲיָרִ֗ים עֹֽבְדֵי֙ הָֽאֲדָמָ֔ה בְּלִ֥יל חָמִ֖יץ יֹאכֵ֑לוּ אֲשֶׁר־ זֹרֶ֥ה בָרַ֖חַת וּבַמִּזְרֶֽה׃ (כז) וְהָיָ֣ה ׀ עַל־ כָּל־ הַ֣ר גָּבֹ֗הַ וְעַל֙ כָּל־ גִּבְעָ֣ה נִשָּׂאָ֔ה פְּלָגִ֖ים יִבְלֵי־ מָ֑יִם בְּיוֹם֙ הֶ֣רֶג רָ֔ב בִּנְפֹ֖ל מִגְדָּלִֽים׃ (כח) וְהָיָ֤ה אֽוֹר־ הַלְּבָנָה֙ כְּא֣וֹר הַֽחַמָּ֔ה וְא֤וֹר הַֽחַמָּה֙ יִהְיֶ֣ה שִׁבְעָתַ֔יִם כְּא֖וֹר שִׁבְעַ֣ת הַיָּמִ֑ים בְּי֗וֹם חֲבֹ֤שׁ יְהוָה֙ אֶת־ שֶׁ֣בֶר עַמּ֔וֹ וּמַ֥חַץ מַכָּת֖וֹ יִרְפָּֽא׃ (כט) הִנֵּ֤ה שֵׁם־ יְהוָה֙ בָּ֣א מִמֶּרְחָ֔ק בֹּעֵ֣ר אַפּ֔וֹ וְכֹ֖בֶד מַשָּׂאָ֑ה שְׂפָתָיו֙ מָ֣לְאוּ זַ֔עַם וּלְשׁוֹנ֖וֹ כְּאֵ֥שׁ אֹכָֽלֶת׃ (ל) וְרוּח֞וֹ כְּנַ֤חַל שׁוֹטֵף֙ עַד־ צַוָּ֣אר יֶֽחֱצֶ֔ה לַהֲנָפָ֥ה גוֹיִ֖ם בְּנָ֣פַת שָׁ֑וְא וְרֶ֣סֶן מַתְעֶ֔ה עַ֖ל לְחָיֵ֥י עַמִּֽים׃ (לא) הַשִּׁיר֙ יִֽהְיֶ֣ה לָכֶ֔ם כְּלֵ֖יל הִתְקַדֶּשׁ־ חָ֑ג וְשִׂמְחַ֣ת לֵבָ֗ב כַּֽהוֹלֵךְ֙ בֶּֽחָלִ֔יל לָב֥וֹא בְהַר־ יְהוָ֖ה אֶל־ צ֥וּר יִשְׂרָאֵֽל׃ (לב) וְהִשְׁמִ֨יעַ יְהוָ֜ה אֶת־ ה֣וֹד קוֹל֗וֹ וְנַ֤חַת זְרוֹעוֹ֙ יַרְאֶ֔ה בְּזַ֣עַף אַ֔ף וְלַ֖הַב אֵ֣שׁ אוֹכֵלָ֑ה נֶ֥פֶץ וָזֶ֖רֶם וְאֶ֥בֶן בָּרָֽד׃ (לג) כִּֽי־ מִקּ֥וֹל יְהוָ֖ה יֵחַ֣ת אַשּׁ֑וּר בַּשֵּׁ֖בֶט יַכֶּֽה׃ (לד) וְהָיָ֗ה כֹּ֤ל מַֽעֲבַר֙ מַטֵּ֣ה מֽוּסָדָ֔ה אֲשֶׁ֨ר יָנִ֤יחַ יְהוָה֙ עָלָ֔יו בְּתֻפִּ֖ים וּבְכִנֹּר֑וֹת וּבְמִלְחֲמ֥וֹת תְּנוּפָ֖ה נִלְחַם־ (בה) בָּֽם׃ (לה) כִּֽי־ עָר֤וּךְ מֵֽאֶתְמוּל֙ תָּפְתֶּ֔ה גַּם־ (הוא) הִ֛יא לַמֶּ֥לֶךְ הוּכָ֖ן הֶעְמִ֣יק הִרְחִ֑ב מְדֻרָתָ֗הּ אֵ֤שׁ וְעֵצִים֙ הַרְבֵּ֔ה נִשְׁמַ֤ת יְהוָה֙ כְּנַ֣חַל גָּפְרִ֔ית בֹּעֲרָ֖ה בָּֽהּ׃ (לו) ה֣וֹי הַיֹּרְדִ֤ים מִצְרַ֙יִם֙ לְעֶזְרָ֔ה עַל־ סוּסִ֖ים יִשָּׁעֵ֑נוּ וַיִּבְטְח֨וּ עַל־ רֶ֜כֶב כִּ֣י רָ֗ב וְעַ֤ל פָּֽרָשִׁים֙ כִּֽי־ עָצְמ֣וּ מְאֹ֔ד וְלֹ֤א שָׁעוּ֙ עַל־ קְד֣וֹשׁ יִשְׂרָאֵ֔ל וְאֶת־ יְהוָ֖ה לֹ֥א דָרָֽשׁוּ׃ (לז) וְגַם־ ה֤וּא חָכָם֙ וַיָּ֣בֵא רָ֔ע וְאֶת־ דְּבָרָ֖יו לֹ֣א הֵסִ֑יר וְקָם֙ עַל־ בֵּ֣ית מְרֵעִ֔ים וְעַל־ עֶזְרַ֖ת פֹּ֥עֲלֵי אָֽוֶן׃ (לח) וּמִצְרַ֤יִם אָדָם֙ וְֽלֹא־ אֵ֔ל וְסוּסֵיהֶ֥ם בָּשָׂ֖ר וְלֹא־ ר֑וּחַ וַֽיהוָ֞ה יַטֶּ֣ה יָד֗וֹ וְכָשַׁ֤ל עוֹזֵר֙ וְנָפַ֣ל עָזֻ֔ר וְיַחְדָּ֖ו בַּל־ יִנָּתֵֽקוּ׃ (כד) כִּ֣י אִם־ שָׁ֞ם אַדִּ֤יר יְהוָה֙ לָ֔נוּ מְקוֹם־ נְהָרִ֥ים יְאֹרִ֖ים רַחֲבֵ֣י יָדָ֑יִם בַּל־ תֵּ֤לֶךְ בּוֹ֙ אֳנִי־ שַׁ֔יִט וְצִ֥י אַדִּ֖יר לֹ֥א יַעַבְרֶֽנּוּ׃ (כה) כִּ֤י יְהוָה֙ שֹׁפְטֵ֔נוּ יְהוָ֖ה מְחֹקְקֵ֑נוּ יְהוָ֥ה מַלְכֵּ֖נוּ ה֥וּא יוֹשִׁיעֵֽנוּ׃ (כו) נִטְּשׁ֖וּ חֲבָלָ֑יִךְ בַּל־ יְחַזְּק֤וּ כֵן־ תָּרְנָם֙ בַּל־ פָּ֣רְשׂוּ נֵ֔ס אָ֣ז חֻלַּ֤ק עַֽד־ שָׁלָל֙ מַרְבֶּ֔ה פִּסְחִ֖ים בָּ֥זְזוּ בַֽז׃ (כז) וּבַל־ יֹאמַ֥ר שָׁכֵ֖ן חָלִ֑יתִי הָעָ֛ם הַיֹּשֵׁ֥ב בָּ֖הּ נְשֻׂ֥א עָוֺֽן׃ (כח) קִרְב֤וּ גוֹיִם֙ לִשְׁמֹ֔עַ וּלְאֻמִּ֖ים הַקְשִׁ֑יבוּ תִּשְׁמַ֤ע הָאָ֙רֶץ֙ וּמְלֹאָ֔הּ תֵּבֵ֖ל וְכָל־ צֶאֱצָאֶֽיהָ׃ (כט) כִּ֣י קֶ֤צֶף לַֽיהוָה֙ עַל־ כָּל־ הַגּוֹיִ֔ם וְחֵמָ֖ה עַל־ כָּל־ צְבָאָ֑ם הֶחֱרִימָ֖ם נְתָנָ֥ם לַטָּֽבַח׃ (ל) וְחַלְלֵיהֶ֣ם יֻשְׁלָ֔כוּ וּפִגְרֵיהֶ֖ם יַעֲלֶ֣ה בָאְשָׁ֑ם וְנָמַ֥סּוּ הָרִ֖ים מִדָּמָֽם׃ (לא) וְנָמַ֙קּוּ֙ כָּל־ צְבָ֣א הַשָּׁמַ֔יִם וְנָגֹ֥לּוּ כַסֵּ֖פֶר הַשָּׁמָ֑יִם וְכָל־ צְבָאָ֣ם יִבּ֔וֹל כִּנְבֹ֤ל עָלֶה֙ מִגֶּ֔פֶן וּכְנֹבֶ֖לֶת מִתְּאֵנָֽה׃ (לב) כִּֽי־ רִוְּתָ֥ה בַשָּׁמַ֖יִם חַרְבִּ֑י הִנֵּה֙ עַל־ אֱד֣וֹם תֵּרֵ֔ד וְעַל־ עַ֥ם חֶרְמִ֖י לְמִשְׁפָּֽט׃ (לג) חֶ֣רֶב לַיהוָ֞ה מָלְאָ֥ה דָם֙ הֻדַּ֣שְׁנָה מֵחֵ֔לֶב מִדַּ֤ם כָּרִים֙ וְעַתּוּדִ֔ים מֵחֵ֖לֶב כִּלְי֣וֹת אֵילִ֑ים כִּ֣י זֶ֤בַח לַֽיהוָה֙ בְּבָצְרָ֔ה וְטֶ֥בַח גָּד֖וֹל בְּאֶ֥רֶץ אֱדֽוֹם׃ (לד) וְיָרְד֤וּ רְאֵמִים֙ עִמָּ֔ם וּפָרִ֖ים עִם־ אַבִּירִ֑ים וְרִוְּתָ֤ה אַרְצָם֙ מִדָּ֔ם וַעֲפָרָ֖ם מֵחֵ֥לֶב יְדֻשָּֽׁן׃ (לה) כִּ֛י י֥וֹם נָקָ֖ם לַֽיהוָ֑ה שְׁנַ֥ת שִׁלּוּמִ֖ים לְרִ֥יב צִיּֽוֹן׃ (לו) וְנֶהֶפְכ֤וּ נְחָלֶ֙יהָ֙ לְזֶ֔פֶת וַעֲפָרָ֖הּ לְגָפְרִ֑ית וְהָיְתָ֣ה אַרְצָ֔הּ לְזֶ֖פֶת בֹּעֵרָֽה׃ (לז) לַ֤יְלָה וְיוֹמָם֙ לֹ֣א תִכְבֶּ֔ה לְעוֹלָ֖ם יַעֲלֶ֣ה עֲשָׁנָ֑הּ מִדּ֤וֹר לָדוֹר֙ תֶּחֱרָ֔ב לְנֵ֣צַח נְצָחִ֔ים אֵ֥ין עֹבֵ֖ר בָּֽהּ׃ (לח) וִירֵשׁ֙וּהָ֙ קָאַ֣ת וְקִפּ֔וֹד וְיַנְשׁ֥וֹף וְעֹרֵ֖ב יִשְׁכְּנוּ־ בָ֑הּ וְנָטָ֥ה עָלֶ֛יהָ קַֽו־ תֹ֖הוּ וְאַבְנֵי־ בֹֽהוּ׃ (לט) חֹרֶ֥יהָ וְאֵֽין־ שָׁ֖ם מְלוּכָ֣ה יִקְרָ֑אוּ וְכָל־ שָׂרֶ֖יהָ יִ֥הְיוּ אָֽפֶס׃ (מ) וְעָלְתָ֤ה אַרְמְנֹתֶ֙יהָ֙ סִירִ֔ים קִמּ֥וֹשׂ וָח֖וֹחַ בְּמִבְצָרֶ֑יהָ וְהָיְתָה֙ נְוֵ֣ה תַנִּ֔ים חָצִ֖יר לִבְנ֥וֹת יַעֲנָֽה׃ (מא) וּפָגְשׁ֤וּ צִיִּים֙ אֶת־ אִיִּ֔ים וְשָׂעִ֖יר עַל־ רֵעֵ֣הוּ יִקְרָ֑א אַךְ־ שָׁם֙ הִרְגִּ֣יעָה לִּילִ֔ית וּמָצְאָ֥ה לָ֖הּ מָנֽוֹחַ׃ (מב) שָׁ֣מָּה קִנְּנָ֤ה קִפּוֹז֙ וַתְּמַלֵּ֔ט וּבָקְעָ֖ה וְדָגְרָ֣ה בְצִלָּ֑הּ אַךְ־ שָׁ֛ם נִקְבְּצ֥וּ דַיּ֖וֹת אִשָּׁ֥ה רְעוּתָֽהּ׃ (מג) דִּרְשׁ֨וּ מֵֽעַל־ סֵ֤פֶר יְהוָה֙ וּֽקְרָ֔אוּ אַחַ֤ת מֵהֵ֙נָּה֙ לֹ֣א נֶעְדָּ֔רָה אִשָּׁ֥ה רְעוּתָ֖הּ לֹ֣א פָקָ֑דוּ כִּֽי־ פִי֙ ה֣וּא צִוָּ֔ה וְרוּח֖וֹ ה֥וּא קִבְּצָֽן׃ (מד) וְהֽוּא־ הִפִּ֤יל לָהֶן֙ גּוֹרָ֔ל וְיָד֛וֹ חִלְּקַ֥תָּה לָהֶ֖ם בַּקָּ֑ו עַד־ עוֹלָם֙ יִֽירָשׁ֔וּהָ לְד֥וֹר וָד֖וֹר יִשְׁכְּנוּ־ בָֽהּ׃ (מה) יְשֻׂשׂ֥וּם מִדְבָּ֖ר וְצִיָּ֑ה וְתָגֵ֧ל עֲרָבָ֛ה וְתִפְרַ֖ח כַּחֲבַצָּֽלֶת׃ (מו) פָּרֹ֨חַ תִּפְרַ֜ח וְתָגֵ֗ל אַ֚ף גִּילַ֣ת וְרַנֵּ֔ן כְּב֤וֹד הַלְּבָנוֹן֙ נִתַּן־ לָ֔הּ הֲדַ֥ר הַכַּרְמֶ֖ל וְהַשָּׁר֑וֹן הֵ֛מָּה יִרְא֥וּ כְבוֹד־ יְהוָ֖ה הֲדַ֥ר אֱלֹהֵֽינוּ׃ (מז) חַזְּק֖וּ יָדַ֣יִם רָפ֑וֹת וּבִרְכַּ֥יִם כֹּשְׁל֖וֹת אַמֵּֽצוּ׃ (מח) אִמְרוּ֙ לְנִמְהֲרֵי־ לֵ֔ב חִזְק֖וּ אַל־ תִּירָ֑אוּ הִנֵּ֤ה אֱלֹֽהֵיכֶם֙ נָקָ֣ם יָב֔וֹא גְּמ֣וּל אֱלֹהִ֔ים ה֥וּא יָב֖וֹא וְיֹשַׁעֲכֶֽם׃ (מט) אָ֥ז תִּפָּקַ֖חְנָה עֵינֵ֣י עִוְרִ֑ים וְאָזְנֵ֥י חֵרְשִׁ֖ים תִּפָּתַֽחְנָה׃ (נ) אָ֣ז יְדַלֵּ֤ג כָּֽאַיָּל֙ פִּסֵּ֔חַ וְתָרֹ֖ן לְשׁ֣וֹן אִלֵּ֑ם כִּֽי־ נִבְקְע֤וּ בַמִּדְבָּר֙ מַ֔יִם וּנְחָלִ֖ים בָּעֲרָבָֽה׃ (נא) וְהָיָ֤ה הַשָּׁרָב֙ לַאֲגַ֔ם וְצִמָּא֖וֹן לְמַבּ֣וּעֵי מָ֑יִם בִּנְוֵ֤ה תַנִּים֙ רִבְצָ֔הּ חָצִ֖יר לְקָנֶ֥ה וָגֹֽמֶא׃ (נב) וְהָיָה־ שָׁ֞ם מַסְל֣וּל וָדֶ֗רֶךְ וְדֶ֤רֶךְ הַקֹּ֙דֶשׁ֙ יִקָּ֣רֵא לָ֔הּ לֹֽא־ יַעַבְרֶ֥נּוּ טָמֵ֖א וְהוּא־ לָ֑מוֹ הֹלֵ֥ךְ דֶּ֛רֶךְ וֶאֱוִילִ֖ים לֹ֥א יִתְעֽוּ׃ (נג) לֹא־ יִהְיֶ֨ה שָׁ֜ם אַרְיֵ֗ה וּפְרִ֤יץ חַיּוֹת֙ בַּֽל־ יַעֲלֶ֔נָּה לֹ֥א תִמָּצֵ֖א שָׁ֑ם וְהָלְכ֖וּ גְּאוּלִֽים׃
