ישעיהו פרק לז לפי החלוקה לסדרים

פרק לז בספר ישעיהו (38 פסוקים) נמצא בישעיהו סדר יד (פסוקים כד–מב) ובישעיהו סדר טו (פסוקים א–יט).

ישעיהו סדר יד

(א) וּפְדוּיֵ֨י יְהוָ֜ה יְשֻׁב֗וּן וּבָ֤אוּ צִיּוֹן֙ בְּרִנָּ֔ה וְשִׂמְחַ֥ת עוֹלָ֖ם עַל־ רֹאשָׁ֑ם שָׂשׂ֤וֹן וְשִׂמְחָה֙ יַשִּׂ֔יגוּ וְנָ֖סוּ יָג֥וֹן וַאֲנָחָֽה׃ (ב) וַיְהִ֡י בְּאַרְבַּע֩ עֶשְׂרֵ֨ה שָׁנָ֜ה לַמֶּ֣לֶךְ חִזְקִיָּ֗הוּ עָלָ֞ה סַנְחֵרִ֤יב מֶֽלֶךְ־ אַשּׁוּר֙ עַ֣ל כָּל־ עָרֵ֧י יְהוּדָ֛ה הַבְּצֻר֖וֹת וַֽיִּתְפְּשֵֽׂם׃ (ג) וַיִּשְׁלַ֣ח מֶֽלֶךְ־ אַשּׁ֣וּר ׀ אֶת־ רַב־ שָׁקֵ֨ה מִלָּכִ֧ישׁ יְרוּשָׁלְַ֛מָה אֶל־ הַמֶּ֥לֶךְ חִזְקִיָּ֖הוּ בְּחֵ֣יל כָּבֵ֑ד וַֽיַּעֲמֹ֗ד בִּתְעָלַת֙ הַבְּרֵכָ֣ה הָעֶלְיוֹנָ֔ה בִּמְסִלַּ֖ת שְׂדֵ֥ה כוֹבֵֽס׃ (ד) וַיֵּצֵ֥א אֵלָ֛יו אֶלְיָקִ֥ים בֶּן־ חִלְקִיָּ֖הוּ אֲשֶׁ֣ר עַל־ הַבָּ֑יִת וְשֶׁבְנָא֙ הַסֹּפֵ֔ר וְיוֹאָ֥ח בֶּן־ אָסָ֖ף הַמַּזְכִּֽיר׃ (ה) וַיֹּ֤אמֶר אֲלֵיהֶם֙ רַב־ שָׁקֵ֔ה אִמְרוּ־ נָ֖א אֶל־ חִזְקִיָּ֑הוּ כֹּֽה־ אָמַ֞ר הַמֶּ֤לֶךְ הַגָּדוֹל֙ מֶ֣לֶךְ אַשּׁ֔וּר מָ֧ה הַבִּטָּח֛וֹן הַזֶּ֖ה אֲשֶׁ֥ר בָּטָֽחְתָּ׃ (ו) אָמַ֙רְתִּי֙ אַךְ־ דְּבַר־ שְׂפָתַ֔יִם עֵצָ֥ה וּגְבוּרָ֖ה לַמִּלְחָמָ֑ה עַתָּה֙ עַל־ מִ֣י בָטַ֔חְתָּ כִּ֥י מָרַ֖דְתָּ בִּֽי׃ (ז) הִנֵּ֣ה בָטַ֡חְתָּ עַל־ מִשְׁעֶנֶת֩ הַקָּנֶ֨ה הָרָצ֤וּץ הַזֶּה֙ עַל־ מִצְרַ֔יִם אֲשֶׁ֨ר יִסָּמֵ֥ךְ אִישׁ֙ עָלָ֔יו וּבָ֥א בְכַפּ֖וֹ וּנְקָבָ֑הּ כֵּ֚ן פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם לְכָֽל־ הַבֹּטְחִ֖ים עָלָֽיו׃ (ח) וְכִי־ תֹאמַ֣ר אֵלַ֔י אֶל־ יְהוָ֥ה אֱלֹהֵ֖ינוּ בָּטָ֑חְנוּ הֲלוֹא־ ה֗וּא אֲשֶׁ֨ר הֵסִ֤יר חִזְקִיָּ֙הוּ֙ אֶת־ בָּמֹתָ֣יו וְאֶת־ מִזְבְּחֹתָ֔יו וַיֹּ֤אמֶר לִֽיהוּדָה֙ וְלִיר֣וּשָׁלִַ֔ם לִפְנֵ֛י הַמִּזְבֵּ֥חַ הַזֶּ֖ה תִּֽשְׁתַּחֲוֽוּ׃ (ט) וְעַתָּה֙ הִתְעָ֣רֶב נָ֔א אֶת־ אֲדֹנִ֖י הַמֶּ֣לֶךְ אַשּׁ֑וּר וְאֶתְּנָ֤ה לְךָ֙ אַלְפַּ֣יִם סוּסִ֔ים אִם־ תּוּכַ֕ל לָ֥תֶת לְךָ֖ רֹכְבִ֥ים עֲלֵיהֶֽם׃ (י) וְאֵ֣יךְ תָּשִׁ֗יב אֵ֠ת פְּנֵ֨י פַחַ֥ת אַחַ֛ד עַבְדֵ֥י אֲדֹנִ֖י הַקְטַנִּ֑ים וַתִּבְטַ֤ח לְךָ֙ עַל־ מִצְרַ֔יִם לְרֶ֖כֶב וּלְפָרָשִֽׁים׃ (יא) וְעַתָּה֙ הֲמִבַּלְעֲדֵ֣י יְהוָ֔ה עָלִ֛יתִי עַל־ הָאָ֥רֶץ הַזֹּ֖את לְהַשְׁחִיתָ֑הּ יְהוָה֙ אָמַ֣ר אֵלַ֔י עֲלֵ֛ה אֶל־ הָאָ֥רֶץ הַזֹּ֖את וְהַשְׁחִיתָֽהּ׃ (יב) וַיֹּ֣אמֶר אֶלְיָקִים֩ וְשֶׁבְנָ֨א וְיוֹאָ֜ח אֶל־ רַב־ שָׁקֵ֗ה דַּבֶּר־ נָ֤א אֶל־ עֲבָדֶ֙יךָ֙ אֲרָמִ֔ית כִּ֥י שֹׁמְעִ֖ים אֲנָ֑חְנוּ וְאַל־ תְּדַבֵּ֤ר אֵלֵ֙ינוּ֙ יְהוּדִ֔ית בְּאָזְנֵ֣י הָעָ֔ם אֲשֶׁ֖ר עַל־ הַחוֹמָֽה׃ (יג) וַיֹּ֣אמֶר רַב־ שָׁקֵ֗ה הַאֶ֨ל אֲדֹנֶ֤יךָ וְאֵלֶ֙יךָ֙ שְׁלָחַ֣נִי אֲדֹנִ֔י לְדַבֵּ֖ר אֶת־ הַדְּבָרִ֣ים הָאֵ֑לֶּה הֲלֹ֣א עַל־ הָאֲנָשִׁ֗ים הַיֹּֽשְׁבִים֙ עַל־ הַ֣חוֹמָ֔ה לֶאֱכֹ֣ל אֶת־ (חראיהם) צוֹאָתָ֗ם וְלִשְׁתּ֛וֹת אֶת־ (שיניהם) מֵימֵ֥י רַגְלֵיהֶ֖ם עִמָּכֶֽם׃ (יד) וַֽיַּעֲמֹד֙ רַב־ שָׁקֵ֔ה וַיִּקְרָ֥א בְקוֹל־ גָּד֖וֹל יְהוּדִ֑ית וַיֹּ֕אמֶר שִׁמְע֗וּ אֶת־ דִּבְרֵ֛י הַמֶּ֥לֶךְ הַגָּד֖וֹל מֶ֥לֶךְ אַשּֽׁוּר׃ (טו) כֹּ֚ה אָמַ֣ר הַמֶּ֔לֶךְ אַל־ יַשִּׁ֥א לָכֶ֖ם חִזְקִיָּ֑הוּ כִּ֥י לֹֽא־ יוּכַ֖ל לְהַצִּ֥יל אֶתְכֶֽם׃ (טז) וְאַל־ יַבְטַ֨ח אֶתְכֶ֤ם חִזְקִיָּ֙הוּ֙ אֶל־ יְהוָ֣ה לֵאמֹ֔ר הַצֵּ֥ל יַצִּילֵ֖נוּ יְהוָ֑ה לֹ֤א תִנָּתֵן֙ הָעִ֣יר הַזֹּ֔את בְּיַ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (יז) אַֽל־ תִּשְׁמְע֖וּ אֶל־ חִזְקִיָּ֑הוּ כִּי֩ כֹ֨ה אָמַ֜ר הַמֶּ֣לֶךְ אַשּׁ֗וּר עֲשֽׂוּ־ אִתִּ֤י בְרָכָה֙ וּצְא֣וּ אֵלַ֔י וְאִכְל֤וּ אִישׁ־ גַּפְנוֹ֙ וְאִ֣ישׁ תְּאֵנָת֔וֹ וּשְׁת֖וּ אִ֥ישׁ מֵי־ בוֹרֽוֹ׃ (יח) עַד־ בֹּאִ֕י וְלָקַחְתִּ֥י אֶתְכֶ֖ם אֶל־ אֶ֣רֶץ כְּאַרְצְכֶ֑ם אֶ֤רֶץ דָּגָן֙ וְתִיר֔וֹשׁ אֶ֥רֶץ לֶ֖חֶם וּכְרָמִֽים׃ (יט) פֶּן־ יַסִּ֨ית אֶתְכֶ֤ם חִזְקִיָּ֙הוּ֙ לֵאמֹ֔ר יְהוָ֖ה יַצִּילֵ֑נוּ הַהִצִּ֜ילוּ אֱלֹהֵ֤י הַגּוֹיִם֙ אִ֣ישׁ אֶת־ אַרְצ֔וֹ מִיַּ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (כ) אַיֵּ֞ה אֱלֹהֵ֤י חֲמָת֙ וְאַרְפָּ֔ד אַיֵּ֖ה אֱלֹהֵ֣י סְפַרְוָ֑יִם וְכִֽי־ הִצִּ֥ילוּ אֶת־ שֹׁמְר֖וֹן מִיָּדִֽי׃ (כא) מִ֗י בְּכָל־ אֱלֹהֵ֤י הָֽאֲרָצוֹת֙ הָאֵ֔לֶּה אֲשֶׁר־ הִצִּ֥ילוּ אֶת־ אַרְצָ֖ם מִיָּדִ֑י כִּֽי־ יַצִּ֧יל יְהוָ֛ה אֶת־ יְרוּשָׁלִַ֖ם מִיָּדִֽי׃ (כב) וַֽיַּחֲרִ֔ישׁוּ וְלֹֽא־ עָנ֥וּ אֹת֖וֹ דָּבָ֑ר כִּֽי־ מִצְוַ֨ת הַמֶּ֥לֶךְ הִ֛יא לֵאמֹ֖ר לֹ֥א תַעֲנֻֽהוּ׃ (כג) וַיָּבֹ֣א אֶלְיָקִ֣ים בֶּן־ חִלְקִיָּ֣הוּ אֲשֶׁר־ עַל־ הַ֠בַּיִת וְשֶׁבְנָ֨א הַסּוֹפֵ֜ר וְיוֹאָ֨ח בֶּן־ אָסָ֧ף הַמַּזְכִּ֛יר אֶל־ חִזְקִיָּ֖הוּ קְרוּעֵ֣י בְגָדִ֑ים וַיַּגִּ֣ידוּ ל֔וֹ אֵ֖ת דִּבְרֵ֥י רַב־ שָׁקֵֽה׃ תחילת פרק לז (כד) וַיְהִ֗י כִּשְׁמֹ֙עַ֙ הַמֶּ֣לֶךְ חִזְקִיָּ֔הוּ וַיִּקְרַ֖ע אֶת־ בְּגָדָ֑יו וַיִּתְכַּ֣ס בַּשָּׂ֔ק וַיָּבֹ֖א בֵּ֥ית יְהוָֽה׃ (כה) וַ֠יִּשְׁלַח אֶת־ אֶלְיָקִ֨ים אֲשֶׁר־ עַל־ הַבַּ֜יִת וְאֵ֣ת ׀ שֶׁבְנָ֣א הַסּוֹפֵ֗ר וְאֵת֙ זִקְנֵ֣י הַכֹּהֲנִ֔ים מִתְכַּסִּ֖ים בַּשַּׂקִּ֑ים אֶל־ יְשַֽׁעְיָ֥הוּ בֶן־ אָמ֖וֹץ הַנָּבִֽיא׃ (כו) וַיֹּאמְר֣וּ אֵלָ֗יו כֹּ֚ה אָמַ֣ר חִזְקִיָּ֔הוּ יוֹם־ צָרָ֧ה וְתוֹכֵחָ֛ה וּנְאָצָ֖ה הַיּ֣וֹם הַזֶּ֑ה כִּ֣י בָ֤אוּ בָנִים֙ עַד־ מַשְׁבֵּ֔ר וְכֹ֥חַ אַ֖יִן לְלֵדָֽה׃ (כז) אוּלַ֡י יִשְׁמַע֩ יְהוָ֨ה אֱלֹהֶ֜יךָ אֵ֣ת ׀ דִּבְרֵ֣י רַב־ שָׁקֵ֗ה אֲשֶׁר֩ שְׁלָח֨וֹ מֶֽלֶךְ־ אַשּׁ֤וּר ׀ אֲדֹנָיו֙ לְחָרֵף֙ אֱלֹהִ֣ים חַ֔י וְהוֹכִ֙יחַ֙ בַּדְּבָרִ֔ים אֲשֶׁ֥ר שָׁמַ֖ע יְהוָ֣ה אֱלֹהֶ֑יךָ וְנָשָׂ֣אתָ תְפִלָּ֔ה בְּעַ֥ד הַשְּׁאֵרִ֖ית הַנִּמְצָאָֽה׃ (כח) וַיָּבֹ֗אוּ עַבְדֵ֛י הַמֶּ֥לֶךְ חִזְקִיָּ֖הוּ אֶל־ יְשַׁעְיָֽהוּ׃ (כט) וַיֹּ֤אמֶר אֲלֵיהֶם֙ יְשַֽׁעְיָ֔הוּ כֹּ֥ה תֹאמְר֖וּן אֶל־ אֲדֹנֵיכֶ֑ם כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ תִּירָא֙ מִפְּנֵ֤י הַדְּבָרִים֙ אֲשֶׁ֣ר שָׁמַ֔עְתָּ אֲשֶׁ֧ר גִּדְּפ֛וּ נַעֲרֵ֥י מֶלֶךְ־ אַשּׁ֖וּר אוֹתִֽי׃ (ל) הִנְנִ֨י נוֹתֵ֥ן בּוֹ֙ ר֔וּחַ וְשָׁמַ֥ע שְׁמוּעָ֖ה וְשָׁ֣ב אֶל־ אַרְצ֑וֹ וְהִפַּלְתִּ֥יו בַּחֶ֖רֶב בְּאַרְצֽוֹ׃ (לא) וַיָּ֙שָׁב֙ רַב־ שָׁקֵ֔ה וַיִּמְצָא֙ אֶת־ מֶ֣לֶךְ אַשּׁ֔וּר נִלְחָ֖ם עַל־ לִבְנָ֑ה כִּ֣י שָׁמַ֔ע כִּ֥י נָסַ֖ע מִלָּכִֽישׁ׃ (לב) וַיִּשְׁמַ֗ע עַל־ תִּרְהָ֤קָה מֶֽלֶךְ־ כּוּשׁ֙ לֵאמֹ֔ר יָצָ֖א לְהִלָּחֵ֣ם אִתָּ֑ךְ וַיִּשְׁמַע֙ וַיִּשְׁלַ֣ח מַלְאָכִ֔ים אֶל־ חִזְקִיָּ֖הוּ לֵאמֹֽר׃ (לג) כֹּ֣ה תֹאמְר֗וּן אֶל־ חִזְקִיָּ֤הוּ מֶֽלֶךְ־ יְהוּדָה֙ לֵאמֹ֔ר אַל־ יַשִּֽׁאֲךָ֣ אֱלֹהֶ֔יךָ אֲשֶׁ֥ר אַתָּ֛ה בּוֹטֵ֥חַ בּ֖וֹ לֵאמֹ֑ר לֹ֤א תִנָּתֵן֙ יְר֣וּשָׁלִַ֔ם בְּיַ֖ד מֶ֥לֶךְ אַשּֽׁוּר׃ (לד) הִנֵּ֣ה ׀ אַתָּ֣ה הַזֶּ֖ה אֲשֶׁ֥ר דִּבֵּֽר׃ (כז) הִנְנִ֣י מֵשִׁ֣יב אֶת־ צֵ֣ל הַֽמַּעֲל֡וֹת אֲשֶׁ֣ר יָרְדָה֩ בְמַעֲל֨וֹת אָחָ֥ז בַּשֶּׁ֛מֶשׁ אֲחֹרַנִּ֖ית עֶ֣שֶׂר מַעֲל֑וֹת וַתָּ֤שָׁב הַשֶּׁ֙מֶשׁ֙ עֶ֣שֶׂר מַעֲל֔וֹת בַּֽמַּעֲל֖וֹת אֲשֶׁ֥ר יָרָֽדָה׃ (כח) מִכְתָּ֖ב לְחִזְקִיָּ֣הוּ מֶֽלֶךְ־ יְהוּדָ֑ה בַּחֲלֹת֕וֹ וַיְחִ֖י מֵחָלְיֽוֹ׃ (כט) אֲנִ֣י אָמַ֗רְתִּי בִּדְמִ֥י יָמַ֛י אֵלֵ֖כָה בְּשַׁעֲרֵ֣י שְׁא֑וֹל פֻּקַּ֖דְתִּי יֶ֥תֶר שְׁנוֹתָֽי׃ (ל) אָמַ֙רְתִּי֙ לֹא־ אֶרְאֶ֣ה יָ֔הּ יָ֖הּ בְּאֶ֣רֶץ הַחַיִּ֑ים לֹא־ אַבִּ֥יט אָדָ֛ם ע֖וֹד עִם־ י֥וֹשְׁבֵי חָֽדֶל׃ (לא) דּוֹרִ֗י נִסַּ֧ע וְנִגְלָ֛ה מִנִּ֖י כְּאֹ֣הֶל רֹעִ֑י קִפַּ֨דְתִּי כָאֹרֵ֤ג חַיַּי֙ מִדַּלָּ֣ה יְבַצְּעֵ֔נִי מִיּ֥וֹם עַד־ לַ֖יְלָה תַּשְׁלִימֵֽנִי׃ (לב) שִׁוִּ֤יתִי עַד־ בֹּ֙קֶר֙ כָּֽאֲרִ֔י כֵּ֥ן יְשַׁבֵּ֖ר כָּל־ עַצְמוֹתָ֑י מִיּ֥וֹם עַד־ לַ֖יְלָה תַּשְׁלִימֵֽנִי׃ (לג) כְּס֤וּס עָגוּר֙ כֵּ֣ן אֲצַפְצֵ֔ף אֶהְגֶּ֖ה כַּיּוֹנָ֑ה דַּלּ֤וּ עֵינַי֙ לַמָּר֔וֹם אֲדֹנָ֖י עָֽשְׁקָה־ לִּ֥י עָרְבֵֽנִי׃ (לד) מָֽה־ אֲדַבֵּ֥ר וְאָֽמַר־ לִ֖י וְה֣וּא עָשָׂ֑ה אֶדַּדֶּ֥ה כָל־ שְׁנוֹתַ֖י עַל־ מַ֥ר נַפְשִֽׁי׃ (לה) אֲדֹנָ֖י עֲלֵיהֶ֣ם יִֽחְי֑וּ וּלְכָל־ בָּהֶן֙ חַיֵּ֣י רוּחִ֔י וְתַחֲלִימֵ֖נִי וְהַחֲיֵֽנִי׃ (לו) הִנֵּ֥ה לְשָׁל֖וֹם מַר־ לִ֣י מָ֑ר וְאַתָּ֞ה חָשַׁ֤קְתָּ נַפְשִׁי֙ מִשַּׁ֣חַת בְּלִ֔י כִּ֥י הִשְׁלַ֛כְתָּ אַחֲרֵ֥י גֵוְךָ֖ כָּל־ חֲטָאָֽי׃ (לז) כִּ֣י לֹ֥א שְׁא֛וֹל תּוֹדֶ֖ךָּ מָ֣וֶת יְהַלְלֶ֑ךָּ לֹֽא־ יְשַׂבְּר֥וּ יֽוֹרְדֵי־ ב֖וֹר אֶל־ אֲמִתֶּֽךָ׃ (לח) חַ֥י חַ֛י ה֥וּא יוֹדֶ֖ךָ כָּמ֣וֹנִי הַיּ֑וֹם אָ֣ב לְבָנִ֔ים יוֹדִ֖יעַ אֶל־ אֲמִתֶּֽךָ׃ (לט) יְהוָ֖ה לְהוֹשִׁיעֵ֑נִי וּנְגִנוֹתַ֧י נְנַגֵּ֛ן כָּל־ יְמֵ֥י חַיֵּ֖ינוּ עַל־ בֵּ֥ית יְהוָֽה׃ (מ) וַיֹּ֣אמֶר יְשַׁעְיָ֔הוּ יִשְׂא֖וּ דְּבֶ֣לֶת תְּאֵנִ֑ים וְיִמְרְח֥וּ עַֽל־ הַשְּׁחִ֖ין וְיֶֽחִי׃ (מא) וַיֹּ֥אמֶר חִזְקִיָּ֖הוּ מָ֣ה א֑וֹת כִּ֥י אֶעֱלֶ֖ה בֵּ֥ית יְהוָֽה׃ (מב) בָּעֵ֣ת הַהִ֡וא שָׁלַ֡ח מְרֹדַ֣ךְ בַּ֠לְאֲדָן בֶּֽן־ בַּלְאֲדָ֧ן מֶֽלֶךְ־ בָּבֶ֛ל סְפָרִ֥ים וּמִנְחָ֖ה אֶל־ חִזְקִיָּ֑הוּ וַיִּשְׁמַ֕ע כִּ֥י חָלָ֖ה וַֽיֶּחֱזָֽק׃ (מג) וַיִּשְׂמַ֣ח עֲלֵיהֶם֮ חִזְקִיָּהוּ֒ וַיַּרְאֵ֣ם אֶת־ בֵּ֣ית (נכתה) נְכֹת֡וֹ אֶת־ הַכֶּסֶף֩ וְאֶת־ הַזָּהָ֨ב וְאֶת־ הַבְּשָׂמִ֜ים וְאֵ֣ת ׀ הַשֶּׁ֣מֶן הַטּ֗וֹב וְאֵת֙ כָּל־ בֵּ֣ית כֵּלָ֔יו וְאֵ֛ת כָּל־ אֲשֶׁ֥ר נִמְצָ֖א בְּאֹֽצְרֹתָ֑יו לֹֽא־ הָיָ֣ה דָבָ֗ר אֲ֠שֶׁר לֹֽא־ הֶרְאָ֧ם חִזְקִיָּ֛הוּ בְּבֵית֖וֹ וּבְכָל־ מֶמְשַׁלְתּֽוֹ׃ (מד) וַיָּבֹא֙ יְשַׁעְיָ֣הוּ הַנָּבִ֔יא אֶל־ הַמֶּ֖לֶךְ חִזְקִיָּ֑הוּ וַיֹּ֨אמֶר אֵלָ֜יו מָ֥ה אָמְר֣וּ ׀ הָאֲנָשִׁ֣ים הָאֵ֗לֶּה וּמֵאַ֙יִן֙ יָבֹ֣אוּ אֵלֶ֔יךָ וַיֹּ֙אמֶר֙ חִזְקִיָּ֔הוּ מֵאֶ֧רֶץ רְחוֹקָ֛ה בָּ֥אוּ אֵלַ֖י מִבָּבֶֽל׃ (מה) וַיֹּ֕אמֶר מָ֥ה רָא֖וּ בְּבֵיתֶ֑ךָ וַיֹּ֣אמֶר חִזְקִיָּ֗הוּ אֵ֣ת כָּל־ אֲשֶׁ֤ר בְּבֵיתִי֙ רָא֔וּ לֹֽא־ הָיָ֥ה דָבָ֛ר אֲשֶׁ֥ר לֹֽא־ הִרְאִיתִ֖ים בְּאוֹצְרֹתָֽי׃ (מו) וַיֹּ֥אמֶר יְשַׁעְיָ֖הוּ אֶל־ חִזְקִיָּ֑הוּ שְׁמַ֖ע דְּבַר־ יְהוָ֥ה צְבָאֽוֹת׃ (מז) הִנֵּה֮ יָמִ֣ים בָּאִים֒ וְנִשָּׂ֣א ׀ כָּל־ אֲשֶׁ֣ר בְּבֵיתֶ֗ךָ וַאֲשֶׁ֨ר אָצְר֧וּ אֲבֹתֶ֛יךָ עַד־ הַיּ֥וֹם הַזֶּ֖ה בָּבֶ֑ל לֹֽא־ יִוָּתֵ֥ר דָּבָ֖ר אָמַ֥ר יְהוָֽה׃ (מח) וּמִבָּנֶ֜יךָ אֲשֶׁ֨ר יֵצְא֧וּ מִמְּךָ֛ אֲשֶׁ֥ר תּוֹלִ֖יד יִקָּ֑חוּ וְהָיוּ֙ סָרִיסִ֔ים בְּהֵיכַ֖ל מֶ֥לֶךְ בָּבֶֽל׃ (מט) וַיֹּ֤אמֶר חִזְקִיָּ֙הוּ֙ אֶֽל־ יְשַׁעְיָ֔הוּ ט֥וֹב דְּבַר־ יְהוָ֖ה אֲשֶׁ֣ר דִּבַּ֑רְתָּ וַיֹּ֕אמֶר כִּ֥י יִהְיֶ֛ה שָׁל֥וֹם וֶאֱמֶ֖ת בְּיָמָֽי׃
