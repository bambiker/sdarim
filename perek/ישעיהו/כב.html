ישעיהו פרק כב לפי החלוקה לסדרים

פרק כב בספר ישעיהו (25 פסוקים) נמצא בישעיהו סדר ח (פסוקים כה–מו) ובישעיהו סדר ט (פסוקים א–ג).

ישעיהו סדר ח

(א) אֲשֶׁ֧ר בֵּרֲכ֛וֹ יְהוָ֥ה צְבָא֖וֹת לֵאמֹ֑ר בָּר֨וּךְ עַמִּ֜י מִצְרַ֗יִם וּמַעֲשֵׂ֤ה יָדַי֙ אַשּׁ֔וּר וְנַחֲלָתִ֖י יִשְׂרָאֵֽל׃ (ב) בִּשְׁנַ֨ת בֹּ֤א תַרְתָּן֙ אַשְׁדּ֔וֹדָה בִּשְׁלֹ֣ח אֹת֔וֹ סַֽרְג֖וֹן מֶ֣לֶךְ אַשּׁ֑וּר וַיִּלָּ֥חֶם בְּאַשְׁדּ֖וֹד וַֽיִּלְכְּדָֽהּ׃ (ג) בָּעֵ֣ת הַהִ֗יא דִּבֶּ֣ר יְהוָה֮ בְּיַ֣ד יְשַׁעְיָ֣הוּ בֶן־ אָמוֹץ֮ לֵאמֹר֒ לֵ֗ךְ וּפִתַּחְתָּ֤ הַשַּׂק֙ מֵעַ֣ל מָתְנֶ֔יךָ וְנַעַלְךָ֥ תַחֲלֹ֖ץ מֵעַ֣ל רַגְלֶ֑יךָ וַיַּ֣עַשׂ כֵּ֔ן הָלֹ֖ךְ עָר֥וֹם וְיָחֵֽף׃ (ד) וַיֹּ֣אמֶר יְהוָ֔ה כַּאֲשֶׁ֥ר הָלַ֛ךְ עַבְדִּ֥י יְשַׁעְיָ֖הוּ עָר֣וֹם וְיָחֵ֑ף שָׁלֹ֤שׁ שָׁנִים֙ א֣וֹת וּמוֹפֵ֔ת עַל־ מִצְרַ֖יִם וְעַל־ כּֽוּשׁ׃ (ה) כֵּ֣ן יִנְהַ֣ג מֶֽלֶךְ־ אַ֠שּׁוּר אֶת־ שְׁבִ֨י מִצְרַ֜יִם וְאֶת־ גָּל֥וּת כּ֛וּשׁ נְעָרִ֥ים וּזְקֵנִ֖ים עָר֣וֹם וְיָחֵ֑ף וַחֲשׂוּפַ֥י שֵׁ֖ת עֶרְוַ֥ת מִצְרָֽיִם׃ (ו) וְחַתּ֖וּ וָבֹ֑שׁוּ מִכּוּשׁ֙ מַבָּטָ֔ם וּמִן־ מִצְרַ֖יִם תִּפְאַרְתָּֽם׃ (ז) וְ֠אָמַר יֹשֵׁ֨ב הָאִ֣י הַזֶּה֮ בַּיּ֣וֹם הַהוּא֒ הִנֵּה־ כֹ֣ה מַבָּטֵ֗נוּ אֲשֶׁר־ נַ֤סְנוּ שָׁם֙ לְעֶזְרָ֔ה לְהִ֨נָּצֵ֔ל מִפְּנֵ֖י מֶ֣לֶךְ אַשּׁ֑וּר וְאֵ֖יךְ נִמָּלֵ֥ט אֲנָֽחְנוּ׃ (ח) מַשָּׂ֖א מִדְבַּר־ יָ֑ם כְּסוּפ֤וֹת בַּנֶּ֙גֶב֙ לַֽחֲלֹ֔ף מִמִּדְבָּ֣ר בָּ֔א מֵאֶ֖רֶץ נוֹרָאָֽה׃ (ט) חָז֥וּת קָשָׁ֖ה הֻגַּד־ לִ֑י הַבּוֹגֵ֤ד ׀ בּוֹגֵד֙ וְהַשּׁוֹדֵ֣ד ׀ שׁוֹדֵ֔ד עֲלִ֤י עֵילָם֙ צוּרִ֣י מָדַ֔י כָּל־ אַנְחָתָ֖ה הִשְׁבַּֽתִּי׃ (י) עַל־ כֵּ֗ן מָלְא֤וּ מָתְנַי֙ חַלְחָלָ֔ה צִירִ֣ים אֲחָז֔וּנִי כְּצִירֵ֖י יֽוֹלֵדָ֑ה נַעֲוֵ֣יתִי מִשְּׁמֹ֔עַ נִבְהַ֖לְתִּי מֵרְאֽוֹת׃ (יא) תָּעָ֣ה לְבָבִ֔י פַּלָּצ֖וּת בִּֽעֲתָ֑תְנִי אֵ֚ת נֶ֣שֶׁף חִשְׁקִ֔י שָׂ֥ם לִ֖י לַחֲרָדָֽה׃ (יב) עָרֹ֧ךְ הַשֻּׁלְחָ֛ן צָפֹ֥ה הַצָּפִ֖ית אָכ֣וֹל שָׁתֹ֑ה ק֥וּמוּ הַשָּׂרִ֖ים מִשְׁח֥וּ מָגֵֽן׃ (יג) כִּ֣י כֹ֥ה אָמַ֛ר אֵלַ֖י אֲדֹנָ֑י לֵ֚ךְ הַעֲמֵ֣ד הַֽמְצַפֶּ֔ה אֲשֶׁ֥ר יִרְאֶ֖ה יַגִּֽיד׃ (יד) וְרָ֣אָה רֶ֗כֶב צֶ֚מֶד פָּֽרָשִׁ֔ים רֶ֥כֶב חֲמ֖וֹר רֶ֣כֶב גָּמָ֑ל וְהִקְשִׁ֥יב קֶ֖שֶׁב רַב־ קָֽשֶׁב׃ (טו) וַיִּקְרָ֖א אַרְיֵ֑ה עַל־ מִצְפֶּ֣ה ׀ אֲדֹנָ֗י אָנֹכִ֞י עֹמֵ֤ד תָּמִיד֙ יוֹמָ֔ם וְעַל־ מִ֨שְׁמַרְתִּ֔י אָנֹכִ֥י נִצָּ֖ב כָּל־ הַלֵּילֽוֹת׃ (טז) וְהִנֵּה־ זֶ֥ה בָא֙ רֶ֣כֶב אִ֔ישׁ צֶ֖מֶד פָּֽרָשִׁ֑ים וַיַּ֣עַן וַיֹּ֗אמֶר נָפְלָ֤ה נָֽפְלָה֙ בָּבֶ֔ל וְכָל־ פְּסִילֵ֥י אֱלֹהֶ֖יהָ שִׁבַּ֥ר לָאָֽרֶץ׃ (יז) מְדֻשָׁתִ֖י וּבֶן־ גָּרְנִ֑י אֲשֶׁ֣ר שָׁמַ֗עְתִּי מֵאֵ֨ת יְהוָ֧ה צְבָא֛וֹת אֱלֹהֵ֥י יִשְׂרָאֵ֖ל הִגַּ֥דְתִּי לָכֶֽם׃ (יח) מַשָּׂ֖א דּוּמָ֑ה אֵלַי֙ קֹרֵ֣א מִשֵּׂעִ֔יר שֹׁמֵר֙ מַה־ מִלַּ֔יְלָה שֹׁמֵ֖ר מַה־ מִלֵּֽיל׃ (יט) אָמַ֣ר שֹׁמֵ֔ר אָתָ֥ה בֹ֖קֶר וְגַם־ לָ֑יְלָה אִם־ תִּבְעָי֥וּן בְּעָ֖יוּ שֻׁ֥בוּ אֵתָֽיוּ׃ (כ) מַשָּׂ֖א בַּעְרָ֑ב בַּיַּ֤עַר בַּעְרַב֙ תָּלִ֔ינוּ אֹֽרְח֖וֹת דְּדָנִֽים׃ (כא) לִקְרַ֥את צָמֵ֖א הֵתָ֣יוּ מָ֑יִם יֹשְׁבֵי֙ אֶ֣רֶץ תֵּימָ֔א בְּלַחְמ֖וֹ קִדְּמ֥וּ נֹדֵֽד׃ (כב) כִּֽי־ מִפְּנֵ֥י חֲרָב֖וֹת נָדָ֑דוּ מִפְּנֵ֣י ׀ חֶ֣רֶב נְטוּשָׁ֗ה וּמִפְּנֵי֙ קֶ֣שֶׁת דְּרוּכָ֔ה וּמִפְּנֵ֖י כֹּ֥בֶד מִלְחָמָֽה׃ (כג) כִּי־ כֹ֛ה אָמַ֥ר אֲדֹנָ֖י אֵלָ֑י בְּע֤וֹד שָׁנָה֙ כִּשְׁנֵ֣י שָׂכִ֔יר וְכָלָ֖ה כָּל־ כְּב֥וֹד קֵדָֽר׃ (כד) וּשְׁאָ֧ר מִסְפַּר־ קֶ֛שֶׁת גִּבּוֹרֵ֥י בְנֵֽי־ קֵדָ֖ר יִמְעָ֑טוּ כִּ֛י יְהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל דִּבֵּֽר׃ תחילת פרק כב (כה) מַשָּׂ֖א גֵּ֣יא חִזָּי֑וֹן מַה־ לָּ֣ךְ אֵפ֔וֹא כִּֽי־ עָלִ֥ית כֻּלָּ֖ךְ לַגַּגּֽוֹת׃ (כו) תְּשֻׁא֣וֹת ׀ מְלֵאָ֗ה עִ֚יר הֽוֹמִיָּ֔ה קִרְיָ֖ה עַלִּיזָ֑ה חֲלָלַ֙יִךְ֙ לֹ֣א חַלְלֵי־ חֶ֔רֶב וְלֹ֖א מֵתֵ֥י מִלְחָמָֽה׃ (כז) כָּל־ קְצִינַ֥יִךְ נָֽדְדוּ־ יַ֖חַד מִקֶּ֣שֶׁת אֻסָּ֑רוּ כָּל־ נִמְצָאַ֙יִךְ֙ אֻסְּר֣וּ יַחְדָּ֔ו מֵרָח֖וֹק בָּרָֽחוּ׃ (כח) עַל־ כֵּ֥ן אָמַ֛רְתִּי שְׁע֥וּ מִנִּ֖י אֲמָרֵ֣ר בַּבֶּ֑כִי אַל־ תָּאִ֣יצוּ לְנַֽחֲמֵ֔נִי עַל־ שֹׁ֖ד בַּת־ עַמִּֽי׃ (כט) כִּ֣י יוֹם֩ מְהוּמָ֨ה וּמְבוּסָ֜ה וּמְבוּכָ֗ה לַֽאדֹנָ֧י יְהוִ֛ה צְבָא֖וֹת בְּגֵ֣יא חִזָּי֑וֹן מְקַרְקַ֥ר קִ֖ר וְשׁ֥וֹעַ אֶל־ הָהָֽר׃ (ל) וְעֵילָם֙ נָשָׂ֣א אַשְׁפָּ֔ה בְּרֶ֥כֶב אָדָ֖ם פָּֽרָשִׁ֑ים וְקִ֥יר עֵרָ֖ה מָגֵֽן׃ (לא) וַיְהִ֥י מִבְחַר־ עֲמָקַ֖יִךְ מָ֣לְאוּ רָ֑כֶב וְהַפָּ֣רָשִׁ֔ים שֹׁ֖ת שָׁ֥תוּ הַשָּֽׁעְרָה׃ (לב) וַיְגַ֕ל אֵ֖ת מָסַ֣ךְ יְהוּדָ֑ה וַתַּבֵּט֙ בַּיּ֣וֹם הַה֔וּא אֶל־ נֶ֖שֶׁק בֵּ֥ית הַיָּֽעַר׃ (לג) וְאֵ֨ת בְּקִיעֵ֧י עִיר־ דָּוִ֛ד רְאִיתֶ֖ם כִּי־ רָ֑בּוּ וַֽתְּקַבְּצ֔וּ אֶת־ מֵ֥י הַבְּרֵכָ֖ה הַתַּחְתּוֹנָֽה׃ (לד) וְאֶת־ בָּתֵּ֥י יְרוּשָׁלִַ֖ם סְפַרְתֶּ֑ם וַתִּתְֿצוּ֙ הַבָּ֣תִּ֔ים לְבַצֵּ֖ר הַחוֹמָֽה׃ (לה) וּמִקְוָ֣ה ׀ עֲשִׂיתֶ֗ם בֵּ֚ין הַחֹ֣מֹתַ֔יִם לְמֵ֖י הַבְּרֵכָ֣ה הַיְשָׁנָ֑ה וְלֹ֤א הִבַּטְתֶּם֙ אֶל־ עֹשֶׂ֔יהָ וְיֹצְרָ֥הּ מֵֽרָח֖וֹק לֹ֥א רְאִיתֶֽם׃ (לו) וַיִּקְרָ֗א אֲדֹנָ֧י יְהוִ֛ה צְבָא֖וֹת בַּיּ֣וֹם הַה֑וּא לִבְכִי֙ וּלְמִסְפֵּ֔ד וּלְקָרְחָ֖ה וְלַחֲגֹ֥ר שָֽׂק׃ (לז) וְהִנֵּ֣ה ׀ שָׂשׂ֣וֹן וְשִׂמְחָ֗ה הָרֹ֤ג ׀ בָּקָר֙ וְשָׁחֹ֣ט צֹ֔אן אָכֹ֥ל בָּשָׂ֖ר וְשָׁת֣וֹת יָ֑יִן אָכ֣וֹל וְשָׁת֔וֹ כִּ֥י מָחָ֖ר נָמֽוּת׃ (לח) וְנִגְלָ֥ה בְאָזְנָ֖י יְהוָ֣ה צְבָא֑וֹת אִם־ יְ֠כֻפַּר הֶעָוֺ֨ן הַזֶּ֤ה לָכֶם֙ עַד־ תְּמֻת֔וּן אָמַ֛ר אֲדֹנָ֥י יְהוִ֖ה צְבָאֽוֹת׃ (לט) כֹּ֥ה אָמַ֛ר אֲדֹנָ֥י יְהוִ֖ה צְבָא֑וֹת לֶךְ־ בֹּא֙ אֶל־ הַסֹּכֵ֣ן הַזֶּ֔ה עַל־ שֶׁבְנָ֖א אֲשֶׁ֥ר עַל־ הַבָּֽיִת׃ (מ) מַה־ לְּךָ֥ פֹה֙ וּמִ֣י לְךָ֣ פֹ֔ה כִּֽי־ חָצַ֧בְתָּ לְּךָ֛ פֹּ֖ה קָ֑בֶר חֹצְבִ֤י מָרוֹם֙ קִבְר֔וֹ חֹקְקִ֥י בַסֶּ֖לַע מִשְׁכָּ֥ן לֽוֹ׃ (מא) הִנֵּ֤ה יְהוָה֙ מְטַלְטֶלְךָ֔ טַלְטֵלָ֖ה גָּ֑בֶר וְעֹטְךָ֖ עָטֹֽה׃ (מב) צָנ֤וֹף יִצְנָפְךָ֙ צְנֵפָ֔ה כַּדּ֕וּר אֶל־ אֶ֖רֶץ רַחֲבַ֣ת יָדָ֑יִם שָׁ֣מָּה תָמ֗וּת וְשָׁ֙מָּה֙ מַרְכְּב֣וֹת כְּבוֹדֶ֔ךָ קְל֖וֹן בֵּ֥ית אֲדֹנֶֽיךָ׃ (מג) וַהֲדַפְתִּ֖יךָ מִמַּצָּבֶ֑ךָ וּמִמַּעֲמָֽדְךָ֖ יֶהֶרְסֶֽךָ׃ (מד) וְהָיָ֖ה בַּיּ֣וֹם הַה֑וּא וְקָרָ֣אתִי לְעַבְדִּ֔י לְאֶלְיָקִ֖ים בֶּן־ חִלְקִיָּֽהוּ׃ (מה) וְהִלְבַּשְׁתִּ֣יו כֻּתָּנְתֶּ֗ךָ וְאַבְנֵֽטְךָ֙ אֲחַזְּקֶ֔נּוּ וּמֶֽמְשֶׁלְתְּךָ֖ אֶתֵּ֣ן בְּיָד֑וֹ וְהָיָ֥ה לְאָ֛ב לְיוֹשֵׁ֥ב יְרוּשָׁלִַ֖ם וּלְבֵ֥ית יְהוּדָֽה׃ (מו) וְנָתַתִּ֛י מַפְתֵּ֥חַ בֵּית־ דָּוִ֖ד עַל־ שִׁכְמ֑וֹ וּפָתַח֙ וְאֵ֣ין סֹגֵ֔ר וְסָגַ֖ר וְאֵ֥ין פֹּתֵֽחַ׃

ישעיהו סדר ט

(א) וּתְקַעְתִּ֥יו יָתֵ֖ד בְּמָק֣וֹם נֶאֱמָ֑ן וְהָיָ֛ה לְכִסֵּ֥א כָב֖וֹד לְבֵ֥ית אָבִֽיו׃ (ב) וְתָל֨וּ עָלָ֜יו כֹּ֣ל ׀ כְּב֣וֹד בֵּית־ אָבִ֗יו הַצֶּֽאֱצָאִים֙ וְהַצְּפִע֔וֹת כֹּ֖ל כְּלֵ֣י הַקָּטָ֑ן מִכְּלֵי֙ הָֽאַגָּנ֔וֹת וְעַ֖ד כָּל־ כְּלֵ֥י הַנְּבָלִֽים׃ (ג) בַּיּ֣וֹם הַה֗וּא נְאֻם֙ יְהוָ֣ה צְבָא֔וֹת תָּמוּשׁ֙ הַיָּתֵ֔ד הַתְּקוּעָ֖ה בְּמָק֣וֹם נֶאֱמָ֑ן וְנִגְדְּעָ֣ה וְנָפְלָ֗ה וְנִכְרַת֙ הַמַּשָּׂ֣א אֲשֶׁר־ עָלֶ֔יהָ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ סוף פרק כב (ד) מַשָּׂ֖א צֹ֑ר הֵילִ֣ילוּ ׀ אֳנִיּ֣וֹת תַּרְשִׁ֗ישׁ כִּֽי־ שֻׁדַּ֤ד מִבַּ֙יִת֙ מִבּ֔וֹא מֵאֶ֥רֶץ כִּתִּ֖ים נִגְלָה־ לָֽמוֹ׃ (ה) דֹּ֖מּוּ יֹ֣שְׁבֵי אִ֑י סֹחֵ֥ר צִיד֛וֹן עֹבֵ֥ר יָ֖ם מִלְאֽוּךְ׃ (ו) וּבְמַ֤יִם רַבִּים֙ זֶ֣רַע שִׁחֹ֔ר קְצִ֥יר יְא֖וֹר תְּבֽוּאָתָ֑הּ וַתְּהִ֖י סְחַ֥ר גּוֹיִֽם׃ (ז) בּ֣וֹשִׁי צִיד֔וֹן כִּֽי־ אָמַ֣ר יָ֔ם מָע֥וֹז הַיָּ֖ם לֵאמֹ֑ר לֹֽא־ חַ֣לְתִּי וְלֹֽא־ יָלַ֗דְתִּי וְלֹ֥א גִדַּ֛לְתִּי בַּחוּרִ֖ים רוֹמַ֥מְתִּי בְתוּלֽוֹת׃ (ח) כַּֽאֲשֶׁר־ שֵׁ֖מַע לְמִצְרָ֑יִם יָחִ֖ילוּ כְּשֵׁ֥מַע צֹֽר׃ (ט) עִבְר֖וּ תַּרְשִׁ֑ישָׁה הֵילִ֖ילוּ יֹ֥שְׁבֵי אִֽי׃ (י) הֲזֹ֥את לָכֶ֖ם עַלִּיזָ֑ה מִֽימֵי־ קֶ֤דֶם קַדְמָתָהּ֙ יֹבִל֣וּהָ רַגְלֶ֔יהָ מֵֽרָח֖וֹק לָגֽוּר׃ (יא) מִ֚י יָעַ֣ץ זֹ֔את עַל־ צֹ֖ר הַמַּֽעֲטִירָ֑ה אֲשֶׁ֤ר סֹחֲרֶ֙יה֙ שָׂרִ֔ים כִּנְעָנֶ֖יהָ נִכְבַּדֵּי־ אָֽרֶץ׃ (יב) יְהוָ֥ה צְבָא֖וֹת יְעָצָ֑הּ לְחַלֵּל֙ גְּא֣וֹן כָּל־ צְבִ֔י לְהָקֵ֖ל כָּל־ נִכְבַּדֵּי־ אָֽרֶץ׃ (יג) עִבְרִ֥י אַרְצֵ֖ךְ כַּיְאֹ֑ר בַּת־ תַּרְשִׁ֕ישׁ אֵ֖ין מֵ֥זַח עֽוֹד׃ (יד) יָדוֹ֙ נָטָ֣ה עַל־ הַיָּ֔ם הִרְגִּ֖יז מַמְלָכ֑וֹת יְהוָה֙ צִוָּ֣ה אֶל־ כְּנַ֔עַן לַשְׁמִ֖ד מָעֻזְנֶֽיהָ׃ (טו) וַיֹּ֕אמֶר לֹֽא־ תוֹסִ֥יפִי ע֖וֹד לַעְל֑וֹז הַֽמְעֻשָּׁקָ֞ה בְּתוּלַ֣ת בַּת־ צִיד֗וֹן (כתיים) כִּתִּים֙ ק֣וּמִי עֲבֹ֔רִי גַּם־ שָׁ֖ם לֹא־ יָנ֥וּחַֽ לָֽךְ׃ (טז) הֵ֣ן ׀ אֶ֣רֶץ כַּשְׂדִּ֗ים זֶ֤ה הָעָם֙ לֹ֣א הָיָ֔ה אַשּׁ֖וּר יְסָדָ֣הּ לְצִיִּ֑ים הֵקִ֣ימוּ (בחיניו) בַחוּנָ֗יו עֹרְרוּ֙ אַרְמְנוֹתֶ֔יהָ שָׂמָ֖הּ לְמַפֵּלָֽה׃ (יז) הֵילִ֖ילוּ אֳנִיּ֣וֹת תַּרְשִׁ֑ישׁ כִּ֥י שֻׁדַּ֖ד מָעֻזְּכֶֽן׃ (יח) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא וְנִשְׁכַּ֤חַת צֹר֙ שִׁבְעִ֣ים שָׁנָ֔ה כִּימֵ֖י מֶ֣לֶךְ אֶחָ֑ד מִקֵּ֞ץ שִׁבְעִ֤ים שָׁנָה֙ יִהְיֶ֣ה לְצֹ֔ר כְּשִׁירַ֖ת הַזּוֹנָֽה׃ (יט) קְחִ֥י כִנּ֛וֹר סֹ֥בִּי עִ֖יר זוֹנָ֣ה נִשְׁכָּחָ֑ה הֵיטִ֤יבִי נַגֵּן֙ הַרְבִּי־ שִׁ֔יר לְמַ֖עַן תִּזָּכֵֽרִי׃ (כ) וְהָיָ֞ה מִקֵּ֣ץ ׀ שִׁבְעִ֣ים שָׁנָ֗ה יִפְקֹ֤ד יְהוָה֙ אֶת־ צֹ֔ר וְשָׁבָ֖ה לְאֶתְנַנָּ֑ה וְזָֽנְתָ֛ה אֶת־ כָּל־ מַמְלְכ֥וֹת הָאָ֖רֶץ עַל־ פְּנֵ֥י הָאֲדָמָֽה׃ (כא) וְהָיָ֨ה סַחְרָ֜הּ וְאֶתְנַנָּ֗הּ קֹ֚דֶשׁ לַֽיהוָ֔ה לֹ֥א יֵֽאָצֵ֖ר וְלֹ֣א יֵֽחָסֵ֑ן כִּ֣י לַיֹּשְׁבִ֞ים לִפְנֵ֤י יְהוָה֙ יִֽהְיֶ֣ה סַחְרָ֔הּ לֶאֱכֹ֥ל לְשָׂבְעָ֖ה וְלִמְכַסֶּ֥ה עָתִֽיק׃ (כב) הִנֵּ֧ה יְהוָ֛ה בּוֹקֵ֥ק הָאָ֖רֶץ וּבֽוֹלְקָ֑הּ וְעִוָּ֣ה פָנֶ֔יהָ וְהֵפִ֖יץ יֹשְׁבֶֽיהָ׃ (כג) וְהָיָ֤ה כָעָם֙ כַּכֹּהֵ֔ן כַּעֶ֙בֶד֙ כַּֽאדֹנָ֔יו כַּשִּׁפְחָ֖ה כַּגְּבִרְתָּ֑הּ כַּקּוֹנֶה֙ כַּמּוֹכֵ֔ר כַּמַּלְוֶה֙ כַּלֹּוֶ֔ה כַּנֹּשֶׁ֕ה כַּאֲשֶׁ֖ר נֹשֶׁ֥א בֽוֹ׃ (כד) הִבּ֧וֹק ׀ תִּבּ֛וֹק הָאָ֖רֶץ וְהִבּ֣וֹז ׀ תִּבּ֑וֹז כִּ֣י יְהוָ֔ה דִּבֶּ֖ר אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (כה) אָבְלָ֤ה נָֽבְלָה֙ הָאָ֔רֶץ אֻמְלְלָ֥ה נָבְלָ֖ה תֵּבֵ֑ל אֻמְלָ֖לוּ מְר֥וֹם עַם־ הָאָֽרֶץ׃ (כו) וְהָאָ֥רֶץ חָנְפָ֖ה תַּ֣חַת יֹשְׁבֶ֑יהָ כִּֽי־ עָבְר֤וּ תוֹרֹת֙ חָ֣לְפוּ חֹ֔ק הֵפֵ֖רוּ בְּרִ֥ית עוֹלָֽם׃ (כז) עַל־ כֵּ֗ן אָלָה֙ אָ֣כְלָה אֶ֔רֶץ וַֽיֶּאְשְׁמ֖וּ יֹ֣שְׁבֵי בָ֑הּ עַל־ כֵּ֗ן חָרוּ֙ יֹ֣שְׁבֵי אֶ֔רֶץ וְנִשְׁאַ֥ר אֱנ֖וֹשׁ מִזְעָֽר׃ (כח) אָבַ֥ל תִּיר֖וֹשׁ אֻמְלְלָה־ גָ֑פֶן נֶאֶנְח֖וּ כָּל־ שִׂמְחֵי־ לֵֽב׃ (כט) שָׁבַת֙ מְשׂ֣וֹשׂ תֻּפִּ֔ים חָדַ֖ל שְׁא֣וֹן עַלִּיזִ֑ים שָׁבַ֖ת מְשׂ֥וֹשׂ כִּנּֽוֹר׃ (ל) בַּשִּׁ֖יר לֹ֣א יִשְׁתּוּ־ יָ֑יִן יֵמַ֥ר שֵׁכָ֖ר לְשֹׁתָֽיו׃ (לא) נִשְׁבְּרָ֖ה קִרְיַת־ תֹּ֑הוּ סֻגַּ֥ר כָּל־ בַּ֖יִת מִבּֽוֹא׃ (לב) צְוָחָ֥ה עַל־ הַיַּ֖יִן בַּֽחוּצ֑וֹת עָֽרְבָה֙ כָּל־ שִׂמְחָ֔ה גָּלָ֖ה מְשׂ֥וֹשׂ הָאָֽרֶץ׃ (לג) נִשְׁאַ֥ר בָּעִ֖יר שַׁמָּ֑ה וּשְׁאִיָּ֖ה יֻכַּת־ שָֽׁעַר׃ (לד) כִּ֣י כֹ֥ה יִהְיֶ֛ה בְּקֶ֥רֶב הָאָ֖רֶץ בְּת֣וֹךְ הָֽעַמִּ֑ים כְּנֹ֣קֶף זַ֔יִת כְּעוֹלֵלֹ֖ת אִם־ כָּלָ֥ה בָצִֽיר׃ (לה) הֵ֛מָּה יִשְׂא֥וּ קוֹלָ֖ם יָרֹ֑נּוּ בִּגְא֣וֹן יְהוָ֔ה צָהֲל֖וּ מִיָּֽם׃ (לו) עַל־ כֵּ֥ן בָּאֻרִ֖ים כַּבְּד֣וּ יְהוָ֑ה בְּאִיֵּ֣י הַיָּ֔ם שֵׁ֥ם יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (לז) מִכְּנַ֨ף הָאָ֜רֶץ זְמִרֹ֤ת שָׁמַ֙עְנוּ֙ צְבִ֣י לַצַּדִּ֔יק וָאֹמַ֛ר רָזִי־ לִ֥י רָֽזִי־ לִ֖י א֣וֹי לִ֑י בֹּגְדִ֣ים בָּגָ֔דוּ וּבֶ֥גֶד בּוֹגְדִ֖ים בָּגָֽדוּ׃ (לח) פַּ֥חַד וָפַ֖חַת וָפָ֑ח עָלֶ֖יךָ יוֹשֵׁ֥ב הָאָֽרֶץ׃ (לט) וְֽ֠הָיָה הַנָּ֞ס מִקּ֤וֹל הַפַּ֙חַד֙ יִפֹּ֣ל אֶל־ הַפַּ֔חַת וְהָֽעוֹלֶה֙ מִתּ֣וֹךְ הַפַּ֔חַת יִלָּכֵ֖ד בַּפָּ֑ח כִּֽי־ אֲרֻבּ֤וֹת מִמָּרוֹם֙ נִפְתָּ֔חוּ וַֽיִּרְעֲשׁ֖וּ מ֥וֹסְדֵי אָֽרֶץ׃ (מ) רֹ֥עָה הִֽתְרֹעֲעָ֖ה הָאָ֑רֶץ פּ֤וֹר הִֽתְפּוֹרְרָה֙ אֶ֔רֶץ מ֥וֹט הִֽתְמוֹטְטָ֖ה אָֽרֶץ׃ (מא) נ֣וֹעַ תָּנ֤וּעַ אֶ֙רֶץ֙ כַּשִּׁכּ֔וֹר וְהִֽתְנוֹדְדָ֖ה כַּמְּלוּנָ֑ה וְכָבַ֤ד עָלֶ֙יהָ֙ פִּשְׁעָ֔הּ וְנָפְלָ֖ה וְלֹא־ תֹסִ֥יף קֽוּם׃ (מב) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא יִפְקֹ֧ד יְהוָ֛ה עַל־ צְבָ֥א הַמָּר֖וֹם בַּמָּר֑וֹם וְעַל־ מַלְכֵ֥י הָאֲדָמָ֖ה עַל־ הָאֲדָמָֽה׃ (מג) וְאֻסְּפ֨וּ אֲסֵפָ֤ה אַסִּיר֙ עַל־ בּ֔וֹר וְסֻגְּר֖וּ עַל־ מַסְגֵּ֑ר וּמֵרֹ֥ב יָמִ֖ים יִפָּקֵֽדוּ׃
