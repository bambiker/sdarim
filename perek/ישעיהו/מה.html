ישעיהו פרק מה לפי החלוקה לסדרים

פרק מה בספר ישעיהו (25 פסוקים) נמצא בישעיהו סדר יח (פסוקים כד–לט) ובישעיהו סדר יט (פסוקים א–ט).

ישעיהו סדר יח

(א) כֹּֽה־ אָמַ֨ר יְהוָ֧ה מֶֽלֶךְ־ יִשְׂרָאֵ֛ל וְגֹאֲל֖וֹ יְהוָ֣ה צְבָא֑וֹת אֲנִ֤י רִאשׁוֹן֙ וַאֲנִ֣י אַחֲר֔וֹן וּמִבַּלְעָדַ֖י אֵ֥ין אֱלֹהִֽים׃ (ב) וּמִֽי־ כָמ֣וֹנִי יִקְרָ֗א וְיַגִּידֶ֤הָ וְיַעְרְכֶ֙הָ֙ לִ֔י מִשּׂוּמִ֖י עַם־ עוֹלָ֑ם וְאֹתִיּ֛וֹת וַאֲשֶׁ֥ר תָּבֹ֖אנָה יַגִּ֥ידוּ לָֽמוֹ׃ (ג) אַֽל־ תִּפְחֲדוּ֙ וְאַל־ תִּרְה֔וּ הֲלֹ֥א מֵאָ֛ז הִשְׁמַעְתִּ֥יךָ וְהִגַּ֖דְתִּי וְאַתֶּ֣ם עֵדָ֑י הֲיֵ֤שׁ אֱל֙וֹהַּ֙ מִבַּלְעָדַ֔י וְאֵ֥ין צ֖וּר בַּל־ יָדָֽעְתִּי׃ (ד) יֹֽצְרֵי־ פֶ֤סֶל כֻּלָּם֙ תֹּ֔הוּ וַחֲמוּדֵיהֶ֖ם בַּל־ יוֹעִ֑ילוּ וְעֵדֵיהֶ֣ם הֵׄ֗מָּׄהׄ בַּל־ יִרְא֛וּ וּבַל־ יֵדְע֖וּ לְמַ֥עַן יֵבֹֽשׁוּ׃ (ה) מִֽי־ יָצַ֥ר אֵ֖ל וּפֶ֣סֶל נָסָ֑ךְ לְבִלְתִּ֖י הוֹעִֽיל׃ (ו) הֵ֤ן כָּל־ חֲבֵרָיו֙ יֵבֹ֔שׁוּ וְחָרָשִׁ֥ים הֵ֖מָּה מֵֽאָדָ֑ם יִֽתְקַבְּצ֤וּ כֻלָּם֙ יַֽעֲמֹ֔דוּ יִפְחֲד֖וּ יֵבֹ֥שׁוּ יָֽחַד׃ (ז) חָרַ֤שׁ בַּרְזֶל֙ מַֽעֲצָ֔ד וּפָעַל֙ בַּפֶּחָ֔ם וּבַמַּקָּב֖וֹת יִצְּרֵ֑הוּ וַיִּפְעָלֵ֙הוּ֙ בִּזְר֣וֹעַ כֹּח֔וֹ גַּם־ רָעֵב֙ וְאֵ֣ין כֹּ֔חַ לֹא־ שָׁ֥תָה מַ֖יִם וַיִּיעָֽף׃ (ח) חָרַ֣שׁ עֵצִים֮ נָ֣טָה קָו֒ יְתָאֲרֵ֣הוּ בַשֶּׂ֔רֶד יַעֲשֵׂ֙הוּ֙ בַּמַּקְצֻע֔וֹת וּבַמְּחוּגָ֖ה יְתָאֳרֵ֑הוּ וַֽיַּעֲשֵׂ֙הוּ֙ כְּתַבְנִ֣ית אִ֔ישׁ כְּתִפְאֶ֥רֶת אָדָ֖ם לָשֶׁ֥בֶת בָּֽיִת׃ (ט) לִכְרָת־ ל֣וֹ אֲרָזִ֔ים וַיִּקַּ֤ח תִּרְזָה֙ וְאַלּ֔וֹן וַיְאַמֶּץ־ ל֖וֹ בַּעֲצֵי־ יָ֑עַר נָטַ֥ע אֹ֖רֶ וְגֶ֥שֶׁם יְגַדֵּֽל׃ (י) וְהָיָ֤ה לְאָדָם֙ לְבָעֵ֔ר וַיִּקַּ֤ח מֵהֶם֙ וַיָּ֔חָם אַף־ יַשִּׂ֖יק וְאָ֣פָה לָ֑חֶם אַף־ יִפְעַל־ אֵל֙ וַיִּשְׁתָּ֔חוּ עָשָׂ֥הוּ פֶ֖סֶל וַיִּסְגָּד־ לָֽמוֹ׃ (יא) חֶצְיוֹ֙ שָׂרַ֣ף בְּמוֹ־ אֵ֔שׁ עַל־ חֶצְיוֹ֙ בָּשָׂ֣ר יֹאכֵ֔ל יִצְלֶ֥ה צָלִ֖י וְיִשְׂבָּ֑ע אַף־ יָחֹם֙ וְיֹאמַ֣ר הֶאָ֔ח חַמּוֹתִ֖י רָאִ֥יתִי אֽוּר׃ (יב) וּשְׁאֵ֣רִית֔וֹ לְאֵ֥ל עָשָׂ֖ה לְפִסְל֑וֹ (יסגוד)־ יִסְגָּד־ ל֤וֹ וְיִשְׁתַּ֙חוּ֙ וְיִתְפַּלֵּ֣ל אֵלָ֔יו וְיֹאמַר֙ הַצִּילֵ֔נִי כִּ֥י אֵלִ֖י אָֽתָּה׃ (יג) לֹ֥א יָדְע֖וּ וְלֹ֣א יָבִ֑ינוּ כִּ֣י טַ֤ח מֵֽרְאוֹת֙ עֵֽינֵיהֶ֔ם מֵהַשְׂכִּ֖יל לִבֹּתָֽם׃ (יד) וְלֹא־ יָשִׁ֣יב אֶל־ לִבּ֗וֹ וְלֹ֨א דַ֥עַת וְלֹֽא־ תְבוּנָה֮ לֵאמֹר֒ חֶצְי֞וֹ שָׂרַ֣פְתִּי בְמוֹ־ אֵ֗שׁ וְ֠אַף אָפִ֤יתִי עַל־ גֶּחָלָיו֙ לֶ֔חֶם אֶצְלֶ֥ה בָשָׂ֖ר וְאֹכֵ֑ל וְיִתְרוֹ֙ לְתוֹעֵבָ֣ה אֶעֱשֶׂ֔ה לְב֥וּל עֵ֖ץ אֶסְגּֽוֹד׃ (טו) רֹעֶ֣ה אֵ֔פֶר לֵ֥ב הוּתַ֖ל הִטָּ֑הוּ וְלֹֽא־ יַצִּ֤יל אֶת־ נַפְשׁוֹ֙ וְלֹ֣א יֹאמַ֔ר הֲל֥וֹא שֶׁ֖קֶר בִּימִינִֽי׃ (טז) זְכָר־ אֵ֣לֶּה יַעֲקֹ֔ב וְיִשְׂרָאֵ֖ל כִּ֣י עַבְדִּי־ אָ֑תָּה יְצַרְתִּ֤יךָ עֶֽבֶד־ לִי֙ אַ֔תָּה יִשְׂרָאֵ֖ל לֹ֥א תִנָּשֵֽׁנִי׃ (יז) מָחִ֤יתִי כָעָב֙ פְּשָׁעֶ֔יךָ וְכֶעָנָ֖ן חַטֹּאותֶ֑יךָ שׁוּבָ֥ה אֵלַ֖י כִּ֥י גְאַלְתִּֽיךָ׃ (יח) רָנּ֨וּ שָׁמַ֜יִם כִּֽי־ עָשָׂ֣ה יְהוָ֗ה הָרִ֙יעוּ֙ תַּחְתִּיּ֣וֹת אָ֔רֶץ פִּצְח֤וּ הָרִים֙ רִנָּ֔ה יַ֖עַר וְכָל־ עֵ֣ץ בּ֑וֹ כִּֽי־ גָאַ֤ל יְהוָה֙ יַֽעֲקֹ֔ב וּבְיִשְׂרָאֵ֖ל יִתְפָּאָֽר׃ (יט) כֹּֽה־ אָמַ֤ר יְהוָה֙ גֹּאֲלֶ֔ךָ וְיֹצֶרְךָ֖ מִבָּ֑טֶן אָנֹכִ֤י יְהוָה֙ עֹ֣שֶׂה כֹּ֔ל נֹטֶ֤ה שָׁמַ֙יִם֙ לְבַדִּ֔י רֹקַ֥ע הָאָ֖רֶץ (מי אתי) מֵאִתִּֽי׃ (כ) מֵפֵר֙ אֹת֣וֹת בַּדִּ֔ים וְקֹסְמִ֖ים יְהוֹלֵ֑ל מֵשִׁ֧יב חֲכָמִ֛ים אָח֖וֹר וְדַעְתָּ֥ם יְשַׂכֵּֽל׃ (כא) מֵקִים֙ דְּבַ֣ר עַבְדּ֔וֹ וַעֲצַ֥ת מַלְאָכָ֖יו יַשְׁלִ֑ים הָאֹמֵ֨ר לִירוּשָׁלִַ֜ם תּוּשָׁ֗ב וּלְעָרֵ֤י יְהוּדָה֙ תִּבָּנֶ֔ינָה וְחָרְבוֹתֶ֖יהָ אֲקוֹמֵֽם׃ (כב) הָאֹמֵ֥ר לַצּוּלָ֖ה חֳרָ֑בִי וְנַהֲרֹתַ֖יִךְ אוֹבִֽישׁ׃ (כג) הָאֹמֵ֤ר לְכ֙וֹרֶשׁ֙ רֹעִ֔י וְכָל־ חֶפְצִ֖י יַשְׁלִ֑ם וְלֵאמֹ֤ר לִירוּשָׁלִַ֙ם֙ תִּבָּנֶ֔ה וְהֵיכָ֖ל תִּוָּסֵֽד׃ תחילת פרק מה (כד) כֹּה־ אָמַ֣ר יְהוָה֮ לִמְשִׁיחוֹ֮ לְכ֣וֹרֶשׁ אֲשֶׁר־ הֶחֱזַ֣קְתִּי בִֽימִינ֗וֹ לְרַד־ לְפָנָיו֙ גּוֹיִ֔ם וּמָתְנֵ֥י מְלָכִ֖ים אֲפַתֵּ֑חַ לִפְתֹּ֤חַ לְפָנָיו֙ דְּלָתַ֔יִם וּשְׁעָרִ֖ים לֹ֥א יִסָּגֵֽרוּ׃ (כה) אֲנִי֙ לְפָנֶ֣יךָ אֵלֵ֔ךְ וַהֲדוּרִ֖ים (אושר) אֲיַשֵּׁ֑ר דַּלְת֤וֹת נְחוּשָׁה֙ אֲשַׁבֵּ֔ר וּבְרִיחֵ֥י בַרְזֶ֖ל אֲגַדֵּֽעַ׃ (כו) וְנָתַתִּ֤י לְךָ֙ אוֹצְר֣וֹת חֹ֔שֶׁךְ וּמַטְמֻנֵ֖י מִסְתָּרִ֑ים לְמַ֣עַן תֵּדַ֗ע כִּֽי־ אֲנִ֧י יְהוָ֛ה הַקּוֹרֵ֥א בְשִׁמְךָ֖ אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (כז) לְמַ֙עַן֙ עַבְדִּ֣י יַעֲקֹ֔ב וְיִשְׂרָאֵ֖ל בְּחִירִ֑י וָאֶקְרָ֤א לְךָ֙ בִּשְׁמֶ֔ךָ אֲכַנְּךָ֖ וְלֹ֥א יְדַעְתָּֽנִי׃ (כח) אֲנִ֤י יְהוָה֙ וְאֵ֣ין ע֔וֹד זוּלָתִ֖י אֵ֣ין אֱלֹהִ֑ים אֲאַזֶּרְךָ֖ וְלֹ֥א יְדַעְתָּֽנִי׃ (כט) לְמַ֣עַן יֵדְע֗וּ מִמִּזְרַח־ שֶׁ֙מֶשׁ֙ וּמִמַּ֣עֲרָבָ֔הּ כִּי־ אֶ֖פֶס בִּלְעָדָ֑י אֲנִ֥י יְהוָ֖ה וְאֵ֥ין עֽוֹד׃ (ל) יוֹצֵ֥ר אוֹר֙ וּבוֹרֵ֣א חֹ֔שֶׁךְ עֹשֶׂ֥ה שָׁל֖וֹם וּב֣וֹרֵא רָ֑ע אֲנִ֥י יְהוָ֖ה עֹשֶׂ֥ה כָל־ אֵֽלֶּה׃ (לא) הַרְעִ֤יפוּ שָׁמַ֙יִם֙ מִמַּ֔עַל וּשְׁחָקִ֖ים יִזְּלוּ־ צֶ֑דֶק תִּפְתַּח־ אֶ֣רֶץ וְיִפְרוּ־ יֶ֗שַׁע וּצְדָקָ֤ה תַצְמִ֙יחַ֙ יַ֔חַד אֲנִ֥י יְהוָ֖ה בְּרָאתִֽיו׃ (לב) ה֗וֹי רָ֚ב אֶת־ יֹ֣צְר֔וֹ חֶ֖רֶשׂ אֶת־ חַרְשֵׂ֣י אֲדָמָ֑ה הֲיֹאמַ֨ר חֹ֤מֶר לְיֹֽצְרוֹ֙ מַֽה־ תַּעֲשֶׂ֔ה וּפָעָלְךָ֖ אֵין־ יָדַ֥יִם לֽוֹ׃ (לג) ה֛וֹי אֹמֵ֥ר לְאָ֖ב מַה־ תּוֹלִ֑יד וּלְאִשָּׁ֖ה מַה־ תְּחִילִֽין׃ (לד) כֹּֽה־ אָמַ֧ר יְהוָ֛ה קְד֥וֹשׁ יִשְׂרָאֵ֖ל וְיֹצְר֑וֹ הָאֹתִיּ֣וֹת שְׁאָל֔וּנִי עַל־ בָּנַ֛י וְעַל־ פֹּ֥עַל יָדַ֖י תְּצַוֻּֽנִי׃ (לה) אָֽנֹכִי֙ עָשִׂ֣יתִי אֶ֔רֶץ וְאָדָ֖ם עָלֶ֣יהָ בָרָ֑אתִי אֲנִ֗י יָדַי֙ נָט֣וּ שָׁמַ֔יִם וְכָל־ צְבָאָ֖ם צִוֵּֽיתִי׃ (לו) אָנֹכִי֙ הַעִירֹתִ֣הֽוּ בְצֶ֔דֶק וְכָל־ דְּרָכָ֖יו אֲיַשֵּׁ֑ר הֽוּא־ יִבְנֶ֤ה עִירִי֙ וְגָלוּתִ֣י יְשַׁלֵּ֔חַ לֹ֤א בִמְחִיר֙ וְלֹ֣א בְשֹׁ֔חַד אָמַ֖ר יְהוָ֥ה צְבָאֽוֹת׃ (לז) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה יְגִ֨יעַ מִצְרַ֥יִם וּֽסְחַר־ כּוּשׁ֮ וּסְבָאִים֮ אַנְשֵׁ֣י מִדָּה֒ עָלַ֤יִךְ יַעֲבֹ֙רוּ֙ וְלָ֣ךְ יִֽהְי֔וּ אַחֲרַ֣יִךְ יֵלֵ֔כוּ בַּזִּקִּ֖ים יַעֲבֹ֑רוּ וְאֵלַ֤יִךְ יִֽשְׁתַּחֲוּוּ֙ אֵלַ֣יִךְ יִתְפַּלָּ֔לוּ אַ֣ךְ בָּ֥ךְ אֵ֛ל וְאֵ֥ין ע֖וֹד אֶ֥פֶס אֱלֹהִֽים׃ (לח) אָכֵ֕ן אַתָּ֖ה אֵ֣ל מִסְתַּתֵּ֑ר אֱלֹהֵ֥י יִשְׂרָאֵ֖ל מוֹשִֽׁיעַ׃ (לט) בּ֥וֹשׁוּ וְגַֽם־ נִכְלְמ֖וּ כֻּלָּ֑ם יַחְדָּו֙ הָלְכ֣וּ בַכְּלִמָּ֔ה חָרָשֵׁ֖י צִירִֽים׃

ישעיהו סדר יט

(א) יִשְׂרָאֵל֙ נוֹשַׁ֣ע בַּיהוָ֔ה תְּשׁוּעַ֖ת עוֹלָמִ֑ים לֹא־ תֵבֹ֥שׁוּ וְלֹא־ תִכָּלְמ֖וּ עַד־ ע֥וֹלְמֵי עַֽד׃ (ב) כִּ֣י כֹ֣ה אָֽמַר־ יְ֠הוָה בּוֹרֵ֨א הַשָּׁמַ֜יִם ה֣וּא הָאֱלֹהִ֗ים יֹצֵ֨ר הָאָ֤רֶץ וְעֹשָׂהּ֙ ה֣וּא כֽוֹנְנָ֔הּ לֹא־ תֹ֥הוּ בְרָאָ֖הּ לָשֶׁ֣בֶת יְצָרָ֑הּ אֲנִ֥י יְהוָ֖ה וְאֵ֥ין עֽוֹד׃ (ג) לֹ֧א בַסֵּ֣תֶר דִּבַּ֗רְתִּי בִּמְקוֹם֙ אֶ֣רֶץ חֹ֔שֶׁךְ לֹ֥א אָמַ֛רְתִּי לְזֶ֥רַע יַעֲקֹ֖ב תֹּ֣הוּ בַקְּשׁ֑וּנִי אֲנִ֤י יְהוָה֙ דֹּבֵ֣ר צֶ֔דֶק מַגִּ֖יד מֵישָׁרִֽים׃ (ד) הִקָּבְצ֥וּ וָבֹ֛אוּ הִֽתְנַגְּשׁ֥וּ יַחְדָּ֖ו פְּלִיטֵ֣י הַגּוֹיִ֑ם לֹ֣א יָדְע֗וּ הַנֹּֽשְׂאִים֙ אֶת־ עֵ֣ץ פִּסְלָ֔ם וּמִתְפַּלְלִ֔ים אֶל־ אֵ֖ל לֹ֥א יוֹשִֽׁיעַ׃ (ה) הַגִּ֣ידוּ וְהַגִּ֔ישׁוּ אַ֥ף יִֽוָּעֲצ֖וּ יַחְדָּ֑ו מִ֣י הִשְׁמִיעַ֩ זֹ֨את מִקֶּ֜דֶם מֵאָ֣ז הִגִּידָ֗הּ הֲל֨וֹא אֲנִ֤י יְהוָה֙ וְאֵֽין־ ע֤וֹד אֱלֹהִים֙ מִבַּלְעָדַ֔י אֵֽל־ צַדִּ֣יק וּמוֹשִׁ֔יעַ אַ֖יִן זוּלָתִֽי׃ (ו) פְּנוּ־ אֵלַ֥י וְהִוָּשְׁע֖וּ כָּל־ אַפְסֵי־ אָ֑רֶץ כִּ֥י אֲנִי־ אֵ֖ל וְאֵ֥ין עֽוֹד׃ (ז) בִּ֣י נִשְׁבַּ֔עְתִּי יָצָ֨א מִפִּ֧י צְדָקָ֛ה דָּבָ֖ר וְלֹ֣א יָשׁ֑וּב כִּי־ לִי֙ תִּכְרַ֣ע כָּל־ בֶּ֔רֶךְ תִּשָּׁבַ֖ע כָּל־ לָשֽׁוֹן׃ (ח) אַ֧ךְ בַּיהוָ֛ה לִ֥י אָמַ֖ר צְדָק֣וֹת וָעֹ֑ז עָדָיו֙ יָב֣וֹא וְיֵבֹ֔שׁוּ כֹּ֖ל הַנֶּחֱרִ֥ים בּֽוֹ׃ (ט) בַּיהוָ֛ה יִצְדְּק֥וּ וְיִֽתְהַלְל֖וּ כָּל־ זֶ֥רַע יִשְׂרָאֵֽל׃ סוף פרק מה (י) כָּרַ֥ע בֵּל֙ קֹרֵ֣ס נְב֔וֹ הָיוּ֙ עֲצַבֵּיהֶ֔ם לַחַיָּ֖ה וְלַבְּהֵמָ֑ה נְשֻׂאֹתֵיכֶ֣ם עֲמוּס֔וֹת מַשָּׂ֖א לַעֲיֵפָֽה׃ (יא) קָרְס֤וּ כָֽרְעוּ֙ יַחְדָּ֔ו לֹ֥א יָכְל֖וּ מַלֵּ֣ט מַשָּׂ֑א וְנַפְשָׁ֖ם בַּשְּׁבִ֥י הָלָֽכָה׃ (יב) שִׁמְע֤וּ אֵלַי֙ בֵּ֣ית יַעֲקֹ֔ב וְכָל־ שְׁאֵרִ֖ית בֵּ֣ית יִשְׂרָאֵ֑ל הַֽעֲמֻסִים֙ מִנִּי־ בֶ֔טֶן הַנְּשֻׂאִ֖ים מִנִּי־ רָֽחַם׃ (יג) וְעַד־ זִקְנָה֙ אֲנִ֣י ה֔וּא וְעַד־ שֵיבָ֖ה אֲנִ֣י אֶסְבֹּ֑ל אֲנִ֤י עָשִׂ֙יתִי֙ וַאֲנִ֣י אֶשָּׂ֔א וַאֲנִ֥י אֶסְבֹּ֖ל וַאֲמַלֵּֽט׃ (יד) לְמִ֥י תְדַמְי֖וּנִי וְתַשְׁו֑וּ וְתַמְשִׁל֖וּנִי וְנִדְמֶֽה׃ (טו) הַזָּלִ֤ים זָהָב֙ מִכִּ֔יס וְכֶ֖סֶף בַּקָּנֶ֣ה יִשְׁקֹ֑לוּ יִשְׂכְּר֤וּ צוֹרֵף֙ וְיַעֲשֵׂ֣הוּ אֵ֔ל יִסְגְּד֖וּ אַף־ יִֽשְׁתַּחֲוּֽוּ׃ (טז) יִ֠שָּׂאֻהוּ עַל־ כָּתֵ֨ף יִסְבְּלֻ֜הוּ וְיַנִּיחֻ֤הוּ תַחְתָּיו֙ וְיַֽעֲמֹ֔ד מִמְּקוֹמ֖וֹ לֹ֣א יָמִ֑ישׁ אַף־ יִצְעַ֤ק אֵלָיו֙ וְלֹ֣א יַעֲנֶ֔ה מִצָּרָת֖וֹ לֹ֥א יוֹשִׁיעֶֽנּוּ׃ (יז) זִכְרוּ־ זֹ֖את וְהִתְאֹשָׁ֑שׁוּ הָשִׁ֥יבוּ פוֹשְׁעִ֖ים עַל־ לֵֽב׃ (יח) זִכְר֥וּ רִאשֹׁנ֖וֹת מֵעוֹלָ֑ם כִּ֣י אָנֹכִ֥י אֵל֙ וְאֵ֣ין ע֔וֹד אֱלֹהִ֖ים וְאֶ֥פֶס כָּמֽוֹנִי׃ (יט) מַגִּ֤יד מֵֽרֵאשִׁית֙ אַחֲרִ֔ית וּמִקֶּ֖דֶם אֲשֶׁ֣ר לֹא־ נַעֲשׂ֑וּ אֹמֵר֙ עֲצָתִ֣י תָק֔וּם וְכָל־ חֶפְצִ֖י אֶעֱשֶֽׂה׃ (כ) קֹרֵ֤א מִמִּזְרָח֙ עַ֔יִט מֵאֶ֥רֶץ מֶרְחָ֖ק אִ֣ישׁ (עצתו) עֲצָתִ֑י אַף־ דִּבַּ֙רְתִּי֙ אַף־ אֲבִיאֶ֔נָּה יָצַ֖רְתִּי אַף־ אֶעֱשֶֽׂנָּה׃ (כא) שִׁמְע֥וּ אֵלַ֖י אַבִּ֣ירֵי לֵ֑ב הָרְחוֹקִ֖ים מִצְּדָקָֽה׃ (כב) קֵרַ֤בְתִּי צִדְקָתִי֙ לֹ֣א תִרְחָ֔ק וּתְשׁוּעָתִ֖י לֹ֣א תְאַחֵ֑ר וְנָתַתִּ֤י בְצִיּוֹן֙ תְּשׁוּעָ֔ה לְיִשְׂרָאֵ֖ל תִּפְאַרְתִּֽי׃ (כג) רְדִ֣י ׀ וּשְׁבִ֣י עַל־ עָפָ֗ר בְּתוּלַת֙ בַּת־ בָּבֶ֔ל שְׁבִי־ לָאָ֥רֶץ אֵין־ כִּסֵּ֖א בַּת־ כַּשְׂדִּ֑ים כִּ֣י לֹ֤א תוֹסִ֙יפִי֙ יִקְרְאוּ־ לָ֔ךְ רַכָּ֖ה וַעֲנֻגָּֽה׃ (כד) קְחִ֥י רֵחַ֖יִם וְטַ֣חֲנִי קָ֑מַח גַּלִּ֨י צַמָּתֵ֧ךְ חֶשְׂפִּי־ שֹׁ֛בֶל גַּלִּי־ שׁ֖וֹק עִבְרִ֥י נְהָרֽוֹת׃ (כה) תִּגָּל֙ עֶרְוָתֵ֔ךְ גַּ֥ם תֵּרָאֶ֖ה חֶרְפָּתֵ֑ךְ נָקָ֣ם אֶקָּ֔ח וְלֹ֥א אֶפְגַּ֖ע אָדָֽם׃ (כו) גֹּאֲלֵ֕נוּ יְהוָ֥ה צְבָא֖וֹת שְׁמ֑וֹ קְד֖וֹשׁ יִשְׂרָאֵֽל׃ (כז) שְׁבִ֥י דוּמָ֛ם וּבֹ֥אִי בַחֹ֖שֶׁךְ בַּת־ כַּשְׂדִּ֑ים כִּ֣י לֹ֤א תוֹסִ֙יפִי֙ יִקְרְאוּ־ לָ֔ךְ גְּבֶ֖רֶת מַמְלָכֽוֹת׃ (כח) קָצַ֣פְתִּי עַל־ עַמִּ֗י חִלַּ֙לְתִּי֙ נַחֲלָתִ֔י וָאֶתְּנֵ֖ם בְּיָדֵ֑ךְ לֹא־ שַׂ֤מְתְּ לָהֶם֙ רַחֲמִ֔ים עַל־ זָקֵ֕ן הִכְבַּ֥דְתְּ עֻלֵּ֖ךְ מְאֹֽד׃ (כט) וַתֹּ֣אמְרִ֔י לְעוֹלָ֖ם אֶהְיֶ֣ה גְבָ֑רֶת עַ֣ד לֹא־ שַׂ֥מְתְּ אֵ֙לֶּה֙ עַל־ לִבֵּ֔ךְ לֹ֥א זָכַ֖רְתְּ אַחֲרִיתָֽהּ׃ (ל) וְעַתָּ֞ה שִׁמְעִי־ זֹ֤את עֲדִינָה֙ הַיּוֹשֶׁ֣בֶת לָבֶ֔טַח הָאֹֽמְרָה֙ בִּלְבָ֔בָהּ אֲנִ֖י וְאַפְסִ֣י ע֑וֹד לֹ֤א אֵשֵׁב֙ אַלְמָנָ֔ה וְלֹ֥א אֵדַ֖ע שְׁכֽוֹל׃ (לא) וְתָבֹאנָה֩ לָּ֨ךְ שְׁתֵּי־ אֵ֥לֶּה רֶ֛גַע בְּי֥וֹם אֶחָ֖ד שְׁכ֣וֹל וְאַלְמֹ֑ן כְּתֻמָּם֙ בָּ֣אוּ עָלַ֔יִךְ בְּרֹ֣ב כְּשָׁפַ֔יִךְ בְּעָצְמַ֥ת חֲבָרַ֖יִךְ מְאֹֽד׃ (לב) וַתִּבְטְחִ֣י בְרָעָתֵ֗ךְ אָמַרְתְּ֙ אֵ֣ין רֹאָ֔נִי חָכְמָתֵ֥ךְ וְדַעְתֵּ֖ךְ הִ֣יא שׁוֹבְבָ֑תֶךְ וַתֹּאמְרִ֣י בְלִבֵּ֔ךְ אֲנִ֖י וְאַפְסִ֥י עֽוֹד׃ (לג) וּבָ֧א עָלַ֣יִךְ רָעָ֗ה לֹ֤א תֵדְעִי֙ שַׁחְרָ֔הּ וְתִפֹּ֤ל עָלַ֙יִךְ֙ הֹוָ֔ה לֹ֥א תוּכְלִ֖י כַּפְּרָ֑הּ וְתָבֹ֨א עָלַ֧יִךְ פִּתְאֹ֛ם שׁוֹאָ֖ה לֹ֥א תֵדָֽעִי׃ (לד) עִמְדִי־ נָ֤א בַחֲבָרַ֙יִךְ֙ וּבְרֹ֣ב כְּשָׁפַ֔יִךְ בַּאֲשֶׁ֥ר יָגַ֖עַתְּ מִנְּעוּרָ֑יִךְ אוּלַ֛י תּוּכְלִ֥י הוֹעִ֖יל אוּלַ֥י תַּעֲרֽוֹצִי׃ (לה) נִלְאֵ֖ית בְּרֹ֣ב עֲצָתָ֑יִךְ יַעַמְדוּ־ נָ֨א וְיוֹשִׁיעֻ֜ךְ (הברו) הֹבְרֵ֣י שָׁמַ֗יִם הַֽחֹזִים֙ בַּכּ֣וֹכָבִ֔ים מֽוֹדִיעִם֙ לֶחֳדָשִׁ֔ים מֵאֲשֶׁ֥ר יָבֹ֖אוּ עָלָֽיִךְ׃ (לו) הִנֵּ֨ה הָי֤וּ כְקַשׁ֙ אֵ֣שׁ שְׂרָפָ֔תַם לֹֽא־ יַצִּ֥ילוּ אֶת־ נַפְשָׁ֖ם מִיַּ֣ד לֶֽהָבָ֑ה אֵין־ גַּחֶ֣לֶת לַחְמָ֔ם א֖וּר לָשֶׁ֥בֶת נֶגְדּֽוֹ׃ (לז) כֵּ֥ן הָיוּ־ לָ֖ךְ אֲשֶׁ֣ר יָגָ֑עַתְּ סֹחֲרַ֣יִךְ מִנְּעוּרַ֗יִךְ אִ֤ישׁ לְעֶבְרוֹ֙ תָּע֔וּ אֵ֖ין מוֹשִׁיעֵֽךְ׃ (לח) שִׁמְעוּ־ זֹ֣את בֵּֽית־ יַעֲקֹ֗ב הַנִּקְרָאִים֙ בְּשֵׁ֣ם יִשְׂרָאֵ֔ל וּמִמֵּ֥י יְהוּדָ֖ה יָצָ֑אוּ הַֽנִּשְׁבָּעִ֣ים ׀ בְּשֵׁ֣ם יְהוָ֗ה וּבֵאלֹהֵ֤י יִשְׂרָאֵל֙ יַזְכִּ֔ירוּ לֹ֥א בֶאֱמֶ֖ת וְלֹ֥א בִצְדָקָֽה׃
