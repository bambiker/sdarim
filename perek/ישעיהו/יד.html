ישעיהו פרק יד לפי החלוקה לסדרים

פרק יד בספר ישעיהו (32 פסוקים) נמצא בישעיהו סדר ה (פסוקים מד–מד) ובישעיהו סדר ו (פסוקים א–לא).

ישעיהו סדר ה

(א) וְנָחָ֥ה עָלָ֖יו ר֣וּחַ יְהוָ֑ה ר֧וּחַ חָכְמָ֣ה וּבִינָ֗ה ר֤וּחַ עֵצָה֙ וּגְבוּרָ֔ה ר֥וּחַ דַּ֖עַת וְיִרְאַ֥ת יְהוָֽה׃ (ב) וַהֲרִיח֖וֹ בְּיִרְאַ֣ת יְהוָ֑ה וְלֹֽא־ לְמַרְאֵ֤ה עֵינָיו֙ יִשְׁפּ֔וֹט וְלֹֽא־ לְמִשְׁמַ֥ע אָזְנָ֖יו יוֹכִֽיחַ׃ (ג) וְשָׁפַ֤ט בְּצֶ֙דֶק֙ דַּלִּ֔ים וְהוֹכִ֥יחַ בְּמִישׁ֖וֹר לְעַנְוֵי־ אָ֑רֶץ וְהִֽכָּה־ אֶ֙רֶץ֙ בְּשֵׁ֣בֶט פִּ֔יו וּבְר֥וּחַ שְׂפָתָ֖יו יָמִ֥ית רָשָֽׁע׃ (ד) וְהָ֥יָה צֶ֖דֶק אֵז֣וֹר מָתְנָ֑יו וְהָאֱמוּנָ֖ה אֵז֥וֹר חֲלָצָֽיו׃ (ה) וְגָ֤ר זְאֵב֙ עִם־ כֶּ֔בֶשׂ וְנָמֵ֖ר עִם־ גְּדִ֣י יִרְבָּ֑ץ וְעֵ֨גֶל וּכְפִ֤יר וּמְרִיא֙ יַחְדָּ֔ו וְנַ֥עַר קָטֹ֖ן נֹהֵ֥ג בָּֽם׃ (ו) וּפָרָ֤ה וָדֹב֙ תִּרְעֶ֔ינָה יַחְדָּ֖ו יִרְבְּצ֣וּ יַלְדֵיהֶ֑ן וְאַרְיֵ֖ה כַּבָּקָ֥ר יֹֽאכַל־ תֶּֽבֶן׃ (ז) וְשִֽׁעֲשַׁ֥ע יוֹנֵ֖ק עַל־ חֻ֣ר פָּ֑תֶן וְעַל֙ מְאוּרַ֣ת צִפְעוֹנִ֔י גָּמ֖וּל יָד֥וֹ הָדָֽה׃ (ח) לֹֽא־ יָרֵ֥עוּ וְלֹֽא־ יַשְׁחִ֖יתוּ בְּכָל־ הַ֣ר קָדְשִׁ֑י כִּֽי־ מָלְאָ֣ה הָאָ֗רֶץ דֵּעָה֙ אֶת־ יְהוָ֔ה כַּמַּ֖יִם לַיָּ֥ם מְכַסִּֽים׃ (ט) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא שֹׁ֣רֶשׁ יִשַׁ֗י אֲשֶׁ֤ר עֹמֵד֙ לְנֵ֣ס עַמִּ֔ים אֵלָ֖יו גּוֹיִ֣ם יִדְרֹ֑שׁוּ וְהָיְתָ֥ה מְנֻחָת֖וֹ כָּבֽוֹד׃ (י) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא יוֹסִ֨יף אֲדֹנָ֤י ׀ שֵׁנִית֙ יָד֔וֹ לִקְנ֖וֹת אֶת־ שְׁאָ֣ר עַמּ֑וֹ אֲשֶׁ֣ר יִשָּׁאֵר֩ מֵאַשּׁ֨וּר וּמִמִּצְרַ֜יִם וּמִפַּתְר֣וֹס וּמִכּ֗וּשׁ וּמֵעֵילָ֤ם וּמִשִּׁנְעָר֙ וּמֵ֣חֲמָ֔ת וּמֵאִיֵּ֖י הַיָּֽם׃ (יא) וְנָשָׂ֥א נֵס֙ לַגּוֹיִ֔ם וְאָסַ֖ף נִדְחֵ֣י יִשְׂרָאֵ֑ל וּנְפֻצ֤וֹת יְהוּדָה֙ יְקַבֵּ֔ץ מֵאַרְבַּ֖ע כַּנְפ֥וֹת הָאָֽרֶץ׃ (יב) וְסָ֙רָה֙ קִנְאַ֣ת אֶפְרַ֔יִם וְצֹרְרֵ֥י יְהוּדָ֖ה יִכָּרֵ֑תוּ אֶפְרַ֙יִם֙ לֹֽא־ יְקַנֵּ֣א אֶת־ יְהוּדָ֔ה וִֽיהוּדָ֖ה לֹֽא־ יָצֹ֥ר אֶת־ אֶפְרָֽיִם׃ (יג) וְעָפ֨וּ בְכָתֵ֤ף פְּלִשְׁתִּים֙ יָ֔מָּה יַחְדָּ֖ו יָבֹ֣זּוּ אֶת־ בְּנֵי־ קֶ֑דֶם אֱד֤וֹם וּמוֹאָב֙ מִשְׁל֣וֹח יָדָ֔ם וּבְנֵ֥י עַמּ֖וֹן מִשְׁמַעְתָּֽם׃ (יד) וְהֶחֱרִ֣ים יְהוָ֗ה אֵ֚ת לְשׁ֣וֹן יָם־ מִצְרַ֔יִם וְהֵנִ֥יף יָד֛וֹ עַל־ הַנָּהָ֖ר בַּעְיָ֣ם רוּח֑וֹ וְהִכָּ֙הוּ֙ לְשִׁבְעָ֣ה נְחָלִ֔ים וְהִדְרִ֖יךְ בַּנְּעָלִֽים׃ (טו) וְהָיְתָ֣ה מְסִלָּ֔ה לִשְׁאָ֣ר עַמּ֔וֹ אֲשֶׁ֥ר יִשָּׁאֵ֖ר מֵֽאַשּׁ֑וּר כַּאֲשֶׁ֤ר הָֽיְתָה֙ לְיִשְׂרָאֵ֔ל בְּי֥וֹם עֲלֹת֖וֹ מֵאֶ֥רֶץ מִצְרָֽיִם׃ (טז) וְאָֽמַרְתָּ֙ בַּיּ֣וֹם הַה֔וּא אוֹדְךָ֣ יְהוָ֔ה כִּ֥י אָנַ֖פְתָּ בִּ֑י יָשֹׁ֥ב אַפְּךָ֖ וּֽתְנַחֲמֵֽנִי׃ (יז) הִנֵּ֨ה אֵ֧ל יְשׁוּעָתִ֛י אֶבְטַ֖ח וְלֹ֣א אֶפְחָ֑ד כִּֽי־ עָזִּ֤י וְזִמְרָת֙ יָ֣הּ יְהוָ֔ה וַֽיְהִי־ לִ֖י לִֽישׁוּעָֽה׃ (יח) וּשְׁאַבְתֶּם־ מַ֖יִם בְּשָׂשׂ֑וֹן מִמַּעַיְנֵ֖י הַיְשׁוּעָֽה׃ (יט) וַאֲמַרְתֶּ֞ם בַּיּ֣וֹם הַה֗וּא הוֹד֤וּ לַֽיהוָה֙ קִרְא֣וּ בִשְׁמ֔וֹ הוֹדִ֥יעוּ בָֽעַמִּ֖ים עֲלִֽילֹתָ֑יו הַזְכִּ֕ירוּ כִּ֥י נִשְׂגָּ֖ב שְׁמֽוֹ׃ (כ) זַמְּר֣וּ יְהוָ֔ה כִּ֥י גֵא֖וּת עָשָׂ֑ה (מידעת) מוּדַ֥עַת זֹ֖את בְּכָל־ הָאָֽרֶץ׃ (כא) צַהֲלִ֥י וָרֹ֖נִּי יוֹשֶׁ֣בֶת צִיּ֑וֹן כִּֽי־ גָד֥וֹל בְּקִרְבֵּ֖ךְ קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (כב) מַשָּׂ֖א בָּבֶ֑ל אֲשֶׁ֣ר חָזָ֔ה יְשַׁעְיָ֖הוּ בֶּן־ אָמֽוֹץ׃ (כג) עַ֤ל הַר־ נִשְׁפֶּה֙ שְֽׂאוּ־ נֵ֔ס הָרִ֥ימוּ ק֖וֹל לָהֶ֑ם הָנִ֣יפוּ יָ֔ד וְיָבֹ֖אוּ פִּתְחֵ֥י נְדִיבִֽים׃ (כד) אֲנִ֥י צִוֵּ֖יתִי לִמְקֻדָּשָׁ֑י גַּ֣ם קָרָ֤אתִי גִבּוֹרַי֙ לְאַפִּ֔י עַלִּיזֵ֖י גַּאֲוָתִֽי׃ (כה) ק֥וֹל הָמ֛וֹן בֶּֽהָרִ֖ים דְּמ֣וּת עַם־ רָ֑ב ק֠וֹל שְׁא֞וֹן מַמְלְכ֤וֹת גּוֹיִם֙ נֶֽאֱסָפִ֔ים יְהוָ֣ה צְבָא֔וֹת מְפַקֵּ֖ד צְבָ֥א מִלְחָמָֽה׃ (כו) בָּאִ֛ים מֵאֶ֥רֶץ מֶרְחָ֖ק מִקְצֵ֣ה הַשָּׁמָ֑יִם יְהוָה֙ וּכְלֵ֣י זַעְמ֔וֹ לְחַבֵּ֖ל כָּל־ הָאָֽרֶץ׃ (כז) הֵילִ֕ילוּ כִּ֥י קָר֖וֹב י֣וֹם יְהוָ֑ה כְּשֹׁ֖ד מִשַּׁדַּ֥י יָבֽוֹא׃ (כח) עַל־ כֵּ֖ן כָּל־ יָדַ֣יִם תִּרְפֶּ֑ינָה וְכָל־ לְבַ֥ב אֱנ֖וֹשׁ יִמָּס׃ (כט) וְֽנִבְהָ֓לוּ ׀ צִירִ֤ים וַֽחֲבָלִים֙ יֹֽאחֵז֔וּן כַּיּוֹלֵדָ֖ה יְחִיל֑וּן אִ֤ישׁ אֶל־ רֵעֵ֙הוּ֙ יִתְמָ֔הוּ פְּנֵ֥י לְהָבִ֖ים פְּנֵיהֶֽם׃ (ל) הִנֵּ֤ה יוֹם־ יְהוָה֙ בָּ֔א אַכְזָרִ֥י וְעֶבְרָ֖ה וַחֲר֣וֹן אָ֑ף לָשׂ֤וּם הָאָ֙רֶץ֙ לְשַׁמָּ֔ה וְחַטָּאֶ֖יהָ יַשְׁמִ֥יד מִמֶּֽנָּה׃ (לא) כִּֽי־ כוֹכְבֵ֤י הַשָּׁמַ֙יִם֙ וּכְסִ֣ילֵיהֶ֔ם לֹ֥א יָהֵ֖לּוּ אוֹרָ֑ם חָשַׁ֤ךְ הַשֶּׁ֙מֶשׁ֙ בְּצֵאת֔וֹ וְיָרֵ֖חַ לֹֽא־ יַגִּ֥יהַ אוֹרֽוֹ׃ (לב) וּפָקַדְתִּ֤י עַל־ תֵּבֵל֙ רָעָ֔ה וְעַל־ רְשָׁעִ֖ים עֲוֺנָ֑ם וְהִשְׁבַּתִּי֙ גְּא֣וֹן זֵדִ֔ים וְגַאֲוַ֥ת עָרִיצִ֖ים אַשְׁפִּֽיל׃ (לג) אוֹקִ֥יר אֱנ֖וֹשׁ מִפָּ֑ז וְאָדָ֖ם מִכֶּ֥תֶם אוֹפִֽיר׃ (לד) עַל־ כֵּן֙ שָׁמַ֣יִם אַרְגִּ֔יז וְתִרְעַ֥שׁ הָאָ֖רֶץ מִמְּקוֹמָ֑הּ בְּעֶבְרַת֙ יְהוָ֣ה צְבָא֔וֹת וּבְי֖וֹם חֲר֥וֹן אַפּֽוֹ׃ (לה) וְהָיָה֙ כִּצְבִ֣י מֻדָּ֔ח וּכְצֹ֖אן וְאֵ֣ין מְקַבֵּ֑ץ אִ֤ישׁ אֶל־ עַמּוֹ֙ יִפְנ֔וּ וְאִ֥ישׁ אֶל־ אַרְצ֖וֹ יָנֽוּסוּ׃ (לו) כָּל־ הַנִּמְצָ֖א יִדָּקֵ֑ר וְכָל־ הַנִּסְפֶּ֖ה יִפּ֥וֹל בֶּחָֽרֶב׃ (לז) וְעֹלְלֵיהֶ֥ם יְרֻטְּשׁ֖וּ לְעֵֽינֵיהֶ֑ם יִשַּׁ֙סּוּ֙ בָּֽתֵּיהֶ֔ם וּנְשֵׁיהֶ֖ם (תשגלנה) תִּשָּׁכַֽבְנָה׃ (לח) הִנְנִ֛י מֵעִ֥יר עֲלֵיהֶ֖ם אֶת־ מָדָ֑י אֲשֶׁר־ כֶּ֙סֶף֙ לֹ֣א יַחְשֹׁ֔בוּ וְזָהָ֖ב לֹ֥א יַחְפְּצוּ־ בֽוֹ׃ (לט) וּקְשָׁת֖וֹת נְעָרִ֣ים תְּרַטַּ֑שְׁנָה וּפְרִי־ בֶ֙טֶן֙ לֹ֣א יְרַחֵ֔מוּ עַל־ בָּנִ֖ים לֹֽא־ תָח֥וּס עֵינָֽם׃ (מ) וְהָיְתָ֤ה בָבֶל֙ צְבִ֣י מַמְלָכ֔וֹת תִּפְאֶ֖רֶת גְּא֣וֹן כַּשְׂדִּ֑ים כְּמַהְפֵּכַ֣ת אֱלֹהִ֔ים אֶת־ סְדֹ֖ם וְאֶת־ עֲמֹרָֽה׃ (מא) לֹֽא־ תֵשֵׁ֣ב לָנֶ֔צַח וְלֹ֥א תִשְׁכֹּ֖ן עַד־ דּ֣וֹר וָד֑וֹר וְלֹֽא־ יַהֵ֥ל שָׁם֙ עֲרָבִ֔י וְרֹעִ֖ים לֹא־ יַרְבִּ֥צוּ שָֽׁם׃ (מב) וְרָבְצוּ־ שָׁ֣ם צִיִּ֔ים וּמָלְא֥וּ בָתֵּיהֶ֖ם אֹחִ֑ים וְשָׁ֤כְנוּ שָׁם֙ בְּנ֣וֹת יַֽעֲנָ֔ה וּשְׂעִירִ֖ים יְרַקְּדוּ־ שָֽׁם׃ (מג) וְעָנָ֤ה אִיִּים֙ בְּאַלְמנוֹתָ֔יו וְתַנִּ֖ים בְּהֵ֣יכְלֵי עֹ֑נֶג וְקָר֤וֹב לָבוֹא֙ עִתָּ֔הּ וְיָמֶ֖יהָ לֹ֥א יִמָּשֵֽׁכוּ׃ תחילת פרק יד (מד) כִּי֩ יְרַחֵ֨ם יְהוָ֜ה אֶֽת־ יַעֲקֹ֗ב וּבָחַ֥ר עוֹד֙ בְּיִשְׂרָאֵ֔ל וְהִנִּיחָ֖ם עַל־ אַדְמָתָ֑ם וְנִלְוָ֤ה הַגֵּר֙ עֲלֵיהֶ֔ם וְנִסְפְּח֖וּ עַל־ בֵּ֥ית יַעֲקֹֽב׃

ישעיהו סדר ו

(א) וּלְקָח֣וּם עַמִּים֮ וֶהֱבִיא֣וּם אֶל־ מְקוֹמָם֒ וְהִֽתְנַחֲל֣וּם בֵּֽית־ יִשְׂרָאֵ֗ל עַ֚ל אַדְמַ֣ת יְהוָ֔ה לַעֲבָדִ֖ים וְלִשְׁפָח֑וֹת וְהָיוּ֙ שֹׁבִ֣ים לְשֹֽׁבֵיהֶ֔ם וְרָד֖וּ בְּנֹגְשֵׂיהֶֽם׃ (ב) וְהָיָ֗ה בְּי֨וֹם הָנִ֤יחַ יְהוָה֙ לְךָ֔ מֵֽעָצְבְּךָ֖ וּמֵרָגְזֶ֑ךָ וּמִן־ הָעֲבֹדָ֥ה הַקָּשָׁ֖ה אֲשֶׁ֥ר עֻבַּד־ בָּֽךְ׃ (ג) וְנָשָׂ֜אתָ הַמָּשָׁ֥ל הַזֶּ֛ה עַל־ מֶ֥לֶךְ בָּבֶ֖ל וְאָמָ֑רְתָּ אֵ֚יךְ שָׁבַ֣ת נֹגֵ֔שׂ שָׁבְתָ֖ה מַדְהֵבָֽה׃ (ד) שָׁבַ֥ר יְהוָ֖ה מַטֵּ֣ה רְשָׁעִ֑ים שֵׁ֖בֶט מֹשְׁלִֽים׃ (ה) מַכֶּ֤ה עַמִּים֙ בְּעֶבְרָ֔ה מַכַּ֖ת בִּלְתִּ֣י סָרָ֑ה רֹדֶ֤ה בָאַף֙ גּוֹיִ֔ם מֻרְדָּ֖ף בְּלִ֥י חָשָֽׂךְ׃ (ו) נָ֥חָה שָׁקְטָ֖ה כָּל־ הָאָ֑רֶץ פָּצְח֖וּ רִנָּֽה׃ (ז) גַּם־ בְּרוֹשִׁ֛ים שָׂמְח֥וּ לְךָ֖ אַרְזֵ֣י לְבָנ֑וֹן מֵאָ֣ז שָׁכַ֔בְתָּ לֹֽא־ יַעֲלֶ֥ה הַכֹּרֵ֖ת עָלֵֽינוּ׃ (ח) שְׁא֗וֹל מִתַּ֛חַת רָגְזָ֥ה לְךָ֖ לִקְרַ֣את בּוֹאֶ֑ךָ עוֹרֵ֨ר לְךָ֤ רְפָאִים֙ כָּל־ עַתּ֣וּדֵי אָ֔רֶץ הֵקִים֙ מִכִּסְאוֹתָ֔ם כֹּ֖ל מַלְכֵ֥י גוֹיִֽם׃ (ט) כֻּלָּ֣ם יַֽעֲנ֔וּ וְיֹאמְר֖וּ אֵלֶ֑יךָ גַּם־ אַתָּ֛ה חֻלֵּ֥יתָ כָמ֖וֹנוּ אֵלֵ֥ינוּ נִמְשָֽׁלְתָּ׃ (י) הוּרַ֥ד שְׁא֛וֹל גְּאוֹנֶ֖ךָ הֶמְיַ֣ת נְבָלֶ֑יךָ תַּחְתֶּ֙יךָ֙ יֻצַּ֣ע רִמָּ֔ה וּמְכַסֶּ֖יךָ תּוֹלֵעָֽה׃ (יא) אֵ֛יךְ נָפַ֥לְתָּ מִשָּׁמַ֖יִם הֵילֵ֣ל בֶּן־ שָׁ֑חַר נִגְדַּ֣עְתָּ לָאָ֔רֶץ חוֹלֵ֖שׁ עַל־ גּוֹיִֽם׃ (יב) וְאַתָּ֞ה אָמַ֤רְתָּ בִֽלְבָבְךָ֙ הַשָּׁמַ֣יִם אֶֽעֱלֶ֔ה מִמַּ֥עַל לְכֽוֹכְבֵי־ אֵ֖ל אָרִ֣ים כִּסְאִ֑י וְאֵשֵׁ֥ב בְּהַר־ מוֹעֵ֖ד בְּיַרְכְּתֵ֥י צָפֽוֹן׃ (יג) אֶעֱלֶ֖ה עַל־ בָּ֣מֳתֵי עָ֑ב אֶדַּמֶּ֖ה לְעֶלְיֽוֹן׃ (יד) אַ֧ךְ אֶל־ שְׁא֛וֹל תּוּרָ֖ד אֶל־ יַרְכְּתֵי־ בֽוֹר׃ (טו) רֹאֶ֙יךָ֙ אֵלֶ֣יךָ יַשְׁגִּ֔יחוּ אֵלֶ֖יךָ יִתְבּוֹנָ֑נוּ הֲזֶ֤ה הָאִישׁ֙ מַרְגִּ֣יז הָאָ֔רֶץ מַרְעִ֖ישׁ מַמְלָכֽוֹת׃ (טז) שָׂ֥ם תֵּבֵ֛ל כַּמִּדְבָּ֖ר וְעָרָ֣יו הָרָ֑ס אֲסִירָ֖יו לֹא־ פָ֥תַח בָּֽיְתָה׃ (יז) כָּל־ מַלְכֵ֥י גוֹיִ֖ם כֻּלָּ֑ם שָׁכְב֥וּ בְכָב֖וֹד אִ֥ישׁ בְּבֵיתֽוֹ׃ (יח) וְאַתָּ֞ה הָשְׁלַ֤כְתָּ מִֽקִּבְרְךָ֙ כְּנֵ֣צֶר נִתְעָ֔ב לְב֥וּשׁ הֲרֻגִ֖ים מְטֹ֣עֲנֵי חָ֑רֶב יוֹרְדֵ֥י אֶל־ אַבְנֵי־ ב֖וֹר כְּפֶ֥גֶר מוּבָֽס׃ (יט) לֹֽא־ תֵחַ֤ד אִתָּם֙ בִּקְבוּרָ֔ה כִּֽי־ אַרְצְךָ֥ שִׁחַ֖תָּ עַמְּךָ֣ הָרָ֑גְתָּ לֹֽא־ יִקָּרֵ֥א לְעוֹלָ֖ם זֶ֥רַע מְרֵעִֽים׃ (כ) הָכִ֧ינוּ לְבָנָ֛יו מַטְבֵּ֖חַ בַּעֲוֺ֣ן אֲבוֹתָ֑ם בַּל־ יָקֻ֙מוּ֙ וְיָ֣רְשׁוּ אָ֔רֶץ וּמָלְא֥וּ פְנֵֽי־ תֵבֵ֖ל עָרִֽים׃ (כא) וְקַמְתִּ֣י עֲלֵיהֶ֔ם נְאֻ֖ם יְהוָ֣ה צְבָא֑וֹת וְהִכְרַתִּ֨י לְבָבֶ֜ל שֵׁ֥ם וּשְׁאָ֛ר וְנִ֥ין וָנֶ֖כֶד נְאֻם־ יְהוָֽה׃ (כב) וְשַׂמְתִּ֛יהָ לְמוֹרַ֥שׁ קִפֹּ֖ד וְאַגְמֵי־ מָ֑יִם וְטֵֽאטֵאתִ֙יהָ֙ בְּמַטְאֲטֵ֣א הַשְׁמֵ֔ד נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כג) נִשְׁבַּ֛ע יְהוָ֥ה צְבָא֖וֹת לֵאמֹ֑ר אִם־ לֹ֞א כַּאֲשֶׁ֤ר דִּמִּ֙יתִי֙ כֵּ֣ן הָיָ֔תָה וְכַאֲשֶׁ֥ר יָעַ֖צְתִּי הִ֥יא תָקֽוּם׃ (כד) לִשְׁבֹּ֤ר אַשּׁוּר֙ בְּאַרְצִ֔י וְעַל־ הָרַ֖י אֲבוּסֶ֑נּוּ וְסָ֤ר מֵֽעֲלֵיהֶם֙ עֻלּ֔וֹ וְסֻ֨בֳּל֔וֹ מֵעַ֥ל שִׁכְמ֖וֹ יָסֽוּר׃ (כה) זֹ֛את הָעֵצָ֥ה הַיְּעוּצָ֖ה עַל־ כָּל־ הָאָ֑רֶץ וְזֹ֛את הַיָּ֥ד הַנְּטוּיָ֖ה עַל־ כָּל־ הַגּוֹיִֽם׃ (כו) כִּֽי־ יְהוָ֧ה צְבָא֛וֹת יָעָ֖ץ וּמִ֣י יָפֵ֑ר וְיָד֥וֹ הַנְּטוּיָ֖ה וּֽמִ֥י יְשִׁיבֶֽנָּה׃ (כז) בִּשְׁנַת־ מ֖וֹת הַמֶּ֣לֶךְ אָחָ֑ז הָיָ֖ה הַמַּשָּׂ֥א הַזֶּֽה׃ (כח) אַֽל־ תִּשְׂמְחִ֤י פְלֶ֙שֶׁת֙ כֻּלֵּ֔ךְ כִּ֥י נִשְׁבַּ֖ר שֵׁ֣בֶט מַכֵּ֑ךְ כִּֽי־ מִשֹּׁ֤רֶשׁ נָחָשׁ֙ יֵ֣צֵא צֶ֔פַע וּפִרְי֖וֹ שָׂרָ֥ף מְעוֹפֵֽף׃ (כט) וְרָעוּ֙ בְּכוֹרֵ֣י דַלִּ֔ים וְאֶבְיוֹנִ֖ים לָבֶ֣טַח יִרְבָּ֑צוּ וְהֵמַתִּ֤י בָֽרָעָב֙ שָׁרְשֵׁ֔ךְ וּשְׁאֵרִיתֵ֖ךְ יַהֲרֹֽג׃ (ל) הֵילִ֤ילִֽי שַׁ֙עַר֙ זַֽעֲקִי־ עִ֔יר נָמ֖וֹג פְּלֶ֣שֶׁת כֻּלֵּ֑ךְ כִּ֤י מִצָּפוֹן֙ עָשָׁ֣ן בָּ֔א וְאֵ֥ין בּוֹדֵ֖ד בְּמוֹעָדָֽיו׃ (לא) וּמַֽה־ יַּעֲנֶ֖ה מַלְאֲכֵי־ ג֑וֹי כִּ֤י יְהוָה֙ יִסַּ֣ד צִיּ֔וֹן וּבָ֥הּ יֶחֱס֖וּ עֲנִיֵּ֥י עַמּֽוֹ׃ סוף פרק יד (לב) מַשָּׂ֖א מוֹאָ֑ב כִּ֠י בְּלֵ֞יל שֻׁדַּ֨ד עָ֤ר מוֹאָב֙ נִדְמָ֔ה כִּ֗י בְּלֵ֛יל שֻׁדַּ֥ד קִיר־ מוֹאָ֖ב נִדְמָֽה׃ (לג) עָלָ֨ה הַבַּ֧יִת וְדִיבֹ֛ן הַבָּמ֖וֹת לְבֶ֑כִי עַל־ נְב֞וֹ וְעַ֤ל מֵֽידְבָא֙ מוֹאָ֣ב יְיֵלִ֔יל בְּכָל־ רֹאשָׁ֣יו קָרְחָ֔ה כָּל־ זָקָ֖ן גְּרוּעָֽה׃ (לד) בְּחוּצֹתָ֖יו חָ֣גְרוּ שָׂ֑ק עַ֣ל גַּגּוֹתֶ֧יהָ וּבִרְחֹבֹתֶ֛יהָ כֻּלֹּ֥ה יְיֵלִ֖יל יֹרֵ֥ד בַּבֶּֽכִי׃ (לה) וַתִּזְעַ֤ק חֶשְׁבּוֹן֙ וְאֶלְעָלֵ֔ה עַד־ יַ֖הַץ נִשְׁמַ֣ע קוֹלָ֑ם עַל־ כֵּ֗ן חֲלֻצֵ֤י מוֹאָב֙ יָרִ֔יעוּ נַפְשׁ֖וֹ יָ֥רְעָה לּֽוֹ׃ (לו) לִבִּי֙ לְמוֹאָ֣ב יִזְעָ֔ק בְּרִיחֶ֕הָ עַד־ צֹ֖עַר עֶגְלַ֣ת שְׁלִשִׁיָּ֑ה כִּ֣י ׀ מַעֲלֵ֣ה הַלּוּחִ֗ית בִּבְכִי֙ יַֽעֲלֶה־ בּ֔וֹ כִּ֚י דֶּ֣רֶךְ חוֹרֹנַ֔יִם זַעֲקַת־ שֶׁ֖בֶר יְעֹעֵֽרוּ׃ (לז) כִּֽי־ מֵ֥י נִמְרִ֖ים מְשַׁמּ֣וֹת יִֽהְי֑וּ כִּֽי־ יָבֵ֤שׁ חָצִיר֙ כָּ֣לָה דֶ֔שֶׁא יֶ֖רֶק לֹ֥א הָיָֽה׃ (לח) עַל־ כֵּ֖ן יִתְרָ֣ה עָשָׂ֑ה וּפְקֻדָּתָ֔ם עַ֛ל נַ֥חַל הָעֲרָבִ֖ים יִשָּׂאֽוּם׃ (לט) כִּֽי־ הִקִּ֥יפָה הַזְּעָקָ֖ה אֶת־ גְּב֣וּל מוֹאָ֑ב עַד־ אֶגְלַ֙יִם֙ יִלְלָתָ֔הּ וּבְאֵ֥ר אֵילִ֖ים יִלְלָתָֽהּ׃ (מ) כִּ֣י מֵ֤י דִימוֹן֙ מָ֣לְאוּ דָ֔ם כִּֽי־ אָשִׁ֥ית עַל־ דִּימ֖וֹן נוֹסָפ֑וֹת לִפְלֵיטַ֤ת מוֹאָב֙ אַרְיֵ֔ה וְלִשְׁאֵרִ֖ית אֲדָמָֽה׃ (מא) שִׁלְחוּ־ כַ֥ר מֹשֵֽׁל־ אֶ֖רֶץ מִסֶּ֣לַע מִדְבָּ֑רָה אֶל־ הַ֖ר בַּת־ צִיּֽוֹן׃ (מב) וְהָיָ֥ה כְעוֹף־ נוֹדֵ֖ד קֵ֣ן מְשֻׁלָּ֑ח תִּֽהְיֶ֙ינָה֙ בְּנ֣וֹת מוֹאָ֔ב מַעְבָּרֹ֖ת לְאַרְנֽוֹן׃ (מג) (הביאו) הָבִ֤יאִי עֵצָה֙ עֲשׂ֣וּ פְלִילָ֔ה שִׁ֧יתִי כַלַּ֛יִל צִלֵּ֖ךְ בְּת֣וֹךְ צָהֳרָ֑יִם סַתְּרִי֙ נִדָּחִ֔ים נֹדֵ֖ד אַל־ תְּגַלִּֽי׃ (מד) יָג֤וּרוּ בָךְ֙ נִדָּחַ֔י מוֹאָ֛ב הֱוִי־ סֵ֥תֶר לָ֖מוֹ מִפְּנֵ֣י שׁוֹדֵ֑ד כִּֽי־ אָפֵ֤ס הַמֵּץ֙ כָּ֣לָה שֹׁ֔ד תַּ֥מּוּ רֹמֵ֖ס מִן־ הָאָֽרֶץ׃
