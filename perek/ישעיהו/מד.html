ישעיהו פרק מד לפי החלוקה לסדרים

פרק מד בספר ישעיהו (28 פסוקים) נמצא בישעיהו סדר יז (פסוקים נז–סא) ובישעיהו סדר יח (פסוקים א–כג).

ישעיהו סדר יז

(א) רִאשׁ֥וֹן לְצִיּ֖וֹן הִנֵּ֣ה הִנָּ֑ם וְלִירוּשָׁלִַ֖ם מְבַשֵּׂ֥ר אֶתֵּֽן׃ (ב) וְאֵ֙רֶא֙ וְאֵ֣ין אִ֔ישׁ וּמֵאֵ֖לֶּה וְאֵ֣ין יוֹעֵ֑ץ וְאֶשְׁאָלֵ֖ם וְיָשִׁ֥יבוּ דָבָֽר׃ (ג) הֵ֣ן כֻּלָּ֔ם אָ֥וֶן אֶ֖פֶס מַעֲשֵׂיהֶ֑ם ר֥וּחַ וָתֹ֖הוּ נִסְכֵּיהֶֽם׃ (ד) הֵ֤ן עַבְדִּי֙ אֶתְמָךְ־ בּ֔וֹ בְּחִירִ֖י רָצְתָ֣ה נַפְשִׁ֑י נָתַ֤תִּי רוּחִי֙ עָלָ֔יו מִשְׁפָּ֖ט לַגּוֹיִ֥ם יוֹצִֽיא׃ (ה) לֹ֥א יִצְעַ֖ק וְלֹ֣א יִשָּׂ֑א וְלֹֽא־ יַשְׁמִ֥יעַ בַּח֖וּץ קוֹלֽוֹ׃ (ו) קָנֶ֤ה רָצוּץ֙ לֹ֣א יִשְׁבּ֔וֹר וּפִשְׁתָּ֥ה כֵהָ֖ה לֹ֣א יְכַבֶּ֑נָּה לֶאֱמֶ֖ת יוֹצִ֥יא מִשְׁפָּֽט׃ (ז) לֹ֤א יִכְהֶה֙ וְלֹ֣א יָר֔וּץ עַד־ יָשִׂ֥ים בָּאָ֖רֶץ מִשְׁפָּ֑ט וּלְתוֹרָת֖וֹ אִיִּ֥ים יְיַחֵֽילוּ׃ (ח) כֹּֽה־ אָמַ֞ר הָאֵ֣ל ׀ יְהוָ֗ה בּוֹרֵ֤א הַשָּׁמַ֙יִם֙ וְנ֣וֹטֵיהֶ֔ם רֹקַ֥ע הָאָ֖רֶץ וְצֶאֱצָאֶ֑יהָ נֹתֵ֤ן נְשָׁמָה֙ לָעָ֣ם עָלֶ֔יהָ וְר֖וּחַ לַהֹלְכִ֥ים בָּֽהּ׃ (ט) אֲנִ֧י יְהוָ֛ה קְרָאתִ֥יךָֽ בְצֶ֖דֶק וְאַחְזֵ֣ק בְּיָדֶ֑ךָ וְאֶצָּרְךָ֗ וְאֶתֶּנְךָ֛ לִבְרִ֥ית עָ֖ם לְא֥וֹר גּוֹיִֽם׃ (י) לִפְקֹ֖חַ עֵינַ֣יִם עִוְר֑וֹת לְהוֹצִ֤יא מִמַּסְגֵּר֙ אַסִּ֔יר מִבֵּ֥ית כֶּ֖לֶא יֹ֥שְׁבֵי חֹֽשֶׁךְ׃ (יא) אֲנִ֥י יְהוָ֖ה ה֣וּא שְׁמִ֑י וּכְבוֹדִי֙ לְאַחֵ֣ר לֹֽא־ אֶתֵּ֔ן וּתְהִלָּתִ֖י לַפְּסִילִֽים׃ (יב) הָרִֽאשֹׁנ֖וֹת הִנֵּה־ בָ֑אוּ וַֽחֲדָשׁוֹת֙ אֲנִ֣י מַגִּ֔יד בְּטֶ֥רֶם תִּצְמַ֖חְנָה אַשְׁמִ֥יע אֶתְכֶֽם׃ (יג) שִׁ֤ירוּ לַֽיהוָה֙ שִׁ֣יר חָדָ֔שׁ תְּהִלָּת֖וֹ מִקְצֵ֣ה הָאָ֑רֶץ יוֹרְדֵ֤י הַיָּם֙ וּמְלֹא֔וֹ אִיִּ֖ים וְיֹשְׁבֵיהֶֽם׃ (יד) יִשְׂא֤וּ מִדְבָּר֙ וְעָרָ֔יו חֲצֵרִ֖ים תֵּשֵׁ֣ב קֵדָ֑ר יָרֹ֙נּוּ֙ יֹ֣שְׁבֵי סֶ֔לַע מֵרֹ֥אשׁ הָרִ֖ים יִצְוָֽחוּ׃ (טו) יָשִׂ֥ימוּ לַֽיהוָ֖ה כָּב֑וֹד וּתְהִלָּת֖וֹ בָּאִיִּ֥ים יַגִּֽידוּ׃ (טז) יְהוָה֙ כַּגִּבּ֣וֹר יֵצֵ֔א כְּאִ֥ישׁ מִלְחָמ֖וֹת יָעִ֣יר קִנְאָ֑ה יָרִ֙יעַ֙ אַף־ יַצְרִ֔יחַ עַל־ אֹיְבָ֖יו יִתְגַּבָּֽר׃ (יז) הֶחֱשֵׁ֙יתִי֙ מֵֽעוֹלָ֔ם אַחֲרִ֖ישׁ אֶתְאַפָּ֑ק כַּיּוֹלֵדָ֣ה אֶפְעֶ֔ה אֶשֹּׁ֥ם וְאֶשְׁאַ֖ף יָֽחַד׃ (יח) אַחֲרִ֤יב הָרִים֙ וּגְבָע֔וֹת וְכָל־ עֶשְׂבָּ֖ם אוֹבִ֑ישׁ וְשַׂמְתִּ֤י נְהָרוֹת֙ לָֽאִיִּ֔ים וַאֲגַמִּ֖ים אוֹבִֽישׁ׃ (יט) וְהוֹלַכְתִּ֣י עִוְרִ֗ים בְּדֶ֙רֶךְ֙ לֹ֣א יָדָ֔עוּ בִּנְתִיב֥וֹת לֹֽא־ יָדְע֖וּ אַדְרִיכֵ֑ם אָשִׂים֩ מַחְשָׁ֨ךְ לִפְנֵיהֶ֜ם לָא֗וֹר וּמַֽעֲקַשִּׁים֙ לְמִישׁ֔וֹר אֵ֚לֶּה הַדְּבָרִ֔ים עֲשִׂיתִ֖ם וְלֹ֥א עֲזַבְתִּֽים׃ (כ) נָסֹ֤גוּ אָחוֹר֙ יֵבֹ֣שׁוּ בֹ֔שֶׁת הַבֹּטְחִ֖ים בַּפָּ֑סֶל הָאֹמְרִ֥ים לְמַסֵּכָ֖ה אַתֶּ֥ם אֱלֹהֵֽינוּ׃ (כא) הַחֵרְשִׁ֖ים שְׁמָ֑עוּ וְהַעִוְרִ֖ים הַבִּ֥יטוּ לִרְאֽוֹת׃ (כב) מִ֤י עִוֵּר֙ כִּ֣י אִם־ עַבְדִּ֔י וְחֵרֵ֖שׁ כְּמַלְאָכִ֣י אֶשְׁלָ֑ח מִ֤י עִוֵּר֙ כִּמְשֻׁלָּ֔ם וְעִוֵּ֖ר כְּעֶ֥בֶד יְהוָֽה׃ (כג) (ראית) רָא֥וֹת רַבּ֖וֹת וְלֹ֣א תִשְׁמֹ֑ר פָּק֥וֹחַ אָזְנַ֖יִם וְלֹ֥א יִשְׁמָֽע׃ (כד) יְהוָ֥ה חָפֵ֖ץ לְמַ֣עַן צִדְק֑וֹ יַגְדִּ֥יל תּוֹרָ֖ה וְיַאְדִּֽיר׃ (כה) וְהוּא֮ עַם־ בָּז֣וּז וְשָׁסוּי֒ הָפֵ֤חַ בַּֽחוּרִים֙ כֻּלָּ֔ם וּבְבָתֵּ֥י כְלָאִ֖ים הָחְבָּ֑אוּ הָי֤וּ לָבַז֙ וְאֵ֣ין מַצִּ֔יל מְשִׁסָּ֖ה וְאֵין־ אֹמֵ֥ר הָשַֽׁב׃ (כו) מִ֥י בָכֶ֖ם יַאֲזִ֣ין זֹ֑את יַקְשִׁ֥ב וְיִשְׁמַ֖ע לְאָחֽוֹר׃ (כז) מִֽי־ נָתַ֨ן (למשוסה) לִמְשִׁסָּ֧ה יַעֲקֹ֛ב וְיִשְׂרָאֵ֥ל לְבֹזְזִ֖ים הֲל֣וֹא יְהוָ֑ה ז֚וּ חָטָ֣אנוּ ל֔וֹ וְלֹֽא־ אָב֤וּ בִדְרָכָיו֙ הָל֔וֹךְ וְלֹ֥א שָׁמְע֖וּ בְּתוֹרָתֽוֹ׃ (כח) וַיִּשְׁפֹּ֤ךְ עָלָיו֙ חֵמָ֣ה אַפּ֔וֹ וֶעֱז֖וּז מִלְחָמָ֑ה וַתְּלַהֲטֵ֤הוּ מִסָּבִיב֙ וְלֹ֣א יָדָ֔ע וַתִּבְעַר־ בּ֖וֹ וְלֹא־ יָשִׂ֥ים עַל־ לֵֽב׃ (כט) וְעַתָּ֞ה כֹּֽה־ אָמַ֤ר יְהוָה֙ בֹּרַאֲךָ֣ יַעֲקֹ֔ב וְיֹצֶרְךָ֖ יִשְׂרָאֵ֑ל אַל־ תִּירָא֙ כִּ֣י גְאַלְתִּ֔יךָ קָרָ֥אתִי בְשִׁמְךָ֖ לִי־ אָֽתָּה׃ (ל) כִּֽי־ תַעֲבֹ֤ר בַּמַּ֙יִם֙ אִתְּךָ־ אָ֔נִי וּבַנְּהָר֖וֹת לֹ֣א יִשְׁטְפ֑וּךָ כִּֽי־ תֵלֵ֤ךְ בְּמוֹ־ אֵשׁ֙ לֹ֣א תִכָּוֶ֔ה וְלֶהָבָ֖ה לֹ֥א תִבְעַר־ בָּֽךְ׃ (לא) כִּ֗י אֲנִי֙ יְהוָ֣ה אֱלֹהֶ֔יךָ קְד֥וֹשׁ יִשְׂרָאֵ֖ל מוֹשִׁיעֶ֑ךָ נָתַ֤תִּי כָפְרְךָ֙ מִצְרַ֔יִם כּ֥וּשׁ וּסְבָ֖א תַּחְתֶּֽיךָ׃ (לב) מֵאֲשֶׁ֨ר יָקַ֧רְתָּ בְעֵינַ֛י נִכְבַּ֖דְתָּ וַאֲנִ֣י אֲהַבְתִּ֑יךָ וְאֶתֵּ֤ן אָדָם֙ תַּחְתֶּ֔יךָ וּלְאֻמִּ֖ים תַּ֥חַת נַפְשֶֽׁךָ׃ (לג) אַל־ תִּירָ֖א כִּ֣י אִתְּךָ־ אָ֑נִי מִמִּזְרָח֙ אָבִ֣יא זַרְעֶ֔ךָ וּמִֽמַּעֲרָ֖ב אֲקַבְּצֶֽךָּ׃ (לד) אֹמַ֤ר לַצָּפוֹן֙ תֵּ֔נִי וּלְתֵימָ֖ן אַל־ תִּכְלָ֑אִי הָבִ֤יאִי בָנַי֙ מֵרָח֔וֹק וּבְנוֹתַ֖י מִקְצֵ֥ה הָאָֽרֶץ׃ (לה) כֹּ֚ל הַנִּקְרָ֣א בִשְׁמִ֔י וְלִכְבוֹדִ֖י בְּרָאתִ֑יו יְצַרְתִּ֖יו אַף־ עֲשִׂיתִֽיו׃ (לו) הוֹצִ֥יא עַם־ עִוֵּ֖ר וְעֵינַ֣יִם יֵ֑שׁ וְחֵרְשִׁ֖ים וְאָזְנַ֥יִם לָֽמוֹ׃ (לז) כָּֽל־ הַגּוֹיִ֞ם נִקְבְּצ֣וּ יַחְדָּ֗ו וְיֵאָֽסְפוּ֙ לְאֻמִּ֔ים מִ֤י בָהֶם֙ יַגִּ֣יד זֹ֔את וְרִֽאשֹׁנ֖וֹת יַשְׁמִיעֻ֑נוּ יִתְּנ֤וּ עֵֽדֵיהֶם֙ וְיִצְדָּ֔קוּ וְיִשְׁמְע֖וּ וְיֹאמְר֥וּ אֱמֶֽת׃ (לח) אַתֶּ֤ם עֵדַי֙ נְאֻם־ יְהוָ֔ה וְעַבְדִּ֖י אֲשֶׁ֣ר בָּחָ֑רְתִּי לְמַ֣עַן תֵּ֠דְעוּ וְתַאֲמִ֨ינוּ לִ֤י וְתָבִ֙ינוּ֙ כִּֽי־ אֲנִ֣י ה֔וּא לְפָנַי֙ לֹא־ נ֣וֹצַר אֵ֔ל וְאַחֲרַ֖י לֹ֥א יִהְיֶֽה׃ (לט) אָנֹכִ֥י אָנֹכִ֖י יְהוָ֑ה וְאֵ֥ין מִבַּלְעָדַ֖י מוֹשִֽׁיעַ׃ (מ) אָנֹכִ֞י הִגַּ֤דְתִּי וְהוֹשַׁ֙עְתִּי֙ וְהִשְׁמַ֔עְתִּי וְאֵ֥ין בָּכֶ֖ם זָ֑ר וְאַתֶּ֥ם עֵדַ֛י נְאֻם־ יְהוָ֖ה וַֽאֲנִי־ אֵֽל׃ (מא) גַּם־ מִיּוֹם֙ אֲנִ֣י ה֔וּא וְאֵ֥ין מִיָּדִ֖י מַצִּ֑יל אֶפְעַ֖ל וּמִ֥י יְשִׁיבֶֽנָּה׃ (מב) כֹּֽה־ אָמַ֧ר יְהוָ֛ה גֹּאַלְכֶ֖ם קְד֣וֹשׁ יִשְׂרָאֵ֑ל לְמַעַנְכֶ֞ם שִׁלַּ֣חְתִּי בָבֶ֗לָה וְהוֹרַדְתִּ֤י בָֽרִיחִים֙ כֻּלָּ֔ם וְכַשְׂדִּ֖ים בָּאֳנִיּ֥וֹת רִנָּתָֽם׃ (מג) אֲנִ֥י יְהוָ֖ה קְדֽוֹשְׁכֶ֑ם בּוֹרֵ֥א יִשְׂרָאֵ֖ל מַלְכְּכֶֽם׃ (מד) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הַנּוֹתֵ֥ן בַּיָּ֖ם דָּ֑רֶךְ וּבְמַ֥יִם עַזִּ֖ים נְתִיבָֽה׃ (מה) הַמּוֹצִ֥יא רֶֽכֶב־ וָס֖וּס חַ֣יִל וְעִזּ֑וּז יַחְדָּ֤ו יִשְׁכְּבוּ֙ בַּל־ יָק֔וּמוּ דָּעֲכ֖וּ כַּפִּשְׁתָּ֥ה כָבֽוּ׃ (מו) אַֽל־ תִּזְכְּר֖וּ רִֽאשֹׁנ֑וֹת וְקַדְמֹנִיּ֖וֹת אַל־ תִּתְבֹּנָֽנוּ׃ (מז) הִנְנִ֨י עֹשֶׂ֤ה חֲדָשָׁה֙ עַתָּ֣ה תִצְמָ֔ח הֲל֖וֹא תֵֽדָע֑וּהָ אַ֣ף אָשִׂ֤ים בַּמִּדְבָּר֙ דֶּ֔רֶךְ בִּֽישִׁמ֖וֹן נְהָרֽוֹת׃ (מח) תְּכַבְּדֵ֙נִי֙ חַיַּ֣ת הַשָּׂדֶ֔ה תַּנִּ֖ים וּבְרִיחֵ֥י בַרְזֶ֖ל אֲגַדֵּֽעַ׃ (כו) וְנָתַתִּ֤י לְךָ֙ אוֹצְר֣וֹת חֹ֔שֶׁךְ וּמַטְמֻנֵ֖י מִסְתָּרִ֑ים לְמַ֣עַן תֵּדַ֗ע כִּֽי־ אֲנִ֧י יְהוָ֛ה הַקּוֹרֵ֥א בְשִׁמְךָ֖ אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (כז) לְמַ֙עַן֙ עַבְדִּ֣י יַעֲקֹ֔ב וְיִשְׂרָאֵ֖ל בְּחִירִ֑י וָאֶקְרָ֤א לְךָ֙ בִּשְׁמֶ֔ךָ אֲכַנְּךָ֖ וְלֹ֥א יְדַעְתָּֽנִי׃ (כח) אֲנִ֤י יְהוָה֙ וְאֵ֣ין ע֔וֹד זוּלָתִ֖י אֵ֣ין אֱלֹהִ֑ים אֲאַזֶּרְךָ֖ וְלֹ֥א יְדַעְתָּֽנִי׃ (כט) לְמַ֣עַן יֵדְע֗וּ מִמִּזְרַח־ שֶׁ֙מֶשׁ֙ וּמִמַּ֣עֲרָבָ֔הּ כִּי־ אֶ֖פֶס בִּלְעָדָ֑י אֲנִ֥י יְהוָ֖ה וְאֵ֥ין עֽוֹד׃ (ל) יוֹצֵ֥ר אוֹר֙ וּבוֹרֵ֣א חֹ֔שֶׁךְ עֹשֶׂ֥ה שָׁל֖וֹם וּב֣וֹרֵא רָ֑ע אֲנִ֥י יְהוָ֖ה עֹשֶׂ֥ה כָל־ אֵֽלֶּה׃ (לא) הַרְעִ֤יפוּ שָׁמַ֙יִם֙ מִמַּ֔עַל וּשְׁחָקִ֖ים יִזְּלוּ־ צֶ֑דֶק תִּפְתַּח־ אֶ֣רֶץ וְיִפְרוּ־ יֶ֗שַׁע וּצְדָקָ֤ה תַצְמִ֙יחַ֙ יַ֔חַד אֲנִ֥י יְהוָ֖ה בְּרָאתִֽיו׃ (לב) ה֗וֹי רָ֚ב אֶת־ יֹ֣צְר֔וֹ חֶ֖רֶשׂ אֶת־ חַרְשֵׂ֣י אֲדָמָ֑ה הֲיֹאמַ֨ר חֹ֤מֶר לְיֹֽצְרוֹ֙ מַֽה־ תַּעֲשֶׂ֔ה וּפָעָלְךָ֖ אֵין־ יָדַ֥יִם לֽוֹ׃ (לג) ה֛וֹי אֹמֵ֥ר לְאָ֖ב מַה־ תּוֹלִ֑יד וּלְאִשָּׁ֖ה מַה־ תְּחִילִֽין׃ (לד) כֹּֽה־ אָמַ֧ר יְהוָ֛ה קְד֥וֹשׁ יִשְׂרָאֵ֖ל וְיֹצְר֑וֹ הָאֹתִיּ֣וֹת שְׁאָל֔וּנִי עַל־ בָּנַ֛י וְעַל־ פֹּ֥עַל יָדַ֖י תְּצַוֻּֽנִי׃ (לה) אָֽנֹכִי֙ עָשִׂ֣יתִי אֶ֔רֶץ וְאָדָ֖ם עָלֶ֣יהָ בָרָ֑אתִי אֲנִ֗י יָדַי֙ נָט֣וּ שָׁמַ֔יִם וְכָל־ צְבָאָ֖ם צִוֵּֽיתִי׃ (לו) אָנֹכִי֙ הַעִירֹתִ֣הֽוּ בְצֶ֔דֶק וְכָל־ דְּרָכָ֖יו אֲיַשֵּׁ֑ר הֽוּא־ יִבְנֶ֤ה עִירִי֙ וְגָלוּתִ֣י יְשַׁלֵּ֔חַ לֹ֤א בִמְחִיר֙ וְלֹ֣א בְשֹׁ֔חַד אָמַ֖ר יְהוָ֥ה צְבָאֽוֹת׃ (לז) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה יְגִ֨יעַ מִצְרַ֥יִם וּֽסְחַר־ כּוּשׁ֮ וּסְבָאִים֮ אַנְשֵׁ֣י מִדָּה֒ עָלַ֤יִךְ יַעֲבֹ֙רוּ֙ וְלָ֣ךְ יִֽהְי֔וּ אַחֲרַ֣יִךְ יֵלֵ֔כוּ בַּזִּקִּ֖ים יַעֲבֹ֑רוּ וְאֵלַ֤יִךְ יִֽשְׁתַּחֲוּוּ֙ אֵלַ֣יִךְ יִתְפַּלָּ֔לוּ אַ֣ךְ בָּ֥ךְ אֵ֛ל וְאֵ֥ין ע֖וֹד אֶ֥פֶס אֱלֹהִֽים׃ (לח) אָכֵ֕ן אַתָּ֖ה אֵ֣ל מִסְתַּתֵּ֑ר אֱלֹהֵ֥י יִשְׂרָאֵ֖ל מוֹשִֽׁיעַ׃ (לט) בּ֥וֹשׁוּ וְגַֽם־ נִכְלְמ֖וּ כֻּלָּ֑ם יַחְדָּו֙ הָלְכ֣וּ בַכְּלִמָּ֔ה חָרָשֵׁ֖י צִירִֽים׃
