ישעיהו פרק סא לפי החלוקה לסדרים

פרק סא בספר ישעיהו (11 פסוקים) נמצא בישעיהו סדר כד (פסוקים מה–נב) ובישעיהו סדר כה (פסוקים א–ג).

ישעיהו סדר כד

(א) אָ֗ז תִּתְעַנַּג֙ עַל־ יְהוָ֔ה וְהִרְכַּבְתִּ֖יךָ עַל־ בָּ֣מֳותֵי אָ֑רֶץ וְהַאֲכַלְתִּ֗יךָ נַחֲלַת֙ יַעֲקֹ֣ב אָבִ֔יךָ כִּ֛י פִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (ב) הֵ֛ן לֹֽא־ קָצְרָ֥ה יַד־ יְהוָ֖ה מֵֽהוֹשִׁ֑יעַ וְלֹא־ כָבְדָ֥ה אָזְנ֖וֹ מִשְּׁמֽוֹעַ׃ (ג) כִּ֤י אִם־ עֲוֺנֹֽתֵיכֶם֙ הָי֣וּ מַבְדִּלִ֔ים בֵּינֵכֶ֕ם לְבֵ֖ין אֱלֹֽהֵיכֶ֑ם וְחַטֹּֽאותֵיכֶ֗ם הִסְתִּ֧ירוּ פָנִ֛ים מִכֶּ֖ם מִשְּׁמֽוֹעַ׃ (ד) כִּ֤י כַפֵּיכֶם֙ נְגֹאֲל֣וּ בַדָּ֔ם וְאֶצְבְּעוֹתֵיכֶ֖ם בֶּֽעָוֺ֑ן שִׂפְתֽוֹתֵיכֶם֙ דִּבְּרוּ־ שֶׁ֔קֶר לְשׁוֹנְכֶ֖ם עַוְלָ֥ה תֶהְגֶּֽה׃ (ה) אֵין־ קֹרֵ֣א בְצֶ֔דֶק וְאֵ֥ין נִשְׁפָּ֖ט בֶּאֱמוּנָ֑ה בָּט֤וֹחַ עַל־ תֹּ֙הוּ֙ וְדַבֶּר־ שָׁ֔וְא הָר֥וֹ עָמָ֖ל וְהוֹלֵ֥יד אָֽוֶן׃ (ו) בֵּיצֵ֤י צִפְעוֹנִי֙ בִּקֵּ֔עוּ וְקוּרֵ֥י עַכָּבִ֖ישׁ יֶאֱרֹ֑גוּ הָאֹכֵ֤ל מִבֵּֽיצֵיהֶם֙ יָמ֔וּת וְהַזּוּרֶ֖ה תִּבָּקַ֥ע אֶפְעֶֽה׃ (ז) קֽוּרֵיהֶם֙ לֹא־ יִהְי֣וּ לְבֶ֔גֶד וְלֹ֥א יִתְכַּסּ֖וּ בְּמַֽעֲשֵׂיהֶ֑ם מַֽעֲשֵׂיהֶם֙ מַֽעֲשֵׂי־ אָ֔וֶן וּפֹ֥עַל חָמָ֖ס בְּכַפֵּיהֶֽם׃ (ח) רַגְלֵיהֶם֙ לָרַ֣ע יָרֻ֔צוּ וִֽימַהֲר֔וּ לִשְׁפֹּ֖ךְ דָּ֣ם נָקִ֑י מַחְשְׁבֽוֹתֵיהֶם֙ מַחְשְׁב֣וֹת אָ֔וֶן שֹׁ֥ד וָשֶׁ֖בֶר בִּמְסִלּוֹתָֽם׃ (ט) דֶּ֤רֶךְ שָׁלוֹם֙ לֹ֣א יָדָ֔עוּ וְאֵ֥ין מִשְׁפָּ֖ט בְּמַעְגְּלוֹתָ֑ם נְתִיבֽוֹתֵיהֶם֙ עִקְּשׁ֣וּ לָהֶ֔ם כֹּ֚ל דֹּרֵ֣ךְ בָּ֔הּ לֹ֥א יָדַ֖ע שָׁלֽוֹם׃ (י) עַל־ כֵּ֗ן רָחַ֤ק מִשְׁפָּט֙ מִמֶּ֔נּוּ וְלֹ֥א תַשִּׂיגֵ֖נוּ צְדָקָ֑ה נְקַוֶּ֤ה לָאוֹר֙ וְהִנֵּה־ חֹ֔שֶׁךְ לִנְגֹה֖וֹת בָּאֲפֵל֥וֹת נְהַלֵּֽךְ׃ (יא) נְגַֽשְׁשָׁ֤ה כַֽעִוְרִים֙ קִ֔יר וּכְאֵ֥ין עֵינַ֖יִם נְגַשֵּׁ֑שָׁה כָּשַׁ֤לְנוּ בַֽצָּהֳרַ֙יִם֙ כַּנֶּ֔שֶׁף בָּאַשְׁמַנִּ֖ים כַּמֵּתִֽים׃ (יב) נֶהֱמֶ֤ה כַדֻּבִּים֙ כֻּלָּ֔נוּ וְכַיּוֹנִ֖ים הָגֹ֣ה נֶהְגֶּ֑ה נְקַוֶּ֤ה לַמִּשְׁפָּט֙ וָאַ֔יִן לִֽישׁוּעָ֖ה רָחֲקָ֥ה מִמֶּֽנּוּ׃ (יג) כִּֽי־ רַבּ֤וּ פְשָׁעֵ֙ינוּ֙ נֶגְדֶּ֔ךָ וְחַטֹּאותֵ֖ינוּ עָ֣נְתָה בָּ֑נוּ כִּֽי־ פְשָׁעֵ֣ינוּ אִתָּ֔נוּ וַעֲוֺנֹתֵ֖ינוּ יְדַֽעֲנֽוּם׃ (יד) פָּשֹׁ֤עַ וְכַחֵשׁ֙ בַּֽיהוָ֔ה וְנָס֖וֹג מֵאַחַ֣ר אֱלֹהֵ֑ינוּ דַּבֶּר־ עֹ֣שֶׁק וְסָרָ֔ה הֹר֧וֹ וְהֹג֛וֹ מִלֵּ֖ב דִּבְרֵי־ שָֽׁקֶר׃ (טו) וְהֻסַּ֤ג אָחוֹר֙ מִשְׁפָּ֔ט וּצְדָקָ֖ה מֵרָח֣וֹק תַּעֲמֹ֑ד כִּֽי־ כָשְׁלָ֤ה בָֽרְחוֹב֙ אֱמֶ֔ת וּנְכֹחָ֖ה לֹא־ תוּכַ֥ל לָבֽוֹא׃ (טז) וַתְּהִ֤י הָֽאֱמֶת֙ נֶעְדֶּ֔רֶת וְסָ֥ר מֵרָ֖ע מִשְׁתּוֹלֵ֑ל וַיַּ֧רְא יְהוָ֛ה וַיֵּ֥רַע בְּעֵינָ֖יו כִּֽי־ אֵ֥ין מִשְׁפָּֽט׃ (יז) וַיַּרְא֙ כִּֽי־ אֵ֣ין אִ֔ישׁ וַיִּשְׁתּוֹמֵ֖ם כִּ֣י אֵ֣ין מַפְגִּ֑יעַ וַתּ֤וֹשַֽׁע לוֹ֙ זְרֹע֔וֹ וְצִדְקָת֖וֹ הִ֥יא סְמָכָֽתְהוּ׃ (יח) וַיִּלְבַּ֤שׁ צְדָקָה֙ כַּשִּׁרְיָ֔ן וְכ֥וֹבַע יְשׁוּעָ֖ה בְּרֹאשׁ֑וֹ וַיִּלְבַּ֞שׁ בִּגְדֵ֤י נָקָם֙ תִּלְבֹּ֔שֶׁת וַיַּ֥עַט כַּמְעִ֖יל קִנְאָֽה׃ (יט) כְּעַ֤ל גְּמֻלוֹת֙ כְּעַ֣ל יְשַׁלֵּ֔ם חֵמָ֣ה לְצָרָ֔יו גְּמ֖וּל לְאֹֽיְבָ֑יו לָאִיִּ֖ים גְּמ֥וּל יְשַׁלֵּֽם׃ (כ) וְיִֽירְא֤וּ מִֽמַּעֲרָב֙ אֶת־ שֵׁ֣ם יְהוָ֔ה וּמִמִּזְרַח־ שֶׁ֖מֶשׁ אֶת־ כְּבוֹד֑וֹ כִּֽי־ יָב֤וֹא כַנָּהָר֙ צָ֔ר ר֥וּחַ יְהוָ֖ה נֹ֥סְסָה בֽוֹ׃ (כא) וּבָ֤א לְצִיּוֹן֙ גּוֹאֵ֔ל וּלְשָׁבֵ֥י פֶ֖שַׁע בְּיַֽעֲקֹ֑ב נְאֻ֖ם יְהוָֽה׃ (כב) וַאֲנִ֗י זֹ֣את בְּרִיתִ֤י אוֹתָם֙ אָמַ֣ר יְהוָ֔ה רוּחִי֙ אֲשֶׁ֣ר עָלֶ֔יךָ וּדְבָרַ֖י אֲשֶׁר־ שַׂ֣מְתִּי בְּפִ֑יךָ לֹֽא־ יָמ֡וּשׁוּ מִפִּיךָ֩ וּמִפִּ֨י זַרְעֲךָ֜ וּמִפִּ֨י זֶ֤רַע זַרְעֲךָ֙ אָמַ֣ר יְהוָ֔ה מֵעַתָּ֖ה וְעַד־ עוֹלָֽם׃ (כג) ק֥וּמִי א֖וֹרִי כִּ֣י בָ֣א אוֹרֵ֑ךְ וּכְב֥וֹד יְהוָ֖ה עָלַ֥יִךְ זָרָֽח׃ (כד) כִּֽי־ הִנֵּ֤ה הַחֹ֙שֶׁךְ֙ יְכַסֶּה־ אֶ֔רֶץ וַעֲרָפֶ֖ל לְאֻמִּ֑ים וְעָלַ֙יִךְ֙ יִזְרַ֣ח יְהוָ֔ה וּכְבוֹד֖וֹ עָלַ֥יִךְ יֵרָאֶֽה׃ (כה) וְהָלְכ֥וּ גוֹיִ֖ם לְאוֹרֵ֑ךְ וּמְלָכִ֖ים לְנֹ֥גַהּ זַרְחֵֽךְ׃ (כו) שְׂאִֽי־ סָבִ֤יב עֵינַ֙יִךְ֙ וּרְאִ֔י כֻּלָּ֖ם נִקְבְּצ֣וּ בָֽאוּ־ לָ֑ךְ בָּנַ֙יִךְ֙ מֵרָח֣וֹק יָבֹ֔אוּ וּבְנֹתַ֖יִךְ עַל־ צַ֥ד תֵּאָמַֽנָה׃ (כז) אָ֤ז תִּרְאִי֙ וְנָהַ֔רְתְּ וּפָחַ֥ד וְרָחַ֖ב לְבָבֵ֑ךְ כִּֽי־ יֵהָפֵ֤ךְ עָלַ֙יִךְ֙ הֲמ֣וֹן יָ֔ם חֵ֥יל גּוֹיִ֖ם יָבֹ֥אוּ לָֽךְ׃ (כח) שִֽׁפְעַ֨ת גְּמַלִּ֜ים תְּכַסֵּ֗ךְ בִּכְרֵ֤י מִדְיָן֙ וְעֵיפָ֔ה כֻּלָּ֖ם מִשְּׁבָ֣א יָבֹ֑אוּ זָהָ֤ב וּלְבוֹנָה֙ יִשָּׂ֔אוּ וּתְהִלֹּ֥ת יְהוָ֖ה יְבַשֵּֽׂרוּ׃ (כט) כָּל־ צֹ֤אן קֵדָר֙ יִקָּ֣בְצוּ לָ֔ךְ אֵילֵ֥י נְבָי֖וֹת יְשָׁרְת֑וּנֶךְ יַעֲל֤וּ עַל־ רָצוֹן֙ מִזְבְּחִ֔י וּבֵ֥ית תִּפְאַרְתִּ֖י אֲפָאֵֽר׃ (ל) מִי־ אֵ֖לֶּה כָּעָ֣ב תְּעוּפֶ֑ינָה וְכַיּוֹנִ֖ים אֶל־ אֲרֻבֹּתֵיהֶֽם׃ (לא) כִּֽי־ לִ֣י ׀ אִיִּ֣ים יְקַוּ֗וּ וָאֳנִיּ֤וֹת תַּרְשִׁישׁ֙ בָּרִ֣אשֹׁנָ֔ה לְהָבִ֤יא בָנַ֙יִךְ֙ מֵֽרָח֔וֹק כַּסְפָּ֥ם וּזְהָבָ֖ם אִתָּ֑ם לְשֵׁם֙ יְהוָ֣ה אֱלֹהַ֔יִךְ וְלִקְד֥וֹשׁ יִשְׂרָאֵ֖ל כִּ֥י פֵאֲרָֽךְ׃ (לב) וּבָנ֤וּ בְנֵֽי־ נֵכָר֙ חֹמֹתַ֔יִךְ וּמַלְכֵיהֶ֖ם יְשָׁרְת֑וּנֶךְ כִּ֤י בְקִצְפִּי֙ הִכִּיתִ֔יךְ וּבִרְצוֹנִ֖י רִֽחַמְתִּֽיךְ׃ (לג) וּפִתְּח֨וּ שְׁעָרַ֧יִךְ תָּמִ֛יד יוֹמָ֥ם וָלַ֖יְלָה לֹ֣א יִסָּגֵ֑רוּ לְהָבִ֤יא אֵלַ֙יִךְ֙ חֵ֣יל גּוֹיִ֔ם וּמַלְכֵיהֶ֖ם נְהוּגִֽים׃ (לד) כִּֽי־ הַגּ֧וֹי וְהַמַּמְלָכָ֛ה אֲשֶׁ֥ר לֹא־ יַעַבְד֖וּךְ יֹאבֵ֑דוּ וְהַגּוֹיִ֖ם חָרֹ֥ב יֶחֱרָֽבוּ׃ (לה) כְּב֤וֹד הַלְּבָנוֹן֙ אֵלַ֣יִךְ יָב֔וֹא בְּר֛וֹשׁ תִּדְהָ֥ר וּתְאַשּׁ֖וּר יַחְדָּ֑ו לְפָאֵר֙ מְק֣וֹם מִקְדָּשִׁ֔י וּמְק֥וֹם רַגְלַ֖י אֲכַבֵּֽד׃ (לו) וְהָלְכ֨וּ אֵלַ֤יִךְ שְׁח֙וֹחַ֙ בְּנֵ֣י מְעַנַּ֔יִךְ וְהִֽשְׁתַּחֲו֛וּ עַל־ כַּפּ֥וֹת רַגְלַ֖יִךְ כָּל־ מְנַֽאֲצָ֑יִךְ וְקָ֤רְאוּ לָךְ֙ עִ֣יר יְהוָ֔ה צִיּ֖וֹן קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (לז) תַּ֧חַת הֱיוֹתֵ֛ךְ עֲזוּבָ֥ה וּשְׂנוּאָ֖ה וְאֵ֣ין עוֹבֵ֑ר וְשַׂמְתִּיךְ֙ לִגְא֣וֹן עוֹלָ֔ם מְשׂ֖וֹשׂ דּ֥וֹר וָדֽוֹר׃ (לח) וְיָנַקְתְּ֙ חֲלֵ֣ב גּוֹיִ֔ם וְשֹׁ֥ד מְלָכִ֖ים תִּינָ֑קִי וְיָדַ֗עַתְּ כִּ֣י אֲנִ֤י יְהוָה֙ מֽוֹשִׁיעֵ֔ךְ וְגֹאֲלֵ֖ךְ אֲבִ֥יר יַעֲקֹֽב׃ (לט) תַּ֣חַת הַנְּחֹ֜שֶׁת אָבִ֣יא זָהָ֗ב וְתַ֤חַת הַבַּרְזֶל֙ אָ֣בִיא כֶ֔סֶף וְתַ֤חַת הָֽעֵצִים֙ נְחֹ֔שֶׁת וְתַ֥חַת הָאֲבָנִ֖ים בַּרְזֶ֑ל וְשַׂמְתִּ֤י פְקֻדָּתֵךְ֙ שָׁל֔וֹם וְנֹגְשַׂ֖יִךְ צְדָקָֽה׃ (מ) לֹא־ יִשָּׁמַ֨ע ע֤וֹד חָמָס֙ בְּאַרְצֵ֔ךְ שֹׁ֥ד וָשֶׁ֖בֶר בִּגְבוּלָ֑יִךְ וְקָרָ֤את יְשׁוּעָה֙ חוֹמֹתַ֔יִךְ וּשְׁעָרַ֖יִךְ (כד) בְּֽכָל־ צָרָתָ֣ם ׀ (לא) ל֣וֹ צָ֗ר וּמַלְאַ֤ךְ פָּנָיו֙ הֽוֹשִׁיעָ֔ם בְּאַהֲבָת֥וֹ וּבְחֶמְלָת֖וֹ ה֣וּא גְאָלָ֑ם וַֽיְנַטְּלֵ֥ם וַֽיְנַשְּׂאֵ֖ם כָּל־ יְמֵ֥י עוֹלָֽם׃ (כה) וְהֵ֛מָּה מָר֥וּ וְעִצְּב֖וּ אֶת־ ר֣וּחַ קָדְשׁ֑וֹ וַיֵּהָפֵ֥ךְ לָהֶ֛ם לְאוֹיֵ֖ב ה֥וּא נִלְחַם־ בָּֽם׃ (כו) וַיִּזְכֹּ֥ר יְמֵֽי־ עוֹלָ֖ם מֹשֶׁ֣ה עַמּ֑וֹ אַיֵּ֣ה ׀ הַֽמַּעֲלֵ֣ם מִיָּ֗ם אֵ֚ת רֹעֵ֣י צֹאנ֔וֹ אַיֵּ֛ה הַשָּׂ֥ם בְּקִרְבּ֖וֹ אֶת־ ר֥וּחַ קָדְשֽׁוֹ׃ (כז) מוֹלִיךְ֙ לִימִ֣ין מֹשֶׁ֔ה זְר֖וֹעַ תִּפְאַרְתּ֑וֹ בּ֤וֹקֵֽעַ מַ֙יִם֙ מִפְּנֵיהֶ֔ם לַעֲשׂ֥וֹת ל֖וֹ שֵׁ֥ם עוֹלָֽם׃ (כח) מוֹלִיכָ֖ם בַּתְּהֹמ֑וֹת כַּסּ֥וּס בַּמִּדְבָּ֖ר לֹ֥א יִכָּשֵֽׁלוּ׃ (כט) כַּבְּהֵמָה֙ בַּבִּקְעָ֣ה תֵרֵ֔ד ר֥וּחַ יְהוָ֖ה תְּנִיחֶ֑נּוּ כֵּ֚ן נִהַ֣גְתָּ עַמְּךָ֔ לַעֲשׂ֥וֹת לְךָ֖ שֵׁ֥ם תִּפְאָֽרֶת׃ (ל) הַבֵּ֤ט מִשָּׁמַ֙יִם֙ וּרְאֵ֔ה מִזְּבֻ֥ל קָדְשְׁךָ֖ וְתִפְאַרְתֶּ֑ךָ אַיֵּ֤ה קִנְאָֽתְךָ֙ וּגְב֣וּרֹתֶ֔ךָ הֲמ֥וֹן מֵעֶ֛יךָ וְֽרַחֲמֶ֖יךָ אֵלַ֥י הִתְאַפָּֽקוּ׃ (לא) כִּֽי־ אַתָּ֣ה אָבִ֔ינוּ כִּ֤י אַבְרָהָם֙ לֹ֣א יְדָעָ֔נוּ וְיִשְׂרָאֵ֖ל לֹ֣א יַכִּירָ֑נוּ אַתָּ֤ה יְהוָה֙ אָבִ֔ינוּ גֹּאֲלֵ֥נוּ מֵֽעוֹלָ֖ם שְׁמֶֽךָ׃ (לב) לָ֣מָּה תַתְעֵ֤נוּ יְהוָה֙ מִדְּרָכֶ֔יךָ תַּקְשִׁ֥יחַ לִבֵּ֖נוּ מִיִּרְאָתֶ֑ךָ שׁ֚וּב לְמַ֣עַן עֲבָדֶ֔יךָ שִׁבְטֵ֖י נַחֲלָתֶֽךָ׃ (לג) לַמִּצְעָ֕ר יָרְשׁ֖וּ עַם־ קָדְשֶׁ֑ךָ צָרֵ֕ינוּ בּוֹסְס֖וּ מִקְדָּשֶֽׁךָ׃ (לד) הָיִ֗ינוּ מֵֽעוֹלָם֙ לֹֽא־ מָשַׁ֣לְתָּ בָּ֔ם לֹֽא־ נִקְרָ֥א שִׁמְךָ֖ עֲלֵיהֶ֑ם לוּא־ קָרַ֤עְתָּ שָׁמַ֙יִם֙ יָרַ֔דְתָּ מִפָּנֶ֖יךָ הָרִ֥ים נָזֹֽלּוּ׃ (לה) כִּקְדֹ֧חַ אֵ֣שׁ הֲמָסִ֗ים מַ֚יִם תִּבְעֶה־ אֵ֔שׁ לְהוֹדִ֥יעַ שִׁמְךָ֖ לְצָרֶ֑יךָ מִפָּנֶ֖יךָ גּוֹיִ֥ם יִרְגָּֽזוּ׃ (לו) בַּעֲשׂוֹתְךָ֥ נוֹרָא֖וֹת לֹ֣א נְקַוֶּ֑ה יָרַ֕דְתָּ מִפָּנֶ֖יךָ הָרִ֥ים נָזֹֽלּוּ׃ (לז) וּמֵעוֹלָ֥ם לֹא־ שָׁמְע֖וּ לֹ֣א הֶאֱזִ֑ינוּ עַ֣יִן לֹֽא־ רָאָ֗תָה אֱלֹהִים֙ זוּלָ֣תְךָ֔ יַעֲשֶׂ֖ה לִמְחַכֵּה־ לֽוֹ׃ (לח) פָּגַ֤עְתָּ אֶת־ שָׂשׂ֙ וְעֹ֣שֵׂה צֶ֔דֶק בִּדְרָכֶ֖יךָ יִזְכְּר֑וּךָ הֵן־ אַתָּ֤ה קָצַ֙פְתָּ֙ וַֽנֶּחֱטָ֔א בָּהֶ֥ם עוֹלָ֖ם וְנִוָּשֵֽׁעַ׃ (לט) וַנְּהִ֤י כַטָּמֵא֙ כֻּלָּ֔נוּ וּכְבֶ֥גֶד עִדִּ֖ים כָּל־ צִדְקֹתֵ֑ינוּ וַנָּ֤בֶל כֶּֽעָלֶה֙ כֻּלָּ֔נוּ וַעֲוֺנֵ֖נוּ כָּר֥וּחַ יִשָּׂאֻֽנוּ׃ (מ) וְאֵין־ קוֹרֵ֣א בְשִׁמְךָ֔ מִתְעוֹרֵ֖ר לְהַחֲזִ֣יק בָּ֑ךְ כִּֽי־ הִסְתַּ֤רְתָּ פָנֶ֙יךָ֙ מִמֶּ֔נּוּ וַתְּמוּגֵ֖נוּ בְּיַד־ עֲוֺנֵֽנוּ׃ (מא) וְעַתָּ֥ה יְהוָ֖ה אָבִ֣ינוּ אָ֑תָּה אֲנַ֤חְנוּ הַחֹ֙מֶר֙ וְאַתָּ֣ה יֹצְרֵ֔נוּ וּמַעֲשֵׂ֥ה יָדְךָ֖ כֻּלָּֽנוּ׃ (מב) אַל־ תִּקְצֹ֤ף יְהוָה֙ עַד־ מְאֹ֔ד וְאַל־ לָעַ֖ד תִּזְכֹּ֣ר עָוֺ֑ן הֵ֥ן הַבֶּט־ נָ֖א עַמְּךָ֥ כֻלָּֽנוּ׃ (מג) עָרֵ֥י קָדְשְׁךָ֖ הָי֣וּ מִדְבָּ֑ר צִיּוֹן֙ מִדְבָּ֣ר הָיָ֔תָה יְרוּשָׁלִַ֖ם שְׁמָמָֽה׃ (מד) בֵּ֧ית קָדְשֵׁ֣נוּ וְתִפְאַרְתֵּ֗נוּ אֲשֶׁ֤ר הִֽלְל֙וּךָ֙ אֲבֹתֵ֔ינוּ הָיָ֖ה לִשְׂרֵ֣פַת אֵ֑שׁ וְכָל־ מַחֲמַדֵּ֖ינוּ הָיָ֥ה לְחָרְבָּֽה׃ (מה) הַעַל־ אֵ֥לֶּה תִתְאַפַּ֖ק יְהוָ֑ה תֶּחֱשֶׁ֥ה וּתְעַנֵּ֖נוּ עַד־ מְאֹֽד׃ (מו) נִדְרַ֙שְׁתִּי֙ לְל֣וֹא שָׁאָ֔לוּ נִמְצֵ֖אתִי לְלֹ֣א בִקְשֻׁ֑נִי אָמַ֙רְתִּי֙ הִנֵּ֣נִי הִנֵּ֔נִי אֶל־ גּ֖וֹי לֹֽא־ קֹרָ֥א בִשְׁמִֽי׃ (מז) פֵּרַ֧שְׂתִּי יָדַ֛י כָּל־ הַיּ֖וֹם אֶל־ עַ֣ם סוֹרֵ֑ר הַהֹלְכִים֙ הַדֶּ֣רֶךְ לֹא־ ט֔וֹב אַחַ֖ר מַחְשְׁבֹתֵיהֶֽם׃ (מח) הָעָ֗ם הַמַּכְעִיסִ֥ים אוֹתִ֛י עַל־ פָּנַ֖י תָּמִ֑יד זֹֽבְחִים֙ בַּגַּנּ֔וֹת וּֽמְקַטְּרִ֖ים עַל־ הַלְּבֵנִֽים׃ (מט) הַיֹּֽשְׁבִים֙ בַּקְּבָרִ֔ים וּבַנְּצוּרִ֖ים יָלִ֑ינוּ הָאֹֽכְלִים֙ בְּשַׂ֣ר הַחֲזִ֔יר (ופרק) וּמְרַ֥ק פִּגֻּלִ֖ים כְּלֵיהֶֽם׃ (נ) הָאֹֽמְרִים֙ קְרַ֣ב אֵלֶ֔יךָ אַל־ תִּגַּשׁ־ בִּ֖י כִּ֣י קְדַשְׁתִּ֑יךָ אֵ֚לֶּה עָשָׁ֣ן בְּאַפִּ֔י אֵ֥שׁ יֹקֶ֖דֶת כָּל־ הַיּֽוֹם׃ (נא) הִנֵּ֥ה כְתוּבָ֖ה לְפָנָ֑י לֹ֤א אֶחֱשֶׂה֙ כִּ֣י אִם־ שִׁלַּ֔מְתִּי וְשִׁלַּמְתִּ֖י עַל־ חֵיקָֽם׃ (נב) עֲ֠וֺנֹתֵיכֶם וַעֲוֺנֹ֨ת אֲבוֹתֵיכֶ֤ם יַחְדָּו֙ אָמַ֣ר יְהוָ֔ה אֲשֶׁ֤ר קִטְּרוּ֙ עַל־ הֶ֣הָרִ֔ים וְעַל־ הַגְּבָע֖וֹת חֵרְפ֑וּנִי וּמַדֹּתִ֧י פְעֻלָּתָ֛ם רִֽאשֹׁנָ֖ה (על)־ אֶל־ חֵיקָֽם׃ (נג) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה כַּאֲשֶׁ֨ר יִמָּצֵ֤א הַתִּירוֹשׁ֙ בָּֽאֶשְׁכּ֔וֹל וְאָמַר֙ אַל־ תַּשְׁחִיתֵ֔הוּ כִּ֥י בְרָכָ֖ה בּ֑וֹ כֵּ֤ן אֶֽעֱשֶׂה֙ לְמַ֣עַן עֲבָדַ֔י לְבִלְתִּ֖י הַֽשְׁחִ֥ית הַכֹּֽל׃
