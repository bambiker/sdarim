ישעיהו פרק נב לפי החלוקה לסדרים

פרק נב בספר ישעיהו (15 פסוקים) נמצא בישעיהו סדר כא (פסוקים לו–מא) ובישעיהו סדר כב (פסוקים א–ט).

ישעיהו סדר כא

(א) וְהַאֲכַלְתִּ֤י אֶת־ מוֹנַ֙יִךְ֙ אֶת־ בְּשָׂרָ֔ם וְכֶעָסִ֖יס דָּמָ֣ם יִשְׁכָּר֑וּן וְיָדְע֣וּ כָל־ בָּשָׂ֗ר כִּ֣י אֲנִ֤י יְהוָה֙ מֽוֹשִׁיעֵ֔ךְ וְגֹאֲלֵ֖ךְ אֲבִ֥יר יַעֲקֹֽב׃ (ב) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אֵ֣י זֶ֠ה סֵ֣פֶר כְּרִית֤וּת אִמְּכֶם֙ אֲשֶׁ֣ר שִׁלַּחְתִּ֔יהָ א֚וֹ מִ֣י מִנּוֹשַׁ֔י אֲשֶׁר־ מָכַ֥רְתִּי אֶתְכֶ֖ם ל֑וֹ הֵ֤ן בַּעֲוֺנֹֽתֵיכֶם֙ נִמְכַּרְתֶּ֔ם וּבְפִשְׁעֵיכֶ֖ם שֻׁלְּחָ֥ה אִמְּכֶֽם׃ (ג) מַדּ֨וּעַ בָּ֜אתִי וְאֵ֣ין אִ֗ישׁ קָרָֽאתִי֮ וְאֵ֣ין עוֹנֶה֒ הֲקָצ֨וֹר קָצְרָ֤ה יָדִי֙ מִפְּד֔וּת וְאִם־ אֵֽין־ בִּ֥י כֹ֖חַ לְהַצִּ֑יל הֵ֣ן בְּגַעֲרָתִ֞י אַחֲרִ֣יב יָ֗ם אָשִׂ֤ים נְהָרוֹת֙ מִדְבָּ֔ר תִּבְאַ֤שׁ דְּגָתָם֙ מֵאֵ֣ין מַ֔יִם וְתָמֹ֖ת בַּצָּמָֽא׃ (ד) אַלְבִּ֥ישׁ שָׁמַ֖יִם קַדְר֑וּת וְשַׂ֖ק אָשִׂ֥ים כְּסוּתָֽם׃ (ה) אֲדֹנָ֣י יְהֹוִ֗ה נָ֤תַן לִי֙ לְשׁ֣וֹן לִמּוּדִ֔ים לָדַ֛עַת לָע֥וּת אֶת־ יָעֵ֖ף דָּבָ֑ר יָעִ֣יר ׀ בַּבֹּ֣קֶר בַּבֹּ֗קֶר יָעִ֥יר לִי֙ אֹ֔זֶן לִשְׁמֹ֖עַ כַּלִּמּוּדִֽים׃ (ו) אֲדֹנָ֤י יְהוִה֙ פָּתַֽח־ לִ֣י אֹ֔זֶן וְאָנֹכִ֖י לֹ֣א מָרִ֑יתִי אָח֖וֹר לֹ֥א נְסוּגֹֽתִי׃ (ז) גֵּוִי֙ נָתַ֣תִּי לְמַכִּ֔ים וּלְחָיַ֖י לְמֹֽרְטִ֑ים פָּנַי֙ לֹ֣א הִסְתַּ֔רְתִּי מִכְּלִמּ֖וֹת וָרֹֽק׃ (ח) וַאדֹנָ֤י יְהוִה֙ יַֽעֲזָר־ לִ֔י עַל־ כֵּ֖ן לֹ֣א נִכְלָ֑מְתִּי עַל־ כֵּ֞ן שַׂ֤מְתִּי פָנַי֙ כַּֽחַלָּמִ֔ישׁ וָאֵדַ֖ע כִּי־ לֹ֥א אֵבֽוֹשׁ׃ (ט) קָרוֹב֙ מַצְדִּיקִ֔י מִֽי־ יָרִ֥יב אִתִּ֖י נַ֣עַמְדָה יָּ֑חַד מִֽי־ בַ֥עַל מִשְׁפָּטִ֖י יִגַּ֥שׁ אֵלָֽי׃ (י) הֵ֣ן אֲדֹנָ֤י יְהוִה֙ יַֽעֲזָר־ לִ֔י מִי־ ה֖וּא יַרְשִׁיעֵ֑נִי הֵ֤ן כֻּלָּם֙ כַּבֶּ֣גֶד יִבְל֔וּ עָ֖שׁ יֹאכְלֵֽם׃ (יא) מִ֤י בָכֶם֙ יְרֵ֣א יְהוָ֔ה שֹׁמֵ֖עַ בְּק֣וֹל עַבְדּ֑וֹ אֲשֶׁ֣ר ׀ הָלַ֣ךְ חֲשֵׁכִ֗ים וְאֵ֥ין נֹ֙גַהּ֙ ל֔וֹ יִבְטַח֙ בְּשֵׁ֣ם יְהוָ֔ה וְיִשָּׁעֵ֖ן בֵּאלֹהָֽיו׃ (יב) הֵ֧ן כֻּלְּכֶ֛ם קֹ֥דְחֵי אֵ֖שׁ מְאַזְּרֵ֣י זִיק֑וֹת לְכ֣וּ ׀ בְּא֣וּר אֶשְׁכֶ֗ם וּבְזִיקוֹת֙ בִּֽעַרְתֶּ֔ם מִיָּדִי֙ הָיְתָה־ זֹּ֣את לָכֶ֔ם לְמַעֲצֵבָ֖ה תִּשְׁכָּבֽוּן׃ (יג) שִׁמְע֥וּ אֵלַ֛י רֹ֥דְפֵי צֶ֖דֶק מְבַקְשֵׁ֣י יְהוָ֑ה הַבִּ֙יטוּ֙ אֶל־ צ֣וּר חֻצַּבְתֶּ֔ם וְאֶל־ מַקֶּ֥בֶת בּ֖וֹר נֻקַּרְתֶּֽם׃ (יד) הַבִּ֙יטוּ֙ אֶל־ אַבְרָהָ֣ם אֲבִיכֶ֔ם וְאֶל־ שָׂרָ֖ה תְּחוֹלֶלְכֶ֑ם כִּי־ אֶחָ֣ד קְרָאתִ֔יו וַאֲבָרְכֵ֖הוּ וְאַרְבֵּֽהוּ׃ (טו) כִּֽי־ נִחַ֨ם יְהוָ֜ה צִיּ֗וֹן נִחַם֙ כָּל־ חָרְבֹתֶ֔יהָ וַיָּ֤שֶׂם מִדְבָּרָהּ֙ כְּעֵ֔דֶן וְעַרְבָתָ֖הּ כְּגַן־ יְהוָ֑ה שָׂשׂ֤וֹן וְשִׂמְחָה֙ יִמָּ֣צֵא בָ֔הּ תּוֹדָ֖ה וְק֥וֹל זִמְרָֽה׃ (טז) הַקְשִׁ֤יבוּ אֵלַי֙ עַמִּ֔י וּלְאוּמִּ֖י אֵלַ֣י הַאֲזִ֑ינוּ כִּ֤י תוֹרָה֙ מֵאִתִּ֣י תֵצֵ֔א וּמִשְׁפָּטִ֔י לְא֥וֹר עַמִּ֖ים אַרְגִּֽיעַ׃ (יז) קָר֤וֹב צִדְקִי֙ יָצָ֣א יִשְׁעִ֔י וּזְרֹעַ֖י עַמִּ֣ים יִשְׁפֹּ֑טוּ אֵלַי֙ אִיִּ֣ים יְקַוּ֔וּ וְאֶל־ זְרֹעִ֖י יְיַחֵלֽוּן׃ (יח) שְׂאוּ֩ לַשָּׁמַ֨יִם עֵֽינֵיכֶ֜ם וְֽהַבִּ֧יטוּ אֶל־ הָאָ֣רֶץ מִתַּ֗חַת כִּֽי־ שָׁמַ֜יִם כֶּעָשָׁ֤ן נִמְלָ֙חוּ֙ וְהָאָ֙רֶץ֙ כַּבֶּ֣גֶד תִּבְלֶ֔ה וְיֹשְׁבֶ֖יהָ כְּמוֹ־ כֵ֣ן יְמוּת֑וּן וִישֽׁוּעָתִי֙ לְעוֹלָ֣ם תִּֽהְיֶ֔ה וְצִדְקָתִ֖י לֹ֥א תֵחָֽת׃ (יט) שִׁמְע֤וּ אֵלַי֙ יֹ֣דְעֵי צֶ֔דֶק עַ֖ם תּוֹרָתִ֣י בְלִבָּ֑ם אַל־ תִּֽירְאוּ֙ חֶרְפַּ֣ת אֱנ֔וֹשׁ וּמִגִּדֻּפֹתָ֖ם אַל־ תֵּחָֽתּוּ׃ (כ) כִּ֤י כַבֶּ֙גֶד֙ יֹאכְלֵ֣ם עָ֔שׁ וְכַצֶּ֖מֶר יֹאכְלֵ֣ם סָ֑ס וְצִדְקָתִי֙ לְעוֹלָ֣ם תִּֽהְיֶ֔ה וִישׁוּעָתִ֖י לְד֥וֹר דּוֹרִֽים׃ (כא) עוּרִ֨י עוּרִ֤י לִבְשִׁי־ עֹז֙ זְר֣וֹעַ יְהוָ֔ה ע֚וּרִי כִּ֣ימֵי קֶ֔דֶם דֹּר֖וֹת עוֹלָמִ֑ים הֲל֥וֹא אַתְּ־ הִ֛יא הַמַּחְצֶ֥בֶת רַ֖הַב מְחוֹלֶ֥לֶת תַּנִּֽין׃ (כב) הֲל֤וֹא אַתְּ־ הִיא֙ הַמַּחֲרֶ֣בֶת יָ֔ם מֵ֖י תְּה֣וֹם רַבָּ֑ה הַשָּׂ֙מָה֙ מַֽעֲמַקֵּי־ יָ֔ם דֶּ֖רֶךְ לַעֲבֹ֥ר גְּאוּלִֽים׃ (כג) וּפְדוּיֵ֨י יְהוָ֜ה יְשׁוּב֗וּן וּבָ֤אוּ צִיּוֹן֙ בְּרִנָּ֔ה וְשִׂמְחַ֥ת עוֹלָ֖ם עַל־ רֹאשָׁ֑ם שָׂשׂ֤וֹן וְשִׂמְחָה֙ יַשִּׂיג֔וּן נָ֖סוּ יָג֥וֹן וַאֲנָחָֽה׃ (כד) אָנֹכִ֧י אָנֹכִ֛י ה֖וּא מְנַחֶמְכֶ֑ם מִֽי־ אַ֤תְּ וַתִּֽירְאִי֙ מֵאֱנ֣וֹשׁ יָמ֔וּת וּמִבֶּן־ אָדָ֖ם חָצִ֥יר יִנָּתֵֽן׃ (כה) וַתִּשְׁכַּ֞ח יְהוָ֣ה עֹשֶׂ֗ךָ נוֹטֶ֣ה שָׁמַיִם֮ וְיֹסֵ֣ד אָרֶץ֒ וַתְּפַחֵ֨ד תָּמִ֜יד כָּל־ הַיּ֗וֹם מִפְּנֵי֙ חֲמַ֣ת הַמֵּצִ֔יק כַּאֲשֶׁ֥ר כּוֹנֵ֖ן לְהַשְׁחִ֑ית וְאַיֵּ֖ה חֲמַ֥ת הַמֵּצִֽיק׃ (כו) מִהַ֥ר צֹעֶ֖ה לְהִפָּתֵ֑חַ וְלֹא־ יָמ֣וּת לַשַּׁ֔חַת וְלֹ֥א יֶחְסַ֖ר לַחְמֽוֹ׃ (כז) וְאָֽנֹכִי֙ יְהוָ֣ה אֱלֹהֶ֔יךָ רֹגַ֣ע הַיָּ֔ם וַיֶּהֱמ֖וּ גַּלָּ֑יו יְהוָ֥ה צְבָא֖וֹת שְׁמֽוֹ׃ (כח) וָאָשִׂ֤ים דְּבָרַי֙ בְּפִ֔יךָ וּבְצֵ֥ל יָדִ֖י כִּסִּיתִ֑יךָ לִנְטֹ֤עַ שָׁמַ֙יִם֙ וְלִיסֹ֣ד אָ֔רֶץ וְלֵאמֹ֥ר לְצִיּ֖וֹן עַמִּי־ אָֽתָּה׃ (כט) הִתְעוֹרְרִ֣י הִֽתְעוֹרְרִ֗י ק֚וּמִי יְר֣וּשָׁלִַ֔ם אֲשֶׁ֥ר שָׁתִ֛ית מִיַּ֥ד יְהוָ֖ה אֶת־ כּ֣וֹס חֲמָת֑וֹ אֶת־ קֻבַּ֜עַת כּ֧וֹס הַתַּרְעֵלָ֛ה שָׁתִ֖ית מָצִֽית׃ (ל) אֵין־ מְנַהֵ֣ל לָ֔הּ מִכָּל־ בָּנִ֖ים יָלָ֑דָה וְאֵ֤ין מַחֲזִיק֙ בְּיָדָ֔הּ מִכָּל־ בָּנִ֖ים גִּדֵּֽלָה׃ (לא) שְׁתַּ֤יִם הֵ֙נָּה֙ קֹֽרְאֹתַ֔יִךְ מִ֖י יָנ֣וּד לָ֑ךְ הַשֹּׁ֧ד וְהַשֶּׁ֛בֶר וְהָרָעָ֥ב וְהַחֶ֖רֶב מִ֥י אֲנַחֲמֵֽךְ׃ (לב) בָּנַ֜יִךְ עֻלְּפ֥וּ שָׁכְב֛וּ בְּרֹ֥אשׁ כָּל־ חוּצ֖וֹת כְּת֣וֹא מִכְמָ֑ר הַֽמְלֵאִ֥ים חֲמַת־ יְהוָ֖ה גַּעֲרַ֥ת אֱלֹהָֽיִךְ׃ (לג) לָכֵ֛ן שִׁמְעִי־ נָ֥א זֹ֖את עֲנִיָּ֑ה וּשְׁכֻרַ֖ת וְלֹ֥א מִיָּֽיִן׃ (לד) כֹּֽה־ אָמַ֞ר אֲדֹנַ֣יִךְ יְהוָ֗ה וֵאלֹהַ֙יִךְ֙ יָרִ֣יב עַמּ֔וֹ הִנֵּ֥ה לָקַ֛חְתִּי מִיָּדֵ֖ךְ אֶת־ כּ֣וֹס הַתַּרְעֵלָ֑ה אֶת־ קֻבַּ֙עַת֙ כּ֣וֹס חֲמָתִ֔י לֹא־ תוֹסִ֥יפִי לִשְׁתּוֹתָ֖הּ עֽוֹד׃ (לה) וְשַׂמְתִּ֙יהָ֙ בְּיַד־ מוֹגַ֔יִךְ אֲשֶׁר־ אָמְר֥וּ לְנַפְשֵׁ֖ךְ שְׁחִ֣י וְנַעֲבֹ֑רָה וַתָּשִׂ֤ימִי כָאָ֙רֶץ֙ גֵּוֵ֔ךְ וְכַח֖וּץ לַעֹבְרִֽים׃ תחילת פרק נב (לו) עוּרִ֥י עוּרִ֛י לִבְשִׁ֥י עֻזֵּ֖ךְ צִיּ֑וֹן לִבְשִׁ֣י ׀ בִּגְדֵ֣י תִפְאַרְתֵּ֗ךְ יְרוּשָׁלִַ֙ם֙ עִ֣יר הַקֹּ֔דֶשׁ כִּ֣י לֹ֥א יוֹסִ֛יף יָבֹא־ בָ֥ךְ ע֖וֹד עָרֵ֥ל וְטָמֵֽא׃ (לז) הִתְנַעֲרִ֧י מֵעָפָ֛ר ק֥וּמִי שְּׁבִ֖י יְרֽוּשָׁלִָ֑ם (התפתחו) הִֽתְפַּתְּחִי֙ מוֹסְרֵ֣י צַוָּארֵ֔ךְ שְׁבִיָּ֖ה בַּת־ צִיּֽוֹן׃ (לח) כִּֽי־ כֹה֙ אָמַ֣ר יְהוָ֔ה חִנָּ֖ם נִמְכַּרְתֶּ֑ם וְלֹ֥א בְכֶ֖סֶף תִּגָּאֵֽלוּ׃ (לט) כִּ֣י כֹ֤ה לֹֽא־ תִקְרַ֖ב אֵלָֽיִךְ׃ (לו) הֵ֣ן גּ֥וֹר יָג֛וּר אֶ֖פֶס מֵֽאוֹתִ֑י מִי־ גָ֥ר אִתָּ֖ךְ עָלַ֥יִךְ יִפּֽוֹל׃ (לז) (הן) הִנֵּ֤ה אָֽנֹכִי֙ בָּרָ֣אתִי חָרָ֔שׁ נֹפֵ֙חַ֙ בְּאֵ֣שׁ פֶּחָ֔ם וּמוֹצִ֥יא כְלִ֖י לְמַעֲשֵׂ֑הוּ וְאָנֹכִ֛י בָּרָ֥אתִי מַשְׁחִ֖ית לְחַבֵּֽל׃ (לח) כָּל־ כְּלִ֞י יוּצַ֤ר עָלַ֙יִךְ֙ לֹ֣א יִצְלָ֔ח וְכָל־ לָשׁ֛וֹן תָּֽקוּם־ אִתָּ֥ךְ לַמִּשְׁפָּ֖ט תַּרְשִׁ֑יעִי זֹ֡את נַחֲלַת֩ עַבְדֵ֨י יְהוָ֧ה וְצִדְקָתָ֛ם מֵאִתִּ֖י נְאֻם־ יְהוָֽה׃ (לט) ה֤וֹי כָּל־ צָמֵא֙ לְכ֣וּ לַמַּ֔יִם וַאֲשֶׁ֥ר אֵֽין־ ל֖וֹ כָּ֑סֶף לְכ֤וּ שִׁבְרוּ֙ וֶֽאֱכֹ֔לוּ וּלְכ֣וּ שִׁבְר֗וּ בְּלוֹא־ כֶ֛סֶף וּבְל֥וֹא מְחִ֖יר יַ֥יִן וְחָלָֽב׃ (מ) לָ֤מָּה תִשְׁקְלוּ־ כֶ֙סֶף֙ בְּֽלוֹא־ לֶ֔חֶם וִיגִיעֲכֶ֖ם בְּל֣וֹא לְשָׂבְעָ֑ה שִׁמְע֨וּ שָׁמ֤וֹעַ אֵלַי֙ וְאִכְלוּ־ ט֔וֹב וְתִתְעַנַּ֥ג בַּדֶּ֖שֶׁן נַפְשְׁכֶֽם׃ (מא) הַטּ֤וּ אָזְנְכֶם֙ וּלְכ֣וּ אֵלַ֔י שִׁמְע֖וּ וּתְחִ֣י נַפְשְׁכֶ֑ם וְאֶכְרְתָ֤ה לָכֶם֙ בְּרִ֣ית עוֹלָ֔ם חַֽסְדֵ֥י דָוִ֖ד הַנֶּאֱמָנִֽים׃ (מב) הֵ֛ן עֵ֥ד לְאוּמִּ֖ים נְתַתִּ֑יו נָגִ֥יד וּמְצַוֵּ֖ה לְאֻמִּֽים׃ (מג) הֵ֣ן גּ֤וֹי לֹֽא־ תֵדַע֙ תִּקְרָ֔א וְג֥וֹי לֹֽא־ יְדָע֖וּךָ אֵלֶ֣יךָ יָר֑וּצוּ לְמַ֙עַן֙ יְהוָ֣ה אֱלֹהֶ֔יךָ וְלִקְד֥וֹשׁ יִשְׂרָאֵ֖ל כִּ֥י פֵאֲרָֽךְ׃ (מד) דִּרְשׁ֥וּ יְהוָ֖ה בְּהִמָּצְא֑וֹ קְרָאֻ֖הוּ בִּֽהְיוֹת֥וֹ קָרֽוֹב׃ (מה) יַעֲזֹ֤ב רָשָׁע֙ דַּרְכּ֔וֹ וְאִ֥ישׁ אָ֖וֶן מַחְשְׁבֹתָ֑יו וְיָשֹׁ֤ב אֶל־ יְהוָה֙ וִֽירַחֲמֵ֔הוּ וְאֶל־ אֱלֹהֵ֖ינוּ כִּֽי־ יַרְבֶּ֥ה לִסְלֽוֹחַ׃ (מו) כִּ֣י לֹ֤א מַחְשְׁבוֹתַי֙ מַחְשְׁב֣וֹתֵיכֶ֔ם וְלֹ֥א דַרְכֵיכֶ֖ם דְּרָכָ֑י נְאֻ֖ם יְהוָֽה׃ (מז) כִּֽי־ גָבְה֥וּ שָׁמַ֖יִם מֵאָ֑רֶץ כֵּ֣ן גָּבְה֤וּ דְרָכַי֙ מִדַּרְכֵיכֶ֔ם וּמַחְשְׁבֹתַ֖י מִמַּחְשְׁבֹתֵיכֶֽם׃ (מח) כִּ֡י כַּאֲשֶׁ֣ר יֵרֵד֩ הַגֶּ֨שֶׁם וְהַשֶּׁ֜לֶג מִן־ הַשָּׁמַ֗יִם וְשָׁ֙מָּה֙ לֹ֣א יָשׁ֔וּב כִּ֚י אִם־ הִרְוָ֣ה אֶת־ הָאָ֔רֶץ וְהוֹלִידָ֖הּ וְהִצְמִיחָ֑הּ וְנָ֤תַן זֶ֙רַע֙ לַזֹּרֵ֔עַ וְלֶ֖חֶם לָאֹכֵֽל׃ (מט) כֵּ֣ן יִֽהְיֶ֤ה דְבָרִי֙ אֲשֶׁ֣ר יֵצֵ֣א מִפִּ֔י לֹֽא־ יָשׁ֥וּב אֵלַ֖י רֵיקָ֑ם כִּ֤י אִם־ עָשָׂה֙ אֶת־ אֲשֶׁ֣ר חָפַ֔צְתִּי וְהִצְלִ֖יחַ אֲשֶׁ֥ר שְׁלַחְתִּֽיו׃ (נ) כִּֽי־ בְשִׂמְחָ֣ה תֵצֵ֔אוּ וּבְשָׁל֖וֹם תּֽוּבָל֑וּן הֶהָרִ֣ים וְהַגְּבָע֗וֹת יִפְצְח֤וּ לִפְנֵיכֶם֙ רִנָּ֔ה וְכָל־ עֲצֵ֥י הַשָּׂדֶ֖ה יִמְחֲאוּ־ כָֽף׃
