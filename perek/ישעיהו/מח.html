ישעיהו פרק מח לפי החלוקה לסדרים

פרק מח בספר ישעיהו (22 פסוקים) נמצא בישעיהו סדר יט (פסוקים לח–לח) ובישעיהו סדר כ (פסוקים א–כא).

ישעיהו סדר יט

(א) יִשְׂרָאֵל֙ נוֹשַׁ֣ע בַּיהוָ֔ה תְּשׁוּעַ֖ת עוֹלָמִ֑ים לֹא־ תֵבֹ֥שׁוּ וְלֹא־ תִכָּלְמ֖וּ עַד־ ע֥וֹלְמֵי עַֽד׃ (ב) כִּ֣י כֹ֣ה אָֽמַר־ יְ֠הוָה בּוֹרֵ֨א הַשָּׁמַ֜יִם ה֣וּא הָאֱלֹהִ֗ים יֹצֵ֨ר הָאָ֤רֶץ וְעֹשָׂהּ֙ ה֣וּא כֽוֹנְנָ֔הּ לֹא־ תֹ֥הוּ בְרָאָ֖הּ לָשֶׁ֣בֶת יְצָרָ֑הּ אֲנִ֥י יְהוָ֖ה וְאֵ֥ין עֽוֹד׃ (ג) לֹ֧א בַסֵּ֣תֶר דִּבַּ֗רְתִּי בִּמְקוֹם֙ אֶ֣רֶץ חֹ֔שֶׁךְ לֹ֥א אָמַ֛רְתִּי לְזֶ֥רַע יַעֲקֹ֖ב תֹּ֣הוּ בַקְּשׁ֑וּנִי אֲנִ֤י יְהוָה֙ דֹּבֵ֣ר צֶ֔דֶק מַגִּ֖יד מֵישָׁרִֽים׃ (ד) הִקָּבְצ֥וּ וָבֹ֛אוּ הִֽתְנַגְּשׁ֥וּ יַחְדָּ֖ו פְּלִיטֵ֣י הַגּוֹיִ֑ם לֹ֣א יָדְע֗וּ הַנֹּֽשְׂאִים֙ אֶת־ עֵ֣ץ פִּסְלָ֔ם וּמִתְפַּלְלִ֔ים אֶל־ אֵ֖ל לֹ֥א יוֹשִֽׁיעַ׃ (ה) הַגִּ֣ידוּ וְהַגִּ֔ישׁוּ אַ֥ף יִֽוָּעֲצ֖וּ יַחְדָּ֑ו מִ֣י הִשְׁמִיעַ֩ זֹ֨את מִקֶּ֜דֶם מֵאָ֣ז הִגִּידָ֗הּ הֲל֨וֹא אֲנִ֤י יְהוָה֙ וְאֵֽין־ ע֤וֹד אֱלֹהִים֙ מִבַּלְעָדַ֔י אֵֽל־ צַדִּ֣יק וּמוֹשִׁ֔יעַ אַ֖יִן זוּלָתִֽי׃ (ו) פְּנוּ־ אֵלַ֥י וְהִוָּשְׁע֖וּ כָּל־ אַפְסֵי־ אָ֑רֶץ כִּ֥י אֲנִי־ אֵ֖ל וְאֵ֥ין עֽוֹד׃ (ז) בִּ֣י נִשְׁבַּ֔עְתִּי יָצָ֨א מִפִּ֧י צְדָקָ֛ה דָּבָ֖ר וְלֹ֣א יָשׁ֑וּב כִּי־ לִי֙ תִּכְרַ֣ע כָּל־ בֶּ֔רֶךְ תִּשָּׁבַ֖ע כָּל־ לָשֽׁוֹן׃ (ח) אַ֧ךְ בַּיהוָ֛ה לִ֥י אָמַ֖ר צְדָק֣וֹת וָעֹ֑ז עָדָיו֙ יָב֣וֹא וְיֵבֹ֔שׁוּ כֹּ֖ל הַנֶּחֱרִ֥ים בּֽוֹ׃ (ט) בַּיהוָ֛ה יִצְדְּק֥וּ וְיִֽתְהַלְל֖וּ כָּל־ זֶ֥רַע יִשְׂרָאֵֽל׃ (י) כָּרַ֥ע בֵּל֙ קֹרֵ֣ס נְב֔וֹ הָיוּ֙ עֲצַבֵּיהֶ֔ם לַחַיָּ֖ה וְלַבְּהֵמָ֑ה נְשֻׂאֹתֵיכֶ֣ם עֲמוּס֔וֹת מַשָּׂ֖א לַעֲיֵפָֽה׃ (יא) קָרְס֤וּ כָֽרְעוּ֙ יַחְדָּ֔ו לֹ֥א יָכְל֖וּ מַלֵּ֣ט מַשָּׂ֑א וְנַפְשָׁ֖ם בַּשְּׁבִ֥י הָלָֽכָה׃ (יב) שִׁמְע֤וּ אֵלַי֙ בֵּ֣ית יַעֲקֹ֔ב וְכָל־ שְׁאֵרִ֖ית בֵּ֣ית יִשְׂרָאֵ֑ל הַֽעֲמֻסִים֙ מִנִּי־ בֶ֔טֶן הַנְּשֻׂאִ֖ים מִנִּי־ רָֽחַם׃ (יג) וְעַד־ זִקְנָה֙ אֲנִ֣י ה֔וּא וְעַד־ שֵיבָ֖ה אֲנִ֣י אֶסְבֹּ֑ל אֲנִ֤י עָשִׂ֙יתִי֙ וַאֲנִ֣י אֶשָּׂ֔א וַאֲנִ֥י אֶסְבֹּ֖ל וַאֲמַלֵּֽט׃ (יד) לְמִ֥י תְדַמְי֖וּנִי וְתַשְׁו֑וּ וְתַמְשִׁל֖וּנִי וְנִדְמֶֽה׃ (טו) הַזָּלִ֤ים זָהָב֙ מִכִּ֔יס וְכֶ֖סֶף בַּקָּנֶ֣ה יִשְׁקֹ֑לוּ יִשְׂכְּר֤וּ צוֹרֵף֙ וְיַעֲשֵׂ֣הוּ אֵ֔ל יִסְגְּד֖וּ אַף־ יִֽשְׁתַּחֲוּֽוּ׃ (טז) יִ֠שָּׂאֻהוּ עַל־ כָּתֵ֨ף יִסְבְּלֻ֜הוּ וְיַנִּיחֻ֤הוּ תַחְתָּיו֙ וְיַֽעֲמֹ֔ד מִמְּקוֹמ֖וֹ לֹ֣א יָמִ֑ישׁ אַף־ יִצְעַ֤ק אֵלָיו֙ וְלֹ֣א יַעֲנֶ֔ה מִצָּרָת֖וֹ לֹ֥א יוֹשִׁיעֶֽנּוּ׃ (יז) זִכְרוּ־ זֹ֖את וְהִתְאֹשָׁ֑שׁוּ הָשִׁ֥יבוּ פוֹשְׁעִ֖ים עַל־ לֵֽב׃ (יח) זִכְר֥וּ רִאשֹׁנ֖וֹת מֵעוֹלָ֑ם כִּ֣י אָנֹכִ֥י אֵל֙ וְאֵ֣ין ע֔וֹד אֱלֹהִ֖ים וְאֶ֥פֶס כָּמֽוֹנִי׃ (יט) מַגִּ֤יד מֵֽרֵאשִׁית֙ אַחֲרִ֔ית וּמִקֶּ֖דֶם אֲשֶׁ֣ר לֹא־ נַעֲשׂ֑וּ אֹמֵר֙ עֲצָתִ֣י תָק֔וּם וְכָל־ חֶפְצִ֖י אֶעֱשֶֽׂה׃ (כ) קֹרֵ֤א מִמִּזְרָח֙ עַ֔יִט מֵאֶ֥רֶץ מֶרְחָ֖ק אִ֣ישׁ (עצתו) עֲצָתִ֑י אַף־ דִּבַּ֙רְתִּי֙ אַף־ אֲבִיאֶ֔נָּה יָצַ֖רְתִּי אַף־ אֶעֱשֶֽׂנָּה׃ (כא) שִׁמְע֥וּ אֵלַ֖י אַבִּ֣ירֵי לֵ֑ב הָרְחוֹקִ֖ים מִצְּדָקָֽה׃ (כב) קֵרַ֤בְתִּי צִדְקָתִי֙ לֹ֣א תִרְחָ֔ק וּתְשׁוּעָתִ֖י לֹ֣א תְאַחֵ֑ר וְנָתַתִּ֤י בְצִיּוֹן֙ תְּשׁוּעָ֔ה לְיִשְׂרָאֵ֖ל תִּפְאַרְתִּֽי׃ (כג) רְדִ֣י ׀ וּשְׁבִ֣י עַל־ עָפָ֗ר בְּתוּלַת֙ בַּת־ בָּבֶ֔ל שְׁבִי־ לָאָ֥רֶץ אֵין־ כִּסֵּ֖א בַּת־ כַּשְׂדִּ֑ים כִּ֣י לֹ֤א תוֹסִ֙יפִי֙ יִקְרְאוּ־ לָ֔ךְ רַכָּ֖ה וַעֲנֻגָּֽה׃ (כד) קְחִ֥י רֵחַ֖יִם וְטַ֣חֲנִי קָ֑מַח גַּלִּ֨י צַמָּתֵ֧ךְ חֶשְׂפִּי־ שֹׁ֛בֶל גַּלִּי־ שׁ֖וֹק עִבְרִ֥י נְהָרֽוֹת׃ (כה) תִּגָּל֙ עֶרְוָתֵ֔ךְ גַּ֥ם תֵּרָאֶ֖ה חֶרְפָּתֵ֑ךְ נָקָ֣ם אֶקָּ֔ח וְלֹ֥א אֶפְגַּ֖ע אָדָֽם׃ (כו) גֹּאֲלֵ֕נוּ יְהוָ֥ה צְבָא֖וֹת שְׁמ֑וֹ קְד֖וֹשׁ יִשְׂרָאֵֽל׃ (כז) שְׁבִ֥י דוּמָ֛ם וּבֹ֥אִי בַחֹ֖שֶׁךְ בַּת־ כַּשְׂדִּ֑ים כִּ֣י לֹ֤א תוֹסִ֙יפִי֙ יִקְרְאוּ־ לָ֔ךְ גְּבֶ֖רֶת מַמְלָכֽוֹת׃ (כח) קָצַ֣פְתִּי עַל־ עַמִּ֗י חִלַּ֙לְתִּי֙ נַחֲלָתִ֔י וָאֶתְּנֵ֖ם בְּיָדֵ֑ךְ לֹא־ שַׂ֤מְתְּ לָהֶם֙ רַחֲמִ֔ים עַל־ זָקֵ֕ן הִכְבַּ֥דְתְּ עֻלֵּ֖ךְ מְאֹֽד׃ (כט) וַתֹּ֣אמְרִ֔י לְעוֹלָ֖ם אֶהְיֶ֣ה גְבָ֑רֶת עַ֣ד לֹא־ שַׂ֥מְתְּ אֵ֙לֶּה֙ עַל־ לִבֵּ֔ךְ לֹ֥א זָכַ֖רְתְּ אַחֲרִיתָֽהּ׃ (ל) וְעַתָּ֞ה שִׁמְעִי־ זֹ֤את עֲדִינָה֙ הַיּוֹשֶׁ֣בֶת לָבֶ֔טַח הָאֹֽמְרָה֙ בִּלְבָ֔בָהּ אֲנִ֖י וְאַפְסִ֣י ע֑וֹד לֹ֤א אֵשֵׁב֙ אַלְמָנָ֔ה וְלֹ֥א אֵדַ֖ע שְׁכֽוֹל׃ (לא) וְתָבֹאנָה֩ לָּ֨ךְ שְׁתֵּי־ אֵ֥לֶּה רֶ֛גַע בְּי֥וֹם אֶחָ֖ד שְׁכ֣וֹל וְאַלְמֹ֑ן כְּתֻמָּם֙ בָּ֣אוּ עָלַ֔יִךְ בְּרֹ֣ב כְּשָׁפַ֔יִךְ בְּעָצְמַ֥ת חֲבָרַ֖יִךְ מְאֹֽד׃ (לב) וַתִּבְטְחִ֣י בְרָעָתֵ֗ךְ אָמַרְתְּ֙ אֵ֣ין רֹאָ֔נִי חָכְמָתֵ֥ךְ וְדַעְתֵּ֖ךְ הִ֣יא שׁוֹבְבָ֑תֶךְ וַתֹּאמְרִ֣י בְלִבֵּ֔ךְ אֲנִ֖י וְאַפְסִ֥י עֽוֹד׃ (לג) וּבָ֧א עָלַ֣יִךְ רָעָ֗ה לֹ֤א תֵדְעִי֙ שַׁחְרָ֔הּ וְתִפֹּ֤ל עָלַ֙יִךְ֙ הֹוָ֔ה לֹ֥א תוּכְלִ֖י כַּפְּרָ֑הּ וְתָבֹ֨א עָלַ֧יִךְ פִּתְאֹ֛ם שׁוֹאָ֖ה לֹ֥א תֵדָֽעִי׃ (לד) עִמְדִי־ נָ֤א בַחֲבָרַ֙יִךְ֙ וּבְרֹ֣ב כְּשָׁפַ֔יִךְ בַּאֲשֶׁ֥ר יָגַ֖עַתְּ מִנְּעוּרָ֑יִךְ אוּלַ֛י תּוּכְלִ֥י הוֹעִ֖יל אוּלַ֥י תַּעֲרֽוֹצִי׃ (לה) נִלְאֵ֖ית בְּרֹ֣ב עֲצָתָ֑יִךְ יַעַמְדוּ־ נָ֨א וְיוֹשִׁיעֻ֜ךְ (הברו) הֹבְרֵ֣י שָׁמַ֗יִם הַֽחֹזִים֙ בַּכּ֣וֹכָבִ֔ים מֽוֹדִיעִם֙ לֶחֳדָשִׁ֔ים מֵאֲשֶׁ֥ר יָבֹ֖אוּ עָלָֽיִךְ׃ (לו) הִנֵּ֨ה הָי֤וּ כְקַשׁ֙ אֵ֣שׁ שְׂרָפָ֔תַם לֹֽא־ יַצִּ֥ילוּ אֶת־ נַפְשָׁ֖ם מִיַּ֣ד לֶֽהָבָ֑ה אֵין־ גַּחֶ֣לֶת לַחְמָ֔ם א֖וּר לָשֶׁ֥בֶת נֶגְדּֽוֹ׃ (לז) כֵּ֥ן הָיוּ־ לָ֖ךְ אֲשֶׁ֣ר יָגָ֑עַתְּ סֹחֲרַ֣יִךְ מִנְּעוּרַ֗יִךְ אִ֤ישׁ לְעֶבְרוֹ֙ תָּע֔וּ אֵ֖ין מוֹשִׁיעֵֽךְ׃ תחילת פרק מח (לח) שִׁמְעוּ־ זֹ֣את בֵּֽית־ יַעֲקֹ֗ב הַנִּקְרָאִים֙ בְּשֵׁ֣ם יִשְׂרָאֵ֔ל וּמִמֵּ֥י יְהוּדָ֖ה יָצָ֑אוּ הַֽנִּשְׁבָּעִ֣ים ׀ בְּשֵׁ֣ם יְהוָ֗ה וּבֵאלֹהֵ֤י יִשְׂרָאֵל֙ יַזְכִּ֔ירוּ לֹ֥א בֶאֱמֶ֖ת וְלֹ֥א בִצְדָקָֽה׃

ישעיהו סדר כ

אַפַּ֗יִם אֶ֚רֶץ יִשְׁתַּ֣חֲווּ לָ֔ךְ וַעֲפַ֥ר רַגְלַ֖יִךְ יְלַחֵ֑כוּ וְיָדַ֙עַתְּ֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה אֲשֶׁ֥ר לֹֽא־ יֵבֹ֖שׁוּ קוָֹֽי׃ (מה) הֲיֻקַּ֥ח מִגִּבּ֖וֹר מַלְק֑וֹחַ וְאִם־ שְׁבִ֥י צַדִּ֖יק יִמָּלֵֽט׃ (מו) כִּי־ כֹ֣ה ׀ אָמַ֣ר יְהוָ֗ה גַּם־ שְׁבִ֤י גִבּוֹר֙ יֻקָּ֔ח וּמַלְק֥וֹחַ עָרִ֖יץ יִמָּלֵ֑ט וְאֶת־ יְרִיבֵךְ֙ אָנֹכִ֣י אָרִ֔יב וְאֶת־ בָּנַ֖יִךְ אָנֹכִ֥י אוֹשִֽׁיעַ׃
