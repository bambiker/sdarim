ישעיהו פרק נח לפי החלוקה לסדרים

פרק נח בספר ישעיהו (14 פסוקים) נמצא בישעיהו סדר כג (פסוקים לה–מז) ובישעיהו סדר כד (פסוקים א–א).

ישעיהו סדר כג

(א) תַּ֤חַת הַֽנַּעֲצוּץ֙ יַעֲלֶ֣ה בְר֔וֹשׁ (תחת) וְתַ֥חַת הַסִּרְפַּ֖ד יַעֲלֶ֣ה הֲדַ֑ס וְהָיָ֤ה לַֽיהוָה֙ לְשֵׁ֔ם לְא֥וֹת עוֹלָ֖ם לֹ֥א יִכָּרֵֽת׃ (ב) כֹּ֚ה אָמַ֣ר יְהוָ֔ה שִׁמְר֥וּ מִשְׁפָּ֖ט וַעֲשׂ֣וּ צְדָקָ֑ה כִּֽי־ קְרוֹבָ֤ה יְשֽׁוּעָתִי֙ לָב֔וֹא וְצִדְקָתִ֖י לְהִגָּלֽוֹת׃ (ג) אַשְׁרֵ֤י אֱנוֹשׁ֙ יַעֲשֶׂה־ זֹּ֔את וּבֶן־ אָדָ֖ם יַחֲזִ֣יק בָּ֑הּ שֹׁמֵ֤ר שַׁבָּת֙ מֵֽחַלְּל֔וֹ וְשֹׁמֵ֥ר יָד֖וֹ מֵעֲשׂ֥וֹת כָּל־ רָֽע׃ (ד) וְאַל־ יֹאמַ֣ר בֶּן־ הַנֵּכָ֗ר הַנִּלְוָ֤ה אֶל־ יְהוָה֙ לֵאמֹ֔ר הַבְדֵּ֧ל יַבְדִּילַ֛נִי יְהוָ֖ה מֵעַ֣ל עַמּ֑וֹ וְאַל־ יֹאמַר֙ הַסָּרִ֔יס הֵ֥ן אֲנִ֖י עֵ֥ץ יָבֵֽשׁ׃ (ה) כִּי־ כֹ֣ה ׀ אָמַ֣ר יְהוָ֗ה לַסָּֽרִיסִים֙ אֲשֶׁ֤ר יִשְׁמְרוּ֙ אֶת־ שַׁבְּתוֹתַ֔י וּבָֽחֲר֖וּ בַּאֲשֶׁ֣ר חָפָ֑צְתִּי וּמַחֲזִיקִ֖ים בִּבְרִיתִֽי׃ (ו) וְנָתַתִּ֨י לָהֶ֜ם בְּבֵיתִ֤י וּבְחֽוֹמֹתַי֙ יָ֣ד וָשֵׁ֔ם ט֖וֹב מִבָּנִ֣ים וּמִבָּנ֑וֹת שֵׁ֤ם עוֹלָם֙ אֶתֶּן־ ל֔וֹ אֲשֶׁ֖ר לֹ֥א יִכָּרֵֽת׃ (ז) וּבְנֵ֣י הַנֵּכָ֗ר הַנִּלְוִ֤ים עַל־ יְהוָה֙ לְשָׁ֣רְת֔וֹ וּֽלְאַהֲבָה֙ אֶת־ שֵׁ֣ם יְהוָ֔ה לִהְי֥וֹת ל֖וֹ לַעֲבָדִ֑ים כָּל־ שֹׁמֵ֤ר שַׁבָּת֙ מֵֽחַלְּל֔וֹ וּמַחֲזִיקִ֖ים בִּבְרִיתִֽי׃ (ח) וַהֲבִיאוֹתִ֞ים אֶל־ הַ֣ר קָדְשִׁ֗י וְשִׂמַּחְתִּים֙ בְּבֵ֣ית תְּפִלָּתִ֔י עוֹלֹתֵיהֶ֧ם וְזִבְחֵיהֶ֛ם לְרָצ֖וֹן עַֽל־ מִזְבְּחִ֑י כִּ֣י בֵיתִ֔י בֵּית־ תְּפִלָּ֥ה יִקָּרֵ֖א לְכָל־ הָעַמִּֽים׃ (ט) נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה מְקַבֵּ֖ץ נִדְחֵ֣י יִשְׂרָאֵ֑ל ע֛וֹד אֲקַבֵּ֥ץ עָלָ֖יו לְנִקְבָּצָֽיו׃ (י) כֹּ֖ל חַיְת֣וֹ שָׂדָ֑י אֵתָ֕יוּ לֶאֱכֹ֥ל כָּל־ חַיְת֖וֹ בַּיָּֽעַר׃ (יא) (צפו) צֹפָ֞יו עִוְרִ֤ים כֻּלָּם֙ לֹ֣א יָדָ֔עוּ כֻּלָּם֙ כְּלָבִ֣ים אִלְּמִ֔ים לֹ֥א יוּכְל֖וּ לִנְבֹּ֑חַ הֹזִים֙ שֹֽׁכְבִ֔ים אֹהֲבֵ֖י לָנֽוּם׃ (יב) וְהַכְּלָבִ֣ים עַזֵּי־ נֶ֗פֶשׁ לֹ֤א יָֽדְעוּ֙ שָׂבְעָ֔ה וְהֵ֣מָּה רֹעִ֔ים לֹ֥א יָדְע֖וּ הָבִ֑ין כֻּלָּם֙ לְדַרְכָּ֣ם פָּנ֔וּ אִ֥ישׁ לְבִצְע֖וֹ מִקָּצֵֽהוּ׃ (יג) אֵתָ֥יוּ אֶקְחָה־ יַ֖יִן וְנִסְבְּאָ֣ה שֵׁכָ֑ר וְהָיָ֤ה כָזֶה֙ י֣וֹם מָחָ֔ר גָּד֖וֹל יֶ֥תֶר מְאֹֽד׃ (יד) הַצַּדִּ֣יק אָבָ֔ד וְאֵ֥ין אִ֖ישׁ שָׂ֣ם עַל־ לֵ֑ב וְאַנְשֵׁי־ חֶ֤סֶד נֶֽאֱסָפִים֙ בְּאֵ֣ין מֵבִ֔ין כִּֽי־ מִפְּנֵ֥י הָרָעָ֖ה נֶאֱסַ֥ף הַצַּדִּֽיק׃ (טו) יָב֣וֹא שָׁל֔וֹם יָנ֖וּחוּ עַל־ מִשְׁכְּבוֹתָ֑ם הֹלֵ֖ךְ נְכֹחֽוֹ׃ (טז) וְאַתֶּ֥ם קִרְבוּ־ הֵ֖נָּה בְּנֵ֣י עֹנְנָ֑ה זֶ֥רַע מְנָאֵ֖ף וַתִּזְנֶֽה׃ (יז) עַל־ מִי֙ תִּתְעַנָּ֔גוּ עַל־ מִ֛י תַּרְחִ֥יבוּ פֶ֖ה תַּאֲרִ֣יכוּ לָשׁ֑וֹן הֲלֽוֹא־ אַתֶּ֥ם יִלְדֵי־ פֶ֖שַׁע זֶ֥רַע שָֽׁקֶר׃ (יח) הַנֵּֽחָמִים֙ בָּֽאֵלִ֔ים תַּ֖חַת כָּל־ עֵ֣ץ רַעֲנָ֑ן שֹׁחֲטֵ֤י הַיְלָדִים֙ בַּנְּחָלִ֔ים תַּ֖חַת סְעִפֵ֥י הַסְּלָעִֽים׃ (יט) בְּחַלְּקֵי־ נַ֣חַל חֶלְקֵ֔ךְ הֵ֥ם הֵ֖ם גּוֹרָלֵ֑ךְ גַּם־ לָהֶ֞ם שָׁפַ֥כְתְּ נֶ֙סֶךְ֙ הֶעֱלִ֣ית מִנְחָ֔ה הַ֥עַל אֵ֖לֶּה אֶנָּחֵֽם׃ (כ) עַ֤ל הַר־ גָּבֹ֙הַּ֙ וְנִשָּׂ֔א שַׂ֖מְתְּ מִשְׁכָּבֵ֑ךְ גַּם־ שָׁ֥ם עָלִ֖ית לִזְבֹּ֥חַ זָֽבַח׃ (כא) וְאַחַ֤ר הַדֶּ֙לֶת֙ וְהַמְּזוּזָ֔ה שַׂ֖מְתְּ זִכְרוֹנֵ֑ךְ כִּ֣י מֵאִתִּ֞י גִּלִּ֣ית וַֽתַּעֲלִ֗י הִרְחַ֤בְתְּ מִשְׁכָּבֵךְ֙ וַתִּכְרָת־ לָ֣ךְ מֵהֶ֔ם אָהַ֥בְתְּ מִשְׁכָּבָ֖ם יָ֥ד חָזִֽית׃ (כב) וַתָּשֻׁ֤רִי לַמֶּ֙לֶךְ֙ בַּשֶּׁ֔מֶן וַתַּרְבִּ֖י רִקֻּחָ֑יִךְ וַתְּשַׁלְּחִ֤י צִרַ֙יִךְ֙ עַד־ מֵ֣רָחֹ֔ק וַתַּשְׁפִּ֖ילִי עַד־ שְׁאֽוֹל׃ (כג) בְּרֹ֤ב דַּרְכֵּךְ֙ יָגַ֔עַתְּ לֹ֥א אָמַ֖רְתְּ נוֹאָ֑שׁ חַיַּ֤ת יָדֵךְ֙ מָצָ֔את עַל־ כֵּ֖ן לֹ֥א חָלִֽית׃ (כד) וְאֶת־ מִ֞י דָּאַ֤גְתְּ וַתִּֽירְאִי֙ כִּ֣י תְכַזֵּ֔בִי וְאוֹתִי֙ לֹ֣א זָכַ֔רְתְּ לֹא־ שַׂ֖מְתְּ עַל־ לִבֵּ֑ךְ הֲלֹ֨א אֲנִ֤י מַחְשֶׁה֙ וּמֵ֣עֹלָ֔ם וְאוֹתִ֖י לֹ֥א תִירָֽאִי׃ (כה) אֲנִ֥י אַגִּ֖יד צִדְקָתֵ֑ךְ וְאֶֽת־ מַעֲשַׂ֖יִךְ וְלֹ֥א יוֹעִילֽוּךְ׃ (כו) בְּזַֽעֲקֵךְ֙ יַצִּילֻ֣ךְ קִבּוּצַ֔יִךְ וְאֶת־ כֻּלָּ֥ם יִשָּׂא־ ר֖וּחַ יִקַּח־ הָ֑בֶל וְהַחוֹסֶ֥ה בִי֙ יִנְחַל־ אֶ֔רֶץ וְיִירַ֖שׁ הַר־ קָדְשִֽׁי׃ (כז) וְאָמַ֥ר סֹֽלּוּ־ סֹ֖לּוּ פַּנּוּ־ דָ֑רֶךְ הָרִ֥ימוּ מִכְשׁ֖וֹל מִדֶּ֥רֶךְ עַמִּֽי׃ (כח) כִּי֩ כֹ֨ה אָמַ֜ר רָ֣ם וְנִשָּׂ֗א שֹׁכֵ֥ן עַד֙ וְקָד֣וֹשׁ שְׁמ֔וֹ מָר֥וֹם וְקָד֖וֹשׁ אֶשְׁכּ֑וֹן וְאֶת־ דַּכָּא֙ וּשְׁפַל־ ר֔וּחַ לְהַחֲיוֹת֙ ר֣וּחַ שְׁפָלִ֔ים וּֽלְהַחֲי֖וֹת לֵ֥ב נִדְכָּאִֽים׃ (כט) כִּ֣י לֹ֤א לְעוֹלָם֙ אָרִ֔יב וְלֹ֥א לָנֶ֖צַח אֶקְּצ֑וֹף כִּי־ ר֙וּחַ֙ מִלְּפָנַ֣י יַֽעֲט֔וֹף וּנְשָׁמ֖וֹת אֲנִ֥י עָשִֽׂיתִי׃ (ל) בַּעֲוֺ֥ן בִּצְע֛וֹ קָצַ֥פְתִּי וְאַכֵּ֖הוּ הַסְתֵּ֣ר וְאֶקְצֹ֑ף וַיֵּ֥לֶךְ שׁוֹבָ֖ב בְּדֶ֥רֶךְ לִבּֽוֹ׃ (לא) דְּרָכָ֥יו רָאִ֖יתִי וְאֶרְפָּאֵ֑הוּ וְאַנְחֵ֕הוּ וַאֲשַׁלֵּ֧ם נִֽחֻמִ֛ים ל֖וֹ וְלַאֲבֵלָֽיו׃ (לב) בּוֹרֵ֖א (נוב) נִ֣יב שְׂפָתָ֑יִם שָׁל֨וֹם ׀ שָׁל֜וֹם לָרָח֧וֹק וְלַקָּר֛וֹב אָמַ֥ר יְהוָ֖ה וּרְפָאתִֽיו׃ (לג) וְהָרְשָׁעִ֖ים כַּיָּ֣ם נִגְרָ֑שׁ כִּ֤י הַשְׁקֵט֙ לֹ֣א יוּכָ֔ל וַיִּגְרְשׁ֥וּ מֵימָ֖יו רֶ֥פֶשׁ וָטִֽיט׃ (לד) אֵ֣ין שָׁל֔וֹם אָמַ֥ר אֱלֹהַ֖י לָרְשָׁעִֽים׃ תחילת פרק נח (לה) קְרָ֤א בְגָרוֹן֙ אַל־ תַּחְשֹׂ֔ךְ כַּשּׁוֹפָ֖ר הָרֵ֣ם קוֹלֶ֑ךָ וְהַגֵּ֤ד לְעַמִּי֙ פִּשְׁעָ֔ם וּלְבֵ֥ית יַעֲקֹ֖ב חַטֹּאתָֽם׃ (לו) וְאוֹתִ֗י י֥וֹם יוֹם֙ יִדְרֹשׁ֔וּן וְדַ֥עַת דְּרָכַ֖י יֶחְפָּצ֑וּן כְּג֞וֹי אֲשֶׁר־ צְדָקָ֣ה עָשָׂ֗ה וּמִשְׁפַּ֤ט אֱלֹהָיו֙ לֹ֣א עָזָ֔ב יִשְׁאָל֙וּנִי֙ מִשְׁפְּטֵי־ צֶ֔דֶק קִרְבַ֥ת אֱלֹהִ֖ים יֶחְפָּצֽוּן׃ (לז) לָ֤מָּה צַּ֙מְנוּ֙ וְלֹ֣א רָאִ֔יתָ עִנִּ֥ינוּ נַפְשֵׁ֖נוּ וְלֹ֣א תֵדָ֑ע הֵ֣ן בְּי֤וֹם צֹֽמְכֶם֙ תִּמְצְאוּ־ חֵ֔פֶץ וְכָל־ עַצְּבֵיכֶ֖ם תִּנְגֹּֽשׂוּ׃ (לח) הֵ֣ן לְרִ֤יב וּמַצָּה֙ תָּצ֔וּמוּ וּלְהַכּ֖וֹת בְּאֶגְרֹ֣ף רֶ֑שַׁע לֹא־ תָצ֣וּמוּ כַיּ֔וֹם לְהַשְׁמִ֥יעַ בַּמָּר֖וֹם קוֹלְכֶֽם׃ (לט) הֲכָזֶ֗ה יִֽהְיֶה֙ צ֣וֹם אֶבְחָרֵ֔הוּ י֛וֹם עַנּ֥וֹת אָדָ֖ם נַפְשׁ֑וֹ הֲלָכֹ֨ף כְּאַגְמֹ֜ן רֹאשׁ֗וֹ וְשַׂ֤ק וָאֵ֙פֶר֙ יַצִּ֔יעַ הֲלָזֶה֙ תִּקְרָא־ צ֔וֹם וְי֥וֹם רָצ֖וֹן לַיהוָֽה׃ (מ) הֲל֣וֹא זֶה֮ צ֣וֹם אֶבְחָרֵהוּ֒ פַּתֵּ֙חַ֙ חַרְצֻבּ֣וֹת רֶ֔שַׁע הַתֵּ֖ר אֲגֻדּ֣וֹת מוֹטָ֑ה וְשַׁלַּ֤ח רְצוּצִים֙ חָפְשִׁ֔ים וְכָל־ מוֹטָ֖ה תְּנַתֵּֽקוּ׃ (מא) הֲל֨וֹא פָרֹ֤ס לָֽרָעֵב֙ לַחְמֶ֔ךָ וַעֲנִיִּ֥ים מְרוּדִ֖ים תָּ֣בִיא בָ֑יִת כִּֽי־ תִרְאֶ֤ה עָרֹם֙ יִשָּׂ֔אוּ וּתְהִלֹּ֥ת יְהוָ֖ה יְבַשֵּֽׂרוּ׃ (כט) כָּל־ צֹ֤אן קֵדָר֙ יִקָּ֣בְצוּ לָ֔ךְ אֵילֵ֥י נְבָי֖וֹת יְשָׁרְת֑וּנֶךְ יַעֲל֤וּ עַל־ רָצוֹן֙ מִזְבְּחִ֔י וּבֵ֥ית תִּפְאַרְתִּ֖י אֲפָאֵֽר׃ (ל) מִי־ אֵ֖לֶּה כָּעָ֣ב תְּעוּפֶ֑ינָה וְכַיּוֹנִ֖ים אֶל־ אֲרֻבֹּתֵיהֶֽם׃ (לא) כִּֽי־ לִ֣י ׀ אִיִּ֣ים יְקַוּ֗וּ וָאֳנִיּ֤וֹת תַּרְשִׁישׁ֙ בָּרִ֣אשֹׁנָ֔ה לְהָבִ֤יא בָנַ֙יִךְ֙ מֵֽרָח֔וֹק כַּסְפָּ֥ם וּזְהָבָ֖ם אִתָּ֑ם לְשֵׁם֙ יְהוָ֣ה אֱלֹהַ֔יִךְ וְלִקְד֥וֹשׁ יִשְׂרָאֵ֖ל כִּ֥י פֵאֲרָֽךְ׃ (לב) וּבָנ֤וּ בְנֵֽי־ נֵכָר֙ חֹמֹתַ֔יִךְ וּמַלְכֵיהֶ֖ם יְשָׁרְת֑וּנֶךְ כִּ֤י בְקִצְפִּי֙ הִכִּיתִ֔יךְ וּבִרְצוֹנִ֖י רִֽחַמְתִּֽיךְ׃ (לג) וּפִתְּח֨וּ שְׁעָרַ֧יִךְ תָּמִ֛יד יוֹמָ֥ם וָלַ֖יְלָה לֹ֣א יִסָּגֵ֑רוּ לְהָבִ֤יא אֵלַ֙יִךְ֙ חֵ֣יל גּוֹיִ֔ם וּמַלְכֵיהֶ֖ם נְהוּגִֽים׃ (לד) כִּֽי־ הַגּ֧וֹי וְהַמַּמְלָכָ֛ה אֲשֶׁ֥ר לֹא־ יַעַבְד֖וּךְ יֹאבֵ֑דוּ וְהַגּוֹיִ֖ם חָרֹ֥ב יֶחֱרָֽבוּ׃ (לה) כְּב֤וֹד הַלְּבָנוֹן֙ אֵלַ֣יִךְ יָב֔וֹא בְּר֛וֹשׁ תִּדְהָ֥ר וּתְאַשּׁ֖וּר יַחְדָּ֑ו לְפָאֵר֙ מְק֣וֹם מִקְדָּשִׁ֔י וּמְק֥וֹם רַגְלַ֖י אֲכַבֵּֽד׃ (לו) וְהָלְכ֨וּ אֵלַ֤יִךְ שְׁח֙וֹחַ֙ בְּנֵ֣י מְעַנַּ֔יִךְ וְהִֽשְׁתַּחֲו֛וּ עַל־ כַּפּ֥וֹת רַגְלַ֖יִךְ כָּל־ מְנַֽאֲצָ֑יִךְ וְקָ֤רְאוּ לָךְ֙ עִ֣יר יְהוָ֔ה צִיּ֖וֹן קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (לז) תַּ֧חַת הֱיוֹתֵ֛ךְ עֲזוּבָ֥ה וּשְׂנוּאָ֖ה וְאֵ֣ין עוֹבֵ֑ר וְשַׂמְתִּיךְ֙ לִגְא֣וֹן עוֹלָ֔ם מְשׂ֖וֹשׂ דּ֥וֹר וָדֽוֹר׃ (לח) וְיָנַקְתְּ֙ חֲלֵ֣ב גּוֹיִ֔ם וְשֹׁ֥ד מְלָכִ֖ים תִּינָ֑קִי וְיָדַ֗עַתְּ כִּ֣י אֲנִ֤י יְהוָה֙ מֽוֹשִׁיעֵ֔ךְ וְגֹאֲלֵ֖ךְ אֲבִ֥יר יַעֲקֹֽב׃ (לט) תַּ֣חַת הַנְּחֹ֜שֶׁת אָבִ֣יא זָהָ֗ב וְתַ֤חַת הַבַּרְזֶל֙ אָ֣בִיא כֶ֔סֶף וְתַ֤חַת הָֽעֵצִים֙ נְחֹ֔שֶׁת וְתַ֥חַת הָאֲבָנִ֖ים בַּרְזֶ֑ל וְשַׂמְתִּ֤י פְקֻדָּתֵךְ֙ שָׁל֔וֹם וְנֹגְשַׂ֖יִךְ צְדָקָֽה׃ (מ) לֹא־ יִשָּׁמַ֨ע ע֤וֹד חָמָס֙ בְּאַרְצֵ֔ךְ שֹׁ֥ד וָשֶׁ֖בֶר בִּגְבוּלָ֑יִךְ וְקָרָ֤את יְשׁוּעָה֙ חוֹמֹתַ֔יִךְ וּשְׁעָרַ֖יִךְ תְּהִלָּֽה׃ (מא) לֹא־ יִֽהְיֶה־ לָּ֨ךְ ע֤וֹד הַשֶּׁ֙מֶשׁ֙ לְא֣וֹר יוֹמָ֔ם וּלְנֹ֕גַהּ הַיָּרֵ֖חַ לֹא־ יָאִ֣יר לָ֑ךְ וְהָיָה־ לָ֤ךְ יְהוָה֙ לְא֣וֹר עוֹלָ֔ם וֵאלֹהַ֖יִךְ לְתִפְאַרְתֵּֽךְ׃ (מב) לֹא־ יָב֥וֹא עוֹד֙ שִׁמְשֵׁ֔ךְ וִירֵחֵ֖ךְ לֹ֣א יֵאָסֵ֑ף כִּ֣י יְהוָ֗ה יִֽהְיֶה־ לָּךְ֙ לְא֣וֹר עוֹלָ֔ם וְשָׁלְמ֖וּ יְמֵ֥י אֶבְלֵֽךְ׃ (מג) וְעַמֵּךְ֙ כֻּלָּ֣ם צַדִּיקִ֔ים לְעוֹלָ֖ם יִ֣ירְשׁוּ אָ֑רֶץ נֵ֧צֶר (מטעו) מַטָּעַ֛י מַעֲשֵׂ֥ה יָדַ֖י לְהִתְפָּאֵֽר׃ (מד) הַקָּטֹן֙ יִֽהְיֶ֣ה לָאֶ֔לֶף וְהַצָּעִ֖יר לְג֣וֹי עָצ֑וּם אֲנִ֥י יְהוָ֖ה בְּעִתָּ֥הּ אֲחִישֶֽׁנָּה׃ (מה) ר֛וּחַ אֲדֹנָ֥י יְהוִ֖ה עָלָ֑י יַ֡עַן מָשַׁח֩ יְהוָ֨ה אֹתִ֜י לְבַשֵּׂ֣ר עֲנָוִ֗ים שְׁלָחַ֙נִי֙ לַחֲבֹ֣שׁ לְנִשְׁבְּרֵי־ לֵ֔ב לִקְרֹ֤א לִשְׁבוּיִם֙ דְּר֔וֹר וְלַאֲסוּרִ֖ים פְּקַח־ קֽוֹחַ׃ (מו) לִקְרֹ֤א שְׁנַת־ רָצוֹן֙ לַֽיהוָ֔ה וְי֥וֹם נָקָ֖ם לֵאלֹהֵ֑ינוּ לְנַחֵ֖ם כָּל־ אֲבֵלִֽים׃ (מז) לָשׂ֣וּם ׀ לַאֲבֵלֵ֣י צִיּ֗וֹן לָתֵת֩ לָהֶ֨ם פְּאֵ֜ר תַּ֣חַת אֵ֗פֶר שֶׁ֤מֶן שָׂשׂוֹן֙ תַּ֣חַת אֵ֔בֶל מַעֲטֵ֣ה תְהִלָּ֔ה תַּ֖חַת ר֣וּחַ כֵּהָ֑ה וְקֹרָ֤א לָהֶם֙ אֵילֵ֣י הַצֶּ֔דֶק מַטַּ֥ע יְהוָ֖ה לְהִתְפָּאֵֽר׃ (מח) וּבָנוּ֙ חָרְב֣וֹת עוֹלָ֔ם שֹׁמְמ֥וֹת רִֽאשֹׁנִ֖ים יְקוֹמֵ֑מוּ וְחִדְּשׁוּ֙ עָ֣רֵי חֹ֔רֶב שֹׁמְמ֖וֹת דּ֥וֹר וָדֽוֹר׃ (מט) וְעָמְד֣וּ זָרִ֔ים וְרָע֖וּ צֹאנְכֶ֑ם וּבְנֵ֣י נֵכָ֔ר אִכָּרֵיכֶ֖ם וְכֹרְמֵיכֶֽם׃ (נ) וְאַתֶּ֗ם כֹּהֲנֵ֤י יְהוָה֙ תִּקָּרֵ֔אוּ מְשָׁרְתֵ֣י אֱלֹהֵ֔ינוּ יֵאָמֵ֖ר לָכֶ֑ם חֵ֤יל גּוֹיִם֙ תֹּאכֵ֔לוּ וּבִכְבוֹדָ֖ם תִּתְיַמָּֽרוּ׃ (נא) תַּ֤חַת בָּשְׁתְּכֶם֙ מִשְׁנֶ֔ה וּכְלִמָּ֖ה יָרֹ֣נּוּ חֶלְקָ֑ם לָכֵ֤ן בְּאַרְצָם֙ מִשְׁנֶ֣ה יִירָ֔שׁוּ שִׂמְחַ֥ת עוֹלָ֖ם תִּֽהְיֶ֥ה לָהֶֽם׃ (נב) כִּ֣י אֲנִ֤י יְהוָה֙ אֹהֵ֣ב מִשְׁפָּ֔ט שֹׂנֵ֥א גָזֵ֖ל בְּעוֹלָ֑ה וְנָתַתִּ֤י פְעֻלָּתָם֙ בֶּאֱמֶ֔ת וּבְרִ֥ית עוֹלָ֖ם אֶכְר֥וֹת לָהֶֽם׃
