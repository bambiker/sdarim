במדבר פרק יח לפי החלוקה לסדרים

פרק יח בספר במדבר (32 פסוקים) נמצא בבמדבר סדר טז (פסוקים יד–לז) ובבמדבר סדר יז (פסוקים א–ח).

במדבר סדר טז

(א) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (ב) דַּבֵּ֣ר ׀ אֶל־ בְּנֵ֣י יִשְׂרָאֵ֗ל וְקַ֣ח מֵֽאִתָּ֡ם מַטֶּ֣ה מַטֶּה֩ לְבֵ֨ית אָ֜ב מֵאֵ֤ת כָּל־ נְשִֽׂיאֵהֶם֙ לְבֵ֣ית אֲבֹתָ֔ם שְׁנֵ֥ים עָשָׂ֖ר מַטּ֑וֹת אִ֣ישׁ אֶת־ שְׁמ֔וֹ תִּכְתֹּ֖ב עַל־ מַטֵּֽהוּ׃ (ג) וְאֵת֙ שֵׁ֣ם אַהֲרֹ֔ן תִּכְתֹּ֖ב עַל־ מַטֵּ֣ה לֵוִ֑י כִּ֚י מַטֶּ֣ה אֶחָ֔ד לְרֹ֖אשׁ בֵּ֥ית אֲבוֹתָֽם׃ (ד) וְהִנַּחְתָּ֖ם בְּאֹ֣הֶל מוֹעֵ֑ד לִפְנֵי֙ הָֽעֵד֔וּת אֲשֶׁ֛ר אִוָּעֵ֥ד לָכֶ֖ם שָֽׁמָּה׃ (ה) וְהָיָ֗ה הָאִ֛ישׁ אֲשֶׁ֥ר אֶבְחַר־ בּ֖וֹ מַטֵּ֣הוּ יִפְרָ֑ח וַהֲשִׁכֹּתִ֣י מֵֽעָלַ֗י אֶת־ תְּלֻנּוֹת֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֛ר הֵ֥ם מַלִּינִ֖ם עֲלֵיכֶֽם׃ (ו) וַיְדַבֵּ֨ר מֹשֶׁ֜ה אֶל־ בְּנֵ֣י יִשְׂרָאֵ֗ל וַיִּתְּנ֣וּ אֵלָ֣יו ׀ כָּֽל־ נְשִֽׂיאֵיהֶ֡ם מַטֶּה֩ לְנָשִׂ֨יא אֶחָ֜ד מַטֶּ֨ה לְנָשִׂ֤יא אֶחָד֙ לְבֵ֣ית אֲבֹתָ֔ם שְׁנֵ֥ים עָשָׂ֖ר מַטּ֑וֹת וּמַטֵּ֥ה אַהֲרֹ֖ן בְּת֥וֹךְ מַטּוֹתָֽם׃ (ז) וַיַּנַּ֥ח מֹשֶׁ֛ה אֶת־ הַמַּטֹּ֖ת לִפְנֵ֣י יְהוָ֑ה בְּאֹ֖הֶל הָעֵדֻֽת׃ (ח) וַיְהִ֣י מִֽמָּחֳרָ֗ת וַיָּבֹ֤א מֹשֶׁה֙ אֶל־ אֹ֣הֶל הָעֵד֔וּת וְהִנֵּ֛ה פָּרַ֥ח מַטֵּֽה־ אַהֲרֹ֖ן לְבֵ֣ית לֵוִ֑י וַיֹּ֤צֵֽא פֶ֙רַח֙ וַיָּ֣צֵֽץ צִ֔יץ וַיִּגְמֹ֖ל שְׁקֵדִֽים׃ (ט) וַיֹּצֵ֨א מֹשֶׁ֤ה אֶת־ כָּל־ הַמַּטֹּת֙ מִלִּפְנֵ֣י יְהוָ֔ה אֶֽל־ כָּל־ בְּנֵ֖י יִשְׂרָאֵ֑ל וַיִּרְא֥וּ וַיִּקְח֖וּ אִ֥ישׁ מַטֵּֽהוּ׃ (י) וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ מֹשֶׁ֗ה הָשֵׁ֞ב אֶת־ מַטֵּ֤ה אַהֲרֹן֙ לִפְנֵ֣י הָעֵד֔וּת לְמִשְׁמֶ֥רֶת לְא֖וֹת לִבְנֵי־ מֶ֑רִי וּתְכַ֧ל תְּלוּנֹּתָ֛ם מֵעָלַ֖י וְלֹ֥א יָמֻֽתוּ׃ (יא) וַיַּ֖עַשׂ מֹשֶׁ֑ה כַּאֲשֶׁ֨ר צִוָּ֧ה יְהוָ֛ה אֹת֖וֹ כֵּ֥ן עָשָֽׂה׃ (יב) וַיֹּֽאמְרוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶל־ מֹשֶׁ֖ה לֵאמֹ֑ר הֵ֥ן גָּוַ֛עְנוּ אָבַ֖דְנוּ כֻּלָּ֥נוּ אָבָֽדְנוּ׃ (יג) כֹּ֣ל הַקָּרֵ֧ב ׀ הַקָּרֵ֛ב אֶל־ מִשְׁכַּ֥ן יְהוָ֖ה יָמ֑וּת הַאִ֥ם תַּ֖מְנוּ לִגְוֺֽעַ׃ תחילת פרק יח (יד) וַיֹּ֤אמֶר יְהוָה֙ אֶֽל־ אַהֲרֹ֔ן אַתָּ֗ה וּבָנֶ֤יךָ וּבֵית־ אָבִ֙יךָ֙ אִתָּ֔ךְ תִּשְׂא֖וּ אֶת־ עֲוֺ֣ן הַמִּקְדָּ֑שׁ וְאַתָּה֙ וּבָנֶ֣יךָ אִתָּ֔ךְ תִּשְׂא֖וּ אֶת־ עֲוֺ֥ן כְּהֻנַּתְכֶֽם׃ (טו) וְגַ֣ם אֶת־ אַחֶיךָ֩ מַטֵּ֨ה לֵוִ֜י שֵׁ֤בֶט אָבִ֙יךָ֙ הַקְרֵ֣ב אִתָּ֔ךְ וְיִלָּו֥וּ עָלֶ֖יךָ וִֽישָׁרְת֑וּךָ וְאַתָּה֙ וּבָנֶ֣יךָ אִתָּ֔ךְ לִפְנֵ֖י אֹ֥הֶל הָעֵדֻֽת׃ (טז) וְשָֽׁמְרוּ֙ מִֽשְׁמַרְתְּךָ֔ וּמִשְׁמֶ֖רֶת כָּל־ הָאֹ֑הֶל אַךְ֩ אֶל־ כְּלֵ֨י הַקֹּ֤דֶשׁ וְאֶל־ הַמִּזְבֵּ֙חַ֙ לֹ֣א יִקְרָ֔בוּ וְלֹֽא־ יָמֻ֥תוּ גַם־ הֵ֖ם גַּם־ אַתֶּֽם׃ (יז) וְנִלְו֣וּ עָלֶ֔יךָ וְשָֽׁמְר֗וּ אֶת־ מִשְׁמֶ֙רֶת֙ אֹ֣הֶל מוֹעֵ֔ד לְכֹ֖ל עֲבֹדַ֣ת הָאֹ֑הֶל וְזָ֖ר לֹא־ יִקְרַ֥ב אֲלֵיכֶֽם׃ (יח) וּשְׁמַרְתֶּ֗ם אֵ֚ת מִשְׁמֶ֣רֶת הַקֹּ֔דֶשׁ וְאֵ֖ת מִשְׁמֶ֣רֶת הַמִּזְבֵּ֑חַ וְלֹֽא־ יִהְיֶ֥ה ע֛וֹד קֶ֖צֶף עַל־ בְּנֵ֥י יִשְׂרָאֵֽל׃ (יט) וַאֲנִ֗י הִנֵּ֤ה לָקַ֙חְתִּי֙ אֶת־ אֲחֵיכֶ֣ם הַלְוִיִּ֔ם מִתּ֖וֹךְ בְּנֵ֣י יִשְׂרָאֵ֑ל לָכֶ֞ם מַתָּנָ֤ה נְתֻנִים֙ לַֽיהוָ֔ה לַעֲבֹ֕ד אֶת־ עֲבֹדַ֖ת אֹ֥הֶל מוֹעֵֽד׃ (כ) וְאַתָּ֣ה וּבָנֶ֣יךָ אִ֠תְּךָ תִּשְׁמְר֨וּ אֶת־ כְּהֻנַּתְכֶ֜ם לְכָל־ דְּבַ֧ר הַמִּזְבֵּ֛חַ וּלְמִבֵּ֥ית לַפָּרֹ֖כֶת וַעֲבַדְתֶּ֑ם עֲבֹדַ֣ת מַתָּנָ֗ה אֶתֵּן֙ אֶת־ כְּהֻנַּתְכֶ֔ם וְהַזָּ֥ר הַקָּרֵ֖ב יוּמָֽת׃ (כא) וַיְדַבֵּ֣ר יְהוָה֮ אֶֽל־ אַהֲרֹן֒ וַאֲנִי֙ הִנֵּ֣ה נָתַ֣תִּֽי לְךָ֔ אֶת־ מִשְׁמֶ֖רֶת תְּרוּמֹתָ֑י לְכָל־ קָדְשֵׁ֣י בְנֵֽי־ יִ֠שְׂרָאֵל לְךָ֨ נְתַתִּ֧ים לְמָשְׁחָ֛ה וּלְבָנֶ֖יךָ לְחָק־ עוֹלָֽם׃ (כב) זֶֽה־ יִהְיֶ֥ה לְךָ֛ מִקֹּ֥דֶשׁ הַקֳּדָשִׁ֖ים מִן־ הָאֵ֑שׁ כָּל־ קָ֠רְבָּנָם לְֽכָל־ מִנְחָתָ֞ם וּלְכָל־ חַטָּאתָ֗ם וּלְכָל־ אֲשָׁמָם֙ אֲשֶׁ֣ר יָשִׁ֣יבוּ לִ֔י קֹ֣דֶשׁ קָֽדָשִׁ֥ים לְךָ֛ ה֖וּא וּלְבָנֶֽיךָ׃ (כג) בְּקֹ֥דֶשׁ הַקֳּדָשִׁ֖ים תֹּאכֲלֶ֑נּוּ כָּל־ זָכָר֙ יֹאכַ֣ל אֹת֔וֹ קֹ֖דֶשׁ יִֽהְיֶה־ לָּֽךְ׃ (כד) וְזֶה־ לְּךָ֞ תְּרוּמַ֣ת מַתָּנָ֗ם לְכָל־ תְּנוּפֹת֮ בְּנֵ֣י יִשְׂרָאֵל֒ לְךָ֣ נְתַתִּ֗ים וּלְבָנֶ֧יךָ וְלִבְנֹתֶ֛יךָ אִתְּךָ֖ לְחָק־ עוֹלָ֑ם כָּל־ טָה֥וֹר בְּבֵיתְךָ֖ יֹאכַ֥ל אֹתֽוֹ׃ (כה) כֹּ֚ל חֵ֣לֶב יִצְהָ֔ר וְכָל־ חֵ֖לֶב תִּיר֣וֹשׁ וְדָגָ֑ן רֵאשִׁיתָ֛ם אֲשֶׁר־ יִתְּנ֥וּ לַֽיהוָ֖ה לְךָ֥ נְתַתִּֽים׃ (כו) בִּכּוּרֵ֞י כָּל־ אֲשֶׁ֧ר בְּאַרְצָ֛ם אֲשֶׁר־ יָבִ֥יאוּ לַיהוָ֖ה לְךָ֣ יִהְיֶ֑ה כָּל־ טָה֥וֹר בְּבֵיתְךָ֖ יֹאכֲלֶֽנּוּ׃ (כז) כָּל־ חֵ֥רֶם בְּיִשְׂרָאֵ֖ל לְךָ֥ יִהְיֶֽה׃ (כח) כָּל־ פֶּ֣טֶר רֶ֠חֶם לְֽכָל־ בָּשָׂ֞ר אֲשֶׁר־ יַקְרִ֧יבוּ לַֽיהוָ֛ה בָּאָדָ֥ם וּבַבְּהֵמָ֖ה יִֽהְיֶה־ לָּ֑ךְ אַ֣ךְ ׀ פָּדֹ֣ה תִפְדֶּ֗ה אֵ֚ת בְּכ֣וֹר הָֽאָדָ֔ם וְאֵ֛ת בְּכֽוֹר־ הַבְּהֵמָ֥ה הַטְּמֵאָ֖ה תִּפְדֶּֽה׃ (כט) וּפְדוּיָו֙ מִבֶּן־ חֹ֣דֶשׁ תִּפְדֶּ֔ה בְּעֶ֨רְכְּךָ֔ כֶּ֛סֶף חֲמֵ֥שֶׁת שְׁקָלִ֖ים בְּשֶׁ֣קֶל הַקֹּ֑דֶשׁ עֶשְׂרִ֥ים גֵּרָ֖ה הֽוּא׃ (ל) אַ֣ךְ בְּֽכוֹר־ שׁ֡וֹר אֽוֹ־ בְכ֨וֹר כֶּ֜שֶׂב אֽוֹ־ בְכ֥וֹר עֵ֛ז לֹ֥א תִפְדֶּ֖ה קֹ֣דֶשׁ הֵ֑ם אֶת־ דָּמָ֞ם תִּזְרֹ֤ק עַל־ הַמִּזְבֵּ֙חַ֙ וְאֶת־ חֶלְבָּ֣ם תַּקְטִ֔יר אִשֶּׁ֛ה לְרֵ֥יחַ נִיחֹ֖חַ לַֽיהוָֽה׃ (לא) וּבְשָׂרָ֖ם יִהְיֶה־ לָּ֑ךְ כַּחֲזֵ֧ה הַתְּנוּפָ֛ה וּכְשׁ֥וֹק הַיָּמִ֖ין לְךָ֥ יִהְיֶֽה׃ (לב) כֹּ֣ל ׀ תְּרוּמֹ֣ת הַקֳּדָשִׁ֗ים אֲשֶׁ֨ר יָרִ֥ימוּ בְנֵֽי־ יִשְׂרָאֵל֮ לַֽיהוָה֒ נָתַ֣תִּֽי לְךָ֗ וּלְבָנֶ֧יךָ וְלִבְנֹתֶ֛יךָ אִתְּךָ֖ לְחָק־ עוֹלָ֑ם בְּרִית֩ מֶ֨לַח עוֹלָ֥ם הִוא֙ לִפְנֵ֣י יְהוָ֔ה לְךָ֖ וּלְזַרְעֲךָ֥ אִתָּֽךְ׃ (לג) וַיֹּ֨אמֶר יְהוָ֜ה אֶֽל־ אַהֲרֹ֗ן בְּאַרְצָם֙ לֹ֣א תִנְחָ֔ל וְחֵ֕לֶק לֹא־ יִהְיֶ֥ה לְךָ֖ בְּתוֹכָ֑ם אֲנִ֤י חֶלְקְךָ֙ וְנַחֲלָ֣תְךָ֔ בְּת֖וֹךְ בְּנֵ֥י יִשְׂרָאֵֽל׃ (לד) וְלִבְנֵ֣י לֵוִ֔י הִנֵּ֥ה נָתַ֛תִּי כָּל־ מַֽעֲשֵׂ֥ר בְּיִשְׂרָאֵ֖ל לְנַחֲלָ֑ה חֵ֤לֶף עֲבֹֽדָתָם֙ אֲשֶׁר־ הֵ֣ם עֹֽבְדִ֔ים אֶת־ עֲבֹדַ֖ת אֹ֥הֶל מוֹעֵֽד׃ (לה) וְלֹא־ יִקְרְב֥וּ ע֛וֹד בְּנֵ֥י יִשְׂרָאֵ֖ל אֶל־ אֹ֣הֶל מוֹעֵ֑ד לָשֵׂ֥את חֵ֖טְא לָמֽוּת׃ (לו) וְעָבַ֨ד הַלֵּוִ֜י ה֗וּא אֶת־ עֲבֹדַת֙ אֹ֣הֶל מוֹעֵ֔ד וְהֵ֖ם יִשְׂא֣וּ עֲוֺנָ֑ם חֻקַּ֤ת עוֹלָם֙ לְדֹרֹ֣תֵיכֶ֔ם וּבְתוֹךְ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לֹ֥א יִנְחֲל֖וּ נַחֲלָֽה׃ (לז) כִּ֞י אֶת־ מַעְשַׂ֣ר בְּנֵֽי־ יִשְׂרָאֵ֗ל אֲשֶׁ֨ר יָרִ֤ימוּ לַֽיהוָה֙ תְּרוּמָ֔ה נָתַ֥תִּי לַלְוִיִּ֖ם לְנַחֲלָ֑ה עַל־ כֵּן֙ אָמַ֣רְתִּי לָהֶ֔ם בְּתוֹךְ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לֹ֥א יִנְחֲל֖וּ נַחֲלָֽה׃

במדבר סדר יז

(א) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (ב) וְאֶל־ הַלְוִיִּ֣ם תְּדַבֵּר֮ וְאָמַרְתָּ֣ אֲלֵהֶם֒ כִּֽי־ תִ֠קְחוּ מֵאֵ֨ת בְּנֵֽי־ יִשְׂרָאֵ֜ל אֶת־ הַֽמַּעֲשֵׂ֗ר אֲשֶׁ֨ר נָתַ֧תִּי לָכֶ֛ם מֵאִתָּ֖ם בְּנַחֲלַתְכֶ֑ם וַהֲרֵמֹתֶ֤ם מִמֶּ֙נּוּ֙ תְּרוּמַ֣ת יְהוָ֔ה מַעֲשֵׂ֖ר מִן־ הַֽמַּעֲשֵֽׂר׃ (ג) וְנֶחְשַׁ֥ב לָכֶ֖ם תְּרוּמַתְכֶ֑ם כַּדָּגָן֙ מִן־ הַגֹּ֔רֶן וְכַֽמְלֵאָ֖ה מִן־ הַיָּֽקֶב׃ (ד) כֵּ֣ן תָּרִ֤ימוּ גַם־ אַתֶּם֙ תְּרוּמַ֣ת יְהוָ֔ה מִכֹּל֙ מַעְשְׂרֹ֣תֵיכֶ֔ם אֲשֶׁ֣ר תִּקְח֔וּ מֵאֵ֖ת בְּנֵ֣י יִשְׂרָאֵ֑ל וּנְתַתֶּ֤ם מִמֶּ֙נּוּ֙ אֶת־ תְּרוּמַ֣ת יְהוָ֔ה לְאַהֲרֹ֖ן הַכֹּהֵֽן׃ (ה) מִכֹּל֙ מַתְּנֹ֣תֵיכֶ֔ם תָּרִ֕ימוּ אֵ֖ת כָּל־ תְּרוּמַ֣ת יְהוָ֑ה מִכָּל־ חֶלְבּ֔וֹ אֶֽת־ מִקְדְּשׁ֖וֹ מִמֶּֽנּוּ׃ (ו) וְאָמַרְתָּ֖ אֲלֵהֶ֑ם בַּהֲרִֽימְכֶ֤ם אֶת־ חֶלְבּוֹ֙ מִמֶּ֔נּוּ וְנֶחְשַׁב֙ לַלְוִיִּ֔ם כִּתְבוּאַ֥ת גֹּ֖רֶן וְכִתְבוּאַ֥ת יָֽקֶב׃ (ז) וַאֲכַלְתֶּ֤ם אֹתוֹ֙ בְּכָל־ מָק֔וֹם אַתֶּ֖ם וּבֵֽיתְכֶ֑ם כִּֽי־ שָׂכָ֥ר הוּא֙ לָכֶ֔ם חֵ֥לֶף עֲבֹֽדַתְכֶ֖ם בְּאֹ֥הֶל מוֹעֵֽד׃ (ח) וְלֹֽא־ תִשְׂא֤וּ עָלָיו֙ חֵ֔טְא בַּהֲרִֽימְכֶ֥ם אֶת־ חֶלְבּ֖וֹ מִמֶּ֑נּוּ וְאֶת־ קָדְשֵׁ֧י בְנֵי־ יִשְׂרָאֵ֛ל לֹ֥א תְחַלְּל֖וּ וְלֹ֥א תָמֽוּתוּ׃ סוף פרק יח (ט) וַיְדַבֵּ֣ר יְהוָ֔ה אֶל־ מֹשֶׁ֥ה וְאֶֽל־ אַהֲרֹ֖ן לֵאמֹֽר׃ (י) זֹ֚את חֻקַּ֣ת הַתּוֹרָ֔ה אֲשֶׁר־ צִוָּ֥ה יְהוָ֖ה לֵאמֹ֑ר דַּבֵּ֣ר ׀ אֶל־ בְּנֵ֣י יִשְׂרָאֵ֗ל וְיִקְח֣וּ אֵלֶיךָ֩ פָרָ֨ה אֲדֻמָּ֜ה תְּמִימָ֗ה אֲשֶׁ֤ר אֵֽין־ בָּהּ֙ מ֔וּם אֲשֶׁ֛ר לֹא־ עָלָ֥ה עָלֶ֖יהָ עֹֽל׃ (יא) וּנְתַתֶּ֣ם אֹתָ֔הּ אֶל־ אֶלְעָזָ֖ר הַכֹּהֵ֑ן וְהוֹצִ֤יא אֹתָהּ֙ אֶל־ מִח֣וּץ לַֽמַּחֲנֶ֔ה וְשָׁחַ֥ט אֹתָ֖הּ לְפָנָֽיו׃ (יב) וְלָקַ֞ח אֶלְעָזָ֧ר הַכֹּהֵ֛ן מִדָּמָ֖הּ בְּאֶצְבָּע֑וֹ וְהִזָּ֞ה אֶל־ נֹ֨כַח פְּנֵ֧י אֹֽהֶל־ מוֹעֵ֛ד מִדָּמָ֖הּ שֶׁ֥בַע פְּעָמִֽים׃ (יג) וְשָׂרַ֥ף אֶת־ הַפָּרָ֖ה לְעֵינָ֑יו אֶת־ עֹרָ֤הּ וְאֶת־ בְּשָׂרָהּ֙ וְאֶת־ דָּמָ֔הּ עַל־ פִּרְשָׁ֖הּ יִשְׂרֹֽף׃ (יד) וְלָקַ֣ח הַכֹּהֵ֗ן עֵ֥ץ אֶ֛רֶז וְאֵז֖וֹב וּשְׁנִ֣י תוֹלָ֑עַת וְהִשְׁלִ֕יךְ אֶל־ תּ֖וֹךְ שְׂרֵפַ֥ת הַפָּרָֽה׃ (טו) וְכִבֶּ֨ס בְּגָדָ֜יו הַכֹּהֵ֗ן וְרָחַ֤ץ בְּשָׂרוֹ֙ בַּמַּ֔יִם וְאַחַ֖ר יָב֣וֹא אֶל־ הַֽמַּחֲנֶ֑ה וְטָמֵ֥א הַכֹּהֵ֖ן עַד־ הָעָֽרֶב׃ (טז) וְהַשֹּׂרֵ֣ף אֹתָ֔הּ יְכַבֵּ֤ס בְּגָדָיו֙ בַּמַּ֔יִם וְרָחַ֥ץ בְּשָׂר֖וֹ בַּמָּ֑יִם וְטָמֵ֖א עַד־ הָעָֽרֶב׃ (יז) וְאָסַ֣ף ׀ אִ֣ישׁ טָה֗וֹר אֵ֚ת אֵ֣פֶר הַפָּרָ֔ה וְהִנִּ֛יחַ מִח֥וּץ לַֽמַּחֲנֶ֖ה בְּמָק֣וֹם טָה֑וֹר וְ֠הָיְתָה לַעֲדַ֨ת בְּנֵֽי־ יִשְׂרָאֵ֧ל לְמִשְׁמֶ֛רֶת לְמֵ֥י נִדָּ֖ה חַטָּ֥את הִֽוא׃ (יח) וְ֠כִבֶּס הָאֹסֵ֨ף אֶת־ אֵ֤פֶר הַפָּרָה֙ אֶת־ בְּגָדָ֔יו וְטָמֵ֖א עַד־ הָעָ֑רֶב וְֽהָיְתָ֞ה לִבְנֵ֣י יִשְׂרָאֵ֗ל וְלַגֵּ֛ר הַגָּ֥ר בְּתוֹכָ֖ם לְחֻקַּ֥ת עוֹלָֽם׃ (יט) הַנֹּגֵ֥עַ בְּמֵ֖ת לְכָל־ נֶ֣פֶשׁ אָדָ֑ם וְטָמֵ֖א שִׁבְעַ֥ת יָמִֽים׃ (כ) ה֣וּא יִתְחַטָּא־ ב֞וֹ בַּיּ֧וֹם הַשְּׁלִישִׁ֛י וּבַיּ֥וֹם הַשְּׁבִיעִ֖י יִטְהָ֑ר וְאִם־ לֹ֨א יִתְחַטָּ֜א בַּיּ֧וֹם הַשְּׁלִישִׁ֛י וּבַיּ֥וֹם הַשְּׁבִיעִ֖י לֹ֥א יִטְהָֽר׃ (כא) כָּֽל־ הַנֹּגֵ֡עַ בְּמֵ֣ת בְּנֶפֶשׁ֩ הָאָדָ֨ם אֲשֶׁר־ יָמ֜וּת וְלֹ֣א יִתְחַטָּ֗א אֶת־ מִשְׁכַּ֤ן יְהוָה֙ טִמֵּ֔א וְנִכְרְתָ֛ה הַנֶּ֥פֶשׁ הַהִ֖וא מִיִּשְׂרָאֵ֑ל כִּי֩ מֵ֨י נִדָּ֜ה לֹא־ זֹרַ֤ק עָלָיו֙ טָמֵ֣א יִהְיֶ֔ה ע֖וֹד טֻמְאָת֥וֹ בֽוֹ׃ (כב) זֹ֚את הַתּוֹרָ֔ה אָדָ֖ם כִּֽי־ יָמ֣וּת בְּאֹ֑הֶל כָּל־ הַבָּ֤א אֶל־ הָאֹ֙הֶל֙ וְכָל־ אֲשֶׁ֣ר בָּאֹ֔הֶל יִטְמָ֖א שִׁבְעַ֥ת יָמִֽים׃ (כג) וְכֹל֙ כְּלִ֣י פָת֔וּחַ אֲשֶׁ֛ר אֵין־ צָמִ֥יד פָּתִ֖יל עָלָ֑יו טָמֵ֖א הֽוּא׃ (כד) וְכֹ֨ל אֲשֶׁר־ יִגַּ֜ע עַל־ פְּנֵ֣י הַשָּׂדֶ֗ה בַּֽחֲלַל־ חֶ֙רֶב֙ א֣וֹ בְמֵ֔ת אֽוֹ־ בְעֶ֥צֶם אָדָ֖ם א֣וֹ בְקָ֑בֶר יִטְמָ֖א שִׁבְעַ֥ת יָמִֽים׃ (כה) וְלָֽקְחוּ֙ לַטָּמֵ֔א מֵעֲפַ֖ר שְׂרֵפַ֣ת הַֽחַטָּ֑את וְנָתַ֥ן עָלָ֛יו מַ֥יִם חַיִּ֖ים אֶל־ כֶּֽלִי׃ (כו) וְלָקַ֨ח אֵז֜וֹב וְטָבַ֣ל בַּמַּיִם֮ אִ֣ישׁ טָהוֹר֒ וְהִזָּ֤ה עַל־ הָאֹ֙הֶל֙ וְעַל־ כָּל־ הַכֵּלִ֔ים וְעַל־ הַנְּפָשׁ֖וֹת אֲשֶׁ֣ר הָֽיוּ־ שָׁ֑ם וְעַל־ הַנֹּגֵ֗עַ בַּעֶ֙צֶם֙ א֣וֹ בֶֽחָלָ֔ל א֥וֹ בַמֵּ֖ת א֥וֹ בַקָּֽבֶר׃ (כז) וְהִזָּ֤ה הַטָּהֹר֙ עַל־ הַטָּמֵ֔א בַּיּ֥וֹם הַשְּׁלִישִׁ֖י וּבַיּ֣וֹם הַשְּׁבִיעִ֑י וְחִטְּאוֹ֙ בַּיּ֣וֹם הַשְּׁבִיעִ֔י וְכִבֶּ֧ס בְּגָדָ֛יו וְרָחַ֥ץ בַּמַּ֖יִם וְטָהֵ֥ר בָּעָֽרֶב׃ (כח) וְאִ֤ישׁ אֲשֶׁר־ יִטְמָא֙ וְלֹ֣א יִתְחַטָּ֔א וְנִכְרְתָ֛ה הַנֶּ֥פֶשׁ הַהִ֖וא מִתּ֣וֹךְ הַקָּהָ֑ל כִּי֩ אֶת־ מִקְדַּ֨שׁ יְהוָ֜ה טִמֵּ֗א מֵ֥י נִדָּ֛ה לֹא־ זֹרַ֥ק עָלָ֖יו טָמֵ֥א הֽוּא׃ (כט) וְהָיְתָ֥ה לָּהֶ֖ם לְחֻקַּ֣ת עוֹלָ֑ם וּמַזֵּ֤ה מֵֽי־ הַנִּדָּה֙ יְכַבֵּ֣ס בְּגָדָ֔יו וְהַנֹּגֵ֙עַ֙ בְּמֵ֣י הַנִּדָּ֔ה יִטְמָ֖א עַד־ הָעָֽרֶב׃ (ל) וְכֹ֛ל אֲשֶׁר־ יִגַּע־ בּ֥וֹ הַטָּמֵ֖א יִטְמָ֑א וְהַנֶּ֥פֶשׁ הַנֹּגַ֖עַת תִּטְמָ֥א עַד־ הָעָֽרֶב׃ (לא) וַיָּבֹ֣אוּ בְנֵֽי־ יִ֠שְׂרָאֵל כָּל־ הָ֨עֵדָ֤ה מִדְבַּר־ צִן֙ בַּחֹ֣דֶשׁ הָֽרִאשׁ֔וֹן וַיֵּ֥שֶׁב הָעָ֖ם בְּקָדֵ֑שׁ וַתָּ֤מָת שָׁם֙ מִרְיָ֔ם וַתִּקָּבֵ֖ר שָֽׁם׃ (לב) וְלֹא־ הָ֥יָה מַ֖יִם לָעֵדָ֑ה וַיִּקָּ֣הֲל֔וּ עַל־ מֹשֶׁ֖ה וְעַֽל־ אַהֲרֹֽן׃ (לג) וַיָּ֥רֶב הָעָ֖ם עִם־ מֹשֶׁ֑ה וַיֹּאמְר֣וּ לֵאמֹ֔ר וְל֥וּ גָוַ֛עְנוּ בִּגְוַ֥ע אַחֵ֖ינוּ לִפְנֵ֥י יְהוָֽה׃ (לד) וְלָמָ֤ה הֲבֵאתֶם֙ אֶת־ קְהַ֣ל יְהוָ֔ה אֶל־ הַמִּדְבָּ֖ר הַזֶּ֑ה לָמ֣וּת שָׁ֔ם אֲנַ֖חְנוּ וּבְעִירֵֽנוּ׃ (לה) וְלָמָ֤ה הֶֽעֱלִיתֻ֙נוּ֙ מִמִּצְרַ֔יִם לְהָבִ֣יא אֹתָ֔נוּ אֶל־ הַמָּק֥וֹם הָרָ֖ע הַזֶּ֑ה לֹ֣א ׀ מְק֣וֹם זֶ֗רַע וּתְאֵנָ֤ה וְגֶ֙פֶן֙ וְרִמּ֔וֹן וּמַ֥יִם אַ֖יִן לִשְׁתּֽוֹת׃ (לו) וַיָּבֹא֩ מֹשֶׁ֨ה וְאַהֲרֹ֜ן מִפְּנֵ֣י הַקָּהָ֗ל אֶל־ פֶּ֙תַח֙ אֹ֣הֶל מוֹעֵ֔ד וַֽיִּפְּל֖וּ עַל־ פְּנֵיהֶ֑ם וַיֵּרָ֥א כְבוֹד־ יְהוָ֖ה אֲלֵיהֶֽם׃ (לז) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (לח) קַ֣ח אֶת־ הַמַּטֶּ֗ה וְהַקְהֵ֤ל אֶת־ הָעֵדָה֙ אַתָּה֙ וְאַהֲרֹ֣ן אָחִ֔יךָ וְדִבַּרְתֶּ֧ם אֶל־ הַסֶּ֛לַע לְעֵינֵיהֶ֖ם וְנָתַ֣ן מֵימָ֑יו וְהוֹצֵאתָ֨ לָהֶ֥ם מַ֙יִם֙ מִן־ הַסֶּ֔לַע וְהִשְׁקִיתָ֥ אֶת־ הָעֵדָ֖ה וְאֶת־ בְּעִירָֽם׃ (לט) וַיִּקַּ֥ח מֹשֶׁ֛ה אֶת־ הַמַּטֶּ֖ה מִלִּפְנֵ֣י יְהוָ֑ה כַּאֲשֶׁ֖ר צִוָּֽהוּ׃ (מ) וַיַּקְהִ֜לוּ מֹשֶׁ֧ה וְאַהֲרֹ֛ן אֶת־ הַקָּהָ֖ל אֶל־ פְּנֵ֣י הַסָּ֑לַע וַיֹּ֣אמֶר לָהֶ֗ם שִׁמְעוּ־ נָא֙ הַמֹּרִ֔ים הֲמִן־ הַסֶּ֣לַע הַזֶּ֔ה נוֹצִ֥יא לָכֶ֖ם מָֽיִם׃ (מא) וַיָּ֨רֶם מֹשֶׁ֜ה אֶת־ יָד֗וֹ וַיַּ֧ךְ אֶת־ הַסֶּ֛לַע בְּמַטֵּ֖הוּ פַּעֲמָ֑יִם וַיֵּצְאוּ֙ מַ֣יִם רַבִּ֔ים וַתֵּ֥שְׁתְּ הָעֵדָ֖ה וּבְעִירָֽם׃ (מב) וַיֹּ֣אמֶר יְהוָה֮ אֶל־ מֹשֶׁ֣ה וְאֶֽל־ אַהֲרֹן֒ יַ֚עַן לֹא־ הֶאֱמַנְתֶּ֣ם בִּ֔י לְהַ֨קְדִּישֵׁ֔נִי לְעֵינֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל לָכֵ֗ן לֹ֤א תָבִ֙יאוּ֙ אֶת־ הַקָּהָ֣ל הַזֶּ֔ה אֶל־ הָאָ֖רֶץ אֲשֶׁר־ נָתַ֥תִּי לָהֶֽם׃ (מג) הֵ֚מָּה מֵ֣י מְרִיבָ֔ה אֲשֶׁר־ רָב֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל אֶת־ יְהוָ֑ה וַיִּקָּדֵ֖שׁ בָּֽם׃
