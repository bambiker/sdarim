משלי פרק כז לפי החלוקה לסדרים

פרק כז בספר משלי (27 פסוקים) נמצא במשלי סדר ז (פסוקים מה–עא).

משלי סדר ז

(א) כְּצִנַּת־ שֶׁ֨לֶג ׀ בְּי֬וֹם קָצִ֗יר צִ֣יר נֶ֭אֱמָן לְשֹׁלְחָ֑יו וְנֶ֖פֶשׁ אֲדֹנָ֣יו יָשִֽׁיב׃ (ב) נְשִׂיאִ֣ים וְ֭רוּחַ וְגֶ֣שֶׁם אָ֑יִן אִ֥ישׁ מִ֝תְהַלֵּ֗ל בְּמַתַּת־ שָֽׁקֶר׃ (ג) בְּאֹ֣רֶךְ אַ֭פַּיִם יְפֻתֶּ֣ה קָצִ֑ין וְלָשׁ֥וֹן רַ֝כָּ֗ה תִּשְׁבָּר־ גָּֽרֶם׃ (ד) דְּבַ֣שׁ מָ֭צָאתָ אֱכֹ֣ל דַּיֶּ֑ךָּ פֶּן־ תִּ֝שְׂבָּעֶ֗נּוּ וַהֲקֵֽאתֽוֹ׃ (ה) הֹקַ֣ר רַ֭גְלְךָ מִבֵּ֣ית רֵעֶ֑ךָ פֶּן־ יִ֝שְׂבָּעֲךָ֗ וּשְׂנֵאֶֽךָ׃ (ו) מֵפִ֣יץ וְ֭חֶרֶב וְחֵ֣ץ שָׁנ֑וּן אִ֥ישׁ עֹנֶ֥ה בְ֝רֵעֵ֗הוּ עֵ֣ד שָֽׁקֶר׃ (ז) שֵׁ֣ן רֹ֭עָה וְרֶ֣גֶל מוּעָ֑דֶת מִבְטָ֥ח בּ֝וֹגֵ֗ד בְּי֣וֹם צָרָֽה׃ (ח) מַ֥עֲדֶה בֶּ֨גֶד ׀ בְּי֣וֹם קָ֭רָה חֹ֣מֶץ עַל־ נָ֑תֶר וְשָׁ֥ר בַּ֝שִּׁרִ֗ים עַ֣ל לֶב־ רָֽע׃ (ט) אִם־ רָעֵ֣ב שֹׂ֭נַאֲךָ הַאֲכִלֵ֣הוּ לָ֑חֶם וְאִם־ צָ֝מֵ֗א הַשְׁקֵ֥הוּ מָֽיִם׃ (י) כִּ֤י גֶֽחָלִ֗ים אַ֭תָּה חֹתֶ֣ה עַל־ רֹאשׁ֑וֹ וַֽ֝יהוָ֗ה יְשַׁלֶּם־ לָֽךְ׃ (יא) ר֣וּחַ צָ֭פוֹן תְּח֣וֹלֵֽל גָּ֑שֶׁם וּפָנִ֥ים נִ֝זְעָמִ֗ים לְשׁ֣וֹן סָֽתֶר׃ (יב) ט֗וֹב שֶׁ֥בֶת עַל־ פִּנַּת־ גָּ֑ג מֵאֵ֥שֶׁת (מדונים) מִ֝דְיָנִ֗ים וּבֵ֥ית חָֽבֶר׃ (יג) מַ֣יִם קָ֭רִים עַל־ נֶ֣פֶשׁ עֲיֵפָ֑ה וּשְׁמוּעָ֥ה ט֝וֹבָ֗ה מֵאֶ֥רֶץ מֶרְחָֽק׃ (יד) מַעְיָ֣ן נִ֭רְפָּשׂ וּמָק֣וֹר מָשְׁחָ֑ת צַ֝דִּ֗יק מָ֣ט לִפְנֵֽי־ רָשָֽׁע׃ (טו) אָ֘כֹ֤ל דְּבַ֣שׁ הַרְבּ֣וֹת לֹא־ ט֑וֹב וְחֵ֖קֶר כְּבֹדָ֣ם כָּבֽוֹד׃ (טז) עִ֣יר פְּ֭רוּצָה אֵ֣ין חוֹמָ֑ה אִ֝֗ישׁ אֲשֶׁ֤ר אֵ֖ין מַעְצָ֣ר לְרוּחֽוֹ׃ (יז) כַּשֶּׁ֤לֶג ׀ בַּקַּ֗יִץ וְכַמָּטָ֥ר בַּקָּצִ֑יר כֵּ֤ן לֹא־ נָאוֶ֖ה לִכְסִ֣יל כָּבֽוֹד׃ (יח) כַּצִּפּ֣וֹר לָ֭נוּד כַּדְּר֣וֹר לָע֑וּף כֵּ֥ן קִֽלְלַ֥ת חִ֝נָּ֗ם (לא) ל֣וֹ תָבֹֽא׃ (יט) שׁ֣וֹט לַ֭סּוּס מֶ֣תֶג לַחֲמ֑וֹר וְ֝שֵׁ֗בֶט לְגֵ֣ו כְּסִילִֽים׃ (כ) אַל־ תַּ֣עַן כְּ֭סִיל כְּאִוַּלְתּ֑וֹ פֶּֽן־ תִּשְׁוֶה־ לּ֥וֹ גַם־ אָֽתָּה׃ (כא) עֲנֵ֣ה כְ֭סִיל כְּאִוַּלְתּ֑וֹ פֶּן־ יִהְיֶ֖ה חָכָ֣ם בְּעֵינָֽיו׃ (כב) מְקַצֶּ֣ה רַ֭גְלַיִם חָמָ֣ס שֹׁתֶ֑ה שֹׁלֵ֖חַ דְּבָרִ֣ים בְּיַד־ כְּסִֽיל׃ (כג) דַּלְי֣וּ שֹׁ֭קַיִם מִפִּסֵּ֑חַ וּ֝מָשָׁ֗ל בְּפִ֣י כְסִילִֽים׃ (כד) כִּצְר֣וֹר אֶ֭בֶן בְּמַרְגֵּמָ֑ה כֵּן־ נוֹתֵ֖ן לִכְסִ֣יל כָּבֽוֹד׃ (כה) ח֭וֹחַ עָלָ֣ה בְיַד־ שִׁכּ֑וֹר וּ֝מָשָׁ֗ל בְּפִ֣י כְסִילִֽים׃ (כו) רַ֥ב מְחֽוֹלֵֽל־ כֹּ֑ל וְשֹׂכֵ֥ר כְּ֝סִ֗יל וְשֹׂכֵ֥ר עֹבְרִֽים׃ (כז) כְּ֭כֶלֶב שָׁ֣ב עַל־ קֵא֑וֹ כְּ֝סִ֗יל שׁוֹנֶ֥ה בְאִוַּלְתּֽוֹ׃ (כח) רָאִ֗יתָ אִ֭ישׁ חָכָ֣ם בְּעֵינָ֑יו תִּקְוָ֖ה לִכְסִ֣יל מִמֶּֽנּוּ׃ (כט) אָמַ֣ר עָ֭צֵל שַׁ֣חַל בַּדָּ֑רֶךְ אֲ֝רִ֗י בֵּ֣ין הָרְחֹבֽוֹת׃ (ל) הַ֭דֶּלֶת תִּסּ֣וֹב עַל־ צִירָ֑הּ וְ֝עָצֵ֗ל עַל־ מִטָּתֽוֹ׃ (לא) טָ֘מַ֤ן עָצֵ֣ל יָ֭דוֹ בַּצַּלָּ֑חַת נִ֝לְאָ֗ה לַֽהֲשִׁיבָ֥הּ אֶל־ פִּֽיו׃ (לב) חָכָ֣ם עָצֵ֣ל בְּעֵינָ֑יו מִ֝שִּׁבְעָ֗ה מְשִׁ֣יבֵי טָֽעַם׃ (לג) מַחֲזִ֥יק בְּאָזְנֵי־ כָ֑לֶב עֹבֵ֥ר מִ֝תְעַבֵּ֗ר עַל־ רִ֥יב לֹּֽא־ לֽוֹ׃ (לד) כְּֽ֭מִתְלַהְלֵהַּ הַיֹּרֶ֥ה זִקִּ֗ים חִצִּ֥ים וָמָֽוֶת׃ (לה) כֵּֽן־ אִ֭ישׁ רִמָּ֣ה אֶת־ רֵעֵ֑הוּ וְ֝אָמַ֗ר הֲ‍ֽלֹא־ מְשַׂחֵ֥ק אָֽנִי׃ (לו) בְּאֶ֣פֶס עֵ֭צִים תִּכְבֶּה־ אֵ֑שׁ וּבְאֵ֥ין נִ֝רְגָּ֗ן יִשְׁתֹּ֥ק מָדֽוֹן׃ (לז) פֶּחָ֣ם לְ֭גֶחָלִים וְעֵצִ֣ים לְאֵ֑שׁ וְאִ֥ישׁ (מדונים) מִ֝דְיָנִ֗ים לְחַרְחַר־ רִֽיב׃ (לח) דִּבְרֵ֣י נִ֭רְגָּן כְּמִֽתְלַהֲמִ֑ים וְ֝הֵ֗ם יָרְד֥וּ חַדְרֵי־ בָֽטֶן׃ (לט) כֶּ֣סֶף סִ֭יגִים מְצֻפֶּ֣ה עַל־ חָ֑רֶשׂ שְׂפָתַ֖יִם דֹּלְקִ֣ים וְלֶב־ רָֽע׃ (מ) (בשפתו) בִּ֭שְׂפָתָיו יִנָּכֵ֣ר שׂוֹנֵ֑א וּ֝בְקִרְבּ֗וֹ יָשִׁ֥ית מִרְמָֽה׃ (מא) כִּֽי־ יְחַנֵּ֣ן ק֭וֹלוֹ אַל־ תַּֽאֲמֶן־ בּ֑וֹ כִּ֤י שֶׁ֖בַע תּוֹעֵב֣וֹת בְּלִבּֽוֹ׃ (מב) תִּכַּסֶּ֣ה שִׂ֭נְאָה בְּמַשָּׁא֑וֹן תִּגָּלֶ֖ה רָעָת֣וֹ בְקָהָֽל׃ (מג) כֹּֽרֶה־ שַּׁ֭חַת בָּ֣הּ יִפֹּ֑ל וְגֹ֥לֵ֥ל אֶ֝בֶן אֵלָ֥יו תָּשֽׁוּב׃ (מד) לְֽשׁוֹן־ שֶׁ֭קֶר יִשְׂנָ֣א דַכָּ֑יו וּפֶ֥ה חָ֝לָ֗ק יַעֲשֶׂ֥ה מִדְחֶֽה׃ תחילת פרק כז (מה) אַֽל־ תִּ֭תְהַלֵּל בְּי֣וֹם מָחָ֑ר כִּ֤י לֹא־ תֵ֝דַ֗ע מַה־ יֵּ֥לֶד יֽוֹם׃ (מו) יְהַלֶּלְךָ֣ זָ֣ר וְלֹא־ פִ֑יךָ נָ֝כְרִ֗י וְאַל־ שְׂפָתֶֽיךָ׃ (מז) כֹּֽבֶד־ אֶ֭בֶן וְנֵ֣טֶל הַח֑וֹל וְכַ֥עַס אֱ֝וִ֗יל כָּבֵ֥ד מִשְּׁנֵיהֶֽם׃ (מח) אַכְזְרִיּ֣וּת חֵ֭מָה וְשֶׁ֣טֶף אָ֑ף וּמִ֥י יַ֝עֲמֹד לִפְנֵ֥י קִנְאָֽה׃ (מט) ט֭וֹבָה תּוֹכַ֣חַת מְגֻלָּ֑ה מֵֽאַהֲבָ֥ה מְסֻתָּֽרֶת׃ (נ) נֶ֭אֱמָנִים פִּצְעֵ֣י אוֹהֵ֑ב וְ֝נַעְתָּר֗וֹת נְשִׁיק֥וֹת שׂוֹנֵֽא׃ (נא) נֶ֣פֶשׁ שְׂ֭בֵעָה תָּב֣וּס נֹ֑פֶת וְנֶ֥פֶשׁ רְ֝עֵבָ֗ה כָּל־ מַ֥ר מָתֽוֹק׃ (נב) כְּ֭צִפּוֹר נוֹדֶ֣דֶת מִן־ קִנָּ֑הּ כֵּֽן־ אִ֝֗ישׁ נוֹדֵ֥ד מִמְּקוֹמֽוֹ׃ (נג) שֶׁ֣מֶן וּ֭קְטֹרֶת יְשַׂמַּֽח־ לֵ֑ב וּמֶ֥תֶק רֵ֝עֵ֗הוּ מֵֽעֲצַת־ נָֽפֶשׁ׃ (נד) רֵֽעֲךָ֨ (ורעה) וְרֵ֪עַ אָבִ֡יךָ אַֽל־ תַּעֲזֹ֗ב וּבֵ֥ית אָחִ֗יךָ אַל־ תָּ֭בוֹא בְּי֣וֹם אֵידֶ֑ךָ ט֥וֹב שָׁכֵ֥ן קָ֝ר֗וֹב מֵאָ֥ח רָחֽוֹק׃ (נה) חֲכַ֣ם בְּ֭נִי וְשַׂמַּ֣ח לִבִּ֑י וְאָשִׁ֖יבָה חֹרְפִ֣י דָבָֽר׃ (נו) עָר֤וּם רָאָ֣ה רָעָ֣ה נִסְתָּ֑ר פְּ֝תָאיִ֗ם עָבְר֥וּ נֶעֱנָֽשׁוּ׃ (נז) קַח־ בִּ֭גְדוֹ כִּי־ עָ֣רַב זָ֑ר וּבְעַ֖ד נָכְרִיָּ֣ה חַבְלֵֽהוּ׃ (נח) מְבָ֘רֵ֤ךְ רֵעֵ֨הוּ ׀ בְּק֣וֹל גָּ֭דוֹל בַּבֹּ֣קֶר הַשְׁכֵּ֑ים קְ֝לָלָ֗ה תֵּחָ֥שֶׁב לֽוֹ׃ (נט) דֶּ֣לֶף ט֭וֹרֵד בְּי֣וֹם סַגְרִ֑יר וְאֵ֥שֶׁת (מדונים) מִ֝דְיָנִ֗ים נִשְׁתָּוָֽה׃ (ס) צֹפְנֶ֥יהָ צָֽפַן־ ר֑וּחַ וְשֶׁ֖מֶן יְמִינ֣וֹ יִקְרָֽא׃ (סא) בַּרְזֶ֣ל בְּבַרְזֶ֣ל יָ֑חַד וְ֝אִ֗ישׁ יַ֣חַד פְּנֵֽי־ רֵעֵֽהוּ׃ (סב) נֹצֵ֣ר תְּ֭אֵנָה יֹאכַ֣ל פִּרְיָ֑הּ וְשֹׁמֵ֖ר אֲדֹנָ֣יו יְכֻבָּֽד׃ (סג) כַּ֭מַּיִם הַפָּנִ֣ים לַפָּנִ֑ים כֵּ֤ן לֵֽב־ הָ֝אָדָ֗ם לָאָדָֽם׃ (סד) שְׁא֣וֹל (ואבדה) וַ֭אֲבַדּוֹ לֹ֣א תִשְׂבַּ֑עְנָה וְעֵינֵ֥י הָ֝אָדָ֗ם לֹ֣א תִשְׂבַּֽעְנָה׃ (סה) מַצְרֵ֣ף לַ֭כֶּסֶף וְכ֣וּר לַזָּהָ֑ב וְ֝אִ֗ישׁ לְפִ֣י מַהֲלָלֽוֹ׃ (סו) אִ֥ם תִּכְתּֽוֹשׁ־ אֶת־ הָאֱוִ֨יל ׀ בַּֽמַּכְתֵּ֡שׁ בְּת֣וֹךְ הָ֭רִיפוֹת בַּֽעֱלִ֑י לֹא־ תָס֥וּר מֵ֝עָלָ֗יו אִוַּלְתּֽוֹ׃ (סז) יָדֹ֣עַ תֵּ֭דַע פְּנֵ֣י צֹאנֶ֑ךָ שִׁ֥ית לִ֝בְּךָ֗ לַעֲדָרִֽים׃ (סח) כִּ֤י לֹ֣א לְעוֹלָ֣ם חֹ֑סֶן וְאִם־ נֵ֝֗זֶר לְד֣וֹר (דור) וָדֽוֹר׃ (סט) גָּלָ֣ה חָ֭צִיר וְנִרְאָה־ דֶ֑שֶׁא וְ֝נֶאֶסְפ֗וּ עִשְּׂב֥וֹת הָרִֽים׃ (ע) כְּבָשִׂ֥ים לִלְבוּשֶׁ֑ךָ וּמְחִ֥יר שָׂ֝דֶ֗ה עַתּוּדִֽים׃ (עא) וְדֵ֤י ׀ חֲלֵ֬ב עִזִּ֗ים לְֽ֭לַחְמְךָ לְלֶ֣חֶם בֵּיתֶ֑ךָ וְ֝חַיִּ֗ים לְנַעֲרוֹתֶֽיךָ׃ סוף פרק כז (עב) נָ֣סוּ וְאֵין־ רֹדֵ֣ף רָשָׁ֑ע וְ֝צַדִּיקִ֗ים כִּכְפִ֥יר יִבְטָֽח׃ (עג) בְּפֶ֣שַֽׁע אֶ֭רֶץ רַבִּ֣ים שָׂרֶ֑יהָ וּבְאָדָ֥ם מֵבִ֥ין יֹ֝דֵ֗עַ כֵּ֣ן יַאֲרִֽיךְ׃ (עד) גֶּ֣בֶר רָ֭שׁ וְעֹשֵׁ֣ק דַּלִּ֑ים מָטָ֥ר סֹ֝חֵ֗ף וְאֵ֣ין לָֽחֶם׃ (עה) עֹזְבֵ֣י ת֭וֹרָה יְהַֽלְל֣וּ רָשָׁ֑ע וְשֹׁמְרֵ֥י ת֝וֹרָ֗ה יִתְגָּ֥רוּ בָֽם׃ (עו) אַנְשֵׁי־ רָ֭ע לֹא־ יָבִ֣ינוּ מִשְׁפָּ֑ט וּמְבַקְשֵׁ֥י יְ֝הוָ֗ה יָבִ֥ינוּ כֹֽל׃ (עז) טֽוֹב־ רָ֭שׁ הוֹלֵ֣ךְ בְּתֻמּ֑וֹ מֵעִקֵּ֥שׁ דְּ֝רָכַ֗יִם וְה֣וּא עָשִֽׁיר׃ (עח) נוֹצֵ֣ר תּ֭וֹרָה בֵּ֣ן מֵבִ֑ין וְרֹעֶה זֽ֝וֹלְלִ֗ים יַכְלִ֥ים אָבִֽיו׃ (עט) מַרְבֶּ֣ה ה֭וֹנוֹ בְּנֶ֣שֶׁךְ (ובתרבית) וְתַרְבִּ֑ית לְחוֹנֵ֖ן דַּלִּ֣ים יִקְבְּצֶֽנּוּ׃ (פ) מֵסִ֣יר אָ֭זְנוֹ מִשְּׁמֹ֣עַ תּוֹרָ֑ה גַּֽם־ תְּ֝פִלָּת֗וֹ תּוֹעֵבָֽה׃ (פא) מַשְׁגֶּ֤ה יְשָׁרִ֨ים ׀ בְּדֶ֥רֶךְ רָ֗ע בִּשְׁחוּת֥וֹ הֽוּא־ יִפּ֑וֹל וּ֝תְמִימִ֗ים יִנְחֲלוּ־ טֽוֹב׃ (פב) חָכָ֣ם בְּ֭עֵינָיו אִ֣ישׁ עָשִׁ֑יר וְדַ֖ל מֵבִ֣ין יַחְקְרֶֽנּוּ׃ (פג) בַּעֲלֹ֣ץ צַ֭דִּיקִים רַבָּ֣ה תִפְאָ֑רֶת וּבְק֥וּם רְ֝שָׁעִ֗ים יְחֻפַּ֥שׂ אָדָֽם׃ (פד) מְכַסֶּ֣ה פְ֭שָׁעָיו לֹ֣א יַצְלִ֑יחַ וּמוֹדֶ֖ה וְעֹזֵ֣ב יְרֻחָֽם׃ (פה) אַשְׁרֵ֣י אָ֭דָם מְפַחֵ֣ד תָּמִ֑יד וּמַקְשֶׁ֥ה לִ֝בּ֗וֹ יִפּ֥וֹל בְּרָעָֽה׃ (פו) אֲרִי־ נֹ֭הֵם וְדֹ֣ב שׁוֹקֵ֑ק מֹשֵׁ֥ל רָ֝שָׁ֗ע עַ֣ל עַם־ דָּֽל׃
