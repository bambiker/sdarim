משלי פרק יח לפי החלוקה לסדרים

פרק יח בספר משלי (24 פסוקים) נמצא במשלי סדר ד (פסוקים קכז–קלה) ובמשלי סדר ה (פסוקים א–טו).

משלי סדר ד

(א) בְּאֵ֣ין אֲ֭לָפִים אֵב֣וּס בָּ֑ר וְרָב־ תְּ֝בוּא֗וֹת בְּכֹ֣חַ שֽׁוֹר׃ (ב) עֵ֣ד אֱ֭מוּנִים לֹ֣א יְכַזֵּ֑ב וְיָפִ֥יחַ כְּ֝זָבִ֗ים עֵ֣ד שָֽׁקֶר׃ (ג) בִּקֶּשׁ־ לֵ֣ץ חָכְמָ֣ה וָאָ֑יִן וְדַ֖עַת לְנָב֣וֹן נָקָֽל׃ (ד) לֵ֣ךְ מִ֭נֶּגֶד לְאִ֣ישׁ כְּסִ֑יל וּבַל־ יָ֝דַ֗עְתָּ שִׂפְתֵי־ דָֽעַת׃ (ה) חָכְמַ֣ת עָ֭רוּם הָבִ֣ין דַּרְכּ֑וֹ וְאִוֶּ֖לֶת כְּסִילִ֣ים מִרְמָֽה׃ (ו) אֱ֭וִלִים יָלִ֣יץ אָשָׁ֑ם וּבֵ֖ין יְשָׁרִ֣ים רָצֽוֹן׃ (ז) לֵ֗ב י֭וֹדֵעַ מָרַּ֣ת נַפְשׁ֑וֹ וּ֝בְשִׂמְחָת֗וֹ לֹא־ יִתְעָ֥רַב זָֽר׃ (ח) בֵּ֣ית רְ֭שָׁעִים יִשָּׁמֵ֑ד וְאֹ֖הֶל יְשָׁרִ֣ים יַפְרִֽיחַ׃ (ט) יֵ֤שׁ דֶּ֣רֶךְ יָ֭שָׁר לִפְנֵי־ אִ֑ישׁ וְ֝אַחֲרִיתָ֗הּ דַּרְכֵי־ מָֽוֶת׃ (י) גַּם־ בִּשְׂח֥וֹק יִכְאַב־ לֵ֑ב וְאַחֲרִיתָ֖הּ שִׂמְחָ֣ה תוּגָֽה׃ (יא) מִדְּרָכָ֣יו יִ֭שְׂבַּע ס֣וּג לֵ֑ב וּ֝מֵעָלָ֗יו אִ֣ישׁ טֽוֹב׃ (יב) פֶּ֭תִי יַאֲמִ֣ין לְכָל־ דָּבָ֑ר וְ֝עָר֗וּם יָבִ֥ין לַאֲשֻׁרֽוֹ׃ (יג) חָכָ֣ם יָ֭רֵא וְסָ֣ר מֵרָ֑ע וּ֝כְסִ֗יל מִתְעַבֵּ֥ר וּבוֹטֵֽחַ׃ (יד) קְֽצַר־ אַ֭פַּיִם יַעֲשֶׂ֣ה אִוֶּ֑לֶת וְאִ֥ישׁ מְ֝זִמּ֗וֹת יִשָּׂנֵֽא׃ (טו) נָחֲל֣וּ פְתָאיִ֣ם אִוֶּ֑לֶת וַֽ֝עֲרוּמִ֗ים יַכְתִּ֥רוּ דָֽעַת׃ (טז) שַׁח֣וּ רָ֭עִים לִפְנֵ֣י טוֹבִ֑ים וּ֝רְשָׁעִ֗ים עַֽל־ שַׁעֲרֵ֥י צַדִּֽיק׃ (יז) גַּם־ לְ֭רֵעֵהוּ יִשָּׂ֣נֵא רָ֑שׁ וְאֹהֲבֵ֖י עָשִׁ֣יר רַבִּֽים׃ (יח) בָּז־ לְרֵעֵ֥הוּ חוֹטֵ֑א וּמְחוֹנֵ֖ן (עניים) עֲנָוִ֣ים אַשְׁרָֽיו׃ (יט) הֲ‍ֽלוֹא־ יִ֭תְעוּ חֹ֣רְשֵׁי רָ֑ע וְחֶ֥סֶד וֶ֝אֱמֶ֗ת חֹ֣רְשֵׁי טֽוֹב׃ (כ) בְּכָל־ עֶ֭צֶב יִהְיֶ֣ה מוֹתָ֑ר וּדְבַר־ שְׂ֝פָתַ֗יִם אַךְ־ לְמַחְסֽוֹר׃ (כא) עֲטֶ֣רֶת חֲכָמִ֣ים עָשְׁרָ֑ם אִוֶּ֖לֶת כְּסִילִ֣ים אִוֶּֽלֶת׃ (כב) מַצִּ֣יל נְ֭פָשׁוֹת עֵ֣ד אֱמֶ֑ת וְיָפִ֖חַ כְּזָבִ֣ים מִרְמָֽה׃ (כג) בְּיִרְאַ֣ת יְ֭הוָה מִבְטַח־ עֹ֑ז וּ֝לְבָנָ֗יו יִהְיֶ֥ה מַחְסֶֽה׃ (כד) יִרְאַ֣ת יְ֭הוָה מְק֣וֹר חַיִּ֑ים לָ֝ס֗וּר מִמֹּ֥קְשֵׁי מָֽוֶת׃ (כה) בְּרָב־ עָ֥ם הַדְרַת־ מֶ֑לֶךְ וּבְאֶ֥פֶס לְ֝אֹ֗ם מְחִתַּ֥ת רָזֽוֹן׃ (כו) אֶ֣רֶךְ אַ֭פַּיִם רַב־ תְּבוּנָ֑ה וּקְצַר־ ר֝֗וּחַ מֵרִ֥ים אִוֶּֽלֶת׃ (כז) חַיֵּ֣י בְ֭שָׂרִים לֵ֣ב מַרְפֵּ֑א וּרְקַ֖ב עֲצָמ֣וֹת קִנְאָֽה׃ (כח) עֹ֣שֵֽׁק דָּ֭ל חֵרֵ֣ף עֹשֵׂ֑הוּ וּ֝מְכַבְּד֗וֹ חֹנֵ֥ן אֶבְיֽוֹן׃ (כט) בְּֽ֭רָעָתוֹ יִדָּחֶ֣ה רָשָׁ֑ע וְחֹסֶ֖ה בְמוֹת֣וֹ צַדִּֽיק׃ (ל) בְּלֵ֣ב נָ֭בוֹן תָּנ֣וּחַ חָכְמָ֑ה וּבְקֶ֥רֶב כְּ֝סִילִ֗ים תִּוָּדֵֽעַ׃ (לא) צְדָקָ֥ה תְרֽוֹמֵֽם־ גּ֑וֹי וְחֶ֖סֶד לְאֻמִּ֣ים חַטָּֽאת׃ (לב) רְֽצוֹן־ מֶ֭לֶךְ לְעֶ֣בֶד מַשְׂכִּ֑יל וְ֝עֶבְרָת֗וֹ תִּהְיֶ֥ה מֵבִֽישׁ׃ (לג) מַֽעֲנֶה־ רַּ֭ךְ יָשִׁ֣יב חֵמָ֑ה וּדְבַר־ עֶ֝֗צֶב יַעֲלֶה־ אָֽף׃ (לד) לְשׁ֣וֹן חֲ֭כָמִים תֵּיטִ֣יב דָּ֑עַת וּפִ֥י כְ֝סִילִ֗ים יַבִּ֥יעַ אִוֶּֽלֶת׃ (לה) בְּֽכָל־ מָ֭קוֹם עֵינֵ֣י יְהוָ֑ה צֹ֝פ֗וֹת רָעִ֥ים וטוֹבִֽים׃ (לו) מַרְפֵּ֣א לָ֭שׁוֹן עֵ֣ץ חַיִּ֑ים וְסֶ֥לֶף בָּ֝֗הּ שֶׁ֣בֶר בְּרֽוּחַ׃ (לז) אֱוִ֗יל יִ֭נְאַץ מוּסַ֣ר אָבִ֑יו וְשֹׁמֵ֖ר תּוֹכַ֣חַת יַעְרִֽם׃ (לח) בֵּ֣ית צַ֭דִּיק חֹ֣סֶן רָ֑ב וּבִתְבוּאַ֖ת רָשָׁ֣ע נֶעְכָּֽרֶת׃ (לט) שִׂפְתֵ֣י חֲ֭כָמִים יְזָ֣רוּ דָ֑עַת וְלֵ֖ב כְּסִילִ֣ים לֹא־ כֵֽן׃ (מ) זֶ֣בַח רְ֭שָׁעִים תּוֹעֲבַ֣ת יְהוָ֑ה וּתְפִלַּ֖ת יְשָׁרִ֣ים רְצוֹנֽוֹ׃ (מא) תּוֹעֲבַ֣ת יְ֭הוָה דֶּ֣רֶךְ רָשָׁ֑ע וּמְרַדֵּ֖ף צְדָקָ֣ה יֶאֱהָֽב׃ (מב) מוּסָ֣ר רָ֭ע לְעֹזֵ֣ב אֹ֑רַח שׂוֹנֵ֖א תוֹכַ֣חַת יָמֽוּת׃ (מג) שְׁא֣וֹל וַ֭אֲבַדּוֹן נֶ֣גֶד יְהוָ֑ה אַ֝֗ף כִּֽי־ לִבּ֥וֹת בְּֽנֵי־ אָדָֽם׃ (מד) לֹ֣א יֶאֱהַב־ לֵ֭ץ הוֹכֵ֣חַֽ ל֑וֹ אֶל־ חֲ֝כָמִ֗ים לֹ֣א יֵלֵֽךְ׃ (מה) לֵ֣ב שָׂ֭מֵחַ יֵיטִ֣ב פָּנִ֑ים וּבְעַצְּבַת־ לֵ֝ב ר֣וּחַ נְכֵאָֽה׃ (מו) לֵ֣ב נָ֭בוֹן יְבַקֶּשׁ־ דָּ֑עַת (ופני) וּפִ֥י כְ֝סִילִ֗ים יִרְעֶ֥ה אִוֶּֽלֶת׃ (מז) כָּל־ יְמֵ֣י עָנִ֣י רָעִ֑ים וְטֽוֹב־ לֵ֝֗ב מִשְׁתֶּ֥ה תָמִֽיד׃ (מח) טוֹב־ מְ֭עַט בְּיִרְאַ֣ת יְהוָ֑ה מֵאוֹצָ֥ר רָ֝֗ב וּמְה֥וּמָה בֽוֹ׃ (מט) ט֤וֹב אֲרֻחַ֣ת יָ֭רָק וְאַהֲבָה־ שָׁ֑ם מִשּׁ֥וֹר אָ֝ב֗וּס וְשִׂנְאָה־ בֽוֹ׃ (נ) אִ֣ישׁ חֵ֭מָה יְגָרֶ֣ה מָד֑וֹן וְאֶ֥רֶך אַ֝פַּ֗יִם יַשְׁקִ֥יט רִֽיב׃ (נא) דֶּ֣רֶךְ עָ֭צֵל כִּמְשֻׂ֣כַת חָ֑דֶק וְאֹ֖רַח יְשָׁרִ֣ים סְלֻלָֽה׃ (נב) בֵּ֣ן חָ֭כָם יְשַׂמַּח־ אָ֑ב וּכְסִ֥יל אָ֝דָ֗ם בּוֹזֶ֥ה אִמּֽוֹ׃ (נג) אִ֭וֶּלֶת שִׂמְחָ֣ה לַחֲסַר־ לֵ֑ב וְאִ֥ישׁ תְּ֝בוּנָ֗ה יְיַשֶׁר־ לָֽכֶת׃ (נד) הָפֵ֣ר מַ֭חֲשָׁבוֹת בְּאֵ֣ין ס֑וֹד וּבְרֹ֖ב יוֹעֲצִ֣ים תָּקֽוּם׃ (נה) שִׂמְחָ֣ה לָ֭אִישׁ בְּמַעֲנֵה־ פִ֑יו וְדָבָ֖ר בְּעִתּ֣וֹ מַה־ טּֽוֹב׃ (נו) אֹ֣רַח חַ֭יִּים לְמַ֣עְלָה לְמַשְׂכִּ֑יל לְמַ֥עַן ס֝֗וּר מִשְּׁא֥וֹל מָֽטָּה׃ (נז) בֵּ֣ית גֵּ֭אִים יִסַּ֥ח ׀ יְהוָ֑ה וְ֝יַצֵּ֗ב גְּב֣וּל אַלְמָנָֽה׃ (נח) תּוֹעֲבַ֣ת יְ֭הוָה מַחְשְׁב֣וֹת רָ֑ע וּ֝טְהֹרִ֗ים אִמְרֵי־ נֹֽעַם׃ (נט) עֹכֵ֣ר בֵּ֭יתוֹ בּוֹצֵ֣עַ בָּ֑צַע וְשׂוֹנֵ֖א מַתָּנֹ֣ת יִחְיֶֽה׃ (ס) לֵ֣ב צַ֭דִּיק יֶהְגֶּ֣ה לַעֲנ֑וֹת וּפִ֥י רְ֝שָׁעִ֗ים יַבִּ֥יעַ רָעֽוֹת׃ (סא) רָח֣וֹק יְ֭הוָה מֵרְשָׁעִ֑ים וּתְפִלַּ֖ת צַדִּיקִ֣ים יִשְׁמָֽע׃ (סב) מְֽאוֹר־ עֵ֭ינַיִם יְשַׂמַּֽח־ לֵ֑ב שְׁמוּעָ֥ה ט֝וֹבָ֗ה תְּדַשֶּׁן־ עָֽצֶם׃ (סג) אֹ֗זֶן שֹׁ֭מַעַת תּוֹכַ֣חַת חַיִּ֑ים בְּקֶ֖רֶב חֲכָמִ֣ים תָּלִֽין׃ (סד) פּוֹרֵ֣עַ מ֭וּסָר מוֹאֵ֣ס נַפְשׁ֑וֹ וְשׁוֹמֵ֥עַ תּ֝וֹכַ֗חַת ק֣וֹנֶה לֵּֽב׃ (סה) יִרְאַ֣ת יְ֭הוָה מוּסַ֣ר חָכְמָ֑ה וְלִפְנֵ֖י כָב֣וֹד עֲנָוָֽה׃ (סו) לְאָדָ֥ם מַֽעַרְכֵי־ לֵ֑ב וּ֝מֵיְהוָ֗ה מַעֲנֵ֥ה לָשֽׁוֹן׃ (סז) כָּֽל־ דַּרְכֵי־ אִ֭ישׁ זַ֣ךְ בְּעֵינָ֑יו וְתֹכֵ֖ן רוּח֣וֹת יְהוָֽה׃ (סח) גֹּ֣ל אֶל־ יְהוָ֣ה מַעֲשֶׂ֑יךָ וְ֝יִכֹּ֗נוּ מַחְשְׁבֹתֶֽיךָ׃ (סט) כֹּ֤ל פָּעַ֣ל יְ֭הוָה לַֽמַּעֲנֵ֑הוּ וְגַם־ רָ֝שָׁ֗ע לְי֣וֹם רָעָֽה׃ (ע) תּוֹעֲבַ֣ת יְ֭הוָה כָּל־ גְּבַהּ־ לֵ֑ב יָ֥ד לְ֝יָ֗ד לֹ֣א יִנָּקֶֽה׃ (עא) בְּחֶ֣סֶד וֶ֭אֱמֶת יְכֻפַּ֣ר עָוֺ֑ן וּבְיִרְאַ֥ת יְ֝הוָ֗ה ס֣וּר מֵרָֽע׃ (עב) בִּרְצ֣וֹת יְ֭הוָה דַּרְכֵי־ אִ֑ישׁ גַּם־ א֝וֹיְבָ֗יו יַשְׁלִ֥ם אִתּֽוֹ׃ (עג) טוֹב־ מְ֭עַט בִּצְדָקָ֑ה מֵרֹ֥ב תְּ֝בוּא֗וֹת בְּלֹ֣א מִשְׁפָּֽט׃ (עד) לֵ֣ב אָ֭דָם יְחַשֵּׁ֣ב דַּרְכּ֑וֹ וַֽ֝יהוָ֗ה יָכִ֥ין צַעֲדֽוֹ׃ (עה) כִּ֥י בִ֝צְדָקָ֗ה יִכּ֥וֹן כִּסֵּֽא׃ (עח) רְצ֣וֹן מְ֭לָכִים שִׂפְתֵי־ צֶ֑דֶק וְדֹבֵ֖ר יְשָׁרִ֣ים יֶאֱהָֽב׃ (עט) חֲמַת־ מֶ֥לֶךְ מַלְאֲכֵי־ מָ֑וֶת וְאִ֖ישׁ חָכָ֣ם יְכַפְּרֶֽנָּה׃ (פ) בְּאוֹר־ פְּנֵי־ וְלֹ֣א יֵעָנֶֽה׃ (פח) מַתָּ֣ן בַּ֭סֵּתֶר יִכְפֶּה־ אָ֑ף וְשֹׁ֥חַד בַּ֝חֵ֗ק חֵמָ֥ה עַזָּֽה׃ (פט) שִׂמְחָ֣ה לַ֭צַּדִּיק עֲשׂ֣וֹת מִשְׁפָּ֑ט וּ֝מְחִתָּ֗ה לְפֹ֣עֲלֵי אָֽוֶן׃ (צ) אָדָ֗ם תּ֭וֹעֶה מִדֶּ֣רֶךְ הַשְׂכֵּ֑ל בִּקְהַ֖ל רְפָאִ֣ים יָנֽוּחַ׃ (צא) אִ֣ישׁ מַ֭חְסוֹר אֹהֵ֣ב שִׂמְחָ֑ה אֹהֵ֥ב יַֽיִן־ וָ֝שֶׁ֗מֶן לֹ֣א יַעֲשִֽׁיר׃ (צב) כֹּ֣פֶר לַצַּדִּ֣יק רָשָׁ֑ע וְתַ֖חַת יְשָׁרִ֣ים בּוֹגֵֽד׃ (צג) ט֗וֹב שֶׁ֥בֶת בְּאֶֽרֶץ־ מִדְבָּ֑ר מֵאֵ֖שֶׁת (מדונים) מִדְיָנִ֣ים וָכָֽעַס׃ (צד) אוֹצָ֤ר ׀ נֶחְמָ֣ד וָ֭שֶׁמֶן בִּנְוֵ֣ה חָכָ֑ם וּכְסִ֖יל אָדָ֣ם יְבַלְּעֶֽנּוּ׃ (צה) רֹ֭דֵף צְדָקָ֣ה וָחָ֑סֶד יִמְצָ֥א חַ֝יִּ֗ים צְדָקָ֥ה וְכָבֽוֹד׃ (צו) עִ֣יר גִּ֭בֹּרִים עָלָ֣ה חָכָ֑ם וַ֝יֹּ֗רֶד עֹ֣ז מִבְטֶחָֽה׃ (צז) שֹׁמֵ֣ר פִּ֭יו וּלְשׁוֹנ֑וֹ שֹׁמֵ֖ר מִצָּר֣וֹת נַפְשֽׁוֹ׃ (צח) זֵ֣ד יָ֭הִיר לֵ֣ץ שְׁמ֑וֹ ע֝וֹשֶׂ֗ה בְּעֶבְרַ֥ת זָדֽוֹן׃ (צט) תַּאֲוַ֣ת עָצֵ֣ל תְּמִיתֶ֑נּוּ כִּֽי־ מֵאֲנ֖וּ יָדָ֣יו לַעֲשֽׂוֹת׃ (ק) כָּל־ הַ֭יּוֹם הִתְאַוָּ֣ה תַאֲוָ֑ה וְצַדִּ֥יק יִ֝תֵּ֗ן וְלֹ֣א יַחְשֹֽׂךְ׃ (קא) זֶ֣בַח רְ֭שָׁעִים תּוֹעֵבָ֑ה אַ֝֗ף כִּֽי־ בְזִמָּ֥ה יְבִיאֶֽנּוּ׃ (קב) עֵד־ כְּזָבִ֥ים יֹאבֵ֑ד וְאִ֥ישׁ שׁ֝וֹמֵ֗עַ לָנֶ֥צַח יְדַבֵּֽר׃ (קג) הֵעֵ֬ז אִ֣ישׁ רָשָׁ֣ע בְּפָנָ֑יו וְ֝יָשָׁ֗ר ה֤וּא ׀ (יכין) יָבִ֬ין (דרכיו) דַּרְכּֽוֹ׃ (קד) אֵ֣ין חָ֭כְמָה וְאֵ֣ין תְּבוּנָ֑ה וְאֵ֥ין עֵ֝צָ֗ה לְנֶ֣גֶד יְהוָֽה׃ (קה) ס֗וּס מ֭וּכָן לְי֣וֹם מִלְחָמָ֑ה וְ֝לַֽיהוָ֗ה הַתְּשׁוּעָֽה׃ (קו) נִבְחָ֣ר שֵׁ֭ם מֵעֹ֣שֶׁר רָ֑ב מִכֶּ֥סֶף וּ֝מִזָּהָ֗ב חֵ֣ן טֽוֹב׃ (קז) עָשִׁ֣יר וָרָ֣שׁ נִפְגָּ֑שׁוּ עֹשֵׂ֖ה כֻלָּ֣ם יְהוָֽה׃ (קח) עָר֤וּם ׀ רָאָ֣ה רָעָ֣ה (ויסתר) וְנִסְתָּ֑ר וּ֝פְתָיִ֗ים עָבְר֥וּ וְֽנֶעֱנָֽשׁוּ׃ (קט) עֵ֣קֶב עֲ֭נָוָה יִרְאַ֣ת יְהוָ֑ה עֹ֖שֶׁר וְכָב֣וֹד וְחַיִּֽים׃ (קי) צִנִּ֣ים פַּ֭חִים בְּדֶ֣רֶךְ עִקֵּ֑שׁ שׁוֹמֵ֥ר נַ֝פְשׁ֗וֹ יִרְחַ֥ק מֵהֶֽם׃ (קיא) חֲנֹ֣ךְ לַ֭נַּעַר עַל־ פִּ֣י דַרְכּ֑וֹ גַּ֥ם כִּֽי־ יַ֝זְקִ֗ין לֹֽא־ יָס֥וּר מִמֶּֽנָּה׃ (קיב) עָ֭שִׁיר בְּרָשִׁ֣ים יִמְשׁ֑וֹל וְעֶ֥בֶד לֹ֝וֶ֗ה לְאִ֣ישׁ מַלְוֶֽה׃ (קיג) זוֹרֵ֣עַ עַ֭וְלָה (יקצור)־ יִקְצָר־ אָ֑וֶן וְשֵׁ֖בֶט עֶבְרָת֣וֹ יִכְלֶֽה׃ (קיד) טֽוֹב־ עַ֭יִן ה֣וּא יְבֹרָ֑ךְ כִּֽי־ נָתַ֖ן מִלַּחְמ֣וֹ לַדָּֽל׃ (קטו) גָּ֣רֵֽשׁ לֵ֭ץ וְיֵצֵ֣א מָד֑וֹן וְ֝יִשְׁבֹּ֗ת דִּ֣ין וְקָלֽוֹן׃ (קטז) אֹהֵ֥ב (טהור)־ טְהָר־ לֵ֑ב חֵ֥ן שְׂ֝פָתָ֗יו רֵעֵ֥הוּ מֶֽלֶךְ׃ (קיז) עֵינֵ֣י יְ֭הוָה נָ֣צְרוּ דָ֑עַת וַ֝יְסַלֵּ֗ף דִּבְרֵ֥י בֹגֵֽד׃ (קיח) אָמַ֣ר עָ֭צֵל אֲרִ֣י בַח֑וּץ בְּת֥וֹךְ רְ֝חֹב֗וֹת אֵֽרָצֵֽחַ׃ (קיט) שׁוּחָ֣ה עֲ֭מֻקָּה פִּ֣י זָר֑וֹת זְע֥וּם יְ֝הוָ֗ה (יפול)־ יִפָּל־ שָֽׁם׃ (קכ) אִ֭וֶּלֶת קְשׁוּרָ֣ה בְלֶב־ נָ֑עַר שֵׁ֥בֶט מ֝וּסָ֗ר יַרְחִיקֶ֥נָּה מִמֶּֽנּוּ׃ (קכא) עֹ֣שֵֽׁק דָּ֭ל לְהַרְבּ֣וֹת ל֑וֹ נֹתֵ֥ן לְ֝עָשִׁ֗יר אַךְ־ לְמַחְסֽוֹר׃ (קכב) הַ֥ט אָזְנְךָ֗ וּ֭שְׁמַע דִּבְרֵ֣י חֲכָמִ֑ים וְ֝לִבְּךָ֗ תָּשִׁ֥ית לְדַעְתִּֽי׃ (קכג) כִּֽי־ נָ֭עִים כִּֽי־ תִשְׁמְרֵ֣ם בְּבִטְנֶ֑ךָ יִכֹּ֥נוּ יַ֝חְדָּ֗ו עַל־ שְׂפָתֶֽיךָ׃ (קכד) לִהְי֣וֹת בַּ֭יהוָה מִבְטַחֶ֑ךָ הוֹדַעְתִּ֖יךָ הַיּ֣וֹם אַף־ אָֽתָּה׃ (קכה) הֲלֹ֤א כָתַ֣בְתִּי לְ֭ךָ (שלשום) שָׁלִישִׁ֑ים בְּמ֖וֹעֵצֹ֣ת וָדָֽעַת׃
