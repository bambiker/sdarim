משלי פרק כה לפי החלוקה לסדרים

פרק כה בספר משלי (28 פסוקים) נמצא במשלי סדר ו (פסוקים עט–צ) ובמשלי סדר ז (פסוקים א–טז).

משלי סדר ו

(א) לְהוֹדִֽיעֲךָ֗ קֹ֭שְׁטְ אִמְרֵ֣י אֱמֶ֑ת לְהָשִׁ֥יב אֲמָרִ֥ים אֱ֝מֶ֗ת לְשֹׁלְחֶֽיךָ׃ (ב) אַֽל־ תִּגְזָל־ דָּ֭ל כִּ֣י דַל־ ה֑וּא וְאַל־ תְּדַכֵּ֖א עָנִ֣י בַשָּֽׁעַר׃ (ג) כִּֽי־ יְ֭הוָה יָרִ֣יב רִיבָ֑ם וְקָבַ֖ע אֶת־ קֹבְעֵיהֶ֣ם נָֽפֶשׁ׃ (ד) אַל־ תִּ֭תְרַע אֶת־ בַּ֣עַל אָ֑ף וְאֶת־ אִ֥ישׁ חֵ֝מוֹת לֹ֣א תָבֽוֹא׃ (ה) פֶּן־ תֶּאֱלַ֥ף (ארחתו) אֹֽרְחֹתָ֑יו וְלָקַחְתָּ֖ מוֹקֵ֣שׁ לְנַפְשֶֽׁךָ׃ (ו) אַל־ תְּהִ֥י בְתֹֽקְעֵי־ כָ֑ף בַּ֝עֹרְבִ֗ים מַשָּׁאֽוֹת׃ (ז) אִם־ אֵֽין־ לְךָ֥ לְשַׁלֵּ֑ם לָ֥מָּה יִקַּ֥ח מִ֝שְׁכָּבְךָ֗ מִתַּחְתֶּֽיךָ׃ (ח) אַל־ תַּ֭סֵּג גְּב֣וּל עוֹלָ֑ם אֲשֶׁ֖ר עָשׂ֣וּ אֲבוֹתֶֽיךָ׃ (ט) חָזִ֡יתָ אִ֤ישׁ ׀ מָ֘הִ֤יר בִּמְלַאכְתּ֗וֹ לִֽפְנֵֽי־ מְלָכִ֥ים יִתְיַצָּ֑ב בַּל־ יִ֝תְיַצֵּב לִפְנֵ֥י חֲשֻׁכִּֽים׃ (י) כִּֽי־ תֵ֭שֵׁב לִלְח֣וֹם אֶת־ מוֹשֵׁ֑ל בִּ֥ין תָּ֝בִ֗ין אֶת־ אֲשֶׁ֥ר לְפָנֶֽיךָ׃ (יא) וְשַׂמְתָּ֣ שַׂכִּ֣ין בְּלֹעֶ֑ךָ אִם־ בַּ֖עַל נֶ֣פֶשׁ אָֽתָּה׃ (יב) אַל־ תִּ֭תְאָו לְמַטְעַמּוֹתָ֑יו וְ֝ה֗וּא לֶ֣חֶם כְּזָבִֽים׃ (יג) אַל־ תִּיגַ֥ע לְֽהַעֲשִׁ֑יר מִֽבִּינָתְךָ֥ חֲדָֽל׃ (יד) (התעוף) הֲתָ֤עִיף עֵינֶ֥יךָ בּ֗וֹ וְֽאֵ֫ינֶ֥נּוּ כִּ֤י עָשֹׂ֣ה יַעֲשֶׂה־ לּ֣וֹ כְנָפַ֑יִם כְּ֝נֶ֗שֶׁר (ועיף) יָע֥וּף הַשָּׁמָֽיִם׃ (טו) אַל־ תִּלְחַ֗ם אֶת־ לֶ֭חֶם רַ֣ע עָ֑יִן וְאַל־ (תתאו) תִּ֝תְאָ֗יו לְמַטְעַמֹּתָֽיו׃ (טז) כִּ֤י ׀ כְּמוֹ־ שָׁעַ֥ר בְּנַפְשׁ֗וֹ כֶּ֫ן־ ה֥וּא אֱכֹ֣ל וּ֭שְׁתֵה יֹ֣אמַר לָ֑ךְ וְ֝לִבּ֗וֹ בַּל־ עִמָּֽךְ׃ (יז) פִּֽתְּךָ־ אָכַ֥לְתָּ תְקִיאֶ֑נָּה וְ֝שִׁחַ֗תָּ דְּבָרֶ֥יךָ הַנְּעִימִֽים׃ (יח) בְּאָזְנֵ֣י כְ֭סִיל אַל־ תְּדַבֵּ֑ר כִּֽי־ יָ֝ב֗וּז לְשֵׂ֣כֶל מִלֶּֽיךָ׃ (יט) אַל־ תַּ֭סֵּג גְּב֣וּל עוֹלָ֑ם וּבִשְׂדֵ֥י יְ֝תוֹמִ֗ים אַל־ תָּבֹֽא׃ (כ) כִּֽי־ גֹאֲלָ֥ם חָזָ֑ק הֽוּא־ יָרִ֖יב אֶת־ רִיבָ֣ם אִתָּֽךְ׃ (כא) הָבִ֣יאָה לַמּוּסָ֣ר לִבֶּ֑ךָ וְ֝אָזְנֶ֗ךָ לְאִמְרֵי־ דָֽעַת׃ (כב) אַל־ תִּמְנַ֣ע מִנַּ֣עַר מוּסָ֑ר כִּֽי־ תַכֶּ֥נּוּ בַ֝שֵּׁ֗בֶט לֹ֣א יָמֽוּת׃ (כג) אַ֭תָּה בַּשֵּׁ֣בֶט תַּכֶּ֑נּוּ וְ֝נַפְשׁ֗וֹ מִשְּׁא֥וֹל תַּצִּֽיל׃ (כד) בְּ֭נִי אִם־ חָכַ֣ם לִבֶּ֑ךָ יִשְׂמַ֖ח לִבִּ֣י גַם־ אָֽנִי׃ (כה) וְתַעְלֹ֥זְנָה כִלְיוֹתָ֑י בְּדַבֵּ֥ר שְׂ֝פָתֶ֗יךָ מֵישָׁרִֽים׃ (כו) אַל־ יְקַנֵּ֣א לִ֭בְּךָ בַּֽחַטָּאִ֑ים כִּ֥י אִם־ בְּיִרְאַת־ יְ֝הוָ֗ה כָּל־ הַיּֽוֹם׃ (כז) כִּ֭י אִם־ יֵ֣שׁ אַחֲרִ֑ית וְ֝תִקְוָתְךָ֗ לֹ֣א תִכָּרֵֽת׃ (כח) שְׁמַע־ אַתָּ֣ה בְנִ֣י וַחֲכָ֑ם וְאַשֵּׁ֖ר בַּדֶּ֣רֶךְ לִבֶּֽךָ׃ (כט) אַל־ תְּהִ֥י בְסֹֽבְאֵי־ יָ֑יִן בְּזֹלֲלֵ֖י בָשָׂ֣ר לָֽמוֹ׃ (ל) כִּי־ סֹבֵ֣א וְ֭זוֹלֵל יִוָּרֵ֑שׁ וּ֝קְרָעִ֗ים תַּלְבִּ֥ישׁ נוּמָֽה׃ (לא) שְׁמַ֣ע לְ֭אָבִיךָ זֶ֣ה יְלָדֶ֑ךָ וְאַל־ תָּ֝ב֗וּז כִּֽי־ זָקְנָ֥ה אִמֶּֽךָ׃ (לב) אֱמֶ֣ת קְ֭נֵה וְאַל־ תִּמְכֹּ֑ר חָכְמָ֖ה וּמוּסָ֣ר וּבִינָֽה׃ (לג) (גול) גִּ֣יל (יגול) יָ֭גִיל אֲבִ֣י צַדִּ֑יק (יולד) וְיוֹלֵ֥ד חָ֝כָ֗ם (וישמח)־ יִשְׂמַח־ בּֽוֹ׃ (לד) יִֽשְׂמַח־ אָבִ֥יךָ וְאִמֶּ֑ךָ וְ֝תָגֵ֗ל יֽוֹלַדְתֶּֽךָ׃ (לה) תְּנָֽה־ בְנִ֣י לִבְּךָ֣ לִ֑י וְ֝עֵינֶ֗יךָ דְּרָכַ֥י (תרצנה) תִּצֹּֽרְנָה׃ (לו) כִּֽי־ שׁוּחָ֣ה עֲמֻקָּ֣ה זוֹנָ֑ה וּבְאֵ֥ר צָ֝רָ֗ה נָכְרִיָּֽה׃ (לז) אַף־ הִ֭יא כְּחֶ֣תֶף תֶּֽאֱרֹ֑ב וּ֝בוֹגְדִ֗ים בְּאָדָ֥ם תּוֹסִֽף׃ (לח) לְמִ֨י א֥וֹי לְמִ֪י אֲב֡וֹי לְמִ֤י (מדונים) מִדְיָנִ֨ים ׀ לְמִ֥י שִׂ֗יחַ לְ֭מִי פְּצָעִ֣ים חִנָּ֑ם לְ֝מִ֗י חַכְלִל֥וּת עֵינָֽיִם׃ (לט) לַֽמְאַחֲרִ֥ים עַל־ הַיָּ֑יִן לַ֝בָּאִ֗ים לַחְקֹ֥ר מִמְסָֽךְ׃ (מ) אַל־ תֵּ֥רֶא יַיִן֮ כִּ֪י יִתְאַ֫דָּ֥ם כִּֽי־ יִתֵּ֣ן (בכיס) בַּכּ֣וֹס עֵינ֑וֹ יִ֝תְהַלֵּ֗ךְ בְּמֵישָׁרִֽים׃ (מא) אַ֭חֲרִיתוֹ כְּנָחָ֣שׁ יִשָּׁ֑ךְ וּֽכְצִפְעֹנִ֥י יַפְרִֽשׁ׃ (מב) עֵ֭ינֶיךָ יִרְא֣וּ זָר֑וֹת וְ֝לִבְּךָ֗ יְדַבֵּ֥ר תַּהְפֻּכֽוֹת׃ (מג) וְ֭הָיִיתָ כְּשֹׁכֵ֣ב בְּלֶב־ יָ֑ם וּ֝כְשֹׁכֵ֗ב בְּרֹ֣אשׁ חִבֵּֽל׃ (מד) הִכּ֥וּנִי בַל־ חָלִיתִי֮ הֲלָמ֗וּנִי בַּל־ יָ֫דָ֥עְתִּי מָתַ֥י אָקִ֑יץ א֝וֹסִ֗יף אֲבַקְשֶׁ֥נּוּ עֽוֹד׃ (מה) אַל־ תְּ֭קַנֵּא בְּאַנְשֵׁ֣י רָעָ֑ה וְאַל־ (תתאו) תִּ֝תְאָ֗יו לִהְי֥וֹת אִתָּֽם׃ (מו) כִּי־ שֹׁ֭ד יֶהְגֶּ֣ה לִבָּ֑ם וְ֝עָמָ֗ל שִׂפְתֵיהֶ֥ם תְּדַבֵּֽרְנָה׃ (מז) בְּ֭חָכְמָה יִבָּ֣נֶה בָּ֑יִת וּ֝בִתְבוּנָ֗ה יִתְכּוֹנָֽן׃ (מח) וּ֭בְדַעַת חֲדָרִ֣ים יִמָּלְא֑וּ כָּל־ ה֖וֹן יָקָ֣ר וְנָעִֽים׃ (מט) גֶּֽבֶר־ חָכָ֥ם בַּע֑וֹז וְאִֽישׁ־ דַּ֝֗עַת מְאַמֶּץ־ כֹּֽחַ׃ (נ) כִּ֣י בְ֭תַחְבֻּלוֹת תַּעֲשֶׂה־ לְּךָ֣ מִלְחָמָ֑ה וּ֝תְשׁוּעָ֗ה בְּרֹ֣ב יוֹעֵֽץ׃ (נא) רָאמ֣וֹת לֶֽאֱוִ֣יל חָכְמ֑וֹת בַּ֝שַּׁ֗עַר לֹ֣א יִפְתַּח־ פִּֽיהוּ׃ (נב) מְחַשֵּׁ֥ב לְהָרֵ֑עַ ל֝֗וֹ בַּֽעַל־ מְזִמּ֥וֹת יִקְרָֽאוּ׃ (נג) זִמַּ֣ת אִוֶּ֣לֶת חַטָּ֑את וְתוֹעֲבַ֖ת לְאָדָ֣ם לֵֽץ׃ (נד) הִ֭תְרַפִּיתָ בְּי֥וֹם צָרָ֗ה צַ֣ר כֹּחֶֽכָה׃ (נה) הַ֭צֵּל לְקֻחִ֣ים לַמָּ֑וֶת וּמָטִ֥ים לַ֝הֶ֗רֶג אִם־ תַּחְשֽׂוֹךְ׃ (נו) כִּֽי־ תֹאמַ֗ר הֵן֮ לֹא־ יָדַ֪עְנ֫וּ זֶ֥ה הֲ‍ֽלֹא־ תֹ֘כֵ֤ן לִבּ֨וֹת ׀ הֽוּא־ יָבִ֗ין וְנֹצֵ֣ר נַ֭פְשְׁךָ ה֣וּא יֵדָ֑ע וְהֵשִׁ֖יב לְאָדָ֣ם כְּפָעֳלֽוֹ׃ (נז) אֱכָל־ בְּנִ֣י דְבַ֣שׁ כִּי־ ט֑וֹב וְנֹ֥פֶת מָ֝ת֗וֹק עַל־ חִכֶּֽךָ׃ (נח) כֵּ֤ן ׀ דְּעֶ֥ה חָכְמָ֗ה לְנַ֫פְשֶׁ֥ךָ אִם־ מָ֭צָאתָ וְיֵ֣שׁ אַחֲרִ֑ית וְ֝תִקְוָתְךָ֗ לֹ֣א תִכָּרֵֽת׃ (נט) אַל־ תֶּאֱרֹ֣ב רָ֭שָׁע לִנְוֵ֣ה צַדִּ֑יק אַֽל־ תְּשַׁדֵּ֥ד רִבְצוֹ׃ (ס) כִּ֤י שֶׁ֨בַע ׀ יִפּ֣וֹל צַדִּ֣יק וָקָ֑ם וּ֝רְשָׁעִ֗ים יִכָּשְׁל֥וּ בְרָעָֽה׃ (סא) בִּנְפֹ֣ל (אויביך) אֽ֭וֹיִבְךָ אַל־ תִּשְׂמָ֑ח וּ֝בִכָּשְׁל֗וֹ אַל־ יָגֵ֥ל לִבֶּֽךָ׃ (סב) פֶּן־ יִרְאֶ֣ה יְ֭הוָה וְרַ֣ע בְּעֵינָ֑יו וְהֵשִׁ֖יב מֵעָלָ֣יו אַפּֽוֹ׃ (סג) אַל־ תִּתְחַ֥ר בַּמְּרֵעִ֑ים אַל־ תְּ֝קַנֵּ֗א בָּרְשָׁעִֽים׃ (סד) כִּ֤י ׀ לֹֽא־ תִהְיֶ֣ה אַחֲרִ֣ית לָרָ֑ע נֵ֖ר רְשָׁעִ֣ים יִדְעָֽךְ׃ (סה) יְרָֽא־ אֶת־ יְהוָ֣ה בְּנִ֣י וָמֶ֑לֶךְ עִם־ שׁ֝וֹנִ֗ים אַל־ תִּתְעָרָֽב׃ (סו) כִּֽי־ פִ֭תְאֹם יָק֣וּם אֵידָ֑ם וּפִ֥יד שְׁ֝נֵיהֶ֗ם מִ֣י יוֹדֵֽעַ׃ (סז) גַּם־ אֵ֥לֶּה לַֽחֲכָמִ֑ים הַֽכֵּר־ פָּנִ֖ים בְּמִשְׁפָּ֣ט בַּל־ טֽוֹב׃ (סח) אֹ֤מֵ֨ר ׀ לְרָשָׁע֮ צַדִּ֪יק אָ֥תָּה יִקְּבֻ֥הוּ עַמִּ֑ים יִזְעָמ֥וּהוּ לְאֻמִּֽים׃ (סט) וְלַמּוֹכִיחִ֥ים יִנְעָ֑ם וַֽ֝עֲלֵיהֶ֗ם תָּב֥וֹא בִרְכַּת־ טֽוֹב׃ (ע) שְׂפָתַ֥יִם יִשָּׁ֑ק מֵ֝שִׁ֗יב דְּבָרִ֥ים נְכֹחִֽים׃ (עא) הָ֘כֵ֤ן בַּח֨וּץ ׀ מְלַאכְתֶּ֗ךָ וְעַתְּדָ֣הּ בַּשָּׂדֶ֣ה לָ֑ךְ אַ֝חַ֗ר וּבָנִ֥יתָ בֵיתֶֽךָ׃ (עב) אַל־ תְּהִ֣י עֵד־ חִנָּ֣ם בְּרֵעֶ֑ךָ וַ֝הֲפִתִּ֗יתָ בִּשְׂפָתֶֽיךָ׃ (עג) אַל־ תֹּאמַ֗ר וְלֹא־ פִ֑יךָ נָ֝כְרִ֗י וְאַל־ שְׂפָתֶֽיךָ׃ (מז) כֹּֽבֶד־ אֶ֭בֶן וְנֵ֣טֶל הַח֑וֹל וְכַ֥עַס אֱ֝וִ֗יל כָּבֵ֥ד מִשְּׁנֵיהֶֽם׃ (מח) אַכְזְרִיּ֣וּת חֵ֭מָה וְשֶׁ֣טֶף אָ֑ף וּמִ֥י יַ֝עֲמֹד לִפְנֵ֥י קִנְאָֽה׃ (מט) ט֭וֹבָה תּוֹכַ֣חַת מְגֻלָּ֑ה מֵֽאַהֲבָ֥ה מְסֻתָּֽרֶת׃ (נ) נֶ֭אֱמָנִים פִּצְעֵ֣י אוֹהֵ֑ב וְ֝נַעְתָּר֗וֹת נְשִׁיק֥וֹת שׂוֹנֵֽא׃ (נא) נֶ֣פֶשׁ שְׂ֭בֵעָה תָּב֣וּס נֹ֑פֶת וְנֶ֥פֶשׁ רְ֝עֵבָ֗ה כָּל־ מַ֥ר מָתֽוֹק׃ (נב) כְּ֭צִפּוֹר נוֹדֶ֣דֶת מִן־ קִנָּ֑הּ כֵּֽן־ אִ֝֗ישׁ נוֹדֵ֥ד מִמְּקוֹמֽוֹ׃ (נג) שֶׁ֣מֶן וּ֭קְטֹרֶת יְשַׂמַּֽח־ לֵ֑ב וּמֶ֥תֶק רֵ֝עֵ֗הוּ מֵֽעֲצַת־ נָֽפֶשׁ׃ (נד) רֵֽעֲךָ֨ (ורעה) וְרֵ֪עַ אָבִ֡יךָ אַֽל־ תַּעֲזֹ֗ב וּבֵ֥ית אָחִ֗יךָ אַל־ תָּ֭בוֹא בְּי֣וֹם אֵידֶ֑ךָ ט֥וֹב שָׁכֵ֥ן קָ֝ר֗וֹב מֵאָ֥ח רָחֽוֹק׃ (נה) חֲכַ֣ם בְּ֭נִי וְשַׂמַּ֣ח לִבִּ֑י וְאָשִׁ֖יבָה חֹרְפִ֣י דָבָֽר׃ (נו) עָר֤וּם רָאָ֣ה רָעָ֣ה נִסְתָּ֑ר פְּ֝תָאיִ֗ם עָבְר֥וּ נֶעֱנָֽשׁוּ׃ (נז) קַח־ בִּ֭גְדוֹ כִּי־ עָ֣רַב זָ֑ר וּבְעַ֖ד נָכְרִיָּ֣ה חַבְלֵֽהוּ׃ (נח) מְבָ֘רֵ֤ךְ רֵעֵ֨הוּ ׀ בְּק֣וֹל גָּ֭דוֹל בַּבֹּ֣קֶר הַשְׁכֵּ֑ים קְ֝לָלָ֗ה תֵּחָ֥שֶׁב לֽוֹ׃ (נט) דֶּ֣לֶף ט֭וֹרֵד בְּי֣וֹם סַגְרִ֑יר וְאֵ֥שֶׁת (מדונים) מִ֝דְיָנִ֗ים נִשְׁתָּוָֽה׃ (ס) צֹפְנֶ֥יהָ צָֽפַן־ ר֑וּחַ וְשֶׁ֖מֶן יְמִינ֣וֹ יִקְרָֽא׃ (סא) בַּרְזֶ֣ל בְּבַרְזֶ֣ל יָ֑חַד וְ֝אִ֗ישׁ יַ֣חַד פְּנֵֽי־ רֵעֵֽהוּ׃ (סב) נֹצֵ֣ר תְּ֭אֵנָה יֹאכַ֣ל פִּרְיָ֑הּ וְשֹׁמֵ֖ר אֲדֹנָ֣יו יְכֻבָּֽד׃ (סג) כַּ֭מַּיִם הַפָּנִ֣ים לַפָּנִ֑ים כֵּ֤ן לֵֽב־ הָ֝אָדָ֗ם לָאָדָֽם׃ (סד) שְׁא֣וֹל (ואבדה) וַ֭אֲבַדּוֹ לֹ֣א תִשְׂבַּ֑עְנָה וְעֵינֵ֥י הָ֝אָדָ֗ם לֹ֣א תִשְׂבַּֽעְנָה׃ (סה) מַצְרֵ֣ף לַ֭כֶּסֶף וְכ֣וּר לַזָּהָ֑ב וְ֝אִ֗ישׁ לְפִ֣י מַהֲלָלֽוֹ׃ (סו) אִ֥ם תִּכְתּֽוֹשׁ־ אֶת־ הָאֱוִ֨יל ׀ בַּֽמַּכְתֵּ֡שׁ בְּת֣וֹךְ הָ֭רִיפוֹת בַּֽעֱלִ֑י לֹא־ תָס֥וּר מֵ֝עָלָ֗יו אִוַּלְתּֽוֹ׃ (סז) יָדֹ֣עַ תֵּ֭דַע פְּנֵ֣י צֹאנֶ֑ךָ שִׁ֥ית לִ֝בְּךָ֗ לַעֲדָרִֽים׃ (סח) כִּ֤י לֹ֣א לְעוֹלָ֣ם חֹ֑סֶן וְאִם־ נֵ֝֗זֶר לְד֣וֹר (דור) וָדֽוֹר׃ (סט) גָּלָ֣ה חָ֭צִיר וְנִרְאָה־ דֶ֑שֶׁא וְ֝נֶאֶסְפ֗וּ עִשְּׂב֥וֹת הָרִֽים׃ (ע) כְּבָשִׂ֥ים לִלְבוּשֶׁ֑ךָ וּמְחִ֥יר שָׂ֝דֶ֗ה עַתּוּדִֽים׃ (עא) וְדֵ֤י ׀ חֲלֵ֬ב עִזִּ֗ים לְֽ֭לַחְמְךָ לְלֶ֣חֶם בֵּיתֶ֑ךָ וְ֝חַיִּ֗ים לְנַעֲרוֹתֶֽיךָ׃ (עב) נָ֣סוּ וְאֵין־ רֹדֵ֣ף רָשָׁ֑ע וְ֝צַדִּיקִ֗ים כִּכְפִ֥יר יִבְטָֽח׃ (עג) בְּפֶ֣שַֽׁע אֶ֭רֶץ רַבִּ֣ים שָׂרֶ֑יהָ וּבְאָדָ֥ם מֵבִ֥ין יֹ֝דֵ֗עַ כֵּ֣ן יַאֲרִֽיךְ׃ (עד) גֶּ֣בֶר רָ֭שׁ וְעֹשֵׁ֣ק דַּלִּ֑ים מָטָ֥ר סֹ֝חֵ֗ף וְאֵ֣ין לָֽחֶם׃ (עה) עֹזְבֵ֣י ת֭וֹרָה יְהַֽלְל֣וּ רָשָׁ֑ע וְשֹׁמְרֵ֥י ת֝וֹרָ֗ה יִתְגָּ֥רוּ בָֽם׃ (עו) אַנְשֵׁי־ רָ֭ע לֹא־ יָבִ֣ינוּ מִשְׁפָּ֑ט וּמְבַקְשֵׁ֥י יְ֝הוָ֗ה יָבִ֥ינוּ כֹֽל׃ (עז) טֽוֹב־ רָ֭שׁ הוֹלֵ֣ךְ בְּתֻמּ֑וֹ מֵעִקֵּ֥שׁ דְּ֝רָכַ֗יִם וְה֣וּא עָשִֽׁיר׃ (עח) נוֹצֵ֣ר תּ֭וֹרָה בֵּ֣ן מֵבִ֑ין וְרֹעֶה זֽ֝וֹלְלִ֗ים יַכְלִ֥ים אָבִֽיו׃ (עט) מַרְבֶּ֣ה ה֭וֹנוֹ בְּנֶ֣שֶׁךְ (ובתרבית) וְתַרְבִּ֑ית לְחוֹנֵ֖ן דַּלִּ֣ים יִקְבְּצֶֽנּוּ׃ (פ) מֵסִ֣יר אָ֭זְנוֹ מִשְּׁמֹ֣עַ תּוֹרָ֑ה גַּֽם־ תְּ֝פִלָּת֗וֹ תּוֹעֵבָֽה׃ (פא) מַשְׁגֶּ֤ה יְשָׁרִ֨ים ׀ בְּדֶ֥רֶךְ רָ֗ע בִּשְׁחוּת֥וֹ הֽוּא־ יִפּ֑וֹל וּ֝תְמִימִ֗ים יִנְחֲלוּ־ טֽוֹב׃ (פב) חָכָ֣ם בְּ֭עֵינָיו אִ֣ישׁ עָשִׁ֑יר וְדַ֖ל מֵבִ֣ין יַחְקְרֶֽנּוּ׃ (פג) בַּעֲלֹ֣ץ צַ֭דִּיקִים רַבָּ֣ה תִפְאָ֑רֶת וּבְק֥וּם רְ֝שָׁעִ֗ים יְחֻפַּ֥שׂ אָדָֽם׃ (פד) מְכַסֶּ֣ה פְ֭שָׁעָיו לֹ֣א יַצְלִ֑יחַ וּמוֹדֶ֖ה וְעֹזֵ֣ב יְרֻחָֽם׃ (פה) אַשְׁרֵ֣י אָ֭דָם מְפַחֵ֣ד תָּמִ֑יד וּמַקְשֶׁ֥ה לִ֝בּ֗וֹ יִפּ֥וֹל בְּרָעָֽה׃ (פו) אֲרִי־ נֹ֭הֵם וְדֹ֣ב שׁוֹקֵ֑ק מֹשֵׁ֥ל רָ֝שָׁ֗ע עַ֣ל עַם־ דָּֽל׃
