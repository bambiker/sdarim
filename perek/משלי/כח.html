משלי פרק כח לפי החלוקה לסדרים

פרק כח בספר משלי (28 פסוקים) נמצא במשלי סדר ז (פסוקים עב–פו) ובמשלי סדר ח (פסוקים א–יג).

משלי סדר ז

(א) כְּצִנַּת־ שֶׁ֨לֶג ׀ בְּי֬וֹם קָצִ֗יר צִ֣יר נֶ֭אֱמָן לְשֹׁלְחָ֑יו וְנֶ֖פֶשׁ אֲדֹנָ֣יו יָשִֽׁיב׃ (ב) נְשִׂיאִ֣ים וְ֭רוּחַ וְגֶ֣שֶׁם אָ֑יִן אִ֥ישׁ מִ֝תְהַלֵּ֗ל בְּמַתַּת־ שָֽׁקֶר׃ (ג) בְּאֹ֣רֶךְ אַ֭פַּיִם יְפֻתֶּ֣ה קָצִ֑ין וְלָשׁ֥וֹן רַ֝כָּ֗ה תִּשְׁבָּר־ גָּֽרֶם׃ (ד) דְּבַ֣שׁ מָ֭צָאתָ אֱכֹ֣ל דַּיֶּ֑ךָּ פֶּן־ תִּ֝שְׂבָּעֶ֗נּוּ וַהֲקֵֽאתֽוֹ׃ (ה) הֹקַ֣ר רַ֭גְלְךָ מִבֵּ֣ית רֵעֶ֑ךָ פֶּן־ יִ֝שְׂבָּעֲךָ֗ וּשְׂנֵאֶֽךָ׃ (ו) מֵפִ֣יץ וְ֭חֶרֶב וְחֵ֣ץ שָׁנ֑וּן אִ֥ישׁ עֹנֶ֥ה בְ֝רֵעֵ֗הוּ עֵ֣ד שָֽׁקֶר׃ (ז) שֵׁ֣ן רֹ֭עָה וְרֶ֣גֶל מוּעָ֑דֶת מִבְטָ֥ח בּ֝וֹגֵ֗ד בְּי֣וֹם צָרָֽה׃ (ח) מַ֥עֲדֶה בֶּ֨גֶד ׀ בְּי֣וֹם קָ֭רָה חֹ֣מֶץ עַל־ נָ֑תֶר וְשָׁ֥ר בַּ֝שִּׁרִ֗ים עַ֣ל לֶב־ רָֽע׃ (ט) אִם־ רָעֵ֣ב שֹׂ֭נַאֲךָ הַאֲכִלֵ֣הוּ לָ֑חֶם וְאִם־ צָ֝מֵ֗א הַשְׁקֵ֥הוּ מָֽיִם׃ (י) כִּ֤י גֶֽחָלִ֗ים אַ֭תָּה חֹתֶ֣ה עַל־ רֹאשׁ֑וֹ וַֽ֝יהוָ֗ה יְשַׁלֶּם־ לָֽךְ׃ (יא) ר֣וּחַ צָ֭פוֹן תְּח֣וֹלֵֽל גָּ֑שֶׁם וּפָנִ֥ים נִ֝זְעָמִ֗ים לְשׁ֣וֹן סָֽתֶר׃ (יב) ט֗וֹב שֶׁ֥בֶת עַל־ פִּנַּת־ גָּ֑ג מֵאֵ֥שֶׁת (מדונים) מִ֝דְיָנִ֗ים וּבֵ֥ית חָֽבֶר׃ (יג) מַ֣יִם קָ֭רִים עַל־ נֶ֣פֶשׁ עֲיֵפָ֑ה וּשְׁמוּעָ֥ה ט֝וֹבָ֗ה מֵאֶ֥רֶץ מֶרְחָֽק׃ (יד) מַעְיָ֣ן נִ֭רְפָּשׂ וּמָק֣וֹר מָשְׁחָ֑ת צַ֝דִּ֗יק מָ֣ט לִפְנֵֽי־ רָשָֽׁע׃ (טו) אָ֘כֹ֤ל דְּבַ֣שׁ הַרְבּ֣וֹת לֹא־ ט֑וֹב וְחֵ֖קֶר כְּבֹדָ֣ם כָּבֽוֹד׃ (טז) עִ֣יר פְּ֭רוּצָה אֵ֣ין חוֹמָ֑ה אִ֝֗ישׁ אֲשֶׁ֤ר אֵ֖ין מַעְצָ֣ר לְרוּחֽוֹ׃ (יז) כַּשֶּׁ֤לֶג ׀ בַּקַּ֗יִץ וְכַמָּטָ֥ר בַּקָּצִ֑יר כֵּ֤ן לֹא־ נָאוֶ֖ה לִכְסִ֣יל כָּבֽוֹד׃ (יח) כַּצִּפּ֣וֹר לָ֭נוּד כַּדְּר֣וֹר לָע֑וּף כֵּ֥ן קִֽלְלַ֥ת חִ֝נָּ֗ם (לא) ל֣וֹ תָבֹֽא׃ (יט) שׁ֣וֹט לַ֭סּוּס מֶ֣תֶג לַחֲמ֑וֹר וְ֝שֵׁ֗בֶט לְגֵ֣ו כְּסִילִֽים׃ (כ) אַל־ תַּ֣עַן כְּ֭סִיל כְּאִוַּלְתּ֑וֹ פֶּֽן־ תִּשְׁוֶה־ לּ֥וֹ גַם־ אָֽתָּה׃ (כא) עֲנֵ֣ה כְ֭סִיל כְּאִוַּלְתּ֑וֹ פֶּן־ יִהְיֶ֖ה חָכָ֣ם בְּעֵינָֽיו׃ (כב) מְקַצֶּ֣ה רַ֭גְלַיִם חָמָ֣ס שֹׁתֶ֑ה שֹׁלֵ֖חַ דְּבָרִ֣ים בְּיַד־ כְּסִֽיל׃ (כג) דַּלְי֣וּ שֹׁ֭קַיִם מִפִּסֵּ֑חַ וּ֝מָשָׁ֗ל בְּפִ֣י כְסִילִֽים׃ (כד) כִּצְר֣וֹר אֶ֭בֶן בְּמַרְגֵּמָ֑ה כֵּן־ נוֹתֵ֖ן לִכְסִ֣יל כָּבֽוֹד׃ (כה) ח֭וֹחַ עָלָ֣ה בְיַד־ שִׁכּ֑וֹר וּ֝מָשָׁ֗ל בְּפִ֣י כְסִילִֽים׃ (כו) רַ֥ב מְחֽוֹלֵֽל־ כֹּ֑ל וְשֹׂכֵ֥ר כְּ֝סִ֗יל וְשֹׂכֵ֥ר עֹבְרִֽים׃ (כז) כְּ֭כֶלֶב שָׁ֣ב עַל־ קֵא֑וֹ כְּ֝סִ֗יל שׁוֹנֶ֥ה בְאִוַּלְתּֽוֹ׃ (כח) רָאִ֗יתָ אִ֭ישׁ חָכָ֣ם בְּעֵינָ֑יו תִּקְוָ֖ה לִכְסִ֣יל מִמֶּֽנּוּ׃ (כט) אָמַ֣ר עָ֭צֵל שַׁ֣חַל בַּדָּ֑רֶךְ אֲ֝רִ֗י בֵּ֣ין הָרְחֹבֽוֹת׃ (ל) הַ֭דֶּלֶת תִּסּ֣וֹב עַל־ צִירָ֑הּ וְ֝עָצֵ֗ל עַל־ מִטָּתֽוֹ׃ (לא) טָ֘מַ֤ן עָצֵ֣ל יָ֭דוֹ בַּצַּלָּ֑חַת נִ֝לְאָ֗ה לַֽהֲשִׁיבָ֥הּ אֶל־ פִּֽיו׃ (לב) חָכָ֣ם עָצֵ֣ל בְּעֵינָ֑יו מִ֝שִּׁבְעָ֗ה מְשִׁ֣יבֵי טָֽעַם׃ (לג) מַחֲזִ֥יק בְּאָזְנֵי־ כָ֑לֶב עֹבֵ֥ר מִ֝תְעַבֵּ֗ר עַל־ רִ֥יב לֹּֽא־ לֽוֹ׃ (לד) כְּֽ֭מִתְלַהְלֵהַּ הַיֹּרֶ֥ה זִקִּ֗ים חִצִּ֥ים וָמָֽוֶת׃ (לה) כֵּֽן־ אִ֭ישׁ רִמָּ֣ה אֶת־ רֵעֵ֑הוּ וְ֝אָמַ֗ר הֲ‍ֽלֹא־ מְשַׂחֵ֥ק אָֽנִי׃ (לו) בְּאֶ֣פֶס עֵ֭צִים תִּכְבֶּה־ אֵ֑שׁ וּבְאֵ֥ין נִ֝רְגָּ֗ן יִשְׁתֹּ֥ק מָדֽוֹן׃ (לז) פֶּחָ֣ם לְ֭גֶחָלִים וְעֵצִ֣ים לְאֵ֑שׁ וְאִ֥ישׁ (מדונים) מִ֝דְיָנִ֗ים לְחַרְחַר־ רִֽיב׃ (לח) דִּבְרֵ֣י נִ֭רְגָּן כְּמִֽתְלַהֲמִ֑ים וְ֝הֵ֗ם יָרְד֥וּ חַדְרֵי־ בָֽטֶן׃ (לט) כֶּ֣סֶף סִ֭יגִים מְצֻפֶּ֣ה עַל־ חָ֑רֶשׂ שְׂפָתַ֖יִם דֹּלְקִ֣ים וְלֶב־ רָֽע׃ (מ) (בשפתו) בִּ֭שְׂפָתָיו יִנָּכֵ֣ר שׂוֹנֵ֑א וּ֝בְקִרְבּ֗וֹ יָשִׁ֥ית מִרְמָֽה׃ (מא) כִּֽי־ יְחַנֵּ֣ן ק֭וֹלוֹ אַל־ תַּֽאֲמֶן־ בּ֑וֹ כִּ֤י שֶׁ֖בַע תּוֹעֵב֣וֹת בְּלִבּֽוֹ׃ (מב) תִּכַּסֶּ֣ה שִׂ֭נְאָה בְּמַשָּׁא֑וֹן תִּגָּלֶ֖ה רָעָת֣וֹ בְקָהָֽל׃ (מג) כֹּֽרֶה־ שַּׁ֭חַת בָּ֣הּ יִפֹּ֑ל וְגֹ֥לֵ֥ל אֶ֝בֶן אֵלָ֥יו תָּשֽׁוּב׃ (מד) לְֽשׁוֹן־ שֶׁ֭קֶר יִשְׂנָ֣א דַכָּ֑יו וּפֶ֥ה חָ֝לָ֗ק יַעֲשֶׂ֥ה מִדְחֶֽה׃ (מה) אַֽל־ תִּ֭תְהַלֵּל בְּי֣וֹם מָחָ֑ר כִּ֤י לֹא־ תֵ֝דַ֗ע מַה־ יֵּ֥לֶד יֽוֹם׃ (מו) יְהַלֶּלְךָ֣ זָ֣ר וְלֹא־ פִ֑יךָ נָ֝כְרִ֗י וְאַל־ שְׂפָתֶֽיךָ׃ (מז) כֹּֽבֶד־ אֶ֭בֶן וְנֵ֣טֶל הַח֑וֹל וְכַ֥עַס אֱ֝וִ֗יל כָּבֵ֥ד מִשְּׁנֵיהֶֽם׃ (מח) אַכְזְרִיּ֣וּת חֵ֭מָה וְשֶׁ֣טֶף אָ֑ף וּמִ֥י יַ֝עֲמֹד לִפְנֵ֥י קִנְאָֽה׃ (מט) ט֭וֹבָה תּוֹכַ֣חַת מְגֻלָּ֑ה מֵֽאַהֲבָ֥ה מְסֻתָּֽרֶת׃ (נ) נֶ֭אֱמָנִים פִּצְעֵ֣י אוֹהֵ֑ב וְ֝נַעְתָּר֗וֹת נְשִׁיק֥וֹת שׂוֹנֵֽא׃ (נא) נֶ֣פֶשׁ שְׂ֭בֵעָה תָּב֣וּס נֹ֑פֶת וְנֶ֥פֶשׁ רְ֝עֵבָ֗ה כָּל־ מַ֥ר מָתֽוֹק׃ (נב) כְּ֭צִפּוֹר נוֹדֶ֣דֶת מִן־ קִנָּ֑הּ כֵּֽן־ אִ֝֗ישׁ נוֹדֵ֥ד מִמְּקוֹמֽוֹ׃ (נג) שֶׁ֣מֶן וּ֭קְטֹרֶת יְשַׂמַּֽח־ לֵ֑ב וּמֶ֥תֶק רֵ֝עֵ֗הוּ מֵֽעֲצַת־ נָֽפֶשׁ׃ (נד) רֵֽעֲךָ֨ (ורעה) וְרֵ֪עַ אָבִ֡יךָ אַֽל־ תַּעֲזֹ֗ב וּבֵ֥ית אָחִ֗יךָ אַל־ תָּ֭בוֹא בְּי֣וֹם אֵידֶ֑ךָ ט֥וֹב שָׁכֵ֥ן קָ֝ר֗וֹב מֵאָ֥ח רָחֽוֹק׃ (נה) חֲכַ֣ם בְּ֭נִי וְשַׂמַּ֣ח לִבִּ֑י וְאָשִׁ֖יבָה חֹרְפִ֣י דָבָֽר׃ (נו) עָר֤וּם רָאָ֣ה רָעָ֣ה נִסְתָּ֑ר פְּ֝תָאיִ֗ם עָבְר֥וּ נֶעֱנָֽשׁוּ׃ (נז) קַח־ בִּ֭גְדוֹ כִּי־ עָ֣רַב זָ֑ר וּבְעַ֖ד נָכְרִיָּ֣ה חַבְלֵֽהוּ׃ (נח) מְבָ֘רֵ֤ךְ רֵעֵ֨הוּ ׀ בְּק֣וֹל גָּ֭דוֹל בַּבֹּ֣קֶר הַשְׁכֵּ֑ים קְ֝לָלָ֗ה תֵּחָ֥שֶׁב לֽוֹ׃ (נט) דֶּ֣לֶף ט֭וֹרֵד בְּי֣וֹם סַגְרִ֑יר וְאֵ֥שֶׁת (מדונים) מִ֝דְיָנִ֗ים נִשְׁתָּוָֽה׃ (ס) צֹפְנֶ֥יהָ צָֽפַן־ ר֑וּחַ וְשֶׁ֖מֶן יְמִינ֣וֹ יִקְרָֽא׃ (סא) בַּרְזֶ֣ל בְּבַרְזֶ֣ל יָ֑חַד וְ֝אִ֗ישׁ יַ֣חַד פְּנֵֽי־ רֵעֵֽהוּ׃ (סב) נֹצֵ֣ר תְּ֭אֵנָה יֹאכַ֣ל פִּרְיָ֑הּ וְשֹׁמֵ֖ר אֲדֹנָ֣יו יְכֻבָּֽד׃ (סג) כַּ֭מַּיִם הַפָּנִ֣ים לַפָּנִ֑ים כֵּ֤ן לֵֽב־ הָ֝אָדָ֗ם לָאָדָֽם׃ (סד) שְׁא֣וֹל (ואבדה) וַ֭אֲבַדּוֹ לֹ֣א תִשְׂבַּ֑עְנָה וְעֵינֵ֥י הָ֝אָדָ֗ם לֹ֣א תִשְׂבַּֽעְנָה׃ (סה) מַצְרֵ֣ף לַ֭כֶּסֶף וְכ֣וּר לַזָּהָ֑ב וְ֝אִ֗ישׁ לְפִ֣י מַהֲלָלֽוֹ׃ (סו) אִ֥ם תִּכְתּֽוֹשׁ־ אֶת־ הָאֱוִ֨יל ׀ בַּֽמַּכְתֵּ֡שׁ בְּת֣וֹךְ הָ֭רִיפוֹת בַּֽעֱלִ֑י לֹא־ תָס֥וּר מֵ֝עָלָ֗יו אִוַּלְתּֽוֹ׃ (סז) יָדֹ֣עַ תֵּ֭דַע פְּנֵ֣י צֹאנֶ֑ךָ שִׁ֥ית לִ֝בְּךָ֗ לַעֲדָרִֽים׃ (סח) כִּ֤י לֹ֣א לְעוֹלָ֣ם חֹ֑סֶן וְאִם־ נֵ֝֗זֶר לְד֣וֹר (דור) וָדֽוֹר׃ (סט) גָּלָ֣ה חָ֭צִיר וְנִרְאָה־ דֶ֑שֶׁא וְ֝נֶאֶסְפ֗וּ עִשְּׂב֥וֹת הָרִֽים׃ (ע) כְּבָשִׂ֥ים לִלְבוּשֶׁ֑ךָ וּמְחִ֥יר שָׂ֝דֶ֗ה עַתּוּדִֽים׃ (עא) וְדֵ֤י ׀ חֲלֵ֬ב עִזִּ֗ים לְֽ֭לַחְמְךָ לְלֶ֣חֶם בֵּיתֶ֑ךָ וְ֝חַיִּ֗ים לְנַעֲרוֹתֶֽיךָ׃ תחילת פרק כח (עב) נָ֣סוּ וְאֵין־ רֹדֵ֣ף רָשָׁ֑ע וְ֝צַדִּיקִ֗ים כִּכְפִ֥יר יִבְטָֽח׃ (עג) בְּפֶ֣שַֽׁע אֶ֭רֶץ רַבִּ֣ים שָׂרֶ֑יהָ וּבְאָדָ֥ם וְאָמַ֗רְתִּי מִ֥י יְה֫וָ֥ה וּפֶֽן־ אִוָּרֵ֥שׁ וְגָנַ֑בְתִּי וְ֝תָפַ֗שְׂתִּי שֵׁ֣ם אֱלֹהָֽי׃ (נ) אַל־ תַּלְשֵׁ֣ן עֶ֭בֶד אֶל־ (אדנו) אֲדֹנָ֑יו פֶּֽן־ יְקַלֶּלְךָ֥ וְאָשָֽׁמְתָּ׃ (נא) דּ֭וֹר אָבִ֣יו יְקַלֵּ֑ל וְאֶת־ אִ֝מּ֗וֹ לֹ֣א יְבָרֵֽךְ׃ (נב) דּ֭וֹר טָה֣וֹר בְּעֵינָ֑יו וּ֝מִצֹּאָת֗וֹ לֹ֣א רֻחָֽץ׃ (נג) דּ֭וֹר מָה־ רָמ֣וּ עֵינָ֑יו וְ֝עַפְעַפָּ֗יו יִנָּשֵֽׂאוּ׃ (נד) דּ֤וֹר ׀ חֲרָב֣וֹת שִׁנָּיו֮ וּֽמַאֲכָל֪וֹת מְֽתַלְּעֹ֫תָ֥יו לֶאֱכֹ֣ל עֲנִיִּ֣ים מֵאֶ֑רֶץ וְ֝אֶבְיוֹנִ֗ים מֵאָדָֽם׃ (נה) לַֽעֲלוּקָ֨ה ׀ שְׁתֵּ֥י בָנוֹת֮ הַ֤ב ׀ הַ֥ב שָׁל֣וֹשׁ הֵ֭נָּה לֹ֣א תִשְׂבַּ֑עְנָה אַ֝רְבַּ֗ע לֹא־ אָ֥מְרוּ הֽוֹן׃ (נו) שְׁאוֹל֮ וְעֹ֪צֶ֫ר רָ֥חַם אֶ֭רֶץ לֹא־ שָׂ֣בְעָה מַּ֑יִם וְ֝אֵ֗שׁ לֹא־ אָ֥מְרָה הֽוֹן׃ (נז) עַ֤יִן ׀ תִּֽלְעַ֣ג לְאָב֮ וְתָב֪וּז לִֽיקֲּהַ֫ת־ אֵ֥ם יִקְּר֥וּהָ עֹרְבֵי־ נַ֑חַל וְֽיֹאכְל֥וּהָ בְנֵי־ נָֽשֶׁר׃ (נח) שְׁלֹשָׁ֣ה הֵ֭מָּה נִפְלְא֣וּ מִמֶּ֑נִּי (וארבע) וְ֝אַרְבָּעָ֗ה לֹ֣א יְדַעְתִּֽים׃ (נט) דֶּ֤רֶךְ הַנֶּ֨שֶׁר ׀ בַּשָּׁמַיִם֮ דֶּ֥רֶךְ נָחָ֗שׁ עֲלֵ֫י צ֥וּר דֶּֽרֶךְ־ אֳנִיָּ֥ה בְלֶב־ יָ֑ם וְדֶ֖רֶךְ גֶּ֣בֶר בְּעַלְמָֽה׃ (ס) כֵּ֤ן ׀ דֶּ֥רֶךְ אִשָּׁ֗ה מְנָ֫אָ֥פֶת אָ֭כְלָה וּמָ֣חֲתָה פִ֑יהָ וְ֝אָמְרָ֗ה לֹֽא־ פָעַ֥לְתִּי אָֽוֶן׃ (סא) תַּ֣חַת שָׁ֭לוֹשׁ רָ֣גְזָה אֶ֑רֶץ וְתַ֥חַת אַ֝רְבַּ֗ע לֹא־ תוּכַ֥ל שְׂאֵֽת׃ (סב) תַּֽחַת־ עֶ֭בֶד כִּ֣י יִמְל֑וֹךְ וְ֝נָבָ֗ל כִּ֣י יִֽשְׂבַּֽע־ לָֽחֶם׃ (סג) תַּ֣חַת שְׂ֭נוּאָה כִּ֣י תִבָּעֵ֑ל וְ֝שִׁפְחָ֗ה כִּֽי־ תִירַ֥שׁ גְּבִרְתָּֽהּ׃ (סד) אַרְבָּ֣עָה הֵ֭ם קְטַנֵּי־ אָ֑רֶץ וְ֝הֵ֗מָּה חֲכָמִ֥ים מְחֻכָּמִֽים׃ (סה) הַ֭נְּמָלִים עַ֣ם לֹא־ עָ֑ז וַיָּכִ֖ינוּ בַקַּ֣יִץ לַחְמָֽם׃ (סו) שְׁ֭פַנִּים עַ֣ם לֹא־ עָצ֑וּם וַיָּשִׂ֖ימוּ בַסֶּ֣לַע בֵּיתָֽם׃ (סז) מֶ֭לֶךְ אֵ֣ין לָאַרְבֶּ֑ה וַיֵּצֵ֖א חֹצֵ֣ץ כֻּלּֽוֹ׃ (סח) שְׂ֭מָמִית בְּיָדַ֣יִם תְּתַפֵּ֑שׂ וְ֝הִ֗יא בְּהֵ֣יכְלֵי מֶֽלֶךְ׃ (סט) שְׁלֹשָׁ֣ה הֵ֭מָּה מֵיטִ֣יבֵי צָ֑עַד וְ֝אַרְבָּעָ֗ה מֵיטִ֥בֵי לָֽכֶת׃ (ע) לַ֭יִשׁ גִּבּ֣וֹר בַּבְּהֵמָ֑ה וְלֹא־ יָ֝שׁ֗וּב מִפְּנֵי־ כֹֽל׃ (עא) זַרְזִ֣יר מָתְנַ֣יִם אוֹ־ תָ֑יִשׁ וּ֝מֶ֗לֶךְ אַלְק֥וּם עִמּֽוֹ׃ (עב) אִם־ נָבַ֥לְתָּ בְהִתְנַשֵּׂ֑א וְאִם־ זַ֝מּ֗וֹתָ יָ֣ד לְפֶֽה׃ (עג) כִּ֤י מִ֪יץ חָלָ֡ב י֘וֹצִ֤יא חֶמְאָ֗ה וּֽמִיץ־ אַ֭ף י֣וֹצִיא דָ֑ם וּמִ֥יץ אַ֝פַּ֗יִם י֣וֹצִיא רִֽיב׃ (עד) דִּ֭בְרֵי לְמוּאֵ֣ל מֶ֑לֶךְ מַ֝שָּׂ֗א אֲ‍ֽשֶׁר־ יִסְּרַ֥תּוּ אִמּֽוֹ׃ (עה) מַה־ בְּ֭רִי וּמַֽה־ בַּר־ בִּטְנִ֑י וּ֝מֶה בַּר־ נְדָרָֽי׃ (עו) אַל־ תִּתֵּ֣ן לַנָּשִׁ֣ים חֵילֶ֑ךָ וּ֝דְרָכֶ֗יךָ לַֽמְח֥וֹת מְלָכִֽין׃ (עז) אַ֤ל לַֽמְלָכִ֨ים ׀ לְֽמוֹאֵ֗ל אַ֣ל לַֽמְלָכִ֣ים שְׁתוֹ־ יָ֑יִן וּ֝לְרוֹזְנִ֗ים (או) אֵ֣י שֵׁכָֽר׃ (עח) פֶּן־ יִ֭שְׁתֶּה וְיִשְׁכַּ֣ח מְחֻקָּ֑ק וִֽ֝ישַׁנֶּה דִּ֣ין כָּל־ בְּנֵי־ עֹֽנִי׃ (עט) תְּנוּ־ שֵׁכָ֣ר לְאוֹבֵ֑ד וְ֝יַיִן לְמָ֣רֵי נָֽפֶשׁ׃ (פ) יִ֭שְׁתֶּה וְיִשְׁכַּ֣ח רִישׁ֑וֹ וַ֝עֲמָל֗וֹ לֹ֣א יִזְכָּר־ עֽוֹד׃ (פא) פְּתַח־ פִּ֥יךָ לְאִלֵּ֑ם אֶל־ דִּ֝֗ין כָּל־ בְּנֵ֥י חֲלֽוֹף׃ (פב) פְּתַח־ פִּ֥יךָ שְׁפָט־ צֶ֑דֶק וְ֝דִ֗ין עָנִ֥י וְאֶבְיֽוֹן׃ (פג) אֵֽשֶׁת־ חַ֭יִל מִ֣י יִמְצָ֑א וְרָחֹ֖ק מִפְּנִינִ֣ים מִכְרָֽהּ׃ (פד) בָּ֣טַח בָּ֭הּ לֵ֣ב בַּעְלָ֑הּ וְ֝שָׁלָ֗ל לֹ֣א יֶחְסָֽר׃ (פה) גְּמָלַ֣תְהוּ ט֣וֹב וְלֹא־ רָ֑ע כֹּ֝֗ל יְמֵ֣י חַיֶּֽיה׃ (פו) דָּ֭רְשָׁה צֶ֣מֶר וּפִשְׁתִּ֑ים וַ֝תַּ֗עַשׂ בְּחֵ֣פֶץ כַּפֶּֽיהָ׃ (פז) הָ֭יְתָה כָּאֳנִיּ֣וֹת סוֹחֵ֑ר מִ֝מֶּרְחָ֗ק תָּבִ֥יא לַחְמָֽהּ׃ (פח) וַתָּ֤קָם ׀ בְּע֬וֹד לַ֗יְלָה וַתִּתֵּ֣ן טֶ֣רֶף לְבֵיתָ֑הּ וְ֝חֹ֗ק לְנַעֲרֹתֶֽיהָ׃ (פט) זָמְמָ֣ה שָׂ֭דֶה וַתִּקָּחֵ֑הוּ מִפְּרִ֥י כַ֝פֶּ֗יהָ (נטע) נָ֣טְעָה כָּֽרֶם׃ (צ) חָֽגְרָ֣ה בְע֣וֹז מָתְנֶ֑יהָ וַ֝תְּאַמֵּ֗ץ זְרֹעוֹתֶֽיהָ׃ (צא) טָ֭עֲמָה כִּי־ ט֣וֹב סַחְרָ֑הּ לֹֽא־ יִכְבֶּ֖ה (בליל) בַלַּ֣יְלָה נֵרָֽהּ׃ (צב) יָ֭דֶיהָ שִׁלְּחָ֣ה בַכִּישׁ֑וֹר וְ֝כַפֶּ֗יהָ תָּ֣מְכוּ פָֽלֶךְ׃ (צג) כַּ֭פָּהּ פָּרְשָׂ֣ה לֶעָנִ֑י וְ֝יָדֶ֗יהָ שִׁלְּחָ֥ה לָֽאֶבְיֽוֹן׃ (צד) לֹא־ תִירָ֣א לְבֵיתָ֣הּ מִשָּׁ֑לֶג כִּ֥י כָל־ בֵּ֝יתָ֗הּ לָבֻ֥שׁ שָׁנִֽים׃ (צה) מַרְבַדִּ֥ים עָֽשְׂתָה־ לָּ֑הּ שֵׁ֖שׁ וְאַרְגָּמָ֣ן לְבוּשָֽׁהּ׃ (צו) נוֹדָ֣ע בַּשְּׁעָרִ֣ים בַּעְלָ֑הּ בְּ֝שִׁבְתּ֗וֹ עִם־ זִקְנֵי־ אָֽרֶץ׃ (צז) סָדִ֣ין עָ֭שְׂתָה וַתִּמְכֹּ֑ר וַ֝חֲג֗וֹר נָתְנָ֥ה לַֽכְּנַעֲנִֽי׃ (צח) עֹז־ וְהָדָ֥ר לְבוּשָׁ֑הּ וַ֝תִּשְׂחַ֗ק לְי֣וֹם אַחֲרֽוֹן׃ (צט) פִּ֭יהָ פָּתְחָ֣ה בְחָכְמָ֑ה וְתֽוֹרַת־ חֶ֝֗סֶד עַל־ לְשׁוֹנָֽהּ׃ (ק) צ֭וֹפִיָּה הֲלִיכ֣וֹת בֵּיתָ֑הּ וְלֶ֥חֶם עַ֝צְל֗וּת לֹ֣א תֹאכֵֽל׃ (קא) קָ֣מוּ בָ֭נֶיהָ וַֽיְאַשְּׁר֑וּהָ בַּ֝עְלָ֗הּ וַֽיְהַֽלְלָהּ׃ (קב) רַבּ֣וֹת בָּ֭נוֹת עָ֣שׂוּ חָ֑יִל וְ֝אַ֗תְּ עָלִ֥ית עַל־ כֻּלָּֽנָה׃ (קג) שֶׁ֣קֶר הַ֭חֵן וְהֶ֣בֶל הַיֹּ֑פִי אִשָּׁ֥ה יִרְאַת־ יְ֝הוָ֗ה הִ֣יא תִתְהַלָּֽל׃ (קד) תְּנוּ־ לָ֭הּ מִפְּרִ֣י יָדֶ֑יהָ וִֽיהַלְל֖וּהָ בַשְּׁעָרִ֣ים מַעֲשֶֽׂיהָ׃
