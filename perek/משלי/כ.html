משלי פרק כ לפי החלוקה לסדרים

פרק כ בספר משלי (30 פסוקים) נמצא במשלי סדר ה (פסוקים מה–עד).

משלי סדר ה

(א) מִגְדַּל־ עֹ֭ז שֵׁ֣ם יְהוָ֑ה בּֽוֹ־ יָר֖וּץ צַדִּ֣יק וְנִשְׂגָּֽב׃ (ב) ה֣וֹן עָ֭שִׁיר קִרְיַ֣ת עֻזּ֑וֹ וּכְחוֹמָ֥ה נִ֝שְׂגָּבָ֗ה בְּמַשְׂכִּיתֽוֹ׃ (ג) לִפְנֵי־ שֶׁ֭בֶר יִגְבַּ֣הּ לֵב־ אִ֑ישׁ וְלִפְנֵ֖י כָב֣וֹד עֲנָוָֽה׃ (ד) מֵשִׁ֣יב דָּ֭בָר בְּטֶ֣רֶם יִשְׁמָ֑ע אִוֶּ֥לֶת הִיא־ ל֝֗וֹ וּכְלִמָּֽה׃ (ה) רֽוּחַ־ אִ֭ישׁ יְכַלְכֵּ֣ל מַחֲלֵ֑הוּ וְר֥וּחַ נְ֝כֵאָ֗ה מִ֣י יִשָּׂאֶֽנָּה׃ (ו) לֵ֣ב נָ֭בוֹן יִקְנֶה־ דָּ֑עַת וְאֹ֥זֶן חֲ֝כָמִ֗ים תְּבַקֶּשׁ־ דָּֽעַת׃ (ז) מַתָּ֣ן אָ֭דָם יַרְחִ֣יב ל֑וֹ וְלִפְנֵ֖י גְדֹלִ֣ים יַנְחֶֽנּוּ׃ (ח) צַדִּ֣יק הָרִאשׁ֣וֹן בְּרִיב֑וֹ (יבא)־ וּבָֽא־ רֵ֝עֵ֗הוּ וַחֲקָרֽוֹ׃ (ט) מִ֭דְיָנִים יַשְׁבִּ֣ית הַגּוֹרָ֑ל וּבֵ֖ין עֲצוּמִ֣ים יַפְרִֽיד׃ (י) אָ֗ח נִפְשָׁ֥ע מִקִּרְיַת־ עֹ֑ז (ומדונים) וּ֝מִדְיָנִ֗ים כִּבְרִ֥יחַ אַרְמֽוֹן׃ (יא) מִפְּרִ֣י פִי־ אִ֭ישׁ תִּשְׂבַּ֣ע בִּטְנ֑וֹ תְּבוּאַ֖ת שְׂפָתָ֣יו יִשְׂבָּֽע׃ (יב) מָ֣וֶת וְ֭חַיִּים בְּיַד־ לָשׁ֑וֹן וְ֝אֹהֲבֶ֗יהָ יֹאכַ֥ל פִּרְיָֽהּ׃ (יג) מָצָ֣א אִ֭שָּׁה מָ֣צָא ט֑וֹב וַיָּ֥פֶק רָ֝צ֗וֹן מֵיְהוָֽה׃ (יד) תַּחֲנוּנִ֥ים יְדַבֶּר־ רָ֑שׁ וְ֝עָשִׁ֗יר יַעֲנֶ֥ה עַזּֽוֹת׃ (טו) אִ֣ישׁ רֵ֭עִים לְהִתְרֹעֵ֑עַ וְיֵ֥שׁ אֹ֝הֵ֗ב דָּבֵ֥ק מֵאָֽח׃ (טז) טֽוֹב־ רָ֭שׁ הוֹלֵ֣ךְ בְּתֻמּ֑וֹ מֵעִקֵּ֥שׁ שְׂ֝פָתָ֗יו וְה֣וּא כְסִֽיל׃ (יז) גַּ֤ם בְּלֹא־ דַ֣עַת נֶ֣פֶשׁ לֹא־ ט֑וֹב וְאָ֖ץ בְּרַגְלַ֣יִם חוֹטֵֽא׃ (יח) אִוֶּ֣לֶת אָ֭דָם תְּסַלֵּ֣ף דַּרְכּ֑וֹ וְעַל־ יְ֝הוָ֗ה יִזְעַ֥ף לִבּֽוֹ׃ (יט) ה֗וֹן יֹ֭סִיף רֵעִ֣ים רַבִּ֑ים וְ֝דָ֗ל מֵרֵ֥עהוּ יִפָּרֵֽד׃ (כ) עֵ֣ד שְׁ֭קָרִים לֹ֣א יִנָּקֶ֑ה וְיָפִ֥יחַ כְּ֝זָבִ֗ים לֹ֣א יִמָּלֵֽט׃ (כא) רַ֭בִּים יְחַלּ֣וּ פְנֵֽי־ נָדִ֑יב וְכָל־ הָ֝רֵ֗עַ לְאִ֣ישׁ מַתָּֽן׃ (כב) כָּ֥ל אֲחֵי־ רָ֨שׁ ׀ שְֽׂנֵאֻ֗הוּ אַ֤ף כִּ֣י מְ֭רֵעֵהוּ רָחֲק֣וּ מִמֶּ֑נּוּ מְרַדֵּ֖ף אֲמָרִ֣ים (לא)־ לוֹ־ הֵֽמָּה׃ (כג) קֹֽנֶה־ לֵּ֭ב אֹהֵ֣ב נַפְשׁ֑וֹ שֹׁמֵ֥ר תְּ֝בוּנָ֗ה לִמְצֹא־ טֽוֹב׃ (כד) עֵ֣ד שְׁ֭קָרִים לֹ֣א יִנָּקֶ֑ה וְיָפִ֖יחַ כְּזָבִ֣ים יֹאבֵֽד׃ (כה) לֹֽא־ נָאוֶ֣ה לִכְסִ֣יל תַּעֲנ֑וּג אַ֝֗ף כִּֽי־ לְעֶ֤בֶד ׀ מְשֹׁ֬ל בְּשָׂרִֽים׃ (כו) שֵׂ֣כֶל אָ֭דָם הֶאֱרִ֣יךְ אַפּ֑וֹ וְ֝תִפאַרְתּ֗וֹ עֲבֹ֣ר עַל־ פָּֽשַׁע׃ (כז) נַ֣הַם כַּ֭כְּפִיר זַ֣עַף מֶ֑לֶךְ וּכְטַ֖ל עַל־ עֵ֣שֶׂב רְצוֹנֽוֹ׃ (כח) הַוֺּ֣ת לְ֭אָבִיו בֵּ֣ן כְּסִ֑יל וְדֶ֥לֶף טֹ֝רֵ֗ד מִדְיְנֵ֥י אִשָּֽׁה׃ (כט) בַּ֣יִת וָ֭הוֹן נַחֲלַ֣ת אָב֑וֹת וּ֝מֵיְהוָ֗ה אִשָּׁ֥ה מַשְׂכָּֽלֶת׃ (ל) עַ֭צְלָה תַּפִּ֣יל תַּרְדֵּמָ֑ה וְנֶ֖פֶשׁ רְמִיָּ֣ה תִרְעָֽב׃ (לא) שֹׁמֵ֣ר מִ֭צְוָה שֹׁמֵ֣ר נַפְשׁ֑וֹ בּוֹזֵ֖ה דְרָכָ֣יו (יומת) יָמֽוּת׃ (לב) מַלְוֵ֣ה יְ֭הוָה ח֣וֹנֵֽן דָּ֑ל וּ֝גְמֻל֗וֹ יְשַׁלֶּם־ לֽוֹ׃ (לג) יַסֵּ֣ר בִּ֭נְךָ כִּי־ יֵ֣שׁ תִּקְוָ֑ה וְאֶל־ הֲ֝מִית֗וֹ אַל־ תִּשָּׂ֥א נַפְשֶֽׁךָ׃ (לד) (גרל)־ גְּֽדָל־ חֵ֭מָה נֹ֣שֵׂא עֹ֑נֶשׁ כִּ֥י אִם־ תַּ֝צִּ֗יל וְע֣וֹד תּוֹסִֽף׃ (לה) שְׁמַ֣ע עֵ֭צָה וְקַבֵּ֣ל מוּסָ֑ר לְ֝מַ֗עַן תֶּחְכַּ֥ם בְּאַחֲרִיתֶֽךָ׃ (לו) רַבּ֣וֹת מַחֲשָׁב֣וֹת בְּלֶב־ אִ֑ישׁ וַעֲצַ֥ת יְ֝הוָ֗ה הִ֣יא תָקֽוּם׃ (לז) תַּאֲוַ֣ת אָדָ֣ם חַסְדּ֑וֹ וְטֽוֹב־ רָ֝שׁ מֵאִ֥ישׁ כָּזָֽב׃ (לח) יִרְאַ֣ת יְהוָ֣ה לְחַיִּ֑ים וְשָׂבֵ֥עַ יָ֝לִ֗ין בַּל־ יִפָּ֥קֶד רָֽע׃ (לט) טָ֘מַ֤ן עָצֵ֣ל יָ֭דוֹ בַּצַּלָּ֑חַת גַּם־ אֶל־ פִּ֝֗יהוּ לֹ֣א יְשִׁיבֶֽנָּה׃ (מ) לֵ֣ץ תַּ֭כֶּה וּפֶ֣תִי יַעְרִ֑ם וְהוֹכִ֥יחַ לְ֝נָב֗וֹן יָבִ֥ין דָּֽעַת׃ (מא) מְֽשַׁדֶּד־ אָ֭ב יַבְרִ֣יחַ אֵ֑ם בֵּ֝֗ן מֵבִ֥ישׁ וּמַחְפִּֽיר׃ (מב) חַֽדַל־ בְּ֭נִי לִשְׁמֹ֣עַ מוּסָ֑ר לִ֝שְׁג֗וֹת מֵֽאִמְרֵי־ דָֽעַת׃ (מג) עֵ֣ד בְּ֭לִיַּעַל יָלִ֣יץ מִשְׁפָּ֑ט וּפִ֥י רְ֝שָׁעִ֗ים יְבַלַּע־ אָֽוֶן׃ (מד) נָכ֣וֹנוּ לַלֵּצִ֣ים שְׁפָטִ֑ים וּ֝מַהֲלֻמ֗וֹת לְגֵ֣ו כְּסִילִֽים׃ תחילת פרק כ (מה) לֵ֣ץ הַ֭יַּין הֹמֶ֣ה שֵׁכָ֑ר וְכָל־ שֹׁ֥גֶה בּ֝֗וֹ לֹ֣א יֶחְכָּֽם׃ (מו) נַ֣הַם כַּ֭כְּפִיר אֵ֣ימַת מֶ֑לֶךְ מִ֝תְעַבְּר֗וֹ חוֹטֵ֥א נַפְשֽׁוֹ׃ (מז) כָּב֣וֹד לָ֭אִישׁ שֶׁ֣בֶת מֵרִ֑יב וְכָל־ אֱ֝וִ֗יל יִתְגַּלָּֽע׃ (מח) מֵ֭חֹרֶף עָצֵ֣ל לֹא־ יַחֲרֹ֑שׁ (ישאל) וְשָׁאַ֖ל בַּקָּצִ֣יר וָאָֽיִן׃ (מט) מַ֣יִם עֲ֭מֻקִּים עֵצָ֣ה בְלֶב־ אִ֑ישׁ וְאִ֖ישׁ תְּבוּנָ֣ה יִדְלֶֽנָּה׃ (נ) רָב־ אָדָ֗ם יִ֭קְרָא אִ֣ישׁ חַסְדּ֑וֹ וְאִ֥ישׁ אֱ֝מוּנִ֗ים מִ֣י יִמְצָֽא׃ (נא) מִתְהַלֵּ֣ךְ בְּתֻמּ֣וֹ צַדִּ֑יק אַשְׁרֵ֖י בָנָ֣יו אַחֲרָֽיו׃ (נב) מֶ֗לֶךְ יוֹשֵׁ֥ב עַל־ כִּסֵּא־ דִ֑ין מְזָרֶ֖ה בְעֵינָ֣יו כָּל־ רָֽע׃ (נג) מִֽי־ יֹ֭אמַר זִכִּ֣יתִי לִבִּ֑י טָ֝הַ֗רְתִּי מֵחַטָּאתִֽי׃ (נד) אֶ֣בֶן וָ֭אֶבֶן אֵיפָ֣ה וְאֵיפָ֑ה תּוֹעֲבַ֥ת יְ֝הוָ֗ה גַּם־ שְׁנֵיהֶֽם׃ (נה) גַּ֣ם בְּ֭מַעֲלָלָיו יִתְנַכֶּר־ נָ֑עַר אִם־ זַ֖ךְ וְאִם־ יָשָׁ֣ר פָּעֳלֽוֹ׃ (נו) אֹ֣זֶן שֹׁ֭מַעַת וְעַ֣יִן רֹאָ֑ה יְ֝הוָ֗ה עָשָׂ֥ה גַם־ שְׁנֵיהֶֽם׃ (נז) אַל־ תֶּֽאֱהַ֣ב שֵׁ֭נָה פֶּן־ תִּוָּרֵ֑שׁ פְּקַ֖ח עֵינֶ֣יךָ שְֽׂבַֽע־ לָֽחֶם׃ (נח) רַ֣ע רַ֭ע יֹאמַ֣ר הַקּוֹנֶ֑ה וְאֹזֵ֥ל ל֝֗וֹ אָ֣ז יִתְהַלָּֽל׃ (נט) יֵ֣שׁ זָ֭הָב וְרָב־ פְּנִינִ֑ים וּכְלִ֥י יְ֝קָ֗ר שִׂפְתֵי־ דָֽעַת׃ (ס) לְֽקַח־ בִּ֭גְדוֹ כִּי־ עָ֣רַב זָ֑ר וּבְעַ֖ד (נכרים) נָכְרִיָּ֣ה חַבְלֵֽהוּ׃ (סא) עָרֵ֣ב לָ֭אִישׁ לֶ֣חֶם שָׁ֑קֶר וְ֝אַחַ֗ר יִמָּֽלֵא־ פִ֥יהוּ חָצָֽץ׃ (סב) מַ֭חֲשָׁבוֹת בְּעֵצָ֣ה תִכּ֑וֹן וּ֝בְתַחְבֻּל֗וֹת עֲשֵׂ֣ה מִלְחָמָֽה׃ (סג) גּֽוֹלֶה־ סּ֭וֹד הוֹלֵ֣ךְ רָכִ֑יל וּלְפֹתֶ֥ה שְׂ֝פָתָ֗יו לֹ֣א תִתְעָרָֽב׃ (סד) מְ֭קַלֵּל אָבִ֣יו וְאִמּ֑וֹ יִֽדְעַ֥ךְ נֵ֝ר֗וֹ (באישון) בֶּאֱשׁ֥וּן חֹֽשֶׁךְ׃ (סה) נַ֭חֲלָה (מבחלת) מְבֹהֶ֣לֶת בָּרִאשֹׁנָ֑ה וְ֝אַחֲרִיתָ֗הּ לֹ֣א תְבֹרָֽךְ׃ (סו) אַל־ תֹּאמַ֥ר אֲשַׁלְּמָה־ רָ֑ע קַוֵּ֥ה לַֽ֝יהוָ֗ה וְיֹ֣שַֽׁע לָֽךְ׃ (סז) תּוֹעֲבַ֣ת יְ֭הוָה אֶ֣בֶן וָאָ֑בֶן וּמֹאזְנֵ֖י מִרְמָ֣ה לֹא־ טֽוֹב׃ (סח) מֵיהוָ֥ה מִצְעֲדֵי־ גָ֑בֶר וְ֝אָדָ֗ם מַה־ יָּבִ֥ין דַּרְכּֽוֹ׃ (סט) מוֹקֵ֣שׁ אָ֭דָם יָ֣לַע קֹ֑דֶשׁ וְאַחַ֖ר נְדָרִ֣ים לְבַקֵּֽר׃ (ע) מְזָרֶ֣ה רְ֭שָׁעִים מֶ֣לֶךְ חָכָ֑ם וַיָּ֖שֶׁב עֲלֵיהֶ֣ם אוֹפָֽן׃ (עא) נֵ֣ר יְ֭הוָה נִשְׁמַ֣ת אָדָ֑ם חֹ֝פֵ֗שׂ כָּל־ חַדְרֵי־ בָֽטֶן׃ (עב) חֶ֣סֶד וֶ֭אֱמֶת יִצְּרוּ־ מֶ֑לֶךְ וְסָעַ֖ד בַּחֶ֣סֶד כִּסְאֽוֹ׃ (עג) תִּפְאֶ֣רֶת בַּחוּרִ֣ים כֹּחָ֑ם וַהֲדַ֖ר זְקֵנִ֣ים שֵׂיבָֽה׃ (עד) חַבֻּר֣וֹת פֶּ֭צַע (תמריק) תַּמְר֣וּק בְּרָ֑ע וּ֝מַכּ֗וֹת חַדְרֵי־ בָֽטֶן׃ סוף פרק כ (עה) פַּלְגֵי־ מַ֣יִם לֶב־ מֶ֭לֶךְ בְּיַד־ יְהוָ֑ה עַֽל־ כָּל־ אֲשֶׁ֖ר יַחְפֹּ֣ץ יַטֶּֽנּוּ׃ (עו) כָּֽל־ דֶּרֶךְ־ אִ֭ישׁ יָשָׁ֣ר בְּעֵינָ֑יו וְתֹכֵ֖ן לִבּ֣וֹת יְהוָֽה׃ (עז) עֲ֭שֹׂה צְדָקָ֣ה וּמִשְׁפָּ֑ט נִבְחָ֖ר לַיהוָ֣ה מִזָּֽבַח׃ (עח) רוּם־ עֵ֭ינַיִם וּרְחַב־ לֵ֑ב נִ֖ר רְשָׁעִ֣ים חַטָּֽאת׃ (עט) מַחְשְׁב֣וֹת חָ֭רוּץ אַךְ־ לְמוֹתָ֑ר וְכָל־ אָ֝֗ץ אַךְ־ לְמַחְסֽוֹר׃ (פ) פֹּ֣עַל א֭וֹצָרוֹת בִּלְשׁ֣וֹן שָׁ֑קֶר הֶ֥בֶל נִ֝דָּ֗ף מְבַקְשֵׁי־ מָֽוֶת׃ (פא) שֹׁד־ רְשָׁעִ֥ים יְגוֹרֵ֑ם כִּ֥י מֵ֝אֲנ֗וּ לַעֲשׂ֥וֹת מִשְׁפָּֽט׃ (פב) הֲפַכְפַּ֬ךְ דֶּ֣רֶךְ אִ֣ישׁ וָזָ֑ר וְ֝זַ֗ךְ יָשָׁ֥ר פָּעֳלֽוֹ׃ (פג) ט֗וֹב לָשֶׁ֥בֶת עַל־ פִּנַּת־ גָּ֑ג מֵאֵ֥שֶׁת מִ֝דְיָנִ֗ים וּבֵ֥ית חָֽבֶר׃ (פד) נֶ֣פֶשׁ רָ֭שָׁע אִוְּתָה־ רָ֑ע לֹא־ יֻחַ֖ן בְּעֵינָ֣יו רֵעֵֽהוּ׃ (פה) בַּעְנָשׁ־ לֵ֭ץ יֶחְכַּם־ פֶּ֑תִי וּבְהַשְׂכִּ֥יל לְ֝חָכָ֗ם יִקַּח־ דָּֽעַת׃ (פו) מַשְׂכִּ֣יל צַ֭דִּיק לְבֵ֣ית רָשָׁ֑ע מְסַלֵּ֖ף רְשָׁעִ֣ים לָרָֽע׃ (פז) אֹטֵ֣ם אָ֭זְנוֹ מִזַּעֲקַת־ דָּ֑ל גַּֽם־ ה֥וּא יִ֝קְרָ֗א וְלֹ֣א יֵעָנֶֽה׃ (פח) מַתָּ֣ן בַּ֭סֵּתֶר יִכְפֶּה־ אָ֑ף וְשֹׁ֥חַד בַּ֝חֵ֗ק חֵמָ֥ה עַזָּֽה׃ (פט) שִׂמְחָ֣ה לַ֭צַּדִּיק עֲשׂ֣וֹת מִשְׁפָּ֑ט וּ֝מְחִתָּ֗ה לְפֹ֣עֲלֵי אָֽוֶן׃ (צ) אָדָ֗ם תּ֭וֹעֶה מִדֶּ֣רֶךְ הַשְׂכֵּ֑ל בִּקְהַ֖ל רְפָאִ֣ים יָנֽוּחַ׃ (צא) אִ֣ישׁ מַ֭חְסוֹר אֹהֵ֣ב שִׂמְחָ֑ה אֹהֵ֥ב יַֽיִן־ וָ֝שֶׁ֗מֶן לֹ֣א יַעֲשִֽׁיר׃ (צב) כֹּ֣פֶר לַצַּדִּ֣יק רָשָׁ֑ע וְתַ֖חַת יְשָׁרִ֣ים בּוֹגֵֽד׃ (צג) ט֗וֹב שֶׁ֥בֶת בְּאֶֽרֶץ־ מִדְבָּ֑ר מֵאֵ֖שֶׁת (מדונים) מִדְיָנִ֣ים וָכָֽעַס׃ (צד) אוֹצָ֤ר ׀ נֶחְמָ֣ד וָ֭שֶׁמֶן בִּנְוֵ֣ה חָכָ֑ם וּכְסִ֖יל אָדָ֣ם יְבַלְּעֶֽנּוּ׃ (צה) רֹ֭דֵף צְדָקָ֣ה וָחָ֑סֶד יִמְצָ֥א חַ֝יִּ֗ים צְדָקָ֥ה וְכָבֽוֹד׃ (צו) עִ֣יר גִּ֭בֹּרִים עָלָ֣ה חָכָ֑ם וַ֝יֹּ֗רֶד עֹ֣ז מִבְטֶחָֽה׃ (צז) שֹׁמֵ֣ר פִּ֭יו וּלְשׁוֹנ֑וֹ שֹׁמֵ֖ר מִצָּר֣וֹת נַפְשֽׁוֹ׃ (צח) זֵ֣ד יָ֭הִיר לֵ֣ץ שְׁמ֑וֹ ע֝וֹשֶׂ֗ה בְּעֶבְרַ֥ת זָדֽוֹן׃ (צט) תַּאֲוַ֣ת עָצֵ֣ל תְּמִיתֶ֑נּוּ כִּֽי־ מֵאֲנ֖וּ יָדָ֣יו לַעֲשֽׂוֹת׃ (ק) כָּל־ הַ֭יּוֹם הִתְאַוָּ֣ה תַאֲוָ֑ה וְצַדִּ֥יק יִ֝תֵּ֗ן וְלֹ֣א יַחְשֹֽׂךְ׃ (קא) זֶ֣בַח רְ֭שָׁעִים תּוֹעֵבָ֑ה אַ֝֗ף כִּֽי־ בְזִמָּ֥ה יְבִיאֶֽנּוּ׃ (קב) עֵד־ כְּזָבִ֥ים יֹאבֵ֑ד וְאִ֥ישׁ שׁ֝וֹמֵ֗עַ לָנֶ֥צַח יְדַבֵּֽר׃ (קג) הֵעֵ֬ז אִ֣ישׁ רָשָׁ֣ע בְּפָנָ֑יו וְ֝יָשָׁ֗ר ה֤וּא ׀ (יכין) יָבִ֬ין (דרכיו) דַּרְכּֽוֹ׃ (קד) אֵ֣ין חָ֭כְמָה וְאֵ֣ין תְּבוּנָ֑ה וְאֵ֥ין עֵ֝צָ֗ה לְנֶ֣גֶד יְהוָֽה׃ (קה) ס֗וּס מ֭וּכָן לְי֣וֹם מִלְחָמָ֑ה וְ֝לַֽיהוָ֗ה הַתְּשׁוּעָֽה׃ (קו) נִבְחָ֣ר שֵׁ֭ם מֵעֹ֣שֶׁר רָ֑ב מִכֶּ֥סֶף וּ֝מִזָּהָ֗ב חֵ֣ן טֽוֹב׃ (קז) עָשִׁ֣יר וָרָ֣שׁ נִפְגָּ֑שׁוּ עֹשֵׂ֖ה כֻלָּ֣ם יְהוָֽה׃ (קח) עָר֤וּם ׀ רָאָ֣ה רָעָ֣ה (ויסתר) וְנִסְתָּ֑ר וּ֝פְתָיִ֗ים עָבְר֥וּ וְֽנֶעֱנָֽשׁוּ׃ (קט) עֵ֣קֶב עֲ֭נָוָה יִרְאַ֣ת יְהוָ֑ה עֹ֖שֶׁר וְכָב֣וֹד וְחַיִּֽים׃ (קי) צִנִּ֣ים פַּ֭חִים בְּדֶ֣רֶךְ עִקֵּ֑שׁ שׁוֹמֵ֥ר נַ֝פְשׁ֗וֹ יִרְחַ֥ק מֵהֶֽם׃ (קיא) חֲנֹ֣ךְ לַ֭נַּעַר עַל־ פִּ֣י דַרְכּ֑וֹ גַּ֥ם כִּֽי־ יַ֝זְקִ֗ין לֹֽא־ יָס֥וּר מִמֶּֽנָּה׃ (קיב) עָ֭שִׁיר בְּרָשִׁ֣ים יִמְשׁ֑וֹל וְעֶ֥בֶד לֹ֝וֶ֗ה לְאִ֣ישׁ מַלְוֶֽה׃ (קיג) זוֹרֵ֣עַ עַ֭וְלָה (יקצור)־ יִקְצָר־ אָ֑וֶן וְשֵׁ֖בֶט עֶבְרָת֣וֹ יִכְלֶֽה׃ (קיד) טֽוֹב־ עַ֭יִן ה֣וּא יְבֹרָ֑ךְ כִּֽי־ נָתַ֖ן מִלַּחְמ֣וֹ לַדָּֽל׃ (קטו) גָּ֣רֵֽשׁ לֵ֭ץ וְיֵצֵ֣א מָד֑וֹן וְ֝יִשְׁבֹּ֗ת דִּ֣ין וְקָלֽוֹן׃ (קטז) אֹהֵ֥ב (טהור)־ טְהָר־ לֵ֑ב חֵ֥ן שְׂ֝פָתָ֗יו רֵעֵ֥הוּ מֶֽלֶךְ׃ (קיז) עֵינֵ֣י יְ֭הוָה נָ֣צְרוּ דָ֑עַת וַ֝יְסַלֵּ֗ף דִּבְרֵ֥י בֹגֵֽד׃ (קיח) אָמַ֣ר עָ֭צֵל אֲרִ֣י בַח֑וּץ בְּת֥וֹךְ רְ֝חֹב֗וֹת אֵֽרָצֵֽחַ׃ (קיט) שׁוּחָ֣ה עֲ֭מֻקָּה פִּ֣י זָר֑וֹת זְע֥וּם יְ֝הוָ֗ה (יפול)־ יִפָּל־ שָֽׁם׃ (קכ) אִ֭וֶּלֶת קְשׁוּרָ֣ה בְלֶב־ נָ֑עַר שֵׁ֥בֶט מ֝וּסָ֗ר יַרְחִיקֶ֥נָּה מִמֶּֽנּוּ׃ (קכא) עֹ֣שֵֽׁק דָּ֭ל לְהַרְבּ֣וֹת ל֑וֹ נֹתֵ֥ן לְ֝עָשִׁ֗יר אַךְ־ לְמַחְסֽוֹר׃ (קכב) הַ֥ט אָזְנְךָ֗ וּ֭שְׁמַע דִּבְרֵ֣י חֲכָמִ֑ים וְ֝לִבְּךָ֗ תָּשִׁ֥ית לְדַעְתִּֽי׃ (קכג) כִּֽי־ נָ֭עִים כִּֽי־ תִשְׁמְרֵ֣ם בְּבִטְנֶ֑ךָ יִכֹּ֥נוּ יַ֝חְדָּ֗ו עַל־ שְׂפָתֶֽיךָ׃ (קכד) לִהְי֣וֹת בַּ֭יהוָה מִבְטַחֶ֑ךָ הוֹדַעְתִּ֖יךָ הַיּ֣וֹם אַף־ אָֽתָּה׃ (קכה) הֲלֹ֤א כָתַ֣בְתִּי לְ֭ךָ (שלשום) שָׁלִישִׁ֑ים בְּמ֖וֹעֵצֹ֣ת וָדָֽעַת׃
