משלי פרק ד לפי החלוקה לסדרים

פרק ד בספר משלי (27 פסוקים) נמצא במשלי סדר א (פסוקים צא–קיז).

משלי סדר א

(א) מִ֭שְׁלֵי שְׁלֹמֹ֣ה בֶן־ דָּוִ֑ד מֶ֝֗לֶךְ יִשְׂרָאֵֽל׃ (ב) לָדַ֣עַת חָכְמָ֣ה וּמוּסָ֑ר לְ֝הָבִ֗ין אִמְרֵ֥י בִינָֽה׃ (ג) לָ֭קַחַת מוּסַ֣ר הַשְׂכֵּ֑ל צֶ֥דֶק וּ֝מִשְׁפָּ֗ט וּמֵישָׁרִֽים׃ (ד) לָתֵ֣ת לִפְתָאיִ֣ם עָרְמָ֑ה לְ֝נַ֗עַר דַּ֣עַת וּמְזִמָּֽה׃ (ה) יִשְׁמַ֣ע חָ֭כָם וְי֣וֹסֶף לֶ֑קַח וְ֝נָב֗וֹן תַּחְבֻּל֥וֹת יִקְנֶֽה׃ (ו) לְהָבִ֣ין מָ֭שָׁל וּמְלִיצָ֑ה דִּבְרֵ֥י חֲ֝כָמִ֗ים וְחִידֹתָֽם׃ (ז) יִרְאַ֣ת יְ֭הוָה רֵאשִׁ֣ית דָּ֑עַת חָכְמָ֥ה וּ֝מוּסָ֗ר אֱוִילִ֥ים בָּֽזוּ׃ (ח) שְׁמַ֣ע בְּ֭נִי מוּסַ֣ר אָבִ֑יךָ וְאַל־ תִּ֝טֹּ֗שׁ תּוֹרַ֥ת אִמֶּֽךָ׃ (ט) כִּ֤י ׀ לִוְיַ֤ת חֵ֓ן הֵ֬ם לְרֹאשֶׁ֑ךָ וַ֝עֲנָקִ֗ים לְגַרְגְּרֹתֶֽיךָ׃ (י) בְּנִ֡י אִם־ יְפַתּ֥וּךָ חַ֝טָּאִ֗ים אַל־ תֹּבֵֽא׃ (יא) אִם־ יֹאמְרוּ֮ לְכָ֪ה אִ֫תָּ֥נוּ נֶאֶרְבָ֥ה לְדָ֑ם נִצְפְּנָ֖ה לְנָקִ֣י חִנָּֽם׃ (יב) נִ֭בְלָעֵם כִּשְׁא֣וֹל חַיִּ֑ים וּ֝תְמִימִ֗ים כְּי֣וֹרְדֵי בֽוֹר׃ (יג) כָּל־ ה֣וֹן יָקָ֣ר נִמְצָ֑א נְמַלֵּ֖א בָתֵּ֣ינוּ שָׁלָֽל׃ (יד) גּ֭וֹרָ֣לְךָ תַּפִּ֣יל בְּתוֹכֵ֑נוּ כִּ֥יס אֶ֝חָ֗ד יִהְיֶ֥ה לְכֻלָּֽנוּ׃ (טו) בְּנִ֗י אַל־ תֵּלֵ֣ךְ בְּדֶ֣רֶךְ אִתָּ֑ם מְנַ֥ע רַ֝גְלְךָ֗ מִנְּתִיבָתָֽם׃ (טז) כִּ֣י רַ֭גְלֵיהֶם לָרַ֣ע יָר֑וּצוּ וִֽ֝ימַהֲר֗וּ לִשְׁפָּךְ־ דָּֽם׃ (יז) כִּֽי־ חִ֭נָּם מְזֹרָ֣ה הָרָ֑שֶׁת בְּ֝עֵינֵ֗י כָל־ בַּ֥עַל כָּנָֽף׃ (יח) וְ֭הֵם לְדָמָ֣ם יֶאֱרֹ֑בוּ יִ֝צְפְּנ֗וּ לְנַפְשֹׁתָֽם׃ (יט) כֵּ֗ן אָ֭רְחוֹת כָּל־ בֹּ֣צֵֽעַ בָּ֑צַע אֶת־ נֶ֖פֶשׁ בְּעָלָ֣יו יִקָּֽח׃ (כ) חָ֭כְמוֹת בַּח֣וּץ תָּרֹ֑נָּה בָּ֝רְחֹב֗וֹת תִּתֵּ֥ן קוֹלָֽהּ׃ (כא) בְּרֹ֥אשׁ הֹמִיּ֗וֹת תִּ֫קְרָ֥א בְּפִתְחֵ֖י שְׁעָרִ֥ים בָּעִ֗יר אֲמָרֶ֥יהָ תֹאמֵֽר׃ (כב) עַד־ מָתַ֣י ׀ פְּתָיִם֮ תְּֽאֵהֲב֫וּ פֶ֥תִי וְלֵצִ֗ים לָ֭צוֹן חָמְד֣וּ לָהֶ֑ם וּ֝כְסִילִ֗ים יִשְׂנְאוּ־ דָֽעַת׃ (כג) תָּשׁ֗וּבוּ לְֽת֫וֹכַחְתִּ֥י הִנֵּ֤ה אַבִּ֣יעָה לָכֶ֣ם רוּחִ֑י אוֹדִ֖יעָה דְבָרַ֣י אֶתְכֶֽם׃ (כד) יַ֣עַן קָ֭רָאתִי וַתְּמָאֵ֑נוּ נָטִ֥יתִי יָ֝דִ֗י וְאֵ֣ין מַקְשִֽׁיב׃ (כה) וַתִּפְרְע֥וּ כָל־ עֲצָתִ֑י וְ֝תוֹכַחְתִּ֗י לֹ֣א אֲבִיתֶֽם׃ (כו) גַּם־ אֲ֭נִי בְּאֵידְכֶ֣ם אֶשְׂחָ֑ק אֶ֝לְעַ֗ג בְּבֹ֣א פַחְדְּכֶֽם׃ (כז) בְּבֹ֤א (כשאוה) כְשׁוֹאָ֨ה ׀ פַּחְדְּכֶ֗ם וְֽ֭אֵידְכֶם כְּסוּפָ֣ה יֶאֱתֶ֑ה בְּבֹ֥א עֲ֝לֵיכֶ֗ם צָרָ֥ה וְצוּקָֽה׃ (כח) אָ֣ז יִ֭קְרָאֻנְנִי וְלֹ֣א אֶֽעֱנֶ֑ה יְ֝שַׁחֲרֻ֗נְנִי וְלֹ֣א יִמְצָאֻֽנְנִי׃ (כט) תַּ֭חַת כִּי־ שָׂ֣נְאוּ דָ֑עַת וְיִרְאַ֥ת יְ֝הֹוָ֗ה לֹ֣א בָחָֽרוּ׃ (ל) לֹא־ אָב֥וּ לַעֲצָתִ֑י נָ֝אֲצ֗וּ כָּל־ תּוֹכַחְתִּֽי׃ (לא) וְֽ֭יֹאכְלוּ מִפְּרִ֣י דַרְכָּ֑ם וּֽמִמֹּעֲצֹ֖תֵיהֶ֣ם יִשְׂבָּֽעוּ׃ (לב) כִּ֤י מְשׁוּבַ֣ת פְּתָיִ֣ם תַּֽהַרְגֵ֑ם וְשַׁלְוַ֖ת כְּסִילִ֣ים תְּאַבְּדֵֽם׃ (לג) וְשֹׁמֵ֣עַֽ לִ֭י יִשְׁכָּן־ בֶּ֑טַח וְ֝שַׁאֲנַ֗ן מִפַּ֥חַד רָעָֽה׃ (לד) בְּ֭נִי אִם־ תִּקַּ֣ח אֲמָרָ֑י וּ֝מִצְוֺתַ֗י תִּצְפֹּ֥ן אִתָּֽךְ׃ (לה) לְהַקְשִׁ֣יב לַֽחָכְמָ֣ה אָזְנֶ֑ךָ תַּטֶּ֥ה לִ֝בְּךָ֗ לַתְּבוּנָֽה׃ (לו) כִּ֤י אִ֣ם לַבִּינָ֣ה תִקְרָ֑א לַ֝תְּבוּנָ֗ה תִּתֵּ֥ן קוֹלֶֽךָ׃ (לז) אִם־ תְּבַקְשֶׁ֥נָּה כַכָּ֑סֶף וְֽכַמַּטְמוֹנִ֥ים תַּחְפְּשֶֽׂנָּה׃ (לח) אָ֗ז תָּ֭בִין יִרְאַ֣ת יְהוָ֑ה וְדַ֖עַת אֱלֹהִ֣ים תִּמְצָֽא׃ (לט) כִּֽי־ יְ֭הוָה יִתֵּ֣ן חָכְמָ֑ה מִ֝פִּ֗יו דַּ֣עַת וּתְבוּנָֽה׃ (מ) (וצפן) יִצְפֹּ֣ן לַ֭יְשָׁרִים תּוּשִׁיָּ֑ה מָ֝גֵ֗ן לְהֹ֣לְכֵי תֹֽם׃ (מא) לִ֭נְצֹר אָרְח֣וֹת מִשְׁפָּ֑ט וְדֶ֖רֶךְ (חסידו) חֲסִידָ֣יו יִשְׁמֹֽר׃ (מב) אָ֗ז תָּ֭בִין צֶ֣דֶק וּמִשְׁפָּ֑ט וּ֝מֵישָׁרִ֗ים כָּל־ מַעְגַּל־ טֽוֹב׃ (מג) כִּֽי־ תָב֣וֹא חָכְמָ֣ה בְלִבֶּ֑ךָ וְ֝דַ֗עַת לְֽנַפְשְׁךָ֥ יִנְעָֽם׃ (מד) מְ֭זִמָּה תִּשְׁמֹ֥ר עָלֶ֗יךָ תְּבוּנָ֥ה תִנְצְרֶֽכָּה׃ (מה) לְ֭הַצִּ֣ילְךָ מִדֶּ֣רֶךְ רָ֑ע מֵ֝אִ֗ישׁ מְדַבֵּ֥ר תַּהְפֻּכֽוֹת׃ (מו) הַ֭עֹ֣זְבִים אָרְח֣וֹת יֹ֑שֶׁר לָ֝לֶ֗כֶת בְּדַרְכֵי־ חֹֽשֶׁךְ׃ (מז) הַ֭שְּׂמֵחִים לַעֲשׂ֥וֹת רָ֑ע יָ֝גִ֗ילוּ בְּֽתַהְפֻּכ֥וֹת רָֽע׃ (מח) אֲשֶׁ֣ר אָרְחֹתֵיהֶ֣ם עִקְּשִׁ֑ים וּ֝נְלוֹזִ֗ים בְּמַעְגְּלוֹתָֽם׃ (מט) לְ֭הַצִּ֣ילְךָ מֵאִשָּׁ֣ה זָרָ֑ה מִ֝נָּכְרִיָּ֗ה אֲמָרֶ֥יהָ הֶחֱלִֽיקָה׃ (נ) הַ֭עֹזֶבֶת אַלּ֣וּף נְעוּרֶ֑יהָ וְאֶת־ בְּרִ֖ית אֱלֹהֶ֣יהָ שָׁכֵֽחָה׃ (נא) כִּ֤י שָׁ֣חָה אֶל־ מָ֣וֶת בֵּיתָ֑הּ וְאֶל־ רְ֝פָאִ֗ים מַעְגְּלֹתֶֽיהָ׃ (נב) כָּל־ בָּ֭אֶיהָ לֹ֣א יְשׁוּב֑וּן וְלֹֽא־ יַ֝שִּׂ֗יגוּ אָרְח֥וֹת חַיִּֽים׃ (נג) לְמַ֗עַן תֵּ֭לֵךְ בְּדֶ֣רֶךְ טוֹבִ֑ים וְאָרְח֖וֹת צַדִּיקִ֣ים תִּשְׁמֹֽר׃ (נד) כִּֽי־ יְשָׁרִ֥ים יִשְׁכְּנוּ אָ֑רֶץ וּ֝תְמִימִ֗ים יִוָּ֥תְרוּ בָֽהּ׃ (נה) וּ֭רְשָׁעִים מֵאֶ֣רֶץ יִכָּרֵ֑תוּ וּ֝בוֹגְדִ֗ים יִסְּח֥וּ מִמֶּֽנָּה׃ (נו) בְּ֭נִי תּוֹרָתִ֣י אַל־ תִּשְׁכָּ֑ח וּ֝מִצְוֺתַ֗י יִצֹּ֥ר לִבֶּֽךָ׃ (נז) כִּ֤י אֹ֣רֶךְ יָ֭מִים וּשְׁנ֣וֹת חַיִּ֑ים וְ֝שָׁל֗וֹם יוֹסִ֥יפוּ לָֽךְ׃ (נח) חֶ֥סֶד וֶאֱמֶ֗ת אַֽל־ יַעַ֫זְבֻ֥ךָ קָשְׁרֵ֥ם עַל־ גַּרְגְּרוֹתֶ֑יךָ כָּ֝תְבֵ֗ם עַל־ ל֥וּחַ לִבֶּֽךָ׃ (נט) וּמְצָא־ חֵ֖ן וְשֵֽׂכֶל־ ט֑וֹב בְּעֵינֵ֖י אֱלֹהִ֣ים וְאָדָֽם׃ (ס) בְּטַ֣ח אֶל־ יְ֭הוָה בְּכָל־ לִבֶּ֑ךָ וְאֶל־ בִּֽ֝ינָתְךָ֗ אַל־ תִּשָּׁעֵֽן׃ (סא) בְּכָל־ דְּרָכֶ֥יךָ דָעֵ֑הוּ וְ֝ה֗וּא יְיַשֵּׁ֥ר אֹֽרְחֹתֶֽיךָ׃ (סב) אַל־ תְּהִ֣י חָכָ֣ם בְּעֵינֶ֑יךָ יְרָ֥א אֶת־ יְ֝הוָ֗ה וְס֣וּר מֵרָֽע׃ (סג) רִ֭פְאוּת תְּהִ֣י לְשָׁרֶּ֑ךָ וְ֝שִׁקּ֗וּי לְעַצְמוֹתֶֽיךָ׃ (סד) כַּבֵּ֣ד אֶת־ יְ֭הוָה מֵהוֹנֶ֑ךָ וּ֝מֵרֵאשִׁ֗ית כָּל־ תְּבוּאָתֶֽךָ׃ (סה) וְיִמָּלְא֣וּ אֲסָמֶ֣יךָ שָׂבָ֑ע וְ֝תִיר֗וֹשׁ יְקָבֶ֥יךָ יִפְרֹֽצוּ׃ (סו) מוּסַ֣ר יְ֭הוָה בְּנִ֣י אַל־ תִּמְאָ֑ס וְאַל־ תָּ֝קֹ֗ץ בְּתוֹכַחְתּֽוֹ׃ (סז) כִּ֤י אֶ֥ת אֲשֶׁ֣ר יֶאֱהַ֣ב יְהוָ֣ה יוֹכִ֑יחַ וּ֝כְאָ֗ב אֶת־ בֵּ֥ן יִרְצֶֽה׃ (סח) אַשְׁרֵ֣י אָ֭דָם מָצָ֣א חָכְמָ֑ה וְ֝אָדָ֗ם יָפִ֥יק תְּבוּנָֽה׃ (סט) כִּ֤י ט֣וֹב סַ֭חְרָהּ מִסְּחַר־ כָּ֑סֶף וּ֝מֵחָר֗וּץ תְּבוּאָתָֽהּ׃ (ע) יְקָ֣רָה הִ֭יא (מפניים) מִפְּנִינִ֑ים וְכָל־ חֲ֝פָצֶ֗יךָ לֹ֣א יִֽשְׁווּ־ בָֽהּ׃ (עא) אֹ֣רֶךְ יָ֭מִים בִּֽימִינָ֑הּ בִּ֝שְׂמֹאולָ֗הּ עֹ֣שֶׁר וְכָבֽוֹד׃ (עב) דְּרָכֶ֥יהָ דַרְכֵי־ נֹ֑עַם וְֽכָל־ נְתִ֖יבוֹתֶ֣יהָ שָׁלֽוֹם׃ (עג) עֵץ־ חַיִּ֣ים הִ֭יא לַמַּחֲזִיקִ֣ים בָּ֑הּ וְֽתֹמְכֶ֥יהָ מְאֻשָּֽׁר׃ (עד) יְֽהוָ֗ה בְּחָכְמָ֥ה יָֽסַד־ אָ֑רֶץ כּוֹנֵ֥ן שָׁ֝מַ֗יִם בִּתְבוּנָֽה׃ (עה) בְּ֭דַעְתּוֹ תְּהוֹמ֣וֹת נִבְקָ֑עוּ וּ֝שְׁחָקִ֗ים יִרְעֲפוּ־ טָֽל׃ (עו) בְּ֭נִי אַל־ יָלֻ֣זוּ מֵעֵינֶ֑יךָ נְצֹ֥ר תֻּ֝שִׁיָּ֗ה וּמְזִמָּֽה׃ (עז) וְיִֽהְי֣וּ חַיִּ֣ים לְנַפְשֶׁ֑ךָ וְ֝חֵ֗ן לְגַרְגְּרֹתֶֽיךָ׃ (עח) אָ֤ז תֵּלֵ֣ךְ לָבֶ֣טַח דַּרְכֶּ֑ךָ וְ֝רַגְלְךָ֗ לֹ֣א תִגּֽוֹף׃ (עט) אִם־ תִּשְׁכַּ֥ב לֹֽא־ תִפְחָ֑ד וְ֝שָׁכַבְתָּ֗ וְֽעָרְבָ֥ה שְׁנָתֶֽךָ׃ (פ) אַל־ תִּ֭ירָא מִפַּ֣חַד פִּתְאֹ֑ם וּמִשֹּׁאַ֥ת רְ֝שָׁעִ֗ים כִּ֣י תָבֹֽא׃ (פא) כִּֽי־ יְ֭הוָה יִהְיֶ֣ה בְכִסְלֶ֑ךָ וְשָׁמַ֖ר רַגְלְךָ֣ מִלָּֽכֶד׃ (פב) אַל־ תִּמְנַע־ ט֥וֹב מִבְּעָלָ֑יו בִּהְי֨וֹת לְאֵ֖ל (ידיך) יָדְךָ֣ לַעֲשֽׂוֹת׃ (פג) אַל־ תֹּ֘אמַ֤ר (לרעיך) לְרֵֽעֲךָ֨ ׀ לֵ֣ךְ וָ֭שׁוּב וּמָחָ֥ר אֶתֵּ֗ן וְיֵ֣שׁ אִתָּֽךְ׃ (פד) אַל־ תַּחֲרֹ֣שׁ עַל־ רֵעֲךָ֣ רָעָ֑ה וְהֽוּא־ יוֹשֵׁ֖ב לָבֶ֣טַח אִתָּֽךְ׃ (פה) אַל־ (תרוב) תָּרִ֣יב עִם־ אָדָ֣ם חִנָּ֑ם אִם־ לֹ֖א גְמָלְךָ֣ רָעָֽה׃ (פו) אַל־ תְּ֭קַנֵּא בְּאִ֣ישׁ חָמָ֑ס וְאַל־ תִּ֝בְחַ֗ר בְּכָל־ דְּרָכָֽיו׃ (פז) כִּ֤י תוֹעֲבַ֣ת יְהוָ֣ה נָל֑וֹז וְֽאֶת־ יְשָׁרִ֥ים סוֹדֽוֹ׃ (פח) מְאֵרַ֣ת יְ֭הוָה בְּבֵ֣ית רָשָׁ֑ע וּנְוֵ֖ה צַדִּיקִ֣ים יְבָרֵֽךְ׃ (פט) אִם־ לַלֵּצִ֥ים הֽוּא־ יָלִ֑יץ (ולעניים) וְ֝לַעֲנָוִ֗ים יִתֶּן־ חֵֽן׃ (צ) כָּ֭בוֹד חֲכָמִ֣ים יִנְחָ֑לוּ וּ֝כְסִילִ֗ים מֵרִ֥ים קָלֽוֹן׃ תחילת פרק ד (צא) שִׁמְע֣וּ בָ֭נִים מ֣וּסַר אָ֑ב וְ֝הַקְשִׁ֗יבוּ לָדַ֥עַת בִּינָֽה׃ (צב) כִּ֤י לֶ֣קַח ט֭וֹב נָתַ֣תִּי לָכֶ֑ם תּֽ֝וֹרָתִ֗י אַֽל־ תַּעֲזֹֽבוּ׃ (צג) כִּי־ בֵ֭ן הָיִ֣יתִי לְאָבִ֑י רַ֥ךְ וְ֝יָחִ֗יד לִפְנֵ֥י אִמִּֽי׃ (צד) וַיֹּרֵ֗נִי וַיֹּ֥אמֶר לִ֗י יִֽתְמָךְ־ דְּבָרַ֥י לִבֶּ֑ךָ שְׁמֹ֖ר מִצְוֺתַ֣י וֶֽחְיֵֽה׃ (צה) קְנֵ֣ה חָ֭כְמָה קְנֵ֣ה בִינָ֑ה אַל־ תִּשְׁכַּ֥ח וְאַל־ תֵּ֝֗ט מֵֽאִמְרֵי־ פִֽי׃ (צו) אַל־ תַּעַזְבֶ֥הָ וְתִשְׁמְרֶ֑ךָּ אֱהָבֶ֥הָ וְתִצְּרֶֽךָּ׃ (צז) רֵאשִׁ֣ית חָ֭כְמָה קְנֵ֣ה חָכְמָ֑ה וּבְכָל־ קִ֝נְיָנְךָ֗ קְנֵ֣ה בִינָֽה׃ (צח) סַלְסְלֶ֥הָ וּֽתְרוֹמְמֶ֑ךָּ תְּ֝כַבֵּ֗דְךָ כִּ֣י תְחַבְּקֶֽנָּה׃ (צט) תִּתֵּ֣ן לְ֭רֹאשְׁךָ לִוְיַת־ חֵ֑ן עֲטֶ֖רֶת תִּפְאֶ֣רֶת תְּמַגְּנֶֽךָּ׃ (ק) שְׁמַ֣ע בְּ֭נִי וְקַ֣ח אֲמָרָ֑י וְיִרְבּ֥וּ לְ֝ךָ֗ שְׁנ֣וֹת חַיִּֽים׃ (קא) בְּדֶ֣רֶךְ חָ֭כְמָה הֹרֵתִ֑יךָ הִ֝דְרַכְתִּ֗יךָ בְּמַעְגְּלֵי־ יֹֽשֶׁר׃ (קב) בְּֽ֭לֶכְתְּךָ לֹא־ יֵצַ֣ר צַעֲדֶ֑ךָ וְאִם־ תָּ֝ר֗וּץ לֹ֣א תִכָּשֵֽׁל׃ (קג) הַחֲזֵ֣ק בַּמּוּסָ֣ר אַל־ תֶּ֑רֶף נִ֝צְּרֶ֗הָ כִּי־ הִ֥יא חַיֶּֽיךָ׃ (קד) בְּאֹ֣רַח רְ֭שָׁעִים אַל־ תָּבֹ֑א וְאַל־ תְּ֝אַשֵּׁ֗ר בְּדֶ֣רֶךְ רָעִֽים׃ (קה) פְּרָעֵ֥הוּ אַל־ תַּעֲבָר־ בּ֑וֹ שְׂטֵ֖ה מֵעָלָ֣יו וַעֲבֽוֹר׃ (קו) כִּ֤י לֹ֣א יִֽ֭שְׁנוּ אִם־ לֹ֣א יָרֵ֑עוּ וְֽנִגְזְלָ֥ה שְׁ֝נָתָ֗ם אִם־ לֹ֥א (יכשולו) יַכְשִֽׁילוּ׃ (קז) כִּ֣י לָ֭חֲמוּ לֶ֣חֶם רֶ֑שַׁע וְיֵ֖ין חֲמָסִ֣ים יִשְׁתּֽוּ׃ (קח) וְאֹ֣רַח צַ֭דִּיקִים כְּא֣וֹר נֹ֑גַהּ הוֹלֵ֥ךְ וָ֝א֗וֹר עַד־ נְכ֥וֹן הַיּֽוֹם׃ (קט) דֶּ֣רֶךְ רְ֭שָׁעִים כָּֽאֲפֵלָ֑ה לֹ֥א יָ֝דְע֗וּ בַּמֶּ֥ה יִכָּשֵֽׁלוּ׃ (קי) בְּ֭נִי לִדְבָרַ֣י הַקְשִׁ֑יבָה לַ֝אֲמָרַ֗י הַט־ אָזְנֶֽךָ׃ (קיא) אַל־ יַלִּ֥יזוּ מֵעֵינֶ֑יךָ שָׁ֝מְרֵ֗ם בְּת֣וֹךְ לְבָבֶֽךָ׃ (קיב) כִּֽי־ חַיִּ֣ים הֵ֭ם לְמֹצְאֵיהֶ֑ם וּֽלְכָל־ בְּשָׂר֥וֹ מַרְפֵּֽא׃ (קיג) מִֽכָּל־ מִ֭שְׁמָר נְצֹ֣ר לִבֶּ֑ךָ כִּֽי־ מִ֝מֶּ֗נּוּ תּוֹצְא֥וֹת חַיִּֽים׃ (קיד) הָסֵ֣ר מִ֭מְּךָ עִקְּשׁ֣וּת פֶּ֑ה וּלְז֥וּת שְׂ֝פָתַ֗יִם הַרְחֵ֥ק מִמֶּֽךָּ׃ (קטו) עֵ֭ינֶיךָ לְנֹ֣כַח יַבִּ֑יטוּ וְ֝עַפְעַפֶּ֗יךָ יַיְשִׁ֥רוּ נֶגְדֶּֽךָ׃ (קטז) פַּ֭לֵּס מַעְגַּ֣ל רַגְלֶ֑ךָ וְֽכָל־ דְּרָכֶ֥יךָ יִכֹּֽנוּ׃ (קיז) אַֽל־ תֵּט־ יָמִ֥ין וּשְׂמֹ֑אול הָסֵ֖ר רַגְלְךָ֣ מֵרָֽע׃ סוף פרק ד (קיח) בְּ֭נִי לְחָכְמָתִ֣י הַקְשִׁ֑יבָה לִ֝תְבוּנָתִ֗י הַט־ אָזְנֶֽךָ׃ (קיט) לִשְׁמֹ֥ר מְזִמּ֑וֹת וְ֝דַ֗עַת שְׂפָתֶ֥יךָ יִנְצֹֽרוּ׃ (קכ) כִּ֤י נֹ֣פֶת תִּ֭טֹּפְנָה שִׂפְתֵ֣י זָרָ֑ה וְחָלָ֖ק מִשֶּׁ֣מֶן חִכָּֽהּ׃ (קכא) וְֽ֭אַחֲרִיתָהּ מָרָ֣ה כַֽלַּעֲנָ֑ה חַ֝דָּ֗ה כְּחֶ֣רֶב פִּיּֽוֹת׃ (קכב) רַ֭גְלֶיהָ יֹרְד֣וֹת מָ֑וֶת שְׁ֝א֗וֹל צְעָדֶ֥יהָ יִתְמֹֽכוּ׃ (קכג) אֹ֣רַח חַ֭יִּים פֶּן־ תְּפַלֵּ֑ס נָע֥וּ מַ֝עְגְּלֹתֶ֗יהָ לֹ֣א תֵדָֽע׃ (קכד) וְעַתָּ֣ה בָ֭נִים שִׁמְעוּ־ לִ֑י וְאַל־ תָּ֝ס֗וּרוּ מֵאִמְרֵי־ פִֽי׃ (קכה) הַרְחֵ֣ק מֵעָלֶ֣יהָ דַרְכֶּ֑ךָ וְאַל־ תִּ֝קְרַ֗ב אֶל־ פֶּ֥תַח בֵּיתָֽהּ׃ (קכו) פֶּן־ תִּתֵּ֣ן לַאֲחֵרִ֣ים הוֹדֶ֑ךָ וּ֝שְׁנֹתֶ֗יךָ לְאַכְזָרִֽי׃ (קכז) פֶּֽן־ יִשְׂבְּע֣וּ זָרִ֣ים כֹּחֶ֑ךָ וַ֝עֲצָבֶ֗יךָ בְּבֵ֣ית נָכְרִֽי׃ (קכח) וְנָהַמְתָּ֥ בְאַחֲרִיתֶ֑ךָ בִּכְל֥וֹת בְּ֝שָׂרְךָ֗ וּשְׁאֵרֶֽךָ׃ (קכט) וְֽאָמַרְתָּ֗ אֵ֭יךְ שָׂנֵ֣אתִי מוּסָ֑ר וְ֝תוֹכַ֗חַת נָאַ֥ץ לִבִּֽי׃ (קל) וְֽלֹא־ שָׁ֭מַעְתִּי בְּק֣וֹל מוֹרָ֑י וְ֝לִֽמְלַמְּדַ֗י לֹא־ הִטִּ֥יתִי אָזְנִֽי׃ (קלא) כִּ֭מְעַט הָיִ֣יתִי בְכָל־ רָ֑ע בְּת֖וֹךְ קָהָ֣ל וְעֵדָֽה׃ (קלב) שְׁתֵה־ מַ֥יִם מִבּוֹרֶ֑ךָ וְ֝נֹזְלִ֗ים מִתּ֥וֹךְ בְּאֵרֶֽךָ׃ (קלג) יָפ֣וּצוּ מַעְיְנֹתֶ֣יךָ ח֑וּצָה בָּ֝רְחֹב֗וֹת פַּלְגֵי־ מָֽיִם׃ (קלד) יִֽהְיוּ־ לְךָ֥ לְבַדֶּ֑ךָ וְאֵ֖ין לְזָרִ֣ים אִתָּֽךְ׃
