משלי פרק ט לפי החלוקה לסדרים

פרק ט בספר משלי (18 פסוקים) נמצא במשלי סדר ב (פסוקים קה–קיד) ובמשלי סדר ג (פסוקים א–ח).

משלי סדר ב

(א) יְהִֽי־ מְקוֹרְךָ֥ בָר֑וּךְ וּ֝שְׂמַ֗ח מֵאֵ֥שֶׁת נְעוּרֶֽךָ׃ (ב) אַיֶּ֥לֶת אֲהָבִ֗ים וְֽיַעֲלַ֫ת־ חֵ֥ן דַּ֭דֶּיהָ יְרַוֻּ֣ךָ בְכָל־ עֵ֑ת בְּ֝אַהֲבָתָ֗הּ תִּשְׁגֶּ֥ה תָמִֽיד׃ (ג) וְלָ֤מָּה תִשְׁגֶּ֣ה בְנִ֣י בְזָרָ֑ה וּ֝תְחַבֵּ֗ק חֵ֣ק נָכְרִיָּֽה׃ (ד) כִּ֤י נֹ֨כַח ׀ עֵינֵ֣י יְ֭הוָה דַּרְכֵי־ אִ֑ישׁ וְֽכָל־ מַעְגְּלֹתָ֥יו מְפַלֵּֽס׃ (ה) עַֽווֹנוֹתָ֗יו יִלְכְּדֻנ֥וֹ אֶת־ הָרָשָׁ֑ע וּבְחַבְלֵ֥י חַ֝טָּאת֗וֹ יִתָּמֵֽךְ׃ (ו) ה֗וּא יָ֭מוּת בְּאֵ֣ין מוּסָ֑ר וּבְרֹ֖ב אִוַּלְתּ֣וֹ יִשְׁגֶּֽה׃ (ז) בְּ֭נִי אִם־ עָרַ֣בְתָּ לְרֵעֶ֑ךָ תָּקַ֖עְתָּ לַזָּ֣ר כַּפֶּֽיךָ׃ (ח) נוֹקַ֥שְׁתָּ בְאִמְרֵי־ פִ֑יךָ נִ֝לְכַּ֗דְתָּ בְּאִמְרֵי־ פִֽיךָ׃ (ט) עֲשֵׂ֨ה זֹ֥את אֵפ֪וֹא ׀ בְּנִ֡י וְֽהִנָּצֵ֗ל כִּ֘י בָ֤אתָ בְכַף־ רֵעֶ֑ךָ לֵ֥ךְ הִ֝תְרַפֵּ֗ס וּרְהַ֥ב רֵעֶֽיךָ׃ (י) אַל־ תִּתֵּ֣ן שֵׁנָ֣ה לְעֵינֶ֑יךָ וּ֝תְנוּמָ֗ה לְעַפְעַפֶּֽיךָ׃ (יא) הִ֭נָּצֵל כִּצְבִ֣י מִיָּ֑ד וּ֝כְצִפּ֗וֹר מִיַּ֥ד יָקֽוּשׁ׃ (יב) לֵֽךְ־ אֶל־ נְמָלָ֥ה עָצֵ֑ל רְאֵ֖ה דְרָכֶ֣יהָ וַחֲכָֽם׃ (יג) אֲשֶׁ֖ר אֵֽין־ לָ֥הּ קָצִ֗ין שֹׁטֵ֥ר וּמֹשֵֽׁל׃ (יד) תָּכִ֣ין בַּקַּ֣יִץ לַחְמָ֑הּ אָגְרָ֥ה בַ֝קָּצִ֗יר מַאֲכָלָֽהּ׃ (טו) עַד־ מָתַ֖י עָצֵ֥ל ׀ תִּשְׁכָּ֑ב מָ֝תַ֗י תָּק֥וּם מִשְּׁנָתֶֽךָ׃ (טז) מְעַ֣ט שֵׁ֭נוֹת מְעַ֣ט תְּנוּמ֑וֹת מְעַ֓ט ׀ חִבֻּ֖ק יָדַ֣יִם לִשְׁכָּֽב׃ (יז) וּבָֽא־ כִמְהַלֵּ֥ךְ רֵאשֶׁ֑ךָ וּ֝מַחְסֹֽרְךָ֗ כְּאִ֣ישׁ מָגֵֽן׃ (יח) אָדָ֣ם בְּ֭לִיַּעַל אִ֣ישׁ אָ֑וֶן ה֝וֹלֵ֗ךְ עִקְּשׁ֥וּת פֶּֽה׃ (יט) קֹרֵ֣ץ בְּ֭עֵינָו מֹלֵ֣ל בְּרַגְלָ֑ו מֹ֝רֶ֗ה בְּאֶצְבְּעֹתָֽיו׃ (כ) תַּֽהְפֻּכ֨וֹת ׀ בְּלִבּ֗וֹ חֹרֵ֣שׁ רָ֣ע בְּכָל־ עֵ֑ת (מדנים) מִדְיָנִ֥ים יְשַׁלֵּֽחַ׃ (כא) עַל־ כֵּ֗ן פִּ֭תְאֹם יָב֣וֹא אֵיד֑וֹ פֶּ֥תַע יִ֝שָּׁבֵ֗ר וְאֵ֣ין מַרְפֵּֽא׃ (כב) שֶׁשׁ־ הֵ֭נָּה שָׂנֵ֣א יְהוָ֑ה וְ֝שֶׁ֗בַע (תועבות) תּוֹעֲבַ֥ת נַפְשֽׁוֹ׃ (כג) עֵינַ֣יִם רָ֭מוֹת לְשׁ֣וֹן שָׁ֑קֶר וְ֝יָדַ֗יִם שֹׁפְכ֥וֹת דָּם־ נָקִֽי׃ (כד) לֵ֗ב חֹ֭רֵשׁ מַחְשְׁב֣וֹת אָ֑וֶן רַגְלַ֥יִם מְ֝מַהֲר֗וֹת לָר֥וּץ לָֽרָעָה׃ (כה) יָפִ֣יחַ כְּ֭זָבִים עֵ֣ד שָׁ֑קֶר וּמְשַׁלֵּ֥חַ מְ֝דָנִ֗ים בֵּ֣ין אַחִֽים׃ (כו) נְצֹ֣ר בְּ֭נִי מִצְוַ֣ת אָבִ֑יךָ וְאַל־ תִּ֝טֹּ֗שׁ תּוֹרַ֥ת אִמֶּֽךָ׃ (כז) קָשְׁרֵ֣ם עַל־ לִבְּךָ֣ תָמִ֑יד עָ֝נְדֵ֗ם עַל־ גַּרְגְּרֹתֶֽךָ׃ (כח) בְּהִתְהַלֶּכְךָ֨ ׀ תַּנְחֶ֬ה אֹתָ֗ךְ בְּֽ֭שָׁכְבְּךָ תִּשְׁמֹ֣ר עָלֶ֑יךָ וַ֝הֲקִיצ֗וֹתָ הִ֣יא תְשִׂיחֶֽךָ׃ (כט) כִּ֤י נֵ֣ר מִ֭צְוָה וְת֣וֹרָה א֑וֹר וְדֶ֥רֶךְ חַ֝יִּ֗ים תּוֹכְח֥וֹת מוּסָֽר׃ (ל) לִ֭שְׁמָרְךָ מֵאֵ֣שֶׁת רָ֑ע מֵֽ֝חֶלְקַ֗ת לָשׁ֥וֹן נָכְרִיָּֽה׃ (לא) אַל־ תַּחְמֹ֣ד יָ֭פְיָהּ בִּלְבָבֶ֑ךָ וְאַל־ תִּ֝קָּֽחֲךָ֗ בְּעַפְעַפֶּֽיהָ׃ (לב) כִּ֤י בְעַד־ אִשָּׁ֥ה זוֹנָ֗ה עַֽד־ כִּכַּ֫ר לָ֥חֶם וְאֵ֥שֶׁת אִ֑ישׁ נֶ֖פֶשׁ יְקָרָ֣ה תָצֽוּד׃ (לג) הֲיַחְתֶּ֤ה אִ֓ישׁ אֵ֬שׁ בְּחֵיק֑וֹ וּ֝בְגָדָ֗יו לֹ֣א תִשָּׂרַֽפְנָה׃ (לד) אִם־ יְהַלֵּ֣ךְ אִ֭ישׁ עַל־ הַגֶּחָלִ֑ים וְ֝רַגְלָ֗יו לֹ֣א תִכָּוֶֽינָה׃ (לה) כֵּ֗ן הַ֭בָּא אֶל־ אֵ֣שֶׁת רֵעֵ֑הוּ לֹ֥א יִ֝נָּקֶ֗ה כָּֽל־ הַנֹּגֵ֥עַ בָּֽהּ׃ (לו) לֹא־ יָב֣וּזוּ לַ֭גַּנָּב כִּ֣י יִגְנ֑וֹב לְמַלֵּ֥א נַ֝פְשׁ֗וֹ כִּ֣י יִרְעָֽב׃ (לז) וְ֭נִמְצָא יְשַׁלֵּ֣ם שִׁבְעָתָ֑יִם אֶת־ כָּל־ ה֖וֹן בֵּית֣וֹ יִתֵּֽן׃ (לח) נֹאֵ֣ף אִשָּׁ֣ה חֲסַר־ לֵ֑ב מַֽשְׁחִ֥ית נַ֝פְשׁ֗וֹ ה֣וּא יַעֲשֶֽׂנָּה׃ (לט) נֶֽגַע־ וְקָל֥וֹן יִמְצָ֑א וְ֝חֶרְפָּת֗וֹ לֹ֣א תִמָּחֶֽה׃ (מ) כִּֽי־ קִנְאָ֥ה חֲמַת־ גָּ֑בֶר וְלֹֽא־ יַ֝חְמ֗וֹל בְּי֣וֹם נָקָֽם׃ (מא) לֹא־ יִ֭שָּׂא פְּנֵ֣י כָל־ כֹּ֑פֶר וְלֹֽא־ יֹ֝אבֶ֗ה כִּ֣י תַרְבֶּה־ שֹֽׁחַד׃ (מב) בְּ֭נִי שְׁמֹ֣ר אֲמָרָ֑י וּ֝מִצְוֺתַ֗י תִּצְפֹּ֥ן אִתָּֽךְ׃ (מג) שְׁמֹ֣ר מִצְוֺתַ֣י וֶחְיֵ֑ה וְ֝תוֹרָתִ֗י כְּאִישׁ֥וֹן עֵינֶֽיךָ׃ (מד) קָשְׁרֵ֥ם עַל־ אֶצְבְּעֹתֶ֑יךָ כָּ֝תְבֵ֗ם עַל־ ל֥וּחַ לִבֶּֽךָ׃ (מה) אֱמֹ֣ר לַֽ֭חָכְמָה אֲחֹ֣תִי אָ֑תְּ וּ֝מֹדָ֗ע לַבִּינָ֥ה תִקְרָֽא׃ (מו) לִ֭שְׁמָרְךָ מֵאִשָּׁ֣ה זָרָ֑ה מִ֝נָּכְרִיָּ֗ה אֲמָרֶ֥יהָ הֶחֱלִֽיקָה׃ (מז) כִּ֭י בְּחַלּ֣וֹן בֵּיתִ֑י בְּעַ֖ד אֶשְׁנַבִּ֣י נִשְׁקָֽפְתִּי׃ (מח) וָאֵ֤רֶא בַפְּתָאיִ֗ם אָ֘בִ֤ינָה בַבָּנִ֗ים נַ֣עַר חֲסַר־ לֵֽב׃ (מט) עֹבֵ֣ר בַּ֭שּׁוּק אֵ֣צֶל פִּנָּ֑הּ וְדֶ֖רֶךְ בֵּיתָ֣הּ יִצְעָֽד׃ (נ) בְּנֶֽשֶׁף־ בְּעֶ֥רֶב י֑וֹם בְּאִישׁ֥וֹן לַ֝֗יְלָה וַאֲפֵלָֽה׃ (נא) וְהִנֵּ֣ה אִ֭שָּׁה לִקְרָאת֑וֹ שִׁ֥ית ז֝וֹנָ֗ה וּנְצֻ֥רַת לֵֽב׃ (נב) הֹמִיָּ֣ה הִ֣יא וְסֹרָ֑רֶת בְּ֝בֵיתָ֗הּ לֹא־ יִשְׁכְּנ֥וּ רַגְלֶֽיהָ׃ (נג) פַּ֤עַם ׀ בַּח֗וּץ פַּ֥עַם בָּרְחֹב֑וֹת וְאֵ֖צֶל כָּל־ פִּנָּ֣ה תֶאֱרֹֽב׃ (נד) וְהֶחֱזִ֣יקָה בּ֭וֹ וְנָ֣שְׁקָה־ לּ֑וֹ הֵעֵ֥זָה פָ֝נֶ֗יהָ וַתֹּ֣אמַר לֽוֹ׃ (נה) זִבְחֵ֣י שְׁלָמִ֣ים עָלָ֑י הַ֝יּ֗וֹם שִׁלַּ֥מְתִּי נְדָרָֽי׃ (נו) עַל־ כֵּ֭ן יָצָ֣אתִי לִקְרָאתֶ֑ךָ לְשַׁחֵ֥ר פָּ֝נֶ֗יךָ וָאֶמְצָאֶֽךָּ׃ (נז) מַ֭רְבַדִּים רָבַ֣דְתִּי עַרְשִׂ֑י חֲ֝טֻב֗וֹת אֵט֥וּן מִצְרָֽיִם׃ (נח) נַ֥פְתִּי מִשְׁכָּבִ֑י מֹ֥ר אֲ֝הָלִ֗ים וְקִנָּמֽוֹן׃ (נט) לְכָ֤ה נִרְוֶ֣ה דֹ֭דִים עַד־ הַבֹּ֑קֶר נִ֝תְעַלְּסָ֗ה בָּאֳהָבִֽים׃ (ס) כִּ֤י אֵ֣ין הָאִ֣ישׁ בְּבֵית֑וֹ הָ֝לַ֗ךְ בְּדֶ֣רֶךְ מֵרָחֽוֹק׃ (סא) צְֽרוֹר־ הַ֭כֶּסֶף לָקַ֣ח בְּיָד֑וֹ לְי֥וֹם הַ֝כֵּ֗סֶא יָבֹ֥א בֵיתֽוֹ׃ (סב) הִ֭טַּתּוּ בְּרֹ֣ב לִקְחָ֑הּ בְּחֵ֥לֶק שְׂ֝פָתֶ֗יהָ תַּדִּיחֶֽנּוּ׃ (סג) ה֤וֹלֵ֥ךְ אַחֲרֶ֗יהָ פִּ֫תְאֹ֥ם כְּ֭שׁוֹר אֶל־ טָ֣בַח יָב֑וֹא וּ֝כְעֶ֗כֶס אֶל־ מוּסַ֥ר אֱוִֽיל׃ (סד) עַ֤ד יְפַלַּ֪ח חֵ֡ץ כְּֽבֵד֗וֹ כְּמַהֵ֣ר צִפּ֣וֹר אֶל־ פָּ֑ח וְלֹֽא־ יָ֝דַ֗ע כִּֽי־ בְנַפְשׁ֥וֹ הֽוּא׃ (סה) וְעַתָּ֣ה בָ֭נִים שִׁמְעוּ־ לִ֑י וְ֝הַקְשִׁ֗יבוּ לְאִמְרֵי־ פִֽי׃ (סו) אַל־ יֵ֣שְׂטְ אֶל־ דְּרָכֶ֣יהָ לִבֶּ֑ךָ אַל־ תֵּ֝תַע בִּנְתִיבוֹתֶֽיהָ׃ (סז) כִּֽי־ רַבִּ֣ים חֲלָלִ֣ים הִפִּ֑ילָה וַ֝עֲצֻמִ֗ים כָּל־ הֲרֻגֶֽיהָ׃ (סח) דַּרְכֵ֣י יֹ֣אכַל ט֑וֹב וְנֶ֖פֶשׁ בֹּגְדִ֣ים חָמָֽס׃ (קב) נֹצֵ֣ר פִּ֭יו שֹׁמֵ֣ר נַפְשׁ֑וֹ פֹּשֵׂ֥ק שְׂ֝פָתָ֗יו מְחִתָּה־ לֽוֹ׃ (קג) מִתְאַוָּ֣ה וָ֭אַיִן נַפְשׁ֣וֹ עָצֵ֑ל וְנֶ֖פֶשׁ חָרֻצִ֣ים תְּדֻשָּֽׁן׃ (קד) דְּבַר־ שֶׁ֭קֶר יִשְׂנָ֣א צַדִּ֑יק וְ֝רָשָׁ֗ע יַבְאִ֥ישׁ וְיַחְפִּֽיר׃ (קה) צְ֭דָקָה תִּצֹּ֣ר תָּם־ דָּ֑רֶךְ וְ֝רִשְׁעָ֗ה תְּסַלֵּ֥ף חַטָּֽאת׃ (קו) יֵ֣שׁ מִ֭תְעַשֵּׁר וְאֵ֣ין כֹּ֑ל מִ֝תְרוֹשֵׁ֗שׁ וְה֣וֹן רָֽב׃ (קז) כֹּ֣פֶר נֶֽפֶשׁ־ אִ֣ישׁ עָשְׁר֑וֹ וְ֝רָ֗שׁ לֹא־ שָׁמַ֥ע גְּעָרָֽה׃ (קח) אוֹר־ צַדִּיקִ֥ים יִשְׂמָ֑ח וְנֵ֖ר רְשָׁעִ֣ים יִדְעָֽךְ׃ (קט) רַק־ בְּ֭זָדוֹן יִתֵּ֣ן מַצָּ֑ה וְאֶת־ נ֖וֹעָצִ֣ים חָכְמָֽה׃ (קי) ה֭וֹן מֵהֶ֣בֶל יִמְעָ֑ט וְקֹבֵ֖ץ עַל־ יָ֣ד יַרְבֶּֽה׃ (קיא) תּוֹחֶ֣לֶת מְ֭מֻשָּׁכָה מַחֲלָה־ לֵ֑ב וְעֵ֥ץ חַ֝יִּ֗ים תַּאֲוָ֥ה בָאָֽה׃ (קיב) בָּ֣ז לְ֭דָבָר יֵחָ֣בֶל ל֑וֹ וִירֵ֥א מִ֝צְוָ֗ה ה֣וּא יְשֻׁלָּֽם׃ (קיג) תּוֹרַ֣ת חָ֭כָם מְק֣וֹר חַיִּ֑ים לָ֝ס֗וּר מִמֹּ֥קְשֵׁי מָֽוֶת׃ (קיד) שֵֽׂכֶל־ ט֭וֹב יִתֶּן־ חֵ֑ן וְדֶ֖רֶךְ בֹּגְדִ֣ים אֵיתָֽן׃ (קטו) כָּל־ עָ֭רוּם יַעֲשֶׂ֣ה בְדָ֑עַת וּ֝כְסִ֗יל יִפְרֹ֥שׂ אִוֶּֽלֶת׃ (קטז) מַלְאָ֣ךְ רָ֭שָׁע יִפֹּ֣ל בְּרָ֑ע וְצִ֖יר אֱמוּנִ֣ים מַרְפֵּֽא׃ (קיז) רֵ֣ישׁ וְ֭קָלוֹן פּוֹרֵ֣עַ מוּסָ֑ר וְשׁוֹמֵ֖ר תּוֹכַ֣חַת יְכֻבָּֽד׃ (קיח) תַּאֲוָ֣ה נִ֭הְיָה תֶּעֱרַ֣ב לְנָ֑פֶשׁ וְתוֹעֲבַ֥ת כְּ֝סִילִ֗ים ס֣וּר מֵרָֽע׃ (קיט) (הלוך) הוֹלֵ֣ךְ אֶת־ חֲכָמִ֣ים (וחכם) יֶחְכָּ֑ם וְרֹעֶ֖ה כְסִילִ֣ים יֵרֽוֹעַ׃ (קכ) חַ֭טָּאִים תְּרַדֵּ֣ף רָעָ֑ה וְאֶת־ צַ֝דִּיקִ֗ים יְשַׁלֶּם־ טֽוֹב׃ (קכא) ט֗וֹב יַנְחִ֥יל בְּנֵֽי־ בָנִ֑ים וְצָפ֥וּן לַ֝צַּדִּ֗יק חֵ֣יל חוֹטֵֽא׃ (קכב) רָב־ אֹ֭כֶל נִ֣יר רָאשִׁ֑ים וְיֵ֥שׁ נִ֝סְפֶּ֗ה בְּלֹ֣א מִשְׁפָּֽט׃ (קכג) חוֹשֵׂ֣ךְ שִׁ֭בְטוֹ שׂוֹנֵ֣א בְנ֑וֹ וְ֝אֹהֲב֗וֹ שִֽׁחֲר֥וֹ מוּסָֽר׃ (קכד) צַדִּ֗יק אֹ֭כֵל לְשֹׂ֣בַע נַפְשׁ֑וֹ וּבֶ֖טֶן רְשָׁעִ֣ים תֶּחְסָֽר׃ (קכה) חַכְמ֣וֹת נָ֭שִׁים בָּנְתָ֣ה בֵיתָ֑הּ וְ֝אִוֶּ֗לֶת בְּיָדֶ֥יהָ תֶהֶרְסֶֽנּוּ׃ (קכו) הוֹלֵ֣ךְ בְּ֭יָשְׁרוֹ יְרֵ֣א יְהוָ֑ה וּנְל֖וֹז דְּרָכָ֣יו בּוֹזֵֽהוּ׃ (קכז) בְּֽפִי־ אֱ֭וִיל חֹ֣טֶר גַּאֲוָ֑ה וְשִׂפְתֵ֥י חֲ֝כָמִ֗ים תִּשְׁמוּרֵֽם׃
