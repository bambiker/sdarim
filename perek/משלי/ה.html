משלי פרק ה לפי החלוקה לסדרים

פרק ה בספר משלי (23 פסוקים) נמצא במשלי סדר א (פסוקים קיח–קלד) ובמשלי סדר ב (פסוקים א–ו).

משלי סדר א

(א) מִ֭שְׁלֵי שְׁלֹמֹ֣ה בֶן־ דָּוִ֑ד מֶ֝֗לֶךְ יִשְׂרָאֵֽל׃ (ב) לָדַ֣עַת חָכְמָ֣ה וּמוּסָ֑ר לְ֝הָבִ֗ין אִמְרֵ֥י בִינָֽה׃ (ג) לָ֭קַחַת מוּסַ֣ר הַשְׂכֵּ֑ל צֶ֥דֶק וּ֝מִשְׁפָּ֗ט וּמֵישָׁרִֽים׃ (ד) לָתֵ֣ת לִפְתָאיִ֣ם עָרְמָ֑ה לְ֝נַ֗עַר דַּ֣עַת וּמְזִמָּֽה׃ (ה) יִשְׁמַ֣ע חָ֭כָם וְי֣וֹסֶף לֶ֑קַח וְ֝נָב֗וֹן תַּחְבֻּל֥וֹת יִקְנֶֽה׃ (ו) לְהָבִ֣ין מָ֭שָׁל וּמְלִיצָ֑ה דִּבְרֵ֥י חֲ֝כָמִ֗ים וְחִידֹתָֽם׃ (ז) יִרְאַ֣ת יְ֭הוָה רֵאשִׁ֣ית דָּ֑עַת חָכְמָ֥ה וּ֝מוּסָ֗ר אֱוִילִ֥ים בָּֽזוּ׃ (ח) שְׁמַ֣ע בְּ֭נִי מוּסַ֣ר אָבִ֑יךָ וְאַל־ תִּ֝טֹּ֗שׁ תּוֹרַ֥ת אִמֶּֽךָ׃ (ט) כִּ֤י ׀ לִוְיַ֤ת חֵ֓ן הֵ֬ם לְרֹאשֶׁ֑ךָ וַ֝עֲנָקִ֗ים לְגַרְגְּרֹתֶֽיךָ׃ (י) בְּנִ֡י אִם־ יְפַתּ֥וּךָ חַ֝טָּאִ֗ים אַל־ תֹּבֵֽא׃ (יא) אִם־ יֹאמְרוּ֮ לְכָ֪ה אִ֫תָּ֥נוּ נֶאֶרְבָ֥ה לְדָ֑ם נִצְפְּנָ֖ה לְנָקִ֣י חִנָּֽם׃ (יב) נִ֭בְלָעֵם כִּשְׁא֣וֹל חַיִּ֑ים וּ֝תְמִימִ֗ים כְּי֣וֹרְדֵי בֽוֹר׃ (יג) כָּל־ ה֣וֹן יָקָ֣ר נִמְצָ֑א נְמַלֵּ֖א בָתֵּ֣ינוּ שָׁלָֽל׃ (יד) גּ֭וֹרָ֣לְךָ תַּפִּ֣יל בְּתוֹכֵ֑נוּ כִּ֥יס אֶ֝חָ֗ד יִהְיֶ֥ה לְכֻלָּֽנוּ׃ (טו) בְּנִ֗י אַל־ תֵּלֵ֣ךְ בְּדֶ֣רֶךְ אִתָּ֑ם מְנַ֥ע רַ֝גְלְךָ֗ מִנְּתִיבָתָֽם׃ (טז) כִּ֣י רַ֭גְלֵיהֶם לָרַ֣ע יָר֑וּצוּ וִֽ֝ימַהֲר֗וּ לִשְׁפָּךְ־ דָּֽם׃ (יז) כִּֽי־ חִ֭נָּם מְזֹרָ֣ה הָרָ֑שֶׁת בְּ֝עֵינֵ֗י כָל־ בַּ֥עַל כָּנָֽף׃ (יח) וְ֭הֵם לְדָמָ֣ם יֶאֱרֹ֑בוּ יִ֝צְפְּנ֗וּ לְנַפְשֹׁתָֽם׃ (יט) כֵּ֗ן אָ֭רְחוֹת כָּל־ בֹּ֣צֵֽעַ בָּ֑צַע אֶת־ נֶ֖פֶשׁ בְּעָלָ֣יו יִקָּֽח׃ (כ) חָ֭כְמוֹת בַּח֣וּץ תָּרֹ֑נָּה בָּ֝רְחֹב֗וֹת תִּתֵּ֥ן קוֹלָֽהּ׃ (כא) בְּרֹ֥אשׁ הֹמִיּ֗וֹת תִּ֫קְרָ֥א בְּפִתְחֵ֖י שְׁעָרִ֥ים בָּעִ֗יר אֲמָרֶ֥יהָ תֹאמֵֽר׃ (כב) עַד־ מָתַ֣י ׀ פְּתָיִם֮ תְּֽאֵהֲב֫וּ פֶ֥תִי וְלֵצִ֗ים לָ֭צוֹן חָמְד֣וּ לָהֶ֑ם וּ֝כְסִילִ֗ים יִשְׂנְאוּ־ דָֽעַת׃ (כג) תָּשׁ֗וּבוּ לְֽת֫וֹכַחְתִּ֥י הִנֵּ֤ה אַבִּ֣יעָה לָכֶ֣ם רוּחִ֑י אוֹדִ֖יעָה דְבָרַ֣י אֶתְכֶֽם׃ (כד) יַ֣עַן קָ֭רָאתִי וַתְּמָאֵ֑נוּ נָטִ֥יתִי יָ֝דִ֗י וְאֵ֣ין מַקְשִֽׁיב׃ (כה) וַתִּפְרְע֥וּ כָל־ עֲצָתִ֑י וְ֝תוֹכַחְתִּ֗י לֹ֣א אֲבִיתֶֽם׃ (כו) גַּם־ אֲ֭נִי בְּאֵידְכֶ֣ם אֶשְׂחָ֑ק אֶ֝לְעַ֗ג בְּבֹ֣א פַחְדְּכֶֽם׃ (כז) בְּבֹ֤א (כשאוה) כְשׁוֹאָ֨ה ׀ פַּחְדְּכֶ֗ם וְֽ֭אֵידְכֶם כְּסוּפָ֣ה יֶאֱתֶ֑ה בְּבֹ֥א עֲ֝לֵיכֶ֗ם צָרָ֥ה וְצוּקָֽה׃ (כח) אָ֣ז יִ֭קְרָאֻנְנִי וְלֹ֣א אֶֽעֱנֶ֑ה יְ֝שַׁחֲרֻ֗נְנִי וְלֹ֣א יִמְצָאֻֽנְנִי׃ (כט) תַּ֭חַת כִּי־ שָׂ֣נְאוּ דָ֑עַת וְיִרְאַ֥ת יְ֝הֹוָ֗ה לֹ֣א בָחָֽרוּ׃ (ל) לֹא־ אָב֥וּ לַעֲצָתִ֑י נָ֝אֲצ֗וּ כָּל־ תּוֹכַחְתִּֽי׃ (לא) וְֽ֭יֹאכְלוּ מִפְּרִ֣י דַרְכָּ֑ם וּֽמִמֹּעֲצֹ֖תֵיהֶ֣ם יִשְׂבָּֽעוּ׃ (לב) כִּ֤י מְשׁוּבַ֣ת פְּתָיִ֣ם תַּֽהַרְגֵ֑ם וְשַׁלְוַ֖ת כְּסִילִ֣ים תְּאַבְּדֵֽם׃ (לג) וְשֹׁמֵ֣עַֽ לִ֭י יִשְׁכָּן־ בֶּ֑טַח וְ֝שַׁאֲנַ֗ן מִפַּ֥חַד רָעָֽה׃ (לד) בְּ֭נִי אִם־ תִּקַּ֣ח אֲמָרָ֑י וּ֝מִצְוֺתַ֗י תִּצְפֹּ֥ן אִתָּֽךְ׃ (לה) לְהַקְשִׁ֣יב לַֽחָכְמָ֣ה אָזְנֶ֑ךָ תַּטֶּ֥ה לִ֝בְּךָ֗ לַתְּבוּנָֽה׃ (לו) כִּ֤י אִ֣ם לַבִּינָ֣ה תִקְרָ֑א לַ֝תְּבוּנָ֗ה תִּתֵּ֥ן קוֹלֶֽךָ׃ (לז) אִם־ תְּבַקְשֶׁ֥נָּה כַכָּ֑סֶף וְֽכַמַּטְמוֹנִ֥ים תַּחְפְּשֶֽׂנָּה׃ (לח) אָ֗ז תָּ֭בִין יִרְאַ֣ת יְהוָ֑ה וְדַ֖עַת אֱלֹהִ֣ים תִּמְצָֽא׃ (לט) כִּֽי־ יְ֭הוָה יִתֵּ֣ן חָכְמָ֑ה מִ֝פִּ֗יו דַּ֣עַת וּתְבוּנָֽה׃ (מ) (וצפן) יִצְפֹּ֣ן לַ֭יְשָׁרִים תּוּשִׁיָּ֑ה מָ֝גֵ֗ן לְהֹ֣לְכֵי תֹֽם׃ (מא) לִ֭נְצֹר אָרְח֣וֹת מִשְׁפָּ֑ט וְדֶ֖רֶךְ (חסידו) חֲסִידָ֣יו יִשְׁמֹֽר׃ (מב) אָ֗ז תָּ֭בִין צֶ֣דֶק וּמִשְׁפָּ֑ט וּ֝מֵישָׁרִ֗ים כָּל־ מַעְגַּל־ טֽוֹב׃ (מג) כִּֽי־ תָב֣וֹא חָכְמָ֣ה בְלִבֶּ֑ךָ וְ֝דַ֗עַת לְֽנַפְשְׁךָ֥ יִנְעָֽם׃ (מד) מְ֭זִמָּה תִּשְׁמֹ֥ר עָלֶ֗יךָ תְּבוּנָ֥ה תִנְצְרֶֽכָּה׃ (מה) לְ֭הַצִּ֣ילְךָ מִדֶּ֣רֶךְ רָ֑ע מֵ֝אִ֗ישׁ מְדַבֵּ֥ר תַּהְפֻּכֽוֹת׃ (מו) הַ֭עֹ֣זְבִים אָרְח֣וֹת יֹ֑שֶׁר לָ֝לֶ֗כֶת בְּדַרְכֵי־ חֹֽשֶׁךְ׃ (מז) הַ֭שְּׂמֵחִים לַעֲשׂ֥וֹת רָ֑ע יָ֝גִ֗ילוּ בְּֽתַהְפֻּכ֥וֹת רָֽע׃ (מח) אֲשֶׁ֣ר אָרְחֹתֵיהֶ֣ם עִקְּשִׁ֑ים וּ֝נְלוֹזִ֗ים בְּמַעְגְּלוֹתָֽם׃ (מט) לְ֭הַצִּ֣ילְךָ מֵאִשָּׁ֣ה זָרָ֑ה מִ֝נָּכְרִיָּ֗ה אֲמָרֶ֥יהָ הֶחֱלִֽיקָה׃ (נ) הַ֭עֹזֶבֶת אַלּ֣וּף נְעוּרֶ֑יהָ וְאֶת־ בְּרִ֖ית אֱלֹהֶ֣יהָ שָׁכֵֽחָה׃ (נא) כִּ֤י שָׁ֣חָה אֶל־ מָ֣וֶת בֵּיתָ֑הּ וְאֶל־ רְ֝פָאִ֗ים מַעְגְּלֹתֶֽיהָ׃ (נב) כָּל־ בָּ֭אֶיהָ לֹ֣א יְשׁוּב֑וּן וְלֹֽא־ יַ֝שִּׂ֗יגוּ אָרְח֥וֹת חַיִּֽים׃ (נג) לְמַ֗עַן תֵּ֭לֵךְ בְּדֶ֣רֶךְ טוֹבִ֑ים וְאָרְח֖וֹת צַדִּיקִ֣ים תִּשְׁמֹֽר׃ (נד) כִּֽי־ יְשָׁרִ֥ים יִשְׁכְּנוּ אָ֑רֶץ וּ֝תְמִימִ֗ים יִוָּ֥תְרוּ בָֽהּ׃ (נה) וּ֭רְשָׁעִים מֵאֶ֣רֶץ יִכָּרֵ֑תוּ וּ֝בוֹגְדִ֗ים יִסְּח֥וּ מִמֶּֽנָּה׃ (נו) בְּ֭נִי תּוֹרָתִ֣י אַל־ תִּשְׁכָּ֑ח וּ֝מִצְוֺתַ֗י יִצֹּ֥ר לִבֶּֽךָ׃ (נז) כִּ֤י אֹ֣רֶךְ יָ֭מִים וּשְׁנ֣וֹת חַיִּ֑ים וְ֝שָׁל֗וֹם יוֹסִ֥יפוּ לָֽךְ׃ (נח) חֶ֥סֶד וֶאֱמֶ֗ת אַֽל־ יַעַ֫זְבֻ֥ךָ קָשְׁרֵ֥ם עַל־ גַּרְגְּרוֹתֶ֑יךָ כָּ֝תְבֵ֗ם עַל־ ל֥וּחַ לִבֶּֽךָ׃ (נט) וּמְצָא־ חֵ֖ן וְשֵֽׂכֶל־ ט֑וֹב בְּעֵינֵ֖י אֱלֹהִ֣ים וְאָדָֽם׃ (ס) בְּטַ֣ח אֶל־ יְ֭הוָה בְּכָל־ לִבֶּ֑ךָ וְאֶל־ בִּֽ֝ינָתְךָ֗ אַל־ תִּשָּׁעֵֽן׃ (סא) בְּכָל־ דְּרָכֶ֥יךָ דָעֵ֑הוּ וְ֝ה֗וּא יְיַשֵּׁ֥ר אֹֽרְחֹתֶֽיךָ׃ (סב) אַל־ תְּהִ֣י חָכָ֣ם בְּעֵינֶ֑יךָ יְרָ֥א אֶת־ יְ֝הוָ֗ה וְס֣וּר מֵרָֽע׃ (סג) רִ֭פְאוּת תְּהִ֣י לְשָׁרֶּ֑ךָ וְ֝שִׁקּ֗וּי לְעַצְמוֹתֶֽיךָ׃ (סד) כַּבֵּ֣ד אֶת־ יְ֭הוָה מֵהוֹנֶ֑ךָ וּ֝מֵרֵאשִׁ֗ית כָּל־ תְּבוּאָתֶֽךָ׃ (סה) וְיִמָּלְא֣וּ אֲסָמֶ֣יךָ שָׂבָ֑ע וְ֝תִיר֗וֹשׁ יְקָבֶ֥יךָ יִפְרֹֽצוּ׃ (סו) מוּסַ֣ר יְ֭הוָה בְּנִ֣י אַל־ תִּמְאָ֑ס וְאַל־ תָּ֝קֹ֗ץ בְּתוֹכַחְתּֽוֹ׃ (סז) כִּ֤י אֶ֥ת אֲשֶׁ֣ר יֶאֱהַ֣ב יְהוָ֣ה יוֹכִ֑יחַ וּ֝כְאָ֗ב אֶת־ בֵּ֥ן יִרְצֶֽה׃ (סח) אַשְׁרֵ֣י אָ֭דָם מָצָ֣א חָכְמָ֑ה וְ֝אָדָ֗ם יָפִ֥יק תְּבוּנָֽה׃ (סט) כִּ֤י ט֣וֹב סַ֭חְרָהּ מִסְּחַר־ כָּ֑סֶף וּ֝מֵחָר֗וּץ תְּבוּאָתָֽהּ׃ (ע) יְקָ֣רָה הִ֭יא (מפניים) מִפְּנִינִ֑ים וְכָל־ חֲ֝פָצֶ֗יךָ לֹ֣א יִֽשְׁווּ־ בָֽהּ׃ (עא) אֹ֣רֶךְ יָ֭מִים בִּֽימִינָ֑הּ בִּ֝שְׂמֹאולָ֗הּ עֹ֣שֶׁר וְכָבֽוֹד׃ (עב) דְּרָכֶ֥יהָ דַרְכֵי־ נֹ֑עַם וְֽכָל־ נְתִ֖יבוֹתֶ֣יהָ שָׁלֽוֹם׃ (עג) עֵץ־ חַיִּ֣ים הִ֭יא לַמַּחֲזִיקִ֣ים בָּ֑הּ וְֽתֹמְכֶ֥יהָ מְאֻשָּֽׁר׃ (עד) יְֽהוָ֗ה בְּחָכְמָ֥ה יָֽסַד־ אָ֑רֶץ כּוֹנֵ֥ן שָׁ֝מַ֗יִם בִּתְבוּנָֽה׃ (עה) תַהְפֻּכ֬וֹת שָׂנֵֽאתִי׃ (פב) לִֽי־ עֵ֭צָה וְתוּשִׁיָּ֑ה אֲנִ֥י בִ֝ינָ֗ה לִ֣י גְבוּרָֽה׃ (פג) בִּ֭י מְלָכִ֣ים יִמְלֹ֑כוּ וְ֝רוֹזְנִ֗ים יְחֹ֣קְקוּ צֶֽדֶק׃ (פד) בִּ֭י שָׂרִ֣ים יָשֹׂ֑רוּ וּ֝נְדִיבִ֗ים כָּל־ שֹׁ֥פְטֵי צֶֽדֶק׃ (פה) אֲ֭נִי (אהביה) אֹהֲבַ֣י אֵהָ֑ב וּ֝מְשַׁחֲרַ֗י יִמְצָאֻֽנְנִי׃ (פו) עֹֽשֶׁר־ וְכָב֥וֹד אִתִּ֑י ה֥וֹן עָ֝תֵ֗ק וּצְדָקָֽה׃ (פז) ט֣וֹב פִּ֭רְיִי מֵחָר֣וּץ וּמִפָּ֑ז וּ֝תְבוּאָתִ֗י מִכֶּ֥סֶף נִבְחָֽר׃ (פח) בְּאֹֽרַח־ צְדָקָ֥ה אֲהַלֵּ֑ך בְּ֝ת֗וֹךְ נְתִיב֥וֹת מִשְׁפָּֽט׃ (פט) לְהַנְחִ֖יל אֹהֲבַ֥י ׀ יֵ֑שׁ וְאֹצְרֹ֖תֵיהֶ֣ם אֲמַלֵּֽא׃ (צ) יְֽהוָ֗ה קָ֭נָנִי רֵאשִׁ֣ית דַּרְכּ֑וֹ קֶ֖דֶם מִפְעָלָ֣יו מֵאָֽז׃ (צא) מֵ֭עוֹלָם נִסַּ֥כְתִּי מֵרֹ֗אשׁ מִקַּדְמֵי־ אָֽרֶץ׃ (צב) בְּאֵין־ תְּהֹמ֥וֹת חוֹלָ֑לְתִּי בְּאֵ֥ין מַ֝עְיָנ֗וֹת נִכְבַּדֵּי־ מָֽיִם׃ (צג) בְּטֶ֣רֶם הָרִ֣ים הָטְבָּ֑עוּ לִפְנֵ֖י גְבָע֣וֹת חוֹלָֽלְתִּי׃ (צד) עַד־ לֹ֣א עָ֭שָׂה אֶ֣רֶץ וְחוּצ֑וֹת וְ֝רֹ֗אשׁ עָפְר֥וֹת תֵּבֵֽל׃ (צה) בַּהֲכִינ֣וֹ שָׁ֭מַיִם שָׁ֣ם אָ֑נִי בְּח֥וּקוֹ ח֝֗וּג עַל־ פְּנֵ֥י תְהֽוֹם׃ (צו) בְּאַמְּצ֣וֹ שְׁחָקִ֣ים מִמָּ֑עַל בַּ֝עֲז֗וֹז עִינ֥וֹת תְּהוֹם׃ (צז) בְּשׂ֘וּמ֤וֹ לַיָּ֨ם ׀ חֻקּ֗וֹ וּ֭מַיִם לֹ֣א יַֽעַבְרוּ־ פִ֑יו בְּ֝חוּק֗וֹ מ֣וֹסְדֵי אָֽרֶץ׃ (צח) וָֽאֶהְיֶ֥ה אֶצְל֗וֹ אָ֫מ֥וֹן וָֽאֶהְיֶ֣ה שַׁ֭עֲשֻׁעִים י֤וֹם ׀ י֑וֹם מְשַׂחֶ֖קֶת לְפָנָ֣יו בְּכָל־ עֵֽת׃ (צט) מְ֭שַׂחֶקֶת בְּתֵבֵ֣ל אַרְצ֑וֹ וְ֝שַׁעֲשֻׁעַ֗י אֶת־ בְּנֵ֥י אָדָֽם׃ (ק) וְעַתָּ֣ה בָ֭נִים שִׁמְעוּ־ לִ֑י וְ֝אַשְׁרֵ֗י דְּרָכַ֥י יִשְׁמֹֽרוּ׃ (קא) שִׁמְע֖וּ מוּסָ֥ר וַחֲכָ֗מוּ וְאַל־ תִּפְרָֽעוּ׃ (קב) אַ֥שְֽׁרֵי אָדָם֮ שֹׁמֵ֪עַֽ֫ לִ֥י לִשְׁקֹ֣ד עַל־ דַּ֭לְתֹתַי י֤וֹם ׀ י֑וֹם לִ֝שְׁמֹ֗ר מְזוּזֹ֥ת פְּתָחָֽי׃ (קג) כִּ֣י מֹ֭צְאִי (מצאי) מָצָ֣א חַיִּ֑ים וַיָּ֥פֶק רָ֝צ֗וֹן מֵיְהוָֽה׃ (קד) וְֽ֭חֹטְאִי חֹמֵ֣ס נַפְשׁ֑וֹ כָּל־ מְ֝שַׂנְאַ֗י אָ֣הֲבוּ מָֽוֶת׃ (קה) חָ֭כְמוֹת בָּנְתָ֣ה בֵיתָ֑הּ חָצְבָ֖ה עַמּוּדֶ֣יהָ שִׁבְעָֽה׃ (קו) טָבְחָ֣ה טִ֭בְחָהּ מָסְכָ֣ה יֵינָ֑הּ אַ֝֗ף עָֽרְכָ֥ה שֻׁלְחָנָֽהּ׃ (קז) שָֽׁלְחָ֣ה נַעֲרֹתֶ֣יהָ תִקְרָ֑א עַל־ גַּ֝פֵּ֗י מְרֹ֣מֵי קָֽרֶת׃ (קח) מִי־ פֶ֭תִי יָסֻ֣ר הֵ֑נָּה חֲסַר־ לֵ֝֗ב אָ֣מְרָה לּֽוֹ׃ (קט) לְ֭כוּ לַחֲמ֣וּ בְֽלַחֲמִ֑י וּ֝שְׁת֗וּ בְּיַ֣יִן מָסָֽכְתִּי׃ (קי) עִזְב֣וּ פְתָאיִ֣ם וִֽחְי֑וּ וְ֝אִשְׁר֗וּ בְּדֶ֣רֶךְ בִּינָֽה׃ (קיא) יֹ֤סֵ֨ר ׀ לֵ֗ץ לֹקֵ֣חַֽ ל֣וֹ קָל֑וֹן וּמוֹכִ֖יחַ לְרָשָׁ֣ע מוּמֽוֹ׃ (קיב) אַל־ תּ֣וֹכַח לֵ֭ץ פֶּן־ יִשְׂנָאֶ֑ךָּ הוֹכַ֥ח לְ֝חָכָ֗ם וְיֶאֱהָבֶֽךָּ׃ (קיג) תֵּ֣ן לְ֭חָכָם וְיֶחְכַּם־ ע֑וֹד הוֹדַ֥ע לְ֝צַדִּ֗יק וְי֣וֹסֶף לֶֽקַח׃ (קיד) תְּחִלַּ֣ת חָ֭כְמָה יִרְאַ֣ת יְהוָ֑ה וְדַ֖עַת קְדֹשִׁ֣ים בִּינָֽה׃
