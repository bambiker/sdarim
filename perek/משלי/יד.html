משלי פרק יד לפי החלוקה לסדרים

פרק יד בספר משלי (35 פסוקים) נמצא במשלי סדר ג (פסוקים קכה–קכז) ובמשלי סדר ד (פסוקים א–לב).

משלי סדר ג

(א) כִּי־ בִ֭י יִרְבּ֣וּ יָמֶ֑יךָ וְיוֹסִ֥יפוּ לְּ֝ךָ֗ שְׁנ֣וֹת חַיִּֽים׃ (ב) אִם־ חָ֭כַמְתָּ חָכַ֣מְתָּ לָּ֑ךְ וְ֝לַ֗צְתָּ לְֽבַדְּךָ֥ תִשָּֽׂא׃ (ג) אֵ֣שֶׁת כְּ֭סִילוּת הֹֽמִיָּ֑ה פְּ֝תַיּ֗וּת וּבַל־ יָ֥דְעָה מָּֽה׃ (ד) וְֽ֭יָשְׁבָה לְפֶ֣תַח בֵּיתָ֑הּ עַל־ כִּ֝סֵּ֗א מְרֹ֣מֵי קָֽרֶת׃ (ה) לִקְרֹ֥א לְעֹֽבְרֵי־ דָ֑רֶךְ הַֽ֝מְיַשְּׁרִ֗ים אֹֽרְחוֹתָֽם׃ (ו) מִי־ פֶ֭תִי יָסֻ֣ר הֵ֑נָּה וַחֲסַר־ לֵ֝֗ב וְאָ֣מְרָה לּֽוֹ׃ (ז) מַֽיִם־ גְּנוּבִ֥ים יִמְתָּ֑קוּ וְלֶ֖חֶם סְתָרִ֣ים יִנְעָֽם׃ (ח) וְֽלֹא־ יָ֭דַע כִּֽי־ רְפָאִ֣ים שָׁ֑ם בְּעִמְקֵ֖י שְׁא֣וֹל קְרֻאֶֽיהָ׃ (ט) מִשְׁלֵ֗י שְׁלֹ֫מֹ֥ה בֵּ֣ן חָ֭כָם יְשַׂמַּח־ אָ֑ב וּבֵ֥ן כְּ֝סִ֗יל תּוּגַ֥ת אִמּֽוֹ׃ (י) לֹא־ י֭וֹעִילוּ אוֹצְר֣וֹת רֶ֑שַׁע וּ֝צְדָקָ֗ה תַּצִּ֥יל מִמָּֽוֶת׃ (יא) לֹֽא־ יַרְעִ֣יב יְ֭הוָה נֶ֣פֶשׁ צַדִּ֑יק וְהַוַּ֖ת רְשָׁעִ֣ים יֶהְדֹּֽף׃ (יב) רָ֗אשׁ עֹשֶׂ֥ה כַף־ רְמִיָּ֑ה וְיַ֖ד חָרוּצִ֣ים תַּעֲשִֽׁיר׃ (יג) אֹגֵ֣ר בַּ֭קַּיִץ בֵּ֣ן מַשְׂכִּ֑יל נִרְדָּ֥ם בַּ֝קָּצִ֗יר בֵּ֣ן מֵבִֽישׁ׃ (יד) בְּ֭רָכוֹת לְרֹ֣אשׁ צַדִּ֑יק וּפִ֥י רְ֝שָׁעִ֗ים יְכַסֶּ֥ה חָמָֽס׃ (טו) זֵ֣כֶר צַ֭דִּיק לִבְרָכָ֑ה וְשֵׁ֖ם רְשָׁעִ֣ים יִרְקָֽב׃ (טז) חֲכַם־ לֵ֭ב יִקַּ֣ח מִצְוֺ֑ת וֶאֱוִ֥יל שְׂ֝פָתַ֗יִם יִלָּבֵֽט׃ (יז) הוֹלֵ֣ךְ בַּ֭תֹּם יֵ֣לֶךְ בֶּ֑טַח וּמְעַקֵּ֥שׁ דְּ֝רָכָ֗יו יִוָּדֵֽעַ׃ (יח) קֹ֣רֵֽץ עַ֭יִן יִתֵּ֣ן עַצָּ֑בֶת וֶאֱוִ֥יל שְׂ֝פָתַ֗יִם יִלָּבֵֽט׃ (יט) מְק֣וֹר חַ֭יִּים פִּ֣י צַדִּ֑יק וּפִ֥י רְ֝שָׁעִ֗ים יְכַסֶּ֥ה חָמָֽס׃ (כ) שִׂ֭נְאָה תְּעוֹרֵ֣ר מְדָנִ֑ים וְעַ֥ל כָּל־ פְּ֝שָׁעִ֗ים תְּכַסֶּ֥ה אַהֲבָֽה׃ (כא) בְּשִׂפְתֵ֣י נָ֭בוֹן תִּמָּצֵ֣א חָכְמָ֑ה וְ֝שֵׁ֗בֶט לְגֵ֣ו חֲסַר־ לֵֽב׃ (כב) חֲכָמִ֥ים יִצְפְּנוּ־ דָ֑עַת וּפִֽי־ אֱ֝וִיל מְחִתָּ֥ה קְרֹבָֽה׃ (כג) ה֣וֹן עָ֭שִׁיר קִרְיַ֣ת עֻזּ֑וֹ מְחִתַּ֖ת דַּלִּ֣ים רֵישָֽׁם׃ (כד) פְּעֻלַּ֣ת צַדִּ֣יק לְחַיִּ֑ים תְּבוּאַ֖ת רָשָׁ֣ע לְחַטָּֽאת׃ (כה) אֹ֣רַח לְ֭חַיִּים שׁוֹמֵ֣ר מוּסָ֑ר וְעוֹזֵ֖ב תּוֹכַ֣חַת מַתְעֶֽה׃ (כו) מְכַסֶּ֣ה שִׂ֭נְאָה שִׂפְתֵי־ שָׁ֑קֶר וּמוֹצִ֥א דִ֝בָּ֗ה ה֣וּא כְסִֽיל׃ (כז) בְּרֹ֣ב דְּ֭בָרִים לֹ֣א יֶחְדַּל־ פָּ֑שַׁע וְחֹשֵׂ֖ךְ שְׂפָתָ֣יו מַשְׂכִּֽיל׃ (כח) כֶּ֣סֶף נִ֭בְחָר לְשׁ֣וֹן צַדִּ֑יק לֵ֖ב רְשָׁעִ֣ים כִּמְעָֽט׃ (כט) שִׂפְתֵ֣י צַ֭דִּיק יִרְע֣וּ רַבִּ֑ים וֶֽ֝אֱוִילִ֗ים בַּחֲסַר־ לֵ֥ב יָמֽוּתוּ׃ (ל) בִּרְכַּ֣ת יְ֭הוָה הִ֣יא תַעֲשִׁ֑יר וְלֹֽא־ יוֹסִ֖ף עֶ֣צֶב עִמָּֽהּ׃ (לא) כִּשְׂח֣וֹק לִ֭כְסִיל עֲשׂ֣וֹת זִמָּ֑ה וְ֝חָכְמָ֗ה לְאִ֣ישׁ תְּבוּנָֽה׃ (לב) מְגוֹרַ֣ת רָ֭שָׁע הִ֣יא תְבוֹאֶ֑נּוּ וְתַאֲוַ֖ת צַדִּיקִ֣ים יִתֵּֽן׃ (לג) כַּעֲב֣וֹר ס֭וּפָה וְאֵ֣ין רָשָׁ֑ע וְ֝צַדִּ֗יק יְס֣וֹד עוֹלָֽם׃ (לד) כַּחֹ֤מֶץ ׀ לַשִּׁנַּ֗יִם וְכֶעָשָׁ֥ן לָעֵינָ֑יִם כֵּ֥ן הֶ֝עָצֵ֗ל לְשֹׁלְחָֽיו׃ (לה) יִרְאַ֣ת יְ֭הוָה תּוֹסִ֣יף יָמִ֑ים וּשְׁנ֖וֹת רְשָׁעִ֣ים תִּקְצֹֽרְנָה׃ (לו) תּוֹחֶ֣לֶת צַדִּיקִ֣ים שִׂמְחָ֑ה וְתִקְוַ֖ת רְשָׁעִ֣ים תֹּאבֵֽד׃ (לז) מָע֣וֹז לַ֭תֹּם דֶּ֣רֶךְ יְהוָ֑ה וּ֝מְחִתָּ֗ה לְפֹ֣עֲלֵי אָֽוֶן׃ (לח) צַדִּ֣יק לְעוֹלָ֣ם בַּל־ יִמּ֑וֹט וּ֝רְשָׁעִ֗ים לֹ֣א יִשְׁכְּנוּ־ אָֽרֶץ׃ (לט) פִּֽי־ צַ֭דִּיק יָנ֣וּב חָכְמָ֑ה וּלְשׁ֥וֹן תַּ֝הְפֻּכ֗וֹת תִּכָּרֵֽת׃ (מ) שִׂפְתֵ֣י צַ֭דִּיק יֵדְע֣וּן רָצ֑וֹן וּפִ֥י רְ֝שָׁעִ֗ים תַּהְפֻּכֽוֹת׃ (מא) מֹאזְנֵ֣י מִ֭רְמָה תּוֹעֲבַ֣ת יְהוָ֑ה וְאֶ֖בֶן שְׁלֵמָ֣ה רְצוֹנֽוֹ׃ (מב) בָּֽא־ זָ֭דוֹן וַיָּבֹ֣א קָל֑וֹן וְֽאֶת־ צְנוּעִ֥ים חָכְמָֽה׃ (מג) תֻּמַּ֣ת יְשָׁרִ֣ים תַּנְחֵ֑ם וְסֶ֖לֶף בּוֹגְדִ֣ים (ושדם) יְשָׁדֵּֽם׃ (מד) לֹא־ יוֹעִ֣יל ה֭וֹן בְּי֣וֹם עֶבְרָ֑ה וּ֝צְדָקָ֗ה תַּצִּ֥יל מִמָּֽוֶת׃ (מה) צִדְקַ֣ת תָּ֭מִים תְּיַשֵּׁ֣ר דַּרְכּ֑וֹ וּ֝בְרִשְׁעָת֗וֹ יִפֹּ֥ל רָשָֽׁע׃ (מו) צִדְקַ֣ת יְ֭שָׁרִים תַּצִּילֵ֑ם וּ֝בְהַוַּ֗ת בֹּגְדִ֥ים יִלָּכֵֽדוּ׃ (מז) בְּמ֤וֹת אָדָ֣ם רָ֭שָׁע תֹּאבַ֣ד תִּקְוָ֑ה וְתוֹחֶ֖לֶת אוֹנִ֣ים אָבָֽדָה׃ (מח) צַ֭דִּיק מִצָּרָ֣ה נֶחֱלָ֑ץ וַיָּבֹ֖א רָשָׁ֣ע תַּחְתָּֽיו׃ (מט) בְּפֶ֗ה חָ֭נֵף יַשְׁחִ֣ת רֵעֵ֑הוּ וּ֝בְדַ֗עַת צַדִּיקִ֥ים יֵחָלֵֽצוּ׃ (נ) בְּט֣וּב צַ֭דִּיקִים תַּעֲלֹ֣ץ קִרְיָ֑ה וּבַאֲבֹ֖ד רְשָׁעִ֣ים רִנָּֽה׃ (נא) בְּבִרְכַּ֣ת יְ֭שָׁרִים תָּר֣וּם קָ֑רֶת וּבְפִ֥י רְ֝שָׁעִ֗ים תֵּהָרֵֽס׃ (נב) בָּז־ לְרֵעֵ֥הוּ חֲסַר־ לֵ֑ב וְאִ֖ישׁ תְּבוּנ֣וֹת יַחֲרִֽישׁ׃ (נג) הוֹלֵ֣ךְ רָ֭כִיל מְגַלֶּה־ סּ֑וֹד וְנֶאֱמַן־ ר֝֗וּחַ מְכַסֶּ֥ה דָבָֽר׃ (נד) בְּאֵ֣ין תַּ֭חְבֻּלוֹת יִפָּל־ עָ֑ם וּ֝תְשׁוּעָ֗ה בְּרֹ֣ב יוֹעֵֽץ׃ (נה) רַע־ יֵ֭רוֹעַ כִּי־ עָ֣רַב זָ֑ר וְשֹׂנֵ֖א תֹקְעִ֣ים בּוֹטֵֽחַ׃ (נו) אֵֽשֶׁת־ חֵ֭ן תִּתְמֹ֣ךְ כָּב֑וֹד וְ֝עָרִיצִ֗ים יִתְמְכוּ־ עֹֽשֶׁר׃ (נז) גֹּמֵ֣ל נַ֭פְשׁוֹ אִ֣ישׁ חָ֑סֶד וְעֹכֵ֥ר שְׁ֝אֵר֗וֹ אַכְזָרִֽי׃ (נח) רָשָׁ֗ע עֹשֶׂ֥ה פְעֻלַּת־ שָׁ֑קֶר וְזֹרֵ֥עַ צְ֝דָקָ֗ה שֶׂ֣כֶר אֱמֶֽת׃ (נט) כֵּן־ צְדָקָ֥ה לְחַיִּ֑ים וּמְרַדֵּ֖ף רָעָ֣ה לְמוֹתֽוֹ׃ (ס) תּוֹעֲבַ֣ת יְ֭הוָה עִקְּשֵׁי־ לֵ֑ב וּ֝רְצוֹנ֗וֹ תְּמִ֣ימֵי דָֽרֶךְ׃ (סא) יָ֣ד לְ֭יָד לֹא־ יִנָּ֣קֶה רָּ֑ע וְזֶ֖רַע צַדִּיקִ֣ים נִמְלָֽט׃ (סב) נֶ֣זֶם זָ֭הָב בְּאַ֣ף חֲזִ֑יר אִשָּׁ֥ה יָ֝פָ֗ה וְסָ֣רַת טָֽעַם׃ (סג) תַּאֲוַ֣ת צַדִּיקִ֣ים אַךְ־ ט֑וֹב תִּקְוַ֖ת רְשָׁעִ֣ים עֶבְרָֽה׃ (סד) יֵ֣שׁ מְ֭פַזֵּר וְנוֹסָ֥ף ע֑וֹד וְחוֹשֵׂ֥ךְ מִ֝יֹּ֗שֶׁר אַךְ־ לְמַחְסֽוֹר׃ (סה) נֶֽפֶשׁ־ בְּרָכָ֥ה תְדֻשָּׁ֑ן וּ֝מַרְוֶ֗ה גַּם־ ה֥וּא יוֹרֶֽא׃ (סו) מֹ֣נֵֽעַ בָּ֭ר יִקְּבֻ֣הוּ לְא֑וֹם וּ֝בְרָכָ֗ה לְרֹ֣אשׁ מַשְׁבִּֽיר׃ (סז) שֹׁ֣חֵֽר ט֭וֹב יְבַקֵּ֣שׁ רָצ֑וֹן וְדֹרֵ֖שׁ רָעָ֣ה תְבוֹאֶֽנּוּ׃ (סח) בּוֹטֵ֣חַ בְּ֭עָשְׁרוֹ ה֣וּא יִפֹּ֑ל וְ֝כֶעָלֶ֗ה צַדִּיקִ֥ים יִפְרָֽחוּ׃ (סט) עוֹכֵ֣ר בֵּ֭יתוֹ יִנְחַל־ ר֑וּחַ וְעֶ֥בֶד אֱ֝וִ֗יל לַחֲכַם־ לֵֽב׃ (ע) פְּֽרִי־ צַ֭דִּיק עֵ֣ץ חַיִּ֑ים וְלֹקֵ֖חַ נְפָשׂ֣וֹת חָכָֽם׃ (עא) הֵ֣ן צַ֭דִּיק בָּאָ֣רֶץ יְשֻׁלָּ֑ם אַ֝֗ף כִּֽי־ רָשָׁ֥ע וְחוֹטֵֽא׃ (עב) אֹהֵ֣ב מ֭וּסָר אֹ֣הֵֽב דָּ֑עַת וְשֹׂנֵ֖א תוֹכַ֣חַת בָּֽעַר׃ (עג) ט֗וֹב יָפִ֣יק רָ֭צוֹן מֵיְהוָ֑ה וְאִ֖ישׁ מְזִמּ֣וֹת יַרְשִֽׁיעַ׃ (עד) לֹא־ יִכּ֣וֹן אָדָ֣ם בְּרֶ֑שַׁע וְשֹׁ֥רֶשׁ צַ֝דִּיקִ֗ים בַּל־ יִמּֽוֹט׃ (עה) אֵֽשֶׁת־ חַ֭יִל עֲטֶ֣רֶת בַּעְלָ֑הּ וּכְרָקָ֖ב בְּעַצְמוֹתָ֣יו מְבִישָֽׁה׃ (עו) מַחְשְׁב֣וֹת צַדִּיקִ֣ים מִשְׁפָּ֑ט תַּחְבֻּל֖וֹת מִלֹּכֵ֥ד עִֽיר׃ (צח) בַּ֭חֵיק יוּטַ֣ל אֶת־ הַגּוֹרָ֑ל וּ֝מֵיְהוָ֗ה כָּל־ מִשְׁפָּטֽוֹ׃ (צט) ט֤וֹב פַּ֣ת חֲ֭רֵבָה וְשַׁלְוָה־ בָ֑הּ מִ֝בַּ֗יִת מָלֵ֥א זִבְחֵי־ רִֽיב׃ (ק) עֶֽבֶד־ מַשְׂכִּ֗יל יִ֭מְשֹׁל בְּבֵ֣ן מֵבִ֑ישׁ וּבְת֥וֹךְ אַ֝חִ֗ים יַחֲלֹ֥ק נַחֲלָֽה׃ (קא) מַצְרֵ֣ף לַ֭כֶּסֶף וְכ֣וּר לַזָּהָ֑ב וּבֹחֵ֖ן לִבּ֣וֹת יְהוָֽה׃ (קב) מֵ֭רַע מַקְשִׁ֣יב עַל־ שְׂפַת־ אָ֑וֶן שֶׁ֥קֶר מֵ֝זִין עַל־ לְשׁ֥וֹן הַוֺּֽת׃ (קג) לֹעֵ֣ג לָ֭רָשׁ חֵרֵ֣ף עֹשֵׂ֑הוּ שָׂמֵ֥חַ לְ֝אֵ֗יד לֹ֣א יִנָּקֶֽה׃ (קד) עֲטֶ֣רֶת זְ֭קֵנִים בְּנֵ֣י בָנִ֑ים וְתִפְאֶ֖רֶת בָּנִ֣ים אֲבוֹתָֽם׃ (קה) לֹא־ נָאוָ֣ה לְנָבָ֣ל שְׂפַת־ יֶ֑תֶר אַ֝֗ף כִּֽי־ לְנָדִ֥יב שְׂפַת־ שָֽׁקֶר׃ (קו) אֶֽבֶן־ חֵ֣ן הַ֭שֹּׁחַד בְּעֵינֵ֣י בְעָלָ֑יו אֶֽל־ כָּל־ אֲשֶׁ֖ר יִפְנֶ֣ה יַשְׂכִּֽיל׃ (קז) מְֽכַסֶּה־ פֶּ֭שַׁע מְבַקֵּ֣שׁ אַהֲבָ֑ה וְשֹׁנֶ֥ה בְ֝דָבָ֗ר מַפְרִ֥יד אַלּֽוּף׃ (קח) תֵּ֣חַת גְּעָרָ֣ה בְמֵבִ֑ין מֵהַכּ֖וֹת כְּסִ֣יל מֵאָֽה׃ (קט) אַךְ־ מְרִ֥י יְבַקֶּשׁ־ רָ֑ע וּמַלְאָ֥ךְ אַ֝כְזָרִ֗י יְשֻׁלַּח־ בּֽוֹ׃ (קי) פָּג֬וֹשׁ דֹּ֣ב שַׁכּ֣וּל בְּאִ֑ישׁ וְאַל־ כְּ֝סִ֗יל בְּאִוַּלְתּֽוֹ׃ (קיא) מֵשִׁ֣יב רָ֭עָה תַּ֣חַת טוֹבָ֑ה לֹא־ (תמיש) תָמ֥וּשׁ רָ֝עָ֗ה מִבֵּיתֽוֹ׃ (קיב) פּ֣וֹטֵֽר מַ֭יִם רֵאשִׁ֣ית מָד֑וֹן וְלִפְנֵ֥י הִ֝תְגַּלַּ֗ע הָרִ֥יב נְטֽוֹשׁ׃ (קיג) מַצְדִּ֣יק רָ֭שָׁע וּמַרְשִׁ֣יעַ צַדִּ֑יק תּוֹעֲבַ֥ת יְ֝הוָ֗ה גַּם־ שְׁנֵיהֶֽם׃ (קיד) לָמָּה־ זֶּ֣ה מְחִ֣יר בְּיַד־ כְּסִ֑יל לִקְנ֖וֹת חָכְמָ֣ה וְלֶב־ אָֽיִן׃ (קטו) בְּכָל־ עֵ֭ת אֹהֵ֣ב הָרֵ֑עַ וְאָ֥ח לְ֝צָרָ֗ה יִוָּלֵֽד׃ (קטז) אָדָ֣ם חֲסַר־ לֵ֭ב תּוֹקֵ֣עַ כָּ֑ף עֹרֵ֥ב עֲ֝רֻבָּ֗ה לִפְנֵ֥י רֵעֵֽהוּ׃ (קיז) אֹ֣הֵֽב פֶּ֭שַׁע אֹהֵ֣ב מַצָּ֑ה מַגְבִּ֥יהַּ פִּ֝תְח֗וֹ מְבַקֶּשׁ־ שָֽׁבֶר׃ (קיח) עִקֶּשׁ־ לֵ֭ב לֹ֣א יִמְצָא־ ט֑וֹב וְנֶהְפָּ֥ךְ בִּ֝לְשׁוֹנ֗וֹ יִפּ֥וֹל בְּרָעָֽה׃ (קיט) יֹלֵ֣ד כְּ֭סִיל לְת֣וּגָה ל֑וֹ וְלֹֽא־ יִ֝שְׂמַ֗ח אֲבִ֣י נָבָֽל׃ (קכ) לֵ֣ב שָׂ֭מֵחַ יֵיטִ֣ב גֵּהָ֑ה וְר֥וּחַ נְ֝כֵאָ֗ה תְּיַבֶּשׁ־ גָּֽרֶם׃ (קכא) שֹׁ֣חַד מֵ֭חֵיק רָשָׁ֣ע יִקָּ֑ח לְ֝הַטּ֗וֹת אָרְח֥וֹת מִשְׁפָּֽט׃ (קכב) אֶת־ פְּנֵ֣י מֵבִ֣ין חָכְמָ֑ה וְעֵינֵ֥י כְ֝סִ֗יל בִּקְצֵה־ אָֽרֶץ׃ (קכג) כַּ֣עַס לְ֭אָבִיו בֵּ֣ן כְּסִ֑יל וּ֝מֶ֗מֶר לְיוֹלַדְתּֽוֹ׃ (קכד) גַּ֤ם עֲנ֣וֹשׁ לַצַּדִּ֣יק לֹא־ ט֑וֹב לְהַכּ֖וֹת נְדִיבִ֣ים עַל־ יֹֽשֶׁר׃ (קכה) חוֹשֵׂ֣ךְ אֲ֭מָרָיו יוֹדֵ֣עַ דָּ֑עַת (וקר)־ יְקַר־ ר֝֗וּחַ אִ֣ישׁ תְּבוּנָֽה׃ (קכו) גַּ֤ם אֱוִ֣יל מַ֭חֲרִישׁ חָכָ֣ם יֵחָשֵׁ֑ב אֹטֵ֖ם שְׂפָתָ֣יו נָבֽוֹן׃ (קכז) לְֽ֭תַאֲוָה יְבַקֵּ֣שׁ נִפְרָ֑ד בְּכָל־ תּ֝וּשִׁיָּ֗ה יִתְגַּלָּֽע׃ (קכח) לֹֽא־ יַחְפֹּ֣ץ כְּ֭סִיל בִּתְבוּנָ֑ה כִּ֝֗י אִֽם־ בְּהִתְגַּלּ֥וֹת לִבּֽוֹ׃ (קכט) בְּֽבוֹא־ רָ֭שָׁע בָּ֣א גַם־ בּ֑וּז וְֽעִם־ קָל֥וֹן חֶרְפָּֽה׃ (קל) מַ֣יִם עֲ֭מֻקִּים דִּבְרֵ֣י פִי־ אִ֑ישׁ נַ֥חַל נֹ֝בֵ֗עַ מְק֣וֹר חָכְמָֽה׃ (קלא) שְׂאֵ֣ת פְּנֵי־ רָשָׁ֣ע לֹא־ ט֑וֹב לְהַטּ֥וֹת צַ֝דִּ֗יק בַּמִּשְׁפָּֽט׃ (קלב) שִׂפְתֵ֣י כְ֭סִיל יָבֹ֣אֽוּ בְרִ֑יב וּ֝פִ֗יו לְֽמַהֲלֻמ֥וֹת יִקְרָֽא׃ (קלג) פִּֽי־ כְ֭סִיל מְחִתָּה־ ל֑וֹ וּ֝שְׂפָתָ֗יו מוֹקֵ֥שׁ נַפְשֽׁוֹ׃ (קלד) דִּבְרֵ֣י נִ֭רְגָּן כְּמִֽתְלַהֲמִ֑ים וְ֝הֵ֗ם יָרְד֥וּ חַדְרֵי־ בָֽטֶן׃ (קלה) גַּ֭ם מִתְרַפֶּ֣ה בִמְלַאכְתּ֑וֹ אָ֥ח ה֝֗וּא לְבַ֣עַל מַשְׁחִֽית׃
