משלי פרק כב לפי החלוקה לסדרים

פרק כב בספר משלי (29 פסוקים) נמצא במשלי סדר ה (פסוקים קו–קכה) ובמשלי סדר ו (פסוקים א–ט).

משלי סדר ה

(א) מִגְדַּל־ עֹ֭ז שֵׁ֣ם יְהוָ֑ה בּֽוֹ־ יָר֖וּץ צַדִּ֣יק וְנִשְׂגָּֽב׃ (ב) ה֣וֹן עָ֭שִׁיר קִרְיַ֣ת עֻזּ֑וֹ וּכְחוֹמָ֥ה נִ֝שְׂגָּבָ֗ה בְּמַשְׂכִּיתֽוֹ׃ (ג) לִפְנֵי־ שֶׁ֭בֶר יִגְבַּ֣הּ לֵב־ אִ֑ישׁ וְלִפְנֵ֖י כָב֣וֹד עֲנָוָֽה׃ (ד) מֵשִׁ֣יב דָּ֭בָר בְּטֶ֣רֶם יִשְׁמָ֑ע אִוֶּ֥לֶת הִיא־ ל֝֗וֹ וּכְלִמָּֽה׃ (ה) רֽוּחַ־ אִ֭ישׁ יְכַלְכֵּ֣ל מַחֲלֵ֑הוּ וְר֥וּחַ נְ֝כֵאָ֗ה מִ֣י יִשָּׂאֶֽנָּה׃ (ו) לֵ֣ב נָ֭בוֹן יִקְנֶה־ דָּ֑עַת וְאֹ֥זֶן חֲ֝כָמִ֗ים תְּבַקֶּשׁ־ דָּֽעַת׃ (ז) מַתָּ֣ן אָ֭דָם יַרְחִ֣יב ל֑וֹ וְלִפְנֵ֖י גְדֹלִ֣ים יַנְחֶֽנּוּ׃ (ח) צַדִּ֣יק הָרִאשׁ֣וֹן בְּרִיב֑וֹ (יבא)־ וּבָֽא־ רֵ֝עֵ֗הוּ וַחֲקָרֽוֹ׃ (ט) מִ֭דְיָנִים יַשְׁבִּ֣ית הַגּוֹרָ֑ל וּבֵ֖ין עֲצוּמִ֣ים יַפְרִֽיד׃ (י) אָ֗ח נִפְשָׁ֥ע מִקִּרְיַת־ עֹ֑ז (ומדונים) וּ֝מִדְיָנִ֗ים כִּבְרִ֥יחַ אַרְמֽוֹן׃ (יא) מִפְּרִ֣י פִי־ אִ֭ישׁ תִּשְׂבַּ֣ע בִּטְנ֑וֹ תְּבוּאַ֖ת שְׂפָתָ֣יו יִשְׂבָּֽע׃ (יב) מָ֣וֶת וְ֭חַיִּים בְּיַד־ לָשׁ֑וֹן וְ֝אֹהֲבֶ֗יהָ יֹאכַ֥ל פִּרְיָֽהּ׃ (יג) מָצָ֣א אִ֭שָּׁה מָ֣צָא ט֑וֹב וַיָּ֥פֶק רָ֝צ֗וֹן מֵיְהוָֽה׃ (יד) תַּחֲנוּנִ֥ים יְדַבֶּר־ רָ֑שׁ וְ֝עָשִׁ֗יר יַעֲנֶ֥ה עַזּֽוֹת׃ (טו) אִ֣ישׁ רֵ֭עִים לְהִתְרֹעֵ֑עַ וְיֵ֥שׁ אֹ֝הֵ֗ב דָּבֵ֥ק מֵאָֽח׃ (טז) טֽוֹב־ רָ֭שׁ הוֹלֵ֣ךְ בְּתֻמּ֑וֹ מֵעִקֵּ֥שׁ שְׂ֝פָתָ֗יו וְה֣וּא כְסִֽיל׃ (יז) גַּ֤ם בְּלֹא־ דַ֣עַת נֶ֣פֶשׁ לֹא־ ט֑וֹב וְאָ֖ץ בְּרַגְלַ֣יִם חוֹטֵֽא׃ (יח) אִוֶּ֣לֶת אָ֭דָם תְּסַלֵּ֣ף דַּרְכּ֑וֹ וְעַל־ יְ֝הוָ֗ה יִזְעַ֥ף לִבּֽוֹ׃ (יט) ה֗וֹן יֹ֭סִיף רֵעִ֣ים רַבִּ֑ים וְ֝דָ֗ל מֵרֵ֥עהוּ יִפָּרֵֽד׃ (כ) עֵ֣ד שְׁ֭קָרִים לֹ֣א יִנָּקֶ֑ה וְיָפִ֥יחַ כְּ֝זָבִ֗ים לֹ֣א יִמָּלֵֽט׃ (כא) רַ֭בִּים יְחַלּ֣וּ פְנֵֽי־ נָדִ֑יב וְכָל־ הָ֝רֵ֗עַ לְאִ֣ישׁ מַתָּֽן׃ (כב) כָּ֥ל אֲחֵי־ רָ֨שׁ ׀ שְֽׂנֵאֻ֗הוּ אַ֤ף כִּ֣י מְ֭רֵעֵהוּ רָחֲק֣וּ מִמֶּ֑נּוּ מְרַדֵּ֖ף אֲמָרִ֣ים (לא)־ לוֹ־ הֵֽמָּה׃ (כג) קֹֽנֶה־ לֵּ֭ב אֹהֵ֣ב נַפְשׁ֑וֹ שֹׁמֵ֥ר תְּ֝בוּנָ֗ה לִמְצֹא־ טֽוֹב׃ (כד) עֵ֣ד שְׁ֭קָרִים לֹ֣א יִנָּקֶ֑ה וְיָפִ֖יחַ כְּזָבִ֣ים יֹאבֵֽד׃ (כה) לֹֽא־ נָאוֶ֣ה לִכְסִ֣יל תַּעֲנ֑וּג אַ֝֗ף כִּֽי־ לְעֶ֤בֶד ׀ מְשֹׁ֬ל בְּשָׂרִֽים׃ (כו) שֵׂ֣כֶל אָ֭דָם הֶאֱרִ֣יךְ אַפּ֑וֹ וְ֝תִפאַרְתּ֗וֹ עֲבֹ֣ר עַל־ פָּֽשַׁע׃ (כז) נַ֣הַם כַּ֭כְּפִיר זַ֣עַף מֶ֑לֶךְ וּכְטַ֖ל עַל־ עֵ֣שֶׂב רְצוֹנֽוֹ׃ (כח) הַוֺּ֣ת לְ֭אָבִיו בֵּ֣ן כְּסִ֑יל וְדֶ֥לֶף טֹ֝רֵ֗ד מִדְיְנֵ֥י אִשָּֽׁה׃ (כט) בַּ֣יִת וָ֭הוֹן נַחֲלַ֣ת אָב֑וֹת וּ֝מֵיְהוָ֗ה אִשָּׁ֥ה מַשְׂכָּֽלֶת׃ (ל) עַ֭צְלָה תַּפִּ֣יל תַּרְדֵּמָ֑ה וְנֶ֖פֶשׁ רְמִיָּ֣ה תִרְעָֽב׃ (לא) שֹׁמֵ֣ר מִ֭צְוָה שֹׁמֵ֣ר נַפְשׁ֑וֹ בּוֹזֵ֖ה דְרָכָ֣יו (יומת) יָמֽוּת׃ (לב) מַלְוֵ֣ה יְ֭הוָה ח֣וֹנֵֽן דָּ֑ל וּ֝גְמֻל֗וֹ יְשַׁלֶּם־ לֽוֹ׃ (לג) יַסֵּ֣ר בִּ֭נְךָ כִּי־ יֵ֣שׁ תִּקְוָ֑ה וְאֶל־ הֲ֝מִית֗וֹ אַל־ תִּשָּׂ֥א נַפְשֶֽׁךָ׃ (לד) (גרל)־ גְּֽדָל־ חֵ֭מָה נֹ֣שֵׂא עֹ֑נֶשׁ כִּ֥י אִם־ תַּ֝צִּ֗יל וְע֣וֹד תּוֹסִֽף׃ (לה) שְׁמַ֣ע עֵ֭צָה וְקַבֵּ֣ל מוּסָ֑ר לְ֝מַ֗עַן תֶּחְכַּ֥ם בְּאַחֲרִיתֶֽךָ׃ (לו) רַבּ֣וֹת מַחֲשָׁב֣וֹת בְּלֶב־ אִ֑ישׁ וַעֲצַ֥ת יְ֝הוָ֗ה הִ֣יא תָקֽוּם׃ (לז) תַּאֲוַ֣ת אָדָ֣ם חַסְדּ֑וֹ וְטֽוֹב־ רָ֝שׁ מֵאִ֥ישׁ כָּזָֽב׃ (לח) יִרְאַ֣ת יְהוָ֣ה לְחַיִּ֑ים וְשָׂבֵ֥עַ יָ֝לִ֗ין בַּל־ יִפָּ֥קֶד רָֽע׃ (לט) טָ֘מַ֤ן עָצֵ֣ל יָ֭דוֹ בַּצַּלָּ֑חַת גַּם־ אֶל־ פִּ֝֗יהוּ לֹ֣א יְשִׁיבֶֽנָּה׃ (מ) לֵ֣ץ תַּ֭כֶּה וּפֶ֣תִי יַעְרִ֑ם וְהוֹכִ֥יחַ לְ֝נָב֗וֹן יָבִ֥ין דָּֽעַת׃ (מא) מְֽשַׁדֶּד־ אָ֭ב יַבְרִ֣יחַ אֵ֑ם בֵּ֝֗ן מֵבִ֥ישׁ וּמַחְפִּֽיר׃ (מב) חַֽדַל־ בְּ֭נִי לִשְׁמֹ֣עַ מוּסָ֑ר לִ֝שְׁג֗וֹת מֵֽאִמְרֵי־ דָֽעַת׃ (מג) עֵ֣ד בְּ֭לִיַּעַל יָלִ֣יץ מִשְׁפָּ֑ט וּפִ֥י רְ֝שָׁעִ֗ים יְבַלַּע־ אָֽוֶן׃ (מד) נָכ֣וֹנוּ לַלֵּצִ֣ים שְׁפָטִ֑ים וּ֝מַהֲלֻמ֗וֹת לְגֵ֣ו כְּסִילִֽים׃ (מה) לֵ֣ץ הַ֭יַּין הֹמֶ֣ה שֵׁכָ֑ר וְכָל־ שֹׁ֥גֶה בּ֝֗וֹ לֹ֣א יֶחְכָּֽם׃ (מו) נַ֣הַם כַּ֭כְּפִיר אֵ֣ימַת מֶ֑לֶךְ מִ֝תְעַבְּר֗וֹ חוֹטֵ֥א נַפְשֽׁוֹ׃ (מז) כָּב֣וֹד לָ֭אִישׁ שֶׁ֣בֶת מֵרִ֑יב וְכָל־ אֱ֝וִ֗יל יִתְגַּלָּֽע׃ (מח) מֵ֭חֹרֶף עָצֵ֣ל לֹא־ יַחֲרֹ֑שׁ (ישאל) וְשָׁאַ֖ל בַּקָּצִ֣יר וָאָֽיִן׃ (מט) מַ֣יִם עֲ֭מֻקִּים עֵצָ֣ה בְלֶב־ אִ֑ישׁ וְאִ֖ישׁ תְּבוּנָ֣ה יִדְלֶֽנָּה׃ (נ) רָב־ אָדָ֗ם יִ֭קְרָא אִ֣ישׁ חַסְדּ֑וֹ וְאִ֥ישׁ אֱ֝מוּנִ֗ים מִ֣י יִמְצָֽא׃ (נא) מִתְהַלֵּ֣ךְ בְּתֻמּ֣וֹ צַדִּ֑יק אַשְׁרֵ֖י בָנָ֣יו אַחֲרָֽיו׃ (נב) מֶ֗לֶךְ יוֹשֵׁ֥ב עַל־ כִּסֵּא־ דִ֑ין מְזָרֶ֖ה בְעֵינָ֣יו כָּל־ רָֽע׃ (נג) מִֽי־ יֹ֭אמַר זִכִּ֣יתִי לִבִּ֑י טָ֝הַ֗רְתִּי מֵחַטָּאתִֽי׃ (נד) אֶ֣בֶן וָ֭אֶבֶן אֵיפָ֣ה וְאֵיפָ֑ה תּוֹעֲבַ֥ת יְ֝הוָ֗ה גַּם־ שְׁנֵיהֶֽם׃ (נה) גַּ֣ם בְּ֭מַעֲלָלָיו יִתְנַכֶּר־ נָ֑עַר אִם־ זַ֖ךְ וְאִם־ יָשָׁ֣ר פָּעֳלֽוֹ׃ (נו) אֹ֣זֶן שֹׁ֭מַעַת וְעַ֣יִן רֹאָ֑ה יְ֝הוָ֗ה עָשָׂ֥ה גַם־ שְׁנֵיהֶֽם׃ (נז) אַל־ תֶּֽאֱהַ֣ב שֵׁ֭נָה פֶּן־ תִּוָּרֵ֑שׁ פְּקַ֖ח עֵינֶ֣יךָ שְֽׂבַֽע־ לָֽחֶם׃ (נח) רַ֣ע רַ֭ע יֹאמַ֣ר הַקּוֹנֶ֑ה וְאֹזֵ֥ל ל֝֗וֹ אָ֣ז יִתְהַלָּֽל׃ (נט) יֵ֣שׁ זָ֭הָב וְרָב־ פְּנִינִ֑ים וּכְלִ֥י יְ֝קָ֗ר שִׂפְתֵי־ דָֽעַת׃ (ס) לְֽקַח־ בִּ֭גְדוֹ כִּי־ עָ֣רַב זָ֑ר וּבְעַ֖ד (נכרים) נָכְרִיָּ֣ה חַבְלֵֽהוּ׃ (סא) עָרֵ֣ב לָ֭אִישׁ לֶ֣חֶם שָׁ֑קֶר וְ֝אַחַ֗ר יִמָּֽלֵא־ פִ֥יהוּ חָצָֽץ׃ (סב) מַ֭חֲשָׁבוֹת בְּעֵצָ֣ה תִכּ֑וֹן וּ֝בְתַחְבֻּל֗וֹת עֲשֵׂ֣ה מִלְחָמָֽה׃ (סג) גּֽוֹלֶה־ סּ֭וֹד הוֹלֵ֣ךְ רָכִ֑יל וּלְפֹתֶ֥ה שְׂ֝פָתָ֗יו לֹ֣א תִתְעָרָֽב׃ (סד) מְ֭קַלֵּל אָבִ֣יו וְאִמּ֑וֹ יִֽדְעַ֥ךְ נֵ֝ר֗וֹ (באישון) בֶּאֱשׁ֥וּן חֹֽשֶׁךְ׃ (סה) נַ֭חֲלָה (מבחלת) מְבֹהֶ֣לֶת בָּרִאשֹׁנָ֑ה וְ֝אַחֲרִיתָ֗הּ לֹ֣א תְבֹרָֽךְ׃ (סו) אַל־ תֹּאמַ֥ר אֲשַׁלְּמָה־ רָ֑ע קַוֵּ֥ה לַֽ֝יהוָ֗ה וְיֹ֣שַֽׁע לָֽךְ׃ (סז) תּוֹעֲבַ֣ת יְ֭הוָה אֶ֣בֶן וָאָ֑בֶן וּמֹאזְנֵ֖י מִרְמָ֣ה לֹא־ טֽוֹב׃ (סח) מֵיהוָ֥ה מִצְעֲדֵי־ גָ֑בֶר וְ֝אָדָ֗ם מַה־ יָּבִ֥ין דַּרְכּֽוֹ׃ (סט) מוֹקֵ֣שׁ אָ֭דָם יָ֣לַע קֹ֑דֶשׁ וְאַחַ֖ר נְדָרִ֣ים לְבַקֵּֽר׃ (ע) מְזָרֶ֣ה רְ֭שָׁעִים מֶ֣לֶךְ חָכָ֑ם וַיָּ֖שֶׁב עֲלֵיהֶ֣ם אוֹפָֽן׃ (עא) נֵ֣ר יְ֭הוָה נִשְׁמַ֣ת אָדָ֑ם חֹ֝פֵ֗שׂ כָּל־ חַדְרֵי־ בָֽטֶן׃ (עב) חֶ֣סֶד
