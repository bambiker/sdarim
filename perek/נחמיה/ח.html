נחמיה פרק ח לפי החלוקה לסדרים

פרק ח בספר נחמיה (18 פסוקים) נמצא בעזרא ונחמיה סדר ז (פסוקים עח–פו) ובעזרא ונחמיה סדר ח (פסוקים א–ט).

עזרא ונחמיה סדר ז

(א) וַתִּשְׁלַם֙ הַֽחוֹמָ֔ה בְּעֶשְׂרִ֥ים וַחֲמִשָּׁ֖ה לֶאֱל֑וּל לַחֲמִשִּׁ֥ים וּשְׁנַ֖יִם יֽוֹם׃ (ב) וַיְהִ֗י כַּאֲשֶׁ֤ר שָֽׁמְעוּ֙ כָּל־ א֣וֹיְבֵ֔ינוּ וַיִּֽרְא֗וּ כָּל־ הַגּוֹיִם֙ אֲשֶׁ֣ר סְבִֽיבֹתֵ֔ינוּ וַיִּפְּל֥וּ מְאֹ֖ד בְּעֵינֵיהֶ֑ם וַיֵּ֣דְע֔וּ כִּ֚י מֵאֵ֣ת אֱלֹהֵ֔ינוּ נֶעֶשְׂתָ֖ה הַמְּלָאכָ֥ה הַזֹּֽאת׃ (ג) גַּ֣ם ׀ בַּיָּמִ֣ים הָהֵ֗ם מַרְבִּ֞ים חֹרֵ֤י יְהוּדָה֙ אִגְּרֹ֣תֵיהֶ֔ם הוֹלְכ֖וֹת עַל־ טוֹבִיָּ֑ה וַאֲשֶׁ֥ר לְטוֹבִיָּ֖ה בָּא֥וֹת אֲלֵיהֶֽם׃ (ד) כִּי־ רַבִּ֣ים בִּֽיהוּדָ֗ה בַּעֲלֵ֤י שְׁבוּעָה֙ ל֔וֹ כִּי־ חָתָ֥ן ה֖וּא לִשְׁכַנְיָ֣ה בֶן־ אָרַ֑ח וִֽיהוֹחָנָ֣ן בְּנ֔וֹ לָקַ֕ח אֶת־ בַּת־ מְשֻׁלָּ֖ם בֶּ֥ן בֶּֽרֶכְיָֽה׃ (ה) גַּ֣ם טוֹבֹתָ֗יו הָי֤וּ אֹמְרִים֙ לְפָנַ֔י וּדְבָרַ֕י הָי֥וּ מוֹצִיאִ֖ים ל֑וֹ אִגְּר֛וֹת שָׁלַ֥ח טוֹבִיָּ֖ה לְיָֽרְאֵֽנִי׃ (ו) וַיְהִ֗י כַּאֲשֶׁ֤ר נִבְנְתָה֙ הַחוֹמָ֔ה וָאַעֲמִ֖יד הַדְּלָת֑וֹת וַיִּפָּֽקְד֛וּ הַשּׁוֹעֲרִ֥ים וְהַמְשֹׁרְרִ֖ים וְהַלְוִיִּֽם׃ (ז) וָאֲצַוֶּ֞ה אֶת־ חֲנָ֣נִי אָחִ֗י וְאֶת־ חֲנַנְיָ֛ה שַׂ֥ר הַבִּירָ֖ה עַל־ יְרוּשָׁלִָ֑ם כִּי־ הוּא֙ כְּאִ֣ישׁ אֱמֶ֔ת וְיָרֵ֥א אֶת־ הָאֱלֹהִ֖ים מֵרַבִּֽים׃ (ח) (ויאמר) וָאֹמַ֣ר לָהֶ֗ם לֹ֣א יִפָּֽתְח֞וּ שַׁעֲרֵ֤י יְרוּשָׁלִַ֙ם֙ עַד־ חֹ֣ם הַשֶּׁ֔מֶשׁ וְעַ֨ד הֵ֥ם עֹמְדִ֛ים יָגִ֥יפוּ הַדְּלָת֖וֹת וֶאֱחֹ֑זוּ וְהַעֲמֵ֗יד מִשְׁמְרוֹת֙ יֹשְׁבֵ֣י יְרוּשָׁלִַ֔ם אִ֚ישׁ בְּמִשְׁמָר֔וֹ וְאִ֖ישׁ נֶ֥גֶד בֵּיתֽוֹ׃ (ט) וְהָעִ֞יר רַחֲבַ֤ת יָדַ֙יִם֙ וּגְדוֹלָ֔ה וְהָעָ֥ם מְעַ֖ט בְּתוֹכָ֑הּ וְאֵ֥ין בָּתִּ֖ים בְּנוּיִֽם׃ (י) וַיִּתֵּ֤ן אֱלֹהַי֙ אֶל־ לִבִּ֔י וָאֶקְבְּצָ֞ה אֶת־ הַחֹרִ֧ים וְאֶת־ הַסְּגָנִ֛ים וְאֶת־ הָעָ֖ם לְהִתְיַחֵ֑שׂ וָֽאֶמְצָ֗א סֵ֤פֶר הַיַּ֙חַשׂ֙ הָעוֹלִ֣ים בָּרִאשׁוֹנָ֔ה וָאֶמְצָ֖א כָּת֥וּב בּֽוֹ׃ (יא) אֵ֣לֶּה ׀ בְּנֵ֣י הַמְּדִינָ֗ה הָעֹלִים֙ מִשְּׁבִ֣י הַגּוֹלָ֔ה אֲשֶׁ֣ר הֶגְלָ֔ה נְבוּכַדְנֶצַּ֖ר מֶ֣לֶךְ בָּבֶ֑ל וַיָּשׁ֧וּבוּ לִֽירוּשָׁלִַ֛ם וְלִיהוּדָ֖ה אִ֥ישׁ לְעִירֽוֹ׃ (יב) הַבָּאִ֣ים עִם־ זְרֻבָּבֶ֗ל יֵשׁ֡וּעַ נְחֶמְיָ֡ה עֲ֠זַרְיָה רַֽעַמְיָ֨ה נַחֲמָ֜נִי מָרְדֳּכַ֥י בִּלְשָׁ֛ן מִסְפֶּ֥רֶת בִּגְוַ֖י נְח֣וּם בַּעֲנָ֑ה מִסְפַּ֕ר אַנְשֵׁ֖י עַ֥ם יִשְׂרָאֵֽל׃ (יג) בְּנֵ֣י פַרְעֹ֔שׁ אַלְפַּ֕יִם מֵאָ֖ה וְשִׁבְעִ֥ים וּשְׁנָֽיִם׃ (יד) בְּנֵ֣י שְׁפַטְיָ֔ה שְׁלֹ֥שׁ מֵא֖וֹת שִׁבְעִ֥ים וּשְׁנָֽיִם׃ (טו) בְּנֵ֣י אָרַ֔ח שֵׁ֥שׁ מֵא֖וֹת חֲמִשִּׁ֥ים וּשְׁנָֽיִם׃ (טז) בְּנֵֽי־ פַחַ֥ת מוֹאָ֛ב לִבְנֵ֥י יֵשׁ֖וּעַ וְיוֹאָ֑ב אַלְפַּ֕יִם וּשְׁמֹנֶ֥ה מֵא֖וֹת שְׁמֹנָ֥ה עָשָֽׂר׃ (יז) בְּנֵ֣י עֵילָ֔ם אֶ֕לֶף מָאתַ֖יִם חֲמִשִּׁ֥ים וְאַרְבָּעָֽה׃ (יח) בְּנֵ֣י זַתּ֔וּא שְׁמֹנֶ֥ה מֵא֖וֹת אַרְבָּעִ֥ים וַחֲמִשָּֽׁה׃ (יט) בְּנֵ֣י זַכָּ֔י שְׁבַ֥ע מֵא֖וֹת וְשִׁשִּֽׁים׃ (כ) בְּנֵ֣י בִנּ֔וּי שֵׁ֥שׁ מֵא֖וֹת אַרְבָּעִ֥ים וּשְׁמֹנָֽה׃ (כא) בְּנֵ֣י בֵבָ֔י שֵׁ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וּשְׁמֹנָֽה׃ (כב) בְּנֵ֣י עַזְגָּ֔ד אַלְפַּ֕יִם שְׁלֹ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וּשְׁנָֽיִם׃ (כג) בְּנֵי֙ אֲדֹ֣נִיקָ֔ם שֵׁ֥שׁ מֵא֖וֹת שִׁשִּׁ֥ים וְשִׁבְעָֽה׃ (כד) בְּנֵ֣י בִגְוָ֔י אַלְפַּ֖יִם שִׁשִּׁ֥ים וְשִׁבְעָֽה׃ (כה) בְּנֵ֣י עָדִ֔ין שֵׁ֥שׁ מֵא֖וֹת חֲמִשִּׁ֥ים וַחֲמִשָּֽׁה׃ (כו) בְּנֵֽי־ אָטֵ֥ר לְחִזְקִיָּ֖ה תִּשְׁעִ֥ים וּשְׁמֹנָֽה׃ (כז) בְּנֵ֣י חָשֻׁ֔ם שְׁלֹ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וּשְׁמֹנָֽה׃ (כח) בְּנֵ֣י בֵצָ֔י שְׁלֹ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וְאַרְבָּעָֽה׃ (כט) בְּנֵ֣י חָרִ֔יף מֵאָ֖ה שְׁנֵ֥ים עָשָֽׂר׃ (ל) בְּנֵ֥י גִבְע֖וֹן תִּשְׁעִ֥ים וַחֲמִשָּֽׁה׃ (לא) אַנְשֵׁ֤י בֵֽית־ לֶ֙חֶם֙ וּנְטֹפָ֔ה מֵאָ֖ה שְׁמֹנִ֥ים וּשְׁמֹנָֽה׃ (לב) אַנְשֵׁ֣י עֲנָת֔וֹת מֵאָ֖ה עֶשְׂרִ֥ים וּשְׁמֹנָֽה׃ (לג) אַנְשֵׁ֥י בֵית־ עַזְמָ֖וֶת אַרְבָּעִ֥ים וּשְׁנָֽיִם׃ (לד) אַנְשֵׁ֨י קִרְיַ֤ת יְעָרִים֙ כְּפִירָ֣ה וּבְאֵר֔וֹת שְׁבַ֥ע מֵא֖וֹת אַרְבָּעִ֥ים וּשְׁלֹשָֽׁה׃ (לה) אַנְשֵׁ֤י הָֽרָמָה֙ וָגָ֔בַע שֵׁ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וְאֶחָֽד׃ (לו) אַנְשֵׁ֣י מִכְמָ֔ס מֵאָ֖ה וְעֶשְׂרִ֥ים וּשְׁנָֽיִם׃ (לז) אַנְשֵׁ֤י בֵֽית־ אֵל֙ וְהָעָ֔י מֵאָ֖ה עֶשְׂרִ֥ים וּשְׁלֹשָֽׁה׃ (לח) אַנְשֵׁ֥י נְב֛וֹ אַחֵ֖ר חֲמִשִּׁ֥ים וּשְׁנָֽיִם׃ (לט) בְּנֵי֙ עֵילָ֣ם אַחֵ֔ר אֶ֕לֶף מָאתַ֖יִם חֲמִשִּׁ֥ים וְאַרְבָּעָֽה׃ (מ) בְּנֵ֣י חָרִ֔ם שְׁלֹ֥שׁ מֵא֖וֹת וְעֶשְׂרִֽים׃ (מא) בְּנֵ֣י יְרֵח֔וֹ שְׁלֹ֥שׁ מֵא֖וֹת אַרְבָּעִ֥ים וַחֲמִשָּֽׁה׃ (מב) בְּנֵי־ לֹד֙ חָדִ֣יד וְאוֹנ֔וֹ שְׁבַ֥ע מֵא֖וֹת וְעֶשְׂרִ֥ים וְאֶחָֽד׃ (מג) בְּנֵ֣י סְנָאָ֔ה שְׁלֹ֣שֶׁת אֲלָפִ֔ים תְּשַׁ֥ע מֵא֖וֹת וּשְׁלֹשִֽׁים׃ (מד) הַֽכֹּהֲנִ֑ים בְּנֵ֤י יְדַֽעְיָה֙ לְבֵ֣ית יֵשׁ֔וּעַ תְּשַׁ֥ע מֵא֖וֹת שִׁבְעִ֥ים וּשְׁלֹשָֽׁה׃ (מה) בְּנֵ֣י אִמֵּ֔ר אֶ֖לֶף חֲמִשִּׁ֥ים וּשְׁנָֽיִם׃ (מו) בְּנֵ֣י פַשְׁח֔וּר אֶ֕לֶף מָאתַ֖יִם אַרְבָּעִ֥ים וְשִׁבְעָֽה׃ (מז) בְּנֵ֣י חָרִ֔ם אֶ֖לֶף שִׁבְעָ֥ה עָשָֽׂר׃ (מח) הַלְוִיִּ֑ם בְּנֵֽי־ יֵשׁ֧וּעַ לְקַדְמִיאֵ֛ל לִבְנֵ֥י לְהוֹדְוָ֖ה שִׁבְעִ֥ים וְאַרְבָּעָֽה׃ (מט) הַֽמְשֹׁרְרִ֑ים בְּנֵ֣י אָסָ֔ף מֵאָ֖ה אַרְבָּעִ֥ים וּשְׁמֹנָֽה׃ (נ) הַשֹּֽׁעֲרִ֗ים בְּנֵֽי־ שַׁלּ֤וּם בְּנֵֽי־ אָטֵר֙ בְּנֵֽי־ טַלְמֹ֣ן בְּנֵֽי־ עַקּ֔וּב בְּנֵ֥י חֲטִיטָ֖א בְּנֵ֣י שֹׁבָ֑י מֵאָ֖ה שְׁלֹשִׁ֥ים וּשְׁמֹנָֽה׃ (נא) הַנְּתִינִ֑ים בְּנֵי־ צִחָ֥א בְנֵי־ חֲשֻׂפָ֖א בְּנֵ֥י טַבָּעֽוֹת׃ (נב) בְּנֵי־ קֵירֹ֥ס בְּנֵי־ סִיעָ֖א בְּנֵ֥י פָדֽוֹן׃ (נג) בְּנֵי־ לְבָנָ֥ה בְנֵי־ חֲגָבָ֖ה בְּנֵ֥י שַׁלְמָֽי׃ (נד) בְּנֵי־ חָנָ֥ן בְּנֵי־ גִדֵּ֖ל בְּנֵי־ גָֽחַר׃ (נה) בְּנֵי־ רְאָיָ֥ה בְנֵי־ רְצִ֖ין בְּנֵ֥י נְקוֹדָֽא׃ (נו) בְּנֵי־ גַזָּ֥ם בְּנֵי־ עֻזָּ֖א בְּנֵ֥י פָסֵֽחַ׃ (נז) בְּנֵי־ בֵסַ֥י בְּנֵי־ מְעוּנִ֖ים בְּנֵ֥י (נפושסים) נְפִֽישְׁסִֽים׃ (נח) בְּנֵי־ בַקְבּ֥וּק בְּנֵֽי־ חֲקוּפָ֖א בְּנֵ֥י חַרְחֽוּר׃ (נט) בְּנֵי־ בַצְלִ֥ית בְּנֵֽי־ מְחִידָ֖א בְּנֵ֥י חַרְשָֽׁא׃ (ס) בְּנֵי־ בַרְק֥וֹס בְּֽנֵי־ סִֽיסְרָ֖א בְּנֵי־ תָֽמַח׃ (סא) בְּנֵ֥י נְצִ֖יחַ בְּנֵ֥י חֲטִיפָֽא׃ (סב) בְּנֵ֖י עַבְדֵ֣י שְׁלֹמֹ֑ה בְּנֵי־ סוֹטַ֥י בְּנֵי־ סוֹפֶ֖רֶת בְּנֵ֥י פְרִידָֽא׃ (סג) וַתִּמְשֹׁ֤ךְ עֲלֵיהֶם֙ שָׁנִ֣ים רַבּ֔וֹת וַתָּ֨עַד בָּ֧ם בְּרוּחֲךָ֛ בְּיַד־ נְבִיאֶ֖יךָ וְלֹ֣א הֶאֱזִ֑ינוּ וַֽתִּתְּנֵ֔ם בְּיַ֖ד עַמֵּ֥י הָאֲרָצֹֽת׃ (מ) וּֽבְרַחֲמֶ֧יךָ הָרַבִּ֛ים לֹֽא־ עֲשִׂיתָ֥ם כָּלָ֖ה וְלֹ֣א עֲזַבְתָּ֑ם כִּ֛י אֵֽל־ חַנּ֥וּן וְרַח֖וּם אָֽתָּה׃ (מא) וְעַתָּ֣ה אֱ֠לֹהֵינוּ הָאֵ֨ל הַגָּד֜וֹל הַגִּבּ֣וֹר וְהַנּוֹרָא֮ שׁוֹמֵ֣ר הַבְּרִ֣ית וְהַחֶסֶד֒ אַל־ יִמְעַ֣ט לְפָנֶ֡יךָ אֵ֣ת כָּל־ הַתְּלָאָ֣ה אֲֽשֶׁר־ מְ֠צָאַתְנוּ לִמְלָכֵ֨ינוּ לְשָׂרֵ֧ינוּ וּלְכֹהֲנֵ֛ינוּ וְלִנְבִיאֵ֥נוּ וְלַאֲבֹתֵ֖ינוּ וּלְכָל־ עַמֶּ֑ךָ מִימֵי֙ מַלְכֵ֣י אַשּׁ֔וּר עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (מב) וְאַתָּ֣ה צַדִּ֔יק עַ֖ל כָּל־ הַבָּ֣א עָלֵ֑ינוּ כִּֽי־ אֱמֶ֥ת עָשִׂ֖יתָ וַאֲנַ֥חְנוּ הִרְשָֽׁעְנוּ׃ (מג) וְאֶת־ מְלָכֵ֤ינוּ שָׂרֵ֙ינוּ֙ כֹּהֲנֵ֣ינוּ וַאֲבֹתֵ֔ינוּ לֹ֥א עָשׂ֖וּ תּוֹרָתֶ֑ךָ וְלֹ֤א הִקְשִׁ֙יבוּ֙ אֶל־ מִצְוֺתֶ֔יךָ וּלְעֵ֣דְוֺתֶ֔יךָ אֲשֶׁ֥ר הַעִידֹ֖תָ בָּהֶֽם׃ (מד) וְהֵ֣ם בְּמַלְכוּתָם֩ וּבְטוּבְךָ֨ הָרָ֜ב אֲשֶׁר־ נָתַ֣תָּ לָהֶ֗ם וּבְאֶ֨רֶץ הָרְחָבָ֧ה וְהַשְּׁמֵנָ֛ה אֲשֶׁר־ נָתַ֥תָּ לִפְנֵיהֶ֖ם לֹ֣א עֲבָד֑וּךָ וְֽלֹא־ שָׁ֔בוּ מִמַּֽעַלְלֵיהֶ֖ם הָרָעִֽים׃ (מה) הִנֵּ֛ה אֲנַ֥חְנוּ הַיּ֖וֹם עֲבָדִ֑ים וְהָאָ֜רֶץ אֲשֶׁר־ נָתַ֣תָּה לַאֲבֹתֵ֗ינוּ לֶאֱכֹ֤ל אֶת־ פִּרְיָהּ֙ וְאֶת־ טוּבָ֔הּ הִנֵּ֛ה אֲנַ֥חְנוּ עֲבָדִ֖ים עָלֶֽיהָ׃ (מו) וּתְבוּאָתָ֣הּ מַרְבָּ֗ה לַמְּלָכִ֛ים אֲשֶׁר־ נָתַ֥תָּה עָלֵ֖ינוּ בְּחַטֹּאותֵ֑ינוּ וְעַ֣ל גְּ֠וִיֹּתֵינוּ מֹשְׁלִ֤ים וּבִבְהֶמְתֵּ֙נוּ֙ כִּרְצוֹנָ֔ם וּבְצָרָ֥ה גְדוֹלָ֖ה אֲנָֽחְנוּ׃
