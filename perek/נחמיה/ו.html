נחמיה פרק ו לפי החלוקה לסדרים

פרק ו בספר נחמיה (19 פסוקים) נמצא בעזרא ונחמיה סדר ו (פסוקים לח–נא) ובעזרא ונחמיה סדר ז (פסוקים א–ה).

עזרא ונחמיה סדר ו

(א) וַנִּבְנֶה֙ אֶת־ הַ֣חוֹמָ֔ה וַתִּקָּשֵׁ֥ר כָּל־ הַחוֹמָ֖ה עַד־ חֶצְיָ֑הּ וַיְהִ֧י לֵ֦ב לָעָ֖ם לַעֲשֽׂוֹת׃ (ב) וַיְהִ֣י כַאֲשֶׁ֣ר שָׁמַ֣ע סַנְבַלַּ֡ט וְ֠טוֹבִיָּה וְהָעַרְבִ֨ים וְהָעַמֹּנִ֜ים וְהָאַשְׁדּוֹדִ֗ים כִּֽי־ עָלְתָ֤ה אֲרוּכָה֙ לְחֹמ֣וֹת יְרוּשָׁלִַ֔ם כִּי־ הֵחֵ֥לּוּ הַפְּרֻצִ֖ים לְהִסָּתֵ֑ם וַיִּ֥חַר לָהֶ֖ם מְאֹֽד׃ (ג) וַיִּקְשְׁר֤וּ כֻלָּם֙ יַחְדָּ֔ו לָב֖וֹא לְהִלָּחֵ֣ם בִּירוּשָׁלִָ֑ם וְלַעֲשׂ֥וֹת ל֖וֹ תּוֹעָֽה׃ (ד) וַנִּתְפַּלֵּ֖ל אֶל־ אֱלֹהֵ֑ינוּ וַנַּעֲמִ֨יד מִשְׁמָ֧ר עֲלֵיהֶ֛ם יוֹמָ֥ם וָלַ֖יְלָה מִפְּנֵיהֶֽם׃ (ה) וַיֹּ֣אמֶר יְהוּדָ֗ה כָּשַׁל֙ כֹּ֣חַ הַסַּבָּ֔ל וְהֶעָפָ֖ר הַרְבֵּ֑ה וַאֲנַ֙חְנוּ֙ לֹ֣א נוּכַ֔ל לִבְנ֖וֹת בַּחוֹמָֽה׃ (ו) וַיֹּאמְר֣וּ צָרֵ֗ינוּ לֹ֤א יֵדְעוּ֙ וְלֹ֣א יִרְא֔וּ עַ֛ד אֲשֶׁר־ נָב֥וֹא אֶל־ תּוֹכָ֖ם וַהֲרַגְנ֑וּם וְהִשְׁבַּ֖תְנוּ אֶת־ הַמְּלָאכָֽה׃ (ז) וַֽיְהִי֙ כַּאֲשֶׁר־ בָּ֣אוּ הַיְּהוּדִ֔ים הַיֹּשְׁבִ֖ים אֶצְלָ֑ם וַיֹּ֤אמְרוּ לָ֙נוּ֙ עֶ֣שֶׂר פְּעָמִ֔ים מִכָּל־ הַמְּקֹמ֖וֹת אֲשֶׁר־ תָּשׁ֥וּבוּ עָלֵֽינוּ׃ (ח) וָֽאַעֲמִ֞יד מִֽתַּחְתִּיּ֧וֹת לַמָּק֛וֹם מֵאַחֲרֵ֥י לַחוֹמָ֖ה (בצחחיים) בַּצְּחִיחִ֑ים וָֽאַעֲמִ֤יד אֶת־ הָעָם֙ לְמִשְׁפָּח֔וֹת עִם־ חַרְבֹתֵיהֶ֛ם רָמְחֵיהֶ֖ם וְקַשְּׁתֹתֵיהֶֽם׃ (ט) וָאֵ֣רֶא וָאָק֗וּם וָאֹמַ֞ר אֶל־ הַחֹרִ֤ים וְאֶל־ הַסְּגָנִים֙ וְאֶל־ יֶ֣תֶר הָעָ֔ם אַל־ תִּֽירְא֖וּ מִפְּנֵיהֶ֑ם אֶת־ אֲדֹנָ֞י הַגָּד֤וֹל וְהַנּוֹרָא֙ זְכֹ֔רוּ וְהִֽלָּחֲמ֗וּ עַל־ אֲחֵיכֶם֙ בְּנֵיכֶ֣ם וּבְנֹתֵיכֶ֔ם נְשֵׁיכֶ֖ם וּבָתֵּיכֶֽם׃ (י) וַיְהִ֞י כַּֽאֲשֶׁר־ שָׁמְע֤וּ אוֹיְבֵ֙ינוּ֙ כִּי־ נ֣וֹדַֽע לָ֔נוּ וַיָּ֥פֶר הָאֱלֹהִ֖ים אֶת־ עֲצָתָ֑ם (ונשוב) וַנָּ֤שָׁב כֻּלָּ֙נוּ֙ אֶל־ הַ֣חוֹמָ֔ה אִ֖ישׁ אֶל־ מְלַאכְתּֽוֹ׃ (יא) וַיְהִ֣י ׀ מִן־ הַיּ֣וֹם הַה֗וּא חֲצִ֣י נְעָרַי֮ עֹשִׂ֣ים בַּמְּלָאכָה֒ וְחֶצְיָ֗ם מַחֲזִיקִים֙ וְהָרְמָחִ֣ים הַמָּגִנִּ֔ים וְהַקְּשָׁת֖וֹת וְהַשִּׁרְיֹנִ֑ים וְהַ֨שָּׂרִ֔ים אַחֲרֵ֖י כָּל־ בֵּ֥ית יְהוּדָֽה׃ (יב) הַבּוֹנִ֧ים בַּחוֹמָ֛ה וְהַנֹּשְׂאִ֥ים בַּסֶּ֖בֶל עֹמְשִׂ֑ים בְּאַחַ֤ת יָדוֹ֙ עֹשֶׂ֣ה בַמְּלָאכָ֔ה וְאַחַ֖ת מַחֲזֶ֥קֶת הַשָּֽׁלַח׃ (יג) וְהַ֨בּוֹנִ֔ים אִ֥ישׁ חַרְבּ֛וֹ אֲסוּרִ֥ים עַל־ מָתְנָ֖יו וּבוֹנִ֑ים וְהַתּוֹקֵ֥עַ בַּשּׁוֹפָ֖ר אֶצְלִֽי׃ (יד) וָאֹמַ֞ר אֶל־ הַחֹרִ֤ים וְאֶל־ הַסְּגָנִים֙ וְאֶל־ יֶ֣תֶר הָעָ֔ם הַמְּלָאכָ֥ה הַרְבֵּ֖ה וּרְחָבָ֑ה וַאֲנַ֗חְנוּ נִפְרָדִים֙ עַל־ הַ֣חוֹמָ֔ה רְחוֹקִ֖ים אִ֥ישׁ מֵאָחִֽיו׃ (טו) בִּמְק֗וֹם אֲשֶׁ֤ר תִּשְׁמְעוּ֙ אֶת־ ק֣וֹל הַשּׁוֹפָ֔ר שָׁ֖מָּה תִּקָּבְצ֣וּ אֵלֵ֑ינוּ אֱלֹהֵ֖ינוּ יִלָּ֥חֶם לָֽנוּ׃ (טז) וַאֲנַ֖חְנוּ עֹשִׂ֣ים בַּמְּלָאכָ֑ה וְחֶצְיָ֗ם מַחֲזִיקִים֙ בָּֽרְמָחִ֔ים מֵעֲל֣וֹת הַשַּׁ֔חַר עַ֖ד צֵ֥את הַכּוֹכָבִֽים׃ (יז) גַּ֣ם בָּעֵ֤ת הַהִיא֙ אָמַ֣רְתִּי לָעָ֔ם אִ֣ישׁ וְנַעֲר֔וֹ יָלִ֖ינוּ בְּת֣וֹךְ יְרוּשָׁלִָ֑ם וְהָֽיוּ־ לָ֧נוּ הַלַּ֛יְלָה מִשְׁמָ֖ר וְהַיּ֥וֹם מְלָאכָֽה׃ (יח) וְאֵ֨ין אֲנִ֜י וְאַחַ֣י וּנְעָרַ֗י וְאַנְשֵׁ֤י הַמִּשְׁמָר֙ אֲשֶׁ֣ר אַחֲרַ֔י אֵין־ אֲנַ֥חְנוּ פֹשְׁטִ֖ים בְּגָדֵ֑ינוּ אִ֖ישׁ שִׁלְח֥וֹ הַמָּֽיִם׃ (יט) וַתְּהִ֨י צַעֲקַ֥ת הָעָ֛ם וּנְשֵׁיהֶ֖ם גְּדוֹלָ֑ה אֶל־ אֲחֵיהֶ֖ם הַיְּהוּדִֽים׃ (כ) וְיֵשׁ֙ אֲשֶׁ֣ר אֹמְרִ֔ים בָּנֵ֥ינוּ וּבְנֹתֵ֖ינוּ אֲנַ֣חְנוּ רַבִּ֑ים וְנִקְחָ֥ה דָגָ֖ן וְנֹאכְלָ֥ה וְנִחְיֶֽה׃ (כא) וְיֵשׁ֙ אֲשֶׁ֣ר אֹמְרִ֔ים שְׂדֹתֵ֛ינוּ וּכְרָמֵ֥ינוּ וּבָתֵּ֖ינוּ אֲנַ֣חְנוּ עֹרְבִ֑ים וְנִקְחָ֥ה דָגָ֖ן בָּרָעָֽב׃ (כב) וְיֵשׁ֙ אֲשֶׁ֣ר אֹמְרִ֔ים לָוִ֥ינוּ כֶ֖סֶף לְמִדַּ֣ת הַמֶּ֑לֶךְ שְׂדֹתֵ֖ינוּ וּכְרָמֵֽינוּ׃ (כג) וְעַתָּ֗ה כִּבְשַׂ֤ר אַחֵ֙ינוּ֙ בְּשָׂרֵ֔נוּ כִּבְנֵיהֶ֖ם בָּנֵ֑ינוּ וְהִנֵּ֣ה אֲנַ֣חְנוּ כֹ֠בְשִׁים אֶת־ בָּנֵ֨ינוּ וְאֶת־ בְּנֹתֵ֜ינוּ לַעֲבָדִ֗ים וְיֵ֨שׁ מִבְּנֹתֵ֤ינוּ נִכְבָּשׁוֹת֙ וְאֵ֣ין לְאֵ֣ל יָדֵ֔נוּ וּשְׂדֹתֵ֥ינוּ וּכְרָמֵ֖ינוּ לַאֲחֵרִֽים׃ (כד) וַיִּ֥חַר לִ֖י מְאֹ֑ד כַּאֲשֶׁ֤ר שָׁמַ֙עְתִּי֙ אֶת־ זַֽעֲקָתָ֔ם וְאֵ֖ת הַדְּבָרִ֥ים הָאֵֽלֶּה׃ (כה) וַיִּמָּלֵ֨ךְ לִבִּ֜י עָלַ֗י וָאָרִ֙יבָה֙ אֶת־ הַחֹרִ֣ים וְאֶת־ הַסְּגָנִ֔ים וָאֹמְרָ֣ה לָהֶ֔ם מַשָּׁ֥א אִישׁ־ בְּאָחִ֖יו אַתֶּ֣ם (נשאים) נֹשִׁ֑ים וָאֶתֵּ֥ן עֲלֵיהֶ֖ם קְהִלָּ֥ה גְדוֹלָֽה׃ (כו) וָאֹמְרָ֣ה לָהֶ֗ם אֲנַ֣חְנוּ קָ֠נִינוּ אֶת־ אַחֵ֨ינוּ הַיְּהוּדִ֜ים הַנִּמְכָּרִ֤ים לַגּוֹיִם֙ כְּדֵ֣י בָ֔נוּ וְגַם־ אַתֶּ֛ם תִּמְכְּר֥וּ אֶת־ אֲחֵיכֶ֖ם וְנִמְכְּרוּ־ לָ֑נוּ וַֽיַּחֲרִ֔ישׁוּ וְלֹ֥א מָצְא֖וּ דָּבָֽר׃ (כז) (ויאמר) וָאוֹמַ֕ר לֹא־ ט֥וֹב הַדָּבָ֖ר אֲשֶׁר־ אַתֶּ֣ם עֹשִׂ֑ים הֲל֞וֹא בְּיִרְאַ֤ת אֱלֹהֵ֙ינוּ֙ תֵּלֵ֔כוּ מֵחֶרְפַּ֖ת הַגּוֹיִ֥ם אוֹיְבֵֽינוּ׃ (כח) וְגַם־ אֲנִי֙ אַחַ֣י וּנְעָרַ֔י נֹשִׁ֥ים בָּהֶ֖ם כֶּ֣סֶף וְדָגָ֑ן נַֽעַזְבָה־ נָּ֖א אֶת־ הַמַּשָּׁ֥א הַזֶּֽה׃ (כט) הָשִׁיבוּ֩ נָ֨א לָהֶ֜ם כְּהַיּ֗וֹם שְׂדֹתֵיהֶ֛ם כַּרְמֵיהֶ֥ם זֵיתֵיהֶ֖ם וּבָתֵּיהֶ֑ם וּמְאַ֨ת הַכֶּ֤סֶף וְהַדָּגָן֙ הַתִּיר֣וֹשׁ וְהַיִּצְהָ֔ר אֲשֶׁ֥ר אַתֶּ֖ם נֹשִׁ֥ים בָּהֶֽם׃ (ל) וַיֹּאמְר֣וּ נָשִׁ֗יב וּמֵהֶם֙ לֹ֣א נְבַקֵּ֔שׁ כֵּ֣ן נַעֲשֶׂ֔ה כַּאֲשֶׁ֖ר אַתָּ֣ה אוֹמֵ֑ר וָאֶקְרָא֙ אֶת־ הַכֹּ֣הֲנִ֔ים וָֽאַשְׁבִּיעֵ֔ם לַעֲשׂ֖וֹת כַּדָּבָ֥ר הַזֶּֽה׃ (לא) גַּם־ חָצְנִ֣י נָעַ֗רְתִּי וָֽאֹמְרָ֡ה כָּ֣כָה יְנַעֵ֪ר הָֽאֱלֹהִ֟ים אֶת־ כָּל־ הָאִישׁ֩ אֲשֶׁ֨ר לֹֽא־ יָקִ֜ים אֶת־ הַדָּבָ֣ר הַזֶּ֗ה מִבֵּיתוֹ֙ וּמִ֣יגִיע֔וֹ וְכָ֛כָה יִהְיֶ֥ה נָע֖וּר וָרֵ֑ק וַיֹּאמְר֨וּ כָֽל־ הַקָּהָ֜ל אָמֵ֗ן וַֽיְהַלְלוּ֙ אֶת־ יְהוָ֔ה וַיַּ֥עַשׂ הָעָ֖ם כַּדָּבָ֥ר הַזֶּֽה׃ (לב) גַּ֞ם מִיּ֣וֹם ׀ אֲשֶׁר־ צִוָּ֣ה אֹתִ֗י לִהְי֣וֹת פֶּחָם֮ בְּאֶ֣רֶץ יְהוּדָה֒ מִשְּׁנַ֣ת עֶשְׂרִ֗ים וְ֠עַד שְׁנַ֨ת שְׁלֹשִׁ֤ים וּשְׁתַּ֙יִם֙ לְאַרְתַּחְשַׁ֣סְתְּא הַמֶּ֔לֶךְ שָׁנִ֖ים שְׁתֵּ֣ים עֶשְׂרֵ֑ה אֲנִ֣י וְאַחַ֔י לֶ֥חֶם הַפֶּ֖חָה לֹ֥א אָכַֽלְתִּי׃ (לג) וְהַפַּחוֹת֩ הָרִאשֹׁנִ֨ים אֲשֶׁר־ לְפָנַ֜י הִכְבִּ֣ידוּ עַל־ הָעָ֗ם וַיִּקְח֨וּ מֵהֶ֜ם בְּלֶ֤חֶם וָיַ֙יִן֙ אַחַר֙ כֶּֽסֶף־ שְׁקָלִ֣ים אַרְבָּעִ֔ים גַּ֥ם נַעֲרֵיהֶ֖ם שָׁלְט֣וּ עַל־ מְשֻׁלָּֽם׃ (פב) וַיִּפְתַּ֨ח עֶזְרָ֤א הַסֵּ֙פֶר֙ לְעֵינֵ֣י כָל־ הָעָ֔ם כִּֽי־ מֵעַ֥ל כָּל־ הָעָ֖ם הָיָ֑ה וּכְפִתְח֖וֹ עָֽמְד֥וּ כָל־ הָעָֽם׃ (פג) וַיְבָ֣רֶךְ עֶזְרָ֔א אֶת־ יְהוָ֥ה הָאֱלֹהִ֖ים הַגָּד֑וֹל וַיַּֽעֲנ֨וּ כָל־ הָעָ֜ם אָמֵ֤ן ׀ אָמֵן֙ בְּמֹ֣עַל יְדֵיהֶ֔ם וַיִּקְּד֧וּ וַיִּשְׁתַּחֲוֻּ֛ לַיהוָ֖ה אַפַּ֥יִם אָֽרְצָה׃ (פד) וְיֵשׁ֡וּעַ וּבָנִ֡י וְשֵׁרֵ֥בְיָ֣ה ׀ יָמִ֡ין עַקּ֡וּב שַׁבְּתַ֣י ׀ הֽוֹדִיָּ֡ה מַעֲשֵׂיָ֡ה קְלִיטָ֣א עֲזַרְיָה֩ יוֹזָבָ֨ד חָנָ֤ן פְּלָאיָה֙ וְהַלְוִיִּ֔ם מְבִינִ֥ים אֶת־ הָעָ֖ם לַתּוֹרָ֑ה וְהָעָ֖ם עַל־ עָמְדָֽם׃ (פה) וַֽיִּקְרְא֥וּ בַסֵּ֛פֶר בְּתוֹרַ֥ת הָאֱלֹהִ֖ים מְפֹרָ֑שׁ וְשׂ֣וֹם שֶׂ֔כֶל וַיָּבִ֖ינוּ בַּמִּקְרָֽא׃ (פו) וַיֹּ֣אמֶר נְחֶמְיָ֣ה ה֣וּא הַתִּרְשָׁ֡תָא וְעֶזְרָ֣א הַכֹּהֵ֣ן ׀ הַסֹּפֵ֡ר וְהַלְוִיִּם֩ הַמְּבִינִ֨ים אֶת־ הָעָ֜ם לְכָל־ הָעָ֗ם הַיּ֤וֹם קָדֹֽשׁ־ הוּא֙ לַיהוָ֣ה אֱלֹהֵיכֶ֔ם אַל־ תִּֽתְאַבְּל֖וּ וְאַל־ תִּבְכּ֑וּ כִּ֤י בוֹכִים֙ כָּל־ הָעָ֔ם כְּשָׁמְעָ֖ם אֶת־ דִּבְרֵ֥י הַתּוֹרָֽה׃
