יחזקאל פרק מז לפי החלוקה לסדרים

פרק מז בספר יחזקאל (23 פסוקים) נמצא ביחזקאל סדר כח (פסוקים לו–מו) וביחזקאל סדר כט (פסוקים א–יב).

יחזקאל סדר כח

(א) וְשֶׂה־ אַחַ֨ת מִן־ הַצֹּ֤אן מִן־ הַמָּאתַ֙יִם֙ מִמַּשְׁקֵ֣ה יִשְׂרָאֵ֔ל לְמִנְחָ֖ה וּלְעוֹלָ֣ה וְלִשְׁלָמִ֑ים לְכַפֵּ֣ר עֲלֵיהֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ב) כֹּ֚ל הָעָ֣ם הָאָ֔רֶץ יִהְי֖וּ אֶל־ הַתְּרוּמָ֣ה הַזֹּ֑את לַנָּשִׂ֖יא בְּיִשְׂרָאֵֽל׃ (ג) וְעַֽל־ הַנָּשִׂ֣יא יִהְיֶ֗ה הָעוֹל֣וֹת וְהַמִּנְחָה֮ וְהַנֵּסֶךְ֒ בַּחַגִּ֤ים וּבֶחֳדָשִׁים֙ וּבַשַּׁבָּת֔וֹת בְּכָֽל־ מוֹעֲדֵ֖י בֵּ֣ית יִשְׂרָאֵ֑ל הֽוּא־ יַעֲשֶׂ֞ה אֶת־ הַחַטָּ֣את וְאֶת־ הַמִּנְחָ֗ה וְאֶת־ הָֽעוֹלָה֙ וְאֶת־ הַשְּׁלָמִ֔ים לְכַפֵּ֖ר בְּעַ֥ד בֵּֽית־ יִשְׂרָאֵֽל׃ (ד) כֹּה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ בָּֽרִאשׁוֹן֙ בְּאֶחָ֣ד לַחֹ֔דֶשׁ תִּקַּ֥ח פַּר־ בֶּן־ בָּקָ֖ר תָּמִ֑ים וְחִטֵּאתָ֖ אֶת־ הַמִּקְדָּֽשׁ׃ (ה) וְלָקַ֨ח הַכֹּהֵ֜ן מִדַּ֣ם הַחַטָּ֗את וְנָתַן֙ אֶל־ מְזוּזַ֣ת הַבַּ֔יִת וְאֶל־ אַרְבַּ֛ע פִּנּ֥וֹת הָעֲזָרָ֖ה לַמִּזְבֵּ֑חַ וְעַ֨ל־ מְזוּזַ֔ת שַׁ֖עַר הֶחָצֵ֥ר הַפְּנִימִֽית׃ (ו) וְכֵ֤ן תַּֽעֲשֶׂה֙ בְּשִׁבְעָ֣ה בַחֹ֔דֶשׁ מֵאִ֥ישׁ שֹׁגֶ֖ה וּמִפֶּ֑תִי וְכִפַּרְתֶּ֖ם אֶת־ הַבָּֽיִת׃ (ז) בָּ֠רִאשׁוֹן בְּאַרְבָּעָ֨ה עָשָׂ֥ר יוֹם֙ לַחֹ֔דֶשׁ יִהְיֶ֥ה לָכֶ֖ם הַפָּ֑סַח חָ֕ג שְׁבֻע֣וֹת יָמִ֔ים מַצּ֖וֹת יֵאָכֵֽל׃ (ח) וְעָשָׂ֤ה הַנָּשִׂיא֙ בַּיּ֣וֹם הַה֔וּא בַּעֲד֕וֹ וּבְעַ֖ד כָּל־ עַ֣ם הָאָ֑רֶץ פַּ֖ר חַטָּֽאת׃ (ט) וְשִׁבְעַ֨ת יְמֵֽי־ הֶחָ֜ג יַעֲשֶׂ֧ה עוֹלָ֣ה לַֽיהוָ֗ה שִׁבְעַ֣ת פָּ֠רִים וְשִׁבְעַ֨ת אֵילִ֤ים תְּמִימִם֙ לַיּ֔וֹם שִׁבְעַ֖ת הַיָּמִ֑ים וְחַטָּ֕את שְׂעִ֥יר עִזִּ֖ים לַיּֽוֹם׃ (י) וּמִנְחָ֗ה אֵיפָ֥ה לַפָּ֛ר וְאֵיפָ֥ה לָאַ֖יִל יַֽעֲשֶׂ֑ה וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (יא) בַּשְּׁבִיעִ֡י בַּחֲמִשָּׁה֩ עָשָׂ֨ר י֤וֹם לַחֹ֙דֶשׁ֙ בֶּחָ֔ג יַעֲשֶׂ֥ה כָאֵ֖לֶּה שִׁבְעַ֣ת הַיָּמִ֑ים כַּֽחַטָּאת֙ כָּעֹלָ֔ה וְכַמִּנְחָ֖ה וְכַשָּֽׁמֶן׃ (יב) כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ שַׁ֜עַר הֶחָצֵ֤ר הַפְּנִימִית֙ הַפֹּנֶ֣ה קָדִ֔ים יִהְיֶ֣ה סָג֔וּר שֵׁ֖שֶׁת יְמֵ֣י הַֽמַּעֲשֶׂ֑ה וּבְי֤וֹם הַשַּׁבָּת֙ יִפָּתֵ֔חַ וּבְי֥וֹם הַחֹ֖דֶשׁ יִפָּתֵֽחַ׃ (יג) וּבָ֣א הַנָּשִׂ֡יא דֶּרֶךְ֩ אוּלָ֨ם הַשַּׁ֜עַר מִח֗וּץ וְעָמַד֙ עַל־ מְזוּזַ֣ת הַשַּׁ֔עַר וְעָשׂ֣וּ הַכֹּהֲנִ֗ים אֶת־ עֽוֹלָתוֹ֙ וְאֶת־ שְׁלָמָ֔יו וְהִֽשְׁתַּחֲוָ֛ה עַל־ מִפְתַּ֥ן הַשַּׁ֖עַר וְיָצָ֑א וְהַשַּׁ֥עַר לֹֽא־ יִסָּגֵ֖ר עַד־ הָעָֽרֶב׃ (יד) וְהִשְׁתַּחֲו֣וּ עַם־ הָאָ֗רֶץ פֶּ֚תַח הַשַּׁ֣עַר הַה֔וּא בַּשַּׁבָּת֖וֹת וּבֶחֳדָשִׁ֑ים לִפְנֵ֖י יְהוָֽה׃ (טו) וְהָ֣עֹלָ֔ה אֲשֶׁר־ יַקְרִ֥ב הַנָּשִׂ֖יא לַֽיהוָ֑ה בְּי֣וֹם הַשַּׁבָּ֗ת שִׁשָּׁ֧ה כְבָשִׂ֛ים תְּמִימִ֖ם וְאַ֥יִל תָּמִֽים׃ (טז) וּמִנְחָה֙ אֵיפָ֣ה לָאַ֔יִל וְלַכְּבָשִׂ֥ים מִנְחָ֖ה מַתַּ֣ת יָד֑וֹ וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (יז) וּבְי֣וֹם הַחֹ֔דֶשׁ פַּ֥ר בֶּן־ בָּקָ֖ר תְּמִימִ֑ם וְשֵׁ֧שֶׁת כְּבָשִׂ֛ם וָאַ֖יִל תְּמִימִ֥ם יִהְיֽוּ׃ (יח) וְאֵיפָ֨ה לַפָּ֜ר וְאֵיפָ֤ה לָאַ֙יִל֙ יַעֲשֶׂ֣ה מִנְחָ֔ה וְלַ֨כְּבָשִׂ֔ים כַּאֲשֶׁ֥ר תַּשִּׂ֖יג יָד֑וֹ וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (יט) וּבְב֖וֹא הַנָּשִׂ֑יא דֶּ֣רֶךְ אוּלָ֤ם הַשַּׁ֙עַר֙ יָב֔וֹא וּבְדַרְכּ֖וֹ יֵצֵֽא׃ (כ) וּבְב֨וֹא עַם־ הָאָ֜רֶץ לִפְנֵ֣י יְהוָה֮ בַּמּֽוֹעֲדִים֒ הַבָּ֡א דֶּרֶךְ־ שַׁ֨עַר צָפ֜וֹן לְהִֽשְׁתַּחֲוֺ֗ת יֵצֵא֙ דֶּרֶךְ־ שַׁ֣עַר נֶ֔גֶב וְהַבָּא֙ דֶּרֶךְ־ שַׁ֣עַר נֶ֔גֶב יֵצֵ֖א דֶּרֶךְ־ שַׁ֣עַר צָפ֑וֹנָה לֹ֣א יָשׁ֗וּב דֶּ֤רֶךְ הַשַּׁ֙עַר֙ אֲשֶׁר־ בָּ֣א ב֔וֹ כִּ֥י נִכְח֖וֹ (יצאו) יֵצֵֽא׃ (כא) וְֽהַנָּשִׂ֑יא בְּתוֹכָ֤ם בְּבוֹאָם֙ יָב֔וֹא וּבְצֵאתָ֖ם יֵצֵֽאוּ׃ (כב) וּבַחַגִּ֣ים וּבַמּוֹעֲדִ֗ים תִּהְיֶ֤ה הַמִּנְחָה֙ אֵיפָ֤ה לַפָּר֙ וְאֵיפָ֣ה לָאַ֔יִל וְלַכְּבָשִׂ֖ים מַתַּ֣ת יָד֑וֹ וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (כג) וְכִֽי־ יַעֲשֶׂה֩ הַנָּשִׂ֨יא נְדָבָ֜ה עוֹלָ֣ה אֽוֹ־ שְׁלָמִים֮ נְדָבָ֣ה לַֽיהוָה֒ וּפָ֣תַֽח ל֗וֹ אֶת הַשַּׁ֙עַר֙ הַפֹּנֶ֣ה קָדִ֔ים וְעָשָׂ֤ה אֶת־ עֹֽלָתוֹ֙ וְאֶת־ שְׁלָמָ֔יו כַּאֲשֶׁ֥ר יַעֲשֶׂ֖ה בְּי֣וֹם הַשַּׁבָּ֑ת וְיָצָ֛א וְסָגַ֥ר אֶת־ הַשַּׁ֖עַר אַחֲרֵ֥י צֵאתֽוֹ׃ (כד) וְכֶ֨בֶשׂ בֶּן־ שְׁנָת֜וֹ תָּמִ֗ים תַּעֲשֶׂ֥ה עוֹלָ֛ה לַיּ֖וֹם לַֽיהֹוָ֑ה בַּבֹּ֥קֶר בַּבֹּ֖קֶר תַּעֲשֶׂ֥ה אֹתֽוֹ׃ (כה) וּמִנְחָה֩ תַעֲשֶׂ֨ה עָלָ֜יו בַּבֹּ֤קֶר בַּבֹּ֙קֶר֙ שִׁשִּׁ֣ית הָֽאֵיפָ֔ה וְשֶׁ֛מֶן שְׁלִישִׁ֥ית הַהִ֖ין לָרֹ֣ס אֶת־ הַסֹּ֑לֶת מִנְחָה֙ לַֽיהוָ֔ה חֻקּ֥וֹת עוֹלָ֖ם תָּמִֽיד׃ (כו) (ועשו) יַעֲשׂ֨וּ אֶת־ הַכֶּ֧בֶשׂ וְאֶת־ הַמִּנְחָ֛ה וְאֶת־ הַשֶּׁ֖מֶן בַּבֹּ֣קֶר בַּבֹּ֑קֶר עוֹלַ֖ת תָּמִֽיד׃ (כז) כֹּה־ אָמַ֞ר אֲדֹנָ֣י יְהֹוִ֗ה כִּֽי־ יִתֵּ֨ן הַנָּשִׂ֤יא מַתָּנָה֙ לְאִ֣ישׁ מִבָּנָ֔יו נַחֲלָת֥וֹ הִ֖יא לְבָנָ֣יו תִּֽהְיֶ֑ה אֲחֻזָּתָ֥ם הִ֖יא בְּנַחֲלָֽה׃ (כח) וְכִֽי־ יִתֵּ֨ן מַתָּנָ֜ה מִנַּחֲלָת֗וֹ לְאַחַד֙ מֵֽעֲבָדָ֔יו וְהָ֤יְתָה לּוֹ֙ עַד־ שְׁנַ֣ת הַדְּר֔וֹר וְשָׁבַ֖ת לַנָּשִׂ֑יא אַ֚ךְ נַחֲלָת֔וֹ בָּנָ֖יו לָהֶ֥ם תִּהְיֶֽה׃ (כט) וְלֹא־ יִקַּ֨ח הַנָּשִׂ֜יא מִנַּחֲלַ֣ת הָעָ֗ם לְהֽוֹנֹתָם֙ מֵאֲחֻזָּתָ֔ם מֵאֲחֻזָּת֖וֹ יַנְחִ֣ל אֶת־ בָּנָ֑יו לְמַ֙עַן֙ אֲשֶׁ֣ר לֹֽא־ יָפֻ֣צוּ עַמִּ֔י אִ֖ישׁ מֵאֲחֻזָּתֽוֹ׃ (ל) וַיְבִיאֵ֣נִי בַמָּבוֹא֮ אֲשֶׁ֣ר עַל־ כֶּ֣תֶף הַשַּׁעַר֒ אֶל־ הַלִּשְׁכ֤וֹת הַקֹּ֙דֶשׁ֙ אֶל־ הַכֹּ֣הֲנִ֔ים הַפֹּנ֖וֹת צָפ֑וֹנָה וְהִנֵּה־ שָׁ֣ם מָק֔וֹם (בירכתם) בַּיַּרְכָתַ֖יִם יָֽמָּה׃ (לא) וַיֹּ֣אמֶר אֵלַ֔י זֶ֣ה הַמָּק֗וֹם אֲשֶׁ֤ר יְבַשְּׁלוּ־ שָׁם֙ הַכֹּ֣הֲנִ֔ים אֶת־ הָאָשָׁ֖ם וְאֶת־ הַחַטָּ֑את אֲשֶׁ֤ר יֹאפוּ֙ אֶת־ הַמִּנְחָ֔ה לְבִלְתִּ֥י הוֹצִ֛יא אֶל־ הֶחָצֵ֥ר הַחִֽיצוֹנָ֖ה לְקַדֵּ֥שׁ אֶת־ הָעָֽם׃ (לב) וַיּוֹצִיאֵ֗נִי אֶל־ הֶֽחָצֵר֙ הַחִ֣יצֹנָ֔ה וַיַּ֣עֲבִירֵ֔נִי אֶל־ אַרְבַּ֖עַת מִקְצוֹעֵ֣י הֶחָצֵ֑ר וְהִנֵּ֤ה חָצֵר֙ בְּמִקְצֹ֣עַ הֶחָצֵ֔ר חָצֵ֖ר בְּמִקְצֹ֥עַ הֶחָצֵֽר׃ (לג) בְּאַרְבַּ֜עַת מִקְצֹע֤וֹת הֶֽחָצֵר֙ חֲצֵר֣וֹת קְטֻר֔וֹת אַרְבָּעִ֣ים אֹ֔רֶךְ וּשְׁלֹשִׁ֖ים רֹ֑חַב מִדָּ֣ה אַחַ֔ת לְאַרְבַּעְתָּ֖ם מְׄהֻׄקְׄצָׄעֽׄוֹׄתׄ׃ (לד) וְט֨וּר סָבִ֥יב בָּהֶ֛ם סָבִ֖יב לְאַרְבַּעְתָּ֑ם וּמְבַשְּׁל֣וֹת עָשׂ֔וּי מִתַּ֥חַת הַטִּיר֖וֹת סָבִֽיב׃ (לה) וַיֹּ֖אמֶר אֵלָ֑י אֵ֚לֶּה בֵּ֣ית הַֽמְבַשְּׁלִ֔ים אֲשֶׁ֧ר יְבַשְּׁלוּ־ שָׁ֛ם מְשָׁרְתֵ֥י הַבַּ֖יִת אֶת־ זֶ֥בַח הָעָֽם׃ תחילת פרק מז (לו) וַיְשִׁבֵנִי֮ אֶל־ פֶּ֣תַח הַבַּיִת֒ וְהִנֵּה־ הַקַּדְמוֹנָ֔ה וְיָרְד֖וּ עַל־ הָֽעֲרָבָ֑ה וּבָ֣אוּ הַיָּ֔מָּה אֶל־ הַיָּ֥מָּה הַמּֽוּצָאִ֖ים (ונרפאו) וְנִרְפּ֥וּ הַמָּֽיִם׃ (מד) וְהָיָ֣ה כָל־ נֶ֣פֶשׁ חַיָּ֣ה ׀ אֲ‍ֽשֶׁר־ יִשְׁרֹ֡ץ אֶ֣ל כָּל־ בִּצֹּאתָ֧יו וּגְבָאָ֛יו וְלֹ֥א יֵרָפְא֖וּ לְמֶ֥לַח נִתָּֽנוּ׃

(א) וְעַל־ הַנַּ֣חַל יַעֲלֶ֣ה עַל־ שְׂפָת֣וֹ מִזֶּ֣ה ׀ וּמִזֶּ֣ה ׀ כָּל־ עֵֽץ־ מַ֠אֲכָל לֹא־ יִבּ֨וֹל עָלֵ֜הוּ וְלֹֽא־ יִתֹּ֣ם פִּרְי֗וֹ לָֽחֳדָשָׁיו֙ יְבַכֵּ֔ר כִּ֣י מֵימָ֔יו מִן־ הַמִּקְדָּ֖שׁ הֵ֣מָּה יֽוֹצְאִ֑ים (והיו) וְהָיָ֤ה פִרְיוֹ֙ לְמַֽאֲכָ֔ל וְעָלֵ֖הוּ לִתְרוּפָֽה׃ (ב) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה גֵּ֤ה גְבוּל֙ אֲשֶׁ֣ר תִּתְנַחֲל֣וּ אֶת־ הָאָ֔רֶץ לִשְׁנֵ֥י עָשָׂ֖ר שִׁבְטֵ֣י יִשְׂרָאֵ֑ל יוֹסֵ֖ף חֲבָלִֽים׃ (ג) וּנְחַלְתֶּ֤ם אוֹתָהּ֙ אִ֣ישׁ כְּאָחִ֔יו אֲשֶׁ֤ר נָשָׂ֙אתִי֙ אֶת־ יָדִ֔י לְתִתָּ֖הּ לַאֲבֹֽתֵיכֶ֑ם וְנָ֨פְלָ֜ה הָאָ֧רֶץ הַזֹּ֛את לָכֶ֖ם בְּנַחֲלָֽה׃ (ד) וְזֶ֖ה גְּב֣וּל הָאָ֑רֶץ לִפְאַ֨ת צָפ֜וֹנָה מִן־ הַיָּ֧ם הַגָּד֛וֹל הַדֶּ֥רֶךְ חֶתְלֹ֖ן לְב֥וֹא צְדָֽדָה׃ (ה) חֲמָ֤ת ׀ בֵּר֙וֹתָה֙ סִבְרַ֔יִם אֲשֶׁר֙ בֵּין־ גְּב֣וּל דַּמֶּ֔שֶׂק וּבֵ֖ין גְּב֣וּל חֲמָ֑ת חָצֵר֙ הַתִּיכ֔וֹן אֲשֶׁ֖ר אֶל־ גְּב֥וּל חַוְרָֽן׃ (ו) וְהָיָ֨ה גְב֜וּל מִן־ הַיָּ֗ם חֲצַ֤ר עֵינוֹן֙ גְּב֣וּל דַּמֶּ֔שֶׂק וְצָפ֥וֹן ׀ צָפ֖וֹנָה וּגְב֣וּל חֲמָ֑ת וְאֵ֖ת פְּאַ֥ת צָפֽוֹן׃ (ז) וּפְאַ֣ת קָדִ֡ים מִבֵּ֣ין חַוְרָ֣ן וּמִבֵּין־ דַּמֶּשֶׂק֩ וּמִבֵּ֨ין הַגִּלְעָ֜ד וּמִבֵּ֨ין אֶ֤רֶץ יִשְׂרָאֵל֙ הַיַּרְדֵּ֔ן מִגְּב֛וּל עַל־ הַיָּ֥ם הַקַּדְמוֹנִ֖י תָּמֹ֑דּוּ וְאֵ֖ת פְּאַ֥ת קָדִֽימָה׃ (ח) וּפְאַת֙ נֶ֣גֶב תֵּימָ֔נָה מִתָּמָ֗ר עַד־ מֵי֙ מְרִיב֣וֹת קָדֵ֔שׁ נַחֲלָ֖ה אֶל־ הַיָּ֣ם הַגָּד֑וֹל וְאֵ֥ת פְּאַת־ תֵּימָ֖נָה נֶֽגְבָּה׃ (ט) וּפְאַת־ יָם֙ הַיָּ֣ם הַגָּד֔וֹל מִגְּב֕וּל עַד־ נֹ֖כַח לְב֣וֹא חֲמָ֑ת זֹ֖את פְּאַת־ יָֽם׃ (י) וְחִלַּקְתֶּ֞ם אֶת־ הָאָ֧רֶץ הַזֹּ֛את לָכֶ֖ם לְשִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (יא) וְהָיָ֗ה תַּפִּ֣לוּ אוֹתָהּ֮ בְּנַחֲלָה֒ לָכֶ֗ם וּלְהַגֵּרִים֙ הַגָּרִ֣ים בְּתוֹכְכֶ֔ם אֲשֶׁר־ הוֹלִ֥דוּ בָנִ֖ים בְּתֽוֹכְכֶ֑ם וְהָי֣וּ לָכֶ֗ם כְּאֶזְרָח֙ בִּבְנֵ֣י יִשְׂרָאֵ֔ל אִתְּכֶם֙ יִפְּל֣וּ בְנַחֲלָ֔ה בְּת֖וֹךְ שִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (יב) וְהָיָ֣ה בַשֵּׁ֔בֶט אֲשֶׁר־ גָּ֥ר הַגֵּ֖ר דַּמֶּ֤שֶׂק צָפ֙וֹנָה֙ אֶל־ יַ֣ד חֲמָ֔ת וְהָיוּ־ ל֧וֹ פְאַת־ קָדִ֛ים הַיָּ֖ם דָּ֥ן אֶחָֽד׃ (יד) וְעַ֣ל ׀ גְּב֣וּל דָּ֗ן מִפְּאַ֥ת קָדִ֛ים עַד־ פְּאַת־ יָ֖מָּה אָשֵׁ֥ר אֶחָֽד׃ (טו) וְעַ֣ל ׀ גְּב֣וּל אָשֵׁ֗ר מִפְּאַ֥ת קָדִ֛ימָה וְעַד־ פְּאַת־ יָ֖מָּה נַפְתָּלִ֥י אֶחָֽד׃ (טז) וְעַ֣ל ׀ גְּב֣וּל נַפְתָּלִ֗י מִפְּאַ֥ת קָדִ֛מָה עַד־ פְּאַת־ יָ֖מָּה מְנַשֶּׁ֥ה אֶחָֽד׃ (יז) וְעַ֣ל ׀ גְּב֣וּל מְנַשֶּׁ֗ה מִפְּאַ֥ת קָדִ֛מָה עַד־ פְּאַת־ יָ֖מָּה אֶפְרַ֥יִם אֶחָֽד׃ (יח) וְעַ֣ל ׀ גְּב֣וּל אֶפְרַ֗יִם מִפְּאַ֥ת קָדִ֛ים וְעַד־ פְּאַת־ יָ֖מָּה רְאוּבֵ֥ן אֶחָֽד׃ (יט) וְעַ֣ל ׀ גְּב֣וּל רְאוּבֵ֗ן מִפְּאַ֥ת קָדִ֛ים עַד־ פְּאַת־ יָ֖מָּה יְהוּדָ֥ה אֶחָֽד׃ (כ) וְעַל֙ גְּב֣וּל יְהוּדָ֔ה מִפְּאַ֥ת קָדִ֖ים עַד־ פְּאַת־ יָ֑מָּה תִּהְיֶ֣ה הַתְּרוּמָ֣ה אֲ‍ֽשֶׁר־ תָּרִ֡ימוּ חֲמִשָּׁה֩ וְעֶשְׂרִ֨ים אֶ֜לֶף רֹ֗חַב וְאֹ֜רֶךְ כְּאַחַ֤ד הַחֲלָקִים֙ מִפְּאַ֤ת קָדִ֙ימָה֙ עַד־ פְּאַת־ יָ֔מָּה וְהָיָ֥ה הַמִּקְדָּ֖שׁ בְּתוֹכֽוֹ׃ (כא) הַתְּרוּמָ֕ה אֲשֶׁ֥ר תָּרִ֖ימוּ לַֽיהוָ֑ה אֹ֗רֶךְ חֲמִשָּׁ֤ה וְעֶשְׂרִים֙ אֶ֔לֶף וְרֹ֖חַב עֲשֶׂ֥רֶת אֲלָפִֽים׃ (כב) וּ֠לְאֵ֜לֶּה תִּהְיֶ֣ה תְרֽוּמַת־ הַקֹּדֶשׁ֮ לַכֹּֽהֲנִים֒ צָפ֜וֹנָה חֲמִשָּׁ֧ה וְעֶשְׂרִ֣ים אֶ֗לֶף וְיָ֙מָּה֙ רֹ֚חַב עֲשֶׂ֣רֶת אֲלָפִ֔ים וְקָדִ֗ימָה רֹ֚חַב עֲשֶׂ֣רֶת אֲלָפִ֔ים וְנֶ֕גְבָּה אֹ֕רֶךְ חֲמִשָּׁ֥ה וְעֶשְׂרִ֖ים אָ֑לֶף וְהָיָ֥ה מִקְדַּשׁ־ יְהוָ֖ה בְּתוֹכֽוֹ׃ (כג) לַכֹּהֲנִ֤ים הַֽמְקֻדָּשׁ֙ מִבְּנֵ֣י צָד֔וֹק אֲשֶׁ֥ר שָׁמְר֖וּ מִשְׁמַרְתִּ֑י אֲשֶׁ֣ר לֹֽא־ תָע֗וּ בִּתְעוֹת֙ בְּנֵ֣י יִשְׂרָאֵ֔ל כַּאֲשֶׁ֥ר תָּע֖וּ הַלְוִיִּֽם׃ (כד) וְהָיְתָ֨ה לָהֶ֧ם תְּרוּמִיָּ֛ה מִתְּרוּמַ֥ת הָאָ֖רֶץ קֹ֣דֶשׁ קָדָשִׁ֑ים אֶל־ גְּב֖וּל הַלְוִיִּֽם׃ (כה) וְהַלְוִיִּ֗ם לְעֻמַּת֙ גְּב֣וּל הַכֹּהֲנִ֔ים חֲמִשָּׁ֨ה וְעֶשְׂרִ֥ים אֶ֙לֶף֙ אֹ֔רֶךְ וְרֹ֖חַב עֲשֶׂ֣רֶת אֲלָפִ֑ים כָּל־ אֹ֗רֶךְ חֲמִשָּׁ֤ה וְעֶשְׂרִים֙ אֶ֔לֶף וְרֹ֖חַב עֲשֶׂ֥רֶת אֲלָפִֽים׃ (כו) וְלֹא־ יִמְכְּר֣וּ מִמֶּ֗נּוּ וְלֹ֥א יָמֵ֛ר וְלֹ֥א (יעבור) יַעֲבִ֖יר רֵאשִׁ֣ית הָאָ֑רֶץ כִּי־ קֹ֖דֶשׁ לַיהוָֽה׃ (כז) וַחֲמֵ֨שֶׁת אֲלָפִ֜ים הַנּוֹתָ֣ר בָּרֹ֗חַב עַל־ פְּנֵ֨י חֲמִשָּׁ֤ה וְעֶשְׂרִים֙ אֶ֔לֶף חֹֽל־ ה֣וּא לָעִ֔יר לְמוֹשָׁ֖ב וּלְמִגְרָ֑שׁ וְהָיְתָ֥ה הָעִ֖יר (בתוכה) בְּתוֹכֽוֹ׃ (כח) וְאֵלֶּה֮ מִדּוֹתֶיהָ֒ פְּאַ֣ת צָפ֗וֹן חֲמֵ֤שׁ מֵאוֹת֙ וְאַרְבַּ֣עַת אֲלָפִ֔ים וּפְאַת־ נֶ֕גֶב חֲמֵ֥שׁ (חמש) מֵא֖וֹת וְאַרְבַּ֣עַת אֲלָפִ֑ים וּמִפְּאַ֣ת קָדִ֗ים חֲמֵ֤שׁ מֵאוֹת֙ וְאַרְבַּ֣עַת אֲלָפִ֔ים וּפְאַת־ יָ֕מָּה חֲמֵ֥שׁ מֵא֖וֹת וְאַרְבַּ֥עַת אֲלָפִֽים׃ (כט) וְהָיָ֣ה מִגְרָשׁ֮ לָעִיר֒ צָפ֙וֹנָה֙ חֲמִשִּׁ֣ים וּמָאתַ֔יִם וְנֶ֖גְבָּה חֲמִשִּׁ֣ים וּמָאתָ֑יִם וְקָדִ֙ימָה֙ חֲמִשִּׁ֣ים וּמָאתַ֔יִם וְיָ֖מָּה חֲמִשִּׁ֥ים וּמָאתָֽיִם׃ (ל) וְהַנּוֹתָ֨ר בָּאֹ֜רֶךְ לְעֻמַּ֣ת ׀ תְּרוּמַ֣ת הַקֹּ֗דֶשׁ עֲשֶׂ֨רֶת אֲלָפִ֤ים קָדִ֙ימָה֙ וַעֲשֶׂ֤רֶת אֲלָפִים֙ יָ֔מָּה וְהָיָ֕ה לְעֻמַּ֖ת תְּרוּמַ֣ת הַקֹּ֑דֶשׁ וְהָיְתָ֤ה (תבואתה) תְבֽוּאָתוֹ֙ לְלֶ֔חֶם לְעֹבְדֵ֖י הָעִֽיר׃ (לא) וְהָעֹבֵ֖ד הָעִ֑יר יַעַבְד֕וּהוּ מִכֹּ֖ל שִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (לב) כָּל־ הַתְּרוּמָ֗ה חֲמִשָּׁ֤ה וְעֶשְׂרִים֙ אֶ֔לֶף בַּחֲמִשָּׁ֥ה וְעֶשְׂרִ֖ים אָ֑לֶף רְבִיעִ֗ית תָּרִ֙ימוּ֙ אֶת־ תְּרוּמַ֣ת הַקֹּ֔דֶשׁ אֶל־ אֲחֻזַּ֖ת הָעִֽיר׃ (לג) וְהַנּוֹתָ֣ר לַנָּשִׂ֣יא מִזֶּ֣ה ׀ וּמִזֶּ֣ה ׀ לִתְרֽוּמַת־ הַקֹּ֣דֶשׁ וְלַאֲחֻזַּ֪ת הָעִ֟יר אֶל־ פְּנֵ֣י חֲמִשָּׁה֩ וְעֶשְׂרִ֨ים אֶ֥לֶף ׀ תְּרוּמָה֮ עַד־ גְּב֣וּל קָדִימָה֒ וְיָ֗מָּה עַל־ פְּ֠נֵי חֲמִשָּׁ֨ה וְעֶשְׂרִ֥ים אֶ֙לֶף֙ עַל־ גְּב֣וּל יָ֔מָּה לְעֻמַּ֥ת חֲלָקִ֖ים לַנָּשִׂ֑יא וְהָֽיְתָה֙ תְּרוּמַ֣ת הַקֹּ֔דֶשׁ וּמִקְדַּ֥שׁ הַבַּ֖יִת (בתוכה) בְּתוֹכֽוֹ׃ (לד) וּמֵאֲחֻזַּ֤ת הַלְוִיִּם֙ וּמֵאֲחֻזַּ֣ת הָעִ֔יר בְּת֛וֹךְ אֲשֶׁ֥ר לַנָּשִׂ֖יא יִֽהְיֶ֑ה בֵּ֣ין ׀ גְּב֣וּל יְהוּדָ֗ה וּבֵין֙ גְּב֣וּל בִּנְיָמִ֔ן לַנָּשִׂ֖יא יִהְיֶֽה׃ (לה) וְיֶ֖תֶר הַשְּׁבָטִ֑ים מִפְּאַ֥ת קָדִ֛ימָה עַד־ פְּאַת־ יָ֖מָּה בִּנְיָמִ֥ן אֶחָֽד׃ (לו) וְעַ֣ל ׀ גְּב֣וּל בִּנְיָמִ֗ן מִפְּאַ֥ת קָדִ֛ימָה עַד־ פְּאַת־ יָ֖מָּה שִׁמְע֥וֹן אֶחָֽד׃ (לז) וְעַ֣ל ׀ גְּב֣וּל שִׁמְע֗וֹן מִפְּאַ֥ת קָדִ֛ימָה עַד־ פְּאַת־ יָ֖מָּה יִשָׂשכָ֥ר אֶחָֽד׃ (לח) וְעַ֣ל ׀ גְּב֣וּל יִשָׂשכָ֗ר מִפְּאַ֥ת קָדִ֛ימָה עַד־ פְּאַת־ יָ֖מָּה זְבוּלֻ֥ן אֶחָֽד׃ (לט) וְעַ֣ל ׀ גְּב֣וּל זְבוּלֻ֗ן מִפְּאַ֥ת קָדִ֛מָה עַד־ פְּאַת־ יָ֖מָּה גָּ֥ד אֶחָֽד׃ (מ) וְעַל֙ גְּב֣וּל גָּ֔ד אֶל־ פְּאַ֖ת נֶ֣גֶב תֵּימָ֑נָה וְהָיָ֨ה גְב֜וּל מִתָּמָ֗ר מֵ֚י מְרִיבַ֣ת קָדֵ֔שׁ נַחֲלָ֖ה עַל־ הַיָּ֥ם הַגָּדֽוֹל׃ (מא) זֹ֥את הָאָ֛רֶץ אֲשֶׁר־ תַּפִּ֥ילוּ מִֽנַּחֲלָ֖ה לְשִׁבְטֵ֣י יִשְׂרָאֵ֑ל וְאֵ֙לֶּה֙ מַחְלְקוֹתָ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (מב) וְאֵ֖לֶּה תּוֹצְאֹ֣ת הָעִ֑יר מִפְּאַ֣ת צָפ֔וֹן חֲמֵ֥שׁ מֵא֛וֹת וְאַרְבַּ֥עַת אֲלָפִ֖ים מִדָּֽה׃ (מג) וְשַׁעֲרֵ֣י הָעִ֗יר עַל־ שְׁמוֹת֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל שְׁעָרִ֥ים שְׁלוֹשָׁ֖ה צָפ֑וֹנָה שַׁ֣עַר רְאוּבֵ֞ן אֶחָ֗ד שַׁ֤עַר יְהוּדָה֙ אֶחָ֔ד שַׁ֥עַר לֵוִ֖י אֶחָֽד׃ (מד) וְאֶל־ פְּאַ֣ת קָדִ֗ימָה חֲמֵ֤שׁ מֵאוֹת֙ וְאַרְבַּ֣עַת אֲלָפִ֔ים וּשְׁעָרִ֖ים שְׁלֹשָׁ֑ה וְשַׁ֨עַר יוֹסֵ֜ף אֶחָ֗ד שַׁ֤עַר בִּנְיָמִן֙ אֶחָ֔ד שַׁ֥עַר דָּ֖ן אֶחָֽד׃ (מה) וּפְאַת־ נֶ֗גְבָּה חֲמֵ֨שׁ מֵא֜וֹת וְאַרְבַּ֤עַת אֲלָפִים֙ מִדָּ֔ה וּשְׁעָרִ֖ים שְׁלֹשָׁ֑ה שַׁ֣עַר שִׁמְע֞וֹן אֶחָ֗ד שַׁ֤עַר יִשָּׂשכָר֙ אֶחָ֔ד שַׁ֥עַר זְבוּלֻ֖ן אֶחָֽד׃ (מו) פְּאַת־ יָ֗מָּה חֲמֵ֤שׁ מֵאוֹת֙ וְאַרְבַּ֣עַת אֲלָפִ֔ים שַֽׁעֲרֵיהֶ֖ם שְׁלֹשָׁ֑ה שַׁ֣עַר גָּ֞ד אֶחָ֗ד שַׁ֤עַר אָשֵׁר֙ אֶחָ֔ד שַׁ֥עַר נַפְתָּלִ֖י אֶחָֽד׃ (מז) סָבִ֕יב שְׁמֹנָ֥ה עָשָׂ֖ר אָ֑לֶף וְשֵׁם־ הָעִ֥יר מִיּ֖וֹם יְהוָ֥ה ׀ שָֽׁמָּה׃
