יחזקאל פרק מב לפי החלוקה לסדרים

פרק מב בספר יחזקאל (20 פסוקים) נמצא ביחזקאל סדר כה (פסוקים לב–מג) וביחזקאל סדר כו (פסוקים א–ח).

יחזקאל סדר כה

(א) וַיְדַבֵּ֖ר אֵלָ֑י זֹ֣ה הַלִּשְׁכָּ֗ה אֲשֶׁ֤ר פָּנֶ֙יהָ֙ דֶּ֣רֶךְ הַדָּר֔וֹם לַכֹּ֣הֲנִ֔ים שֹׁמְרֵ֖י מִשְׁמֶ֥רֶת הַבָּֽיִת׃ (ב) וְהַלִּשְׁכָּ֗ה אֲשֶׁ֤ר פָּנֶ֙יהָ֙ דֶּ֣רֶךְ הַצָּפ֔וֹן לַכֹּ֣הֲנִ֔ים שֹׁמְרֵ֖י מִשְׁמֶ֣רֶת הַמִּזְבֵּ֑חַ הֵ֣מָּה בְנֵֽי־ צָד֗וֹק הַקְּרֵבִ֧ים מִבְּנֵֽי־ לֵוִ֛י אֶל־ יְהוָ֖ה לְשָׁרְתֽוֹ׃ (ג) וַיָּ֨מָד אֶת־ הֶחָצֵ֜ר אֹ֣רֶךְ ׀ מֵאָ֣ה אַמָּ֗ה וְרֹ֛חַב מֵאָ֥ה אַמָּ֖ה מְרֻבָּ֑עַת וְהַמִּזְבֵּ֖חַ לִפְנֵ֥י הַבָּֽיִת׃ (ד) וַיְבִאֵנִי֮ אֶל־ אֻלָ֣ם הַבַּיִת֒ וַיָּ֙מָד֙ אֵ֣ל אֻלָ֔ם חָמֵ֤שׁ אַמּוֹת֙ מִפֹּ֔ה וְחָמֵ֥שׁ אַמּ֖וֹת מִפֹּ֑ה וְרֹ֣חַב הַשַּׁ֔עַר שָׁלֹ֤שׁ אַמּוֹת֙ מִפּ֔וֹ וְשָׁלֹ֥שׁ אַמּ֖וֹת מִפּֽוֹ׃ (ה) אֹ֣רֶךְ הָאֻלָ֞ם עֶשְׂרִ֣ים אַמָּ֗ה וְרֹ֙חַב֙ עַשְׁתֵּ֣י עֶשְׂרֵ֣ה אַמָּ֔ה וּבַֽמַּעֲל֔וֹת אֲשֶׁ֥ר יַעֲל֖וּ אֵלָ֑יו וְעַמֻּדִים֙ אֶל־ הָ֣אֵילִ֔ים אֶחָ֥ד מִפֹּ֖ה וְאֶחָ֥ד מִפֹּֽה׃ (ו) וַיְבִיאֵ֖נִי אֶל־ הַהֵיכָ֑ל וַיָּ֣מָד אֶת־ הָאֵילִ֗ים שֵׁשׁ־ אַמּ֨וֹת רֹ֧חַב־ מִפּ֛וֹ וְשֵׁשׁ־ אַמּֽוֹת־ רֹ֥חַב מִפּ֖וֹ רֹ֥חַב הָאֹֽהֶל׃ (ז) וְרֹ֣חַב הַפֶּתַח֮ עֶ֣שֶׂר אַמּוֹת֒ וְכִתְפ֣וֹת הַפֶּ֔תַח חָמֵ֤שׁ אַמּוֹת֙ מִפּ֔וֹ וְחָמֵ֥שׁ אַמּ֖וֹת מִפּ֑וֹ וַיָּ֤מָד אָרְכּוֹ֙ אַרְבָּעִ֣ים אַמָּ֔ה וְרֹ֖חַב עֶשְׂרִ֥ים אַמָּֽה׃ (ח) וּבָ֣א לִפְנִ֔ימָה וַיָּ֥מָד אֵֽיל־ הַפֶּ֖תַח שְׁתַּ֣יִם אַמּ֑וֹת וְהַפֶּ֙תַח֙ שֵׁ֣שׁ אַמּ֔וֹת וְרֹ֥חַב הַפֶּ֖תַח שֶׁ֥בַע אַמּֽוֹת׃ (ט) וַיָּ֨מָד אֶת־ אָרְכּ֜וֹ עֶשְׂרִ֣ים אַמָּ֗ה וְרֹ֛חַב עֶשְׂרִ֥ים אַמָּ֖ה אֶל־ פְּנֵ֣י הַֽהֵיכָ֑ל וַיֹּ֣אמֶר אֵלַ֔י זֶ֖ה קֹ֥דֶשׁ הַקֳּדָשִֽׁים׃ (י) וַיָּ֥מָד קִֽיר־ הַבַּ֖יִת שֵׁ֣שׁ אַמּ֑וֹת וְרֹ֣חַב הַצֵּלָע֩ אַרְבַּ֨ע אַמּ֜וֹת סָבִ֧יב ׀ סָבִ֛יב לַבַּ֖יִת סָבִֽיב׃ (יא) וְהַצְּלָעוֹת֩ צֵלָ֨ע אֶל־ צֵלָ֜ע שָׁל֧וֹשׁ וּשְׁלֹשִׁ֣ים פְּעָמִ֗ים וּ֠בָאוֹת בַּקִּ֨יר אֲשֶׁר־ לַבַּ֧יִת לַצְּלָע֛וֹת סָבִ֥יב ׀ סָבִ֖יב לִהְי֣וֹת אֲחוּזִ֑ים וְלֹֽא־ יִהְי֥וּ אֲחוּזִ֖ים בְּקִ֥יר הַבָּֽיִת׃ (יב) וְֽרָחֲבָ֡ה וְֽנָסְבָה֩ לְמַ֨עְלָה לְמַ֜עְלָה לַצְּלָע֗וֹת כִּ֣י מֽוּסַב־ הַ֠בַּיִת לְמַ֨עְלָה לְמַ֜עְלָה סָבִ֤יב ׀ סָבִיב֙ לַבַּ֔יִת עַל־ כֵּ֥ן רֹֽחַב־ לַבַּ֖יִת לְמָ֑עְלָה וְכֵ֧ן הַתַּחְתּוֹנָ֛ה יַעֲלֶ֥ה עַל־ הָעֶלְיוֹנָ֖ה לַתִּיכוֹנָֽה׃ (יג) וְרָאִ֧יתִי לַבַּ֛יִת גֹּ֖בַהּ סָבִ֣יב ׀ סָבִ֑יב (מיסדות) מוּסְד֤וֹת הַצְּלָעוֹת֙ מְל֣וֹ הַקָּנֶ֔ה שֵׁ֥שׁ אַמּ֖וֹת אַצִּֽילָה׃ (יד) רֹ֣חַב הַקִּ֧יר אֲ‍ֽשֶׁר־ לַצֵּלָ֛ע אֶל־ הַח֖וּץ חָמֵ֣שׁ אַמּ֑וֹת וַאֲשֶׁ֣ר מֻנָּ֔ח בֵּ֥ית צְלָע֖וֹת אֲשֶׁ֥ר לַבָּֽיִת׃ (טו) וּבֵ֨ין הַלְּשָׁכ֜וֹת רֹ֣חַב עֶשְׂרִ֥ים אַמָּ֛ה סָבִ֥יב לַבַּ֖יִת סָבִ֥יב ׀ סָבִֽיב׃ (טז) וּפֶ֤תַח הַצֵּלָע֙ לַמֻּנָּ֔ח פֶּ֤תַח אֶחָד֙ דֶּ֣רֶךְ הַצָּפ֔וֹן וּפֶ֥תַח אֶחָ֖ד לַדָּר֑וֹם וְרֹ֙חַב֙ מְק֣וֹם הַמֻּנָּ֔ח חָמֵ֥שׁ אַמּ֖וֹת סָבִ֥יב ׀ סָבִֽיב׃ (יז) וְהַבִּנְיָ֡ן אֲשֶׁר֩ אֶל־ פְּנֵ֨י הַגִּזְרָ֜ה פְּאַ֣ת דֶּֽרֶךְ־ הַיָּ֗ם רֹ֚חַב שִׁבְעִ֣ים אַמָּ֔ה וְקִ֧יר הַבִּנְיָ֛ן חָֽמֵשׁ־ אַמּ֥וֹת רֹ֖חַב סָבִ֣יב ׀ סָבִ֑יב וְאָרְכּ֖וֹ תִּשְׁעִ֥ים אַמָּֽה׃ (יח) וּמָדַ֣ד אֶת־ הַבַּ֔יִת אֹ֖רֶךְ מֵאָ֣ה אַמָּ֑ה וְהַגִּזְרָ֤ה וְהַבִּנְיָה֙ וְקִ֣ירוֹתֶ֔יהָ אֹ֖רֶךְ מֵאָ֥ה אַמָּֽה׃ (יט) וְרֹחַב֩ פְּנֵ֨י הַבַּ֧יִת וְהַגִּזְרָ֛ה לַקָּדִ֖ים מֵאָ֥ה אַמָּֽה׃ (כ) וּמָדַ֣ד אֹֽרֶךְ־ הַ֠בִּנְיָן אֶל־ פְּנֵ֨י הַגִּזְרָ֜ה אֲשֶׁ֨ר עַל־ אַחֲרֶ֧יהָ (ואתוקיהא) וְאַתִּיקֶ֛יהָא מִפּ֥וֹ וּמִפּ֖וֹ מֵאָ֣ה אַמָּ֑ה וְהַֽהֵיכָל֙ הַפְּנִימִ֔י וְאֻֽלַמֵּ֖י הֶחָצֵֽר׃ (כא) הַסִּפִּ֡ים וְהַחַלּוֹנִ֣ים הָ֠אֲטֻמוֹת וְהָאַתִּיקִ֤ים ׀ סָבִיב֙ לִשְׁלָשְׁתָּ֔ם נֶ֧גֶד הַסַּ֛ף שְׂחִ֥יף עֵ֖ץ סָבִ֣יב ׀ סָבִ֑יב וְהָאָ֙רֶץ֙ עַד־ הַֽחַלֹּנ֔וֹת וְהַֽחַלֹּנ֖וֹת מְכֻסּֽוֹת׃ (כב) עַל־ מֵעַ֣ל הַפֶּ֡תַח וְעַד־ הַבַּיִת֩ הַפְּנִימִ֨י וְלַח֜וּץ וְאֶל־ כָּל־ הַקִּ֨יר סָבִ֧יב ׀ סָבִ֛יב בַּפְּנִימִ֥י וּבַחִיצ֖וֹן מִדּֽוֹת׃ (כג) וְעָשׂ֥וּי כְּרוּבִ֖ים וְתִֽמֹרִ֑ים וְתִֽמֹרָה֙ בֵּין־ כְּר֣וּב לִכְר֔וּב וּשְׁנַ֥יִם פָּנִ֖ים לַכְּרֽוּב׃ (כד) וּפְנֵ֨י אָדָ֤ם אֶל־ הַתִּֽמֹרָה֙ מִפּ֔וֹ וּפְנֵֽי־ כְפִ֥יר אֶל־ הַתִּֽמֹרָ֖ה מִפּ֑וֹ עָשׂ֥וּי אֶל־ כָּל־ הַבַּ֖יִת סָבִ֥יב ׀ סָבִֽיב׃ (כה) מֵהָאָ֙רֶץ֙ עַד־ מֵעַ֣ל הַפֶּ֔תַח הַכְּרוּבִ֥ים וְהַתִּֽמֹרִ֖ים עֲשׂוּיִ֑ם וְקִ֖יר הַׄהֵׄיׄכָֽׄלׄ׃ (כו) הַֽהֵיכָ֖ל מְזוּזַ֣ת רְבֻעָ֑ה וּפְנֵ֣י הַקֹּ֔דֶשׁ הַמַּרְאֶ֖ה כַּמַּרְאֶֽה׃ (כז) הַמִּזְבֵּ֡חַ עֵ֣ץ שָׁלוֹשׁ֩ אַמּ֨וֹת גָּבֹ֜הַּ וְאָרְכּ֣וֹ שְׁתַּֽיִם־ אַמּ֗וֹת וּמִקְצֹֽעוֹתָיו֙ ל֔וֹ וְאָרְכּ֥וֹ וְקִֽירֹתָ֖יו עֵ֑ץ וַיְדַבֵּ֣ר אֵלַ֔י זֶ֚ה הַשֻּׁלְחָ֔ן אֲשֶׁ֖ר לִפְנֵ֥י יְהוָֽה׃ (כח) וּשְׁתַּ֧יִם דְּלָת֛וֹת לַֽהֵיכָ֖ל וְלַקֹּֽדֶשׁ׃ (כט) וּשְׁתַּ֥יִם דְּלָת֖וֹת לַדְּלָת֑וֹת שְׁ֚תַּיִם מוּסַבּ֣וֹת דְּלָת֔וֹת שְׁ֚תַּיִם לְדֶ֣לֶת אֶחָ֔ת וּשְׁתֵּ֥י דְלָת֖וֹת לָאַחֶֽרֶת׃ (ל) וַעֲשׂוּיָ֨ה אֲלֵיהֶ֜ן אֶל־ דַּלְת֤וֹת הַֽהֵיכָל֙ כְּרוּבִ֣ים וְתִֽמֹרִ֔ים כַּאֲשֶׁ֥ר עֲשׂוּיִ֖ם לַקִּיר֑וֹת וְעָ֥ב עֵ֛ץ אֶל־ פְּנֵ֥י הָאוּלָ֖ם מֵהַחֽוּץ׃ (לא) וְחַלּוֹנִ֨ים אֲטֻמ֤וֹת וְתִֽמֹרִים֙ מִפּ֣וֹ וּמִפּ֔וֹ אֶל־ כִּתְפ֖וֹת הָֽאוּלָ֑ם וְצַלְע֥וֹת הַבַּ֖יִת וְהָעֻבִּֽים׃ תחילת פרק מב (לב) וַיּוֹצִאֵ֗נִי אֶל־ הֶֽחָצֵר֙ הַחִ֣יצוֹנָ֔ה הַדֶּ֖רֶךְ דֶּ֣רֶךְ הַצָּפ֑וֹן וַיְבִאֵ֣נִי אֶל־ הַלִּשְׁכָּ֗ה אֲשֶׁ֨ר נֶ֧גֶד הַגִּזְרָ֛ה וַאֲשֶֽׁר־ נֶ֥גֶד הַבִּנְיָ֖ן אֶל־ הַצָּפֽוֹן׃ (לג) אֶל־ פְּנֵי־ אֹ֙רֶךְ֙ אַמּ֣וֹת הַמֵּאָ֔ה פֶּ֖תַח הַצָּפ֑וֹן וְהָרֹ֖חַב חֲמִשִּׁ֥ים אַמּֽוֹת׃ (לד) נֶ֣גֶד הָֽעֶשְׂרִ֗ים אֲשֶׁר֙ לֶחָצֵ֣ר הַפְּנִימִ֔י וְנֶ֣גֶד רִֽצְפָ֔ה אֲשֶׁ֖ר לֶחָצֵ֣ר הַחִֽיצוֹנָ֑ה אַתִּ֥יק אֶל־ פְּנֵֽי־ אַתִּ֖יק בַּשְּׁלִשִֽׁים׃ (לה) וְלִפְנֵ֨י הַלְּשָׁכ֜וֹת מַהֲלַךְ֩ עֶ֨שֶׂר אַמּ֥וֹת רֹ֙חַב֙ אֶל־ הַפְּנִימִ֔ית דֶּ֖רֶךְ אַמָּ֣ה אֶחָ֑ת וּפִתְחֵיהֶ֖ם לַצָּפֽוֹן׃ (לו) וְהַלְּשָׁכ֥וֹת הָעֶלְיוֹנֹ֖ת קְצֻר֑וֹת כִּֽי־ יוֹכְל֨וּ אַתִּיקִ֜ים מֵהֵ֗נָה מֵֽהַתַּחְתֹּנ֛וֹת וּמֵהַתִּֽכֹנ֖וֹת בִּנְיָֽן׃ (לז) כִּ֤י מְשֻׁלָּשׁוֹת֙ הֵ֔נָּה וְאֵ֤ין לָהֶן֙ עַמּוּדִ֔ים כְּעַמּוּדֵ֖י הַחֲצֵר֑וֹת עַל־ כֵּ֣ן נֶאֱצַ֗ל מֵהַתַּחְתּוֹנ֛וֹת וּמֵהַתִּֽיכֹנ֖וֹת מֵהָאָֽרֶץ׃ (לח) וְגָדֵ֤ר אֲשֶׁר־ לַחוּץ֙ לְעֻמַּ֣ת הַלְּשָׁכ֔וֹת דֶּ֛רֶךְ הֶחָצֵ֥ר הַחִֽצוֹנָ֖ה אֶל־ פְּנֵ֣י הַלְּשָׁכ֑וֹת אָרְכּ֖וֹ חֲמִשִּׁ֥ים אַמָּֽה׃ (לט) כִּֽי־ אֹ֣רֶךְ הַלְּשָׁכ֗וֹת אֲשֶׁ֛ר לֶחָצֵ֥ר הַחִֽצוֹנָ֖ה חֲמִשִּׁ֣ים אַמָּ֑ה וְהִנֵּ֛ה עַל־ פְּנֵ֥י הַהֵיכָ֖ל מֵאָ֥ה אַמָּֽה׃ (מ) מִדָּמ֗וֹ וְנָ֨תַתָּ֜ה עַל־ אַרְבַּ֤ע קַרְנֹתָיו֙ וְאֶל־ אַרְבַּע֙ פִּנּ֣וֹת הָעֲזָרָ֔ה וְאֶֽל־ הַגְּב֖וּל סָבִ֑יב וְחִטֵּאתָ֥ אוֹת֖וֹ וְכִפַּרְתָּֽהוּ׃ (כט) וְלָ֣קַחְתָּ֔ אֵ֖ת הַפָּ֣ר הַֽחַטָּ֑את וּשְׂרָפוֹ֙ בְּמִפְקַ֣ד הַבַּ֔יִת מִח֖וּץ לַמִּקְדָּֽשׁ׃ (ל) וּבַיּוֹם֙ הַשֵּׁנִ֔י תַּקְרִ֛יב שְׂעִיר־ עִזִּ֥ים תָּמִ֖ים לְחַטָּ֑את וְחִטְּאוּ֙ אֶת־ הַמִּזְבֵּ֔חַ כַּאֲשֶׁ֥ר חִטְּא֖וּ בַּפָּֽר׃ (לא) בְּכַלּוֹתְךָ֖ מֵֽחַטֵּ֑א תַּקְרִיב֙ פַּ֣ר בֶּן־ בָּקָ֣ר תָּמִ֔ים וְאַ֥יִל מִן־ הַצֹּ֖אן תָּמִֽים׃ (לב) וְהִקְרַבְתָּ֖ם לִפְנֵ֣י יְהוָ֑ה וְהִשְׁלִ֨יכוּ הַכֹּהֲנִ֤ים עֲלֵיהֶם֙ מֶ֔לַח וְהֶעֱל֥וּ אוֹתָ֛ם עֹלָ֖ה לַֽיהוָֽה׃ (לג) שִׁבְעַ֣ת יָמִ֔ים תַּעֲשֶׂ֥ה שְׂעִיר־ חַטָּ֖את לַיּ֑וֹם וּפַ֧ר בֶּן־ בָּקָ֛ר וְאַ֥יִל מִן־ הַצֹּ֖אן תְּמִימִ֥ים יַעֲשֽׂוּ׃ (לד) שִׁבְעַ֣ת יָמִ֗ים יְכַפְּרוּ֙ אֶת־ הַמִּזְבֵּ֔חַ וְטִֽהֲר֖וּ אֹת֑וֹ וּמִלְא֖וּ (ידו) יָדָֽיו׃
