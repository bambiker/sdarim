יחזקאל פרק ה לפי החלוקה לסדרים

פרק ה בספר יחזקאל (17 פסוקים) נמצא ביחזקאל סדר ב (פסוקים לד–נ).

יחזקאל סדר ב

(א) וַתִּשָּׂאֵ֣נִי ר֔וּחַ וָאֶשְׁמַ֣ע אַחֲרַ֔י ק֖וֹל רַ֣עַשׁ גָּד֑וֹל בָּר֥וּךְ כְּבוֹד־ יְהוָ֖ה מִמְּקוֹמֽוֹ׃ (ב) וְק֣וֹל ׀ כַּנְפֵ֣י הַחַיּ֗וֹת מַשִּׁיקוֹת֙ אִשָּׁ֣ה אֶל־ אֲחוֹתָ֔הּ וְק֥וֹל הָאוֹפַנִּ֖ים לְעֻמָּתָ֑ם וְק֖וֹל רַ֥עַשׁ גָּדֽוֹל׃ (ג) וְר֥וּחַ נְשָׂאַ֖תְנִי וַתִּקָּחֵ֑נִי וָאֵלֵ֥ךְ מַר֙ בַּחֲמַ֣ת רוּחִ֔י וְיַד־ יְהוָ֥ה עָלַ֖י חָזָֽקָה׃ (ד) וָאָב֨וֹא אֶל־ הַגּוֹלָ֜ה תֵּ֣ל אָ֠בִיב הַיֹּשְׁבִ֤ים אֶֽל־ נְהַר־ כְּבָר֙ (ואשר) וָֽאֵשֵׁ֔ב הֵ֖מָּה יוֹשְׁבִ֣ים שָׁ֑ם וָאֵשֵׁ֥ב שָׁ֛ם שִׁבְעַ֥ת יָמִ֖ים מַשְׁמִ֥ים בְּתוֹכָֽם׃ (ה) וַיְהִ֕י מִקְצֵ֖ה שִׁבְעַ֣ת יָמִ֑ים וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ו) בֶּן־ אָדָ֕ם צֹפֶ֥ה נְתַתִּ֖יךָ לְבֵ֣ית יִשְׂרָאֵ֑ל וְשָׁמַעְתָּ֤ מִפִּי֙ דָּבָ֔ר וְהִזְהַרְתָּ֥ אוֹתָ֖ם מִמֶּֽנִּי׃ (ז) בְּאָמְרִ֤י לָֽרָשָׁע֙ מ֣וֹת תָּמ֔וּת וְלֹ֣א הִזְהַרְתּ֗וֹ וְלֹ֥א דִבַּ֛רְתָּ לְהַזְהִ֥יר רָשָׁ֛ע מִדַּרְכּ֥וֹ הָרְשָׁעָ֖ה לְחַיֹּת֑וֹ ה֤וּא רָשָׁע֙ בַּעֲוֺנ֣וֹ יָמ֔וּת וְדָמ֖וֹ מִיָּדְךָ֥ אֲבַקֵּֽשׁ׃ (ח) וְאַתָּה֙ כִּֽי־ הִזְהַ֣רְתָּ רָשָׁ֔ע וְלֹא־ שָׁב֙ מֵֽרִשְׁע֔וֹ וּמִדַּרְכּ֖וֹ הָרְשָׁעָ֑ה ה֚וּא בַּעֲוֺנ֣וֹ יָמ֔וּת וְאַתָּ֖ה אֶֽת־ נַפְשְׁךָ֥ הִצַּֽלְתָּ׃ (ט) וּבְשׁ֨וּב צַדִּ֤יק מִצִּדְקוֹ֙ וְעָ֣שָׂה עָ֔וֶל וְנָתַתִּ֥י מִכְשׁ֛וֹל לְפָנָ֖יו ה֣וּא יָמ֑וּת כִּ֣י לֹ֤א הִזְהַרְתּוֹ֙ בְּחַטָּאת֣וֹ יָמ֔וּת וְלֹ֣א תִזָּכַ֗רְןָ צִדְקֹתָו֙ אֲשֶׁ֣ר עָשָׂ֔ה וְדָמ֖וֹ מִיָּדְךָ֥ אֲבַקֵּֽשׁ׃ (י) וְאַתָּ֞ה כִּ֧י הִזְהַרְתּ֣וֹ צַדִּ֗יק לְבִלְתִּ֥י חֲטֹ֛א צַדִּ֖יק וְה֣וּא לֹא־ חָטָ֑א חָי֤וֹ יִֽחְיֶה֙ כִּ֣י נִזְהָ֔ר וְאַתָּ֖ה אֶֽת־ נַפְשְׁךָ֥ הִצַּֽלְתָּ׃ (יא) וַתְּהִ֥י עָלַ֛י שָׁ֖ם יַד־ יְהוָ֑ה וַיֹּ֣אמֶר אֵלַ֗י ק֥וּם צֵא֙ אֶל־ הַבִּקְעָ֔ה וְשָׁ֖ם אֲדַבֵּ֥ר אוֹתָֽךְ׃ (יב) וָאָקוּם֮ וָאֵצֵ֣א אֶל־ הַבִּקְעָה֒ וְהִנֵּה־ שָׁ֤ם כְּבוֹד־ יְהוָה֙ עֹמֵ֔ד כַּכָּב֕וֹד אֲשֶׁ֥ר רָאִ֖יתִי עַל־ נְהַר־ כְּבָ֑ר וָאֶפֹּ֖ל עַל־ פָּנָֽי׃ (יג) וַתָּבֹא־ בִ֣י ר֔וּחַ וַתַּעֲמִדֵ֖נִי עַל־ רַגְלָ֑י וַיְדַבֵּ֤ר אֹתִי֙ וַיֹּ֣אמֶר אֵלַ֔י בֹּ֥א הִסָּגֵ֖ר בְּת֥וֹךְ בֵּיתֶֽךָ׃ (יד) וְאַתָּ֣ה בֶן־ אָדָ֗ם הִנֵּ֨ה נָתְנ֤וּ עָלֶ֙יךָ֙ עֲבוֹתִ֔ים וַאֲסָר֖וּךָ בָּהֶ֑ם וְלֹ֥א תֵצֵ֖א בְּתוֹכָֽם׃ (טו) וּלְשֽׁוֹנְךָ֙ אַדְבִּ֣יק אֶל־ חִכֶּ֔ךָ וְנֶֽאֱלַ֔מְתָּ וְלֹא־ תִֽהְיֶ֥ה לָהֶ֖ם לְאִ֣ישׁ מוֹכִ֑יחַ כִּ֛י בֵּ֥ית מְרִ֖י הֵֽמָּה׃ (טז) וּֽבְדַבְּרִ֤י אֽוֹתְךָ֙ אֶפְתַּ֣ח אֶת־ פִּ֔יךָ וְאָמַרְתָּ֣ אֲלֵיהֶ֔ם כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהֹוִ֑ה הַשֹּׁמֵ֤עַ ׀ יִשְׁמָע֙ וְהֶחָדֵ֣ל ׀ יֶחְדָּ֔ל כִּ֛י בֵּ֥ית מְרִ֖י הֵֽמָּה׃ (יז) וְאַתָּ֤ה בֶן־ אָדָם֙ קַח־ לְךָ֣ לְבֵנָ֔ה וְנָתַתָּ֥ה אוֹתָ֖הּ לְפָנֶ֑יךָ וְחַקּוֹתָ֥ עָלֶ֛יהָ עִ֖יר אֶת־ יְרוּשָׁלִָֽם׃ (יח) וְנָתַתָּ֨ה עָלֶ֜יהָ מָצ֗וֹר וּבָנִ֤יתָ עָלֶ֙יהָ֙ דָּיֵ֔ק וְשָׁפַכְתָּ֥ עָלֶ֖יהָ סֹֽלְלָ֑ה וְנָתַתָּ֨ה עָלֶ֧יהָ מַחֲנ֛וֹת וְשִׂים־ עָלֶ֥יהָ כָּרִ֖ים סָבִֽיב׃ (יט) וְאַתָּ֤ה קַח־ לְךָ֙ מַחֲבַ֣ת בַּרְזֶ֔ל וְנָתַתָּ֤ה אוֹתָהּ֙ קִ֣יר בַּרְזֶ֔ל בֵּינְךָ֖ וּבֵ֣ין הָעִ֑יר וַהֲכִינֹתָה֩ אֶת־ פָּנֶ֨יךָ אֵלֶ֜יהָ וְהָיְתָ֤ה בַמָּצוֹר֙ וְצַרְתָּ֣ עָלֶ֔יהָ א֥וֹת הִ֖יא לְבֵ֥ית יִשְׂרָאֵֽל׃ (כ) וְאַתָּ֤ה שְׁכַב֙ עַל־ צִדְּךָ֣ הַשְּׂמָאלִ֔י וְשַׂמְתָּ֛ אֶת־ עֲוֺ֥ן בֵּֽית־ יִשְׂרָאֵ֖ל עָלָ֑יו מִסְפַּ֤ר הַיָּמִים֙ אֲשֶׁ֣ר תִּשְׁכַּ֣ב עָלָ֔יו תִּשָּׂ֖א אֶת־ עֲוֺנָֽם׃ (כא) וַאֲנִ֗י נָתַ֤תִּֽי לְךָ֙ אֶת־ שְׁנֵ֣י עֲוֺנָ֔ם לְמִסְפַּ֣ר יָמִ֔ים שְׁלֹשׁ־ מֵא֥וֹת וְתִשְׁעִ֖ים י֑וֹם וְנָשָׂ֖אתָ עֲוֺ֥ן בֵּֽית־ יִשְׂרָאֵֽל׃ (כב) וְכִלִּיתָ֣ אֶת־ אֵ֗לֶּה וְשָׁ֨כַבְתָּ֜ עַל־ צִדְּךָ֤ (הימיני) הַיְמָנִי֙ שֵׁנִ֔ית וְנָשָׂ֖אתָ אֶת־ עֲוֺ֣ן בֵּית־ יְהוּדָ֑ה אַרְבָּעִ֣ים י֔וֹם י֧וֹם לַשָּׁנָ֛ה י֥וֹם לַשָּׁנָ֖ה נְתַתִּ֥יו לָֽךְ׃ (כג) וְאֶל־ מְצ֤וֹר יְרוּשָׁלִַ֙ם֙ תָּכִ֣ין פָּנֶ֔יךָ וּֽזְרֹעֲךָ֖ חֲשׂוּפָ֑ה וְנִבֵּאתָ֖ עָלֶֽיהָ׃ (כד) וְהִנֵּ֛ה נָתַ֥תִּי עָלֶ֖יךָ עֲבוֹתִ֑ים וְלֹֽא־ תֵהָפֵ֤ךְ מִֽצִּדְּךָ֙ אֶל־ צִדֶּ֔ךָ עַד־ כַּלּוֹתְךָ֖ יְמֵ֥י מְצוּרֶֽךָ׃ (כה) וְאַתָּ֣ה קַח־ לְךָ֡ חִטִּ֡ין וּ֠שְׂעֹרִים וּפ֨וֹל וַעֲדָשִׁ֜ים וְדֹ֣חַן וְכֻסְּמִ֗ים וְנָתַתָּ֤ה אוֹתָם֙ בִּכְלִ֣י אֶחָ֔ד וְעָשִׂ֧יתָ אוֹתָ֛ם לְךָ֖ לְלָ֑חֶם מִסְפַּ֨ר הַיָּמִ֜ים אֲשֶׁר־ אַתָּ֣ה ׀ שׁוֹכֵ֣ב עַֽל־ צִדְּךָ֗ שְׁלֹשׁ־ מֵא֧וֹת וְתִשְׁעִ֛ים י֖וֹם תֹּאכֲלֶֽנּוּ׃ (כו) וּמַאֲכָֽלְךָ֙ אֲשֶׁ֣ר תֹּאכֲלֶ֔נּוּ בְּמִשְׁק֕וֹל עֶשְׂרִ֥ים שֶׁ֖קֶל לַיּ֑וֹם מֵעֵ֥ת עַד־ עֵ֖ת תֹּאכֲלֶֽנּוּ׃ (כז) וּמַ֛יִם בִּמְשׂוּרָ֥ה תִשְׁתֶּ֖ה שִׁשִּׁ֣ית הַהִ֑ין מֵעֵ֥ת עַד־ עֵ֖ת תִּשְׁתֶּֽה׃ (כח) וְעֻגַ֥ת שְׂעֹרִ֖ים תֹּֽאכֲלֶ֑נָּה וְהִ֗יא בְּגֶֽלְלֵי֙ צֵאַ֣ת הָֽאָדָ֔ם תְּעֻגֶ֖נָה לְעֵינֵיהֶֽם׃ (כט) וַיֹּ֣אמֶר יְהוָ֔ה כָּ֣כָה יֹאכְל֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶת־ לַחְמָ֖ם טָמֵ֑א בַּגּוֹיִ֕ם אֲשֶׁ֥ר אַדִּיחֵ֖ם שָֽׁם׃ (ל) וָאֹמַ֗ר אֲהָהּ֙ אֲדֹנָ֣י יְהוִ֔ה הִנֵּ֥ה נַפְשִׁ֖י לֹ֣א מְטֻמָּאָ֑ה וּנְבֵלָ֨ה וּטְרֵפָ֤ה לֹֽא־ אָכַ֙לְתִּי֙ מִנְּעוּרַ֣י וְעַד־ עַ֔תָּה וְלֹא־ בָ֥א בְּפִ֖י בְּשַׂ֥ר פִּגּֽוּל׃ (לא) וַיֹּ֣אמֶר אֵלַ֔י רְאֵ֗ה נָתַ֤תִּֽי לְךָ֙ אֶת־ (צפועי) צְפִיעֵ֣י הַבָּקָ֔ר תַּ֖חַת גֶּלְלֵ֣י הָֽאָדָ֑ם וְעָשִׂ֥יתָ אֶֽת־ לַחְמְךָ֖ עֲלֵיהֶֽם׃ (לב) וַיֹּ֣אמֶר אֵלַ֗י בֶּן־ אָדָם֙ הִנְנִ֨י שֹׁבֵ֤ר מַטֵּה־ לֶ֙חֶם֙ בִּיר֣וּשָׁלִַ֔ם וְאָכְלוּ־ לֶ֥חֶם בְּמִשְׁקָ֖ל וּבִדְאָגָ֑ה וּמַ֕יִם בִּמְשׂוּרָ֥ה וּבְשִׁמָּמ֖וֹן יִשְׁתּֽוּ׃ (לג) לְמַ֥עַן יַחְסְר֖וּ לֶ֣חֶם וָמָ֑יִם וְנָשַׁ֙מּוּ֙ אִ֣ישׁ וְאָחִ֔יו וְנָמַ֖קּוּ בַּעֲוֺנָֽם׃ תחילת פרק ה (לד) וְאַתָּ֨ה בֶן־ אָדָ֜ם קַח־ לְךָ֣ ׀ חֶ֣רֶב חַדָּ֗ה תַּ֤עַר הַגַּלָּבִים֙ תִּקָּחֶ֣נָּה לָּ֔ךְ וְהַעֲבַרְתָּ֥ עַל־ רֹאשְׁךָ֖ וְעַל־ זְקָנֶ֑ךָ וְלָקַחְתָּ֥ לְךָ֛ מֹאזְנֵ֥י מִשְׁקָ֖ל וְחִלַּקְתָּֽם׃ (לה) שְׁלִשִׁ֗ית בָּא֤וּר תַּבְעִיר֙ בְּת֣וֹךְ הָעִ֔יר כִּמְלֹ֖את יְמֵ֣י הַמָּצ֑וֹר וְלָֽקַחְתָּ֣ אֶת־ הַשְּׁלִשִׁ֗ית תַּכֶּ֤ה בַחֶ֙רֶב֙ סְבִ֣יבוֹתֶ֔יהָ וְהַשְּׁלִשִׁית֙ תִּזְרֶ֣ה לָר֔וּחַ וְחֶ֖רֶב אָרִ֥יק אַחֲרֵיהֶֽם׃ (לו) וְלָקַחְתָּ֥ מִשָּׁ֖ם מְעַ֣ט בְּמִסְפָּ֑ר וְצַרְתָּ֥ אוֹתָ֖ם בִּכְנָפֶֽיךָ׃ (לז) וּמֵהֶם֙ ע֣וֹד תִּקָּ֔ח וְהִשְׁלַכְתָּ֤ אוֹתָם֙ אֶל־ תּ֣וֹךְ הָאֵ֔שׁ וְשָׂרַפְתָּ֥ אֹתָ֖ם בָּאֵ֑שׁ מִמֶּ֥נּוּ תֵצֵא־ אֵ֖שׁ אֶל־ כָּל־ בֵּ֥ית יִשְׂרָאֵֽל׃ (לח) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהֹוִ֔ה זֹ֚את יְר֣וּשָׁלִַ֔ם בְּת֥וֹךְ הַגּוֹיִ֖ם שַׂמְתִּ֑יהָ וּסְבִיבוֹתֶ֖יהָ אֲרָצֽוֹת׃ (לט) וַתֶּ֨מֶר אֶת־ מִשְׁפָּטַ֤י לְרִשְׁעָה֙ מִן־ הַגּוֹיִ֔ם וְאֶ֨ת־ חֻקּוֹתַ֔י מִן־ הָאֲרָצ֖וֹת אֲשֶׁ֣ר סְבִיבוֹתֶ֑יהָ כִּ֤י בְמִשְׁפָּטַי֙ מָאָ֔סוּ וְחֻקּוֹתַ֖י לֹא־ הָלְכ֥וּ בָהֶֽם׃ (מ) לָכֵ֞ן כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה יַ֤עַן הֲמָנְכֶם֙ מִן־ הַגּוֹיִם֙ אֲשֶׁ֣ר סְבִיבֽוֹתֵיכֶ֔ם בְּחֻקּוֹתַי֙ לֹ֣א הֲלַכְתֶּ֔ם וְאֶת־ מִשְׁפָּטַ֖י לֹ֣א עֲשִׂיתֶ֑ם וּֽכְמִשְׁפְּטֵ֧י הַגּוֹיִ֛ם אֲשֶׁ֥ר סְבִיבוֹתֵיכֶ֖ם לֹ֥א עֲשִׂיתֶֽם׃ (מא) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י עָלַ֖יִךְ גַּם־ אָ֑נִי וְעָשִׂ֧יתִי בְתוֹכֵ֛ךְ מִשְׁפָּטִ֖ים לְעֵינֵ֥י הַגּוֹיִֽם׃ (מב) וְעָשִׂ֣יתִי בָ֗ךְ אֵ֚ת אֲשֶׁ֣ר לֹֽא־ עָשִׂ֔יתִי וְאֵ֛ת אֲשֶֽׁר־ לֹֽא־ אֶעֱשֶׂ֥ה כָמֹ֖הוּ ע֑וֹד יַ֖עַן כָּל־ תּוֹעֲבֹתָֽיִךְ׃ (מג) לָכֵ֗ן אָב֞וֹת יֹאכְל֤וּ בָנִים֙ בְּתוֹכֵ֔ךְ וּבָנִ֖ים יֹאכְל֣וּ אֲבוֹתָ֑ם וְעָשִׂ֤יתִי בָךְ֙ שְׁפָטִ֔ים וְזֵרִיתִ֥י אֶת־ כָּל־ שְׁאֵרִיתֵ֖ךְ לְכָל־ רֽוּחַ׃ (מד) לָכֵ֣ן חַי־ אָ֗נִי נְאֻם֮ אֲדֹנָ֣י יְהוִה֒ אִם־ לֹ֗א יַ֚עַן אֶת־ מִקְדָּשִׁ֣י טִמֵּ֔את בְּכָל־ שִׁקּוּצַ֖יִךְ וּבְכָל־ תּוֹעֲבֹתָ֑יִךְ וְגַם־ אֲנִ֤י אֶגְרַע֙ וְלֹא־ תָח֣וֹס עֵינִ֔י וְגַם־ אֲנִ֖י לֹ֥א אֶחְמֽוֹל׃ (מה) שְׁלִשִׁתֵ֞יךְ בַּדֶּ֣בֶר יָמ֗וּתוּ וּבָֽרָעָב֙ יִכְל֣וּ בְתוֹכֵ֔ךְ וְהַ֨שְּׁלִשִׁ֔ית בַּחֶ֖רֶב יִפְּל֣וּ סְבִיבוֹתָ֑יִךְ וְהַשְּׁלִישִׁית֙ לְכָל־ ר֣וּחַ אֱזָרֶ֔ה וְחֶ֖רֶב אָרִ֥יק אַחֲרֵיהֶֽם׃ (מו) וְכָלָ֣ה אַפִּ֗י וַהֲנִחוֹתִ֧י חֲמָתִ֛י בָּ֖ם וְהִנֶּחָ֑מְתִּי וְֽיָדְע֞וּ כִּי־ אֲנִ֣י יְהוָ֗ה דִּבַּ֙רְתִּי֙ בְּקִנְאָתִ֔י בְּכַלּוֹתִ֥י חֲמָתִ֖י בָּֽם׃ (מז) וְאֶתְּנֵךְ֙ לְחָרְבָּ֣ה וּלְחֶרְפָּ֔ה בַּגּוֹיִ֖ם אֲשֶׁ֣ר סְבִיבוֹתָ֑יִךְ לְעֵינֵ֖י כָּל־ עוֹבֵֽר׃ (מח) וְֽהָ֨יְתָ֜ה חֶרְפָּ֤ה וּגְדוּפָה֙ מוּסָ֣ר וּמְשַׁמָּ֔ה לַגּוֹיִ֖ם אֲשֶׁ֣ר סְבִיבוֹתָ֑יִךְ בַּעֲשׂוֹתִי֩ בָ֨ךְ שְׁפָטִ֜ים בְּאַ֤ף וּבְחֵמָה֙ וּבְתֹכְח֣וֹת חֵמָ֔ה אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (מט) בְּֽשַׁלְּחִ֡י אֶת־ חִצֵּי֩ הָרָעָ֨ב הָרָעִ֤ים בָּהֶם֙ אֲשֶׁ֣ר הָי֣וּ לְמַשְׁחִ֔ית אֲשֶׁר־ אֲשַׁלַּ֥ח אוֹתָ֖ם לְשַֽׁחֶתְכֶ֑ם וְרָעָב֙ אֹסֵ֣ף עֲלֵיכֶ֔ם וְשָׁבַרְתִּ֥י לָכֶ֖ם מַטֵּה־ לָֽחֶם׃ (נ) וְשִׁלַּחְתִּ֣י עֲ֠לֵיכֶם רָעָ֞ב וְחַיָּ֤ה רָעָה֙ וְשִׁכְּלֻ֔ךְ וְדֶ֥בֶר וָדָ֖ם יַעֲבָר־ בָּ֑ךְ וְחֶ֙רֶב֙ אָבִ֣יא עָלַ֔יִךְ אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ סוף פרק ה
