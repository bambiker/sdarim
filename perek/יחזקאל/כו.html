יחזקאל פרק כו לפי החלוקה לסדרים

פרק כו בספר יחזקאל (21 פסוקים) נמצא ביחזקאל סדר טו (פסוקים כב–מ) וביחזקאל סדר טז (פסוקים א–ב).

יחזקאל סדר טו

(א) וְהָיָ֨ה יְחֶזְקֵ֤אל לָכֶם֙ לְמוֹפֵ֔ת כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה תַּעֲשׂ֑וּ בְּבֹאָ֕הּ וִֽידַעְתֶּ֕ם כִּ֥י אֲנִ֖י אֲדֹנָ֥י יְהוִֽה׃ (ב) וְאַתָּ֣ה בֶן־ אָדָ֔ם הֲל֗וֹא בְּי֨וֹם קַחְתִּ֤י מֵהֶם֙ אֶת־ מָ֣עוּזָּ֔ם מְשׂ֖וֹשׂ תִּפְאַרְתָּ֑ם אֶת־ מַחְמַ֤ד עֵֽינֵיהֶם֙ וְאֶת־ מַשָּׂ֣א נַפְשָׁ֔ם בְּנֵיהֶ֖ם וּבְנוֹתֵיהֶֽם׃ (ג) בַּיּ֣וֹם הַה֔וּא יָב֥וֹא הַפָּלִ֖יט אֵלֶ֑יךָ לְהַשְׁמָע֖וּת אָזְנָֽיִם׃ (ד) בַּיּ֣וֹם הַה֗וּא יִפָּ֤תַח פִּ֙יךָ֙ אֶת־ הַפָּלִ֔יט וּתְדַבֵּ֕ר וְלֹ֥א תֵֽאָלֵ֖ם ע֑וֹד וְהָיִ֤יתָ לָהֶם֙ לְמוֹפֵ֔ת וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (ה) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ו) בֶּן־ אָדָ֕ם שִׂ֥ים פָּנֶ֖יךָ אֶל־ בְּנֵ֣י עַמּ֑וֹן וְהִנָּבֵ֖א עֲלֵיהֶֽם׃ (ז) וְאָֽמַרְתָּ֙ לִבְנֵ֣י עַמּ֔וֹן שִׁמְע֖וּ דְּבַר־ אֲדֹנָ֣י יְהוִ֑ה כֹּה־ אָמַ֣ר אֲדֹנָ֣י יְהוִ֡ה יַעַן֩ אָמְרֵ֨ךְ הֶאָ֜ח אֶל־ מִקְדָּשִׁ֣י כִֽי־ נִחָ֗ל וְאֶל־ אַדְמַ֤ת יִשְׂרָאֵל֙ כִּ֣י נָשַׁ֔מָּה וְאֶל־ בֵּ֣ית יְהוּדָ֔ה כִּ֥י הָלְכ֖וּ בַּגּוֹלָֽה׃ (ח) לָכֵ֡ן הִנְנִי֩ נֹתְנָ֨ךְ לִבְנֵי־ קֶ֜דֶם לְמֽוֹרָשָׁ֗ה וְיִשְּׁב֤וּ טִירֽוֹתֵיהֶם֙ בָּ֔ךְ וְנָ֥תְנוּ בָ֖ךְ מִשְׁכְּנֵיהֶ֑ם הֵ֚מָּה יֹאכְל֣וּ פִרְיֵ֔ךְ וְהֵ֖מָּה יִשְׁתּ֥וּ חֲלָבֵֽךְ׃ (ט) וְנָתַתִּ֤י אֶת־ רַבָּה֙ לִנְוֵ֣ה גְמַלִּ֔ים וְאֶת־ בְּנֵ֥י עַמּ֖וֹן לְמִרְבַּץ־ צֹ֑אן וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (י) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֚עַן מַחְאֲךָ֣ יָ֔ד וְרַקְעֲךָ֖ בְּרָ֑גֶל וַתִּשְׂמַ֤ח בְּכָל־ שָֽׁאטְךָ֙ בְּנֶ֔פֶשׁ אֶל־ אַדְמַ֖ת יִשְׂרָאֵֽל׃ (יא) לָכֵ֡ן הִנְנִי֩ נָטִ֨יתִי אֶת־ יָדִ֜י עָלֶ֗יךָ וּנְתַתִּ֤יךָֽ־ (לבג) לְבַז֙ לַגּוֹיִ֔ם וְהִכְרַתִּ֙יךָ֙ מִן־ הָ֣עַמִּ֔ים וְהַאֲבַדְתִּ֖יךָ מִן־ הָאֲרָצ֑וֹת אַשְׁמִ֣ידְךָ֔ וְיָדַעְתָּ֖ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (יב) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה יַ֗עַן אֲמֹ֤ר מוֹאָב֙ וְשֵׂעִ֔יר הִנֵּ֥ה כְּכָֽל־ הַגּוֹיִ֖ם בֵּ֥ית יְהוּדָֽה׃ (יג) לָכֵן֩ הִנְנִ֨י פֹתֵ֜חַ אֶת־ כֶּ֤תֶף מוֹאָב֙ מֵהֶ֣עָרִ֔ים מֵֽעָרָ֖יו מִקָּצֵ֑הוּ צְבִ֗י אֶ֚רֶץ בֵּ֣ית הַיְשִׁימֹ֔ת בַּ֥עַל מְע֖וֹן (וקריתמה) וְקִרְיָתָֽיְמָה׃ (יד) לִבְנֵי־ קֶ֙דֶם֙ עַל־ בְּנֵ֣י עַמּ֔וֹן וּנְתַתִּ֖יהָ לְמֽוֹרָשָׁ֑ה לְמַ֛עַן לֹֽא־ תִזָּכֵ֥ר בְּנֵֽי־ עַמּ֖וֹן בַּגּוֹיִֽם׃ (טו) וּבְמוֹאָ֖ב אֶעֱשֶׂ֣ה שְׁפָטִ֑ים וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (טז) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֣עַן עֲשׂ֥וֹת אֱד֛וֹם בִּנְקֹ֥ם נָקָ֖ם לְבֵ֣ית יְהוּדָ֑ה וַיֶּאְשְׁמ֥וּ אָשׁ֖וֹם וְנִקְּמ֥וּ בָהֶֽם׃ (יז) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה וְנָטִ֤תִי יָדִי֙ עַל־ אֱד֔וֹם וְהִכְרַתִּ֥י מִמֶּ֖נָּה אָדָ֣ם וּבְהֵמָ֑ה וּנְתַתִּ֤יהָ חָרְבָּה֙ מִתֵּימָ֔ן וּדְדָ֖נֶה בַּחֶ֥רֶב יִפֹּֽלוּ׃ (יח) וְנָתַתִּ֨י אֶת־ נִקְמָתִ֜י בֶּאֱד֗וֹם בְּיַד֙ עַמִּ֣י יִשְׂרָאֵ֔ל וְעָשׂ֣וּ בֶאֱד֔וֹם כְּאַפִּ֖י וְכַחֲמָתִ֑י וְיָֽדְעוּ֙ אֶת־ נִקְמָתִ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יט) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֛עַן עֲשׂ֥וֹת פְּלִשְׁתִּ֖ים בִּנְקָמָ֑ה וַיִּנָּקְמ֤וּ נָקָם֙ בִּשְׁאָ֣ט בְּנֶ֔פֶשׁ לְמַשְׁחִ֖ית אֵיבַ֥ת עוֹלָֽם׃ (כ) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֨י נוֹטֶ֤ה יָדִי֙ עַל־ פְּלִשְׁתִּ֔ים וְהִכְרַתִּ֖י אֶת־ כְּרֵתִ֑ים וְהַ֣אֲבַדְתִּ֔י אֶת־ שְׁאֵרִ֖ית ח֥וֹף הַיָּֽם׃ (כא) וְעָשִׂ֤יתִי בָם֙ נְקָמ֣וֹת גְּדֹל֔וֹת בְּתוֹכְח֖וֹת חֵמָ֑ה וְיָֽדְעוּ֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה בְּתִתִּ֥י אֶת־ נִקְמָתִ֖י בָּֽם׃ תחילת פרק כו (כב) וַיְהִ֛י בְּעַשְׁתֵּֽי־ עֶשְׂרֵ֥ה שָׁנָ֖ה בְּאֶחָ֣ד לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כג) בֶּן־ אָדָ֗ם יַ֠עַן אֲשֶׁר־ אָ֨מְרָה צֹּ֤ר עַל־ יְרוּשָׁלִַ֙ם֙ הֶאָ֔ח נִשְׁבְּרָ֛ה דַּלְת֥וֹת הָעַמִּ֖ים נָסֵ֣בָּה אֵלָ֑י אִמָּלְאָ֖ה הָחֳרָֽבָה׃ (כד) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י עָלַ֖יִךְ צֹ֑ר וְהַעֲלֵיתִ֤י עָלַ֙יִךְ֙ גּוֹיִ֣ם רַבִּ֔ים כְּהַעֲל֥וֹת הַיָּ֖ם לְגַלָּֽיו׃ (כה) וְשִׁחֲת֞וּ חֹמ֣וֹת צֹ֗ר וְהָֽרְסוּ֙ מִגְדָּלֶ֔יהָ וְסִֽחֵיתִ֥י עֲפָרָ֖הּ מִמֶּ֑נָּה וְנָתַתִּ֥י אוֹתָ֖הּ לִצְחִ֥יחַ סָֽלַע׃ (כו) מִשְׁטַ֨ח חֲרָמִ֤ים תִּֽהְיֶה֙ בְּת֣וֹךְ הַיָּ֔ם כִּ֚י אֲנִ֣י דִבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה וְהָיְתָ֥ה לְבַ֖ז לַגּוֹיִֽם׃ (כז) וּבְנוֹתֶ֙יהָ֙ אֲשֶׁ֣ר בַּשָּׂדֶ֔ה בַּחֶ֖רֶב תֵּהָרַ֑גְנָה וְיָדְע֖וּ כִּי־ אֲנִ֥י יְהוָֽה׃ (כח) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֧י מֵבִ֣יא אֶל־ צֹ֗ר נְבוּכַדְרֶאצַּ֧ר מֶֽלֶךְ־ בָּבֶ֛ל מִצָּפ֖וֹן מֶ֣לֶךְ מְלָכִ֑ים בְּס֛וּס וּבְרֶ֥כֶב וּבְפָרָשִׁ֖ים וְקָהָ֥ל וְעַם־ רָֽב׃ (כט) בְּנוֹתַ֥יִךְ בַּשָּׂדֶ֖ה בַּחֶ֣רֶב יַהֲרֹ֑ג וְנָתַ֨ן עָלַ֜יִךְ דָּיֵ֗ק וְשָׁפַ֤ךְ עָלַ֙יִךְ֙ סֹֽלְלָ֔ה וְהֵקִ֥ים עָלַ֖יִךְ צִנָּֽה׃ (ל) וּמְחִ֣י קָֽבָלּ֔וֹ יִתֵּ֖ן בְּחֹֽמוֹתָ֑יִךְ וּמִ֨גְדְּלֹתַ֔יִךְ יִתֹּ֖ץ בְּחַרְבוֹתָֽיו׃ (לא) מִשִּׁפְעַ֥ת סוּסָ֖יו יְכַסֵּ֣ךְ אֲבָקָ֑ם מִקּוֹל֩ פָּרַ֨שׁ וְגַלְגַּ֜ל וָרֶ֗כֶב תִּרְעַ֙שְׁנָה֙ חֽוֹמוֹתַ֔יִךְ בְּבֹאוֹ֙ בִּשְׁעָרַ֔יִךְ כִּמְבוֹאֵ֖י עִ֥יר מְבֻקָּעָֽה׃ (לב) בְּפַרְס֣וֹת סוּסָ֔יו יִרְמֹ֖ס אֶת־ כָּל־ חֽוּצוֹתָ֑יִךְ עַמֵּךְ֙ בַּחֶ֣רֶב יַהֲרֹ֔ג וּמַצְּב֥וֹת עֻזֵּ֖ךְ לָאָ֥רֶץ תֵּרֵֽד׃ (לג) וְשָׁלְל֣וּ חֵילֵ֗ךְ וּבָֽזְזוּ֙ רְכֻלָּתֵ֔ךְ וְהָֽרְסוּ֙ חוֹמוֹתַ֔יִךְ וּבָתֵּ֥י חֶמְדָּתֵ֖ךְ יִתֹּ֑צוּ וַאֲבָנַ֤יִךְ וְעֵצַ֙יִךְ֙ וַֽעֲפָרֵ֔ךְ בְּת֥וֹךְ מַ֖יִם יָשִֽׂימוּ׃ (לד) וְהִשְׁבַּתִּ֖י הֲמ֣וֹן שִׁירָ֑יִךְ וְק֣וֹל כִּנּוֹרַ֔יִךְ לֹ֥א יִשָּׁמַ֖ע עֽוֹד׃ (לה) וּנְתַתִּ֞יךְ לִצְחִ֣יחַ סֶ֗לַע מִשְׁטַ֤ח חֲרָמִים֙ תִּֽהְיֶ֔ה לֹ֥א תִבָּנֶ֖ה ע֑וֹד כִּ֣י אֲנִ֤י יְהוָה֙ דִּבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לו) כֹּ֥ה אָמַ֛ר אֲדֹנָ֥י יְהוִ֖ה לְצ֑וֹר הֲלֹ֣א ׀ מִקּ֣וֹל מַפַּלְתֵּ֗ךְ בֶּאֱנֹ֨ק חָלָ֜ל בֵּהָ֤רֵֽג הֶ֙רֶג֙ בְּתוֹכֵ֔ךְ יִרְעֲשׁ֖וּ הָאִיִּֽים׃ (לז) וְֽיָרְד֞וּ מֵעַ֣ל כִּסְאוֹתָ֗ם כֹּ֚ל נְשִׂיאֵ֣י הַיָּ֔ם וְהֵסִ֙ירוּ֙ אֶת־ מְעִ֣ילֵיהֶ֔ם וְאֶת־ בִּגְדֵ֥י רִקְמָתָ֖ם יִפְשֹׁ֑טוּ חֲרָד֤וֹת ׀ יִלְבָּ֙שׁוּ֙ עַל־ הָאָ֣רֶץ יֵשֵׁ֔בוּ וְחָֽרְדוּ֙ לִרְגָעִ֔ים וְשָׁמְמ֖וּ עָלָֽיִךְ׃ (לח) וְנָשְׂא֨וּ עָלַ֤יִךְ קִינָה֙ וְאָ֣מְרוּ לָ֔ךְ אֵ֣יךְ אָבַ֔דְתְּ נוֹשֶׁ֖בֶת מִיַּמִּ֑ים הָעִ֣יר הַהֻלָּ֗לָה אֲשֶׁר֩ הָיְתָ֨ה חֲזָקָ֤ה בַיָּם֙ הִ֣יא וְיֹשְׁבֶ֔יהָ אֲשֶׁר־ נָתְנ֥וּ חִתִּיתָ֖ם לְכָל־ (א) וְהוֹרַדְתִּיךְ֩ אֶת־ י֨וֹרְדֵי ב֜וֹר אֶל־ עַ֣ם עוֹלָ֗ם וְ֠הוֹשַׁבְתִּיךְ בְּאֶ֨רֶץ תַּחְתִּיּ֜וֹת כָּחֳרָב֤וֹת מֵֽעוֹלָם֙ אֶת־ י֣וֹרְדֵי ב֔וֹר לְמַ֖עַן לֹ֣א תֵשֵׁ֑בִי וְנָתַתִּ֥י צְבִ֖י בְּאֶ֥רֶץ חַיִּֽים׃ (ב) בַּלָּה֥וֹת אֶתְּנֵ֖ךְ וְאֵינֵ֑ךְ וּֽתְבֻקְשִׁ֗י וְלֹֽא־ תִמָּצְאִ֥י עוֹד֙ לְעוֹלָ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהֹוִֽה׃ סוף פרק כו (ג) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ד) וְאַתָּ֣ה בֶן־ אָדָ֔ם שָׂ֥א עַל־ צֹ֖ר קִינָֽה׃ (ה) וְאָמַרְתָּ֣ לְצ֗וֹר (הישבתי) הַיֹּשֶׁ֙בֶת֙ עַל־ מְבוֹאֹ֣ת יָ֔ם רֹכֶ֙לֶת֙ הָֽעַמִּ֔ים אֶל־ אִיִּ֖ים רַבִּ֑ים כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה צ֕וֹר אַ֣תְּ אָמַ֔רְתְּ אֲנִ֖י כְּלִ֥ילַת יֹֽפִי׃ (ו) בְּלֵ֥ב יַמִּ֖ים גְּבוּלָ֑יִךְ בֹּנַ֕יִךְ כָּלְל֖וּ יָפְיֵֽךְ׃ (ז) בְּרוֹשִׁ֤ים מִשְּׂנִיר֙ בָּ֣נוּ לָ֔ךְ אֵ֖ת כָּל־ לֻֽחֹתָ֑יִם אֶ֤רֶז מִלְּבָנוֹן֙ לָקָ֔חוּ לַעֲשׂ֥וֹת לִהְי֥וֹת לָ֖ךְ לְנֵ֑ס תְּכֵ֧לֶת וְאַרְגָּמָ֛ן מֵאִיֵּ֥י אֱלִישָׁ֖ה הָיָ֥ה מְכַסֵּֽךְ׃ (י) יֹשְׁבֵ֤י צִידוֹן֙ וְאַרְוַ֔ד הָי֥וּ שָׁטִ֖ים לָ֑ךְ חֲכָמַ֤יִךְ צוֹר֙ הָ֣יוּ בָ֔ךְ הֵ֖מָּה חֹבְלָֽיִךְ׃ (יא) בַּלָּה֣וֹת הָיִ֔ית וְאֵינֵ֖ךְ עַד־ עוֹלָֽם׃ (לט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (מ) בֶּן־ אָדָ֡ם אֱמֹר֩ לִנְגִ֨יד צֹ֜ר כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה יַ֣עַן גָּבַ֤הּ לִבְּךָ֙ וַתֹּ֙אמֶר֙ אֵ֣ל אָ֔נִי מוֹשַׁ֧ב אֱלֹהִ֛ים יָשַׁ֖בְתִּי בְּלֵ֣ב יַמִּ֑ים וְאַתָּ֤ה אָדָם֙ וְֽלֹא־ אֵ֔ל וַתִּתֵּ֥ן לִבְּךָ֖ כְּלֵ֥ב אֱלֹהִֽים׃ (מא) הִנֵּ֥ה חָכָ֛ם אַתָּ֖ה (מדנאל) מִדָּֽנִיֵּ֑אל כָּל־ סָת֖וּם לֹ֥א עֲמָמֽוּךָ׃ (מב) בְּחָכְמָֽתְךָ֙ וּבִתְבוּנָ֣תְךָ֔ עָשִׂ֥יתָ לְּךָ֖ חָ֑יִל וַתַּ֛עַשׂ זָהָ֥ב וָכֶ֖סֶף בְּאוֹצְרוֹתֶֽיךָ׃ (מג) בְּרֹ֧ב חָכְמָתְךָ֛ בִּרְכֻלָּתְךָ֖ הִרְבִּ֣יתָ חֵילֶ֑ךָ וַיִּגְבַּ֥הּ לְבָבְךָ֖ בְּחֵילֶֽךָ׃ (מד) לָכֵ֕ן כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה יַ֛עַן תִּתְּךָ֥ אֶת־ לְבָבְךָ֖ כְּלֵ֥ב אֱלֹהִֽים׃ (מה) לָכֵ֗ן הִנְנִ֨י מֵבִ֤יא עָלֶ֙יךָ֙ זָרִ֔ים עָרִיצֵ֖י גּוֹיִ֑ם וְהֵרִ֤יקוּ חַרְבוֹתָם֙ עַל־ יְפִ֣י חָכְמָתֶ֔ךָ וְחִלְּל֖וּ יִפְעָתֶֽךָ׃ (מו) לַשַּׁ֖חַת יֽוֹרִד֑וּךָ וָמַ֛תָּה מְמוֹתֵ֥י חָלָ֖ל בְּלֵ֥ב יַמִּֽים׃ (מז) הֶאָמֹ֤ר תֹּאמַר֙ אֱלֹהִ֣ים אָ֔נִי לִפְנֵ֖י הֹֽרְגֶ֑ךָ וְאַתָּ֥ה אָדָ֛ם וְלֹא־ אֵ֖ל בְּיַ֥ד מְחַלְלֶֽיךָ׃ (מח) מוֹתֵ֧י עֲרֵלִ֛ים תָּמ֖וּת בְּיַד־ זָרִ֑ים כִּ֚י אֲנִ֣י דִבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (מט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (נ) בֶּן־ אָדָ֕ם שָׂ֥א קִינָ֖ה עַל־ מֶ֣לֶךְ צ֑וֹר וְאָמַ֣רְתָּ לּ֗וֹ כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה אַתָּה֙ חוֹתֵ֣ם תָּכְנִ֔ית מָלֵ֥א חָכְמָ֖ה וּכְלִ֥יל יֹֽפִי׃
