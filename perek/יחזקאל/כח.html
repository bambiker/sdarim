יחזקאל פרק כח לפי החלוקה לסדרים

פרק כח בספר יחזקאל (26 פסוקים) נמצא ביחזקאל סדר טז (פסוקים לט–נ) וביחזקאל סדר יז (פסוקים א–יד).

יחזקאל סדר טז

(א) וְהוֹרַדְתִּיךְ֩ אֶת־ י֨וֹרְדֵי ב֜וֹר אֶל־ עַ֣ם עוֹלָ֗ם וְ֠הוֹשַׁבְתִּיךְ בְּאֶ֨רֶץ תַּחְתִּיּ֜וֹת כָּחֳרָב֤וֹת מֵֽעוֹלָם֙ אֶת־ י֣וֹרְדֵי ב֔וֹר לְמַ֖עַן לֹ֣א תֵשֵׁ֑בִי וְנָתַתִּ֥י צְבִ֖י בְּאֶ֥רֶץ חַיִּֽים׃ (ב) בַּלָּה֥וֹת אֶתְּנֵ֖ךְ וְאֵינֵ֑ךְ וּֽתְבֻקְשִׁ֗י וְלֹֽא־ תִמָּצְאִ֥י עוֹד֙ לְעוֹלָ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהֹוִֽה׃ (ג) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ד) וְאַתָּ֣ה בֶן־ אָדָ֔ם שָׂ֥א עַל־ צֹ֖ר קִינָֽה׃ (ה) וְאָמַרְתָּ֣ לְצ֗וֹר (הישבתי) הַיֹּשֶׁ֙בֶת֙ עַל־ מְבוֹאֹ֣ת יָ֔ם רֹכֶ֙לֶת֙ הָֽעַמִּ֔ים אֶל־ אִיִּ֖ים רַבִּ֑ים כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה צ֕וֹר אַ֣תְּ אָמַ֔רְתְּ אֲנִ֖י כְּלִ֥ילַת יֹֽפִי׃ (ו) בְּלֵ֥ב יַמִּ֖ים גְּבוּלָ֑יִךְ בֹּנַ֕יִךְ כָּלְל֖וּ יָפְיֵֽךְ׃ (ז) בְּרוֹשִׁ֤ים מִשְּׂנִיר֙ בָּ֣נוּ לָ֔ךְ אֵ֖ת כָּל־ לֻֽחֹתָ֑יִם אֶ֤רֶז מִלְּבָנוֹן֙ לָקָ֔חוּ לַעֲשׂ֥וֹת תֹּ֖רֶן עָלָֽיִךְ׃ (ח) אַלּוֹנִים֙ מִבָּ֔שָׁן עָשׂ֖וּ מִשּׁוֹטָ֑יִךְ קַרְשֵׁ֤ךְ עָֽשׂוּ־ שֵׁן֙ בַּת־ אֲשֻׁרִ֔ים מֵאִיֵּ֖י (כתים) כִּתִּיִּֽים׃ (ט) שֵׁשׁ־ בְּרִקְמָ֤ה מִמִּצְרַ֙יִם֙ הָיָ֣ה מִפְרָשֵׂ֔ךְ לִהְי֥וֹת לָ֖ךְ לְנֵ֑ס תְּכֵ֧לֶת וְאַרְגָּמָ֛ן מֵאִיֵּ֥י אֱלִישָׁ֖ה הָיָ֥ה מְכַסֵּֽךְ׃ (י) יֹשְׁבֵ֤י צִידוֹן֙ וְאַרְוַ֔ד הָי֥וּ שָׁטִ֖ים לָ֑ךְ חֲכָמַ֤יִךְ צוֹר֙ הָ֣יוּ בָ֔ךְ הֵ֖מָּה חֹבְלָֽיִךְ׃ (יא) זִקְנֵ֨י גְבַ֤ל וַחֲכָמֶ֙יהָ֙ הָ֣יוּ בָ֔ךְ מַחֲזִיקֵ֖י בִּדְקֵ֑ךְ כָּל־ אֳנִיּ֨וֹת הַיָּ֤ם וּמַלָּֽחֵיהֶם֙ הָ֣יוּ בָ֔ךְ לַעֲרֹ֖ב מַעֲרָבֵֽךְ׃ (יב) פָּרַ֨ס וְל֤וּד וּפוּט֙ הָי֣וּ בְחֵילֵ֔ךְ אַנְשֵׁ֖י מִלְחַמְתֵּ֑ךְ מָגֵ֤ן וְכוֹבַע֙ תִּלּוּ־ בָ֔ךְ הֵ֖מָּה נָתְנ֥וּ הֲדָרֵֽךְ׃ (יג) בְּנֵ֧י אַרְוַ֣ד וְחֵילֵ֗ךְ עַל־ חוֹמוֹתַ֙יִךְ֙ סָבִ֔יב וְגַ֨מָּדִ֔ים בְּמִגְדְּלוֹתַ֖יִךְ הָי֑וּ שִׁלְטֵיהֶ֞ם תִּלּ֤וּ עַל־ חוֹמוֹתַ֙יִךְ֙ סָבִ֔יב הֵ֖מָּה כָּלְל֥וּ יָפְיֵֽךְ׃ (יד) תַּרְשִׁ֥ישׁ סֹחַרְתֵּ֖ךְ מֵרֹ֣ב כָּל־ ה֑וֹן בְּכֶ֤סֶף בַּרְזֶל֙ בְּדִ֣יל וְעוֹפֶ֔רֶת נָתְנ֖וּ עִזְבוֹנָֽיִךְ׃ (טו) יָוָ֤ן תֻּבַל֙ וָמֶ֔שֶׁךְ הֵ֖מָּה רֹֽכְלָ֑יִךְ בְּנֶ֤פֶשׁ אָדָם֙ וּכְלֵ֣י נְחֹ֔שֶׁת נָתְנ֖וּ מַעֲרָבֵֽךְ׃ (טז) מִבֵּ֖ית תּוֹגַרְמָ֑ה סוּסִ֤ים וּפָֽרָשִׁים֙ וּפְרָדִ֔ים נָתְנ֖וּ עִזְבוֹנָֽיִךְ׃ (יז) בְּנֵ֤י דְדָן֙ רֹֽכְלַ֔יִךְ אִיִּ֥ים רַבִּ֖ים סְחֹרַ֣ת יָדֵ֑ךְ קַרְנ֥וֹת שֵׁן֙ (והובנים) וְהָבְנִ֔ים הֵשִׁ֖יבוּ אֶשְׁכָּרֵֽךְ׃ (יח) אֲרָ֥ם סֹחַרְתֵּ֖ךְ מֵרֹ֣ב מַעֲשָׂ֑יִךְ בְּ֠נֹפֶךְ אַרְגָּמָ֨ן וְרִקְמָ֤ה וּבוּץ֙ וְרָאמֹ֣ת וְכַדְכֹּ֔ד נָתְנ֖וּ בְּעִזְבוֹנָֽיִךְ׃ (יט) יְהוּדָה֙ וְאֶ֣רֶץ יִשְׂרָאֵ֔ל הֵ֖מָּה רֹכְלָ֑יִךְ בְּחִטֵּ֣י מִ֠נִּית וּפַנַּ֨ג וּדְבַ֤שׁ וָשֶׁ֙מֶן֙ וָצֹ֔רִי נָתְנ֖וּ מַעֲרָבֵֽךְ׃ (כ) דַּמֶּ֧שֶׂק סֹחַרְתֵּ֛ךְ בְּרֹ֥ב מַעֲשַׂ֖יִךְ מֵרֹ֣ב כָּל־ ה֑וֹן בְּיֵ֥ין חֶלְבּ֖וֹן וְצֶ֥מֶר צָֽחַר׃ (כא) וְדָ֤ן וְיָוָן֙ מְאוּזָּ֔ל בְּעִזְבוֹנַ֖יִךְ נָתָ֑נּוּ בַּרְזֶ֤ל עָשׁוֹת֙ קִדָּ֣ה וְקָנֶ֔ה בְּמַעֲרָבֵ֖ךְ הָיָֽה׃ (כב) דְּדָן֙ רֹֽכַלְתֵּ֔ךְ בְבִגְדֵי־ חֹ֖פֶשׁ לְרִכְבָּֽה׃ (כג) עֲרַב֙ וְכָל־ נְשִׂיאֵ֣י קֵדָ֔ר הֵ֖מָּה סֹחֲרֵ֣י יָדֵ֑ךְ בְּכָרִ֤ים וְאֵילִים֙ וְעַתּוּדִ֔ים בָּ֖ם סֹחֲרָֽיִךְ׃ (כד) רֹכְלֵ֤י שְׁבָא֙ וְרַעְמָ֔ה הֵ֖מָּה רֹכְלָ֑יִךְ בְּרֹ֨אשׁ כָּל־ בֹּ֜שֶׂם וּבְכָל־ אֶ֤בֶן יְקָרָה֙ וְזָהָ֔ב נָתְנ֖וּ עִזְבוֹנָֽיִךְ׃ (כה) חָרָ֤ן וְכַנֵּה֙ וָעֶ֔דֶן רֹכְלֵ֖י שְׁבָ֑א אַשּׁ֖וּר כִּלְמַ֥ד רֹכַלְתֵּֽךְ׃ (כו) הֵ֤מָּה רֹכְלַ֙יִךְ֙ בְּמַכְלֻלִ֔ים בִּגְלוֹמֵי֙ תְּכֵ֣לֶת וְרִקְמָ֔ה וּבְגִנְזֵ֖י בְּרֹמִ֑ים בַּחֲבָלִ֧ים חֲבֻשִׁ֛ים וַאֲרֻזִ֖ים בְּמַרְכֻלְתֵּֽךְ׃ (כז) אֳנִיּ֣וֹת תַּרְשִׁ֔ישׁ שָׁרוֹתַ֖יִךְ מַעֲרָבֵ֑ךְ וַתִּמָּלְאִ֧י וַֽתִּכְבְּדִ֛י מְאֹ֖ד בְּלֵ֥ב־ יַמִּֽים׃ (כח) בְּמַ֤יִם רַבִּים֙ הֱבִיא֔וּךְ הַשָּׁטִ֖ים אֹתָ֑ךְ ר֚וּחַ הַקָּדִ֔ים שְׁבָרֵ֖ךְ בְּלֵ֥ב יַמִּֽים׃ (כט) הוֹנֵךְ֙ וְעִזְבוֹנַ֔יִךְ מַעֲרָבֵ֕ךְ מַלָּחַ֖יִךְ וְחֹבְלָ֑יִךְ מַחֲזִיקֵ֣י בִדְקֵ֣ך וְֽעֹרְבֵ֣י מַ֠עֲרָבֵךְ וְכָל־ אַנְשֵׁ֨י מִלְחַמְתֵּ֜ךְ אֲשֶׁר־ בָּ֗ךְ וּבְכָל־ קְהָלֵךְ֙ אֲשֶׁ֣ר בְּתוֹכֵ֔ךְ יִפְּלוּ֙ בְּלֵ֣ב יַמִּ֔ים בְּי֖וֹם מַפַּלְתֵּֽךְ׃ (ל) לְק֖וֹל זַעֲקַ֣ת חֹבְלָ֑יִךְ יִרְעֲשׁ֖וּ מִגְרֹשֽׁוֹת׃ (לא) וְֽיָרְד֞וּ מֵאָנִיּֽוֹתֵיהֶ֗ם כֹּ֚ל תֹּפְשֵׂ֣י מָשׁ֔וֹט מַלָּחִ֕ים כֹּ֖ל חֹבְלֵ֣י הַיָּ֑ם אֶל־ הָאָ֖רֶץ יַעֲמֹֽדוּ׃ (לב) וְהִשְׁמִ֤יעוּ עָלַ֙יִךְ֙ בְּקוֹלָ֔ם וְיִזְעֲק֖וּ מָרָ֑ה וְיַעֲל֤וּ עָֽפָר֙ עַל־ רָ֣אשֵׁיהֶ֔ם בָּאֵ֖פֶר יִתְפַּלָּֽשׁוּ׃ (לג) וְהִקְרִ֤יחוּ אֵלַ֙יִךְ֙ קָרְחָ֔ה וְחָגְר֖וּ שַׂקִּ֑ים וּבָכ֥וּ אֵלַ֛יִךְ בְּמַר־ נֶ֖פֶשׁ מִסְפֵּ֥ד מָֽר׃ (לד) וְנָשְׂא֨וּ אֵלַ֤יִךְ בְּנִיהֶם֙ קִינָ֔ה וְקוֹנְנ֖וּ עָלָ֑יִךְ מִ֣י כְצ֔וֹר כְּדֻמָ֖ה בְּת֥וֹךְ הַיָּֽם׃ (לה) בְּצֵ֤את עִזְבוֹנַ֙יִךְ֙ מִיַּמִּ֔ים הִשְׂבַּ֖עַתְּ עַמִּ֣ים רַבִּ֑ים בְּרֹ֤ב הוֹנַ֙יִךְ֙ וּמַ֣עֲרָבַ֔יִךְ הֶעֱשַׁ֖רְתְּ מַלְכֵי־ אָֽרֶץ׃ (לו) עֵ֛ת נִשְׁבֶּ֥רֶת מִיַּמִּ֖ים בְּמַֽעֲמַקֵּי־ מָ֑יִם מַעֲרָבֵ֥ךְ וְכָל־ קְהָלֵ֖ךְ בְּתוֹכֵ֥ךְ נָפָֽלוּ׃ (לז) כֹּ֚ל יֹשְׁבֵ֣י הָאִיִּ֔ים שָׁמְמ֖וּ עָלָ֑יִךְ וּמַלְכֵיהֶם֙ שָׂ֣עֲרוּ שַׂ֔עַר רָעֲמ֖וּ פָּנִֽים׃ (לח) סֹֽחֲרִים֙ בָּ֣עַמִּ֔ים שָׁרְק֖וּ עָלָ֑יִךְ בַּלָּה֣וֹת הָיִ֔ית וְאֵינֵ֖ךְ עַד־ עוֹלָֽם׃ תחילת פרק כח (לט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (מ) בֶּן־ אָדָ֡ם אֱמֹר֩ לִנְגִ֨יד צֹ֜ר כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה יַ֣עַן גָּבַ֤הּ לִבְּךָ֙ וַתֹּ֙אמֶר֙ אֵ֣ל אָ֔נִי מוֹשַׁ֧ב אֱלֹהִ֛ים יָשַׁ֖בְתִּי בְּלֵ֣ב יַמִּ֑ים וְאַתָּ֤ה אָדָם֙ וְֽלֹא־ אֵ֔ל וַתִּתֵּ֥ן לִבְּךָ֖ כְּלֵ֥ב אֱלֹהִֽים׃ (מא) הִנֵּ֥ה חָכָ֛ם אַתָּ֖ה (מדנאל) מִדָּֽנִיֵּ֑אל כָּל־ סָת֖וּם לֹ֥א עֲמָמֽוּךָ׃ (מב) בְּחָכְמָֽתְךָ֙ וּבִתְבוּנָ֣תְךָ֔ עָשִׂ֥יתָ לְּךָ֖ חָ֑יִל וַתַּ֛עַשׂ זָהָ֥ב וָכֶ֖סֶף בְּאוֹצְרוֹתֶֽיךָ׃ (מג) בְּרֹ֧ב חָכְמָתְךָ֛ בִּרְכֻלָּתְךָ֖ הִרְבִּ֣יתָ חֵילֶ֑ךָ וַיִּגְבַּ֥הּ לְבָבְךָ֖ בְּחֵילֶֽךָ׃ (מד) לָכֵ֕ן כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה יַ֛עַן תִּתְּךָ֥ אֶת־ לְבָבְךָ֖ כְּלֵ֥ב אֱלֹהִֽים׃ (מה) לָכֵ֗ן הִנְנִ֨י מֵבִ֤יא עָלֶ֙יךָ֙ זָרִ֔ים עָרִיצֵ֖י גּוֹיִ֑ם וְהֵרִ֤יקוּ חַרְבוֹתָם֙ עַל־ יְפִ֣י חָכְמָתֶ֔ךָ וְחִלְּל֖וּ יִפְעָתֶֽךָ׃ (מו) לַשַּׁ֖חַת יֽוֹרִד֑וּךָ וָמַ֛תָּה מְמוֹתֵ֥י חָלָ֖ל בְּלֵ֥ב יַמִּֽים׃ (מז) הֶאָמֹ֤ר תֹּאמַר֙ אֱלֹהִ֣ים אָ֔נִי לִפְנֵ֖י לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לב) בֶּן־ אָדָ֗ם נְבוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּ֠בֶל הֶעֱבִ֨יד אֶת־ חֵיל֜וֹ עֲבֹדָ֤ה גְדֹלָה֙ אֶל־ צֹ֔ר כָּל־ רֹ֣אשׁ מֻקְרָ֔ח וְכָל־ כָּתֵ֖ף מְרוּטָ֑ה וְ֠שָׂכָר לֹא־ הָ֨יָה ל֤וֹ וּלְחֵילוֹ֙ מִצֹּ֔ר עַל־ הָעֲבֹדָ֖ה אֲשֶׁר־ עָבַ֥ד עָלֶֽיהָ׃ (לג) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י נֹתֵ֛ן לִנְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל אֶת־ אֶ֣רֶץ מִצְרָ֑יִם וְנָשָׂ֨א הֲמֹנָ֜הּ וְשָׁלַ֤ל שְׁלָלָהּ֙ וּבָזַ֣ז בִּזָּ֔הּ וְהָיְתָ֥ה שָׂכָ֖ר לְחֵילֽוֹ׃ (לד) פְּעֻלָּתוֹ֙ אֲשֶׁר־ עָ֣בַד בָּ֔הּ נָתַ֥תִּי ל֖וֹ אֶת־ אֶ֣רֶץ מִצְרָ֑יִם אֲשֶׁר֙ עָ֣שׂוּ לִ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃
