יחזקאל פרק כט לפי החלוקה לסדרים

פרק כט בספר יחזקאל (21 פסוקים) נמצא ביחזקאל סדר יז (פסוקים טו–לד) וביחזקאל סדר יח (פסוקים א–א).

יחזקאל סדר יז

(א) בְּעֵ֨דֶן גַּן־ אֱלֹהִ֜ים הָיִ֗יתָ כָּל־ אֶ֨בֶן יְקָרָ֤ה מְסֻכָתֶ֙ךָ֙ אֹ֣דֶם פִּטְדָ֞ה וְיָהֲלֹ֗ם תַּרְשִׁ֥ישׁ שֹׁ֙הַם֙ וְיָ֣שְׁפֵ֔ה סַפִּ֣יר נֹ֔פֶךְ וּבָרְקַ֖ת וְזָהָ֑ב מְלֶ֨אכֶת תֻּפֶּ֤יךָ וּנְקָבֶ֙יךָ֙ בָּ֔ךְ בְּי֥וֹם הִבָּרַאֲךָ֖ כּוֹנָֽנוּ׃ (ב) אַ֨תְּ־ כְּר֔וּב מִמְשַׁ֖ח הַסּוֹכֵ֑ךְ וּנְתַתִּ֗יךָ בְּהַ֨ר קֹ֤דֶשׁ אֱלֹהִים֙ הָיִ֔יתָ בְּת֥וֹךְ אַבְנֵי־ אֵ֖שׁ הִתְהַלָּֽכְתָּ׃ (ג) תָּמִ֤ים אַתָּה֙ בִּדְרָכֶ֔יךָ מִיּ֖וֹם הִבָּֽרְאָ֑ךְ עַד־ נִמְצָ֥א עַוְלָ֖תָה בָּֽךְ׃ (ד) בְּרֹ֣ב רְכֻלָּתְךָ֗ מָל֧וּ תוֹכְךָ֛ חָמָ֖ס וַֽתֶּחֱטָ֑א וָאֶחַלֶּלְךָ֩ מֵהַ֨ר אֱלֹהִ֤ים וָֽאַבֶּדְךָ֙ כְּר֣וּב הַסֹּכֵ֔ךְ מִתּ֖וֹךְ אַבְנֵי־ אֵֽשׁ׃ (ה) גָּבַ֤הּ לִבְּךָ֙ בְּיָפְיֶ֔ךָ שִׁחַ֥תָּ חָכְמָתְךָ֖ עַל־ יִפְעָתֶ֑ךָ עַל־ אֶ֣רֶץ הִשְׁלַכְתִּ֗יךָ לִפְנֵ֧י מְלָכִ֛ים נְתַתִּ֖יךָ לְרַ֥אֲוָה בָֽךְ׃ (ו) מֵרֹ֣ב עֲוֺנֶ֗יךָ בְּעֶ֙וֶל֙ רְכֻלָּ֣תְךָ֔ חִלַּ֖לְתָּ מִקְדָּשֶׁ֑יךָ וָֽאוֹצִא־ אֵ֤שׁ מִתּֽוֹכְךָ֙ הִ֣יא אֲכָלַ֔תְךָ וָאֶתֶּנְךָ֤ לְאֵ֙פֶר֙ עַל־ הָאָ֔רֶץ לְעֵינֵ֖י כָּל־ רֹאֶֽיךָ׃ (ז) כָּל־ יוֹדְעֶ֙יךָ֙ בָּֽעַמִּ֔ים שָׁמְמ֖וּ עָלֶ֑יךָ בַּלָּה֣וֹת הָיִ֔יתָ וְאֵינְךָ֖ עַד־ עוֹלָֽם׃ (ח) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ט) בֶּן־ אָדָ֕ם שִׂ֥ים פָּנֶ֖יךָ אֶל־ צִיד֑וֹן וְהִנָּבֵ֖א עָלֶֽיהָ׃ (י) וְאָמַרְתָּ֗ כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֤י עָלַ֙יִךְ֙ צִיד֔וֹן וְנִכְבַּדְתִּ֖י בְּתוֹכֵ֑ךְ וְֽיָדְע֞וּ כִּֽי־ אֲנִ֣י יְהוָ֗ה בַּעֲשׂ֥וֹתִי בָ֛הּ שְׁפָטִ֖ים וְנִקְדַּ֥שְׁתִּי בָֽהּ׃ (יא) וְשִׁלַּחְתִּי־ בָ֞הּ דֶּ֤בֶר וָדָם֙ בְּח֣וּצוֹתֶ֔יהָ וְנִפְלַ֤ל חָלָל֙ בְּתוֹכָ֔הּ בְּחֶ֥רֶב עָלֶ֖יהָ מִסָּבִ֑יב וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (יב) וְלֹֽא־ יִהְיֶ֨ה ע֜וֹד לְבֵ֣ית יִשְׂרָאֵ֗ל סִלּ֤וֹן מַמְאִיר֙ וְק֣וֹץ מַכְאִ֔ב מִכֹּל֙ סְבִ֣יבֹתָ֔ם הַשָּׁאטִ֖ים אוֹתָ֑ם וְיָ֣דְע֔וּ כִּ֥י אֲנִ֖י אֲדֹנָ֥י יְהוִֽה׃ (יג) כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ בְּקַבְּצִ֣י ׀ אֶת־ בֵּ֣ית יִשְׂרָאֵ֗ל מִן־ הָֽעַמִּים֙ אֲשֶׁ֣ר נָפֹ֣צוּ בָ֔ם וְנִקְדַּ֥שְׁתִּי בָ֖ם לְעֵינֵ֣י הַגּוֹיִ֑ם וְיָֽשְׁבוּ֙ עַל־ אַדְמָתָ֔ם אֲשֶׁ֥ר נָתַ֖תִּי לְעַבְדִּ֥י לְיַעֲקֹֽב׃ (יד) וְיָשְׁב֣וּ עָלֶיהָ֮ לָבֶטַח֒ וּבָנ֤וּ בָתִּים֙ וְנָטְע֣וּ כְרָמִ֔ים וְיָשְׁב֖וּ לָבֶ֑טַח בַּעֲשׂוֹתִ֣י שְׁפָטִ֗ים בְּכֹ֨ל הַשָּׁאטִ֤ים אֹתָם֙ מִסְּבִ֣יבוֹתָ֔ם וְיָ֣דְע֔וּ כִּ֛י אֲנִ֥י יְהוָ֖ה אֱלֹהֵיהֶֽם׃ תחילת פרק כט (טו) בַּשָּׁנָה֙ הָעֲשִׂירִ֔ית בָּעֲשִׂרִ֕י בִּשְׁנֵ֥ים עָשָׂ֖ר לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (טז) בֶּן־ אָדָ֕ם שִׂ֣ים פָּנֶ֔יךָ עַל־ פַּרְעֹ֖ה מֶ֣לֶךְ מִצְרָ֑יִם וְהִנָּבֵ֣א עָלָ֔יו וְעַל־ מִצְרַ֖יִם כֻּלָּֽהּ׃ (יז) דַּבֵּ֨ר וְאָמַרְתָּ֜ כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה הִנְנִ֤י עָלֶ֙יךָ֙ פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם הַתַּנִּים֙ הַגָּד֔וֹל הָרֹבֵ֖ץ בְּת֣וֹךְ יְאֹרָ֑יו אֲשֶׁ֥ר אָמַ֛ר לִ֥י יְאֹרִ֖י וַאֲנִ֥י עֲשִׂיתִֽנִי׃ (יח) וְנָתַתִּ֤י (חחיים) חַחִים֙ בִּלְחָיֶ֔יךָ וְהִדְבַּקְתִּ֥י דְגַת־ יְאֹרֶ֖יךָ בְּקַשְׂקְשֹׂתֶ֑יךָ וְהַעֲלִיתִ֙יךָ֙ מִתּ֣וֹךְ יְאֹרֶ֔יךָ וְאֵת֙ כָּל־ דְּגַ֣ת יְאֹרֶ֔יךָ בְּקַשְׂקְשֹׂתֶ֖יךָ תִּדְבָּֽק׃ (יט) וּנְטַשְׁתִּ֣יךָ הַמִּדְבָּ֗רָה אוֹתְךָ֙ וְאֵת֙ כָּל־ דְּגַ֣ת יְאֹרֶ֔יךָ עַל־ פְּנֵ֤י הַשָּׂדֶה֙ תִּפּ֔וֹל לֹ֥א תֵאָסֵ֖ף וְלֹ֣א תִקָּבֵ֑ץ לְחַיַּ֥ת הָאָ֛רֶץ וּלְע֥וֹף הַשָּׁמַ֖יִם נְתַתִּ֥יךָ לְאָכְלָֽה׃ (כ) וְיָֽדְעוּ֙ כָּל־ יֹשְׁבֵ֣י מִצְרַ֔יִם כִּ֖י אֲנִ֣י יְהוָ֑ה יַ֧עַן הֱיוֹתָ֛ם מִשְׁעֶ֥נֶת קָנֶ֖ה לְבֵ֥ית יִשְׂרָאֵֽל׃ (כא) בְּתָפְשָׂ֨ם בְּךָ֤ (בכפך) בַכַּף֙ תֵּר֔וֹץ וּבָקַעְתָּ֥ לָהֶ֖ם כָּל־ כָּתֵ֑ף וּבְהִֽשָּׁעֲנָ֤ם עָלֶ֙יךָ֙ תִּשָּׁבֵ֔ר וְהַעֲמַדְתָּ֥ לָהֶ֖ם כָּל־ מָתְנָֽיִם׃ (כב) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֛י מֵבִ֥יא עָלַ֖יִךְ חָ֑רֶב וְהִכְרַתִּ֥י מִמֵּ֖ךְ אָדָ֥ם וּבְהֵמָֽה׃ (כג) וְהָיְתָ֤ה אֶֽרֶץ־ מִצְרַ֙יִם֙ לִשְׁמָמָ֣ה וְחָרְבָּ֔ה וְיָדְע֖וּ כִּֽי־ אֲנִ֣י יְהוָ֑ה יַ֧עַן אָמַ֛ר יְאֹ֥ר לִ֖י וַאֲנִ֥י עָשִֽׂיתִי׃ (כד) לָכֵ֛ן הִנְנִ֥י אֵלֶ֖יךָ וְאֶל־ יְאֹרֶ֑יךָ וְנָתַתִּ֞י אֶת־ אֶ֣רֶץ מִצְרַ֗יִם לְחָרְבוֹת֙ חֹ֣רֶב שְׁמָמָ֔ה מִמִּגְדֹּ֥ל סְוֵנֵ֖ה וְעַד־ גְּב֥וּל כּֽוּשׁ׃ (כה) לֹ֤א תַעֲבָר־ בָּהּ֙ רֶ֣גֶל אָדָ֔ם וְרֶ֥גֶל בְּהֵמָ֖ה לֹ֣א תַעֲבָר־ בָּ֑הּ וְלֹ֥א תֵשֵׁ֖ב אַרְבָּעִ֥ים שָׁנָֽה׃ (כו) וְנָתַתִּ֣י אֶת־ אֶרֶץ֩ מִצְרַ֨יִם שְׁמָמָ֜ה בְּת֣וֹךְ ׀ אֲרָצ֣וֹת נְשַׁמּ֗וֹת וְעָרֶ֙יהָ֙ בְּת֨וֹךְ עָרִ֤ים מָֽחֳרָבוֹת֙ תִּֽהְיֶ֣יןָ שְׁמָמָ֔ה אַרְבָּעִ֖ים שָׁנָ֑ה וַהֲפִצֹתִ֤י אֶת־ מִצְרַ֙יִם֙ בַּגּוֹיִ֔ם וְֽזֵרִיתִ֖ים בָּאֲרָצֽוֹת׃ (כז) כִּ֛י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה מִקֵּ֞ץ אַרְבָּעִ֤ים שָׁנָה֙ אֲקַבֵּ֣ץ אֶת־ מִצְרַ֔יִם מִן־ הָעַמִּ֖ים אֲשֶׁר־ נָפֹ֥צוּ שָֽׁמָּה׃ (כח) וְשַׁבְתִּי֙ אֶת־ שְׁב֣וּת מִצְרַ֔יִם וַהֲשִׁבֹתִ֤י אֹתָם֙ אֶ֣רֶץ פַּתְר֔וֹס עַל־ אֶ֖רֶץ מְכֽוּרָתָ֑ם וְהָ֥יוּ שָׁ֖ם מַמְלָכָ֥ה שְׁפָלָֽה׃ (כט) מִן־ הַמַּמְלָכוֹת֙ תִּהְיֶ֣ה שְׁפָלָ֔ה וְלֹֽא־ תִתְנַשֵּׂ֥א ע֖וֹד עַל־ הַגּוֹיִ֑ם וְהִ֨מְעַטְתִּ֔ים לְבִלְתִּ֖י רְד֥וֹת בַּגּוֹיִֽם׃ (ל) וְלֹ֣א יִֽהְיֶה־ עוֹד֩ לְבֵ֨ית יִשְׂרָאֵ֤ל לְמִבְטָח֙ מַזְכִּ֣יר עָוֺ֔ן בִּפְנוֹתָ֖ם אַחֲרֵיהֶ֑ם וְיָ֣דְע֔וּ כִּ֥י אֲנִ֖י אֲדֹנָ֥י יְהוִֽה׃ (לא) וַיְהִ֗י בְּעֶשְׂרִ֤ים וָשֶׁ֙בַע֙ שָׁנָ֔ה בָּֽרִאשׁ֖וֹן בְּאֶחָ֣ד לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לב) בֶּן־ אָדָ֗ם נְבוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּ֠בֶל הֶעֱבִ֨יד אֶת־ חֵיל֜וֹ עֲבֹדָ֤ה גְדֹלָה֙ אֶל־ צֹ֔ר כָּל־ רֹ֣אשׁ מֻקְרָ֔ח וְכָל־ כָּתֵ֖ף מְרוּטָ֑ה וְ֠שָׂכָר לֹא־ הָ֨יָה ל֤וֹ וּלְחֵילוֹ֙ מִצֹּ֔ר עַל־ הָעֲבֹדָ֖ה אֲשֶׁר־ עָבַ֥ד עָלֶֽיהָ׃ (לג) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י נֹתֵ֛ן לִנְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל אֶת־ אֶ֣רֶץ מִצְרָ֑יִם וְנָשָׂ֨א הֲמֹנָ֜הּ וְשָׁלַ֤ל שְׁלָלָהּ֙ וּבָזַ֣ז בִּזָּ֔הּ וְהָיְתָ֥ה שָׂכָ֖ר לְחֵילֽוֹ׃ (לד) פְּעֻלָּתוֹ֙ אֲשֶׁר־ עָ֣בַד בָּ֔הּ נָתַ֥תִּי ל֖וֹ אֶת־ אֶ֣רֶץ מִצְרָ֑יִם אֲשֶׁר֙ עָ֣שׂוּ לִ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃

יחזקאל סדר יח

(א) בַּיּ֣וֹם הַה֗וּא אַצְמִ֤יחַ קֶ֙רֶן֙ לְבֵ֣ית יִשְׂרָאֵ֔ל וּלְךָ֛ אֶתֵּ֥ן פִּתְחֽוֹן־ פֶּ֖ה בְּתוֹכָ֑ם וְיָדְע֖וּ כִּי־ אֲנִ֥י יְהוָֽה׃ סוף פרק כט (ב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ג) בֶּן־ אָדָ֕ם הִנָּבֵא֙ וְאָ֣מַרְתָּ֔ כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הֵילִ֖ילוּ הָ֥הּ לַיּֽוֹם׃ (ד) כִּֽי־ קָר֣וֹב י֔וֹם וְקָר֥וֹב י֖וֹם לַֽיהוָ֑ה י֣וֹם עָנָ֔ן עֵ֥ת גּוֹיִ֖ם יִֽהְיֶֽה׃ (ה) וּבָאָ֥ה חֶ֙רֶב֙ בְּמִצְרַ֔יִם וְהָיְתָ֤ה חַלְחָלָה֙ בְּכ֔וּשׁ בִּנְפֹ֥ל חָלָ֖ל בְּמִצְרָ֑יִם וְלָקְח֣וּ הֲמוֹנָ֔הּ וְנֶהֶרְס֖וּ יְסוֹדֹתֶֽיהָ׃ (ו) כּ֣וּשׁ וּפ֤וּט וְלוּד֙ וְכָל־ הָעֶ֣רֶב וְכ֔וּב וּבְנֵ֖י אֶ֣רֶץ הַבְּרִ֑ית אִתָּ֖ם בַּחֶ֥רֶב יִפֹּֽלוּ׃ (ז) כֹּ֚ה אָמַ֣ר יְהוָ֔ה וְנָֽפְלוּ֙ סֹמְכֵ֣י מִצְרַ֔יִם וְיָרַ֖ד גְּא֣וֹן עֻזָּ֑הּ מִמִּגְדֹּ֣ל סְוֵנֵ֗ה בַּחֶ֙רֶב֙ יִפְּלוּ־ בָ֔הּ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ח) וְנָשַׁ֕מּוּ בְּת֖וֹךְ אֲרָצ֣וֹת נְשַׁמּ֑וֹת וְעָרָ֕יו בְּתוֹךְ־ עָרִ֥ים נַחֲרָב֖וֹת תִּֽהְיֶֽינָה׃ (ט) וְיָדְע֖וּ כִּֽי־ אֲנִ֣י יְהוָ֑ה בְּתִתִּי־ אֵ֣שׁ בְּמִצְרַ֔יִם וְנִשְׁבְּר֖וּ כָּל־ עֹזְרֶֽיהָ׃ (י) בַּיּ֣וֹם הַה֗וּא יֵצְא֨וּ מַלְאָכִ֤ים מִלְּפָנַי֙ בַּצִּ֔ים לְהַחֲרִ֖יד אֶת־ כּ֣וּשׁ בֶּ֑טַח וְהָיְתָ֨ה חַלְחָלָ֤ה בָהֶם֙ בְּי֣וֹם מִצְרַ֔יִם כִּ֥י הִנֵּ֖ה בָּאָֽה׃ (יא) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה וְהִשְׁבַּתִּי֙ אֶת־ הֲמ֣וֹן מִצְרַ֔יִם בְּיַ֖ד נְבוּכַדְרֶאצַּ֥ר מֶלֶךְ־ בָּבֶֽל׃ (יב) ה֠וּא וְעַמּ֤וֹ אִתּוֹ֙ עָרִיצֵ֣י גוֹיִ֔ם מֽוּבָאִ֖ים לְשַׁחֵ֣ת הָאָ֑רֶץ וְהֵרִ֤יקוּ חַרְבוֹתָם֙ עַל־ מִצְרַ֔יִם וּמָלְא֥וּ אֶת־ הָאָ֖רֶץ חָלָֽל׃ (יג) וְנָתַתִּ֤י יְאֹרִים֙ חָֽרָבָ֔ה וּמָכַרְתִּ֥י אֶת־ הָאָ֖רֶץ בְּיַד־ רָעִ֑ים וַהֲשִׁמֹּתִ֞י אֶ֤רֶץ וּמְלֹאָהּ֙ בְּיַד־ זָרִ֔ים אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (יד) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה וְהַאֲבַדְתִּ֨י גִלּוּלִ֜ים וְהִשְׁבַּתִּ֤י אֱלִילִים֙ מִנֹּ֔ף וְנָשִׂ֥יא מֵאֶֽרֶץ־ מִצְרַ֖יִם לֹ֣א יִֽהְיֶה־ ע֑וֹד וְנָתַתִּ֥י יִרְאָ֖ה בְּאֶ֥רֶץ מִצְרָֽיִם׃ (טו) וַהֲשִׁמֹּתִי֙ אֶת־ פַּתְר֔וֹס וְנָתַ֥תִּי אֵ֖שׁ בְּצֹ֑עַן וְעָשִׂ֥יתִי שְׁפָטִ֖ים בְּנֹֽא׃ (טז) וְשָׁפַכְתִּ֣י חֲמָתִ֔י עַל־ סִ֖ין מָע֣וֹז מִצְרָ֑יִם וְהִכְרַתִּ֖י אֶת־ הֲמ֥וֹן נֹֽא׃ (יז) וְנָתַ֤תִּי אֵשׁ֙ בְּמִצְרַ֔יִם ח֤וּל (תחיל) תָּחוּל֙ סִ֔ין וְנֹ֖א תִּהְיֶ֣ה לְהִבָּקֵ֑עַ וְנֹ֖ף צָרֵ֥י יוֹמָֽם׃ (יח) בַּח֥וּרֵי אָ֛וֶן וּפִי־ בֶ֖סֶת בַּחֶ֣רֶב יִפֹּ֑לוּ וְהֵ֖נָּה בַּשְּׁבִ֥י תֵלַֽכְנָה׃ (יט) וּבִֽתְחַפְנְחֵס֙ חָשַׂ֣ךְ הַיּ֔וֹם בְּשִׁבְרִי־ שָׁם֙ אֶת־ מֹט֣וֹת מִצְרַ֔יִם וְנִשְׁבַּת־ בָּ֖הּ גְּא֣וֹן עֻזָּ֑הּ הִ֚יא עָנָ֣ן יְכַסֶּ֔נָּה וּבְנוֹתֶ֖יהָ בַּשְּׁבִ֥י תֵלַֽכְנָה׃ (כ) וְעָשִׂ֥יתִי שְׁפָטִ֖ים בְּמִצְרָ֑יִם וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כא) וַיְהִ֗י בְּאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה בָּֽרִאשׁ֖וֹן בְּשִׁבְעָ֣ה לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כב) בֶּן־ אָדָ֕ם אֶת־ זְר֛וֹעַ פַּרְעֹ֥ה מֶֽלֶךְ־ מִצְרַ֖יִם שָׁבָ֑רְתִּי וְהִנֵּ֣ה לֹֽא־ חֻ֠בְּשָׁה לָתֵ֨ת רְפֻא֜וֹת לָשׂ֥וּם חִתּ֛וּל לְחָבְשָׁ֥הּ לְחָזְקָ֖הּ לִתְפֹּ֥שׂ בֶּחָֽרֶב׃ (כג) לָכֵ֞ן כֹּה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה הִנְנִי֙ אֶל־ פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם וְשָֽׁבַרְתִּי֙ אֶת־ זְרֹ֣עֹתָ֔יו אֶת־ הַחֲזָקָ֖ה וְאֶת־ הַנִּשְׁבָּ֑רֶת וְהִפַּלְתִּ֥י אֶת־ הַחֶ֖רֶב מִיָּדֽוֹ׃ (כד) וַהֲפִצוֹתִ֥י אֶת־ מִצְרַ֖יִם בַּגּוֹיִ֑ם וְזֵרִיתִ֖ם בָּאֲרָצֽוֹת׃ (כה) וְחִזַּקְתִּ֗י אֶת־ זְרֹעוֹת֙ מֶ֣לֶךְ בָּבֶ֔ל וְנָתַתִּ֥י אֶת־ חַרְבִּ֖י בְּיָד֑וֹ וְשָׁבַרְתִּי֙ אֶת־ זְרֹע֣וֹת פַּרְעֹ֔ה וְנָאַ֛ק נַאֲק֥וֹת חָלָ֖ל לְפָנָֽיו׃ (כו) וְהַחֲזַקְתִּ֗י אֶת־ זְרֹעוֹת֙ מֶ֣לֶךְ בָּבֶ֔ל וּזְרֹע֥וֹת פַּרְעֹ֖ה תִּפֹּ֑לְנָה וְֽיָדְע֞וּ כִּֽי־ אֲנִ֣י יְהוָ֗ה בְּתִתִּ֤י חַרְבִּי֙ בְּיַ֣ד מֶֽלֶךְ־ בָּבֶ֔ל וְנָטָ֥ה אוֹתָ֖הּ אֶל־ אֶ֥רֶץ מִצְרָֽיִם׃ (כז) וַהֲפִצוֹתִ֤י אֶת־ מִצְרַ֙יִם֙ בַּגּוֹיִ֔ם וְזֵרִיתִ֥י אוֹתָ֖ם בָּאֲרָצ֑וֹת וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כח) וַיְהִ֗י בְּאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה בַּשְּׁלִישִׁ֖י בְּאֶחָ֣ד לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כט) בֶּן־ אָדָ֕ם אֱמֹ֛ר אֶל־ פַּרְעֹ֥ה מֶֽלֶךְ־ מִצְרַ֖יִם וְאֶל־ הֲמוֹנ֑וֹ אֶל־ מִ֖י דָּמִ֥יתָ בְגָדְלֶֽךָ׃ (ל) הִנֵּ֨ה אַשּׁ֜וּר אֶ֣רֶז בַּלְּבָנ֗וֹן יְפֵ֥ה עָנָ֛ף וְחֹ֥רֶשׁ מֵצַ֖ל וּגְבַ֣הּ קוֹמָ֑ה וּבֵ֣ין עֲבֹתִ֔ים הָיְתָ֖ה צַמַּרְתּֽוֹ׃ (לא) מַ֣יִם גִּדְּל֔וּהוּ תְּה֖וֹם רֹֽמְמָ֑תְהוּ אֶת־ נַהֲרֹתֶ֗יהָ הֹלֵךְ֙ סְבִיב֣וֹת מַטָּעָ֔הּ וְאֶת־ תְּעָלֹתֶ֣יהָ שִׁלְחָ֔ה אֶ֖ל כָּל־ עֲצֵ֥י הַשָּׂדֶֽה׃ (לב) עַל־ כֵּן֙ גָּבְהָ֣א קֹמָת֔וֹ מִכֹּ֖ל עֲצֵ֣י הַשָּׂדֶ֑ה וַתִּרְבֶּ֨ינָה סַֽרְעַפֹּתָ֜יו וַתֶּאֱרַ֧כְנָה (פארתו) פֹארֹתָ֛יו מִמַּ֥יִם רַבִּ֖ים בְּשַׁלְּחֽוֹ׃ (לג) בִּסְעַפֹּתָ֤יו קִֽנְנוּ֙ כָּל־ ע֣וֹף הַשָּׁמַ֔יִם וְתַ֤חַת פֹּֽארֹתָיו֙ יָֽלְד֔וּ כֹּ֖ל חַיַּ֣ת הַשָּׂדֶ֑ה וּבְצִלּוֹ֙ יֵֽשְׁב֔וּ כֹּ֖ל גּוֹיִ֥ם רַבִּֽים׃ (לד) וַיְּיִ֣ף בְּגָדְל֔וֹ בְּאֹ֖רֶךְ דָּֽלִיּוֹתָ֑יו כִּֽי־ הָיָ֥ה שָׁרְשׁ֖וֹ אֶל־ מַ֥יִם רַבִּֽים׃ (לה) אֲרָזִ֣ים לֹֽא־ עֲמָמֻהוּ֮ בְּגַן־ אֱלֹהִים֒ בְּרוֹשִׁ֗ים לֹ֤א דָמוּ֙ אֶל־ סְעַפֹּתָ֔יו וְעַרְמֹנִ֥ים לֹֽא־ הָי֖וּ כְּפֹֽארֹתָ֑יו כָּל־ עֵץ֙ בְּגַן־ אֱלֹהִ֔ים לֹא־ דָמָ֥ה אֵלָ֖יו בְּיָפְיֽוֹ׃ (לו) יָפֶ֣ה עֲשִׂיתִ֔יו בְּרֹ֖ב דָּֽלִיּוֹתָ֑יו וַיְקַנְאֻ֙הוּ֙ כָּל־ עֲצֵי־ עֵ֔דֶן אֲשֶׁ֖ר בְּגַ֥ן הָאֱלֹהִֽים׃ (לז) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֕עַן אֲשֶׁ֥ר גָּבַ֖הְתָּ בְּקוֹמָ֑ה וַיִּתֵּ֤ן צַמַּרְתּוֹ֙ אֶל־ בֵּ֣ין עֲבוֹתִ֔ים וְרָ֥ם לְבָב֖וֹ בְּגָבְהֽוֹ׃ (לח) וְאֶ֨תְּנֵ֔הוּ בְּיַ֖ד אֵ֣יל גּוֹיִ֑ם עָשׂ֤וֹ יַֽעֲשֶׂה֙ ל֔וֹ כְּרִשְׁע֖וֹ גֵּרַשְׁתִּֽהוּ׃ (לט) וַיִּכְרְתֻ֧הוּ זָרִ֛ים עָרִיצֵ֥י גוֹיִ֖ם וַֽיִּטְּשֻׁ֑הוּ אֶל־ הֶ֠הָרִים וּבְכָל־ גֵּ֨אָי֜וֹת נָפְל֣וּ דָלִיּוֹתָ֗יו וַתִּשָּׁבַ֤רְנָה פֹֽארֹתָיו֙ בְּכֹל֙ אֲפִיקֵ֣י הָאָ֔רֶץ וַיֵּרְד֧וּ מִצִּלּ֛וֹ כָּל־ עַמֵּ֥י הָאָ֖רֶץ וַֽיִּטְּשֻֽׁהוּ׃ (מ) עַל־ מַפַּלְתּ֥וֹ יִשְׁכְּנ֖וּ כָּל־ ע֣וֹף הַשָּׁמָ֑יִם וְאֶל־ פֹּארֹתָ֣יו הָי֔וּ כֹּ֖ל חַיַּ֥ת הַשָּׂדֶֽה׃ (מא) לְמַ֡עַן אֲשֶׁר֩ לֹא־ יִגְבְּה֨וּ בְקוֹמָתָ֜ם כָּל־ עֲצֵי־ מַ֗יִם וְלֹֽא־ יִתְּנ֤וּ אֶת־ צַמַּרְתָּם֙ אֶל־ בֵּ֣ין עֲבֹתִ֔ים וְלֹֽא־ יַעַמְד֧וּ אֵלֵיהֶ֛ם בְּגָבְהָ֖ם כָּל־ שֹׁ֣תֵי מָ֑יִם כִּֽי־ כֻלָּם֩ נִתְּנ֨וּ לַמָּ֜וֶת אֶל־ אֶ֣רֶץ תַּחְתִּ֗ית בְּת֛וֹךְ בְּנֵ֥י אָדָ֖ם אֶל־ י֥וֹרְדֵי בֽוֹר׃ (מב) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה בְּי֨וֹם רִדְתּ֤וֹ שְׁא֙וֹלָה֙ הֶאֱבַ֜לְתִּי כִּסֵּ֤תִי עָלָיו֙ אֶת־ תְּה֔וֹם וָֽאֶמְנַע֙ נַהֲרוֹתֶ֔יהָ וַיִּכָּלְא֖וּ מַ֣יִם רַבִּ֑ים וָאַקְדִּ֤ר עָלָיו֙ לְבָנ֔וֹן וְכָל־ עֲצֵ֥י הַשָּׂדֶ֖ה עָלָ֥יו עֻלְפֶּֽה׃ (מג) מִקּ֤וֹל מַפַּלְתּוֹ֙ הִרְעַ֣שְׁתִּי גוֹיִ֔ם בְּהוֹרִדִ֥י אֹת֛וֹ שְׁא֖וֹלָה אֶת־ י֣וֹרְדֵי ב֑וֹר וַיִּנָּ֨חֲמ֜וּ בְּאֶ֤רֶץ תַּחְתִּית֙ כָּל־ עֲצֵי־ עֵ֔דֶן מִבְחַ֥ר וְטוֹב־ לְבָנ֖וֹן כָּל־ שֹׁ֥תֵי מָֽיִם׃ (מד) גַּם־ הֵ֗ם אִתּ֛וֹ יָרְד֥וּ שְׁא֖וֹלָה אֶל־ חַלְלֵי־ חָ֑רֶב וּזְרֹע֛וֹ יָשְׁב֥וּ בְצִלּ֖וֹ בְּת֥וֹךְ גּוֹיִֽם׃ (מה) אֶל־ מִ֨י דָמִ֥יתָ כָּ֛כָה בְּכָב֥וֹד וּבְגֹ֖דֶל בַּעֲצֵי־ עֵ֑דֶן וְהוּרַדְתָּ֨ אֶת־ עֲצֵי־ עֵ֜דֶן אֶל־ אֶ֣רֶץ תַּחְתִּ֗ית בְּת֨וֹךְ עֲרֵלִ֤ים תִּשְׁכַּב֙ אֶת־ חַלְלֵי־ חֶ֔רֶב ה֤וּא פַרְעֹה֙ וְכָל־ הֲמוֹנֹ֔ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃
