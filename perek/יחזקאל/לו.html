יחזקאל פרק לו לפי החלוקה לסדרים

פרק לו בספר יחזקאל (38 פסוקים) נמצא ביחזקאל סדר כא (פסוקים כב–מה) וביחזקאל סדר כב (פסוקים א–יד).

יחזקאל סדר כא

(א) וְנָתַתִּ֥י אוֹתָ֛ם וּסְבִיב֥וֹת גִּבְעָתִ֖י בְּרָכָ֑ה וְהוֹרַדְתִּ֤י הַגֶּ֙שֶׁם֙ בְּעִתּ֔וֹ גִּשְׁמֵ֥י בְרָכָ֖ה יִֽהְיֽוּ׃ (ב) וְנָתַן֩ עֵ֨ץ הַשָּׂדֶ֜ה אֶת־ פִּרְי֗וֹ וְהָאָ֙רֶץ֙ תִּתֵּ֣ן יְבוּלָ֔הּ וְהָי֥וּ עַל־ אַדְמָתָ֖ם לָבֶ֑טַח וְֽיָדְע֞וּ כִּי־ אֲנִ֣י יְהוָ֗ה בְּשִׁבְרִי֙ אֶת־ מֹט֣וֹת עֻלָּ֔ם וְהִ֨צַּלְתִּ֔ים מִיַּ֖ד הָעֹבְדִ֥ים בָּהֶֽם׃ (ג) וְלֹא־ יִהְי֨וּ ע֥וֹד בַּז֙ לַגּוֹיִ֔ם וְחַיַּ֥ת הָאָ֖רֶץ לֹ֣א תֹאכְלֵ֑ם וְיָשְׁב֥וּ לָבֶ֖טַח וְאֵ֥ין מַחֲרִֽיד׃ (ד) וַהֲקִמֹתִ֥י לָהֶ֛ם מַטָּ֖ע לְשֵׁ֑ם וְלֹֽא־ יִהְי֨וּ ע֜וֹד אֲסֻפֵ֤י רָעָב֙ בָּאָ֔רֶץ וְלֹֽא־ יִשְׂא֥וּ ע֖וֹד כְּלִמַּ֥ת הַגּוֹיִֽם׃ (ה) וְיָדְע֗וּ כִּ֣י אֲנִ֧י יְהוָ֛ה אֱלֹהֵיהֶ֖ם אִתָּ֑ם וְהֵ֗מָּה עַמִּי֙ בֵּ֣ית יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ו) וְאַתֵּ֥ן צֹאנִ֛י צֹ֥אן מַרְעִיתִ֖י אָדָ֣ם אַתֶּ֑ם אֲנִי֙ אֱלֹ֣הֵיכֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ח) בֶּן־ אָדָ֕ם שִׂ֥ים פָּנֶ֖יךָ עַל־ הַ֣ר שֵׂעִ֑יר וְהִנָּבֵ֖א עָלָֽיו׃ (ט) וְאָמַ֣רְתָּ לּ֗וֹ כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י אֵלֶ֖יךָ הַר־ שֵׂעִ֑יר וְנָטִ֤יתִי יָדִי֙ עָלֶ֔יךָ וּנְתַתִּ֖יךָ שְׁמָמָ֥ה וּמְשַׁמָּֽה׃ (י) עָרֶ֙יךָ֙ חָרְבָּ֣ה אָשִׂ֔ים וְאַתָּ֖ה שְׁמָמָ֣ה תִֽהְיֶ֑ה וְיָדַעְתָּ֖ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (יא) יַ֗עַן הֱי֤וֹת לְךָ֙ אֵיבַ֣ת עוֹלָ֔ם וַתַּגֵּ֥ר אֶת־ בְּנֵֽי־ יִשְׂרָאֵ֖ל עַל־ יְדֵי־ חָ֑רֶב בְּעֵ֣ת אֵידָ֔ם בְּעֵ֖ת עֲוֺ֥ן קֵֽץ׃ (יב) לָכֵ֣ן חַי־ אָ֗נִי נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה כִּֽי־ לְדָ֥ם אֶעֶשְׂךָ֖ וְדָ֣ם יִרְדֲּפֶ֑ךָ אִם־ לֹ֥א דָ֛ם שָׂנֵ֖אתָ וְדָ֥ם יִרְדֲּפֶֽךָ׃ (יג) וְנָֽתַתִּי֙ אֶת־ הַ֣ר שֵׂעִ֔יר לְשִֽׁמְמָ֖ה וּשְׁמָמָ֑ה וְהִכְרַתִּ֥י מִמֶּ֖נּוּ עֹבֵ֥ר וָשָֽׁב׃ (יד) וּמִלֵּאתִ֥י אֶת־ הָרָ֖יו חֲלָלָ֑יו גִּבְעוֹתֶ֤יךָ וְגֵאוֹתֶ֙יךָ֙ וְכָל־ אֲפִיקֶ֔יךָ חַלְלֵי־ חֶ֖רֶב יִפְּל֥וּ בָהֶֽם׃ (טו) שִֽׁמְמ֤וֹת עוֹלָם֙ אֶתֶּנְךָ֔ וְעָרֶ֖יךָ לֹ֣א (תישבנה) תָשֹׁ֑בְנָה וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (טז) יַ֣עַן אֲ֠מָרְךָ אֶת־ שְׁנֵ֨י הַגּוֹיִ֜ם וְאֶת־ שְׁתֵּ֧י הָאֲרָצ֛וֹת לִ֥י תִהְיֶ֖ינָה וִֽירַשְׁנ֑וּהָ וַֽיהוָ֖ה שָׁ֥ם הָיָֽה׃ (יז) לָכֵ֣ן חַי־ אָ֗נִי נְאֻם֮ אֲדֹנָ֣י יְהוִה֒ וְעָשִׂ֗יתִי כְּאַפְּךָ֙ וּכְקִנְאָ֣תְךָ֔ אֲשֶׁ֣ר עָשִׂ֔יתָה מִשִּׂנְאָתֶ֖יךָ בָּ֑ם וְנוֹדַ֥עְתִּי בָ֖ם כַּאֲשֶׁ֥ר אֶשְׁפְּטֶֽךָ׃ (יח) וְֽיָדַעְתָּ֮ כִּֽי־ אֲנִ֣י יְהוָה֒ שָׁמַ֣עְתִּי ׀ אֶת־ כָּל־ נָאָֽצוֹתֶ֗יךָ אֲשֶׁ֥ר אָמַ֛רְתָּ עַל־ הָרֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֣ר ׀ (שממה) שָׁמֵ֑מוּ לָ֥נוּ נִתְּנ֖וּ לְאָכְלָֽה׃ (יט) וַתַּגְדִּ֤ילוּ עָלַי֙ בְּפִיכֶ֔ם וְהַעְתַּרְתֶּ֥ם עָלַ֖י דִּבְרֵיכֶ֑ם אֲנִ֖י שָׁמָֽעְתִּי׃ (כ) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה כִּשְׂמֹ֙חַ֙ כָּל־ הָאָ֔רֶץ שְׁמָמָ֖ה אֶעֱשֶׂה־ לָּֽךְ׃ (כא) כְּשִׂמְחָ֨תְךָ֜ לְנַחְלַ֧ת בֵּֽית־ יִשְׂרָאֵ֛ל עַ֥ל אֲשֶׁר־ שָׁמֵ֖מָה כֵּ֣ן אֶעֱשֶׂה־ לָּ֑ךְ שְׁמָמָ֨ה תִֽהְיֶ֤ה הַר־ שֵׂעִיר֙ וְכָל־ אֱד֣וֹם כֻּלָּ֔הּ וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ תחילת פרק לו (כב) וְאַתָּ֣ה בֶן־ אָדָ֔ם הִנָּבֵ֖א אֶל־ הָרֵ֣י יִשְׂרָאֵ֑ל וְאָ֣מַרְתָּ֔ הָרֵי֙ יִשְׂרָאֵ֔ל שִׁמְע֖וּ דְּבַר־ יְהוָֽה׃ (כג) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֣עַן אָמַ֧ר הָאוֹיֵ֛ב עֲלֵיכֶ֖ם הֶאָ֑ח וּבָמ֣וֹת עוֹלָ֔ם לְמֽוֹרָשָׁ֖ה הָ֥יְתָה לָּֽנוּ׃ (כד) לָכֵן֙ הִנָּבֵ֣א וְאָמַרְתָּ֔ כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה יַ֣עַן בְּיַ֡עַן שַׁמּוֹת֩ וְשָׁאֹ֨ף אֶתְכֶ֜ם מִסָּבִ֗יב לִֽהְיוֹתְכֶ֤ם מֽוֹרָשָׁה֙ לִשְׁאֵרִ֣ית הַגּוֹיִ֔ם וַתֵּֽעֲל֛וּ עַל־ שְׂפַ֥ת לָשׁ֖וֹן וְדִבַּת־ עָֽם׃ (כה) לָכֵן֙ הָרֵ֣י יִשְׂרָאֵ֔ל שִׁמְע֖וּ דְּבַר־ אֲדֹנָ֣י יְהוִ֑ה כֹּֽה־ אָמַ֣ר אֲדֹנָ֣י יְ֠הוִה לֶהָרִ֨ים וְלַגְּבָע֜וֹת לָאֲפִיקִ֣ים וְלַגֵּאָי֗וֹת וְלֶחֳרָב֤וֹת הַשֹּֽׁמְמוֹת֙ וְלֶעָרִ֣ים הַנֶּעֱזָב֔וֹת אֲשֶׁ֨ר הָי֤וּ לְבַז֙ וּלְלַ֔עַג לִשְׁאֵרִ֥ית הַגּוֹיִ֖ם אֲשֶׁ֥ר מִסָּבִֽיב׃ (כו) לָכֵ֗ן כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ אִם־ לֹ֠א בְּאֵ֨שׁ קִנְאָתִ֥י דִבַּ֛רְתִּי עַל־ שְׁאֵרִ֥ית הַגּוֹיִ֖ם וְעַל־ אֱד֣וֹם כֻּלָּ֑א אֲשֶׁ֣ר נָתְנֽוּ־ אֶת־ אַרְצִ֣י ׀ לָ֠הֶם לְמ֨וֹרָשָׁ֜ה בְּשִׂמְחַ֤ת כָּל־ לֵבָב֙ בִּשְׁאָ֣ט נֶ֔פֶשׁ לְמַ֥עַן מִגְרָשָׁ֖הּ לָבַֽז׃ (כז) לָכֵ֕ן הִנָּבֵ֖א עַל־ אַדְמַ֣ת יִשְׂרָאֵ֑ל וְאָמַרְתָּ֡ לֶהָרִ֣ים וְ֠לַגְּבָעוֹת לָאֲפִיקִ֨ים וְלַגֵּאָי֜וֹת כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה הִנְנִ֨י בְקִנְאָתִ֤י וּבַחֲמָתִי֙ דִּבַּ֔רְתִּי יַ֛עַן כְּלִמַּ֥ת גּוֹיִ֖ם נְשָׂאתֶֽם׃ (כח) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה אֲנִ֖י נָשָׂ֣אתִי אֶת־ יָדִ֑י אִם־ לֹ֤א הַגּוֹיִם֙ אֲשֶׁ֣ר לָכֶ֣ם מִסָּבִ֔יב הֵ֖מָּה כְּלִמָּתָ֥ם יִשָּֽׂאוּ׃ (כט) וְאַתֶּ֞ם הָרֵ֤י יִשְׂרָאֵל֙ עַנְפְּכֶ֣ם תִּתֵּ֔נוּ וּפֶרְיְכֶ֥ם תִּשְׂא֖וּ לְעַמִּ֣י יִשְׂרָאֵ֑ל כִּ֥י קֵרְב֖וּ לָבֽוֹא׃ (ל) כִּ֖י הִנְנִ֣י אֲלֵיכֶ֑ם וּפָנִ֣יתִי אֲלֵיכֶ֔ם וְנֶעֱבַדְתֶּ֖ם וְנִזְרַעְתֶּֽם׃ (לא) וְהִרְבֵּיתִ֤י עֲלֵיכֶם֙ אָדָ֔ם כָּל־ בֵּ֥ית יִשְׂרָאֵ֖ל כֻּלֹּ֑ה וְנֹֽשְׁבוּ֙ הֶֽעָרִ֔ים וְהֶחֳרָב֖וֹת תִּבָּנֶֽינָה׃ (לב) וְהִרְבֵּיתִ֧י עֲלֵיכֶ֛ם אָדָ֥ם וּבְהֵמָ֖ה וְרָב֣וּ וּפָר֑וּ וְהוֹשַׁבְתִּ֨י אֶתְכֶ֜ם כְּקַדְמֽוֹתֵיכֶ֗ם וְהֵטִֽבֹתִי֙ מֵרִאשֹׁ֣תֵיכֶ֔ם וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (לג) וְהוֹלַכְתִּי֩ עֲלֵיכֶ֨ם אָדָ֜ם אֶת־ עַמִּ֤י יִשְׂרָאֵל֙ וִֽירֵשׁ֔וּךָ וְהָיִ֥יתָ לָהֶ֖ם לְנַחֲלָ֑ה וְלֹא־ תוֹסִ֥ף ע֖וֹד לְשַׁכְּלָֽם׃ (לד) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֚עַן אֹמְרִ֣ים לָכֶ֔ם אֹכֶ֥לֶת אָדָ֖ם (אתי) אָ֑תְּ וּמְשַׁכֶּ֥לֶת (גויך) גּוֹיַ֖יִךְ הָיִֽית׃ (לה) לָכֵ֗ן אָדָם֙ לֹא־ תֹ֣אכְלִי ע֔וֹד (וגויך) וְגוֹיַ֖יִךְ לֹ֣א (תכשלי)־ תְשַׁכְּלִי־ ע֑וֹד נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לו) וְלֹא־ אַשְׁמִ֨יעַ אֵלַ֤יִךְ עוֹד֙ כְּלִמַּ֣ת הַגּוֹיִ֔ם וְחֶרְפַּ֥ת עַמִּ֖ים לֹ֣א תִשְׂאִי־ ע֑וֹד (וגויך) וְגוֹיַ֙יִךְ֙ לֹא־ תַכְשִׁ֣לִי ע֔וֹד נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לח) בֶּן־ אָדָ֗ם בֵּ֤ית יִשְׂרָאֵל֙ יֹשְׁבִ֣ים עַל־ אַדְמָתָ֔ם וַיְטַמְּא֣וּ אוֹתָ֔הּ בְּדַרְכָּ֖ם וּבַעֲלִֽילוֹתָ֑ם כְּטֻמְאַת֙ הַנִּדָּ֔ה הָיְתָ֥ה דַרְכָּ֖ם לְפָנָֽי׃ (לט) וָאֶשְׁפֹּ֤ךְ חֲמָתִי֙ עֲלֵיהֶ֔ם עַל־ הַדָּ֖ם אֲשֶׁר־ שָׁפְכ֣וּ עַל־ הָאָ֑רֶץ וּבְגִלּוּלֵיהֶ֖ם טִמְּאֽוּהָ׃ (מ) וָאָפִ֤יץ אֹתָם֙ בַּגּוֹיִ֔ם וַיִּזָּר֖וּ בָּאֲרָצ֑וֹת כְּדַרְכָּ֥ם וְכַעֲלִילוֹתָ֖ם שְׁפַטְתִּֽים׃ (מא) וַיָּב֗וֹא אֶל־ הַגּוֹיִם֙ אֲשֶׁר־ בָּ֣אוּ שָׁ֔ם וַֽיְחַלְּל֖וּ אֶת־ שֵׁ֣ם קָדְשִׁ֑י בֶּאֱמֹ֤ר יִשְׂרָאֵ֗ל כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה לֹ֧א לְמַעַנְכֶ֛ם אֲנִ֥י עֹשֶׂ֖ה בֵּ֣ית יִשְׂרָאֵ֑ל כִּ֤י אִם־ לְשֵׁם־ קָדְשִׁי֙ אֲשֶׁ֣ר חִלַּלְתֶּ֔ם בַּגּוֹיִ֖ם אֲשֶׁר־ בָּ֥אתֶם שָֽׁם׃ (מד) וְקִדַּשְׁתִּ֞י (א) וְזָרַקְתִּ֧י עֲלֵיכֶ֛ם מַ֥יִם טְהוֹרִ֖ים וּטְהַרְתֶּ֑ם מִכֹּ֧ל טֻמְאוֹתֵיכֶ֛ם וּמִכָּל־ גִּלּ֥וּלֵיכֶ֖ם אֲטַהֵ֥ר אֶתְכֶֽם׃ (ב) וְנָתַתִּ֤י לָכֶם֙ לֵ֣ב חָדָ֔שׁ וְר֥וּחַ חֲדָשָׁ֖ה אֶתֵּ֣ן בְּקִרְבְּכֶ֑ם וַהֲסִ֨רֹתִ֜י אֶת־ לֵ֤ב הָאֶ֙בֶן֙ מִבְּשַׂרְכֶ֔ם וְנָתַתִּ֥י לָכֶ֖ם לֵ֥ב בָּשָֽׂר׃ (ג) וְאֶת־ רוּחִ֖י אֶתֵּ֣ן בְּקִרְבְּכֶ֑ם וְעָשִׂ֗יתִי אֵ֤ת אֲשֶׁר־ בְּחֻקַּי֙ תֵּלֵ֔כוּ וּמִשְׁפָּטַ֥י תִּשְׁמְר֖וּ וַעֲשִׂיתֶֽם׃ (ד) וִישַׁבְתֶּ֣ם בָּאָ֔רֶץ אֲשֶׁ֥ר נָתַ֖תִּי לַאֲבֹֽתֵיכֶ֑ם וִהְיִ֤יתֶם לִי֙ לְעָ֔ם וְאָ֣נֹכִ֔י אֶהְיֶ֥ה לָכֶ֖ם לֵאלֹהִֽים׃ (ה) וְהוֹשַׁעְתִּ֣י אֶתְכֶ֔ם מִכֹּ֖ל טֻמְאֽוֹתֵיכֶ֑ם וְקָרָ֤אתִי אֶל־ הַדָּגָן֙ וְהִרְבֵּיתִ֣י אֹת֔וֹ וְלֹא־ אֶתֵּ֥ן עֲלֵיכֶ֖ם רָעָֽב׃ (ו) וְהִרְבֵּיתִי֙ אֶת־ פְּרִ֣י הָעֵ֔ץ וּתְנוּבַ֖ת הַשָּׂדֶ֑ה לְמַ֗עַן אֲ֠שֶׁר לֹ֣א תִקְח֥וּ ע֛וֹד חֶרְפַּ֥ת רָעָ֖ב בַּגּוֹיִֽם׃ (ז) וּזְכַרְתֶּם֙ אֶת־ דַּרְכֵיכֶ֣ם הָרָעִ֔ים וּמַעַלְלֵיכֶ֖ם אֲשֶׁ֣ר לֹֽא־ טוֹבִ֑ים וּנְקֹֽטֹתֶם֙ בִּפְנֵיכֶ֔ם עַ֚ל עֲוֺנֹ֣תֵיכֶ֔ם וְעַ֖ל תּוֹעֲבֽוֹתֵיכֶֽם׃ (ח) לֹ֧א לְמַעַנְכֶ֣ם אֲנִֽי־ עֹשֶׂ֗ה נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה יִוָּדַ֖ע לָכֶ֑ם בּ֧וֹשׁוּ וְהִכָּלְמ֛וּ מִדַּרְכֵיכֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל׃ (ט) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה בְּיוֹם֙ טַהֲרִ֣י אֶתְכֶ֔ם מִכֹּ֖ל עֲוֺנֽוֹתֵיכֶ֑ם וְהֽוֹשַׁבְתִּי֙ אֶת־ הֶ֣עָרִ֔ים וְנִבְנ֖וּ הֶחֳרָבֽוֹת׃ (י) וְהָאָ֥רֶץ הַנְּשַׁמָּ֖ה תֵּֽעָבֵ֑ד תַּ֚חַת אֲשֶׁ֣ר הָיְתָ֣ה שְׁמָמָ֔ה לְעֵינֵ֖י כָּל־ עוֹבֵֽר׃ (יא) וְאָמְר֗וּ הָאָ֤רֶץ הַלֵּ֙זוּ֙ הַנְּשַׁמָּ֔ה הָיְתָ֖ה כְּגַן־ עֵ֑דֶן וְהֶעָרִ֧ים הֶחֳרֵב֛וֹת וְהַֽנְשַׁמּ֥וֹת וְהַנֶּהֱרָס֖וֹת בְּצוּר֥וֹת יָשָֽׁבוּ׃ (יב) וְיָדְע֣וּ הַגּוֹיִ֗ם אֲשֶׁ֣ר יִֽשָּׁאֲרוּ֮ לְיוֹסֵף֙ עֵ֣ץ אֶפְרַ֔יִם וְכָל־ בֵּ֥ית יִשְׂרָאֵ֖ל (חברו) חֲבֵרָֽיו׃ (לא) וְקָרַ֨ב אֹתָ֜ם אֶחָ֧ד אֶל־ אֶחָ֛ד לְךָ֖ לְעֵ֣ץ אֶחָ֑ד וְהָי֥וּ לַאֲחָדִ֖ים בְּיָדֶֽךָ׃ (לב) וְכַֽאֲשֶׁר֙ יֹאמְר֣וּ אֵלֶ֔יךָ בְּנֵ֥י עַמְּךָ֖ לֵאמֹ֑ר הֲלֽוֹא־ תַגִּ֥יד לָ֖נוּ מָה־ אֵ֥לֶּה לָּֽךְ׃ (לג) דַּבֵּ֣ר אֲלֵהֶ֗ם כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנֵּה֩ אֲנִ֨י לֹקֵ֜חַ אֶת־ עֵ֤ץ יוֹסֵף֙ אֲשֶׁ֣ר בְּיַד־ אֶפְרַ֔יִם וְשִׁבְטֵ֥י יִשְׂרָאֵ֖ל (חברו) חֲבֵרָ֑יו וְנָתַתִּי֩ אוֹתָ֨ם עָלָ֜יו אֶת־ עֵ֣ץ יְהוּדָ֗ה וַֽעֲשִׂיתִם֙ לְעֵ֣ץ אֶחָ֔ד וְהָי֥וּ אֶחָ֖ד בְּיָדִֽי׃ (לד) וְהָי֨וּ הָעֵצִ֜ים אֲ‍ֽשֶׁר־ תִּכְתֹּ֧ב עֲלֵיהֶ֛ם בְּיָדְךָ֖ לְעֵינֵיהֶֽם׃ (לה) וְדַבֵּ֣ר אֲלֵיהֶ֗ם כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנֵּ֨ה אֲנִ֤י לֹקֵ֙חַ֙ אֶת־ בְּנֵ֣י יִשְׂרָאֵ֔ל מִבֵּ֥ין הַגּוֹיִ֖ם אֲשֶׁ֣ר הָֽלְכוּ־ שָׁ֑ם וְקִבַּצְתִּ֤י אֹתָם֙ מִסָּבִ֔יב וְהֵבֵאתִ֥י אוֹתָ֖ם אֶל־ אַדְמָתָֽם׃ (לו) וְעָשִׂ֣יתִי אֹ֠תָם לְג֨וֹי אֶחָ֤ד בָּאָ֙רֶץ֙ בְּהָרֵ֣י יִשְׂרָאֵ֔ל וּמֶ֧לֶךְ אֶחָ֛ד יִֽהְיֶ֥ה לְכֻלָּ֖ם לְמֶ֑לֶךְ וְלֹ֤א (יהיה)־ יִֽהְיוּ־ עוֹד֙ לִשְׁנֵ֣י גוֹיִ֔ם וְלֹ֨א יֵחָ֥צוּ ע֛וֹד לִשְׁתֵּ֥י מַמְלָכ֖וֹת עֽוֹד׃ (לז) וְלֹ֧א יִֽטַמְּא֣וּ ע֗וֹד בְּגִלּֽוּלֵיהֶם֙ וּבְשִׁקּ֣וּצֵיהֶ֔ם וּבְכֹ֖ל פִּשְׁעֵיהֶ֑ם וְהוֹשַׁעְתִּ֣י אֹתָ֗ם מִכֹּ֤ל מוֹשְׁבֹֽתֵיהֶם֙ אֲשֶׁ֣ר חָטְא֣וּ בָהֶ֔ם וְטִהַרְתִּ֤י אוֹתָם֙ וְהָיוּ־ לִ֣י לְעָ֔ם וַאֲנִ֕י אֶהְיֶ֥ה לָהֶ֖ם לֵאלֹהִֽים׃ (לח) וְעַבְדִּ֤י דָוִד֙ מֶ֣לֶךְ עֲלֵיהֶ֔ם וְרוֹעֶ֥ה אֶחָ֖ד יִהְיֶ֣ה לְכֻלָּ֑ם וּבְמִשְׁפָּטַ֣י יֵלֵ֔כוּ וְחֻקֹּתַ֥י יִשְׁמְר֖וּ וְעָשׂ֥וּ אוֹתָֽם׃ (לט) וְיָשְׁב֣וּ עַל־ הָאָ֗רֶץ אֲשֶׁ֤ר נָתַ֙תִּי֙ לְעַבְדִּ֣י לְיַֽעֲקֹ֔ב אֲשֶׁ֥ר יָֽשְׁבוּ־ בָ֖הּ אֲבֽוֹתֵיכֶ֑ם וְיָשְׁב֣וּ עָלֶ֡יהָ הֵ֠מָּה וּבְנֵיהֶ֞ם וּבְנֵ֤י בְנֵיהֶם֙ עַד־ עוֹלָ֔ם וְדָוִ֣ד עַבְדִּ֔י נָשִׂ֥יא לָהֶ֖ם לְעוֹלָֽם׃ (מ) וְכָרַתִּ֤י לָהֶם֙ בְּרִ֣ית שָׁל֔וֹם בְּרִ֥ית עוֹלָ֖ם יִהְיֶ֣ה אוֹתָ֑ם וּנְתַתִּים֙ וְהִרְבֵּיתִ֣י אוֹתָ֔ם וְנָתַתִּ֧י אֶת־ מִקְדָּשִׁ֛י בְּתוֹכָ֖ם לְעוֹלָֽם׃ (מא) וְהָיָ֤ה מִשְׁכָּנִי֙ עֲלֵיהֶ֔ם וְהָיִ֥יתִי לָהֶ֖ם לֵֽאלֹהִ֑ים וְהֵ֖מָּה יִֽהְיוּ־ לִ֥י לְעָֽם׃
