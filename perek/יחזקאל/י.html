יחזקאל פרק י לפי החלוקה לסדרים

פרק י בספר יחזקאל (22 פסוקים) נמצא ביחזקאל סדר ה (פסוקים א–כב).

יחזקאל סדר ה

תחילת פרק י (א) וָאֶרְאֶ֗ה וְהִנֵּ֤ה אֶל־ הָרָקִ֙יעַ֙ אֲשֶׁר֙ עַל־ רֹ֣אשׁ הַכְּרֻבִ֔ים כְּאֶ֣בֶן סַפִּ֔יר כְּמַרְאֵ֖ה דְּמ֣וּת כִּסֵּ֑א נִרְאָ֖ה עֲלֵיהֶֽם׃ (ב) וַיֹּ֜אמֶר אֶל־ הָאִ֣ישׁ ׀ לְבֻ֣שׁ הַבַּדִּ֗ים וַיֹּ֡אמֶר בֹּא֩ אֶל־ בֵּינ֨וֹת לַגַּלְגַּ֜ל אֶל־ תַּ֣חַת לַכְּר֗וּב וּמַלֵּ֨א חָפְנֶ֤יךָ גַֽחֲלֵי־ אֵשׁ֙ מִבֵּינ֣וֹת לַכְּרֻבִ֔ים וּזְרֹ֖ק עַל־ הָעִ֑יר וַיָּבֹ֖א לְעֵינָֽי׃ (ג) וְהַכְּרֻבִ֗ים עֹֽמְדִ֛ים מִימִ֥ין לַבַּ֖יִת בְּבֹא֣וֹ הָאִ֑ישׁ וְהֶעָנָ֣ן מָלֵ֔א אֶת־ הֶחָצֵ֖ר הַפְּנִימִֽית׃ (ד) וַיָּ֤רָם כְּבוֹד־ יְהוָה֙ מֵעַ֣ל הַכְּר֔וּב עַ֖ל מִפְתַּ֣ן הַבָּ֑יִת וַיִּמָּלֵ֤א הַבַּ֙יִת֙ אֶת־ הֶ֣עָנָ֔ן וְהֶֽחָצֵר֙ מָֽלְאָ֔ה אֶת־ נֹ֖גַהּ כְּב֥וֹד יְהוָֽה׃ (ה) וְקוֹל֙ כַּנְפֵ֣י הַכְּרוּבִ֔ים נִשְׁמַ֕ע עַד־ הֶחָצֵ֖ר הַחִיצֹנָ֑ה כְּק֥וֹל אֵל־ שַׁדַּ֖י בְּדַבְּרֽוֹ׃ (ו) וַיְהִ֗י בְּצַוֺּתוֹ֙ אֶת־ הָאִ֤ישׁ לְבֻֽשׁ־ הַבַּדִּים֙ לֵאמֹ֔ר קַ֥ח אֵשׁ֙ מִבֵּינ֣וֹת לַגַּלְגַּ֔ל מִבֵּינ֖וֹת לַכְּרוּבִ֑ים וַיָּבֹא֙ וַֽיַּעֲמֹ֔ד אֵ֖צֶל הָאוֹפָֽן׃ (ז) וַיִּשְׁלַח֩ הַכְּר֨וּב אֶת־ יָד֜וֹ מִבֵּינ֣וֹת לַכְּרוּבִ֗ים אֶל־ הָאֵשׁ֙ אֲשֶׁר֙ בֵּינ֣וֹת הַכְּרֻבִ֔ים וַיִּשָּׂא֙ וַיִּתֵּ֔ן אֶל־ חָפְנֵ֖י לְבֻ֣שׁ הַבַּדִּ֑ים וַיִּקַּ֖ח וַיֵּצֵֽא׃ (ח) וַיֵּרָ֖א לַכְּרֻבִ֑ים תַּבְנִית֙ יַד־ אָדָ֔ם תַּ֖חַת כַּנְפֵיהֶֽם׃ (ט) וָאֶרְאֶ֗ה וְהִנֵּ֨ה אַרְבָּעָ֣ה אוֹפַנִּים֮ אֵ֣צֶל הַכְּרוּבִים֒ אוֹפַ֣ן אֶחָ֗ד אֵ֚צֶל הַכְּר֣וּב אֶחָ֔ד וְאוֹפַ֣ן אֶחָ֔ד אֵ֖צֶל הַכְּר֣וּב אֶחָ֑ד וּמַרְאֵה֙ הָא֣וֹפַנִּ֔ים כְּעֵ֖ין אֶ֥בֶן תַּרְשִֽׁישׁ׃ (י) וּמַ֨רְאֵיהֶ֔ם דְּמ֥וּת אֶחָ֖ד לְאַרְבַּעְתָּ֑ם כַּאֲשֶׁ֛ר יִהְיֶ֥ה הָאוֹפַ֖ן בְּת֥וֹךְ הָאוֹפָֽן׃ (יא) בְּלֶכְתָּ֗ם אֶל־ אַרְבַּ֤עַת רִבְעֵיהֶם֙ יֵלֵ֔כוּ לֹ֥א יִסַּ֖בּוּ בְּלֶכְתָּ֑ם כִּ֣י הַמָּק֞וֹם אֲשֶׁר־ יִפְנֶ֤ה הָרֹאשׁ֙ אַחֲרָ֣יו יֵלֵ֔כוּ לֹ֥א יִסַּ֖בּוּ בְּלֶכְתָּֽם׃ (יב) וְכָל־ בְּשָׂרָם֙ וְגַבֵּהֶ֔ם וִֽידֵיהֶ֖ם וְכַנְפֵיהֶ֑ם וְהָאֽוֹפַנִּ֗ים מְלֵאִ֤ים עֵינַ֙יִם֙ סָבִ֔יב לְאַרְבַּעְתָּ֖ם אוֹפַנֵּיהֶֽם׃ (יג) לָא֖וֹפַנִּ֑ים לָהֶ֛ם קוֹרָ֥א הַגַּלְגַּ֖ל בְּאָזְנָֽי׃ (יד) וְאַרְבָּעָ֥ה פָנִ֖ים לְאֶחָ֑ד פְּנֵ֨י הָאֶחָ֜ד פְּנֵ֣י הַכְּר֗וּב וּפְנֵ֤י הַשֵּׁנִי֙ פְּנֵ֣י אָדָ֔ם וְהַשְּׁלִישִׁי֙ פְּנֵ֣י אַרְיֵ֔ה וְהָרְבִיעִ֖י פְּנֵי־ נָֽשֶׁר׃ (טו) וַיֵּרֹ֖מּוּ הַכְּרוּבִ֑ים הִ֣יא הַחַיָּ֔ה אֲשֶׁ֥ר רָאִ֖יתִי בִּֽנְהַר־ כְּבָֽר׃ (טז) וּבְלֶ֙כֶת֙ הַכְּרוּבִ֔ים יֵלְכ֥וּ הָאוֹפַנִּ֖ים אֶצְלָ֑ם וּבִשְׂאֵ֨ת הַכְּרוּבִ֜ים אֶת־ כַּנְפֵיהֶ֗ם לָרוּם֙ מֵעַ֣ל הָאָ֔רֶץ לֹא־ יִסַּ֧בּוּ הָאוֹפַנִּ֛ים גַּם־ הֵ֖ם מֵאֶצְלָֽם׃ (יז) בְּעָמְדָ֣ם יַעֲמֹ֔דוּ וּבְרוֹמָ֖ם יֵר֣וֹמּוּ אוֹתָ֑ם כִּ֛י ר֥וּחַ הַחַיָּ֖ה בָּהֶֽם׃ (יח) וַיֵּצֵא֙ כְּב֣וֹד יְהוָ֔ה מֵעַ֖ל מִפְתַּ֣ן הַבָּ֑יִת וַֽיַּעֲמֹ֖ד עַל־ הַכְּרוּבִֽים׃ (יט) וַיִּשְׂא֣וּ הַכְּרוּבִ֣ים אֶת־ כַּ֠נְפֵיהֶם וַיֵּר֨וֹמּוּ מִן־ הָאָ֤רֶץ לְעֵינַי֙ בְּצֵאתָ֔ם וְהָאֽוֹפַנִּ֖ים לְעֻמָּתָ֑ם וַֽיַּעֲמֹ֗ד פֶּ֣תַח שַׁ֤עַר בֵּית־ יְהוָה֙ הַקַּדְמוֹנִ֔י וּכְב֧וֹד אֱלֹהֵֽי־ יִשְׂרָאֵ֛ל עֲלֵיהֶ֖ם מִלְמָֽעְלָה׃ (כ) הִ֣יא הַחַיָּ֗ה אֲשֶׁ֥ר רָאִ֛יתִי תַּ֥חַת אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בִּֽנְהַר־ כְּבָ֑ר וָאֵדַ֕ע כִּ֥י כְרוּבִ֖ים הֵֽמָּה׃ (כא) אַרְבָּעָ֨ה אַרְבָּעָ֤ה פָנִים֙ לְאֶחָ֔ד וְאַרְבַּ֥ע כְּנָפַ֖יִם לְאֶחָ֑ד וּדְמוּת֙ יְדֵ֣י אָדָ֔ם תַּ֖חַת כַּנְפֵיהֶֽם׃ (כב) וּדְמ֣וּת פְּנֵיהֶ֔ם הֵ֣מָּה הַפָּנִ֗ים אֲשֶׁ֤ר רָאִ֙יתִי֙ עַל־ נְהַר־ כְּבָ֔ר מַרְאֵיהֶ֖ם וְאוֹתָ֑ם אִ֛ישׁ אֶל־ עֵ֥בֶר פָּנָ֖יו יֵלֵֽכוּ׃ סוף פרק י (כג) וַתִּשָּׂ֨א אֹתִ֜י ר֗וּחַ וַתָּבֵ֣א אֹ֠תִי אֶל־ שַׁ֨עַר בֵּית־ יְהוָ֤ה הַקַּדְמוֹנִי֙ הַפּוֹנֶ֣ה קָדִ֔ימָה וְהִנֵּה֙ בְּפֶ֣תַח הַשַּׁ֔עַר עֶשְׂרִ֥ים וַחֲמִשָּׁ֖ה אִ֑ישׁ וָאֶרְאֶ֨ה בְתוֹכָ֜ם אֶת־ יַאֲזַנְיָ֧ה בֶן־ עַזֻּ֛ר וְאֶת־ פְּלַטְיָ֥הוּ בֶן־ בְּנָיָ֖הוּ שָׂרֵ֖י הָעָֽם׃ (כד) וַיֹּ֖אמֶר אֵלָ֑י בֶּן־ אָדָ֕ם אֵ֣לֶּה הָאֲנָשִׁ֞ים הַחֹשְׁבִ֥ים אָ֛וֶן וְהַיֹּעֲצִ֥ים עֲצַת־ רָ֖ע בָּעִ֥יר הַזֹּֽאת׃ (כה) הָאֹ֣מְרִ֔ים לֹ֥א בְקָר֖וֹב בְּנ֣וֹת בָּתִּ֑ים הִ֣יא הַסִּ֔יר וַאֲנַ֖חְנוּ הַבָּשָֽׂר׃ (כו) לָכֵ֖ן הִנָּבֵ֣א עֲלֵיהֶ֑ם הִנָּבֵ֖א בֶּן־ אָדָֽם׃ (כז) וַתִּפֹּ֣ל עָלַי֮ ר֣וּחַ יְהוָה֒ וַיֹּ֣אמֶר אֵלַ֗י אֱמֹר֙ כֹּה־ אָמַ֣ר יְהוָ֔ה כֵּ֥ן אֲמַרְתֶּ֖ם בֵּ֣ית יִשְׂרָאֵ֑ל וּמַעֲל֥וֹת רֽוּחֲכֶ֖ם אֲנִ֥י יְדַעְתִּֽיהָ׃ (כח) הִרְבֵּיתֶ֥ם חַלְלֵיכֶ֖ם בָּעִ֣יר הַזֹּ֑את וּמִלֵּאתֶ֥ם חוּצֹתֶ֖יהָ חָלָֽל׃ (כט) לָכֵ֗ן כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ חַלְלֵיכֶם֙ אֲשֶׁ֣ר שַׂמְתֶּ֣ם בְּתוֹכָ֔הּ הֵ֥מָּה הַבָּשָׂ֖ר וְהִ֣יא הַסִּ֑יר וְאֶתְכֶ֖ם הוֹצִ֥יא מִתּוֹכָֽהּ׃ (ל) חֶ֖רֶב יְרֵאתֶ֑ם וְחֶ֙רֶב֙ אָבִ֣יא עֲלֵיכֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לא) וְהוֹצֵאתִ֤י אֶתְכֶם֙ מִתּוֹכָ֔הּ וְנָתַתִּ֥י אֶתְכֶ֖ם בְּיַד־ זָרִ֑ים וְעָשִׂ֛יתִי בָכֶ֖ם שְׁפָטִֽים׃ (לב) בַּחֶ֣רֶב תִּפֹּ֔לוּ עַל־ גְּב֥וּל יִשְׂרָאֵ֖ל אֶשְׁפּ֣וֹט אֶתְכֶ֑ם וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (לג) הִ֗יא לֹֽא־ תִהְיֶ֤ה לָכֶם֙ לְסִ֔יר וְאַתֶּ֛ם תִּהְי֥וּ בְתוֹכָ֖הּ לְבָשָׂ֑ר אֶל־ גְּב֥וּל יִשְׂרָאֵ֖ל אֶשְׁפֹּ֥ט אֶתְכֶֽם׃ (לד) וִֽידַעְתֶּם֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה אֲשֶׁ֤ר בְּחֻקַּי֙ לֹ֣א הֲלַכְתֶּ֔ם וּמִשְׁפָּטַ֖י לֹ֣א עֲשִׂיתֶ֑ם וּֽכְמִשְׁפְּטֵ֧י הַגּוֹיִ֛ם אֲשֶׁ֥ר סְבִיבוֹתֵיכֶ֖ם עֲשִׂיתֶֽם׃ (לה) וַֽיְהִי֙ כְּהִנָּ֣בְאִ֔י וּפְלַטְיָ֥הוּ בֶן־ בְּנָיָ֖ה מֵ֑ת וָאֶפֹּ֨ל עַל־ פָּנַ֜י וָאֶזְעַ֣ק קוֹל־ גָּד֗וֹל וָאֹמַר֙ אֲהָהּ֙ אֲדֹנָ֣י יְהוִ֔ה כָּלָה֙ אַתָּ֣ה עֹשֶׂ֔ה אֵ֖ת שְׁאֵרִ֥ית יִשְׂרָאֵֽל׃ (לו) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לז) בֶּן־ אָדָ֗ם אַחֶ֤יךָ אַחֶ֙יךָ֙ אַנְשֵׁ֣י גְאֻלָּתֶ֔ךָ וְכָל־ בֵּ֥ית יִשְׂרָאֵ֖ל כֻּלֹּ֑ה אֲשֶׁר֩ אָמְר֨וּ לָהֶ֜ם יֹשְׁבֵ֣י יְרוּשָׁלִַ֗ם רַֽחֲקוּ֙ מֵעַ֣ל יְהוָ֔ה לָ֥נוּ הִ֛יא נִתְּנָ֥ה הָאָ֖רֶץ לְמוֹרָשָֽׁה׃ (לח) לָכֵ֣ן אֱמֹ֗ר כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ כִּ֤י הִרְחַקְתִּים֙ בַּגּוֹיִ֔ם וְכִ֥י הֲפִֽיצוֹתִ֖ים בָּאֲרָצ֑וֹת וָאֱהִ֤י לָהֶם֙ לְמִקְדָּ֣שׁ מְעַ֔ט בָּאֲרָצ֖וֹת אֲשֶׁר־ בָּ֥אוּ שָֽׁם׃ (לט) לָכֵ֣ן אֱמֹ֗ר כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ וְקִבַּצְתִּ֤י אֶתְכֶם֙ מִן־ הָ֣עַמִּ֔ים וְאָסַפְתִּ֣י אֶתְכֶ֔ם מִן־ הָ֣אֲרָצ֔וֹת אֲשֶׁ֥ר נְפֹצוֹתֶ֖ם בָּהֶ֑ם וְנָתַתִּ֥י לָכֶ֖ם אֶת־ אַדְמַ֥ת יִשְׂרָאֵֽל׃ (מ) וּבָ֖אוּ־ שָׁ֑מָּה וְהֵסִ֜ירוּ אֶת־ כָּל־ שִׁקּוּצֶ֛יהָ וְאֶת־ כָּל־ תּוֹעֲבוֹתֶ֖יהָ מִמֶּֽנָּה׃ (מא) וְנָתַתִּ֤י לָהֶם֙ לֵ֣ב אֶחָ֔ד וְר֥וּחַ חֲדָשָׁ֖ה אֶתֵּ֣ן בְּקִרְבְּכֶ֑ם וַהֲסִ֨רֹתִ֜י לֵ֤ב הָאֶ֙בֶן֙ מִבְּשָׂרָ֔ם וְנָתַתִּ֥י לָהֶ֖ם לֵ֥ב בָּשָֽׂר׃
