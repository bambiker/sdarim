יחזקאל פרק לז לפי החלוקה לסדרים

פרק לז בספר יחזקאל (28 פסוקים) נמצא ביחזקאל סדר כב (פסוקים טו–מא) וביחזקאל סדר כג (פסוקים א–א).

יחזקאל סדר כב

(א) וְזָרַקְתִּ֧י עֲלֵיכֶ֛ם מַ֥יִם טְהוֹרִ֖ים וּטְהַרְתֶּ֑ם מִכֹּ֧ל טֻמְאוֹתֵיכֶ֛ם וּמִכָּל־ גִּלּ֥וּלֵיכֶ֖ם אֲטַהֵ֥ר אֶתְכֶֽם׃ (ב) וְנָתַתִּ֤י לָכֶם֙ לֵ֣ב חָדָ֔שׁ וְר֥וּחַ חֲדָשָׁ֖ה אֶתֵּ֣ן בְּקִרְבְּכֶ֑ם וַהֲסִ֨רֹתִ֜י אֶת־ לֵ֤ב הָאֶ֙בֶן֙ מִבְּשַׂרְכֶ֔ם וְנָתַתִּ֥י לָכֶ֖ם לֵ֥ב בָּשָֽׂר׃ (ג) וְאֶת־ רוּחִ֖י אֶתֵּ֣ן בְּקִרְבְּכֶ֑ם וְעָשִׂ֗יתִי אֵ֤ת אֲשֶׁר־ בְּחֻקַּי֙ תֵּלֵ֔כוּ וּמִשְׁפָּטַ֥י תִּשְׁמְר֖וּ וַעֲשִׂיתֶֽם׃ (ד) וִישַׁבְתֶּ֣ם בָּאָ֔רֶץ אֲשֶׁ֥ר נָתַ֖תִּי לַאֲבֹֽתֵיכֶ֑ם וִהְיִ֤יתֶם לִי֙ לְעָ֔ם וְאָ֣נֹכִ֔י אֶהְיֶ֥ה לָכֶ֖ם לֵאלֹהִֽים׃ (ה) וְהוֹשַׁעְתִּ֣י אֶתְכֶ֔ם מִכֹּ֖ל טֻמְאֽוֹתֵיכֶ֑ם וְקָרָ֤אתִי אֶל־ הַדָּגָן֙ וְהִרְבֵּיתִ֣י אֹת֔וֹ וְלֹא־ אֶתֵּ֥ן עֲלֵיכֶ֖ם רָעָֽב׃ (ו) וְהִרְבֵּיתִי֙ אֶת־ פְּרִ֣י הָעֵ֔ץ וּתְנוּבַ֖ת הַשָּׂדֶ֑ה לְמַ֗עַן אֲ֠שֶׁר לֹ֣א תִקְח֥וּ ע֛וֹד חֶרְפַּ֥ת רָעָ֖ב בַּגּוֹיִֽם׃ (ז) וּזְכַרְתֶּם֙ אֶת־ דַּרְכֵיכֶ֣ם הָרָעִ֔ים וּמַעַלְלֵיכֶ֖ם אֲשֶׁ֣ר לֹֽא־ טוֹבִ֑ים וּנְקֹֽטֹתֶם֙ בִּפְנֵיכֶ֔ם עַ֚ל עֲוֺנֹ֣תֵיכֶ֔ם וְעַ֖ל תּוֹעֲבֽוֹתֵיכֶֽם׃ (ח) לֹ֧א לְמַעַנְכֶ֣ם אֲנִֽי־ עֹשֶׂ֗ה נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה יִוָּדַ֖ע לָכֶ֑ם בּ֧וֹשׁוּ וְהִכָּלְמ֛וּ מִדַּרְכֵיכֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל׃ (ט) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה בְּיוֹם֙ טַהֲרִ֣י אֶתְכֶ֔ם מִכֹּ֖ל עֲוֺנֽוֹתֵיכֶ֑ם וְהֽוֹשַׁבְתִּי֙ אֶת־ הֶ֣עָרִ֔ים וְנִבְנ֖וּ הֶחֳרָבֽוֹת׃ (י) וְהָאָ֥רֶץ הַנְּשַׁמָּ֖ה תֵּֽעָבֵ֑ד תַּ֚חַת אֲשֶׁ֣ר הָיְתָ֣ה שְׁמָמָ֔ה לְעֵינֵ֖י כָּל־ עוֹבֵֽר׃ (יא) וְאָמְר֗וּ הָאָ֤רֶץ הַלֵּ֙זוּ֙ הַנְּשַׁמָּ֔ה הָיְתָ֖ה כְּגַן־ עֵ֑דֶן וְהֶעָרִ֧ים הֶחֳרֵב֛וֹת וְהַֽנְשַׁמּ֥וֹת וְהַנֶּהֱרָס֖וֹת בְּצוּר֥וֹת יָשָֽׁבוּ׃ (יב) וְיָדְע֣וּ הַגּוֹיִ֗ם אֲשֶׁ֣ר יִֽשָּׁאֲרוּ֮ סְבִיבוֹתֵיכֶם֒ כִּ֣י ׀ אֲנִ֣י יְהוָ֗ה בָּנִ֙יתִי֙ הַנֶּ֣הֱרָס֔וֹת נָטַ֖עְתִּי הַנְּשַׁמָּ֑ה אֲנִ֥י יְהוָ֖ה דִּבַּ֥רְתִּי וְעָשִֽׂיתִי׃ (יג) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה ע֗וֹד זֹ֛את אִדָּרֵ֥שׁ לְבֵֽית־ יִשְׂרָאֵ֖ל לַעֲשׂ֣וֹת לָהֶ֑ם אַרְבֶּ֥ה אֹתָ֛ם כַּצֹּ֖אן אָדָֽם׃ (יד) כְּצֹ֣אן קָֽדָשִׁ֗ים כְּצֹ֤אן יְרוּשָׁלִַ֙ם֙ בְּמ֣וֹעֲדֶ֔יהָ כֵּ֤ן תִּהְיֶ֙ינָה֙ הֶעָרִ֣ים הֶחֳרֵב֔וֹת מְלֵא֖וֹת צֹ֣אן אָדָ֑ם וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ תחילת פרק לז (טו) הָיְתָ֣ה עָלַי֮ יַד־ יְהוָה֒ וַיּוֹצִאֵ֤נִי בְר֙וּחַ֙ יְהוָ֔ה וַיְנִיחֵ֖נִי בְּת֣וֹךְ הַבִּקְעָ֑ה וְהִ֖יא מְלֵאָ֥ה עֲצָמֽוֹת׃ (טז) וְהֶעֱבִירַ֥נִי עֲלֵיהֶ֖ם סָבִ֣יב ׀ סָבִ֑יב וְהִנֵּ֨ה רַבּ֤וֹת מְאֹד֙ עַל־ פְּנֵ֣י הַבִּקְעָ֔ה וְהִנֵּ֖ה יְבֵשׁ֥וֹת מְאֹֽד׃ (יז) וַיֹּ֣אמֶר אֵלַ֔י בֶּן־ אָדָ֕ם הֲתִחְיֶ֖ינָה הָעֲצָמ֣וֹת הָאֵ֑לֶּה וָאֹמַ֕ר אֲדֹנָ֥י יְהוִ֖ה אַתָּ֥ה יָדָֽעְתָּ׃ (יח) וַיֹּ֣אמֶר אֵלַ֔י הִנָּבֵ֖א עַל־ הָעֲצָמ֣וֹת הָאֵ֑לֶּה וְאָמַרְתָּ֣ אֲלֵיהֶ֔ם הָעֲצָמוֹת֙ הַיְבֵשׁ֔וֹת שִׁמְע֖וּ דְּבַר־ יְהוָֽה׃ (יט) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה לָעֲצָמ֖וֹת הָאֵ֑לֶּה הִנֵּ֨ה אֲנִ֜י מֵבִ֥יא בָכֶ֛ם ר֖וּחַ וִחְיִיתֶֽם׃ (כ) וְנָתַתִּי֩ עֲלֵיכֶ֨ם גִּדִ֜ים וְֽהַעֲלֵתִ֧י עֲלֵיכֶ֣ם בָּשָׂ֗ר וְקָרַמְתִּ֤י עֲלֵיכֶם֙ ע֔וֹר וְנָתַתִּ֥י בָכֶ֛ם ר֖וּחַ וִחְיִיתֶ֑ם וִידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כא) וְנִבֵּ֖אתִי כַּאֲשֶׁ֣ר צֻוֵּ֑יתִי וַֽיְהִי־ ק֤וֹל כְּהִנָּֽבְאִי֙ וְהִנֵּה־ רַ֔עַשׁ וַתִּקְרְב֣וּ עֲצָמ֔וֹת עֶ֖צֶם אֶל־ עַצְמֽוֹ׃ (כב) וְרָאִ֜יתִי וְהִנֵּֽה־ עֲלֵיהֶ֤ם גִּדִים֙ וּבָשָׂ֣ר עָלָ֔ה וַיִּקְרַ֧ם עֲלֵיהֶ֛ם ע֖וֹר מִלְמָ֑עְלָה וְר֖וּחַ אֵ֥ין בָּהֶֽם׃ (כג) וַיֹּ֣אמֶר אֵלַ֔י הִנָּבֵ֖א אֶל־ הָר֑וּחַ הִנָּבֵ֣א בֶן־ אָ֠דָם וְאָמַרְתָּ֨ אֶל־ הָר֜וּחַ כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה מֵאַרְבַּ֤ע רוּחוֹת֙ בֹּ֣אִי הָר֔וּחַ וּפְחִ֛י בַּהֲרוּגִ֥ים הָאֵ֖לֶּה וְיִֽחְיֽוּ׃ (כד) וְהִנַּבֵּ֖אתִי כַּאֲשֶׁ֣ר צִוָּ֑נִי וַתָּבוֹא֩ בָהֶ֨ם הָר֜וּחַ וַיִּֽחְי֗וּ וַיַּֽעַמְדוּ֙ עַל־ רַגְלֵיהֶ֔ם חַ֖יִל גָּד֥וֹל מְאֹד־ מְאֹֽד׃ (כה) וַיֹּאמֶר֮ אֵלַי֒ בֶּן־ אָדָ֕ם הָעֲצָמ֣וֹת הָאֵ֔לֶּה כָּל־ בֵּ֥ית יִשְׂרָאֵ֖ל הֵ֑מָּה הִנֵּ֣ה אֹמְרִ֗ים יָבְשׁ֧וּ עַצְמוֹתֵ֛ינוּ וְאָבְדָ֥ה תִקְוָתֵ֖נוּ נִגְזַ֥רְנוּ לָֽנוּ׃ (כו) לָכֵן֩ הִנָּבֵ֨א וְאָמַרְתָּ֜ אֲלֵיהֶ֗ם כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנֵּה֩ אֲנִ֨י פֹתֵ֜חַ אֶת־ קִבְרֽוֹתֵיכֶ֗ם וְהַעֲלֵיתִ֥י אֶתְכֶ֛ם מִקִּבְרוֹתֵיכֶ֖ם עַמִּ֑י וְהֵבֵאתִ֥י אֶתְכֶ֖ם אֶל־ אַדְמַ֥ת יִשְׂרָאֵֽל׃ (כז) וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֣י יְהוָ֑ה בְּפִתְחִ֣י אֶת־ קִבְרֽוֹתֵיכֶ֗ם וּבְהַעֲלוֹתִ֥י אֶתְכֶ֛ם מִקִּבְרוֹתֵיכֶ֖ם עַמִּֽי׃ (כח) וְנָתַתִּ֨י רוּחִ֤י בָכֶם֙ וִחְיִיתֶ֔ם וְהִנַּחְתִּ֥י אֶתְכֶ֖ם עַל־ אַדְמַתְכֶ֑ם וִידַעְתֶּ֞ם כִּי־ אֲנִ֧י יְהוָ֛ה דִּבַּ֥רְתִּי וְעָשִׂ֖יתִי נְאֻם־ יְהוָֽה׃ (כט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ל) וְאַתָּ֣ה בֶן־ אָדָ֗ם קַח־ לְךָ֙ עֵ֣ץ אֶחָ֔ד וּכְתֹ֤ב עָלָיו֙ לִֽיהוּדָ֔ה וְלִבְנֵ֥י יִשְׂרָאֵ֖ל (חברו) חֲבֵרָ֑יו וּלְקַח֙ עֵ֣ץ אֶחָ֔ד וּכְת֣וֹב עָלָ֗יו לְיוֹסֵף֙ עֵ֣ץ אֶפְרַ֔יִם וְכָל־ בֵּ֥ית יִשְׂרָאֵ֖ל (חברו) חֲבֵרָֽיו׃ (לא) וְקָרַ֨ב אֹתָ֜ם אֶחָ֧ד אֶל־ אֶחָ֛ד לְךָ֖ לְעֵ֣ץ אֶחָ֑ד וְהָי֥וּ לַאֲחָדִ֖ים בְּיָדֶֽךָ׃ (לב) וְכַֽאֲשֶׁר֙ יֹאמְר֣וּ אֵלֶ֔יךָ בְּנֵ֥י עַמְּךָ֖ לֵאמֹ֑ר הֲלֽוֹא־ תַגִּ֥יד לָ֖נוּ מָה־ אֵ֥לֶּה לָּֽךְ׃ (לג) דַּבֵּ֣ר אֲלֵהֶ֗ם כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנֵּה֩ אֲנִ֨י לֹקֵ֜חַ אֶת־ עֵ֤ץ יוֹסֵף֙ אֲשֶׁ֣ר בְּיַד־ אֶפְרַ֔יִם וְשִׁבְטֵ֥י יִשְׂרָאֵ֖ל (חברו) חֲבֵרָ֑יו וְנָתַתִּי֩ אוֹתָ֨ם עָלָ֜יו אֶת־ עֵ֣ץ יְהוּדָ֗ה וַֽעֲשִׂיתִם֙ לְעֵ֣ץ אֶחָ֔ד וְהָי֥וּ אֶחָ֖ד בְּיָדִֽי׃ (לד) וְהָי֨וּ הָעֵצִ֜ים אֲ‍ֽשֶׁר־ תִּכְתֹּ֧ב עֲלֵיהֶ֛ם בְּיָדְךָ֖ לְעֵינֵיהֶֽם׃ (לה) וְדַבֵּ֣ר אֲלֵיהֶ֗ם כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנֵּ֨ה אֲנִ֤י לֹקֵ֙חַ֙ אֶת־ בְּנֵ֣י יִשְׂרָאֵ֔ל מִבֵּ֥ין הַגּוֹיִ֖ם אֲשֶׁ֣ר הָֽלְכוּ־ שָׁ֑ם וְקִבַּצְתִּ֤י אֹתָם֙ מִסָּבִ֔יב וְהֵבֵאתִ֥י אוֹתָ֖ם אֶל־ אַדְמָתָֽם׃ (לו) וְעָשִׂ֣יתִי אֹ֠תָם לְג֨וֹי אֶחָ֤ד בָּאָ֙רֶץ֙ בְּהָרֵ֣י יִשְׂרָאֵ֔ל וּמֶ֧לֶךְ אֶחָ֛ד יִֽהְיֶ֥ה לְכֻלָּ֖ם לְמֶ֑לֶךְ וְלֹ֤א (יהיה)־ יִֽהְיוּ־ עוֹד֙ לִשְׁנֵ֣י גוֹיִ֔ם וְלֹ֨א יֵחָ֥צוּ ע֛וֹד לִשְׁתֵּ֥י מַמְלָכ֖וֹת עֽוֹד׃ (לז) וְלֹ֧א יִֽטַמְּא֣וּ ע֗וֹד בְּגִלּֽוּלֵיהֶם֙ וּבְשִׁקּ֣וּצֵיהֶ֔ם וּבְכֹ֖ל פִּשְׁעֵיהֶ֑ם וְהוֹשַׁעְתִּ֣י אֹתָ֗ם מִכֹּ֤ל מוֹשְׁבֹֽתֵיהֶם֙ אֲשֶׁ֣ר חָטְא֣וּ בָהֶ֔ם וְטִהַרְתִּ֤י אוֹתָם֙ וְהָיוּ־ לִ֣י לְעָ֔ם וַאֲנִ֕י אֶהְיֶ֥ה לָהֶ֖ם לֵאלֹהִֽים׃ (לח) וְעַבְדִּ֤י דָוִד֙ מֶ֣לֶךְ עֲלֵיהֶ֔ם וְרוֹעֶ֥ה אֶחָ֖ד יִהְיֶ֣ה לְכֻלָּ֑ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ל) וְשִׁלַּחְתִּי־ אֵ֣שׁ בְּמָג֔וֹג וּבְיֹשְׁבֵ֥י הָאִיִּ֖ים לָבֶ֑טַח וְיָדְע֖וּ כִּי־ אֲנִ֥י יְהוָֽה׃ (לא) וְאֶת־ שֵׁ֨ם קָדְשִׁ֜י אוֹדִ֗יעַ בְּתוֹךְ֙ עַמִּ֣י יִשְׂרָאֵ֔ל וְלֹֽא־ אַחֵ֥ל אֶת־ שֵׁם־ קָדְשִׁ֖י ע֑וֹד וְיָדְע֤וּ הַגּוֹיִם֙ כִּי־ אֲנִ֣י יְהוָ֔ה קָד֖וֹשׁ בְּיִשְׂרָאֵֽל׃ (לב) הִנֵּ֤ה בָאָה֙ וְנִֽהְיָ֔תָה נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה ה֥וּא הַיּ֖וֹם אֲשֶׁ֥ר דִּבַּֽרְתִּי׃ (לג) וְֽיָצְא֞וּ יֹשְׁבֵ֣י ׀ עָרֵ֣י יִשְׂרָאֵ֗ל וּבִעֲר֡וּ וְ֠הִשִּׂיקוּ בְּנֶ֨שֶׁק וּמָגֵ֤ן וְצִנָּה֙ בְּקֶ֣שֶׁת וּבְחִצִּ֔ים וּבְמַקֵּ֥ל יָ֖ד וּבְרֹ֑מַח וּבִעֲר֥וּ בָהֶ֛ם אֵ֖שׁ שֶׁ֥בַע שָׁנִֽים׃ (לד) וְלֹֽא־ יִשְׂא֨וּ עֵצִ֜ים מִן־ הַשָּׂדֶ֗ה וְלֹ֤א יַחְטְבוּ֙ מִן־ הַיְּעָרִ֔ים כִּ֥י בַנֶּ֖שֶׁק יְבַֽעֲרוּ־ אֵ֑שׁ וְשָׁלְל֣וּ אֶת־ שֹׁלְלֵיהֶ֗ם וּבָֽזְזוּ֙ אֶת־ בֹּ֣זְזֵיהֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לה) וְהָיָ֣ה בַיּ֣וֹם הַה֡וּא אֶתֵּ֣ן לְגוֹג֩ ׀ מְקֽוֹם־ שָׁ֨ם קֶ֜בֶר בְּיִשְׂרָאֵ֗ל גֵּ֤י הָעֹֽבְרִים֙ קִדְמַ֣ת הַיָּ֔ם וְחֹסֶ֥מֶת הִ֖יא אֶת־ הָעֹֽבְרִ֑ים וְקָ֣בְרוּ שָׁ֗ם אֶת־ גּוֹג֙ וְאֶת־ כָּל־ הֲמוֹנֹ֔ה וְקָ֣רְא֔וּ גֵּ֖יא הֲמ֥וֹן גּֽוֹג׃ (לו) וּקְבָרוּם֙ בֵּ֣ית יִשְׂרָאֵ֔ל לְמַ֖עַן טַהֵ֣ר אֶת־ הָאָ֑רֶץ שִׁבְעָ֖ה חֳדָשִֽׁים׃ (לז) וְקָֽבְרוּ֙ כָּל־ עַ֣ם הָאָ֔רֶץ וְהָיָ֥ה לָהֶ֖ם לְשֵׁ֑ם י֚וֹם הִכָּ֣בְדִ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לח) וְאַנְשֵׁ֨י תָמִ֤יד יַבְדִּ֙ילוּ֙ עֹבְרִ֣ים בָּאָ֔רֶץ מְקַבְּרִ֣ים אֶת־ הָעֹבְרִ֗ים אֶת־ הַנּוֹתָרִ֛ים עַל־ פְּנֵ֥י הָאָ֖רֶץ לְטַֽהֲרָ֑הּ מִקְצֵ֥ה שִׁבְעָֽה־ חֳדָשִׁ֖ים יַחְקֹֽרוּ׃ (לט) וְעָבְר֤וּ הָעֹֽבְרִים֙ בָּאָ֔רֶץ וְרָאָה֙ עֶ֣צֶם אָדָ֔ם וּבָנָ֥ה אֶצְל֖וֹ צִיּ֑וּן עַ֣ד קָבְר֤וּ אֹתוֹ֙ הַֽמְקַבְּרִ֔ים אֶל־ גֵּ֖יא הֲמ֥וֹן גּֽוֹג׃ (מ) וְגַ֥ם שֶׁם־ עִ֛יר הֲמוֹנָ֖ה וְטִהֲר֥וּ הָאָֽרֶץ׃ (מא) וְאַתָּ֨ה בֶן־ אָדָ֜ם כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה אֱמֹר֩ לְצִפּ֨וֹר כָּל־ כָּנָ֜ף וּלְכֹ֣ל ׀ חַיַּ֣ת הַשָּׂדֶ֗ה הִקָּבְצ֤וּ וָבֹ֙אוּ֙ הֵאָסְפ֣וּ מִסָּבִ֔יב עַל־ זִבְחִ֗י אֲשֶׁ֨ר אֲנִ֜י זֹבֵ֤חַ לָכֶם֙ זֶ֣בַח גָּד֔וֹל עַ֖ל הָרֵ֣י יִשְׂרָאֵ֑ל וַאֲכַלְתֶּ֥ם בָּשָׂ֖ר וּשְׁתִ֥יתֶם דָּֽם׃ (מב) בְּשַׂ֤ר גִּבּוֹרִים֙ תֹּאכֵ֔לוּ וְדַם־ נְשִׂיאֵ֥י הָאָ֖רֶץ תִּשְׁתּ֑וּ אֵילִ֨ים כָּרִ֤ים וְעַתּוּדִים֙ פָּרִ֔ים מְרִיאֵ֥י בָשָׁ֖ן כֻּלָּֽם׃ (מג) וַאֲכַלְתֶּם־ חֵ֣לֶב לְשָׂבְעָ֔ה וּשְׁתִ֥יתֶם דָּ֖ם לְשִׁכָּר֑וֹן מִזִּבְחִ֖י אֲשֶׁר־ זָבַ֥חְתִּי לָכֶֽם׃ (מד) וּשְׂבַעְתֶּ֤ם עַל־ שֻׁלְחָנִי֙ ס֣וּס וָרֶ֔כֶב גִּבּ֖וֹר וְכָל־ אִ֣ישׁ מִלְחָמָ֑ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (מה) וְנָתַתִּ֥י אֶת־ כְּבוֹדִ֖י בַּגּוֹיִ֑ם וְרָא֣וּ כָל־ הַגּוֹיִ֗ם אֶת־ מִשְׁפָּטִי֙ אֲשֶׁ֣ר עָשִׂ֔יתִי וְאֶת־ יָדִ֖י אֲשֶׁר־ שַׂ֥מְתִּי בָהֶֽם׃
