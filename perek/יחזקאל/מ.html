יחזקאל פרק מ לפי החלוקה לסדרים

פרק מ בספר יחזקאל (49 פסוקים) נמצא ביחזקאל סדר כד (פסוקים ט–נב) וביחזקאל סדר כה (פסוקים א–ה).

יחזקאל סדר כד

(א) וְיָֽדְעוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל כִּ֛י אֲנִ֥י יְהוָ֖ה אֱלֹֽהֵיהֶ֑ם מִן־ הַיּ֥וֹם הַה֖וּא וָהָֽלְאָה׃ (ב) וְיָדְע֣וּ הַ֠גּוֹיִם כִּ֣י בַעֲוֺנָ֞ם גָּל֣וּ בֵֽית־ יִשְׂרָאֵ֗ל עַ֚ל אֲשֶׁ֣ר מָֽעֲלוּ־ בִ֔י וָאַסְתִּ֥ר פָּנַ֖י מֵהֶ֑ם וָֽאֶתְּנֵם֙ בְּיַ֣ד צָרֵיהֶ֔ם וַיִּפְּל֥וּ בַחֶ֖רֶב כֻּלָּֽם׃ (ג) כְּטֻמְאָתָ֥ם וּכְפִשְׁעֵיהֶ֖ם עָשִׂ֣יתִי אֹתָ֑ם וָאַסְתִּ֥ר פָּנַ֖י מֵהֶֽם׃ (ד) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה עַתָּ֗ה אָשִׁיב֙ אֶת־ (שבית) שְׁב֣וּת יַֽעֲקֹ֔ב וְרִֽחַמְתִּ֖י כָּל־ בֵּ֣ית יִשְׂרָאֵ֑ל וְקִנֵּאתִ֖י לְשֵׁ֥ם קָדְשִֽׁי׃ (ה) וְנָשׂוּ֙ אֶת־ כְּלִמָּתָ֔ם וְאֶת־ כָּל־ מַעֲלָ֖ם אֲשֶׁ֣ר מָעֲלוּ־ בִ֑י בְּשִׁבְתָּ֧ם עַל־ אַדְמָתָ֛ם לָבֶ֖טַח וְאֵ֥ין מַחֲרִֽיד׃ (ו) בְּשׁוֹבְבִ֤י אוֹתָם֙ מִן־ הָ֣עַמִּ֔ים וְקִבַּצְתִּ֣י אֹתָ֔ם מֵֽאַרְצ֖וֹת אֹֽיְבֵיהֶ֑ם וְנִקְדַּ֣שְׁתִּי בָ֔ם לְעֵינֵ֖י הַגּוֹיִ֥ם רַבִּֽים׃ (ז) וְיָדְע֗וּ כִּ֣י אֲנִ֤י יְהוָה֙ אֱלֹ֣הֵיהֶ֔ם בְּהַגְלוֹתִ֤י אֹתָם֙ אֶל־ הַגּוֹיִ֔ם וְכִנַּסְתִּ֖ים עַל־ אַדְמָתָ֑ם וְלֹֽא־ אוֹתִ֥יר ע֛וֹד מֵהֶ֖ם שָֽׁם׃ (ח) וְלֹֽא־ אַסְתִּ֥יר ע֛וֹד פָּנַ֖י מֵהֶ֑ם אֲשֶׁ֨ר שָׁפַ֤כְתִּי אֶת־ רוּחִי֙ עַל־ בֵּ֣ית יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ תחילת פרק מ (ט) בְּעֶשְׂרִ֣ים וְחָמֵ֣שׁ שָׁנָ֣ה לְ֠גָלוּתֵנוּ בְּרֹ֨אשׁ הַשָּׁנָ֜ה בֶּעָשׂ֣וֹר לַחֹ֗דֶשׁ בְּאַרְבַּ֤ע עֶשְׂרֵה֙ שָׁנָ֔ה אַחַ֕ר אֲשֶׁ֥ר הֻכְּתָ֖ה הָעִ֑יר בְּעֶ֣צֶם ׀ הַיּ֣וֹם הַזֶּ֗ה הָיְתָ֤ה עָלַי֙ יַד־ יְהוָ֔ה וַיָּבֵ֥א אֹתִ֖י שָֽׁמָּה׃ (י) בְּמַרְא֣וֹת אֱלֹהִ֔ים הֱבִיאַ֖נִי אֶל־ אֶ֣רֶץ יִשְׂרָאֵ֑ל וַיְנִיחֵ֗נִי אֶל־ הַ֤ר גָּבֹ֙הַּ֙ מְאֹ֔ד וְעָלָ֥יו כְּמִבְנֵה־ עִ֖יר מִנֶּֽגֶב׃ (יא) וַיָּבֵ֨יא אוֹתִ֜י שָׁ֗מָּה וְהִנֵּה־ אִישׁ֙ מַרְאֵ֙הוּ֙ כְּמַרְאֵ֣ה נְחֹ֔שֶׁת וּפְתִיל־ פִּשְׁתִּ֥ים בְּיָד֖וֹ וּקְנֵ֣ה הַמִּדָּ֑ה וְה֥וּא עֹמֵ֖ד בַּשָּֽׁעַר׃ (יב) וַיְדַבֵּ֨ר אֵלַ֜י הָאִ֗ישׁ בֶּן־ אָדָ֡ם רְאֵ֣ה בְעֵינֶיךָ֩ וּבְאָזְנֶ֨יךָ שְּׁמָ֜ע וְשִׂ֣ים לִבְּךָ֗ לְכֹ֤ל אֲשֶׁר־ אֲנִי֙ מַרְאֶ֣ה אוֹתָ֔ךְ כִּ֛י לְמַ֥עַן הַרְאוֹתְכָ֖ה הֻבָ֣אתָה הֵ֑נָּה הַגֵּ֛ד אֶת־ כָּל־ אֲשֶׁר־ אַתָּ֥ה רֹאֶ֖ה לְבֵ֥ית יִשְׂרָאֵֽל׃ (יג) וְהִנֵּ֥ה חוֹמָ֛ה מִח֥וּץ לַבַּ֖יִת סָבִ֣יב ׀ סָבִ֑יב וּבְיַ֨ד הָאִ֜ישׁ קְנֵ֣ה הַמִּדָּ֗ה שֵׁשׁ־ אַמּ֤וֹת בָּֽאַמָּה֙ וָטֹ֔פַח וַיָּ֜מָד אֶת־ רֹ֤חַב הַבִּנְיָן֙ קָנֶ֣ה אֶחָ֔ד וְקוֹמָ֖ה קָנֶ֥ה אֶחָֽד׃ (יד) וַיָּב֗וֹא אֶל־ שַׁ֙עַר֙ אֲשֶׁ֤ר פָּנָיו֙ דֶּ֣רֶךְ הַקָּדִ֔ימָה וַיַּ֖עַל (במעלותו) בְּמַֽעֲלוֹתָ֑יו וַיָּ֣מָד ׀ אֶת־ סַ֣ף הַשַּׁ֗עַר קָנֶ֤ה אֶחָד֙ רֹ֔חַב וְאֵת֙ סַ֣ף אֶחָ֔ד קָנֶ֥ה אֶחָ֖ד רֹֽחַב׃ (טו) וְהַתָּ֗א קָנֶ֨ה אֶחָ֥ד אֹ֙רֶךְ֙ וְקָנֶ֤ה אֶחָד֙ רֹ֔חַב וּבֵ֥ין הַתָּאִ֖ים חָמֵ֣שׁ אַמּ֑וֹת וְסַ֣ף הַ֠שַּׁעַר מֵאֵ֨צֶל אוּלָ֥ם הַשַּׁ֛עַר מֵֽהַבַּ֖יִת קָנֶ֥ה אֶחָֽד׃ (טז) וַיָּ֜מָד אֶת־ אֻלָ֥ם הַשַּׁ֛עַר מֵהַבַּ֖יִת קָנֶ֥ה אֶחָֽד׃ (יז) וַיָּ֜מָד אֶת־ אֻלָ֤ם הַשַּׁ֙עַר֙ שְׁמֹנֶ֣ה אַמּ֔וֹת (ואילו) וְאֵילָ֖יו שְׁתַּ֣יִם אַמּ֑וֹת וְאֻלָ֥ם הַשַּׁ֖עַר מֵהַבָּֽיִת׃ (יח) וְתָאֵ֨י הַשַּׁ֜עַר דֶּ֣רֶךְ הַקָּדִ֗ים שְׁלֹשָׁ֤ה מִפֹּה֙ וּשְׁלֹשָׁ֣ה מִפֹּ֔ה מִדָּ֥ה אַחַ֖ת לִשְׁלָשְׁתָּ֑ם וּמִדָּ֥ה אַחַ֛ת לָאֵילִ֖ם מִפֹּ֥ה וּמִפּֽוֹ׃ (יט) וַיָּ֛מָד אֶת־ רֹ֥חַב פֶּֽתַח־ הַשַּׁ֖עַר עֶ֣שֶׂר אַמּ֑וֹת אֹ֣רֶךְ הַשַּׁ֔עַר שְׁל֥וֹשׁ עֶשְׂרֵ֖ה אַמּֽוֹת׃ (כ) וּגְב֞וּל לִפְנֵ֤י הַתָּאוֹת֙ אַמָּ֣ה אֶחָ֔ת וְאַמָּה־ אַחַ֥ת גְּב֖וּל מִפֹּ֑ה וְהַתָּ֕א שֵׁשׁ־ אַמּ֣וֹת מִפּ֔וֹ וְשֵׁ֥שׁ אַמּ֖וֹת מִפּֽוֹ׃ (כא) וַיָּ֣מָד אֶת־ הַשַּׁ֗עַר מִגַּ֤ג הַתָּא֙ לְגַגּ֔וֹ רֹ֕חַב עֶשְׂרִ֥ים וְחָמֵ֖שׁ אַמּ֑וֹת פֶּ֖תַח נֶ֥גֶד פָּֽתַח׃ (כב) וַיַּ֥עַשׂ אֶת־ אֵילִ֖ים שִׁשִּׁ֣ים אַמָּ֑ה וְאֶל־ אֵיל֙ הֶֽחָצֵ֔ר הַשַּׁ֖עַר סָבִ֥יב ׀ סָבִֽיב׃ (כג) וְעַ֗ל פְּנֵי֙ הַשַּׁ֣עַר (היאתון) הָֽאִית֔וֹן עַל־ לִפְנֵ֕י אֻלָ֥ם הַשַּׁ֖עַר הַפְּנִימִ֑י חֲמִשִּׁ֖ים אַמָּֽה׃ (כד) וְחַלֹּנ֣וֹת אֲטֻמ֣וֹת אֶֽל־ הַתָּאִ֡ים וְאֶל֩ אֵלֵיהֵ֨מָה לִפְנִ֤ימָה לַשַּׁ֙עַר֙ סָבִ֣יב ׀ סָבִ֔יב וְכֵ֖ן לָאֵֽלַמּ֑וֹת וְחַלּוֹנ֞וֹת סָבִ֤יב ׀ סָבִיב֙ לִפְנִ֔ימָה וְאֶל־ אַ֖יִל תִּמֹרִֽים׃ (כה) וַיְבִיאֵ֗נִי אֶל־ הֶֽחָצֵר֙ הַחִ֣יצוֹנָ֔ה וְהִנֵּ֤ה לְשָׁכוֹת֙ וְרִֽצְפָ֔ה עָשׂ֥וּי לֶחָצֵ֖ר סָבִ֣יב ׀ סָבִ֑יב שְׁלֹשִׁ֥ים לְשָׁכ֖וֹת אֶל־ הָרִֽצְפָֽה׃ (כו) וְהָרִֽצְפָה֙ אֶל־ כֶּ֣תֶף הַשְּׁעָרִ֔ים לְעֻמַּ֖ת אֹ֣רֶךְ הַשְּׁעָרִ֑ים הָרִֽצְפָ֖ה הַתַּחְתּוֹנָֽה׃ (כז) וַיָּ֣מָד רֹ֡חַב מִלִּפְנֵי֩ הַשַּׁ֨עַר הַתַּחְתּ֜וֹנָה לִפְנֵ֨י הֶחָצֵ֧ר הַפְּנִימִ֛י מִח֖וּץ מֵאָ֣ה אַמָּ֑ה הַקָּדִ֖ים וְהַצָּפֽוֹן׃ (כח) וְהַשַּׁ֗עַר אֲשֶׁ֤ר פָּנָיו֙ דֶּ֣רֶךְ הַצָּפ֔וֹן לֶחָצֵ֖ר הַחִֽיצוֹנָ֑ה מָדַ֥ד אָרְכּ֖וֹ וְרָחְבּֽוֹ׃ (כט) (ותאו) וְתָאָ֗יו שְׁלוֹשָׁ֤ה מִפּוֹ֙ וּשְׁלֹשָׁ֣ה מִפּ֔וֹ (ואילו) וְאֵילָ֤יו (ואלמו) וְאֵֽלַמָּיו֙ הָיָ֔ה כְּמִדַּ֖ת הַשַּׁ֣עַר הָרִאשׁ֑וֹן חֲמִשִּׁ֤ים אַמָּה֙ אָרְכּ֔וֹ וְרֹ֕חַב חָמֵ֥שׁ וְעֶשְׂרִ֖ים בָּאַמָּֽה׃ (ל) (וחלונו) וְחַלּוֹנָ֤יו (ואלמו) וְאֵֽלַמָּיו֙ (ותמרו) וְתִ֣מֹרָ֔יו כְּמִדַּ֣ת הַשַּׁ֔עַר אֲשֶׁ֥ר פָּנָ֖יו דֶּ֣רֶךְ הַקָּדִ֑ים וּבְמַעֲל֥וֹת שֶׁ֙בַע֙ יַֽעֲלוּ־ ב֔וֹ (ואילמו) וְאֵֽילַמָּ֖יו לִפְנֵיהֶֽם׃ (לא) וְשַׁ֙עַר֙ לֶחָצֵ֣ר הַפְּנִימִ֔י נֶ֣גֶד הַשַּׁ֔עַר לַצָּפ֖וֹן וְלַקָּדִ֑ים וַיָּ֧מָד מִשַּׁ֛עַר אֶל־ שַׁ֖עַר מֵאָ֥ה אַמָּֽה׃ (לב) וַיּוֹלִכֵ֙נִי֙ דֶּ֣רֶךְ הַדָּר֔וֹם וְהִנֵּה־ שַׁ֖עַר דֶּ֣רֶךְ הַדָּר֑וֹם וּמָדַ֤ד (אילו) אֵילָיו֙ (ואילמו) וְאֵ֣ילַמָּ֔יו כַּמִּדּ֖וֹת הָאֵֽלֶּה׃ (לג) וְחַלּוֹנִ֨ים ל֤וֹ (ולאילמו) וּלְאֵֽילַמָּיו֙ סָבִ֣יב ׀ סָבִ֔יב כְּהַחֲלֹּנ֖וֹת הָאֵ֑לֶּה חֲמִשִּׁ֤ים אַמָּה֙ אֹ֔רֶךְ וְרֹ֕חַב חָמֵ֥שׁ וְעֶשְׂרִ֖ים אַמָּֽה׃ (לד) וּמַעֲל֤וֹת שִׁבְעָה֙ (עלותו) עֹֽלוֹתָ֔יו (ואלמו) וְאֵֽלַמָּ֖יו לִפְנֵיהֶ֑ם וְתִמֹרִ֣ים ל֗וֹ אֶחָ֥ד מִפּ֛וֹ וְאֶחָ֥ד מִפּ֖וֹ אֶל־ (אילו) אֵילָֽיו׃ (לה) וְשַׁ֛עַר לֶחָצֵ֥ר הַפְּנִימִ֖י דֶּ֣רֶךְ הַדָּר֑וֹם וַיָּ֨מָד מִשַּׁ֧עַר אֶל־ הַשַּׁ֛עַר דֶּ֥רֶךְ הַדָּר֖וֹם מֵאָ֥ה אַמּֽוֹת׃ (לו) וַיְבִיאֵ֛נִי אֲשֶׁ֨ר עַל־ אַחֲרֶ֧יהָ (ואתוקיהא) וְאַתִּיקֶ֛יהָא מִפּ֥וֹ וּמִפּ֖וֹ מֵאָ֣ה אַמָּ֑ה וְהַֽהֵיכָל֙ הַפְּנִימִ֔י וְאֻֽלַמֵּ֖י הֶחָצֵֽר׃ (כא) הַסִּפִּ֡ים וְהַחַלּוֹנִ֣ים הָ֠אֲטֻמוֹת וְהָאַתִּיקִ֤ים ׀ סָבִיב֙ לִשְׁלָשְׁתָּ֔ם נֶ֧גֶד הַסַּ֛ף שְׂחִ֥יף עֵ֖ץ סָבִ֣יב ׀ סָבִ֑יב וְהָאָ֙רֶץ֙ עַד־ הַֽחַלֹּנ֔וֹת וְהַֽחַלֹּנ֖וֹת מְכֻסּֽוֹת׃ (כב) עַל־ מֵעַ֣ל הַפֶּ֡תַח וְעַד־ הַבַּיִת֩ הַפְּנִימִ֨י וְלַח֜וּץ וְאֶל־ כָּל־ הַקִּ֨יר סָבִ֧יב ׀ סָבִ֛יב בַּפְּנִימִ֥י וּבַחִיצ֖וֹן מִדּֽוֹת׃ (כג) וְעָשׂ֥וּי כְּרוּבִ֖ים וְתִֽמֹרִ֑ים וְתִֽמֹרָה֙ בֵּין־ כְּר֣וּב לִכְר֔וּב וּשְׁנַ֥יִם פָּנִ֖ים לַכְּרֽוּב׃ (כד) וּפְנֵ֨י אָדָ֤ם אֶל־ הַתִּֽמֹרָה֙ מִפּ֔וֹ וּפְנֵֽי־ כְפִ֥יר אֶל־ הַתִּֽמֹרָ֖ה מִפּ֑וֹ עָשׂ֥וּי אֶל־ כָּל־ הַבַּ֖יִת סָבִ֥יב ׀ סָבִֽיב׃ (כה) מֵהָאָ֙רֶץ֙ עַד־ מֵעַ֣ל הַפֶּ֔תַח הַכְּרוּבִ֥ים וְהַתִּֽמֹרִ֖ים עֲשׂוּיִ֑ם וְקִ֖יר הַׄהֵׄיׄכָֽׄלׄ׃ (כו) הַֽהֵיכָ֖ל מְזוּזַ֣ת רְבֻעָ֑ה וּפְנֵ֣י הַקֹּ֔דֶשׁ הַמַּרְאֶ֖ה כַּמַּרְאֶֽה׃ (כז) הַמִּזְבֵּ֡חַ עֵ֣ץ שָׁלוֹשׁ֩ אַמּ֨וֹת גָּבֹ֜הַּ וְאָרְכּ֣וֹ שְׁתַּֽיִם־ אַמּ֗וֹת וּמִקְצֹֽעוֹתָיו֙ ל֔וֹ וְאָרְכּ֥וֹ וְקִֽירֹתָ֖יו עֵ֑ץ וַיְדַבֵּ֣ר אֵלַ֔י זֶ֚ה הַשֻּׁלְחָ֔ן אֲשֶׁ֖ר לִפְנֵ֥י יְהוָֽה׃ (כח) וּשְׁתַּ֧יִם דְּלָת֛וֹת לַֽהֵיכָ֖ל וְלַקֹּֽדֶשׁ׃ (כט) וּשְׁתַּ֥יִם דְּלָת֖וֹת לַדְּלָת֑וֹת שְׁ֚תַּיִם מוּסַבּ֣וֹת דְּלָת֔וֹת שְׁ֚תַּיִם לְדֶ֣לֶת אֶחָ֔ת וּשְׁתֵּ֥י דְלָת֖וֹת לָאַחֶֽרֶת׃ (ל) וַעֲשׂוּיָ֨ה אֲלֵיהֶ֜ן אֶל־ דַּלְת֤וֹת הַֽהֵיכָל֙ כְּרוּבִ֣ים וְתִֽמֹרִ֔ים כַּאֲשֶׁ֥ר עֲשׂוּיִ֖ם לַקִּיר֑וֹת וְעָ֥ב עֵ֛ץ אֶל־ פְּנֵ֥י הָאוּלָ֖ם מֵהַחֽוּץ׃ (לא) וְחַלּוֹנִ֨ים אֲטֻמ֤וֹת וְתִֽמֹרִים֙ מִפּ֣וֹ וּמִפּ֔וֹ אֶל־ כִּתְפ֖וֹת הָֽאוּלָ֑ם וְצַלְע֥וֹת הַבַּ֖יִת וְהָעֻבִּֽים׃ (לב) וַיּוֹצִאֵ֗נִי אֶל־ הֶֽחָצֵר֙ הַחִ֣יצוֹנָ֔ה הַדֶּ֖רֶךְ דֶּ֣רֶךְ הַצָּפ֑וֹן וַיְבִאֵ֣נִי אֶל־ הַלִּשְׁכָּ֗ה אֲשֶׁ֨ר נֶ֧גֶד הַגִּזְרָ֛ה וַאֲשֶֽׁר־ נֶ֥גֶד הַבִּנְיָ֖ן אֶל־ הַצָּפֽוֹן׃ (לג) אֶל־ פְּנֵי־ אֹ֙רֶךְ֙ אַמּ֣וֹת הַמֵּאָ֔ה פֶּ֖תַח הַצָּפ֑וֹן וְהָרֹ֖חַב חֲמִשִּׁ֥ים אַמּֽוֹת׃ (לד) נֶ֣גֶד הָֽעֶשְׂרִ֗ים אֲשֶׁר֙ לֶחָצֵ֣ר הַפְּנִימִ֔י וְנֶ֣גֶד רִֽצְפָ֔ה אֲשֶׁ֖ר לֶחָצֵ֣ר הַחִֽיצוֹנָ֑ה אַתִּ֥יק אֶל־ פְּנֵֽי־ אַתִּ֖יק בַּשְּׁלִשִֽׁים׃ (לה) וְלִפְנֵ֨י הַלְּשָׁכ֜וֹת מַהֲלַךְ֩ עֶ֨שֶׂר אַמּ֥וֹת רֹ֙חַב֙ אֶל־ הַפְּנִימִ֔ית דֶּ֖רֶךְ אַמָּ֣ה אֶחָ֑ת וּפִתְחֵיהֶ֖ם לַצָּפֽוֹן׃ (לו) וְהַלְּשָׁכ֥וֹת הָעֶלְיוֹנֹ֖ת קְצֻר֑וֹת כִּֽי־ יוֹכְל֨וּ אַתִּיקִ֜ים מֵהֵ֗נָה מֵֽהַתַּחְתֹּנ֛וֹת וּמֵהַתִּֽכֹנ֖וֹת בִּנְיָֽן׃ (לז) כִּ֤י מְשֻׁלָּשׁוֹת֙ הֵ֔נָּה וְאֵ֤ין לָהֶן֙ עַמּוּדִ֔ים כְּעַמּוּדֵ֖י הַחֲצֵר֑וֹת עַל־ כֵּ֣ן נֶאֱצַ֗ל מֵהַתַּחְתּוֹנ֛וֹת וּמֵהַתִּֽיכֹנ֖וֹת מֵהָאָֽרֶץ׃ (לח) וְגָדֵ֤ר אֲשֶׁר־ לַחוּץ֙ לְעֻמַּ֣ת הַלְּשָׁכ֔וֹת דֶּ֛רֶךְ הֶחָצֵ֥ר הַחִֽצוֹנָ֖ה אֶל־ פְּנֵ֣י הַלְּשָׁכ֑וֹת אָרְכּ֖וֹ חֲמִשִּׁ֥ים אַמָּֽה׃ (לט) כִּֽי־ אֹ֣רֶךְ הַלְּשָׁכ֗וֹת אֲשֶׁ֛ר לֶחָצֵ֥ר הַחִֽצוֹנָ֖ה חֲמִשִּׁ֣ים אַמָּ֑ה וְהִנֵּ֛ה עַל־ פְּנֵ֥י הַהֵיכָ֖ל מֵאָ֥ה אַמָּֽה׃ (מ) (ומתחתה לשכות) וּמִתַּ֖חַת הַלְּשָׁכ֣וֹת הָאֵ֑לֶּה (המבוא) הַמֵּבִיא֙ מֵֽהַקָּדִ֔ים בְּבֹא֣וֹ לָהֵ֔נָּה מֵֽהֶחָצֵ֖ר הַחִצֹנָֽה׃ (מא) בְּרֹ֣חַב ׀ גֶּ֣דֶר הֶחָצֵ֗ר דֶּ֧רֶךְ הַקָּדִ֛ים אֶל־ פְּנֵ֧י הַגִּזְרָ֛ה וְאֶל־ פְּנֵ֥י הַבִּנְיָ֖ן לְשָׁכֽוֹת׃ (מב) וְדֶ֙רֶךְ֙ לִפְנֵיהֶ֔ם כְּמַרְאֵ֣ה הַלְּשָׁכ֗וֹת אֲשֶׁר֙ דֶּ֣רֶךְ הַצָּפ֔וֹן כְּאָרְכָּ֖ן כֵּ֣ן רָחְבָּ֑ן וְכֹל֙ מוֹצָ֣אֵיהֶ֔ן וּכְמִשְׁפְּטֵיהֶ֖ן וּכְפִתְחֵיהֶֽן׃ (מג) וּכְפִתְחֵ֣י הַלְּשָׁכ֗וֹת אֲשֶׁר֙ דֶּ֣רֶךְ הַדָּר֔וֹם פֶּ֖תַח בְּרֹ֣אשׁ דָּ֑רֶךְ דֶּ֗רֶךְ בִּפְנֵי֙ הַגְּדֶ֣רֶת הֲגִינָ֔ה דֶּ֥רֶךְ הַקָּדִ֖ים בְּבוֹאָֽן׃
