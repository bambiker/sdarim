יחזקאל פרק כג לפי החלוקה לסדרים

פרק כג בספר יחזקאל (49 פסוקים) נמצא ביחזקאל סדר יג (פסוקים יז–מב) וביחזקאל סדר יד (פסוקים א–כג).

יחזקאל סדר יג

(א) וְנִחַ֥לְתְּ בָּ֖ךְ לְעֵינֵ֣י גוֹיִ֑ם וְיָדַ֖עַתְּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (ב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ג) בֶּן־ אָדָ֕ם הָיוּ־ לִ֥י בֵֽית־ יִשְׂרָאֵ֖ל (לסוג) לְסִ֑יג כֻּלָּ֡ם נְ֠חֹשֶׁת וּבְדִ֨יל וּבַרְזֶ֤ל וְעוֹפֶ֙רֶת֙ בְּת֣וֹךְ כּ֔וּר סִגִ֥ים כֶּ֖סֶף הָיֽוּ׃ (ד) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֛עַן הֱי֥וֹת כֻּלְּכֶ֖ם לְסִגִ֑ים לָכֵן֙ הִנְנִ֣י קֹבֵ֣ץ אֶתְכֶ֔ם אֶל־ תּ֖וֹךְ יְרוּשָׁלִָֽם׃ (ה) קְבֻ֣צַת כֶּ֡סֶף וּ֠נְחֹשֶׁת וּבַרְזֶ֨ל וְעוֹפֶ֤רֶת וּבְדִיל֙ אֶל־ תּ֣וֹךְ כּ֔וּר לָפַֽחַת־ עָלָ֥יו אֵ֖שׁ לְהַנְתִּ֑יךְ כֵּ֤ן אֶקְבֹּץ֙ בְּאַפִּ֣י וּבַחֲמָתִ֔י וְהִנַּחְתִּ֥י וְהִתַּכְתִּ֖י אֶתְכֶֽם׃ (ו) וְכִנַּסְתִּ֣י אֶתְכֶ֔ם וְנָפַחְתִּ֥י עֲלֵיכֶ֖ם בְּאֵ֣שׁ עֶבְרָתִ֑י וְנִתַּכְתֶּ֖ם בְּתוֹכָֽהּ׃ (ז) כְּהִתּ֥וּךְ כֶּ֙סֶף֙ בְּת֣וֹךְ כּ֔וּר כֵּ֖ן תֻּתְּכ֣וּ בְתוֹכָ֑הּ וִֽידַעְתֶּם֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה שָׁפַ֥כְתִּי חֲמָתִ֖י עֲלֵיכֶֽם׃ (ח) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ט) בֶּן־ אָדָ֕ם אֱמָר־ לָ֕הּ אַ֣תְּ אֶ֔רֶץ לֹ֥א מְטֹהָרָ֖ה הִ֑יא לֹ֥א גֻשְׁמָ֖הּ בְּי֥וֹם זָֽעַם׃ (י) קֶ֤שֶׁר נְבִיאֶ֙יהָ֙ בְּתוֹכָ֔הּ כַּאֲרִ֥י שׁוֹאֵ֖ג טֹ֣רֵֽף טָ֑רֶף נֶ֣פֶשׁ אָכָ֗לוּ חֹ֤סֶן וִיקָר֙ יִקָּ֔חוּ אַלְמְנוֹתֶ֖יהָ הִרְבּ֥וּ בְתוֹכָֽהּ׃ (יא) כֹּהֲנֶ֜יהָ חָמְס֣וּ תוֹרָתִי֮ וַיְחַלְּל֣וּ קָדָשַׁי֒ בֵּֽין־ קֹ֤דֶשׁ לְחֹל֙ לֹ֣א הִבְדִּ֔ילוּ וּבֵין־ הַטָּמֵ֥א לְטָה֖וֹר לֹ֣א הוֹדִ֑יעוּ וּמִשַׁבְּתוֹתַי֙ הֶעְלִ֣ימוּ עֵֽינֵיהֶ֔ם וָאֵחַ֖ל בְּתוֹכָֽם׃ (יב) שָׂרֶ֣יהָ בְקִרְבָּ֔הּ כִּזְאֵבִ֖ים טֹ֣רְפֵי טָ֑רֶף לִשְׁפָּךְ־ דָּם֙ לְאַבֵּ֣ד נְפָשׁ֔וֹת לְמַ֖עַן בְּצֹ֥עַ בָּֽצַע׃ (יג) וּנְבִיאֶ֗יהָ טָח֤וּ לָהֶם֙ תָּפֵ֔ל חֹזִ֣ים שָׁ֔וְא וְקֹסְמִ֥ים לָהֶ֖ם כָּזָ֑ב אֹמְרִ֗ים כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה וַֽיהוָ֖ה לֹ֥א דִבֵּֽר׃ (יד) עַ֤ם הָאָ֙רֶץ֙ עָ֣שְׁקוּ עֹ֔שֶׁק וְגָזְל֖וּ גָּזֵ֑ל וְעָנִ֤י וְאֶבְיוֹן֙ הוֹנ֔וּ וְאֶת־ הַגֵּ֥ר עָשְׁק֖וּ בְּלֹ֥א מִשְׁפָּֽט׃ (טו) וָאֲבַקֵּ֣שׁ מֵהֶ֡ם אִ֣ישׁ גֹּֽדֵר־ גָּדֵר֩ וְעֹמֵ֨ד בַּפֶּ֧רֶץ לְפָנַ֛י בְּעַ֥ד הָאָ֖רֶץ לְבִלְתִּ֣י שַׁחֲתָ֑הּ וְלֹ֖א מָצָֽאתִי׃ (טז) וָאֶשְׁפֹּ֤ךְ עֲלֵיהֶם֙ זַעְמִ֔י בְּאֵ֥שׁ עֶבְרָתִ֖י כִּלִּיתִ֑ים דַּרְכָּם֙ בְּרֹאשָׁ֣ם נָתַ֔תִּי נְאֻ֖ם אֲדֹנָ֥י יְהֹוִֽה׃ תחילת פרק כג (יז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יח) בֶּן־ אָדָ֑ם שְׁתַּ֣יִם נָשִׁ֔ים בְּנ֥וֹת אֵם־ אַחַ֖ת הָיֽוּ׃ (יט) וַתִּזְנֶ֣ינָה בְמִצְרַ֔יִם בִּנְעוּרֵיהֶ֖ן זָנ֑וּ שָׁ֚מָּה מֹעֲכ֣וּ שְׁדֵיהֶ֔ן וְשָׁ֣ם עִשּׂ֔וּ דַּדֵּ֖י בְּתוּלֵיהֶֽן׃ (כ) וּשְׁמוֹתָ֗ן אָהֳלָ֤ה הַגְּדוֹלָה֙ וְאָהֳלִיבָ֣ה אֲחוֹתָ֔הּ וַתִּֽהְיֶ֣ינָה לִ֔י וַתֵּלַ֖דְנָה בָּנִ֣ים וּבָנ֑וֹת וּשְׁמוֹתָ֕ן שֹׁמְר֣וֹן אָהֳלָ֔ה וִירוּשָׁלִַ֖ם אָהֳלִיבָֽה׃ (כא) וַתִּ֥זֶן אָהֳלָ֖ה תַּחְתָּ֑י וַתַּעְגַּב֙ עַֽל־ מְאַהֲבֶ֔יהָ אֶל־ אַשּׁ֖וּר קְרוֹבִֽים׃ (כב) לְבֻשֵׁ֤י תְכֵ֙לֶת֙ פַּח֣וֹת וּסְגָנִ֔ים בַּח֥וּרֵי חֶ֖מֶד כֻּלָּ֑ם פָּרָשִׁ֕ים רֹכְבֵ֖י סוּסִֽים׃ (כג) וַתִּתֵּ֤ן תַּזְנוּתֶ֙יהָ֙ עֲלֵיהֶ֔ם מִבְחַ֥ר בְּנֵֽי־ אַשּׁ֖וּר כֻּלָּ֑ם וּבְכֹ֧ל אֲשֶׁר־ עָֽגְבָ֛ה בְּכָל־ גִּלּוּלֵיהֶ֖ם נִטְמָֽאָה׃ (כד) וְאֶת־ תַּזְנוּתֶ֤יהָ מִמִּצְרַ֙יִם֙ לֹ֣א עָזָ֔בָה כִּ֤י אוֹתָהּ֙ שָׁכְב֣וּ בִנְעוּרֶ֔יהָ וְהֵ֥מָּה עִשּׂ֖וּ דַּדֵּ֣י בְתוּלֶ֑יהָ וַיִּשְׁפְּכ֥וּ תַזְנוּתָ֖ם עָלֶֽיהָ׃ (כה) לָכֵ֥ן נְתַתִּ֖יהָ בְּיַד־ מְאַֽהֲבֶ֑יהָ בְּיַד֙ בְּנֵ֣י אַשּׁ֔וּר אֲשֶׁ֥ר עָגְבָ֖ה עֲלֵיהֶֽם׃ (כו) הֵמָּה֮ גִּלּ֣וּ עֶרְוָתָהּ֒ בָּנֶ֤יהָ וּבְנוֹתֶ֙יהָ֙ לָקָ֔חוּ וְאוֹתָ֖הּ בַּחֶ֣רֶב הָרָ֑גוּ וַתְּהִי־ שֵׁם֙ לַנָּשִׁ֔ים וּשְׁפוּטִ֖ים עָ֥שׂוּ בָֽהּ׃ (כז) וַתֵּ֙רֶא֙ אֲחוֹתָ֣הּ אָהֳלִיבָ֔ה וַתַּשְׁחֵ֥ת עַגְבָתָ֖הּ מִמֶּ֑נָּה וְאֶת־ תַּ֨זְנוּתֶ֔יהָ מִזְּנוּנֵ֖י אֲחוֹתָֽהּ׃ (כח) אֶל־ בְּנֵי֩ אַשּׁ֨וּר עָגָ֜בָה פַּח֨וֹת וּסְגָנִ֤ים קְרֹבִים֙ לְבֻשֵׁ֣י מִכְל֔וֹל פָּרָשִׁ֖ים רֹכְבֵ֣י סוּסִ֑ים בַּח֥וּרֵי חֶ֖מֶד כֻּלָּֽם׃ (כט) וָאֵ֖רֶא כִּ֣י נִטְמָ֑אָה דֶּ֥רֶךְ אֶחָ֖ד לִשְׁתֵּיהֶֽן׃ (ל) וַתּ֖וֹסֶף אֶל־ תַּזְנוּתֶ֑יהָ וַתֵּ֗רֶא אַנְשֵׁי֙ מְחֻקֶּ֣ה עַל־ הַקִּ֔יר צַלְמֵ֣י (כשדיים) כַשְׂדִּ֔ים חֲקֻקִ֖ים בַּשָּׁשַֽׁר׃ (לא) חֲגוֹרֵ֨י אֵז֜וֹר בְּמָתְנֵיהֶ֗ם סְרוּחֵ֤י טְבוּלִים֙ בְּרָ֣אשֵׁיהֶ֔ם מַרְאֵ֥ה שָׁלִשִׁ֖ים כֻּלָּ֑ם דְּמ֤וּת בְּנֵֽי־ בָבֶל֙ כַּשְׂדִּ֔ים אֶ֖רֶץ מוֹלַדְתָּֽם׃ (לב) (ותעגב) וַתַּעְגְּבָ֥ה עֲלֵיהֶ֖ם לְמַרְאֵ֣ה עֵינֶ֑יהָ וַתִּשְׁלַ֧ח מַלְאָכִ֛ים אֲלֵיהֶ֖ם כַּשְׂדִּֽימָה׃ (לג) וַיָּבֹ֨אוּ אֵלֶ֤יהָ בְנֵֽי־ בָבֶל֙ לְמִשְׁכַּ֣ב דֹּדִ֔ים וַיְטַמְּא֥וּ אוֹתָ֖הּ בְּתַזְנוּתָ֑ם וַתִּ֨טְמָא־ בָ֔ם וַתֵּ֥קַע נַפְשָׁ֖הּ מֵהֶֽם׃ (לד) וַתְּגַל֙ תַּזְנוּתֶ֔יהָ וַתְּגַ֖ל אֶת־ עֶרְוָתָ֑הּ וַתֵּ֤קַע נַפְשִׁי֙ מֵֽעָלֶ֔יהָ כַּאֲשֶׁ֛ר נָקְעָ֥ה נַפְשִׁ֖י מֵעַ֥ל אֲחוֹתָֽהּ׃ (לה) וַתַּרְבֶּ֖ה אֶת־ תַּזְנוּתֶ֑יהָ לִזְכֹּר֙ אֶת־ יְמֵ֣י נְעוּרֶ֔יהָ אֲשֶׁ֥ר זָנְתָ֖ה בְּאֶ֥רֶץ מִצְרָֽיִם׃ (לו) וַֽתַּעְגְּבָ֔ה עַ֖ל פִּֽלַגְשֵׁיהֶ֑ם אֲשֶׁ֤ר בְּשַׂר־ חֲמוֹרִים֙ בְּשָׂרָ֔ם וְזִרְמַ֥ת סוּסִ֖ים זִרְמָתָֽם׃ (לז) וַֽתִּפְקְדִ֔י אֵ֖ת זִמַּ֣ת נְעוּרָ֑יִךְ בַּעְשׂ֤וֹת מִמִּצְרַ֙יִם֙ דַּדַּ֔יִךְ לְמַ֖עַן שְׁדֵ֥י נְעוּרָֽיִךְ׃ (לח) לָכֵ֣ן אָהֳלִיבָ֗ה כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנְנִ֨י מֵעִ֤יר אֶת־ מְאַהֲבַ֙יִךְ֙ עָלַ֔יִךְ אֵ֛ת אֲשֶׁר־ נָקְעָ֥ה נַפְשֵׁ֖ךְ מֵהֶ֑ם וַהֲבֵאתִ֥ים עָלַ֖יִךְ מִסָּבִֽיב׃ (לט) בְּנֵ֧י בָבֶ֣ל וְכָל־ כַּשְׂדִּ֗ים פְּק֤וֹד וְשׁ֙וֹעַ֙ וְק֔וֹעַ כָּל־ בְּנֵ֥י אַשּׁ֖וּר אוֹתָ֑ם בַּח֨וּרֵי חֶ֜מֶד פַּח֤וֹת וּסְגָנִים֙ כֻּלָּ֔ם שָֽׁלִשִׁים֙ וּקְרוּאִ֔ים רֹכְבֵ֥י סוּסִ֖ים כֻּלָּֽם׃ (מ) וּבָ֣אוּ עָלַ֡יִךְ הֹ֠צֶן רֶ֤כֶב וְגַלְגַּל֙ וּבִקְהַ֣ל עַמִּ֔ים צִנָּ֤ה וּמָגֵן֙ וְקוֹבַ֔ע יָשִׂ֥ימוּ עָלַ֖יִךְ סָבִ֑יב וְנָתַתִּ֤י לִפְנֵיהֶם֙ מִשְׁפָּ֔ט וּשְׁפָט֖וּךְ בְּמִשְׁפְּטֵיהֶֽם׃ (מא) וְנָתַתִּ֨י קִנְאָתִ֜י בָּ֗ךְ וְעָשׂ֤וּ אוֹתָךְ֙ בְּחֵמָ֔ה אַפֵּ֤ךְ וְאָזְנַ֙יִךְ֙ יָסִ֔ירוּ וְאַחֲרִיתֵ֖ךְ בַּחֶ֣רֶב תִּפּ֑וֹל הֵ֗מָּה בָּנַ֤יִךְ וּבְנוֹתַ֙יִךְ֙ יִקָּ֔חוּ וְאַחֲרִיתֵ֖ךְ תֵּאָכֵ֥ל בָּאֵֽשׁ׃ (מב) וְהִפְשִׁיט֖וּךְ אֶת־ בְּגָדָ֑יִךְ וְלָקְח֖וּ כְּלֵ֥י תִפְאַרְתֵּֽךְ׃

יחזקאל סדר יד

תִסְפֹּד֙ וְלֹ֣א תִבְכֶּ֔ה וְל֥וֹא תָב֖וֹא דִּמְעָתֶֽךָ׃ (מ) הֵאָנֵ֣ק ׀ דֹּ֗ם מֵתִים֙ אֵ֣בֶל לֹֽא־ תַֽעֲשֶׂ֔ה פְאֵֽרְךָ֙ חֲב֣וֹשׁ עָלֶ֔יךָ וּנְעָלֶ֖יךָ תָּשִׂ֣ים בְּרַגְלֶ֑יךָ וְלֹ֤א תַעְטֶה֙ עַל־ שָׂפָ֔ם וְלֶ֥חֶם אֲנָשִׁ֖ים לֹ֥א תֹאכֵֽל׃ (מא) וָאֲדַבֵּ֤ר אֶל־ הָעָם֙ בַּבֹּ֔קֶר וַתָּ֥מָת אִשְׁתִּ֖י בָּעָ֑רֶב וָאַ֥עַשׂ בַּבֹּ֖קֶר כַּאֲשֶׁ֥ר צֻוֵּֽיתִי׃ (מב) וַיֹּאמְר֥וּ אֵלַ֖י הָעָ֑ם הֲלֹֽא־ תַגִּ֥יד לָ֙נוּ֙ מָה־ אֵ֣לֶּה לָּ֔נוּ כִּ֥י אַתָּ֖ה עֹשֶֽׂה׃ (מג) וָאֹמַ֖ר אֲלֵיהֶ֑ם דְּבַ֨ר־ יְהוָ֔ה הָיָ֥ה אֵלַ֖י לֵאמֹֽר׃ (מד) אֱמֹ֣ר ׀ לְבֵ֣ית יִשְׂרָאֵ֗ל כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנְנִ֨י מְחַלֵּ֤ל אֶת־ מִקְדָּשִׁי֙ גְּא֣וֹן עֻזְּכֶ֔ם מַחְמַ֥ד עֵֽינֵיכֶ֖ם וּמַחְמַ֣ל נַפְשְׁכֶ֑ם וּבְנֵיכֶ֧ם וּבְנֽוֹתֵיכֶ֛ם אֲשֶׁ֥ר עֲזַבְתֶּ֖ם בַּחֶ֥רֶב יִפֹּֽלוּ׃ (מה) וַעֲשִׂיתֶ֖ם כַּאֲשֶׁ֣ר עָשִׂ֑יתִי עַל־ שָׂפָם֙ לֹ֣א תַעְט֔וּ וְלֶ֥חֶם אֲנָשִׁ֖ים לֹ֥א תֹאכֵֽלוּ׃ (מו) וּפְאֵרֵכֶ֣ם עַל־ רָאשֵׁיכֶ֗ם וְנַֽעֲלֵיכֶם֙ בְּרַגְלֵיכֶ֔ם לֹ֥א תִסְפְּד֖וּ וְלֹ֣א תִבְכּ֑וּ וּנְמַקֹּתֶם֙ בַּעֲוֺנֹ֣תֵיכֶ֔ם וּנְהַמְתֶּ֖ם אִ֥ישׁ אֶל־ אָחִֽיו׃
