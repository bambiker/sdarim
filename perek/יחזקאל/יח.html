יחזקאל פרק יח לפי החלוקה לסדרים

פרק יח בספר יחזקאל (32 פסוקים) נמצא ביחזקאל סדר ט (פסוקים כט–לו) וביחזקאל סדר י (פסוקים א–כד).

יחזקאל סדר ט

(א) וְזָכַרְתִּ֨י אֲנִ֧י אֶת־ בְּרִיתִ֛י אוֹתָ֖ךְ בִּימֵ֣י נְעוּרָ֑יִךְ וַהֲקִמוֹתִ֥י לָ֖ךְ בְּרִ֥ית עוֹלָֽם׃ (ב) וְזָכַ֣רְתְּ אֶת־ דְּרָכַיִךְ֮ וְנִכְלַמְתְּ֒ בְּקַחְתֵּ֗ךְ אֶת־ אֲחוֹתַ֙יִךְ֙ הַגְּדֹל֣וֹת מִמֵּ֔ךְ אֶל־ הַקְּטַנּ֖וֹת מִמֵּ֑ךְ וְנָתַתִּ֨י אֶתְהֶ֥ן לָ֛ךְ לְבָנ֖וֹת וְלֹ֥א מִבְּרִיתֵֽךְ׃ (ג) וַהֲקִימוֹתִ֥י אֲנִ֛י אֶת־ בְּרִיתִ֖י אִתָּ֑ךְ וְיָדַ֖עַתְּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (ד) לְמַ֤עַן תִּזְכְּרִי֙ וָבֹ֔שְׁתְּ וְלֹ֨א יִֽהְיֶה־ לָּ֥ךְ עוֹד֙ פִּתְח֣וֹן פֶּ֔ה מִפְּנֵ֖י כְּלִמָּתֵ֑ךְ בְּכַפְּרִי־ לָךְ֙ לְכָל־ אֲשֶׁ֣ר עָשִׂ֔ית נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ה) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ו) בֶּן־ אָדָ֕ם ח֥וּד חִידָ֖ה וּמְשֹׁ֣ל מָשָׁ֑ל אֶל־ בֵּ֖ית יִשְׂרָאֵֽל׃ (ז) וְאָמַרְתָּ֞ כֹּה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה הַנֶּ֤שֶׁר הַגָּדוֹל֙ גְּד֤וֹל הַכְּנָפַ֙יִם֙ אֶ֣רֶךְ הָאֵ֔בֶר מָלֵא֙ הַנּוֹצָ֔ה אֲשֶׁר־ ל֖וֹ הָֽרִקְמָ֑ה בָּ֚א אֶל־ הַלְּבָנ֔וֹן וַיִּקַּ֖ח אֶת־ צַמֶּ֥רֶת הָאָֽרֶז׃ (ח) אֵ֛ת רֹ֥אשׁ יְנִֽיקוֹתָ֖יו קָטָ֑ף וַיְבִיאֵ֙הוּ֙ אֶל־ אֶ֣רֶץ כְּנַ֔עַן בְּעִ֥יר רֹכְלִ֖ים שָׂמֽוֹ׃ (ט) וַיִּקַּח֙ מִזֶּ֣רַע הָאָ֔רֶץ וַֽיִּתְּנֵ֖הוּ בִּשְׂדֵה־ זָ֑רַע קָ֚ח עַל־ מַ֣יִם רַבִּ֔ים צַפְצָפָ֖ה שָׂמֽוֹ׃ (י) וַיִּצְמַ֡ח וַיְהִי֩ לְגֶ֨פֶן סֹרַ֜חַת שִׁפְלַ֣ת קוֹמָ֗ה לִפְנ֤וֹת דָּלִיּוֹתָיו֙ אֵלָ֔יו וְשָׁרָשָׁ֖יו תַּחְתָּ֣יו יִֽהְי֑וּ וַתְּהִ֣י לְגֶ֔פֶן וַתַּ֣עַשׂ בַּדִּ֔ים וַתְּשַׁלַּ֖ח פֹּארֽוֹת׃ (יא) וַיְהִ֤י נֶֽשֶׁר־ אֶחָד֙ גָּד֔וֹל גְּד֥וֹל כְּנָפַ֖יִם וְרַב־ נוֹצָ֑ה וְהִנֵּה֩ הַגֶּ֨פֶן הַזֹּ֜את כָּֽפְנָ֧ה שָׁרֳשֶׁ֣יהָ עָלָ֗יו וְדָֽלִיּוֹתָיו֙ שִׁלְחָה־ לּ֔וֹ לְהַשְׁק֣וֹת אוֹתָ֔הּ מֵעֲרֻג֖וֹת מַטָּעָֽהּ׃ (יב) אֶל־ שָׂ֥דֶה טּ֛וֹב אֶל־ מַ֥יִם רַבִּ֖ים הִ֣יא שְׁתוּלָ֑ה לַעֲשׂ֤וֹת עָנָף֙ וְלָשֵׂ֣את פֶּ֔רִי לִהְי֖וֹת לְגֶ֥פֶן אַדָּֽרֶת׃ (יג) אֱמֹ֗ר כֹּ֥ה אָמַ֛ר אֲדֹנָ֥י יְהֹוִ֖ה תִּצְלָ֑ח הֲלוֹא֩ אֶת־ שָׁרָשֶׁ֨יהָ יְנַתֵּ֜ק וְאֶת־ פִּרְיָ֣הּ ׀ יְקוֹסֵ֣ס וְיָבֵ֗שׁ כָּל־ טַרְפֵּ֤י צִמְחָהּ֙ תִּיבָ֔שׁ וְלֹֽא־ בִזְרֹ֤עַ גְּדוֹלָה֙ וּבְעַם־ רָ֔ב לְמַשְׂא֥וֹת אוֹתָ֖הּ מִשָּׁרָשֶֽׁיהָ׃ (יד) וְהִנֵּ֥ה שְׁתוּלָ֖ה הֲתִצְלָ֑ח הֲלוֹא֩ כְגַ֨עַת בָּ֜הּ ר֤וּחַ הַקָּדִים֙ תִּיבַ֣שׁ יָבֹ֔שׁ עַל־ עֲרֻגֹ֥ת צִמְחָ֖הּ תִּיבָֽשׁ׃ (טו) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (טז) אֱמָר־ נָא֙ לְבֵ֣ית הַמֶּ֔רִי הֲלֹ֥א יְדַעְתֶּ֖ם מָה־ אֵ֑לֶּה אֱמֹ֗ר הִנֵּה־ בָ֨א מֶֽלֶךְ־ בָּבֶ֤ל יְרוּשָׁלִַ֙ם֙ וַיִּקַּ֤ח אֶת־ מַלְכָּהּ֙ וְאֶת־ שָׂרֶ֔יהָ וַיָּבֵ֥א אוֹתָ֛ם אֵלָ֖יו בָּבֶֽלָה׃ (יז) וַיִּקַּח֙ מִזֶּ֣רַע הַמְּלוּכָ֔ה וַיִּכְרֹ֥ת אִתּ֖וֹ בְּרִ֑ית וַיָּבֵ֤א אֹתוֹ֙ בְּאָלָ֔ה וְאֶת־ אֵילֵ֥י הָאָ֖רֶץ לָקָֽח׃ (יח) לִֽהְיוֹת֙ מַמְלָכָ֣ה שְׁפָלָ֔ה לְבִלְתִּ֖י הִתְנַשֵּׂ֑א לִשְׁמֹ֥ר אֶת־ בְּרִית֖וֹ לְעָמְדָֽהּ׃ (יט) וַיִּמְרָד־ בּ֗וֹ לִשְׁלֹ֤חַ מַלְאָכָיו֙ מִצְרַ֔יִם לָֽתֶת־ ל֥וֹ סוּסִ֖ים וְעַם־ רָ֑ב הֲיִצְלָ֤ח הֲיִמָּלֵט֙ הָעֹשֵׂ֣ה אֵ֔לֶּה וְהֵפֵ֥ר בְּרִ֖ית וְנִמְלָֽט׃ (כ) חַי־ אָ֗נִי נְאֻם֮ אֲדֹנָ֣י יְהוִה֒ אִם־ לֹ֗א בִּמְקוֹם֙ הַמֶּ֙לֶךְ֙ הַמַּמְלִ֣יךְ אֹת֔וֹ אֲשֶׁ֤ר בָּזָה֙ אֶת־ אָ֣לָת֔וֹ וַאֲשֶׁ֥ר הֵפֵ֖ר אֶת־ בְּרִית֑וֹ אִתּ֥וֹ בְתוֹךְ־ בָּבֶ֖ל יָמֽוּת׃ (כא) וְלֹא֩ בְחַ֨יִל גָּד֜וֹל וּבְקָהָ֣ל רָ֗ב יַעֲשֶׂ֨ה אוֹת֤וֹ פַרְעֹה֙ בַּמִּלְחָמָ֔ה בִּשְׁפֹּ֥ךְ סֹלְלָ֖ה וּבִבְנ֣וֹת דָּיֵ֑ק לְהַכְרִ֖ית נְפָשׁ֥וֹת רַבּֽוֹת׃ (כב) וּבָזָ֥ה אָלָ֖ה לְהָפֵ֣ר בְּרִ֑ית וְהִנֵּ֨ה נָתַ֥ן יָד֛וֹ וְכָל־ אֵ֥לֶּה עָשָׂ֖ה לֹ֥א יִמָּלֵֽט׃ (כג) לָכֵ֞ן כֹּה־ אָמַ֨ר אֲדֹנָ֣י יְהוִה֮ חַי־ אָנִי֒ אִם־ לֹ֗א אָֽלָתִי֙ אֲשֶׁ֣ר בָּזָ֔ה וּבְרִיתִ֖י אֲשֶׁ֣ר הֵפִ֑יר וּנְתַתִּ֖יו בְּרֹאשֽׁוֹ׃ (כד) וּפָרַשְׂתִּ֤י עָלָיו֙ רִשְׁתִּ֔י וְנִתְפַּ֖שׂ בִּמְצֽוּדָתִ֑י וַהֲבִיאוֹתִ֣יהוּ בָבֶ֗לָה וְנִשְׁפַּטְתִּ֤י אִתּוֹ֙ שָׁ֔ם מַעֲל֖וֹ אֲשֶׁ֥ר מָֽעַל־ בִּֽי׃ (כה) וְאֵ֨ת כָּל־ (מברחו) מִבְרָחָ֤יו בְּכָל־ אֲגַפָּיו֙ בַּחֶ֣רֶב יִפֹּ֔לוּ וְהַנִּשְׁאָרִ֖ים לְכָל־ ר֣וּחַ יִפָּרֵ֑שׂוּ וִידַעְתֶּ֕ם כִּ֛י אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (כו) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה וְלָקַ֣חְתִּי אָ֗נִי מִצַּמֶּ֧רֶת הָאֶ֛רֶז הָרָמָ֖ה וְנָתָ֑תִּי מֵרֹ֤אשׁ יֹֽנְקוֹתָיו֙ רַ֣ךְ אֶקְטֹ֔ף וְשָׁתַ֣לְתִּי אָ֔נִי עַ֥ל הַר־ גָּבֹ֖הַ וְתָלֽוּל׃ (כז) בְּהַ֨ר מְר֤וֹם יִשְׂרָאֵל֙ אֶשְׁתֳּלֶ֔נּוּ וְנָשָׂ֤א עָנָף֙ וְעָ֣שָׂה פֶ֔רִי וְהָיָ֖ה לְאֶ֣רֶז אַדִּ֑יר וְשָׁכְנ֣וּ תַחְתָּ֗יו כֹּ֚ל צִפּ֣וֹר כָּל־ כָּנָ֔ף בְּצֵ֥ל דָּלִיּוֹתָ֖יו תִּשְׁכֹּֽנָּה׃ (כח) וְֽיָדְע֞וּ כָּל־ עֲצֵ֣י הַשָּׂדֶ֗ה כִּ֣י אֲנִ֤י יְהוָה֙ הִשְׁפַּ֣לְתִּי ׀ עֵ֣ץ גָּבֹ֗הַ הִגְבַּ֙הְתִּי֙ עֵ֣ץ שָׁפָ֔ל הוֹבַ֙שְׁתִּי֙ עֵ֣ץ לָ֔ח וְהִפְרַ֖חְתִּי עֵ֣ץ יָבֵ֑שׁ אֲנִ֥י יְהוָ֖ה דִּבַּ֥רְתִּי וְעָשִֽׂיתִי׃ תחילת פרק יח (כט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ל) מַה־ לָּכֶ֗ם אַתֶּם֙ מֹֽשְׁלִים֙ אֶת־ הַמָּשָׁ֣ל הַזֶּ֔ה עַל־ אַדְמַ֥ת יִשְׂרָאֵ֖ל לֵאמֹ֑ר אָבוֹת֙ יֹ֣אכְלוּ בֹ֔סֶר וְשִׁנֵּ֥י הַבָּנִ֖ים תִּקְהֶֽינָה׃ (לא) חַי־ אָ֕נִי נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה אִם־ יִֽהְיֶ֨ה לָכֶ֜ם ע֗וֹד מְשֹׁ֛ל הַמָּשָׁ֥ל הַזֶּ֖ה בְּיִשְׂרָאֵֽל׃ (לב) הֵ֤ן כָּל־ הַנְּפָשׁוֹת֙ לִ֣י הֵ֔נָּה כְּנֶ֧פֶשׁ הָאָ֛ב וּכְנֶ֥פֶשׁ הַבֵּ֖ן לִי־ הֵ֑נָּה הַנֶּ֥פֶשׁ הַחֹטֵ֖את הִ֥יא תָמֽוּת׃ (לג) וְאִ֖ישׁ כִּי־ יִהְיֶ֣ה צַדִּ֑יק וְעָשָׂ֥ה מִשְׁפָּ֖ט וּצְדָקָֽה׃ (לד) אֶל־ הֶֽהָרִים֙ לֹ֣א אָכָ֔ל וְעֵינָיו֙ לֹ֣א נָשָׂ֔א אֶל־ גִּלּוּלֵ֖י בֵּ֣ית יִשְׂרָאֵ֑ל וְאֶת־ אֵ֤שֶׁת רֵעֵ֙הוּ֙ לֹ֣א טִמֵּ֔א וְאֶל־ אִשָּׁ֥ה נִדָּ֖ה לֹ֥א יִקְרָֽב׃ (לה) וְאִישׁ֙ לֹ֣א יוֹנֶ֔ה חֲבֹלָת֥וֹ חוֹב֙ יָשִׁ֔יב גְּזֵלָ֖ה לֹ֣א יִגְזֹ֑ל לַחְמוֹ֙ לְרָעֵ֣ב יִתֵּ֔ן וְעֵירֹ֖ם יְכַסֶּה־ בָּֽגֶד׃ (לו) בַּנֶּ֣שֶׁךְ לֹֽא־ יִתֵּ֗ן וְתַרְבִּית֙ לֹ֣א יִקָּ֔ח מֵעָ֖וֶל יָשִׁ֣יב יָד֑וֹ מִשְׁפַּ֤ט אֱמֶת֙ יַֽעֲשֶׂ֔ה בֵּ֥ין אִ֖ישׁ לְאִֽישׁ׃

יחזקאל סדר י

(א) בְּחֻקּוֹתַ֧י יְהַלֵּ֛ךְ וּמִשְׁפָּטַ֥י שָׁמַ֖ר לַעֲשׂ֣וֹת אֱמֶ֑ת צַדִּ֥יק הוּא֙ חָיֹ֣ה יִֽחְיֶ֔ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ב) וְהוֹלִ֥יד בֵּן־ פָּרִ֖יץ שֹׁפֵ֣ךְ דָּ֑ם וְעָ֣שָׂה אָ֔ח מֵאַחַ֖ד מֵאֵֽלֶּה׃ (ג) וְה֕וּא אֶת־ כָּל־ אֵ֖לֶּה לֹ֣א עָשָׂ֑ה כִּ֣י גַ֤ם אֶל־ הֶֽהָרִים֙ אָכַ֔ל וְאֶת־ אֵ֥שֶׁת רֵעֵ֖הוּ טִמֵּֽא׃ (ד) עָנִ֤י וְאֶבְיוֹן֙ הוֹנָ֔ה גְּזֵל֣וֹת גָּזָ֔ל חֲבֹ֖ל לֹ֣א יָשִׁ֑יב וְאֶל־ הַגִּלּוּלִים֙ נָשָׂ֣א עֵינָ֔יו תּוֹעֵבָ֖ה עָשָֽׂה׃ (ה) בַּנֶּ֧שֶׁךְ נָתַ֛ן וְתַרְבִּ֥ית לָקַ֖ח וָחָ֑י לֹ֣א יִֽחְיֶ֗ה אֵ֣ת כָּל־ הַתּוֹעֵב֤וֹת הָאֵ֙לֶּה֙ עָשָׂ֔ה מ֣וֹת יוּמָ֔ת דָּמָ֖יו בּ֥וֹ יִהְיֶֽה׃ (ו) וְהִנֵּה֙ הוֹלִ֣יד בֵּ֔ן וַיַּ֕רְא אֶת־ כָּל־ חַטֹּ֥את אָבִ֖יו אֲשֶׁ֣ר עָשָׂ֑ה וַיִּרְאֶ֕ה וְלֹ֥א יַעֲשֶׂ֖ה כָּהֵֽן׃ (ז) עַל־ הֶֽהָרִים֙ לֹ֣א אָכָ֔ל וְעֵינָיו֙ לֹ֣א נָשָׂ֔א אֶל־ גִּלּוּלֵ֖י בֵּ֣ית יִשְׂרָאֵ֑ל אֶת־ אֵ֥שֶׁת רֵעֵ֖הוּ לֹ֥א טִמֵּֽא׃ (ח) וְאִישׁ֙ לֹ֣א הוֹנָ֔ה חֲבֹל֙ לֹ֣א חָבָ֔ל וּגְזֵלָ֖ה לֹ֣א גָזָ֑ל לַחְמוֹ֙ לְרָעֵ֣ב נָתָ֔ן וְעֵר֖וֹם כִּסָּה־ בָֽגֶד׃ (ט) מֵעָנִ֞י הֵשִׁ֣יב יָד֗וֹ נֶ֤שֶׁךְ וְתַרְבִּית֙ לֹ֣א לָקָ֔ח מִשְׁפָּטַ֣י עָשָׂ֔ה בְּחֻקּוֹתַ֖י הָלָ֑ךְ ה֗וּא לֹ֥א יָמ֛וּת בַּעֲוֺ֥ן אָבִ֖יו חָיֹ֥ה יִחְיֶֽה׃ (י) אָבִ֞יו כִּֽי־ עָ֣שַׁק עֹ֗שֶׁק גָּזַל֙ גֵּ֣זֶל אָ֔ח וַאֲשֶׁ֥ר לֹא־ ט֛וֹב עָשָׂ֖ה בְּת֣וֹךְ עַמָּ֑יו וְהִנֵּה־ מֵ֖ת בַּעֲוֺנֽוֹ׃ (יא) וַאֲמַרְתֶּ֕ם מַדֻּ֛עַ לֹא־ נָשָׂ֥א הַבֵּ֖ן בַּעֲוֺ֣ן הָאָ֑ב וְהַבֵּ֞ן מִשְׁפָּ֧ט וּצְדָקָ֣ה עָשָׂ֗ה אֵ֣ת כָּל־ חֻקּוֹתַ֥י שָׁמַ֛ר וַיַּעֲשֶׂ֥ה אֹתָ֖ם חָיֹ֥ה יִחְיֶֽה׃ (יב) הַנֶּ֥פֶשׁ הַחֹטֵ֖את הִ֣יא תָמ֑וּת בֵּ֞ן לֹא־ יִשָּׂ֣א ׀ בַּעֲוֺ֣ן הָאָ֗ב וְאָב֙ לֹ֤א יִשָּׂא֙ בַּעֲוֺ֣ן הַבֵּ֔ן צִדְקַ֤ת הַצַּדִּיק֙ עָלָ֣יו תִּֽהְיֶ֔ה וְרִשְׁעַ֥ת (רשע) הָרָשָׁ֖ע עָלָ֥יו תִּֽהְיֶֽה׃ (יג) וְהָרָשָׁ֗ע כִּ֤י יָשׁוּב֙ מִכָּל־ (חטאתו) חַטֹּאתָיו֙ אֲשֶׁ֣ר עָשָׂ֔ה וְשָׁמַר֙ אֶת־ כָּל־ חֻקּוֹתַ֔י וְעָשָׂ֥ה מִשְׁפָּ֖ט וּצְדָקָ֑ה חָיֹ֥ה יִחְיֶ֖ה לֹ֥א יָמֽוּת׃ (יד) כָּל־ פְּשָׁעָיו֙ אֲשֶׁ֣ר עָשָׂ֔ה לֹ֥א יִזָּכְר֖וּ ל֑וֹ בְּצִדְקָת֥וֹ אֲשֶׁר־ עָשָׂ֖ה יִֽחְיֶֽה׃ (טו) הֶחָפֹ֤ץ אֶחְפֹּץ֙ מ֣וֹת רָשָׁ֔ע נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה הֲל֛וֹא בְּשׁוּב֥וֹ מִדְּרָכָ֖יו וְחָיָֽה׃ (טז) וּבְשׁ֨וּב צַדִּ֤יק מִצִּדְקָתוֹ֙ וְעָ֣שָׂה עָ֔וֶל כְּכֹ֨ל הַתּוֹעֵב֜וֹת אֲשֶׁר־ עָשָׂ֧ה הָרָשָׁ֛ע יַעֲשֶׂ֖ה וָחָ֑י כָּל־ (צדקתו) צִדְקֹתָ֤יו אֲשֶׁר־ עָשָׂה֙ לֹ֣א תִזָּכַ֔רְנָה בְּמַעֲל֧וֹ אֲשֶׁר־ מָעַ֛ל וּבְחַטָּאת֥וֹ אֲשֶׁר־ חָטָ֖א בָּ֥ם יָמֽוּת׃ (יז) וַאֲמַרְתֶּ֕ם לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י שִׁמְעוּ־ נָא֙ בֵּ֣ית יִשְׂרָאֵ֔ל הֲדַרְכִּי֙ לֹ֣א יִתָּכֵ֔ן הֲלֹ֥א דַרְכֵיכֶ֖ם לֹ֥א יִתָּכֵֽנוּ׃ (יח) בְּשׁוּב־ צַדִּ֧יק מִצִּדְקָת֛וֹ וְעָ֥שָׂה עָ֖וֶל וּמֵ֣ת עֲלֵיהֶ֑ם בְּעַוְל֥וֹ אֲשֶׁר־ עָשָׂ֖ה יָמֽוּת׃ (יט) וּבְשׁ֣וּב רָשָׁ֗ע מֵֽרִשְׁעָתוֹ֙ אֲשֶׁ֣ר עָשָׂ֔ה וַיַּ֥עַשׂ מִשְׁפָּ֖ט וּצְדָקָ֑ה ה֖וּא אֶת־ נַפְשׁ֥וֹ יְחַיֶּֽה׃ (כ) וַיִּרְאֶ֣ה (וישוב) וַיָּ֔שָׁב מִכָּל־ פְּשָׁעָ֖יו אֲשֶׁ֣ר עָשָׂ֑ה חָי֥וֹ יִחְיֶ֖ה לֹ֥א יָמֽוּת׃ (כא) וְאָֽמְרוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י הַדְּרָכַ֞י לֹ֤א יִתָּֽכְנּוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל הֲלֹ֥א דַרְכֵיכֶ֖ם לֹ֥א יִתָּכֵֽן׃ (כב) לָכֵן֩ אִ֨ישׁ כִּדְרָכָ֜יו אֶשְׁפֹּ֤ט אֶתְכֶם֙ בֵּ֣ית יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה שׁ֤וּבוּ וְהָשִׁ֙יבוּ֙ מִכָּל־ פִּשְׁעֵיכֶ֔ם וְלֹֽא־ יִהְיֶ֥ה לָכֶ֛ם לְמִכְשׁ֖וֹל עָוֺֽן׃ (כג) הַשְׁלִ֣יכוּ מֵעֲלֵיכֶ֗ם אֶת־ כָּל־ פִּשְׁעֵיכֶם֙ אֲשֶׁ֣ר פְּשַׁעְתֶּ֣ם בָּ֔ם וַעֲשׂ֥וּ לָכֶ֛ם לֵ֥ב חָדָ֖שׁ וְר֣וּחַ חֲדָשָׁ֑ה וְלָ֥מָּה תָמֻ֖תוּ בֵּ֥ית יִשְׂרָאֵֽל׃ (כד) כִּ֣י לֹ֤א אֶחְפֹּץ֙ בְּמ֣וֹת הַמֵּ֔ת נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה וְהָשִׁ֖יבוּ וִֽחְיֽוּ׃ סוף פרק יח (כה) וְאַתָּה֙ שָׂ֣א קִינָ֔ה אֶל־ נְשִׂיאֵ֖י יִשְׂרָאֵֽל׃ (כו) וְאָמַרְתָּ֗ מָ֤ה אִמְּךָ֙ לְבִיָּ֔א בֵּ֥ין אֲרָי֖וֹת רָבָ֑צָה בְּת֥וֹךְ כְּפִרִ֖ים רִבְּתָ֥ה גוּרֶֽיהָ׃ (כז) וַתַּ֛עַל אֶחָ֥ד מִגֻּרֶ֖יהָ כְּפִ֣יר הָיָ֑ה וַיִּלְמַ֥ד לִטְרָף־ טֶ֖רֶף אָדָ֥ם אָכָֽל׃ (כח) וַיִּשְׁמְע֥וּ אֵלָ֛יו גּוֹיִ֖ם בְּשַׁחְתָּ֣ם נִתְפָּ֑שׂ וַיְבִאֻ֥הוּ בַֽחַחִ֖ים אֶל־ אֶ֥רֶץ מִצְרָֽיִם׃ (כט) וַתֵּ֙רֶא֙ כִּ֣י נֽוֹחֲלָ֔ה אָבְדָ֖ה תִּקְוָתָ֑הּ וַתִּקַּ֛ח אֶחָ֥ד מִגֻּרֶ֖יהָ כְּפִ֥יר שָׂמָֽתְהוּ׃ (ל) וַיִּתְהַלֵּ֥ךְ בְּתוֹךְ־ אֲרָי֖וֹת כְּפִ֣יר הָיָ֑ה וַיִּלְמַ֥ד לִטְרָף־ טֶ֖רֶף אָדָ֥ם אָכָֽל׃ (לא) וַיֵּ֙דַע֙ אַלְמְנוֹתָ֔יו וְעָרֵיהֶ֖ם הֶחֱרִ֑יב וַתֵּ֤שַׁם אֶ֙רֶץ֙ וּמְלֹאָ֔הּ מִקּ֖וֹל שַׁאֲגָתֽוֹ׃ (לב) וַיִּתְּנ֨וּ עָלָ֥יו גּוֹיִ֛ם סָבִ֖יב מִמְּדִינ֑וֹת וַֽיִּפְרְשׂ֥וּ עָלָ֛יו רִשְׁתָּ֖ם בְּשַׁחְתָּ֥ם נִתְפָּֽשׂ׃ (לג) וַֽיִּתְּנֻ֤הוּ בַסּוּגַר֙ בַּֽחַחִ֔ים וַיְבִאֻ֖הוּ אֶל־ מֶ֣לֶךְ בָּבֶ֑ל יְבִאֻ֙הוּ֙ בַּמְּצֹד֔וֹת לְמַ֗עַן לֹא־ יִשָּׁמַ֥ע קוֹל֛וֹ ע֖וֹד אֶל־ הָרֵ֥י יִשְׂרָאֵֽל׃ (לד) אִמְּךָ֥ כַגֶּ֛פֶן בְּדָמְךָ֖ עַל־ מַ֣יִם שְׁתוּלָ֑ה פֹּֽרִיָּה֙ וַֽעֲנֵפָ֔ה הָיְתָ֖ה מִמַּ֥יִם רַבִּֽים׃ (לה) וַיִּֽהְיוּ־ לָ֞הּ מַטּ֣וֹת עֹ֗ז אֶל־ שִׁבְטֵי֙ מֹֽשְׁלִ֔ים וַתִּגְבַּ֥הּ קֽוֹמָת֖וֹ עַל־ בֵּ֣ין עֲבֹתִ֑ים וַיֵּרָ֣א בְגָבְה֔וֹ בְּרֹ֖ב דָּלִיֹּתָֽיו׃ (לו) וַתֻּתַּ֤שׁ בְּחֵמָה֙ לָאָ֣רֶץ הֻשְׁלָ֔כָה וְר֥וּחַ הַקָּדִ֖ים הוֹבִ֣ישׁ פִּרְיָ֑הּ הִתְפָּרְק֧וּ וְיָבֵ֛שׁוּ מַטֵּ֥ה עֻזָּ֖הּ אֵ֥שׁ אֲכָלָֽתְהוּ׃ (לז) וְעַתָּ֖ה שְׁתוּלָ֣ה בַמִּדְבָּ֑ר בְּאֶ֖רֶץ צִיָּ֥ה וְצָמָֽא׃ (לח) וַתֵּצֵ֨א אֵ֜שׁ מִמַּטֵּ֤ה בַדֶּ֙יהָ֙ פִּרְיָ֣הּ אָכָ֔לָה וְלֹא־ הָ֥יָה בָ֛הּ מַטֵּה־ עֹ֖ז שֵׁ֣בֶט לִמְשׁ֑וֹל קִ֥ינָה הִ֖יא וַתְּהִ֥י לְקִינָֽה׃
