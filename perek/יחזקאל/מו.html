יחזקאל פרק מו לפי החלוקה לסדרים

פרק מו בספר יחזקאל (24 פסוקים) נמצא ביחזקאל סדר כח (פסוקים יב–לה).

יחזקאל סדר כח

(א) וְשֶׂה־ אַחַ֨ת מִן־ הַצֹּ֤אן מִן־ הַמָּאתַ֙יִם֙ מִמַּשְׁקֵ֣ה יִשְׂרָאֵ֔ל לְמִנְחָ֖ה וּלְעוֹלָ֣ה וְלִשְׁלָמִ֑ים לְכַפֵּ֣ר עֲלֵיהֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ב) כֹּ֚ל הָעָ֣ם הָאָ֔רֶץ יִהְי֖וּ אֶל־ הַתְּרוּמָ֣ה הַזֹּ֑את לַנָּשִׂ֖יא בְּיִשְׂרָאֵֽל׃ (ג) וְעַֽל־ הַנָּשִׂ֣יא יִהְיֶ֗ה הָעוֹל֣וֹת וְהַמִּנְחָה֮ וְהַנֵּסֶךְ֒ בַּחַגִּ֤ים וּבֶחֳדָשִׁים֙ וּבַשַּׁבָּת֔וֹת בְּכָֽל־ מוֹעֲדֵ֖י בֵּ֣ית יִשְׂרָאֵ֑ל הֽוּא־ יַעֲשֶׂ֞ה אֶת־ הַחַטָּ֣את וְאֶת־ הַמִּנְחָ֗ה וְאֶת־ הָֽעוֹלָה֙ וְאֶת־ הַשְּׁלָמִ֔ים לְכַפֵּ֖ר בְּעַ֥ד בֵּֽית־ יִשְׂרָאֵֽל׃ (ד) כֹּה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ בָּֽרִאשׁוֹן֙ בְּאֶחָ֣ד לַחֹ֔דֶשׁ תִּקַּ֥ח פַּר־ בֶּן־ בָּקָ֖ר תָּמִ֑ים וְחִטֵּאתָ֖ אֶת־ הַמִּקְדָּֽשׁ׃ (ה) וְלָקַ֨ח הַכֹּהֵ֜ן מִדַּ֣ם הַחַטָּ֗את וְנָתַן֙ אֶל־ מְזוּזַ֣ת הַבַּ֔יִת וְאֶל־ אַרְבַּ֛ע פִּנּ֥וֹת הָעֲזָרָ֖ה לַמִּזְבֵּ֑חַ וְעַ֨ל־ מְזוּזַ֔ת שַׁ֖עַר הֶחָצֵ֥ר הַפְּנִימִֽית׃ (ו) וְכֵ֤ן תַּֽעֲשֶׂה֙ בְּשִׁבְעָ֣ה בַחֹ֔דֶשׁ מֵאִ֥ישׁ שֹׁגֶ֖ה וּמִפֶּ֑תִי וְכִפַּרְתֶּ֖ם אֶת־ הַבָּֽיִת׃ (ז) בָּ֠רִאשׁוֹן בְּאַרְבָּעָ֨ה עָשָׂ֥ר יוֹם֙ לַחֹ֔דֶשׁ יִהְיֶ֥ה לָכֶ֖ם הַפָּ֑סַח חָ֕ג שְׁבֻע֣וֹת יָמִ֔ים מַצּ֖וֹת יֵאָכֵֽל׃ (ח) וְעָשָׂ֤ה הַנָּשִׂיא֙ בַּיּ֣וֹם הַה֔וּא בַּעֲד֕וֹ וּבְעַ֖ד כָּל־ עַ֣ם הָאָ֑רֶץ פַּ֖ר חַטָּֽאת׃ (ט) וְשִׁבְעַ֨ת יְמֵֽי־ הֶחָ֜ג יַעֲשֶׂ֧ה עוֹלָ֣ה לַֽיהוָ֗ה שִׁבְעַ֣ת פָּ֠רִים וְשִׁבְעַ֨ת אֵילִ֤ים תְּמִימִם֙ לַיּ֔וֹם שִׁבְעַ֖ת הַיָּמִ֑ים וְחַטָּ֕את שְׂעִ֥יר עִזִּ֖ים לַיּֽוֹם׃ (י) וּמִנְחָ֗ה אֵיפָ֥ה לַפָּ֛ר וְאֵיפָ֥ה לָאַ֖יִל יַֽעֲשֶׂ֑ה וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (יא) בַּשְּׁבִיעִ֡י בַּחֲמִשָּׁה֩ עָשָׂ֨ר י֤וֹם לַחֹ֙דֶשׁ֙ בֶּחָ֔ג יַעֲשֶׂ֥ה כָאֵ֖לֶּה שִׁבְעַ֣ת הַיָּמִ֑ים כַּֽחַטָּאת֙ כָּעֹלָ֔ה וְכַמִּנְחָ֖ה וְכַשָּֽׁמֶן׃ תחילת פרק מו (יב) כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ שַׁ֜עַר הֶחָצֵ֤ר הַפְּנִימִית֙ הַפֹּנֶ֣ה קָדִ֔ים יִהְיֶ֣ה סָג֔וּר שֵׁ֖שֶׁת יְמֵ֣י הַֽמַּעֲשֶׂ֑ה וּבְי֤וֹם הַשַּׁבָּת֙ יִפָּתֵ֔חַ וּבְי֥וֹם הַחֹ֖דֶשׁ יִפָּתֵֽחַ׃ (יג) וּבָ֣א הַנָּשִׂ֡יא דֶּרֶךְ֩ אוּלָ֨ם הַשַּׁ֜עַר מִח֗וּץ וְעָמַד֙ עַל־ מְזוּזַ֣ת הַשַּׁ֔עַר וְעָשׂ֣וּ הַכֹּהֲנִ֗ים אֶת־ עֽוֹלָתוֹ֙ וְאֶת־ שְׁלָמָ֔יו וְהִֽשְׁתַּחֲוָ֛ה עַל־ מִפְתַּ֥ן הַשַּׁ֖עַר וְיָצָ֑א וְהַשַּׁ֥עַר לֹֽא־ יִסָּגֵ֖ר עַד־ הָעָֽרֶב׃ (יד) וְהִשְׁתַּחֲו֣וּ עַם־ הָאָ֗רֶץ פֶּ֚תַח הַשַּׁ֣עַר הַה֔וּא בַּשַּׁבָּת֖וֹת וּבֶחֳדָשִׁ֑ים לִפְנֵ֖י יְהוָֽה׃ (טו) וְהָ֣עֹלָ֔ה אֲשֶׁר־ יַקְרִ֥ב הַנָּשִׂ֖יא לַֽיהוָ֑ה בְּי֣וֹם הַשַּׁבָּ֗ת שִׁשָּׁ֧ה כְבָשִׂ֛ים תְּמִימִ֖ם וְאַ֥יִל תָּמִֽים׃ (טז) וּמִנְחָה֙ אֵיפָ֣ה לָאַ֔יִל וְלַכְּבָשִׂ֥ים מִנְחָ֖ה מַתַּ֣ת יָד֑וֹ וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (יז) וּבְי֣וֹם הַחֹ֔דֶשׁ פַּ֥ר בֶּן־ בָּקָ֖ר תְּמִימִ֑ם וְשֵׁ֧שֶׁת כְּבָשִׂ֛ם וָאַ֖יִל תְּמִימִ֥ם יִהְיֽוּ׃ (יח) וְאֵיפָ֨ה לַפָּ֜ר וְאֵיפָ֤ה לָאַ֙יִל֙ יַעֲשֶׂ֣ה מִנְחָ֔ה וְלַ֨כְּבָשִׂ֔ים כַּאֲשֶׁ֥ר תַּשִּׂ֖יג יָד֑וֹ וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (יט) וּבְב֖וֹא הַנָּשִׂ֑יא דֶּ֣רֶךְ אוּלָ֤ם הַשַּׁ֙עַר֙ יָב֔וֹא וּבְדַרְכּ֖וֹ יֵצֵֽא׃ (כ) וּבְב֨וֹא עַם־ הָאָ֜רֶץ לִפְנֵ֣י יְהוָה֮ בַּמּֽוֹעֲדִים֒ הַבָּ֡א דֶּרֶךְ־ שַׁ֨עַר צָפ֜וֹן לְהִֽשְׁתַּחֲוֺ֗ת יֵצֵא֙ דֶּרֶךְ־ שַׁ֣עַר נֶ֔גֶב וְהַבָּא֙ דֶּרֶךְ־ שַׁ֣עַר נֶ֔גֶב יֵצֵ֖א דֶּרֶךְ־ שַׁ֣עַר צָפ֑וֹנָה לֹ֣א יָשׁ֗וּב דֶּ֤רֶךְ הַשַּׁ֙עַר֙ אֲשֶׁר־ בָּ֣א ב֔וֹ כִּ֥י נִכְח֖וֹ (יצאו) יֵצֵֽא׃ (כא) וְֽהַנָּשִׂ֑יא בְּתוֹכָ֤ם בְּבוֹאָם֙ יָב֔וֹא וּבְצֵאתָ֖ם יֵצֵֽאוּ׃ (כב) וּבַחַגִּ֣ים וּבַמּוֹעֲדִ֗ים תִּהְיֶ֤ה הַמִּנְחָה֙ אֵיפָ֤ה לַפָּר֙ וְאֵיפָ֣ה לָאַ֔יִל וְלַכְּבָשִׂ֖ים מַתַּ֣ת יָד֑וֹ וְשֶׁ֖מֶן הִ֥ין לָאֵיפָֽה׃ (כג) וְכִֽי־ יַעֲשֶׂה֩ הַנָּשִׂ֨יא נְדָבָ֜ה עוֹלָ֣ה אֽוֹ־ שְׁלָמִים֮ נְדָבָ֣ה לַֽיהוָה֒ וּפָ֣תַֽח ל֗וֹ אֶת הַשַּׁ֙עַר֙ הַפֹּנֶ֣ה קָדִ֔ים וְעָשָׂ֤ה אֶת־ עֹֽלָתוֹ֙ וְאֶת־ שְׁלָמָ֔יו כַּאֲשֶׁ֥ר יַעֲשֶׂ֖ה בְּי֣וֹם הַשַּׁבָּ֑ת וְיָצָ֛א וְסָגַ֥ר אֶת־ הַשַּׁ֖עַר אַחֲרֵ֥י צֵאתֽוֹ׃ (כד) וְכֶ֨בֶשׂ בֶּן־ שְׁנָת֜וֹ תָּמִ֗ים תַּעֲשֶׂ֥ה עוֹלָ֛ה לַיּ֖וֹם לַֽיהֹוָ֑ה בַּבֹּ֥קֶר בַּבֹּ֖קֶר תַּעֲשֶׂ֥ה אֹתֽוֹ׃ (כה) וּמִנְחָה֩ תַעֲשֶׂ֨ה עָלָ֜יו בַּבֹּ֤קֶר בַּבֹּ֙קֶר֙ שִׁשִּׁ֣ית הָֽאֵיפָ֔ה וְשֶׁ֛מֶן שְׁלִישִׁ֥ית הַהִ֖ין לָרֹ֣ס אֶת־ הַסֹּ֑לֶת מִנְחָה֙ לַֽיהוָ֔ה חֻקּ֥וֹת עוֹלָ֖ם תָּמִֽיד׃ (כו) (ועשו) יַעֲשׂ֨וּ אֶת־ הַכֶּ֧בֶשׂ וְאֶת־ הַמִּנְחָ֛ה וְאֶת־ הַשֶּׁ֖מֶן בַּבֹּ֣קֶר בַּבֹּ֑קֶר עוֹלַ֖ת תָּמִֽיד׃ (כז) כֹּה־ אָמַ֞ר אֲדֹנָ֣י יְהֹוִ֗ה כִּֽי־ יִתֵּ֨ן הַנָּשִׂ֤יא מַתָּנָה֙ לְאִ֣ישׁ מִבָּנָ֔יו נַחֲלָת֥וֹ הִ֖יא לְבָנָ֣יו תִּֽהְיֶ֑ה אֲחֻזָּתָ֥ם הִ֖יא בְּנַחֲלָֽה׃ (כח) וְכִֽי־ יִתֵּ֨ן מַתָּנָ֜ה מִנַּחֲלָת֗וֹ לְאַחַד֙ מֵֽעֲבָדָ֔יו וְהָ֤יְתָה לּוֹ֙ עַד־ שְׁנַ֣ת הַדְּר֔וֹר וְשָׁבַ֖ת לַנָּשִׂ֑יא אַ֚ךְ נַחֲלָת֔וֹ בָּנָ֖יו לָהֶ֥ם תִּהְיֶֽה׃ (כט) וְלֹא־ יִקַּ֨ח הַנָּשִׂ֜יא מִנַּחֲלַ֣ת הָעָ֗ם לְהֽוֹנֹתָם֙ מֵאֲחֻזָּתָ֔ם מֵאֲחֻזָּת֖וֹ יַנְחִ֣ל אֶת־ בָּנָ֑יו לְמַ֙עַן֙ אֲשֶׁ֣ר לֹֽא־ יָפֻ֣צוּ עַמִּ֔י אִ֖ישׁ מֵאֲחֻזָּתֽוֹ׃ (ל) וַיְבִיאֵ֣נִי בַמָּבוֹא֮ אֲשֶׁ֣ר עַל־ כֶּ֣תֶף הַשַּׁעַר֒ אֶל־ הַלִּשְׁכ֤וֹת הַקֹּ֙דֶשׁ֙ אֶל־ הַכֹּ֣הֲנִ֔ים הַפֹּנ֖וֹת צָפ֑וֹנָה וְהִנֵּה־ שָׁ֣ם מָק֔וֹם (בירכתם) בַּיַּרְכָתַ֖יִם יָֽמָּה׃ (לא) וַיֹּ֣אמֶר אֵלַ֔י זֶ֣ה הַמָּק֗וֹם אֲשֶׁ֤ר יְבַשְּׁלוּ־ שָׁם֙ הַכֹּ֣הֲנִ֔ים אֶת־ הָאָשָׁ֖ם וְאֶת־ הַחַטָּ֑את אֲשֶׁ֤ר יֹאפוּ֙ אֶת־ הַמִּנְחָ֔ה לְבִלְתִּ֥י הוֹצִ֛יא אֶל־ הֶחָצֵ֥ר הַחִֽיצוֹנָ֖ה לְקַדֵּ֥שׁ אֶת־ הָעָֽם׃ (לב) וַיּוֹצִיאֵ֗נִי אֶל־ הֶֽחָצֵר֙ הַחִ֣יצֹנָ֔ה וַיַּ֣עֲבִירֵ֔נִי אֶל־ אַרְבַּ֖עַת מִקְצוֹעֵ֣י הֶחָצֵ֑ר וְהִנֵּ֤ה חָצֵר֙ בְּמִקְצֹ֣עַ הֶחָצֵ֔ר חָצֵ֖ר בְּמִקְצֹ֥עַ הֶחָצֵֽר׃ (לג) בְּאַרְבַּ֜עַת מִקְצֹע֤וֹת הֶֽחָצֵר֙ חֲצֵר֣וֹת קְטֻר֔וֹת אַרְבָּעִ֣ים אֹ֔רֶךְ וּשְׁלֹשִׁ֖ים רֹ֑חַב מִדָּ֣ה אַחַ֔ת לְאַרְבַּעְתָּ֖ם מְׄהֻׄקְׄצָׄעֽׄוֹׄתׄ׃ (לד) וְט֨וּר סָבִ֥יב בָּהֶ֛ם סָבִ֖יב לְאַרְבַּעְתָּ֑ם וּמְבַשְּׁל֣וֹת עָשׂ֔וּי מִתַּ֥חַת הַטִּיר֖וֹת סָבִֽיב׃ (לה) וַיֹּ֖אמֶר אֵלָ֑י אֵ֚לֶּה בֵּ֣ית הַֽמְבַשְּׁלִ֔ים אֲשֶׁ֧ר יְבַשְּׁלוּ־ שָׁ֛ם מְשָׁרְתֵ֥י הַבַּ֖יִת אֶת־ זֶ֥בַח הָעָֽם׃ סוף פרק מו (לו) וַיְשִׁבֵנִי֮ אֶל־ פֶּ֣תַח הַבַּיִת֒ וְהִנֵּה־ מַ֣יִם יֹצְאִ֗ים מִתַּ֨חַת מִפְתַּ֤ן הַבַּ֙יִת֙ קָדִ֔ימָה כִּֽי־ פְנֵ֥י הַבַּ֖יִת קָדִ֑ים וְהַמַּ֣יִם יֹרְדִ֗ים מִתַּ֜חַת מִכֶּ֤תֶף הַבַּ֙יִת֙ הַיְמָנִ֔ית מִנֶּ֖גֶב לַמִּזְבֵּֽחַ׃ (לז) וַיּוֹצִאֵנִי֮ דֶּֽרֶךְ־ שַׁ֣עַר צָפוֹנָה֒ וַיְסִבֵּ֙נִי֙ דֶּ֣רֶךְ ח֔וּץ אֶל־ שַׁ֣עַר הַח֔וּץ דֶּ֖רֶךְ הַפּוֹנֶ֣ה קָדִ֑ים וְהִנֵּה־ מַ֣יִם מְפַכִּ֔ים מִן־ הַכָּתֵ֖ף הַיְמָנִֽית׃ (לח) בְּצֵאת־ הָאִ֥ישׁ קָדִ֖ים וְקָ֣ו בְּיָד֑וֹ וַיָּ֤מָד אֶ֙לֶף֙ בָּֽאַמָּ֔ה וַיַּעֲבִרֵ֥נִי בַמַּ֖יִם מֵ֥י אָפְסָֽיִם׃ (לט) וַיָּ֣מָד אֶ֔לֶף וַיַּעֲבִרֵ֥נִי בַמַּ֖יִם מַ֣יִם בִּרְכָּ֑יִם וַיָּ֣מָד אֶ֔לֶף וַיַּעֲבִרֵ֖נִי מֵ֥י מָתְנָֽיִם׃ (מ) וַיָּ֣מָד אֶ֔לֶף נַ֕חַל אֲשֶׁ֥ר לֹֽא־ אוּכַ֖ל לַעֲבֹ֑ר כִּֽי־ גָא֤וּ הַמַּ֙יִם֙ מֵ֣י שָׂ֔חוּ נַ֖חַל אֲשֶׁ֥ר לֹֽא־ יֵעָבֵֽר׃ (מא) וַיֹּ֥אמֶר אֵלַ֖י הֲרָאִ֣יתָ בֶן־ אָדָ֑ם וַיּוֹלִכֵ֥נִי וַיְשִׁבֵ֖נִי שְׂפַ֥ת הַנָּֽחַל׃ (מב) בְּשׁוּבֵ֕נִי וְהִנֵּה֙ אֶל־ שְׂפַ֣ת הַנַּ֔חַל עֵ֖ץ רַ֣ב מְאֹ֑ד מִזֶּ֖ה וּמִזֶּֽה׃ (מג) וַיֹּ֣אמֶר אֵלַ֗י הַמַּ֤יִם הָאֵ֙לֶּה֙ יוֹצְאִ֗ים אֶל־ הַגְּלִילָה֙ הַקַּדְמוֹנָ֔ה וְיָרְד֖וּ עַל־ הָֽעֲרָבָ֑ה וּבָ֣אוּ הַיָּ֔מָּה אֶל־ הַיָּ֥מָּה הַמּֽוּצָאִ֖ים (ונרפאו) וְנִרְפּ֥וּ הַמָּֽיִם׃ (מד) וְהָיָ֣ה כָל־ נֶ֣פֶשׁ חַיָּ֣ה ׀ אֲ‍ֽשֶׁר־ יִשְׁרֹ֡ץ אֶ֣ל כָּל־ אֲשֶׁר֩ יָב֨וֹא שָׁ֤ם נַחֲלַ֙יִם֙ יִֽחְיֶ֔ה וְהָיָ֥ה הַדָּגָ֖ה רַבָּ֣ה מְאֹ֑ד כִּי֩ בָ֨אוּ שָׁ֜מָּה הַמַּ֣יִם הָאֵ֗לֶּה וְיֵרָֽפְאוּ֙ וָחָ֔י כֹּ֛ל אֲשֶׁר־ יָ֥בוֹא שָׁ֖מָּה הַנָּֽחַל׃ (מה) וְהָיָה֩ (יעמדו) עָמְד֨וּ עָלָ֜יו דַּוָּגִ֗ים מֵעֵ֥ין גֶּ֙דִי֙ וְעַד־ עֵ֣ין עֶגְלַ֔יִם מִשְׁט֥וֹחַ לַֽחֲרָמִ֖ים יִהְי֑וּ לְמִינָה֙ תִּהְיֶ֣ה דְגָתָ֔ם כִּדְגַ֛ת הַיָּ֥ם הַגָּד֖וֹל רַבָּ֥ה מְאֹֽד׃ (מו) (בצאתו) בִּצֹּאתָ֧יו וּגְבָאָ֛יו וְלֹ֥א יֵרָפְא֖וּ לְמֶ֥לַח נִתָּֽנוּ׃
