יחזקאל פרק כד לפי החלוקה לסדרים

פרק כד בספר יחזקאל (27 פסוקים) נמצא ביחזקאל סדר יד (פסוקים כד–מו) וביחזקאל סדר טו (פסוקים א–ד).

יחזקאל סדר יד

(א) וְהִשְׁבַּתִּ֤י זִמָּתֵךְ֙ מִמֵּ֔ךְ וְאֶת־ זְנוּתֵ֖ךְ מֵאֶ֣רֶץ מִצְרָ֑יִם וְלֹֽא־ תִשְׂאִ֤י עֵינַ֙יִךְ֙ אֲלֵיהֶ֔ם וּמִצְרַ֖יִם לֹ֥א תִזְכְּרִי־ עֽוֹד׃ (ב) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִי֙ נֹֽתְנָ֔ךְ בְּיַ֖ד אֲשֶׁ֣ר שָׂנֵ֑את בְּיַ֛ד אֲשֶׁר־ נָקְעָ֥ה נַפְשֵׁ֖ךְ מֵהֶֽם׃ (ג) וְעָשׂ֨וּ אוֹתָ֜ךְ בְּשִׂנְאָ֗ה וְלָקְחוּ֙ כָּל־ יְגִיעֵ֔ךְ וַעֲזָב֖וּךְ עֵירֹ֣ם וְעֶרְיָ֑ה וְנִגְלָה֙ עֶרְוַ֣ת זְנוּנַ֔יִךְ וְזִמָּתֵ֖ךְ וְתַזְנוּתָֽיִךְ׃ (ד) עָשֹׂ֥ה אֵ֖לֶּה לָ֑ךְ בִּזְנוֹתֵךְ֙ אַחֲרֵ֣י גוֹיִ֔ם עַ֥ל אֲשֶׁר־ נִטְמֵ֖את בְּגִלּוּלֵיהֶֽם׃ (ה) בְּדֶ֥רֶךְ אֲחוֹתֵ֖ךְ הָלָ֑כְתְּ וְנָתַתִּ֥י כוֹסָ֖הּ בְּיָדֵֽךְ׃ (ו) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהֹוִ֔ה כּ֤וֹס אֲחוֹתֵךְ֙ תִּשְׁתִּ֔י הָעֲמֻקָּ֖ה וְהָרְחָבָ֑ה תִּהְיֶ֥ה לִצְחֹ֛ק וּלְלַ֖עַג מִרְבָּ֥ה לְהָכִֽיל׃ (ז) שִׁכָּר֥וֹן וְיָג֖וֹן תִּמָּלֵ֑אִי כּ֚וֹס שַׁמָּ֣ה וּשְׁמָמָ֔ה כּ֖וֹס אֲחוֹתֵ֥ךְ שֹׁמְרֽוֹן׃ (ח) וְשָׁתִ֨ית אוֹתָ֜הּ וּמָצִ֗ית וְאֶת־ חֲרָשֶׂ֛יהָ תְּגָרֵ֖מִי וְשָׁדַ֣יִךְ תְּנַתֵּ֑קִי כִּ֚י אֲנִ֣י דִבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ט) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֚עַן שָׁכַ֣חַתְּ אוֹתִ֔י וַתַּשְׁלִ֥יכִי אוֹתִ֖י אַחֲרֵ֣י גַוֵּ֑ךְ וְגַם־ אַ֛תְּ שְׂאִ֥י זִמָּתֵ֖ךְ וְאֶת־ תַּזְנוּתָֽיִךְ׃ (י) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י בֶּן־ אָדָ֕ם הֲתִשְׁפּ֥וֹט אֶֽת־ אָהֳלָ֖ה וְאֶת־ אָהֳלִיבָ֑ה וְהַגֵּ֣ד לָהֶ֔ן אֵ֖ת תוֹעֲבוֹתֵיהֶֽן׃ (יא) כִּ֣י נִאֵ֗פוּ וְדָם֙ בִּֽידֵיהֶ֔ן וְאֶת־ גִּלּֽוּלֵיהֶ֖ן נִאֵ֑פוּ וְגַ֤ם אֶת־ בְּנֵיהֶן֙ אֲשֶׁ֣ר יָֽלְדוּ־ לִ֔י הֶעֱבִ֥ירוּ לָהֶ֖ם לְאָכְלָֽה׃ (יב) ע֥וֹד זֹ֖את עָ֣שׂוּ לִ֑י טִמְּא֤וּ אֶת־ מִקְדָּשִׁי֙ בַּיּ֣וֹם הַה֔וּא וְאֶת־ שַׁבְּתוֹתַ֖י חִלֵּֽלוּ׃ (יג) וּֽבְשַׁחֲטָ֤ם אֶת־ בְּנֵיהֶם֙ לְגִלּ֣וּלֵיהֶ֔ם וַיָּבֹ֧אוּ אֶל־ מִקְדָּשִׁ֛י בַּיּ֥וֹם הַה֖וּא לְחַלְּל֑וֹ וְהִנֵּה־ כֹ֥ה עָשׂ֖וּ בְּת֥וֹךְ בֵּיתִֽי׃ (יד) וְאַ֗ף כִּ֤י תִשְׁלַ֙חְנָה֙ לַֽאֲנָשִׁ֔ים בָּאִ֖ים מִמֶּרְחָ֑ק אֲשֶׁ֨ר מַלְאָ֜ךְ שָׁל֤וּחַ אֲלֵיהֶם֙ וְהִנֵּה־ בָ֔אוּ לַאֲשֶׁ֥ר רָחַ֛צְתְּ כָּחַ֥לְתְּ עֵינַ֖יִךְ וְעָ֥דִית עֶֽדִי׃ (טו) וְיָשַׁבְתְּ֙ עַל־ מִטָּ֣ה כְבוּדָּ֔ה וְשֻׁלְחָ֥ן עָר֖וּךְ לְפָנֶ֑יהָ וּקְטָרְתִּ֥י וְשַׁמְנִ֖י שַׂ֥מְתְּ עָלֶֽיהָ׃ (טז) וְק֣וֹל הָמוֹן֮ שָׁלֵ֣ו בָהּ֒ וְאֶל־ אֲנָשִׁים֙ מֵרֹ֣ב אָדָ֔ם מוּבָאִ֥ים (סובאים) סָבָאִ֖ים מִמִּדְבָּ֑ר וַֽיִּתְּנ֤וּ צְמִידִים֙ אֶל־ יְדֵיהֶ֔ן וַעֲטֶ֥רֶת תִּפְאֶ֖רֶת עַל־ רָאשֵׁיהֶֽן׃ (יז) וָאֹמַ֕ר לַבָּלָ֖ה נִֽאוּפִ֑ים (עת) עַתָּ֛ה (יזנה) יִזְנ֥וּ תַזְנוּתֶ֖הָ וָהִֽיא׃ (יח) וַיָּב֣וֹא אֵלֶ֔יהָ כְּב֖וֹא אֶל־ אִשָּׁ֣ה זוֹנָ֑ה כֵּ֣ן בָּ֗אוּ אֶֽל־ אָֽהֳלָה֙ וְאֶל־ אָ֣הֳלִיבָ֔ה אִשֹּׁ֖ת הַזִּמָּֽה׃ (יט) וַאֲנָשִׁ֣ים צַדִּיקִ֗ם הֵ֚מָּה יִשְׁפְּט֣וּ אֽוֹתְהֶ֔ם מִשְׁפַּט֙ נֹֽאֲפ֔וֹת וּמִשְׁפַּ֖ט שֹׁפְכ֣וֹת דָּ֑ם כִּ֤י נֹֽאֲפֹת֙ הֵ֔נָּה וְדָ֖ם בִּֽידֵיהֶֽן׃ (כ) כִּ֛י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הַעֲלֵ֤ה עֲלֵיהֶם֙ קָהָ֔ל וְנָתֹ֥ן אֶתְהֶ֖ן לְזַעֲוָ֥ה וְלָבַֽז׃ (כא) וְרָגְמ֨וּ עֲלֵיהֶ֥ן אֶ֙בֶן֙ קָהָ֔ל וּבָרֵ֥א אוֹתְהֶ֖ן בְּחַרְבוֹתָ֑ם בְּנֵיהֶ֤ם וּבְנֽוֹתֵיהֶם֙ יַהֲרֹ֔גוּ וּבָתֵּיהֶ֖ן בָּאֵ֥שׁ יִשְׂרֹֽפוּ׃ (כב) וְהִשְׁבַּתִּ֥י זִמָּ֖ה מִן־ הָאָ֑רֶץ וְנִֽוַּסְּרוּ֙ כָּל־ הַנָּשִׁ֔ים וְלֹ֥א תַעֲשֶׂ֖ינָה כְּזִמַּתְכֶֽנָה׃ (כג) וְנָתְנ֤וּ זִמַּתְכֶ֙נָה֙ עֲלֵיכֶ֔ן וַחֲטָאֵ֥י גִלּוּלֵיכֶ֖ן תִּשֶּׂ֑אינָה וִידַעְתֶּ֕ם כִּ֥י אֲנִ֖י אֲדֹנָ֥י יְהוִֽה׃ תחילת פרק כד (כד) וַיְהִי֩ דְבַר־ יְהוָ֨ה אֵלַ֜י בַּשָּׁנָ֤ה הַתְּשִׁיעִית֙ בַּחֹ֣דֶשׁ הָעֲשִׂירִ֔י בֶּעָשׂ֥וֹר לַחֹ֖דֶשׁ לֵאמֹֽר׃ (כה) בֶּן־ אָדָ֗ם (כתוב)־ כְּתָב־ לְךָ֙ אֶת־ שֵׁ֣ם הַיּ֔וֹם אֶת־ עֶ֖צֶם הַיּ֣וֹם הַזֶּ֑ה סָמַ֤ךְ מֶֽלֶךְ־ בָּבֶל֙ אֶל־ יְר֣וּשָׁלִַ֔ם בְּעֶ֖צֶם הַיּ֥וֹם הַזֶּֽה׃ (כו) וּמְשֹׁ֤ל אֶל־ בֵּית־ הַמֶּ֙רִי֙ מָשָׁ֔ל וְאָמַרְתָּ֣ אֲלֵיהֶ֔ם כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה שְׁפֹ֤ת הַסִּיר֙ שְׁפֹ֔ת וְגַם־ יְצֹ֥ק בּ֖וֹ מָֽיִם׃ (כז) אֱסֹ֤ף נְתָחֶ֙יהָ֙ אֵלֶ֔יהָ כָּל־ נֵ֥תַח ט֖וֹב יָרֵ֣ךְ וְכָתֵ֑ף מִבְחַ֥ר עֲצָמִ֖ים מַלֵּֽא׃ (כח) מִבְחַ֤ר הַצֹּאן֙ לָק֔וֹחַ וְגַ֛ם דּ֥וּר הָעֲצָמִ֖ים תַּחְתֶּ֑יהָ רַתַּ֣ח רְתָחֶ֔יהָ גַּם־ בָּשְׁל֥וּ עֲצָמֶ֖יהָ בְּתוֹכָֽהּ׃ (כט) לָכֵ֞ן כֹּה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה אוֹי֮ עִ֣יר הַדָּמִים֒ סִ֚יר אֲשֶׁ֣ר חֶלְאָתָ֣ה בָ֔הּ וְחֶ֨לְאָתָ֔הּ לֹ֥א יָצְאָ֖ה מִמֶּ֑נָּה לִנְתָחֶ֤יהָ לִנְתָחֶ֙יהָ֙ הוֹצִיאָ֔הּ לֹא־ נָפַ֥ל עָלֶ֖יהָ גּוֹרָֽל׃ (ל) כִּ֤י דָמָהּ֙ בְּתוֹכָ֣הּ הָיָ֔ה עַל־ צְחִ֥יחַ סֶ֖לַע שָׂמָ֑תְהוּ לֹ֤א שְׁפָכַ֙תְהוּ֙ עַל־ הָאָ֔רֶץ לְכַסּ֥וֹת עָלָ֖יו עָפָֽר׃ (לא) לְהַעֲל֤וֹת חֵמָה֙ לִנְקֹ֣ם נָקָ֔ם נָתַ֥תִּי אֶת־ דָּמָ֖הּ עַל־ צְחִ֣יחַ סָ֑לַע לְבִלְתִּ֖י הִכָּסֽוֹת׃ (לב) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה א֖וֹי עִ֣יר הַדָּמִ֑ים גַּם־ אֲנִ֖י אַגְדִּ֥יל הַמְּדוּרָֽה׃ (לג) הַרְבֵּ֤ה הָעֵצִים֙ הַדְלֵ֣ק הָאֵ֔שׁ הָתֵ֖ם הַבָּשָׂ֑ר וְהַרְקַח֙ הַמֶּרְקָחָ֔ה וְהָעֲצָמ֖וֹת יֵחָֽרוּ׃ (לד) וְהַעֲמִידֶ֥הָ עַל־ גֶּחָלֶ֖יהָ רֵקָ֑ה לְמַ֨עַן תֵּחַ֜ם וְחָ֣רָה נְחֻשְׁתָּ֗הּ וְנִתְּכָ֤ה בְתוֹכָהּ֙ טֻמְאָתָ֔הּ תִּתֻּ֖ם חֶלְאָתָֽהּ׃ (לה) תְּאֻנִ֖ים הֶלְאָ֑ת וְלֹֽא־ תֵצֵ֤א מִמֶּ֙נָּה֙ רַבַּ֣ת חֶלְאָתָ֔הּ בְּאֵ֖שׁ חֶלְאָתָֽהּ׃ (לו) בְּטֻמְאָתֵ֖ךְ זִמָּ֑ה יַ֤עַן טִֽהַרְתִּיךְ֙ וְלֹ֣א טָהַ֔רְתְּ מִטֻּמְאָתֵךְ֙ לֹ֣א תִטְהֲרִי־ ע֔וֹד עַד־ הֲנִיחִ֥י אֶת־ חֲמָתִ֖י בָּֽךְ׃ (לז) אֲנִ֨י יְהוָ֤ה דִּבַּ֙רְתִּי֙ בָּאָ֣ה וְעָשִׂ֔יתִי לֹֽא־ אֶפְרַ֥ע וְלֹא־ אָח֖וּס וְלֹ֣א אֶנָּחֵ֑ם כִּדְרָכַ֤יִךְ וְכַעֲלִילוֹתַ֙יִךְ֙ שְׁפָט֔וּךְ נְאֻ֖ם אֲדֹנָ֥י יְהֹוִֽה׃ (לח) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לט) בֶּן־ אָדָ֕ם הִנְנִ֨י לֹקֵ֧חַ מִמְּךָ֛ אֶת־ מַחְמַ֥ד עֵינֶ֖יךָ בְּמַגֵּפָ֑ה וְלֹ֤א תִסְפֹּד֙ וְלֹ֣א תִבְכֶּ֔ה וְל֥וֹא תָב֖וֹא דִּמְעָתֶֽךָ׃ (מ) הֵאָנֵ֣ק ׀ דֹּ֗ם מֵתִים֙ אֵ֣בֶל לֹֽא־ תַֽעֲשֶׂ֔ה פְאֵֽרְךָ֙ חֲב֣וֹשׁ עָלֶ֔יךָ וּנְעָלֶ֖יךָ תָּשִׂ֣ים בְּרַגְלֶ֑יךָ וְלֹ֤א תַעְטֶה֙ עַל־ שָׂפָ֔ם וְלֶ֥חֶם אֲנָשִׁ֖ים לֹ֥א תֹאכֵֽל׃ (מא) וָאֲדַבֵּ֤ר אֶל־ הָעָם֙ בַּבֹּ֔קֶר וַתָּ֥מָת אִשְׁתִּ֖י בָּעָ֑רֶב וָאַ֥עַשׂ בַּבֹּ֖קֶר כַּאֲשֶׁ֥ר צֻוֵּֽיתִי׃ (מב) וַיֹּאמְר֥וּ אֵלַ֖י הָעָ֑ם הֲלֹֽא־ (א) וְהָיָ֨ה יְחֶזְקֵ֤אל לָכֶם֙ לְמוֹפֵ֔ת כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה תַּעֲשׂ֑וּ בְּבֹאָ֕הּ וִֽידַעְתֶּ֕ם כִּ֥י אֲנִ֖י אֲדֹנָ֥י יְהוִֽה׃ (ב) וְאַתָּ֣ה בֶן־ אָדָ֔ם הֲל֗וֹא בְּי֨וֹם קַחְתִּ֤י מֵהֶם֙ אֶת־ מָ֣עוּזָּ֔ם מְשׂ֖וֹשׂ תִּפְאַרְתָּ֑ם אֶת־ מַחְמַ֤ד עֵֽינֵיהֶם֙ וְאֶת־ מַשָּׂ֣א נַפְשָׁ֔ם בְּנֵיהֶ֖ם וּבְנוֹתֵיהֶֽם׃ (ג) בַּיּ֣וֹם הַה֔וּא יָב֥וֹא הַפָּלִ֖יט אֵלֶ֑יךָ לְהַשְׁמָע֖וּת אָזְנָֽיִם׃ (ד) בַּיּ֣וֹם הַה֗וּא יִפָּ֤תַח פִּ֙יךָ֙ אֶת־ הַפָּלִ֔יט וּתְדַבֵּ֕ר וְלֹ֥א תֵֽאָלֵ֖ם ע֑וֹד וְהָיִ֤יתָ לָהֶם֙ לְמוֹפֵ֔ת וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ סוף פרק כד (ה) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ו) בֶּן־ אָדָ֕ם שִׂ֥ים פָּנֶ֖יךָ אֶל־ בְּנֵ֣י עַמּ֑וֹן וְהִנָּבֵ֖א עֲלֵיהֶֽם׃ (ז) וְאָֽמַרְתָּ֙ לִבְנֵ֣י עַמּ֔וֹן שִׁמְע֖וּ דְּבַר־ אֲדֹנָ֣י יְהוִ֑ה כֹּה־ אָמַ֣ר אֲדֹנָ֣י יְהוִ֡ה יַעַן֩ אָמְרֵ֨ךְ הֶאָ֜ח אֶל־ מִקְדָּשִׁ֣י כִֽי־ נִחָ֗ל וְאֶל־ אַדְמַ֤ת יִשְׂרָאֵל֙ כִּ֣י נָשַׁ֔מָּה וְאֶל־ בֵּ֣ית יְהוּדָ֔ה כִּ֥י הָלְכ֖וּ בַּגּוֹלָֽה׃ (ח) לָכֵ֡ן הִנְנִי֩ נֹתְנָ֨ךְ לִבְנֵי־ קֶ֜דֶם לְמֽוֹרָשָׁ֗ה וְיִשְּׁב֤וּ טִירֽוֹתֵיהֶם֙ בָּ֔ךְ וְנָ֥תְנוּ בָ֖ךְ מִשְׁכְּנֵיהֶ֑ם הֵ֚מָּה יֹאכְל֣וּ פִרְיֵ֔ךְ וְהֵ֖מָּה יִשְׁתּ֥וּ חֲלָבֵֽךְ׃ (ט) וְנָתַתִּ֤י אֶת־ רַבָּה֙ לִנְוֵ֣ה גְמַלִּ֔ים וְאֶת־ בְּנֵ֥י עַמּ֖וֹן לְמִרְבַּץ־ צֹ֑אן וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (י) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֚עַן מַחְאֲךָ֣ יָ֔ד וְרַקְעֲךָ֖ בְּרָ֑גֶל וַתִּשְׂמַ֤ח בְּכָל־ שָֽׁאטְךָ֙ בְּנֶ֔פֶשׁ אֶל־ אַדְמַ֖ת יִשְׂרָאֵֽל׃ (יא) לָכֵ֡ן הִנְנִי֩ נָטִ֨יתִי אֶת־ יָדִ֜י עָלֶ֗יךָ וּנְתַתִּ֤יךָֽ־ (לבג) לְבַז֙ לַגּוֹיִ֔ם וְהִכְרַתִּ֙יךָ֙ מִן־ הָ֣עַמִּ֔ים וְהַאֲבַדְתִּ֖יךָ מִן־ הָאֲרָצ֑וֹת אַשְׁמִ֣ידְךָ֔ וְיָדַעְתָּ֖ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (יב) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה יַ֗עַן אֲמֹ֤ר מוֹאָב֙ וְשֵׂעִ֔יר הִנֵּ֥ה כְּכָֽל־ הַגּוֹיִ֖ם בֵּ֥ית יְהוּדָֽה׃ (יג) לָכֵן֩ הִנְנִ֨י פֹתֵ֜חַ אֶת־ כֶּ֤תֶף מוֹאָב֙ מֵהֶ֣עָרִ֔ים מֵֽעָרָ֖יו מִקָּצֵ֑הוּ צְבִ֗י אֶ֚רֶץ בֵּ֣ית הַיְשִׁימֹ֔ת בַּ֥עַל מְע֖וֹן (וקריתמה) וְקִרְיָתָֽיְמָה׃ (יד) לִבְנֵי־ קֶ֙דֶם֙ עַל־ בְּנֵ֣י עַמּ֔וֹן וּנְתַתִּ֖יהָ בִּנְקֹ֥ם נָקָ֖ם לְבֵ֣ית יְהוּדָ֑ה וַיֶּאְשְׁמ֥וּ אָשׁ֖וֹם וְנִקְּמ֥וּ בָהֶֽם׃ (יז) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה וְנָטִ֤תִי יָדִי֙ עַל־ אֱד֔וֹם וְהִכְרַתִּ֥י מִמֶּ֖נָּה אָדָ֣ם וּבְהֵמָ֑ה וּנְתַתִּ֤יהָ חָרְבָּה֙ מִתֵּימָ֔ן וּדְדָ֖נֶה בַּחֶ֥רֶב יִפֹּֽלוּ׃ (יח) וְנָתַתִּ֨י אֶת־ נִקְמָתִ֜י בֶּאֱד֗וֹם בְּיַד֙ עַמִּ֣י יִשְׂרָאֵ֔ל וְעָשׂ֣וּ בֶאֱד֔וֹם כְּאַפִּ֖י וְכַחֲמָתִ֑י וְיָֽדְעוּ֙ אֶת־ נִקְמָתִ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יט) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֛עַן עֲשׂ֥וֹת פְּלִשְׁתִּ֖ים בִּנְקָמָ֑ה וַיִּנָּקְמ֤וּ נָקָם֙ בִּשְׁאָ֣ט בְּנֶ֔פֶשׁ לְמַשְׁחִ֖ית אֵיבַ֥ת עוֹלָֽם׃ (כ) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֨י נוֹטֶ֤ה יָדִי֙ עַל־ פְּלִשְׁתִּ֔ים וְהִכְרַתִּ֖י אֶת־ כְּרֵתִ֑ים וְהַ֣אֲבַדְתִּ֔י אֶת־ שְׁאֵרִ֖ית ח֥וֹף הַיָּֽם׃ (כא) וְעָשִׂ֤יתִי בָם֙ נְקָמ֣וֹת גְּדֹל֔וֹת בְּתוֹכְח֖וֹת חֵמָ֑ה וְיָֽדְעוּ֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה בְּתִתִּ֥י אֶת־ נִקְמָתִ֖י בָּֽם׃ (כב) וַיְהִ֛י בְּעַשְׁתֵּֽי־ עֶשְׂרֵ֥ה שָׁנָ֖ה בְּאֶחָ֣ד לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כג) בֶּן־ אָדָ֗ם יַ֠עַן אֲשֶׁר־ אָ֨מְרָה צֹּ֤ר עַל־ יְרוּשָׁלִַ֙ם֙ הֶאָ֔ח נִשְׁבְּרָ֛ה דַּלְת֥וֹת הָעַמִּ֖ים נָסֵ֣בָּה אֵלָ֑י אִמָּלְאָ֖ה הָחֳרָֽבָה׃ (כד) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י עָלַ֖יִךְ צֹ֑ר וְהַעֲלֵיתִ֤י עָלַ֙יִךְ֙ גּוֹיִ֣ם רַבִּ֔ים כְּהַעֲל֥וֹת הַיָּ֖ם לְגַלָּֽיו׃ (כה) וְשִׁחֲת֞וּ חֹמ֣וֹת צֹ֗ר וְהָֽרְסוּ֙ מִגְדָּלֶ֔יהָ וְסִֽחֵיתִ֥י עֲפָרָ֖הּ מִמֶּ֑נָּה וְנָתַתִּ֥י אוֹתָ֖הּ לִצְחִ֥יחַ סָֽלַע׃ (כו) מִשְׁטַ֨ח חֲרָמִ֤ים תִּֽהְיֶה֙ בְּת֣וֹךְ הַיָּ֔ם כִּ֚י אֲנִ֣י דִבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה וְהָיְתָ֥ה לְבַ֖ז לַגּוֹיִֽם׃ (כז) וּבְנוֹתֶ֙יהָ֙ אֲשֶׁ֣ר בַּשָּׂדֶ֔ה בַּחֶ֖רֶב תֵּהָרַ֑גְנָה וְיָדְע֖וּ כִּי־ אֲנִ֥י יְהוָֽה׃ (כח) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֧י מֵבִ֣יא אֶל־ צֹ֗ר נְבוּכַדְרֶאצַּ֧ר מֶֽלֶךְ־ בָּבֶ֛ל מִצָּפ֖וֹן מֶ֣לֶךְ מְלָכִ֑ים בְּס֛וּס וּבְרֶ֥כֶב וּבְפָרָשִׁ֖ים וְקָהָ֥ל וְעַם־ רָֽב׃ (כט) בְּנוֹתַ֥יִךְ בַּשָּׂדֶ֖ה בַּחֶ֣רֶב יַהֲרֹ֑ג וְנָתַ֨ן עָלַ֜יִךְ דָּיֵ֗ק וְשָׁפַ֤ךְ עָלַ֙יִךְ֙ סֹֽלְלָ֔ה וְהֵקִ֥ים עָלַ֖יִךְ צִנָּֽה׃ (ל) וּמְחִ֣י קָֽבָלּ֔וֹ יִתֵּ֖ן בְּחֹֽמוֹתָ֑יִךְ וּמִ֨גְדְּלֹתַ֔יִךְ יִתֹּ֖ץ בְּחַרְבוֹתָֽיו׃ (לא) מִשִּׁפְעַ֥ת סוּסָ֖יו יְכַסֵּ֣ךְ אֲבָקָ֑ם מִקּוֹל֩ פָּרַ֨שׁ וְגַלְגַּ֜ל וָרֶ֗כֶב תִּרְעַ֙שְׁנָה֙ חֽוֹמוֹתַ֔יִךְ בְּבֹאוֹ֙ בִּשְׁעָרַ֔יִךְ כִּמְבוֹאֵ֖י עִ֥יר מְבֻקָּעָֽה׃ (לב) בְּפַרְס֣וֹת סוּסָ֔יו יִרְמֹ֖ס אֶת־ כָּל־ חֽוּצוֹתָ֑יִךְ עַמֵּךְ֙ בַּחֶ֣רֶב יַהֲרֹ֔ג וּמַצְּב֥וֹת עֻזֵּ֖ךְ לָאָ֥רֶץ תֵּרֵֽד׃ (לג) וְשָׁלְל֣וּ חֵילֵ֗ךְ וּבָֽזְזוּ֙ רְכֻלָּתֵ֔ךְ וְהָֽרְסוּ֙ חוֹמוֹתַ֔יִךְ וּבָתֵּ֥י חֶמְדָּתֵ֖ךְ יִתֹּ֑צוּ וַאֲבָנַ֤יִךְ וְעֵצַ֙יִךְ֙ וַֽעֲפָרֵ֔ךְ בְּת֥וֹךְ מַ֖יִם יָשִֽׂימוּ׃ (לד) וְהִשְׁבַּתִּ֖י הֲמ֣וֹן שִׁירָ֑יִךְ וְק֣וֹל כִּנּוֹרַ֔יִךְ לֹ֥א יִשָּׁמַ֖ע עֽוֹד׃ (לה) וּנְתַתִּ֞יךְ לִצְחִ֣יחַ סֶ֗לַע מִשְׁטַ֤ח חֲרָמִים֙ תִּֽהְיֶ֔ה לֹ֥א תִבָּנֶ֖ה ע֑וֹד כִּ֣י אֲנִ֤י יְהוָה֙ דִּבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לו) כֹּ֥ה אָמַ֛ר אֲדֹנָ֥י יְהוִ֖ה לְצ֑וֹר הֲלֹ֣א ׀ מִקּ֣וֹל מַפַּלְתֵּ֗ךְ בֶּאֱנֹ֨ק חָלָ֜ל בֵּהָ֤רֵֽג הֶ֙רֶג֙ בְּתוֹכֵ֔ךְ יִרְעֲשׁ֖וּ הָאִיִּֽים׃ (לז) וְֽיָרְד֞וּ מֵעַ֣ל כִּסְאוֹתָ֗ם כֹּ֚ל נְשִׂיאֵ֣י הַיָּ֔ם וְהֵסִ֙ירוּ֙ אֶת־ מְעִ֣ילֵיהֶ֔ם וְאֶת־ בִּגְדֵ֥י רִקְמָתָ֖ם יִפְשֹׁ֑טוּ חֲרָד֤וֹת ׀ יִלְבָּ֙שׁוּ֙ עַל־ הָאָ֣רֶץ יֵשֵׁ֔בוּ וְחָֽרְדוּ֙ לִרְגָעִ֔ים וְשָׁמְמ֖וּ עָלָֽיִךְ׃ (לח) וְנָשְׂא֨וּ עָלַ֤יִךְ קִינָה֙ וְאָ֣מְרוּ לָ֔ךְ אֵ֣יךְ אָבַ֔דְתְּ נוֹשֶׁ֖בֶת מִיַּמִּ֑ים הָעִ֣יר הַהֻלָּ֗לָה אֲשֶׁר֩ הָיְתָ֨ה חֲזָקָ֤ה בַיָּם֙ הִ֣יא וְיֹשְׁבֶ֔יהָ אֲשֶׁר־ נָתְנ֥וּ חִתִּיתָ֖ם לְכָל־ יוֹשְׁבֶֽיהָ׃ (לט) עַתָּה֙ יֶחְרְד֣וּ הָֽאִיִּ֔ן י֖וֹם מַפַּלְתֵּ֑ךְ וְנִבְהֲל֛וּ הָאִיִּ֥ים אֲשֶׁר־ בַּיָּ֖ם מִצֵּאתֵֽךְ׃ (מ) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה בְּתִתִּ֤י אֹתָךְ֙ עִ֣יר נֶחֱרֶ֔בֶת כֶּעָרִ֖ים אֲשֶׁ֣ר לֹֽא־ נוֹשָׁ֑בוּ בְּהַעֲל֤וֹת עָלַ֙יִךְ֙ אֶת־ תְּה֔וֹם וְכִסּ֖וּךְ הַמַּ֥יִם הָרַבִּֽים׃
