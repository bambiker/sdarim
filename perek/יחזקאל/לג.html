פרק לג בספר יחזקאל (33 פסוקים) נמצא ביחזקאל סדר יט (פסוקים לג–מז) וביחזקאל סדר כ (פסוקים א–יח).

יחזקאל סדר יט

(א) וַֽיְהִי֙ בִּשְׁתֵּ֣י עֶשְׂרֵ֣ה שָׁנָ֔ה בִּשְׁנֵי־ עָשָׂ֥ר חֹ֖דֶשׁ בְּאֶחָ֣ד לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ב) בֶּן־ אָדָ֗ם שָׂ֤א קִינָה֙ עַל־ פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם וְאָמַרְתָּ֣ אֵלָ֔יו כְּפִ֥יר גּוֹיִ֖ם נִדְמֵ֑יתָ וְאַתָּה֙ כַּתַּנִּ֣ים בַּיַּמִּ֔ים וַתָּ֣גַח בְּנַהֲרוֹתֶ֗יךָ וַתִּדְלַח־ מַ֙יִם֙ בְּרַגְלֶ֔יךָ וַתִּרְפֹּ֖ס נַהֲרוֹתָֽם׃ (ג) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה וּפָרַשְׂתִּ֤י עָלֶ֙יךָ֙ אֶת־ רִשְׁתִּ֔י בִּקְהַ֖ל עַמִּ֣ים רַבִּ֑ים וְהֶעֱל֖וּךָ בְּחֶרְמִֽי׃ (ד) וּנְטַשְׁתִּ֣יךָ בָאָ֔רֶץ עַל־ פְּנֵ֥י הַשָּׂדֶ֖ה אֲטִילֶ֑ךָ וְהִשְׁכַּנְתִּ֤י עָלֶ֙יךָ֙ כָּל־ ע֣וֹף הַשָּׁמַ֔יִם וְהִשְׂבַּעְתִּ֥י מִמְּךָ֖ חַיַּ֥ת כָּל־ הָאָֽרֶץ׃ (ה) וְנָתַתִּ֥י אֶת־ בְּשָׂרְךָ֖ עַל־ הֶֽהָרִ֑ים וּמִלֵּאתִ֥י הַגֵּאָי֖וֹת רָמוּתֶֽךָ׃ (ו) וְהִשְׁקֵיתִ֨י אֶ֧רֶץ צָפָתְךָ֛ מִדָּמְךָ֖ אֶל־ הֶֽהָרִ֑ים וַאֲפִקִ֖ים יִמָּלְא֥וּן מִמֶּֽךָּ׃ (ז) וְכִסֵּיתִ֤י בְכַבּֽוֹתְךָ֙ שָׁמַ֔יִם וְהִקְדַּרְתִּ֖י אֶת־ כֹּֽכְבֵיהֶ֑ם שֶׁ֚מֶשׁ בֶּעָנָ֣ן אֲכַסֶּ֔נּוּ וְיָרֵ֖חַ לֹא־ יָאִ֥יר אוֹרֽוֹ׃ (ח) כָּל־ מְא֤וֹרֵי אוֹר֙ בַּשָּׁמַ֔יִם אַקְדִּירֵ֖ם עָלֶ֑יךָ וְנָתַ֤תִּי חֹ֙שֶׁךְ֙ עַֽל־ אַרְצְךָ֔ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ט) וְהִ֨כְעַסְתִּ֔י לֵ֖ב עַמִּ֣ים רַבִּ֑ים בַּהֲבִיאִ֤י שִׁבְרְךָ֙ בַּגּוֹיִ֔ם עַל־ אֲרָצ֖וֹת אֲשֶׁ֥ר לֹֽא־ יְדַעְתָּֽם׃ (י) וַהֲשִׁמּוֹתִ֨י עָלֶ֜יךָ עַמִּ֣ים רַבִּ֗ים וּמַלְכֵיהֶם֙ יִשְׂעֲר֤וּ עָלֶ֙יךָ֙ שַׂ֔עַר בְּעוֹפְפִ֥י חַרְבִּ֖י עַל־ פְּנֵיהֶ֑ם וְחָרְד֤וּ לִרְגָעִים֙ אִ֣ישׁ לְנַפְשׁ֔וֹ בְּי֖וֹם מַפַּלְתֶּֽךָ׃ (יא) כִּ֛י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה חֶ֥רֶב מֶֽלֶךְ־ בָּבֶ֖ל תְּבוֹאֶֽךָ׃ (יב) בְּחַרְב֤וֹת גִּבּוֹרִים֙ אַפִּ֣יל הֲמוֹנֶ֔ךָ עָרִיצֵ֥י גוֹיִ֖ם כֻּלָּ֑ם וְשָֽׁדְדוּ֙ אֶת־ גְּא֣וֹן מִצְרַ֔יִם וְנִשְׁמַ֖ד כָּל־ הֲמוֹנָֽהּ׃ (יג) וְהַֽאֲבַדְתִּי֙ אֶת־ כָּל־ בְּהֶמְתָּ֔הּ מֵעַ֖ל מַ֣יִם רַבִּ֑ים וְלֹ֨א תִדְלָחֵ֤ם רֶֽגֶל־ אָדָם֙ ע֔וֹד וּפַרְס֥וֹת בְּהֵמָ֖ה לֹ֥א תִדְלָחֵֽם׃ (יד) אָ֚ז אַשְׁקִ֣יעַ מֵֽימֵיהֶ֔ם וְנַהֲרוֹתָ֖ם כַּשֶּׁ֣מֶן אוֹלִ֑יךְ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (טו) בְּתִתִּי֩ אֶת־ אֶ֨רֶץ מִצְרַ֜יִם שְׁמָמָ֣ה וּנְשַׁמָּ֗ה אֶ֚רֶץ מִמְּלֹאָ֔הּ בְּהַכּוֹתִ֖י אֶת־ כָּל־ י֣וֹשְׁבֵי בָ֑הּ וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (טז) קִינָ֥ה הִיא֙ וְק֣וֹנְנ֔וּהָ בְּנ֥וֹת הַגּוֹיִ֖ם תְּקוֹנֵ֣נָּה אוֹתָ֑הּ עַל־ מִצְרַ֤יִם וְעַל־ כָּל־ הֲמוֹנָהּ֙ תְּקוֹנֵ֣נָּה אוֹתָ֔הּ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יז) וַֽיְהִי֙ בִּשְׁתֵּ֣י עֶשְׂרֵ֣ה שָׁנָ֔ה בַּחֲמִשָּׁ֥ה עָשָׂ֖ר לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יח) בֶּן־ אָדָ֕ם נְהֵ֛ה עַל־ הֲמ֥וֹן מִצְרַ֖יִם וְהוֹרִדֵ֑הוּ א֠וֹתָהּ וּבְנ֨וֹת גּוֹיִ֧ם אַדִּרִ֛ם אֶל־ אֶ֥רֶץ תַּחְתִּיּ֖וֹת אֶת־ י֥וֹרְדֵי בֽוֹר׃ (יט) מִמִּ֖י נָעָ֑מְתָּ רְדָ֥ה וְהָשְׁכְּבָ֖ה אֶת־ עֲרֵלִֽים׃ (כ) בְּת֥וֹךְ חַלְלֵי־ חֶ֖רֶב יִפֹּ֑לוּ חֶ֣רֶב נִתָּ֔נָה מָשְׁכ֥וּ אוֹתָ֖הּ וְכָל־ הֲמוֹנֶֽיהָ׃ (כא) יְדַבְּרוּ־ ל֞וֹ אֵלֵ֧י גִבּוֹרִ֛ים מִתּ֥וֹךְ שְׁא֖וֹל אֶת־ עֹֽזְרָ֑יו יָֽרְד֛וּ שָׁכְב֥וּ הָעֲרֵלִ֖ים חַלְלֵי־ חָֽרֶב׃ (כב) שָׁ֤ם אַשּׁוּר֙ וְכָל־ קְהָלָ֔הּ סְבִֽיבוֹתָ֖יו קִבְרֹתָ֑יו כֻּלָּ֣ם חֲלָלִ֔ים הַנֹּפְלִ֖ים בֶּחָֽרֶב׃ (כג) אֲשֶׁ֨ר נִתְּנ֤וּ קִבְרֹתֶ֙יהָ֙ בְּיַרְכְּתֵי־ ב֔וֹר וַיְהִ֣י קְהָלָ֔הּ סְבִיב֖וֹת קְבֻרָתָ֑הּ כֻּלָּ֤ם חֲלָלִים֙ נֹפְלִ֣ים בַּחֶ֔רֶב אֲשֶׁר־ נָתְנ֥וּ חִתִּ֖ית בְּאֶ֥רֶץ חַיִּֽים׃ (כד) שָׁ֤ם עֵילָם֙ וְכָל־ הֲמוֹנָ֔הּ סְבִיב֖וֹת קְבֻרָתָ֑הּ כֻּלָּ֣ם חֲלָלִים֩ הַנֹּפְלִ֨ים בַּחֶ֜רֶב אֲ‍ֽשֶׁר־ יָרְד֥וּ עֲרֵלִ֣ים ׀ אֶל־ אֶ֣רֶץ תַּחְתִּיּ֗וֹת אֲשֶׁ֨ר נָתְנ֤וּ חִתִּיתָם֙ בְּאֶ֣רֶץ חַיִּ֔ים וַיִּשְׂא֥וּ כְלִמָּתָ֖ם אֶת־ י֥וֹרְדֵי בֽוֹר׃ (כה) בְּת֣וֹךְ חֲ֠לָלִים נָתְנ֨וּ מִשְׁכָּ֥ב לָהּ֙ בְּכָל־ הֲמוֹנָ֔הּ סְבִֽיבוֹתָ֖יו קִבְרֹתֶ֑הָ כֻּלָּ֣ם עֲרֵלִ֣ים חַלְלֵי־ חֶ֡רֶב כִּֽי־ נִתַּ֨ן חִתִּיתָ֜ם בְּאֶ֣רֶץ חַיִּ֗ים וַיִּשְׂא֤וּ כְלִמָּתָם֙ אֶת־ י֣וֹרְדֵי ב֔וֹר בְּת֥וֹךְ חֲלָלִ֖ים נִתָּֽן׃ (כו) שָׁ֣ם מֶ֤שֶׁךְ תֻּבַל֙ וְכָל־ הֲמוֹנָ֔הּ סְבִֽיבוֹתָ֖יו קִבְרוֹתֶ֑יהָ כֻּלָּ֤ם עֲרֵלִים֙ מְחֻ֣לְלֵי חֶ֔רֶב כִּֽי־ נָתְנ֥וּ חִתִּיתָ֖ם בְּאֶ֥רֶץ חַיִּֽים׃ (כז) וְלֹ֤א יִשְׁכְּבוּ֙ אֶת־ גִּבּוֹרִ֔ים נֹפְלִ֖ים מֵעֲרֵלִ֑ים אֲשֶׁ֣ר יָרְדֽוּ־ שְׁא֣וֹל בִּכְלֵֽי־ מִלְחַמְתָּם֩ וַיִּתְּנ֨וּ אֶת־ חַרְבוֹתָ֜ם תַּ֣חַת רָאשֵׁיהֶ֗ם וַתְּהִ֤י עֲוֺֽנֹתָם֙ עַל־ עַצְמוֹתָ֔ם כִּֽי־ חִתִּ֥ית גִּבּוֹרִ֖ים בְּאֶ֥רֶץ חַיִּֽים׃ (כח) וְאַתָּ֗ה בְּת֧וֹךְ עֲרֵלִ֛ים תִּשָּׁבַ֥ר וְתִשְׁכַּ֖ב אֶת־ חַלְלֵי־ חָֽרֶב׃ (כט) שָׁ֣מָּה אֱד֗וֹם מְלָכֶ֙יהָ֙ וְכָל־ נְשִׂיאֶ֔יהָ אֲשֶׁר־ נִתְּנ֥וּ בִגְבוּרָתָ֖ם אֶת־ חַלְלֵי־ חָ֑רֶב הֵ֛מָּה אֶת־ עֲרֵלִ֥ים יִשְׁכָּ֖בוּ וְאֶת־ יֹ֥רְדֵי בֽוֹר׃ (ל) שָׁ֣מָּה נְסִיכֵ֥י צָפ֛וֹן כֻּלָּ֖ם וְכָל־ צִֽדֹנִ֑י אֲשֶׁר־ יָרְד֣וּ אֶת־ חֲלָלִ֗ים בְּחִתִּיתָ֤ם מִגְבֽוּרָתָם֙ בּוֹשִׁ֔ים וַיִּשְׁכְּב֤וּ עֲרֵלִים֙ אֶת־ חַלְלֵי־ חֶ֔רֶב וַיִּשְׂא֥וּ כְלִמָּתָ֖ם אֶת־ י֥וֹרְדֵי בֽוֹר׃ (לא) אוֹתָם֙ יִרְאֶ֣ה פַרְעֹ֔ה וְנִחַ֖ם עַל־ כָּל־ (המונה) הֲמוֹנ֑וֹ חַלְלֵי־ חֶ֙רֶב֙ פַּרְעֹ֣ה וְכָל־ חֵיל֔וֹ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לב) כִּֽי־ נָתַ֥תִּי אֶת־ (חתיתו) חִתִּיתִ֖י בְּאֶ֣רֶץ חַיִּ֑ים וְהֻשְׁכַּב֩ בְּת֨וֹךְ עֲרֵלִ֜ים אֶת־ חַלְלֵי־ חֶ֗רֶב פַּרְעֹה֙ וְכָל־ הֲמוֹנֹ֔ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ תחילת פרק לג (לג) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לד) בֶּן־ אָדָ֗ם דַּבֵּ֤ר אֶל־ בְּנֵֽי־ עַמְּךָ֙ וְאָמַרְתָּ֣ אֲלֵיהֶ֔ם אֶ֕רֶץ כִּֽי־ אָבִ֥יא עָלֶ֖יהָ חָ֑רֶב וְלָקְח֨וּ עַם־ הָאָ֜רֶץ אִ֤ישׁ אֶחָד֙ מִקְצֵיהֶ֔ם וְנָתְנ֥וּ אֹת֛וֹ לָהֶ֖ם לְצֹפֶֽה׃ (לה) וְרָאָ֥ה אֶת־ הַחֶ֖רֶב בָּאָ֣ה עַל־ הָאָ֑רֶץ וְתָקַ֥ע בַּשּׁוֹפָ֖ר וְהִזְהִ֥יר אֶת־ הָעָֽם׃ (לו) וְשָׁמַ֨ע הַשֹּׁמֵ֜עַ אֶת־ ק֤וֹל הַשּׁוֹפָר֙ וְלֹ֣א נִזְהָ֔ר וַתָּ֥בוֹא חֶ֖רֶב וַתִּקָּחֵ֑הוּ דָּמ֥וֹ בְרֹאשׁ֖וֹ יִֽהְיֶֽה׃ (לז) אֵת֩ ק֨וֹל הַשּׁוֹפָ֤ר שָׁמַע֙ וְלֹ֣א נִזְהָ֔ר דָּמ֖וֹ בּ֣וֹ יִֽהְיֶ֑ה וְה֥וּא נִזְהָ֖ר נַפְשׁ֥וֹ מִלֵּֽט׃ (לח) וְ֠הַצֹּפֶה כִּֽי־ יִרְאֶ֨ה אֶת־ הַחֶ֜רֶב בָּאָ֗ה וְלֹֽא־ תָקַ֤ע בַּשּׁוֹפָר֙ וְהָעָ֣ם לֹֽא־ נִזְהָ֔ר וַתָּב֣וֹא חֶ֔רֶב וַתִּקַּ֥ח עַמְּךָ֙ צִדְקַ֣ת הַצַּדִּ֗יק לֹ֤א תַצִּילֶ֙נּוּ֙ בְּי֣וֹם פִּשְׁע֔וֹ וְרִשְׁעַ֤ת הָֽרָשָׁע֙ לֹֽא־ יִכָּ֣שֶׁל בָּ֔הּ בְּי֖וֹם שׁוּב֣וֹ מֵֽרִשְׁע֑וֹ וְצַדִּ֗יק לֹ֥א יוּכַ֛ל לִֽחְי֥וֹת בָּ֖הּ בְּי֥וֹם חֲטֹאתֽוֹ׃ (מה) בְּאָמְרִ֤י לַצַּדִּיק֙ חָיֹ֣ה יִֽחְיֶ֔ה וְהֽוּא־ בָטַ֥ח עַל־ צִדְקָת֖וֹ וְעָ֣שָׂה עָ֑וֶל כָּל־ (צדקתו) צִדְקֹתָיו֙ לֹ֣א תִזָּכַ֔רְנָה וּבְעַוְל֥וֹ אֲשֶׁר־ עָשָׂ֖ה בּ֥וֹ יָמֽוּת׃ (מו) (א) כָּל־ (חטאתו) חַטֹּאתָיו֙ אֲשֶׁ֣ר חָטָ֔א לֹ֥א תִזָּכַ֖רְנָה ל֑וֹ מִשְׁפָּ֧ט וּצְדָקָ֛ה עָשָׂ֖ה חָי֥וֹ יִֽחְיֶֽה׃ (ב) וְאָמְרוּ֙ בְּנֵ֣י עַמְּךָ֔ לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י וְהֵ֖מָּה דַּרְכָּ֥ם לֹֽא־ יִתָּכֵֽן׃ (ג) בְּשׁוּב־ צַדִּ֥יק מִצִּדְקָת֖וֹ וְעָ֣שָׂה עָ֑וֶל וּמֵ֖ת בָּהֶֽם׃ (ד) וּבְשׁ֤וּב רָשָׁע֙ מֵֽרִשְׁעָת֔וֹ וְעָשָׂ֥ה מִשְׁפָּ֖ט וּצְדָקָ֑ה עֲלֵיהֶ֖ם ה֥וּא יִֽחְיֶֽה׃ (ה) וַאֲמַרְתֶּ֕ם לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י אִ֧ישׁ כִּדְרָכָ֛יו אֶשְׁפּ֥וֹט אֶתְכֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל (ו) וַיְהִ֞י בִּשְׁתֵּ֧י עֶשְׂרֵ֣ה שָׁנָ֗ה בָּעֲשִׂרִ֛י בַּחֲמִשָּׁ֥ה לַחֹ֖דֶשׁ לְגָלוּתֵ֑נוּ בָּא־ אֵלַ֨י הַפָּלִ֧יט מִירוּשָׁלִַ֛ם לֵאמֹ֖ר הֻכְּתָ֥ה הָעִֽיר׃ (ז) וְיַד־ יְהוָה֩ הָיְתָ֨ה אֵלַ֜י בָּעֶ֗רֶב לִפְנֵי֙ בּ֣וֹא הַפָּלִ֔יט וַיִּפְתַּ֣ח אֶת־ פִּ֔י עַד־ בּ֥וֹא אֵלַ֖י בַּבֹּ֑קֶר וַיִּפָּ֣תַח פִּ֔י וְלֹ֥א נֶאֱלַ֖מְתִּי עֽוֹד׃ (ח) וַיְהִ֥י דְבַר־ יְהֹוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ט) בֶּן־ אָדָ֗ם יֹ֠שְׁבֵי הֶחֳרָב֨וֹת הָאֵ֜לֶּה עַל־ אַדְמַ֤ת יִשְׂרָאֵל֙ אֹמְרִ֣ים לֵאמֹ֔ר אֶחָד֙ הָיָ֣ה אַבְרָהָ֔ם וַיִּירַ֖שׁ אֶת־ הָאָ֑רֶץ וַאֲנַ֣חְנוּ רַבִּ֔ים לָ֛נוּ נִתְּנָ֥ה הָאָ֖רֶץ לְמוֹרָשָֽׁה׃ (י) לָכֵן֩ אֱמֹ֨ר אֲלֵיהֶ֜ם כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה עַל־ הַדָּ֧ם ׀ תֹּאכֵ֛לוּ וְעֵינֵכֶ֛ם תִּשְׂא֥וּ אֶל־ גִּלּוּלֵיכֶ֖ם וְדָ֣ם תִּשְׁפֹּ֑כוּ וְהָאָ֖רֶץ תִּירָֽשׁוּ׃ (יא) עֲמַדְתֶּ֤ם עַֽל־ חַרְבְּכֶם֙ עֲשִׂיתֶ֣ן תּוֹעֵבָ֔ה וְאִ֛ישׁ אֶת־ אֵ֥שֶׁת רֵעֵ֖הוּ טִמֵּאתֶ֑ם וְהָאָ֖רֶץ תִּירָֽשׁוּ׃ (יב) כֹּֽה־ תֹאמַ֨ר אֲלֵהֶ֜ם כֹּה־ אָמַ֨ר אֲדֹנָ֣י יְהוִה֮ חַי־ אָנִי֒ אִם־ לֹ֞א אֲשֶׁ֤ר בֶּֽחֳרָבוֹת֙ בַּחֶ֣רֶב יִפֹּ֔לוּ וַֽאֲשֶׁר֙ עַל־ פְּנֵ֣י הַשָּׂדֶ֔ה לַחַיָּ֥ה נְתַתִּ֖יו לְאָכְל֑וֹ וַאֲשֶׁ֛ר בַּמְּצָד֥וֹת וּבַמְּעָר֖וֹת בַּדֶּ֥בֶר יָמֽוּתוּ׃ (יג) וְנָתַתִּ֤י אֶת־ הָאָ֙רֶץ֙ שְׁמָמָ֣ה וּמְשַׁמָּ֔ה וְנִשְׁבַּ֖ת גְּא֣וֹן עֻזָּ֑הּ וְשָֽׁמְמ֛וּ הָרֵ֥י יִשְׂרָאֵ֖ל מֵאֵ֥ין עוֹבֵֽר׃ (יד) וְיָדְע֖וּ כִּֽי־ אֲנִ֣י יְהוָ֑ה בְּתִתִּ֤י אֶת־ הָאָ֙רֶץ֙ שְׁמָמָ֣ה וּמְשַׁמָּ֔ה עַ֥ל כָּל־ תּוֹעֲבֹתָ֖ם אֲשֶׁ֥ר עָשֽׂוּ׃ (טו) וְאַתָּ֣ה בֶן־ אָדָ֔ם בְּנֵ֣י עַמְּךָ֗ הַנִּדְבָּרִ֤ים בְּךָ֙ אֵ֣צֶל הַקִּיר֔וֹת וּבְפִתְחֵ֖י הַבָּתִּ֑ים וְדִבֶּר־ חַ֣ד אֶת־ אַחַ֗ד אִ֤ישׁ אֶת־ אָחִיו֙ לֵאמֹ֔ר בֹּֽאוּ־ נָ֣א וְשִׁמְע֔וּ מָ֣ה הַדָּבָ֔ר הַיּוֹצֵ֖א מֵאֵ֥ת יְהוָֽה׃ (טז) וְיָב֣וֹאוּ אֵ֠לֶיךָ כִּמְבוֹא־ עָ֞ם וְיֵשְׁב֤וּ לְפָנֶ֙יךָ֙ עַמִּ֔י וְשָֽׁמְעוּ֙ אֶת־ דְּבָרֶ֔יךָ וְאוֹתָ֖ם לֹ֣א יַֽעֲשׂ֑וּ כִּֽי־ עֲגָבִ֤ים בְּפִיהֶם֙ הֵ֣מָּה עֹשִׂ֔ים אַחֲרֵ֥י בִצְעָ֖ם לִבָּ֥ם הֹלֵֽךְ׃ (יז) וְהִנְּךָ֤ לָהֶם֙ כְּשִׁ֣יר עֲגָבִ֔ים יְפֵ֥ה ק֖וֹל וּמֵטִ֣ב נַגֵּ֑ן וְשָֽׁמְעוּ֙ אֶת־ דְּבָרֶ֔יךָ וְעֹשִׂ֥ים אֵינָ֖ם אוֹתָֽם׃ (יח) וּבְבֹאָ֑הּ הִנֵּ֣ה בָאָ֔ה וְיָ֣דְע֔וּ כִּ֥י נָבִ֖יא הָיָ֥ה בְתוֹכָֽם׃ סוף פרק לג (יט) הָיוּ֙ רֹעִ֣ים אוֹתָ֔ם הֲל֣וֹא הַצֹּ֔אן יִרְע֖וּ הָרֹעִֽים׃ (כא) אֶת־ הַחֵ֤לֶב תֹּאכֵ֙לוּ֙ וְאֶת־ הַצֶּ֣מֶר תִּלְבָּ֔שׁוּ הַבְּרִיאָ֖ה תִּזְבָּ֑חוּ הַצֹּ֖אן לֹ֥א תִרְעֽוּ׃ (כב) אֶֽת־ הַנַּחְלוֹת֩ לֹ֨א וּבְפָֽרֶךְ׃ (כג) וַתְּפוּצֶ֖ינָה מִבְּלִ֣י רֹעֶ֑ה וַתִּהְיֶ֧ינָה לְאָכְלָ֛ה לְכָל־ חַיַּ֥ת הַשָּׂדֶ֖ה וַתְּפוּצֶֽינָה׃ (כד) יִשְׁגּ֤וּ צֹאנִי֙ בְּכָל־ הֶ֣הָרִ֔ים וְעַ֖ל כָּל־ גִּבְעָ֣ה רָמָ֑ה וְעַ֨ל כָּל־ פְּנֵ֤י הָאָ֙רֶץ֙ נָפֹ֣צוּ צֹאנִ֔י וְאֵ֥ין דּוֹרֵ֖שׁ וְאֵ֥ין מְבַקֵּֽשׁ׃ (כה) לָכֵ֣ן רֹעִ֔ים שִׁמְע֖וּ אֶת־ דְּבַ֥ר יְהוָֽה׃ (כו) חַי־ אָ֜נִי נְאֻ֣ם ׀ אֲדֹנָ֣י יְהוִ֗ה אִם־ לֹ֣א יַ֣עַן הֱיֽוֹת־ צֹאנִ֣י ׀ לָבַ֡ז וַתִּֽהְיֶינָה֩ צֹאנִ֨י לְאָכְלָ֜ה לְכָל־ חַיַּ֤ת הַשָּׂדֶה֙ מֵאֵ֣ין רֹעֶ֔ה וְלֹֽא־ דָרְשׁ֥וּ רֹעַ֖י אֶת־ צֹאנִ֑י וַיִּרְע֤וּ הָֽרֹעִים֙ אוֹתָ֔ם וְאֶת־ צֹאנִ֖י לֹ֥א רָעֽוּ׃ (כז) לָכֵן֙ הָֽרֹעִ֔ים שִׁמְע֖וּ דְּבַר־ יְהוָֽה׃ (כח) כֹּה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה הִנְנִ֨י אֶֽל־ הָרֹעִ֜ים וְֽדָרַשְׁתִּ֧י אֶת־ צֹאנִ֣י מִיָּדָ֗ם וְהִשְׁבַּתִּים֙ מֵרְע֣וֹת צֹ֔אן וְלֹא־ יִרְע֥וּ ע֛וֹד הָרֹעִ֖ים אוֹתָ֑ם וְהִצַּלְתִּ֤י צֹאנִי֙ מִפִּיהֶ֔ם וְלֹֽא־ תִהְיֶ֥יןָ לָהֶ֖ם לְאָכְלָֽה׃ (כט) כִּ֛י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הִנְנִי־ אָ֕נִי וְדָרַשְׁתִּ֥י אֶת־ צֹאנִ֖י וּבִקַּרְתִּֽים׃ (ל) כְּבַקָּרַת֩ רֹעֶ֨ה עֶדְר֜וֹ בְּיוֹם־ הֱיוֹת֤וֹ בְתוֹךְ־ צֹאנוֹ֙ נִפְרָשׁ֔וֹת כֵּ֖ן אֲבַקֵּ֣ר אֶת־ צֹאנִ֑י וְהִצַּלְתִּ֣י אֶתְהֶ֗ם מִכָּל־ הַמְּקוֹמֹת֙ אֲשֶׁ֣ר נָפֹ֣צוּ שָׁ֔ם בְּי֥וֹם עָנָ֖ן וַעֲרָפֶֽל׃ (לא) וְהוֹצֵאתִ֣ים מִן־ הָעַמִּ֗ים וְקִבַּצְתִּים֙ מִן־ הָ֣אֲרָצ֔וֹת וַהֲבִיאֹתִ֖ים אֶל־ אַדְמָתָ֑ם וּרְעִיתִים֙ אֶל־ הָרֵ֣י יִשְׂרָאֵ֔ל בָּאֲפִיקִ֕ים וּבְכֹ֖ל מוֹשְׁבֵ֥י הָאָֽרֶץ׃ (לב) בְּמִרְעֶה־ טּוֹב֙ אֶרְעֶ֣ה אֹתָ֔ם וּבְהָרֵ֥י מְרֽוֹם־ יִשְׂרָאֵ֖ל יִהְיֶ֣ה נְוֵהֶ֑ם שָׁ֤ם תִּרְבַּ֙צְנָה֙ בְּנָ֣וֶה טּ֔וֹב וּמִרְעֶ֥ה שָׁמֵ֛ן תִּרְעֶ֖ינָה אֶל־ הָרֵ֥י יִשְׂרָאֵֽל׃ (לג) אֲנִ֨י אֶרְעֶ֤ה צֹאנִי֙ וַאֲנִ֣י אַרְבִּיצֵ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לד) אֶת־ הָאֹבֶ֤דֶת אֲבַקֵּשׁ֙ וְאֶת־ הַנִּדַּ֣חַת אָשִׁ֔יב וְלַנִּשְׁבֶּ֣רֶת אֶחֱבֹ֔שׁ וְאֶת־ הַחוֹלָ֖ה אֲחַזֵּ֑ק וְאֶת־ הַשְּׁמֵנָ֧ה וְאֶת־ הַחֲזָקָ֛ה אַשְׁמִ֖יד אֶרְעֶ֥נָּה בְמִשְׁפָּֽט׃ (לה) וְאַתֵּ֣נָה צֹאנִ֔י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הִנְנִ֤י שֹׁפֵט֙ בֵּֽין־ שֶׂ֣ה לָשֶׂ֔ה לָאֵילִ֖ים וְלָעַתּוּדִֽים׃ (לו) הַמְעַ֣ט מִכֶּ֗ם הַמִּרְעֶ֤ה הַטּוֹב֙ תִּרְע֔וּ וְיֶ֙תֶר֙ מִרְעֵיכֶ֔ם תִּרְמְס֖וּ בְּרַגְלֵיכֶ֑ם וּמִשְׁקַע־ מַ֣יִם תִּשְׁתּ֔וּ וְאֵת֙ הַנּ֣וֹתָרִ֔ים בְּרַגְלֵיכֶ֖ם תִּרְפֹּשֽׂוּן׃ (לז) וְצֹאנִ֑י מִרְמַ֤ס רַגְלֵיכֶם֙ תִּרְעֶ֔ינָה וּמִרְפַּ֥שׂ רַגְלֵיכֶ֖ם תִּשְׁתֶּֽינָה׃ (לח) לָכֵ֗ן כֹּ֥ה אָמַ֛ר אֲדֹנָ֥י יְהוִ֖ה אֲלֵיהֶ֑ם הִנְנִי־ אָ֕נִי וְשָֽׁפַטְתִּי֙ בֵּֽין־ שֶׂ֣ה בִרְיָ֔ה וּבֵ֥ין שֶׂ֖ה רָזָֽה׃ (לט) יַ֗עַן בְּצַ֤ד וּבְכָתֵף֙ תֶּהְדֹּ֔פוּ וּבְקַרְנֵיכֶ֥ם תְּנַגְּח֖וּ כָּל־ הַנַּחְל֑וֹת עַ֣ד אֲשֶׁ֧ר הֲפִיצוֹתֶ֛ם אוֹתָ֖נָה אֶל־ הַחֽוּצָה׃ (מ) וְהוֹשַׁעְתִּ֣י לְצֹאנִ֔י וְלֹֽא־ תִהְיֶ֥ינָה ע֖וֹד לָבַ֑ז וְשָׁ֣פַטְתִּ֔י בֵּ֥ין שֶׂ֖ה לָשֶֽׂה׃ (מא) וַהֲקִמֹתִ֨י עֲלֵיהֶ֜ם רֹעֶ֤ה אֶחָד֙ וְרָעָ֣ה אֶתְהֶ֔ן אֵ֖ת עַבְדִּ֣י דָוִ֑יד ה֚וּא יִרְעֶ֣ה אֹתָ֔ם וְהֽוּא־ יִהְיֶ֥ה לָהֶ֖ן לְרֹעֶֽה׃ (מב) וַאֲנִ֣י יְהוָ֗ה אֶהְיֶ֤ה לָהֶם֙ לֵֽאלֹהִ֔ים וְעַבְדִּ֥י דָוִ֖ד נָשִׂ֣יא בְתוֹכָ֑ם אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (מג) וְכָרַתִּ֤י לָהֶם֙ בְּרִ֣ית שָׁל֔וֹם וְהִשְׁבַּתִּ֥י חַיָּֽה־ רָעָ֖ה מִן־ הָאָ֑רֶץ וְיָשְׁב֤וּ בַמִּדְבָּר֙ לָבֶ֔טַח וְיָשְׁנ֖וּ בַּיְּעָרִֽים׃
