יחזקאל פרק כב לפי החלוקה לסדרים

פרק כב בספר יחזקאל (31 פסוקים) נמצא ביחזקאל סדר יב (פסוקים מב–נו) וביחזקאל סדר יג (פסוקים א–טז).

יחזקאל סדר יב

(א) בְּרֵ֣יחַ נִיחֹחַ֮ אֶרְצֶ֣ה אֶתְכֶם֒ בְּהוֹצִיאִ֤י אֶתְכֶם֙ מִן־ הָ֣עַמִּ֔ים וְקִבַּצְתִּ֣י אֶתְכֶ֔ם מִן־ הָ֣אֲרָצ֔וֹת אֲשֶׁ֥ר נְפֹצֹתֶ֖ם בָּ֑ם וְנִקְדַּשְׁתִּ֥י בָכֶ֖ם לְעֵינֵ֥י הַגּוֹיִֽם׃ (ב) וִֽידַעְתֶּם֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה בַּהֲבִיאִ֥י אֶתְכֶ֖ם אֶל־ אַדְמַ֣ת יִשְׂרָאֵ֑ל אֶל־ הָאָ֗רֶץ אֲשֶׁ֤ר נָשָׂ֙אתִי֙ אֶת־ יָדִ֔י לָתֵ֥ת אוֹתָ֖הּ לַאֲבֽוֹתֵיכֶֽם׃ (ג) וּזְכַרְתֶּם־ שָׁ֗ם אֶת־ דַּרְכֵיכֶם֙ וְאֵת֙ כָּל־ עֲלִיל֣וֹתֵיכֶ֔ם אֲשֶׁ֥ר נִטְמֵאתֶ֖ם בָּ֑ם וּנְקֹֽטֹתֶם֙ בִּפְנֵיכֶ֔ם בְּכָל־ רָעוֹתֵיכֶ֖ם אֲשֶׁ֥ר עֲשִׂיתֶֽם׃ (ד) וִֽידַעְתֶּם֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה בַּעֲשׂוֹתִ֥י אִתְּכֶ֖ם לְמַ֣עַן שְׁמִ֑י לֹא֩ כְדַרְכֵיכֶ֨ם הָרָעִ֜ים וְכַעֲלִילֽוֹתֵיכֶ֤ם הַנִּשְׁחָתוֹת֙ בֵּ֣ית יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ה) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ו) בֶּן־ אָדָ֗ם שִׂ֤ים פָּנֶ֙יךָ֙ דֶּ֣רֶךְ תֵּימָ֔נָה וְהַטֵּ֖ף אֶל־ דָּר֑וֹם וְהִנָּבֵ֛א אֶל־ יַ֥עַר הַשָּׂדֶ֖ה נֶֽגֶב׃ (ז) וְאָֽמַרְתָּ֙ לְיַ֣עַר הַנֶּ֔גֶב שְׁמַ֖ע דְּבַר־ יְהוָ֑ה כֹּֽה־ אָמַ֣ר אֲדֹנָ֣י יְהוִ֡ה הִנְנִ֣י מַֽצִּית־ בְּךָ֣ ׀ אֵ֡שׁ וְאָכְלָ֣ה בְךָ֣ כָל־ עֵֽץ־ לַח֩ וְכָל־ עֵ֨ץ יָבֵ֤שׁ לֹֽא־ תִכְבֶּה֙ לַהֶ֣בֶת שַׁלְהֶ֔בֶת וְנִצְרְבוּ־ בָ֥הּ כָּל־ פָּנִ֖ים מִנֶּ֥גֶב צָפֽוֹנָה׃ (ח) וְרָאוּ֙ כָּל־ בָּשָׂ֔ר כִּ֛י אֲנִ֥י יְהוָ֖ה בִּֽעַרְתִּ֑יהָ לֹ֖א תִּכְבֶּֽה׃ (ט) וָאֹמַ֕ר אֲהָ֖הּ אֲדֹנָ֣י יְהוִ֑ה הֵ֚מָּה אֹמְרִ֣ים לִ֔י הֲלֹ֛א מְמַשֵּׁ֥ל מְשָׁלִ֖ים הֽוּא׃ (י) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יא) בֶּן־ אָדָ֗ם שִׂ֤ים פָּנֶ֙יךָ֙ אֶל־ יְר֣וּשָׁלִַ֔ם וְהַטֵּ֖ף אֶל־ מִקְדָּשִׁ֑ים וְהִנָּבֵ֖א אֶל־ אַדְמַ֥ת יִשְׂרָאֵֽל׃ (יב) וְאָמַרְתָּ֞ לְאַדְמַ֣ת יִשְׂרָאֵ֗ל כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֣י אֵלַ֔יִךְ וְהוֹצֵאתִ֥י חַרְבִּ֖י מִתַּעְרָ֑הּ וְהִכְרַתִּ֥י מִמֵּ֖ךְ צַדִּ֥יק וְרָשָֽׁע׃ (יג) יַ֛עַן אֲשֶׁר־ הִכְרַ֥תִּי מִמֵּ֖ךְ צַדִּ֣יק וְרָשָׁ֑ע לָ֠כֵן תֵּצֵ֨א חַרְבִּ֧י מִתַּעְרָ֛הּ אֶל־ כָּל־ בָּשָׂ֖ר מִנֶּ֥גֶב צָפֽוֹן׃ (יד) וְיָֽדְעוּ֙ כָּל־ בָּשָׂ֔ר כִּ֚י אֲנִ֣י יְהוָ֔ה הוֹצֵ֥אתִי חַרְבִּ֖י מִתַּעְרָ֑הּ לֹ֥א תָשׁ֖וּב עֽוֹד׃ (טו) וְאַתָּ֥ה בֶן־ אָדָ֖ם הֵֽאָנַ֑ח בְּשִׁבְר֤וֹן מָתְנַ֙יִם֙ וּבִמְרִיר֔וּת תֵּֽאָנַ֖ח לְעֵינֵיהֶֽם׃ (טז) וְהָיָה֙ כִּֽי־ יֹאמְר֣וּ אֵלֶ֔יךָ עַל־ מָ֖ה אַתָּ֣ה נֶאֱנָ֑ח וְאָמַרְתָּ֡ אֶל־ שְׁמוּעָ֣ה כִֽי־ בָאָ֡ה וְנָמֵ֣ס כָּל־ לֵב֩ וְרָפ֨וּ כָל־ יָדַ֜יִם וְכִהֲתָ֣ה כָל־ ר֗וּחַ וְכָל־ בִּרְכַּ֙יִם֙ תֵּלַ֣כְנָה מַּ֔יִם הִנֵּ֤ה בָאָה֙ וְנִֽהְיָ֔תָה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יח) בֶּן־ אָדָ֕ם הִנָּבֵא֙ וְאָ֣מַרְתָּ֔ כֹּ֖ה אָמַ֣ר אֲדֹנָ֑י אֱמֹ֕ר חֶ֥רֶב חֶ֛רֶב הוּחַ֖דָּה וְגַם־ מְרוּטָֽה׃ (יט) לְמַ֨עַן טְבֹ֤חַ טֶ֙בַח֙ הוּחַ֔דָּה לְמַעַן־ הֱיֵה־ לָ֥הּ בָּ֖רָק מֹרָ֑טָּה א֣וֹ נָשִׂ֔ישׂ שֵׁ֥בֶט בְּנִ֖י מֹאֶ֥סֶת כָּל־ עֵֽץ׃ (כ) וַיִּתֵּ֥ן אֹתָ֛הּ לְמָרְטָ֖ה לִתְפֹּ֣שׂ בַּכָּ֑ף הִֽיא־ הוּחַ֤דָּה חֶ֙רֶב֙ וְהִ֣יא מֹרָ֔טָּה לָתֵ֥ת אוֹתָ֖הּ בְּיַד־ הוֹרֵֽג׃ (כא) זְעַ֤ק וְהֵילֵל֙ בֶּן־ אָדָ֔ם כִּי־ הִיא֙ הָיתָ֣ה בְעַמִּ֔י הִ֖יא בְּכָל־ נְשִׂיאֵ֣י יִשְׂרָאֵ֑ל מְגוּרֵ֤י אֶל־ חֶ֙רֶב֙ הָי֣וּ אֶת־ עַמִּ֔י לָכֵ֖ן סְפֹ֥ק אֶל־ יָרֵֽךְ׃ (כב) כִּ֣י בֹ֔חַן וּמָ֕ה אִם־ גַּם־ שֵׁ֥בֶט מֹאֶ֖סֶת לֹ֣א יִֽהְיֶ֑ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (כג) וְאַתָּ֣ה בֶן־ אָדָ֔ם הִנָּבֵ֕א וְהַ֖ךְ כַּ֣ף אֶל־ כָּ֑ף וְתִכָּפֵ֞ל חֶ֤רֶב שְׁלִישִׁ֙תָה֙ חֶ֣רֶב חֲלָלִ֔ים הִ֗יא חֶ֚רֶב חָלָ֣ל הַגָּד֔וֹל הַחֹדֶ֖רֶת לָהֶֽם׃ (כד) לְמַ֣עַן ׀ לָמ֣וּג לֵ֗ב וְהַרְבֵּה֙ הַמִּכְשֹׁלִ֔ים עַ֚ל כָּל־ שַׁ֣עֲרֵיהֶ֔ם נָתַ֖תִּי אִבְחַת־ חָ֑רֶב אָ֛ח עֲשׂוּיָ֥ה לְבָרָ֖ק מְעֻטָּ֥ה לְטָֽבַח׃ (כה) הִתְאַחֲדִ֥י הֵימִ֖נִי הָשִׂ֣ימִי הַשְׂמִ֑ילִי אָ֖נָה פָּנַ֥יִךְ מֻעָדֽוֹת׃ (כו) וְגַם־ אֲנִ֗י אַכֶּ֤ה כַפִּי֙ אֶל־ כַּפִּ֔י וַהֲנִחֹתִ֖י חֲמָתִ֑י אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (כז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כח) וְאַתָּ֨ה בֶן־ אָדָ֜ם שִׂים־ לְךָ֣ ׀ שְׁנַ֣יִם דְּרָכִ֗ים לָבוֹא֙ חֶ֣רֶב מֶֽלֶךְ־ בָּבֶ֔ל מֵאֶ֥רֶץ אֶחָ֖ד יֵצְא֣וּ שְׁנֵיהֶ֑ם וְיָ֣ד בָּרֵ֔א בְּרֹ֥אשׁ דֶּֽרֶךְ־ עִ֖יר בָּרֵֽא׃ (כט) דֶּ֣רֶךְ תָּשִׂ֔ים לָב֣וֹא חֶ֔רֶב אֵ֖ת רַבַּ֣ת בְּנֵֽי־ עַמּ֑וֹן וְאֶת־ יְהוּדָ֥ה בִירוּשָׁלִַ֖ם בְּצוּרָֽה׃ (ל) כִּֽי־ עָמַ֨ד מֶלֶךְ־ בָּבֶ֜ל אֶל־ אֵ֣ם הַדֶּ֗רֶךְ בְּרֹ֛אשׁ שְׁנֵ֥י הַדְּרָכִ֖ים לִקְסָם־ קָ֑סֶם קִלְקַ֤ל בַּֽחִצִּים֙ שָׁאַ֣ל בַּתְּרָפִ֔ים רָאָ֖ה בַּכָּבֵֽד׃ (לא) בִּֽימִינ֞וֹ הָיָ֣ה ׀ הַקֶּ֣סֶם יְרוּשָׁלִַ֗ם לָשׂ֤וּם כָּרִים֙ לִפְתֹּ֤חַ פֶּה֙ בְּרֶ֔צַח לְהָרִ֥ים ק֖וֹל בִּתְרוּעָ֑ה לָשׂ֤וּם כָּרִים֙ עַל־ שְׁעָרִ֔ים לִשְׁפֹּ֥ךְ סֹלְלָ֖ה לִבְנ֥וֹת דָּיֵֽק׃ (לב) וְהָיָ֨ה לָהֶ֤ם (כקסום)־ כִּקְסָם־ שָׁוְא֙ בְּעֵ֣ינֵיהֶ֔ם שְׁבֻעֵ֥י שְׁבֻע֖וֹת לָהֶ֑ם וְהֽוּא־ מַזְכִּ֥יר עָוֺ֖ן לְהִתָּפֵֽשׂ׃ (לג) לָכֵ֗ן כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ יַ֗עַן הַזְכַּרְכֶם֙ עֲוֺ֣נְכֶ֔ם בְּהִגָּל֣וֹת פִּשְׁעֵיכֶ֗ם לְהֵֽרָאוֹת֙ חַטֹּ֣אותֵיכֶ֔ם בְּכֹ֖ל עֲלִילֽוֹתֵיכֶ֑ם יַ֚עַן הִזָּ֣כֶרְכֶ֔ם בַּכַּ֖ף תִּתָּפֵֽשׂוּ׃ (לד) וְאַתָּה֙ חָלָ֣ל רָשָׁ֔ע נְשִׂ֖יא יִשְׂרָאֵ֑ל אֲשֶׁר־ בָּ֣א יוֹמ֔וֹ בְּעֵ֖ת עֲוֺ֥ן קֵֽץ׃ (לה) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הָסִיר֙ הַמִּצְנֶ֔פֶת וְהָרִ֖ים הָֽעֲטָרָ֑ה זֹ֣את לֹא־ זֹ֔את הַשָּׁפָ֣לָה הַגְבֵּ֔הַ וְהַגָּבֹ֖הַ הַשְׁפִּֽיל׃ (לו) עַוָּ֥ה עַוָּ֖ה עַוָּ֣ה אֲשִׂימֶ֑נָּה גַּם־ זֹאת֙ לֹ֣א הָיָ֔ה עַד־ בֹּ֛א אֲשֶׁר־ ל֥וֹ הַמִּשְׁפָּ֖ט וּנְתַתִּֽיו׃ (לז) וְאַתָּ֣ה בֶן־ אָדָ֗ם הִנָּבֵ֤א וְאָֽמַרְתָּ֙ כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהֹוִ֔ה אֶל־ בְּנֵ֥י עַמּ֖וֹן וְאֶל־ חֶרְפָּתָ֑ם וְאָמַרְתָּ֗ חֶ֣רֶב חֶ֤רֶב פְּתוּחָה֙ לְטֶ֣בַח מְרוּטָ֔ה לְהָכִ֖יל לְמַ֥עַן בָּרָֽק׃ (לח) בַּחֲז֥וֹת לָךְ֙ שָׁ֔וְא בִּקְסָם־ לָ֖ךְ כָּזָ֑ב לָתֵ֣ת אוֹתָ֗ךְ אֶֽל־ צַוְּארֵי֙ חַֽלְלֵ֣י רְשָׁעִ֔ים אֲשֶׁר־ בָּ֣א יוֹמָ֔ם בְּעֵ֖ת עֲוֺ֥ן קֵֽץ׃ (לט) הָשַׁ֖ב אֶל־ תַּעְרָ֑הּ בִּמְק֧וֹם אֲשֶׁר־ נִבְרֵ֛את בְּאֶ֥רֶץ מְכֻרוֹתַ֖יִךְ אֶשְׁפֹּ֥ט אֹתָֽךְ׃ (מ) וְשָׁפַכְתִּ֤י עָלַ֙יִךְ֙ זַעְמִ֔י בְּאֵ֥שׁ עֶבְרָתִ֖י אָפִ֣יחַ עָלָ֑יִךְ וּנְתַתִּ֗יךְ בְּיַד֙ אֲנָשִׁ֣ים בֹּֽעֲרִ֔ים חָרָשֵׁ֖י מַשְׁחִֽית׃ (מא) לָאֵ֤שׁ תִּֽהְיֶה֙ לְאָכְלָ֔ה לָקָ֔חוּ וְאוֹתָ֖הּ בַּחֶ֣רֶב הָרָ֑גוּ וַתְּהִי־ שֵׁם֙ לַנָּשִׁ֔ים וּשְׁפוּטִ֖ים עָ֥שׂוּ בָֽהּ׃ (כז) וַתֵּ֙רֶא֙ אֲחוֹתָ֣הּ אָהֳלִיבָ֔ה וַתַּשְׁחֵ֥ת עַגְבָתָ֖הּ מִמֶּ֑נָּה וְאֶת־ תַּ֨זְנוּתֶ֔יהָ מִזְּנוּנֵ֖י אֲחוֹתָֽהּ׃ (כח) אֶל־ בְּנֵי֩ אַשּׁ֨וּר עָגָ֜בָה פַּח֨וֹת וּסְגָנִ֤ים קְרֹבִים֙ לְבֻשֵׁ֣י מִכְל֔וֹל פָּרָשִׁ֖ים רֹכְבֵ֣י סוּסִ֑ים בַּח֥וּרֵי חֶ֖מֶד כֻּלָּֽם׃ (כט) וָאֵ֖רֶא כִּ֣י נִטְמָ֑אָה דֶּ֥רֶךְ אֶחָ֖ד לִשְׁתֵּיהֶֽן׃ (ל) וַתּ֖וֹסֶף אֶל־ תַּזְנוּתֶ֑יהָ וַתֵּ֗רֶא אַנְשֵׁי֙ מְחֻקֶּ֣ה עַל־ הַקִּ֔יר צַלְמֵ֣י (כשדיים) כַשְׂדִּ֔ים חֲקֻקִ֖ים בַּשָּׁשַֽׁר׃ (לא) חֲגוֹרֵ֨י אֵז֜וֹר בְּמָתְנֵיהֶ֗ם סְרוּחֵ֤י טְבוּלִים֙ בְּרָ֣אשֵׁיהֶ֔ם מַרְאֵ֥ה שָׁלִשִׁ֖ים כֻּלָּ֑ם דְּמ֤וּת בְּנֵֽי־ בָבֶל֙ כַּשְׂדִּ֔ים אֶ֖רֶץ מוֹלַדְתָּֽם׃ (לב) (ותעגב) וַתַּעְגְּבָ֥ה עֲלֵיהֶ֖ם לְמַרְאֵ֣ה עֵינֶ֑יהָ וַתִּשְׁלַ֧ח מַלְאָכִ֛ים אֲלֵיהֶ֖ם כַּשְׂדִּֽימָה׃ (לג) וַיָּבֹ֨אוּ אֵלֶ֤יהָ בְנֵֽי־ בָבֶל֙ לְמִשְׁכַּ֣ב דֹּדִ֔ים וַיְטַמְּא֥וּ אוֹתָ֖הּ בְּתַזְנוּתָ֑ם וַתִּ֨טְמָא־ בָ֔ם וַתֵּ֥קַע נַפְשָׁ֖הּ מֵהֶֽם׃ (לד) וַתְּגַל֙ תַּזְנוּתֶ֔יהָ וַתְּגַ֖ל אֶת־ עֶרְוָתָ֑הּ וַתֵּ֤קַע נַפְשִׁי֙ מֵֽעָלֶ֔יהָ כַּאֲשֶׁ֛ר נָקְעָ֥ה נַפְשִׁ֖י מֵעַ֥ל אֲחוֹתָֽהּ׃ (לה) וַתַּרְבֶּ֖ה אֶת־ תַּזְנוּתֶ֑יהָ לִזְכֹּר֙ אֶת־ יְמֵ֣י נְעוּרֶ֔יהָ אֲשֶׁ֥ר זָנְתָ֖ה בְּאֶ֥רֶץ מִצְרָֽיִם׃ (לו) וַֽתַּעְגְּבָ֔ה עַ֖ל פִּֽלַגְשֵׁיהֶ֑ם אֲשֶׁ֤ר בְּשַׂר־ חֲמוֹרִים֙ בְּשָׂרָ֔ם וְזִרְמַ֥ת סוּסִ֖ים זִרְמָתָֽם׃ (לז) וַֽתִּפְקְדִ֔י אֵ֖ת זִמַּ֣ת נְעוּרָ֑יִךְ בַּעְשׂ֤וֹת מִמִּצְרַ֙יִם֙ דַּדַּ֔יִךְ לְמַ֖עַן שְׁדֵ֥י נְעוּרָֽיִךְ׃ (לח) לָכֵ֣ן אָהֳלִיבָ֗ה כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִנְנִ֨י מֵעִ֤יר אֶת־ מְאַהֲבַ֙יִךְ֙ עָלַ֔יִךְ אֵ֛ת אֲשֶׁר־ נָקְעָ֥ה נַפְשֵׁ֖ךְ מֵהֶ֑ם וַהֲבֵאתִ֥ים עָלַ֖יִךְ מִסָּבִֽיב׃ (לט) בְּנֵ֧י בָבֶ֣ל וְכָל־ כַּשְׂדִּ֗ים פְּק֤וֹד וְשׁ֙וֹעַ֙ וְק֔וֹעַ כָּל־ בְּנֵ֥י אַשּׁ֖וּר אוֹתָ֑ם בַּח֨וּרֵי חֶ֜מֶד פַּח֤וֹת וּסְגָנִים֙ כֻּלָּ֔ם שָֽׁלִשִׁים֙ וּקְרוּאִ֔ים רֹכְבֵ֥י סוּסִ֖ים כֻּלָּֽם׃ (מ) וּבָ֣אוּ עָלַ֡יִךְ הֹ֠צֶן רֶ֤כֶב וְגַלְגַּל֙ וּבִקְהַ֣ל עַמִּ֔ים צִנָּ֤ה וּמָגֵן֙ וְקוֹבַ֔ע יָשִׂ֥ימוּ עָלַ֖יִךְ סָבִ֑יב וְנָתַתִּ֤י לִפְנֵיהֶם֙ מִשְׁפָּ֔ט וּשְׁפָט֖וּךְ בְּמִשְׁפְּטֵיהֶֽם׃ (מא) וְנָתַתִּ֨י קִנְאָתִ֜י בָּ֗ךְ וְעָשׂ֤וּ אוֹתָךְ֙ בְּחֵמָ֔ה אַפֵּ֤ךְ וְאָזְנַ֙יִךְ֙ יָסִ֔ירוּ וְאַחֲרִיתֵ֖ךְ בַּחֶ֣רֶב תִּפּ֑וֹל הֵ֗מָּה בָּנַ֤יִךְ וּבְנוֹתַ֙יִךְ֙ יִקָּ֔חוּ וְאַחֲרִיתֵ֖ךְ תֵּאָכֵ֥ל בָּאֵֽשׁ׃ (מב) וְהִפְשִׁיט֖וּךְ אֶת־ בְּגָדָ֑יִךְ וְלָקְח֖וּ כְּלֵ֥י תִפְאַרְתֵּֽךְ׃
