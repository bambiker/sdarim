פרק כט בספר דברי הימים ב (36 פסוקים) נמצא בדברי הימים סדר כא (פסוקים נט–סח) ובדברי הימים סדר כב (פסוקים א–כו).

דברי הימים סדר כא

(א) ה֚וּא בָּנָ֣ה אֶת־ אֵיל֔וֹת וַיְשִׁיבֶ֖הָ לִֽיהוּדָ֑ה אַחֲרֵ֥י שְׁכַֽב־ הַמֶּ֖לֶךְ עִם־ אֲבֹתָֽיו׃ (ב) בֶּן־ שֵׁ֨שׁ עֶשְׂרֵ֤ה שָׁנָה֙ עֻזִּיָּ֣הוּ בְמָלְכ֔וֹ וַחֲמִשִּׁ֤ים וּשְׁתַּ֙יִם֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ (יכיליה) יְכָלְיָ֖ה מִן־ יְרוּשָׁלִָֽם׃ (ג) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה אֲמַצְיָ֥הוּ אָבִֽיו׃ (ד) וַיְהִי֙ לִדְרֹ֣שׁ אֱלֹהִ֔ים בִּימֵ֣י זְכַרְיָ֔הוּ הַמֵּבִ֖ין בִּרְאֹ֣ת הָאֱלֹהִ֑ים וּבִימֵי֙ דָּרְשׁ֣וֹ אֶת־ יְהוָ֔ה הִצְלִיח֖וֹ הָאֱלֹהִֽים׃ (ה) וַיֵּצֵא֙ וַיִּלָּ֣חֶם בַּפְּלִשְׁתִּ֔ים וַיִּפְרֹ֞ץ אֶת־ ח֣וֹמַת גַּ֗ת וְאֵת֙ חוֹמַ֣ת יַבְנֵ֔ה וְאֵ֖ת חוֹמַ֣ת אַשְׁדּ֑וֹד וַיִּבְנֶ֣ה עָרִ֔ים בְּאַשְׁדּ֖וֹד וּבַפְּלִשְׁתִּֽים׃ (ו) וַיַּעְזְרֵ֨הוּ הָֽאֱלֹהִ֜ים עַל־ פְּלִשְׁתִּ֧ים וְעַל־ (הערביים) הָֽעַרְבִ֛ים הַיֹּשְׁבִ֥ים בְּגוּר־ בָּ֖עַל וְהַמְּעוּנִֽים׃ (ז) וַיִּתְּנ֧וּ הָֽעַמּוֹנִ֛ים מִנְחָ֖ה לְעֻזִּיָּ֑הוּ וַיֵּ֤לֶךְ שְׁמוֹ֙ עַד־ לְב֣וֹא מִצְרַ֔יִם כִּ֥י הֶחֱזִ֖יק עַד־ לְמָֽעְלָה׃ (ח) וַיִּ֨בֶן עֻזִּיָּ֤הוּ מִגְדָּלִים֙ בִּיר֣וּשָׁלִַ֔ם עַל־ שַׁ֧עַר הַפִּנָּ֛ה וְעַל־ שַׁ֥עַר הַגַּ֖יְא וְעַל־ הַמִּקְצ֑וֹעַ וַֽיְחַזְּקֵֽם׃ (ט) וַיִּ֨בֶן מִגְדָּלִ֜ים בַּמִּדְבָּ֗ר וַיַּחְצֹב֙ בֹּר֣וֹת רַבִּ֔ים כִּ֤י מִקְנֶה־ רַּב֙ הָ֣יָה ל֔וֹ וּבַשְּׁפֵלָ֖ה וּבַמִּישׁ֑וֹר אִכָּרִ֣ים וְכֹֽרְמִ֗ים בֶּהָרִים֙ וּבַכַּרְמֶ֔ל כִּֽי־ אֹהֵ֥ב אֲדָמָ֖ה הָיָֽה׃ (י) וַיְהִ֣י לְעֻזִּיָּ֡הוּ חַיִל֩ עֹשֵׂ֨ה מִלְחָמָ֜ה יוֹצְאֵ֧י צָבָ֣א לִגְד֗וּד בְּמִסְפַּר֙ פְּקֻדָּתָ֔ם בְּיַד֙ (יעואל) יְעִיאֵ֣ל הַסּוֹפֵ֔ר וּמַעֲשֵׂיָ֖הוּ הַשּׁוֹטֵ֑ר עַ֚ל יַד־ חֲנַנְיָ֔הוּ מִשָּׂרֵ֖י הַמֶּֽלֶךְ׃ (יא) כֹּ֠ל מִסְפַּ֞ר רָאשֵׁ֤י הָאָבוֹת֙ לְגִבּ֣וֹרֵי חָ֔יִל אַלְפַּ֖יִם וְשֵׁ֥שׁ מֵאֽוֹת׃ (יב) וְעַל־ יָדָם֩ חֵ֨יל צָבָ֜א שְׁלֹ֧שׁ מֵא֣וֹת אֶ֗לֶף וְשִׁבְעַ֤ת אֲלָפִים֙ וַחֲמֵ֣שׁ מֵא֔וֹת עוֹשֵׂ֥י מִלְחָמָ֖ה בְּכֹ֣חַ חָ֑יִל לַעְזֹ֥ר לַמֶּ֖לֶךְ עַל־ הָאוֹיֵֽב׃ (יג) וַיָּכֶן֩ לָהֶ֨ם עֻזִּיָּ֜הוּ לְכָל־ הַצָּבָ֗א מָגִנִּ֤ים וּרְמָחִים֙ וְכ֣וֹבָעִ֔ים וְשִׁרְיֹנ֖וֹת וּקְשָׁת֑וֹת וּלְאַבְנֵ֖י קְלָעִֽים׃ (יד) וַיַּ֣עַשׂ ׀ בִּירוּשָׁלִַ֨ם חִשְּׁבֹנ֜וֹת מַחֲשֶׁ֣בֶת חוֹשֵׁ֗ב לִהְי֤וֹת עַל־ הַמִּגְדָּלִים֙ וְעַל־ הַפִּנּ֔וֹת לִירוֹא֙ בַּֽחִצִּ֔ים וּבָאֲבָנִ֖ים גְּדֹל֑וֹת וַיֵּצֵ֤א שְׁמוֹ֙ עַד־ לְמֵ֣רָח֔וֹק כִּֽי־ הִפְלִ֥יא לְהֵעָזֵ֖ר עַ֥ד כִּֽי־ חָזָֽק׃ (טו) וּכְחֶזְקָת֗וֹ גָּבַ֤הּ לִבּוֹ֙ עַד־ לְהַשְׁחִ֔ית וַיִּמְעַ֖ל בַּיהוָ֣ה אֱלֹהָ֑יו וַיָּבֹא֙ אֶל־ הֵיכַ֣ל יְהוָ֔ה לְהַקְטִ֖יר עַל־ מִזְבַּ֥ח הַקְּטֹֽרֶת׃ (טז) וַיָּבֹ֥א אַחֲרָ֖יו עֲזַרְיָ֣הוּ הַכֹּהֵ֑ן וְעִמּ֞וֹ כֹּהֲנִ֧ים ׀ לַיהוָ֛ה שְׁמוֹנִ֖ים בְּנֵי־ חָֽיִל׃ (יז) וַיַּעַמְד֞וּ עַל־ עֻזִּיָּ֣הוּ הַמֶּ֗לֶךְ וַיֹּ֤אמְרוּ לוֹ֙ לֹא־ לְךָ֣ עֻזִּיָּ֗הוּ לְהַקְטִיר֙ לַֽיהוָ֔ה כִּ֣י לַכֹּהֲנִ֧ים בְּנֵי־ אַהֲרֹ֛ן הַמְקֻדָּשִׁ֖ים לְהַקְטִ֑יר צֵ֤א מִן־ הַמִּקְדָּשׁ֙ כִּ֣י מָעַ֔לְתָּ וְלֹֽא־ לְךָ֥ לְכָב֖וֹד מֵיְהוָ֥ה אֱלֹהִֽים׃ (יח) וַיִּזְעַף֙ עֻזִּיָּ֔הוּ וּבְיָד֥וֹ מִקְטֶ֖רֶת לְהַקְטִ֑יר וּבְזַעְפּ֣וֹ עִם־ הַכֹּהֲנִ֗ים וְ֠הַצָּרַעַת זָרְחָ֨ה בְמִצְח֜וֹ לִפְנֵ֤י הַכֹּֽהֲנִים֙ בְּבֵ֣ית יְהוָ֔ה מֵעַ֖ל לְמִזְבַּ֥ח הַקְּטֹֽרֶת׃ (יט) וַיִּ֣פֶן אֵלָ֡יו עֲזַרְיָהוּ֩ כֹהֵ֨ן הָרֹ֜אשׁ וְכָל־ הַכֹּהֲנִ֗ים וְהִנֵּה־ ה֤וּא מְצֹרָע֙ בְּמִצְח֔וֹ וַיַּבְהִל֖וּהוּ מִשָּׁ֑ם וְגַם־ הוּא֙ נִדְחַ֣ף לָצֵ֔את כִּ֥י נִגְּע֖וֹ יְהוָֽה׃ (כ) וַיְהִי֩ עֻזִּיָּ֨הוּ הַמֶּ֜לֶךְ מְצֹרָ֣ע ׀ עַד־ י֣וֹם מוֹת֗וֹ וַיֵּ֜שֶׁב בֵּ֤ית (החפשות) הַֽחָפְשִׁית֙ מְצֹרָ֔ע כִּ֥י נִגְזַ֖ר מִבֵּ֣ית יְהוָ֑ה וְיוֹתָ֤ם בְּנוֹ֙ עַל־ בֵּ֣ית הַמֶּ֔לֶךְ שׁוֹפֵ֖ט אֶת־ עַ֥ם הָאָֽרֶץ׃ (כא) וְיֶ֙תֶר֙ דִּבְרֵ֣י עֻזִּיָּ֔הוּ הָרִאשֹׁנִ֖ים וְהָאֲחֲרֹנִ֑ים כָּתַ֛ב יְשַֽׁעְיָ֥הוּ בֶן־ אָמ֖וֹץ הַנָּבִֽיא׃ (כב) וַיִּשְׁכַּ֨ב עֻזִּיָּ֜הוּ עִם־ אֲבֹתָ֗יו וַיִּקְבְּר֨וּ אֹת֤וֹ עִם־ אֲבֹתָיו֙ בִּשְׂדֵ֤ה הַקְּבוּרָה֙ אֲשֶׁ֣ר לַמְּלָכִ֔ים כִּ֥י אָמְר֖וּ מְצוֹרָ֣ע ה֑וּא וַיִּמְלֹ֛ךְ יוֹתָ֥ם בְּנ֖וֹ תַּחְתָּֽיו׃ (כג) בֶּן־ עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָה֙ יוֹתָ֣ם בְּמָלְכ֔וֹ וְשֵׁשׁ־ עֶשְׂרֵ֣ה שָׁנָ֔ה מָלַ֖ךְ בִּֽירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ יְרוּשָׁ֖ה בַּת־ צָדֽוֹק׃ (כד) וַיַּ֨עַשׂ הַיָּשָׁ֜ר בְּעֵינֵ֣י יְהוָ֗ה כְּכֹ֤ל אֲשֶׁר־ עָשָׂה֙ עֻזִּיָּ֣הוּ אָבִ֔יו רַ֕ק לֹא־ בָ֖א אֶל־ הֵיכַ֣ל יְהוָ֑ה וְע֥וֹד הָעָ֖ם מַשְׁחִיתִֽים׃ (כה) ה֗וּא בָּנָ֛ה אֶת־ שַׁ֥עַר בֵּית־ יְהוָ֖ה הָעֶלְי֑וֹן וּבְחוֹמַ֥ת הָעֹ֛פֶל בָּנָ֖ה לָרֹֽב׃ (כו) וְעָרִ֥ים בָּנָ֖ה בְּהַר־ יְהוּדָ֑ה וּבֶחֳרָשִׁ֣ים בָּנָ֔ה בִּֽירָנִיּ֖וֹת וּמִגְדָּלִֽים׃ (כז) וְ֠הוּא נִלְחַ֞ם עִם־ מֶ֣לֶךְ בְּנֵי־ עַמּוֹן֮ וַיֶּחֱזַ֣ק עֲלֵיהֶם֒ וַיִּתְּנוּ־ ל֨וֹ בְנֵֽי־ עַמּ֜וֹן בַּשָּׁנָ֣ה הַהִ֗יא מֵאָה֙ כִּכַּר־ כֶּ֔סֶף וַעֲשֶׂ֨רֶת אֲלָפִ֤ים כֹּרִים֙ חִטִּ֔ים וּשְׂעוֹרִ֖ים עֲשֶׂ֣רֶת אֲלָפִ֑ים זֹ֗את הֵשִׁ֤יבוּ לוֹ֙ בְּנֵ֣י עַמּ֔וֹן וּבַשָּׁנָ֥ה הַשֵּׁנִ֖ית וְהַשְּׁלִשִֽׁית׃ (כח) וַיִּתְחַזֵּ֖ק יוֹתָ֑ם כִּ֚י הֵכִ֣ין דְּרָכָ֔יו לִפְנֵ֖י יְהוָ֥ה אֱלֹהָֽיו׃ (כט) וְיֶתֶר֙ דִּבְרֵ֣י יוֹתָ֔ם וְכָל־ מִלְחֲמֹתָ֖יו וּדְרָכָ֑יו הִנָּ֣ם כְּתוּבִ֔ים עַל־ סֵ֥פֶר מַלְכֵֽי־ יִשְׂרָאֵ֖ל וִיהוּדָֽה׃ (ל) בֶּן־ עֶשְׂרִ֧ים וְחָמֵ֛שׁ שָׁנָ֖ה הָיָ֣ה בְמָלְכ֑וֹ וְשֵׁשׁ־ עֶשְׂרֵ֣ה שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָֽם׃ (לא) וַיִּשְׁכַּ֤ב יוֹתָם֙ עִם־ אֲבֹתָ֔יו וַיִּקְבְּר֥וּ אֹת֖וֹ בְּעִ֣יר דָּוִ֑יד וַיִּמְלֹ֛ךְ אָחָ֥ז בְּנ֖וֹ תַּחְתָּֽיו׃ (לב) בֶּן־ עֶשְׂרִ֤ים שָׁנָה֙ אָחָ֣ז בְּמָלְכ֔וֹ וְשֵׁשׁ־ עֶשְׂרֵ֣ה שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְלֹא־ עָשָׂ֧ה הַיָּשָׁ֛ר בְּעֵינֵ֥י יְהוָ֖ה כְּדָוִ֥יד אָבִֽיו׃ (לג) וַיֵּ֕לֶךְ בְּדַרְכֵ֖י מַלְכֵ֣י יִשְׂרָאֵ֑ל וְגַ֧ם מַסֵּכ֛וֹת עָשָׂ֖ה לַבְּעָלִֽים׃ (לד) וְה֥וּא הִקְטִ֖יר בְּגֵ֣יא בֶן־ הִנֹּ֑ם וַיַּבְעֵ֤ר אֶת־ בָּנָיו֙ בָּאֵ֔שׁ כְּתֹֽעֲבוֹת֙ הַגּוֹיִ֔ם אֲשֶׁר֙ הֹרִ֣ישׁ יְהוָ֔ה מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (לה) וַיְזַבֵּ֧חַ וַיְקַטֵּ֛ר בַּבָּמ֖וֹת וְעַל־ הַגְּבָע֑וֹת וְתַ֖חַת כָּל־ עֵ֥ץ רַעֲנָֽן׃ (לו) וַֽיִּתְּנֵ֜הוּ יְהוָ֣ה אֱלֹהָיו֮ בְּיַ֣ד מֶ֣לֶךְ אֲרָם֒ וַיַּ֨כּוּ־ ב֔וֹ וַיִּשְׁבּ֤וּ וְעֶשְׂרִ֥ים אֶ֛לֶף בְּי֥וֹם אֶחָ֖ד הַכֹּ֣ל בְּנֵי־ חָ֑יִל בְּעָזְבָ֕ם אֶת־ יְהוָ֖ה אֱלֹהֵ֥י אֲבוֹתָֽם׃ (לח) וַֽיַּהֲרֹ֞ג זִכְרִ֣י ׀ גִּבּ֣וֹר אֶפְרַ֗יִם אֶת־ מַעֲשֵׂיָ֙הוּ֙ בֶּן־ הַמֶּ֔לֶךְ וְאֶת־ עַזְרִיקָ֖ם נְגִ֣יד הַבָּ֑יִת וְאֶת־ אֶלְקָנָ֖ה מִשְׁנֵ֥ה הַמֶּֽלֶךְ׃ (לט) וַיִּשְׁבּוּ֩ בְנֵֽי־ יִשְׂרָאֵ֨ל מֵֽאֲחֵיהֶ֜ם מָאתַ֣יִם אֶ֗לֶף נָשִׁים֙ בָּנִ֣ים וּבָנ֔וֹת וְגַם־ שָׁלָ֥ל רָ֖ב בָּזְז֣וּ מֵהֶ֑ם וַיָּבִ֥יאוּ אֶת־ הַשָּׁלָ֖ל לְשֹׁמְרֽוֹן׃ (מ) וְ֠שָׁם הָיָ֨ה נָבִ֥יא לַֽיהוָה֮ עֹדֵ֣ד שְׁמוֹ֒ וַיֵּצֵ֗א לִפְנֵ֤י הַצָּבָא֙ הַבָּ֣א לְשֹׁמְר֔וֹן וַיֹּ֣אמֶר לָהֶ֗ם הִ֠נֵּה בַּחֲמַ֨ת יְהוָ֧ה אֱלֹהֵֽי־ אֲבוֹתֵיכֶ֛ם עַל־ יְהוּדָ֖ה נְתָנָ֣ם בְּיֶדְכֶ֑ם וַתַּֽהַרְגוּ־ בָ֣ם בְזַ֔עַף עַ֥ד לַשָּׁמַ֖יִם הִגִּֽיעַ׃ (מא) וְ֠עַתָּה בְּנֵֽי־ יְהוּדָ֤ה וִֽירוּשָׁלִַ֙ם֙ אַתֶּ֣ם אֹמְרִ֔ים לִכְבֹּ֛שׁ לַעֲבָדִ֥ים וְלִשְׁפָח֖וֹת לָכֶ֑ם הֲלֹ֤א רַק־ אַתֶּם֙ עִמָּכֶ֣ם אֲשָׁמ֔וֹת לַיהוָ֖ה אֱלֹהֵיכֶֽם׃ (מב) וְעַתָּ֣ה שְׁמָע֔וּנִי וְהָשִׁ֙יבוּ֙ הַשִּׁבְיָ֔ה אֲשֶׁ֥ר שְׁבִיתֶ֖ם מֵאֲחֵיכֶ֑ם כִּ֛י חֲר֥וֹן אַף־ יְהוָ֖ה עֲלֵיכֶֽם׃ (מג) וַיָּקֻ֨מוּ אֲנָשִׁ֜ים מֵרָאשֵׁ֣י בְנֵֽי־ אֶפְרַ֗יִם עֲזַרְיָ֤הוּ בֶן־ יְהֽוֹחָנָן֙ בֶּרֶכְיָ֣הוּ בֶן־ מְשִׁלֵּמ֔וֹת וִֽיחִזְקִיָּ֙הוּ֙ בֶּן־ שַׁלֻּ֔ם וַעֲמָשָׂ֖א בֶּן־ חַדְלָ֑י עַל־ הַבָּאִ֖ים מִן־ הַצָּבָֽא׃ (מד) וַיֹּאמְר֣וּ לָהֶ֗ם לֹא־ תָבִ֤יאוּ אֶת־ הַשִּׁבְיָה֙ הֵ֔נָּה כִּי֩ לְאַשְׁמַ֨ת יְהוָ֤ה עָלֵ֙ינוּ֙ אַתֶּ֣ם אֹמְרִ֔ים לְהֹסִ֥יף עַל־ חַטֹּאתֵ֖ינוּ וְעַל־ אַשְׁמָתֵ֑ינוּ כִּֽי־ רַבָּ֤ה אַשְׁמָה֙ לָ֔נוּ וַחֲר֥וֹן אָ֖ף עַל־ יִשְׂרָאֵֽל׃ (מה) וַיַּעֲזֹ֣ב הֶֽחָל֗וּץ אֶת־ הַשִּׁבְיָה֙ וְאֶת־ הַבִּזָּ֔ה לִפְנֵ֥י הַשָּׂרִ֖ים וְכָל־ הַקָּהָֽל׃ (מו) וַיָּקֻ֣מוּ הָאֲנָשִׁים֩ אֲשֶׁר־ נִקְּב֨וּ בְשֵׁמ֜וֹת וַיַּחֲזִ֣יקוּ בַשִּׁבְיָ֗ה וְכָֽל־ מַעֲרֻמֵּיהֶם֮ הִלְבִּ֣ישׁוּ מִן־ הַשָּׁלָל֒ וַיַּלְבִּשׁ֣וּם וַ֠יַּנְעִלוּם וַיַּאֲכִל֨וּם וַיַּשְׁק֜וּם וַיְסֻכ֗וּם וַיְנַהֲל֤וּם בַּחֲמֹרִים֙ לְכָל־ כּוֹשֵׁ֔ל וַיְבִיא֛וּם יְרֵח֥וֹ עִיר־ הַתְּמָרִ֖ים אֵ֣צֶל אֲחֵיהֶ֑ם וַיָּשׁ֖וּבוּ שֹׁמְרֽוֹן׃ (מז) בָּעֵ֣ת הַהִ֗יא שָׁלַ֞ח הַמֶּ֧לֶךְ אָחָ֛ז עַל־ מַלְכֵ֥י אַשּׁ֖וּר לַעְזֹ֥ר לֽוֹ׃ (מח) וְע֥וֹד אֲדוֹמִ֖ים בָּ֑אוּ וַיַּכּ֥וּ בִיהוּדָ֖ה וַיִּשְׁבּוּ־ שֶֽׁבִי׃ (מט) וּפְלִשְׁתִּ֣ים פָּשְׁט֗וּ בְּעָרֵ֨י הַשְּׁפֵלָ֣ה וְהַנֶּגֶב֮ לִֽיהוּדָה֒ וַֽ֠יִּלְכְּדוּ אֶת־ בֵּֽית־ שֶׁ֨מֶשׁ וְאֶת־ אַיָּל֜וֹן וְאֶת־ הַגְּדֵר֗וֹת וְאֶת־ שׂוֹכ֤וֹ וּבְנוֹתֶ֙יהָ֙ וְאֶת־ תִּמְנָ֣ה וּבְנוֹתֶ֔יהָ וְאֶת־ גִּמְז֖וֹ וְאֶת־ בְּנֹתֶ֑יהָ וַיֵּשְׁב֖וּ שָֽׁם׃ (נ) כִּֽי־ הִכְנִ֤יעַ יְהוָה֙ אֶת־ יְהוּדָ֔ה בַּעֲב֖וּר אָחָ֣ז מֶֽלֶךְ־ יִשְׂרָאֵ֑ל כִּ֤י הִפְרִ֙יעַ֙ בִּֽיהוּדָ֔ה וּמָע֥וֹל מַ֖עַל בַּיהוָֽה׃ (נא) וַיָּבֹ֣א עָלָ֔יו תִּלְּגַ֥ת פִּלְנְאֶ֖סֶר מֶ֣לֶךְ אַשּׁ֑וּר וַיָּ֥צַר ל֖וֹ וְלֹ֥א חֲזָקֽוֹ׃ (נב) כִּֽי־ חָלַ֤ק אָחָז֙ אֶת־ בֵּ֣ית יְהוָ֔ה וְאֶת־ בֵּ֥ית הַמֶּ֖לֶךְ וְהַשָּׂרִ֑ים וַיִּתֵּן֙ לְמֶ֣לֶךְ אַשּׁ֔וּר וְלֹ֥א לְעֶזְרָ֖ה לֽוֹ׃ (נג) וּבְעֵת֙ הָצֵ֣ר ל֔וֹ וַיּ֖וֹסֶף לִמְע֣וֹל בַּיהוָ֑ה ה֖וּא הַמֶּ֥לֶךְ אָחָֽז׃ (נד) וַיִּזְבַּ֗ח לֵֽאלֹהֵ֣י דַרְמֶשֶׂק֮ הַמַּכִּ֣ים בּוֹ֒ וַיֹּ֗אמֶר כִּ֠י אֱלֹהֵ֤י מַלְכֵֽי־ אֲרָם֙ הֵ֚ם מַעְזְרִ֣ים אוֹתָ֔ם לָהֶ֥ם אֲזַבֵּ֖חַ וְיַעְזְר֑וּנִי וְהֵ֛ם הָֽיוּ־ ל֥וֹ לְהַכְשִׁיל֖וֹ וּלְכָל־ יִשְׂרָאֵֽל׃ (נה) וַיֶּאֱסֹ֨ף אָחָ֜ז אֶת־ כְּלֵ֣י בֵית־ הָֽאֱלֹהִ֗ים וַיְקַצֵּץ֙ אֶת־ כְּלֵ֣י בֵית־ הָֽאֱלֹהִ֔ים וַיִּסְגֹּ֖ר אֶת־ דַּלְת֣וֹת בֵּית־ יְהוָ֑ה וַיַּ֨עַשׂ ל֧וֹ מִזְבְּח֛וֹת בְּכָל־ פִּנָּ֖ה בִּירוּשָׁלִָֽם׃ (נו) וּבְכָל־ עִ֨יר וָעִ֤יר לִֽיהוּדָה֙ עָשָׂ֣ה בָמ֔וֹת לְקַטֵּ֖ר לֵֽאלֹהִ֣ים אֲחֵרִ֑ים וַיַּכְעֵ֕ס אֶת־ יְהוָ֖ה אֱלֹהֵ֥י אֲבֹתָֽיו׃ (נז) וְיֶ֤תֶר דְּבָרָיו֙ וְכָל־ דְּרָכָ֔יו הָרִאשֹׁנִ֖ים וְהָאַחֲרוֹנִ֑ים הִנָּ֣ם כְּתוּבִ֔ים עַל־ סֵ֥פֶר מַלְכֵֽי־ יְהוּדָ֖ה וְיִשְׂרָאֵֽל׃ (נח) וַיִּשְׁכַּ֨ב אָחָ֜ז עִם־ אֲבֹתָ֗יו וַֽיִּקְבְּרֻ֤הוּ בָעִיר֙ בִּיר֣וּשָׁלִַ֔ם כִּ֚י לֹ֣א הֱבִיאֻ֔הוּ לְקִבְרֵ֖י מַלְכֵ֣י יִשְׂרָאֵ֑ל וַיִּמְלֹ֛ךְ יְחִזְקִיָּ֥הֽוּ בְנ֖וֹ תַּחְתָּֽיו׃ תחילת פרק כט (נט) יְחִזְקִיָּ֣הוּ מָלַ֗ךְ בֶּן־ עֶשְׂרִ֤ים וְחָמֵשׁ֙ שָׁנָ֔ה וְעֶשְׂרִ֤ים וָתֵ֙שַׁע֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ אֲבִיָּ֖ה בַּת־ זְכַרְיָֽהוּ׃ (ס) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה דָּוִ֥יד אָבִֽיו׃ (סא) ה֣וּא בַשָּׁנָה֩ הָרִאשׁוֹנָ֨ה לְמָלְכ֜וֹ בַּחֹ֣דֶשׁ הָרִאשׁ֗וֹן פָּתַ֛ח אֶת־ דַּלְת֥וֹת בֵּית־ יְהוָ֖ה וַֽיְחַזְּקֵֽם׃ (סב) וַיָּבֵ֥א אֶת־ הַכֹּהֲנִ֖ים וְאֶת־ הַלְוִיִּ֑ם וַיַּֽאַסְפֵ֖ם לִרְח֥וֹב הַמִּזְרָֽח׃ (סג) וַיֹּ֥אמֶר לָהֶ֖ם שְׁמָע֣וּנִי הַלְוִיִּ֑ם עַתָּ֣ה הִֽתְקַדְּשׁ֗וּ וְקַדְּשׁוּ֙ אֶת־ בֵּ֤ית יְהוָה֙ אֱלֹהֵ֣י אֲבֹתֵיכֶ֔ם וְהוֹצִ֥יאוּ אֶת־ הַנִּדָּ֖ה מִן־ הַקֹּֽדֶשׁ׃ (סד) כִּֽי־ מָעֲל֣וּ אֲבֹתֵ֗ינוּ וְעָשׂ֥וּ הָרַ֛ע בְּעֵינֵ֥י יְהוָֽה־ אֱלֹהֵ֖ינוּ וַיַּֽעַזְבֻ֑הוּ וַיַּסֵּ֧בּוּ פְנֵיהֶ֛ם מִמִּשְׁכַּ֥ן יְהוָ֖ה וַיִּתְּנוּ־ עֹֽרֶף׃ (סה) גַּ֣ם סָֽגְר֞וּ דַּלְת֣וֹת הָאוּלָ֗ם וַיְכַבּוּ֙ אֶת־ הַנֵּר֔וֹת וּקְטֹ֖רֶת לֹ֣א הִקְטִ֑ירוּ וְעֹלָה֙ לֹא־ הֶעֱל֣וּ בַקֹּ֔דֶשׁ לֵאלֹהֵ֖י יִשְׂרָאֵֽל׃ (סו) וַיְהִי֙ קֶ֣צֶף יְהוָ֔ה עַל־ יְהוּדָ֖ה וִירוּשָׁלִָ֑ם וַיִּתְּנֵ֤ם (לזועה) לְזַֽעֲוָה֙ לְשַׁמָּ֣ה וְלִשְׁרֵקָ֔ה כַּאֲשֶׁ֛ר אַתֶּ֥ם רֹאִ֖ים בְּעֵינֵיכֶֽם׃ (סז) וְהִנֵּ֛ה נָפְל֥וּ אֲבוֹתֵ֖ינוּ בֶּחָ֑רֶב וּבָנֵ֨ינוּ וּבְנוֹתֵ֧ינוּ וְנָשֵׁ֛ינוּ בַּשְּׁבִ֖י עַל־ זֹֽאת׃ (סח) עַתָּה֙ עִם־ לְבָבִ֔י לִכְר֣וֹת בְּרִ֔ית לַיהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וְיָשֹׁ֥ב מִמֶּ֖נּוּ חֲר֥וֹן אַפּֽוֹ׃

וְיוֹאֵ֣ל בֶּן־ עֲזַרְיָהוּ֮ מִן־ בְּנֵ֣י הַקְּהָתִי֒ וּמִן־ בְּנֵ֣י מְרָרִ֔י קִ֚ישׁ בֶּן־ עַבְדִּ֔י וַעֲזַרְיָ֖הוּ בֶּן־ יְהַלֶּלְאֵ֑ל וּמִן־ הַגֵּ֣רְשֻׁנִּ֔י יוֹאָח֙ בֶּן־ זִמָּ֔ה וְעֵ֖דֶן בֶּן־ יוֹאָֽח׃ (ג) וּמִן־ בְּנֵי֙ אֱלִ֣יצָפָ֔ן שִׁמְרִ֖י (ויעואל) וִיעִיאֵ֑ל וּמִן־ בְּנֵ֣י אָסָ֔ף זְכַרְיָ֖הוּ וּמַתַּנְיָֽהוּ׃ (ד) וּמִן־ בְּנֵ֥י הֵימָ֖ן (יחואל) יְחִיאֵ֣ל וְשִׁמְעִ֑י וּמִן־ בְּנֵ֣י יְדוּת֔וּן שְׁמַֽעְיָ֖ה וְעֻזִּיאֵֽל׃ (ה) וַיַּֽאַסְפ֤וּ אֶת־ אֲחֵיהֶם֙ וַיִּֽתְקַדְּשׁ֔וּ וַיָּבֹ֥אוּ כְמִצְוַת־ הַמֶּ֖לֶךְ בְּדִבְרֵ֣י יְהוָ֑ה לְטַהֵ֖ר בֵּ֥ית יְהוָֽה׃ (ו) וַיָּבֹ֣אוּ הַ֠כֹּהֲנִים לִפְנִ֣ימָה בֵית־ יְהוָה֮ לְטַהֵר֒ וַיּוֹצִ֗יאוּ אֵ֤ת כָּל־ הַטֻּמְאָה֙ אֲשֶׁ֤ר מָֽצְאוּ֙ בְּהֵיכַ֣ל יְהוָ֔ה לַחֲצַ֖ר בֵּ֣ית יְהוָ֑ה וַֽיְקַבְּלוּ֙ הַלְוִיִּ֔ם לְהוֹצִ֥יא לְנַֽחַל־ קִדְר֖וֹן חֽוּצָה׃ (ז) וַ֠יָּחֵלּוּ בְּאֶחָ֞ד לַחֹ֣דֶשׁ הָרִאשׁוֹן֮ לְקַדֵּשׁ֒ וּבְי֧וֹם שְׁמוֹנָ֣ה לַחֹ֗דֶשׁ בָּ֚אוּ לְאוּלָ֣ם יְהוָ֔ה וַיְקַדְּשׁ֥וּ אֶת־ בֵּית־ יְהוָ֖ה לְיָמִ֣ים שְׁמוֹנָ֑ה וּבְי֨וֹם שִׁשָּׁ֥ה עָשָׂ֛ר לַחֹ֥דֶשׁ הָרִאשׁ֖וֹן כִּלּֽוּ׃ (ח) וַיָּב֤וֹאוּ פְנִ֙ימָה֙ אֶל־ חִזְקִיָּ֣הוּ הַמֶּ֔לֶךְ וַיֹּ֣אמְר֔וּ טִהַ֖רְנוּ אֶת־ כָּל־ בֵּ֣ית יְהוָ֑ה אֶת־ מִזְבַּ֤ח הָעוֹלָה֙ וְאֶת־ כָּל־ כֵּלָ֔יו וְאֶת־ שֻׁלְחַ֥ן הַֽמַּעֲרֶ֖כֶת וְאֶת־ כָּל־ כֵּלָֽיו׃ (ט) וְאֵ֣ת כָּל־ הַכֵּלִ֗ים אֲשֶׁ֣ר הִזְנִיחַ֩ הַמֶּ֨לֶךְ אָחָ֧ז בְּמַלְכוּת֛וֹ בְּמַעֲל֖וֹ הֵכַ֣נּוּ וְהִקְדָּ֑שְׁנוּ וְהִנָּ֕ם לִפְנֵ֖י מִזְבַּ֥ח יְהוָֽה׃ (י) וַיַּשְׁכֵּם֙ יְחִזְקִיָּ֣הוּ הַמֶּ֔לֶךְ וַיֶּאֱסֹ֕ף אֵ֖ת שָׂרֵ֣י הָעִ֑יר וַיַּ֖עַל בֵּ֥ית יְהוָֽה׃ (יא) וַיָּבִ֣יאוּ פָרִים־ שִׁבְעָה֩ וְאֵילִ֨ים שִׁבְעָ֜ה וּכְבָשִׂ֣ים שִׁבְעָ֗ה וּצְפִירֵ֨י עִזִּ֤ים שִׁבְעָה֙ לְחַטָּ֔את עַל־ הַמַּמְלָכָ֥ה וְעַל־ הַמִּקְדָּ֖שׁ וְעַל־ יְהוּדָ֑ה וַיֹּ֗אמֶר לִבְנֵ֤י אַהֲרֹן֙ הַכֹּ֣הֲנִ֔ים לְהַעֲל֖וֹת עַל־ מִזְבַּ֥ח יְהוָֽה׃ (יב) וַֽיִּשְׁחֲטוּ֙ הַבָּקָ֔ר וַיְקַבְּל֤וּ הַכֹּֽהֲנִים֙ אֶת־ הַדָּ֔ם וַֽיִּזְרְק֖וּ הַמִּזְבֵּ֑חָה וַיִּשְׁחֲט֣וּ הָאֵלִ֗ים וַיִּזְרְק֤וּ הַדָּם֙ הַמִּזְבֵּ֔חָה וַֽיִּשְׁחֲטוּ֙ הַכְּבָשִׂ֔ים וַיִּזְרְק֥וּ הַדָּ֖ם הַמִּזְבֵּֽחָה׃ (יג) וַיַּגִּ֙ישׁוּ֙ אֶת־ שְׂעִירֵ֣י הַֽחַטָּ֔את לִפְנֵ֥י הַמֶּ֖לֶךְ וְהַקָּהָ֑ל וַיִּסְמְכ֥וּ יְדֵיהֶ֖ם עֲלֵיהֶֽם׃ (יד) וַיִּשְׁחָטוּם֙ הַכֹּ֣הֲנִ֔ים וַֽיְחַטְּא֤וּ אֶת־ דָּמָם֙ הַמִּזְבֵּ֔חָה לְכַפֵּ֖ר עַל־ כָּל־ יִשְׂרָאֵ֑ל כִּ֤י לְכָל־ יִשְׂרָאֵל֙ אָמַ֣ר הַמֶּ֔לֶךְ הָעוֹלָ֖ה וְהַחַטָּֽאת׃ (טו) וַיַּֽעֲמֵ֨ד אֶת־ הַלְוִיִּ֜ם בֵּ֣ית יְהוָ֗ה בִּמְצִלְתַּ֙יִם֙ בִּנְבָלִ֣ים וּבְכִנֹּר֔וֹת בְּמִצְוַ֥ת דָּוִ֛יד וְגָ֥ד חֹזֵֽה־ הַמֶּ֖לֶךְ וְנָתָ֣ן הַנָּבִ֑יא כִּ֧י בְיַד־ יְהוָ֛ה הַמִּצְוָ֖ה בְּיַד־ נְבִיאָֽיו׃ (טז) וַיַּֽעַמְד֤וּ הַלְוִיִּם֙ בִּכְלֵ֣י דָוִ֔יד וְהַכֹּהֲנִ֖ים בַּחֲצֹצְרֽוֹת׃ (יז) וַיֹּ֙אמֶר֙ חִזְקִיָּ֔הוּ לְהַעֲל֥וֹת הָעֹלָ֖ה לְהַמִּזְבֵּ֑חַ וּבְעֵ֞ת הֵחֵ֣ל הָֽעוֹלָ֗ה הֵחֵ֤ל שִׁיר־ יְהוָה֙ וְהַחֲצֹ֣צְר֔וֹת וְעַ֨ל־ יְדֵ֔י כְּלֵ֖י דָּוִ֥יד מֶֽלֶךְ־ יִשְׂרָאֵֽל׃ (יח) וְכָל־ הַקָּהָל֙ מִֽשְׁתַּחֲוִ֔ים וְהַשִּׁ֣יר מְשׁוֹרֵ֔ר וְהַחֲצֹצְר֖וֹת (מחצצרים) מַחְצְרִ֑ים הַכֹּ֕ל עַ֖ד לִכְל֥וֹת הָעֹלָֽה׃ (יט) וּכְכַלּ֖וֹת לְהַעֲל֑וֹת כָּרְע֗וּ הַמֶּ֛לֶךְ וְכָֽל־ הַנִּמְצְאִ֥ים אִתּ֖וֹ וַיִּֽשְׁתַּחֲוֽוּ׃ (כ) וַ֠יֹּאמֶר יְחִזְקִיָּ֨הוּ הַמֶּ֤לֶךְ וְהַשָּׂרִים֙ לַלְוִיִּ֔ם לְהַלֵּל֙ לַֽיהוָ֔ה בְּדִבְרֵ֥י דָוִ֖יד וְאָסָ֣ף הַחֹזֶ֑ה וַֽיְהַלְלוּ֙ עַד־ לְשִׂמְחָ֔ה וַֽיִּקְּד֖וּ וַיִּֽשְׁתַּחֲוֽוּ׃ (כא) וַיַּ֨עַן יְחִזְקִיָּ֜הוּ וַיֹּ֗אמֶר עַתָּ֨ה מִלֵּאתֶ֤ם יֶדְכֶם֙ לַיהוָ֔ה גֹּ֧שׁוּ וְהָבִ֛יאוּ זְבָחִ֥ים וְתוֹד֖וֹת לְבֵ֣ית יְהוָ֑ה וַיָּבִ֤יאוּ הַקָּהָל֙ זְבָחִ֣ים וְתוֹד֔וֹת וְכָל־ נְדִ֥יב לֵ֖ב עֹלֽוֹת׃ (כב) וַיְהִ֞י מִסְפַּ֣ר הָעֹלָה֮ אֲשֶׁ֣ר הֵבִ֣יאוּ הַקָּהָל֒ בָּקָ֣ר שִׁבְעִ֔ים אֵילִ֥ים מֵאָ֖ה כְּבָשִׂ֣ים מָאתָ֑יִם לְעֹלָ֥ה לַיהוָ֖ה כָּל־ אֵֽלֶּה׃ (כג) וְֽהַקֳּדָשִׁ֑ים בָּקָר֙ שֵׁ֣שׁ מֵא֔וֹת וְצֹ֖אן שְׁלֹ֥שֶׁת אֲלָפִֽים׃ (כד) רַ֤ק הַכֹּֽהֲנִים֙ הָי֣וּ לִמְעָ֔ט וְלֹ֣א יָֽכְל֔וּ לְהַפְשִׁ֖יט אֶת־ כָּל־ הָעֹל֑וֹת וַֽיְּחַזְּק֞וּם אֲחֵיהֶ֣ם הַלְוִיִּ֗ם עַד־ כְּל֤וֹת הַמְּלָאכָה֙ וְעַ֣ד יִתְקַדְּשׁ֣וּ הַכֹּֽהֲנִ֔ים כִּ֤י הַלְוִיִּם֙ יִשְׁרֵ֣י לֵבָ֔ב לְהִתְקַדֵּ֖שׁ מֵֽהַכֹּהֲנִֽים׃ (כה) וְגַם־ עֹלָ֨ה לָרֹ֜ב בְּחֶלְבֵ֧י הַשְּׁלָמִ֛ים וּבַנְּסָכִ֖ים לָעֹלָ֑ה וַתִּכּ֖וֹן עֲבוֹדַ֥ת בֵּית־ יְהוָֽה׃ (כו) וַיִּשְׂמַ֤ח יְחִזְקִיָּ֙הוּ֙ וְכָל־ הָעָ֔ם עַ֛ל הַהֵכִ֥ין הָאֱלֹהִ֖ים לָעָ֑ם כִּ֥י בְּפִתְאֹ֖ם הָיָ֥ה הַדָּבָֽר׃ סוף פרק כט (כז) וַיִּשְׁלַ֨ח יְחִזְקִיָּ֜הוּ עַל־ כָּל־ יִשְׂרָאֵ֣ל וִֽיהוּדָ֗ה וְגַֽם־ אִגְּרוֹת֙ כָּתַב֙ עַל־ אֶפְרַ֣יִם וּמְנַשֶּׁ֔ה לָב֥וֹא לְבֵית־ יְהוָ֖ה בִּֽירוּשָׁלִָ֑ם לַעֲשׂ֣וֹת פֶּ֔סַח לַיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (כח) וַיִּוָּעַ֨ץ הַמֶּ֧לֶךְ וְשָׂרָ֛יו וְכָל־ הַקָּהָ֖ל בִּירוּשָׁלִָ֑ם לַעֲשׂ֥וֹת הַפֶּ֖סַח בַּחֹ֥דֶשׁ הַשֵּׁנִֽי׃ (כט) כִּ֣י לֹ֧א יָכְל֛וּ לַעֲשֹׂת֖וֹ בָּעֵ֣ת הַהִ֑יא כִּ֤י הַכֹּהֲנִים֙ לֹֽא־ הִתְקַדְּשׁ֣וּ לְמַדַּ֔י וְהָעָ֖ם לֹא־ נֶאֶסְפ֥וּ לִֽירוּשָׁלִָֽם׃ (ל) וַיִּישַׁ֥ר הַדָּבָ֖ר בְּעֵינֵ֣י הַמֶּ֑לֶךְ וּבְעֵינֵ֖י כָּל־ הַקָּהָֽל׃ (לא) וַיַּֽעֲמִ֣ידוּ דָבָ֗ר לְהַעֲבִ֨יר ק֤וֹל בְּכָל־ יִשְׂרָאֵל֙ מִבְּאֵֽר־ שֶׁ֣בַע וְעַד־ דָּ֔ן לָב֞וֹא לַעֲשׂ֥וֹת פֶּ֛סַח לַיהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בִּירוּשָׁלִָ֑ם כִּ֣י לֹ֥א לָרֹ֛ב עָשׂ֖וּ כַּכָּתֽוּב׃ (לב) וַיֵּלְכוּ֩ הָרָצִ֨ים בָּֽאִגְּר֜וֹת מִיַּ֧ד הַמֶּ֣לֶךְ וְשָׂרָ֗יו בְּכָל־ יִשְׂרָאֵל֙ וִֽיהוּדָ֔ה וּכְמִצְוַ֥ת הַמֶּ֖לֶךְ לֵאמֹ֑ר בְּנֵ֣י יִשְׂרָאֵ֗ל שׁ֚וּבוּ אֶל־ יְהוָ֗ה אֱלֹהֵי֙ אַבְרָהָם֙ יִצְחָ֣ק וְיִשְׂרָאֵ֔ל וְיָשֹׁב֙ אֶל־ הַפְּלֵיטָ֔ה הַנִּשְׁאֶ֣רֶת לָכֶ֔ם מִכַּ֖ף מַלְכֵ֥י אַשּֽׁוּר׃ (לג) וְאַל־ תִּֽהְי֗וּ כַּאֲבֽוֹתֵיכֶם֙ וְכַ֣אֲחֵיכֶ֔ם אֲשֶׁ֣ר מָעֲל֔וּ בַּיהוָ֖ה אֱלֹהֵ֣י אֲבוֹתֵיהֶ֑ם וַיִּתְּנֵ֣ם לְשַׁמָּ֔ה כַּאֲשֶׁ֖ר אַתֶּ֥ם רֹאִֽים׃ (לד) עַתָּ֕ה אַל־ תַּקְשׁ֥וּ עָרְפְּכֶ֖ם כַּאֲבוֹתֵיכֶ֑ם תְּנוּ־ יָ֣ד לַיהוָ֗ה וּבֹ֤אוּ לְמִקְדָּשׁוֹ֙ אֲשֶׁ֣ר הִקְדִּ֣ישׁ לְעוֹלָ֔ם וְעִבְדוּ֙ אֶת־ יְהוָ֣ה אֱלֹהֵיכֶ֔ם וְיָשֹׁ֥ב מִכֶּ֖ם חֲר֥וֹן אַפּֽוֹ׃ (לה) כִּ֣י בְשׁוּבְכֶ֞ם עַל־ יְהוָ֗ה אֲחֵיכֶ֨ם וּבְנֵיכֶ֤ם לְרַחֲמִים֙ לִפְנֵ֣י שֽׁוֹבֵיהֶ֔ם וְלָשׁ֖וּב לָאָ֣רֶץ הַזֹּ֑את כִּֽי־ חַנּ֤וּן וְרַחוּם֙ יְהוָ֣ה אֱלֹהֵיכֶ֔ם וְלֹא־ יָסִ֤יר פָּנִים֙ מִכֶּ֔ם אִם־ תָּשׁ֖וּבוּ אֵלָֽיו׃ (לו) וַיִּֽהְי֨וּ הָרָצִ֜ים עֹבְרִ֨ים מֵעִ֧יר ׀ לָעִ֛יר בְּאֶֽרֶץ־ אֶפְרַ֥יִם וּמְנַשֶּׁ֖ה וְעַד־ זְבֻל֑וּן וַיִּֽהְיוּ֙ מַשְׂחִיקִ֣ים עֲלֵיהֶ֔ם וּמַלְעִגִ֖ים בָּֽם׃ (לז) אַךְ־ אֲנָשִׁ֛ים מֵאָשֵׁ֥ר וּמְנַשֶּׁ֖ה וּמִזְּבֻל֑וּן נִֽכְנְע֔וּ וַיָּבֹ֖אוּ לִירוּשָׁלִָֽם׃ (לח) גַּ֣ם בִּיהוּדָ֗ה הָֽיְתָה֙ יַ֣ד הָאֱלֹהִ֔ים לָתֵ֥ת לָהֶ֖ם לֵ֣ב אֶחָ֑ד לַעֲשׂ֞וֹת מִצְוַ֥ת הַמֶּ֛לֶךְ וְהַשָּׂרִ֖ים בִּדְבַ֥ר יְהוָֽה׃ (לט) וַיֵּֽאָסְפ֤וּ יְרוּשָׁלִַ֙ם֙ עַם־ רָ֔ב לַעֲשׂ֛וֹת אֶת־ חַ֥ג הַמַּצּ֖וֹת בַּחֹ֣דֶשׁ הַשֵּׁנִ֑י קָהָ֖ל לָרֹ֥ב מְאֹֽד׃ (מ) וַיָּקֻ֕מוּ וַיָּסִ֙ירוּ֙ אֶת־ הַֽמִּזְבְּח֔וֹת אֲשֶׁ֖ר בִּירוּשָׁלִָ֑ם וְאֵ֤ת כָּל־ הַֽמְקַטְּרוֹת֙ הֵסִ֔ירוּ וַיַּשְׁלִ֖יכוּ לְנַ֥חַל קִדְרֽוֹן׃ (מא) וַיִּשְׁחֲט֣וּ הַפֶּ֔סַח בְּאַרְבָּעָ֥ה עָשָׂ֖ר לַחֹ֣דֶשׁ הַשֵּׁנִ֑י וְהַכֹּהֲנִ֨ים וְהַלְוִיִּ֤ם נִכְלְמוּ֙ וַיִּֽתְקַדְּשׁ֔וּ וַיָּבִ֥יאוּ עֹל֖וֹת בֵּ֥ית יְהוָֽה׃ (מב) וַיַּֽעַמְד֤וּ עַל־ עָמְדָם֙ כְּמִשְׁפָּטָ֔ם כְּתוֹרַ֖ת מֹשֶׁ֣ה אִישׁ־ הָאֱלֹהִ֑ים הַכֹּֽהֲנִים֙ זֹרְקִ֣ים אֶת־ הַדָּ֔ם מִיַּ֖ד הַלְוִיִּֽם׃ (מג) כִּי־ רַבַּ֥ת בַּקָּהָ֖ל אֲשֶׁ֣ר לֹא־ הִתְקַדָּ֑שׁוּ וְהַלְוִיִּ֞ם עַל־ שְׁחִיטַ֣ת הַפְּסָחִ֗ים לְכֹל֙ לֹ֣א טָה֔וֹר לְהַקְדִּ֖ישׁ לַיהוָֽה׃ (מד) כִּ֣י מַרְבִּ֣ית הָעָ֡ם רַ֠בַּת מֵֽאֶפְרַ֨יִם וּמְנַשֶּׁ֜ה יִשָּׂשכָ֤ר וּזְבֻלוּן֙ לֹ֣א הִטֶּהָ֔רוּ כִּֽי־ אָכְל֥וּ אֶת־ הַפֶּ֖סַח בְּלֹ֣א כַכָּת֑וּב כִּי֩ הִתְפַּלֵּ֨ל יְחִזְקִיָּ֤הוּ עֲלֵיהֶם֙ לֵאמֹ֔ר יְהוָ֥ה הַטּ֖וֹב יְכַפֵּ֥ר בְּעַֽד׃ (מה) כָּל־ לְבָב֣וֹ הֵכִ֔ין לִדְר֛וֹשׁ הָאֱלֹהִ֥ים ׀ יְהוָ֖ה אֱלֹהֵ֣י אֲבוֹתָ֑יו וְלֹ֖א כְּטָהֳרַ֥ת הַקֹּֽדֶשׁ׃ (מו) וַיִּשְׁמַ֤ע יְהוָה֙ אֶל־ יְחִזְקִיָּ֔הוּ וַיִּרְפָּ֖א אֶת־ הָעָֽם׃ (מז) וַיַּעֲשׂ֣וּ בְנֵֽי־ יִ֠שְׂרָאֵל הַנִּמְצְאִ֨ים בִּירוּשָׁלִַ֜ם אֶת־ חַ֧ג הַמַּצּ֛וֹת שִׁבְעַ֥ת יָמִ֖ים בְּשִׂמְחָ֣ה גְדוֹלָ֑ה וּֽמְהַלְלִ֣ים לַ֠יהוָה י֣וֹם ׀ בְּי֞וֹם הַלְוִיִּ֧ם וְהַכֹּהֲנִ֛ים בִּכְלֵי־ עֹ֖ז לַיהוָֽה׃ (מח) וַיְדַבֵּ֣ר יְחִזְקִיָּ֗הוּ עַל־ לֵב֙ כָּל־ הַלְוִיִּ֔ם הַמַּשְׂכִּילִ֥ים שֵֽׂכֶל־ ט֖וֹב לַיהוָ֑ה וַיֹּאכְל֤וּ אֶת־ הַמּוֹעֵד֙ שִׁבְעַ֣ת הַיָּמִ֔ים מְזַבְּחִים֙ זִבְחֵ֣י שְׁלָמִ֔ים וּמִ֨תְוַדִּ֔ים לַיהוָ֖ה אֱלֹהֵ֥י אֲבוֹתֵיהֶֽם׃ (מט) וַיִּוָּֽעֲצוּ֙ כָּל־ הַקָּהָ֔ל לַעֲשׂ֕וֹת שִׁבְעַ֥ת יָמִ֖ים אֲחֵרִ֑ים וַיַּֽעֲשׂ֥וּ שִׁבְעַת־ יָמִ֖ים שִׂמְחָֽה׃ (נ) כִּ֣י חִזְקִיָּ֣הוּ מֶֽלֶךְ־ יְ֠הוּדָה הֵרִ֨ים לַקָּהָ֜ל אֶ֣לֶף פָּרִים֮ וְשִׁבְעַ֣ת אֲלָפִ֣ים צֹאן֒ וְהַשָּׂרִ֞ים הֵרִ֤ימוּ לַקָּהָל֙ פָּרִ֣ים אֶ֔לֶף וְצֹ֖אן עֲשֶׂ֣רֶת אֲלָפִ֑ים וַיִּֽתְקַדְּשׁ֥וּ כֹהֲנִ֖ים לָרֹֽב׃ (נא) וַֽיִּשְׂמְח֣וּ ׀ כָּל־ קְהַ֣ל יְהוּדָ֗ה וְהַכֹּהֲנִים֙ וְהַלְוִיִּ֔ם וְכָל־ הַקָּהָ֖ל הַבָּאִ֣ים מִיִּשְׂרָאֵ֑ל וְהַגֵּרִ֗ים הַבָּאִים֙ מֵאֶ֣רֶץ יִשְׂרָאֵ֔ל וְהַיּוֹשְׁבִ֖ים בִּיהוּדָֽה׃ (נב) וַתְּהִ֥י שִׂמְחָֽה־ גְדוֹלָ֖ה בִּֽירוּשָׁלִָ֑ם כִּ֠י מִימֵ֞י שְׁלֹמֹ֤ה בֶן־ דָּוִיד֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל לֹ֥א כָזֹ֖את בִּירוּשָׁלִָֽם׃ (נג) וַיָּקֻ֜מוּ הַכֹּהֲנִ֤ים הַלְוִיִּם֙ וַיְבָרֲכ֣וּ אֶת־ הָעָ֔ם וַיִּשָּׁמַ֖ע בְּקוֹלָ֑ם וַתָּב֧וֹא תְפִלָּתָ֛ם לִמְע֥וֹן קָדְשׁ֖וֹ לַשָּׁמָֽיִם׃ (נד) וּכְכַלּ֣וֹת כָּל־ זֹ֗את יָצְא֨וּ כָּל־ יִשְׂרָאֵ֥ל הַֽנִּמְצְאִים֮ לְעָרֵ֣י יְהוּדָה֒ וַיְשַׁבְּר֣וּ הַמַּצֵּב֣וֹת וַיְגַדְּע֣וּ הָאֲשֵׁרִ֡ים וַיְנַתְּצ֣וּ אֶת־ הַ֠בָּמוֹת וְאֶת־ הַֽמִּזְבְּחֹ֞ת מִכָּל־ יְהוּדָ֧ה וּבִנְיָמִ֛ן וּבְאֶפְרַ֥יִם וּמְנַשֶּׁ֖ה עַד־ לְכַלֵּ֑ה וַיָּשׁ֜וּבוּ כָּל־ בְּנֵ֧י יִשְׂרָאֵ֛ל אִ֥ישׁ לַאֲחֻזָּת֖וֹ לְעָרֵיהֶֽם׃ (נה) וַיַּעֲמֵ֣ד יְחִזְקִיָּ֡הוּ אֶת־ מַחְלְק֣וֹת הַכֹּהֲנִ֣ים וְ֠הַלְוִיִּם עַֽל־ מַחְלְקוֹתָ֞ם אִ֣ישׁ ׀ כְּפִ֣י עֲבֹדָת֗וֹ לַכֹּהֲנִים֙ וְלַלְוִיִּ֔ם לְעֹלָ֖ה וְלִשְׁלָמִ֑ים לְשָׁרֵת֙ וּלְהֹד֣וֹת וּלְהַלֵּ֔ל בְּשַׁעֲרֵ֖י מַחֲנ֥וֹת יְהוָֽה׃ (נו) וּמְנָת֩ הַמֶּ֨לֶךְ מִן־ רְכוּשׁ֜וֹ לָעֹל֗וֹת לְעֹלוֹת֙ הַבֹּ֣קֶר וְהָעֶ֔רֶב וְהָ֣עֹל֔וֹת לַשַּׁבָּת֖וֹת וְלֶחֳדָשִׁ֣ים וְלַמֹּעֲדִ֑ים כַּכָּת֖וּב בְּתוֹרַ֥ת יְהוָֽה׃ (נז) וַיֹּ֤אמֶר לָעָם֙ לְיוֹשְׁבֵ֣י יְרוּשָׁלִַ֔ם לָתֵ֕ת מְנָ֥ת הַכֹּהֲנִ֖ים וְהַלְוִיִּ֑ם לְמַ֥עַן יֶחֶזְק֖וּ בְּתוֹרַ֥ת יְהוָֽה׃ (נח) וְכִפְרֹ֣ץ הַדָּבָ֗ר הִרְבּ֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ רֵאשִׁ֣ית דָּגָ֗ן תִּיר֤וֹשׁ וְיִצְהָר֙ וּדְבַ֔שׁ וְכֹ֖ל תְּבוּאַ֣ת שָׂדֶ֑ה וּמַעְשַׂ֥ר הַכֹּ֛ל לָרֹ֖ב הֵבִֽיאוּ׃ (נט) וּבְנֵ֧י יִשְׂרָאֵ֣ל וִֽיהוּדָ֗ה הַיּֽוֹשְׁבִים֮ בְּעָרֵ֣י יְהוּדָה֒ גַּם־ הֵ֗ם מַעְשַׂ֤ר בָּקָר֙ וָצֹ֔אן וּמַעְשַׂ֣ר קָֽדָשִׁ֔ים הַמְקֻדָּשִׁ֖ים לַיהוָ֣ה אֱלֹהֵיהֶ֑ם הֵבִ֕יאוּ וַֽיִּתְּנ֖וּ עֲרֵמ֥וֹת עֲרֵמֽוֹת׃ (ס) בַּחֹ֙דֶשׁ֙ הַשְּׁלִשִׁ֔י הֵחֵ֥לּוּ הָעֲרֵמ֖וֹת לְיִסּ֑וֹד וּבַחֹ֥דֶשׁ הַשְּׁבִיעִ֖י כִּלּֽוּ׃ (סא) וַיָּבֹ֙אוּ֙ יְחִזְקִיָּ֣הוּ וְהַשָּׂרִ֔ים וַיִּרְא֖וּ אֶת־ הָעֲרֵמ֑וֹת וַֽיְבָרֲכוּ֙ אֶת־ יְהוָ֔ה וְאֵ֖ת עַמּ֥וֹ יִשְׂרָאֵֽל׃ (סב) וַיִּדְרֹ֣שׁ יְחִזְקִיָּ֗הוּ עַל־ הַכֹּֽהֲנִ֛ים
