דברי הימים סדר יח

(א) וַיֹּ֕אמֶר לָכֵ֖ן שִׁמְע֣וּ דְבַר־ יְהוָ֑ה רָאִ֤יתִי אֶת־ יְהוָה֙ יוֹשֵׁ֣ב עַל־ כִּסְא֔וֹ וְכָל־ צְבָ֤א הַשָּׁמַ֙יִם֙ עֹֽמְדִ֔ים עַל־ יְמִינ֖וֹ וּשְׂמֹאלֽוֹ׃ (ב) וַיֹּ֣אמֶר יְהוָ֗ה מִ֤י יְפַתֶּה֙ אֶת־ אַחְאָ֣ב מֶֽלֶךְ־ יִשְׂרָאֵ֔ל וְיַ֕עַל וְיִפֹּ֖ל בְּרָמ֣וֹת גִּלְעָ֑ד וַיֹּ֕אמֶר זֶ֚ה אֹמֵ֣ר כָּ֔כָה וְזֶ֖ה אֹמֵ֥ר כָּֽכָה׃ (ג) וַיֵּצֵ֣א הָר֗וּחַ וַֽיַּעֲמֹד֙ לִפְנֵ֣י יְהוָ֔ה וַיֹּ֖אמֶר אֲנִ֣י אֲפַתֶּ֑נּוּ וַיֹּ֧אמֶר יְהוָ֛ה אֵלָ֖יו בַּמָּֽה׃ (ד) וַיֹּ֗אמֶר אֵצֵא֙ וְהָיִ֙יתִי֙ לְר֣וּחַ שֶׁ֔קֶר בְּפִ֖י כָּל־ נְבִיאָ֑יו וַיֹּ֗אמֶר תְּפַתֶּה֙ וְגַם־ תּוּכָ֔ל צֵ֖א וַעֲשֵׂה־ כֵֽן׃ (ה) וְעַתָּ֗ה הִנֵּ֨ה נָתַ֤ן יְהוָה֙ ר֣וּחַ שֶׁ֔קֶר בְּפִ֖י נְבִיאֶ֣יךָ אֵ֑לֶּה וַֽיהוָ֔ה דִּבֶּ֥ר עָלֶ֖יךָ רָעָֽה׃ (ו) וַיִּגַּשׁ֙ צִדְקִיָּ֣הוּ בֶֽן־ כְּנַעֲנָ֔ה וַיַּ֥ךְ אֶת־ מִיכָ֖יְהוּ עַל־ הַלֶּ֑חִי וַיֹּ֗אמֶר אֵ֣י זֶ֤ה הַדֶּ֙רֶךְ֙ עָבַ֧ר רֽוּחַ־ יְהוָ֛ה מֵאִתִּ֖י לְדַבֵּ֥ר אֹתָֽךְ׃ (ז) וַיֹּ֣אמֶר מִיכָ֔יְהוּ הִנְּךָ֥ רֹאֶ֖ה בַּיּ֣וֹם הַה֑וּא אֲשֶׁ֥ר תָּב֛וֹא חֶ֥דֶר בְּחֶ֖דֶר לְהֵחָבֵֽא׃ (ח) וַיֹּ֙אמֶר֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל קְחוּ֙ אֶת־ מִיכָ֔יְהוּ וַהֲשִׁיבֻ֖הוּ אֶל־ אָמ֣וֹן שַׂר־ הָעִ֑יר וְאֶל־ יוֹאָ֖שׁ בֶּן־ הַמֶּֽלֶךְ׃ (ט) וַאֲמַרְתֶּ֗ם כֹּ֚ה אָמַ֣ר הַמֶּ֔לֶךְ שִׂ֥ימוּ זֶ֖ה בֵּ֣ית הַכֶּ֑לֶא וְהַאֲכִלֻ֜הוּ לֶ֤חֶם לַ֙חַץ֙ וּמַ֣יִם לַ֔חַץ עַ֖ד שׁוּבִ֥י בְשָׁלֽוֹם׃ (י) וַיֹּ֣אמֶר מִיכָ֔יְהוּ אִם־ שׁ֤וֹב תָּשׁוּב֙ בְּשָׁל֔וֹם לֹא־ דִבֶּ֥ר יְהוָ֖ה בִּ֑י וַיֹּ֕אמֶר שִׁמְע֖וּ עַמִּ֥ים כֻּלָּֽם׃ (יא) וַיַּ֧עַל מֶֽלֶךְ־ יִשְׂרָאֵ֛ל וִֽיהוֹשָׁפָ֥ט מֶֽלֶךְ־ יְהוּדָ֖ה אֶל־ רָמֹ֥ת גִּלְעָֽד׃ (יב) וַיֹּאמֶר֩ מֶ֨לֶךְ יִשְׂרָאֵ֜ל אֶל־ יְהוֹשָׁפָ֗ט הִתְחַפֵּשׂ֙ וָב֣וֹא בַמִּלְחָמָ֔ה וְאַתָּ֖ה לְבַ֣שׁ בְּגָדֶ֑יךָ וַיִּתְחַפֵּשׂ֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיָּבֹ֖אוּ בַּמִּלְחָמָֽה׃ (יג) וּמֶ֣לֶךְ אֲרָ֡ם צִוָּה֩ אֶת־ שָׂרֵ֨י הָרֶ֤כֶב אֲשֶׁר־ לוֹ֙ לֵאמֹ֔ר לֹ֚א תִּלָּ֣חֲמ֔וּ אֶת־ הַקָּטֹ֖ן אֶת־ הַגָּד֑וֹל כִּ֛י אִֽם־ אֶת־ מֶ֥לֶךְ יִשְׂרָאֵ֖ל לְבַדּֽוֹ׃ (יד) וַיְהִ֡י כִּרְאוֹת֩ שָׂרֵ֨י הָרֶ֜כֶב אֶת־ יְהוֹשָׁפָ֗ט וְהֵ֤מָּה אָֽמְרוּ֙ מֶ֣לֶךְ יִשְׂרָאֵ֣ל ה֔וּא וַיָּסֹ֥בּוּ עָלָ֖יו לְהִלָּחֵ֑ם וַיִּזְעַ֤ק יְהֽוֹשָׁפָט֙ וַֽיהוָ֣ה עֲזָר֔וֹ וַיְסִיתֵ֥ם אֱלֹהִ֖ים מִמֶּֽנּוּ׃ (טו) וַיְהִ֗י כִּרְאוֹת֙ שָׂרֵ֣י הָרֶ֔כֶב כִּ֥י לֹא־ הָיָ֖ה מֶ֣לֶךְ יִשְׂרָאֵ֑ל וַיָּשֻׁ֖בוּ מֵאַחֲרָֽיו׃ (טז) וְאִ֗ישׁ מָשַׁ֤ךְ בַּקֶּ֙שֶׁת֙ לְתֻמּ֔וֹ וַיַּךְ֙ אֶת־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל בֵּ֥ין הַדְּבָקִ֖ים וּבֵ֣ין הַשִּׁרְיָ֑ן וַיֹּ֣אמֶר לָֽרַכָּ֗ב הֲפֹ֧ךְ (ידיך) יָדְךָ֛ וְהוֹצֵאתַ֥נִי מִן־ הַֽמַּחֲנֶ֖ה כִּ֥י הָחֳלֵֽיתִי׃ (יז) וַתַּ֤עַל הַמִּלְחָמָה֙ בַּיּ֣וֹם הַה֔וּא וּמֶ֣לֶךְ יִשְׂרָאֵ֗ל הָיָ֨ה מַעֲמִ֧יד בַּמֶּרְכָּבָ֛ה נֹ֥כַח אֲרָ֖ם עַד־ הָעָ֑רֶב וַיָּ֕מָת לְעֵ֖ת בּ֥וֹא הַשָּֽׁמֶשׁ׃ (יח) וַ֠יָּשָׁב יְהוֹשָׁפָ֨ט מֶֽלֶךְ־ יְהוּדָ֧ה אֶל־ בֵּית֛וֹ בְּשָׁל֖וֹם לִֽירוּשָׁלִָֽם׃ (יט) וַיֵּצֵ֣א אֶל־ פָּנָ֗יו יֵה֣וּא בֶן־ חֲנָ֘נִי֮ הַחֹזֶה֒ וַיֹּ֙אמֶר֙ אֶל־ הַמֶּ֣לֶךְ יְהוֹשָׁפָ֔ט הֲלָרָשָׁ֣ע לַעְזֹ֔ר וּלְשֹׂנְאֵ֥י יְהוָ֖ה תֶּאֱהָ֑ב וּבָזֹאת֙ עָלֶ֣יךָ קֶּ֔צֶף מִלִּפְנֵ֖י יְהוָֽה׃ (כ) אֲבָ֕ל דְּבָרִ֥ים טוֹבִ֖ים נִמְצְא֣וּ עִמָּ֑ךְ כִּֽי־ בִעַ֤רְתָּ הָאֲשֵׁרוֹת֙ מִן־ הָאָ֔רֶץ וַהֲכִינ֥וֹתָ לְבָבְךָ֖ לִדְרֹ֥שׁ הָֽאֱלֹהִֽים׃ (כא) וַיֵּ֥שֶׁב יְהוֹשָׁפָ֖ט בִּירוּשָׁלִָ֑ם וַיָּ֜שָׁב וַיֵּצֵ֣א בָעָ֗ם מִבְּאֵ֥ר שֶׁ֙בַע֙ עַד־ הַ֣ר אֶפְרַ֔יִם וַיְשִׁיבֵ֕ם אֶל־ יְהוָ֖ה אֱלֹהֵ֥י אֲבוֹתֵיהֶֽם׃ (כב) וַיַּעֲמֵ֨ד שֹֽׁפְטִ֜ים בָּאָ֗רֶץ בְּכָל־ עָרֵ֧י יְהוּדָ֛ה הַבְּצֻר֖וֹת לְעִ֥יר וָעִֽיר׃ (כג) וַיֹּ֣אמֶר אֶל־ הַשֹּֽׁפְטִ֗ים רְאוּ֙ מָֽה־ אַתֶּ֣ם עֹשִׂ֔ים כִּ֣י לֹ֧א לְאָדָ֛ם תִּשְׁפְּט֖וּ כִּ֣י לַיהוָ֑ה וְעִמָּכֶ֖ם בִּדְבַ֥ר מִשְׁפָּֽט׃ (כד) וְעַתָּ֕ה יְהִ֥י פַֽחַד־ יְהוָ֖ה עֲלֵיכֶ֑ם שִׁמְר֣וּ וַעֲשׂ֔וּ כִּֽי־ אֵ֞ין עִם־ יְהוָ֣ה אֱלֹהֵ֗ינוּ עַוְלָ֛ה וּמַשֹּׂ֥א פָנִ֖ים וּמִקַּח־ שֹֽׁחַד׃ (כה) וְגַ֣ם בִּ֠ירוּשָׁלִַם הֶעֱמִ֨יד יְהוֹשָׁפָ֜ט מִן־ הַלְוִיִּ֣ם וְהַכֹּהֲנִ֗ים וּמֵרָאשֵׁ֤י הָאָבוֹת֙ לְיִשְׂרָאֵ֔ל לְמִשְׁפַּ֥ט יְהוָ֖ה וְלָרִ֑יב וַיָּשֻׁ֖בוּ יְרוּשָׁלִָֽם׃ (כו) וַיְצַ֥ו עֲלֵיהֶ֖ם לֵאמֹ֑ר כֹּ֤ה תַעֲשׂוּן֙ בְּיִרְאַ֣ת יְהוָ֔ה בֶּאֱמוּנָ֖ה וּבְלֵבָ֥ב שָׁלֵֽם׃ (כז) וְכָל־ רִיב֩ אֲשֶׁר־ יָב֨וֹא עֲלֵיכֶ֜ם מֵאֲחֵיכֶ֣ם ׀ הַיֹּשְׁבִ֣ים בְּעָרֵיהֶ֗ם בֵּֽין־ דָּ֣ם ׀ לְדָם֮ בֵּין־ תּוֹרָ֣ה לְמִצְוָה֮ לְחֻקִּ֣ים וּלְמִשְׁפָּטִים֒ וְהִזְהַרְתֶּ֣ם אֹתָ֔ם וְלֹ֤א יֶאְשְׁמוּ֙ לַיהוָ֔ה וְהָֽיָה־ קֶ֥צֶף עֲלֵיכֶ֖ם וְעַל־ אֲחֵיכֶ֑ם כֹּ֥ה תַעֲשׂ֖וּן וְלֹ֥א תֶאְשָֽׁמוּ׃ (כח) וְהִנֵּ֡ה אֲמַרְיָ֣הוּ כֹהֵן֩ הָרֹ֨אשׁ עֲלֵיכֶ֜ם לְכֹ֣ל דְּבַר־ יְהוָ֗ה וּזְבַדְיָ֨הוּ בֶן־ יִשְׁמָעֵ֜אל הַנָּגִ֤יד לְבֵית־ יְהוּדָה֙ לְכֹ֣ל דְּבַר־ הַמֶּ֔לֶךְ וְשֹׁטְרִ֥ים הַלְוִיִּ֖ם לִפְנֵיכֶ֑ם חִזְק֣וּ וַעֲשׂ֔וּ וִיהִ֥י יְהוָ֖ה עִם־ הַטּֽוֹב׃ תחילת פרק כ (כט) וַיְהִ֣י אַֽחֲרֵיכֵ֡ן בָּ֣אוּ בְנֵי־ מוֹאָב֩ וּבְנֵ֨י עַמּ֜וֹן וְעִמָּהֶ֧ם ׀ מֵֽהָעַמּוֹנִ֛ים עַל־ יְהוֹשָׁפָ֖ט לַמִּלְחָמָֽה׃ (ל) וַיָּבֹ֗אוּ וַיַּגִּ֤ידוּ לִֽיהוֹשָׁפָט֙ לֵאמֹ֔ר בָּ֣א עָלֶ֜יךָ הָמ֥וֹן רָ֛ב מֵעֵ֥בֶר לַיָּ֖ם מֵאֲרָ֑ם וְהִנָּם֙ בְּחַֽצְצ֣וֹן תָּמָ֔ר הִ֖יא עֵ֥ין גֶּֽדִי׃ (לא) וַיִּרָ֕א וַיִּתֵּ֧ן יְהוֹשָׁפָ֛ט אֶת־ פָּנָ֖יו לִדְר֣וֹשׁ לַיהוָ֑ה וַיִּקְרָא־ צ֖וֹם עַל־ כָּל־ יְהוּדָֽה׃ (לב) וַיִּקָּבְצ֣וּ יְהוּדָ֔ה לְבַקֵּ֖שׁ מֵֽיְהוָ֑ה גַּ֚ם מִכָּל־ עָרֵ֣י יְהוּדָ֔ה בָּ֖אוּ לְבַקֵּ֥שׁ אֶת־ יְהוָֽה׃ (לג) וַיַּעֲמֹ֣ד יְהוֹשָׁפָ֗ט בִּקְהַ֧ל יְהוּדָ֛ה וִירוּשָׁלִַ֖ם בְּבֵ֣ית יְהוָ֑ה לִפְנֵ֖י הֶחָצֵ֥ר הַחֲדָשָֽׁה׃ (לד) וַיֹּאמַ֗ר יְהוָ֞ה אֱלֹהֵ֤י אֲבֹתֵ֙ינוּ֙ הֲלֹ֨א אַתָּֽה־ ה֤וּא אֱלֹהִים֙ בַּשָּׁמַ֔יִם וְאַתָּ֣ה בְּעֶצְי֥וֹן גָּֽבֶר׃ (ז) וַיִּתְנַבֵּ֞א אֱלִיעֶ֤זֶר בֶּן־ דֹּדָוָ֙הוּ֙ מִמָּ֣רֵשָׁ֔ה עַל־ יְהוֹשָׁפָ֖ט לֵאמֹ֑ר כְּהִֽתְחַבֶּרְךָ֣ עִם־ אֲחַזְיָ֗הוּ פָּרַ֤ץ יְהוָה֙ אֶֽת־ מַעֲשֶׂ֔יךָ וַיִּשָּׁבְר֣וּ אֳנִיּ֔וֹת וְלֹ֥א עָצְר֖וּ לָלֶ֥כֶת אֶל־ תַּרְשִֽׁישׁ׃ סוף פרק כ (ח) וַיִּשְׁכַּ֤ב יְהֽוֹשָׁפָט֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֥ר עִם־ אֲבֹתָ֖יו בְּעִ֣יר דָּוִ֑יד וַיִּמְלֹ֛ךְ יְהוֹרָ֥ם בְּנ֖וֹ וַעֲתַלְיָ֖ה מֹלֶ֥כֶת עַל־ הָאָֽרֶץ׃ (מ) וּבַשָּׁנָ֨ה הַשְּׁבִעִ֜ית הִתְחַזַּ֣ק יְהוֹיָדָ֗ע וַיִּקַּ֣ח אֶת־ שָׂרֵ֣י הַמֵּא֡וֹת לַעֲזַרְיָ֣הוּ בֶן־ יְרֹחָ֡ם וּלְיִשְׁמָעֵ֣אל בֶּן־ יְ֠הוֹחָנָן וְלַֽעֲזַרְיָ֨הוּ בֶן־ עוֹבֵ֜ד וְאֶת־ מַעֲשֵׂיָ֧הוּ בֶן־ עֲדָיָ֛הוּ וְאֶת־ אֱלִישָׁפָ֥ט בֶּן־ זִכְרִ֖י עִמּ֥וֹ בַבְּרִֽית׃ (מא) וַיָּסֹ֙בּוּ֙ בִּֽיהוּדָ֔ה וַיִּקְבְּצ֤וּ אֶת־ הַלְוִיִּם֙ מִכָּל־ עָרֵ֣י יְהוּדָ֔ה וְרָאשֵׁ֥י הָאָב֖וֹת לְיִשְׂרָאֵ֑ל וַיָּבֹ֖אוּ אֶל־ יְרוּשָׁלִָֽם׃ (מב) וַיִּכְרֹ֨ת כָּל־ הַקָּהָ֥ל בְּרִ֛ית בְּבֵ֥ית הָאֱלֹהִ֖ים עִם־ הַמֶּ֑לֶךְ וַיֹּ֣אמֶר לָהֶ֗ם הִנֵּ֤ה בֶן־ הַמֶּ֙לֶךְ֙ יִמְלֹ֔ךְ כַּאֲשֶׁ֛ר דִּבֶּ֥ר יְהוָ֖ה עַל־ בְּנֵ֥י דָוִֽיד׃ (מג) זֶ֥ה הַדָּבָ֖ר אֲשֶׁ֣ר תַּעֲשׂ֑וּ הַשְּׁלִשִׁ֨ית מִכֶּ֜ם בָּאֵ֣י הַשַּׁבָּ֗ת לַכֹּֽהֲנִים֙ וְלַלְוִיִּ֔ם לְשֹֽׁעֲרֵ֖י הַסִּפִּֽים׃ (מד) וְהַשְּׁלִשִׁית֙ בְּבֵ֣ית הַמֶּ֔לֶךְ וְהַשְּׁלִשִׁ֖ית בְּשַׁ֣עַר הַיְס֑וֹד וְכָל־ הָעָ֔ם בְּחַצְר֖וֹת בֵּ֥ית יְהוָֽה׃ (מה) וְאַל־ יָב֣וֹא בֵית־ יְהוָ֗ה כִּ֤י אִם־ הַכֹּֽהֲנִים֙ וְהַמְשָׁרְתִ֣ים לַלְוִיִּ֔ם הֵ֥מָּה יָבֹ֖אוּ כִּי־ קֹ֣דֶשׁ הֵ֑מָּה וְכָל־ הָעָ֔ם יִשְׁמְר֖וּ מִשְׁמֶ֥רֶת יְהוָֽה׃ (מו) וְהִקִּיפוּ֩ הַלְוִיִּ֨ם אֶת־ הַמֶּ֜לֶךְ סָבִ֗יב אִ֚ישׁ וְכֵלָ֣יו בְּיָד֔וֹ וְהַבָּ֥א אֶל־ הַבַּ֖יִת יוּמָ֑ת וִֽהְי֥וּ אֶת־ הַמֶּ֖לֶךְ בְּבֹא֥וֹ וּבְצֵאתֽוֹ׃ (מז) וַיַּעֲשׂ֨וּ הַלְוִיִּ֜ם וְכָל־ יְהוּדָ֗ה כְּכֹ֣ל אֲשֶׁר־ צִוָּה֮ יְהוֹיָדָ֣ע הַכֹּהֵן֒ וַיִּקְחוּ֙ אִ֣ישׁ אֶת־ אֲנָשָׁ֔יו בָּאֵ֣י הַשַּׁבָּ֔ת עִ֖ם יוֹצְאֵ֣י הַשַּׁבָּ֑ת כִּ֣י לֹ֥א פָטַ֛ר יְהוֹיָדָ֥ע הַכֹּהֵ֖ן אֶת־ הַֽמַּחְלְקֽוֹת׃ (מח) וַיִּתֵּן֩ יְהוֹיָדָ֨ע הַכֹּהֵ֜ן לְשָׂרֵ֣י הַמֵּא֗וֹת אֶת־ הַֽחֲנִיתִים֙ וְאֶת־ הַמָּגִנּוֹת֙ וְאֶת־ הַשְּׁלָטִ֔ים אֲשֶׁ֖ר לַמֶּ֣לֶךְ דָּוִ֑יד אֲשֶׁ֖ר בֵּ֥ית הָאֱלֹהִֽים׃ (מט) וַיַּעֲמֵ֨ד אֶת־ כָּל־ הָעָ֜ם וְאִ֣ישׁ ׀ שִׁלְח֣וֹ בְיָד֗וֹ מִכֶּ֨תֶף הַבַּ֤יִת הַיְמָנִית֙ עַד־ כֶּ֤תֶף הַבַּ֙יִת֙ הַשְּׂמָאלִ֔ית לַמִּזְבֵּ֖חַ וְלַבָּ֑יִת עַל־ הַמֶּ֖לֶךְ סָבִֽיב׃
