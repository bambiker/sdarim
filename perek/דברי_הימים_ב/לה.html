דברי הימים ב פרק לה לפי החלוקה לסדרים

פרק לה בספר דברי הימים ב (27 פסוקים) נמצא בדברי הימים סדר כד (פסוקים לג–לז) ובדברי הימים סדר כה (פסוקים א–כב).

דברי הימים סדר כד

(א) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה וַיֵּ֗לֶךְ בְּדַרְכֵי֙ דָּוִ֣יד אָבִ֔יו וְלֹא־ סָ֖ר יָמִ֥ין וּשְׂמֹֽאול׃ (ב) וּבִשְׁמוֹנֶ֨ה שָׁנִ֜ים לְמָלְכ֗וֹ וְהוּא֙ עוֹדֶ֣נּוּ נַ֔עַר הֵחֵ֕ל לִדְר֕וֹשׁ לֵאלֹהֵ֖י דָּוִ֣יד אָבִ֑יו וּבִשְׁתֵּ֧ים עֶשְׂרֵ֣ה שָׁנָ֗ה הֵחֵל֙ לְטַהֵ֔ר אֶת־ יְהוּדָה֙ וִיר֣וּשָׁלִַ֔ם מִן־ הַבָּמוֹת֙ וְהָ֣אֲשֵׁרִ֔ים וְהַפְּסִלִ֖ים וְהַמַּסֵּכֽוֹת׃ (ג) וַיְנַתְּצ֣וּ לְפָנָ֗יו אֵ֚ת מִזְבְּח֣וֹת הַבְּעָלִ֔ים וְהַֽחַמָּנִ֛ים אֲשֶׁר־ לְמַ֥עְלָה מֵעֲלֵיהֶ֖ם גִּדֵּ֑עַ וְ֠הָאֲשֵׁרִים וְהַפְּסִלִ֤ים וְהַמַּסֵּכוֹת֙ שִׁבַּ֣ר וְהֵדַ֔ק וַיִּזְרֹק֙ עַל־ פְּנֵ֣י הַקְּבָרִ֔ים הַזֹּבְחִ֖ים לָהֶֽם׃ (ד) וְעַצְמוֹת֙ כֹּֽהֲנִ֔ים שָׂרַ֖ף עַל־ (מזבחותים) מִזְבְּחוֹתָ֑ם וַיְטַהֵ֥ר אֶת־ יְהוּדָ֖ה וְאֶת־ יְרוּשָׁלִָֽם׃ (ה) וּבְעָרֵ֨י מְנַשֶּׁ֧ה וְאֶפְרַ֛יִם וְשִׁמְע֖וֹן וְעַד־ נַפְתָּלִ֑י (בהר בתיהם) בְּחַרְבֹתֵיהֶ֖ם סָבִֽיב׃ (ו) וַיְנַתֵּ֣ץ אֶת־ הַֽמִּזְבְּח֗וֹת וְאֶת־ הָאֲשֵׁרִ֤ים וְהַפְּסִלִים֙ כִּתַּ֣ת לְהֵדַ֔ק וְכָל־ הַֽחַמָּנִ֥ים גִּדַּ֖ע בְּכָל־ אֶ֣רֶץ יִשְׂרָאֵ֑ל וַיָּ֖שָׁב לִירוּשָׁלִָֽם׃ (ז) וּבִשְׁנַ֨ת שְׁמוֹנֶ֤ה עֶשְׂרֵה֙ לְמָלְכ֔וֹ לְטַהֵ֥ר הָאָ֖רֶץ וְהַבָּ֑יִת שָׁ֠לַח אֶת־ שָׁפָ֨ן בֶּן־ אֲצַלְיָ֜הוּ וְאֶת־ מַעֲשֵׂיָ֣הוּ שַׂר־ הָעִ֗יר וְ֠אֵת יוֹאָ֤ח בֶּן־ יֽוֹאָחָז֙ הַמַּזְכִּ֔יר לְחַזֵּ֕ק אֶת־ בֵּ֖ית יְהוָ֥ה אֱלֹהָֽיו׃ (ח) וַיָּבֹ֜אוּ אֶל־ חִלְקִיָּ֣הוּ ׀ הַכֹּהֵ֣ן הַגָּד֗וֹל וַֽיִּתְּנוּ֮ אֶת־ הַכֶּסֶף֮ הַמּוּבָ֣א בֵית־ אֱלֹהִים֒ אֲשֶׁ֣ר אָסְפֽוּ־ הַלְוִיִּם֩ שֹׁמְרֵ֨י הַסַּ֜ף מִיַּ֧ד מְנַשֶּׁ֣ה וְאֶפְרַ֗יִם וּמִכֹּל֙ שְׁאֵרִ֣ית יִשְׂרָאֵ֔ל וּמִכָּל־ יְהוּדָ֖ה וּבִנְיָמִ֑ן (וישבי) וַיָּשֻׁ֖בוּ יְרֽוּשָׁלִָֽם׃ (ט) וַֽיִּתְּנ֗וּ עַל־ יַד֙ עֹשֵׂ֣ה הַמְּלָאכָ֔ה הַמֻּפְקָדִ֖ים בְּבֵ֣ית יְהוָ֑ה וַיִּתְּנ֨וּ אֹת֜וֹ עוֹשֵׂ֣י הַמְּלָאכָ֗ה אֲשֶׁ֤ר עֹשִׂים֙ בְּבֵ֣ית יְהוָ֔ה לִבְדּ֥וֹק וּלְחַזֵּ֖ק הַבָּֽיִת׃ (י) וַֽיִּתְּנ֗וּ לֶחָֽרָשִׁים֙ וְלַבֹּנִ֔ים לִקְנוֹת֙ אַבְנֵ֣י מַחְצֵ֔ב וְעֵצִ֖ים לַֽמְחַבְּר֑וֹת וּלְקָרוֹת֙ אֶת־ הַבָּ֣תִּ֔ים אֲשֶׁ֥ר הִשְׁחִ֖יתוּ מַלְכֵ֥י יְהוּדָֽה׃ (יא) וְהָאֲנָשִׁים֩ עֹשִׂ֨ים בֶּאֱמוּנָ֜ה בַּמְּלָאכָ֗ה וַעֲלֵיהֶ֣ם ׀ מֻ֠פְקָדִים יַ֣חַת וְעֹבַדְיָ֤הוּ הַלְוִיִּם֙ מִן־ בְּנֵ֣י מְרָרִ֔י וּזְכַרְיָ֧ה וּמְשֻׁלָּ֛ם מִן־ בְּנֵ֥י הַקְּהָתִ֖ים לְנַצֵּ֑חַ וְהַ֨לְוִיִּ֔ם כָּל־ מֵבִ֖ין בִּכְלֵי־ שִֽׁיר׃ (יב) וְעַ֣ל הַסַּבָּלִ֗ים וּֽמְנַצְּחִים֙ לְכֹל֙ עֹשֵׂ֣ה מְלָאכָ֔ה לַעֲבוֹדָ֖ה וַעֲבוֹדָ֑ה וּמֵֽהַלְוִיִּ֔ם סוֹפְרִ֥ים וְשֹׁטְרִ֖ים וְשׁוֹעֲרִֽים׃ (יג) וּבְהוֹצִיאָ֣ם אֶת־ הַכֶּ֔סֶף הַמּוּבָ֖א בֵּ֣ית יְהוָ֑ה מָצָא֙ חִלְקִיָּ֣הוּ הַכֹּהֵ֔ן אֶת־ סֵ֥פֶר תּֽוֹרַת־ יְהוָ֖ה בְּיַד־ מֹשֶֽׁה׃ (יד) וַיַּ֣עַן חִלְקִיָּ֗הוּ וַיֹּ֙אמֶר֙ אֶל־ שָׁפָ֣ן הַסּוֹפֵ֔ר סֵ֧פֶר הַתּוֹרָ֛ה מָצָ֖אתִי בְּבֵ֣ית יְהוָ֑ה וַיִּתֵּ֧ן חִלְקִיָּ֛הוּ אֶת־ הַסֵּ֖פֶר אֶל־ שָׁפָֽן׃ (טו) וַיָּבֵ֨א שָׁפָ֤ן אֶת־ הַסֵּ֙פֶר֙ אֶל־ הַמֶּ֔לֶךְ וַיָּ֨שֶׁב ע֧וֹד אֶת־ הַמֶּ֛לֶךְ דָּבָ֖ר לֵאמֹ֑ר כֹּ֛ל אֲשֶׁר־ נִתַּ֥ן בְּיַד־ עֲבָדֶ֖יךָ הֵ֥ם עֹשִֽׂים׃ (טז) וַיַּתִּ֕יכוּ אֶת־ הַכֶּ֖סֶף הַנִּמְצָ֣א בְּבֵית־ יְהוָ֑ה וַֽיִּתְּנ֗וּהוּ עַל־ יַד֙ הַמֻּפְקָדִ֔ים וְעַל־ יַ֖ד עוֹשֵׂ֥י הַמְּלָאכָֽה׃ (יז) וַיַּגֵּ֞ד שָׁפָ֤ן הַסּוֹפֵר֙ לַמֶּ֣לֶךְ לֵאמֹ֔ר סֵ֚פֶר נָ֣תַן לִ֔י חִלְקִיָּ֖הוּ הַכֹּהֵ֑ן וַיִּקְרָא־ ב֥וֹ שָׁפָ֖ן לִפְנֵ֥י הַמֶּֽלֶךְ׃ (יח) וַיְהִי֙ כִּשְׁמֹ֣עַ הַמֶּ֔לֶךְ אֵ֖ת דִּבְרֵ֣י הַתּוֹרָ֑ה וַיִּקְרַ֖ע אֶת־ בְּגָדָֽיו׃ (יט) וַיְצַ֣ו הַמֶּ֡לֶךְ אֶת־ חִלְקִיָּ֡הוּ וְאֶת־ אֲחִיקָ֣ם בֶּן־ שָׁ֠פָן וְאֶת־ עַבְדּ֨וֹן בֶּן־ מִיכָ֜ה וְאֵ֣ת ׀ שָׁפָ֣ן הַסּוֹפֵ֗ר וְאֵ֛ת עֲשָׂיָ֥ה עֶֽבֶד־ הַמֶּ֖לֶךְ לֵאמֹֽר׃ (כ) לְכוּ֩ דִרְשׁ֨וּ אֶת־ יְהוָ֜ה בַּעֲדִ֗י וּבְעַד֙ הַנִּשְׁאָר֙ בְּיִשְׂרָאֵ֣ל וּבִֽיהוּדָ֔ה עַל־ דִּבְרֵ֥י הַסֵּ֖פֶר אֲשֶׁ֣ר נִמְצָ֑א כִּֽי־ גְדוֹלָ֤ה חֲמַת־ יְהוָה֙ אֲשֶׁ֣ר נִתְּכָ֣ה בָ֔נוּ עַל֩ אֲשֶׁ֨ר לֹא־ שָׁמְר֤וּ אֲבוֹתֵ֙ינוּ֙ אֶת־ דְּבַ֣ר יְהוָ֔ה לַעֲשׂ֕וֹת כְּכָל־ הַכָּת֖וּב עַל־ הַסֵּ֥פֶר הַזֶּֽה׃ (כא) וַיֵּ֨לֶךְ חִלְקִיָּ֜הוּ וַאֲשֶׁ֣ר הַמֶּ֗לֶךְ אֶל־ חֻלְדָּ֨ה הַנְּבִיאָ֜ה אֵ֣שֶׁת ׀ שַׁלֻּ֣ם בֶּן־ (תוקהת) תָּקְהַ֗ת בֶּן־ חַסְרָה֙ שׁוֹמֵ֣ר הַבְּגָדִ֔ים וְהִ֛יא יוֹשֶׁ֥בֶת בִּירוּשָׁלִַ֖ם בַּמִּשְׁנֶ֑ה וַיְדַבְּר֥וּ אֵלֶ֖יהָ כָּזֹֽאת׃ (כב) וַתֹּ֣אמֶר לָהֶ֔ם כֹּה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל אִמְר֣וּ לָאִ֔ישׁ אֲשֶׁר־ שָׁלַ֥ח אֶתְכֶ֖ם אֵלָֽי׃ (כג) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֨י מֵבִ֥יא רָעָ֛ה עַל־ הַמָּק֥וֹם הַזֶּ֖ה וְעַל־ יוֹשְׁבָ֑יו אֵ֤ת כָּל־ הָאָלוֹת֙ הַכְּתוּב֣וֹת עַל־ הַסֵּ֔פֶר אֲשֶׁ֣ר קָֽרְא֔וּ לִפְנֵ֖י מֶ֥לֶךְ יְהוּדָֽה׃ (כד) תַּ֣חַת ׀ אֲשֶׁ֣ר עֲזָב֗וּנִי (ויקטירו) וַֽיְקַטְּרוּ֙ לֵֽאלֹהִ֣ים אֲחֵרִ֔ים לְמַ֙עַן֙ הַכְעִיסֵ֔נִי בְּכֹ֖ל מַעֲשֵׂ֣י יְדֵיהֶ֑ם וְתִתַּ֧ךְ חֲמָתִ֛י בַּמָּק֥וֹם הַזֶּ֖ה וְלֹ֥א תִכְבֶּֽה׃ (כה) וְאֶל־ מֶ֣לֶךְ יְהוּדָ֗ה הַשֹּׁלֵ֤חַ אֶתְכֶם֙ לִדְר֣וֹשׁ בַּֽיהוָ֔ה כֹּ֥ה תֹאמְר֖וּ אֵלָ֑יו כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הַדְּבָרִ֖ים אֲשֶׁ֥ר שָׁמָֽעְתָּ׃ (כו) יַ֠עַן רַךְ־ לְבָ֨בְךָ֜ וַתִּכָּנַ֣ע ׀ מִלִּפְנֵ֣י אֱלֹהִ֗ים בְּשָׁמְעֲךָ֤ אֶת־ דְּבָרָיו֙ עַל־ הַמָּק֤וֹם הַזֶּה֙ וְעַל־ יֹ֣שְׁבָ֔יו וַתִּכָּנַ֣ע לְפָנַ֔י וַתִּקְרַ֥ע אֶת־ בְּגָדֶ֖יךָ וַתֵּ֣בְךְּ לְפָנָ֑י וְגַם־ אֲנִ֥י שָׁמַ֖עְתִּי נְאֻם־ יְהוָֽה׃ (כז) הִנְנִ֨י אֹֽסִפְךָ֜ אֶל־ אֲבֹתֶ֗יךָ וְנֶאֱסַפְתָּ֣ אֶל־ קִבְרֹתֶיךָ֮ בְּשָׁלוֹם֒ וְלֹא־ תִרְאֶ֣ינָה עֵינֶ֔יךָ בְּכֹל֙ הָֽרָעָ֔ה אֲשֶׁ֨ר אֲנִ֥י מֵבִ֛יא עַל־ הַמָּק֥וֹם הַזֶּ֖ה וְעַל־ יֹשְׁבָ֑יו וַיָּשִׁ֥יבוּ אֶת־ הַמֶּ֖לֶךְ דָּבָֽר׃ (כח) וַיִּשְׁלַ֖ח הַמֶּ֑לֶךְ וַיֶּאֱסֹ֕ף אֶת־ כָּל־ זִקְנֵ֥י יְהוּדָ֖ה וִירוּשָׁלִָֽם׃ (כט) וַיַּ֣עַל הַמֶּ֣לֶךְ בֵּית־ יְ֠הוָה וְכָל־ אִ֨ישׁ יְהוּדָ֜ה וְיֹשְׁבֵ֣י יְרוּשָׁלִַ֗ם וְהַכֹּֽהֲנִים֙ וְהַלְוִיִּ֔ם וְכָל־ הָעָ֖ם מִגָּד֣וֹל וְעַד־ קָטָ֑ן וַיִּקְרָ֣א בְאָזְנֵיהֶ֗ם אֶת־ כָּל־ דִּבְרֵי֙ סֵ֣פֶר הַבְּרִ֔ית הַנִּמְצָ֖א בֵּ֥ית יְהוָֽה׃ (ל) וַיַּעֲמֹ֨ד הַמֶּ֜לֶךְ עַל־ עָמְד֗וֹ וַיִּכְרֹ֣ת אֶֽת־ הַבְּרִית֮ לִפְנֵ֣י יְהוָה֒ לָלֶ֜כֶת אַחֲרֵ֣י יְהוָ֗ה וְלִשְׁמ֤וֹר אֶת־ מִצְוֺתָיו֙ וְהָאַחֲרֹנִ֑ים הִנָּ֣ם כְּתוּבִ֔ים עַל־ סֵ֥פֶר מַלְכֵֽי־ יִשְׂרָאֵ֖ל וִיהוּדָֽה׃ סוף פרק לה (כג) וַיִּקְחוּ֙ עַם־ הָאָ֔רֶץ אֶת־ יְהוֹאָחָ֖ז בֶּן־ יֹאשִׁיָּ֑הוּ וַיַּמְלִיכֻ֥הוּ תַֽחַת־ אָבִ֖יו בִּירוּשָׁלִָֽם׃ (כד) בֶּן־ שָׁל֧וֹשׁ וְעֶשְׂרִ֛ים שָׁנָ֖ה יוֹאָחָ֣ז בְּמָלְכ֑וֹ וּשְׁלֹשָׁ֣ה חֳדָשִׁ֔ים מָלַ֖ךְ בִּירוּשָׁלִָֽם׃ (כה) וַיְסִירֵ֥הוּ מֶֽלֶךְ־ מִצְרַ֖יִם בִּֽירוּשָׁלִָ֑ם וַֽיַּעֲנֹשׁ֙ אֶת־ הָאָ֔רֶץ מֵאָ֥ה כִכַּר־ כֶּ֖סֶף וְכִכַּ֥ר זָהָֽב׃ (כו) וַיַּמְלֵ֨ךְ מֶֽלֶךְ־ מִצְרַ֜יִם אֶת־ אֶלְיָקִ֣ים אָחִ֗יו עַל־ יְהוּדָה֙ וִיר֣וּשָׁלִַ֔ם וַיַּסֵּ֥ב אֶת־ שְׁמ֖וֹ יְהוֹיָקִ֑ים וְאֶת־ יוֹאָחָ֤ז אָחִיו֙ לָקַ֣ח נְכ֔וֹ וַיְבִיאֵ֖הוּ מִצְרָֽיְמָה׃ (כז) בֶּן־ עֶשְׂרִ֨ים וְחָמֵ֤שׁ שָׁנָה֙ יְהוֹיָקִ֣ים בְּמָלְכ֔וֹ וְאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וַיַּ֣עַשׂ הָרַ֔ע בְּעֵינֵ֖י יְהוָ֥ה אֱלֹהָֽיו׃ (כח) עָלָ֣יו עָלָ֔ה נְבוּכַדְנֶאצַּ֖ר מֶ֣לֶךְ בָּבֶ֑ל וַיַּֽאַסְרֵ֙הוּ֙ בַּֽנְחֻשְׁתַּ֔יִם לְהֹלִיכ֖וֹ בָּבֶֽלָה׃ (כט) וּמִכְּלֵי֙ בֵּ֣ית יְהוָ֔ה הֵבִ֥יא נְבוּכַדְנֶאצַּ֖ר לְבָבֶ֑ל וַיִּתְּנֵ֥ם בְּהֵיכָל֖וֹ בְּבָבֶֽל׃ (ל) וְיֶתֶר֩ דִּבְרֵ֨י יְהֽוֹיָקִ֜ים וְתֹֽעֲבֹתָ֤יו אֲשֶׁר־ עָשָׂה֙ וְהַנִּמְצָ֣א עָלָ֔יו הִנָּ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר מַלְכֵ֥י יִשְׂרָאֵ֖ל וִֽיהוּדָ֑ה וַיִּמְלֹ֛ךְ יְהוֹיָכִ֥ין בְּנ֖וֹ תַּחְתָּֽיו׃ (לא) בֶּן־ שְׁמוֹנֶ֤ה שָׁנִים֙ יְהוֹיָכִ֣ין בְּמָלְכ֔וֹ וּשְׁלֹשָׁ֤ה חֳדָשִׁים֙ וַעֲשֶׂ֣רֶת יָמִ֔ים מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֥י יְהוָֽה׃ (לב) וְלִתְשׁוּבַ֣ת הַשָּׁנָ֗ה שָׁלַח֙ הַמֶּ֣לֶךְ נְבֽוּכַדְנֶאצַּ֔ר וַיְבִאֵ֣הוּ בָבֶ֔לָה עִם־ כְּלֵ֖י חֶמְדַּ֣ת בֵּית־ יְהוָ֑ה וַיַּמְלֵךְ֙ אֶת־ צִדְקִיָּ֣הוּ אָחִ֔יו עַל־ יְהוּדָ֖ה וִֽירוּשָׁלִָֽם׃
(לג) בֶּן־ עֶשְׂרִ֧ים וְאַחַ֛ת שָׁנָ֖ה צִדְקִיָּ֣הוּ בְמָלְכ֑וֹ וְאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָֽם׃ (לד) וַיַּ֣עַשׂ הָרַ֔ע בְּעֵינֵ֖י יְהוָ֣ה אֱלֹהָ֑יו לֹ֣א נִכְנַ֗ע מִלִּפְנֵ֛י יִרְמְיָ֥הוּ הַנָּבִ֖יא מִפִּ֥י יְהוָֽה׃ (לה) וְ֠גַם בַּמֶּ֤לֶךְ נְבֽוּכַדְנֶאצַּר֙ מָרָ֔ד אֲשֶׁ֥ר הִשְׁבִּיע֖וֹ בֵּֽאלֹהִ֑ים וַיֶּ֤קֶשׁ אֶת־ עָרְפּוֹ֙ וַיְאַמֵּ֣ץ אֶת־ לְבָב֔וֹ מִשּׁ֕וּב אֶל־ יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (לו) גַּ֠ם כָּל־ שָׂרֵ֨י הַכֹּהֲנִ֤ים וְהָעָם֙ הִרְבּ֣וּ לִמְעָול־ מַ֔עַל כְּכֹ֖ל תֹּעֲב֣וֹת הַגּוֹיִ֑ם וַֽיְטַמְּאוּ֙ אֶת־ בֵּ֣ית יְהוָ֔ה אֲשֶׁ֥ר הִקְדִּ֖ישׁ בִּירוּשָׁלִָֽם׃ (לז) וַיִּשְׁלַ֡ח יְהוָה֩ אֱלֹהֵ֨י אֲבוֹתֵיהֶ֧ם עֲלֵיהֶ֛ם בְּיַ֥ד מַלְאָכָ֖יו הַשְׁכֵּ֣ם וְשָׁל֑וֹחַ כִּֽי־ חָמַ֥ל עַל־ עַמּ֖וֹ וְעַל־ מְעוֹנֽוֹ׃ (לח) וַיִּֽהְי֤וּ מַלְעִבִים֙ בְּמַלְאֲכֵ֣י הָאֱלֹהִ֔ים וּבוֹזִ֣ים דְּבָרָ֔יו וּמִֽתַּעְתְּעִ֖ים בִּנְבִאָ֑יו עַ֠ד עֲל֧וֹת חֲמַת־ יְהוָ֛ה בְּעַמּ֖וֹ עַד־ לְאֵ֥ין מַרְפֵּֽא׃ (לט) וַיַּ֨עַל עֲלֵיהֶ֜ם אֶת־ מֶ֣לֶךְ (כשדיים) כַּשְׂדִּ֗ים וַיַּהֲרֹ֨ג בַּחוּרֵיהֶ֤ם בַּחֶ֙רֶב֙ בְּבֵ֣ית מִקְדָּשָׁ֔ם וְלֹ֥א חָמַ֛ל עַל־ בָּח֥וּר וּבְתוּלָ֖ה זָקֵ֣ן וְיָשֵׁ֑שׁ הַכֹּ֖ל נָתַ֥ן בְּיָדֽוֹ׃ (מ) וְ֠כֹל כְּלֵ֞י בֵּ֤ית הָאֱלֹהִים֙ הַגְּדֹלִ֣ים וְהַקְּטַנִּ֔ים וְאֹֽצְרוֹת֙ בֵּ֣ית יְהוָ֔ה וְאֹצְר֥וֹת הַמֶּ֖לֶךְ וְשָׂרָ֑יו הַכֹּ֖ל הֵבִ֥יא בָבֶֽל׃ (מא) וַֽיִּשְׂרְפוּ֙ אֶת־ בֵּ֣ית הָאֱלֹהִ֔ים וַֽיְנַתְּצ֔וּ אֵ֖ת חוֹמַ֣ת יְרוּשָׁלִָ֑ם וְכָל־ אַרְמְנוֹתֶ֙יהָ֙ שָׂרְפ֣וּ בָאֵ֔שׁ וְכָל־ כְּלֵ֥י מַחֲמַדֶּ֖יהָ לְהַשְׁחִֽית׃ (מב) וַיֶּ֛גֶל הַשְּׁאֵרִ֥ית מִן־ הַחֶ֖רֶב אֶל־ בָּבֶ֑ל וַֽיִּהְיוּ־ ל֤וֹ וּלְבָנָיו֙ לַעֲבָדִ֔ים עַד־ מְלֹ֖ךְ מַלְכ֥וּת פָּרָֽס׃ (מג) לְמַלֹּ֤אות דְּבַר־ יְהוָה֙ בְּפִ֣י יִרְמְיָ֔הוּ עַד־ רָצְתָ֥ה הָאָ֖רֶץ אֶת־ שַׁבְּתוֹתֶ֑יהָ כָּל־ יְמֵ֤י הָשַּׁמָּה֙ שָׁבָ֔תָה לְמַלֹּ֖אות שִׁבְעִ֥ים שָׁנָֽה׃ (מד) וּבִשְׁנַ֣ת אַחַ֗ת לְכ֙וֹרֶשׁ֙ מֶ֣לֶךְ פָּרַ֔ס לִכְל֥וֹת דְּבַר־ יְהוָ֖ה בְּפִ֣י יִרְמְיָ֑הוּ הֵעִ֣יר יְהוָ֗ה אֶת־ ר֙וּחַ֙ כּ֣וֹרֶשׁ מֶֽלֶךְ־ פָּרַ֔ס וַיַּֽעֲבֶר־ קוֹל֙ בְּכָל־ מַלְכוּת֔וֹ וְגַם־ בְּמִכְתָּ֖ב לֵאמֹֽר׃       (מה) כֹּה־ אָמַ֞ר כּ֣וֹרֶשׁ ׀ מֶ֣לֶךְ פָּרַ֗ס כָּל־ מַמְלְכ֤וֹת הָאָ֙רֶץ֙ נָ֣תַן לִ֗י יְהוָה֙ אֱלֹהֵ֣י הַשָּׁמַ֔יִם וְהֽוּא־ פָקַ֤ד עָלַי֙ לִבְנֽוֹת־ ל֣וֹ בַ֔יִת בִּירוּשָׁלִַ֖ם אֲשֶׁ֣ר בִּֽיהוּדָ֑ה מִֽי־ בָכֶ֣ם מִכָּל־ עַמּ֗וֹ יְהוָ֧ה אֱלֹהָ֛יו עִמּ֖וֹ וְיָֽעַל׃
