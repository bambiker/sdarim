פרק יב בספר דברי הימים ב (16 פסוקים) נמצא בדברי הימים סדר טו (פסוקים נא–סא) ובדברי הימים סדר טז (פסוקים א–ה).

דברי הימים סדר טו

(א) וְהֵ֣ם מְבִיאִ֣ים אִ֣ישׁ מִנְחָת֡וֹ כְּלֵ֣י כֶסֶף֩ וּכְלֵ֨י זָהָ֤ב וּשְׂלָמוֹת֙ נֵ֣שֶׁק וּבְשָׂמִ֔ים סוּסִ֖ים וּפְרָדִ֑ים דְּבַר־ שָׁנָ֖ה בְּשָׁנָֽה׃ (ב) וַיְהִ֨י לִשְׁלֹמֹ֜ה אַרְבַּעַת֩ אֲלָפִ֨ים אֻֽרְי֤וֹת סוּסִים֙ וּמַרְכָּב֔וֹת וּשְׁנֵים־ עָשָׂ֥ר אֶ֖לֶף פָּרָשִׁ֑ים וַיַּנִּיחֵם֙ בְּעָרֵ֣י הָרֶ֔כֶב וְעִם־ הַמֶּ֖לֶךְ בִּירוּשָׁלִָֽם׃ (ג) וַיְהִ֥י מוֹשֵׁ֖ל בְּכָל־ הַמְּלָכִ֑ים מִן־ הַנָּהָר֙ וְעַד־ אֶ֣רֶץ פְּלִשְׁתִּ֔ים וְעַ֖ד גְּב֥וּל מִצְרָֽיִם׃ (ד) וַיִּתֵּ֨ן הַמֶּ֧לֶךְ אֶת־ הַכֶּ֛סֶף בִּירוּשָׁלִַ֖ם כָּאֲבָנִ֑ים וְאֵ֣ת הָאֲרָזִ֗ים נָתַ֛ן כַּשִּׁקְמִ֥ים אֲשֶׁר־ בַּשְּׁפֵלָ֖ה לָרֹֽב׃ (ה) וּמוֹצִיאִ֨ים סוּסִ֧ים מִמִּצְרַ֛יִם לִשְׁלֹמֹ֖ה וּמִכָּל־ הָאֲרָצֽוֹת׃ (ו) וּשְׁאָר֙ דִּבְרֵ֣י שְׁלֹמֹ֔ה הָרִאשֹׁנִ֖ים וְהָאֲחֲרוֹנִ֑ים הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ דִּבְרֵי֙ נָתָ֣ן הַנָּבִ֔יא וְעַל־ נְבוּאַ֞ת אֲחִיָּ֣ה הַשִּֽׁילוֹנִ֗י וּבַחֲזוֹת֙ (יעדי) יֶעְדּ֣וֹ הַחֹזֶ֔ה עַל־ יָרָבְעָ֖ם בֶּן־ נְבָֽט׃ (ז) וַיִּמְלֹ֨ךְ שְׁלֹמֹ֧ה בִֽירוּשָׁלִַ֛ם עַל־ כָּל־ יִשְׂרָאֵ֖ל אַרְבָּעִ֥ים שָׁנָֽה׃ (ח) וַיִּשְׁכַּ֤ב שְׁלֹמֹה֙ עִם־ אֲבֹתָ֔יו וַֽיִּקְבְּרֻ֔הוּ בְּעִ֖יר דָּוִ֣יד אָבִ֑יו וַיִּמְלֹ֛ךְ רְחַבְעָ֥ם בְּנ֖וֹ תַּחְתָּֽיו׃ (ט) וַיֵּ֥לֶךְ רְחַבְעָ֖ם שְׁכֶ֑מָה כִּ֥י שְׁכֶ֛ם בָּ֥אוּ כָל־ יִשְׂרָאֵ֖ל לְהַמְלִ֥יךְ אֹתֽוֹ׃ (י) וַיְהִ֞י כִּשְׁמֹ֨עַ יָרָבְעָ֤ם בֶּן־ נְבָט֙ וְה֣וּא בְמִצְרַ֔יִם אֲשֶׁ֣ר בָּרַ֔ח מִפְּנֵ֖י שְׁלֹמֹ֣ה הַמֶּ֑לֶךְ וַיָּ֥שָׁב יָרָבְעָ֖ם מִמִּצְרָֽיִם׃ (יא) וַֽיִּשְׁלְחוּ֙ וַיִּקְרְאוּ־ ל֔וֹ וַיָּבֹ֥א יָרָבְעָ֖ם וְכָל־ יִשְׂרָאֵ֑ל וַֽיְדַבְּר֔וּ אֶל־ רְחַבְעָ֖ם לֵאמֹֽר׃ (יב) אָבִ֖יךָ הִקְשָׁ֣ה אֶת־ עֻלֵּ֑נוּ וְעַתָּ֡ה הָקֵל֩ מֵעֲבֹדַ֨ת אָבִ֜יךָ הַקָּשָׁ֗ה וּמֵעֻלּ֧וֹ הַכָּבֵ֛ד אֲשֶׁר־ נָתַ֥ן עָלֵ֖ינוּ וְנַֽעַבְדֶֽךָּ׃ (יג) וַיֹּ֣אמֶר אֲלֵהֶ֗ם ע֛וֹד שְׁלֹ֥שֶׁת יָמִ֖ים וְשׁ֣וּבוּ אֵלָ֑י וַיֵּ֖לֶךְ הָעָֽם׃ (יד) וַיִּוָּעַ֞ץ הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ הַזְּקֵנִים֙ אֲשֶׁר־ הָי֣וּ עֹֽמְדִ֗ים לִפְנֵי֙ שְׁלֹמֹ֣ה אָבִ֔יו בִּֽהְיֹת֥וֹ חַ֖י לֵאמֹ֑ר אֵ֚יךְ אַתֶּ֣ם נֽוֹעָצִ֔ים לְהָשִׁ֥יב לָֽעָם־ הַזֶּ֖ה דָּבָֽר׃ (טו) וַיְדַבְּר֨וּ אֵלָ֜יו לֵאמֹ֗ר אִם־ תִּֽהְיֶ֨ה לְט֜וֹב לְהָעָ֤ם הַזֶּה֙ וּרְצִיתָ֔ם וְדִבַּרְתָּ֥ אֲלֵהֶ֖ם דְּבָרִ֣ים טוֹבִ֑ים וְהָי֥וּ לְךָ֛ עֲבָדִ֖ים כָּל־ הַיָּמִֽים׃ (טז) וַֽיַּעֲזֹ֛ב אֶת־ עֲצַ֥ת הַזְּקֵנִ֖ים אֲשֶׁ֣ר יְעָצֻ֑הוּ וַיִּוָּעַ֗ץ אֶת־ הַיְלָדִים֙ אֲשֶׁ֣ר גָּדְל֣וּ אִתּ֔וֹ הָעֹמְדִ֖ים לְפָנָֽיו׃ (יז) וַיֹּ֣אמֶר אֲלֵהֶ֗ם מָ֚ה אַתֶּ֣ם נֽוֹעָצִ֔ים וְנָשִׁ֥יב דָּבָ֖ר אֶת־ הָעָ֣ם הַזֶּ֑ה אֲשֶׁ֨ר דִּבְּר֤וּ אֵלַי֙ לֵאמֹ֔ר הָקֵל֙ מִן־ הָעֹ֔ל אֲשֶׁר־ נָתַ֥ן אָבִ֖יךָ עָלֵֽינוּ׃ (יח) וַיְדַבְּר֣וּ אִתּ֗וֹ הַיְלָדִים֙ אֲשֶׁ֨ר גָּדְל֣וּ אִתּוֹ֮ לֵאמֹר֒ כֹּֽה־ תֹאמַ֡ר לָעָם֩ אֲשֶׁר־ דִּבְּר֨וּ אֵלֶ֜יךָ לֵאמֹ֗ר אָבִ֙יךָ֙ הִכְבִּ֣יד אֶת־ עֻלֵּ֔נוּ וְאַתָּ֖ה הָקֵ֣ל מֵעָלֵ֑ינוּ כֹּ֚ה תֹּאמַ֣ר אֲלֵהֶ֔ם קָֽטָנִּ֥י עָבָ֖ה מִמָּתְנֵ֥י אָבִֽי׃ (יט) וְעַתָּ֗ה אָבִי֙ הֶעְמִ֤יס עֲלֵיכֶם֙ עֹ֣ל כָּבֵ֔ד וַאֲנִ֖י אֹסִ֣יף עַֽל־ עֻלְּכֶ֑ם אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֖י בָּֽעֲקְרַבִּֽים׃ (כ) וַיָּבֹ֨א יָרָבְעָ֧ם וְכָל־ הָעָ֛ם אֶל־ רְחַבְעָ֖ם בַּיּ֣וֹם הַשְּׁלִשִׁ֑י כַּאֲשֶׁ֨ר דִּבֶּ֤ר הַמֶּ֙לֶךְ֙ לֵאמֹ֔ר שׁ֥וּבוּ אֵלַ֖י בַּיּ֥וֹם הַשְּׁלִשִֽׁי׃ (כא) וַיַּעֲנֵ֥ם הַמֶּ֖לֶךְ קָשָׁ֑ה וַֽיַּעֲזֹב֙ הַמֶּ֣לֶךְ רְחַבְעָ֔ם אֵ֖ת עֲצַ֥ת הַזְּקֵנִֽים׃ (כב) וַיְדַבֵּ֣ר אֲלֵהֶ֗ם כַּעֲצַ֤ת הַיְלָדִים֙ לֵאמֹ֔ר אַכְבִּיד֙ אֶֽת־ עֻלְּכֶ֔ם וַאֲנִ֖י אֹסִ֣יף עָלָ֑יו אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֖י בָּעֲקְרַבִּֽים׃ (כג) וְלֹֽא־ שָׁמַ֥ע הַמֶּ֖לֶךְ אֶל־ הָעָ֑ם כִּֽי־ הָיְתָ֤ה נְסִבָּה֙ מֵעִ֣ם הָֽאֱלֹהִ֔ים לְמַעַן֩ הָקִ֨ים יְהוָ֜ה אֶת־ דְּבָר֗וֹ אֲשֶׁ֤ר דִּבֶּר֙ בְּיַד֙ אֲחִיָּ֣הוּ הַשִּֽׁלוֹנִ֔י אֶל־ יָרָבְעָ֖ם בֶּן־ נְבָֽט׃ (כד) וְכָל־ יִשְׂרָאֵ֗ל כִּ֠י לֹא־ שָׁמַ֣ע הַמֶּלֶךְ֮ לָהֶם֒ וַיָּשִׁ֣יבוּ הָעָ֣ם אֶת־ הַמֶּ֣לֶךְ ׀ לֵאמֹ֡ר מַה־ לָּנוּ֩ חֵ֨לֶק בְּדָוִ֜יד וְלֹֽא־ נַחֲלָ֣ה בְּבֶן־ יִשַׁ֗י אִ֤ישׁ לְאֹהָלֶ֙יךָ֙ יִשְׂרָאֵ֔ל עַתָּ֕ה רְאֵ֥ה בֵיתְךָ֖ דָּוִ֑יד וַיֵּ֥לֶךְ כָּל־ יִשְׂרָאֵ֖ל לְאֹהָלָֽיו׃ (כה) וּבְנֵ֣י יִשְׂרָאֵ֔ל הַיֹּֽשְׁבִ֖ים בְּעָרֵ֣י יְהוּדָ֑ה וַיִּמְלֹ֥ךְ עֲלֵיהֶ֖ם רְחַבְעָֽם׃ (כו) וַיִּשְׁלַ֞ח הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ הֲדֹרָם֙ אֲשֶׁ֣ר עַל־ הַמַּ֔ס וַיִּרְגְּמוּ־ ב֧וֹ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶ֖בֶן וַיָּמֹ֑ת וְהַמֶּ֣לֶךְ רְחַבְעָ֗ם הִתְאַמֵּץ֙ לַֽעֲל֣וֹת בַּמֶּרְכָּבָ֔ה לָנ֖וּס יְרוּשָׁלִָֽם׃ (כז) וַיִּפְשְׁע֤וּ יִשְׂרָאֵל֙ בְּבֵ֣ית דָּוִ֔יד עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כח) וַיָּבֹ֣א רְחַבְעָם֮ יְרוּשָׁלִַם֒ וַיַּקְהֵל֩ אֶת־ בֵּ֨ית יְהוּדָ֜ה וּבִנְיָמִ֗ן מֵאָ֨ה וּשְׁמוֹנִ֥ים אֶ֛לֶף בָּח֖וּר עֹשֵׂ֣ה מִלְחָמָ֑ה לְהִלָּחֵם֙ עִם־ יִשְׂרָאֵ֔ל לְהָשִׁ֥יב אֶת־ הַמַּמְלָכָ֖ה לִרְחַבְעָֽם׃ (כט) וַיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ שְׁמַֽעְיָ֥הוּ אִישׁ־ הָאֱלֹהִ֖ים לֵאמֹֽר׃ (ל) אֱמֹ֕ר אֶל־ רְחַבְעָ֥ם בֶּן־ שְׁלֹמֹ֖ה מֶ֣לֶךְ יְהוּדָ֑ה וְאֶל֙ כָּל־ יִשְׂרָאֵ֔ל בִּיהוּדָ֥ה וּבִנְיָמִ֖ן לֵאמֹֽר׃ (לא) כֹּ֣ה אָמַ֣ר יְהוָ֡ה לֹא־ תַעֲלוּ֩ וְלֹא־ תִלָּ֨חֲמ֜וּ עִם־ אֲחֵיכֶ֗ם שׁ֚וּבוּ אִ֣ישׁ לְבֵית֔וֹ כִּ֧י מֵֽאִתִּ֛י נִהְיָ֖ה הַדָּבָ֣ר הַזֶּ֑ה וַֽיִּשְׁמְעוּ֙ אֶת־ דִּבְרֵ֣י יְהוָ֔ה וַיָּשֻׁ֖בוּ מִלֶּ֥כֶת אֶל־ יָרָבְעָֽם׃ (לב) וַיֵּ֥שֶׁב רְחַבְעָ֖ם בִּירוּשָׁלִָ֑ם וַיִּ֧בֶן עָרִ֛ים לְמָצ֖וֹר בִּיהוּדָֽה׃ (לג) וַיִּ֧בֶן אֶת־ בֵּֽית־ לֶ֛חֶם וְאֶת־ עֵיטָ֖ם וְאֶת־ תְּקֽוֹעַ׃ (לד) וְאֶת־ בֵּֽית־ צ֥וּר וְאֶת־ שׂוֹכ֖וֹ וְאֶת־ עֲדֻלָּֽם׃ (לה) וְאֶת־ גַּ֥ת וְאֶת־ מָרֵשָׁ֖ה וְאֶת־ זִֽיף׃ (לו) וְאֶת־ אֲדוֹרַ֥יִם וְאֶת־ לָכִ֖ישׁ וְאֶת־ עֲזֵקָֽה׃ (לז) וְאֶת־ צָרְעָה֙ וְאֶת־ אַיָּל֔וֹן וְאֶת־ חֶבְר֔וֹן אֲשֶׁ֥ר בִּיהוּדָ֖ה וּבְבִנְיָמִ֑ן עָרֵ֖י מְצֻרֽוֹת׃ (לח) וַיְחַזֵּ֖ק אֶת־ הַמְּצֻר֑וֹת וַיִּתֵּ֤ן בָּהֶם֙ נְגִידִ֔ים וְאֹצְר֥וֹת מַאֲכָ֖ל וְשֶׁ֥מֶן וָיָֽיִן׃ (לט) וּבְכָל־ עִ֤יר וָעִיר֙ צִנּ֣וֹת וּרְמָחִ֔ים וַֽיְחַזְּקֵ֖ם לְהַרְבֵּ֣ה מְאֹ֑ד וַיְהִי־ ל֖וֹ יְהוּדָ֥ה וּבִנְיָמִֽן׃ (מ) וְהַכֹּהֲנִים֙ וְהַלְוִיִּ֔ם אֲשֶׁ֖ר בְּכָל־ יִשְׂרָאֵ֑ל הִֽתְיַצְּב֥וּ עָלָ֖יו יִרְפּ֖וּ יְדֵיכֶ֑ם כִּ֛י יֵ֥שׁ שָׂכָ֖ר לִפְעֻלַּתְכֶֽם׃ (נ) וְכִשְׁמֹ֨עַ אָסָ֜א הַדְּבָרִ֣ים הָאֵ֗לֶּה וְהַנְּבוּאָה֮ עֹדֵ֣ד הַנָּבִיא֒ הִתְחַזַּ֗ק וַיַּעֲבֵ֤ר הַשִּׁקּוּצִים֙ מִכָּל־ אֶ֤רֶץ יְהוּדָה֙ וּבִנְיָמִ֔ן וּמִן־ הֶ֣עָרִ֔ים אֲשֶׁ֥ר לָכַ֖ד מֵהַ֣ר אֶפְרָ֑יִם וַיְחַדֵּשׁ֙ אֶת־ מִזְבַּ֣ח יְהוָ֔ה אֲשֶׁ֕ר לִפְנֵ֖י אוּלָ֥ם יְהוָֽה׃ (נא) וַיִּקְבֹּ֗ץ אֶת־ כָּל־ יְהוּדָה֙ וּבִנְיָמִ֔ן וְהַגָּרִים֙ עִמָּהֶ֔ם מֵאֶפְרַ֥יִם וּמְנַשֶּׁ֖ה וּמִשִּׁמְע֑וֹן כִּֽי־ נָפְל֨וּ עָלָ֤יו מִיִּשְׂרָאֵל֙ לָרֹ֔ב בִּרְאֹתָ֕ם כִּֽי־ יְהוָ֥ה אֱלֹהָ֖יו עִמּֽוֹ׃ (נב) וַיִּקָּבְצ֥וּ יְרוּשָׁלִַ֖ם בַּחֹ֣דֶשׁ הַשְּׁלִישִׁ֑י לִשְׁנַ֥ת חֲמֵשׁ־ עֶשְׂרֵ֖ה לְמַלְכ֥וּת אָסָֽא׃ (נג) וַיִּזְבְּח֤וּ לַיהוָה֙ בַּיּ֣וֹם הַה֔וּא מִן־ הַשָּׁלָ֖ל הֵבִ֑יאוּ בָּקָר֙ שְׁבַ֣ע מֵא֔וֹת וְצֹ֖אן שִׁבְעַ֥ת אֲלָפִֽים׃ (נד) וַיָּבֹ֣אוּ בַבְּרִ֔ית לִדְר֕וֹשׁ אֶת־ יְהוָ֖ה אֱלֹהֵ֣י אֲבוֹתֵיהֶ֑ם בְּכָל־ לְבָבָ֖ם וּבְכָל־ נַפְשָֽׁם׃ (נה) וְכֹ֨ל אֲשֶׁ֧ר לֹֽא־ יִדְרֹ֛שׁ לַיהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל יוּמָ֑ת לְמִן־ קָטֹן֙ וְעַד־ גָּד֔וֹל לְמֵאִ֖ישׁ וְעַד־ אִשָּֽׁה׃ (נו) וַיִּשָּֽׁבְעוּ֙ לַיהוָ֔ה בְּק֥וֹל גָּד֖וֹל וּבִתְרוּעָ֑ה וּבַחֲצֹצְר֖וֹת וּבְשׁוֹפָרֽוֹת׃
