דברי הימים ב פרק ג לפי החלוקה לסדרים

פרק ג בספר דברי הימים ב (17 פסוקים) נמצא בדברי הימים סדר יב (פסוקים טז–לב).

דברי הימים סדר יב

(א) הִנֵּה֩ אֲנִ֨י בֽוֹנֶה־ בַּ֜יִת לְשֵׁ֣ם ׀ יְהוָ֣ה אֱלֹהָ֗י לְהַקְדִּ֣ישׁ ל֡וֹ לְהַקְטִ֣יר לְפָנָ֣יו קְטֹֽרֶת־ סַמִּים֩ וּמַעֲרֶ֨כֶת תָּמִ֤יד וְעֹלוֹת֙ לַבֹּ֣קֶר וְלָעֶ֔רֶב לַשַּׁבָּתוֹת֙ וְלֶ֣חֳדָשִׁ֔ים וּֽלְמוֹעֲדֵ֖י יְהוָ֣ה אֱלֹהֵ֑ינוּ לְעוֹלָ֖ם זֹ֥את עַל־ יִשְׂרָאֵֽל׃ (ב) וְהַבַּ֛יִת אֲשֶׁר־ אֲנִ֥י בוֹנֶ֖ה גָּד֑וֹל כִּֽי־ גָד֥וֹל אֱלֹהֵ֖ינוּ מִכָּל־ הָאֱלֹהִֽים׃ (ג) וּמִ֤י יַעֲצָר־ כֹּ֙חַ֙ לִבְנֽוֹת־ ל֣וֹ בַ֔יִת כִּ֧י הַשָּׁמַ֛יִם וּשְׁמֵ֥י הַשָּׁמַ֖יִם לֹ֣א יְכַלְכְּלֻ֑הוּ וּמִ֤י אֲנִי֙ אֲשֶׁ֣ר אֶבְנֶה־ לּ֣וֹ בַ֔יִת כִּ֖י אִם־ לְהַקְטִ֥יר לְפָנָֽיו׃ (ד) וְעַתָּ֡ה שְֽׁלַֽח־ לִ֣י אִישׁ־ חָכָ֡ם לַעֲשׂוֹת֩ בַּזָּהָ֨ב וּבַכֶּ֜סֶף וּבַנְּחֹ֣שֶׁת וּבַבַּרְזֶ֗ל וּבָֽאַרְגְּוָן֙ וְכַרְמִ֣יל וּתְכֵ֔לֶת וְיֹדֵ֖עַ לְפַתֵּ֣חַ פִּתּוּחִ֑ים עִם־ הַֽחֲכָמִ֗ים אֲשֶׁ֤ר עִמִּי֙ בִּֽיהוּדָ֣ה וּבִֽירוּשָׁלִַ֔ם אֲשֶׁ֥ר הֵכִ֖ין דָּוִ֥יד אָבִֽי׃ (ה) וּֽשְׁלַֽח־ לִי֩ עֲצֵ֨י אֲרָזִ֜ים בְּרוֹשִׁ֣ים וְאַלְגּוּמִּים֮ מֵֽהַלְּבָנוֹן֒ כִּ֚י אֲנִ֣י יָדַ֔עְתִּי אֲשֶׁ֤ר עֲבָדֶ֙יךָ֙ יֽוֹדְעִ֔ים לִכְר֖וֹת עֲצֵ֣י לְבָנ֑וֹן וְהִנֵּ֥ה עֲבָדַ֖י עִם־ עֲבָדֶֽיךָ׃ (ו) וּלְהָכִ֥ין לִ֛י עֵצִ֖ים לָרֹ֑ב כִּ֥י הַבַּ֛יִת אֲשֶׁר־ אֲנִ֥י בוֹנֶ֖ה גָּד֥וֹל וְהַפְלֵֽא׃ (ז) וְהִנֵּ֣ה לַֽחֹטְבִ֣ים ׀ לְֽכֹרְתֵ֣י ׀ הָעֵצִ֡ים נָתַתִּי֩ חִטִּ֨ים ׀ מַכּ֜וֹת לַעֲבָדֶ֗יךָ כֹּרִים֙ עֶשְׂרִ֣ים אֶ֔לֶף וּשְׂעֹרִ֕ים כֹּרִ֖ים עֶשְׂרִ֣ים אָ֑לֶף וְיַ֗יִן בַּתִּים֙ עֶשְׂרִ֣ים אֶ֔לֶף וְשֶׁ֕מֶן בַּתִּ֖ים עֶשְׂרִ֥ים אָֽלֶף׃ (ח) וַיֹּ֨אמֶר חוּרָ֤ם מֶֽלֶךְ־ צֹר֙ בִּכְתָ֔ב וַיִּשְׁלַ֖ח אֶל־ שְׁלֹמֹ֑ה בְּאַהֲבַ֤ת יְהוָה֙ אֶת־ עַמּ֔וֹ נְתָנְךָ֥ עֲלֵיהֶ֖ם מֶֽלֶךְ׃ (ט) וַיֹּאמֶר֮ חוּרָם֒ בָּר֤וּךְ יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֣ר עָשָׂ֔ה אֶת־ הַשָּׁמַ֖יִם וְאֶת־ הָאָ֑רֶץ אֲשֶׁ֣ר נָתַן֩ לְדָוִ֨יד הַמֶּ֜לֶךְ בֵּ֣ן חָכָ֗ם יוֹדֵ֙עַ֙ שֵׂ֣כֶל וּבִינָ֔ה אֲשֶׁ֤ר יִבְנֶה־ בַּ֙יִת֙ לַיהוָ֔ה וּבַ֖יִת לְמַלְכוּתֽוֹ׃ (י) וְעַתָּ֗ה שָׁלַ֧חְתִּי אִישׁ־ חָכָ֛ם יוֹדֵ֥עַ בִּינָ֖ה לְחוּרָ֥ם אָבִֽי׃ (יא) בֶּן־ אִשָּׁ֞ה מִן־ בְּנ֣וֹת דָּ֗ן וְאָבִ֣יו אִישׁ־ צֹרִ֡י יוֹדֵ֡עַ לַעֲשׂ֣וֹת בַּזָּֽהָב־ וּ֠בַכֶּסֶף בַּנְּחֹ֨שֶׁת בַּבַּרְזֶ֜ל בָּאֲבָנִ֣ים וּבָעֵצִ֗ים בָּאַרְגָּמָ֤ן בַּתְּכֵ֙לֶת֙ וּבַבּ֣וּץ וּבַכַּרְמִ֔יל וּלְפַתֵּ֙חַ֙ כָּל־ פִּתּ֔וּחַ וְלַחְשֹׁ֖ב כָּל־ מַחֲשָׁ֑בֶת אֲשֶׁ֤ר יִנָּֽתֶן־ לוֹ֙ עִם־ חֲכָמֶ֔יךָ וְֽחַכְמֵ֔י אֲדֹנִ֖י דָּוִ֥יד אָבִֽיךָ׃ (יב) וְ֠עַתָּה הַחִטִּ֨ים וְהַשְּׂעֹרִ֜ים הַשֶּׁ֤מֶן וְהַיַּ֙יִן֙ אֲשֶׁ֣ר אָמַ֣ר אֲדֹנִ֔י יִשְׁלַ֖ח לַעֲבָדָֽיו׃ (יג) וַ֠אֲנַחְנוּ נִכְרֹ֨ת עֵצִ֤ים מִן־ הַלְּבָנוֹן֙ כְּכָל־ צָרְכֶּ֔ךָ וּנְבִיאֵ֥ם לְךָ֛ רַפְסֹד֖וֹת עַל־ יָ֣ם יָפ֑וֹ וְאַתָּ֛ה תַּעֲלֶ֥ה אֹתָ֖ם יְרוּשָׁלִָֽם׃ (יד) וַיִּסְפֹּ֣ר שְׁלֹמֹ֗ה כָּל־ הָאֲנָשִׁ֤ים הַגֵּירִים֙ אֲשֶׁר֙ בְּאֶ֣רֶץ יִשְׂרָאֵ֔ל אַחֲרֵ֣י הַסְּפָ֔ר אֲשֶׁ֥ר סְפָרָ֖ם דָּוִ֣יד אָבִ֑יו וַיִּמָּצְא֗וּ מֵאָ֤ה וַחֲמִשִּׁים֙ אֶ֔לֶף וּשְׁלֹ֥שֶׁת אֲלָפִ֖ים וְשֵׁ֥שׁ מֵאֽוֹת׃ (טו) וַיַּ֨עַשׂ מֵהֶ֜ם שִׁבְעִ֥ים אֶ֙לֶף֙ סַבָּ֔ל וּשְׁמֹנִ֥ים אֶ֖לֶף חֹצֵ֣ב בָּהָ֑ר וּשְׁלֹ֤שֶׁת אֲלָפִים֙ וְשֵׁ֣שׁ מֵא֔וֹת מְנַצְּחִ֖ים לְהַעֲבִ֥יד אֶת־ הָעָֽם׃ תחילת פרק ג (טז) וַיָּ֣חֶל שְׁלֹמֹ֗ה לִבְנ֤וֹת אֶת־ בֵּית־ יְהוָה֙ בִּיר֣וּשָׁלִַ֔ם בְּהַר֙ הַמּ֣וֹרִיָּ֔ה אֲשֶׁ֥ר נִרְאָ֖ה לְדָוִ֣יד אָבִ֑יהוּ אֲשֶׁ֤ר הֵכִין֙ בִּמְק֣וֹם דָּוִ֔יד בְּגֹ֖רֶן אָרְנָ֥ן הַיְבוּסִֽי׃ (יז) וַ֠יָּחֶל לִבְנ֞וֹת בַּחֹ֤דֶשׁ הַשֵּׁנִי֙ בַּשֵּׁנִ֔י בִּשְׁנַ֥ת אַרְבַּ֖ע לְמַלְכוּתֽוֹ׃ (יח) וְאֵ֙לֶּה֙ הוּסַ֣ד שְׁלֹמֹ֔ה לִבְנ֖וֹת אֶת־ בֵּ֣ית הָאֱלֹהִ֑ים הָאֹ֡רֶךְ אַמּ֞וֹת בַּמִּדָּ֤ה הָרִֽאשׁוֹנָה֙ אַמּ֣וֹת שִׁשִּׁ֔ים וְרֹ֖חַב אַמּ֥וֹת עֶשְׂרִֽים׃ (יט) וְהָאוּלָ֡ם אֲשֶׁר֩ עַל־ פְּנֵ֨י הָאֹ֜רֶךְ עַל־ פְּנֵ֤י רֹֽחַב־ הַבַּ֙יִת֙ אַמּ֣וֹת עֶשְׂרִ֔ים וְהַגֹּ֖בַהּ מֵאָ֣ה וְעֶשְׂרִ֑ים וַיְצַפֵּ֥הוּ מִפְּנִ֖ימָה זָהָ֥ב טָהֽוֹר׃ (כ) וְאֵ֣ת ׀ הַבַּ֣יִת הַגָּד֗וֹל חִפָּה֙ עֵ֣ץ בְּרוֹשִׁ֔ים וַיְחַפֵּ֖הוּ זָהָ֣ב ט֑וֹב וַיַּ֧עַל עָלָ֛יו תִּמֹרִ֖ים וְשַׁרְשְׁרֽוֹת׃ (כא) וַיְצַ֧ף אֶת־ הַבַּ֛יִת אֶ֥בֶן יְקָרָ֖ה לְתִפְאָ֑רֶת וְהַזָּהָ֖ב זְהַ֥ב פַּרְוָֽיִם׃ (כב) וַיְחַ֨ף אֶת־ הַבַּ֜יִת הַקֹּר֧וֹת הַסִּפִּ֛ים וְקִֽירוֹתָ֥יו וְדַלְתוֹתָ֖יו זָהָ֑ב וּפִתַּ֥ח כְּרוּבִ֖ים עַל־ הַקִּירֽוֹת׃ (כג) וַיַּ֙עַשׂ֙ אֶת־ בֵּֽית־ קֹ֣דֶשׁ הַקֳּדָשִׁ֔ים אָרְכּ֞וֹ עַל־ פְּנֵ֤י רֹֽחַב־ הַבַּ֙יִת֙ אַמּ֣וֹת עֶשְׂרִ֔ים וְרָחְבּ֖וֹ אַמּ֣וֹת עֶשְׂרִ֑ים וַיְחַפֵּ֙הוּ֙ זָהָ֣ב ט֔וֹב לְכִכָּרִ֖ים שֵׁ֥שׁ מֵאֽוֹת׃ (כד) וּמִשְׁקָ֛ל לְמִסְמְר֥וֹת לִשְׁקָלִ֖ים חֲמִשִּׁ֣ים זָהָ֑ב וְהָעֲלִיּ֖וֹת חִפָּ֥ה זָהָֽב׃ (כה) וַיַּ֜עַשׂ בְּבֵֽית־ קֹ֤דֶשׁ הַקֳּדָשִׁים֙ כְּרוּבִ֣ים שְׁנַ֔יִם מַעֲשֵׂ֖ה צַעֲצֻעִ֑ים וַיְצַפּ֥וּ אֹתָ֖ם זָהָֽב׃ (כו) וְכַנְפֵי֙ הַכְּרוּבִ֔ים אָרְכָּ֖ם אַמּ֣וֹת עֶשְׂרִ֑ים כְּנַ֨ף הָאֶחָ֜ד לְאַמּ֣וֹת חָמֵ֗שׁ מַגַּ֙עַת֙ לְקִ֣יר הַבַּ֔יִת וְהַכָּנָ֤ף הָאַחֶ֙רֶת֙ אַמּ֣וֹת חָמֵ֔שׁ מַגִּ֕יעַ לִכְנַ֖ף הַכְּר֥וּב הָאַחֵֽר׃ (כז) וּכְנַ֨ף הַכְּר֤וּב הָאֶחָד֙ אַמּ֣וֹת חָמֵ֔שׁ מַגִּ֖יעַ לְקִ֣יר הַבָּ֑יִת וְהַכָּנָ֤ף הָאַחֶ֙רֶת֙ אַמּ֣וֹת חָמֵ֔שׁ דְּבֵקָ֕ה לִכְנַ֖ף הַכְּר֥וּב הָאַחֵֽר׃ (כח) כַּנְפֵי֙ הַכְּרוּבִ֣ים הָאֵ֔לֶּה פֹּֽרְשִׂ֖ים אַמּ֣וֹת עֶשְׂרִ֑ים וְהֵ֛ם עֹמְדִ֥ים עַל־ רַגְלֵיהֶ֖ם וּפְנֵיהֶ֥ם לַבָּֽיִת׃ (כט) וַיַּ֙עַשׂ֙ אֶת־ הַפָּרֹ֔כֶת תְּכֵ֥לֶת וְאַרְגָּמָ֖ן וְכַרְמִ֣יל וּב֑וּץ וַיַּ֥עַל עָלָ֖יו כְּרוּבִֽים׃ (ל) וַיַּ֜עַשׂ לִפְנֵ֤י הַבַּ֙יִת֙ עַמּוּדִ֣ים שְׁנַ֔יִם אַמּ֕וֹת שְׁלֹשִׁ֥ים וְחָמֵ֖שׁ אֹ֑רֶךְ וְהַצֶּ֥פֶת אֲשֶׁר־ עַל־ רֹאשׁ֖וֹ אַמּ֥וֹת חָמֵֽשׁ׃ (לא) וַיַּ֤עַשׂ שַׁרְשְׁרוֹת֙ בַּדְּבִ֔יר וַיִּתֵּ֖ן עַל־ רֹ֣אשׁ הָעַמֻּדִ֑ים וַיַּ֤עַשׂ רִמּוֹנִים֙ מֵאָ֔ה וַיִּתֵּ֖ן בַּֽשַּׁרְשְׁרֽוֹת׃ (לב) וַיָּ֤קֶם אֶת־ הָֽעַמּוּדִים֙ עַל־ פְּנֵ֣י הַהֵיכָ֔ל אֶחָ֥ד מִיָּמִ֖ין וְאֶחָ֣ד מֵֽהַשְּׂמֹ֑אול וַיִּקְרָ֤א שֵׁם־ (הימיני) הַיְמָנִי֙ יָכִ֔ין וְשֵׁ֥ם הַשְּׂמָאלִ֖י בֹּֽעַז׃ סוף פרק ג (לג) וַיַּ֙עַשׂ֙ מִזְבַּ֣ח נְחֹ֔שֶׁת עֶשְׂרִ֤ים אַמָּה֙ אָרְכּ֔וֹ וְעֶשְׂרִ֥ים אַמָּ֖ה רָחְבּ֑וֹ וְעֶ֥שֶׂר אַמּ֖וֹת קוֹמָתֽוֹ׃ (לד) וַיַּ֥עַשׂ אֶת־ הַיָּ֖ם מוּצָ֑ק עֶ֣שֶׂר בָּֽ֠אַמָּה מִשְּׂפָת֨וֹ אֶל־ שְׂפָת֜וֹ עָג֣וֹל ׀ סָבִ֗יב וְחָמֵ֤שׁ בָּֽאַמָּה֙ קֽוֹמָת֔וֹ וְקָו֙ שְׁלֹשִׁ֣ים בָּֽאַמָּ֔ה יָסֹ֥ב אֹת֖וֹ סָבִֽיב׃ (לה) וּדְמ֣וּת בְּקָרִים֩ תַּ֨חַת ל֜וֹ סָבִ֤יב ׀ סָבִיב֙ סוֹבְבִ֣ים אֹת֔וֹ עֶ֚שֶׂר בָּֽאַמָּ֔ה מַקִּיפִ֥ים אֶת־ הַיָּ֖ם סָבִ֑יב שְׁנַ֤יִם טוּרִים֙ הַבָּקָ֔ר יְצוּקִ֖ים בְּמֻֽצַקְתּֽוֹ׃ (לו) עוֹמֵ֞ד עַל־ שְׁנֵ֧ים עָשָׂ֣ר בָּקָ֗ר שְׁלֹשָׁ֣ה פֹנִ֣ים ׀ צָפ֡וֹנָה וּשְׁלוֹשָׁה֩ פֹנִ֨ים ׀ יָ֜מָּה וּשְׁלֹשָׁ֣ה ׀ פֹּנִ֣ים נֶ֗גְבָּה וּשְׁלֹשָׁה֙ פֹּנִ֣ים מִזְרָ֔חָה וְהַיָּ֥ם עֲלֵיהֶ֖ם מִלְמָ֑עְלָה וְכָל־ אֲחֹרֵיהֶ֖ם בָּֽיְתָה׃ (לז) וְעָבְי֣וֹ טֶ֔פַח וּשְׂפָתוֹ֙ כְּמַעֲשֵׂ֣ה שְׂפַת־ כּ֔וֹס פֶּ֖רַח שֽׁוֹשַׁנָּ֑ה מַחֲזִ֣יק בַּתִּ֔ים שְׁלֹ֥שֶׁת אֲלָפִ֖ים יָכִֽיל׃ (לח) וַיַּ֣עַשׂ כִּיּוֹרִים֮ עֲשָׂרָה֒ וַ֠יִּתֵּן חֲמִשָּׁ֨ה מִיָּמִ֜ין וַחֲמִשָּׁ֤ה מִשְּׂמֹאול֙ לְרָחְצָ֣ה בָהֶ֔ם אֶת־ מַעֲשֵׂ֥ה הָעוֹלָ֖ה יָדִ֣יחוּ בָ֑ם וְהַיָּ֕ם לְרָחְצָ֥ה לַכֹּהֲנִ֖ים בּֽוֹ׃ (לט) וַ֠יַּעַשׂ אֶת־ מְנֹר֧וֹת הַזָּהָ֛ב עֶ֖שֶׂר כְּמִשְׁפָּטָ֑ם וַיִּתֵּן֙ בַּֽהֵיכָ֔ל חָמֵ֥שׁ מִיָּמִ֖ין וְחָמֵ֥שׁ מִשְּׂמֹֽאול׃ (מ) וַיַּ֣עַשׂ שֻׁלְחָנוֹת֮ עֲשָׂרָה֒ וַיַּנַּח֙ בַּֽהֵיכָ֔ל חֲמִשָּׁ֥ה מִיָּמִ֖ין וַחֲמִשָּׁ֣ה מִשְּׂמֹ֑אול וַיַּ֛עַשׂ מִזְרְקֵ֥י זָהָ֖ב מֵאָֽה׃ (מא) וַיַּ֙עַשׂ֙ חֲצַ֣ר הַכֹּהֲנִ֔ים וְהָעֲזָרָ֖ה הַגְּדוֹלָ֑ה וּדְלָת֧וֹת לָעֲזָרָ֛ה וְדַלְתוֹתֵיהֶ֖ם צִפָּ֥ה נְחֹֽשֶׁת׃ (מב) וְאֶת־ הַיָּ֗ם נָתַ֞ן מִכֶּ֧תֶף הַיְמָנִ֛ית קֵ֖דְמָה מִמּ֥וּל נֶֽגְבָּה׃ (מג) וַיַּ֣עַשׂ חוּרָ֔ם אֶת־ הַ֨סִּיר֔וֹת וְאֶת־ הַיָּעִ֖ים וְאֶת־ הַמִּזְרָק֑וֹת וַיְכַ֣ל (חירם) חוּרָ֗ם לַעֲשׂוֹת֙ אֶת־ הַמְּלָאכָ֔ה אֲשֶׁ֥ר עָשָׂ֛ה לַמֶּ֥לֶךְ שְׁלֹמֹ֖ה בְּבֵ֥ית הָאֱלֹהִֽים׃ (מד) עַמּוּדִ֣ים שְׁנַ֔יִם וְהַגֻּלּ֧וֹת וְהַכֹּתָר֛וֹת עַל־ רֹ֥אשׁ הָעַמּוּדִ֖ים שְׁתָּ֑יִם וְהַשְּׂבָכ֣וֹת שְׁתַּ֔יִם לְכַסּ֗וֹת אֶת־ שְׁתֵּי֙ גֻּלּ֣וֹת הַכֹּֽתָר֔וֹת אֲשֶׁ֖ר עַל־ רֹ֥אשׁ הָֽעַמּוּדִֽים׃ (מה) וְאֶת־ הָֽרִמּוֹנִ֛ים אַרְבַּ֥ע מֵא֖וֹת לִשְׁתֵּ֣י הַשְּׂבָכ֑וֹת שְׁנַ֨יִם טוּרִ֤ים רִמּוֹנִים֙ לַשְּׂבָכָ֣ה הָאֶחָ֔ת לְכַסּ֗וֹת אֶת־ שְׁתֵּי֙ גֻּלּ֣וֹת הַכֹּֽתָר֔וֹת אֲשֶׁ֖ר עַל־ פְּנֵ֥י הָעַמּוּדִֽים׃ (מו) וְאֶת־ הַמְּכֹנ֖וֹת עָשָׂ֑ה וְאֶת־ הַכִּיֹּר֥וֹת עָשָׂ֖ה עַל־ הַמְּכֹנֽוֹת׃ (מז) אֶת־ הַיָּ֖ם אֶחָ֑ד וְאֶת־ הַבָּקָ֥ר שְׁנֵים־ עָשָׂ֖ר תַּחְתָּֽיו׃ (מח) וְאֶת־ הַ֠סִּירוֹת וְאֶת־ הַיָּעִ֤ים וְאֶת־ הַמִּזְלָגוֹת֙ וְאֶת־ כָּל־ כְּלֵיהֶ֔ם עָשָׂ֞ה חוּרָ֥ם אָבִ֛יו לַמֶּ֥לֶךְ שְׁלֹמֹ֖ה לְבֵ֣ית יְהוָ֑ה נְחֹ֖שֶׁת מָרֽוּק׃ (מט) בְּכִכַּ֤ר הַיַּרְדֵּן֙ יְצָקָ֣ם הַמֶּ֔לֶךְ בַּעֲבִ֖י הָאֲדָמָ֑ה בֵּ֥ין סֻכּ֖וֹת וּבֵ֥ין צְרֵדָֽתָה׃ (נ) וַיַּ֧עַשׂ שְׁלֹמֹ֛ה כָּל־ הַכֵּלִ֥ים הָאֵ֖לֶּה לָרֹ֣ב מְאֹ֑ד כִּ֛י לֹ֥א נֶחְקַ֖ר מִשְׁקַ֥ל הַנְּחֹֽשֶׁת׃ (נא) וַיַּ֣עַשׂ שְׁלֹמֹ֔ה אֵ֚ת כָּל־ הַכֵּלִ֔ים אֲשֶׁ֖ר בֵּ֣ית הָאֱלֹהִ֑ים וְאֵת֙ מִזְבַּ֣ח הַזָּהָ֔ב וְאֶת־ הַשֻּׁלְחָנ֔וֹת וַעֲלֵיהֶ֖ם לֶ֥חֶם הַפָּנִֽים׃ (נב) וְאֶת־ הַמְּנֹר֞וֹת וְנֵרֹתֵיהֶ֗ם לְבַעֲרָ֧ם כַּמִּשְׁפָּ֛ט לִפְנֵ֥י הַדְּבִ֖יר זָהָ֥ב סָגֽוּר׃ (נג) וְהַפֶּ֧רַח וְהַנֵּר֛וֹת וְהַמֶּלְקַחַ֖יִם זָהָ֑ב ה֖וּא מִכְל֥וֹת זָהָֽב׃ (נד) וְהַֽמְזַמְּר֧וֹת וְהַמִּזְרָק֛וֹת וְהַכַּפּ֥וֹת וְהַמַּחְתּ֖וֹת זָהָ֣ב סָג֑וּר וּפֶ֣תַח הַ֠בַּיִת דַּלְתוֹתָ֨יו הַפְּנִימִיּ֜וֹת לְקֹ֣דֶשׁ הַקֳּדָשִׁ֗ים וְדַלְתֵ֥י הַבַּ֛יִת לַהֵיכָ֖ל זָהָֽב׃ (נה) וַתִּשְׁלַם֙ כָּל־ הַמְּלָאכָ֔ה אֲשֶׁר־ עָשָׂ֥ה שְׁלֹמֹ֖ה לְבֵ֣ית יְהוָ֑ה וַיָּבֵ֨א שְׁלֹמֹ֜ה אֶת־ קָדְשֵׁ֣י ׀ דָּוִ֣יד אָבִ֗יו וְאֶת־ הַכֶּ֤סֶף וְאֶת־ הַזָּהָב֙ וְאֶת־ כָּל־ הַכֵּלִ֔ים נָתַ֕ן בְּאֹצְר֖וֹת בֵּ֥ית הָאֱלֹהִֽים׃ (נו) אָז֩ יַקְהֵ֨יל שְׁלֹמֹ֜ה אֶת־ זִקְנֵ֣י יִשְׂרָאֵ֗ל וְאֶת־ כָּל־ רָאשֵׁ֨י הַמַּטּ֜וֹת נְשִׂיאֵ֧י הָאָב֛וֹת לִבְנֵ֥י יִשְׂרָאֵ֖ל אֶל־ יְרוּשָׁלִָ֑ם לְֽהַעֲל֞וֹת אֶת־ אֲר֧וֹן בְּרִית־ יְהוָ֛ה מֵעִ֥יר דָּוִ֖יד הִ֥יא צִיּֽוֹן׃ (נז) וַיִּקָּהֲל֧וּ אֶל־ הַמֶּ֛לֶךְ כָּל־ אִ֥ישׁ יִשְׂרָאֵ֖ל בֶּחָ֑ג ה֖וּא הַחֹ֥דֶשׁ הַשְּׁבִעִֽי׃ (נח) וַיָּבֹ֕אוּ כֹּ֖ל זִקְנֵ֣י יִשְׂרָאֵ֑ל וַיִּשְׂא֥וּ הַלְוִיִּ֖ם אֶת־ הָאָרֽוֹן׃ (נט) וַיַּעֲל֤וּ אֶת־ הָאָרוֹן֙ וְאֶת־ אֹ֣הֶל מוֹעֵ֔ד וְאֶת־ כָּל־ כְּלֵ֥י הַקֹּ֖דֶשׁ אֲשֶׁ֣ר בָּאֹ֑הֶל הֶעֱל֣וּ אֹתָ֔ם הַכֹּהֲנִ֖ים הַלְוִיִּֽם׃ (ס) וְהַמֶּ֣לֶךְ שְׁלֹמֹ֗ה וְכָל־ עֲדַ֧ת יִשְׂרָאֵ֛ל הַנּוֹעָדִ֥ים עָלָ֖יו לִפְנֵ֣י הָאָר֑וֹן מְזַבְּחִים֙ צֹ֣אן וּבָקָ֔ר אֲשֶׁ֧ר לֹֽא־ יִסָּפְר֛וּ וְלֹ֥א יִמָּנ֖וּ מֵרֹֽב׃ (סא) וַיָּבִ֣יאוּ הַ֠כֹּהֲנִים אֶת־ אֲר֨וֹן בְּרִית־ יְהוָ֧ה אֶל־ מְקוֹמ֛וֹ אֶל־ דְּבִ֥יר הַבַּ֖יִת אֶל־ קֹ֣דֶשׁ הַקְּדָשִׁ֑ים אֶל־ תַּ֖חַת כַּנְפֵ֥י הַכְּרוּבִֽים׃ (סב) וַיִּהְי֤וּ הַכְּרוּבִים֙ פֹּרְשִׂ֣ים כְּנָפַ֔יִם עַל־ מְק֖וֹם הָאָר֑וֹן וַיְכַסּ֧וּ הַכְּרוּבִ֛ים עַל־ הָאָר֥וֹן וְעַל־ בַּדָּ֖יו מִלְמָֽעְלָה׃ (סג) וַֽיַּאֲרִיכוּ֮ הַבַּדִּים֒ וַיֵּרָאוּ֩ רָאשֵׁ֨י הַבַּדִּ֤ים מִן־ הָאָרוֹן֙ עַל־ פְּנֵ֣י הַדְּבִ֔יר וְלֹ֥א יֵרָא֖וּ הַח֑וּצָה וַֽיְהִי־ שָׁ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (סד) אֵ֚ין בָּֽאָר֔וֹן רַ֚ק שְׁנֵ֣י הַלֻּח֔וֹת אֲשֶׁר־ נָתַ֥ן מֹשֶׁ֖ה בְּחֹרֵ֑ב אֲשֶׁ֨ר כָּרַ֤ת יְהוָה֙ עִם־ בְּנֵ֣י יִשְׂרָאֵ֔ל בְּצֵאתָ֖ם מִמִּצְרָֽיִם׃ (סה) וַיְהִ֕י בְּצֵ֥את הַכֹּהֲנִ֖ים מִן־ הַקֹּ֑דֶשׁ כִּ֠י כָּל־ הַכֹּהֲנִ֤ים הַֽנִּמְצְאִים֙ הִתְקַדָּ֔שׁוּ אֵ֖ין לִשְׁמ֥וֹר לְמַחְלְקֽוֹת׃ (סו) וְהַלְוִיִּ֣ם הַמְשֹׁרֲרִ֣ים לְכֻלָּ֡ם לְאָסָ֡ף לְהֵימָ֣ן לִֽ֠ידֻתוּן וְלִבְנֵיהֶ֨ם וְלַאֲחֵיהֶ֜ם מְלֻבָּשִׁ֣ים בּ֗וּץ בִּמְצִלְתַּ֙יִם֙ וּבִנְבָלִ֣ים וְכִנֹּר֔וֹת עֹמְדִ֖ים מִזְרָ֣ח לַמִּזְבֵּ֑חַ וְעִמָּהֶ֤ם כֹּֽהֲנִים֙ לְמֵאָ֣ה וְעֶשְׂרִ֔ים (מחצררים) מַחְצְרִ֖ים בַּחֲצֹֽצְרֽוֹת׃ (סז) וַיְהִ֣י כְ֠אֶחָד (למחצצרים) לַמְחַצְּרִ֨ים וְלַמְשֹֽׁרֲרִ֜ים לְהַשְׁמִ֣יעַ קוֹל־ אֶחָ֗ד לְהַלֵּ֣ל וּלְהֹדוֹת֮ לַיהוָה֒ וּכְהָרִ֣ים ק֠וֹל בַּחֲצֹצְר֨וֹת וּבִמְצִלְתַּ֜יִם וּבִכְלֵ֣י הַשִּׁ֗יר וּבְהַלֵּ֤ל לַיהוָה֙ כִּ֣י ט֔וֹב כִּ֥י לְעוֹלָ֖ם חַסְדּ֑וֹ וְהַבַּ֛יִת מָלֵ֥א עָנָ֖ן בֵּ֥ית יְהוָֽה׃
