עובדיה פרק א לפי החלוקה לסדרים

פרק א בספר עובדיה (21 פסוקים) נמצא בתרי עשר סדר ח (פסוקים לג–נב) ובתרי עשר סדר ט (פסוקים א–א).

תרי עשר סדר ח

(א) וַיִּקָּחֵ֣נִי יְהוָ֔ה מֵאַחֲרֵ֖י הַצֹּ֑אן וַיֹּ֤אמֶר אֵלַי֙ יְהוָ֔ה לֵ֥ךְ הִנָּבֵ֖א אֶל־ עַמִּ֥י יִשְׂרָאֵֽל׃ (ב) וְעַתָּ֖ה שְׁמַ֣ע דְּבַר־ יְהוָ֑ה אַתָּ֣ה אֹמֵ֗ר לֹ֤א תִנָּבֵא֙ עַל־ יִשְׂרָאֵ֔ל וְלֹ֥א תַטִּ֖יף עַל־ בֵּ֥ית יִשְׂחָֽק׃ (ג) לָכֵ֞ן כֹּה־ אָמַ֣ר יְהוָ֗ה אִשְׁתְּךָ֞ בָּעִ֤יר תִּזְנֶה֙ וּבָנֶ֤יךָ וּבְנֹתֶ֙יךָ֙ בַּחֶ֣רֶב יִפֹּ֔לוּ וְאַדְמָתְךָ֖ בַּחֶ֣בֶל תְּחֻלָּ֑ק וְאַתָּ֗ה עַל־ אֲדָמָ֤ה טְמֵאָה֙ תָּמ֔וּת וְיִ֨שְׂרָאֵ֔ל גָּלֹ֥ה יִגְלֶ֖ה מֵעַ֥ל אַדְמָתֽוֹ׃ (ד) כֹּ֥ה הִרְאַ֖נִי אֲדֹנָ֣י יְהוִ֑ה וְהִנֵּ֖ה כְּל֥וּב קָֽיִץ׃ (ה) וַיֹּ֗אמֶר מָֽה־ אַתָּ֤ה רֹאֶה֙ עָמ֔וֹס וָאֹמַ֖ר כְּל֣וּב קָ֑יִץ וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י בָּ֤א הַקֵּץ֙ אֶל־ עַמִּ֣י יִשְׂרָאֵ֔ל לֹא־ אוֹסִ֥יף ע֖וֹד עֲב֥וֹר לֽוֹ׃ (ו) וְהֵילִ֜ילוּ שִׁיר֤וֹת הֵיכָל֙ בַּיּ֣וֹם הַה֔וּא נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה רַ֣ב הַפֶּ֔גֶר בְּכָל־ מָק֖וֹם הִשְׁלִ֥יךְ הָֽס׃ (ז) שִׁמְעוּ־ זֹ֕את הַשֹּׁאֲפִ֖ים אֶבְי֑וֹן וְלַשְׁבִּ֖ית (ענוי)־ עֲנִיֵּי־ אָֽרֶץ׃ (ח) לֵאמֹ֗ר מָתַ֞י יַעֲבֹ֤ר הַחֹ֙דֶשׁ֙ וְנַשְׁבִּ֣ירָה שֶּׁ֔בֶר וְהַשַּׁבָּ֖ת וְנִפְתְּחָה־ בָּ֑ר לְהַקְטִ֤ין אֵיפָה֙ וּלְהַגְדִּ֣יל שֶׁ֔קֶל וּלְעַוֵּ֖ת מֹאזְנֵ֥י מִרְמָֽה׃ (ט) לִקְנ֤וֹת בַּכֶּ֙סֶף֙ דַּלִּ֔ים וְאֶבְי֖וֹן בַּעֲב֣וּר נַעֲלָ֑יִם וּמַפַּ֥ל בַּ֖ר נַשְׁבִּֽיר׃ (י) נִשְׁבַּ֥ע יְהוָ֖ה בִּגְא֣וֹן יַעֲקֹ֑ב אִם־ אֶשְׁכַּ֥ח לָנֶ֖צַח כָּל־ מַעֲשֵׂיהֶֽם׃ (יא) הַ֤עַל זֹאת֙ לֹֽא־ תִרְגַּ֣ז הָאָ֔רֶץ וְאָבַ֖ל כָּל־ יוֹשֵׁ֣ב בָּ֑הּ וְעָלְתָ֤ה כָאֹר֙ כֻּלָּ֔הּ וְנִגְרְשָׁ֥ה (ונשקה) וְנִשְׁקְעָ֖ה כִּיא֥וֹר מִצְרָֽיִם׃ (יב) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה וְהֵבֵאתִ֥י הַשֶּׁ֖מֶשׁ בַּֽצָּהֳרָ֑יִם וְהַחֲשַׁכְתִּ֥י לָאָ֖רֶץ בְּי֥וֹם אֽוֹר׃ (יג) וְהָפַכְתִּ֨י חַגֵּיכֶ֜ם לְאֵ֗בֶל וְכָל־ שִֽׁירֵיכֶם֙ לְקִינָ֔ה וְהַעֲלֵיתִ֤י עַל־ כָּל־ מָתְנַ֙יִם֙ שָׂ֔ק וְעַל־ כָּל־ רֹ֖אשׁ קָרְחָ֑ה וְשַׂמְתִּ֙יהָ֙ כְּאֵ֣בֶל יָחִ֔יד וְאַחֲרִיתָ֖הּ כְּי֥וֹם מָֽר׃ (יד) הִנֵּ֣ה ׀ יָמִ֣ים בָּאִ֗ים נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה וְהִשְׁלַחְתִּ֥י רָעָ֖ב בָּאָ֑רֶץ לֹֽא־ רָעָ֤ב לַלֶּ֙חֶם֙ וְלֹֽא־ צָמָ֣א לַמַּ֔יִם כִּ֣י אִם־ לִשְׁמֹ֔עַ אֵ֖ת דִּבְרֵ֥י יְהוָֽה׃ (טו) וְנָעוּ֙ מִיָּ֣ם עַד־ יָ֔ם וּמִצָּפ֖וֹן וְעַד־ מִזְרָ֑ח יְשֽׁוֹטְט֛וּ לְבַקֵּ֥שׁ אֶת־ דְּבַר־ יְהוָ֖ה וְלֹ֥א יִמְצָֽאוּ׃ (טז) בַּיּ֨וֹם הַה֜וּא תִּ֠תְעַלַּפְנָה הַבְּתוּלֹ֧ת הַיָּפ֛וֹת וְהַבַּחוּרִ֖ים בַּצָּמָֽא׃ (יז) הַנִּשְׁבָּעִים֙ בְּאַשְׁמַ֣ת שֹֽׁמְר֔וֹן וְאָמְר֗וּ חֵ֤י אֱלֹהֶ֙יךָ֙ דָּ֔ן וְחֵ֖י דֶּ֣רֶךְ בְּאֵֽר־ שָׁ֑בַע וְנָפְל֖וּ וְלֹא־ יָק֥וּמוּ עֽוֹד׃ (יח) רָאִ֨יתִי אֶת־ אֲדֹנָ֜י נִצָּ֣ב עַֽל־ הַמִּזְבֵּ֗חַ וַיֹּאמֶר֩ הַ֨ךְ הַכַּפְתּ֜וֹר וְיִרְעֲשׁ֣וּ הַסִּפִּ֗ים וּבְצַ֙עַם֙ בְּרֹ֣אשׁ כֻּלָּ֔ם וְאַחֲרִיתָ֖ם בַּחֶ֣רֶב אֶהֱרֹ֑ג לֹֽא־ יָנ֤וּס לָהֶם֙ נָ֔ס וְלֹֽא־ יִמָּלֵ֥ט לָהֶ֖ם פָּלִֽיט׃ (יט) אִם־ יַחְתְּר֣וּ בִשְׁא֔וֹל מִשָּׁ֖ם יָדִ֣י תִקָּחֵ֑ם וְאִֽם־ יַעֲלוּ֙ הַשָּׁמַ֔יִם מִשָּׁ֖ם אוֹרִידֵֽם׃ (כ) וְאִם־ יֵחָֽבְאוּ֙ בְּרֹ֣אשׁ הַכַּרְמֶ֔ל מִשָּׁ֥ם אֲחַפֵּ֖שׂ וּלְקַחְתִּ֑ים וְאִם־ יִסָּ֨תְר֜וּ מִנֶּ֤גֶד עֵינַי֙ בְּקַרְקַ֣ע הַיָּ֔ם מִשָּׁ֛ם אֲצַוֶּ֥ה אֶת־ הַנָּחָ֖שׁ וּנְשָׁכָֽם׃ (כא) וְאִם־ יֵלְכ֤וּ בַשְּׁבִי֙ לִפְנֵ֣י אֹֽיבֵיהֶ֔ם מִשָּׁ֛ם אֲצַוֶּ֥ה אֶת־ הַחֶ֖רֶב וַהֲרָגָ֑תַם וְשַׂמְתִּ֨י עֵינִ֧י עֲלֵיהֶ֛ם לְרָעָ֖ה וְלֹ֥א לְטוֹבָֽה׃ (כב) וַאדֹנָ֨י יְהוִ֜ה הַצְּבָא֗וֹת הַנּוֹגֵ֤עַ בָּאָ֙רֶץ֙ וַתָּמ֔וֹג וְאָבְל֖וּ כָּל־ י֣וֹשְׁבֵי בָ֑הּ וְעָלְתָ֤ה כַיְאֹר֙ כֻּלָּ֔הּ וְשָׁקְעָ֖ה כִּיאֹ֥ר מִצְרָֽיִם (כג) הַבּוֹנֶ֤ה בַשָּׁמַ֙יִם֙ (מעלותו) מַעֲלוֹתָ֔יו וַאֲגֻדָּת֖וֹ עַל־ אֶ֣רֶץ יְסָדָ֑הּ הַקֹּרֵ֣א לְמֵֽי־ הַיָּ֗ם וַֽיִּשְׁפְּכֵ֛ם עַל־ פְּנֵ֥י הָאָ֖רֶץ יְהוָ֥ה שְׁמֽוֹ׃ (כד) הֲל֣וֹא כִבְנֵי֩ כֻשִׁיִּ֨ים אַתֶּ֥ם לִ֛י בְּנֵ֥י יִשְׂרָאֵ֖ל נְאֻם־ יְהוָ֑ה הֲל֣וֹא אֶת־ יִשְׂרָאֵ֗ל הֶעֱלֵ֙יתִי֙ מֵאֶ֣רֶץ מִצְרַ֔יִם וּפְלִשְׁתִּיִּ֥ים מִכַּפְתּ֖וֹר וַאֲרָ֥ם מִקִּֽיר׃ (כה) הִנֵּ֞ה עֵינֵ֣י ׀ אֲדֹנָ֣י יְהוִ֗ה בַּמַּמְלָכָה֙ הַֽחַטָּאָ֔ה וְהִשְׁמַדְתִּ֣י אֹתָ֔הּ מֵעַ֖ל פְּנֵ֣י הָאֲדָמָ֑ה אֶ֗פֶס כִּ֠י לֹ֣א הַשְׁמֵ֥יד אַשְׁמִ֛יד אֶת־ בֵּ֥ית יַעֲקֹ֖ב נְאֻם־ יְהוָֽה׃ (כו) כִּֽי־ הִנֵּ֤ה אָֽנֹכִי֙ מְצַוֶּ֔ה וַהֲנִע֥וֹתִי בְכָֽל־ הַגּוֹיִ֖ם אֶת־ בֵּ֣ית יִשְׂרָאֵ֑ל כַּאֲשֶׁ֤ר יִנּ֙וֹעַ֙ בַּכְּבָרָ֔ה וְלֹֽא־ יִפּ֥וֹל צְר֖וֹר אָֽרֶץ׃ (כז) בַּחֶ֣רֶב יָמ֔וּתוּ כֹּ֖ל חַטָּאֵ֣י עַמִּ֑י הָאֹמְרִ֗ים לֹֽא־ תַגִּ֧ישׁ וְתַקְדִּ֛ים בַּעֲדֵ֖ינוּ הָרָעָֽה׃ (כח) בַּיּ֣וֹם הַה֔וּא אָקִ֛ים אֶת־ סֻכַּ֥ת דָּוִ֖יד הַנֹּפֶ֑לֶת וְגָדַרְתִּ֣י אֶת־ פִּרְצֵיהֶ֗ן וַהֲרִֽסֹתָיו֙ אָקִ֔ים וּבְנִיתִ֖יהָ כִּימֵ֥י עוֹלָֽם׃ (כט) לְמַ֨עַן יִֽירְשׁ֜וּ אֶת־ שְׁאֵרִ֤ית אֱדוֹם֙ וְכָל־ הַגּוֹיִ֔ם אֲשֶׁר־ נִקְרָ֥א שְׁמִ֖י עֲלֵיהֶ֑ם נְאֻם־ יְהוָ֖ה עֹ֥שֶׂה זֹּֽאת׃ (ל) הִנֵּ֨ה יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וְנִגַּ֤שׁ חוֹרֵשׁ֙ בַּקֹּצֵ֔ר וְדֹרֵ֥ךְ עֲנָבִ֖ים בְּמֹשֵׁ֣ךְ הַזָּ֑רַע וְהִטִּ֤יפוּ הֶֽהָרִים֙ עָסִ֔יס וְכָל־ הַגְּבָע֖וֹת תִּתְמוֹגַֽגְנָה׃ (לא) וְשַׁבְתִּי֮ אֶת־ שְׁב֣וּת עַמִּ֣י יִשְׂרָאֵל֒ וּבָנ֞וּ עָרִ֤ים נְשַׁמּוֹת֙ וְיָשָׁ֔בוּ וְנָטְע֣וּ כְרָמִ֔ים וְשָׁת֖וּ אֶת־ יֵינָ֑ם וְעָשׂ֣וּ גַנּ֔וֹת וְאָכְל֖וּ אֶת־ פְּרִיהֶֽם׃ (לב) וּנְטַעְתִּ֖ים עַל־ אַדְמָתָ֑ם וְלֹ֨א יִנָּתְשׁ֜וּ ע֗וֹד מֵעַ֤ל אַדְמָתָם֙ אֲשֶׁ֣ר נָתַ֣תִּי לָהֶ֔ם אָמַ֖ר יְהוָ֥ה אֱלֹהֶֽיךָ׃ תחילת פרק א (לג) חֲז֖וֹן עֹֽבַדְיָ֑ה כֹּֽה־ אָמַר֩ אֲדֹנָ֨י יְהוִ֜ה לֶאֱד֗וֹם שְׁמוּעָ֨ה שָׁמַ֜עְנוּ מֵאֵ֤ת יְהוָה֙ וְצִיר֙ בַּגּוֹיִ֣ם שֻׁלָּ֔ח ק֛וּמוּ וְנָק֥וּמָה עָלֶיהָ לַמִּלְחָמָֽה׃ (לד) הִנֵּ֥ה קָטֹ֛ן נְתַתִּ֖יךָ בַּגּוֹיִ֑ם בָּז֥וּי אַתָּ֖ה מְאֹֽד׃ (לה) זְד֤וֹן לִבְּךָ֙ הִשִּׁיאֶ֔ךָ שֹׁכְנִ֥י בְחַגְוֵי־ סֶּ֖לַע מְר֣וֹם שִׁבְתּ֑וֹ אֹמֵ֣ר בְּלִבּ֔וֹ מִ֥י יוֹרִדֵ֖נִי אָֽרֶץ׃ (לו) אִם־ תַּגְבִּ֣יהַּ כַּנֶּ֔שֶׁר וְאִם־ בֵּ֥ין כּֽוֹכָבִ֖ים שִׂ֣ים קִנֶּ֑ךָ מִשָּׁ֥ם אוֹרִֽידְךָ֖ נְאֻם־ יְהוָֽה׃ (לז) אִם־ גַּנָּבִ֤ים בָּאֽוּ־ לְךָ֙ אִם־ שׁ֣וֹדְדֵי לַ֔יְלָה אֵ֣יךְ נִדְמֵ֔יתָה הֲל֥וֹא יִגְנְב֖וּ דַּיָּ֑ם אִם־ בֹּֽצְרִים֙ בָּ֣אוּ לָ֔ךְ הֲל֖וֹא יַשְׁאִ֥ירוּ עֹלֵלֽוֹת׃ (לח) אֵ֚יךְ נֶחְפְּשׂ֣וּ עֵשָׂ֔ו נִבְע֖וּ מַצְפֻּנָֽיו׃ (לט) עַֽד־ הַגְּב֣וּל שִׁלְּח֗וּךָ כֹּ֚ל אַנְשֵׁ֣י בְרִיתֶ֔ךָ הִשִּׁיא֛וּךָ יָכְל֥וּ לְךָ֖ אַנְשֵׁ֣י שְׁלֹמֶ֑ךָ לַחְמְךָ֗ יָשִׂ֤ימוּ מָזוֹר֙ עַד־ נֶ֔פֶשׁ תְּה֖וֹם יְסֹבְבֵ֑נִי ס֖וּף חָב֥וּשׁ לְרֹאשִֽׁי׃ (כד) לְקִצְבֵ֤י הָרִים֙ יָרַ֔דְתִּי הָאָ֛רֶץ בְּרִחֶ֥יהָ בַעֲדִ֖י לְעוֹלָ֑ם וַתַּ֧עַל מִשַּׁ֛חַת חַיַּ֖י יְהוָ֥ה אֱלֹהָֽי׃ (כה) בְּהִתְעַטֵּ֤ף עָלַי֙ נַפְשִׁ֔י אֶת־ יְהוָ֖ה זָכָ֑רְתִּי וַתָּב֤וֹא אֵלֶ֙יךָ֙ תְּפִלָּתִ֔י אֶל־ הֵיכַ֖ל קָדְשֶֽׁךָ׃ (כו) מְשַׁמְּרִ֖ים הַבְלֵי־ שָׁ֑וְא חַסְדָּ֖ם יַעֲזֹֽבוּ׃ (כז) וַאֲנִ֗י בְּק֤וֹל תּוֹדָה֙ אֶזְבְּחָה־ לָּ֔ךְ אֲשֶׁ֥ר נָדַ֖רְתִּי אֲשַׁלֵּ֑מָה יְשׁוּעָ֖תָה לַיהוָֽה׃ (כח) וַיֹּ֥אמֶר יְהוָ֖ה לַדָּ֑ג וַיָּקֵ֥א אֶת־ יוֹנָ֖ה אֶל־ הַיַּבָּשָֽׁה׃ (כט) וַיְהִ֧י דְבַר־ יְהוָ֛ה אֶל־ יוֹנָ֖ה שֵׁנִ֥ית לֵאמֹֽר׃ (ל) ק֛וּם לֵ֥ךְ אֶל־ נִֽינְוֵ֖ה הָעִ֣יר הַגְּדוֹלָ֑ה וִּקְרָ֤א אֵלֶ֙יהָ֙ אֶת־ הַקְּרִיאָ֔ה אֲשֶׁ֥ר אָנֹכִ֖י דֹּבֵ֥ר אֵלֶֽיךָ׃ (לא) וַיָּ֣קָם יוֹנָ֗ה וַיֵּ֛לֶךְ אֶל־ נִֽינְוֶ֖ה כִּדְבַ֣ר יְהוָ֑ה וְנִֽינְוֵ֗ה הָיְתָ֤ה עִיר־ גְּדוֹלָה֙ לֵֽאלֹהִ֔ים מַהֲלַ֖ךְ שְׁלֹ֥שֶׁת יָמִֽים׃ (לב) וַיָּ֤חֶל יוֹנָה֙ לָב֣וֹא בָעִ֔יר מַהֲלַ֖ךְ י֣וֹם אֶחָ֑ד וַיִּקְרָא֙ וַיֹּאמַ֔ר ע֚וֹד אַרְבָּעִ֣ים י֔וֹם וְנִֽינְוֵ֖ה נֶהְפָּֽכֶת׃ (לג) וַֽיַּאֲמִ֛ינוּ אַנְשֵׁ֥י נִֽינְוֵ֖ה בֵּֽאלֹהִ֑ים וַיִּקְרְאוּ־ צוֹם֙ וַיִּלְבְּשׁ֣וּ שַׂקִּ֔ים מִגְּדוֹלָ֖ם וְעַד־ קְטַנָּֽם׃ (לד) וַיִּגַּ֤ע הַדָּבָר֙ אֶל־ מֶ֣לֶך נִֽינְוֵ֔ה וַיָּ֙קָם֙ מִכִּסְא֔וֹ וַיַּעֲבֵ֥ר אַדַּרְתּ֖וֹ מֵֽעָלָ֑יו וַיְכַ֣ס שַׂ֔ק וַיֵּ֖שֶׁב עַל־ הָאֵֽפֶר׃ (לה) וַיַּזְעֵ֗ק וַיֹּ֙אמֶר֙ בְּנִֽינְוֵ֔ה מִטַּ֧עַם הַמֶּ֛לֶךְ וּגְדֹלָ֖יו לֵאמֹ֑ר הָאָדָ֨ם וְהַבְּהֵמָ֜ה הַבָּקָ֣ר וְהַצֹּ֗אן אַֽל־ יִטְעֲמוּ֙ מְא֔וּמָה אַ֨ל־ יִרְע֔וּ וּמַ֖יִם אַל־ יִשְׁתּֽוּ׃ (לו) וְיִתְכַּסּ֣וּ שַׂקִּ֗ים הָֽאָדָם֙ וְהַבְּהֵמָ֔ה וְיִקְרְא֥וּ אֶל־ אֱלֹהִ֖ים בְּחָזְקָ֑ה וְיָשֻׁ֗בוּ אִ֚ישׁ מִדַּרְכּ֣וֹ הָֽרָעָ֔ה וּמִן־ הֶחָמָ֖ס אֲשֶׁ֥ר בְּכַפֵּיהֶֽם׃ (לז) מִֽי־ יוֹדֵ֣עַ יָשׁ֔וּב וְנִחַ֖ם הָאֱלֹהִ֑ים וְשָׁ֛ב מֵחֲר֥וֹן אַפּ֖וֹ וְלֹ֥א נֹאבֵֽד׃ (לח) וַיַּ֤רְא הָֽאֱלֹהִים֙ אֶֽת־ מַ֣עֲשֵׂיהֶ֔ם כִּי־ שָׁ֖בוּ מִדַּרְכָּ֣ם הָרָעָ֑ה וַיִּנָּ֣חֶם הָאֱלֹהִ֗ים עַל־ הָרָעָ֛ה אֲשֶׁר־ דִּבֶּ֥ר לַעֲשׂוֹת־ לָהֶ֖ם וְלֹ֥א עָשָֽׂה׃ (לט) וַיֵּ֥רַע אֶל־ יוֹנָ֖ה רָעָ֣ה גְדוֹלָ֑ה וַיִּ֖חַר לֽוֹ׃ (מ) וַיִּתְפַּלֵּ֨ל אֶל־ יְהוָ֜ה וַיֹּאמַ֗ר אָנָּ֤ה יְהוָה֙ הֲלוֹא־ זֶ֣ה דְבָרִ֗י עַד־ הֱיוֹתִי֙ עַל־ אַדְמָתִ֔י עַל־ כֵּ֥ן קִדַּ֖מְתִּי לִבְרֹ֣חַ תַּרְשִׁ֑ישָׁה כִּ֣י יָדַ֗עְתִּי כִּ֤י אַתָּה֙ אֵֽל־ חַנּ֣וּן וְרַח֔וּם אֶ֤רֶךְ אַפַּ֙יִם֙ וְרַב־ חֶ֔סֶד וְנִחָ֖ם עַל־ הָרָעָֽה׃ (מא) וְעַתָּ֣ה יְהוָ֔ה קַח־ נָ֥א אֶת־ נַפְשִׁ֖י מִמֶּ֑נִּי כִּ֛י ט֥וֹב מוֹתִ֖י מֵחַיָּֽי׃ (מב) וַיֹּ֣אמֶר יְהוָ֔ה הַהֵיטֵ֖ב חָ֥רָה לָֽךְ׃ (מג) וַיֵּצֵ֤א יוֹנָה֙ מִן־ הָעִ֔יר וַיֵּ֖שֶׁב מִקֶּ֣דֶם לָעִ֑יר וַיַּעַשׂ֩ ל֨וֹ שָׁ֜ם סֻכָּ֗ה וַיֵּ֤שֶׁב תַּחְתֶּ֙יהָ֙ בַּצֵּ֔ל עַ֚ד אֲשֶׁ֣ר יִרְאֶ֔ה מַה־ יִּהְיֶ֖ה בָּעִֽיר׃ (מד) וַיְמַ֣ן יְהוָֽה־ אֱ֠לֹהִים קִיקָי֞וֹן וַיַּ֣עַל ׀ מֵעַ֣ל לְיוֹנָ֗ה לִֽהְי֥וֹת צֵל֙ עַל־ רֹאשׁ֔וֹ לְהַצִּ֥יל ל֖וֹ מֵרָֽעָת֑וֹ וַיִּשְׂמַ֥ח יוֹנָ֛ה עַל־ הַקִּֽיקָי֖וֹן שִׂמְחָ֥ה גְדוֹלָֽה׃ (מה) וַיְמַ֤ן הָֽאֱלֹהִים֙ תּוֹלַ֔עַת בַּעֲל֥וֹת הַשַּׁ֖חַר לַֽמָּחֳרָ֑ת וַתַּ֥ךְ אֶת־ הַקִּֽיקָי֖וֹן וַיִּיבָֽשׁ׃ (מו) וַיְהִ֣י ׀ כִּזְרֹ֣חַ הַשֶּׁ֗מֶשׁ וַיְמַ֨ן אֱלֹהִ֜ים ר֤וּחַ קָדִים֙ חֲרִישִׁ֔ית וַתַּ֥ךְ הַשֶּׁ֛מֶשׁ עַל־ רֹ֥אשׁ יוֹנָ֖ה וַיִּתְעַלָּ֑ף וַיִּשְׁאַ֤ל אֶת־ נַפְשׁוֹ֙ לָמ֔וּת וַיֹּ֕אמֶר ט֥וֹב מוֹתִ֖י מֵחַיָּֽי׃ (מז) וַיֹּ֤אמֶר אֱלֹהִים֙ אֶל־ יוֹנָ֔ה הַהֵיטֵ֥ב חָרָֽה־ לְךָ֖ עַל־ הַקִּֽיקָי֑וֹן וַיֹּ֕אמֶר הֵיטֵ֥ב חָֽרָה־ לִ֖י עַד־ מָֽוֶת׃ (מח) וַיֹּ֣אמֶר יְהוָ֔ה אַתָּ֥ה חַ֙סְתָּ֙ עַל־ הַקִּ֣יקָי֔וֹן אֲשֶׁ֛ר לֹא־ עָמַ֥לְתָּ בּ֖וֹ וְלֹ֣א גִדַּלְתּ֑וֹ שֶׁבִּן־ לַ֥יְלָה הָיָ֖ה וּבִן־ לַ֥יְלָה אָבָֽד׃ (מט) וַֽאֲנִי֙ לֹ֣א אָח֔וּס עַל־ נִינְוֵ֖ה הָעִ֣יר הַגְּדוֹלָ֑ה אֲשֶׁ֣ר יֶשׁ־ בָּ֡הּ הַרְבֵּה֩ מִֽשְׁתֵּים־ עֶשְׂרֵ֨ה רִבּ֜וֹ אָדָ֗ם אֲשֶׁ֤ר לֹֽא־ יָדַע֙ בֵּין־ יְמִינ֣וֹ לִשְׂמֹאל֔וֹ וּבְהֵמָ֖ה רַבָּֽה׃
