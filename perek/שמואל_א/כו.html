שמואל א פרק כו לפי החלוקה לסדרים

פרק כו בספר שמואל א (25 פסוקים) נמצא בשמואל סדר יז (פסוקים יג–לו) ובשמואל סדר יח (פסוקים א–א).

שמואל סדר יז

(א) וּבָר֥וּךְ טַעְמֵ֖ךְ וּבְרוּכָ֣ה אָ֑תְּ אֲשֶׁ֨ר כְּלִתִ֜נִי הַיּ֤וֹם הַזֶּה֙ מִבּ֣וֹא בְדָמִ֔ים וְהֹשֵׁ֥עַ יָדִ֖י לִֽי׃ (ב) וְאוּלָ֗ם חַי־ יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֣ר מְנָעַ֔נִי מֵהָרַ֖ע אֹתָ֑ךְ כִּ֣י ׀ לוּלֵ֣י מִהַ֗רְתְּ (ותבאתי) וַתָּבֹאת֙ לִקְרָאתִ֔י כִּ֣י אִם־ נוֹתַ֧ר לְנָבָ֛ל עַד־ א֥וֹר הַבֹּ֖קֶר מַשְׁתִּ֥ין בְּקִֽיר׃ (ג) וַיִּקַּ֤ח דָּוִד֙ מִיָּדָ֔הּ אֵ֥ת אֲשֶׁר־ הֵבִ֖יאָה ל֑וֹ וְלָ֣הּ אָמַ֗ר עֲלִ֤י לְשָׁלוֹם֙ לְבֵיתֵ֔ךְ רְאִי֙ שָׁמַ֣עְתִּי בְקוֹלֵ֔ךְ וָאֶשָּׂ֖א פָּנָֽיִךְ׃ (ד) וַתָּבֹ֣א אֲבִיגַ֣יִל ׀ אֶל־ נָבָ֡ל וְהִנֵּה־ לוֹ֩ מִשְׁתֶּ֨ה בְּבֵית֜וֹ כְּמִשְׁתֵּ֣ה הַמֶּ֗לֶךְ וְלֵ֤ב נָבָל֙ ט֣וֹב עָלָ֔יו וְה֥וּא שִׁכֹּ֖ר עַד־ מְאֹ֑ד וְלֹֽא־ הִגִּ֣ידָה לּ֗וֹ דָּבָ֥ר קָטֹ֛ן וְגָד֖וֹל עַד־ א֥וֹר הַבֹּֽקֶר׃ (ה) וַיְהִ֣י בַבֹּ֗קֶר בְּצֵ֤את הַיַּ֙יִן֙ מִנָּבָ֔ל וַתַּגֶּד־ ל֣וֹ אִשְׁתּ֔וֹ אֶת־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיָּ֤מָת לִבּוֹ֙ בְּקִרְבּ֔וֹ וְה֖וּא הָיָ֥ה לְאָֽבֶן׃ (ו) וַיְהִ֖י כַּעֲשֶׂ֣רֶת הַיָּמִ֑ים וַיִּגֹּ֧ף יְהוָ֛ה אֶת־ נָבָ֖ל וַיָּמֹֽת׃ (ז) וַיִּשְׁמַ֣ע דָּוִד֮ כִּ֣י מֵ֣ת נָבָל֒ וַיֹּ֡אמֶר בָּר֣וּךְ יְהוָ֡ה אֲשֶׁ֣ר רָב֩ אֶת־ רִ֨יב חֶרְפָּתִ֜י מִיַּ֣ד נָבָ֗ל וְאֶת־ עַבְדּוֹ֙ חָשַׂ֣ךְ מֵֽרָעָ֔ה וְאֵת֙ רָעַ֣ת נָבָ֔ל הֵשִׁ֥יב יְהוָ֖ה בְּרֹאשׁ֑וֹ וַיִּשְׁלַ֤ח דָּוִד֙ וַיְדַבֵּ֣ר בַּאֲבִיגַ֔יִל לְקַחְתָּ֥הּ ל֖וֹ לְאִשָּֽׁה׃ (ח) וַיָּבֹ֜אוּ עַבְדֵ֥י דָוִ֛ד אֶל־ אֲבִיגַ֖יִל הַכַּרְמֶ֑לָה וַיְדַבְּר֤וּ אֵלֶ֙יהָ֙ לֵאמֹ֔ר דָּוִד֙ שְׁלָחָ֣נוּ אֵלַ֔יִךְ לְקַחְתֵּ֥ךְ ל֖וֹ לְאִשָּֽׁה׃ (ט) וַתָּ֕קָם וַתִּשְׁתַּ֥חוּ אַפַּ֖יִם אָ֑רְצָה וַתֹּ֗אמֶר הִנֵּ֤ה אֲמָֽתְךָ֙ לְשִׁפְחָ֔ה לִרְחֹ֕ץ רַגְלֵ֖י עַבְדֵ֥י אֲדֹנִֽי׃ (י) וַתְּמַהֵ֞ר וַתָּ֣קָם אֲבִיגַ֗יִל וַתִּרְכַּב֙ עַֽל־ הַחֲמ֔וֹר וְחָמֵשׁ֙ נַעֲרֹתֶ֔יהָ הַהֹלְכ֖וֹת לְרַגְלָ֑הּ וַתֵּ֗לֶךְ אַֽחֲרֵי֙ מַלְאֲכֵ֣י דָוִ֔ד וַתְּהִי־ ל֖וֹ לְאִשָּֽׁה׃ (יא) וְאֶת־ אֲחִינֹ֛עַם לָקַ֥ח דָּוִ֖ד מִֽיִּזְרְעֶ֑אל וַתִּהְיֶ֛יןָ גַּֽם־ שְׁתֵּיהֶ֥ן ל֖וֹ לְנָשִֽׁים׃ (יב) וְשָׁא֗וּל נָתַ֛ן אֶת־ מִיכַ֥ל בִּתּ֖וֹ אֵ֣שֶׁת דָּוִ֑ד לְפַלְטִ֥י בֶן־ לַ֖יִשׁ אֲשֶׁ֥ר מִגַּלִּֽים׃ תחילת פרק כו (יג) וַיָּבֹ֤אוּ הַזִּפִים֙ אֶל־ שָׁא֔וּל הַגִּבְעָ֖תָה לֵאמֹ֑ר הֲל֨וֹא דָוִ֤ד מִסְתַּתֵּר֙ בְּגִבְעַ֣ת הַחֲכִילָ֔ה עַ֖ל פְּנֵ֥י הַיְשִׁימֹֽן׃ (יד) וַיָּ֣קָם שָׁא֗וּל וַיֵּ֙רֶד֙ אֶל־ מִדְבַּר־ זִ֔יף וְאִתּ֛וֹ שְׁלֹֽשֶׁת־ אֲלָפִ֥ים אִ֖ישׁ בְּחוּרֵ֣י יִשְׂרָאֵ֑ל לְבַקֵּ֥שׁ אֶת־ דָּוִ֖ד בְּמִדְבַּר־ זִֽיף׃ (טו) וַיִּ֨חַן שָׁא֜וּל בְּגִבְעַ֣ת הַחֲכִילָ֗ה אֲשֶׁ֛ר עַל־ פְּנֵ֥י הַיְשִׁימֹ֖ן עַל־ הַדָּ֑רֶךְ וְדָוִד֙ יֹשֵׁ֣ב בַּמִּדְבָּ֔ר וַיַּ֕רְא כִּ֣י בָ֥א שָׁא֛וּל אַחֲרָ֖יו הַמִּדְבָּֽרָה׃ (טז) וַיִּשְׁלַ֥ח דָּוִ֖ד מְרַגְּלִ֑ים וַיֵּ֕דַע כִּֽי־ בָ֥א שָׁא֖וּל אֶל־ נָכֽוֹן׃ (יז) וַיָּ֣קָם דָּוִ֗ד וַיָּבֹא֮ אֶֽל־ הַמָּקוֹם֮ אֲשֶׁ֣ר חָנָה־ שָׁ֣ם שָׁאוּל֒ וַיַּ֣רְא דָּוִ֗ד אֶת־ הַמָּקוֹם֙ אֲשֶׁ֣ר שָֽׁכַב־ שָׁ֣ם שָׁא֔וּל וְאַבְנֵ֥ר בֶּן־ נֵ֖ר שַׂר־ צְבָא֑וֹ וְשָׁאוּל֙ שֹׁכֵ֣ב בַּמַּעְגָּ֔ל וְהָעָ֖ם חֹנִ֥ים (סביבתו) סְבִיבֹתָֽיו׃ (יח) וַיַּ֨עַן דָּוִ֜ד וַיֹּ֣אמֶר ׀ אֶל־ אֲחִימֶ֣לֶךְ הַחִתִּ֗י וְאֶל־ אֲבִישַׁ֨י בֶּן־ צְרוּיָ֜ה אֲחִ֤י יוֹאָב֙ לֵאמֹ֔ר מִֽי־ יֵרֵ֥ד אִתִּ֛י אֶל־ שָׁא֖וּל אֶל־ הַֽמַּחֲנֶ֑ה וַיֹּ֣אמֶר אֲבִישַׁ֔י אֲנִ֖י אֵרֵ֥ד עִמָּֽךְ׃ (יט) וַיָּבֹא֩ דָוִ֨ד וַאֲבִישַׁ֥י ׀ אֶל־ הָעָם֮ לַיְלָה֒ וְהִנֵּ֣ה שָׁא֗וּל שֹׁכֵ֤ב יָשֵׁן֙ בַּמַּעְגָּ֔ל וַחֲנִית֥וֹ מְעוּכָֽה־ בָאָ֖רֶץ (מראשתו) מְרַאֲשֹׁתָ֑יו וְאַבְנֵ֣ר וְהָעָ֔ם שֹׁכְבִ֖ים (סביבתו) סְבִיבֹתָֽיו׃ (כ) וַיֹּ֤אמֶר אֲבִישַׁי֙ אֶל־ דָּוִ֔ד סִגַּ֨ר אֱלֹהִ֥ים הַיּ֛וֹם אֶת־ אוֹיִבְךָ֖ בְּיָדֶ֑ךָ וְעַתָּה֩ אַכֶּ֨נּוּ נָ֜א בַּחֲנִ֤ית וּבָאָ֙רֶץ֙ פַּ֣עַם אַחַ֔ת וְלֹ֥א אֶשְׁנֶ֖ה לֽוֹ׃ (כא) וַיֹּ֧אמֶר דָּוִ֛ד אֶל־ אֲבִישַׁ֖י אַל־ תַּשְׁחִיתֵ֑הוּ כִּ֠י מִ֣י שָׁלַ֥ח יָד֛וֹ בִּמְשִׁ֥יחַ יְהוָ֖ה וְנִקָּֽה׃ (כב) וַיֹּ֤אמֶר דָּוִד֙ חַי־ יְהוָ֔ה כִּ֥י אִם־ יְהוָ֖ה יִגָּפֶ֑נּוּ אֽוֹ־ יוֹמ֤וֹ יָבוֹא֙ וָמֵ֔ת א֧וֹ בַמִּלְחָמָ֛ה יֵרֵ֖ד וְנִסְפָּֽה׃ (כג) חָלִ֤ילָה לִּי֙ מֵֽיהוָ֔ה מִשְּׁלֹ֥חַ יָדִ֖י בִּמְשִׁ֣יחַ יְהוָ֑ה וְ֠עַתָּה קַח־ נָ֨א אֶֽת־ הַחֲנִ֜ית אֲשֶׁ֧ר (מראשתו) מְרַאֲשֹׁתָ֛יו וְאֶת־ צַפַּ֥חַת הַמַּ֖יִם וְנֵ֥לֲכָה לָּֽנוּ׃ (כד) וַיִּקַּח֩ דָּוִ֨ד אֶֽת־ הַחֲנִ֜ית וְאֶת־ צַפַּ֤חַת הַמַּ֙יִם֙ מֵרַאֲשֹׁתֵ֣י שָׁא֔וּל וַיֵּלְכ֖וּ לָהֶ֑ם וְאֵ֣ין רֹאֶה֩ וְאֵ֨ין יוֹדֵ֜עַ וְאֵ֣ין מֵקִ֗יץ כִּ֤י כֻלָּם֙ יְשֵׁנִ֔ים כִּ֚י תַּרְדֵּמַ֣ת יְהוָ֔ה נָפְלָ֖ה עֲלֵיהֶֽם׃ (כה) וַיַּעֲבֹ֤ר דָּוִד֙ הָעֵ֔בֶר וַיַּעֲמֹ֥ד עַל־ רֹאשׁ־ הָהָ֖ר מֵֽרָחֹ֑ק רַ֥ב הַמָּק֖וֹם בֵּינֵיהֶֽם׃ (כו) וַיִּקְרָ֨א דָוִ֜ד אֶל־ הָעָ֗ם וְאֶל־ אַבְנֵ֤ר בֶּן־ נֵר֙ לֵאמֹ֔ר הֲל֥וֹא תַעֲנֶ֖ה אַבְנֵ֑ר וַיַּ֤עַן אַבְנֵר֙ וַיֹּ֔אמֶר מִ֥י אַתָּ֖ה קָרָ֥אתָ אֶל־ הַמֶּֽלֶךְ׃ (כז) וַיֹּאמֶר֩ דָּוִ֨ד אֶל־ אַבְנֵ֜ר הֲלוֹא־ אִ֣ישׁ אַתָּ֗ה וּמִ֤י כָמ֙וֹךָ֙ בְּיִשְׂרָאֵ֔ל וְלָ֙מָּה֙ לֹ֣א שָׁמַ֔רְתָּ אֶל־ אֲדֹנֶ֖יךָ הַמֶּ֑לֶךְ כִּי־ בָא֙ אַחַ֣ד הָעָ֔ם לְהַשְׁחִ֖ית אֶת־ הַמֶּ֥לֶךְ אֲדֹנֶֽיךָ׃ (כח) לֹא־ ט֞וֹב הַדָּבָ֣ר הַזֶּה֮ אֲשֶׁ֣ר עָשִׂיתָ֒ חַי־ יְהוָ֗ה כִּ֤י בְנֵי־ מָ֙וֶת֙ אַתֶּ֔ם אֲשֶׁ֧ר לֹֽא־ שְׁמַרְתֶּ֛ם עַל־ אֲדֹנֵיכֶ֖ם עַל־ מְשִׁ֣יחַ יְהוָ֑ה וְעַתָּ֣ה ׀ רְאֵ֗ה אֵֽי־ חֲנִ֥ית הַמֶּ֛לֶךְ וְאֶת־ צַפַּ֥חַת הַמַּ֖יִם אֲשֶׁ֥ר (מראשתו) מְרַאֲשֹׁתָֽיו׃ (כט) וַיַּכֵּ֤ר שָׁאוּל֙ אֶת־ ק֣וֹל דָּוִ֔ד וַיֹּ֕אמֶר הֲקוֹלְךָ֥ זֶ֖ה בְּנִ֣י דָוִ֑ד וַיֹּ֣אמֶר דָּוִ֔ד קוֹלִ֖י אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (ל) וַיֹּ֕אמֶר לָ֥מָּה זֶּ֛ה אֲדֹנִ֥י רֹדֵ֖ף אַחֲרֵ֣י עַבְדּ֑וֹ כִּ֚י מֶ֣ה עָשִׂ֔יתִי וּמַה־ בְּיָדִ֖י רָעָֽה׃ (לא) וְעַתָּ֗ה יִֽשְׁמַֽע־ נָא֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ אֵ֖ת דִּבְרֵ֣י עַבְדּ֑וֹ אִם־ יְהוָ֞ה הֱסִֽיתְךָ֥ בִי֙ יָרַ֣ח מִנְחָ֔ה וְאִ֣ם ׀ בְּנֵ֣י הָאָדָ֗ם אֲרוּרִ֥ים הֵם֙ לִפְנֵ֣י יְהוָ֔ה כִּֽי־ גֵרְשׁ֣וּנִי הַיּ֗וֹם מֵהִסְתַּפֵּ֜חַ בְּנַחֲלַ֤ת יְהוָה֙ לֵאמֹ֔ר לֵ֥ךְ עֲבֹ֖ד אֱלֹהִ֥ים אֲחֵרִֽים׃ (לב) וְעַתָּ֗ה אַל־ יִפֹּ֤ל דָּֽמִי֙ אַ֔רְצָה מִנֶּ֖גֶד פְּנֵ֣י יְהוָ֑ה כִּֽי־ יָצָ֞א מֶ֣לֶךְ יִשְׂרָאֵ֗ל לְבַקֵּשׁ֙ אֶת־ פַּרְעֹ֣שׁ אֶחָ֔ד כַּאֲשֶׁ֛ר יִרְדֹּ֥ף הַקֹּרֵ֖א בֶּהָרִֽים׃ (לג) וַיֹּאמֶר֩ שָׁא֨וּל חָטָ֜אתִי שׁ֣וּב בְּנִֽי־ דָוִ֗ד כִּ֠י לֹֽא־ אָרַ֤ע לְךָ֙ ע֔וֹד תַּ֠חַת אֲשֶׁ֨ר יָקְרָ֥ה נַפְשִׁ֛י בְּעֵינֶ֖יךָ הַיּ֣וֹם הַזֶּ֑ה הִנֵּ֥ה הִסְכַּ֛לְתִּי אֹתִ֑י וַיֹּ֣אמֶר שָׁ֠אוּל צַר־ לִ֨י מְאֹ֜ד וּפְלִשְׁתִּ֣ים ׀ נִלְחָמִ֣ים בִּ֗י וֵֽאלֹהִ֞ים סָ֤ר מֵֽעָלַי֙ וְלֹֽא־ עָנָ֣נִי ע֗וֹד גַּ֤ם בְּיַֽד־ הַנְּבִיאִם֙ גַּם־ בַּ֣חֲלֹמ֔וֹת וָאֶקְרָאֶ֣ה לְךָ֔ לְהוֹדִיעֵ֖נִי מָ֥ה אֶעֱשֶֽׂה׃ (כט) וַיֹּ֣אמֶר שְׁמוּאֵ֔ל וְלָ֖מָּה תִּשְׁאָלֵ֑נִי וַיהוָ֛ה סָ֥ר מֵעָלֶ֖יךָ וַיְהִ֥י עָרֶֽךָ׃ (ל) וַיַּ֤עַשׂ יְהוָה֙ ל֔וֹ כַּאֲשֶׁ֖ר דִּבֶּ֣ר בְּיָדִ֑י וַיִּקְרַ֨ע יְהוָ֤ה אֶת־ הַמַּמְלָכָה֙ מִיָּדֶ֔ךָ וַֽיִּתְּנָ֖הּ לְרֵעֲךָ֥ לְדָוִֽד׃ (לא) כַּאֲשֶׁ֤ר לֹֽא־ שָׁמַ֙עְתָּ֙ בְּק֣וֹל יְהוָ֔ה וְלֹֽא־ עָשִׂ֥יתָ חֲרוֹן־ אַפּ֖וֹ בַּעֲמָלֵ֑ק עַל־ כֵּן֙ הַדָּבָ֣ר הַזֶּ֔ה עָשָֽׂה־ לְךָ֥ יְהוָ֖ה הַיּ֥וֹם הַזֶּֽה׃ (לב) וְיִתֵּ֣ן יְ֠הוָה גַּ֣ם אֶת־ יִשְׂרָאֵ֤ל עִמְּךָ֙ בְּיַד־ פְּלִשְׁתִּ֔ים וּמָחָ֕ר אַתָּ֥ה וּבָנֶ֖יךָ עִמִּ֑י גַּ֚ם אֶת־ מַחֲנֵ֣ה יִשְׂרָאֵ֔ל יִתֵּ֥ן יְהוָ֖ה בְּיַד־ פְּלִשְׁתִּֽים׃ (לג) וַיְמַהֵ֣ר שָׁא֗וּל וַיִּפֹּ֤ל מְלֹא־ קֽוֹמָתוֹ֙ אַ֔רְצָה וַיִּרָ֥א מְאֹ֖ד מִדִּבְרֵ֣י שְׁמוּאֵ֑ל גַּם־ כֹּ֙חַ֙ לֹא־ הָ֣יָה ב֔וֹ כִּ֣י לֹ֤א אָכַל֙ לֶ֔חֶם כָּל־ הַיּ֖וֹם וְכָל־ הַלָּֽיְלָה׃ (לד) וַתָּב֤וֹא הָֽאִשָּׁה֙ אֶל־ שָׁא֔וּל וַתֵּ֖רֶא כִּי־ נִבְהַ֣ל מְאֹ֑ד וַתֹּ֣אמֶר אֵלָ֗יו הִנֵּ֨ה שָׁמְעָ֤ה שִׁפְחָֽתְךָ֙ בְּקוֹלֶ֔ךָ וָאָשִׂ֤ים נַפְשִׁי֙ בְּכַפִּ֔י וָֽאֶשְׁמַע֙ אֶת־ דְּבָרֶ֔יךָ אֲשֶׁ֥ר דִּבַּ֖רְתָּ אֵלָֽי׃ (לה) וְעַתָּ֗ה שְׁמַֽע־ נָ֤א גַם־ אַתָּה֙ בְּק֣וֹל שִׁפְחָתֶ֔ךָ וְאָשִׂ֧מָה לְפָנֶ֛יךָ פַּת־ לֶ֖חֶם וֶאֱכ֑וֹל וִיהִ֤י בְךָ֙ כֹּ֔חַ כִּ֥י תֵלֵ֖ךְ בַּדָּֽרֶךְ׃ (לו) וַיְמָאֵ֗ן וַיֹּ֙אמֶר֙ לֹ֣א אֹכַ֔ל וַיִּפְרְצוּ־ ב֤וֹ עֲבָדָיו֙ וְגַם־ הָ֣אִשָּׁ֔ה וַיִּשְׁמַ֖ע לְקֹלָ֑ם וַיָּ֙קָם֙ מֵֽהָאָ֔רֶץ וַיֵּ֖שֶׁב אֶל־ הַמִּטָּֽה׃
