שמואל א פרק טו לפי החלוקה לסדרים

פרק טו בספר שמואל א (35 פסוקים) נמצא בשמואל סדר ח (פסוקים לא–מו) ובשמואל סדר ט (פסוקים א–יט).

שמואל סדר ח

(א) וַיּ֧וֹשַׁע יְהוָ֛ה בַּיּ֥וֹם הַה֖וּא אֶת־ יִשְׂרָאֵ֑ל וְהַ֨מִּלְחָמָ֔ה עָבְרָ֖ה אֶת־ בֵּ֥ית אָֽוֶן׃ (ב) וְאִֽישׁ־ יִשְׂרָאֵ֥ל נִגַּ֖שׂ בַּיּ֣וֹם הַה֑וּא וַיֹּאֶל֩ שָׁא֨וּל אֶת־ הָעָ֜ם לֵאמֹ֗ר אָר֣וּר הָ֠אִישׁ אֲשֶׁר־ יֹ֨אכַל לֶ֜חֶם עַד־ הָעֶ֗רֶב וְנִקַּמְתִּי֙ מֵאֹ֣יְבַ֔י וְלֹֽא טָעַ֥ם כָּל־ הָעָ֖ם לָֽחֶם׃ (ג) וְכָל־ הָאָ֖רֶץ בָּ֣אוּ בַיָּ֑עַר וַיְהִ֥י דְבַ֖שׁ עַל־ פְּנֵ֥י הַשָּׂדֶֽה׃ (ד) וַיָּבֹ֤א הָעָם֙ אֶל־ הַיַּ֔עַר וְהִנֵּ֖ה הֵ֣לֶךְ דְּבָ֑שׁ וְאֵין־ מַשִּׂ֤יג יָדוֹ֙ אֶל־ פִּ֔יו כִּֽי־ יָרֵ֥א הָעָ֖ם אֶת־ הַשְּׁבֻעָֽה׃ (ה) וְיוֹנָתָ֣ן לֹֽא־ שָׁמַ֗ע בְּהַשְׁבִּ֣יעַ אָבִיו֮ אֶת־ הָעָם֒ וַיִּשְׁלַ֗ח אֶת־ קְצֵ֤ה הַמַּטֶּה֙ אֲשֶׁ֣ר בְּיָד֔וֹ וַיִּטְבֹּ֥ל אוֹתָ֖הּ בְּיַעְרַ֣ת הַדְּבָ֑שׁ וַיָּ֤שֶׁב יָדוֹ֙ אֶל־ פִּ֔יו (ותראנה) וַתָּאֹ֖רְנָה עֵינָֽיו׃ (ו) וַיַּעַן֩ אִ֨ישׁ מֵֽהָעָ֜ם וַיֹּ֗אמֶר הַשְׁבֵּעַ֩ הִשְׁבִּ֨יעַ אָבִ֤יךָ אֶת־ הָעָם֙ לֵאמֹ֔ר אָר֥וּר הָאִ֛ישׁ אֲשֶׁר־ יֹ֥אכַל לֶ֖חֶם הַיּ֑וֹם וַיָּ֖עַף הָעָֽם׃ (ז) וַיֹּ֙אמֶר֙ יֽוֹנָתָ֔ן עָכַ֥ר אָבִ֖י אֶת־ הָאָ֑רֶץ רְאוּ־ נָא֙ כִּֽי־ אֹ֣רוּ עֵינַ֔י כִּ֣י טָעַ֔מְתִּי מְעַ֖ט דְּבַ֥שׁ הַזֶּֽה׃ (ח) אַ֗ף כִּ֡י לוּא֩ אָכֹ֨ל אָכַ֤ל הַיּוֹם֙ הָעָ֔ם מִשְּׁלַ֥ל אֹיְבָ֖יו אֲשֶׁ֣ר מָצָ֑א כִּ֥י עַתָּ֛ה לֹֽא־ רָבְתָ֥ה מַכָּ֖ה בַּפְּלִשְׁתִּֽים׃ (ט) וַיַּכּ֞וּ בַּיּ֤וֹם הַהוּא֙ בַּפְּלִשְׁתִּ֔ים מִמִּכְמָ֖שׂ אַיָּלֹ֑נָה וַיָּ֥עַף הָעָ֖ם מְאֹֽד׃ (י) (ויעש) וַיַּ֤עַט הָעָם֙ אֶל־ (שלל) הַשָּׁלָ֔ל וַיִּקְח֨וּ צֹ֧אן וּבָקָ֛ר וּבְנֵ֥י בָקָ֖ר וַיִּשְׁחֲטוּ־ אָ֑רְצָה וַיֹּ֥אכַל הָעָ֖ם עַל־ הַדָּֽם׃ (יא) וַיַּגִּ֤ידוּ לְשָׁאוּל֙ לֵאמֹ֔ר הִנֵּ֥ה הָעָ֛ם חֹטִ֥אים לַֽיהוָ֖ה לֶאֱכֹ֣ל עַל־ הַדָּ֑ם וַיֹּ֣אמֶר בְּגַדְתֶּ֔ם גֹּֽלּוּ־ אֵלַ֥י הַיּ֖וֹם אֶ֥בֶן גְּדוֹלָֽה׃ (יב) וַיֹּ֣אמֶר שָׁא֣וּל פֻּ֣צוּ בָעָ֡ם וַאֲמַרְתֶּ֣ם לָהֶ֡ם הַגִּ֣ישׁוּ אֵלַי֩ אִ֨ישׁ שׁוֹר֜וֹ וְאִ֣ישׁ שְׂיֵ֗הוּ וּשְׁחַטְתֶּ֤ם בָּזֶה֙ וַאֲכַלְתֶּ֔ם וְלֹֽא־ תֶחֶטְא֥וּ לַֽיהוָ֖ה לֶאֱכֹ֣ל אֶל־ הַדָּ֑ם וַיַּגִּ֨שׁוּ כָל־ הָעָ֜ם אִ֣ישׁ שׁוֹר֧וֹ בְיָד֛וֹ הַלַּ֖יְלָה וַיִּשְׁחֲטוּ־ שָֽׁם׃ (יג) וַיִּ֧בֶן שָׁא֛וּל מִזְבֵּ֖חַ לַֽיהוָ֑ה אֹת֣וֹ הֵחֵ֔ל לִבְנ֥וֹת מִזְבֵּ֖חַ לַֽיהוָֽה׃ (יד) וַיֹּ֣אמֶר שָׁא֡וּל נֵרְדָ֣ה אַחֲרֵי֩ פְלִשְׁתִּ֨ים ׀ לַ֜יְלָה וְֽנָבֹ֥זָה בָהֶ֣ם ׀ עַד־ א֣וֹר הַבֹּ֗קֶר וְלֹֽא־ נַשְׁאֵ֤ר בָּהֶם֙ אִ֔ישׁ וַיֹּ֣אמְר֔וּ כָּל־ הַטּ֥וֹב בְּעֵינֶ֖יךָ עֲשֵׂ֑ה וַיֹּ֙אמֶר֙ הַכֹּהֵ֔ן נִקְרְבָ֥ה הֲלֹ֖ם אֶל־ הָאֱלֹהִֽים׃ (טו) וַיִּשְׁאַ֤ל שָׁאוּל֙ בֵּֽאלֹהִ֔ים הַֽאֵרֵד֙ אַחֲרֵ֣י פְלִשְׁתִּ֔ים הֲתִתְּנֵ֖ם בְּיַ֣ד יִשְׂרָאֵ֑ל וְלֹ֥א עָנָ֖הוּ בַּיּ֥וֹם הַהֽוּא׃ (טז) וַיֹּ֣אמֶר שָׁא֔וּל גֹּ֣שֽׁוּ הֲלֹ֔ם כֹּ֖ל פִּנּ֣וֹת הָעָ֑ם וּדְע֣וּ וּרְא֔וּ בַּמָּ֗ה הָֽיְתָ֛ה הַחַטָּ֥את הַזֹּ֖את הַיּֽוֹם׃ (יז) כִּ֣י חַי־ יְהוָ֗ה הַמּוֹשִׁ֙יעַ֙ אֶת־ יִשְׂרָאֵ֔ל כִּ֧י אִם־ יֶשְׁנ֛וֹ בְּיוֹנָתָ֥ן בְּנִ֖י כִּ֣י מ֣וֹת יָמ֑וּת וְאֵ֥ין עֹנֵ֖הוּ מִכָּל־ הָעָֽם׃ (יח) וַיֹּ֣אמֶר אֶל־ כָּל־ יִשְׂרָאֵ֗ל אַתֶּם֙ תִּֽהְיוּ֙ לְעֵ֣בֶר אֶחָ֔ד וַֽאֲנִי֙ וְיוֹנָתָ֣ן בְּנִ֔י נִהְיֶ֖ה לְעֵ֣בֶר אֶחָ֑ד וַיֹּאמְר֤וּ הָעָם֙ אֶל־ שָׁא֔וּל הַטּ֥וֹב בְּעֵינֶ֖יךָ עֲשֵֽׂה׃ (יט) וַיֹּ֣אמֶר שָׁא֗וּל אֶל־ יְהוָ֛ה אֱלֹהֵ֥י יִשְׂרָאֵ֖ל הָ֣בָה תָמִ֑ים וַיִּלָּכֵ֧ד יוֹנָתָ֛ן וְשָׁא֖וּל וְהָעָ֥ם יָצָֽאוּ׃ (כ) וַיֹּ֣אמֶר שָׁא֔וּל הַפִּ֕ילוּ בֵּינִ֕י וּבֵ֖ין יוֹנָתָ֣ן בְּנִ֑י וַיִּלָּכֵ֖ד יוֹנָתָֽן׃ (כא) וַיֹּ֤אמֶר שָׁאוּל֙ אֶל־ י֣וֹנָתָ֔ן הַגִּ֥ידָה לִּ֖י מֶ֣ה עָשִׂ֑יתָה וַיַּגֶּד־ ל֣וֹ יוֹנָתָ֗ן וַיֹּאמֶר֩ טָעֹ֨ם טָעַ֜מְתִּי בִּקְצֵ֨ה הַמַּטֶּ֧ה אֲשֶׁר־ בְּיָדִ֛י מְעַ֥ט דְּבַ֖שׁ הִנְנִ֥י אָמֽוּת׃ (כב) וַיֹּ֣אמֶר שָׁא֔וּל כֹּֽה־ יַעֲשֶׂ֥ה אֱלֹהִ֖ים וְכֹ֣ה יוֹסִ֑ף כִּֽי־ מ֥וֹת תָּמ֖וּת יוֹנָתָֽן׃ (כג) וַיֹּ֨אמֶר הָעָ֜ם אֶל־ שָׁא֗וּל הֲ‍ֽיוֹנָתָ֤ן ׀ יָמוּת֙ אֲשֶׁ֣ר עָ֠שָׂה הַיְשׁוּעָ֨ה הַגְּדוֹלָ֣ה הַזֹּאת֮ בְּיִשְׂרָאֵל֒ חָלִ֗ילָה חַי־ יְהוָה֙ אִם־ יִפֹּ֞ל מִשַּׂעֲרַ֤ת רֹאשׁוֹ֙ אַ֔רְצָה כִּֽי־ עִם־ אֱלֹהִ֥ים עָשָׂ֖ה הַיּ֣וֹם הַזֶּ֑ה וַיִּפְדּ֥וּ הָעָ֛ם אֶת־ יוֹנָתָ֖ן וְלֹא־ מֵֽת׃ (כד) וַיַּ֣עַל שָׁא֔וּל מֵאַחֲרֵ֖י פְּלִשְׁתִּ֑ים וּפְלִשְׁתִּ֖ים הָלְכ֥וּ לִמְקוֹמָֽם׃ (כה) וְשָׁא֛וּל לָכַ֥ד הַמְּלוּכָ֖ה עַל־ יִשְׂרָאֵ֑ל וַיִּלָּ֣חֶם סָבִ֣יב ׀ בְּֽכָל־ אֹיְבָ֡יו בְּמוֹאָ֣ב ׀ וּבִבְנֵי־ עַמּ֨וֹן וּבֶאֱד֜וֹם וּבְמַלְכֵ֤י צוֹבָה֙ וּבַפְּלִשְׁתִּ֔ים וּבְכֹ֥ל אֲשֶׁר־ יִפְנֶ֖ה יַרְשִֽׁיעַ׃ (כו) וַיַּ֣עַשׂ חַ֔יִל וַיַּ֖ךְ אֶת־ עֲמָלֵ֑ק וַיַּצֵּ֥ל אֶת־ יִשְׂרָאֵ֖ל מִיַּ֥ד שֹׁסֵֽהוּ׃ (כז) וַיִּֽהְיוּ֙ בְּנֵ֣י שָׁא֔וּל יוֹנָתָ֥ן וְיִשְׁוִ֖י וּמַלְכִּי־ שׁ֑וּעַ וְשֵׁם֙ שְׁתֵּ֣י בְנֹתָ֔יו שֵׁ֤ם הַבְּכִירָה֙ מֵרַ֔ב וְשֵׁ֥ם הַקְּטַנָּ֖ה מִיכַֽל׃ (כח) וְשֵׁם֙ אֵ֣שֶׁת שָׁא֔וּל אֲחִינֹ֖עַם בַּת־ אֲחִימָ֑עַץ וְשֵׁ֤ם שַׂר־ צְבָאוֹ֙ אֲבִינֵ֔ר בֶּן־ נֵ֖ר דּ֥וֹד שָׁאֽוּל׃ (כט) וְקִ֧ישׁ אֲבִֽי־ שָׁא֛וּל וְנֵ֥ר אֲבִֽי־ אַבְנֵ֖ר בֶּן־ אֲבִיאֵֽל׃ (ל) וַתְּהִ֤י הַמִּלְחָמָה֙ חֲזָקָ֣ה עַל־ פְּלִשְׁתִּ֔ים כֹּ֖ל יְמֵ֣י שָׁא֑וּל וְרָאָ֨ה שָׁא֜וּל כָּל־ אִ֤ישׁ גִּבּוֹר֙ וְכָל־ בֶּן־ חַ֔יִל וַיַּאַסְפֵ֖הוּ אֵלָֽיו׃ תחילת פרק טו (לא) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ שָׁא֔וּל אֹתִ֨י שָׁלַ֤ח יְהוָה֙ לִמְשָׁחֳךָ֣ לְמֶ֔לֶךְ עַל־ עַמּ֖וֹ עַל־ יִשְׂרָאֵ֑ל וְעַתָּ֣ה שְׁמַ֔ע לְק֖וֹל דִּבְרֵ֥י יְהוָֽה׃ (לב) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת פָּקַ֕דְתִּי אֵ֛ת אֲשֶׁר־ עָשָׂ֥ה עֲמָלֵ֖ק לְיִשְׂרָאֵ֑ל אֲשֶׁר־ שָׂ֥ם לוֹ֙ בַּדֶּ֔רֶךְ בַּעֲלֹת֖וֹ מִמִּצְרָֽיִם׃ (לג) עַתָּה֩ לֵ֨ךְ וְהִכִּֽיתָ֜ה אֶת־ עֲמָלֵ֗ק וְהַֽחֲרַמְתֶּם֙ אֶת־ כָּל־ אֲשֶׁר־ ל֔וֹ וְלֹ֥א תַחְמֹ֖ל עָלָ֑יו וְהֵמַתָּ֞ה מֵאִ֣ישׁ עַד־ אִשָּׁ֗ה מֵֽעֹלֵל֙ וְעַד־ יוֹנֵ֔ק מִשּׁ֣וֹר וְעַד־ שֶׂ֔ה מִגָּמָ֖ל וְעַד־ חֲמֽוֹר׃ (לד) וַיְשַׁמַּ֤ע שָׁאוּל֙ אֶת־ הָעָ֔ם וַֽיִּפְקְדֵם֙ בַּטְּלָאִ֔ים מָאתַ֥יִם אֶ֖לֶף רַגְלִ֑י וַעֲשֶׂ֥רֶת אֲלָפִ֖ים אֶת־ אִ֥ישׁ יְהוּדָֽה׃ (לה) וַיָּבֹ֥א שָׁא֖וּל עַד־ עִ֣יר עֲמָלֵ֑ק וַיָּ֖רֶב בַּנָּֽחַל׃ (לו) וַיֹּ֣אמֶר שָׁא֣וּל אֶֽל־ הַקֵּינִ֡י לְכוּ֩ סֻּ֨רוּ רְד֜וּ מִתּ֣וֹךְ עֲמָלֵקִ֗י פֶּן־ אֹֽסִפְךָ֙ עִמּ֔וֹ וְאַתָּ֞ה עָשִׂ֤יתָה חֶ֙סֶד֙ עִם־ כָּל־ בְּנֵ֣י יִשְׂרָאֵ֔ל בַּעֲלוֹתָ֖ם מִמִּצְרָ֑יִם וַיָּ֥סַר קֵינִ֖י מִתּ֥וֹךְ עֲמָלֵֽק׃ (לז) וַיַּ֥ךְ שָׁא֖וּל אֶת־ עֲמָלֵ֑ק מֵֽחֲוִילָה֙ בּוֹאֲךָ֣ שׁ֔וּר אֲשֶׁ֖ר עַל־ פְּנֵ֥י מִצְרָֽיִם׃ (לח) בְּבוֹאָ֔ם וַיַּ֖רְא אֶת־ אֱלִיאָ֑ב וַיֹּ֕אמֶר אַ֛ךְ נֶ֥גֶד יְהוָ֖ה מְשִׁיחֽוֹ׃ (כו) וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ שְׁמוּאֵ֗ל אַל־ תַּבֵּ֧ט אֶל־ מַרְאֵ֛הוּ וְאֶל־ גְּבֹ֥הַּ קוֹמָת֖וֹ כִּ֣י מְאַסְתִּ֑יהוּ כִּ֣י ׀ לֹ֗א אֲשֶׁ֤ר יִרְאֶה֙ הָאָדָ֔ם כִּ֤י הָֽאָדָם֙ יִרְאֶ֣ה לַעֵינַ֔יִם וַיהוָ֖ה יִרְאֶ֥ה לַלֵּבָֽב׃ (כז) וַיִּקְרָ֤א יִשַׁי֙ אֶל־ אֲבִ֣ינָדָ֔ב וַיַּעֲבִרֵ֖הוּ לִפְנֵ֣י שְׁמוּאֵ֑ל וַיֹּ֕אמֶר גַּם־ בָּזֶ֖ה לֹֽא־ בָחַ֥ר יְהוָֽה׃ (כח) וַיַּעֲבֵ֥ר יִשַׁ֖י שַׁמָּ֑ה וַיֹּ֕אמֶר גַּם־ בָּזֶ֖ה לֹא־ בָחַ֥ר יְהוָֽה׃ (כט) וַיַּעֲבֵ֥ר יִשַׁ֛י שִׁבְעַ֥ת בָּנָ֖יו לִפְנֵ֣י שְׁמוּאֵ֑ל וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ יִשַׁ֔י לֹא־ בָחַ֥ר יְהוָ֖ה בָּאֵֽלֶּה׃ (ל) וַיֹּ֨אמֶר שְׁמוּאֵ֣ל אֶל־ יִשַׁי֮ הֲתַ֣מּוּ הַנְּעָרִים֒ וַיֹּ֗אמֶר ע֚וֹד שָׁאַ֣ר הַקָּטָ֔ן וְהִנֵּ֥ה רֹעֶ֖ה בַּצֹּ֑אן וַיֹּ֨אמֶר שְׁמוּאֵ֤ל אֶל־ יִשַׁי֙ שִׁלְחָ֣ה וְקָחֶ֔נּוּ כִּ֥י לֹא־ נָסֹ֖ב עַד־ בֹּא֥וֹ פֹֽה׃ (לא) וַיִּשְׁלַ֤ח וַיְבִיאֵ֙הוּ֙ וְה֣וּא אַדְמוֹנִ֔י עִם־ יְפֵ֥ה עֵינַ֖יִם וְט֣וֹב רֹ֑אִי וַיֹּ֧אמֶר יְהוָ֛ה ק֥וּם מְשָׁחֵ֖הוּ כִּֽי־ זֶ֥ה הֽוּא׃ (לב) וַיִּקַּ֨ח שְׁמוּאֵ֜ל אֶת־ קֶ֣רֶן הַשֶּׁ֗מֶן וַיִּמְשַׁ֣ח אֹתוֹ֮ בְּקֶ֣רֶב אֶחָיו֒ וַתִּצְלַ֤ח רֽוּחַ־ יְהוָה֙ אֶל־ דָּוִ֔ד מֵהַיּ֥וֹם הַה֖וּא וָמָ֑עְלָה וַיָּ֣קָם שְׁמוּאֵ֔ל וַיֵּ֖לֶךְ הָרָמָֽתָה׃ (לג) וְר֧וּחַ יְהוָ֛ה סָ֖רָה מֵעִ֣ם שָׁא֑וּל וּבִֽעֲתַ֥תּוּ רֽוּחַ־ רָעָ֖ה מֵאֵ֥ת יְהוָֽה׃ (לד) וַיֹּאמְר֥וּ עַבְדֵֽי־ שָׁא֖וּל אֵלָ֑יו הִנֵּה־ נָ֧א רֽוּחַ־ אֱלֹהִ֛ים רָעָ֖ה מְבַעִתֶּֽךָ׃ (לה) יֹאמַר־ נָ֤א אֲדֹנֵ֙נוּ֙ עֲבָדֶ֣יךָ לְפָנֶ֔יךָ יְבַקְשׁ֕וּ אִ֕ישׁ יֹדֵ֖עַ מְנַגֵּ֣ן בַּכִּנּ֑וֹר וְהָיָ֗ה בִּֽהְי֨וֹת עָלֶ֤יךָ רֽוּחַ־ אֱלֹהִים֙ רָעָ֔ה וְנִגֵּ֥ן בְּיָד֖וֹ וְט֥וֹב לָֽךְ׃ (לו) וַיֹּ֥אמֶר שָׁא֖וּל אֶל־ עֲבָדָ֑יו רְאוּ־ נָ֣א לִ֗י אִ֚ישׁ מֵיטִ֣יב לְנַגֵּ֔ן וַהֲבִיאוֹתֶ֖ם אֵלָֽי׃
