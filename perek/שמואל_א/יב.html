שמואל א פרק יב לפי החלוקה לסדרים

פרק יב בספר שמואל א (25 פסוקים) נמצא בשמואל סדר ו (פסוקים כ–מ) ובשמואל סדר ז (פסוקים א–ד).

שמואל סדר ו

(א) וַיֹּ֨אמֶר שְׁמוּאֵ֜ל אֶל־ כָּל־ הָעָ֗ם הַרְּאִיתֶם֙ אֲשֶׁ֣ר בָּֽחַר־ בּ֣וֹ יְהוָ֔ה כִּ֛י אֵ֥ין כָּמֹ֖הוּ בְּכָל־ הָעָ֑ם וַיָּרִ֧עוּ כָל־ הָעָ֛ם וַיֹּאמְר֖וּ יְחִ֥י הַמֶּֽלֶךְ׃ (ב) וַיְדַבֵּ֨ר שְׁמוּאֵ֜ל אֶל־ הָעָ֗ם אֵ֚ת מִשְׁפַּ֣ט הַמְּלֻכָ֔ה וַיִּכְתֹּ֣ב בַּסֵּ֔פֶר וַיַּנַּ֖ח לִפְנֵ֣י יְהוָ֑ה וַיְשַׁלַּ֧ח שְׁמוּאֵ֛ל אֶת־ כָּל־ הָעָ֖ם אִ֥ישׁ לְבֵיתֽוֹ׃ (ג) וְגַ֨ם־ שָׁא֔וּל הָלַ֥ךְ לְבֵית֖וֹ גִּבְעָ֑תָה וַיֵּלְכ֣וּ עִמּ֔וֹ הַחַ֕יִל אֲשֶׁר־ נָגַ֥ע אֱלֹהִ֖ים בְּלִבָּֽם׃ (ד) וּבְנֵ֧י בְלִיַּ֣עַל אָמְר֗וּ מַה־ יֹּשִׁעֵ֙נוּ֙ זֶ֔ה וַיִּבְזֻ֕הוּ וְלֹֽא־ הֵבִ֥יאוּ ל֖וֹ מִנְחָ֑ה וַיְהִ֖י כְּמַחֲרִֽישׁ׃ (ה) וַיַּ֗עַל נָחָשׁ֙ הָֽעַמּוֹנִ֔י וַיִּ֖חַן עַל־ יָבֵ֣שׁ גִּלְעָ֑ד וַיֹּ֨אמְר֜וּ כָּל־ אַנְשֵׁ֤י יָבֵישׁ֙ אֶל־ נָחָ֔שׁ כְּרָת־ לָ֥נוּ בְרִ֖ית וְנַעַבְדֶֽךָּ׃ (ו) וַיֹּ֣אמֶר אֲלֵיהֶ֗ם נָחָשׁ֙ הָעַמּוֹנִ֔י בְּזֹאת֙ אֶכְרֹ֣ת לָכֶ֔ם בִּנְק֥וֹר לָכֶ֖ם כָּל־ עֵ֣ין יָמִ֑ין וְשַׂמְתִּ֥יהָ חֶרְפָּ֖ה עַל־ כָּל־ יִשְׂרָאֵֽל׃ (ז) וַיֹּאמְר֨וּ אֵלָ֜יו זִקְנֵ֣י יָבֵ֗ישׁ הֶ֤רֶף לָ֙נוּ֙ שִׁבְעַ֣ת יָמִ֔ים וְנִשְׁלְחָה֙ מַלְאָכִ֔ים בְּכֹ֖ל גְּב֣וּל יִשְׂרָאֵ֑ל וְאִם־ אֵ֥ין מוֹשִׁ֛יעַ אֹתָ֖נוּ וְיָצָ֥אנוּ אֵלֶֽיךָ׃ (ח) וַיָּבֹ֤אוּ הַמַּלְאָכִים֙ גִּבְעַ֣ת שָׁא֔וּל וַיְדַבְּר֥וּ הַדְּבָרִ֖ים בְּאָזְנֵ֣י הָעָ֑ם וַיִּשְׂא֧וּ כָל־ הָעָ֛ם אֶת־ קוֹלָ֖ם וַיִּבְכּֽוּ׃ (ט) וְהִנֵּ֣ה שָׁא֗וּל בָּ֣א אַחֲרֵ֤י הַבָּקָר֙ מִן־ הַשָּׂדֶ֔ה וַיֹּ֣אמֶר שָׁא֔וּל מַה־ לָּעָ֖ם כִּ֣י יִבְכּ֑וּ וַיְסַ֨פְּרוּ־ ל֔וֹ אֶת־ דִּבְרֵ֖י אַנְשֵׁ֥י יָבֵֽישׁ׃ (י) וַתִּצְלַ֤ח רֽוּחַ־ אֱלֹהִים֙ עַל־ שָׁא֔וּל (בשמעו) כְּשָׁמְע֖וֹ אֶת־ הַדְּבָרִ֣ים הָאֵ֑לֶּה וַיִּ֥חַר אַפּ֖וֹ מְאֹֽד׃ (יא) וַיִּקַּח֩ צֶ֨מֶד בָּקָ֜ר וַֽיְנַתְּחֵ֗הוּ וַיְשַׁלַּ֞ח בְּכָל־ גְּב֣וּל יִשְׂרָאֵל֮ בְּיַ֣ד הַמַּלְאָכִ֣ים ׀ לֵאמֹר֒ אֲשֶׁר֩ אֵינֶ֨נּוּ יֹצֵ֜א אַחֲרֵ֤י שָׁאוּל֙ וְאַחַ֣ר שְׁמוּאֵ֔ל כֹּ֥ה יֵעָשֶׂ֖ה לִבְקָר֑וֹ וַיִּפֹּ֤ל פַּֽחַד־ יְהוָה֙ עַל־ הָעָ֔ם וַיֵּצְא֖וּ כְּאִ֥ישׁ אֶחָֽד׃ (יב) וַֽיִּפְקְדֵ֖ם בְּבָ֑זֶק וַיִּהְי֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ שְׁלֹ֣שׁ מֵא֣וֹת אֶ֔לֶף וְאִ֥ישׁ יְהוּדָ֖ה שְׁלֹשִׁ֥ים אָֽלֶף׃ (יג) וַיֹּאמְר֞וּ לַמַּלְאָכִ֣ים הַבָּאִ֗ים כֹּ֤ה תֹֽאמְרוּן֙ לְאִישׁ֙ יָבֵ֣ישׁ גִּלְעָ֔ד מָחָ֛ר תִּהְיֶֽה־ לָכֶ֥ם תְּשׁוּעָ֖ה (בחם) כְּחֹ֣ם הַשָּׁ֑מֶשׁ וַיָּבֹ֣אוּ הַמַּלְאָכִ֗ים וַיַּגִּ֛ידוּ לְאַנְשֵׁ֥י יָבֵ֖ישׁ וַיִּשְׂמָֽחוּ׃ (יד) וַֽיֹּאמְרוּ֙ אַנְשֵׁ֣י יָבֵ֔ישׁ מָחָ֖ר נֵצֵ֣א אֲלֵיכֶ֑ם וַעֲשִׂיתֶ֣ם לָּ֔נוּ כְּכָל־ הַטּ֖וֹב בְּעֵינֵיכֶֽם׃ (טו) וַיְהִ֣י מִֽמָּחֳרָ֗ת וַיָּ֨שֶׂם שָׁא֣וּל אֶת־ הָעָם֮ שְׁלֹשָׁ֣ה רָאשִׁים֒ וַיָּבֹ֤אוּ בְתוֹךְ־ הַֽמַּחֲנֶה֙ בְּאַשְׁמֹ֣רֶת הַבֹּ֔קֶר וַיַּכּ֥וּ אֶת־ עַמּ֖וֹן עַד־ חֹ֣ם הַיּ֑וֹם וַיְהִ֤י הַנִּשְׁאָרִים֙ וַיָּפֻ֔צוּ וְלֹ֥א נִשְׁאֲרוּ־ בָ֖ם שְׁנַ֥יִם יָֽחַד׃ (טז) וַיֹּ֤אמֶר הָעָם֙ אֶל־ שְׁמוּאֵ֔ל מִ֣י הָאֹמֵ֔ר שָׁא֖וּל יִמְלֹ֣ךְ עָלֵ֑ינוּ תְּנ֥וּ הָאֲנָשִׁ֖ים וּנְמִיתֵֽם׃ (יז) וַיֹּ֣אמֶר שָׁא֔וּל לֹֽא־ יוּמַ֥ת אִ֖ישׁ בַּיּ֣וֹם הַזֶּ֑ה כִּ֥י הַיּ֛וֹם עָשָֽׂה־ יְהוָ֥ה תְּשׁוּעָ֖ה בְּיִשְׂרָאֵֽל׃ (יח) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ הָעָ֔ם לְכ֖וּ וְנֵלְכָ֣ה הַגִּלְגָּ֑ל וּנְחַדֵּ֥שׁ שָׁ֖ם הַמְּלוּכָֽה׃ (יט) וַיֵּלְכ֨וּ כָל־ הָעָ֜ם הַגִּלְגָּ֗ל וַיַּמְלִכוּ֩ שָׁ֨ם אֶת־ שָׁא֜וּל לִפְנֵ֤י יְהוָה֙ בַּגִּלְגָּ֔ל וַיִּזְבְּחוּ־ שָׁ֛ם זְבָחִ֥ים שְׁלָמִ֖ים לִפְנֵ֣י יְהוָ֑ה וַיִּשְׂמַ֨ח שָׁ֥ם שָׁא֛וּל וְכָל־ אַנְשֵׁ֥י יִשְׂרָאֵ֖ל עַד־ מְאֹֽד׃ תחילת פרק יב (כ) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ כָּל־ יִשְׂרָאֵ֔ל הִנֵּה֙ שָׁמַ֣עְתִּי בְקֹֽלְכֶ֔ם לְכֹ֥ל אֲשֶׁר־ אֲמַרְתֶּ֖ם לִ֑י וָאַמְלִ֥יךְ עֲלֵיכֶ֖ם מֶֽלֶךְ׃ (כא) וְעַתָּ֞ה הִנֵּ֥ה הַמֶּ֣לֶךְ ׀ מִתְהַלֵּ֣ךְ לִפְנֵיכֶ֗ם וַאֲנִי֙ זָקַ֣נְתִּי וָשַׂ֔בְתִּי וּבָנַ֖י הִנָּ֣ם אִתְּכֶ֑ם וַאֲנִי֙ הִתְהַלַּ֣כְתִּי לִפְנֵיכֶ֔ם מִנְּעֻרַ֖י עַד־ הַיּ֥וֹם הַזֶּֽה׃ (כב) הִנְנִ֣י עֲנ֣וּ בִי֩ נֶ֨גֶד יְהוָ֜ה וְנֶ֣גֶד מְשִׁיח֗וֹ אֶת־ שׁוֹר֩ ׀ מִ֨י לָקַ֜חְתִּי וַחֲמ֧וֹר מִ֣י לָקַ֗חְתִּי וְאֶת־ מִ֤י עָשַׁ֙קְתִּי֙ אֶת־ מִ֣י רַצּ֔וֹתִי וּמִיַּד־ מִי֙ לָקַ֣חְתִּי כֹ֔פֶר וְאַעְלִ֥ים עֵינַ֖י בּ֑וֹ וְאָשִׁ֖יב לָכֶֽם׃ (כג) וַיֹּ֣אמְר֔וּ לֹ֥א עֲשַׁקְתָּ֖נוּ וְלֹ֣א רַצּוֹתָ֑נוּ וְלֹֽא־ לָקַ֥חְתָּ מִיַּד־ אִ֖ישׁ מְאֽוּמָה׃ (כד) וַיֹּ֨אמֶר אֲלֵיהֶ֜ם עֵ֧ד יְהוָ֣ה בָּכֶ֗ם וְעֵ֤ד מְשִׁיחוֹ֙ הַיּ֣וֹם הַזֶּ֔ה כִּ֣י לֹ֧א מְצָאתֶ֛ם בְּיָדִ֖י מְא֑וּמָה וַיֹּ֖אמֶר עֵֽד׃ (כה) וַיֹּ֥אמֶר שְׁמוּאֵ֖ל אֶל־ הָעָ֑ם יְהוָ֗ה אֲשֶׁ֤ר עָשָׂה֙ אֶת־ מֹשֶׁ֣ה וְאֶֽת־ אַהֲרֹ֔ן וַאֲשֶׁ֧ר הֶעֱלָ֛ה אֶת־ אֲבֹתֵיכֶ֖ם מֵאֶ֥רֶץ מִצְרָֽיִם׃ (כו) וְעַתָּ֗ה הִֽתְיַצְּב֛וּ וְאִשָּׁפְטָ֥ה אִתְּכֶ֖ם לִפְנֵ֣י יְהוָ֑ה אֵ֚ת כָּל־ צִדְק֣וֹת יְהוָ֔ה אֲשֶׁר־ עָשָׂ֥ה אִתְּכֶ֖ם וְאֶת־ אֲבוֹתֵיכֶֽם׃ (כז) כַּֽאֲשֶׁר־ בָּ֥א יַעֲקֹ֖ב מִצְרָ֑יִם וַיִּזְעֲק֤וּ אֲבֽוֹתֵיכֶם֙ אֶל־ יְהוָ֔ה וַיִּשְׁלַ֨ח יְהוָ֜ה אֶת־ מֹשֶׁ֣ה וְאֶֽת־ אַהֲרֹ֗ן וַיּוֹצִ֤יאוּ אֶת־ אֲבֹֽתֵיכֶם֙ מִמִּצְרַ֔יִם וַיֹּשִׁב֖וּם בַּמָּק֥וֹם הַזֶּֽה׃ (כח) וַֽיִּשְׁכְּח֖וּ אֶת־ יְהוָ֣ה אֱלֹהֵיהֶ֑ם וַיִּמְכֹּ֣ר אֹתָ֡ם בְּיַ֣ד סִֽיסְרָא֩ שַׂר־ צְבָ֨א חָצ֜וֹר וּבְיַד־ פְּלִשְׁתִּ֗ים וּבְיַד֙ מֶ֣לֶךְ מוֹאָ֔ב וַיִּֽלָּחֲמ֖וּ בָּֽם׃ (כט) וַיִּזְעֲק֤וּ אֶל־ יְהוָה֙ (ויאמר) וַיֹּאמְר֣וּ חָטָ֔אנוּ כִּ֤י עָזַ֙בְנוּ֙ אֶת־ יְהוָ֔ה וַנַּעֲבֹ֥ד אֶת־ הַבְּעָלִ֖ים וְאֶת־ הָעַשְׁתָּר֑וֹת וְעַתָּ֗ה הַצִּילֵ֛נוּ מִיַּ֥ד אֹיְבֵ֖ינוּ וְנַעַבְדֶֽךָּ׃ (ל) וַיִּשְׁלַ֤ח יְהוָה֙ אֶת־ יְרֻבַּ֣עַל וְאֶת־ בְּדָ֔ן וְאֶת־ יִפְתָּ֖ח וְאֶת־ שְׁמוּאֵ֑ל וַיַּצֵּ֨ל אֶתְכֶ֜ם מִיַּ֤ד אֹֽיְבֵיכֶם֙ מִסָּבִ֔יב וַתֵּשְׁב֖וּ בֶּֽטַח׃ (לא) וַתִּרְא֗וּ כִּֽי־ נָחָ֞שׁ מֶ֣לֶךְ בְּנֵֽי־ עַמּוֹן֮ בָּ֣א עֲלֵיכֶם֒ וַתֹּ֣אמְרוּ לִ֔י לֹ֕א כִּי־ מֶ֖לֶךְ יִמְלֹ֣ךְ עָלֵ֑ינוּ וַיהוָ֥ה אֱלֹהֵיכֶ֖ם מַלְכְּכֶֽם׃ (לב) וְעַתָּ֗ה הִנֵּ֥ה הַמֶּ֛לֶךְ אֲשֶׁ֥ר בְּחַרְתֶּ֖ם אֲשֶׁ֣ר שְׁאֶלְתֶּ֑ם וְהִנֵּ֨ה נָתַ֧ן יְהוָ֛ה עֲלֵיכֶ֖ם מֶֽלֶךְ׃ (לג) אִם־ תִּֽירְא֣וּ אֶת־ יְהוָ֗ה וַעֲבַדְתֶּ֤ם אֹתוֹ֙ וּשְׁמַעְתֶּ֣ם בְּקֹל֔וֹ וְלֹ֥א תַמְר֖וּ אֶת־ פִּ֣י יְהוָ֑ה וִהְיִתֶ֣ם גַּם־ אַתֶּ֗ם וְגַם־ הַמֶּ֙לֶךְ֙ אֲשֶׁ֣ר מָלַ֣ךְ עֲלֵיכֶ֔ם אַחַ֖ר יְהוָ֥ה אֱלֹהֵיכֶֽם׃ (לד) וְאִם־ לֹ֤א תִשְׁמְעוּ֙ בְּק֣וֹל יְהוָ֔ה וּמְרִיתֶ֖ם אֶת־ פִּ֣י יְהוָ֑ה וְהָיְתָ֧ה יַד־ יְהוָ֛ה בָּכֶ֖ם וּבַאֲבֹתֵיכֶֽם׃ (לה) וְאֶת־ יוֹנָתָ֑ן וַתִּמָּצֵ֣א לְשָׁא֔וּל וּלְיוֹנָתָ֖ן בְּנֽוֹ׃ (כז) וַיֵּצֵא֙ מַצַּ֣ב פְּלִשְׁתִּ֔ים אֶֽל־ מַעֲבַ֖ר מִכְמָֽשׂ׃ (כח) וַיְהִ֣י הַיּ֗וֹם וַיֹּ֨אמֶר יוֹנָתָ֤ן בֶּן־ שָׁאוּל֙ אֶל־ הַנַּ֙עַר֙ נֹשֵׂ֣א כֵלָ֔יו לְכָ֗ה וְנַעְבְּרָה֙ אֶל־ מַצַּ֣ב פְּלִשְׁתִּ֔ים אֲשֶׁ֖ר מֵעֵ֣בֶר הַלָּ֑ז וּלְאָבִ֖יו לֹ֥א הִגִּֽיד׃ (כט) וְשָׁא֗וּל יוֹשֵׁב֙ בִּקְצֵ֣ה הַגִּבְעָ֔ה תַּ֥חַת הָרִמּ֖וֹן אֲשֶׁ֣ר בְּמִגְר֑וֹן וְהָעָם֙ אֲשֶׁ֣ר עִמּ֔וֹ כְּשֵׁ֥שׁ מֵא֖וֹת אִֽישׁ׃ (ל) וַאֲחִיָּ֣ה בֶן־ אֲחִט֡וּב אֲחִ֡י אִיכָב֣וֹד ׀ בֶּן־ פִּינְחָ֨ס בֶּן־ עֵלִ֜י כֹּהֵ֧ן ׀ יְהוָ֛ה בְּשִׁל֖וֹ נֹשֵׂ֣א אֵפ֑וֹד וְהָעָם֙ לֹ֣א יָדַ֔ע כִּ֥י הָלַ֖ךְ יוֹנָתָֽן׃ (לא) וּבֵ֣ין הַֽמַּעְבְּר֗וֹת אֲשֶׁ֨ר בִּקֵּ֤שׁ יֽוֹנָתָן֙ לַֽעֲבֹר֙ עַל־ מַצַּ֣ב פְּלִשְׁתִּ֔ים שֵׁן־ הַסֶּ֤לַע מֵהָעֵ֙בֶר֙ מִזֶּ֔ה וְשֵׁן־ הַסֶּ֥לַע מֵהָעֵ֖בֶר מִזֶּ֑ה וְשֵׁ֤ם הָֽאֶחָד֙ בּוֹצֵ֔ץ וְשֵׁ֥ם הָאֶחָ֖ד סֶֽנֶּה׃ (לב) הַשֵּׁ֧ן הָאֶחָ֛ד מָצ֥וּק מִצָּפ֖וֹן מ֣וּל מִכְמָ֑שׂ וְהָאֶחָ֥ד מִנֶּ֖גֶב מ֥וּל גָּֽבַע׃ (לג) וַיֹּ֨אמֶר יְהוֹנָתָ֜ן אֶל־ הַנַּ֣עַר ׀ נֹשֵׂ֣א כֵלָ֗יו לְכָה֙ וְנַעְבְּרָ֗ה אֶל־ מַצַּב֙ הָעֲרֵלִ֣ים הָאֵ֔לֶּה אוּלַ֛י יַעֲשֶׂ֥ה יְהוָ֖ה לָ֑נוּ כִּ֣י אֵ֤ין לַֽיהוָה֙ מַעְצ֔וֹר לְהוֹשִׁ֥יעַ בְּרַ֖ב א֥וֹ בִמְעָֽט׃ (לד) וַיֹּ֤אמֶר לוֹ֙ נֹשֵׂ֣א כֵלָ֔יו עֲשֵׂ֖ה כָּל־ אֲשֶׁ֣ר בִּלְבָבֶ֑ךָ נְטֵ֣ה לָ֔ךְ הִנְנִ֥י עִמְּךָ֖ כִּלְבָבֶֽךָ׃ (לה) וַיֹּ֙אמֶר֙ יְה֣וֹנָתָ֔ן הִנֵּ֛ה אֲנַ֥חְנוּ עֹבְרִ֖ים אֶל־ הָאֲנָשִׁ֑ים וְנִגְלִ֖ינוּ אֲלֵיהֶֽם׃ (לו) אִם־ כֹּ֤ה יֹֽאמְרוּ֙ אֵלֵ֔ינוּ דֹּ֕מּוּ עַד־ הַגִּיעֵ֖נוּ אֲלֵיכֶ֑ם וְעָמַ֣דְנוּ תַחְתֵּ֔ינוּ וְלֹ֥א נַעֲלֶ֖ה אֲלֵיהֶֽם׃ (לז) וְאִם־ כֹּ֨ה יֹאמְר֜וּ עֲל֤וּ עָלֵ֙ינוּ֙ וְעָלִ֔ינוּ כִּֽי־ נְתָנָ֥ם יְהוָ֖ה בְּיָדֵ֑נוּ וְזֶה־ לָּ֖נוּ הָאֽוֹת׃ (לח) וַיִּגָּל֣וּ שְׁנֵיהֶ֔ם אֶל־ מַצַּ֖ב פְּלִשְׁתִּ֑ים וַיֹּאמְר֣וּ פְלִשְׁתִּ֔ים הִנֵּ֤ה עִבְרִים֙ יֹֽצְאִ֔ים מִן־ הַחֹרִ֖ים אֲשֶׁ֥ר הִתְחַבְּאוּ־ שָֽׁם׃ (לט) וַיַּעֲנוּ֩ אַנְשֵׁ֨י הַמַּצָּבָ֜ה אֶת־ יוֹנָתָ֣ן ׀ וְאֶת־ נֹשֵׂ֣א כֵלָ֗יו וַיֹּֽאמְרוּ֙ עֲל֣וּ אֵלֵ֔ינוּ וְנוֹדִ֥יעָה אֶתְכֶ֖ם דָּבָ֑ר וַיֹּ֨אמֶר יוֹנָתָ֜ן אֶל־ נֹשֵׂ֤א כֵלָיו֙ עֲלֵ֣ה אַחֲרַ֔י כִּֽי־ נְתָנָ֥ם יְהוָ֖ה בְּיַ֥ד יִשְׂרָאֵֽל׃ (מ) וַיַּ֣עַל יוֹנָתָ֗ן עַל־ יָדָיו֙ וְעַל־ רַגְלָ֔יו וְנֹשֵׂ֥א כֵלָ֖יו אַחֲרָ֑יו וַֽיִּפְּלוּ֙ לִפְנֵ֣י יוֹנָתָ֔ן וְנֹשֵׂ֥א כֵלָ֖יו מְמוֹתֵ֥ת אַחֲרָֽיו׃ (מא) וַתְּהִ֞י הַמַּכָּ֣ה הָרִאשֹׁנָ֗ה אֲשֶׁ֨ר הִכָּ֧ה יוֹנָתָ֛ן וְנֹשֵׂ֥א כֵלָ֖יו כְּעֶשְׂרִ֣ים אִ֑ישׁ כְּבַחֲצִ֥י מַעֲנָ֖ה צֶ֥מֶד שָׂדֶֽה׃ (מב) וַתְּהִי֩ חֲרָדָ֨ה בַמַּחֲנֶ֤ה בַשָּׂדֶה֙ וּבְכָל־ הָעָ֔ם הַמַּצָּב֙ וְהַמַּשְׁחִ֔ית חָרְד֖וּ גַּם־ הֵ֑מָּה וַתִּרְגַּ֣ז הָאָ֔רֶץ וַתְּהִ֖י לְחֶרְדַּ֥ת אֱלֹהִֽים׃ (מג) וַיִּרְא֤וּ הַצֹּפִים֙ לְשָׁא֔וּל בְּגִבְעַ֖ת בִּנְיָמִ֑ן וְהִנֵּ֧ה הֶהָמ֛וֹן נָמ֖וֹג וַיֵּ֥לֶךְ וַהֲלֹֽם׃ (מד) וַיֹּ֣אמֶר שָׁא֗וּל לָעָם֙ אֲשֶׁ֣ר אִתּ֔וֹ פִּקְדוּ־ נָ֣א וּרְא֔וּ מִ֖י הָלַ֣ךְ מֵעִמָּ֑נוּ וַֽיִּפְקְד֔וּ וְהִנֵּ֛ה אֵ֥ין יוֹנָתָ֖ן וְנֹשֵׂ֥א כֵלָֽיו׃ (מה) וַיֹּ֤אמֶר שָׁאוּל֙ לַֽאֲחִיָּ֔ה הַגִּ֖ישָׁה אֲר֣וֹן הָאֱלֹהִ֑ים כִּֽי־ הָיָ֞ה אֲר֧וֹן הָאֱלֹהִ֛ים בַּיּ֥וֹם הַה֖וּא וּבְנֵ֥י יִשְׂרָאֵֽל׃ (מו) וַיְהִ֗י עַ֣ד דִּבֶּ֤ר שָׁאוּל֙ אֶל־ הַכֹּהֵ֔ן וְהֶהָמ֗וֹן אֲשֶׁר֙ בְּמַחֲנֵ֣ה פְלִשְׁתִּ֔ים וַיֵּ֥לֶךְ הָל֖וֹךְ וָרָ֑ב וַיֹּ֧אמֶר שָׁא֛וּל אֶל־ הַכֹּהֵ֖ן אֱסֹ֥ף יָדֶֽךָ׃ (מז) וַיִּזָּעֵ֣ק שָׁא֗וּל וְכָל־ הָעָם֙ אֲשֶׁ֣ר אִתּ֔וֹ וַיָּבֹ֖אוּ עַד־ הַמִּלְחָמָ֑ה וְהִנֵּ֨ה הָיְתָ֜ה חֶ֤רֶב אִישׁ֙ בְּרֵעֵ֔הוּ מְהוּמָ֖ה גְּדוֹלָ֥ה מְאֹֽד׃ (מח) וְהָעִבְרִ֗ים הָי֤וּ לַפְּלִשְׁתִּים֙ כְּאֶתְמ֣וֹל שִׁלְשׁ֔וֹם אֲשֶׁ֨ר עָל֥וּ עִמָּ֛ם בַּֽמַּחֲנֶ֖ה סָבִ֑יב וְגַם־ הֵ֗מָּה לִֽהְיוֹת֙ עִם־ יִשְׂרָאֵ֔ל אֲשֶׁ֥ר עִם־ שָׁא֖וּל וְיוֹנָתָֽן׃ (מט) וְכֹל֩ אִ֨ישׁ יִשְׂרָאֵ֜ל הַמִּֽתְחַבְּאִ֤ים בְּהַר־ אֶפְרַ֙יִם֙ שָֽׁמְע֔וּ כִּֽי־ נָ֖סוּ פְּלִשְׁתִּ֑ים וַֽיַּדְבְּק֥וּ גַם־ הֵ֛מָּה אַחֲרֵיהֶ֖ם בַּמִּלְחָמָֽה׃
