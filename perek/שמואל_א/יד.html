שמואל א פרק יד לפי החלוקה לסדרים

פרק יד בספר שמואל א (52 פסוקים) נמצא בשמואל סדר ז (פסוקים כח–מט) ובשמואל סדר ח (פסוקים א–ל).

שמואל סדר ז

(א) כִּ֠י לֹֽא־ יִטֹּ֤שׁ יְהוָה֙ אֶת־ עַמּ֔וֹ בַּעֲב֖וּר שְׁמ֣וֹ הַגָּד֑וֹל כִּ֚י הוֹאִ֣יל יְהוָ֔ה לַעֲשׂ֥וֹת אֶתְכֶ֛ם ל֖וֹ לְעָֽם׃ (ב) גַּ֣ם אָנֹכִ֗י חָלִ֤ילָה לִּי֙ מֵחֲטֹ֣א לַֽיהוָ֔ה מֵחֲדֹ֖ל לְהִתְפַּלֵּ֣ל בַּעַדְכֶ֑ם וְהוֹרֵיתִ֣י אֶתְכֶ֔ם בְּדֶ֥רֶךְ הַטּוֹבָ֖ה וְהַיְשָׁרָֽה׃ (ג) אַ֣ךְ ׀ יְר֣אוּ אֶת־ יְהוָ֗ה וַעֲבַדְתֶּ֥ם אֹת֛וֹ בֶּאֱמֶ֖ת בְּכָל־ לְבַבְכֶ֑ם כִּ֣י רְא֔וּ אֵ֥ת אֲשֶׁר־ הִגְדִּ֖ל עִמָּכֶֽם׃ (ד) וְאִם־ הָרֵ֖עַ תָּרֵ֑עוּ גַּם־ אַתֶּ֥ם גַּֽם־ מַלְכְּכֶ֖ם תִּסָּפֽוּ׃ (ה) בֶּן־ שָׁנָ֖ה שָׁא֣וּל בְּמָלְכ֑וֹ וּשְׁתֵּ֣י שָׁנִ֔ים מָלַ֖ךְ עַל־ יִשְׂרָאֵֽל׃ (ו) וַיִּבְחַר־ ל֨וֹ שָׁא֜וּל שְׁלֹ֣שֶׁת אֲלָפִים֮ מִיִּשְׂרָאֵל֒ וַיִּהְי֨וּ עִם־ שָׁא֜וּל אַלְפַּ֗יִם בְּמִכְמָשׂ֙ וּבְהַ֣ר בֵּֽית־ אֵ֔ל וְאֶ֗לֶף הָיוּ֙ עִם־ י֣וֹנָתָ֔ן בְּגִבְעַ֖ת בִּנְיָמִ֑ין וְיֶ֣תֶר הָעָ֔ם שִׁלַּ֖ח אִ֥ישׁ לְאֹהָלָֽיו׃ (ז) וַיַּ֣ךְ יוֹנָתָ֗ן אֵ֣ת נְצִ֤יב פְּלִשְׁתִּים֙ אֲשֶׁ֣ר בְּגֶ֔בַע וַֽיִּשְׁמְע֖וּ פְּלִשְׁתִּ֑ים וְשָׁאוּל֩ תָּקַ֨ע בַּשּׁוֹפָ֤ר בְּכָל־ הָאָ֙רֶץ֙ לֵאמֹ֔ר יִשְׁמְע֖וּ הָעִבְרִֽים׃ (ח) וְכָל־ יִשְׂרָאֵ֞ל שָׁמְע֣וּ לֵאמֹ֗ר הִכָּ֤ה שָׁאוּל֙ אֶת־ נְצִ֣יב פְּלִשְׁתִּ֔ים וְגַם־ נִבְאַשׁ יִשְׂרָאֵ֖ל בַּפְּלִשְׁתִּ֑ים וַיִּצָּעֲק֥וּ הָעָ֛ם אַחֲרֵ֥י שָׁא֖וּל הַגִּלְגָּֽל׃ (ט) וּפְלִשְׁתִּ֞ים נֶאֶסְפ֣וּ ׀ לְהִלָּחֵ֣ם עִם־ יִשְׂרָאֵ֗ל שְׁלֹשִׁ֨ים אֶ֤לֶף רֶ֙כֶב֙ וְשֵׁ֤שֶׁת אֲלָפִים֙ פָּרָשִׁ֔ים וְעָ֕ם כַּח֛וֹל אֲשֶׁ֥ר עַל־ שְׂפַֽת־ הַיָּ֖ם לָרֹ֑ב וַֽיַּעֲלוּ֙ וַיַּחֲנ֣וּ בְמִכְמָ֔שׂ קִדְמַ֖ת בֵּ֥ית אָֽוֶן׃ (י) וְאִ֨ישׁ יִשְׂרָאֵ֤ל רָאוּ֙ כִּ֣י צַר־ ל֔וֹ כִּ֥י נִגַּ֖שׂ הָעָ֑ם וַיִּֽתְחַבְּא֣וּ הָעָ֗ם בַּמְּעָר֤וֹת וּבַֽחֲוָחִים֙ וּבַסְּלָעִ֔ים וּבַצְּרִחִ֖ים וּבַבֹּרֽוֹת׃ (יא) וְעִבְרִ֗ים עָֽבְרוּ֙ אֶת־ הַיַּרְדֵּ֔ן אֶ֥רֶץ גָּ֖ד וְגִלְעָ֑ד וְשָׁאוּל֙ עוֹדֶ֣נּוּ בַגִּלְגָּ֔ל וְכָל־ הָעָ֖ם חָרְד֥וּ אַחֲרָֽיו׃ (יב) (וייחל) וַיּ֣וֹחֶל ׀ שִׁבְעַ֣ת יָמִ֗ים לַמּוֹעֵד֙ אֲשֶׁ֣ר שְׁמוּאֵ֔ל וְלֹא־ בָ֥א שְׁמוּאֵ֖ל הַגִּלְגָּ֑ל וַיָּ֥פֶץ הָעָ֖ם מֵעָלָֽיו׃ (יג) וַיֹּ֣אמֶר שָׁא֔וּל הַגִּ֣שׁוּ אֵלַ֔י הָעֹלָ֖ה וְהַשְּׁלָמִ֑ים וַיַּ֖עַל הָעֹלָֽה׃ (יד) וַיְהִ֗י כְּכַלֹּתוֹ֙ לְהַעֲל֣וֹת הָעֹלָ֔ה וְהִנֵּ֥ה שְׁמוּאֵ֖ל בָּ֑א וַיֵּצֵ֥א שָׁא֛וּל לִקְרָאת֖וֹ לְבָרֲכֽוֹ׃ (טו) וַיֹּ֥אמֶר שְׁמוּאֵ֖ל מֶ֣ה עָשִׂ֑יתָ וַיֹּ֣אמֶר שָׁא֡וּל כִּֽי־ רָאִיתִי֩ כִֽי־ נָפַ֨ץ הָעָ֜ם מֵעָלַ֗י וְאַתָּה֙ לֹא־ בָ֙אתָ֙ לְמוֹעֵ֣ד הַיָּמִ֔ים וּפְלִשְׁתִּ֖ים נֶאֱסָפִ֥ים מִכְמָֽשׂ׃ (טז) וָאֹמַ֗ר עַ֠תָּה יֵרְד֨וּ פְלִשְׁתִּ֤ים אֵלַי֙ הַגִּלְגָּ֔ל וּפְנֵ֥י יְהוָ֖ה לֹ֣א חִלִּ֑יתִי וָֽאֶתְאַפַּ֔ק וָאַעֲלֶ֖ה הָעֹלָֽה׃ (יז) וַיֹּ֧אמֶר שְׁמוּאֵ֛ל אֶל־ שָׁא֖וּל נִסְכָּ֑לְתָּ לֹ֣א שָׁמַ֗רְתָּ אֶת־ מִצְוַ֞ת יְהוָ֤ה אֱלֹהֶ֙יךָ֙ אֲשֶׁ֣ר צִוָּ֔ךְ כִּ֣י עַתָּ֗ה הֵכִ֨ין יְהוָ֧ה אֶת־ מַֽמְלַכְתְּךָ֛ אֶל־ יִשְׂרָאֵ֖ל עַד־ עוֹלָֽם׃ (יח) וְעַתָּ֖ה מַמְלַכְתְּךָ֣ לֹא־ תָק֑וּם בִּקֵּשׁ֩ יְהוָ֨ה ל֜וֹ אִ֣ישׁ כִּלְבָב֗וֹ וַיְצַוֵּ֨הוּ יְהוָ֤ה לְנָגִיד֙ עַל־ עַמּ֔וֹ כִּ֚י לֹ֣א שָׁמַ֔רְתָּ אֵ֥ת אֲשֶֽׁר־ צִוְּךָ֖ יְהוָֽה׃ (יט) וַיָּ֣קָם שְׁמוּאֵ֗ל וַיַּ֛עַל מִן־ הַגִּלְגָּ֖ל גִּבְעַ֣ת בִּנְיָמִ֑ן וַיִּפְקֹ֣ד שָׁא֗וּל אֶת־ הָעָם֙ הַנִּמְצְאִ֣ים עִמּ֔וֹ כְּשֵׁ֥שׁ מֵא֖וֹת אִֽישׁ׃ (כ) וְשָׁא֞וּל וְיוֹנָתָ֣ן בְּנ֗וֹ וְהָעָם֙ הַנִּמְצָ֣א עִמָּ֔ם יֹשְׁבִ֖ים בְּגֶ֣בַע בִּנְיָמִ֑ן וּפְלִשְׁתִּ֖ים חָנ֥וּ בְמִכְמָֽשׂ׃ (כא) וַיֵּצֵ֧א הַמַּשְׁחִ֛ית מִמַּחֲנֵ֥ה פְלִשְׁתִּ֖ים שְׁלֹשָׁ֣ה רָאשִׁ֑ים הָרֹ֨אשׁ אֶחָ֥ד יִפְנֶ֛ה אֶל־ דֶּ֥רֶךְ עָפְרָ֖ה אֶל־ אֶ֥רֶץ שׁוּעָֽל׃ (כב) וְהָרֹ֤אשׁ אֶחָד֙ יִפְנֶ֔ה דֶּ֖רֶךְ בֵּ֣ית חֹר֑וֹן וְהָרֹ֨אשׁ אֶחָ֤ד יִפְנֶה֙ דֶּ֣רֶךְ הַגְּב֔וּל הַנִּשְׁקָ֛ף עַל־ גֵּ֥י הַצְּבֹעִ֖ים הַמִּדְבָּֽרָה׃ (כג) וְחָרָשׁ֙ לֹ֣א יִמָּצֵ֔א בְּכֹ֖ל אֶ֣רֶץ יִשְׂרָאֵ֑ל כִּֽי־ (אמר) אָמְר֣וּ פְלִשְׁתִּ֔ים פֶּ֚ן יַעֲשׂ֣וּ הָעִבְרִ֔ים חֶ֖רֶב א֥וֹ חֲנִֽית׃ (כד) וַיֵּרְד֥וּ כָל־ יִשְׂרָאֵ֖ל הַפְּלִשְׁתִּ֑ים לִ֠לְטוֹשׁ אִ֣ישׁ אֶת־ מַחֲרַשְׁתּ֤וֹ וְאֶת־ אֵתוֹ֙ וְאֶת־ קַרְדֻּמּ֔וֹ וְאֵ֖ת מַחֲרֵשָׁתֽוֹ׃ (כה) וְֽהָיְתָ֞ה הַפְּצִ֣ירָה פִ֗ים לַמַּֽחֲרֵשֹׁת֙ וְלָ֣אֵתִ֔ים וְלִשְׁלֹ֥שׁ קִלְּשׁ֖וֹן וּלְהַקַּרְדֻּמִּ֑ים וּלְהַצִּ֖יב הַדָּרְבָֽן׃ (כו) וְהָיָה֙ בְּי֣וֹם מִלְחֶ֔מֶת וְלֹ֨א נִמְצָ֜א חֶ֤רֶב וַחֲנִית֙ בְּיַ֣ד כָּל־ הָעָ֔ם אֲשֶׁ֥ר אֶת־ שָׁא֖וּל וְאֶת־ יוֹנָתָ֑ן וַתִּמָּצֵ֣א לְשָׁא֔וּל וּלְיוֹנָתָ֖ן בְּנֽוֹ׃ (כז) וַיֵּצֵא֙ מַצַּ֣ב פְּלִשְׁתִּ֔ים אֶֽל־ מַעֲבַ֖ר מִכְמָֽשׂ׃ תחילת פרק יד (כח) וַיְהִ֣י הַיּ֗וֹם וַיֹּ֨אמֶר יוֹנָתָ֤ן בֶּן־ שָׁאוּל֙ אֶל־ הַנַּ֙עַר֙ נֹשֵׂ֣א כֵלָ֔יו לְכָ֗ה וְנַעְבְּרָה֙ אֶל־ מַצַּ֣ב פְּלִשְׁתִּ֔ים אֲשֶׁ֖ר מֵעֵ֣בֶר הַלָּ֑ז וּלְאָבִ֖יו לֹ֥א הִגִּֽיד׃ (כט) וְשָׁא֗וּל יוֹשֵׁב֙ בִּקְצֵ֣ה הַגִּבְעָ֔ה תַּ֥חַת הָרִמּ֖וֹן אֲשֶׁ֣ר בְּמִגְר֑וֹן וְהָעָם֙ אֲשֶׁ֣ר עִמּ֔וֹ כְּשֵׁ֥שׁ מֵא֖וֹת אִֽישׁ׃ (ל) וַאֲחִיָּ֣ה בֶן־ אֲחִט֡וּב אֲחִ֡י אִיכָב֣וֹד ׀ בֶּן־ פִּינְחָ֨ס בֶּן־ עֵלִ֜י כֹּהֵ֧ן ׀ יְהוָ֛ה בְּשִׁל֖וֹ נֹשֵׂ֣א אֵפ֑וֹד וְהָעָם֙ לֹ֣א יָדַ֔ע כִּ֥י הָלַ֖ךְ יוֹנָתָֽן׃ (לא) וּבֵ֣ין הַֽמַּעְבְּר֗וֹת אֲשֶׁ֨ר בִּקֵּ֤שׁ יֽוֹנָתָן֙ לַֽעֲבֹר֙ עַל־ מַצַּ֣ב פְּלִשְׁתִּ֔ים שֵׁן־ הַסֶּ֤לַע מֵהָעֵ֙בֶר֙ מִזֶּ֔ה וְשֵׁן־ הַסֶּ֥לַע מֵהָעֵ֖בֶר מִזֶּ֑ה וְשֵׁ֤ם הָֽאֶחָד֙ בּוֹצֵ֔ץ וְשֵׁ֥ם הָאֶחָ֖ד סֶֽנֶּה׃ (לב) הַשֵּׁ֧ן הָאֶחָ֛ד מָצ֥וּק מִצָּפ֖וֹן מ֣וּל מִכְמָ֑שׂ וְהָאֶחָ֥ד מִנֶּ֖גֶב מ֥וּל גָּֽבַע׃ (לג) וַיֹּ֨אמֶר יְהוֹנָתָ֜ן אֶל־ הַנַּ֣עַר ׀ נֹשֵׂ֣א כֵלָ֗יו לְכָה֙ וְנַעְבְּרָ֗ה אֶל־ מַצַּב֙ הָעֲרֵלִ֣ים הָאֵ֔לֶּה אוּלַ֛י יַעֲשֶׂ֥ה יְהוָ֖ה לָ֑נוּ כִּ֣י אֵ֤ין לַֽיהוָה֙ מַעְצ֔וֹר לְהוֹשִׁ֥יעַ בְּרַ֖ב א֥וֹ בִמְעָֽט׃ (לד) וַיֹּ֤אמֶר לוֹ֙ נֹשֵׂ֣א כֵלָ֔יו עֲשֵׂ֖ה כָּל־ אֲשֶׁ֣ר בִּלְבָבֶ֑ךָ נְטֵ֣ה לָ֔ךְ הִנְנִ֥י עִמְּךָ֖ כִּלְבָבֶֽךָ׃ (לה) וַיֹּ֙אמֶר֙ יְה֣וֹנָתָ֔ן הִנֵּ֛ה אֲנַ֥חְנוּ עֹבְרִ֖ים אֶל־ הָאֲנָשִׁ֑ים וְנִגְלִ֖ינוּ אֲלֵיהֶֽם׃ (לו) אִם־ כֹּ֤ה יֹֽאמְרוּ֙ אֵלֵ֔ינוּ דֹּ֕מּוּ עַד־ הַגִּיעֵ֖נוּ אֲלֵיכֶ֑ם וְעָמַ֣דְנוּ תַחְתֵּ֔ינוּ וְלֹ֥א נַעֲלֶ֖ה אֲלֵיהֶֽם׃ (לז) וְאִם־ כֹּ֨ה יֹאמְר֜וּ עֲל֤וּ עָלֵ֙ינוּ֙ וְעָלִ֔ינוּ כִּֽי־ נְתָנָ֥ם יְהוָ֖ה יוֹנָתָ֛ן וְשָׁא֖וּל וְהָעָ֥ם יָצָֽאוּ׃ (כ) וַיֹּ֣אמֶר שָׁא֔וּל הַפִּ֕ילוּ בֵּינִ֕י וּבֵ֖ין יוֹנָתָ֣ן בְּנִ֑י וַיִּלָּכֵ֖ד יוֹנָתָֽן׃ (כא) וַיֹּ֤אמֶר שָׁאוּל֙ אֶל־ י֣וֹנָתָ֔ן הַגִּ֥ידָה לִּ֖י מֶ֣ה עָשִׂ֑יתָה וַיַּגֶּד־ ל֣וֹ יוֹנָתָ֗ן וַיֹּאמֶר֩ טָעֹ֨ם טָעַ֜מְתִּי בִּקְצֵ֨ה הַמַּטֶּ֧ה אֲשֶׁר־ בְּיָדִ֛י מְעַ֥ט דְּבַ֖שׁ הִנְנִ֥י אָמֽוּת׃ (כב) וַיֹּ֣אמֶר שָׁא֔וּל כֹּֽה־ יַעֲשֶׂ֥ה אֱלֹהִ֖ים וְכֹ֣ה יוֹסִ֑ף כִּֽי־ מ֥וֹת תָּמ֖וּת יוֹנָתָֽן׃ (כג) וַיֹּ֨אמֶר הָעָ֜ם אֶל־ שָׁא֗וּל הֲ‍ֽיוֹנָתָ֤ן ׀ יָמוּת֙ אֲשֶׁ֣ר עָ֠שָׂה הַיְשׁוּעָ֨ה הַגְּדוֹלָ֣ה הַזֹּאת֮ בְּיִשְׂרָאֵל֒ חָלִ֗ילָה חַי־ יְהוָה֙ אִם־ יִפֹּ֞ל מִשַּׂעֲרַ֤ת רֹאשׁוֹ֙ אַ֔רְצָה כִּֽי־ עִם־ אֱלֹהִ֥ים עָשָׂ֖ה הַיּ֣וֹם הַזֶּ֑ה וַיִּפְדּ֥וּ הָעָ֛ם אֶת־ יוֹנָתָ֖ן וְלֹא־ מֵֽת׃ (כד) וַיַּ֣עַל שָׁא֔וּל מֵאַחֲרֵ֖י פְּלִשְׁתִּ֑ים וּפְלִשְׁתִּ֖ים הָלְכ֥וּ לִמְקוֹמָֽם׃ (כה) וְשָׁא֛וּל לָכַ֥ד הַמְּלוּכָ֖ה עַל־ יִשְׂרָאֵ֑ל וַיִּלָּ֣חֶם סָבִ֣יב ׀ בְּֽכָל־ אֹיְבָ֡יו בְּמוֹאָ֣ב ׀ וּבִבְנֵי־ עַמּ֨וֹן וּבֶאֱד֜וֹם וּבְמַלְכֵ֤י צוֹבָה֙ וּבַפְּלִשְׁתִּ֔ים וּבְכֹ֥ל אֲשֶׁר־ יִפְנֶ֖ה יַרְשִֽׁיעַ׃ (כו) וַיַּ֣עַשׂ חַ֔יִל וַיַּ֖ךְ אֶת־ עֲמָלֵ֑ק וַיַּצֵּ֥ל אֶת־ יִשְׂרָאֵ֖ל מִיַּ֥ד שֹׁסֵֽהוּ׃ (כז) וַיִּֽהְיוּ֙ בְּנֵ֣י שָׁא֔וּל יוֹנָתָ֥ן וְיִשְׁוִ֖י וּמַלְכִּי־ שׁ֑וּעַ וְשֵׁם֙ שְׁתֵּ֣י בְנֹתָ֔יו שֵׁ֤ם הַבְּכִירָה֙ מֵרַ֔ב וְשֵׁ֥ם הַקְּטַנָּ֖ה מִיכַֽל׃ (כח) וְשֵׁם֙ אֵ֣שֶׁת שָׁא֔וּל אֲחִינֹ֖עַם בַּת־ אֲחִימָ֑עַץ וְשֵׁ֤ם שַׂר־ צְבָאוֹ֙ אֲבִינֵ֔ר בֶּן־ נֵ֖ר דּ֥וֹד שָׁאֽוּל׃ (כט) וְקִ֧ישׁ אֲבִֽי־ שָׁא֛וּל וְנֵ֥ר אֲבִֽי־ אַבְנֵ֖ר בֶּן־ אֲבִיאֵֽל׃ (ל) וַתְּהִ֤י הַמִּלְחָמָה֙ חֲזָקָ֣ה עַל־ פְּלִשְׁתִּ֔ים כֹּ֖ל יְמֵ֣י שָׁא֑וּל וְרָאָ֨ה שָׁא֜וּל כָּל־ אִ֤ישׁ גִּבּוֹר֙ וְכָל־ בֶּן־ חַ֔יִל וַיַּאַסְפֵ֖הוּ אֵלָֽיו׃ סוף פרק יד (לא) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ שָׁא֔וּל אֹתִ֨י שָׁלַ֤ח יְהוָה֙ לִמְשָׁחֳךָ֣ לְמֶ֔לֶךְ עַל־ עַמּ֖וֹ עַל־ יִשְׂרָאֵ֑ל וְעַתָּ֣ה שְׁמַ֔ע לְק֖וֹל דִּבְרֵ֥י יְהוָֽה׃ (לב) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת פָּקַ֕דְתִּי אֵ֛ת אֲשֶׁר־ עָשָׂ֥ה עֲמָלֵ֖ק לְיִשְׂרָאֵ֑ל אֲשֶׁר־ שָׂ֥ם לוֹ֙ בַּדֶּ֔רֶךְ בַּעֲלֹת֖וֹ מִמִּצְרָֽיִם׃ (לג) עַתָּה֩ לֵ֨ךְ וְהִכִּֽיתָ֜ה אֶת־ עֲמָלֵ֗ק וְהַֽחֲרַמְתֶּם֙ אֶת־ כָּל־ אֲשֶׁר־ ל֔וֹ וְלֹ֥א תַחְמֹ֖ל עָלָ֑יו וְהֵמַתָּ֞ה מֵאִ֣ישׁ עַד־ אִשָּׁ֗ה מֵֽעֹלֵל֙ וְעַד־ יוֹנֵ֔ק מִשּׁ֣וֹר וְעַד־ שֶׂ֔ה מִגָּמָ֖ל וְעַד־ חֲמֽוֹר׃ (לד) וַיְשַׁמַּ֤ע שָׁאוּל֙ אֶת־ הָעָ֔ם וַֽיִּפְקְדֵם֙ בַּטְּלָאִ֔ים מָאתַ֥יִם אֶ֖לֶף רַגְלִ֑י וַעֲשֶׂ֥רֶת אֲלָפִ֖ים אֶת־ אִ֥ישׁ יְהוּדָֽה׃ (לה) וַיָּבֹ֥א שָׁא֖וּל עַד־ עִ֣יר עֲמָלֵ֑ק וַיָּ֖רֶב בַּנָּֽחַל׃ (לו) וַיֹּ֣אמֶר שָׁא֣וּל אֶֽל־ הַקֵּינִ֡י לְכוּ֩ סֻּ֨רוּ רְד֜וּ מִתּ֣וֹךְ עֲמָלֵקִ֗י פֶּן־ אֹֽסִפְךָ֙ עִמּ֔וֹ וְאַתָּ֞ה עָשִׂ֤יתָה חֶ֙סֶד֙ עִם־ כָּל־ בְּנֵ֣י יִשְׂרָאֵ֔ל בַּעֲלוֹתָ֖ם מִמִּצְרָ֑יִם וַיָּ֥סַר קֵינִ֖י מִתּ֥וֹךְ עֲמָלֵֽק׃ (לז) וַיַּ֥ךְ שָׁא֖וּל אֶת־ עֲמָלֵ֑ק מֵֽחֲוִילָה֙ בּוֹאֲךָ֣ שׁ֔וּר אֲשֶׁ֖ר עַל־ פְּנֵ֥י מִצְרָֽיִם׃ (לח) וַיִּתְפֹּ֛שׂ אֶת־ אֲגַ֥ג מֶֽלֶךְ־ עֲמָלֵ֖ק חָ֑י וְאֶת־ כָּל־ הָעָ֖ם הֶחֱרִ֥ים לְפִי־ חָֽרֶב׃ (לט) וַיַּחְמֹל֩ שָׁא֨וּל וְהָעָ֜ם עַל־ אֲגָ֗ג וְעַל־ מֵיטַ֣ב הַצֹּאן֩ וְהַבָּקָ֨ר וְהַמִּשְׁנִ֤ים וְעַל־ הַכָּרִים֙ וְעַל־ כָּל־ הַטּ֔וֹב וְלֹ֥א אָב֖וּ הַחֲרִימָ֑ם וְכָל־ הַמְּלָאכָ֛ה נְמִבְזָ֥ה וְנָמֵ֖ס אֹתָ֥הּ הֶחֱרִֽימוּ׃ (מ) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ שְׁמוּאֵ֖ל לֵאמֹֽר׃ (מא) נִחַ֗מְתִּי כִּֽי־ הִמְלַ֤כְתִּי אֶת־ שָׁאוּל֙ לְמֶ֔לֶךְ כִּֽי־ שָׁב֙ מֵאַֽחֲרַ֔י וְאֶת־ דְּבָרַ֖י לֹ֣א הֵקִ֑ים וַיִּ֙חַר֙ לִשְׁמוּאֵ֔ל וַיִּזְעַ֥ק אֶל־ יְהוָ֖ה כָּל־ הַלָּֽיְלָה׃ (מב) וַיַּשְׁכֵּ֧ם שְׁמוּאֵ֛ל לִקְרַ֥את שָׁא֖וּל בַּבֹּ֑קֶר וַיֻּגַּ֨ד לִשְׁמוּאֵ֜ל לֵאמֹ֗ר בָּֽא־ שָׁא֤וּל הַכַּרְמֶ֙לָה֙ וְהִנֵּ֨ה מַצִּ֥יב לוֹ֙ יָ֔ד וַיִּסֹּב֙ וַֽיַּעֲבֹ֔ר וַיֵּ֖רֶד הַגִּלְגָּֽל׃ (מג) וַיָּבֹ֥א שְׁמוּאֵ֖ל אֶל־ שָׁא֑וּל וַיֹּ֧אמֶר ל֣וֹ שָׁא֗וּל בָּר֤וּךְ אַתָּה֙ לַֽיהוָ֔ה הֲקִימֹ֖תִי אֶת־ דְּבַ֥ר יְהוָֽה׃ (מד) וַיֹּ֣אמֶר שְׁמוּאֵ֔ל וּמֶ֛ה קֽוֹל־ הַצֹּ֥אן הַזֶּ֖ה בְּאָזְנָ֑י וְק֣וֹל הַבָּקָ֔ר אֲשֶׁ֥ר אָנֹכִ֖י שֹׁמֵֽעַ׃ (מה) וַיֹּ֨אמֶר שָׁא֜וּל מֵעֲמָלֵקִ֣י הֱבִיא֗וּם אֲשֶׁ֨ר חָמַ֤ל הָעָם֙ עַל־ מֵיטַ֤ב הַצֹּאן֙ וְהַבָּקָ֔ר לְמַ֥עַן זְבֹ֖חַ לַיהוָ֣ה אֱלֹהֶ֑יךָ וְאֶת־ הַיּוֹתֵ֖ר הֶחֱרַֽמְנוּ׃ (מו) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ שָׁא֔וּל הֶ֚רֶף וְאַגִּ֣ידָה לְּךָ֔ אֵת֩ אֲשֶׁ֨ר דִּבֶּ֧ר יְהוָ֛ה אֵלַ֖י הַלָּ֑יְלָה (ויאמרו) וַיֹּ֥אמֶר ל֖וֹ דַּבֵּֽר׃
