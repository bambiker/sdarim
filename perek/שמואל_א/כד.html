שמואל א פרק כד לפי החלוקה לסדרים

פרק כד בספר שמואל א (23 פסוקים) נמצא בשמואל סדר טו (פסוקים כו–מד) ובשמואל סדר טז (פסוקים א–ד).

שמואל סדר טו

(א) וַיּ֨וֹסֶף ע֤וֹד דָּוִד֙ לִשְׁאֹ֣ל בַּֽיהוָ֔ה וַֽיַּעֲנֵ֖הוּ יְהוָ֑ה וַיֹּ֗אמֶר ק֚וּם רֵ֣ד קְעִילָ֔ה כִּֽי־ אֲנִ֥י נֹתֵ֛ן אֶת־ פְּלִשְׁתִּ֖ים בְּיָדֶֽךָ׃ (ב) וַיֵּ֣לֶךְ דָּוִד֩ (ואנשו) וַאֲנָשָׁ֨יו קְעִילָ֜ה וַיִּלָּ֣חֶם בַּפְּלִשְׁתִּ֗ים וַיִּנְהַג֙ אֶת־ מִקְנֵיהֶ֔ם וַיַּ֥ךְ בָּהֶ֖ם מַכָּ֣ה גְדוֹלָ֑ה וַיֹּ֣שַׁע דָּוִ֔ד אֵ֖ת יֹשְׁבֵ֥י קְעִילָֽה׃ (ג) וַיְהִ֗י בִּ֠בְרֹחַ אֶבְיָתָ֧ר בֶּן־ אֲחִימֶ֛לֶךְ אֶל־ דָּוִ֖ד קְעִילָ֑ה אֵפ֖וֹד יָרַ֥ד בְּיָדֽוֹ׃ (ד) וַיֻּגַּ֣ד לְשָׁא֔וּל כִּי־ בָ֥א דָוִ֖ד קְעִילָ֑ה וַיֹּ֣אמֶר שָׁא֗וּל נִכַּ֨ר אֹת֤וֹ אֱלֹהִים֙ בְּיָדִ֔י כִּ֚י נִסְגַּ֣ר לָב֔וֹא בְּעִ֖יר דְּלָתַ֥יִם וּבְרִֽיחַ׃ (ה) וַיְשַׁמַּ֥ע שָׁא֛וּל אֶת־ כָּל־ הָעָ֖ם לַמִּלְחָמָ֑ה לָרֶ֣דֶת קְעִילָ֔ה לָצ֥וּר אֶל־ דָּוִ֖ד וְאֶל־ אֲנָשָֽׁיו׃ (ו) וַיֵּ֣דַע דָּוִ֔ד כִּ֣י עָלָ֔יו שָׁא֖וּל מַחֲרִ֣ישׁ הָרָעָ֑ה וַיֹּ֙אמֶר֙ אֶל־ אֶבְיָתָ֣ר הַכֹּהֵ֔ן הַגִּ֖ישָׁה הָאֵפֽוֹד׃ (ז) וַיֹּאמֶר֮ דָּוִד֒ יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל שָׁמֹ֤עַ שָׁמַע֙ עַבְדְּךָ֔ כִּֽי־ מְבַקֵּ֥שׁ שָׁא֖וּל לָב֣וֹא אֶל־ קְעִילָ֑ה לְשַׁחֵ֥ת לָעִ֖יר בַּעֲבוּרִֽי׃ (ח) הֲיַסְגִּרֻ֣נִי בַעֲלֵי֩ קְעִילָ֨ה בְיָד֜וֹ הֲיֵרֵ֣ד שָׁא֗וּל כַּֽאֲשֶׁר֙ שָׁמַ֣ע עַבְדֶּ֔ךָ יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הַגֶּד־ נָ֖א לְעַבְדֶּ֑ךָ וַיֹּ֥אמֶר יְהוָ֖ה יֵרֵֽד׃ (ט) וַיֹּ֣אמֶר דָּוִ֔ד הֲיַסְגִּ֜רוּ בַּעֲלֵ֧י קְעִילָ֛ה אֹתִ֥י וְאֶת־ אֲנָשַׁ֖י בְּיַד־ שָׁא֑וּל וַיֹּ֥אמֶר יְהוָ֖ה יַסְגִּֽירוּ׃ (י) וַיָּקָם֩ דָּוִ֨ד וַאֲנָשָׁ֜יו כְּשֵׁשׁ־ מֵא֣וֹת אִ֗ישׁ וַיֵּצְאוּ֙ מִקְּעִלָ֔ה וַיִּֽתְהַלְּכ֖וּ בַּאֲשֶׁ֣ר יִתְהַלָּ֑כוּ וּלְשָׁא֣וּל הֻגַּ֗ד כִּֽי־ נִמְלַ֤ט דָּוִד֙ מִקְּעִילָ֔ה וַיֶּחְדַּ֖ל לָצֵֽאת׃ (יא) וַיֵּ֨שֶׁב דָּוִ֤ד בַּמִּדְבָּר֙ בַּמְּצָד֔וֹת וַיֵּ֥שֶׁב בָּהָ֖ר בְּמִדְבַּר־ זִ֑יף וַיְבַקְשֵׁ֤הוּ שָׁאוּל֙ כָּל־ הַיָּמִ֔ים וְלֹֽא־ נְתָנ֥וֹ אֱלֹהִ֖ים בְּיָדֽוֹ׃ (יב) וַיַּ֣רְא דָוִ֔ד כִּֽי־ יָצָ֥א שָׁא֖וּל לְבַקֵּ֣שׁ אֶת־ נַפְשׁ֑וֹ וְדָוִ֥ד בְּמִדְבַּר־ זִ֖יף בַּחֹֽרְשָׁה׃ (יג) וַיָּ֙קָם֙ יְהוֹנָתָ֣ן בֶּן־ שָׁא֔וּל וַיֵּ֥לֶךְ אֶל־ דָּוִ֖ד חֹ֑רְשָׁה וַיְחַזֵּ֥ק אֶת־ יָד֖וֹ בֵּאלֹהִֽים׃ (יד) וַיֹּ֨אמֶר אֵלָ֜יו אַל־ תִּירָ֗א כִּ֠י לֹ֤א תִֽמְצָאֲךָ֙ יַ֚ד שָׁא֣וּל אָבִ֔י וְאַתָּה֙ תִּמְלֹ֣ךְ עַל־ יִשְׂרָאֵ֔ל וְאָנֹכִ֖י אֶֽהְיֶה־ לְּךָ֣ לְמִשְׁנֶ֑ה וְגַם־ שָׁא֥וּל אָבִ֖י יֹדֵ֥עַ כֵּֽן׃ (טו) וַיִּכְרְת֧וּ שְׁנֵיהֶ֛ם בְּרִ֖ית לִפְנֵ֣י יְהוָ֑ה וַיֵּ֤שֶׁב דָּוִד֙ בַּחֹ֔רְשָׁה וִיהוֹנָתָ֖ן הָלַ֥ךְ לְבֵיתֽוֹ׃ (טז) וַיַּעֲל֤וּ זִפִים֙ אֶל־ שָׁא֔וּל הַגִּבְעָ֖תָה לֵאמֹ֑ר הֲל֣וֹא דָ֠וִד מִסְתַּתֵּ֨ר עִמָּ֤נוּ בַמְּצָדוֹת֙ בַּחֹ֔רְשָׁה בְּגִבְעַת֙ הַֽחֲכִילָ֔ה אֲשֶׁ֖ר מִימִ֥ין הַיְשִׁימֽוֹן׃ (יז) וְ֠עַתָּה לְכָל־ אַוַּ֨ת נַפְשְׁךָ֥ הַמֶּ֛לֶךְ לָרֶ֖דֶת רֵ֑ד וְלָ֥נוּ הַסְגִּיר֖וֹ בְּיַ֥ד הַמֶּֽלֶךְ׃ (יח) וַיֹּ֣אמֶר שָׁא֔וּל בְּרוּכִ֥ים אַתֶּ֖ם לַֽיהוָ֑ה כִּ֥י חֲמַלְתֶּ֖ם עָלָֽי׃ (יט) לְכוּ־ נָ֞א הָכִ֣ינוּ ע֗וֹד וּדְע֤וּ וּרְאוּ֙ אֶת־ מְקוֹמוֹ֙ אֲשֶׁ֣ר תִּֽהְיֶ֣ה רַגְל֔וֹ מִ֥י רָאָ֖הוּ שָׁ֑ם כִּ֚י אָמַ֣ר אֵלַ֔י עָר֥וֹם יַעְרִ֖ם הֽוּא׃ (כ) וּרְא֣וּ וּדְע֗וּ מִכֹּ֤ל הַמַּֽחֲבֹאִים֙ אֲשֶׁ֣ר יִתְחַבֵּ֣א שָׁ֔ם וְשַׁבְתֶּ֤ם אֵלַי֙ אֶל־ נָכ֔וֹן וְהָלַכְתִּ֖י אִתְּכֶ֑ם וְהָיָה֙ אִם־ יֶשְׁנ֣וֹ בָאָ֔רֶץ וְחִפַּשְׂתִּ֣י אֹת֔וֹ בְּכֹ֖ל אַלְפֵ֥י יְהוּדָֽה׃ (כא) וַיָּק֛וּמוּ וַיֵּלְכ֥וּ זִ֖יפָה לִפְנֵ֣י שָׁא֑וּל וְדָוִ֨ד וַאֲנָשָׁ֜יו בְּמִדְבַּ֤ר מָעוֹן֙ בָּעֲרָבָ֔ה אֶ֖ל יְמִ֥ין הַיְשִׁימֽוֹן׃ (כב) וַיֵּ֨לֶךְ שָׁא֣וּל וַאֲנָשָׁיו֮ לְבַקֵּשׁ֒ וַיַּגִּ֣דוּ לְדָוִ֔ד וַיֵּ֣רֶד הַסֶּ֔לַע וַיֵּ֖שֶׁב בְּמִדְבַּ֣ר מָע֑וֹן וַיִּשְׁמַ֣ע שָׁא֔וּל וַיִּרְדֹּ֥ף אַחֲרֵֽי־ דָוִ֖ד מִדְבַּ֥ר מָעֽוֹן׃ (כג) וַיֵּ֨לֶךְ שָׁא֜וּל מִצַּ֤ד הָהָר֙ מִזֶּ֔ה וְדָוִ֧ד וַאֲנָשָׁ֛יו מִצַּ֥ד הָהָ֖ר מִזֶּ֑ה וַיְהִ֨י דָוִ֜ד נֶחְפָּ֤ז לָלֶ֙כֶת֙ מִפְּנֵ֣י שָׁא֔וּל וְשָׁא֣וּל וַאֲנָשָׁ֗יו עֹֽטְרִ֛ים אֶל־ דָּוִ֥ד וְאֶל־ אֲנָשָׁ֖יו לְתָפְשָֽׂם׃ (כד) וּמַלְאָ֣ךְ בָּ֔א אֶל־ שָׁא֖וּל לֵאמֹ֑ר מַהֲרָ֣ה וְלֵ֔כָה כִּֽי־ פָשְׁט֥וּ פְלִשְׁתִּ֖ים עַל־ הָאָֽרֶץ׃ (כה) וַיָּ֣שָׁב שָׁא֗וּל מִרְדֹף֙ אַחֲרֵ֣י דָוִ֔ד וַיֵּ֖לֶךְ לִקְרַ֣את פְּלִשְׁתִּ֑ים עַל־ כֵּ֗ן קָֽרְאוּ֙ לַמָּק֣וֹם הַה֔וּא סֶ֖לַע הַֽמַּחְלְקֽוֹת׃ תחילת פרק כד (כו) וַיַּ֥עַל דָּוִ֖ד מִשָּׁ֑ם וַיֵּ֖שֶׁב בִּמְצָד֥וֹת עֵֽין־ גֶּֽדִי׃ (כז) וַיְהִ֗י כַּֽאֲשֶׁר֙ שָׁ֣ב שָׁא֔וּל מֵאַחֲרֵ֖י פְּלִשְׁתִּ֑ים וַיַּגִּ֤דוּ לוֹ֙ לֵאמֹ֔ר הִנֵּ֣ה דָוִ֔ד בְּמִדְבַּ֖ר עֵ֥ין גֶּֽדִי׃ (כח) וַיִּקַּ֣ח שָׁא֗וּל שְׁלֹ֧שֶׁת אֲלָפִ֛ים אִ֥ישׁ בָּח֖וּר מִכָּל־ יִשְׂרָאֵ֑ל וַיֵּ֗לֶךְ לְבַקֵּ֤שׁ אֶת־ דָּוִד֙ וַֽאֲנָשָׁ֔יו עַל־ פְּנֵ֖י צוּרֵ֥י הַיְּעֵלִֽים׃ (כט) וַ֠יָּבֹא אֶל־ גִּדְר֨וֹת הַצֹּ֤אן עַל־ הַדֶּ֙רֶךְ֙ וְשָׁ֣ם מְעָרָ֔ה וַיָּבֹ֥א שָׁא֖וּל לְהָסֵ֣ךְ אֶת־ רַגְלָ֑יו וְדָוִד֙ וַאֲנָשָׁ֔יו בְּיַרְכְּתֵ֥י הַמְּעָרָ֖ה יֹשְׁבִֽים׃ (ל) וַיֹּאמְרוּ֩ אַנְשֵׁ֨י דָוִ֜ד אֵלָ֗יו הִנֵּ֨ה הַיּ֜וֹם אֲ‍ֽשֶׁר־ אָמַ֧ר יְהוָ֣ה אֵלֶ֗יךָ הִנֵּ֨ה אָנֹכִ֜י נֹתֵ֤ן אֶת־ (איביך) אֹֽיִבְךָ֙ בְּיָדֶ֔ךָ וְעָשִׂ֣יתָ לּ֔וֹ כַּאֲשֶׁ֖ר יִטַ֣ב בְּעֵינֶ֑יךָ וַיָּ֣קָם דָּוִ֗ד וַיִּכְרֹ֛ת אֶת־ כְּנַֽף־ הַמְּעִ֥יל אֲשֶׁר־ לְשָׁא֖וּל בַּלָּֽט׃ (לא) וַֽיְהִי֙ אַֽחֲרֵי־ כֵ֔ן וַיַּ֥ךְ לֵב־ דָּוִ֖ד אֹת֑וֹ עַ֚ל אֲשֶׁ֣ר כָּרַ֔ת אֶת־ כָּנָ֖ף אֲשֶׁ֥ר לְשָׁאֽוּל׃ (לב) וַיֹּ֨אמֶר לַאֲנָשָׁ֜יו חָלִ֧ילָה לִּ֣י מֵֽיהוָ֗ה אִם־ אֶעֱשֶׂה֩ אֶת־ הַדָּבָ֨ר הַזֶּ֤ה לַֽאדֹנִי֙ לִמְשִׁ֣יחַ יְהוָ֔ה לִשְׁלֹ֥חַ יָדִ֖י בּ֑וֹ כִּֽי־ מְשִׁ֥יחַ יְהוָ֖ה הֽוּא׃ (לג) וַיְשַׁסַּ֨ע דָּוִ֤ד אֶת־ אֲנָשָׁיו֙ בַּדְּבָרִ֔ים וְלֹ֥א נְתָנָ֖ם לָק֣וּם אֶל־ שָׁא֑וּל וְשָׁא֛וּל קָ֥ם מֵהַמְּעָרָ֖ה וַיֵּ֥לֶךְ בַּדָּֽרֶךְ׃ (לד) וַיָּ֨קָם דָּוִ֜ד אַחֲרֵי־ כֵ֗ן וַיֵּצֵא֙ (מן)־ (המערה) מֵֽהַמְּעָרָ֔ה וַיִּקְרָ֧א אַֽחֲרֵי־ שָׁא֛וּל לֵאמֹ֖ר אֲדֹנִ֣י הַמֶּ֑לֶךְ וַיַּבֵּ֤ט שָׁאוּל֙ אַֽחֲרָ֔יו וַיִּקֹּ֨ד דָּוִ֥ד אַפַּ֛יִם אַ֖רְצָה וַיִּשְׁתָּֽחוּ׃ (לה) וַיֹּ֤אמֶר דָּוִד֙ לְשָׁא֔וּל לָ֧מָּה תִשְׁמַ֛ע אֶת־ דִּבְרֵ֥י אָדָ֖ם לֵאמֹ֑ר הִנֵּ֣ה דָוִ֔ד מְבַקֵּ֖שׁ רָעָתֶֽךָ׃ (לו) הִנֵּה֩ הַיּ֨וֹם הַזֶּ֜ה רָא֣וּ עֵינֶ֗יךָ אֵ֣ת אֲשֶׁר־ נְתָנְךָ֩ יְהוָ֨ה ׀ הַיּ֤וֹם ׀ בְּיָדִי֙ בַּמְּעָרָ֔ה וְאָמַ֥ר לַהֲרָגֲךָ֖ וַתָּ֣חָס עָלֶ֑יךָ וָאֹמַ֗ר לֹא־ אֶשְׁלַ֤ח יָדִי֙ בַּֽאדֹנִ֔י כִּי־ מְשִׁ֥יחַ יְהוָ֖ה הֽוּא׃ (לז) אַךְ֩ לַשֶּׁ֨קֶר שָׁמַ֜רְתִּי אֶֽת־ כָּל־ אֲשֶׁ֤ר לָזֶה֙ בַּמִּדְבָּ֔ר וְלֹא־ נִפְקַ֥ד מִכָּל־ אֲשֶׁר־ ל֖וֹ מְא֑וּמָה וַיָּֽשֶׁב־ לִ֥י רָעָ֖ה תַּ֥חַת טוֹבָֽה׃ (כו) כֹּה־ יַעֲשֶׂ֧ה אֱלֹהִ֛ים לְאֹיְבֵ֥י דָוִ֖ד וְכֹ֣ה יֹסִ֑יף אִם־ אַשְׁאִ֧יר מִכָּל־ אֲשֶׁר־ ל֛וֹ עַד־ הַבֹּ֖קֶר מַשְׁתִּ֥ין בְּקִֽיר׃ (כז) וַתֵּ֤רֶא אֲבִיגַ֙יִל֙ אֶת־ דָּוִ֔ד וַתְּמַהֵ֕ר וַתֵּ֖רֶד מֵעַ֣ל הַחֲמ֑וֹר וַתִּפֹּ֞ל לְאַפֵּ֤י דָוִד֙ עַל־ פָּנֶ֔יהָ וַתִּשְׁתַּ֖חוּ אָֽרֶץ׃ (כח) וַתִּפֹּל֙ עַל־ רַגְלָ֔יו וַתֹּ֕אמֶר בִּי־ אֲנִ֥י אֲדֹנִ֖י הֶֽעָוֺ֑ן וּֽתְדַבֶּר־ נָ֤א אֲמָֽתְךָ֙ בְּאָזְנֶ֔יךָ וּשְׁמַ֕ע אֵ֖ת דִּבְרֵ֥י אֲמָתֶֽךָ׃ (כט) אַל־ נָ֣א יָשִׂ֣ים אֲדֹנִ֣י ׀ אֶת־ לִבּ֡וֹ אֶל־ אִישׁ֩ הַבְּלִיַּ֨עַל הַזֶּ֜ה עַל־ נָבָ֗ל כִּ֤י כִשְׁמוֹ֙ כֶּן־ ה֔וּא נָבָ֣ל שְׁמ֔וֹ וּנְבָלָ֖ה עִמּ֑וֹ וַֽאֲנִי֙ אֲמָ֣תְךָ֔ לֹ֥א רָאִ֛יתִי אֶת־ נַעֲרֵ֥י אֲדֹנִ֖י אֲשֶׁ֥ר שָׁלָֽחְתָּ׃ (ל) וְעַתָּ֣ה אֲדֹנִ֗י חַי־ יְהוָ֤ה וְחֵֽי־ נַפְשְׁךָ֙ אֲשֶׁ֨ר מְנָעֲךָ֤ יְהוָה֙ מִבּ֣וֹא בְדָמִ֔ים וְהוֹשֵׁ֥עַ יָדְךָ֖ לָ֑ךְ וְעַתָּ֗ה יִֽהְי֤וּ כְנָבָל֙ אֹיְבֶ֔יךָ וְהַֽמְבַקְשִׁ֥ים אֶל־ אֲדֹנִ֖י רָעָֽה׃ (לא) וְעַתָּה֙ הַבְּרָכָ֣ה הַזֹּ֔את אֲשֶׁר־ הֵבִ֥יא שִׁפְחָתְךָ֖ לַֽאדֹנִ֑י וְנִתְּנָה֙ לַנְּעָרִ֔ים הַמִּֽתְהַלְּכִ֖ים בְּרַגְלֵ֥י אֲדֹנִֽי׃ (לב) שָׂ֥א נָ֖א לְפֶ֣שַׁע אֲמָתֶ֑ךָ כִּ֣י עָשֹֽׂה־ יַעֲשֶׂה֩ יְהוָ֨ה לַֽאדֹנִ֜י בַּ֣יִת נֶאֱמָ֗ן כִּי־ מִלְחֲמ֤וֹת יְהוָה֙ אֲדֹנִ֣י נִלְחָ֔ם וְרָעָ֛ה לֹא־ תִמָּצֵ֥א בְךָ֖ מִיָּמֶֽיךָ׃ (לג) וַיָּ֤קָם אָדָם֙ לִרְדָפְךָ֔ וּלְבַקֵּ֖שׁ אֶת־ נַפְשֶׁ֑ךָ וְֽהָיְתָה֩ נֶ֨פֶשׁ אֲדֹנִ֜י צְרוּרָ֣ה ׀ בִּצְר֣וֹר הַחַיִּ֗ים אֵ֚ת יְהוָ֣ה אֱלֹהֶ֔יךָ וְאֵ֨ת נֶ֤פֶשׁ אֹיְבֶ֙יךָ֙ יְקַלְּעֶ֔נָּה בְּת֖וֹךְ כַּ֥ף הַקָּֽלַע׃ (לד) וְהָיָ֗ה כִּֽי־ יַעֲשֶׂ֤ה יְהוָה֙ לַֽאדֹנִ֔י כְּכֹ֛ל אֲשֶׁר־ דִּבֶּ֥ר אֶת־ הַטּוֹבָ֖ה עָלֶ֑יךָ וְצִוְּךָ֥ לְנָגִ֖יד עַל־ יִשְׂרָאֵֽל׃ (לה) וְלֹ֣א תִהְיֶ֣ה זֹ֣את ׀ לְךָ֡ לְפוּקָה֩ וּלְמִכְשׁ֨וֹל לֵ֜ב לַאדֹנִ֗י וְלִשְׁפָּךְ־ דָּם֙ חִנָּ֔ם וּלְהוֹשִׁ֥יעַ אֲדֹנִ֖י ל֑וֹ וְהֵיטִ֤ב יְהוָה֙ לַֽאדֹנִ֔י וְזָכַרְתָּ֖ אֶת־ אֲמָתֶֽךָ׃ (לו) וַיֹּ֥אמֶר דָּוִ֖ד לַאֲבִיגַ֑ל בָּר֤וּךְ יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֧ר שְׁלָחֵ֛ךְ הַיּ֥וֹם הַזֶּ֖ה לִקְרָאתִֽי׃
