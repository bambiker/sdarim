שמואל א פרק יח לפי החלוקה לסדרים

פרק יח בספר שמואל א (30 פסוקים) נמצא בשמואל סדר יא (פסוקים כג–לה) ובשמואל סדר יב (פסוקים א–יז).

שמואל סדר יא

(א) וַיֹּאמֶר֮ דָּוִד֒ יְהוָ֗ה אֲשֶׁ֨ר הִצִּלַ֜נִי מִיַּ֤ד הָֽאֲרִי֙ וּמִיַּ֣ד הַדֹּ֔ב ה֣וּא יַצִּילֵ֔נִי מִיַּ֥ד הַפְּלִשְׁתִּ֖י הַזֶּ֑ה וַיֹּ֨אמֶר שָׁא֤וּל אֶל־ דָּוִד֙ לֵ֔ךְ וַֽיהוָ֖ה יִהְיֶ֥ה עִמָּֽךְ׃ (ב) וַיַּלְבֵּ֨שׁ שָׁא֤וּל אֶת־ דָּוִד֙ מַדָּ֔יו וְנָתַ֛ן ק֥וֹבַע נְחֹ֖שֶׁת עַל־ רֹאשׁ֑וֹ וַיַּלְבֵּ֥שׁ אֹת֖וֹ שִׁרְיֽוֹן׃ (ג) וַיַּחְגֹּ֣ר דָּוִ֣ד אֶת־ חַ֠רְבּוֹ מֵעַ֨ל לְמַדָּ֜יו וַיֹּ֣אֶל לָלֶכֶת֮ כִּ֣י לֹֽא־ נִסָּה֒ וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ שָׁא֗וּל לֹ֥א אוּכַ֛ל לָלֶ֥כֶת בָּאֵ֖לֶּה כִּ֣י לֹ֣א נִסִּ֑יתִי וַיְסִרֵ֥ם דָּוִ֖ד מֵעָלָֽיו׃ (ד) וַיִּקַּ֨ח מַקְל֜וֹ בְּיָד֗וֹ וַיִּבְחַר־ ל֣וֹ חֲמִשָּׁ֣ה חַלֻּקֵֽי־ אֲבָנִ֣ים ׀ מִן־ הַנַּ֡חַל וַיָּ֣שֶׂם אֹ֠תָם בִּכְלִ֨י הָרֹעִ֧ים אֲשֶׁר־ ל֛וֹ וּבַיַּלְק֖וּט וְקַלְּע֣וֹ בְיָד֑וֹ וַיִּגַּ֖שׁ אֶל־ הַפְּלִשְׁתִּֽי׃ (ה) וַיֵּ֙לֶךְ֙ הַפְּלִשְׁתִּ֔י הֹלֵ֥ךְ וְקָרֵ֖ב אֶל־ דָּוִ֑ד וְהָאִ֛ישׁ נֹשֵׂ֥א הַצִּנָּ֖ה לְפָנָֽיו׃ (ו) וַיַּבֵּ֧ט הַפְּלִשְׁתִּ֛י וַיִּרְאֶ֥ה אֶת־ דָּוִ֖ד וַיִּבְזֵ֑הוּ כִּֽי־ הָיָ֣ה נַ֔עַר וְאַדְמֹנִ֖י עִם־ יְפֵ֥ה מַרְאֶֽה׃ (ז) וַיֹּ֤אמֶר הַפְּלִשְׁתִּי֙ אֶל־ דָּוִ֔ד הֲכֶ֣לֶב אָנֹ֔כִי כִּֽי־ אַתָּ֥ה בָֽא־ אֵלַ֖י בַּמַּקְל֑וֹת וַיְקַלֵּ֧ל הַפְּלִשְׁתִּ֛י אֶת־ דָּוִ֖ד בֵּאלֹהָֽיו׃ (ח) וַיֹּ֥אמֶר הַפְּלִשְׁתִּ֖י אֶל־ דָּוִ֑ד לְכָ֣ה אֵלַ֔י וְאֶתְּנָה֙ אֶת־ בְּשָׂ֣רְךָ֔ לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֥ת הַשָּׂדֶֽה׃ (ט) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ הַפְּלִשְׁתִּ֔י אַתָּה֙ בָּ֣א אֵלַ֔י בְּחֶ֖רֶב וּבַחֲנִ֣ית וּבְכִיד֑וֹן וְאָנֹכִ֣י בָֽא־ אֵלֶ֗יךָ בְּשֵׁם֙ יְהוָ֣ה צְבָא֔וֹת אֱלֹהֵ֛י מַעַרְכ֥וֹת יִשְׂרָאֵ֖ל אֲשֶׁ֥ר חֵרַֽפְתָּ׃ (י) הַיּ֣וֹם הַזֶּ֡ה יְסַגֶּרְךָ֩ יְהוָ֨ה בְּיָדִ֜י וְהִכִּיתִ֗ךָ וַהֲסִרֹתִ֤י אֶת־ רֹֽאשְׁךָ֙ מֵעָלֶ֔יךָ וְנָ֨תַתִּ֜י פֶּ֣גֶר מַחֲנֵ֤ה פְלִשְׁתִּים֙ הַיּ֣וֹם הַזֶּ֔ה לְע֥וֹף הַשָּׁמַ֖יִם וּלְחַיַּ֣ת הָאָ֑רֶץ וְיֵֽדְעוּ֙ כָּל־ הָאָ֔רֶץ כִּ֛י יֵ֥שׁ אֱלֹהִ֖ים לְיִשְׂרָאֵֽל׃ (יא) וְיֵֽדְעוּ֙ כָּל־ הַקָּהָ֣ל הַזֶּ֔ה כִּֽי־ לֹ֛א בְּחֶ֥רֶב וּבַחֲנִ֖ית יְהוֹשִׁ֣יעַ יְהוָ֑ה כִּ֤י לַֽיהוָה֙ הַמִּלְחָמָ֔ה וְנָתַ֥ן אֶתְכֶ֖ם בְּיָדֵֽנוּ׃ (יב) וְהָיָה֙ כִּֽי־ קָ֣ם הַפְּלִשְׁתִּ֔י וַיֵּ֥לֶךְ וַיִּקְרַ֖ב לִקְרַ֣את דָּוִ֑ד וַיְמַהֵ֣ר דָּוִ֔ד וַיָּ֥רָץ הַמַּעֲרָכָ֖ה לִקְרַ֥את הַפְּלִשְׁתִּֽי׃ (יג) וַיִּשְׁלַח֩ דָּוִ֨ד אֶת־ יָד֜וֹ אֶל־ הַכֶּ֗לִי וַיִּקַּ֨ח מִשָּׁ֥ם אֶ֙בֶן֙ וַיְקַלַּ֔ע וַיַּ֥ךְ אֶת־ הַפְּלִשְׁתִּ֖י אֶל־ מִצְח֑וֹ וַתִּטְבַּ֤ע הָאֶ֙בֶן֙ בְּמִצְח֔וֹ וַיִּפֹּ֥ל עַל־ פָּנָ֖יו אָֽרְצָה׃ (יד) וַיֶּחֱזַ֨ק דָּוִ֤ד מִן־ הַפְּלִשְׁתִּי֙ בַּקֶּ֣לַע וּבָאֶ֔בֶן וַיַּ֥ךְ אֶת־ הַפְּלִשְׁתִּ֖י וַיְמִיתֵ֑הוּ וְחֶ֖רֶב אֵ֥ין בְּיַד־ דָּוִֽד׃ (טו) וַיָּ֣רָץ דָּ֠וִד וַיַּעֲמֹ֨ד אֶל־ הַפְּלִשְׁתִּ֜י וַיִּקַּ֣ח אֶת־ חַ֠רְבּוֹ וַֽיִּשְׁלְפָ֤הּ מִתַּעְרָהּ֙ וַיְמֹ֣תְתֵ֔הוּ וַיִּכְרָת־ בָּ֖הּ אֶת־ רֹאשׁ֑וֹ וַיִּרְא֧וּ הַפְּלִשְׁתִּ֛ים כִּֽי־ מֵ֥ת גִּבּוֹרָ֖ם וַיָּנֻֽסוּ׃ (טז) וַיָּקֻ֣מוּ אַנְשֵׁי֩ יִשְׂרָאֵ֨ל וִיהוּדָ֜ה וַיָּרִ֗עוּ וַֽיִּרְדְּפוּ֙ אֶת־ הַפְּלִשְׁתִּ֔ים עַד־ בּוֹאֲךָ֣ גַ֔יְא וְעַ֖ד שַׁעֲרֵ֣י עֶקְר֑וֹן וַֽיִּפְּל֞וּ חַֽלְלֵ֤י פְלִשְׁתִּים֙ בְּדֶ֣רֶךְ שַׁעֲרַ֔יִם וְעַד־ גַּ֖ת וְעַד־ עֶקְרֽוֹן׃ (יז) וַיָּשֻׁ֙בוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל מִדְּלֹ֖ק אַחֲרֵ֣י פְלִשְׁתִּ֑ים וַיָּשֹׁ֖סּוּ אֶת־ מַחֲנֵיהֶֽם׃ (יח) וַיִּקַּ֤ח דָּוִד֙ אֶת־ רֹ֣אשׁ הַפְּלִשְׁתִּ֔י וַיְבִאֵ֖הוּ יְרוּשָׁלִָ֑ם וְאֶת־ כֵּלָ֖יו שָׂ֥ם בְּאָהֳלֽוֹ׃ (יט) וְכִרְא֨וֹת שָׁא֜וּל אֶת־ דָּוִ֗ד יֹצֵא֙ לִקְרַ֣את הַפְּלִשְׁתִּ֔י אָמַ֗ר אֶל־ אַבְנֵר֙ שַׂ֣ר הַצָּבָ֔א בֶּן־ מִי־ זֶ֥ה הַנַּ֖עַר אַבְנֵ֑ר וַיֹּ֣אמֶר אַבְנֵ֔ר חֵֽי־ נַפְשְׁךָ֥ הַמֶּ֖לֶךְ אִם־ יָדָֽעְתִּי׃ (כ) וַיֹּ֖אמֶר הַמֶּ֑לֶךְ שְׁאַ֣ל אַתָּ֔ה בֶּן־ מִי־ זֶ֖ה הָעָֽלֶם׃ (כא) וּכְשׁ֣וּב דָּוִ֗ד מֵֽהַכּוֹת֙ אֶת־ הַפְּלִשְׁתִּ֔י וַיִּקַּ֤ח אֹתוֹ֙ אַבְנֵ֔ר וַיְבִאֵ֖הוּ לִפְנֵ֣י שָׁא֑וּל וְרֹ֥אשׁ הַפְּלִשְׁתִּ֖י בְּיָדֽוֹ׃ (כב) וַיֹּ֤אמֶר אֵלָיו֙ שָׁא֔וּל בֶּן־ מִ֥י אַתָּ֖ה הַנָּ֑עַר וַיֹּ֣אמֶר דָּוִ֔ד בֶּֽן־ עַבְדְּךָ֥ יִשַׁ֖י בֵּ֥ית הַלַּחְמִֽי׃ תחילת פרק יח (כג) וַיְהִ֗י כְּכַלֹּתוֹ֙ לְדַבֵּ֣ר אֶל־ שָׁא֔וּל וְנֶ֙פֶשׁ֙ יְה֣וֹנָתָ֔ן נִקְשְׁרָ֖ה בְּנֶ֣פֶשׁ דָּוִ֑ד (ויאהבו) וַיֶּאֱהָבֵ֥הוּ יְהוֹנָתָ֖ן כְּנַפְשֽׁוֹ׃ (כד) וַיִּקָּחֵ֥הוּ שָׁא֖וּל בַּיּ֣וֹם הַה֑וּא וְלֹ֣א נְתָנ֔וֹ לָשׁ֖וּב בֵּ֥ית אָבִֽיו׃ (כה) וַיִּכְרֹ֧ת יְהוֹנָתָ֛ן וְדָוִ֖ד בְּרִ֑ית בְּאַהֲבָת֥וֹ אֹת֖וֹ כְּנַפְשֽׁוֹ׃ (כו) וַיִּתְפַּשֵּׁ֣ט יְהוֹנָתָ֗ן אֶֽת־ הַמְּעִיל֙ אֲשֶׁ֣ר עָלָ֔יו וַֽיִּתְּנֵ֖הוּ לְדָוִ֑ד וּמַדָּ֕יו וְעַד־ חַרְבּ֥וֹ וְעַד־ קַשְׁתּ֖וֹ וְעַד־ חֲגֹרֽוֹ׃ (כז) וַיֵּצֵ֨א דָוִ֜ד בְּכֹל֩ אֲשֶׁ֨ר יִשְׁלָחֶ֤נּוּ שָׁאוּל֙ יַשְׂכִּ֔יל וַיְשִׂמֵ֣הוּ שָׁא֔וּל עַ֖ל אַנְשֵׁ֣י הַמִּלְחָמָ֑ה וַיִּיטַב֙ בְּעֵינֵ֣י כָל־ הָעָ֔ם וְגַ֕ם בְּעֵינֵ֖י עַבְדֵ֥י שָׁאֽוּל׃ (כח) וַיְהִ֣י בְּבוֹאָ֗ם בְּשׁ֤וּב דָּוִד֙ מֵהַכּ֣וֹת אֶת־ הַפְּלִשְׁתִּ֔י וַתֵּצֶ֨אנָה הַנָּשִׁ֜ים מִכָּל־ עָרֵ֤י יִשְׂרָאֵל֙ (לשור) לָשִׁ֣יר וְהַמְּחֹל֔וֹת לִקְרַ֖את שָׁא֣וּל הַמֶּ֑לֶךְ בְּתֻפִּ֥ים בְּשִׂמְחָ֖ה וּבְשָׁלִשִֽׁים׃ (כט) וַֽתַּעֲנֶ֛ינָה הַנָּשִׁ֥ים הַֽמְשַׂחֲק֖וֹת וַתֹּאמַ֑רְןָ הִכָּ֤ה שָׁאוּל֙ (באלפו) בַּאֲלָפָ֔יו וְדָוִ֖ד בְּרִבְבֹתָֽיו׃ (ל) וַיִּ֨חַר לְשָׁא֜וּל מְאֹ֗ד וַיֵּ֤רַע בְּעֵינָיו֙ הַדָּבָ֣ר הַזֶּ֔ה וַיֹּ֗אמֶר נָתְנ֤וּ לְדָוִד֙ רְבָב֔וֹת וְלִ֥י נָתְנ֖וּ הָאֲלָפִ֑ים וְע֥וֹד ל֖וֹ אַ֥ךְ הַמְּלוּכָֽה׃ (לא) וַיְהִ֥י שָׁא֖וּל (עון) עוֹיֵ֣ן אֶת־ דָּוִ֑ד מֵהַיּ֥וֹם הַה֖וּא וָהָֽלְאָה׃ (לב) וַיְהִ֣י מִֽמָּחֳרָ֗ת וַתִּצְלַ֣ח רוּחַ֩ אֱלֹהִ֨ים ׀ רָעָ֤ה ׀ אֶל־ שָׁאוּל֙ וַיִּתְנַבֵּ֣א בְתוֹךְ־ הַבַּ֔יִת וְדָוִ֛ד מְנַגֵּ֥ן בְּיָד֖וֹ כְּי֣וֹם ׀ בְּי֑וֹם וְהַחֲנִ֖ית בְּיַד־ שָׁאֽוּל׃ (לג) וַיָּ֤טֶל שָׁאוּל֙ אֶֽת־ הַחֲנִ֔ית וַיֹּ֕אמֶר אַכֶּ֥ה בְדָוִ֖ד וּבַקִּ֑יר וַיִּסֹּ֥ב דָּוִ֛ד מִפָּנָ֖יו פַּעֲמָֽיִם׃ (לד) וַיִּרָ֥א שָׁא֖וּל מִלִּפְנֵ֣י דָוִ֑ד כִּֽי־ הָיָ֤ה יְהוָה֙ עִמּ֔וֹ וּמֵעִ֥ם שָׁא֖וּל סָֽר׃ (לה) וַיְסִרֵ֤הוּ שָׁאוּל֙ מֵֽעִמּ֔וֹ וַיְשִׂמֵ֥הוּ ל֖וֹ שַׂר־ אָ֑לֶף וַיֵּצֵ֥א וַיָּבֹ֖א לִפְנֵ֥י הָעָֽם׃

שמואל סדר יב

(א) וַיְהִ֥י דָוִ֛ד לְכָל־ דָּרְכָ֖ו מַשְׂכִּ֑יל וַֽיהוָ֖ה עִמּֽוֹ׃ (ב) וַיַּ֣רְא שָׁא֔וּל אֲשֶׁר־ ה֖וּא מַשְׂכִּ֣יל מְאֹ֑ד וַיָּ֖גָר מִפָּנָֽיו׃ (ג) וְכָל־ יִשְׂרָאֵל֙ וִיהוּדָ֔ה אֹהֵ֖ב אֶת־ דָּוִ֑ד כִּֽי־ ה֛וּא יוֹצֵ֥א וָבָ֖א לִפְנֵיהֶֽם׃ (ד) וַיֹּ֨אמֶר שָׁא֜וּל אֶל־ דָּוִ֗ד הִנֵּה֩ בִתִּ֨י הַגְּדוֹלָ֤ה מֵרַב֙ אֹתָהּ֙ אֶתֶּן־ לְךָ֣ לְאִשָּׁ֔ה אַ֚ךְ הֱיֵה־ לִּ֣י לְבֶן־ חַ֔יִל וְהִלָּחֵ֖ם מִלְחֲמ֣וֹת יְהוָ֑ה וְשָׁא֣וּל אָמַ֗ר אַל־ תְּהִ֤י יָדִי֙ בּ֔וֹ וּתְהִי־ ב֖וֹ יַד־ פְּלִשְׁתִּֽים׃ (ה) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ שָׁא֗וּל מִ֤י אָֽנֹכִי֙ וּמִ֣י חַיַּ֔י מִשְׁפַּ֥חַת אָבִ֖י בְּיִשְׂרָאֵ֑ל כִּֽי־ אֶהְיֶ֥ה חָתָ֖ן לַמֶּֽלֶךְ׃ (ו) וַיְהִ֗י בְּעֵ֥ת תֵּ֛ת אֶת־ מֵרַ֥ב בַּת־ שָׁא֖וּל לְדָוִ֑ד וְהִ֧יא נִתְּנָ֛ה לְעַדְרִיאֵ֥ל הַמְּחֹלָתִ֖י לְאִשָּֽׁה׃ (ז) וַתֶּאֱהַ֛ב מִיכַ֥ל בַּת־ שָׁא֖וּל אֶת־ דָּוִ֑ד וַיַּגִּ֣דוּ לְשָׁא֔וּל וַיִּשַׁ֥ר הַדָּבָ֖ר בְּעֵינָֽיו׃ (ח) וַיֹּ֨אמֶר שָׁא֜וּל אֶתְּנֶ֤נָּה לּוֹ֙ וּתְהִי־ ל֣וֹ לְמוֹקֵ֔שׁ וּתְהִי־ ב֖וֹ יַד־ פְּלִשְׁתִּ֑ים וַיֹּ֤אמֶר שָׁאוּל֙ אֶל־ דָּוִ֔ד בִּשְׁתַּ֛יִם תִּתְחַתֵּ֥ן בִּ֖י הַיּֽוֹם׃ (ט) וַיְצַ֨ו שָׁא֜וּל אֶת־ עֲבָדָ֗ו דַּבְּר֨וּ אֶל־ דָּוִ֤ד בַּלָּט֙ לֵאמֹ֔ר הִנֵּ֨ה חָפֵ֤ץ בְּךָ֙ הַמֶּ֔לֶךְ וְכָל־ עֲבָדָ֖יו אֲהֵב֑וּךָ וְעַתָּ֖ה הִתְחַתֵּ֥ן בַּמֶּֽלֶךְ׃ (י) וַֽיְדַבְּר֞וּ עַבְדֵ֤י שָׁאוּל֙ בְּאָזְנֵ֣י דָוִ֔ד אֶת־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיֹּ֣אמֶר דָּוִ֗ד הַֽנְקַלָּ֤ה בְעֵֽינֵיכֶם֙ הִתְחַתֵּ֣ן בַּמֶּ֔לֶךְ וְאָנֹכִ֖י אִֽישׁ־ רָ֥שׁ וְנִקְלֶֽה׃ (יא) וַיַּגִּ֜דוּ עַבְדֵ֥י שָׁא֛וּל ל֖וֹ לֵאמֹ֑ר כַּדְּבָרִ֥ים הָאֵ֖לֶּה דִּבֶּ֥ר דָּוִֽד׃ (יב) וַיֹּ֨אמֶר שָׁא֜וּל כֹּֽה־ תֹאמְר֣וּ לְדָוִ֗ד אֵֽין־ חֵ֤פֶץ לַמֶּ֙לֶךְ֙ בְּמֹ֔הַר כִּ֗י בְּמֵאָה֙ עָרְל֣וֹת פְּלִשְׁתִּ֔ים לְהִנָּקֵ֖ם בְּאֹיְבֵ֣י הַמֶּ֑לֶךְ וְשָׁא֣וּל חָשַׁ֔ב לְהַפִּ֥יל אֶת־ דָּוִ֖ד בְּיַד־ פְּלִשְׁתִּֽים׃ (יג) וַיַּגִּ֨דוּ עֲבָדָ֤יו לְדָוִד֙ אֶת־ הַדְּבָרִ֣ים הָאֵ֔לֶּה וַיִּשַׁ֤ר הַדָּבָר֙ בְּעֵינֵ֣י דָוִ֔ד לְהִתְחַתֵּ֖ן בַּמֶּ֑לֶךְ וְלֹ֥א מָלְא֖וּ הַיָּמִֽים׃ (יד) וַיָּ֨קָם דָּוִ֜ד וַיֵּ֣לֶךְ ׀ ה֣וּא וַאֲנָשָׁ֗יו וַיַּ֣ךְ בַּפְּלִשְׁתִּים֮ מָאתַ֣יִם אִישׁ֒ וַיָּבֵ֤א דָוִד֙ אֶת־ עָרְלֹ֣תֵיהֶ֔ם וַיְמַלְא֣וּם לַמֶּ֔לֶךְ לְהִתְחַתֵּ֖ן בַּמֶּ֑לֶךְ וַיִּתֶּן־ ל֥וֹ שָׁא֛וּל אֶת־ מִיכַ֥ל בִּתּ֖וֹ לְאִשָּֽׁה׃ (טו) וַיַּ֤רְא שָׁאוּל֙ וַיֵּ֔דַע כִּ֥י יְהוָ֖ה עִם־ דָּוִ֑ד וּמִיכַ֥ל בַּת־ שָׁא֖וּל אֲהֵבַֽתְהוּ׃ (טז) וַיֹּ֣אסֶף שָׁא֗וּל לֵרֹ֛א מִפְּנֵ֥י דָוִ֖ד ע֑וֹד וַיְהִ֥י שָׁא֛וּל אֹיֵ֥ב אֶת־ דָּוִ֖ד כָּל־ הַיָּמִֽים׃ (יז) וַיֵּצְא֖וּ שָׂרֵ֣י פְלִשְׁתִּ֑ים וַיְהִ֣י ׀ מִדֵּ֣י צֵאתָ֗ם שָׂכַ֤ל דָּוִד֙ מִכֹּל֙ עַבְדֵ֣י שָׁא֔וּל וַיִּיקַ֥ר שְׁמ֖וֹ מְאֹֽד׃ סוף פרק יח (יח) וַיְדַבֵּ֣ר שָׁא֗וּל אֶל־ יוֹנָתָ֤ן בְּנוֹ֙ וְאֶל־ כָּל־ עֲבָדָ֔יו לְהָמִ֖ית אֶת־ דָּוִ֑ד וִיהֽוֹנָתָן֙ בֶּן־ שָׁא֔וּל חָפֵ֥ץ בְּדָוִ֖ד מְאֹֽד׃ (יט) וַיַּגֵּ֤ד יְהוֹנָתָן֙ לְדָוִ֣ד לֵאמֹ֔ר מְבַקֵּ֛שׁ שָׁא֥וּל אָבִ֖י לַהֲמִיתֶ֑ךָ וְעַתָּה֙ הִשָּֽׁמֶר־ נָ֣א בַבֹּ֔קֶר וְיָשַׁבְתָּ֥ בַסֵּ֖תֶר וְנַחְבֵּֽאתָ׃ (כ) וַאֲנִ֨י אֵצֵ֜א וְעָמַדְתִּ֣י לְיַד־ אָבִ֗י בַּשָּׂדֶה֙ אֲשֶׁ֣ר אַתָּ֣ה שָׁ֔ם וַאֲנִ֕י אֲדַבֵּ֥ר בְּךָ֖ אֶל־ אָבִ֑י וְרָאִ֥יתִי מָ֖ה וְהִגַּ֥דְתִּי לָֽךְ׃ (כא) וַיְדַבֵּ֨ר יְהוֹנָתָ֤ן בְּדָוִד֙ ט֔וֹב אֶל־ שָׁא֖וּל אָבִ֑יו וַיֹּ֣אמֶר אֵ֠לָיו אַל־ יֶחֱטָ֨א הַמֶּ֜לֶךְ בְּעַבְדּ֣וֹ בְדָוִ֗ד כִּ֣י ל֤וֹא חָטָא֙ לָ֔ךְ וְכִ֥י מַעֲשָׂ֖יו טוֹב־ לְךָ֥ מְאֹֽד׃ (כב) וַיָּשֶׂם֩ אֶת־ נַפְשׁ֨וֹ בְכַפּ֜וֹ וַיַּ֣ךְ אֶת־ הַפְּלִשְׁתִּ֗י וַיַּ֨עַשׂ יְהוָ֜ה תְּשׁוּעָ֤ה גְדוֹלָה֙ לְכָל־ יִשְׂרָאֵ֔ל רָאִ֖יתָ וַתִּשְׂמָ֑ח וְלָ֤מָּה תֶֽחֱטָא֙ בְּדָ֣ם נָקִ֔י לְהָמִ֥ית אֶת־ דָּוִ֖ד חִנָּֽם׃ (כג) וַיִּשְׁמַ֥ע שָׁא֖וּל בְּק֣וֹל יְהוֹנָתָ֑ן וַיִּשָּׁבַ֣ע שָׁא֔וּל חַי־ יְהוָ֖ה אִם־ יוּמָֽת׃ (כד) וַיִּקְרָ֤א יְהוֹנָתָן֙ לְדָוִ֔ד וַיַּגֶּד־ לוֹ֙ יְה֣וֹנָתָ֔ן אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיָּבֵ֨א יְהוֹנָתָ֤ן אֶת־ דָּוִד֙ אֶל־ שָׁא֔וּל וַיְהִ֥י לְפָנָ֖יו כְּאֶתְמ֥וֹל שִׁלְשֽׁוֹם׃ (כה) וַתּ֥וֹסֶף הַמִּלְחָמָ֖ה לִֽהְי֑וֹת וַיֵּצֵ֨א דָוִ֜ד וַיִּלָּ֣חֶם בַּפְּלִשְׁתִּ֗ים וַיַּ֤ךְ בָּהֶם֙ מַכָּ֣ה גְדוֹלָ֔ה וַיָּנֻ֖סוּ מִפָּנָֽיו׃ (כו) וַתְּהִי֩ ר֨וּחַ יְהוָ֤ה ׀ רָעָה֙ אֶל־ שָׁא֔וּל וְהוּא֙ בְּבֵית֣וֹ יוֹשֵׁ֔ב וַחֲנִית֖וֹ בְּיָד֑וֹ וְדָוִ֖ד מְנַגֵּ֥ן בְּיָֽד׃ (כז) וַיְבַקֵּ֨שׁ שָׁא֜וּל לְהַכּ֤וֹת בַּֽחֲנִית֙ בְּדָוִ֣ד וּבַקִּ֔יר וַיִּפְטַר֙ מִפְּנֵ֣י שָׁא֔וּל וַיַּ֥ךְ אֶֽת־ הַחֲנִ֖ית בַּקִּ֑יר וְדָוִ֛ד נָ֥ס וַיִּמָּלֵ֖ט בַּלַּ֥יְלָה הֽוּא׃ (כח) וַיִּשְׁלַח֩ שָׁא֨וּל מַלְאָכִ֜ים אֶל־ בֵּ֤ית דָּוִד֙ לְשָׁמְר֔וֹ וְלַהֲמִית֖וֹ בַּבֹּ֑קֶר וַתַּגֵּ֣ד לְדָוִ֗ד מִיכַ֤ל אִשְׁתּוֹ֙ לֵאמֹ֔ר אִם־ אֵ֨ינְךָ֜ מְמַלֵּ֤ט אֶֽת־ נַפְשְׁךָ֙ הַלַּ֔יְלָה מָחָ֖ר אַתָּ֥ה מוּמָֽת׃ (כט) וַתֹּ֧רֶד מִיכַ֛ל אֶת־ דָּוִ֖ד בְּעַ֣ד הַחַלּ֑וֹן וַיֵּ֥לֶךְ וַיִּבְרַ֖ח וַיִּמָּלֵֽט׃ (ל) וַתִּקַּ֨ח מִיכַ֜ל אֶת־ הַתְּרָפִ֗ים וַתָּ֙שֶׂם֙ אֶל־ הַמִּטָּ֔ה וְאֵת֙ כְּבִ֣יר הָֽעִזִּ֔ים שָׂ֖מָה מְרַֽאֲשֹׁתָ֑יו וַתְּכַ֖ס בַּבָּֽגֶד׃ (לא) וַיִּשְׁלַ֥ח שָׁא֛וּל מַלְאָכִ֖ים לָקַ֣חַת אֶת־ דָּוִ֑ד וַתֹּ֖אמֶר חֹלֶ֥ה הֽוּא׃ (לב) וַיִּשְׁלַ֤ח שָׁאוּל֙ אֶת־ הַמַּלְאָכִ֔ים לִרְא֥וֹת אֶת־ דָּוִ֖ד לֵאמֹ֑ר הַעֲל֨וּ אֹת֧וֹ בַמִּטָּ֛ה אֵלַ֖י לַהֲמִתֽוֹ׃ (לג) וַיָּבֹ֙אוּ֙ הַמַּלְאָכִ֔ים וְהִנֵּ֥ה הַתְּרָפִ֖ים אֶל־ הַמִּטָּ֑ה וּכְבִ֥יר הָעִזִּ֖ים מְרַאֲשֹׁתָֽיו׃ (לד) וַיֹּ֨אמֶר שָׁא֜וּל אֶל־ מִיכַ֗ל לָ֤מָּה כָּ֙כָה֙ רִמִּיתִ֔נִי וַתְּשַׁלְּחִ֥י אֶת־ אֹיְבִ֖י וַיִּמָּלֵ֑ט וַתֹּ֤אמֶר מִיכַל֙ אֶל־ שָׁא֔וּל הוּא־ אָמַ֥ר אֵלַ֛י שַׁלְּחִ֖נִי לָמָ֥ה אֲמִיתֵֽךְ׃ (לה) וְדָוִ֨ד בָּרַ֜ח וַיִּמָּלֵ֗ט וַיָּבֹ֤א אֶל־ שְׁמוּאֵל֙ הָרָמָ֔תָה וַיַּ֨גֶּד־ ל֔וֹ אֵ֛ת כָּל־ אֲשֶׁ֥ר עָֽשָׂה־ ל֖וֹ שָׁא֑וּל וַיֵּ֤לֶךְ הוּא֙ וּשְׁמוּאֵ֔ל וַיֵּשְׁב֖וּ (בנוית) בְּנָֽיוֹת׃ (לו) וַיֻּגַּ֥ד לְשָׁא֖וּל לֵאמֹ֑ר הִנֵּ֣ה דָוִ֔ד (בנוית) בְּנָי֖וֹת בָּרָמָֽה׃ (לז) וַיִּשְׁלַ֨ח שָׁא֣וּל מַלְאָכִים֮ לָקַ֣חַת אֶת־ דָּוִד֒ וַיַּ֗רְא אֶֽת־ לַהֲקַ֤ת הַנְּבִיאִים֙ נִבְּאִ֔ים וּשְׁמוּאֵ֕ל עֹמֵ֥ד נִצָּ֖ב עֲלֵיהֶ֑ם וַתְּהִ֞י עַֽל־ מַלְאֲכֵ֤י שָׁאוּל֙ ר֣וּחַ אֱלֹהִ֔ים וַיִּֽתְנַבְּא֖וּ גַּם־ הֵֽמָּה׃ (לח) וַיַּגִּ֣דוּ לְשָׁא֗וּל וַיִּשְׁלַח֙ מַלְאָכִ֣ים אֲחֵרִ֔ים וַיִּֽתְנַבְּא֖וּ גַּם־ הֵ֑מָּה וַיֹּ֣סֶף שָׁא֗וּל וַיִּשְׁלַח֙ מַלְאָכִ֣ים שְׁלִשִׁ֔ים וַיִּֽתְנַבְּא֖וּ גַּם־ הֵֽמָּה׃ (לט) וַיֵּ֨לֶךְ גַּם־ ה֜וּא הָרָמָ֗תָה וַיָּבֹא֙ עַד־ בּ֤וֹר הַגָּדוֹל֙ אֲשֶׁ֣ר בַּשֶּׂ֔כוּ וַיִּשְׁאַ֣ל וַיֹּ֔אמֶר אֵיפֹ֥ה שְׁמוּאֵ֖ל וְדָוִ֑ד וַיֹּ֕אמֶר הִנֵּ֖ה (בנוית) בְּנָי֥וֹת בָּרָמָֽה׃ (מ) וַיֵּ֣לֶךְ שָׁ֔ם אֶל־ (נוית) נָי֖וֹת בָּרָמָ֑ה וַתְּהִי֩ עָלָ֨יו גַּם־ ה֜וּא ר֣וּחַ אֱלֹהִ֗ים וַיֵּ֤לֶךְ הָלוֹךְ֙ וַיִּתְנַבֵּ֔א עַד־ בֹּא֖וֹ (בנוית) בְּנָי֥וֹת בָּרָמָֽה׃ (מא) וַיִּפְשַׁ֨ט גַּם־ ה֜וּא בְּגָדָ֗יו וַיִּתְנַבֵּ֤א גַם־ הוּא֙ לִפְנֵ֣י שְׁמוּאֵ֔ל וַיִּפֹּ֣ל עָרֹ֔ם כָּל־ הַיּ֥וֹם הַה֖וּא וְכָל־ הַלָּ֑יְלָה עַל־ כֵּן֙ יֹֽאמְר֔וּ הֲגַ֥ם שָׁא֖וּל בַּנְּבִיאִֽם׃ (מב) וַיִּבְרַ֣ח דָּוִ֔ד (מנוות) מִנָּי֖וֹת בָּרָמָ֑ה וַיָּבֹ֞א וַיֹּ֣אמֶר ׀ לִפְנֵ֣י יְהוֹנָתָ֗ן מֶ֤ה עָשִׂ֙יתִי֙ מֶֽה־ עֲוֺנִ֤י וּמֶֽה־ חַטָּאתִי֙ לִפְנֵ֣י אָבִ֔יךָ כִּ֥י מְבַקֵּ֖שׁ אֶת־ נַפְשִֽׁי׃ (מג) וַיֹּ֨אמֶר ל֣וֹ חָלִילָה֮ לֹ֣א תָמוּת֒ הִנֵּ֡ה (לו)־ (עשה) לֹֽא־ יַעֲשֶׂ֨ה אָבִ֜י דָּבָ֣ר גָּד֗וֹל א֚וֹ דָּבָ֣ר קָטֹ֔ן וְלֹ֥א יִגְלֶ֖ה אֶת־ אָזְנִ֑י וּמַדּוּעַ֩ יַסְתִּ֨יר אָבִ֥י מִמֶּ֛נִּי אֶת־ הַדָּבָ֥ר הַזֶּ֖ה אֵ֥ין זֹֽאת׃ (מד) וַיִּשָּׁבַ֨ע ע֜וֹד דָּוִ֗ד וַיֹּ֙אמֶר֙ יָדֹ֨עַ יָדַ֜ע אָבִ֗יךָ כִּֽי־ מָצָ֤אתִי חֵן֙ בְּעֵינֶ֔יךָ וַיֹּ֛אמֶר אַל־ יֵֽדַע־ זֹ֥את יְהוֹנָתָ֖ן פֶּן־ יֵֽעָצֵ֑ב וְאוּלָ֗ם חַי־ יְהוָה֙ וְחֵ֣י נַפְשֶׁ֔ךָ כִּ֣י כְפֶ֔שַׂע בֵּינִ֖י וּבֵ֥ין הַמָּֽוֶת׃
