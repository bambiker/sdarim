שמואל א פרק ג לפי החלוקה לסדרים

פרק ג בספר שמואל א (21 פסוקים) נמצא בשמואל סדר ב (פסוקים כח–מו) ובשמואל סדר ג (פסוקים א–ב).

שמואל סדר ב

(א) יְהוָ֞ה יֵחַ֣תּוּ (מריבו) מְרִיבָ֗יו (עלו) עָלָיו֙ בַּשָּׁמַ֣יִם יַרְעֵ֔ם יְהוָ֖ה יָדִ֣ין אַפְסֵי־ אָ֑רֶץ וְיִתֶּן־ עֹ֣ז לְמַלְכּ֔וֹ וְיָרֵ֖ם קֶ֥רֶן מְשִׁיחֽוֹ׃ (ב) וַיֵּ֧לֶךְ אֶלְקָנָ֛ה הָרָמָ֖תָה עַל־ בֵּית֑וֹ וְהַנַּ֗עַר הָיָ֤ה מְשָׁרֵת֙ אֶת־ יְהוָ֔ה אֶת־ פְּנֵ֖י עֵלִ֥י הַכֹּהֵֽן׃ (ג) וּבְנֵ֥י עֵלִ֖י בְּנֵ֣י בְלִיָּ֑עַל לֹ֥א יָדְע֖וּ אֶת־ יְהוָֽה׃ (ד) וּמִשְׁפַּ֥ט הַכֹּהֲנִ֖ים אֶת־ הָעָ֑ם כָּל־ אִ֞ישׁ זֹבֵ֣חַ זֶ֗בַח וּבָ֨א נַ֤עַר הַכֹּהֵן֙ כְּבַשֵּׁ֣ל הַבָּשָׂ֔ר וְהַמַּזְלֵ֛ג שְׁלֹ֥שׁ־ הַשִּׁנַּ֖יִם בְּיָדֽוֹ׃ (ה) וְהִכָּ֨ה בַכִּיּ֜וֹר א֣וֹ בַדּ֗וּד א֤וֹ בַקַּלַּ֙חַת֙ א֣וֹ בַפָּר֔וּר כֹּ֚ל אֲשֶׁ֣ר יַעֲלֶ֣ה הַמַּזְלֵ֔ג יִקַּ֥ח הַכֹּהֵ֖ן בּ֑וֹ כָּ֚כָה יַעֲשׂ֣וּ לְכָל־ יִשְׂרָאֵ֔ל הַבָּאִ֥ים שָׁ֖ם בְּשִׁלֹֽה׃ (ו) גַּם֮ בְּטֶרֶם֮ יַקְטִר֣וּן אֶת־ הַחֵלֶב֒ וּבָ֣א ׀ נַ֣עַר הַכֹּהֵ֗ן וְאָמַר֙ לָאִ֣ישׁ הַזֹּבֵ֔חַ תְּנָ֣ה בָשָׂ֔ר לִצְל֖וֹת לַכֹּהֵ֑ן וְלֹֽא־ יִקַּ֧ח מִמְּךָ֛ בָּשָׂ֥ר מְבֻשָּׁ֖ל כִּ֥י אִם־ חָֽי׃ (ז) וַיֹּ֨אמֶר אֵלָ֜יו הָאִ֗ישׁ קַטֵּ֨ר יַקְטִיר֤וּן כַּיּוֹם֙ הַחֵ֔לֶב וְקַ֨ח־ לְךָ֔ כַּאֲשֶׁ֥ר תְּאַוֶּ֖ה נַפְשֶׁ֑ךָ וְאָמַ֥ר ׀ (לו) לֹא֙ כִּ֚י עַתָּ֣ה תִתֵּ֔ן וְאִם־ לֹ֖א לָקַ֥חְתִּי בְחָזְקָֽה׃ (ח) וַתְּהִ֨י חַטַּ֧את הַנְּעָרִ֛ים גְּדוֹלָ֥ה מְאֹ֖ד אֶת־ פְּנֵ֣י יְהוָ֑ה כִּ֤י נִֽאֲצוּ֙ הָֽאֲנָשִׁ֔ים אֵ֖ת מִנְחַ֥ת יְהוָֽה׃ (ט) וּשְׁמוּאֵ֕ל מְשָׁרֵ֖ת אֶת־ פְּנֵ֣י יְהוָ֑ה נַ֕עַר חָג֖וּר אֵפ֥וֹד בָּֽד׃ (י) וּמְעִ֤יל קָטֹן֙ תַּעֲשֶׂה־ לּ֣וֹ אִמּ֔וֹ וְהַעַלְתָ֥ה ל֖וֹ מִיָּמִ֣ים ׀ יָמִ֑ימָה בַּֽעֲלוֹתָהּ֙ אֶת־ אִישָׁ֔הּ לִזְבֹּ֖חַ אֶת־ זֶ֥בַח הַיָּמִֽים׃ (יא) וּבֵרַ֨ךְ עֵלִ֜י אֶת־ אֶלְקָנָ֣ה וְאֶת־ אִשְׁתּ֗וֹ וְאָמַר֙ יָשֵׂם֩ יְהוָ֨ה לְךָ֥ זֶ֙רַע֙ מִן־ הָאִשָּׁ֣ה הַזֹּ֔את תַּ֚חַת הַשְּׁאֵלָ֔ה אֲשֶׁ֥ר שָׁאַ֖ל לַֽיהוָ֑ה וְהָלְכ֖וּ לִמְקֹמֽוֹ׃ (יב) כִּֽי־ פָקַ֤ד יְהוָה֙ אֶת־ חַנָּ֔ה וַתַּ֛הַר וַתֵּ֥לֶד שְׁלֹשָֽׁה־ בָנִ֖ים וּשְׁתֵּ֣י בָנ֑וֹת וַיִּגְדַּ֛ל הַנַּ֥עַר שְׁמוּאֵ֖ל עִם־ יְהוָֽה׃ (יג) וְעֵלִ֖י זָקֵ֣ן מְאֹ֑ד וְשָׁמַ֗ע אֵת֩ כָּל־ אֲשֶׁ֨ר יַעֲשׂ֤וּן בָּנָיו֙ לְכָל־ יִשְׂרָאֵ֔ל וְאֵ֤ת אֲשֶֽׁר־ יִשְׁכְּבוּן֙ אֶת־ הַנָּשִׁ֔ים הַצֹּ֣בְא֔וֹת פֶּ֖תַח אֹ֥הֶל מוֹעֵֽד׃ (יד) וַיֹּ֣אמֶר לָהֶ֔ם לָ֥מָּה תַעֲשׂ֖וּן כַּדְּבָרִ֣ים הָאֵ֑לֶּה אֲשֶׁ֨ר אָנֹכִ֤י שֹׁמֵ֙עַ֙ אֶת־ דִּבְרֵיכֶ֣ם רָעִ֔ים מֵאֵ֖ת כָּל־ הָעָ֥ם אֵֽלֶּה׃ (טו) אַ֖ל בָּנָ֑י כִּ֠י לֽוֹא־ טוֹבָ֤ה הַשְּׁמֻעָה֙ אֲשֶׁ֣ר אָנֹכִ֣י שֹׁמֵ֔עַ מַעֲבִרִ֖ים עַם־ יְהוָֽה׃ (טז) אִם־ יֶחֱטָ֨א אִ֤ישׁ לְאִישׁ֙ וּפִֽלְל֣וֹ אֱלֹהִ֔ים וְאִ֤ם לַֽיהוָה֙ יֶֽחֱטָא־ אִ֔ישׁ מִ֖י יִתְפַּלֶּל־ ל֑וֹ וְלֹ֤א יִשְׁמְעוּ֙ לְק֣וֹל אֲבִיהֶ֔ם כִּֽי־ חָפֵ֥ץ יְהוָ֖ה לַהֲמִיתָֽם׃ (יז) וְהַנַּ֣עַר שְׁמוּאֵ֔ל הֹלֵ֥ךְ וְגָדֵ֖ל וָט֑וֹב גַּ֚ם עִם־ יְהוָ֔ה וְגַ֖ם עִם־ אֲנָשִֽׁים׃ (יח) וַיָּבֹ֥א אִישׁ־ אֱלֹהִ֖ים אֶל־ עֵלִ֑י וַיֹּ֣אמֶר אֵלָ֗יו כֹּ֚ה אָמַ֣ר יְהוָ֔ה הֲנִגְלֹ֤ה נִגְלֵ֙יתִי֙ אֶל־ בֵּ֣ית אָבִ֔יךָ בִּֽהְיוֹתָ֥ם בְּמִצְרַ֖יִם לְבֵ֥ית פַּרְעֹֽה׃ (יט) וּבָחֹ֣ר אֹ֠תוֹ מִכָּל־ שִׁבְטֵ֨י יִשְׂרָאֵ֥ל לִי֙ לְכֹהֵ֔ן לַעֲל֣וֹת עַֽל־ מִזְבְּחִ֗י לְהַקְטִ֥יר קְטֹ֛רֶת לָשֵׂ֥את אֵפ֖וֹד לְפָנָ֑י וָֽאֶתְּנָה֙ לְבֵ֣ית אָבִ֔יךָ אֶת־ כָּל־ אִשֵּׁ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (כ) לָ֣מָּה תִבְעֲט֗וּ בְּזִבְחִי֙ וּבְמִנְחָתִ֔י אֲשֶׁ֥ר צִוִּ֖יתִי מָע֑וֹן וַתְּכַבֵּ֤ד אֶת־ בָּנֶ֙יךָ֙ מִמֶּ֔נִּי לְהַבְרִֽיאֲכֶ֗ם מֵרֵאשִׁ֛ית כָּל־ מִנְחַ֥ת יִשְׂרָאֵ֖ל לְעַמִּֽי׃ (כא) לָכֵ֗ן נְאֻם־ יְהוָה֮ אֱלֹהֵ֣י יִשְׂרָאֵל֒ אָמ֣וֹר אָמַ֔רְתִּי בֵּֽיתְךָ֙ וּבֵ֣ית אָבִ֔יךָ יִתְהַלְּכ֥וּ לְפָנַ֖י עַד־ עוֹלָ֑ם וְעַתָּ֤ה נְאֻם־ יְהוָה֙ חָלִ֣ילָה לִּ֔י כִּֽי־ מְכַבְּדַ֥י אֲכַבֵּ֖ד וּבֹזַ֥י יֵקָֽלּוּ׃ (כב) הִנֵּה֙ יָמִ֣ים בָּאִ֔ים וְגָֽדַעְתִּי֙ אֶת־ זְרֹ֣עֲךָ֔ וְאֶת־ זְרֹ֖עַ בֵּ֣ית אָבִ֑יךָ מִֽהְי֥וֹת זָקֵ֖ן בְּבֵיתֶֽךָ׃ (כג) וְהִבַּטְתָּ֙ צַ֣ר מָע֔וֹן בְּכֹ֥ל אֲשֶׁר־ יֵיטִ֖יב אֶת־ יִשְׂרָאֵ֑ל וְלֹֽא־ יִהְיֶ֥ה זָקֵ֛ן בְּבֵיתְךָ֖ כָּל־ הַיָּמִֽים׃ (כד) וְאִ֗ישׁ לֹֽא־ אַכְרִ֤ית לְךָ֙ מֵעִ֣ם מִזְבְּחִ֔י לְכַלּ֥וֹת אֶת־ עֵינֶ֖יךָ וְלַאֲדִ֣יב אֶת־ נַפְשֶׁ֑ךָ וְכָל־ מַרְבִּ֥ית בֵּיתְךָ֖ יָמ֥וּתוּ אֲנָשִֽׁים׃ (כה) וְזֶה־ לְּךָ֣ הָא֗וֹת אֲשֶׁ֤ר יָבֹא֙ אֶל־ שְׁנֵ֣י בָנֶ֔יךָ אֶל־ חָפְנִ֖י וּפִֽינְחָ֑ס בְּי֥וֹם אֶחָ֖ד יָמ֥וּתוּ שְׁנֵיהֶֽם׃ (כו) וַהֲקִימֹתִ֥י לִי֙ כֹּהֵ֣ן נֶאֱמָ֔ן כַּאֲשֶׁ֛ר בִּלְבָבִ֥י וּבְנַפְשִׁ֖י יַעֲשֶׂ֑ה וּבָנִ֤יתִי לוֹ֙ בַּ֣יִת נֶאֱמָ֔ן וְהִתְהַלֵּ֥ךְ לִפְנֵֽי־ מְשִׁיחִ֖י כָּל־ הַיָּמִֽים׃ (כז) וְהָיָ֗ה כָּל־ הַנּוֹתָר֙ בְּבֵ֣יתְךָ֔ יָבוֹא֙ לְהִשְׁתַּחֲוֺ֣ת ל֔וֹ לַאֲג֥וֹרַת כֶּ֖סֶף וְכִכַּר־ לָ֑חֶם וְאָמַ֗ר סְפָחֵ֥נִי נָ֛א אֶל־ אַחַ֥ת הַכְּהֻנּ֖וֹת לֶאֱכֹ֥ל פַּת־ לָֽחֶם׃ תחילת פרק ג (כח) וְהַנַּ֧עַר שְׁמוּאֵ֛ל מְשָׁרֵ֥ת אֶת־ יְהוָ֖ה לִפְנֵ֣י עֵלִ֑י וּדְבַר־ יְהוָ֗ה הָיָ֤ה יָקָר֙ בַּיָּמִ֣ים הָהֵ֔ם אֵ֥ין חָז֖וֹן נִפְרָֽץ׃ (כט) וַֽיְהִי֙ בַּיּ֣וֹם הַה֔וּא וְעֵלִ֖י שֹׁכֵ֣ב בִּמְקֹמ֑וֹ (ועינו) וְעֵינָיו֙ הֵחֵ֣לּוּ כֵה֔וֹת לֹ֥א יוּכַ֖ל לִרְאֽוֹת׃ (ל) וְנֵ֤ר אֱלֹהִים֙ טֶ֣רֶם יִכְבֶּ֔ה וּשְׁמוּאֵ֖ל שֹׁכֵ֑ב בְּהֵיכַ֣ל יְהוָ֔ה אֲשֶׁר־ שָׁ֖ם אֲר֥וֹן אֱלֹהִֽים׃ (לא) וַיִּקְרָ֧א יְהוָ֛ה אֶל־ שְׁמוּאֵ֖ל וַיֹּ֥אמֶר הִנֵּֽנִי׃ (לב) וַיָּ֣רָץ אֶל־ עֵלִ֗י וַיֹּ֤אמֶר הִנְנִי֙ כִּֽי־ קָרָ֣אתָ לִּ֔י וַיֹּ֥אמֶר לֹֽא־ קָרָ֖אתִי שׁ֣וּב שְׁכָ֑ב וַיֵּ֖לֶךְ וַיִּשְׁכָּֽב׃ (לג) וַיֹּ֣סֶף יְהוָ֗ה קְרֹ֣א עוֹד֮ שְׁמוּאֵל֒ וַיָּ֤קָם שְׁמוּאֵל֙ וַיֵּ֣לֶךְ אֶל־ עֵלִ֔י וַיֹּ֣אמֶר הִנְנִ֔י כִּ֥י קָרָ֖אתָ לִ֑י וַיֹּ֛אמֶר לֹֽא־ קָרָ֥אתִי בְנִ֖י שׁ֥וּב שְׁכָֽב׃ (לד) וּשְׁמוּאֵ֕ל טֶ֖רֶם יָדַ֣ע אֶת־ יְהוָ֑ה וְטֶ֛רֶם יִגָּלֶ֥ה אֵלָ֖יו דְּבַר־ יְהוָֽה׃ (לה) וַיֹּ֨סֶף יְהוָ֥ה קְרֹא־ שְׁמוּאֵל֮ בַּשְּׁלִשִׁית֒ וַיָּ֙קָם֙ וַיֵּ֣לֶךְ אֶל־ עֵלִ֔י וַיֹּ֣אמֶר הִנְנִ֔י כִּ֥י קָרָ֖אתָ לִ֑י וַיָּ֣בֶן עֵלִ֔י כִּ֥י יְהוָ֖ה קֹרֵ֥א לַנָּֽעַר׃ (לו) וַיֹּ֨אמֶר עֵלִ֣י לִשְׁמוּאֵל֮ לֵ֣ךְ שְׁכָב֒ וְהָיָה֙ אִם־ יִקְרָ֣א אֵלֶ֔יךָ וְאָֽמַרְתָּ֙ דַּבֵּ֣ר יְהוָ֔ה כִּ֥י שֹׁמֵ֖עַ עַבְדֶּ֑ךָ וַיֵּ֣לֶךְ שְׁמוּאֵ֔ל וַיִּשְׁכַּ֖ב בִּמְקוֹמֽוֹ׃ (לז) וַיָּבֹ֤א יְהוָה֙ וַיִּתְיַצַּ֔ב וַיִּקְרָ֥א כְפַֽעַם־ בְּפַ֖עַם שְׁמוּאֵ֣ל ׀ שְׁמוּאֵ֑ל וַיֹּ֤אמֶר שְׁמוּאֵל֙ דַּבֵּ֔ר כִּ֥י שֹׁמֵ֖עַ עַבְדֶּֽךָ׃ (לח) וַיֹּ֤אמֶר יְהוָה֙ אֶל־ שְׁמוּאֵ֔ל הִנֵּ֧ה אָנֹכִ֛י עֹשֶׂ֥ה דָבָ֖ר בְּיִשְׂרָאֵ֑ל אֲשֶׁר֙ כָּל־ שֹׁ֣מְע֔וֹ אֲר֥וֹן הָאֱלֹהִֽים׃ (כה) וּפְלִשְׁתִּים֙ לָֽקְח֔וּ אֵ֖ת אֲר֣וֹן הָאֱלֹהִ֑ים וַיְבִאֻ֛הוּ מֵאֶ֥בֶן הָעֵ֖זֶר אַשְׁדּֽוֹדָה׃ (כו) וַיִּקְח֤וּ פְלִשְׁתִּים֙ אֶת־ אֲר֣וֹן הָאֱלֹהִ֔ים וַיָּבִ֥יאוּ אֹת֖וֹ בֵּ֣ית דָּג֑וֹן וַיַּצִּ֥יגוּ אֹת֖וֹ אֵ֥צֶל דָּגֽוֹן׃ (כז) וַיַּשְׁכִּ֤מוּ אַשְׁדּוֹדִים֙ מִֽמָּחֳרָ֔ת וְהִנֵּ֣ה דָג֗וֹן נֹפֵ֤ל לְפָנָיו֙ אַ֔רְצָה לִפְנֵ֖י אֲר֣וֹן יְהוָ֑ה וַיִּקְחוּ֙ אֶת־ דָּג֔וֹן וַיָּשִׁ֥בוּ אֹת֖וֹ לִמְקוֹמֽוֹ׃ (כח) וַיַּשְׁכִּ֣מוּ בַבֹּקֶר֮ מִֽמָּחֳרָת֒ וְהִנֵּ֣ה דָג֗וֹן נֹפֵ֤ל לְפָנָיו֙ אַ֔רְצָה לִפְנֵ֖י אֲר֣וֹן יְהוָ֑ה וְרֹ֨אשׁ דָּג֜וֹן וּשְׁתֵּ֣י ׀ כַּפּ֣וֹת יָדָ֗יו כְּרֻתוֹת֙ אֶל־ הַמִּפְתָּ֔ן רַ֥ק דָּג֖וֹן נִשְׁאַ֥ר עָלָֽיו׃ (כט) עַל־ כֵּ֡ן לֹֽא־ יִדְרְכוּ֩ כֹהֲנֵ֨י דָג֜וֹן וְכָֽל־ הַבָּאִ֧ים בֵּית־ דָּג֛וֹן עַל־ מִפְתַּ֥ן דָּג֖וֹן בְּאַשְׁדּ֑וֹד עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (ל) וַתִּכְבַּ֧ד יַד־ יְהוָ֛ה אֶל־ הָאַשְׁדּוֹדִ֖ים וַיְשִׁמֵּ֑ם וַיַּ֤ךְ אֹתָם֙ (בעפלים) בַּטְּחֹרִ֔ים אֶת־ אַשְׁדּ֖וֹד וְאֶת־ גְּבוּלֶֽיהָ׃ (לא) וַיִּרְא֥וּ אַנְשֵֽׁי־ אַשְׁדּ֖וֹד כִּֽי־ כֵ֑ן וְאָמְר֗וּ לֹֽא־ יֵשֵׁ֞ב אֲר֨וֹן אֱלֹהֵ֤י יִשְׂרָאֵל֙ עִמָּ֔נוּ כִּֽי־ קָשְׁתָ֤ה יָדוֹ֙ עָלֵ֔ינוּ וְעַ֖ל דָּג֥וֹן אֱלֹהֵֽינוּ׃ (לב) וַיִּשְׁלְח֡וּ וַיַּאַסְפוּ֩ אֶת־ כָּל־ סַרְנֵ֨י פְלִשְׁתִּ֜ים אֲלֵיהֶ֗ם וַיֹּֽאמְרוּ֙ מַֽה־ נַּעֲשֶׂ֗ה לַֽאֲרוֹן֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל וַיֹּ֣אמְר֔וּ גַּ֣ת יִסֹּ֔ב אֲר֖וֹן אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וַיַּסֵּ֕בּוּ אֶת־ אֲר֖וֹן אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (לג) וַיְהִ֞י אַחֲרֵ֣י ׀ הֵסַ֣בּוּ אֹת֗וֹ וַתְּהִ֨י יַד־ יְהוָ֤ה ׀ בָּעִיר֙ מְהוּמָה֙ גְּדוֹלָ֣ה מְאֹ֔ד וַיַּךְ֙ אֶת־ אַנְשֵׁ֣י הָעִ֔יר מִקָּטֹ֖ן וְעַד־ גָּד֑וֹל וַיִּשָּׂתְר֥וּ לָהֶ֖ם (עפלים) טְחֹרִֽים׃ (לד) וַֽיְשַׁלְּח֛וּ אֶת־ אֲר֥וֹן הָאֱלֹהִ֖ים עֶקְר֑וֹן וַיְהִ֗י כְּב֨וֹא אֲר֤וֹן הָאֱלֹהִים֙ עֶקְר֔וֹן וַיִּזְעֲק֨וּ הָֽעֶקְרֹנִ֜ים לֵאמֹ֗ר הֵסַ֤בּוּ אֵלַי֙ אֶת־ אֲרוֹן֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל לַהֲמִיתֵ֖נִי וְאֶת־ עַמִּֽי׃ (לה) וַיִּשְׁלְח֨וּ וַיַּאַסְפ֜וּ אֶת־ כָּל־ סַרְנֵ֣י פְלִשְׁתִּ֗ים וַיֹּֽאמְרוּ֙ שַׁלְּח֞וּ אֶת־ אֲר֨וֹן אֱלֹהֵ֤י יִשְׂרָאֵל֙ וְיָשֹׁ֣ב לִמְקֹמ֔וֹ וְלֹֽא־ יָמִ֥ית אֹתִ֖י וְאֶת־ עַמִּ֑י כִּֽי־ הָיְתָ֤ה מְהֽוּמַת־ מָ֙וֶת֙ בְּכָל־ הָעִ֔יר כָּבְדָ֥ה מְאֹ֛ד יַ֥ד הָאֱלֹהִ֖ים שָֽׁם׃ (לו) וְהָֽאֲנָשִׁים֙ אֲשֶׁ֣ר לֹא־ מֵ֔תוּ הֻכּ֖וּ (בעפלים) בַּטְּחֹרִ֑ים וַתַּ֛עַל שַֽׁוְעַ֥ת הָעִ֖יר הַשָּׁמָֽיִם׃ (לז) וַיְהִ֧י אֲרוֹן־ יְהוָ֛ה בִּשְׂדֵ֥ה פְלִשְׁתִּ֖ים שִׁבְעָ֥ה חֳדָשִֽׁים׃ (לח) וַיִּקְרְא֣וּ פְלִשְׁתִּ֗ים לַכֹּהֲנִ֤ים וְלַקֹּֽסְמִים֙ לֵאמֹ֔ר מַֽה־ נַּעֲשֶׂ֖ה לַאֲר֣וֹן יְהוָ֑ה הוֹדִעֻ֕נוּ בַּמֶּ֖ה נְשַׁלְּחֶ֥נּוּ לִמְקוֹמֽוֹ׃ (לט) וַיֹּאמְר֗וּ אִֽם־ מְשַׁלְּחִ֞ים אֶת־ אֲר֨וֹן אֱלֹהֵ֤י יִשְׂרָאֵל֙ אַל־ תְּשַׁלְּח֤וּ אֹתוֹ֙ רֵיקָ֔ם כִּֽי־ הָשֵׁ֥ב תָּשִׁ֛יבוּ ל֖וֹ אָשָׁ֑ם אָ֤ז תֵּרָֽפְאוּ֙ וְנוֹדַ֣ע לָכֶ֔ם לָ֛מָּה לֹא־ תָס֥וּר יָד֖וֹ מִכֶּֽם׃ (מ) וַיֹּאמְר֗וּ מָ֣ה הָאָשָׁם֮ אֲשֶׁ֣ר נָשִׁ֣יב לוֹ֒ וַיֹּאמְר֗וּ מִסְפַּר֙ סַרְנֵ֣י פְלִשְׁתִּ֔ים חֲמִשָּׁה֙ (עפלי) טְחֹרֵ֣י זָהָ֔ב וַחֲמִשָּׁ֖ה עַכְבְּרֵ֣י זָהָ֑ב כִּֽי־ מַגֵּפָ֥ה אַחַ֛ת לְכֻלָּ֖ם וּלְסַרְנֵיכֶֽם׃ (מא) וַעֲשִׂיתֶם֩ צַלְמֵ֨י (עפליכם) טְחֹרֵיכֶ֜ם וְצַלְמֵ֣י עַכְבְּרֵיכֶ֗ם הַמַּשְׁחִיתִם֙ אֶת־ הָאָ֔רֶץ וּנְתַתֶּ֛ם לֵאלֹהֵ֥י יִשְׂרָאֵ֖ל כָּב֑וֹד אוּלַ֗י יָקֵ֤ל אֶת־ יָדוֹ֙ מֵֽעֲלֵיכֶ֔ם וּמֵעַ֥ל אֱלֹהֵיכֶ֖ם וּמֵעַ֥ל אַרְצְכֶֽם׃ (מב) וְלָ֤מָּה תְכַבְּדוּ֙ אֶת־ לְבַבְכֶ֔ם כַּאֲשֶׁ֧ר כִּבְּד֛וּ מִצְרַ֥יִם וּפַרְעֹ֖ה אֶת־ לִבָּ֑ם הֲלוֹא֙ כַּאֲשֶׁ֣ר הִתְעַלֵּ֣ל בָּהֶ֔ם וַֽיְשַׁלְּח֖וּם וַיֵּלֵֽכוּ׃ (מג) וְעַתָּ֗ה קְח֨וּ וַעֲשׂ֜וּ עֲגָלָ֤ה חֲדָשָׁה֙ אֶחָ֔ת וּשְׁתֵּ֤י פָרוֹת֙ עָל֔וֹת אֲשֶׁ֛ר לֹא־ עָלָ֥ה עֲלֵיהֶ֖ם עֹ֑ל וַאֲסַרְתֶּ֤ם אֶת־ הַפָּרוֹת֙ בָּעֲגָלָ֔ה וַהֲשֵׁיבֹתֶ֧ם בְּנֵיהֶ֛ם מֵאַחֲרֵיהֶ֖ם הַבָּֽיְתָה׃ (מד) וּלְקַחְתֶּ֞ם אֶת־ אֲר֣וֹן יְהוָ֗ה וּנְתַתֶּ֤ם אֹתוֹ֙ אֶל־ הָ֣עֲגָלָ֔ה וְאֵ֣ת ׀ כְּלֵ֣י הַזָּהָ֗ב אֲשֶׁ֨ר הֲשֵׁבֹתֶ֥ם לוֹ֙ אָשָׁ֔ם תָּשִׂ֥ימוּ בָאַרְגַּ֖ז מִצִּדּ֑וֹ וְשִׁלַּחְתֶּ֥ם אֹת֖וֹ וְהָלָֽךְ׃ (מה) וּרְאִיתֶ֗ם אִם־ דֶּ֨רֶךְ גְּבוּל֤וֹ יַֽעֲלֶה֙ בֵּ֣ית שֶׁ֔מֶשׁ ה֚וּא עָ֣שָׂה לָ֔נוּ אֶת־ הָרָעָ֥ה הַגְּדוֹלָ֖ה הַזֹּ֑את וְאִם־ לֹ֗א וְיָדַ֙עְנוּ֙ כִּ֣י לֹ֤א יָדוֹ֙ נָ֣גְעָה בָּ֔נוּ מִקְרֶ֥ה ה֖וּא הָ֥יָה לָֽנוּ׃ (מו) וַיַּעֲשׂ֤וּ הָאֲנָשִׁים֙ כֵּ֔ן וַיִּקְח֗וּ שְׁתֵּ֤י פָרוֹת֙ עָל֔וֹת וַיַּאַסְר֖וּם בָּעֲגָלָ֑ה וְאֶת־ בְּנֵיהֶ֖ם כָּל֥וּ בַבָּֽיִת׃ (מז) וַיָּשִׂ֛מוּ אֶת־ אֲר֥וֹן יְהוָ֖ה אֶל־ הָעֲגָלָ֑ה וְאֵ֣ת הָאַרְגַּ֗ז וְאֵת֙ עַכְבְּרֵ֣י הַזָּהָ֔ב וְאֵ֖ת צַלְמֵ֥י טְחֹרֵיהֶֽם׃ (מח) וַיִשַּׁ֨רְנָה הַפָּר֜וֹת בַּדֶּ֗רֶךְ עַל־ דֶּ֙רֶךְ֙ בֵּ֣ית שֶׁ֔מֶשׁ בִּמְסִלָּ֣ה אַחַ֗ת הָלְכ֤וּ הָלֹךְ֙ וְגָע֔וֹ וְלֹא־ סָ֖רוּ יָמִ֣ין וּשְׂמֹ֑אול וְסַרְנֵ֤י פְלִשְׁתִּים֙ הֹלְכִ֣ים אַחֲרֵיהֶ֔ם עַד־ גְּב֖וּל בֵּ֥ית שָֽׁמֶשׁ׃ (מט) וּבֵ֣ית שֶׁ֔מֶשׁ קֹצְרִ֥ים קְצִיר־ חִטִּ֖ים בָּעֵ֑מֶק וַיִּשְׂא֣וּ אֶת־ עֵינֵיהֶ֗ם וַיִּרְאוּ֙ אֶת־ הָ֣אָר֔וֹן וַֽיִּשְׂמְח֖וּ לִרְאֽוֹת׃
