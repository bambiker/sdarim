שמואל א פרק כג לפי החלוקה לסדרים

פרק כג בספר שמואל א (28 פסוקים) נמצא בשמואל סדר יד (פסוקים מא–מג) ובשמואל סדר טו (פסוקים א–כה).

שמואל סדר יד

(א) וַיֹּ֧אמֶר יְהוֹנָתָ֛ן לְדָוִ֖ד לֵ֣ךְ לְשָׁל֑וֹם אֲשֶׁר֩ נִשְׁבַּ֨עְנוּ שְׁנֵ֜ינוּ אֲנַ֗חְנוּ בְּשֵׁ֤ם יְהוָה֙ לֵאמֹ֔ר יְהוָ֞ה יִֽהְיֶ֣ה ׀ בֵּינִ֣י וּבֵינֶ֗ךָ וּבֵ֥ין זַרְעִ֛י וּבֵ֥ין זַרְעֲךָ֖ עַד־ עוֹלָֽם׃ (ב) וַיָּ֖קָם וַיֵּלַ֑ךְ וִיהוֹנָתָ֖ן בָּ֥א הָעִֽיר׃ (ג) וַיָּבֹ֤א דָוִד֙ נֹ֔בֶה אֶל־ אֲחִימֶ֖לֶךְ הַכֹּהֵ֑ן וַיֶּחֱרַ֨ד אֲחִימֶ֜לֶךְ לִקְרַ֣את דָּוִ֗ד וַיֹּ֤אמֶר לוֹ֙ מַדּ֤וּעַ אַתָּה֙ לְבַדֶּ֔ךָ וְאִ֖ישׁ אֵ֥ין אִתָּֽךְ׃ (ד) וַיֹּ֨אמֶר דָּוִ֜ד לַאֲחִימֶ֣לֶךְ הַכֹּהֵ֗ן הַמֶּלֶךְ֮ צִוַּ֣נִי דָבָר֒ וַיֹּ֣אמֶר אֵלַ֗י אִ֣ישׁ אַל־ יֵ֧דַע מְא֛וּמָה אֶת־ הַדָּבָ֛ר אֲשֶׁר־ אָנֹכִ֥י שֹׁלֵֽחֲךָ֖ וַאֲשֶׁ֣ר צִוִּיתִ֑ךָ וְאֶת־ הַנְּעָרִ֣ים יוֹדַ֔עְתִּי אֶל־ מְק֥וֹם פְּלֹנִ֖י אַלְמוֹנִֽי׃ (ה) וְעַתָּ֗ה מַה־ יֵּ֧שׁ תַּֽחַת־ יָדְךָ֛ חֲמִשָּׁה־ לֶ֖חֶם תְּנָ֣ה בְיָדִ֑י א֖וֹ הַנִּמְצָֽא׃ (ו) וַיַּ֨עַן הַכֹּהֵ֤ן אֶת־ דָּוִד֙ וַיֹּ֔אמֶר אֵֽין־ לֶ֥חֶם חֹ֖ל אֶל־ תַּ֣חַת יָדִ֑י כִּֽי־ אִם־ לֶ֤חֶם קֹ֙דֶשׁ֙ יֵ֔שׁ אִם־ נִשְׁמְר֥וּ הַנְּעָרִ֖ים אַ֥ךְ מֵאִשָּֽׁה׃ (ז) וַיַּעַן֩ דָּוִ֨ד אֶת־ הַכֹּהֵ֜ן וַיֹּ֣אמֶר ל֗וֹ כִּ֣י אִם־ אִשָּׁ֤ה עֲצֻֽרָה־ לָ֙נוּ֙ כִּתְמ֣וֹל שִׁלְשֹׁ֔ם בְּצֵאתִ֕י וַיִּהְי֥וּ כְלֵֽי־ הַנְּעָרִ֖ים קֹ֑דֶשׁ וְהוּא֙ דֶּ֣רֶךְ חֹ֔ל וְאַ֕ף כִּ֥י הַיּ֖וֹם יִקְדַּ֥שׁ בַּכֶּֽלִי׃ (ח) וַיִּתֶּן־ ל֥וֹ הַכֹּהֵ֖ן קֹ֑דֶשׁ כִּי֩ לֹא־ הָ֨יָה שָׁ֜ם לֶ֗חֶם כִּֽי־ אִם־ לֶ֤חֶם הַפָּנִים֙ הַמּֽוּסָרִים֙ מִלִּפְנֵ֣י יְהוָ֔ה לָשׂוּם֙ לֶ֣חֶם חֹ֔ם בְּי֖וֹם הִלָּקְחֽוֹ׃ (ט) וְשָׁ֡ם אִישׁ֩ מֵעַבְדֵ֨י שָׁא֜וּל בַּיּ֣וֹם הַה֗וּא נֶעְצָר֙ לִפְנֵ֣י יְהוָ֔ה וּשְׁמ֖וֹ דֹּאֵ֣ג הָאֲדֹמִ֑י אַבִּ֥יר הָרֹעִ֖ים אֲשֶׁ֥ר לְשָׁאֽוּל׃ (י) וַיֹּ֤אמֶר דָּוִד֙ לַאֲחִימֶ֔לֶךְ וְאִ֛ין יֶשׁ־ פֹּ֥ה תַֽחַת־ יָדְךָ֖ חֲנִ֣ית אוֹ־ חָ֑רֶב כִּ֣י גַם־ חַרְבִּ֤י וְגַם־ כֵּלַי֙ לֹֽא־ לָקַ֣חְתִּי בְיָדִ֔י כִּֽי־ הָיָ֥ה דְבַר־ הַמֶּ֖לֶךְ נָחֽוּץ׃ (יא) וַיֹּ֣אמֶר הַכֹּהֵ֗ן חֶרֶב֩ גָּלְיָ֨ת הַפְּלִשְׁתִּ֜י אֲשֶׁר־ הִכִּ֣יתָ ׀ בְּעֵ֣מֶק הָאֵלָ֗ה הִנֵּה־ הִ֞יא לוּטָ֣ה בַשִּׂמְלָה֮ אַחֲרֵ֣י הָאֵפוֹד֒ אִם־ אֹתָ֤הּ תִּֽקַּח־ לְךָ֙ קָ֔ח כִּ֣י אֵ֥ין אַחֶ֛רֶת זוּלָתָ֖הּ בָּזֶ֑ה וַיֹּ֧אמֶר דָּוִ֛ד אֵ֥ין כָּמ֖וֹהָ תְּנֶ֥נָּה לִּֽי׃ (יב) וַיָּ֣קָם דָּוִ֔ד וַיִּבְרַ֥ח בַּיּוֹם־ הַה֖וּא מִפְּנֵ֣י שָׁא֑וּל וַיָּבֹ֕א אֶל־ אָכִ֖ישׁ מֶ֥לֶךְ גַּֽת׃ (יג) וַיֹּ֨אמְר֜וּ עַבְדֵ֤י אָכִישׁ֙ אֵלָ֔יו הֲלוֹא־ זֶ֥ה דָוִ֖ד מֶ֣לֶךְ הָאָ֑רֶץ הֲל֣וֹא לָזֶ֗ה יַעֲנ֤וּ בַמְּחֹלוֹת֙ לֵאמֹ֔ר הִכָּ֤ה שָׁאוּל֙ (באלפו) בַּאֲלָפָ֔יו וְדָוִ֖ד (ברבבתו) בְּרִבְבֹתָֽיו׃ (יד) וַיָּ֧שֶׂם דָּוִ֛ד אֶת־ הַדְּבָרִ֥ים הָאֵ֖לֶּה בִּלְבָב֑וֹ וַיִּרָ֣א מְאֹ֔ד מִפְּנֵ֖י אָכִ֥ישׁ מֶֽלֶךְ־ גַּֽת׃ (טו) וַיְשַׁנּ֤וֹ אֶת־ טַעְמוֹ֙ בְּעֵ֣ינֵיהֶ֔ם וַיִּתְהֹלֵ֖ל בְּיָדָ֑ם (ויתו) וַיְתָיו֙ עַל־ דַּלְת֣וֹת הַשַּׁ֔עַר וַיּ֥וֹרֶד רִיר֖וֹ אֶל־ זְקָנֽוֹ׃ (טז) וַיֹּ֥אמֶר אָכִ֖ישׁ אֶל־ עֲבָדָ֑יו הִנֵּ֤ה תִרְאוּ֙ אִ֣ישׁ מִשְׁתַּגֵּ֔עַ לָ֛מָּה תָּבִ֥יאוּ אֹת֖וֹ אֵלָֽי׃ (יז) חֲסַ֤ר מְשֻׁגָּעִים֙ אָ֔נִי כִּי־ הֲבֵאתֶ֣ם אֶת־ זֶ֔ה לְהִשְׁתַּגֵּ֖עַ עָלָ֑י הֲזֶ֖ה יָב֥וֹא אֶל־ בֵּיתִֽי׃ (יח) וַיֵּ֤לֶךְ דָּוִד֙ מִשָּׁ֔ם וַיִּמָּלֵ֖ט אֶל־ מְעָרַ֣ת עֲדֻלָּ֑ם וַיִּשְׁמְע֤וּ אֶחָיו֙ וְכָל־ בֵּ֣ית אָבִ֔יו וַיֵּרְד֥וּ אֵלָ֖יו שָֽׁמָּה׃ (יט) וַיִּֽתְקַבְּצ֣וּ אֵ֠לָיו כָּל־ אִ֨ישׁ מָצ֜וֹק וְכָל־ אִ֨ישׁ אֲשֶׁר־ ל֤וֹ נֹשֶׁא֙ וְכָל־ אִ֣ישׁ מַר־ נֶ֔פֶשׁ וַיְהִ֥י עֲלֵיהֶ֖ם לְשָׂ֑ר וַיִּהְי֣וּ עִמּ֔וֹ כְּאַרְבַּ֥ע מֵא֖וֹת אִֽישׁ׃ (כ) וַיֵּ֧לֶךְ דָּוִ֛ד מִשָּׁ֖ם מִצְפֵּ֣ה מוֹאָ֑ב וַיֹּ֣אמֶר ׀ אֶל־ מֶ֣לֶךְ מוֹאָ֗ב יֵֽצֵא־ נָ֞א אָבִ֤י וְאִמִּי֙ אִתְּכֶ֔ם עַ֚ד אֲשֶׁ֣ר אֵדַ֔ע מַה־ יַּֽעֲשֶׂה־ לִּ֖י אֱלֹהִֽים׃ (כא) וַיַּנְחֵ֕ם אֶת־ פְּנֵ֖י מֶ֣לֶךְ מוֹאָ֑ב וַיֵּשְׁב֣וּ עִמּ֔וֹ כָּל־ יְמֵ֥י הֱיוֹת־ דָּוִ֖ד בַּמְּצוּדָֽה׃ (כב) וַיֹּאמֶר֩ גָּ֨ד הַנָּבִ֜יא אֶל־ דָּוִ֗ד לֹ֤א תֵשֵׁב֙ בַּמְּצוּדָ֔ה לֵ֥ךְ וּבָֽאתָ־ לְּךָ֖ אֶ֣רֶץ יְהוּדָ֑ה וַיֵּ֣לֶךְ דָּוִ֔ד וַיָּבֹ֖א יַ֥עַר חָֽרֶת׃ (כג) וַיִּשְׁמַ֣ע שָׁא֔וּל כִּ֚י נוֹדַ֣ע דָּוִ֔ד וַאֲנָשִׁ֖ים אֲשֶׁ֣ר אִתּ֑וֹ וְשָׁאוּל֩ יוֹשֵׁ֨ב בַּגִּבְעָ֜ה תַּֽחַת־ הָאֶ֤שֶׁל בָּֽרָמָה֙ וַחֲנִית֣וֹ בְיָד֔וֹ וְכָל־ עֲבָדָ֖יו נִצָּבִ֥ים עָלָֽיו׃ (כד) וַיֹּ֣אמֶר שָׁא֗וּל לַֽעֲבָדָיו֙ הַנִּצָּבִ֣ים עָלָ֔יו שִׁמְעוּ־ נָ֖א בְּנֵ֣י יְמִינִ֑י גַּם־ לְכֻלְּכֶ֗ם יִתֵּ֤ן בֶּן־ יִשַׁי֙ שָׂד֣וֹת וּכְרָמִ֔ים לְכֻלְּכֶ֣ם יָשִׂ֔ים שָׂרֵ֥י אֲלָפִ֖ים וְשָׂרֵ֥י מֵאֽוֹת׃ (כה) כִּי֩ קְשַׁרְתֶּ֨ם כֻּלְּכֶ֜ם עָלַ֗י וְאֵין־ גֹּלֶ֤ה אֶת־ אָזְנִי֙ בִּכְרָת־ בְּנִ֣י עִם־ בֶּן־ יִשַׁ֔י וְאֵין־ חֹלֶ֥ה מִכֶּ֛ם עָלַ֖י וְגֹלֶ֣ה אֶת־ אָזְנִ֑י כִּ֣י הֵקִים֩ בְּנִ֨י אֶת־ עַבְדִּ֥י עָלַ֛י לְאֹרֵ֖ב כַּיּ֥וֹם הַזֶּֽה׃ (כו) וַיַּ֜עַן דֹּאֵ֣ג הָאֲדֹמִ֗י וְה֛וּא נִצָּ֥ב עַל־ עַבְדֵֽי־ שָׁא֖וּל וַיֹּאמַ֑ר רָאִ֙יתִי֙ אֶת־ בֶּן־ יִשַׁ֔י בָּ֣א נֹ֔בֶה אֶל־ אֲחִימֶ֖לֶךְ בֶּן־ אֲחִטֽוּב׃ (כז) וַיִּשְׁאַל־ לוֹ֙ בַּֽיהוָ֔ה וְצֵידָ֖ה נָ֣תַן ל֑וֹ וְאֵ֗ת חֶ֛רֶב גָּלְיָ֥ת הַפְּלִשְׁתִּ֖י נָ֥תַן לֽוֹ׃ (כח) וַיִּשְׁלַ֣ח הַמֶּ֡לֶךְ לִקְרֹא֩ אֶת־ אֲחִימֶ֨לֶךְ בֶּן־ אֲחִיט֜וּב הַכֹּהֵ֗ן וְאֵ֨ת כָּל־ בֵּ֥ית אָבִ֛יו הַכֹּהֲנִ֖ים אֲשֶׁ֣ר בְּנֹ֑ב וַיָּבֹ֥אוּ כֻלָּ֖ם אֶל־ הַמֶּֽלֶךְ׃ (כט) וַיֹּ֣אמֶר שָׁא֔וּל שְֽׁמַֽע־ נָ֖א בֶּן־ אֲחִיט֑וּב וַיֹּ֖אמֶר הִנְנִ֥י אֲדֹנִֽי׃ (ל) וַיֹּ֤אמֶר (אלו) אֵלָיו֙ שָׁא֔וּל לָ֚מָּה קְשַׁרְתֶּ֣ם עָלַ֔י אַתָּ֖ה וּבֶן־ יִשָׁ֑י בְּתִתְּךָ֨ ל֜וֹ לֶ֣חֶם וְחֶ֗רֶב וְשָׁא֥וֹל לוֹ֙ בֵּֽאלֹהִ֔ים לָק֥וּם אֵלַ֛י לְאֹרֵ֖ב כַּיּ֥וֹם הַזֶּֽה׃ (לא) וַיַּ֧עַן אֲחִימֶ֛לֶךְ אֶת־ הַמֶּ֖לֶךְ וַיֹּאמַ֑ר וּמִ֤י בְכָל־ עֲבָדֶ֙יךָ֙ כְּדָוִ֣ד נֶאֱמָ֔ן וַחֲתַ֥ן הַמֶּ֛לֶךְ וְסָ֥ר אֶל־ מִשְׁמַעְתֶּ֖ךָ וְנִכְבָּ֥ד בְּבֵיתֶֽךָ׃ (לב) הַיּ֧וֹם הַחִלֹּ֛תִי (לשאול)־ לִשְׁאָל־ ל֥וֹ בֵאלֹהִ֖ים חָלִ֣ילָה לִּ֑י אַל־ יָשֵׂם֩ הַמֶּ֨לֶךְ בְּעַבְדּ֤וֹ דָבָר֙ בְּכָל־ בֵּ֣ית אָבִ֔י כִּ֠י לֹֽא־ יָדַ֤ע עַבְדְּךָ֙ בְּכָל־ זֹ֔את דָּבָ֥ר קָטֹ֖ן א֥וֹ גָדֽוֹל׃ (לג) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ מ֥וֹת תָּמ֖וּת אֲחִימֶ֑לֶךְ אַתָּ֖ה וְכָל־ בֵּ֥ית אָבִֽיךָ׃ (לד) וַיֹּ֣אמֶר הַמֶּ֡לֶךְ לָרָצִים֩ הַנִּצָּבִ֨ים עָלָ֜יו סֹ֥בּוּ וְהָמִ֣יתוּ ׀ כֹּהֲנֵ֣י יְהוָ֗ה כִּ֤י גַם־ יָדָם֙ עִם־ דָּוִ֔ד וְכִ֤י יָֽדְעוּ֙ כִּֽי־ בֹרֵ֣חַ ה֔וּא וְלֹ֥א גָל֖וּ אֶת־ (אזנו) אָזְנִ֑י וְלֹֽא־ אָב֞וּ עַבְדֵ֤י הַמֶּ֙לֶךְ֙ לִשְׁלֹ֣חַ וַאֲנָשָׁ֨יו קְעִילָ֜ה וַיִּלָּ֣חֶם בַּפְּלִשְׁתִּ֗ים וַיִּנְהַג֙ אֶת־ מִקְנֵיהֶ֔ם וַיַּ֥ךְ בָּהֶ֖ם מַכָּ֣ה גְדוֹלָ֑ה וַיֹּ֣שַׁע דָּוִ֔ד אֵ֖ת יֹשְׁבֵ֥י קְעִילָֽה׃ (ג) וַיְהִ֗י בִּ֠בְרֹחַ אֶבְיָתָ֧ר בֶּן־ אֲחִימֶ֛לֶךְ אֶל־ דָּוִ֖ד קְעִילָ֑ה אֵפ֖וֹד יָרַ֥ד בְּיָדֽוֹ׃ (ד) וַיֻּגַּ֣ד לְשָׁא֔וּל כִּי־ בָ֥א דָוִ֖ד קְעִילָ֑ה וַיֹּ֣אמֶר שָׁא֗וּל נִכַּ֨ר אֹת֤וֹ אֱלֹהִים֙ בְּיָדִ֔י כִּ֚י נִסְגַּ֣ר לָב֔וֹא דָּוִ֥ד וְאֶל־ אֲנָשָׁ֖יו לְתָפְשָֽׂם׃ (כד) וּמַלְאָ֣ךְ בָּ֔א אֶל־ שָׁא֖וּל לֵאמֹ֑ר מַהֲרָ֣ה וְלֵ֔כָה כִּֽי־ פָשְׁט֥וּ פְלִשְׁתִּ֖ים עַל־ הָאָֽרֶץ׃ (כה) וַיָּ֣שָׁב שָׁא֗וּל מִרְדֹף֙ אַחֲרֵ֣י דָוִ֔ד וַיֵּ֖לֶךְ לִקְרַ֣את פְּלִשְׁתִּ֑ים עַל־ כֵּ֗ן קָֽרְאוּ֙ לַמָּק֣וֹם הַה֔וּא סֶ֖לַע הַֽמַּחְלְקֽוֹת׃ סוף פרק כג (כו) וַיַּ֥עַל דָּוִ֖ד מִשָּׁ֑ם וַיֵּ֖שֶׁב בִּמְצָד֥וֹת עֵֽין־ גֶּֽדִי׃ (כז) וַיְהִ֗י כַּֽאֲשֶׁר֙ שָׁ֣ב שָׁא֔וּל מֵאַחֲרֵ֖י פְּלִשְׁתִּ֑ים וַיַּגִּ֤דוּ לוֹ֙ לֵאמֹ֔ר הִנֵּ֣ה דָוִ֔ד בְּמִדְבַּ֖ר עֵ֥ין גֶּֽדִי׃ (כח) וַיִּקַּ֣ח שָׁא֗וּל שְׁלֹ֧שֶׁת אֲלָפִ֛ים אִ֥ישׁ בָּח֖וּר מִכָּל־ יִשְׂרָאֵ֑ל וַיֵּ֗לֶךְ לְבַקֵּ֤שׁ אֶת־ דָּוִד֙ וַֽאֲנָשָׁ֔יו עַל־ פְּנֵ֖י צוּרֵ֥י הַיְּעֵלִֽים׃ (כט) וַ֠יָּבֹא אֶל־ גִּדְר֨וֹת הַצֹּ֤אן עַל־ הַדֶּ֙רֶךְ֙ וְשָׁ֣ם מְעָרָ֔ה וַיָּבֹ֥א שָׁא֖וּל לְהָסֵ֣ךְ אֶת־ רַגְלָ֑יו וְדָוִד֙ וַאֲנָשָׁ֔יו בְּיַרְכְּתֵ֥י הַמְּעָרָ֖ה יֹשְׁבִֽים׃ (ל) וַיֹּאמְרוּ֩ אַנְשֵׁ֨י דָוִ֜ד אֵלָ֗יו הִנֵּ֨ה הַיּ֜וֹם אֲ‍ֽשֶׁר־ אָמַ֧ר יְהוָ֣ה אֵלֶ֗יךָ הִנֵּ֨ה אָנֹכִ֜י נֹתֵ֤ן אֶת־ (איביך) אֹֽיִבְךָ֙ בְּיָדֶ֔ךָ וְעָשִׂ֣יתָ לּ֔וֹ כַּאֲשֶׁ֖ר יִטַ֣ב בְּעֵינֶ֑יךָ וַיָּ֣קָם דָּוִ֗ד וַיִּכְרֹ֛ת אֶת־ כְּנַֽף־ הַמְּעִ֥יל אֲשֶׁר־ לְשָׁא֖וּל בַּלָּֽט׃ (לא) וַֽיְהִי֙ אַֽחֲרֵי־ כֵ֔ן וַיַּ֥ךְ לֵב־ דָּוִ֖ד אֹת֑וֹ עַ֚ל אֲשֶׁ֣ר כָּרַ֔ת אֶת־ כָּנָ֖ף אֲשֶׁ֥ר לְשָׁאֽוּל׃ (לב) וַיֹּ֨אמֶר לַאֲנָשָׁ֜יו חָלִ֧ילָה לִּ֣י מֵֽיהוָ֗ה אִם־ אֶעֱשֶׂה֩ אֶת־ הַדָּבָ֨ר הַזֶּ֤ה לַֽאדֹנִי֙ לִמְשִׁ֣יחַ יְהוָ֔ה לִשְׁלֹ֥חַ יָדִ֖י בּ֑וֹ כִּֽי־ מְשִׁ֥יחַ יְהוָ֖ה הֽוּא׃ (לג) וַיְשַׁסַּ֨ע דָּוִ֤ד אֶת־ אֲנָשָׁיו֙ בַּדְּבָרִ֔ים וְלֹ֥א נְתָנָ֖ם לָק֣וּם אֶל־ שָׁא֑וּל וְשָׁא֛וּל קָ֥ם מֵהַמְּעָרָ֖ה וַיֵּ֥לֶךְ בַּדָּֽרֶךְ׃ (לד) וַיָּ֨קָם דָּוִ֜ד אַחֲרֵי־ כֵ֗ן וַיֵּצֵא֙ (מן)־ (המערה) מֵֽהַמְּעָרָ֔ה וַיִּקְרָ֧א אַֽחֲרֵי־ שָׁא֛וּל לֵאמֹ֖ר אֲדֹנִ֣י הַמֶּ֑לֶךְ וַיַּבֵּ֤ט שָׁאוּל֙ אַֽחֲרָ֔יו וַיִּקֹּ֨ד דָּוִ֥ד אַפַּ֛יִם אַ֖רְצָה וַיִּשְׁתָּֽחוּ׃ (לה) וַיֹּ֤אמֶר דָּוִד֙ לְשָׁא֔וּל לָ֧מָּה תִשְׁמַ֛ע אֶת־ דִּבְרֵ֥י אָדָ֖ם לֵאמֹ֑ר הִנֵּ֣ה דָוִ֔ד מְבַקֵּ֖שׁ רָעָתֶֽךָ׃ (לו) הִנֵּה֩ הַיּ֨וֹם הַזֶּ֜ה רָא֣וּ עֵינֶ֗יךָ אֵ֣ת אֲשֶׁר־ נְתָנְךָ֩ יְהוָ֨ה ׀ הַיּ֤וֹם ׀ בְּיָדִי֙ בַּמְּעָרָ֔ה וְאָמַ֥ר לַהֲרָגֲךָ֖ וַתָּ֣חָס עָלֶ֑יךָ וָאֹמַ֗ר לֹא־ אֶשְׁלַ֤ח יָדִי֙ בַּֽאדֹנִ֔י כִּי־ מְשִׁ֥יחַ יְהוָ֖ה הֽוּא׃ (לז) וְאָבִ֣י רְאֵ֔ה גַּ֗ם רְאֵ֛ה אֶת־ כְּנַ֥ף מְעִילְךָ֖ בְּיָדִ֑י כִּ֡י בְּכָרְתִי֩ אֶת־ כְּנַ֨ף מְעִֽילְךָ֜ וְלֹ֣א הֲרַגְתִּ֗יךָ דַּ֤ע וּרְאֵה֙ כִּי֩ אֵ֨ין בְּיָדִ֜י רָעָ֤ה וָפֶ֙שַׁע֙ וְלֹא־ חָטָ֣אתִי לָ֔ךְ וְאַתָּ֛ה צֹדֶ֥ה אֶת־ נַפְשִׁ֖י לְקַחְתָּֽהּ׃ (לח) יִשְׁפֹּ֤ט יְהוָה֙ בֵּינִ֣י וּבֵינֶ֔ךָ וּנְקָמַ֥נִי יְהוָ֖ה מִמֶּ֑ךָּ וְיָדִ֖י לֹ֥א תִֽהְיֶה־ בָּֽךְ׃ (לט) כַּאֲשֶׁ֣ר יֹאמַ֗ר מְשַׁל֙ הַקַּדְמֹנִ֔י מֵרְשָׁעִ֖ים יֵ֣צֵא רֶ֑שַׁע וְיָדִ֖י לֹ֥א תִֽהְיֶה־ בָּֽךְ׃ (מ) אַחֲרֵ֨י מִ֤י יָצָא֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל אַחֲרֵ֥י מִ֖י אַתָּ֣ה רֹדֵ֑ף אַֽחֲרֵי֙ כֶּ֣לֶב מֵ֔ת אַחֲרֵ֖י פַּרְעֹ֥שׁ אֶחָֽד׃ (מא) וְהָיָ֤ה יְהוָה֙ לְדַיָּ֔ן וְשָׁפַ֖ט בֵּינִ֣י וּבֵינֶ֑ךָ וְיֵ֙רֶא֙ וְיָרֵ֣ב אֶת־ רִיבִ֔י וְיִשְׁפְּטֵ֖נִי מִיָּדֶֽךָ׃ (מב) וַיְהִ֣י ׀ כְּכַלּ֣וֹת דָּוִ֗ד לְדַבֵּ֞ר אֶת־ הַדְּבָרִ֤ים הָאֵ֙לֶּה֙ אֶל־ שָׁא֔וּל וַיֹּ֣אמֶר שָׁא֔וּל הֲקֹלְךָ֥ זֶ֖ה בְּנִ֣י דָוִ֑ד וַיִּשָּׂ֥א שָׁא֛וּל קֹל֖וֹ וַיֵּֽבְךְּ׃ (מג) וַיֹּ֙אמֶר֙ אֶל־ דָּוִ֔ד צַדִּ֥יק אַתָּ֖ה מִמֶּ֑נִּי כִּ֤י אַתָּה֙ גְּמַלְתַּ֣נִי הַטּוֹבָ֔ה וַאֲנִ֖י גְּמַלְתִּ֥יךָ הָרָעָֽה׃ (מד) (ואת) וְאַתָּה֙ הִגַּ֣דְתָּ הַיּ֔וֹם אֵ֛ת אֲשֶׁר־ עָשִׂ֥יתָה אִתִּ֖י טוֹבָ֑ה אֵת֩ אֲשֶׁ֨ר סִגְּרַ֧נִי יְהוָ֛ה בְּיָדְךָ֖ וְלֹ֥א הֲרַגְתָּֽנִי׃
