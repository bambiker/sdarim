שמואל א פרק ל לפי החלוקה לסדרים

פרק ל בספר שמואל א (31 פסוקים) נמצא בשמואל סדר יט (פסוקים יד–לז) ובשמואל סדר כ (פסוקים א–ז).

שמואל סדר יט

(א) וְלָאִשָּׁ֤ה עֵֽגֶל־ מַרְבֵּק֙ בַּבַּ֔יִת וַתְּמַהֵ֖ר וַתִּזְבָּחֵ֑הוּ וַתִּקַּח־ קֶ֣מַח וַתָּ֔לָשׁ וַתֹּפֵ֖הוּ מַצּֽוֹת׃ (ב) וַתַּגֵּ֧שׁ לִפְנֵֽי־ שָׁא֛וּל וְלִפְנֵ֥י עֲבָדָ֖יו וַיֹּאכֵ֑לוּ וַיָּקֻ֥מוּ וַיֵּלְכ֖וּ בַּלַּ֥יְלָה הַהֽוּא׃ (ג) וַיִּקְבְּצ֧וּ פְלִשְׁתִּ֛ים אֶת־ כָּל־ מַחֲנֵיהֶ֖ם אֲפֵ֑קָה וְיִשְׂרָאֵ֣ל חֹנִ֔ים בַּעַ֖יִן אֲשֶׁ֥ר בְּיִזְרְעֶֽאל׃ (ד) וְסַרְנֵ֤י פְלִשְׁתִּים֙ עֹֽבְרִ֔ים לְמֵא֖וֹת וְלַאֲלָפִ֑ים וְדָוִ֣ד וַאֲנָשָׁ֗יו עֹֽבְרִ֛ים בָּאַחֲרֹנָ֖ה עִם־ אָכִֽישׁ׃ (ה) וַיֹּֽאמְרוּ֙ שָׂרֵ֣י פְלִשְׁתִּ֔ים מָ֖ה הָעִבְרִ֣ים הָאֵ֑לֶּה וַיֹּ֨אמֶר אָכִ֜ישׁ אֶל־ שָׂרֵ֣י פְלִשְׁתִּ֗ים הֲלֽוֹא־ זֶ֨ה דָוִ֜ד עֶ֣בֶד ׀ שָׁא֣וּל מֶֽלֶךְ־ יִשְׂרָאֵ֗ל אֲשֶׁ֨ר הָיָ֤ה אִתִּי֙ זֶ֤ה יָמִים֙ אוֹ־ זֶ֣ה שָׁנִ֔ים וְלֹֽא־ מָצָ֤אתִי בוֹ֙ מְא֔וּמָה מִיּ֥וֹם נָפְל֖וֹ עַד־ הַיּ֥וֹם הַזֶּֽה׃ (ו) וַיִּקְצְפ֨וּ עָלָ֜יו שָׂרֵ֣י פְלִשְׁתִּ֗ים וַיֹּ֣אמְרוּ לוֹ֩ שָׂרֵ֨י פְלִשְׁתִּ֜ים הָשֵׁ֣ב אֶת־ הָאִ֗ישׁ וְיָשֹׁב֙ אֶל־ מְקוֹמוֹ֙ אֲשֶׁ֣ר הִפְקַדְתּ֣וֹ שָׁ֔ם וְלֹֽא־ יֵרֵ֤ד עִמָּ֙נוּ֙ בַּמִּלְחָמָ֔ה וְלֹא־ יִֽהְיֶה־ לָּ֥נוּ לְשָׂטָ֖ן בַּמִּלְחָמָ֑ה וּבַמֶּ֗ה יִתְרַצֶּ֥ה זֶה֙ אֶל־ אֲדֹנָ֔יו הֲל֕וֹא בְּרָאשֵׁ֖י הָאֲנָשִׁ֥ים הָהֵֽם׃ (ז) הֲלוֹא־ זֶ֣ה דָוִ֔ד אֲשֶׁ֧ר יַעֲנוּ־ ל֛וֹ בַּמְּחֹל֖וֹת לֵאמֹ֑ר הִכָּ֤ה שָׁאוּל֙ בַּֽאֲלָפָ֔יו וְדָוִ֖ד (ברבבתו) בְּרִבְבֹתָֽיו׃ (ח) וַיִּקְרָ֨א אָכִ֜ישׁ אֶל־ דָּוִ֗ד וַיֹּ֣אמֶר אֵ֠לָיו חַי־ יְהוָ֞ה כִּי־ יָשָׁ֣ר אַתָּ֗ה וְט֣וֹב בְּ֠עֵינַי צֵאתְךָ֨ וּבֹאֲךָ֤ אִתִּי֙ בַּֽמַּחֲנֶ֔ה כִּ֠י לֹֽא־ מָצָ֤אתִֽי בְךָ֙ רָעָ֔ה מִיּ֛וֹם בֹּאֲךָ֥ אֵלַ֖י עַד־ הַיּ֣וֹם הַזֶּ֑ה וּבְעֵינֵ֥י הַסְּרָנִ֖ים לֹֽא־ ט֥וֹב אָֽתָּה׃ (ט) וְעַתָּ֥ה שׁ֖וּב וְלֵ֣ךְ בְּשָׁל֑וֹם וְלֹֽא־ תַעֲשֶׂ֣ה רָ֔ע בְּעֵינֵ֖י סַרְנֵ֥י פְלִשְׁתִּֽים׃ (י) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ אָכִ֗ישׁ כִּ֣י מֶ֤ה עָשִׂ֙יתִי֙ וּמַה־ מָּצָ֣אתָ בְעַבְדְּךָ֔ מִיּוֹם֙ אֲשֶׁ֣ר הָיִ֣יתִי לְפָנֶ֔יךָ עַ֖ד הַיּ֣וֹם הַזֶּ֑ה כִּ֣י לֹ֤א אָבוֹא֙ וְנִלְחַ֔מְתִּי בְּאֹיְבֵ֖י אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (יא) וַיַּ֣עַן אָכִישׁ֮ וַיֹּ֣אמֶר אֶל־ דָּוִד֒ יָדַ֕עְתִּי כִּ֣י ט֥וֹב אַתָּ֛ה בְּעֵינַ֖י כְּמַלְאַ֣ךְ אֱלֹהִ֑ים אַ֣ךְ שָׂרֵ֤י פְלִשְׁתִּים֙ אָֽמְר֔וּ לֹֽא־ יַעֲלֶ֥ה עִמָּ֖נוּ בַּמִּלְחָמָֽה׃ (יב) וְעַתָּה֙ הַשְׁכֵּ֣ם בַּבֹּ֔קֶר וְעַבְדֵ֥י אֲדֹנֶ֖יךָ אֲשֶׁר־ בָּ֣אוּ אִתָּ֑ךְ וְהִשְׁכַּמְתֶּ֣ם בַּבֹּ֔קֶר וְא֥וֹר לָכֶ֖ם וָלֵֽכוּ׃ (יג) וַיַּשְׁכֵּ֨ם דָּוִ֜ד ה֤וּא וַֽאֲנָשָׁיו֙ לָלֶ֣כֶת בַּבֹּ֔קֶר לָשׁ֖וּב אֶל־ אֶ֣רֶץ פְּלִשְׁתִּ֑ים וּפְלִשְׁתִּ֖ים עָל֥וּ יִזְרְעֶֽאל׃ תחילת פרק ל (יד) וַיְהִ֞י בְּבֹ֨א דָוִ֧ד וַאֲנָשָׁ֛יו צִֽקְלַ֖ג בַּיּ֣וֹם הַשְּׁלִישִׁ֑י וַעֲמָלֵקִ֣י פָֽשְׁט֗וּ אֶל־ נֶ֙גֶב֙ וְאֶל־ צִ֣קְלַ֔ג וַיַּכּוּ֙ אֶת־ צִ֣קְלַ֔ג וַיִּשְׂרְפ֥וּ אֹתָ֖הּ בָּאֵֽשׁ׃ (טו) וַיִּשְׁבּ֨וּ אֶת־ הַנָּשִׁ֤ים אֲשֶׁר־ בָּהּ֙ מִקָּטֹ֣ן וְעַד־ גָּד֔וֹל לֹ֥א הֵמִ֖יתוּ אִ֑ישׁ וַיִּֽנְהֲג֔וּ וַיֵּלְכ֖וּ לְדַרְכָּֽם׃ (טז) וַיָּבֹ֨א דָוִ֤ד וַֽאֲנָשָׁיו֙ אֶל־ הָעִ֔יר וְהִנֵּ֥ה שְׂרוּפָ֖ה בָּאֵ֑שׁ וּנְשֵׁיהֶ֛ם וּבְנֵיהֶ֥ם וּבְנֹתֵיהֶ֖ם נִשְׁבּֽוּ׃ (יז) וַיִּשָּׂ֨א דָוִ֜ד וְהָעָ֧ם אֲשֶׁר־ אִתּ֛וֹ אֶת־ קוֹלָ֖ם וַיִּבְכּ֑וּ עַ֣ד אֲשֶׁ֧ר אֵין־ בָּהֶ֛ם כֹּ֖חַ לִבְכּֽוֹת׃ (יח) וּשְׁתֵּ֥י נְשֵֽׁי־ דָוִ֖ד נִשְׁבּ֑וּ אֲחִינֹ֙עַם֙ הַיִּזְרְעֵלִ֔ית וַאֲבִיגַ֕יִל אֵ֖שֶׁת נָבָ֥ל הַֽכַּרְמְלִֽי׃ (יט) וַתֵּ֨צֶר לְדָוִ֜ד מְאֹ֗ד כִּֽי־ אָמְר֤וּ הָעָם֙ לְסָקְל֔וֹ כִּֽי־ מָ֙רָה֙ נֶ֣פֶשׁ כָּל־ הָעָ֔ם אִ֖ישׁ עַל־ (בנו) בָּנָ֣יו וְעַל־ בְּנֹתָ֑יו וַיִּתְחַזֵּ֣ק דָּוִ֔ד בַּיהוָ֖ה אֱלֹהָֽיו׃ (כ) וַיֹּ֣אמֶר דָּוִ֗ד אֶל־ אֶבְיָתָ֤ר הַכֹּהֵן֙ בֶּן־ אֲחִימֶ֔לֶךְ הַגִּֽישָׁה־ נָּ֥א לִ֖י הָאֵפֹ֑ד וַיַּגֵּ֧שׁ אֶבְיָתָ֛ר אֶת־ הָאֵפֹ֖ד אֶל־ דָּוִֽד׃ (כא) וַיִּשְׁאַ֨ל דָּוִ֤ד בַּֽיהוָה֙ לֵאמֹ֔ר אֶרְדֹּ֛ף אַחֲרֵ֥י הַגְּדוּד־ הַזֶּ֖ה הַֽאַשִּׂגֶ֑נּוּ וַיֹּ֤אמֶר לוֹ֙ רְדֹ֔ף כִּֽי־ הַשֵּׂ֥ג תַּשִּׂ֖יג וְהַצֵּ֥ל תַּצִּֽיל׃ (כב) וַיֵּ֣לֶךְ דָּוִ֗ד ה֚וּא וְשֵׁשׁ־ מֵא֥וֹת אִישׁ֙ אֲשֶׁ֣ר אִתּ֔וֹ וַיָּבֹ֖אוּ עַד־ נַ֣חַל הַבְּשׂ֑וֹר וְהַנּֽוֹתָרִ֖ים עָמָֽדוּ׃ (כג) וַיִּרְדֹּ֣ף דָּוִ֔ד ה֖וּא וְאַרְבַּע־ מֵא֣וֹת אִ֑ישׁ וַיַּֽעַמְדוּ֙ מָאתַ֣יִם אִ֔ישׁ אֲשֶׁ֣ר פִּגְּר֔וּ מֵעֲבֹ֖ר אֶת־ נַ֥חַל הַבְּשֽׂוֹר׃ (כד) וַֽיִּמְצְא֤וּ אִישׁ־ מִצְרִי֙ בַּשָּׂדֶ֔ה וַיִּקְח֥וּ אֹת֖וֹ אֶל־ דָּוִ֑ד וַיִּתְּנוּ־ ל֥וֹ לֶ֙חֶם֙ וַיֹּ֔אכַל וַיַּשְׁקֻ֖הוּ מָֽיִם׃ (כה) וַיִּתְּנוּ־ לוֹ֩ פֶ֨לַח דְּבֵלָ֜ה וּשְׁנֵ֤י צִמֻּקִים֙ וַיֹּ֔אכַל וַתָּ֥שָׁב רוּח֖וֹ אֵלָ֑יו כִּ֠י לֹֽא־ אָ֤כַל לֶ֙חֶם֙ וְלֹא־ שָׁ֣תָה מַ֔יִם שְׁלֹשָׁ֥ה יָמִ֖ים וּשְׁלֹשָׁ֥ה לֵילֽוֹת׃ (כו) וַיֹּ֨אמֶר ל֤וֹ דָוִד֙ לְֽמִי־ אַ֔תָּה וְאֵ֥י מִזֶּ֖ה אָ֑תָּה וַיֹּ֜אמֶר נַ֧עַר מִצְרִ֣י אָנֹ֗כִי עֶ֚בֶד לְאִ֣ישׁ עֲמָֽלֵקִ֔י וַיַּעַזְבֵ֧נִי אֲדֹנִ֛י כִּ֥י חָלִ֖יתִי הַיּ֥וֹם שְׁלֹשָֽׁה׃ (כז) אֲנַ֡חְנוּ פָּשַׁ֜טְנוּ נֶ֧גֶב הַכְּרֵתִ֛י וְעַל־ אֲשֶׁ֥ר לִֽיהוּדָ֖ה וְעַל־ נֶ֣גֶב כָּלֵ֑ב וְאֶת־ צִקְלַ֖ג שָׂרַ֥פְנוּ בָאֵֽשׁ׃ (כח) וַיֹּ֤אמֶר אֵלָיו֙ דָּוִ֔ד הֲתוֹרִדֵ֖נִי אֶל־ הַגְּד֣וּד הַזֶּ֑ה וַיֹּ֡אמֶר הִשָּׁבְעָה֩ לִּ֨י בֵֽאלֹהִ֜ים אִם־ תְּמִיתֵ֗נִי וְאִם־ תַּסְגִּרֵ֙נִי֙ בְּיַד־ אֲדֹנִ֔י וְאוֹרִֽדְךָ֖ אֶל־ הַגְּד֥וּד הַזֶּֽה׃ (כט) וַיֹּ֣רִדֵ֔הוּ וְהִנֵּ֥ה נְטֻשִׁ֖ים עַל־ פְּנֵ֣י כָל־ הָאָ֑רֶץ אֹכְלִ֤ים וְשֹׁתִים֙ וְחֹ֣גְגִ֔ים בְּכֹל֙ הַשָּׁלָ֣ל הַגָּד֔וֹל אֲשֶׁ֥ר לָקְח֛וּ מֵאֶ֥רֶץ פְּלִשְׁתִּ֖ים וּמֵאֶ֥רֶץ יְהוּדָֽה׃ (ל) וַיַּכֵּ֥ם דָּוִ֛ד מֵהַנֶּ֥שֶׁף וְעַד־ הָעֶ֖רֶב לְמָֽחֳרָתָ֑ם וְלֹֽא־ נִמְלַ֤ט מֵהֶם֙ אִ֔ישׁ כִּי֩ אִם־ אַרְבַּ֨ע מֵא֧וֹת אִֽישׁ־ נַ֛עַר אֲשֶׁר־ רָכְב֥וּ עַל־ הַגְּמַלִּ֖ים וַיָּנֻֽסוּ׃ (לא) וַיַּצֵּ֣ל דָּוִ֔ד אֵ֛ת כָּל־ אֲשֶׁ֥ר לָקְח֖וּ עֲמָלֵ֑ק וְאֶת־ שְׁתֵּ֥י נָשָׁ֖יו הִצִּ֥יל דָּוִֽד׃ (לב) וְלֹ֣א נֶעְדַּר־ לָ֠הֶם מִן־ הַקָּטֹ֨ן וְעַד־ הַגָּד֜וֹל וְעַד־ בָּנִ֤ים וּבָנוֹת֙ וּמִשָּׁלָ֔ל וְעַ֛ד כָּל־ אֲשֶׁ֥ר לָקְח֖וּ לָהֶ֑ם הַכֹּ֖ל הֵשִׁ֥יב דָּוִֽד׃ (לג) וַיִּקַּ֣ח דָּוִ֔ד אֶת־ כָּל־ הַצֹּ֖אן וְהַבָּקָ֑ר נָהֲג֗וּ לִפְנֵי֙ הַמִּקְנֶ֣ה הַה֔וּא וַיֹּ֣אמְר֔וּ זֶ֖ה שְׁלַ֥ל דָּוִֽד׃ (לד) וַיָּבֹ֣א דָוִ֗ד אֶל־ מָאתַ֨יִם כִּ֛י לֹ֥א יִֽחְיֶ֖ה אַחֲרֵ֣י נִפְל֑וֹ וָאֶקַּ֞ח הַנֵּ֣זֶר ׀ אֲשֶׁ֣ר עַל־ רֹאשׁ֗וֹ וְאֶצְעָדָה֙ אֲשֶׁ֣ר עַל־ זְרֹע֔וֹ וָאֲבִיאֵ֥ם אֶל־ אֲדֹנִ֖י הֵֽנָּה׃ (לא) וַיַּחֲזֵ֥ק דָּוִ֛ד (בבגדו) בִּבְגָדָ֖יו וַיִּקְרָעֵ֑ם וְגַ֥ם כָּל־ הָאֲנָשִׁ֖ים אֲשֶׁ֥ר אִתּֽוֹ׃ (לב) וַֽיִּסְפְּדוּ֙ וַיִּבְכּ֔וּ וַיָּצֻ֖מוּ עַד־ הָעָ֑רֶב עַל־ שָׁא֞וּל וְעַל־ יְהוֹנָתָ֣ן בְּנ֗וֹ וְעַל־ עַ֤ם יְהוָה֙ וְעַל־ בֵּ֣ית יִשְׂרָאֵ֔ל כִּ֥י נָפְל֖וּ בֶּחָֽרֶב׃ (לג) וַיֹּ֣אמֶר דָּוִ֗ד אֶל־ הַנַּ֙עַר֙ הַמַּגִּ֣יד ל֔וֹ אֵ֥י מִזֶּ֖ה אָ֑תָּה וַיֹּ֕אמֶר בֶּן־ אִ֛ישׁ גֵּ֥ר עֲמָלֵקִ֖י אָנֹֽכִי׃ (לד) וַיֹּ֥אמֶר אֵלָ֖יו דָּוִ֑ד אֵ֚יךְ לֹ֣א יָרֵ֔אתָ לִשְׁלֹ֙חַ֙ יָֽדְךָ֔ לְשַׁחֵ֖ת אֶת־ מְשִׁ֥יחַ יְהוָֽה׃ (לה) וַיִּקְרָ֣א דָוִ֗ד לְאַחַד֙ מֵֽהַנְּעָרִ֔ים וַיֹּ֖אמֶר גַּ֣שׁ פְּגַע־ בּ֑וֹ וַיַּכֵּ֖הוּ וַיָּמֹֽת׃ (לו) וַיֹּ֤אמֶר אֵלָיו֙ דָּוִ֔ד (דמיך) דָּמְךָ֖ עַל־ רֹאשֶׁ֑ךָ כִּ֣י פִ֗יךָ עָנָ֤ה בְךָ֙ לֵאמֹ֔ר אָנֹכִ֥י מֹתַ֖תִּי אֶת־ מְשִׁ֥יחַ יְהוָֽה׃ (לז) וַיְקֹנֵ֣ן דָּוִ֔ד אֶת־ הַקִּינָ֖ה הַזֹּ֑את עַל־ שָׁא֖וּל וְעַל־ יְהוֹנָתָ֥ן בְּנֽוֹ׃ (לח) וַיֹּ֕אמֶר לְלַמֵּ֥ד בְּנֵֽי־ יְהוּדָ֖ה קָ֑שֶׁת הִנֵּ֥ה כְתוּבָ֖ה עַל־ סֵ֥פֶר הַיָּשָֽׁר׃ (לט) הַצְּבִי֙ יִשְׂרָאֵ֔ל עַל־ בָּמוֹתֶ֖יךָ חָלָ֑ל אֵ֖יךְ נָפְל֥וּ גִבּוֹרִֽים׃ (מ) אַל־ תַּגִּ֣ידוּ בְגַ֔ת אַֽל־ תְּבַשְּׂר֖וּ בְּחוּצֹ֣ת אַשְׁקְל֑וֹן פֶּן־ תִּשְׂמַ֙חְנָה֙ בְּנ֣וֹת פְּלִשְׁתִּ֔ים פֶּֽן־ תַּעֲלֹ֖זְנָה בְּנ֥וֹת הָעֲרֵלִֽים׃ (מא) הָרֵ֣י בַגִּלְבֹּ֗עַ אַל־ טַ֧ל וְאַל־ מָטָ֛ר עֲלֵיכֶ֖ם וּשְׂדֵ֣י תְרוּמֹ֑ת כִּ֣י שָׁ֤ם נִגְעַל֙ מָגֵ֣ן גִּבּוֹרִ֔ים מָגֵ֣ן שָׁא֔וּל בְּלִ֖י מָשִׁ֥יחַ בַּשָּֽׁמֶן׃ (מב) מִדַּ֣ם חֲלָלִ֗ים מֵחֵ֙לֶב֙ גִּבּוֹרִ֔ים קֶ֚שֶׁת יְה֣וֹנָתָ֔ן לֹ֥א נָשׂ֖וֹג אָח֑וֹר וְחֶ֣רֶב שָׁא֔וּל לֹ֥א תָשׁ֖וּב רֵיקָֽם׃ (מג) שָׁא֣וּל וִיהוֹנָתָ֗ן הַנֶּאֱהָבִ֤ים וְהַנְּעִימִם֙ בְּחַיֵּיהֶ֔ם וּבְמוֹתָ֖ם לֹ֣א נִפְרָ֑דוּ מִנְּשָׁרִ֣ים קַ֔לּוּ מֵאֲרָי֖וֹת גָּבֵֽרוּ׃ (מד) בְּנוֹת֙ יִשְׂרָאֵ֔ל אֶל־ שָׁא֖וּל בְּכֶ֑ינָה הַמַּלְבִּֽשְׁכֶ֤ם שָׁנִי֙ עִם־ עֲדָנִ֔ים הַֽמַּעֲלֶה֙ עֲדִ֣י זָהָ֔ב עַ֖ל לְבוּשְׁכֶֽן׃ (מה) אֵ֚יךְ נָפְל֣וּ גִבֹּרִ֔ים בְּת֖וֹךְ הַמִּלְחָמָ֑ה יְה֣וֹנָתָ֔ן עַל־ בָּמוֹתֶ֖יךָ חָלָֽל׃ (מו) צַר־ לִ֣י עָלֶ֗יךָ אָחִי֙ יְה֣וֹנָתָ֔ן נָעַ֥מְתָּ לִּ֖י מְאֹ֑ד נִפְלְאַ֤תָה אַהֲבָֽתְךָ֙ לִ֔י מֵאַהֲבַ֖ת נָשִֽׁים׃ (מז) אֵ֚יךְ נָפְל֣וּ גִבּוֹרִ֔ים וַיֹּאבְד֖וּ כְּלֵ֥י מִלְחָמָֽה׃ (מח) וַיְהִ֣י אַֽחֲרֵי־ כֵ֗ן וַיִּשְׁאַל֩ דָּוִ֨ד בַּֽיהוָ֤ה ׀ לֵאמֹר֙ הַאֶעֱלֶ֗ה בְּאַחַת֙ עָרֵ֣י יְהוּדָ֔ה וַיֹּ֧אמֶר יְהוָ֛ה אֵלָ֖יו עֲלֵ֑ה וַיֹּ֧אמֶר דָּוִ֛ד אָ֥נָה אֶעֱלֶ֖ה וַיֹּ֥אמֶר חֶבְרֹֽנָה׃ (מט) וַיַּ֤עַל שָׁם֙ דָּוִ֔ד וְגַ֖ם שְׁתֵּ֣י נָשָׁ֑יו אֲחִינֹ֙עַם֙ הַיִּזְרְעֵלִ֔ית וַאֲבִיגַ֕יִל אֵ֖שֶׁת נָבָ֥ל הַֽכַּרְמְלִֽי׃ (נ) וַאֲנָשָׁ֧יו אֲשֶׁר־ עִמּ֛וֹ הֶעֱלָ֥ה דָוִ֖ד אִ֣ישׁ וּבֵית֑וֹ וַיֵּשְׁב֖וּ בְּעָרֵ֥י חֶבְרֽוֹן׃ (נא) וַיָּבֹ֙אוּ֙ אַנְשֵׁ֣י יְהוּדָ֔ה וַיִּמְשְׁחוּ־ שָׁ֧ם אֶת־ דָּוִ֛ד לְמֶ֖לֶךְ עַל־ בֵּ֣ית יְהוּדָ֑ה וַיַּגִּ֤דוּ לְדָוִד֙ לֵאמֹ֔ר אַנְשֵׁי֙ יָבֵ֣ישׁ גִּלְעָ֔ד אֲשֶׁ֥ר קָבְר֖וּ אֶת־ שָׁאֽוּל׃ (נב) וַיִּשְׁלַ֤ח דָּוִד֙ מַלְאָכִ֔ים אֶל־ אַנְשֵׁ֖י יָבֵ֣ישׁ גִּלְעָ֑ד וַיֹּ֣אמֶר אֲלֵיהֶ֗ם בְּרֻכִ֤ים אַתֶּם֙ לַֽיהוָ֔ה אֲשֶׁ֨ר עֲשִׂיתֶ֜ם הַחֶ֣סֶד הַזֶּ֗ה עִם־ אֲדֹֽנֵיכֶם֙ עִם־ שָׁא֔וּל וַֽתִּקְבְּר֖וּ אֹתֽוֹ׃ (נג) וְעַתָּ֕ה יַֽעַשׂ־ יְהוָ֥ה עִמָּכֶ֖ם חֶ֣סֶד וֶאֱמֶ֑ת וְגַ֣ם אָנֹכִ֗י אֶעֱשֶׂ֤ה אִתְּכֶם֙ הַטּוֹבָ֣ה הַזֹּ֔את אֲשֶׁ֥ר עֲשִׂיתֶ֖ם הַדָּבָ֥ר הַזֶּֽה׃
