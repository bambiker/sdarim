פרק ט בספר דניאל (27 פסוקים) נמצא בדניאל סדר ה (פסוקים עה–עז) ובדניאל סדר ו (פסוקים א–כד).

דניאל סדר ה

(א) וְ֠דָנִיֵּאל כְּדִ֨י יְדַ֜ע דִּֽי־ רְשִׁ֤ים כְּתָבָא֙ עַ֣ל לְבַיְתֵ֔הּ וְכַוִּ֨ין פְּתִיחָ֥ן לֵהּ֙ בְּעִלִּיתֵ֔הּ נֶ֖גֶד יְרוּשְׁלֶ֑ם וְזִמְנִין֩ תְּלָתָ֨ה בְיוֹמָ֜א ה֣וּא ׀ בָּרֵ֣ךְ עַל־ בִּרְכ֗וֹהִי וּמְצַלֵּ֤א וּמוֹדֵא֙ קֳדָ֣ם אֱלָהֵ֔הּ כָּל־ קֳבֵל֙ דִּֽי־ הֲוָ֣א עָבֵ֔ד מִן־ קַדְמַ֖ת דְּנָֽה׃ (ב) אֱ֠דַיִן גֻּבְרַיָּ֤א אִלֵּךְ֙ הַרְגִּ֔שׁוּ וְהַשְׁכַּ֖חוּ לְדָנִיֵּ֑אל בָּעֵ֥א וּמִתְחַנַּ֖ן קֳדָ֥ם אֱלָהֵֽהּ׃ (ג) בֵּ֠אדַיִן קְרִ֨יבוּ וְאָמְרִ֥ין קֳדָם־ מַלְכָּא֮ עַל־ אֱסָ֣ר מַלְכָּא֒ הֲלָ֧א אֱסָ֣ר רְשַׁ֗מְתָּ דִּ֣י כָל־ אֱנָ֡שׁ דִּֽי־ יִבְעֵה֩ מִן־ כָּל־ אֱלָ֨הּ וֶֽאֱנָ֜שׁ עַד־ יוֹמִ֣ין תְּלָתִ֗ין לָהֵן֙ מִנָּ֣ךְ מַלְכָּ֔א יִתְרְמֵ֕א לְג֖וֹב אַרְיָותָ֑א עָנֵ֨ה מַלְכָּ֜א וְאָמַ֗ר יַצִּיבָ֧א מִלְּתָ֛א כְּדָת־ מָדַ֥י וּפָרַ֖ס דִּי־ לָ֥א תֶעְדֵּֽא׃ (ד) בֵּ֠אדַיִן עֲנ֣וֹ וְאָמְרִין֮ קֳדָ֣ם מַלְכָּא֒ דִּ֣י דָנִיֵּ֡אל דִּי֩ מִן־ בְּנֵ֨י גָלוּתָ֜א דִּ֣י יְה֗וּד לָא־ שָׂ֨ם (עליך) עֲלָ֤ךְ מַלְכָּא֙ טְעֵ֔ם וְעַל־ אֱסָרָ֖א דִּ֣י רְשַׁ֑מְתָּ וְזִמְנִ֤ין תְּלָתָה֙ בְּיוֹמָ֔א בָּעֵ֖א בָּעוּתֵֽהּ׃ (ה) אֱדַ֨יִן מַלְכָּ֜א כְּדִ֧י מִלְּתָ֣א שְׁמַ֗ע שַׂגִּיא֙ בְּאֵ֣שׁ עֲל֔וֹהִי וְעַ֧ל דָּנִיֵּ֛אל שָׂ֥ם בָּ֖ל לְשֵׁיזָבוּתֵ֑הּ וְעַד֙ מֶֽעָלֵ֣י שִׁמְשָׁ֔א הֲוָ֥א מִשְׁתַּדַּ֖ר לְהַצָּלוּתֵֽהּ׃ (ו) בֵּאדַ֙יִן֙ גֻּבְרַיָּ֣א אִלֵּ֔ךְ הַרְגִּ֖שׁוּ עַל־ מַלְכָּ֑א וְאָמְרִ֣ין לְמַלְכָּ֗א דַּ֤ע מַלְכָּא֙ דִּֽי־ דָת֙ לְמָדַ֣י וּפָרַ֔ס דִּֽי־ כָל־ אֱסָ֥ר וּקְיָ֛ם דִּֽי־ מַלְכָּ֥א יְהָקֵ֖ים לָ֥א לְהַשְׁנָיָֽה׃ (ז) בֵּאדַ֜יִן מַלְכָּ֣א אֲמַ֗ר וְהַיְתִיו֙ לְדָ֣נִיֵּ֔אל וּרְמ֕וֹ לְגֻבָּ֖א דִּ֣י אַרְיָוָתָ֑א עָנֵ֤ה מַלְכָּא֙ וְאָמַ֣ר לְדָנִיֵּ֔אל אֱלָהָ֗ךְ דִּ֣י (אנתה) אַ֤נְתְּ פָּֽלַֽח־ לֵהּ֙ בִּתְדִירָ֔א ה֖וּא יְשֵׁיזְבִנָּֽךְ׃ (ח) וְהֵיתָ֙יִת֙ אֶ֣בֶן חֲדָ֔ה וְשֻׂמַ֖ת עַל־ פֻּ֣ם גֻּבָּ֑א וְחַתְמַ֨הּ מַלְכָּ֜א בְּעִזְקְתֵ֗הּ וּבְעִזְקָת֙ רַבְרְבָנ֔וֹהִי דִּ֛י לָא־ תִשְׁנֵ֥א צְב֖וּ בְּדָנִיֵּֽאל׃ (ט) אֱ֠דַיִן אֲזַ֨ל מַלְכָּ֤א לְהֵֽיכְלֵהּ֙ וּבָ֣ת טְוָ֔ת וְדַחֲוָ֖ן לָא־ הַנְעֵ֣ל קָֽדָמ֑וֹהִי וְשִׁנְתֵּ֖הּ נַדַּ֥ת עֲלֽוֹהִי׃ (י) בֵּאדַ֣יִן מַלְכָּ֔א בִּשְׁפַּרְפָּרָ֖א יְק֣וּם בְּנָגְהָ֑א וּבְהִ֨תְבְּהָלָ֔ה לְגֻבָּ֥א דִֽי־ אַרְיָוָתָ֖א אֲזַֽל׃ (יא) וּכְמִקְרְבֵ֣הּ לְגֻבָּ֔א לְדָ֣נִיֵּ֔אל בְּקָ֥ל עֲצִ֖יב זְעִ֑ק עָנֵ֨ה מַלְכָּ֜א וְאָמַ֣ר לְדָנִיֵּ֗אל דָּֽנִיֵּאל֙ עֲבֵד֙ אֱלָהָ֣א חַיָּ֔א אֱלָהָ֗ךְ דִּ֣י (אנתה) אַ֤נְתְּ פָּֽלַֽח־ לֵהּ֙ בִּתְדִירָ֔א הַיְכִ֥ל לְשֵׁיזָבוּתָ֖ךְ מִן־ אַרְיָוָתָֽא׃ (יב) אֱדַ֙יִן֙ דָּנִיֶּ֔אל עִם־ מַלְכָּ֖א מַלִּ֑ל מַלְכָּ֖א לְעָלְמִ֥ין חֱיִֽי׃ (יג) אֱלָהִ֞י שְׁלַ֣ח מַלְאֲכֵ֗הּ וּֽסֲגַ֛ר פֻּ֥ם אַרְיָוָתָ֖א וְלָ֣א חַבְּל֑וּנִי כָּל־ קֳבֵ֗ל דִּ֤י קָֽדָמ֙וֹהִי֙ זָכוּ֙ הִשְׁתְּכַ֣חַת לִ֔י וְאַ֤ף (קדמיך) קָֽדָמָךְ֙ מַלְכָּ֔א חֲבוּלָ֖ה לָ֥א עַבְדֵֽת׃ (יד) בֵּאדַ֣יִן מַלְכָּ֗א שַׂגִּיא֙ טְאֵ֣ב עֲל֔וֹהִי וּלְדָ֣נִיֵּ֔אל אֲמַ֖ר לְהַנְסָקָ֣ה מִן־ גֻּבָּ֑א וְהֻסַּ֨ק דָּנִיֵּ֜אל מִן־ גֻּבָּ֗א וְכָל־ חֲבָל֙ לָא־ הִשְׁתְּכַ֣ח בֵּ֔הּ דִּ֖י הֵימִ֥ן בֵּאלָהֵֽהּ׃ (טו) וַאֲמַ֣ר מַלְכָּ֗א וְהַיְתִ֞יו גֻּבְרַיָּ֤א אִלֵּךְ֙ דִּֽי־ אֲכַ֤לוּ קַרְצ֙וֹהִי֙ דִּ֣י דָֽנִיֵּ֔אל וּלְגֹ֤ב אַרְיָוָתָא֙ רְמ֔וֹ אִנּ֖וּן בְּנֵיה֣וֹן וּנְשֵׁיה֑וֹן וְלָֽא־ מְט֞וֹ לְאַרְעִ֣ית גֻּבָּ֗א עַ֠ד דִּֽי־ שְׁלִ֤טֽוּ בְהוֹן֙ אַרְיָ֣וָתָ֔א וְכָל־ גַּרְמֵיה֖וֹן הַדִּֽקוּ׃ (טז) בֵּאדַ֜יִן דָּרְיָ֣וֶשׁ מַלְכָּ֗א כְּ֠תַב לְֽכָל־ עַֽמְמַיָּ֞א אֻמַיָּ֧א וְלִשָּׁנַיָּ֛א דִּֽי־ (דארין) דָיְרִ֥ין בְּכָל־ אַרְעָ֖א שְׁלָמְכ֥וֹן יִשְׂגֵּֽא׃ (יז) מִן־ קֳדָמַי֮ שִׂ֣ים טְעֵם֒ דִּ֣י ׀ בְּכָל־ שָׁלְטָ֣ן מַלְכוּתִ֗י לֶהֱוֺ֤ן (זאעין) זָיְעִין֙ וְדָ֣חֲלִ֔ין מִן־ קֳדָ֖ם אֱלָהֵ֣הּ דִּי־ דָֽנִיֵּ֑אל דִּי־ ה֣וּא ׀ אֱלָהָ֣א חַיָּ֗א וְקַיָּם֙ לְעָ֣לְמִ֔ין וּמַלְכוּתֵהּ֙ דִּֽי־ לָ֣א תִתְחַבַּ֔ל וְשָׁלְטָנֵ֖הּ עַד־ סוֹפָֽא׃ (יח) מְשֵׁיזִ֣ב וּמַצִּ֗ל וְעָבֵד֙ אָתִ֣ין וְתִמְהִ֔ין בִּשְׁמַיָּ֖א וּבְאַרְעָ֑א דִּ֚י שֵׁיזִ֣יב לְדָֽנִיֵּ֔אל מִן־ יַ֖ד אַרְיָוָתָֽא׃ (יט) וְדָנִיֵּ֣אל דְּנָ֔ה הַצְלַ֖ח בְּמַלְכ֣וּת דָּרְיָ֑וֶשׁ וּבְמַלְכ֖וּת כּ֥וֹרֶשׁ (פרסיא) פָּרְסָאָֽה׃ (כ) בִּשְׁנַ֣ת חֲדָ֗ה לְבֵלְאשַׁצַּר֙ מֶ֣לֶךְ בָּבֶ֔ל דָּנִיֵּאל֙ חֵ֣לֶם חֲזָ֔ה וְחֶזְוֵ֥י רֵאשֵׁ֖הּ עַֽל־ מִשְׁכְּבֵ֑הּ בֵּאדַ֙יִן֙ חֶלְמָ֣א כְתַ֔ב רֵ֥אשׁ מִלִּ֖ין אֲמַֽר׃ (כא) עָנֵ֤ה דָנִיֵּאל֙ וְאָמַ֔ר חָזֵ֥ה הֲוֵ֛ית בְּחֶזְוִ֖י עִם־ לֵֽילְיָ֑א וַאֲר֗וּ אַרְבַּע֙ רוּחֵ֣י שְׁמַיָּ֔א מְגִיחָ֖ן לְיַמָּ֥א רַבָּֽא׃ (כב) וְאַרְבַּ֤ע חֵיוָן֙ רַבְרְבָ֔ן סָלְקָ֖ן מִן־ יַמָּ֑א שָׁנְיָ֖ן דָּ֥א מִן־ דָּֽא׃ (כג) קַדְמָיְתָ֣א כְאַרְיֵ֔ה וְגַפִּ֥ין דִּֽי־ נְשַׁ֖ר לַ֑הּ חָזֵ֣ה הֲוֵ֡ית עַד֩ דִּי־ מְּרִ֨יטוּ גַפַּ֜יהּ וּנְטִ֣ילַת מִן־ אַרְעָ֗א וְעַל־ רַגְלַ֙יִן֙ כֶּאֱנָ֣שׁ הֳקִימַ֔ת וּלְבַ֥ב אֱנָ֖שׁ יְהִ֥יב לַֽהּ׃ (כד) וַאֲר֣וּ חֵיוָה֩ אָחֳרִ֨י תִנְיָנָ֜ה דָּמְיָ֣ה לְדֹ֗ב וְלִשְׂטַר־ חַד֙ הֳקִמַ֔ת וּתְלָ֥ת עִלְעִ֛ין בְּפֻמַּ֖הּ בֵּ֣ין (שניה) שִׁנַּ֑הּ וְכֵן֙ אָמְרִ֣ין לַ֔הּ ק֥וּמִֽי אֲכֻ֖לִי בְּשַׂ֥ר שַׂגִּֽיא׃ (כה) בָּאתַ֨ר דְּנָ֜ה חָזֵ֣ה הֲוֵ֗ית וַאֲר֤וּ אָֽחֳרִי֙ כִּנְמַ֔ר וְלַ֨הּ גַּפִּ֥ין אַרְבַּ֛ע דִּי־ ע֖וֹף עַל־ (גביה) גַּבַּ֑הּ וְאַרְבְּעָ֤ה רֵאשִׁין֙ לְחֵ֣יוְתָ֔א וְשָׁלְטָ֖ן יְהִ֥יב לַֽהּ׃ (כו) בָּאתַ֣ר דְּנָה֩ חָזֵ֨ה הֲוֵ֜ית בְּחֶזְוֵ֣י לֵֽילְיָ֗א וַאֲר֣וּ חֵיוָ֣ה (רביעיה) רְֽבִיעָאָ֡ה דְּחִילָה֩ וְאֵֽימְתָנִ֨י וְתַקִּיפָ֜א יַתִּ֗ירָא וְשִׁנַּ֨יִן דִּֽי־ פַרְזֶ֥ל לַהּ֙ רַבְרְבָ֔ן אָֽכְלָ֣ה וּמַדֱּקָ֔ה וּשְׁאָרָ֖א (ברגליה) בְּרַגְלַ֣הּ רָפְסָ֑ה וְהִ֣יא מְשַׁנְּיָ֗ה מִן־ כָּל־ חֵֽיוָתָא֙ דִּ֣י קָֽדָמַ֔יהּ וְקַרְנַ֥יִן עֲשַׂ֖ר לַֽהּ׃ אֵלַ֔י כְּעֵ֖ת מִנְחַת־ עָֽרֶב׃ (יט) וַיָּ֖בֶן וַיְדַבֵּ֣ר עִמִּ֑י וַיֹּאמַ֕ר דָּנִיֵּ֕אל עַתָּ֥ה יָצָ֖אתִי לְהַשְׂכִּילְךָ֥ בִינָֽה׃ (כ) בִּתְחִלַּ֨ת תַּחֲנוּנֶ֜יךָ יָצָ֣א דָבָ֗ר וַאֲנִי֙ בָּ֣אתִי לְהַגִּ֔יד כִּ֥י חֲמוּד֖וֹת אָ֑תָּה וּבִין֙ בַּדָּבָ֔ר וְהָבֵ֖ן בַּמַּרְאֶֽה׃ (כא) שָׁבֻעִ֨ים שִׁבְעִ֜ים נֶחְתַּ֥ךְ עַֽל־ עַמְּךָ֣ ׀ וְעַל־ עִ֣יר קָדְשֶׁ֗ךָ לְכַלֵּ֨א הַפֶּ֜שַׁע (ולחתם) וּלְהָתֵ֤ם (חטאות) חַטָּאת֙ וּלְכַפֵּ֣ר עָוֺ֔ן וּלְהָבִ֖יא צֶ֣דֶק עֹֽלָמִ֑ים וְלַחְתֹּם֙ חָז֣וֹן וְנָבִ֔יא וְלִמְשֹׁ֖חַ קֹ֥דֶשׁ קָֽדָשִֽׁים׃ (כב) וְתֵדַ֨ע וְתַשְׂכֵּ֜ל מִן־ מֹצָ֣א דָבָ֗ר לְהָשִׁיב֙ וְלִבְנ֤וֹת יְרֽוּשָׁלִַ֙ם֙ עַד־ מָשִׁ֣יחַ נָגִ֔יד שָׁבֻעִ֖ים שִׁבְעָ֑ה וְשָׁבֻעִ֞ים שִׁשִּׁ֣ים וּשְׁנַ֗יִם תָּשׁוּב֙ וְנִבְנְתָה֙ רְח֣וֹב וְחָר֔וּץ וּבְצ֖וֹק הָעִתִּֽים׃ (כג) וְאַחֲרֵ֤י הַשָּׁבֻעִים֙ שִׁשִּׁ֣ים וּשְׁנַ֔יִם יִכָּרֵ֥ת מָשִׁ֖יחַ וְאֵ֣ין ל֑וֹ וְהָעִ֨יר וְהַקֹּ֜דֶשׁ יַ֠שְׁחִית עַ֣ם נָגִ֤יד הַבָּא֙ וְקִצּ֣וֹ בַשֶּׁ֔טֶף וְעַד֙ קֵ֣ץ מִלְחָמָ֔ה נֶחֱרֶ֖צֶת שֹׁמֵמֽוֹת׃ (כד) וְהִגְבִּ֥יר בְּרִ֛ית לָרַבִּ֖ים שָׁב֣וּעַ אֶחָ֑ד וַחֲצִ֨י הַשָּׁב֜וּעַ יַשְׁבִּ֣ית ׀ זֶ֣בַח וּמִנְחָ֗ה וְעַ֨ל כְּנַ֤ף שִׁקּוּצִים֙ מְשֹׁמֵ֔ם וְעַד־ כָּלָה֙ וְנֶ֣חֱרָצָ֔ה תִּתַּ֖ךְ עַל־ שֹׁמֵֽם׃ סוף פרק ט (כה) בִּשְׁנַ֣ת שָׁל֗וֹשׁ לְכ֙וֹרֶשׁ֙ מֶ֣לֶךְ פָּרַ֔ס דָּבָר֙ נִגְלָ֣ה לְדָֽנִיֵּ֔אל אֲשֶׁר־ נִקְרָ֥א שְׁמ֖וֹ בֵּלְטְשַׁאצַּ֑ר וֶאֱמֶ֤ת הַדָּבָר֙ וְצָבָ֣א גָד֔וֹל וּבִין֙ אֶת־ הַדָּבָ֔ר וּבִ֥ינָה ל֖וֹ בַּמַּרְאֶֽה׃ (כו) בַּיָּמִ֖ים הָהֵ֑ם אֲנִ֤י דָֽנִיֵּאל֙ הָיִ֣יתִי מִתְאַבֵּ֔ל שְׁלֹשָׁ֥ה שָׁבֻעִ֖ים יָמִֽים׃ (כז) לֶ֣חֶם חֲמֻד֞וֹת לֹ֣א אָכַ֗לְתִּי וּבָשָׂ֥ר וָיַ֛יִן לֹא־ בָ֥א אֶל־ פִּ֖י וְס֣וֹךְ לֹא־ סָ֑כְתִּי עַד־ מְלֹ֕את שְׁלֹ֥שֶׁת שָׁבֻעִ֖ים יָמִֽים׃ (כח) וּבְי֛וֹם עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה לַחֹ֣דֶשׁ הָרִאשׁ֑וֹן וַאֲנִ֗י הָיִ֛יתִי עַ֣ל יַ֧ד הַנָּהָ֛ר הַגָּד֖וֹל ה֥וּא חִדָּֽקֶל׃ (כט) וָאֶשָּׂ֤א אֶת־ עֵינַי֙ וָאֵ֔רֶא וְהִנֵּ֥ה אִישׁ־ אֶחָ֖ד לָב֣וּשׁ בַּדִּ֑ים וּמָתְנָ֥יו חֲגֻרִ֖ים בְּכֶ֥תֶם אוּפָֽז׃ (ל) וּגְוִיָּת֣וֹ כְתַרְשִׁ֗ישׁ וּפָנָ֞יו כְּמַרְאֵ֤ה בָרָק֙ וְעֵינָיו֙ כְּלַפִּ֣ידֵי אֵ֔שׁ וּזְרֹֽעֹתָיו֙ וּמַרְגְּלֹתָ֔יו כְּעֵ֖ין נְחֹ֣שֶׁת קָלָ֑ל וְק֥וֹל דְּבָרָ֖יו כְּק֥וֹל הָמֽוֹן׃ (לא) וְרָאִיתִי֩ אֲנִ֨י דָנִיֵּ֤אל לְבַדִּי֙ אֶת־ הַמַּרְאָ֔ה וְהָאֲנָשִׁים֙ אֲשֶׁ֣ר הָי֣וּ עִמִּ֔י לֹ֥א רָא֖וּ אֶת־ הַמַּרְאָ֑ה אֲבָ֗ל חֲרָדָ֤ה גְדֹלָה֙ נָפְלָ֣ה עֲלֵיהֶ֔ם וַֽיִּבְרְח֖וּ בְּהֵחָבֵֽא׃ (לב) וַאֲנִי֙ נִשְׁאַ֣רְתִּי לְבַדִּ֔י וָֽאֶרְאֶ֗ה אֶת־ הַמַּרְאָ֤ה הַגְּדֹלָה֙ הַזֹּ֔את וְלֹ֥א נִשְׁאַר־ בִּ֖י כֹּ֑ח וְהוֹדִ֗י נֶהְפַּ֤ךְ עָלַי֙ לְמַשְׁחִ֔ית וְלֹ֥א עָצַ֖רְתִּי כֹּֽחַ׃ (לג) וָאֶשְׁמַ֖ע אֶת־ ק֣וֹל דְּבָרָ֑יו וּכְשָׁמְעִי֙ אֶת־ ק֣וֹל דְּבָרָ֔יו וַאֲנִ֗י הָיִ֛יתִי נִרְדָּ֥ם עַל־ פָּנַ֖י וּפָנַ֥י אָֽרְצָה׃ (לד) וְהִנֵּה־ יָ֖ד נָ֣גְעָה בִּ֑י וַתְּנִיעֵ֥נִי עַל־ בִּרְכַּ֖י וְכַפּ֥וֹת יָדָֽי׃ (לה) וַיֹּ֣אמֶר אֵלַ֡י דָּנִיֵּ֣אל אִישׁ־ חֲ֠מֻדוֹת הָבֵ֨ן בַּדְּבָרִ֜ים אֲשֶׁר֩ אָנֹכִ֨י דֹבֵ֤ר אֵלֶ֙יךָ֙ וַעֲמֹ֣ד עַל־ עָמְדֶ֔ךָ כִּ֥י עַתָּ֖ה שֻׁלַּ֣חְתִּי אֵלֶ֑יךָ וּבְדַבְּר֥וֹ עִמִּ֛י אֶת־ הַדָּבָ֥ר הַזֶּ֖ה עָמַ֥דְתִּי מַרְעִֽיד׃ (לו) וַיֹּ֣אמֶר אֵלַי֮ אַל־ תִּירָ֣א דָנִיֵּאל֒ כִּ֣י ׀ מִן־ הַיּ֣וֹם הָרִאשׁ֗וֹן אֲשֶׁ֨ר נָתַ֧תָּ אֶֽת־ לִבְּךָ֛ לְהָבִ֧ין וּלְהִתְעַנּ֛וֹת לִפְנֵ֥י אֱלֹהֶ֖יךָ נִשְׁמְע֣וּ דְבָרֶ֑יךָ וַאֲנִי־ בָ֖אתִי בִּדְבָרֶֽיךָ׃ (לז) וְשַׂ֣ר ׀ מַלְכ֣וּת פָּרַ֗ס עֹמֵ֤ד לְנֶגְדִּי֙ עֶשְׂרִ֣ים וְאֶחָ֣ד י֔וֹם וְהִנֵּ֣ה מִֽיכָאֵ֗ל אַחַ֛ד הַשָּׂרִ֥ים הָרִאשֹׁנִ֖ים בָּ֣א לְעָזְרֵ֑נִי וַאֲנִי֙ נוֹתַ֣רְתִּי שָׁ֔ם אֵ֖צֶל מַלְכֵ֥י פָרָֽס׃ (לח) וּבָ֙אתִי֙ לַהֲבִ֣ינְךָ֔ אֵ֛ת אֲשֶׁר־ יִקְרָ֥ה לְעַמְּךָ֖ בְּאַחֲרִ֣ית הַיָּמִ֑ים כִּי־ ע֥וֹד חָז֖וֹן לַיָּמִֽים׃ (לט) וּבְדַבְּר֣וֹ עִמִּ֔י כַּדְּבָרִ֖ים הָאֵ֑לֶּה נָתַ֧תִּי פָנַ֛י אַ֖רְצָה וְנֶאֱלָֽמְתִּי׃ (מ) וְהִנֵּ֗ה כִּדְמוּת֙ בְּנֵ֣י אָדָ֔ם נֹגֵ֖עַ עַל־ שְׂפָתָ֑י וָאֶפְתַּח־ פִּ֗י וָאֲדַבְּרָה֙ וָאֹֽמְרָה֙ אֶל־ הָעֹמֵ֣ד לְנֶגְדִּ֔י אֲדֹנִ֗י בַּמַּרְאָה֙ נֶהֶפְכ֤וּ צִירַי֙ עָלַ֔י וְלֹ֥א עָצַ֖רְתִּי כֹּֽחַ׃ (מא) וְהֵ֣יךְ יוּכַ֗ל עֶ֤בֶד אֲדֹנִי֙ זֶ֔ה לְדַבֵּ֖ר עִם־ אֲדֹ֣נִי זֶ֑ה וַאֲנִ֤י מֵעַ֙תָּה֙ לֹֽא־ יַעֲמָד־ בִּ֣י כֹ֔חַ וּנְשָׁמָ֖ה לֹ֥א נִשְׁאֲרָה־ בִֽי׃ (מב) וַיֹּ֧סֶף וַיִּגַּע־ בִּ֛י כְּמַרְאֵ֥ה אָדָ֖ם וַֽיְחַזְּקֵֽנִי׃ (מג) וַיֹּ֜אמֶר אַל־ תִּירָ֧א אִישׁ־ חֲמֻד֛וֹת שָׁל֥וֹם לָ֖ךְ חֲזַ֣ק וַחֲזָ֑ק וּֽבְדַבְּר֤וֹ עִמִּי֙ הִתְחַזַּ֔קְתִּי וָאֹֽמְרָ֛ה יְדַבֵּ֥ר אֲדֹנִ֖י כִּ֥י חִזַּקְתָּֽנִי׃ (מד) וַיֹּ֗אמֶר הֲיָדַ֙עְתָּ֙ לָמָּה־ בָּ֣אתִי אֵלֶ֔יךָ וְעַתָּ֣ה אָשׁ֔וּב לְהִלָּחֵ֖ם עִם־ שַׂ֣ר פָּרָ֑ס וַאֲנִ֣י יוֹצֵ֔א וְהִנֵּ֥ה שַׂר־ יָוָ֖ן בָּֽא׃
