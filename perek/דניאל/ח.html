דניאל פרק ח לפי החלוקה לסדרים

פרק ח בספר דניאל (27 פסוקים) נמצא בדניאל סדר ה (פסוקים מח–עד).

דניאל סדר ה

(א) וְ֠דָנִיֵּאל כְּדִ֨י יְדַ֜ע דִּֽי־ רְשִׁ֤ים כְּתָבָא֙ עַ֣ל לְבַיְתֵ֔הּ וְכַוִּ֨ין פְּתִיחָ֥ן לֵהּ֙ בְּעִלִּיתֵ֔הּ נֶ֖גֶד יְרוּשְׁלֶ֑ם וְזִמְנִין֩ תְּלָתָ֨ה בְיוֹמָ֜א ה֣וּא ׀ בָּרֵ֣ךְ עַל־ בִּרְכ֗וֹהִי וּמְצַלֵּ֤א וּמוֹדֵא֙ קֳדָ֣ם אֱלָהֵ֔הּ כָּל־ קֳבֵל֙ דִּֽי־ הֲוָ֣א עָבֵ֔ד מִן־ קַדְמַ֖ת דְּנָֽה׃ (ב) אֱ֠דַיִן גֻּבְרַיָּ֤א אִלֵּךְ֙ הַרְגִּ֔שׁוּ וְהַשְׁכַּ֖חוּ לְדָנִיֵּ֑אל בָּעֵ֥א וּמִתְחַנַּ֖ן קֳדָ֥ם אֱלָהֵֽהּ׃ (ג) בֵּ֠אדַיִן קְרִ֨יבוּ וְאָמְרִ֥ין קֳדָם־ מַלְכָּא֮ עַל־ אֱסָ֣ר מַלְכָּא֒ הֲלָ֧א אֱסָ֣ר רְשַׁ֗מְתָּ דִּ֣י כָל־ אֱנָ֡שׁ דִּֽי־ יִבְעֵה֩ מִן־ כָּל־ אֱלָ֨הּ וֶֽאֱנָ֜שׁ עַד־ יוֹמִ֣ין תְּלָתִ֗ין לָהֵן֙ מִנָּ֣ךְ מַלְכָּ֔א יִתְרְמֵ֕א לְג֖וֹב אַרְיָותָ֑א עָנֵ֨ה מַלְכָּ֜א וְאָמַ֗ר יַצִּיבָ֧א מִלְּתָ֛א כְּדָת־ מָדַ֥י וּפָרַ֖ס דִּי־ לָ֥א תֶעְדֵּֽא׃ (ד) בֵּ֠אדַיִן עֲנ֣וֹ וְאָמְרִין֮ קֳדָ֣ם מַלְכָּא֒ דִּ֣י דָנִיֵּ֡אל דִּי֩ מִן־ בְּנֵ֨י גָלוּתָ֜א דִּ֣י יְה֗וּד לָא־ שָׂ֨ם (עליך) עֲלָ֤ךְ מַלְכָּא֙ טְעֵ֔ם וְעַל־ אֱסָרָ֖א דִּ֣י רְשַׁ֑מְתָּ וְזִמְנִ֤ין תְּלָתָה֙ בְּיוֹמָ֔א בָּעֵ֖א בָּעוּתֵֽהּ׃ (ה) אֱדַ֨יִן מַלְכָּ֜א כְּדִ֧י מִלְּתָ֣א שְׁמַ֗ע שַׂגִּיא֙ בְּאֵ֣שׁ עֲל֔וֹהִי וְעַ֧ל דָּנִיֵּ֛אל שָׂ֥ם בָּ֖ל לְשֵׁיזָבוּתֵ֑הּ וְעַד֙ מֶֽעָלֵ֣י שִׁמְשָׁ֔א הֲוָ֥א מִשְׁתַּדַּ֖ר לְהַצָּלוּתֵֽהּ׃ (ו) בֵּאדַ֙יִן֙ גֻּבְרַיָּ֣א אִלֵּ֔ךְ הַרְגִּ֖שׁוּ עַל־ מַלְכָּ֑א וְאָמְרִ֣ין לְמַלְכָּ֗א דַּ֤ע מַלְכָּא֙ דִּֽי־ דָת֙ לְמָדַ֣י וּפָרַ֔ס דִּֽי־ כָל־ אֱסָ֥ר וּקְיָ֛ם דִּֽי־ מַלְכָּ֥א יְהָקֵ֖ים לָ֥א לְהַשְׁנָיָֽה׃ (ז) בֵּאדַ֜יִן מַלְכָּ֣א אֲמַ֗ר וְהַיְתִיו֙ לְדָ֣נִיֵּ֔אל וּרְמ֕וֹ לְגֻבָּ֖א דִּ֣י אַרְיָוָתָ֑א עָנֵ֤ה מַלְכָּא֙ וְאָמַ֣ר לְדָנִיֵּ֔אל אֱלָהָ֗ךְ דִּ֣י (אנתה) אַ֤נְתְּ פָּֽלַֽח־ לֵהּ֙ בִּתְדִירָ֔א ה֖וּא יְשֵׁיזְבִנָּֽךְ׃ (ח) וְהֵיתָ֙יִת֙ אֶ֣בֶן חֲדָ֔ה וְשֻׂמַ֖ת עַל־ פֻּ֣ם גֻּבָּ֑א וְחַתְמַ֨הּ מַלְכָּ֜א בְּעִזְקְתֵ֗הּ וּבְעִזְקָת֙ רַבְרְבָנ֔וֹהִי דִּ֛י לָא־ תִשְׁנֵ֥א צְב֖וּ בְּדָנִיֵּֽאל׃ (ט) אֱ֠דַיִן אֲזַ֨ל מַלְכָּ֤א לְהֵֽיכְלֵהּ֙ וּבָ֣ת טְוָ֔ת וְדַחֲוָ֖ן לָא־ הַנְעֵ֣ל קָֽדָמ֑וֹהִי וְשִׁנְתֵּ֖הּ נַדַּ֥ת עֲלֽוֹהִי׃ (י) בֵּאדַ֣יִן מַלְכָּ֔א בִּשְׁפַּרְפָּרָ֖א יְק֣וּם בְּנָגְהָ֑א וּבְהִ֨תְבְּהָלָ֔ה לְגֻבָּ֥א דִֽי־ אַרְיָוָתָ֖א אֲזַֽל׃ (יא) וּכְמִקְרְבֵ֣הּ לְגֻבָּ֔א לְדָ֣נִיֵּ֔אל בְּקָ֥ל עֲצִ֖יב זְעִ֑ק עָנֵ֨ה מַלְכָּ֜א וְאָמַ֣ר לְדָנִיֵּ֗אל דָּֽנִיֵּאל֙ עֲבֵד֙ אֱלָהָ֣א חַיָּ֔א אֱלָהָ֗ךְ דִּ֣י (אנתה) אַ֤נְתְּ פָּֽלַֽח־ לֵהּ֙ בִּתְדִירָ֔א הַיְכִ֥ל לְשֵׁיזָבוּתָ֖ךְ מִן־ אַרְיָוָתָֽא׃ (יב) אֱדַ֙יִן֙ דָּנִיֶּ֔אל עִם־ מַלְכָּ֖א מַלִּ֑ל מַלְכָּ֖א לְעָלְמִ֥ין חֱיִֽי׃ (יג) אֱלָהִ֞י שְׁלַ֣ח מַלְאֲכֵ֗הּ וּֽסֲגַ֛ר פֻּ֥ם אַרְיָוָתָ֖א וְלָ֣א חַבְּל֑וּנִי כָּל־ קֳבֵ֗ל דִּ֤י קָֽדָמ֙וֹהִי֙ זָכוּ֙ הִשְׁתְּכַ֣חַת לִ֔י וְאַ֤ף (קדמיך) קָֽדָמָךְ֙ מַלְכָּ֔א חֲבוּלָ֖ה לָ֥א עַבְדֵֽת׃ (יד) בֵּאדַ֣יִן מַלְכָּ֗א שַׂגִּיא֙ טְאֵ֣ב עֲל֔וֹהִי וּלְדָ֣נִיֵּ֔אל אֲמַ֖ר לְהַנְסָקָ֣ה מִן־ גֻּבָּ֑א וְהֻסַּ֨ק דָּנִיֵּ֜אל מִן־ גֻּבָּ֗א וְכָל־ חֲבָל֙ לָא־ הִשְׁתְּכַ֣ח בֵּ֔הּ דִּ֖י הֵימִ֥ן בֵּאלָהֵֽהּ׃ (טו) וַאֲמַ֣ר מַלְכָּ֗א וְהַיְתִ֞יו גֻּבְרַיָּ֤א אִלֵּךְ֙ דִּֽי־ אֲכַ֤לוּ קַרְצ֙וֹהִי֙ דִּ֣י דָֽנִיֵּ֔אל וּלְגֹ֤ב אַרְיָוָתָא֙ רְמ֔וֹ אִנּ֖וּן בְּנֵיה֣וֹן וּנְשֵׁיה֑וֹן וְלָֽא־ מְט֞וֹ לְאַרְעִ֣ית גֻּבָּ֗א עַ֠ד דִּֽי־ שְׁלִ֤טֽוּ בְהוֹן֙ אַרְיָ֣וָתָ֔א וְכָל־ גַּרְמֵיה֖וֹן הַדִּֽקוּ׃ (טז) בֵּאדַ֜יִן דָּרְיָ֣וֶשׁ מַלְכָּ֗א כְּ֠תַב לְֽכָל־ עַֽמְמַיָּ֞א אֻמַיָּ֧א וְלִשָּׁנַיָּ֛א דִּֽי־ (דארין) דָיְרִ֥ין בְּכָל־ אַרְעָ֖א שְׁלָמְכ֥וֹן יִשְׂגֵּֽא׃ (יז) מִן־ קֳדָמַי֮ שִׂ֣ים טְעֵם֒ דִּ֣י ׀ בְּכָל־ שָׁלְטָ֣ן מַלְכוּתִ֗י לֶהֱוֺ֤ן (זאעין) זָיְעִין֙ וְדָ֣חֲלִ֔ין מִן־ קֳדָ֖ם אֱלָהֵ֣הּ דִּי־ דָֽנִיֵּ֑אל דִּי־ ה֣וּא ׀ אֱלָהָ֣א חַיָּ֗א וְקַיָּם֙ לְעָ֣לְמִ֔ין וּמַלְכוּתֵהּ֙ דִּֽי־ לָ֣א תִתְחַבַּ֔ל וְשָׁלְטָנֵ֖הּ עַד־ סוֹפָֽא׃ (יח) מְשֵׁיזִ֣ב וּמַצִּ֗ל וְעָבֵד֙ אָתִ֣ין וְתִמְהִ֔ין בִּשְׁמַיָּ֖א וּבְאַרְעָ֑א דִּ֚י שֵׁיזִ֣יב לְדָֽנִיֵּ֔אל מִן־ יַ֖ד אַרְיָוָתָֽא׃ (יט) וְדָנִיֵּ֣אל דְּנָ֔ה הַצְלַ֖ח בְּמַלְכ֣וּת דָּרְיָ֑וֶשׁ וּבְמַלְכ֖וּת כּ֥וֹרֶשׁ (פרסיא) פָּרְסָאָֽה׃ (כ) בִּשְׁנַ֣ת חֲדָ֗ה לְבֵלְאשַׁצַּר֙ מֶ֣לֶךְ בָּבֶ֔ל דָּנִיֵּאל֙ חֵ֣לֶם חֲזָ֔ה וְחֶזְוֵ֥י רֵאשֵׁ֖הּ עַֽל־ מִשְׁכְּבֵ֑הּ בֵּאדַ֙יִן֙ חֶלְמָ֣א כְתַ֔ב רֵ֥אשׁ מִלִּ֖ין אֲמַֽר׃ (כא) עָנֵ֤ה דָנִיֵּאל֙ וְאָמַ֔ר חָזֵ֥ה הֲוֵ֛ית בְּחֶזְוִ֖י עִם־ לֵֽילְיָ֑א וַאֲר֗וּ אַרְבַּע֙ רוּחֵ֣י שְׁמַיָּ֔א מְגִיחָ֖ן לְיַמָּ֥א רַבָּֽא׃ (כב) וְאַרְבַּ֤ע חֵיוָן֙ רַבְרְבָ֔ן סָלְקָ֖ן מִן־ יַמָּ֑א שָׁנְיָ֖ן דָּ֥א מִן־ דָּֽא׃ (כג) קַדְמָיְתָ֣א כְאַרְיֵ֔ה וְגַפִּ֥ין דִּֽי־ נְשַׁ֖ר לַ֑הּ חָזֵ֣ה הֲוֵ֡ית עַד֩ דִּי־ מְּרִ֨יטוּ גַפַּ֜יהּ וּנְטִ֣ילַת מִן־ אַרְעָ֗א וְעַל־ רַגְלַ֙יִן֙ כֶּאֱנָ֣שׁ הֳקִימַ֔ת וּלְבַ֥ב אֱנָ֖שׁ יְהִ֥יב לַֽהּ׃ (כד) וַאֲר֣וּ חֵיוָה֩ אָחֳרִ֨י תִנְיָנָ֜ה דָּמְיָ֣ה לְדֹ֗ב וְלִשְׂטַר־ חַד֙ הֳקִמַ֔ת וּתְלָ֥ת עִלְעִ֛ין בְּפֻמַּ֖הּ בֵּ֣ין (שניה) שִׁנַּ֑הּ וְכֵן֙ אָמְרִ֣ין לַ֔הּ ק֥וּמִֽי אֲכֻ֖לִי בְּשַׂ֥ר שַׂגִּֽיא׃ (כה) בָּאתַ֨ר דְּנָ֜ה חָזֵ֣ה הֲוֵ֗ית וַאֲר֤וּ אָֽחֳרִי֙ כִּנְמַ֔ר וְלַ֨הּ גַּפִּ֥ין אַרְבַּ֛ע דִּי־ ע֖וֹף עַל־ (גביה) גַּבַּ֑הּ וְאַרְבְּעָ֤ה רֵאשִׁין֙ לְחֵ֣יוְתָ֔א וְשָׁלְטָ֖ן יְהִ֥יב לַֽהּ׃ (כו) בָּאתַ֣ר דְּנָה֩ חָזֵ֨ה הֲוֵ֜ית בְּחֶזְוֵ֣י לֵֽילְיָ֗א וַאֲר֣וּ חֵיוָ֣ה (רביעיה) רְֽבִיעָאָ֡ה דְּחִילָה֩ וְאֵֽימְתָנִ֨י וְתַקִּיפָ֜א יַתִּ֗ירָא וְשִׁנַּ֨יִן דִּֽי־ פַרְזֶ֥ל לַהּ֙ רַבְרְבָ֔ן אָֽכְלָ֣ה וּמַדֱּקָ֔ה וּשְׁאָרָ֖א (ברגליה) בְּרַגְלַ֣הּ רָפְסָ֑ה וְהִ֣יא מְשַׁנְּיָ֗ה מִן־ כָּל־ חֵֽיוָתָא֙ דִּ֣י קָֽדָמַ֔יהּ וְקַרְנַ֥יִן עֲשַׂ֖ר לַֽהּ׃ (כז) מִשְׂתַּכַּ֨ל הֲוֵ֜ית בְּקַרְנַיָּ֗א וַ֠אֲלוּ קֶ֣רֶן אָחֳרִ֤י זְעֵירָה֙ סִלְקָ֣ת (ביניהון) בֵּֽינֵיהֵ֔ן וּתְלָ֗ת מִן־ קַרְנַיָּא֙ קַדְמָ֣יָתָ֔א (אתעקרו) אֶתְעֲקַ֖רָה מִן־ (קדמיה) קֳדָמַ֑הּ וַאֲל֨וּ עַיְנִ֜ין כְּעַיְנֵ֤י אֲנָשָׁא֙ בְּקַרְנָא־ דָ֔א וּפֻ֖ם מְמַלִּ֥ל רַבְרְבָֽן׃ (כח) חָזֵ֣ה הֲוֵ֗ית עַ֣ד דִּ֤י כָרְסָוָן֙ רְמִ֔יו וְעַתִּ֥יק יוֹמִ֖ין יְתִ֑ב לְבוּשֵׁ֣הּ ׀ כִּתְלַ֣ג חִוָּ֗ר וַתַּפֵּ֥ל אַ֛רְצָה מִן־ הַצָּבָ֥א וּמִן־ הַכּוֹכָבִ֖ים וַֽתִּרְמְסֵֽם׃ (נח) וְעַ֥ד שַֽׂר־ הַצָּבָ֖א הִגְדִּ֑יל וּמִמֶּ֙נּוּ֙ (הרים) הוּרַ֣ם הַתָּמִ֔יד וְהֻשְׁלַ֖ךְ מְכ֥וֹן מִקְדָּשֽׁוֹ׃ (נט) וְצָבָ֛א תִּנָּתֵ֥ן עַל־ הַתָּמִ֖יד בְּפָ֑שַׁע וְתַשְׁלֵ֤ךְ אֱמֶת֙ אַ֔רְצָה וְעָשְׂתָ֖ה וְהִצְלִֽיחָה׃ (ס) וָאֶשְׁמְעָ֥ה אֶֽחָד־ קָד֖וֹשׁ מְדַבֵּ֑ר וַיֹּאמֶר֩ אֶחָ֨ד קָד֜וֹשׁ לַפַּֽלְמוֹנִ֣י הַֽמְדַבֵּ֗ר עַד־ מָתַ֞י הֶחָז֤וֹן הַתָּמִיד֙ וְהַפֶּ֣שַׁע שֹׁמֵ֔ם תֵּ֛ת וְקֹ֥דֶשׁ וְצָבָ֖א מִרְמָֽס׃ (סא) וַיֹּ֣אמֶר אֵלַ֔י עַ֚ד עֶ֣רֶב בֹּ֔קֶר אַלְפַּ֖יִם וּשְׁלֹ֣שׁ מֵא֑וֹת וְנִצְדַּ֖ק קֹֽדֶשׁ׃ (סב) וַיְהִ֗י בִּרְאֹתִ֛י אֲנִ֥י דָנִיֵּ֖אל אֶת־ הֶחָז֑וֹן וָאֲבַקְשָׁ֣ה בִינָ֔ה וְהִנֵּ֛ה עֹמֵ֥ד לְנֶגְדִּ֖י כְּמַרְאֵה־ גָֽבֶר׃ (סג) וָאֶשְׁמַ֥ע קוֹל־ אָדָ֖ם בֵּ֣ין אוּלָ֑י וַיִּקְרָא֙ וַיֹּאמַ֔ר גַּבְרִיאֵ֕ל הָבֵ֥ן לְהַלָּ֖ז אֶת־ הַמַּרְאֶֽה׃ (סד) וַיָּבֹא֙ אֵ֣צֶל עָמְדִ֔י וּבְבֹא֣וֹ נִבְעַ֔תִּי וָאֶפְּלָ֖ה עַל־ פָּנָ֑י וַיֹּ֤אמֶר אֵלַי֙ הָבֵ֣ן בֶּן־ אָדָ֔ם כִּ֖י לְעֶת־ קֵ֥ץ הֶחָזֽוֹן׃ (סה) וּבְדַבְּר֣וֹ עִמִּ֔י נִרְדַּ֥מְתִּי עַל־ פָּנַ֖י אָ֑רְצָה וַיִּ֨גַּע־ בִּ֔י וַיַּֽעֲמִידֵ֖נִי עַל־ עָמְדִֽי׃ (סו) וַיֹּ֙אמֶר֙ הִנְנִ֣י מוֹדִֽיעֲךָ֔ אֵ֥ת אֲשֶׁר־ יִהְיֶ֖ה בְּאַחֲרִ֣ית הַזָּ֑עַם כִּ֖י לְמוֹעֵ֥ד קֵֽץ׃ (סז) הָאַ֥יִל אֲשֶׁר־ רָאִ֖יתָ בַּ֣עַל הַקְּרָנָ֑יִם מַלְכֵ֖י מָדַ֥י וּפָרָֽס׃ (סח) וְהַצָּפִ֥יר הַשָּׂעִ֖יר מֶ֣לֶךְ יָוָ֑ן וְהַקֶּ֤רֶן הַגְּדוֹלָה֙ אֲשֶׁ֣ר בֵּין־ עֵינָ֔יו ה֖וּא הַמֶּ֥לֶךְ הָרִאשֽׁוֹן׃ (סט) וְהַ֨נִּשְׁבֶּ֔רֶת וַתַּֽעֲמֹ֥דְנָה אַרְבַּ֖ע תַּחְתֶּ֑יהָ אַרְבַּ֧ע מַלְכֻי֛וֹת מִגּ֥וֹי יַעֲמֹ֖דְנָה וְלֹ֥א בְכֹחֽוֹ׃ (ע) וּֽבְאַחֲרִית֙ מַלְכוּתָ֔ם כְּהָתֵ֖ם הַפֹּשְׁעִ֑ים יַעֲמֹ֛ד מֶ֥לֶךְ עַז־ פָּנִ֖ים וּמֵבִ֥ין חִידֽוֹת׃ (עא) וְעָצַ֤ם כֹּחוֹ֙ וְלֹ֣א בְכֹח֔וֹ וְנִפְלָא֥וֹת יַשְׁחִ֖ית וְהִצְלִ֣יחַ וְעָשָׂ֑ה וְהִשְׁחִ֥ית עֲצוּמִ֖ים וְעַם־ קְדֹשִֽׁים׃ (עב) וְעַל־ שִׂכְל֗וֹ וְהִצְלִ֤יחַ מִרְמָה֙ בְּיָד֔וֹ וּבִלְבָב֣וֹ יַגְדִּ֔יל וּבְשַׁלְוָ֖ה יַשְׁחִ֣ית רַבִּ֑ים וְעַ֤ל־ שַׂר־ שָׂרִים֙ יַעֲמֹ֔ד וּבְאֶ֥פֶס יָ֖ד יִשָּׁבֵֽר׃ (עג) וּמַרְאֵ֨ה הָעֶ֧רֶב וְהַבֹּ֛קֶר אֲשֶׁ֥ר נֶאֱמַ֖ר אֱמֶ֣ת ה֑וּא וְאַתָּה֙ סְתֹ֣ם הֶֽחָז֔וֹן כִּ֖י לְיָמִ֥ים רַבִּֽים׃ (עד) וַאֲנִ֣י דָנִיֵּ֗אל נִהְיֵ֤יתִי וְנֶֽחֱלֵ֙יתִי֙ יָמִ֔ים וָאָק֕וּם וָאֶֽעֱשֶׂ֖ה אֶת־ מְלֶ֣אכֶת הַמֶּ֑לֶךְ וָאֶשְׁתּוֹמֵ֥ם עַל־ הַמַּרְאֶ֖ה וְאֵ֥ין מֵבִֽין׃ סוף פרק ח (עה) בִּשְׁנַ֣ת אַחַ֗ת לְדָרְיָ֛וֶשׁ בֶּן־ אֲחַשְׁוֵר֖וֹשׁ מִזֶּ֣רַע מָדָ֑י אֲשֶׁ֣ר הָמְלַ֔ךְ עַ֖ל מַלְכ֥וּת כַּשְׂדִּֽים׃ (עו) בִּשְׁנַ֤ת אַחַת֙ לְמָלְכ֔וֹ אֲנִי֙ דָּֽנִיֵּ֔אל בִּינֹ֖תִי בַּסְּפָרִ֑ים מִסְפַּ֣ר הַשָּׁנִ֗ים אֲשֶׁ֨ר הָיָ֤ה דְבַר־ יְהוָה֙ אֶל־ יִרְמִיָ֣ה הַנָּבִ֔יא לְמַלֹּ֛אות לְחָרְב֥וֹת יְרוּשָׁלִַ֖ם שִׁבְעִ֥ים שָׁנָֽה׃ (עז) וָאֶתְּנָ֣ה אֶת־ פָּנַ֗י אֶל־ אֲדֹנָי֙ הָֽאֱלֹהִ֔ים לְבַקֵּ֥שׁ תְּפִלָּ֖ה וְתַחֲנוּנִ֑ים בְּצ֖וֹם וְשַׂ֥ק וָאֵֽפֶר׃
