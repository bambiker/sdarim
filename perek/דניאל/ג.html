פרק ג בספר דניאל (33 פסוקים) נמצא בדניאל סדר ב (פסוקים טז–מד) ובדניאל סדר ג (פסוקים א–ד).

דניאל סדר ב

(א) בֵּאדַ֣יִן דָּ֣קוּ כַחֲדָ֡ה פַּרְזְלָא֩ חַסְפָּ֨א נְחָשָׁ֜א כַּסְפָּ֣א וְדַהֲבָ֗א וַהֲווֹ֙ כְּע֣וּר מִן־ אִדְּרֵי־ קַ֔יִט וּנְשָׂ֤א הִמּוֹן֙ רוּחָ֔א וְכָל־ אֲתַ֖ר לָא־ הִשְׁתֲּכַ֣ח לְה֑וֹן וְאַבְנָ֣א ׀ דִּֽי־ מְחָ֣ת לְצַלְמָ֗א הֲוָ֛ת לְט֥וּר רַ֖ב וּמְלָ֥ת כָּל־ אַרְעָֽא׃ (ב) דְּנָ֣ה חֶלְמָ֔א וּפִשְׁרֵ֖הּ נֵאמַ֥ר קֳדָם־ מַלְכָּֽא׃ (ג) אַ֣נְתְּה מַלְכָּ֔א מֶ֖לֶךְ מַלְכַיָּ֑א דִּ֚י אֱלָ֣הּ שְׁמַיָּ֔א מַלְכוּתָ֥א חִסְנָ֛א וְתָקְפָּ֥א וִֽיקָרָ֖א יְהַב־ לָֽךְ׃ (ד) וּבְכָל־ דִּ֣י (דארין) דָֽיְרִ֣ין בְּֽנֵי־ אֲ֠נָשָׁא חֵיוַ֨ת בָּרָ֤א וְעוֹף־ שְׁמַיָּא֙ יְהַ֣ב בִּידָ֔ךְ וְהַשְׁלְטָ֖ךְ בְּכָלְּה֑וֹן אַנְתְּה־ ה֔וּא רֵאשָׁ֖ה דִּ֥י דַהֲבָֽא׃ (ה) וּבָתְרָ֗ךְ תְּק֛וּם מַלְכ֥וּ אָחֳרִ֖י אֲרַ֣עא מִנָּ֑ךְ וּמַלְכ֨וּ (תליתיא) תְלִיתָאָ֤ה אָחֳרִי֙ דִּ֣י נְחָשָׁ֔א דִּ֥י תִשְׁלַ֖ט בְּכָל־ אַרְעָֽא׃ (ו) וּמַלְכוּ֙ (רביעיה) רְבִ֣יעָאָ֔ה תֶּהֱוֵ֥א תַקִּיפָ֖ה כְּפַרְזְלָ֑א כָּל־ קֳבֵ֗ל דִּ֤י פַרְזְלָא֙ מְהַדֵּ֤ק וְחָשֵׁל֙ כֹּ֔לָּא וּֽכְפַרְזְלָ֛א דִּֽי־ מְרָעַ֥ע כָּל־ אִלֵּ֖ין תַּדִּ֥ק וְתֵרֹֽעַ׃ (ז) וְדִֽי־ חֲזַ֜יְתָה רַגְלַיָּ֣א וְאֶצְבְּעָתָ֗א (מנהון) מִנְּהֵ֞ן חֲסַ֤ף דִּֽי־ פֶחָר֙ (ומנהון) וּמִנְּהֵ֣ין פַּרְזֶ֔ל מַלְכ֤וּ פְלִיגָה֙ תֶּהֱוֵ֔ה וּמִן־ נִצְבְּתָ֥א דִ֥י פַרְזְלָ֖א לֶֽהֱוֵא־ בַ֑הּ כָּל־ קֳבֵל֙ דִּ֣י חֲזַ֔יְתָה פַּ֨רְזְלָ֔א מְעָרַ֖ב בַּחֲסַ֥ף טִינָֽא׃ (ח) וְאֶצְבְּעָת֙ רַגְלַיָּ֔א (מנהון) מִנְּהֵ֥ין פַּרְזֶ֖ל (ומנהון) וּמִנְּהֵ֣ין חֲסַ֑ף מִן־ קְצָ֤ת מַלְכוּתָא֙ תֶּהֱוֵ֣ה תַקִּיפָ֔ה וּמִנַּ֖הּ תֶּהֱוֵ֥ה תְבִירָֽה׃ (ט) (די) וְדִ֣י חֲזַ֗יְתָ פַּרְזְלָא֙ מְעָרַב֙ בַּחֲסַ֣ף טִינָ֔א מִתְעָרְבִ֤ין לֶהֱוֺן֙ בִּזְרַ֣ע אֲנָשָׁ֔א וְלָֽא־ לֶהֱוֺ֥ן דָּבְקִ֖ין דְּנָ֣ה עִם־ דְּנָ֑ה הֵֽא־ כְדִ֣י פַרְזְלָ֔א לָ֥א מִתְעָרַ֖ב עִם־ חַסְפָּֽא׃ (י) וּֽבְיוֹמֵיה֞וֹן דִּ֧י מַלְכַיָּ֣א אִנּ֗וּן יְקִים֩ אֱלָ֨הּ שְׁמַיָּ֤א מַלְכוּ֙ דִּ֤י לְעָלְמִין֙ לָ֣א תִתְחַבַּ֔ל וּמַ֨לְכוּתָ֔ה לְעַ֥ם אָחֳרָ֖ן לָ֣א תִשְׁתְּבִ֑ק תַּדִּ֤ק וְתָסֵיף֙ כָּל־ אִלֵּ֣ין מַלְכְוָתָ֔א וְהִ֖יא תְּק֥וּם לְעָלְמַיָּֽא׃ (יא) כָּל־ קֳבֵ֣ל דִּֽי־ חֲזַ֡יְתָ דִּ֣י מִטּוּרָא֩ אִתְגְּזֶ֨רֶת אֶ֜בֶן דִּי־ לָ֣א בִידַ֗יִן וְ֠הַדֶּקֶת פַּרְזְלָ֨א נְחָשָׁ֤א חַסְפָּא֙ כַּסְפָּ֣א וְדַהֲבָ֔א אֱלָ֥הּ רַב֙ הוֹדַ֣ע לְמַלְכָּ֔א מָ֛ה דִּ֥י לֶהֱוֵ֖א אַחֲרֵ֣י דְנָ֑ה וְיַצִּ֥יב חֶלְמָ֖א וּמְהֵימַ֥ן פִּשְׁרֵֽהּ׃ (יב) בֵּ֠אדַיִן מַלְכָּ֤א נְבֽוּכַדְנֶצַּר֙ נְפַ֣ל עַל־ אַנְפּ֔וֹהִי וּלְדָנִיֵּ֖אל סְגִ֑ד וּמִנְחָה֙ וְנִ֣יחֹחִ֔ין אֲמַ֖ר לְנַסָּ֥כָה לֵֽהּ׃ (יג) עָנֵה֩ מַלְכָּ֨א לְדָנִיֵּ֜אל וְאָמַ֗ר מִן־ קְשֹׁט֙ דִּ֣י אֱלָהֲכ֗וֹן ה֣וּא אֱלָ֧הּ אֱלָהִ֛ין וּמָרֵ֥א מַלְכִ֖ין וְגָלֵ֣ה רָזִ֑ין דִּ֣י יְכֵ֔לְתָּ לְמִגְלֵ֖א רָזָ֥ה דְנָֽה׃ (יד) אֱדַ֨יִן מַלְכָּ֜א לְדָנִיֵּ֣אל רַבִּ֗י וּמַתְּנָ֨ן רַבְרְבָ֤ן שַׂגִּיאָן֙ יְהַב־ לֵ֔הּ וְהַ֨שְׁלְטֵ֔הּ עַ֖ל כָּל־ מְדִינַ֣ת בָּבֶ֑ל וְרַב־ סִגְנִ֔ין עַ֖ל כָּל־ חַכִּימֵ֥י בָבֶֽל׃ (טו) וְדָנִיֵּאל֙ בְּעָ֣א מִן־ מַלְכָּ֔א וּמַנִּ֗י עַ֤ל עֲבִֽידְתָּא֙ דִּ֚י מְדִינַ֣ת בָּבֶ֔ל לְשַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ וְדָנִיֵּ֖אל בִּתְרַ֥ע מַלְכָּֽא׃ תחילת פרק ג (טז) נְבוּכַדְנֶצַּ֣ר מַלְכָּ֗א עֲבַד֙ צְלֵ֣ם דִּֽי־ דְהַ֔ב רוּמֵהּ֙ אַמִּ֣ין שִׁתִּ֔ין פְּתָיֵ֖הּ אַמִּ֣ין שִׁ֑ת אֲקִימֵהּ֙ בְּבִקְעַ֣ת דּוּרָ֔א בִּמְדִינַ֖ת בָּבֶֽל׃ (יז) וּנְבוּכַדְנֶצַּ֣ר מַלְכָּ֡א שְׁלַ֡ח לְמִכְנַ֣שׁ ׀ לַֽאֲחַשְׁדַּרְפְּנַיָּ֡א סִגְנַיָּ֣א וּֽפַחֲוָתָ֡א אֲדַרְגָּזְרַיָּא֩ גְדָ֨בְרַיָּ֤א דְּתָבְרַיָּא֙ תִּפְתָּיֵ֔א וְכֹ֖ל שִׁלְטֹנֵ֣י מְדִֽינָתָ֑א לְמֵתֵא֙ לַחֲנֻכַּ֣ת צַלְמָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֥ר מַלְכָּֽא׃ (יח) בֵּאדַ֡יִן מִֽתְכַּנְּשִׁ֡ין אֲחַשְׁדַּרְפְּנַיָּ֡א סִגְנַיָּ֣א וּֽפַחֲוָתָ֡א אֲדַרְגָּזְרַיָּ֣א גְדָבְרַיָּא֩ דְּתָ֨בְרַיָּ֜א תִּפְתָּיֵ֗א וְכֹל֙ שִׁלְטֹנֵ֣י מְדִֽינָתָ֔א לַחֲנֻכַּ֣ת צַלְמָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֣ר מַלְכָּ֑א (וקאמין) וְקָֽיְמִין֙ לָקֳבֵ֣ל צַלְמָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּֽר׃ (יט) וְכָרוֹזָ֖א קָרֵ֣א בְחָ֑יִל לְכ֤וֹן אָֽמְרִין֙ עַֽמְמַיָּ֔א אֻמַּיָּ֖א וְלִשָּׁנַיָּֽא׃ (כ) בְּעִדָּנָ֡א דִּֽי־ תִשְׁמְע֡וּן קָ֣ל קַרְנָ֣א מַ֠שְׁרוֹקִיתָא (קיתרוס) קַתְר֨וֹס סַבְּכָ֤א פְּסַנְתֵּרִין֙ סוּמְפֹּ֣נְיָ֔ה וְכֹ֖ל זְנֵ֣י זְמָרָ֑א תִּפְּל֤וּן וְתִסְגְּדוּן֙ לְצֶ֣לֶם דַּהֲבָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֥ר מַלְכָּֽא׃ (כא) וּמַן־ דִּי־ לָ֥א יִפֵּ֖ל וְיִסְגֻּ֑ד בַּהּ־ שַׁעֲתָ֣א יִתְרְמֵ֔א לְגֽוֹא־ אַתּ֥וּן נוּרָ֖א יָקִֽדְתָּֽא׃ (כב) כָּל־ קֳבֵ֣ל דְּנָ֡ה בֵּהּ־ זִמְנָ֡א כְּדִ֣י שָֽׁמְעִ֣ין כָּֽל־ עַמְמַיָּ֡א קָ֣ל קַרְנָא֩ מַשְׁר֨וֹקִיתָ֜א (קיתרס) קַתְר֤וֹס שַׂבְּכָא֙ פְּסַנְטֵרִ֔ין וְכֹ֖ל זְנֵ֣י זְמָרָ֑א נָֽפְלִ֨ין כָּֽל־ עַֽמְמַיָּ֜א אֻמַיָּ֣א וְלִשָּׁנַיָּ֗א סָֽגְדִין֙ לְצֶ֣לֶם דַּהֲבָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֥ר מַלְכָּֽא׃ (כג) כָּל־ קֳבֵ֤ל דְּנָה֙ בֵּהּ־ זִמְנָ֔א קְרִ֖בוּ גֻּבְרִ֣ין כַּשְׂדָּאִ֑ין וַאֲכַ֥לוּ קַרְצֵיה֖וֹן דִּ֥י יְהוּדָיֵֽא׃ (כד) עֲנוֹ֙ וְאָ֣מְרִ֔ין לִנְבוּכַדְנֶצַּ֖ר מַלְכָּ֑א מַלְכָּ֖א לְעָלְמִ֥ין חֱיִֽי׃ (כה) (אנתה) אַ֣נְתְּ מַלְכָּא֮ שָׂ֣מְתָּ טְּעֵם֒ דִּ֣י כָל־ אֱנָ֡שׁ דִּֽי־ יִשְׁמַ֡ע קָ֣ל קַרְנָ֣א מַ֠שְׁרֹקִיתָא (קיתרס) קַתְר֨וֹס שַׂבְּכָ֤א פְסַנְתֵּרִין֙ (וסיפניה) וְסוּפֹּ֣נְיָ֔ה וְכֹ֖ל זְנֵ֣י זְמָרָ֑א יִפֵּ֥ל וְיִסְגֻּ֖ד לְצֶ֥לֶם דַּהֲבָֽא׃ (כו) וּמַן־ דִּי־ לָ֥א יִפֵּ֖ל וְיִסְגֻּ֑ד יִתְרְמֵ֕א לְגֽוֹא־ אַתּ֥וּן נוּרָ֖א יָקִֽדְתָּֽא׃ (כז) אִיתַ֞י גֻּבְרִ֣ין יְהוּדָאיִ֗ן דִּֽי־ מַנִּ֤יתָ יָתְהוֹן֙ עַל־ עֲבִידַת֙ מְדִינַ֣ת בָּבֶ֔ל שַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ גֻּבְרַיָּ֣א אִלֵּ֗ךְ לָא־ שָׂ֨מֽוּ (עליך) עֲלָ֤ךְ מַלְכָּא֙ טְעֵ֔ם (לאלהיך) לֵֽאלָהָךְ֙ לָ֣א פָלְחִ֔ין וּלְצֶ֧לֶם מַלְכָּ֗א עֲבַד֙ לְחֶ֣ם רַ֔ב לְרַבְרְבָנ֖וֹהִי אֲלַ֑ף וְלָקֳבֵ֥ל אַלְפָּ֖א חַמְרָ֥א שָׁתֵֽה׃ (מ) בֵּלְשַׁאצַּ֞ר אֲמַ֣ר ׀ בִּטְעֵ֣ם חַמְרָ֗א לְהַיְתָיָה֙ לְמָאנֵי֙ דַּהֲבָ֣א וְכַסְפָּ֔א דִּ֤י הַנְפֵּק֙ נְבוּכַדְנֶצַּ֣ר אֲב֔וּהִי מִן־ הֵיכְלָ֖א דִּ֣י בִירוּשְׁלֶ֑ם וְיִשְׁתּ֣וֹן בְּה֗וֹן מַלְכָּא֙ וְרַבְרְבָנ֔וֹהִי שֵׁגְלָתֵ֖הּ וּלְחֵנָתֵֽהּ׃ (מא) בֵּאדַ֗יִן הַיְתִיו֙ מָאנֵ֣י דַהֲבָ֔א דִּ֣י הַנְפִּ֗קוּ מִן־ הֵֽיכְלָ֛א דִּֽי־ בֵ֥ית אֱלָהָ֖א דִּ֣י בִירֽוּשְׁלֶ֑ם וְאִשְׁתִּ֣יו בְּה֗וֹן מַלְכָּא֙ וְרַבְרְבָנ֔וֹהִי שֵׁגְלָתֵ֖הּ וּלְחֵנָתֵֽהּ׃ (מב) אִשְׁתִּ֖יו חַמְרָ֑א וְ֠שַׁבַּחוּ לֵֽאלָהֵ֞י דַּהֲבָ֧א וְכַסְפָּ֛א נְחָשָׁ֥א פַרְזְלָ֖א אָעָ֥א וְאַבְנָֽא׃ (מג) בַּהּ־ שַׁעֲתָ֗ה (נפקו) נְפַ֙קָה֙ אֶצְבְּעָן֙ דִּ֣י יַד־ אֱנָ֔שׁ וְכָֽתְבָן֙ לָקֳבֵ֣ל נֶבְרַשְׁתָּ֔א עַל־ גִּירָ֕א דִּֽי־ כְתַ֥ל הֵיכְלָ֖א דִּ֣י מַלְכָּ֑א וּמַלְכָּ֣א חָזֵ֔ה פַּ֥ס יְדָ֖ה דִּ֥י כָתְבָֽה׃ (מד) אֱדַ֤יִן מַלְכָּא֙ זִיוֺ֣הִי שְׁנ֔וֹהִי וְרַעיֹנֹ֖הִי יְבַהֲלוּנֵּ֑הּ וְקִטְרֵ֤י חַרְצֵהּ֙ מִשְׁתָּרַ֔יִן וְאַ֨רְכֻבָּתֵ֔הּ דָּ֥א לְדָ֖א נָֽקְשָֽׁן׃ (מה) קָרֵ֤א מַלְכָּא֙ בְּחַ֔יִל לְהֶֽעָלָה֙ לְאָ֣שְׁפַיָּ֔א (כשדיא) כַּשְׂדָּאֵ֖י וְגָזְרַיָּ֑א עָנֵ֨ה מַלְכָּ֜א וְאָמַ֣ר ׀ לְחַכִּימֵ֣י בָבֶ֗ל דִּ֣י כָל־ אֱ֠נָשׁ דִּֽי־ יִקְרֵ֞ה כְּתָבָ֣ה דְנָ֗ה וּפִשְׁרֵהּ֙ יְחַוִּנַּ֔נִי אַרְגְּוָנָ֣א יִלְבַּ֗שׁ (והמונכא) וְהַֽמְנִיכָ֤א דִֽי־ דַהֲבָא֙ עַֽל־ צַוְּארֵ֔הּ וְתַלְתִּ֥י בְמַלְכוּתָ֖א יִשְׁלַֽט׃ (מו) אֱדַ֙יִן֙ (עללין) עָֽלִּ֔ין כֹּ֖ל חַכִּימֵ֣י מַלְכָּ֑א וְלָֽא־ כָהֲלִ֤ין כְּתָבָא֙ לְמִקְרֵ֔א (ופשרא) וּפִשְׁרֵ֖הּ לְהוֹדָעָ֥ה לְמַלְכָּֽא׃ (מז) אֱ֠דַיִן מַלְכָּ֤א בֵלְשַׁאצַּר֙ שַׂגִּ֣יא מִתְבָּהַ֔ל וְזִיוֺ֖הִי שָׁנַ֣יִן עֲל֑וֹהִי וְרַבְרְבָנ֖וֹהִי מִֽשְׁתַּבְּשִֽׁין׃ (מח) מַלְכְּתָ֕א לָקֳבֵ֨ל מִלֵּ֤י מַלְכָּא֙ וְרַבְרְבָנ֔וֹהִי לְבֵ֥ית מִשְׁתְּיָ֖א (עללת) עַלַּ֑ת עֲנָ֨ת מַלְכְּתָ֜א וַאֲמֶ֗רֶת מַלְכָּא֙ לְעָלְמִ֣ין חֱיִ֔י אַֽל־ יְבַהֲלוּךְ֙ רַעְיוֹנָ֔ךְ וְזִיוָ֖יךְ אַל־ יִשְׁתַּנּֽוֹ׃ (מט) אִיתַ֨י גְּבַ֜ר בְּמַלְכוּתָ֗ךְ דִּ֠י ר֣וּחַ אֱלָהִ֣ין קַדִּישִׁין֮ בֵּהּ֒ וּבְיוֹמֵ֣י אֲב֗וּךְ נַהִיר֧וּ וְשָׂכְלְתָנ֛וּ וְחָכְמָ֥ה כְּחָכְמַת־ אֱלָהִ֖ין הִשְׁתְּכַ֣חַת בֵּ֑הּ וּמַלְכָּ֤א נְבֻֽכַדְנֶצַּר֙ אֲב֔וּךְ רַ֧ב חַרְטֻמִּ֣ין אָֽשְׁפִ֗ין כַּשְׂדָּאִין֙ גָּזְרִ֔ין הֲקִימֵ֖הּ אֲב֥וּךְ מַלְכָּֽא׃
