דניאל פרק י לפי החלוקה לסדרים

פרק י בספר דניאל (21 פסוקים) נמצא בדניאל סדר ו (פסוקים כה–מד) ובדניאל סדר ז (פסוקים א–א).

דניאל סדר ו

(א) וָֽאֶתְפַּֽלְלָ֛ה לַיהוָ֥ה אֱלֹהַ֖י וָאֶתְוַדֶּ֑ה וָאֹֽמְרָ֗ה אָנָּ֤א אֲדֹנָי֙ הָאֵ֤ל הַגָּדוֹל֙ וְהַנּוֹרָ֔א שֹׁמֵ֤ר הַבְּרִית֙ וְֽהַחֶ֔סֶד לְאֹהֲבָ֖יו וּלְשֹׁמְרֵ֥י מִצְוֺתָֽיו׃ (ב) חָטָ֥אנוּ וְעָוִ֖ינוּ (והרשענו) הִרְשַׁ֣עְנוּ וּמָרָ֑דְנוּ וְס֥וֹר מִמִּצְוֺתֶ֖ךָ וּמִמִּשְׁפָּטֶֽיךָ׃ (ג) וְלֹ֤א שָׁמַ֙עְנוּ֙ אֶל־ עֲבָדֶ֣יךָ הַנְּבִיאִ֔ים אֲשֶׁ֤ר דִּבְּרוּ֙ בְּשִׁמְךָ֔ אֶל־ מְלָכֵ֥ינוּ שָׂרֵ֖ינוּ וַאֲבֹתֵ֑ינוּ וְאֶ֖ל כָּל־ עַ֥ם הָאָֽרֶץ׃ (ד) לְךָ֤ אֲדֹנָי֙ הַצְּדָקָ֔ה וְלָ֛נוּ בֹּ֥שֶׁת הַפָּנִ֖ים כַּיּ֣וֹם הַזֶּ֑ה לְאִ֤ישׁ יְהוּדָה֙ וּלְיוֹשְׁבֵ֣י יְרֽוּשָׁלִַ֔ם וּֽלְכָל־ יִשְׂרָאֵ֞ל הַקְּרֹבִ֣ים וְהָרְחֹקִ֗ים בְּכָל־ הָֽאֲרָצוֹת֙ אֲשֶׁ֣ר הִדַּחְתָּ֣ם שָׁ֔ם בְּמַעֲלָ֖ם אֲשֶׁ֥ר מָֽעֲלוּ־ בָֽךְ׃ (ה) יְהוָ֗ה לָ֚נוּ בֹּ֣שֶׁת הַפָּנִ֔ים לִמְלָכֵ֥ינוּ לְשָׂרֵ֖ינוּ וְלַאֲבֹתֵ֑ינוּ אֲשֶׁ֥ר חָטָ֖אנוּ לָֽךְ׃ (ו) לַֽאדֹנָ֣י אֱלֹהֵ֔ינוּ הָרַחֲמִ֖ים וְהַסְּלִח֑וֹת כִּ֥י מָרַ֖דְנוּ בּֽוֹ׃ (ז) וְלֹ֣א שָׁמַ֔עְנוּ בְּק֖וֹל יְהוָ֣ה אֱלֹהֵ֑ינוּ לָלֶ֤כֶת בְּתֽוֹרֹתָיו֙ אֲשֶׁ֣ר נָתַ֣ן לְפָנֵ֔ינוּ בְּיַ֖ד עֲבָדָ֥יו הַנְּבִיאִֽים׃ (ח) וְכָל־ יִשְׂרָאֵ֗ל עָֽבְרוּ֙ אֶת־ תּ֣וֹרָתֶ֔ךָ וְס֕וֹר לְבִלְתִּ֖י שְׁמ֣וֹעַ בְּקֹלֶ֑ךָ וַתִּתַּ֨ךְ עָלֵ֜ינוּ הָאָלָ֣ה וְהַשְּׁבֻעָ֗ה אֲשֶׁ֤ר כְּתוּבָה֙ בְּתוֹרַת֙ מֹשֶׁ֣ה עֶֽבֶד־ הָֽאֱלֹהִ֔ים כִּ֥י חָטָ֖אנוּ לֽוֹ׃ (ט) וַיָּ֜קֶם אֶת־ (דבריו) דְּבָר֣וֹ ׀ אֲשֶׁר־ דִּבֶּ֣ר עָלֵ֗ינוּ וְעַ֤ל שֹֽׁפְטֵ֙ינוּ֙ אֲשֶׁ֣ר שְׁפָט֔וּנוּ לְהָבִ֥יא עָלֵ֖ינוּ רָעָ֣ה גְדֹלָ֑ה אֲשֶׁ֣ר לֹֽא־ נֶעֶשְׂתָ֗ה תַּ֚חַת כָּל־ הַשָּׁמַ֔יִם כַּאֲשֶׁ֥ר נֶעֶשְׂתָ֖ה בִּירוּשָׁלִָֽם׃ (י) כַּאֲשֶׁ֤ר כָּתוּב֙ בְּתוֹרַ֣ת מֹשֶׁ֔ה אֵ֛ת כָּל־ הָרָעָ֥ה הַזֹּ֖את בָּ֣אָה עָלֵ֑ינוּ וְלֹֽא־ חִלִּ֜ינוּ אֶת־ פְּנֵ֣י ׀ יְהוָ֣ה אֱלֹהֵ֗ינוּ לָשׁוּב֙ מֵֽעֲוֺנֵ֔נוּ וּלְהַשְׂכִּ֖יל בַּאֲמִתֶּֽךָ׃ (יא) וַיִּשְׁקֹ֤ד יְהוָה֙ עַל־ הָ֣רָעָ֔ה וַיְבִיאֶ֖הָ עָלֵ֑ינוּ כִּֽי־ צַדִּ֞יק יְהוָ֣ה אֱלֹהֵ֗ינוּ עַל־ כָּל־ מַֽעֲשָׂיו֙ אֲשֶׁ֣ר עָשָׂ֔ה וְלֹ֥א שָׁמַ֖עְנוּ בְּקֹלֽוֹ׃ (יב) וְעַתָּ֣ה ׀ אֲדֹנָ֣י אֱלֹהֵ֗ינוּ אֲשֶׁר֩ הוֹצֵ֨אתָ אֶֽת־ עַמְּךָ֜ מֵאֶ֤רֶץ מִצְרַ֙יִם֙ בְּיָ֣ד חֲזָקָ֔ה וַתַּֽעַשׂ־ לְךָ֥ שֵׁ֖ם כַּיּ֣וֹם הַזֶּ֑ה חָטָ֖אנוּ רָשָֽׁעְנוּ׃ (יג) אֲדֹנָ֗י כְּכָל־ צִדְקֹתֶ֙ךָ֙ יָֽשָׁב־ נָ֤א אַפְּךָ֙ וַחֲמָ֣תְךָ֔ מֵעִֽירְךָ֥ יְרוּשָׁלִַ֖ם הַר־ קָדְשֶׁ֑ךָ כִּ֤י בַחֲטָאֵ֙ינוּ֙ וּבַעֲוֺנ֣וֹת אֲבֹתֵ֔ינוּ יְרוּשָׁלִַ֧ם וְעַמְּךָ֛ לְחֶרְפָּ֖ה לְכָל־ סְבִיבֹתֵֽינוּ׃ (יד) וְעַתָּ֣ה ׀ שְׁמַ֣ע אֱלֹהֵ֗ינוּ אֶל־ תְּפִלַּ֤ת עַבְדְּךָ֙ וְאֶל־ תַּ֣חֲנוּנָ֔יו וְהָאֵ֣ר פָּנֶ֔יךָ עַל־ מִקְדָּשְׁךָ֖ הַשָּׁמֵ֑ם לְמַ֖עַן אֲדֹנָֽי׃ (טו) הַטֵּ֨ה אֱלֹהַ֥י ׀ אָזְנְךָ֮ וּֽשֲׁמָע֒ (פקחה) פְּקַ֣ח עֵינֶ֗יךָ וּרְאֵה֙ שֹֽׁמְמֹתֵ֔ינוּ וְהָעִ֕יר אֲשֶׁר־ נִקְרָ֥א שִׁמְךָ֖ עָלֶ֑יהָ כִּ֣י ׀ לֹ֣א עַל־ צִדְקֹתֵ֗ינוּ אֲנַ֨חְנוּ מַפִּילִ֤ים תַּחֲנוּנֵ֙ינוּ֙ לְפָנֶ֔יךָ כִּ֖י עַל־ רַחֲמֶ֥יךָ הָרַבִּֽים׃ (טז) אֲדֹנָ֤י ׀ שְׁמָ֙עָה֙ אֲדֹנָ֣י ׀ סְלָ֔חָה אֲדֹנָ֛י הַֽקֲשִׁ֥יבָה וַעֲשֵׂ֖ה אַל־ תְּאַחַ֑ר לְמַֽעֲנְךָ֣ אֱלֹהַ֔י כִּֽי־ שִׁמְךָ֣ נִקְרָ֔א עַל־ עִירְךָ֖ וְעַל־ עַמֶּֽךָ׃ (יז) וְע֨וֹד אֲנִ֤י מְדַבֵּר֙ וּמִתְפַּלֵּ֔ל וּמִתְוַדֶּה֙ חַטָּאתִ֔י וְחַטַּ֖את עַמִּ֣י יִשְׂרָאֵ֑ל וּמַפִּ֣יל תְּחִנָּתִ֗י לִפְנֵי֙ יְהוָ֣ה אֱלֹהַ֔י עַ֖ל הַר־ קֹ֥דֶשׁ אֱלֹהָֽי׃ (יח) וְע֛וֹד אֲנִ֥י מְדַבֵּ֖ר בַּתְּפִלָּ֑ה וְהָאִ֣ישׁ גַּבְרִיאֵ֡ל אֲשֶׁר֩ רָאִ֨יתִי בֶחָז֤וֹן בַּתְּחִלָּה֙ מֻעָ֣ף בִּיעָ֔ף נֹגֵ֣עַ אֵלַ֔י כְּעֵ֖ת מִנְחַת־ עָֽרֶב׃ (יט) וַיָּ֖בֶן וַיְדַבֵּ֣ר עִמִּ֑י וַיֹּאמַ֕ר דָּנִיֵּ֕אל עַתָּ֥ה יָצָ֖אתִי לְהַשְׂכִּילְךָ֥ בִינָֽה׃ (כ) בִּתְחִלַּ֨ת תַּחֲנוּנֶ֜יךָ יָצָ֣א דָבָ֗ר וַאֲנִי֙ בָּ֣אתִי לְהַגִּ֔יד כִּ֥י חֲמוּד֖וֹת אָ֑תָּה וּבִין֙ בַּדָּבָ֔ר וְהָבֵ֖ן בַּמַּרְאֶֽה׃ (כא) שָׁבֻעִ֨ים שִׁבְעִ֜ים נֶחְתַּ֥ךְ עַֽל־ עַמְּךָ֣ ׀ וְעַל־ עִ֣יר קָדְשֶׁ֗ךָ לְכַלֵּ֨א הַפֶּ֜שַׁע (ולחתם) וּלְהָתֵ֤ם (חטאות) חַטָּאת֙ וּלְכַפֵּ֣ר עָוֺ֔ן וּלְהָבִ֖יא צֶ֣דֶק עֹֽלָמִ֑ים וְלַחְתֹּם֙ חָז֣וֹן וְנָבִ֔יא וְלִמְשֹׁ֖חַ קֹ֥דֶשׁ קָֽדָשִֽׁים׃ (כב) וְתֵדַ֨ע וְתַשְׂכֵּ֜ל מִן־ מֹצָ֣א דָבָ֗ר לְהָשִׁיב֙ וְלִבְנ֤וֹת יְרֽוּשָׁלִַ֙ם֙ עַד־ מָשִׁ֣יחַ נָגִ֔יד שָׁבֻעִ֖ים שִׁבְעָ֑ה וְשָׁבֻעִ֞ים שִׁשִּׁ֣ים וּשְׁנַ֗יִם תָּשׁוּב֙ וְנִבְנְתָה֙ רְח֣וֹב וְחָר֔וּץ וּבְצ֖וֹק הָעִתִּֽים׃ (כג) וְאַחֲרֵ֤י הַשָּׁבֻעִים֙ שִׁשִּׁ֣ים וּשְׁנַ֔יִם יִכָּרֵ֥ת מָשִׁ֖יחַ וְאֵ֣ין ל֑וֹ וְהָעִ֨יר וְהַקֹּ֜דֶשׁ יַ֠שְׁחִית עַ֣ם נָגִ֤יד הַבָּא֙ וְקִצּ֣וֹ בַשֶּׁ֔טֶף וְעַד֙ קֵ֣ץ מִלְחָמָ֔ה נֶחֱרֶ֖צֶת שֹׁמֵמֽוֹת׃ (כד) וְהִגְבִּ֥יר בְּרִ֛ית לָרַבִּ֖ים שָׁב֣וּעַ אֶחָ֑ד וַחֲצִ֨י הַשָּׁב֜וּעַ יַשְׁבִּ֣ית ׀ זֶ֣בַח וּמִנְחָ֗ה וְעַ֨ל כְּנַ֤ף שִׁקּוּצִים֙ מְשֹׁמֵ֔ם וְעַד־ כָּלָה֙ וְנֶ֣חֱרָצָ֔ה תִּתַּ֖ךְ עַל־ שֹׁמֵֽם׃ תחילת פרק י (כה) בִּשְׁנַ֣ת שָׁל֗וֹשׁ לְכ֙וֹרֶשׁ֙ מֶ֣לֶךְ פָּרַ֔ס דָּבָר֙ נִגְלָ֣ה לְדָֽנִיֵּ֔אל אֲשֶׁר־ נִקְרָ֥א שְׁמ֖וֹ בֵּלְטְשַׁאצַּ֑ר וֶאֱמֶ֤ת הַדָּבָר֙ וְצָבָ֣א גָד֔וֹל וּבִין֙ אֶת־ הַדָּבָ֔ר וּבִ֥ינָה ל֖וֹ בַּמַּרְאֶֽה׃ (כו) בַּיָּמִ֖ים הָהֵ֑ם אֲנִ֤י דָֽנִיֵּאל֙ הָיִ֣יתִי מִתְאַבֵּ֔ל שְׁלֹשָׁ֥ה שָׁבֻעִ֖ים יָמִֽים׃ (כז) לֶ֣חֶם חֲמֻד֞וֹת לֹ֣א אָכַ֗לְתִּי וּבָשָׂ֥ר וָיַ֛יִן לֹא־ בָ֥א אֶל־ פִּ֖י וְס֣וֹךְ לֹא־ סָ֑כְתִּי עַד־ מְלֹ֕את שְׁלֹ֥שֶׁת שָׁבֻעִ֖ים יָמִֽים׃ (כח) וּבְי֛וֹם עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה לַחֹ֣דֶשׁ הָרִאשׁ֑וֹן וַאֲנִ֗י הָיִ֛יתִי עַ֣ל יַ֧ד הַנָּהָ֛ר הַגָּד֖וֹל ה֥וּא חִדָּֽקֶל׃ (כט) וָאֶשָּׂ֤א אֶת־ עֵינַי֙ וָאֵ֔רֶא וְהִנֵּ֥ה אִישׁ־ אֶחָ֖ד לָב֣וּשׁ בַּדִּ֑ים וּמָתְנָ֥יו חֲגֻרִ֖ים בְּכֶ֥תֶם אוּפָֽז׃ (ל) וּגְוִיָּת֣וֹ כְתַרְשִׁ֗ישׁ וּפָנָ֞יו כְּמַרְאֵ֤ה בָרָק֙ וְעֵינָיו֙ כְּלַפִּ֣ידֵי אֵ֔שׁ וּזְרֹֽעֹתָיו֙ וּמַרְגְּלֹתָ֔יו כְּעֵ֖ין נְחֹ֣שֶׁת קָלָ֑ל וְק֥וֹל דְּבָרָ֖יו כְּק֥וֹל הָמֽוֹן׃ (לא) וְרָאִיתִי֩ אֲנִ֨י דָנִיֵּ֤אל לְבַדִּי֙ אֶת־ הַמַּרְאָ֔ה וְהָאֲנָשִׁים֙ אֲשֶׁ֣ר הָי֣וּ עִמִּ֔י לֹ֥א רָא֖וּ אֶת־ הַמַּרְאָ֑ה אֲבָ֗ל חֲרָדָ֤ה גְדֹלָה֙ נָפְלָ֣ה עֲלֵיהֶ֔ם וַֽיִּבְרְח֖וּ בְּהֵחָבֵֽא׃ (לב) וַאֲנִי֙ נִשְׁאַ֣רְתִּי לְבַדִּ֔י וָֽאֶרְאֶ֗ה אֶת־ הַמַּרְאָ֤ה הַגְּדֹלָה֙ הַזֹּ֔את וְלֹ֥א יִשָּׁבֵ֔ר וְלֹ֥א בְאַפַּ֖יִם וְלֹ֥א בְמִלְחָמָֽה׃ (כב) וְעָמַ֤ד עַל־ כַּנּוֹ֙ נִבְזֶ֔ה וְלֹא־ נָתְנ֥וּ עָלָ֖יו ה֣וֹד מַלְכ֑וּת וּבָ֣א בְשַׁלְוָ֔ה וְהֶחֱזִ֥יק מַלְכ֖וּת בַּחֲלַקְלַקּֽוֹת׃ (כג) וּזְרֹע֥וֹת הַשֶּׁ֛טֶף יִשָּׁטְפ֥וּ מִלְּפָנָ֖יו וְיִשָּׁבֵ֑רוּ וְגַ֖ם נְגִ֥יד בְּרִֽית׃ (כד) וּמִן־ הִֽתְחַבְּר֥וּת אֵלָ֖יו יַעֲשֶׂ֣ה מִרְמָ֑ה וְעָלָ֥ה וְעָצַ֖ם בִּמְעַט־ גּֽוֹי׃ (כה) בְּשַׁלְוָ֞ה וּבְמִשְׁמַנֵּ֣י מְדִינָה֮ יָבוֹא֒ וְעָשָׂ֗ה אֲשֶׁ֨ר לֹא־ עָשׂ֤וּ אֲבֹתָיו֙ וַאֲב֣וֹת אֲבֹתָ֔יו בִּזָּ֧ה וְשָׁלָ֛ל וּרְכ֖וּשׁ לָהֶ֣ם יִבְז֑וֹר וְעַ֧ל מִבְצָרִ֛ים יְחַשֵּׁ֥ב מַחְשְׁבֹתָ֖יו וְעַד־ עֵֽת׃ (כו) וְיָעֵר֩ כֹּח֨וֹ וּלְבָב֜וֹ עַל־ מֶ֣לֶךְ הַנֶּגֶב֮ בְּחַ֣יִל גָּדוֹל֒ וּמֶ֣לֶךְ הַנֶּ֗גֶב יִתְגָּרֶה֙ לַמִּלְחָמָ֔ה בְּחַֽיִל־ גָּד֥וֹל וְעָצ֖וּם עַד־ מְאֹ֑ד וְלֹ֣א יַעֲמֹ֔ד כִּֽי־ יַחְשְׁב֥וּ עָלָ֖יו מַחֲשָׁבֽוֹת׃ (כז) וְאֹכְלֵ֧י פַת־ בָּג֛וֹ יִשְׁבְּר֖וּהוּ וְחֵיל֣וֹ יִשְׁט֑וֹף וְנָפְל֖וּ חֲלָלִ֥ים רַבִּֽים׃ (כח) וּשְׁנֵיהֶ֤ם הַמְּלָכִים֙ לְבָבָ֣ם לְמֵרָ֔ע וְעַל־ שֻׁלְחָ֥ן אֶחָ֖ד כָּזָ֣ב יְדַבֵּ֑רוּ וְלֹ֣א תִצְלָ֔ח כִּי־ ע֥וֹד קֵ֖ץ לַמּוֹעֵֽד׃ (כט) וְיָשֹׁ֤ב אַרְצוֹ֙ בִּרְכ֣וּשׁ גָּד֔וֹל וּלְבָב֖וֹ עַל־ בְּרִ֣ית קֹ֑דֶשׁ וְעָשָׂ֖ה וְשָׁ֥ב לְאַרְצֽוֹ׃ (ל) לַמּוֹעֵ֥ד יָשׁ֖וּב וּבָ֣א בַנֶּ֑גֶב וְלֹֽא־ תִהְיֶ֥ה כָרִאשֹׁנָ֖ה וְכָאַחֲרֹנָֽה׃ (לא) וּבָ֨אוּ ב֜וֹ צִיִּ֤ים כִּתִּים֙ וְנִכְאָ֔ה וְשָׁ֛ב וְזָעַ֥ם עַל־ בְּרִֽית־ ק֖וֹדֶשׁ וְעָשָׂ֑ה וְשָׁ֣ב וְיָבֵ֔ן עַל־ עֹזְבֵ֖י בְּרִ֥ית קֹֽדֶשׁ׃ (לב) וּזְרֹעִ֖ים מִמֶּ֣נּוּ יַעֲמֹ֑דוּ וְחִלְּל֞וּ הַמִּקְדָּ֤שׁ הַמָּעוֹז֙ וְהֵסִ֣ירוּ הַתָּמִ֔יד וְנָתְנ֖וּ הַשִּׁקּ֥וּץ מְשׁוֹמֵֽם׃ (לג) וּמַרְשִׁיעֵ֣י בְרִ֔ית יַחֲנִ֖יף בַּחֲלַקּ֑וֹת וְעַ֛ם יֹדְעֵ֥י אֱלֹהָ֖יו יַחֲזִ֥קוּ וְעָשֽׂוּ׃ (לד) וּמַשְׂכִּ֣ילֵי עָ֔ם יָבִ֖ינוּ לָֽרַבִּ֑ים וְנִכְשְׁל֞וּ בְּחֶ֧רֶב וּבְלֶהָבָ֛ה בִּשְׁבִ֥י וּבְבִזָּ֖ה יָמִֽים׃ (לה) וּבְהִכָּ֣שְׁלָ֔ם יֵעָזְר֖וּ עֵ֣זֶר מְעָ֑ט וְנִלְו֧וּ עֲלֵיהֶ֛ם רַבִּ֖ים בַּחֲלַקְלַקּֽוֹת׃ (לו) וּמִן־ הַמַּשְׂכִּילִ֣ים יִכָּֽשְׁל֗וּ לִצְר֥וֹף בָּהֶ֛ם וּלְבָרֵ֥ר וְלַלְבֵּ֖ן עַד־ עֵ֣ת קֵ֑ץ כִּי־ ע֖וֹד לַמּוֹעֵֽד׃ (לז) וְעָשָׂ֨ה כִרְצוֹנ֜וֹ הַמֶּ֗לֶךְ וְיִתְרוֹמֵ֤ם וְיִתְגַּדֵּל֙ עַל־ כָּל־ אֵ֔ל וְעַל֙ אֵ֣ל אֵלִ֔ים יְדַבֵּ֖ר נִפְלָא֑וֹת וְהִצְלִ֙יחַ֙ עַד־ כָּ֣לָה זַ֔עַם כִּ֥י נֶחֱרָצָ֖ה נֶעֱשָֽׂתָה׃ (לח) וְעַל־ אֱלֹהֵ֤י אֲבֹתָיו֙ לֹ֣א יָבִ֔ין וְעַל־ חֶמְדַּ֥ת נָשִׁ֛ים וְעַֽל־ כָּל־ אֱל֖וֹהַּ לֹ֣א יָבִ֑ין כִּ֥י עַל־ כֹּ֖ל יִתְגַּדָּֽל׃ (לט) וְלֶאֱלֹ֙הַּ֙ מָֽעֻזִּ֔ים עַל־ כַּנּ֖וֹ יְכַבֵּ֑ד וְלֶאֱל֜וֹהַּ אֲשֶׁ֧ר לֹא־ יְדָעֻ֣הוּ אֲבֹתָ֗יו יְכַבֵּ֛ד בְּזָהָ֥ב וּבְכֶ֛סֶף וּבְאֶ֥בֶן יְקָרָ֖ה וּבַחֲמֻדֽוֹת׃ (מ) וְעָשָׂ֞ה לְמִבְצְרֵ֤י מָֽעֻזִּים֙ עִם־ אֱל֣וֹהַּ נֵכָ֔ר אֲשֶׁ֥ר (הכיר) יַכִּ֖יר יַרְבֶּ֣ה כָב֑וֹד וְהִמְשִׁילָם֙ בָּֽרַבִּ֔ים וַאֲדָמָ֖ה יְחַלֵּ֥ק בִּמְחִֽיר׃ (מא) וּבְעֵ֣ת קֵ֗ץ יִתְנַגַּ֤ח עִמּוֹ֙ מֶ֣לֶךְ הַנֶּ֔גֶב וְיִשְׂתָּעֵ֨ר עָלָ֜יו מֶ֣לֶךְ הַצָּפ֗וֹן בְּרֶ֙כֶב֙ וּבְפָ֣רָשִׁ֔ים וּבָאֳנִיּ֖וֹת רַבּ֑וֹת וּבָ֥א בַאֲרָצ֖וֹת וְשָׁטַ֥ף וְעָבָֽר׃ (מב) וּבָא֙ בְּאֶ֣רֶץ הַצְּבִ֔י וְרַבּ֖וֹת יִכָּשֵׁ֑לוּ וְאֵ֙לֶּה֙ יִמָּלְט֣וּ מִיָּד֔וֹ אֱד֣וֹם וּמוֹאָ֔ב וְרֵאשִׁ֖ית בְּנֵ֥י עַמּֽוֹן׃ (מג) וְיִשְׁלַ֥ח יָד֖וֹ בַּאֲרָצ֑וֹת וְאֶ֣רֶץ מִצְרַ֔יִם לֹ֥א תִהְיֶ֖ה לִפְלֵיטָֽה׃ (מד) וּמָשַׁ֗ל בְּמִכְמַנֵּי֙ הַזָּהָ֣ב וְהַכֶּ֔סֶף וּבְכֹ֖ל חֲמֻד֣וֹת מִצְרָ֑יִם וְלֻבִ֥ים וְכֻשִׁ֖ים בְּמִצְעָדָֽיו׃ (מה) וּשְׁמֻע֣וֹת יְבַהֲלֻ֔הוּ מִמִּזְרָ֖ח וּמִצָּפ֑וֹן וְיָצָא֙ בְּחֵמָ֣א גְדֹלָ֔ה לְהַשְׁמִ֥יד וּֽלְהַחֲרִ֖ים רַבִּֽים׃ (מו) וְיִטַּע֙ אָהֳלֶ֣י אַפַּדְנ֔וֹ בֵּ֥ין יַמִּ֖ים לְהַר־ צְבִי־ קֹ֑דֶשׁ וּבָא֙ עַד־ קִצּ֔וֹ וְאֵ֥ין עוֹזֵ֖ר לֽוֹ׃ (מז) וּבָעֵ֣ת הַהִיא֩ יַעֲמֹ֨ד מִֽיכָאֵ֜ל הַשַּׂ֣ר הַגָּד֗וֹל הָעֹמֵד֮ עַל־ בְּנֵ֣י עַמֶּךָ֒ וְהָיְתָה֙ עֵ֣ת צָרָ֔ה אֲשֶׁ֤ר לֹֽא־ נִהְיְתָה֙ מִֽהְי֣וֹת גּ֔וֹי עַ֖ד הָעֵ֣ת הַהִ֑יא וּבָעֵ֤ת הַהִיא֙ יִמָּלֵ֣ט עַמְּךָ֔ כָּל־ הַנִּמְצָ֖א כָּת֥וּב בַּסֵּֽפֶר׃ (מח) וְרַבִּ֕ים מִיְּשֵׁנֵ֥י אַדְמַת־ עָפָ֖ר יָקִ֑יצוּ אֵ֚לֶּה לְחַיֵּ֣י עוֹלָ֔ם וְאֵ֥לֶּה לַחֲרָפ֖וֹת לְדִרְא֥וֹן עוֹלָֽם׃ (מט) וְהַ֨מַּשְׂכִּלִ֔ים יַזְהִ֖רוּ כְּזֹ֣הַר הָרָקִ֑יעַ וּמַצְדִּיקֵי֙ הָֽרַבִּ֔ים כַּכּוֹכָבִ֖ים לְעוֹלָ֥ם וָעֶֽד׃ (נ) וְאַתָּ֣ה דָֽנִיֵּ֗אל סְתֹ֧ם הַדְּבָרִ֛ים וַחֲתֹ֥ם הַסֵּ֖פֶר עַד־ עֵ֣ת קֵ֑ץ יְשֹׁטְט֥וּ רַבִּ֖ים וְתִרְבֶּ֥ה הַדָּֽעַת׃ (נא) וְרָאִ֙יתִי֙ אֲנִ֣י דָנִיֵּ֔אל וְהִנֵּ֛ה שְׁנַ֥יִם אֲחֵרִ֖ים עֹמְדִ֑ים אֶחָ֥ד הֵ֙נָּה֙ לִשְׂפַ֣ת הַיְאֹ֔ר וְאֶחָ֥ד הֵ֖נָּה לִשְׂפַ֥ת הַיְאֹֽר׃ (נב) וַיֹּ֗אמֶר לָאִישׁ֙ לְב֣וּשׁ הַבַּדִּ֔ים אֲשֶׁ֥ר מִמַּ֖עַל לְמֵימֵ֣י הַיְאֹ֑ר עַד־ מָתַ֖י קֵ֥ץ הַפְּלָאֽוֹת׃ (נג) וָאֶשְׁמַ֞ע אֶת־ הָאִ֣ישׁ ׀ לְב֣וּשׁ הַבַּדִּ֗ים אֲשֶׁ֣ר מִמַּעַל֮ לְמֵימֵ֣י הַיְאֹר֒ וַיָּ֨רֶם יְמִינ֤וֹ וּשְׂמֹאלוֹ֙ אֶל־ הַשָּׁמַ֔יִם וַיִּשָּׁבַ֖ע בְּחֵ֣י הָעוֹלָ֑ם כִּי֩ לְמוֹעֵ֨ד מֽוֹעֲדִ֜ים וָחֵ֗צִי וּכְכַלּ֛וֹת נַפֵּ֥ץ יַד־ עַם־ קֹ֖דֶשׁ תִּכְלֶ֥ינָה כָל־ אֵֽלֶּה׃ (נד) וַאֲנִ֥י שָׁמַ֖עְתִּי וְלֹ֣א אָבִ֑ין וָאֹ֣מְרָ֔ה אֲדֹנִ֕י מָ֥ה אַחֲרִ֖ית אֵֽלֶּה׃ (נה) וַיֹּ֖אמֶר לֵ֣ךְ דָּנִיֵּ֑אל כִּֽי־ סְתֻמִ֧ים וַחֲתֻמִ֛ים הַדְּבָרִ֖ים עַד־ עֵ֥ת קֵֽץ׃ (נו) יִ֠תְבָּֽרֲרוּ וְיִֽתְלַבְּנ֤וּ וְיִצָּֽרְפוּ֙ רַבִּ֔ים וְהִרְשִׁ֣יעוּ רְשָׁעִ֔ים וְלֹ֥א יָבִ֖ינוּ כָּל־ רְשָׁעִ֑ים וְהַמַּשְׂכִּלִ֖ים יָבִֽינוּ׃ (נז) וּמֵעֵת֙ הוּסַ֣ר הַתָּמִ֔יד וְלָתֵ֖ת שִׁקּ֣וּץ שֹׁמֵ֑ם יָמִ֕ים אֶ֖לֶף מָאתַ֥יִם וְתִשְׁעִֽים׃ (נח) אַשְׁרֵ֥י הַֽמְחַכֶּ֖ה וְיַגִּ֑יעַ לְיָמִ֕ים אֶ֕לֶף שְׁלֹ֥שׁ מֵא֖וֹת שְׁלֹשִׁ֥ים וַחֲמִשָּֽׁה׃ (נט) וְאַתָּ֖ה לֵ֣ךְ לַקֵּ֑ץ וְתָנ֛וּחַ וְתַעֲמֹ֥ד לְגֹרָלְךָ֖ לְקֵ֥ץ הַיָּמִֽין׃
