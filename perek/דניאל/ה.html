דניאל פרק ה לפי החלוקה לסדרים

פרק ה בספר דניאל (30 פסוקים) נמצא בדניאל סדר ג (פסוקים לט–מט) ובדניאל סדר ד (פסוקים א–יט).

דניאל סדר ג

(א) בֵּאדַ֣יִן מַלְכָּ֗א הַצְלַ֛ח לְשַׁדְרַ֥ךְ מֵישַׁ֛ךְ וַעֲבֵ֥ד נְג֖וֹ בִּמְדִינַ֥ת בָּבֶֽל׃ (ב) נְבוּכַדְנֶצַּ֣ר מַלְכָּ֗א לְֽכָל־ עַֽמְמַיָּ֞א אֻמַיָּ֧א וְלִשָּׁנַיָּ֛א דִּֽי־ (דארין) דָיְרִ֥ין בְּכָל־ אַרְעָ֖א שְׁלָמְכ֥וֹן יִשְׂגֵּֽא׃ (ג) אָֽתַיָּא֙ וְתִמְהַיָּ֔א דִּ֚י עֲבַ֣ד עִמִּ֔י אֱלָהָ֖א (עליא) עִלָּאָ֑ה שְׁפַ֥ר קָֽדָמַ֖י לְהַחֲוָיָֽה׃ (ד) אָת֙וֹהִי֙ כְּמָ֣ה רַבְרְבִ֔ין וְתִמְה֖וֹהִי כְּמָ֣ה תַקִּיפִ֑ין מַלְכוּתֵהּ֙ מַלְכ֣וּת עָלַ֔ם וְשָׁלְטָנֵ֖הּ עִם־ דָּ֥ר וְדָֽר׃ (ה) אֲנָ֣ה נְבוּכַדְנֶצַּ֗ר שְׁלֵ֤ה הֲוֵית֙ בְּבֵיתִ֔י וְרַעְנַ֖ן בְּהֵיכְלִֽי׃ (ו) חֵ֥לֶם חֲזֵ֖ית וִֽידַחֲלִנַּ֑נִי וְהַרְהֹרִין֙ עַֽל־ מִשְׁכְּבִ֔י וְחֶזְוֵ֥י רֵאשִׁ֖י יְבַהֲלֻנַּֽנִי׃ (ז) וּמִנִּי֙ שִׂ֣ים טְעֵ֔ם לְהַנְעָלָ֣ה קָֽדָמַ֔י לְכֹ֖ל חַכִּימֵ֣י בָבֶ֑ל דִּֽי־ פְשַׁ֥ר חֶלְמָ֖א יְהֽוֹדְעֻנַּֽנִי׃ (ח) בֵּאדַ֣יִן (עללין) עָלִּ֗ין חַרְטֻמַיָּא֙ אָֽשְׁפַיָּ֔א (כשדיא) כַּשְׂדָּאֵ֖י וְגָזְרַיָּ֑א וְחֶלְמָ֗א אָמַ֤ר אֲנָה֙ קֳדָ֣מֵיה֔וֹן וּפִשְׁרֵ֖הּ לָא־ מְהוֹדְעִ֥ין לִֽי׃ (ט) וְעַ֣ד אָחֳרֵ֡ין עַל֩ קָֽדָמַ֨י דָּנִיֵּ֜אל דִּֽי־ שְׁמֵ֤הּ בֵּלְטְשַׁאצַּר֙ כְּשֻׁ֣ם אֱלָהִ֔י וְדִ֛י רֽוּחַ־ אֱלָהִ֥ין קַדִּישִׁ֖ין בֵּ֑הּ וְחֶלְמָ֖א קָֽדָמ֥וֹהִי אַמְרֵֽת׃ (י) בֵּלְטְשַׁאצַּר֮ רַ֣ב חַרְטֻמַיָּא֒ דִּ֣י ׀ אֲנָ֣ה יִדְעֵ֗ת דִּ֠י ר֣וּחַ אֱלָהִ֤ין קַדִּישִׁין֙ בָּ֔ךְ וְכָל־ רָ֖ז לָא־ אָנֵ֣ס לָ֑ךְ חֶזְוֵ֨י חֶלְמִ֧י דִֽי־ חֲזֵ֛ית וּפִשְׁרֵ֖הּ אֱמַֽר׃ (יא) וְחֶזְוֵ֥י רֵאשִׁ֖י עַֽל־ מִשְׁכְּבִ֑י חָזֵ֣ה הֲוֵ֔ית וַאֲל֥וּ אִילָ֛ן בְּג֥וֹא אַרְעָ֖א וְרוּמֵ֥הּ שַׂגִּֽיא׃ (יב) רְבָ֥ה אִֽילָנָ֖א וּתְקִ֑ף וְרוּמֵהּ֙ יִמְטֵ֣א לִשְׁמַיָּ֔א וַחֲזוֹתֵ֖הּ לְס֥וֹף כָּל־ אַרְעָֽא׃ (יג) עָפְיֵ֤הּ שַׁפִּיר֙ וְאִנְבֵּ֣הּ שַׂגִּ֔יא וּמָז֨וֹן לְכֹ֖לָּא־ בֵ֑הּ תְּחֹת֜וֹהִי תַּטְלֵ֣ל ׀ חֵיוַ֣ת בָּרָ֗א וּבְעַנְפ֙וֹהִי֙ (ידרון) יְדוּרָן֙ צִפֲּרֵ֣י שְׁמַיָּ֔א וּמִנֵּ֖הּ יִתְּזִ֥ין כָּל־ בִּשְׂרָֽא׃ (יד) חָזֵ֥ה הֲוֵ֛ית בְּחֶזְוֵ֥י רֵאשִׁ֖י עַֽל־ מִשְׁכְּבִ֑י וַאֲלוּ֙ עִ֣יר וְקַדִּ֔ישׁ מִן־ שְׁמַיָּ֖א נָחִֽת׃ (טו) קָרֵ֨א בְחַ֜יִל וְכֵ֣ן אָמַ֗ר גֹּ֤דּוּ אִֽילָנָא֙ וְקַצִּ֣צוּ עַנְפ֔וֹהִי אַתַּ֥רוּ עָפְיֵ֖הּ וּבַדַּ֣רוּ אִנְבֵּ֑הּ תְּנֻ֤ד חֵֽיוְתָא֙ מִן־ תַּחְתּ֔וֹהִי וְצִפְּרַיָּ֖א מִן־ עַנְפֽוֹהִי׃ (טז) בְּרַ֨ם עִקַּ֤ר שָׁרְשׁ֙וֹהִי֙ בְּאַרְעָ֣א שְׁבֻ֔קוּ וּבֶֽאֱסוּר֙ דִּֽי־ פַרְזֶ֣ל וּנְחָ֔שׁ בְּדִתְאָ֖א דִּ֣י בָרָ֑א וּבְטַ֤ל שְׁמַיָּא֙ יִצְטַבַּ֔ע וְעִם־ חֵיוְתָ֥א חֲלָקֵ֖הּ בַּעֲשַׂ֥ב אַרְעָֽא׃ (יז) לִבְבֵהּ֙ מִן־ (אנושא) אֲנָשָׁ֣א יְשַׁנּ֔וֹן וּלְבַ֥ב חֵיוָ֖ה יִתְיְהִ֣ב לֵ֑הּ וְשִׁבְעָ֥ה עִדָּנִ֖ין יַחְלְפ֥וּן עֲלֽוֹהִי׃ (יח) בִּגְזֵרַ֤ת עִירִין֙ פִּתְגָמָ֔א וּמֵאמַ֥ר קַדִּישִׁ֖ין שְׁאֵֽלְתָ֑א עַד־ דִּבְרַ֡ת דִּ֣י יִנְדְּע֣וּן חַ֠יַּיָּא דִּֽי־ שַׁלִּ֨יט (עליא) עִלָּאָ֜ה בְּמַלְכ֣וּת (אנושא) אֲנָשָׁ֗א וּלְמַן־ דִּ֤י יִצְבֵּא֙ יִתְּנִנַּ֔הּ וּשְׁפַ֥ל אֲנָשִׁ֖ים יְקִ֥ים (עליה) עֲלַֽהּ׃ (יט) דְּנָה֙ חֶלְמָ֣א חֲזֵ֔ית אֲנָ֖ה מַלְכָּ֣א נְבוּכַדְנֶצַּ֑ר (ואנתה) וְאַ֨נְתְּ בֵּלְטְשַׁאצַּ֜ר פִּשְׁרֵ֣א ׀ אֱמַ֗ר כָּל־ קֳבֵל֙ דִּ֣י ׀ כָּל־ חַכִּימֵ֣י מַלְכוּתִ֗י לָֽא־ יָכְלִ֤ין פִּשְׁרָא֙ לְהוֹדָ֣עֻתַ֔נִי (ואנתה) וְאַ֣נְתְּ כָּהֵ֔ל דִּ֛י רֽוּחַ־ אֱלָהִ֥ין קַדִּישִׁ֖ין בָּֽךְ׃ (כ) אֱדַ֨יִן דָּֽנִיֵּ֜אל דִּֽי־ שְׁמֵ֣הּ בֵּלְטְשַׁאצַּ֗ר אֶשְׁתּוֹמַם֙ כְּשָׁעָ֣ה חֲדָ֔ה וְרַעְיֹנֹ֖הִי יְבַהֲלֻנֵּ֑הּ עָנֵ֨ה מַלְכָּ֜א וְאָמַ֗ר בֵּלְטְשַׁאצַּר֙ חֶלְמָ֤א וּפִשְׁרֵא֙ אַֽל־ יְבַהֲלָ֔ךְ עָנֵ֤ה בֵלְטְשַׁאצַּר֙ וְאָמַ֔ר (מראי) מָרִ֕י חֶלְמָ֥א (לשנאיך) לְשָֽׂנְאָ֖ךְ וּפִשְׁרֵ֥הּ (לעריך) לְעָרָֽךְ׃ (כא) אִֽילָנָא֙ דִּ֣י חֲזַ֔יְתָ דִּ֥י רְבָ֖ה וּתְקִ֑ף וְרוּמֵהּ֙ יִמְטֵ֣א לִשְׁמַיָּ֔א וַחֲזוֹתֵ֖הּ לְכָל־ אַרְעָֽא׃ (כב) וְעָפְיֵ֤הּ שַׁפִּיר֙ וְאִנְבֵּ֣הּ שַׂגִּ֔יא וּמָז֨וֹן לְכֹ֖לָּא־ בֵ֑הּ תְּחֹת֗וֹהִי תְּדוּר֙ חֵיוַ֣ת בָּרָ֔א וּבְעַנְפ֕וֹהִי יִשְׁכְּנָ֖ן צִפֲּרֵ֥י שְׁמַיָּֽא׃ (כג) (אנתה)־ אַנְתְּ־ ה֣וּא מַלְכָּ֔א דִּ֥י רְבַ֖ית וּתְקֵ֑פְתְּ וּרְבוּתָ֤ךְ רְבָת֙ וּמְטָ֣ת לִשְׁמַיָּ֔א וְשָׁלְטָנָ֖ךְ לְס֥וֹף אַרְעָֽא׃ (כד) וְדִ֣י חֲזָ֣ה מַלְכָּ֡א עִ֣יר וְקַדִּ֣ישׁ נָחִ֣ת ׀ מִן־ שְׁמַיָּ֡א וְאָמַר֩ גֹּ֨דּוּ אִֽילָנָ֜א וְחַבְּל֗וּהִי בְּרַ֨ם עִקַּ֤ר שָׁרְשׁ֙וֹהִי֙ בְּאַרְעָ֣א שְׁבֻ֔קוּ וּבֶאֱסוּר֙ דִּֽי־ פַרְזֶ֣ל וּנְחָ֔שׁ בְּדִתְאָ֖א דִּ֣י בָרָ֑א וּבְטַ֧ל שְׁמַיָּ֣א יִצְטַבַּ֗ע וְעִם־ חֵיוַ֤ת בָּרָא֙ חֲלָקֵ֔הּ עַ֛ד דִּֽי־ שִׁבְעָ֥ה עִדָּנִ֖ין יַחְלְפ֥וּן עֲלֽוֹהִי׃ (כה) דְּנָ֥ה פִשְׁרָ֖א מַלְכָּ֑א וּגְזֵרַ֤ת (עליא) עִלָּאָה֙ הִ֔יא דִּ֥י מְטָ֖ת עַל־ (מראי) מָרִ֥י מַלְכָּֽא׃ (כו) וְלָ֣ךְ טָֽרְדִ֣ין מִן־ אֲנָשָׁ֡א וְעִם־ חֵיוַ֣ת בָּרָא֩ לֶהֱוֵ֨ה מְדֹרָ֜ךְ וְעִשְׂבָּ֥א כְתוֹרִ֣ין ׀ לָ֣ךְ יְטַֽעֲמ֗וּן וּמִטַּ֤ל שְׁמַיָּא֙ לָ֣ךְ מְצַבְּעִ֔ין וְשִׁבְעָ֥ה עִדָּנִ֖ין יַחְלְפ֣וּן עֲלָ֑יךְ עַ֣ד דִּֽי־ תִנְדַּ֗ע דִּֽי־ שַׁלִּ֤יט עִלָּיָא֙ בְּמַלְכ֣וּת אֲנָשָׁ֔א וּלְמַן־ דִּ֥י יִצְבֵּ֖א יִתְּנִנַּֽהּ׃ (כז) וְדִ֣י אֲמַ֗רוּ לְמִשְׁבַּ֞ק עִקַּ֤ר שָׁרְשׁ֙וֹהִי֙ דִּ֣י אִֽילָנָ֔א מַלְכוּתָ֖ךְ לָ֣ךְ קַיָּמָ֑ה מִן־ דִּ֣י תִנְדַּ֔ע דִּ֥י שַׁלִּטִ֖ן שְׁמַיָּֽא׃ (כח) לָהֵ֣ן מַלְכָּ֗א מִלְכִּי֙ יִשְׁפַּ֣ר (עליך) עֲלָ֔ךְ (וחטיך) וַחֲטָאָךְ֙ בְּצִדְקָ֣ה פְרֻ֔ק וַעֲוָיָתָ֖ךְ בְּמִחַ֣ן עֲנָ֑יִן הֵ֛ן תֶּהֱוֵ֥א אַרְכָ֖ה לִשְׁלֵוְתָֽךְ׃ (כט) כֹּ֣לָּא מְּטָ֔א עַל־ נְבוּכַדְנֶצַּ֖ר מַלְכָּֽא׃ (ל) לִקְצָ֥ת יַרְחִ֖ין תְּרֵֽי־ עֲשַׂ֑ר עַל־ הֵיכַ֧ל מַלְכוּתָ֛א דִּ֥י בָבֶ֖ל מְהַלֵּ֥ךְ הֲוָֽה׃ (לא) עָנֵ֤ה מַלְכָּא֙ וְאָמַ֔ר הֲלָ֥א דָא־ הִ֖יא בָּבֶ֣ל רַבְּתָ֑א דִּֽי־ אֲנָ֤ה בֱנַיְתַהּ֙ לְבֵ֣ית מַלְכ֔וּ בִּתְקַ֥ף חִסְנִ֖י וְלִיקָ֥ר הַדְרִֽי׃ (לב) ע֗וֹד מִלְּתָא֙ בְּפֻ֣ם מַלְכָּ֔א קָ֖ל מִן־ שְׁמַיָּ֣א נְפַ֑ל לָ֤ךְ אָֽמְרִין֙ נְבוּכַדְנֶצַּ֣ר מַלְכָּ֔א מַלְכוּתָ֖ה עֲדָ֥ת מִנָּֽךְ׃ (לג) וּמִן־ אֲנָשָׁא֩ לָ֨ךְ טָֽרְדִ֜ין וְֽעִם־ חֵיוַ֧ת בָּרָ֣א מְדֹרָ֗ךְ עִשְׂבָּ֤א כְתוֹרִין֙ לָ֣ךְ יְטַעֲמ֔וּן וְשִׁבְעָ֥ה עִדָּנִ֖ין יַחְלְפ֣וּן (עליך) עֲלָ֑ךְ עַ֣ד דִּֽי־ תִנְדַּ֗ע דִּֽי־ שַׁלִּ֤יט עִלָּיָא֙ בְּמַלְכ֣וּת אֲנָשָׁ֔א וּלְמַן־ דִּ֥י יִצְבֵּ֖א וְדָחֲלִ֖ין מִן־ קֳדָמ֑וֹהִי דִּֽי־ הֲוָ֨ה צָבֵ֜א הֲוָ֣א קָטֵ֗ל וְדִֽי־ הֲוָ֤ה צָבֵא֙ הֲוָ֣ה מַחֵ֔א וְדִֽי־ הֲוָ֤ה צָבֵא֙ הֲוָ֣ה מָרִ֔ים וְדִֽי־ הֲוָ֥ה צָבֵ֖א הֲוָ֥ה מַשְׁפִּֽיל׃ (ט) וּכְדִי֙ רִ֣ם לִבְבֵ֔הּ וְרוּחֵ֖הּ תִּֽקְפַ֣ת לַהֲזָדָ֑ה הָנְחַת֙ מִן־ כָּרְסֵ֣א מַלְכוּתֵ֔הּ וִֽיקָרָ֖ה הֶעְדִּ֥יוּ מִנֵּֽהּ׃ (י) וּמִן־ בְּנֵי֩ אֲנָשָׁ֨א טְרִ֜יד וְלִבְבֵ֣הּ ׀ עִם־ חֵיוְתָ֣א (שוי) שַׁוִּ֗יְו וְעִם־ עֲרָֽדַיָּא֙ מְדוֹרֵ֔הּ עִשְׂבָּ֤א כְתוֹרִין֙ יְטַ֣עֲמוּנֵּ֔הּ וּמִטַּ֥ל שְׁמַיָּ֖א גִּשְׁמֵ֣הּ יִצְטַבַּ֑ע עַ֣ד דִּֽי־ יְדַ֗ע דִּֽי־ שַׁלִּ֞יט אֱלָהָ֤א (עליא) עִלָּאָה֙ בְּמַלְכ֣וּת אֲנָשָׁ֔א וּלְמַן־ דִּ֥י יִצְבֵּ֖ה יְהָקֵ֥ים (עליה) עֲלַֽהּ׃ (יא) (ואנתה) וְאַ֤נְתְּ בְּרֵהּ֙ בֵּלְשַׁאצַּ֔ר לָ֥א הַשְׁפֵּ֖לְתְּ לִבְבָ֑ךְ כָּל־ קֳבֵ֕ל דִּ֥י כָל־ דְּנָ֖ה יְדַֽעְתָּ׃ (יב) וְעַ֣ל מָרֵֽא־ שְׁמַיָּ֣א ׀ הִתְרוֹמַ֡מְתָּ וּלְמָֽאנַיָּ֨א דִֽי־ בַיְתֵ֜הּ הַיְתִ֣יו (קדמיך) קָֽדָמָ֗ךְ (ואנתה) וְאַ֨נְתְּ (ורברבניך) וְרַבְרְבָנָ֜ךְ שֵֽׁגְלָתָ֣ךְ וּלְחֵנָתָךְ֮ חַמְרָא֮ שָׁתַ֣יִן בְּהוֹן֒ וְלֵֽאלָהֵ֣י כַסְפָּֽא־ וְ֠דַהֲבָא נְחָשָׁ֨א פַרְזְלָ֜א אָעָ֣א וְאַבְנָ֗א דִּ֠י לָֽא־ חָזַ֧יִן וְלָא־ שָׁמְעִ֛ין וְלָ֥א יָדְעִ֖ין שַׁבַּ֑חְתָּ וְלֵֽאלָהָ֞א דִּֽי־ נִשְׁמְתָ֥ךְ בִּידֵ֛הּ וְכָל־ אֹרְחָתָ֥ךְ לֵ֖הּ לָ֥א הַדַּֽרְתָּ׃ (יג) בֵּאדַ֙יִן֙ מִן־ קֳדָמ֔וֹהִי שְׁלִ֖יַחַ פַּסָּ֣א דִֽי־ יְדָ֑א וּכְתָבָ֥א דְנָ֖ה רְשִֽׁים׃ (יד) וּדְנָ֥ה כְתָבָ֖א דִּ֣י רְשִׁ֑ים מְנֵ֥א מְנֵ֖א תְּקֵ֥ל וּפַרְסִֽין׃ (טו) דְּנָ֖ה פְּשַֽׁר־ מִלְּתָ֑א מְנֵ֕א מְנָֽה־ אֱלָהָ֥א מַלְכוּתָ֖ךְ וְהַשְׁלְמַֽהּ׃ (טז) תְּקֵ֑ל תְּקִ֥ילְתָּה בְמֹֽאזַנְיָ֖א וְהִשְׁתְּכַ֥חַתְּ חַסִּֽיר׃ (יז) פְּרֵ֑ס פְּרִיסַת֙ מַלְכוּתָ֔ךְ וִיהִיבַ֖ת לְמָדַ֥י וּפָרָֽס׃ (יח) בֵּאדַ֣יִן ׀ אֲמַ֣ר בֵּלְשַׁאצַּ֗ר וְהַלְבִּ֤ישׁוּ לְדָֽנִיֵּאל֙ אַרְגְּוָנָ֔א (והמונכא) וְהַֽמְנִיכָ֥א דִֽי־ דַהֲבָ֖א עַֽל־ צַוְּארֵ֑הּ וְהַכְרִ֣זֽוּ עֲל֔וֹהִי דִּֽי־ לֶהֱוֵ֥א שַׁלִּ֛יט תַּלְתָּ֖א בְּמַלְכוּתָֽא׃ (יט) בֵּ֚הּ בְּלֵ֣ילְיָ֔א קְטִ֕יל בֵּלְאשַׁצַּ֖ר מַלְכָּ֥א (כשדיא) כַשְׂדָּאָֽה׃ סוף פרק ה (כ) וְדָרְיָ֙וֶשׁ֙ מָֽדָיָ֔א קַבֵּ֖ל מַלְכוּתָ֑א כְּבַ֥ר שְׁנִ֖ין שִׁתִּ֥ין וְתַרְתֵּֽין׃ (כא) שְׁפַר֙ קֳדָ֣ם דָּרְיָ֔וֶשׁ וַהֲקִים֙ עַל־ מַלְכוּתָ֔א לַאֲחַשְׁדַּרְפְּנַיָּ֖א מְאָ֣ה וְעֶשְׂרִ֑ין דִּ֥י לֶהֱוֺ֖ן בְּכָל־ מַלְכוּתָֽא׃ (כב) וְעֵ֤לָּא מִנְּהוֹן֙ סָרְכִ֣ין תְּלָתָ֔א דִּ֥י דָנִיֵּ֖אל חַֽד־ מִנְּה֑וֹן דִּֽי־ לֶהֱוֺ֞ן אֲחַשְׁדַּרְפְּנַיָּ֣א אִלֵּ֗ין יָהֲבִ֤ין לְהוֹן֙ טַעְמָ֔א וּמַלְכָּ֖א לָֽא־ לֶהֱוֵ֥א נָזִֽק׃ (כג) אֱדַ֙יִן֙ דָּנִיֵּ֣אל דְּנָ֔ה הֲוָ֣א מִתְנַצַּ֔ח עַל־ סָרְכַיָּ֖א וַאֲחַשְׁדַּרְפְּנַיָּ֑א כָּל־ קֳבֵ֗ל דִּ֣י ר֤וּחַ יַתִּירָא֙ בֵּ֔הּ וּמַלְכָּ֣א עֲשִׁ֔ית לַהֲקָמוּתֵ֖הּ עַל־ כָּל־ מַלְכוּתָֽא׃ (כד) אֱדַ֨יִן סָֽרְכַיָּ֜א וַאֲחַשְׁדַּרְפְּנַיָּ֗א הֲו֨וֹ בָעַ֧יִן עִלָּ֛ה לְהַשְׁכָּחָ֥ה לְדָנִיֵּ֖אל מִצַּ֣ד מַלְכוּתָ֑א וְכָל־ עִלָּ֨ה וּשְׁחִיתָ֜ה לָא־ יָכְלִ֣ין לְהַשְׁכָּחָ֗ה כָּל־ קֳבֵל֙ דִּֽי־ מְהֵימַ֣ן ה֔וּא וְכָל־ שָׁלוּ֙ וּשְׁחִיתָ֔ה לָ֥א הִשְׁתְּכַ֖חַת עֲלֽוֹהִי׃ (כה) אֱ֠דַיִן גֻּבְרַיָּ֤א אִלֵּךְ֙ אָֽמְרִ֔ין דִּ֣י לָ֧א נְהַשְׁכַּ֛ח לְדָנִיֵּ֥אל דְּנָ֖ה כָּל־ עִלָּ֑א לָהֵ֕ן הַשְׁכַּ֥חְנָֽה עֲל֖וֹהִי בְּדָ֥ת אֱלָהֵֽהּ׃ (כו) אֱ֠דַיִן סָרְכַיָּ֤א וַאֲחַשְׁדַּרְפְּנַיָּא֙ אִלֵּ֔ן הַרְגִּ֖שׁוּ עַל־ מַלְכָּ֑א וְכֵן֙ אָמְרִ֣ין לֵ֔הּ דָּרְיָ֥וֶשׁ מַלְכָּ֖א לְעָלְמִ֥ין חֱיִֽי׃ (כז) אִתְיָעַ֜טוּ כֹּ֣ל ׀ סָרְכֵ֣י מַלְכוּתָ֗א סִגְנַיָּ֤א וַֽאֲחַשְׁדַּרְפְּנַיָּא֙ הַדָּֽבְרַיָּ֣א וּפַחֲוָתָ֔א לְקַיָּמָ֤ה קְיָם֙ מַלְכָּ֔א וּלְתַקָּפָ֖ה אֱסָ֑ר דִּ֣י כָל־ דִּֽי־ יִבְעֵ֣ה בָ֠עוּ מִן־ כָּל־ אֱלָ֨הּ וֶֽאֱנָ֜שׁ עַד־ יוֹמִ֣ין תְּלָתִ֗ין לָהֵן֙ מִנָּ֣ךְ מַלְכָּ֔א יִתְרְמֵ֕א לְגֹ֖ב אַרְיָוָתָֽא׃ (כח) כְּעַ֣ן מַלְכָּ֔א תְּקִ֥ים אֱסָרָ֖א וְתִרְשֻׁ֣ם כְּתָבָ֑א דִּ֣י לָ֧א לְהַשְׁנָיָ֛ה כְּדָת־ מָדַ֥י וּפָרַ֖ס דִּי־ לָ֥א תֶעְדֵּֽא׃ (כט) כָּל־ קֳבֵ֖ל דְּנָ֑ה מַלְכָּא֙ דָּֽרְיָ֔וֶשׁ רְשַׁ֥ם כְּתָבָ֖א וֶאֱסָרָֽא׃
