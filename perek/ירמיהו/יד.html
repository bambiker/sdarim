ירמיהו פרק יד לפי החלוקה לסדרים

פרק יד בספר ירמיהו (22 פסוקים) נמצא בירמיהו סדר ז (פסוקים לא–נא) ובירמיהו סדר ח (פסוקים א–א).

ירמיהו סדר ז

(א) וְהָיָ֗ה אַֽחֲרֵי֙ נָתְשִׁ֣י אוֹתָ֔ם אָשׁ֖וּב וְרִֽחַמְתִּ֑ים וַהֲשִׁבֹתִ֛ים אִ֥ישׁ לְנַחֲלָת֖וֹ וְאִ֥ישׁ לְאַרְצֽוֹ׃ (ב) וְהָיָ֡ה אִם־ לָמֹ֣ד יִלְמְדוּ֩ אֶת־ דַּֽרְכֵ֨י עַמִּ֜י לְהִשָּׁבֵ֤עַ בִּשְׁמִי֙ חַי־ יְהוָ֔ה כַּאֲשֶׁ֤ר לִמְּדוּ֙ אֶת־ עַמִּ֔י לְהִשָּׁבֵ֖עַ בַּבָּ֑עַל וְנִבְנ֖וּ בְּת֥וֹךְ עַמִּֽי׃ (ג) וְאִ֖ם לֹ֣א יִשְׁמָ֑עוּ וְנָ֨תַשְׁתִּ֜י אֶת־ הַגּ֥וֹי הַה֛וּא נָת֥וֹשׁ וְאַבֵּ֖ד נְאֻם־ יְהוָֽה׃ (ד) כֹּֽה־ אָמַ֨ר יְהוָ֜ה אֵלַ֗י הָל֞וֹךְ וְקָנִ֤יתָ לְּךָ֙ אֵז֣וֹר פִּשְׁתִּ֔ים וְשַׂמְתּ֖וֹ עַל־ מָתְנֶ֑יךָ וּבַמַּ֖יִם לֹ֥א תְבִאֵֽהוּ׃ (ה) וָאֶקְנֶ֥ה אֶת־ הָאֵז֖וֹר כִּדְבַ֣ר יְהוָ֑ה וָאָשִׂ֖ם עַל־ מָתְנָֽי׃ (ו) וַיְהִ֧י דְבַר־ יְהוָ֛ה אֵלַ֖י שֵׁנִ֥ית לֵאמֹֽר׃ (ז) קַ֧ח אֶת־ הָאֵז֛וֹר אֲשֶׁ֥ר קָנִ֖יתָ אֲשֶׁ֣ר עַל־ מָתְנֶ֑יךָ וְקוּם֙ לֵ֣ךְ פְּרָ֔תָה וְטָמְנֵ֥הוּ שָׁ֖ם בִּנְקִ֥יק הַסָּֽלַע׃ (ח) וָאֵלֵ֕ךְ וָאֶטְמְנֵ֖הוּ בִּפְרָ֑ת כַּאֲשֶׁ֛ר צִוָּ֥ה יְהוָ֖ה אוֹתִֽי׃ (ט) וַיְהִ֕י מִקֵּ֖ץ יָמִ֣ים רַבִּ֑ים וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י ק֚וּם לֵ֣ךְ פְּרָ֔תָה וְקַ֤ח מִשָּׁם֙ אֶת־ הָ֣אֵז֔וֹר אֲשֶׁ֥ר צִוִּיתִ֖יךָ לְטָמְנוֹ־ שָֽׁם׃ (י) וָאֵלֵ֣ךְ פְּרָ֔תָה וָאֶחְפֹּ֗ר וָֽאֶקַּח֙ אֶת־ הָ֣אֵז֔וֹר מִן־ הַמָּק֖וֹם אֲשֶׁר־ טְמַנְתִּ֣יו שָׁ֑מָּה וְהִנֵּה֙ נִשְׁחַ֣ת הָאֵז֔וֹר לֹ֥א יִצְלַ֖ח לַכֹּֽל׃ (יא) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יב) כֹּ֖ה אָמַ֣ר יְהוָ֑ה כָּ֠כָה אַשְׁחִ֞ית אֶת־ גְּא֧וֹן יְהוּדָ֛ה וְאֶת־ גְּא֥וֹן יְרוּשָׁלִַ֖ם הָרָֽב׃ (יג) הָעָם֩ הַזֶּ֨ה הָרָ֜ע הַֽמֵּאֲנִ֣ים ׀ לִשְׁמ֣וֹעַ אֶת־ דְּבָרַ֗י הַהֹֽלְכִים֙ בִּשְׁרִר֣וּת לִבָּ֔ם וַיֵּלְכ֗וּ אַֽחֲרֵי֙ אֱלֹהִ֣ים אֲחֵרִ֔ים לְעָבְדָ֖ם וּלְהִשְׁתַּחֲוֺ֣ת לָהֶ֑ם וִיהִי֙ כָּאֵז֣וֹר הַזֶּ֔ה אֲשֶׁ֥ר לֹא־ יִצְלַ֖ח לַכֹּֽל׃ (יד) כִּ֡י כַּאֲשֶׁר֩ יִדְבַּ֨ק הָאֵז֜וֹר אֶל־ מָתְנֵי־ אִ֗ישׁ כֵּ֣ן הִדְבַּ֣קְתִּי אֵ֠לַי אֶת־ כָּל־ בֵּ֨ית יִשְׂרָאֵ֜ל וְאֶת־ כָּל־ בֵּ֤ית יְהוּדָה֙ נְאֻם־ יְהוָ֔ה לִֽהְי֥וֹת לִי֙ לְעָ֔ם וּלְשֵׁ֥ם וְלִתְהִלָּ֖ה וּלְתִפְאָ֑רֶת וְלֹ֖א שָׁמֵֽעוּ׃ (טו) וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם אֶת־ הַדָּבָ֣ר הַזֶּ֗ה כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל כָּל־ נֵ֖בֶל יִמָּ֣לֵא יָ֑יִן וְאָמְר֣וּ אֵלֶ֔יךָ הֲיָדֹ֙עַ֙ לֹ֣א נֵדַ֔ע כִּ֥י כָל־ נֵ֖בֶל יִמָּ֥לֵא יָֽיִן׃ (טז) וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם כֹּֽה־ אָמַ֣ר יְהוָ֗ה הִנְנִ֣י מְמַלֵּ֣א אֶת־ כָּל־ יֹשְׁבֵ֣י הָאָ֪רֶץ הַזֹּ֟את וְאֶת־ הַמְּלָכִ֣ים הַיֹּשְׁבִים֩ לְדָוִ֨ד עַל־ כִּסְא֜וֹ וְאֶת־ הַכֹּהֲנִ֣ים וְאֶת־ הַנְּבִיאִ֗ים וְאֵ֛ת כָּל־ יֹשְׁבֵ֥י יְרוּשָׁלִָ֖ם שִׁכָּרֽוֹן׃ (יז) וְנִפַּצְתִּים֩ אִ֨ישׁ אֶל־ אָחִ֜יו וְהָאָב֧וֹת וְהַבָּנִ֛ים יַחְדָּ֖ו נְאֻם־ יְהוָ֑ה לֹֽא־ אֶחְמ֧וֹל וְלֹֽא־ אָח֛וּס וְלֹ֥א אֲרַחֵ֖ם מֵהַשְׁחִיתָֽם׃ (יח) שִׁמְע֥וּ וְהַאֲזִ֖ינוּ אַל־ תִּגְבָּ֑הוּ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (יט) תְּנוּ֩ לַיהוָ֨ה אֱלֹהֵיכֶ֤ם כָּבוֹד֙ בְּטֶ֣רֶם יַחְשִׁ֔ךְ וּבְטֶ֛רֶם יִֽתְנַגְּפ֥וּ רַגְלֵיכֶ֖ם עַל־ הָ֣רֵי נָ֑שֶׁף וְקִוִּיתֶ֤ם לְאוֹר֙ וְשָׂמָ֣הּ לְצַלְמָ֔וֶת (ישית) וְשִׁ֖ית לַעֲרָפֶֽל׃ (כ) וְאִם֙ לֹ֣א תִשְׁמָע֔וּהָ בְּמִסְתָּרִ֥ים תִּבְכֶּֽה־ נַפְשִׁ֖י מִפְּנֵ֣י גֵוָ֑ה וְדָמֹ֨עַ תִּדְמַ֜ע וְתֵרַ֤ד עֵינִי֙ דִּמְעָ֔ה כִּ֥י נִשְׁבָּ֖ה עֵ֥דֶר יְהוָֽה׃ (כא) אֱמֹ֥ר לַמֶּ֛לֶךְ וְלַגְּבִירָ֖ה הַשְׁפִּ֣ילוּ שֵׁ֑בוּ כִּ֤י יָרַד֙ מַרְאֲשׁ֣וֹתֵיכֶ֔ם עֲטֶ֖רֶת תִּֽפְאַרְתְּכֶֽם׃ (כב) עָרֵ֥י הַנֶּ֛גֶב סֻגְּר֖וּ וְאֵ֣ין פֹּתֵ֑חַ הָגְלָ֧ת יְהוּדָ֛ה כֻּלָּ֖הּ הָגְלָ֥ת שְׁלוֹמִֽים׃ (כג) (שאי) שְׂא֤וּ עֵֽינֵיכֶם֙ (וראי) וּרְא֔וּ הַבָּאִ֖ים מִצָּפ֑וֹן אַיֵּ֗ה הָעֵ֙דֶר֙ נִתַּן־ לָ֔ךְ צֹ֖אן תִּפְאַרְתֵּֽךְ׃ (כד) מַה־ תֹּֽאמְרִי֙ כִּֽי־ יִפְקֹ֣ד עָלַ֔יִךְ וְ֠אַתְּ לִמַּ֨דְתְּ אֹתָ֥ם עָלַ֛יִךְ אַלֻּפִ֖ים לְרֹ֑אשׁ הֲל֤וֹא חֲבָלִים֙ יֹאחֱז֔וּךְ כְּמ֖וֹ אֵ֥שֶׁת לֵדָֽה׃ (כה) וְכִ֤י תֹאמְרִי֙ בִּלְבָבֵ֔ךְ מַדּ֖וּעַ קְרָאֻ֣נִי אֵ֑לֶּה בְּרֹ֧ב עֲוֺנֵ֛ךְ נִגְל֥וּ שׁוּלַ֖יִךְ נֶחְמְס֥וּ עֲקֵבָֽיִךְ׃ (כו) הֲיַהֲפֹ֤ךְ כּוּשִׁי֙ עוֹר֔וֹ וְנָמֵ֖ר חֲבַרְבֻּרֹתָ֑יו גַּם־ אַתֶּם֙ תּוּכְל֣וּ לְהֵיטִ֔יב לִמֻּדֵ֖י הָרֵֽעַ׃ (כז) וַאֲפִיצֵ֖ם כְּקַשׁ־ עוֹבֵ֑ר לְר֖וּחַ מִדְבָּֽר׃ (כח) זֶ֣ה גוֹרָלֵ֧ךְ מְנָת־ מִדַּ֛יִךְ מֵֽאִתִּ֖י נְאֻם־ יְהוָ֑ה אֲשֶׁר֙ שָׁכַ֣חַתְּ אוֹתִ֔י וַֽתִּבְטְחִ֖י בַּשָּֽׁקֶר׃ (כט) וְגַם־ אֲנִ֛י חָשַׂ֥פְתִּי שׁוּלַ֖יִךְ עַל־ פָּנָ֑יִךְ וְנִרְאָ֖ה קְלוֹנֵֽךְ׃ (ל) נִֽאֻפַ֤יִךְ וּמִצְהֲלוֹתַ֙יִךְ֙ זִמַּ֣ת זְנוּתֵ֔ךְ עַל־ גְּבָעוֹת֙ בַּשָּׂדֶ֔ה רָאִ֖יתִי שִׁקּוּצָ֑יִךְ א֥וֹי לָךְ֙ יְר֣וּשָׁלִַ֔ם לֹ֣א תִטְהֲרִ֔י אַחֲרֵ֥י מָתַ֖י עֹֽד׃ תחילת פרק יד (לא) אֲשֶׁ֨ר הָיָ֤ה דְבַר־ יְהוָה֙ אֶֽל־ יִרְמְיָ֔הוּ עַל־ דִּבְרֵ֖י הַבַּצָּרֽוֹת׃ (לב) אָבְלָ֣ה יְהוּדָ֔ה וּשְׁעָרֶ֥יהָ אֻמְלְל֖וּ קָדְר֣וּ לָאָ֑רֶץ וְצִוְחַ֥ת יְרוּשָׁלִַ֖ם עָלָֽתָה׃ (לג) וְאַדִּ֣רֵיהֶ֔ם שָׁלְח֥וּ (צעוריהם) צְעִירֵיהֶ֖ם לַמָּ֑יִם בָּ֣אוּ עַל־ גֵּבִ֞ים לֹא־ מָ֣צְאוּ מַ֗יִם שָׁ֤בוּ כְלֵיהֶם֙ רֵיקָ֔ם בֹּ֥שׁוּ וְהָכְלְמ֖וּ וְחָפ֥וּ רֹאשָֽׁם׃ (לד) בַּעֲב֤וּר הָאֲדָמָה֙ חַ֔תָּה כִּ֛י לֹא־ הָיָ֥ה גֶ֖שֶׁם בָּאָ֑רֶץ בֹּ֥שׁוּ אִכָּרִ֖ים חָפ֥וּ רֹאשָֽׁם׃ (לה) כִּ֤י גַם־ אַיֶּ֙לֶת֙ בַּשָּׂדֶ֔ה יָלְדָ֖ה וְעָז֑וֹב כִּ֥י לֹֽא־ הָיָ֖ה דֶּֽשֶׁא׃ (לו) וּפְרָאִים֙ עָמְד֣וּ עַל־ שְׁפָיִ֔ם שָׁאֲפ֥וּ ר֖וּחַ כַּתַּנִּ֑ים כָּל֥וּ עֵינֵיהֶ֖ם כִּי־ אֵ֥ין עֵֽשֶׂב׃ (לז) אִם־ עֲוֺנֵ֙ינוּ֙ עָ֣נוּ בָ֔נוּ יְהוָ֕ה עֲשֵׂ֖ה לְמַ֣עַן שְׁמֶ֑ךָ כִּֽי־ רַבּ֥וּ מְשׁוּבֹתֵ֖ינוּ לְךָ֥ חָטָֽאנוּ׃ (לח) מִקְוֵה֙ יִשְׂרָאֵ֔ל מֽוֹשִׁיע֖וֹ בְּעֵ֣ת צָרָ֑ה לָ֤מָּה תִֽהְיֶה֙ כְּגֵ֣ר בָּאָ֔רֶץ וּכְאֹרֵ֖חַ נָטָ֥ה לָלֽוּן׃ (לט) לָ֤מָּה תִֽהְיֶה֙ כְּאִ֣ישׁ נִדְהָ֔ם כְּגִבּ֖וֹר לֹא־ יוּכַ֣ל לְהוֹשִׁ֑יעַ וְאַתָּ֧ה בְקִרְבֵּ֣נוּ יְהוָ֗ה וְשִׁמְךָ֛ עָלֵ֥ינוּ נִקְרָ֖א אַל־ תַּנִּחֵֽנוּ׃ (מ) כֹּֽה־ אָמַ֨ר יְהוָ֜ה לָעָ֣ם הַזֶּ֗ה כֵּ֤ן אָֽהֲבוּ֙ לָנ֔וּעַ רַגְלֵיהֶ֖ם לֹ֣א חָשָׂ֑כוּ וַיהוָה֙ לֹ֣א רָצָ֔ם עַתָּה֙ יִזְכֹּ֣ר עֲוֺנָ֔ם וְיִפְקֹ֖ד חַטֹּאתָֽם׃ (מא) וַיֹּ֥אמֶר יְהוָ֖ה אֵלָ֑י אַל־ תִּתְפַּלֵּ֛ל בְּעַד־ הָעָ֥ם הַזֶּ֖ה לְטוֹבָֽה׃ (מב) כִּ֣י יָצֻ֗מוּ אֵינֶ֤נִּי שֹׁמֵ֙עַ֙ אֶל־ רִנָּתָ֔ם וְכִ֧י יַעֲל֛וּ עֹלָ֥ה וּמִנְחָ֖ה אֵינֶ֣נִּי רֹצָ֑ם כִּ֗י בַּחֶ֙רֶב֙ וְקֶ֤סֶם (ואלול) וֶֽאֱלִיל֙ (ותרמות) וְתַרְמִ֣ית לִבָּ֔ם הֵ֖מָּה מִֽתְנַבְּאִ֥ים לָכֶֽם׃ (מה) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה עַֽל־ הַנְּבִאִ֞ים הַנִּבְּאִ֣ים בִּשְׁמִי֮ וַאֲנִ֣י לֹֽא־ שְׁלַחְתִּים֒ הַשָּׂדֶ֗ה וְהִנֵּה֙ חַלְלֵי־ חֶ֔רֶב וְאִם֙ בָּ֣אתִי הָעִ֔יר וְהִנֵּ֖ה תַּחֲלוּאֵ֣י רָעָ֑ב כִּֽי־ גַם־ נָבִ֧יא גַם־ כֹּהֵ֛ן סָחֲר֥וּ אֶל־ אֶ֖רֶץ וְלֹ֥א יָדָֽעוּ׃ (מט) הֲמָאֹ֨ס מָאַ֜סְתָּ אֶת־ יְהוּדָ֗ה אִם־ בְּצִיּוֹן֙ גָּעֲלָ֣ה נַפְשֶׁ֔ךָ מַדּ֙וּעַ֙ הִכִּיתָ֔נוּ וְאֵ֥ין לָ֖נוּ מַרְפֵּ֑א קַוֵּ֤ה לְשָׁלוֹם֙ וְאֵ֣ין ט֔וֹב וּלְעֵ֥ת מַרְפֵּ֖א וְהִנֵּ֥ה בְעָתָֽה׃ (נ) יָדַ֧עְנוּ יְהוָ֛ה (א) הֲיֵ֨שׁ בְּהַבְלֵ֤י הַגּוֹיִם֙ מַגְשִׁמִ֔ים וְאִם־ הַשָּׁמַ֖יִם יִתְּנ֣וּ רְבִבִ֑ים הֲלֹ֨א אַתָּה־ ה֜וּא יְהוָ֤ה אֱלֹהֵ֙ינוּ֙ וּ֨נְקַוֶּה־ לָּ֔ךְ כִּֽי־ אַתָּ֥ה עָשִׂ֖יתָ אֶת־ כָּל־ אֵֽלֶּה׃ סוף פרק יד (ב) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י אִם־ יַעֲמֹ֨ד מֹשֶׁ֤ה וּשְׁמוּאֵל֙ לְפָנַ֔י אֵ֥ין נַפְשִׁ֖י אֶל־ הָעָ֣ם הַזֶּ֑ה שַׁלַּ֥ח מֵֽעַל־ פָּנַ֖י וְיֵצֵֽאוּ׃ (ג) וְהָיָ֛ה כִּֽי־ יֹאמְר֥וּ אֵלֶ֖יךָ אָ֣נָה נֵצֵ֑א וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם כֹּֽה־ אָמַ֣ר יְהוָ֗ה אֲשֶׁ֨ר לַמָּ֤וֶת לַמָּ֙וֶת֙ וַאֲשֶׁ֤ר לַחֶ֙רֶב֙ לַחֶ֔רֶב וַאֲשֶׁ֤ר לָֽרָעָב֙ לָֽרָעָ֔ב וַאֲשֶׁ֥ר לַשְּׁבִ֖י לַשֶּֽׁבִי׃ (ד) וּפָקַדְתִּ֨י עֲלֵיהֶ֜ם אֲנִ֛י לְהוֹשִֽׁיעֲךָ֥ וּלְהַצִּילֶ֖ךָ נְאֻם־ יְהוָֽה׃ (כב) וְהִצַּלְתִּ֖יךָ מִיַּ֣ד רָעִ֑ים וּפְדִתִ֖יךָ מִכַּ֥ף עָרִצִֽים׃ (כג) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כד) לֹֽא־ תִקַּ֥ח לְךָ֖ אִשָּׁ֑ה וְלֹֽא־ יִהְי֤וּ לְךָ֙ בָּנִ֣ים וּבָנ֔וֹת בַּמָּק֖וֹם הַזֶּֽה׃ (כה) כִּי־ כֹ֣ה ׀ אָמַ֣ר יְהוָ֗ה עַל־ הַבָּנִים֙ וְעַל־ הַבָּנ֔וֹת הַיִּלּוֹדִ֖ים בַּמָּק֣וֹם הַזֶּ֑ה וְעַֽל־ אִמֹּתָ֞ם הַיֹּלְד֣וֹת אוֹתָ֗ם וְעַל־ אֲבוֹתָ֛ם הַמּוֹלִדִ֥ים אוֹתָ֖ם בָּאָ֥רֶץ הַזֹּֽאת׃ (כו) מְמוֹתֵ֨י תַחֲלֻאִ֜ים יָמֻ֗תוּ לֹ֤א יִסָּֽפְדוּ֙ וְלֹ֣א יִקָּבֵ֔רוּ לְדֹ֛מֶן עַל־ פְּנֵ֥י הָאֲדָמָ֖ה יִֽהְי֑וּ וּבַחֶ֤רֶב וּבָֽרָעָב֙ יִכְל֔וּ וְהָיְתָ֤ה נִבְלָתָם֙ לְמַאֲכָ֔ל לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֥ת הָאָֽרֶץ׃ (כז) כִּֽי־ כֹ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ תָּבוֹא֙ בֵּ֣ית מַרְזֵ֔חַ וְאַל־ תֵּלֵ֣ךְ לִסְפּ֔וֹד וְאַל־ תָּנֹ֖ד לָהֶ֑ם כִּֽי־ אָסַ֨פְתִּי אֶת־ שְׁלוֹמִ֜י מֵאֵ֨ת הָעָ֤ם־ הַזֶּה֙ נְאֻם־ יְהוָ֔ה אֶת־ הַחֶ֖סֶד וְאֶת־ הָֽרַחֲמִֽים׃ (כח) וּמֵ֨תוּ גְדֹלִ֧ים וּקְטַנִּ֛ים בָּאָ֥רֶץ הַזֹּ֖את לֹ֣א יִקָּבֵ֑רוּ וְלֹֽא־ יִסְפְּד֣וּ לָהֶ֔ם וְלֹ֣א יִתְגֹּדַ֔ד וְלֹ֥א יִקָּרֵ֖חַ לָהֶֽם׃ (כט) וְלֹֽא־ יִפְרְס֥וּ לָהֶ֛ם עַל־ אֵ֖בֶל לְנַחֲמ֣וֹ עַל־ מֵ֑ת וְלֹֽא־ יַשְׁק֤וּ אוֹתָם֙ כּ֣וֹס תַּנְחוּמִ֔ים עַל־ אָבִ֖יו וְעַל־ אִמּֽוֹ׃ (ל) וּבֵית־ מִשְׁתֶּ֥ה לֹא־ תָב֖וֹא לָשֶׁ֣בֶת אוֹתָ֑ם לֶאֱכֹ֖ל וְלִשְׁתּֽוֹת׃ (לא) כִּי֩ כֹ֨ה אָמַ֜ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֨י מַשְׁבִּ֜ית מִן־ הַמָּק֥וֹם הַזֶּ֛ה לְעֵינֵיכֶ֖ם וּבִֽימֵיכֶ֑ם ק֤וֹל שָׂשׂוֹן֙ וְק֣וֹל שִׂמְחָ֔ה ק֥וֹל חָתָ֖ן וְק֥וֹל כַּלָּֽה׃ (לב) וְהָיָ֗ה כִּ֤י תַגִּיד֙ לָעָ֣ם הַזֶּ֔ה אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וְאָמְר֣וּ אֵלֶ֗יךָ עַל־ מֶה֩ דִבֶּ֨ר יְהוָ֤ה עָלֵ֙ינוּ֙ אֵ֣ת כָּל־ הָרָעָ֤ה הַגְּדוֹלָה֙ הַזֹּ֔את וּמֶ֤ה עֲוֺנֵ֙נוּ֙ וּמֶ֣ה חַטָּאתֵ֔נוּ אֲשֶׁ֥ר חָטָ֖אנוּ לַֽיהוָ֥ה אֱלֹהֵֽינוּ׃ (לג) וְאָמַרְתָּ֣ אֲלֵיהֶ֗ם עַל֩ אֲשֶׁר־ עָזְב֨וּ אֲבוֹתֵיכֶ֤ם אוֹתִי֙ נְאֻם־ יְהוָ֔ה וַיֵּלְכ֗וּ אַֽחֲרֵי֙ אֱלֹהִ֣ים אֲחֵרִ֔ים וַיַּעַבְד֖וּם וַיִּשְׁתַּחֲו֣וּ לָהֶ֑ם וְאֹתִ֣י עָזָ֔בוּ וְאֶת־ תּוֹרָתִ֖י לֹ֥א שָׁמָֽרוּ׃ (לד) וְאַתֶּ֛ם הֲרֵעֹתֶ֥ם לַעֲשׂ֖וֹת מֵאֲבֽוֹתֵיכֶ֑ם וְהִנְּכֶ֣ם הֹלְכִ֗ים אִ֚ישׁ אַֽחֲרֵי֙ שְׁרִר֣וּת לִבּֽוֹ־ הָרָ֔ע לְבִלְתִּ֖י שְׁמֹ֥עַ אֵלָֽי׃ (לה) וְהֵטַלְתִּ֣י אֶתְכֶ֗ם מֵעַל֙ הָאָ֣רֶץ הַזֹּ֔את עַל־ הָאָ֕רֶץ אֲשֶׁר֙ לֹ֣א יְדַעְתֶּ֔ם אַתֶּ֖ם וַאֲבֽוֹתֵיכֶ֑ם וַעֲבַדְתֶּם־ שָׁ֞ם אֶת־ אֱלֹהִ֤ים אֲחֵרִים֙ יוֹמָ֣ם וָלַ֔יְלָה אֲשֶׁ֛ר לֹֽא־ אֶתֵּ֥ן לָכֶ֖ם חֲנִינָֽה׃ (לו) לָכֵ֛ן הִנֵּֽה־ יָמִ֥ים בָּאִ֖ים נְאֻם־ יְהוָ֑ה וְלֹֽא־ יֵאָמֵ֥ר עוֹד֙ חַי־ יְהוָ֔ה אֲשֶׁ֧ר הֶעֱלָ֛ה אֶת־ בְּנֵ֥י יִשְׂרָאֵ֖ל מֵאֶ֥רֶץ מִצְרָֽיִם׃ (לז) כִּ֣י אִם־ חַי־ יְהוָ֗ה אֲשֶׁ֨ר הֶעֱלָ֜ה אֶת־ בְּנֵ֤י יִשְׂרָאֵל֙ מֵאֶ֣רֶץ צָפ֔וֹן וּמִכֹּל֙ הָֽאֲרָצ֔וֹת אֲשֶׁ֥ר הִדִּיחָ֖ם שָׁ֑מָּה וַהֲשִֽׁבֹתִים֙ עַל־ אַדְמָתָ֔ם אֲשֶׁ֥ר נָתַ֖תִּי לַאֲבוֹתָֽם׃ (לח) הִנְנִ֨י שֹׁלֵ֜חַ (לדוגים) לְדַיָּגִ֥ים רַבִּ֛ים נְאֻם־ יְהוָ֖ה וְדִיג֑וּם וְאַֽחֲרֵי־ כֵ֗ן אֶשְׁלַח֙ לְרַבִּ֣ים צַיָּדִ֔ים וְצָד֞וּם מֵעַ֤ל כָּל־ הַר֙ וּמֵעַ֣ל כָּל־ גִּבְעָ֔ה וּמִנְּקִיקֵ֖י הַסְּלָעִֽים׃ (לט) כִּ֤י עֵינַי֙ עַל־ כָּל־ דַּרְכֵיהֶ֔ם לֹ֥א נִסְתְּר֖וּ מִלְּפָנָ֑י וְלֹֽא־ נִצְפַּ֥ן עֲוֺנָ֖ם מִנֶּ֥גֶד עֵינָֽי׃ (מ) וְשִׁלַּמְתִּ֣י רִֽאשׁוֹנָ֗ה מִשְׁנֵ֤ה עֲוֺנָם֙ וְחַטָּאתָ֔ם עַ֖ל חַלְּלָ֣ם אֶת־ אַרְצִ֑י בְּנִבְלַ֤ת שִׁקּֽוּצֵיהֶם֙ וְתוֹעֲב֣וֹתֵיהֶ֔ם מָלְא֖וּ אֶת־ נַחֲלָתִֽי׃ (מא) יְהוָ֞ה עֻזִּ֧י וּמָעֻזִּ֛י וּמְנוּסִ֖י בְּי֣וֹם צָרָ֑ה אֵלֶ֗יךָ גּוֹיִ֤ם יָבֹ֙אוּ֙ מֵֽאַפְסֵי־ אָ֔רֶץ וְיֹאמְר֗וּ אַךְ־ שֶׁ֙קֶר֙ נָחֲל֣וּ אֲבוֹתֵ֔ינוּ הֶ֖בֶל וְאֵֽין־ בָּ֥ם מוֹעִֽיל׃ (מב) הֲיַעֲשֶׂה־ לּ֥וֹ אָדָ֖ם אֱלֹהִ֑ים וְהֵ֖מָּה לֹ֥א אֱלֹהִֽים׃ (מג) לָכֵן֙ הִנְנִ֣י מֽוֹדִיעָ֔ם בַּפַּ֣עַם הַזֹּ֔את אוֹדִיעֵ֥ם אֶת־ יָדִ֖י וְאֶת־ גְּבֽוּרָתִ֑י וְיָדְע֖וּ כִּֽי־ שְׁמִ֥י יְהוָֽה׃ (מד) חַטַּ֣את יְהוּדָ֗ה כְּתוּבָ֛ה בְּעֵ֥ט בַּרְזֶ֖ל בְּצִפֹּ֣רֶן שָׁמִ֑יר חֲרוּשָׁה֙ עַל־ ל֣וּחַ לִבָּ֔ם וּלְקַרְנ֖וֹת מִזְבְּחוֹתֵיכֶֽם׃ (מה) כִּזְכֹּ֤ר בְּנֵיהֶם֙ מִזְבְּחוֹתָ֔ם וַאֲשֵׁרֵיהֶ֖ם עַל־ עֵ֣ץ רַֽעֲנָ֑ן עַ֖ל גְּבָע֥וֹת הַגְּבֹהֽוֹת׃ (מו) הֲרָרִי֙ בַּשָּׂדֶ֔ה חֵילְךָ֥ כָל־ אוֹצְרוֹתֶ֖יךָ לָבַ֣ז אֶתֵּ֑ן בָּמֹתֶ֕יךָ בְּחַטָּ֖את בְּכָל־ גְּבוּלֶֽיךָ׃ (מז) וְשָׁמַטְתָּ֗ה וּבְךָ֙ מִנַּחֲלָֽתְךָ֙ אֲשֶׁ֣ר נָתַ֣תִּי לָ֔ךְ וְהַעֲבַדְתִּ֙יךָ֙ אֶת־ אֹ֣יְבֶ֔יךָ בָּאָ֖רֶץ אֲשֶׁ֣ר לֹֽא־ יָדָ֑עְתָּ כִּֽי־ אֵ֛שׁ קְדַחְתֶּ֥ם בְּאַפִּ֖י עַד־ עוֹלָ֥ם תּוּקָֽד׃ (מח) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אָר֤וּר הַגֶּ֙בֶר֙ אֲשֶׁ֣ר יִבְטַ֣ח בָּֽאָדָ֔ם וְשָׂ֥ם בָּשָׂ֖ר זְרֹע֑וֹ וּמִן־ יְהוָ֖ה יָס֥וּר לִבּֽוֹ׃ (מט) וְהָיָה֙ כְּעַרְעָ֣ר בָּֽעֲרָבָ֔ה וְלֹ֥א יִרְאֶ֖ה כִּי־ יָב֣וֹא ט֑וֹב וְשָׁכַ֤ן חֲרֵרִים֙ בַּמִּדְבָּ֔ר אֶ֥רֶץ מְלֵחָ֖ה וְלֹ֥א תֵשֵֽׁב׃
