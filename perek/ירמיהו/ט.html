ירמיהו פרק ט לפי החלוקה לסדרים

פרק ט בספר ירמיהו (25 פסוקים) נמצא בירמיהו סדר ה (פסוקים לו–נז) ובירמיהו סדר ו (פסוקים א–ג).

ירמיהו סדר ה

(א) כִּ֣י אִֽם־ אֶת־ הַדָּבָ֣ר הַ֠זֶּה צִוִּ֨יתִי אוֹתָ֤ם לֵאמֹר֙ שִׁמְע֣וּ בְקוֹלִ֔י וְהָיִ֤יתִי לָכֶם֙ לֵֽאלֹהִ֔ים וְאַתֶּ֖ם תִּֽהְיוּ־ לִ֣י לְעָ֑ם וַהֲלַכְתֶּ֗ם בְּכָל־ הַדֶּ֙רֶךְ֙ אֲשֶׁ֣ר אֲצַוֶּ֣ה אֶתְכֶ֔ם לְמַ֖עַן יִיטַ֥ב לָכֶֽם׃ (ב) וְלֹ֤א שָֽׁמְעוּ֙ וְלֹֽא־ הִטּ֣וּ אֶת־ אָזְנָ֔ם וַיֵּֽלְכוּ֙ בְּמֹ֣עֵצ֔וֹת בִּשְׁרִר֖וּת לִבָּ֣ם הָרָ֑ע וַיִּהְי֥וּ לְאָח֖וֹר וְלֹ֥א לְפָנִֽים׃ (ג) לְמִן־ הַיּ֗וֹם אֲשֶׁ֨ר יָצְא֤וּ אֲבֽוֹתֵיכֶם֙ מֵאֶ֣רֶץ מִצְרַ֔יִם עַ֖ד הַיּ֣וֹם הַזֶּ֑ה וָאֶשְׁלַ֤ח אֲלֵיכֶם֙ אֶת־ כָּל־ עֲבָדַ֣י הַנְּבִיאִ֔ים י֖וֹם הַשְׁכֵּ֥ם וְשָׁלֹֽחַ׃ (ד) וְל֤וֹא שָׁמְעוּ֙ אֵלַ֔י וְלֹ֥א הִטּ֖וּ אֶת־ אָזְנָ֑ם וַיַּקְשׁוּ֙ אֶת־ עָרְפָּ֔ם הֵרֵ֖עוּ מֵאֲבוֹתָֽם׃ (ה) וְדִבַּרְתָּ֤ אֲלֵיהֶם֙ אֶת־ כָּל־ הַדְּבָרִ֣ים הָאֵ֔לֶּה וְלֹ֥א יִשְׁמְע֖וּ אֵלֶ֑יךָ וְקָרָ֥אתָ אֲלֵיהֶ֖ם וְלֹ֥א יַעֲנֽוּכָה׃ (ו) וְאָמַרְתָּ֣ אֲלֵיהֶ֗ם זֶ֤ה הַגּוֹי֙ אֲשֶׁ֣ר לֽוֹא־ שָׁמְע֗וּ בְּקוֹל֙ יְהוָ֣ה אֱלֹהָ֔יו וְלֹ֥א לָקְח֖וּ מוּסָ֑ר אָֽבְדָה֙ הָֽאֱמוּנָ֔ה וְנִכְרְתָ֖ה מִפִּיהֶֽם׃ (ז) גָּזִּ֤י נִזְרֵךְ֙ וְֽהַשְׁלִ֔יכִי וּשְׂאִ֥י עַל־ שְׁפָיִ֖ם קִינָ֑ה כִּ֚י מָאַ֣ס יְהוָ֔ה וַיִּטֹּ֖שׁ אֶת־ דּ֥וֹר עֶבְרָתֽוֹ׃ (ח) כִּֽי־ עָשׂ֨וּ בְנֵי־ יְהוּדָ֥ה הָרַ֛ע בְּעֵינַ֖י נְאֻום־ יְהוָ֑ה שָׂ֣מוּ שִׁקּֽוּצֵיהֶ֗ם בַּבַּ֛יִת אֲשֶׁר־ נִקְרָא־ שְׁמִ֥י עָלָ֖יו לְטַמְּאֽוֹ׃ (ט) וּבָנ֞וּ בָּמ֣וֹת הַתֹּ֗פֶת אֲשֶׁר֙ בְּגֵ֣יא בֶן־ הִנֹּ֔ם לִשְׂרֹ֛ף אֶת־ בְּנֵיהֶ֥ם וְאֶת־ בְּנֹתֵיהֶ֖ם בָּאֵ֑שׁ אֲשֶׁר֙ לֹ֣א צִוִּ֔יתִי וְלֹ֥א עָלְתָ֖ה עַל־ לִבִּֽי׃ (י) לָכֵ֞ן הִנֵּֽה־ יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וְלֹא־ יֵאָמֵ֨ר ע֤וֹד הַתֹּ֙פֶת֙ וְגֵ֣יא בֶן־ הִנֹּ֔ם כִּ֖י אִם־ גֵּ֣יא הַהֲרֵגָ֑ה וְקָבְר֥וּ בְתֹ֖פֶת מֵאֵ֥ין מָקֽוֹם׃ (יא) וְֽהָ֨יְתָ֜ה נִבְלַ֨ת הָעָ֤ם הַזֶּה֙ לְמַֽאֲכָ֔ל לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֣ת הָאָ֑רֶץ וְאֵ֖ין מַחֲרִֽיד׃ (יב) וְהִשְׁבַּתִּ֣י ׀ מֵעָרֵ֣י יְהוּדָ֗ה וּמֵֽחֻצוֹת֙ יְר֣וּשָׁלִַ֔ם ק֤וֹל שָׂשׂוֹן֙ וְק֣וֹל שִׂמְחָ֔ה ק֥וֹל חָתָ֖ן וְק֣וֹל כַּלָּ֑ה כִּ֥י לְחָרְבָּ֖ה תִּהְיֶ֥ה הָאָֽרֶץ׃ (יג) בָּעֵ֣ת הַהִ֣יא נְאֻם־ יְהוָ֡ה (ויציאו) יוֹצִ֣יאוּ אֶת־ עַצְמ֣וֹת מַלְכֵֽי־ יְהוּדָ֣ה וְאֶת־ עַצְמוֹת־ שָׂרָיו֩ וְאֶת־ עַצְמ֨וֹת הַכֹּהֲנִ֜ים וְאֵ֣ת ׀ עַצְמ֣וֹת הַנְּבִיאִ֗ים וְאֵ֛ת עַצְמ֥וֹת יוֹשְׁבֵֽי־ יְרוּשָׁלִָ֖ם מִקִּבְרֵיהֶֽם׃ (יד) וּשְׁטָחוּם֩ לַשֶּׁ֨מֶשׁ וְלַיָּרֵ֜חַ וּלְכֹ֣ל ׀ צְבָ֣א הַשָּׁמַ֗יִם אֲשֶׁ֨ר אֲהֵב֜וּם וַאֲשֶׁ֤ר עֲבָדוּם֙ וַֽאֲשֶׁר֙ הָלְכ֣וּ אַֽחֲרֵיהֶ֔ם וַאֲשֶׁ֣ר דְּרָשׁ֔וּם וַאֲשֶׁ֥ר הִֽשְׁתַּחֲו֖וּ לָהֶ֑ם לֹ֤א יֵאָֽסְפוּ֙ וְלֹ֣א יִקָּבֵ֔רוּ לְדֹ֛מֶן עַל־ פְּנֵ֥י הָאֲדָמָ֖ה יִֽהְיֽוּ׃ (טו) וְנִבְחַ֥ר מָ֙וֶת֙ מֵֽחַיִּ֔ים לְכֹ֗ל הַשְּׁאֵרִית֙ הַנִּשְׁאָרִ֔ים מִן־ הַמִּשְׁפָּחָ֥ה הָֽרָעָ֖ה הַזֹּ֑את בְּכָל־ הַמְּקֹמ֤וֹת הַנִּשְׁאָרִים֙ אֲשֶׁ֣ר הִדַּחְתִּ֣ים שָׁ֔ם נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (טז) וְאָמַרְתָּ֣ אֲלֵיהֶ֗ם כֹּ֚ה אָמַ֣ר יְהוָ֔ה הֲיִפְּל֖וּ וְלֹ֣א יָק֑וּמוּ אִם־ יָשׁ֖וּב וְלֹ֥א יָשֽׁוּב׃ (יז) מַדּ֨וּעַ שׁוֹבְבָ֜ה הָעָ֥ם הַזֶּ֛ה יְרוּשָׁלִַ֖ם מְשֻׁבָ֣ה נִצַּ֑חַת הֶחֱזִ֙יקוּ֙ בַּתַּרְמִ֔ית מֵאֲנ֖וּ לָשֽׁוּב׃ (יח) הִקְשַׁ֤בְתִּי וָֽאֶשְׁמָע֙ לוֹא־ כֵ֣ן יְדַבֵּ֔רוּ אֵ֣ין אִ֗ישׁ נִחָם֙ עַל־ רָ֣עָת֔וֹ לֵאמֹ֖ר מֶ֣ה עָשִׂ֑יתִי כֻּלֹּ֗ה שָׁ֚ב (במרצותם) בִּמְר֣וּצָתָ֔ם כְּס֥וּס שׁוֹטֵ֖ף בַּמִּלְחָמָֽה׃ (יט) גַּם־ חֲסִידָ֣ה בַשָּׁמַ֗יִם יָֽדְעָה֙ מֽוֹעֲדֶ֔יהָ וְתֹ֤ר (וסוס) וְסִיס֙ וְעָג֔וּר שָׁמְר֖וּ אֶת־ עֵ֣ת בֹּאָ֑נָה וְעַמִּ֕י לֹ֣א יָֽדְע֔וּ אֵ֖ת מִשְׁפַּ֥ט יְהוָֽה׃ (כ) אֵיכָ֤ה תֹֽאמְרוּ֙ חֲכָמִ֣ים אֲנַ֔חְנוּ וְתוֹרַ֥ת יְהוָ֖ה אִתָּ֑נוּ אָכֵן֙ הִנֵּ֣ה לַשֶּׁ֣קֶר עָשָׂ֔ה עֵ֖ט שֶׁ֥קֶר סֹפְרִֽים׃ (כא) הֹבִ֣ישׁוּ חֲכָמִ֔ים חַ֖תּוּ וַיִּלָּכֵ֑דוּ הִנֵּ֤ה בִדְבַר־ יְהוָה֙ מָאָ֔סוּ וְחָכְמַֽת־ מֶ֖ה לָהֶֽם׃ (כב) לָכֵן֩ אֶתֵּ֨ן אֶת־ נְשֵׁיהֶ֜ם לַאֲחֵרִ֗ים שְׂדֽוֹתֵיהֶם֙ לְי֣וֹרְשִׁ֔ים כִּ֤י מִקָּטֹן֙ וְעַד־ גָּד֔וֹל כֻּלֹּ֖ה בֹּצֵ֣עַ בָּ֑צַע מִנָּבִיא֙ וְעַד־ כֹּהֵ֔ן כֻּלֹּ֖ה עֹ֥שֶׂה שָּֽׁקֶר׃ (כג) וַיְרַפּ֞וּ אֶת־ שֶׁ֤בֶר בַּת־ עַמִּי֙ עַל־ נְקַלָּ֔ה לֵאמֹ֖ר שָׁל֣וֹם ׀ שָׁל֑וֹם וְאֵ֖ין שָׁלֽוֹם׃ (כד) הֹבִ֕שׁוּ כִּ֥י תוֹעֵבָ֖ה עָשׂ֑וּ גַּם־ בּ֣וֹשׁ לֹֽא־ יֵבֹ֗שׁוּ וְהִכָּלֵם֙ לֹ֣א יָדָ֔עוּ לָכֵ֞ן יִפְּל֣וּ בַנֹּפְלִ֗ים בְּעֵ֧ת פְּקֻדָּתָ֛ם יִכָּשְׁל֖וּ אָמַ֥ר יְהוָֽה׃ (כה) אָסֹ֥ף אֲסִיפֵ֖ם נְאֻם־ יְהֹוָ֑ה אֵין֩ עֲנָבִ֨ים בַּגֶּ֜פֶן וְאֵ֧ין תְּאֵנִ֣ים בַּתְּאֵנָ֗ה וְהֶֽעָלֶה֙ נָבֵ֔ל וָאֶתֵּ֥ן לָהֶ֖ם יַעַבְרֽוּם׃ (כו) עַל־ מָה֙ אֲנַ֣חְנוּ יֹֽשְׁבִ֔ים הֵֽאָסְפ֗וּ וְנָב֛וֹא אֶל־ עָרֵ֥י הַמִּבְצָ֖ר וְנִדְּמָה־ שָּׁ֑ם כִּי֩ יְהוָ֨ה אֱלֹהֵ֤ינוּ הֲדִמָּ֙נוּ֙ וַיַּשְׁקֵ֣נוּ מֵי־ רֹ֔אשׁ כִּ֥י חָטָ֖אנוּ לַיהוָֽה׃ (כז) קַוֵּ֥ה לְשָׁל֖וֹם וְאֵ֣ין ט֑וֹב לְעֵ֥ת מַרְפֵּ֖ה וְהִנֵּ֥ה בְעָתָֽה׃ (כח) מִדָּ֤ן נִשְׁמַע֙ נַחְרַ֣ת סוּסָ֗יו מִקּוֹל֙ מִצְהֲל֣וֹת אַבִּירָ֔יו רָעֲשָׁ֖ה כָּל־ הָאָ֑רֶץ וַיָּב֗וֹאוּ וַיֹּֽאכְלוּ֙ אֶ֣רֶץ וּמְלוֹאָ֔הּ עִ֖יר וְיֹ֥שְׁבֵי בָֽהּ׃ (כט) כִּי֩ הִנְנִ֨י מְשַׁלֵּ֜חַ בָּכֶ֗ם נְחָשִׁים֙ צִפְעֹנִ֔ים אֲשֶׁ֥ר אֵין־ לָהֶ֖ם לָ֑חַשׁ וְנִשְּׁכ֥וּ אֶתְכֶ֖ם נְאֻם־ יְהוָֽה׃ (ל) מַבְלִ֥יגִיתִ֖י עֲלֵ֣י יָג֑וֹן עָלַ֖י לִבִּ֥י דַוָּֽי׃ (לא) הִנֵּה־ ק֞וֹל שַֽׁוְעַ֣ת בַּת־ עַמִּ֗י מֵאֶ֙רֶץ֙ מַרְחַקִּ֔ים הַֽיהוָה֙ אֵ֣ין בְּצִיּ֔וֹן אִם־ מַלְכָּ֖הּ אֵ֣ין בָּ֑הּ מַדּ֗וּעַ הִכְעִס֛וּנִי בִּפְסִלֵיהֶ֖ם בְּהַבְלֵ֥י נֵכָֽר׃ (לב) עָבַ֥ר קָצִ֖יר כָּ֣לָה קָ֑יִץ וַאֲנַ֖חְנוּ ל֥וֹא נוֹשָֽׁעְנוּ׃ (לג) עַל־ שֶׁ֥בֶר בַּת־ עַמִּ֖י הָשְׁבָּ֑רְתִּי קָדַ֕רְתִּי שַׁמָּ֖ה הֶחֱזִקָֽתְנִי׃ (לד) הַצֳרִי֙ אֵ֣ין בְּגִלְעָ֔ד אִם־ רֹפֵ֖א אֵ֣ין שָׁ֑ם כִּ֗י מַדּ֙וּעַ֙ לֹ֣א עָֽלְתָ֔ה אֲרֻכַ֖ת בַּת־ עַמִּֽי׃ (לה) מִֽי־ יִתֵּ֤ן רֹאשִׁי֙ מַ֔יִם וְעֵינִ֖י מְק֣וֹר דִּמְעָ֑ה וְאֶבְכֶּה֙ יוֹמָ֣ם וָלַ֔יְלָה אֵ֖ת חַֽלְלֵ֥י בַת־ עַמִּֽי׃ תחילת פרק ט (לו) מִֽי־ יִתְּנֵ֣נִי בַמִּדְבָּ֗ר גַם־ אַחֶ֣יךָ וּבֵית־ אָבִ֗יךָ גַּם־ הֵ֙מָּה֙ בָּ֣גְדוּ בָ֔ךְ גַּם־ הֵ֛מָּה קָרְא֥וּ אַחֲרֶ֖יךָ מָלֵ֑א אַל־ תַּאֲמֵ֣ן בָּ֔ם כִּֽי־ יְדַבְּר֥וּ אֵלֶ֖יךָ טוֹבֽוֹת׃ (נח) עָזַ֙בְתִּי֙ אֶת־ בֵּיתִ֔י נָטַ֖שְׁתִּי אֶת־ נַחֲלָתִ֑י נָתַ֛תִּי אֶת־ יְדִד֥וּת נַפְשִׁ֖י בְּכַ֥ף אֹיְבֶֽיהָ׃ (נט) הָיְתָה־ לִּ֥י נַחֲלָתִ֖י כְּאַרְיֵ֣ה בַיָּ֑עַר נָתְנָ֥ה עָלַ֛י בְּקוֹלָ֖הּ עַל־ כֵּ֥ן שְׂנֵאתִֽיהָ׃ (ס) הַעַ֨יִט צָב֤וּעַ נַחֲלָתִי֙ לִ֔י הַעַ֖יִט סָבִ֣יב עָלֶ֑יהָ לְכ֗וּ אִסְפ֛וּ כָּל־ חַיַּ֥ת הַשָּׂדֶ֖ה הֵתָ֥יוּ לְאָכְלָֽה׃ (סא) רֹעִ֤ים רַבִּים֙ שִֽׁחֲת֣וּ כַרְמִ֔י בֹּסְס֖וּ אֶת־ חֶלְקָתִ֑י נָֽתְנ֛וּ אֶת־ חֶלְקַ֥ת חֶמְדָּתִ֖י לְמִדְבַּ֥ר שְׁמָמָֽה׃ (סב) שָׂמָהּ֙ לִשְׁמָמָ֔ה אָבְלָ֥ה עָלַ֖י שְׁמֵמָ֑ה נָשַׁ֙מָּה֙ כָּל־ הָאָ֔רֶץ כִּ֛י אֵ֥ין אִ֖ישׁ שָׂ֥ם עַל־ לֵֽב׃ (סג) עַֽל־ כָּל־ שְׁפָיִ֣ם בַּמִּדְבָּ֗ר בָּ֚אוּ שֹֽׁדְדִ֔ים כִּ֣י חֶ֤רֶב לַֽיהוָה֙ אֹֽכְלָ֔ה מִקְצֵה־ אֶ֖רֶץ וְעַד־ קְצֵ֣ה הָאָ֑רֶץ אֵ֥ין שָׁל֖וֹם לְכָל־ בָּשָֽׂר׃ (סד) זָרְע֤וּ חִטִּים֙ וְקֹצִ֣ים קָצָ֔רוּ נֶחְל֖וּ לֹ֣א יוֹעִ֑לוּ וּבֹ֙שׁוּ֙ מִתְּבוּאֹ֣תֵיכֶ֔ם מֵחֲר֖וֹן אַף־ יְהוָֽה׃ (סה) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה עַל־ כָּל־ שְׁכֵנַי֙ הָֽרָעִ֔ים הַנֹּֽגְעִים֙ בַּֽנַּחֲלָ֔ה אֲשֶׁר־ הִנְחַ֥לְתִּי אֶת־ עַמִּ֖י אֶת־ יִשְׂרָאֵ֑ל הִנְנִ֤י נֹֽתְשָׁם֙ מֵעַ֣ל אַדְמָתָ֔ם וְאֶת־ בֵּ֥ית יְהוּדָ֖ה אֶתּ֥וֹשׁ מִתּוֹכָֽם׃
