ירמיהו פרק לה לפי החלוקה לסדרים

פרק לה בספר ירמיהו (19 פסוקים) נמצא בירמיהו סדר כ (פסוקים לד–מב) ובירמיהו סדר כא (פסוקים א–י).

ירמיהו סדר כ

(א) בַּיָּמִ֤ים הָהֵם֙ תִּוָּשַׁ֣ע יְהוּדָ֔ה וִירוּשָׁלִַ֖ם תִּשְׁכּ֣וֹן לָבֶ֑טַח וְזֶ֥ה אֲשֶׁר־ יִקְרָא־ לָ֖הּ יְהוָ֥ה ׀ צִדְקֵֽנוּ׃ (ב) כִּי־ כֹ֖ה אָמַ֣ר יְהוָ֑ה לֹֽא־ יִכָּרֵ֣ת לְדָוִ֔ד אִ֕ישׁ יֹשֵׁ֖ב עַל־ כִּסֵּ֥א בֵֽית־ יִשְׂרָאֵֽל׃ (ג) וְלַכֹּהֲנִים֙ הַלְוִיִּ֔ם לֹֽא־ יִכָּרֵ֥ת אִ֖ישׁ מִלְּפָנָ֑י מַעֲלֶ֨ה עוֹלָ֜ה וּמַקְטִ֥יר מִנְחָ֛ה וְעֹ֥שֶׂה־ זֶּ֖בַח כָּל־ הַיָּמִֽים׃ (ד) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶֽל־ יִרְמְיָ֖הוּ לֵאמֽוֹר׃ (ה) כֹּ֚ה אָמַ֣ר יְהוָ֔ה אִם־ תָּפֵ֙רוּ֙ אֶת־ בְּרִיתִ֣י הַיּ֔וֹם וְאֶת־ בְּרִיתִ֖י הַלָּ֑יְלָה וּלְבִלְתִּ֛י הֱי֥וֹת יֽוֹמָם־ וָלַ֖יְלָה בְּעִתָּֽם׃ (ו) גַּם־ בְּרִיתִ֤י תֻפַר֙ אֶת־ דָּוִ֣ד עַבְדִּ֔י מִהְיֽוֹת־ ל֥וֹ בֵ֖ן מֹלֵ֣ךְ עַל־ כִּסְא֑וֹ וְאֶת־ הַלְוִיִּ֥ם הַכֹּהֲנִ֖ים מְשָׁרְתָֽי׃ (ז) אֲשֶׁ֤ר לֹֽא־ יִסָּפֵר֙ צְבָ֣א הַשָּׁמַ֔יִם וְלֹ֥א יִמַּ֖ד ח֣וֹל הַיָּ֑ם כֵּ֣ן אַרְבֶּ֗ה אֶת־ זֶ֙רַע֙ דָּוִ֣ד עַבְדִּ֔י וְאֶת־ הַלְוִיִּ֖ם מְשָׁרְתֵ֥י אֹתִֽי׃ (ח) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶֽל־ יִרְמְיָ֖הוּ לֵאמֹֽר׃ (ט) הֲל֣וֹא רָאִ֗יתָ מָֽה־ הָעָ֤ם הַזֶּה֙ דִּבְּר֣וּ לֵאמֹ֔ר שְׁתֵּ֣י הַמִּשְׁפָּח֗וֹת אֲשֶׁ֨ר בָּחַ֧ר יְהוָ֛ה בָּהֶ֖ם וַיִּמְאָסֵ֑ם וְאֶת־ עַמִּי֙ יִנְאָצ֔וּן מִֽהְי֥וֹת ע֖וֹד גּ֥וֹי לִפְנֵיהֶֽם׃ (י) כֹּ֚ה אָמַ֣ר יְהוָ֔ה אִם־ לֹ֥א בְרִיתִ֖י יוֹמָ֣ם וָלָ֑יְלָה חֻקּ֛וֹת שָׁמַ֥יִם וָאָ֖רֶץ לֹא־ שָֽׂמְתִּי׃ (יא) גַּם־ זֶ֣רַע יַעֲקוֹב֩ וְדָוִ֨ד עַבְדִּ֜י אֶמְאַ֗ס מִקַּ֤חַת מִזַּרְעוֹ֙ מֹֽשְׁלִ֔ים אֶל־ זֶ֥רַע אַבְרָהָ֖ם יִשְׂחָ֣ק וְיַעֲקֹ֑ב כִּֽי־ (אשוב) אָשִׁ֥יב אֶת־ שְׁבוּתָ֖ם וְרִחַמְתִּֽים׃ (יב) הַדָּבָ֛ר אֲשֶׁר־ הָיָ֥ה אֶֽל־ יִרְמְיָ֖הוּ מֵאֵ֣ת יְהוָ֑ה וּנְבוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֣ל ׀ וְכָל־ חֵיל֡וֹ וְכָל־ מַמְלְכ֣וֹת אֶרֶץ֩ מֶמְשֶׁ֨לֶת יָד֜וֹ וְכָל־ הָעַמִּ֗ים נִלְחָמִ֧ים עַל־ יְרוּשָׁלִַ֛ם וְעַל־ כָּל־ עָרֶ֖יהָ לֵאמֹֽר׃ (יג) כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הָלֹךְ֙ וְאָ֣מַרְתָּ֔ אֶל־ צִדְקִיָּ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה וְאָמַרְתָּ֣ אֵלָ֗יו כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֨י נֹתֵ֜ן אֶת־ הָעִ֤יר הַזֹּאת֙ בְּיַ֣ד מֶֽלֶךְ־ בָּבֶ֔ל וּשְׂרָפָ֖הּ בָּאֵֽשׁ׃ (יד) וְאַתָּ֗ה לֹ֚א תִמָּלֵט֙ מִיָּד֔וֹ כִּ֚י תָּפֹ֣שׂ תִּתָּפֵ֔שׂ וּבְיָד֖וֹ תִּנָּתֵ֑ן וְֽ֠עֵינֶיךָ אֶת־ עֵינֵ֨י מֶֽלֶךְ־ בָּבֶ֜ל תִּרְאֶ֗ינָה וּפִ֛יהוּ אֶת־ פִּ֥יךָ יְדַבֵּ֖ר וּבָבֶ֥ל תָּבֽוֹא׃ (טו) אַ֚ךְ שְׁמַ֣ע דְּבַר־ יְהוָ֔ה צִדְקִיָּ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה כֹּֽה־ אָמַ֤ר יְהוָה֙ עָלֶ֔יךָ לֹ֥א תָמ֖וּת בֶּחָֽרֶב׃ (טז) בְּשָׁל֣וֹם תָּמ֗וּת וּֽכְמִשְׂרְפ֣וֹת אֲ֠בוֹתֶיךָ הַמְּלָכִ֨ים הָרִֽאשֹׁנִ֜ים אֲשֶׁר־ הָי֣וּ לְפָנֶ֗יךָ כֵּ֚ן יִשְׂרְפוּ־ לָ֔ךְ וְה֥וֹי אָד֖וֹן יִסְפְּדוּ־ לָ֑ךְ כִּֽי־ דָבָ֥ר אֲנִֽי־ דִבַּ֖רְתִּי נְאֻם־ יְהוָֽה׃ (יז) וַיְדַבֵּר֙ יִרְמְיָ֣הוּ הַנָּבִ֔יא אֶל־ צִדְקִיָּ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה אֵ֛ת כָּל־ הַדְּבָרִ֥ים הָאֵ֖לֶּה בִּירוּשָׁלִָֽם׃ (יח) וְחֵ֣יל מֶֽלֶךְ־ בָּבֶ֗ל נִלְחָמִים֙ עַל־ יְר֣וּשָׁלִַ֔ם וְעַ֛ל כָּל־ עָרֵ֥י יְהוּדָ֖ה הַנּֽוֹתָר֑וֹת אֶל־ לָכִישׁ֙ וְאֶל־ עֲזֵקָ֔ה כִּ֣י הֵ֗נָּה נִשְׁאֲר֛וּ בְּעָרֵ֥י יְהוּדָ֖ה עָרֵ֥י מִבְצָֽר׃ (יט) הַדָּבָ֛ר אֲשֶׁר־ הָיָ֥ה אֶֽל־ יִרְמְיָ֖הוּ מֵאֵ֣ת יְהוָ֑ה אַחֲרֵ֡י כְּרֹת֩ הַמֶּ֨לֶךְ צִדְקִיָּ֜הוּ בְּרִ֗ית אֶת־ כָּל־ הָעָם֙ אֲשֶׁ֣ר בִּירֽוּשָׁלִַ֔ם לִקְרֹ֥א לָהֶ֖ם דְּרֽוֹר׃ (כ) לְ֠שַׁלַּח אִ֣ישׁ אֶת־ עַבְדּ֞וֹ וְאִ֧ישׁ אֶת־ שִׁפְחָת֛וֹ הָעִבְרִ֥י וְהָעִבְרִיָּ֖ה חָפְשִׁ֑ים לְבִלְתִּ֧י עֲבָד־ בָּ֛ם בִּיהוּדִ֥י אָחִ֖יהוּ אִֽישׁ׃ (כא) וַיִּשְׁמְעוּ֩ כָל־ הַשָּׂרִ֨ים וְכָל־ הָעָ֜ם אֲשֶׁר־ בָּ֣אוּ בַבְּרִ֗ית לְ֠שַׁלַּח אִ֣ישׁ אֶת־ עַבְדּ֞וֹ וְאִ֤ישׁ אֶת־ שִׁפְחָתוֹ֙ חָפְשִׁ֔ים לְבִלְתִּ֥י עֲבָד־ בָּ֖ם ע֑וֹד וַֽיִּשְׁמְע֖וּ וַיְשַׁלֵּֽחוּ׃ (כב) וַיָּשׁ֙וּבוּ֙ אַחֲרֵי־ כֵ֔ן וַיָּשִׁ֗בוּ אֶת־ הָֽעֲבָדִים֙ וְאֶת־ הַשְּׁפָח֔וֹת אֲשֶׁ֥ר שִׁלְּח֖וּ חָפְשִׁ֑ים (ויכבישום) וַֽיִּכְבְּשׁ֔וּם לַעֲבָדִ֖ים וְלִשְׁפָחֽוֹת׃ (כג) וַיְהִ֤י דְבַר־ יְהוָה֙ אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (כד) כֹּֽה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל אָנֹכִ֗י כָּרַ֤תִּֽי בְרִית֙ אֶת־ אֲב֣וֹתֵיכֶ֔ם בְּי֨וֹם הוֹצִאִ֤י אוֹתָם֙ מֵאֶ֣רֶץ מִצְרַ֔יִם מִבֵּ֥ית עֲבָדִ֖ים לֵאמֹֽר׃ (כה) מִקֵּ֣ץ שֶׁ֣בַע שָׁנִ֡ים תְּֽשַׁלְּח֡וּ אִישׁ֩ אֶת־ אָחִ֨יו הָעִבְרִ֜י אֲשֶֽׁר־ יִמָּכֵ֣ר לְךָ֗ וַעֲבָֽדְךָ֙ שֵׁ֣שׁ שָׁנִ֔ים וְשִׁלַּחְתּ֥וֹ חָפְשִׁ֖י מֵֽעִמָּ֑ךְ וְלֹֽא־ שָׁמְע֤וּ אֲבֽוֹתֵיכֶם֙ אֵלַ֔י וְלֹ֥א הִטּ֖וּ אֶת־ אָזְנָֽם׃ (כו) וַתָּשֻׁ֨בוּ אַתֶּ֜ם הַיּ֗וֹם וַתַּעֲשׂ֤וּ אֶת־ הַיָּשָׁר֙ בְּעֵינַ֔י לִקְרֹ֥א דְר֖וֹר אִ֣ישׁ לְרֵעֵ֑הוּ וַתִּכְרְת֤וּ בְרִית֙ לְפָנַ֔י בַּבַּ֕יִת אֲשֶׁר־ נִקְרָ֥א שְׁמִ֖י עָלָֽיו׃ (כז) וַתָּשֻׁ֙בוּ֙ וַתְּחַלְּל֣וּ אֶת־ שְׁמִ֔י וַתָּשִׁ֗בוּ אִ֤ישׁ אֶת־ עַבְדּוֹ֙ וְאִ֣ישׁ אֶת־ שִׁפְחָת֔וֹ אֲשֶׁר־ שִׁלַּחְתֶּ֥ם חָפְשִׁ֖ים לְנַפְשָׁ֑ם וַתִּכְבְּשׁ֣וּ אֹתָ֔ם לִֽהְי֣וֹת לָכֶ֔ם לַעֲבָדִ֖ים וְלִשְׁפָחֽוֹת׃ (כח) לָכֵן֮ כֹּה־ אָמַ֣ר יְהוָה֒ אַתֶּם֙ לֹֽא־ שְׁמַעְתֶּ֣ם אֵלַ֔י לִקְרֹ֣א דְר֔וֹר אִ֥ישׁ לְאָחִ֖יו וְאִ֣ישׁ לְרֵעֵ֑הוּ הִנְנִ֣י קֹרֵא֩ לָכֶ֨ם דְּר֜וֹר נְאֻם־ יְהוָ֗ה אֶל־ הַחֶ֙רֶב֙ אֶל־ הַדֶּ֣בֶר וְאֶל־ הָרָעָ֔ב וְנָתַתִּ֤י אֶתְכֶם֙ (לזועה) לְזַעֲוָ֔ה לְכֹ֖ל מַמְלְכ֥וֹת הָאָֽרֶץ׃ (כט) וְנָתַתִּ֣י אֶת־ הָאֲנָשִׁ֗ים הָעֹֽבְרִים֙ אֶת־ בְּרִתִ֔י אֲשֶׁ֤ר לֹֽא־ הֵקִ֙ימוּ֙ אֶת־ דִּבְרֵ֣י הַבְּרִ֔ית אֲשֶׁ֥ר כָּרְת֖וּ לְפָנָ֑י הָעֵ֙גֶל֙ אֲשֶׁ֣ר כָּרְת֣וּ לִשְׁנַ֔יִם וַיַּעַבְר֖וּ בֵּ֥ין בְּתָרָֽיו׃ (ל) שָׂרֵ֨י יְהוּדָ֜ה וְשָׂרֵ֣י יְרוּשָׁלִַ֗ם הַסָּֽרִסִים֙ וְהַכֹּ֣הֲנִ֔ים וְכֹ֖ל עַ֣ם הָאָ֑רֶץ הָעֹ֣בְרִ֔ים בֵּ֖ין בִּתְרֵ֥י הָעֵֽגֶל׃ (לא) וְנָתַתִּ֤י אוֹתָם֙ בְּיַ֣ד אֹֽיְבֵיהֶ֔ם וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשָׁ֑ם וְהָיְתָ֤ה נִבְלָתָם֙ לְמַֽאֲכָ֔ל לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֥ת הָאָֽרֶץ׃ (לב) וְאֶת־ צִדְקִיָּ֨הוּ מֶֽלֶךְ־ יְהוּדָ֜ה וְאֶת־ שָׂרָ֗יו אֶתֵּן֙ בְּיַ֣ד אֹֽיְבֵיהֶ֔ם וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשָׁ֑ם וּבְיַד חֵ֚יל מֶ֣לֶךְ בָּבֶ֔ל הָעֹלִ֖ים מֵעֲלֵיכֶֽם׃ (לג) הִנְנִ֨י מְצַוֶּ֜ה נְאֻם־ יְהוָ֗ה וַהֲשִׁ֨בֹתִ֜ים אֶל־ הָעִ֤יר הַזֹּאת֙ וְנִלְחֲמ֣וּ עָלֶ֔יהָ וּלְכָד֖וּהָ וּשְׂרָפֻ֣הָ בָאֵ֑שׁ וְאֶת־ עָרֵ֧י יְהוּדָ֛ה אֶתֵּ֥ן שְׁמָמָ֖ה מֵאֵ֥ין יֹשֵֽׁב׃ תחילת פרק לה (לד) הַדָּבָ֛ר אֲשֶׁר־ הָיָ֥ה אֶֽל־ יִרְמְיָ֖הוּ מֵאֵ֣ת יְהוָ֑ה בִּימֵ֨י יְהוֹיָקִ֧ים בֶּן־ יֹאשִׁיָּ֛הוּ מֶ֥לֶךְ יְהוּדָ֖ה לֵאמֹֽר׃ (לה) הָלוֹךְ֮ אֶל־ בֵּ֣ית הָרֵכָבִים֒ יִרְמְיָ֖הוּ בֵּ֣ית יְהוָ֑ה בְּלִשְׁכַּ֡ת גְּמַרְיָהוּ֩ בֶן־ שָׁפָ֨ן הַסֹּפֵ֜ר בֶּחָצֵ֣ר הָעֶלְי֗וֹן פֶּ֣תַח שַׁ֤עַר בֵּית־ יְהוָה֙ הֶֽחָדָ֔שׁ בְּאָזְנֵ֖י כָּל־ הָעָֽם׃ (כא) וַ֠יִּשְׁמַ֗ע מִכָ֨יְהוּ בֶן־ גְּמַרְיָ֧הוּ בֶן־ שָׁפָ֛ן אֶת־ כָּל־ דִּבְרֵ֥י יְהוָ֖ה מֵעַ֥ל הַסֵּֽפֶר׃ (כב) וַיֵּ֤רֶד בֵּית־ הַמֶּ֙לֶךְ֙ עַל־ לִשְׁכַּ֣ת הַסֹּפֵ֔ר וְהִ֨נֵּה־ שָׁ֔ם כָּל־ הַשָּׂרִ֖ים יֽוֹשְׁבִ֑ים אֱלִישָׁמָ֣ע הַסֹּפֵ֡ר וּדְלָיָ֣הוּ בֶן־ שְׁ֠מַעְיָהוּ וְאֶלְנָתָ֨ן בֶּן־ עַכְבּ֜וֹר וּגְמַרְיָ֧הוּ בֶן־ שָׁפָ֛ן וְצִדְקִיָּ֥הוּ בֶן־ חֲנַנְיָ֖הוּ וְכָל־ הַשָּׂרִֽים׃ (כג) וַיַּגֵּ֤ד לָהֶם֙ מִכָ֔יְהוּ אֵ֥ת כָּל־ הַדְּבָרִ֖ים אֲשֶׁ֣ר שָׁמֵ֑עַ בִּקְרֹ֥א בָר֛וּךְ בַּסֵּ֖פֶר בְּאָזְנֵ֥י הָעָֽם׃ (כד) וַיִּשְׁלְח֨וּ כָל־ הַשָּׂרִ֜ים אֶל־ בָּר֗וּךְ אֶת־ יְהוּדִ֡י בֶּן־ נְ֠תַנְיָהוּ בֶּן־ שֶׁלֶמְיָ֣הוּ בֶן־ כּוּשִׁי֮ לֵאמֹר֒ הַמְּגִלָּ֗ה אֲשֶׁ֨ר קָרָ֤אתָ בָּהּ֙ בְּאָזְנֵ֣י הָעָ֔ם קָחֶ֥נָּה בְיָדְךָ֖ וָלֵ֑ךְ וַ֠יִּקַּח בָּר֨וּךְ בֶּן־ נֵרִיָּ֤הוּ אֶת־ הַמְּגִלָּה֙ בְּיָד֔וֹ וַיָּבֹ֖א אֲלֵיהֶֽם׃ (כה) וַיֹּאמְר֣וּ אֵלָ֔יו שֵׁ֣ב נָ֔א וּקְרָאֶ֖נָּה בְּאָזְנֵ֑ינוּ וַיִּקְרָ֥א בָר֖וּךְ בְּאָזְנֵיהֶֽם׃ (כו) וַיְהִ֗י כְּשָׁמְעָם֙ אֶת־ כָּל־ הַדְּבָרִ֔ים פָּחֲד֖וּ אִ֣ישׁ אֶל־ רֵעֵ֑הוּ וַיֹּֽאמְרוּ֙ אֶל־ בָּר֔וּךְ הַגֵּ֤יד נַגִּיד֙ לַמֶּ֔לֶךְ אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (כז) וְאֶ֨ת־ בָּר֔וּךְ שָׁאֲל֖וּ לֵאמֹ֑ר הַגֶּד־ נָ֣א לָ֔נוּ אֵ֗יךְ כָּתַ֛בְתָּ אֶת־ כָּל־ הַדְּבָרִ֥ים הָאֵ֖לֶּה מִפִּֽיו׃ (כח) וַיֹּ֤אמֶר לָהֶם֙ בָּר֔וּךְ מִפִּיו֙ יִקְרָ֣א אֵלַ֔י אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַאֲנִ֛י כֹּתֵ֥ב עַל־ הַסֵּ֖פֶר בַּדְּיֽוֹ׃ (כט) וַיֹּאמְר֤וּ הַשָּׂרִים֙ אֶל־ בָּר֔וּךְ לֵ֥ךְ הִסָּתֵ֖ר אַתָּ֣ה וְיִרְמְיָ֑הוּ וְאִ֥ישׁ אַל־ יֵדַ֖ע אֵיפֹ֥ה אַתֶּֽם׃ (ל) וַיָּבֹ֤אוּ אֶל־ הַמֶּ֙לֶךְ֙ חָצֵ֔רָה וְאֶת־ הַמְּגִלָּ֣ה הִפְקִ֔דוּ בְּלִשְׁכַּ֖ת אֱלִישָׁמָ֣ע הַסֹּפֵ֑ר וַיַּגִּ֙ידוּ֙ בְּאָזְנֵ֣י הַמֶּ֔לֶךְ אֵ֖ת כָּל־ הַדְּבָרִֽים׃ (לא) וַיִּשְׁלַ֨ח הַמֶּ֜לֶךְ אֶת־ יְהוּדִ֗י לָקַ֙חַת֙ אֶת־ הַמְּגִלָּ֔ה וַיִּ֨קָּחֶ֔הָ מִלִּשְׁכַּ֖ת אֱלִישָׁמָ֣ע הַסֹּפֵ֑ר וַיִּקְרָאֶ֤הָ יְהוּדִי֙ בְּאָזְנֵ֣י הַמֶּ֔לֶךְ וּבְאָזְנֵי֙ כָּל־ הַשָּׂרִ֔ים הָעֹמְדִ֖ים מֵעַ֥ל הַמֶּֽלֶךְ׃ (לב) וְהַמֶּ֗לֶךְ יוֹשֵׁב֙ בֵּ֣ית הַחֹ֔רֶף בַּחֹ֖דֶשׁ הַתְּשִׁיעִ֑י וְאֶת־ הָאָ֖ח לְפָנָ֥יו מְבֹעָֽרֶת׃ (לג) וַיְהִ֣י ׀ כִּקְר֣וֹא יְהוּדִ֗י שָׁלֹ֣שׁ דְּלָתוֹת֮ וְאַרְבָּעָה֒ יִֽקְרָעֶ֙הָ֙ בְּתַ֣עַר הַסֹּפֵ֔ר וְהַשְׁלֵ֕ךְ אֶל־ הָאֵ֖שׁ אֲשֶׁ֣ר אֶל־ הָאָ֑ח עַד־ תֹּם֙ כָּל־ הַמְּגִלָּ֔ה עַל־ הָאֵ֖שׁ אֲשֶׁ֥ר עַל־ הָאָֽח׃ (לד) וְלֹ֣א פָחֲד֔וּ וְלֹ֥א קָרְע֖וּ אֶת־ בִּגְדֵיהֶ֑ם הַמֶּ֙לֶךְ֙ וְכָל־ עֲבָדָ֔יו הַשֹּׁ֣מְעִ֔ים אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (לה) וְגַם֩ אֶלְנָתָ֨ן וּדְלָיָ֤הוּ וּגְמַרְיָ֙הוּ֙ הִפְגִּ֣עוּ בַמֶּ֔לֶךְ לְבִלְתִּ֥י שְׂרֹ֖ף אֶת־ הַמְּגִלָּ֑ה וְלֹ֥א שָׁמַ֖ע אֲלֵיהֶֽם׃
