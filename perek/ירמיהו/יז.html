ירמיהו פרק יז לפי החלוקה לסדרים

פרק יז בספר ירמיהו (27 פסוקים) נמצא בירמיהו סדר ח (פסוקים מד–מט) ובירמיהו סדר ט (פסוקים א–כא).

ירמיהו סדר ח

(א) הֲיֵ֨שׁ בְּהַבְלֵ֤י הַגּוֹיִם֙ מַגְשִׁמִ֔ים וְאִם־ הַשָּׁמַ֖יִם יִתְּנ֣וּ רְבִבִ֑ים הֲלֹ֨א אַתָּה־ ה֜וּא יְהוָ֤ה אֱלֹהֵ֙ינוּ֙ וּ֨נְקַוֶּה־ לָּ֔ךְ כִּֽי־ אַתָּ֥ה עָשִׂ֖יתָ אֶת־ כָּל־ אֵֽלֶּה׃ (ב) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י אִם־ יַעֲמֹ֨ד מֹשֶׁ֤ה וּשְׁמוּאֵל֙ לְפָנַ֔י אֵ֥ין נַפְשִׁ֖י אֶל־ הָעָ֣ם הַזֶּ֑ה שַׁלַּ֥ח מֵֽעַל־ פָּנַ֖י וְיֵצֵֽאוּ׃ (ג) וְהָיָ֛ה כִּֽי־ יֹאמְר֥וּ אֵלֶ֖יךָ אָ֣נָה נֵצֵ֑א וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם כֹּֽה־ אָמַ֣ר יְהוָ֗ה אֲשֶׁ֨ר לַמָּ֤וֶת לַמָּ֙וֶת֙ וַאֲשֶׁ֤ר לַחֶ֙רֶב֙ לַחֶ֔רֶב וַאֲשֶׁ֤ר לָֽרָעָב֙ לָֽרָעָ֔ב וַאֲשֶׁ֥ר לַשְּׁבִ֖י לַשֶּֽׁבִי׃ (ד) וּפָקַדְתִּ֨י עֲלֵיהֶ֜ם אַרְבַּ֤ע מִשְׁפָּחוֹת֙ נְאֻם־ יְהוָ֔ה אֶת־ הַחֶ֣רֶב לַֽהֲרֹ֔ג וְאֶת־ הַכְּלָבִ֖ים לִסְחֹ֑ב וְאֶת־ ע֧וֹף הַשָּׁמַ֛יִם וְאֶת־ בֶּהֱמַ֥ת הָאָ֖רֶץ לֶאֱכֹ֥ל וּלְהַשְׁחִֽית׃ (ה) וּנְתַתִּ֣ים (לזועה) לְזַֽעֲוָ֔ה לְכֹ֖ל מַמְלְכ֣וֹת הָאָ֑רֶץ בִּ֠גְלַל מְנַשֶּׁ֤ה בֶן־ יְחִזְקִיָּ֙הוּ֙ מֶ֣לֶךְ יְהוּדָ֔ה עַ֥ל אֲשֶׁר־ עָשָׂ֖ה בִּירוּשָׁלִָֽם׃ (ו) כִּ֠י מִֽי־ יַחְמֹ֤ל עָלַ֙יִךְ֙ יְר֣וּשָׁלִַ֔ם וּמִ֖י יָנ֣וּד לָ֑ךְ וּמִ֣י יָס֔וּר לִשְׁאֹ֥ל לְשָׁלֹ֖ם לָֽךְ׃ (ז) אַ֣תְּ נָטַ֥שְׁתְּ אֹתִ֛י נְאֻם־ יְהוָ֖ה אָח֣וֹר תֵּלֵ֑כִי וָאַ֨ט אֶת־ יָדִ֤י עָלַ֙יִךְ֙ וָֽאַשְׁחִיתֵ֔ךְ נִלְאֵ֖יתִי הִנָּחֵֽם׃ (ח) וָאֶזְרֵ֥ם בְּמִזְרֶ֖ה בְּשַׁעֲרֵ֣י הָאָ֑רֶץ שִׁכַּ֤לְתִּי אִבַּ֙דְתִּי֙ אֶת־ עַמִּ֔י מִדַּרְכֵיהֶ֖ם לוֹא־ שָֽׁבוּ׃ (ט) עָֽצְמוּ־ לִ֤י אַלְמְנֹתָו֙ מֵח֣וֹל יַמִּ֔ים הֵבֵ֨אתִי לָהֶ֥ם עַל־ אֵ֛ם בָּח֖וּר שֹׁדֵ֣ד בַּֽצָּהֳרָ֑יִם הִפַּ֤לְתִּי עָלֶ֙יהָ֙ פִּתְאֹ֔ם עִ֖יר וּבֶהָלֽוֹת׃ (י) אֻמְלְלָ֞ה יֹלֶ֣דֶת הַשִּׁבְעָ֗ה נָפְחָ֥ה נַפְשָׁ֛הּ (באה) בָּ֥א שִׁמְשָׁ֛הּ בְּעֹ֥ד יוֹמָ֖ם בּ֣וֹשָׁה וְחָפֵ֑רָה וּשְׁאֵֽרִיתָ֗ם לַחֶ֧רֶב אֶתֵּ֛ן לִפְנֵ֥י אֹיְבֵיהֶ֖ם נְאֻם־ יְהוָֽה׃ (יא) אֽוֹי־ לִ֣י אִמִּ֔י כִּ֣י יְלִדְתִּ֗נִי אִ֥ישׁ רִ֛יב וְאִ֥ישׁ מָד֖וֹן לְכָל־ הָאָ֑רֶץ לֹֽא־ נָשִׁ֥יתִי וְלֹא־ נָֽשׁוּ־ בִ֖י כֻּלֹּ֥ה מְקַלְלַֽונִי׃ (יב) אָמַ֣ר יְהוָ֔ה אִם־ לֹ֥א (שרותך) שֵֽׁרִיתִ֖יךָ לְט֑וֹב אִם־ ל֣וֹא ׀ הִפְגַּ֣עְתִּֽי בְךָ֗ בְּעֵ֥ת־ רָעָ֛ה וּבְעֵ֥ת צָרָ֖ה אֶת־ הָאֹיֵֽב׃ (יג) הֲיָרֹ֨עַ בַּרְזֶ֧ל ׀ בַּרְזֶ֛ל מִצָּפ֖וֹן וּנְחֹֽשֶׁת׃ (יד) חֵילְךָ֧ וְאוֹצְרוֹתֶ֛יךָ לָבַ֥ז אֶתֵּ֖ן לֹ֣א בִמְחִ֑יר וּבְכָל־ חַטֹּאותֶ֖יךָ וּבְכָל־ גְּבוּלֶֽיךָ׃ (טו) וְהַֽעֲבַרְתִּי֙ אֶת־ אֹ֣יְבֶ֔יךָ בְּאֶ֖רֶץ לֹ֣א יָדָ֑עְתָּ כִּֽי־ אֵ֛שׁ קָדְחָ֥ה בְאַפִּ֖י עֲלֵיכֶ֥ם תּוּקָֽד׃ (טז) אַתָּ֧ה יָדַ֣עְתָּ יְהוָ֗ה זָכְרֵ֤נִי וּפָקְדֵ֙נִי֙ וְהִנָּ֤קֶם לִי֙ מֵרֹ֣דְפַ֔י אַל־ לְאֶ֥רֶךְ אַפְּךָ֖ תִּקָּחֵ֑נִי דַּ֕ע שְׂאֵתִ֥י עָלֶ֖יךָ חֶרְפָּֽה׃ (יז) נִמְצְא֤וּ דְבָרֶ֙יךָ֙ וָאֹ֣כְלֵ֔ם וַיְהִ֤י (דבריך) דְבָֽרְךָ֙ לִ֔י לְשָׂשׂ֖וֹן וּלְשִׂמְחַ֣ת לְבָבִ֑י כִּֽי־ נִקְרָ֤א שִׁמְךָ֙ עָלַ֔י יְהוָ֖ה אֱלֹהֵ֥י צְבָאֽוֹת׃ (יח) לֹֽא־ יָשַׁ֥בְתִּי בְסוֹד־ מְשַׂחֲקִ֖ים וָֽאֶעְלֹ֑ז מִפְּנֵ֤י יָֽדְךָ֙ בָּדָ֣ד יָשַׁ֔בְתִּי כִּֽי־ זַ֖עַם מִלֵּאתָֽנִי׃ (יט) לָ֣מָּה הָיָ֤ה כְאֵבִי֙ נֶ֔צַח וּמַכָּתִ֖י אֲנוּשָׁ֑ה֙ מֵֽאֲנָה֙ הֵֽרָפֵ֔א הָי֨וֹ תִֽהְיֶ֥ה לִי֙ כְּמ֣וֹ אַכְזָ֔ב מַ֖יִם לֹ֥א נֶאֱמָֽנוּ׃ (כ) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה אִם־ תָּשׁ֤וּב וַאֲשִֽׁיבְךָ֙ לְפָנַ֣י תַּֽעֲמֹ֔ד וְאִם־ תּוֹצִ֥יא יָקָ֛ר מִזּוֹלֵ֖ל כְּפִ֣י תִֽהְיֶ֑ה יָשֻׁ֤בוּ הֵ֙מָּה֙ אֵלֶ֔יךָ וְאַתָּ֖ה לֹֽא־ תָשׁ֥וּב אֲלֵיהֶֽם׃ (כא) וּנְתַתִּ֜יךָ לָעָ֣ם הַזֶּ֗ה לְחוֹמַ֤ת נְחֹ֙שֶׁת֙ בְּצוּרָ֔ה וְנִלְחֲמ֥וּ אֵלֶ֖יךָ וְלֹא־ י֣וּכְלוּ לָ֑ךְ כִּֽי־ אִתְּךָ֥ אֲנִ֛י לְהוֹשִֽׁיעֲךָ֥ וּלְהַצִּילֶ֖ךָ נְאֻם־ יְהוָֽה׃ (כב) וְהִצַּלְתִּ֖יךָ מִיַּ֣ד רָעִ֑ים וּפְדִתִ֖יךָ מִכַּ֥ף עָרִצִֽים׃ (כג) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כד) לֹֽא־ תִקַּ֥ח לְךָ֖ אִשָּׁ֑ה וְלֹֽא־ יִהְי֤וּ לְךָ֙ בָּנִ֣ים וּבָנ֔וֹת בַּמָּק֖וֹם הַזֶּֽה׃ (כה) כִּי־ כֹ֣ה ׀ אָמַ֣ר יְהוָ֗ה עַל־ הַבָּנִים֙ וְעַל־ הַבָּנ֔וֹת הַיִּלּוֹדִ֖ים בַּמָּק֣וֹם הַזֶּ֑ה וְעַֽל־ אִמֹּתָ֞ם הַיֹּלְד֣וֹת אוֹתָ֗ם וְעַל־ אֲבוֹתָ֛ם הַמּוֹלִדִ֥ים אוֹתָ֖ם בָּאָ֥רֶץ הַזֹּֽאת׃ (כו) מְמוֹתֵ֨י תַחֲלֻאִ֜ים יָמֻ֗תוּ לֹ֤א יִסָּֽפְדוּ֙ וְלֹ֣א יִקָּבֵ֔רוּ לְדֹ֛מֶן עַל־ פְּנֵ֥י הָאֲדָמָ֖ה יִֽהְי֑וּ וּבַחֶ֤רֶב וּבָֽרָעָב֙ יִכְל֔וּ וְהָיְתָ֤ה נִבְלָתָם֙ לְמַאֲכָ֔ל לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֥ת הָאָֽרֶץ׃ (כז) כִּֽי־ כֹ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ תָּבוֹא֙ בֵּ֣ית מַרְזֵ֔חַ וְאַל־ תֵּלֵ֣ךְ לִסְפּ֔וֹד וְאַל־ תָּנֹ֖ד לָהֶ֑ם כִּֽי־ אָסַ֨פְתִּי אֶת־ שְׁלוֹמִ֜י מֵאֵ֨ת הָעָ֤ם־ הַזֶּה֙ נְאֻם־ יְהוָ֔ה אֶת־ הַחֶ֖סֶד וְאֶת־ הָֽרַחֲמִֽים׃ (כח) וּמֵ֨תוּ גְדֹלִ֧ים וּקְטַנִּ֛ים בָּאָ֥רֶץ הַזֹּ֖את לֹ֣א יִקָּבֵ֑רוּ וְלֹֽא־ יִסְפְּד֣וּ לָהֶ֔ם וְלֹ֣א יִתְגֹּדַ֔ד וְלֹ֥א יִקָּרֵ֖חַ לָהֶֽם׃ (כט) וְלֹֽא־ יִפְרְס֥וּ לָהֶ֛ם עַל־ אֵ֖בֶל לְנַחֲמ֣וֹ עַל־ מֵ֑ת וְלֹֽא־ יַשְׁק֤וּ אוֹתָם֙ כּ֣וֹס תַּנְחוּמִ֔ים עַל־ אָבִ֖יו וְעַל־ אִמּֽוֹ׃ (ל) וּבֵית־ מִשְׁתֶּ֥ה לֹא־ תָב֖וֹא לָשֶׁ֣בֶת אוֹתָ֑ם לֶאֱכֹ֖ל וְלִשְׁתּֽוֹת׃ (לא) כִּי֩ כֹ֨ה אָמַ֜ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֨י מַשְׁבִּ֜ית מִן־ הַמָּק֥וֹם הַזֶּ֛ה לְעֵינֵיכֶ֖ם וּבִֽימֵיכֶ֑ם ק֤וֹל שָׂשׂוֹן֙ וְק֣וֹל שִׂמְחָ֔ה ק֥וֹל חָתָ֖ן וְק֥וֹל כַּלָּֽה׃ (לב) וְהָיָ֗ה כִּ֤י תַגִּיד֙ לָעָ֣ם הַזֶּ֔ה אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וְאָמְר֣וּ אֵלֶ֗יךָ עַל־ מֶה֩ דִבֶּ֨ר יְהוָ֤ה עָלֵ֙ינוּ֙ אֵ֣ת כָּל־ הָרָעָ֤ה הַגְּדוֹלָה֙ הַזֹּ֔את וּמֶ֤ה עֲוֺנֵ֙נוּ֙ וּמֶ֣ה חַטָּאתֵ֔נוּ אֲשֶׁ֥ר חָטָ֖אנוּ לַֽיהוָ֥ה אֱלֹהֵֽינוּ׃ (לג) וְאָמַרְתָּ֣ אֲלֵיהֶ֗ם עַל֩ אֲשֶׁר־ עָזְב֨וּ אֲבוֹתֵיכֶ֤ם אוֹתִי֙ נְאֻם־ יְהוָ֔ה וַיֵּלְכ֗וּ אַֽחֲרֵי֙ אֱלֹהִ֣ים אֲחֵרִ֔ים וַיַּעַבְד֖וּם וַיִּשְׁתַּחֲו֣וּ לָהֶ֑ם וְאֹתִ֣י עָזָ֔בוּ וְאֶת־ תּוֹרָתִ֖י לֹ֥א שָׁמָֽרוּ׃ (לד) וְאַתֶּ֛ם הֲרֵעֹתֶ֥ם לַעֲשׂ֖וֹת מֵאֲבֽוֹתֵיכֶ֑ם וְהִנְּכֶ֣ם הֹלְכִ֗ים אִ֚ישׁ אַֽחֲרֵי֙ שְׁרִר֣וּת לִבּֽוֹ־ הָרָ֔ע לְבִלְתִּ֖י שְׁמֹ֥עַ אֵלָֽי׃ (לה) וְהֵטַלְתִּ֣י אֶתְכֶ֗ם מֵעַל֙ הָאָ֣רֶץ הַזֹּ֔את עַל־ הָאָ֕רֶץ אֲשֶׁר֙ לֹ֣א יְדַעְתֶּ֔ם אַתֶּ֖ם וַאֲבֽוֹתֵיכֶ֑ם וַעֲבַדְתֶּם־ שָׁ֞ם אֶת־ אֱלֹהִ֤ים אֲחֵרִים֙ יוֹמָ֣ם וָלַ֔יְלָה אֲשֶׁ֛ר לֹֽא־ אֶתֵּ֥ן לָכֶ֖ם חֲנִינָֽה׃ (לו) לָכֵ֛ן הִנֵּֽה־ יָמִ֥ים בָּאִ֖ים נְאֻם־ יְהוָ֑ה וְלֹֽא־ יֵאָמֵ֥ר עוֹד֙ חַי־ יְהוָ֔ה אֲשֶׁ֧ר הֶעֱלָ֛ה אֶת־ בְּנֵ֥י יִשְׂרָאֵ֖ל מֵאֶ֥רֶץ מִצְרָֽיִם׃ (לז) כִּ֣י אִם־ חַי־ יְהוָ֗ה אֲשֶׁ֨ר הֶעֱלָ֜ה אֶת־ בְּנֵ֤י יִשְׂרָאֵל֙ מֵאֶ֣רֶץ צָפ֔וֹן הַדָּבָר֙ אֲשֶׁ֣ר הָיָ֣ה אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (כג) ק֥וּם וְיָרַדְתָּ֖ בֵּ֣ית הַיּוֹצֵ֑ר וְשָׁ֖מָּה אַשְׁמִֽיעֲךָ֥ אֶת־ דְּבָרָֽי׃ (כד) וָאֵרֵ֖ד בֵּ֣ית הַיּוֹצֵ֑ר (והנהו) וְהִנֵּה־ ה֛וּא עֹשֶׂ֥ה מְלָאכָ֖ה עַל־ הָאָבְנָֽיִם׃ (כה) וְנִשְׁחַ֣ת הַכְּלִ֗י אֲשֶׁ֨ר ה֥וּא עֹשֶׂ֛ה בַּחֹ֖מֶר בְּיַ֣ד הַיּוֹצֵ֑ר וְשָׁ֗ב וַֽיַּעֲשֵׂ֙הוּ֙ כְּלִ֣י אַחֵ֔ר כַּאֲשֶׁ֥ר יָשַׁ֛ר בְּעֵינֵ֥י הַיּוֹצֵ֖ר לַעֲשֽׂוֹת׃ (כו) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֽוֹר׃ (כז) הֲכַיּוֹצֵ֨ר הַזֶּ֜ה לֹא־ אוּכַ֨ל לַעֲשׂ֥וֹת לָכֶ֛ם בֵּ֥ית יִשְׂרָאֵ֖ל נְאֻם־ יְהוָ֑ה הִנֵּ֤ה כַחֹ֙מֶר֙ בְּיַ֣ד הַיּוֹצֵ֔ר כֵּן־ אַתֶּ֥ם בְּיָדִ֖י בֵּ֥ית יִשְׂרָאֵֽל׃ (כח) רֶ֣גַע אֲדַבֵּ֔ר עַל־ גּ֖וֹי וְעַל־ מַמְלָכָ֑ה לִנְת֥וֹשׁ וְלִנְת֖וֹץ וּֽלְהַאֲבִֽיד׃ (כט) וְשָׁב֙ הַגּ֣וֹי הַה֔וּא מֵרָ֣עָת֔וֹ אֲשֶׁ֥ר דִּבַּ֖רְתִּי עָלָ֑יו וְנִֽחַמְתִּי֙ עַל־ הָ֣רָעָ֔ה אֲשֶׁ֥ר חָשַׁ֖בְתִּי לַעֲשׂ֥וֹת לֽוֹ׃ (ל) וְרֶ֣גַע אֲדַבֵּ֔ר עַל־ גּ֖וֹי וְעַל־ מַמְלָכָ֑ה לִבְנֹ֖ת וְלִנְטֹֽעַ׃ (לא) וְעָשָׂ֤ה (הרעה) הָרַע֙ בְּעֵינַ֔י לְבִלְתִּ֖י שְׁמֹ֣עַ בְּקוֹלִ֑י וְנִֽחַמְתִּי֙ עַל־ הַטּוֹבָ֔ה אֲשֶׁ֥ר אָמַ֖רְתִּי לְהֵיטִ֥יב אוֹתֽוֹ׃ (לב) וְעַתָּ֡ה אֱמָר־ נָ֣א אֶל־ אִישׁ־ יְהוּדָה֩ וְעַל־ יוֹשְׁבֵ֨י יְרוּשָׁלִַ֜ם לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנֵּ֨ה אָנֹכִ֜י יוֹצֵ֤ר עֲלֵיכֶם֙ רָעָ֔ה וְחֹשֵׁ֥ב עֲלֵיכֶ֖ם מַֽחֲשָׁבָ֑ה שׁ֣וּבוּ נָ֗א אִ֚ישׁ מִדַּרְכּ֣וֹ הָֽרָעָ֔ה וְהֵיטִ֥יבוּ דַרְכֵיכֶ֖ם וּמַעַלְלֵיכֶֽם׃ (לג) וְאָמְר֖וּ נוֹאָ֑שׁ כִּֽי־ אַחֲרֵ֤י מַחְשְׁבוֹתֵ֙ינוּ֙ נֵלֵ֔ךְ וְאִ֛ישׁ שְׁרִר֥וּת לִבּֽוֹ־ הָרָ֖ע נַעֲשֶֽׂה׃ (לד) לָכֵ֗ן כֹּ֚ה אָמַ֣ר יְהוָ֔ה שַֽׁאֲלוּ־ נָא֙ בַּגּוֹיִ֔ם מִ֥י שָׁמַ֖ע כָּאֵ֑לֶּה שַֽׁעֲרֻרִת֙ עָשְׂתָ֣ה מְאֹ֔ד בְּתוּלַ֖ת יִשְׂרָאֵֽל׃ (לה) הֲיַעֲזֹ֥ב מִצּ֛וּר שָׂדַ֖י שֶׁ֣לֶג לְבָנ֑וֹן אִם־ יִנָּתְשׁ֗וּ מַ֛יִם זָרִ֥ים קָרִ֖ים נוֹזְלִֽים׃ (לו) כִּֽי־ שְׁכֵחֻ֥נִי עַמִּ֖י לַשָּׁ֣וְא יְקַטֵּ֑רוּ וַיַּכְשִׁל֤וּם בְּדַרְכֵיהֶם֙ שְׁבִילֵ֣י עוֹלָ֔ם לָלֶ֣כֶת נְתִיב֔וֹת דֶּ֖רֶךְ לֹ֥א סְלוּלָֽה׃ (לז) לָשׂ֥וּם אַרְצָ֛ם לְשַׁמָּ֖ה (שרוקת) שְׁרִיק֣וֹת עוֹלָ֑ם כֹּ֚ל עוֹבֵ֣ר עָלֶ֔יהָ יִשֹּׁ֖ם וְיָנִ֥יד בְּרֹאשֽׁוֹ׃ (לח) כְּרֽוּחַ־ קָדִ֥ים אֲפִיצֵ֖ם לִפְנֵ֣י אוֹיֵ֑ב עֹ֧רֶף וְלֹֽא־ פָנִ֛ים אֶרְאֵ֖ם בְּי֥וֹם אֵידָֽם׃ (לט) וַיֹּאמְר֗וּ לְכ֨וּ וְנַחְשְׁבָ֣ה עַֽל־ יִרְמְיָהוּ֮ מַחֲשָׁבוֹת֒ כִּי֩ לֹא־ תֹאבַ֨ד תּוֹרָ֜ה מִכֹּהֵ֗ן וְעֵצָה֙ מֵֽחָכָ֔ם וְדָבָ֖ר מִנָּבִ֑יא לְכוּ֙ וְנַכֵּ֣הוּ בַלָּשׁ֔וֹן וְאַל־ נַקְשִׁ֖יבָה אֶל־ כָּל־ דְּבָרָֽיו׃
