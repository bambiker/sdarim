ירמיהו פרק מט לפי החלוקה לסדרים

פרק מט בספר ירמיהו (39 פסוקים) נמצא בירמיהו סדר כז (פסוקים נז–נז) ובירמיהו סדר כח (פסוקים א–לח).

ירמיהו סדר כז

(א) וְ֠אַתָּה אַל־ תִּירָ֞א עַבְדִּ֤י יַֽעֲקֹב֙ וְאַל־ תֵּחַ֣ת יִשְׂרָאֵ֔ל כִּ֠י הִנְנִ֤י מוֹשִֽׁעֲךָ֙ מֵֽרָח֔וֹק וְאֶֽת־ זַרְעֲךָ֖ מֵאֶ֣רֶץ שִׁבְיָ֑ם וְשָׁ֧ב יַעֲק֛וֹב וְשָׁקַ֥ט וְשַׁאֲנַ֖ן וְאֵ֥ין מַחֲרִֽיד׃ (ב) אַ֠תָּה אַל־ תִּירָ֞א עַבְדִּ֤י יַֽעֲקֹב֙ נְאֻם־ יְהוָ֔ה כִּ֥י אִתְּךָ֖ אָ֑נִי כִּי֩ אֶעֱשֶׂ֨ה כָלָ֜ה בְּכָֽל־ הַגּוֹיִ֣ם ׀ אֲשֶׁ֧ר הִדַּחְתִּ֣יךָ שָׁ֗מָּה וְאֹֽתְךָ֙ לֹא־ אֶעֱשֶׂ֣ה כָלָ֔ה וְיִסַּרְתִּ֙יךָ֙ לַמִּשְׁפָּ֔ט וְנַקֵּ֖ה לֹ֥א אֲנַקֶּֽךָּ׃ (ג) אֲשֶׁ֨ר הָיָ֧ה דְבַר־ יְהוָ֛ה אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא אֶל־ פְּלִשְׁתִּ֑ים בְּטֶ֛רֶם יַכֶּ֥ה פַרְעֹ֖ה אֶת־ עַזָּֽה׃ (ד) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה הִנֵּה־ מַ֜יִם עֹלִ֤ים מִצָּפוֹן֙ וְהָיוּ֙ לְנַ֣חַל שׁוֹטֵ֔ף וְיִשְׁטְפוּ֙ אֶ֣רֶץ וּמְלוֹאָ֔הּ עִ֖יר וְיֹ֣שְׁבֵי בָ֑הּ וְזָֽעֲקוּ֙ הָֽאָדָ֔ם וְהֵילִ֕ל כֹּ֖ל יוֹשֵׁ֥ב הָאָֽרֶץ׃ (ה) מִקּ֗וֹל שַֽׁעֲטַת֙ פַּרְס֣וֹת אַבִּירָ֔יו מֵרַ֣עַשׁ לְרִכְבּ֔וֹ הֲמ֖וֹן גַּלְגִּלָּ֑יו לֹֽא־ הִפְנ֤וּ אָבוֹת֙ אֶל־ בָּנִ֔ים מֵֽרִפְי֖וֹן יָדָֽיִם׃ (ו) עַל־ הַיּ֗וֹם הַבָּא֙ לִשְׁד֣וֹד אֶת־ כָּל־ פְּלִשְׁתִּ֔ים לְהַכְרִ֤ית לְצֹר֙ וּלְצִיד֔וֹן כֹּ֖ל שָׂרִ֣יד עֹזֵ֑ר כִּֽי־ שֹׁדֵ֤ד יְהוָה֙ אֶת־ פְּלִשְׁתִּ֔ים שְׁאֵרִ֖ית אִ֥י כַפְתּֽוֹר׃ (ז) בָּ֤אָה קָרְחָה֙ אֶל־ עַזָּ֔ה נִדְמְתָ֥ה אַשְׁקְל֖וֹן שְׁאֵרִ֣ית עִמְקָ֑ם עַד־ מָתַ֖י תִּתְגּוֹדָֽדִי׃ (ח) ה֗וֹי חֶ֚רֶב לַֽיהוָ֔ה עַד־ אָ֖נָה לֹ֣א תִשְׁקֹ֑טִי הֵאָֽסְפִי֙ אַל־ תַּעְרֵ֔ךְ הֵרָגְעִ֖י וָדֹֽמִּי׃ (ט) אֵ֣יךְ תִּשְׁקֹ֔טִי וַֽיהוָ֖ה צִוָּה־ לָ֑הּ אֶֽל־ אַשְׁקְל֛וֹן וְאֶל־ ח֥וֹף הַיָּ֖ם שָׁ֥ם יְעָדָֽהּ׃ (י) לְמוֹאָ֡ב כֹּֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל ה֤וֹי אֶל־ נְבוֹ֙ כִּ֣י שֻׁדָּ֔דָה הֹבִ֥ישָׁה נִלְכְּדָ֖ה קִרְיָתָ֑יִם הֹבִ֥ישָׁה הַמִּשְׂגָּ֖ב וָחָֽתָּה׃ (יא) אֵ֣ין עוֹד֮ תְּהִלַּ֣ת מוֹאָב֒ בְּחֶשְׁבּ֗וֹן חָשְׁב֤וּ עָלֶ֙יהָ֙ רָעָ֔ה לְכ֖וּ וְנַכְרִיתֶ֣נָּה מִגּ֑וֹי גַּם־ מַדְמֵ֣ן תִּדֹּ֔מִּי אַחֲרַ֖יִךְ תֵּ֥לֶךְ חָֽרֶב׃ (יב) ק֥וֹל צְעָקָ֖ה מֵחֹֽרוֹנָ֑יִם שֹׁ֖ד וָשֶׁ֥בֶר גָּדֽוֹל׃ (יג) נִשְׁבְּרָ֖ה מוֹאָ֑ב הִשְׁמִ֥יעוּ זְּעָקָ֖ה (צעוריה) צְעִירֶֽיהָ׃ (יד) כִּ֚י מַעֲלֵ֣ה (הלחות) הַלּוּחִ֔ית בִּבְכִ֖י יַֽעֲלֶה־ בֶּ֑כִי כִּ֚י בְּמוֹרַ֣ד חוֹרֹנַ֔יִם צָרֵ֥י צַֽעֲקַת־ שֶׁ֖בֶר שָׁמֵֽעוּ׃ (טו) נֻ֖סוּ מַלְּט֣וּ נַפְשְׁכֶ֑ם וְתִֽהְיֶ֕ינָה כַּעֲרוֹעֵ֖ר בַּמִּדְבָּֽר׃ (טז) כִּ֠י יַ֣עַן בִּטְחֵ֤ךְ בְּמַעֲשַׂ֙יִךְ֙ וּבְא֣וֹצְרוֹתַ֔יִךְ גַּם־ אַ֖תְּ תִּלָּכֵ֑דִי וְיָצָ֤א (כמיש) כְמוֹשׁ֙ בַּגּוֹלָ֔ה כֹּהֲנָ֥יו וְשָׂרָ֖יו (יחד) יַחְדָּֽיו׃ (יז) וְיָבֹ֨א שֹׁדֵ֜ד אֶל־ כָּל־ עִ֗יר וְעִיר֙ לֹ֣א תִמָּלֵ֔ט וְאָבַ֥ד הָעֵ֖מֶק וְנִשְׁמַ֣ד הַמִּישֹׁ֑ר אֲשֶׁ֖ר אָמַ֥ר יְהוָֽה׃ (יח) תְּנוּ־ צִ֣יץ לְמוֹאָ֔ב כִּ֥י נָצֹ֖א תֵּצֵ֑א וְעָרֶ֙יהָ֙ לְשַׁמָּ֣ה תִֽהְיֶ֔ינָה מֵאֵ֥ין יוֹשֵׁ֖ב בָּהֵֽן׃ (יט) אָר֗וּר עֹשֶׂ֛ה מְלֶ֥אכֶת יְהוָ֖ה רְמִיָּ֑ה וְאָר֕וּר מֹנֵ֥עַ חַרְבּ֖וֹ מִדָּֽם׃ (כ) שַׁאֲנַ֨ן מוֹאָ֜ב מִנְּעוּרָ֗יו וְשֹׁקֵ֥ט הוּא֙ אֶל־ שְׁמָרָ֔יו וְלֹֽא־ הוּרַ֤ק מִכְּלִי֙ אֶל־ כֶּ֔לִי וּבַגּוֹלָ֖ה לֹ֣א הָלָ֑ךְ עַל־ כֵּ֗ן עָמַ֤ד טַעְמוֹ֙ בּ֔וֹ וְרֵיח֖וֹ לֹ֥א נָמָֽר׃ (כא) לָכֵ֞ן הִנֵּ֖ה־ יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וְשִׁלַּחְתִּי־ ל֥וֹ צֹעִ֖ים וְצֵעֻ֑הוּ וְכֵלָ֣יו יָרִ֔יקוּ וְנִבְלֵיהֶ֖ם יְנַפֵּֽצוּ׃ (כב) וּבֹ֥שׁ מוֹאָ֖ב מִכְּמ֑וֹשׁ כַּאֲשֶׁר־ בֹּ֙שׁוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל מִבֵּ֥ית אֵ֖ל מִבְטֶחָֽם׃ (כג) אֵ֚יךְ תֹּֽאמְר֔וּ גִּבּוֹרִ֖ים אֲנָ֑חְנוּ וְאַנְשֵׁי־ חַ֖יִל לַמִּלְחָמָֽה׃ (כד) שֻׁדַּ֤ד מוֹאָב֙ וְעָרֶ֣יהָ עָלָ֔ה וּמִבְחַ֥ר בַּֽחוּרָ֖יו יָרְד֣וּ לַטָּ֑בַח נְאֻ֨ם־ הַמֶּ֔לֶךְ יְהוָ֥ה צְבָא֖וֹת שְׁמֽוֹ׃ (כה) קָר֥וֹב אֵיד־ מוֹאָ֖ב לָב֑וֹא וְרָ֣עָת֔וֹ מִהֲרָ֖ה מְאֹֽד׃ (כו) נֻ֤דוּ לוֹ֙ כָּל־ סְבִיבָ֔יו וְכֹ֖ל יֹדְעֵ֣י שְׁמ֑וֹ אִמְר֗וּ אֵיכָ֤ה נִשְׁבַּר֙ מַטֵּה־ עֹ֔ז מַקֵּ֖ל תִּפְאָרָֽה׃ (כז) רְדִ֤י מִכָּבוֹד֙ (ישבי) וּשְׁבִ֣י בַצָּמָ֔א יֹשֶׁ֖בֶת בַּת־ דִּיב֑וֹן כִּֽי־ שֹׁדֵ֤ד מוֹאָב֙ עָ֣לָה בָ֔ךְ שִׁחֵ֖ת מִבְצָרָֽיִךְ׃ (כח) אֶל־ דֶּ֛רֶךְ עִמְדִ֥י וְצַפִּ֖י יוֹשֶׁ֣בֶת עֲרוֹעֵ֑ר שַׁאֲלִי־ נָ֣ס וְנִמְלָ֔טָה אִמְרִ֖י מַה־ נִּֽהְיָֽתָה׃ (כט) הֹבִ֥ישׁ מוֹאָ֛ב כִּֽי־ חַ֖תָּה (הילילי) הֵילִ֣ילוּ ׀ (וזעקי) וּֽזְעָ֑קוּ הַגִּ֣ידוּ בְאַרְנ֔וֹן כִּ֥י שֻׁדַּ֖ד מוֹאָֽב׃ (ל) וּמִשְׁפָּ֥ט בָּ֖א אֶל־ אֶ֣רֶץ הַמִּישֹׁ֑ר אֶל־ חֹל֥וֹן וְאֶל־ יַ֖הְצָה וְעַל־ (מופעת) מֵיפָֽעַת׃ (לא) וְעַל־ דִּיב֣וֹן וְעַל־ נְב֔וֹ וְעַל־ בֵּ֖ית דִּבְלָתָֽיִם׃ (לב) וְעַ֧ל קִרְיָתַ֛יִם וְעַל־ בֵּ֥ית גָּמ֖וּל וְעַל־ בֵּ֥ית מְעֽוֹן׃ (לג) וְעַל־ קְרִיּ֖וֹת וְעַל־ בָּצְרָ֑ה וְעַ֗ל כָּל־ עָרֵי֙ אֶ֣רֶץ מוֹאָ֔ב הָרְחֹק֖וֹת וְהַקְּרֹבֽוֹת׃ (לד) נִגְדְּעָה֙ קֶ֣רֶן מוֹאָ֔ב וּזְרֹע֖וֹ נִשְׁבָּ֑רָה נְאֻ֖ם יְהוָֽה׃ (לה) הַשְׁכִּירֻ֕הוּ כִּ֥י עַל־ יְהוָ֖ה הִגְדִּ֑יל וְסָפַ֤ק מוֹאָב֙ בְּקִיא֔וֹ וְהָיָ֥ה לִשְׂחֹ֖ק גַּם־ הֽוּא׃ (לו) וְאִ֣ם ׀ ל֣וֹא הַשְּׂחֹ֗ק הָיָ֤ה לְךָ֙ יִשְׂרָאֵ֔ל אִם־ בְּגַנָּבִ֖ים (נמצאה) נִמְצָ֑א כִּֽי־ מִדֵּ֧י דְבָרֶ֛יךָ בּ֖וֹ תִּתְנוֹדָֽד׃ (לז) עִזְב֤וּ עָרִים֙ וְשִׁכְנ֣וּ בַּסֶּ֔לַע יֹשְׁבֵ֖י מוֹאָ֑ב וִֽהְי֣וּ כְיוֹנָ֔ה תְּקַנֵּ֖ן בְּעֶבְרֵ֥י פִי־ פָֽחַת׃ (לח) שָׁמַ֥עְנוּ גְאוֹן־ מוֹאָ֖ב גֵּאֶ֣ה מְאֹ֑ד גָּבְה֧וֹ וּגְאוֹנ֛וֹ וְגַאֲוָת֖וֹ וְרֻ֥ם לִבּֽוֹ׃ (לט) אֲנִ֤י יָדַ֙עְתִּי֙ נְאֻם־ יְהוָ֔ה עֶבְרָת֖וֹ וְלֹא־ כֵ֑ן בַּדָּ֖יו לֹא־ כֵ֥ן עָשֽׂוּ׃ (מ) עַל־ כֵּן֙ עַל־ מוֹאָ֣ב אֲיֵלִ֔יל וּלְמוֹאָ֥ב כֻּלֹּ֖ה אֶזְעָ֑ק אֶל־ אַנְשֵׁ֥י קִֽיר־ חֶ֖רֶשׂ יֶהְגֶּֽה׃ (מא) מִבְּכִ֨י יַעְזֵ֤ר אֶבְכֶּה־ לָּךְ֙ הַגֶּ֣פֶן שִׂבְמָ֔ה נְטִֽישֹׁתַ֙יִךְ֙ עָ֣בְרוּ יָ֔ם עַ֛ד יָ֥ם יַעְזֵ֖ר נָגָ֑עוּ עַל־ קֵיצֵ֥ךְ וְעַל־ בְּצִירֵ֖ךְ שֹׁדֵ֥ד נָפָֽל׃ (מב) וְנֶאֶסְפָ֨ה שִׂמְחָ֥ה וָגִ֛יל מִכַּרְמֶ֖ל וּמֵאֶ֣רֶץ מוֹאָ֑ב וְיַ֙יִן֙ מִיקָבִ֣ים הִשְׁבַּ֔תִּי לֹֽא־ יִדְרֹ֣ךְ הֵידָ֔ד הֵידָ֖ד לֹ֥א הֵידָֽד׃ (מג) מִזַּעֲקַ֨ת חֶשְׁבּ֜וֹן עַד־ אֶלְעָלֵ֗ה עַד־ יַ֙הַץ֙ נָתְנ֣וּ קוֹלָ֔ם מִצֹּ֙עַר֙ עַד־ חֹ֣רֹנַ֔יִם עֶגְלַ֖ת שְׁלִֽשִׁיָּ֑ה כִּ֚י גַּם־ מֵ֣י נִמְרִ֔ים לִמְשַׁמּ֖וֹת יִהְיֽוּ׃ (מד) וְהִשְׁבַּתִּ֥י לְמוֹאָ֖ב נְאֻם־ יְהוָ֑ה מַעֲלֶ֣ה בָמָ֔ה וּמַקְטִ֖יר לֵאלֹהָֽיו׃ (מה) עַל־ כֵּ֞ן לִבִּ֤י לְמוֹאָב֙ כַּחֲלִלִ֣ים (א) לָכֵ֡ן הִנֵּה֩ יָמִ֨ים בָּאִ֜ים נְאֻם־ יְהוָ֗ה וְ֠הִשְׁמַעְתִּי אֶל־ רַבַּ֨ת בְּנֵי־ עַמּ֜וֹן תְּרוּעַ֣ת מִלְחָמָ֗ה וְהָֽיְתָה֙ לְתֵ֣ל שְׁמָמָ֔ה וּבְנֹתֶ֖יהָ בָּאֵ֣שׁ תִּצַּ֑תְנָה וְיָרַ֧שׁ יִשְׂרָאֵ֛ל אֶת־ יֹרְשָׁ֖יו אָמַ֥ר יְהוָֽה׃ (ב) הֵילִ֨ילִי חֶשְׁבּ֜וֹן כִּ֣י שֻׁדְּדָה־ עַ֗י צְעַקְנָה֮ בְּנ֣וֹת רַבָּה֒ חֲגֹ֣רְנָה שַׂקִּ֔ים סְפֹ֕דְנָה וְהִתְשׁוֹטַ֖טְנָה בַּגְּדֵר֑וֹת כִּ֤י מַלְכָּם֙ בַּגּוֹלָ֣ה יֵלֵ֔ךְ כֹּהֲנָ֥יו וְשָׂרָ֖יו יַחְדָּֽיו׃ (ג) מַה־ תִּתְהַֽלְלִי֙ בָּֽעֲמָקִ֔ים זָ֣ב עִמְקֵ֔ךְ הַבַּ֖ת הַשּֽׁוֹבֵבָ֑ה הַבֹּֽטְחָה֙ בְּאֹ֣צְרֹתֶ֔יהָ מִ֖י יָב֥וֹא אֵלָֽי׃ (ד) הִנְנִי֩ מֵבִ֨יא עָלַ֜יִךְ פַּ֗חַד נְאֻם־ אֲדֹנָ֧י יְהוִ֛ה צְבָא֖וֹת מִכָּל־ סְבִיבָ֑יִךְ וְנִדַּחְתֶּם֙ אִ֣ישׁ לְפָנָ֔יו וְאֵ֥ין מְקַבֵּ֖ץ לַנֹּדֵֽד׃ (ה) וְאַחֲרֵי־ כֵ֗ן אָשִׁ֛יב אֶת־ שְׁב֥וּת בְּנֵֽי־ עַמּ֖וֹן נְאֻם־ יְהוָֽה׃ (ו) לֶאֱד֗וֹם כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת הַאֵ֥ין ע֛וֹד חָכְמָ֖ה בְּתֵימָ֑ן אָבְדָ֤ה עֵצָה֙ מִבָּנִ֔ים נִסְרְחָ֖ה חָכְמָתָֽם׃ (ז) נֻ֤סוּ הָפְנוּ֙ הֶעְמִ֣יקוּ לָשֶׁ֔בֶת יֹשְׁבֵ֖י דְּדָ֑ן כִּ֣י אֵ֥יד עֵשָׂ֛ו הֵבֵ֥אתִי עָלָ֖יו עֵ֥ת פְּקַדְתִּֽיו׃ (ח) אִם־ בֹּֽצְרִים֙ בָּ֣אוּ לָ֔ךְ לֹ֥א יַשְׁאִ֖רוּ עֽוֹלֵל֑וֹת אִם־ גַּנָּבִ֥ים בַּלַּ֖יְלָה הִשְׁחִ֥יתוּ קוֹלָֽהּ׃ (כא) הִנֵּ֤ה כַנֶּ֙שֶׁר֙ יַעֲלֶ֣ה וְיִדְאֶ֔ה וְיִפְרֹ֥שׂ כְּנָפָ֖יו עַל־ בָּצְרָ֑ה וְֽ֠הָיָה לֵ֞ב גִּבּוֹרֵ֤י אֱדוֹם֙ בַּיּ֣וֹם הַה֔וּא כְּלֵ֖ב אִשָּׁ֥ה מְצֵרָֽה׃ (כב) לְדַמֶּ֗שֶׂק אֲחָזַ֖תָּה כַּיּוֹלֵדָֽה׃ (כד) אֵ֥יךְ לֹֽא־ עֻזְּבָ֖ה עִ֣יר (תהלה) תְּהִלָּ֑ת קִרְיַ֖ת מְשׂוֹשִֽׂי׃ (כה) לָכֵ֛ן יִפְּל֥וּ בַחוּרֶ֖יהָ בִּרְחֹבֹתֶ֑יהָ וְכָל־ אַנְשֵׁ֨י הַמִּלְחָמָ֤ה יִדַּ֙מּוּ֙ בַּיּ֣וֹם הַה֔וּא נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כו) וְהִצַּ֥תִּי אֵ֖שׁ בְּחוֹמַ֣ת דַּמָּ֑שֶׂק וְאָכְלָ֖ה אַרְמְנ֥וֹת בֶּן־ הֲדָֽד׃ (כז) לְקֵדָ֣ר ׀ וּֽלְמַמְלְכ֣וֹת חָצ֗וֹר אֲשֶׁ֤ר הִכָּה֙ (נבוכדראצור) נְבֽוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֔ל כֹּ֖ה אָמַ֣ר יְהוָ֑ה ק֚וּמוּ עֲל֣וּ אֶל־ קֵדָ֔ר וְשָׁדְד֖וּ אֶת־ בְּנֵי־ קֶֽדֶם׃ (כח) אָהֳלֵיהֶ֤ם וְצֹאנָם֙ יִקָּ֔חוּ יְרִיעוֹתֵיהֶ֧ם וְכָל־ כְּלֵיהֶ֛ם וּגְמַלֵּיהֶ֖ם יִשְׂא֣וּ לָהֶ֑ם וְקָרְא֧וּ עֲלֵיהֶ֛ם מָג֖וֹר מִסָּבִֽיב׃ (כט) נֻסוּ֩ נֻּ֨דוּ מְאֹ֜ד הֶעְמִ֧יקוּ לָשֶׁ֛בֶת יֹשְׁבֵ֥י חָצ֖וֹר נְאֻם־ יְהוָ֑ה כִּֽי־ יָעַ֨ץ עֲלֵיכֶ֜ם נְבוּכַדְרֶאצַּ֤ר מֶֽלֶךְ־ בָּבֶל֙ עֵצָ֔ה וְחָשַׁ֥ב (עליהם) עֲלֵיכֶ֖ם מַחֲשָׁבָֽה׃ (ל) ק֣וּמוּ עֲל֗וּ אֶל־ גּ֥וֹי שְׁלֵ֛יו יוֹשֵׁ֥ב לָבֶ֖טַח נְאֻם־ יְהוָ֑ה לֹא־ דְלָתַ֧יִם וְלֹֽא־ בְרִ֛יחַ ל֖וֹ בָּדָ֥ד יִשְׁכֹּֽנוּ׃ (לא) וְהָי֨וּ גְמַלֵּיהֶ֜ם לָבַ֗ז וַהֲמ֤וֹן מִקְנֵיהֶם֙ לְשָׁלָ֔ל וְזֵרִתִ֥ים לְכָל־ ר֖וּחַ קְצוּצֵ֣י פֵאָ֑ה וּמִכָּל־ עֲבָרָ֛יו אָבִ֥יא אֶת־ אֵידָ֖ם נְאֻם־ יְהוָֽה׃ (לב) וְהָיְתָ֨ה חָצ֜וֹר לִמְע֥וֹן תַּנִּ֛ים שְׁמָמָ֖ה עַד־ עוֹלָ֑ם לֹֽא־ יֵשֵׁ֥ב שָׁם֙ אִ֔ישׁ וְלֹֽא־ יָג֥וּר בָּ֖הּ בֶּן־ אָדָֽם׃ (לג) אֲשֶׁ֨ר הָיָ֧ה דְבַר־ יְהוָ֛ה אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא אֶל־ עֵילָ֑ם בְּרֵאשִׁ֗ית מַלְכ֛וּת צִדְקִיָּ֥ה מֶֽלֶךְ־ יְהוּדָ֖ה לֵאמֹֽר׃ (לד) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת הִנְנִ֥י שֹׁבֵ֖ר אֶת־ קֶ֣שֶׁת עֵילָ֑ם רֵאשִׁ֖ית גְּבוּרָתָֽם׃ (לה) וְהֵבֵאתִ֨י אֶל־ עֵילָ֜ם אַרְבַּ֣ע רוּח֗וֹת מֵֽאַרְבַּע֙ קְצ֣וֹת הַשָּׁמַ֔יִם וְזֵ֣רִתִ֔ים לְכֹ֖ל הָרֻח֣וֹת הָאֵ֑לֶּה וְלֹֽא־ יִהְיֶ֣ה הַגּ֔וֹי אֲשֶׁ֛ר לֹֽא־ יָב֥וֹא שָׁ֖ם נִדְּחֵ֥י (עולם) עֵילָֽם׃ (לו) וְהַחְתַּתִּ֣י אֶת־ עֵ֠ילָם לִפְנֵ֨י אֹיְבֵיהֶ֜ם וְלִפְנֵ֣י ׀ מְבַקְשֵׁ֣י נַפְשָׁ֗ם וְהֵבֵאתִ֨י עֲלֵיהֶ֧ם ׀ רָעָ֛ה אֶת־ חֲר֥וֹן אַפִּ֖י נְאֻם־ יְהוָ֑ה וְשִׁלַּחְתִּ֤י אַֽחֲרֵיהֶם֙ אֶת־ הַחֶ֔רֶב עַ֥ד כַּלּוֹתִ֖י אוֹתָֽם׃ (לז) וְשַׂמְתִּ֥י כִסְאִ֖י בְּעֵילָ֑ם וְהַאֲבַדְתִּ֥י מִשָּׁ֛ם מֶ֥לֶךְ וְשָׂרִ֖ים נְאֻם־ יְהוָֽה׃ (לח) וְהָיָ֣ה ׀ בְּאַחֲרִ֣ית הַיָּמִ֗ים (אשוב) אָשִׁ֛יב אֶת־ (שבית) שְׁב֥וּת עֵילָ֖ם נְאֻם־ יְהוָֽה׃ סוף פרק מט (לט) הַדָּבָ֗ר אֲשֶׁ֨ר דִּבֶּ֧ר יְהוָ֛ה אֶל־ בָּבֶ֖ל אֶל־ אֶ֣רֶץ כַּשְׂדִּ֑ים בְּיַ֖ד יִרְמְיָ֥הוּ הַנָּבִֽיא׃ (מ) הַגִּ֨ידוּ בַגּוֹיִ֤ם וְהַשְׁמִ֙יעוּ֙ וּֽשְׂאוּ־ נֵ֔ס הַשְׁמִ֖יעוּ אַל־ תְּכַחֵ֑דוּ אִמְרוּ֩ נִלְכְּדָ֨ה בָבֶ֜ל הֹבִ֥ישׁ בֵּל֙ חַ֣ת מְרֹדָ֔ךְ הֹבִ֣ישׁוּ עֲצַבֶּ֔יהָ חַ֖תּוּ גִּלּוּלֶֽיהָ׃ (מא) כִּ֣י עָלָה֩ עָלֶ֨יהָ גּ֜וֹי מִצָּפ֗וֹן הֽוּא־ יָשִׁ֤ית אֶת־ אַרְצָהּ֙ לְשַׁמָּ֔ה וְלֹֽא־ יִהְיֶ֥ה יוֹשֵׁ֖ב בָּ֑הּ מֵאָדָ֥ם וְעַד־ בְּהֵמָ֖ה נָ֥דוּ הָלָֽכוּ׃ (מב) בַּיָּמִ֨ים הָהֵ֜מָּה וּבָעֵ֤ת הַהִיא֙ נְאֻם־ יְהוָ֔ה יָבֹ֧אוּ בְנֵֽי־ יִשְׂרָאֵ֛ל הֵ֥מָּה וּבְנֵֽי־ יְהוּדָ֖ה יַחְדָּ֑ו הָל֤וֹךְ וּבָכוֹ֙ יֵלֵ֔כוּ וְאֶת־ יְהוָ֥ה אֱלֹהֵיהֶ֖ם יְבַקֵּֽשׁוּ׃
