ירמיהו פרק ג לפי החלוקה לסדרים

פרק ג בספר ירמיהו (25 פסוקים) נמצא בירמיהו סדר א (פסוקים נז–נט) ובירמיהו סדר ב (פסוקים א–כב).

ירמיהו סדר א

(א) דִּבְרֵ֥י יִרְמְיָ֖הוּ בֶּן־ חִלְקִיָּ֑הוּ מִן־ הַכֹּֽהֲנִים֙ אֲשֶׁ֣ר בַּעֲנָת֔וֹת בְּאֶ֖רֶץ בִּנְיָמִֽן׃ (ב) אֲשֶׁ֨ר הָיָ֤ה דְבַר־ יְהוָה֙ אֵלָ֔יו בִּימֵ֛י יֹאשִׁיָּ֥הוּ בֶן־ אָמ֖וֹן מֶ֣לֶךְ יְהוּדָ֑ה בִּשְׁלֹשׁ־ עֶשְׂרֵ֥ה שָׁנָ֖ה לְמָלְכֽוֹ׃ (ג) וַיְהִ֗י בִּימֵ֨י יְהוֹיָקִ֤ים בֶּן־ יֹאשִׁיָּ֙הוּ֙ מֶ֣לֶךְ יְהוּדָ֔ה עַד־ תֹּם֙ עַשְׁתֵּ֣י עֶשְׂרֵ֣ה שָׁנָ֔ה לְצִדְקִיָּ֥הוּ בֶן־ יֹאשִׁיָּ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה עַד־ גְּל֥וֹת יְרוּשָׁלִַ֖ם בַּחֹ֥דֶשׁ הַחֲמִישִֽׁי׃ (ד) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ה) בְּטֶ֨רֶם (אצורך) אֶצָּרְךָ֤ בַבֶּ֙טֶן֙ יְדַעְתִּ֔יךָ וּבְטֶ֛רֶם תֵּצֵ֥א מֵרֶ֖חֶם הִקְדַּשְׁתִּ֑יךָ נָבִ֥יא לַגּוֹיִ֖ם נְתַתִּֽיךָ׃ (ו) וָאֹמַ֗ר אֲהָהּ֙ אֲדֹנָ֣י יְהֹוִ֔ה הִנֵּ֥ה לֹא־ יָדַ֖עְתִּי דַּבֵּ֑ר כִּי־ נַ֖עַר אָנֹֽכִי׃ (ז) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י אַל־ תֹּאמַ֖ר נַ֣עַר אָנֹ֑כִי כִּ֠י עַֽל־ כָּל־ אֲשֶׁ֤ר אֶֽשְׁלָחֲךָ֙ תֵּלֵ֔ךְ וְאֵ֛ת כָּל־ אֲשֶׁ֥ר אֲצַוְּךָ֖ תְּדַבֵּֽר׃ (ח) אַל־ תִּירָ֖א מִפְּנֵיהֶ֑ם כִּֽי־ אִתְּךָ֥ אֲנִ֛י לְהַצִּלֶ֖ךָ נְאֻם־ יְהוָֽה׃ (ט) וַיִּשְׁלַ֤ח יְהוָה֙ אֶת־ יָד֔וֹ וַיַּגַּ֖ע עַל־ פִּ֑י וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י הִנֵּ֛ה נָתַ֥תִּי דְבָרַ֖י בְּפִֽיךָ׃ (י) רְאֵ֞ה הִפְקַדְתִּ֣יךָ ׀ הַיּ֣וֹם הַזֶּ֗ה עַל־ הַגּוֹיִם֙ וְעַל־ הַמַּמְלָכ֔וֹת לִנְת֥וֹשׁ וְלִנְת֖וֹץ וּלְהַאֲבִ֣יד וְלַהֲר֑וֹס לִבְנ֖וֹת וְלִנְטֽוֹעַ׃ (יא) וַיְהִ֤י דְבַר־ יְהוָה֙ אֵלַ֣י לֵאמֹ֔ר מָה־ אַתָּ֥ה רֹאֶ֖ה יִרְמְיָ֑הוּ וָאֹמַ֕ר מַקֵּ֥ל שָׁקֵ֖ד אֲנִ֥י רֹאֶֽה׃ (יב) וַיֹּ֧אמֶר יְהוָ֛ה אֵלַ֖י הֵיטַ֣בְתָּ לִרְא֑וֹת כִּֽי־ שֹׁקֵ֥ד אֲנִ֛י עַל־ דְּבָרִ֖י לַעֲשֹׂתֽוֹ׃ (יג) וַיְהִ֨י דְבַר־ יְהוָ֤ה ׀ אֵלַי֙ שֵׁנִ֣ית לֵאמֹ֔ר מָ֥ה אַתָּ֖ה רֹאֶ֑ה וָאֹמַ֗ר סִ֤יר נָפ֙וּחַ֙ אֲנִ֣י רֹאֶ֔ה וּפָנָ֖יו מִפְּנֵ֥י צָפֽוֹנָה׃ (יד) וַיֹּ֥אמֶר יְהוָ֖ה אֵלָ֑י מִצָּפוֹן֙ תִּפָּתַ֣ח הָרָעָ֔ה עַ֥ל כָּל־ יֹשְׁבֵ֖י הָאָֽרֶץ׃ (טו) כִּ֣י ׀ הִנְנִ֣י קֹרֵ֗א לְכָֽל־ מִשְׁפְּח֛וֹת מַמְלְכ֥וֹת צָפ֖וֹנָה נְאֻם־ יְהוָ֑ה וּבָ֡אוּ וְֽנָתְנוּ֩ אִ֨ישׁ כִּסְא֜וֹ פֶּ֣תַח ׀ שַׁעֲרֵ֣י יְרוּשָׁלִַ֗ם וְעַ֤ל כָּל־ חוֹמֹתֶ֙יהָ֙ סָבִ֔יב וְעַ֖ל כָּל־ עָרֵ֥י יְהוּדָֽה׃ (טז) וְדִבַּרְתִּ֤י מִשְׁפָּטַי֙ אוֹתָ֔ם עַ֖ל כָּל־ רָעָתָ֑ם אֲשֶׁ֣ר עֲזָב֗וּנִי וַֽיְקַטְּרוּ֙ לֵאלֹהִ֣ים אֲחֵרִ֔ים וַיִּֽשְׁתַּחֲו֖וּ לְמַעֲשֵׂ֥י יְדֵיהֶֽם׃ (יז) וְאַתָּה֙ תֶּאְזֹ֣ר מָתְנֶ֔יךָ וְקַמְתָּ֙ וְדִבַּרְתָּ֣ אֲלֵיהֶ֔ם אֵ֛ת כָּל־ אֲשֶׁ֥ר אָנֹכִ֖י אֲצַוֶּ֑ךָּ אַל־ תֵּחַת֙ מִפְּנֵיהֶ֔ם פֶּֽן־ אֲחִתְּךָ֖ לִפְנֵיהֶֽם׃ (יח) וַאֲנִ֞י הִנֵּ֧ה נְתַתִּ֣יךָ הַיּ֗וֹם לְעִ֨יר מִבְצָ֜ר וּלְעַמּ֥וּד בַּרְזֶ֛ל וּלְחֹמ֥וֹת נְחֹ֖שֶׁת עַל־ כָּל־ הָאָ֑רֶץ לְמַלְכֵ֤י יְהוּדָה֙ לְשָׂרֶ֔יהָ לְכֹהֲנֶ֖יהָ וּלְעַ֥ם הָאָֽרֶץ׃ (יט) וְנִלְחֲמ֥וּ אֵלֶ֖יךָ וְלֹא־ י֣וּכְלוּ לָ֑ךְ כִּֽי־ אִתְּךָ֥ אֲנִ֛י נְאֻם־ יְהוָ֖ה לְהַצִּילֶֽךָ׃ (כ) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כא) הָלֹ֡ךְ וְקָֽרָאתָ֩ בְאָזְנֵ֨י יְרוּשָׁלִַ֜ם לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה זָכַ֤רְתִּי לָךְ֙ חֶ֣סֶד נְעוּרַ֔יִךְ אַהֲבַ֖ת כְּלוּלֹתָ֑יִךְ לֶכְתֵּ֤ךְ אַחֲרַי֙ בַּמִּדְבָּ֔ר בְּאֶ֖רֶץ לֹ֥א זְרוּעָֽה׃ (כב) קֹ֤דֶשׁ יִשְׂרָאֵל֙ לַיהוָ֔ה רֵאשִׁ֖ית תְּבוּאָתֹ֑ה כָּל־ אֹכְלָ֣יו יֶאְשָׁ֔מוּ רָעָ֛ה תָּבֹ֥א אֲלֵיהֶ֖ם נְאֻם־ יְהוָֽה׃ (כג) שִׁמְע֥וּ דְבַר־ יְהוָ֖ה בֵּ֣ית יַעֲקֹ֑ב וְכָֽל־ מִשְׁפְּח֖וֹת בֵּ֥ית יִשְׂרָאֵֽל׃ (כד) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה מַה־ מָּצְא֨וּ אֲבוֹתֵיכֶ֥ם בִּי֙ עָ֔וֶל כִּ֥י רָחֲק֖וּ מֵעָלָ֑י וַיֵּֽלְכ֛וּ אַחֲרֵ֥י הַהֶ֖בֶל וַיֶּהְבָּֽלוּ׃ (כה) וְלֹ֣א אָמְר֔וּ אַיֵּ֣ה יְהוָ֔ה הַמַּעֲלֶ֥ה אֹתָ֖נוּ מֵאֶ֣רֶץ מִצְרָ֑יִם הַמּוֹלִ֨יךְ אֹתָ֜נוּ בַּמִּדְבָּ֗ר בְּאֶ֨רֶץ עֲרָבָ֤ה וְשׁוּחָה֙ בְּאֶ֙רֶץ֙ צִיָּ֣ה וְצַלְמָ֔וֶת בְּאֶ֗רֶץ לֹֽא־ עָ֤בַר בָּהּ֙ אִ֔ישׁ וְלֹֽא־ יָשַׁ֥ב אָדָ֖ם שָֽׁם׃ (כו) וָאָבִ֤יא אֶתְכֶם֙ אֶל־ אֶ֣רֶץ הַכַּרְמֶ֔ל לֶאֱכֹ֥ל פִּרְיָ֖הּ וְטוּבָ֑הּ וַתָּבֹ֙אוּ֙ וַתְּטַמְּא֣וּ אֶת־ אַרְצִ֔י וְנַחֲלָתִ֥י שַׂמְתֶּ֖ם לְתוֹעֵבָֽה׃ (כז) הַכֹּהֲנִ֗ים לֹ֤א אָֽמְרוּ֙ אַיֵּ֣ה יְהוָ֔ה וְתֹפְשֵׂ֤י הַתּוֹרָה֙ לֹ֣א יְדָע֔וּנִי וְהָרֹעִ֖ים פָּ֣שְׁעוּ בִ֑י וְהַנְּבִיאִים֙ נִבְּא֣וּ בַבַּ֔עַל וְאַחֲרֵ֥י לֹֽא־ יוֹעִ֖לוּ הָלָֽכוּ׃ (כח) לָכֵ֗ן עֹ֛ד אָרִ֥יב אִתְּכֶ֖ם נְאֻם־ יְהוָ֑ה וְאֶת־ בְּנֵ֥י בְנֵיכֶ֖ם אָרִֽיב׃ (כט) כִּ֣י עִבְר֞וּ אִיֵּ֤י כִתִּיִּים֙ וּרְא֔וּ וְקֵדָ֛ר שִׁלְח֥וּ וְהִֽתְבּוֹנְנ֖וּ מְאֹ֑ד וּרְא֕וּ הֵ֥ן הָיְתָ֖ה כָּזֹֽאת׃ (ל) הַהֵימִ֥יר גּוֹי֙ אֱלֹהִ֔ים וְהֵ֖מָּה לֹ֣א אֱלֹהִ֑ים וְעַמִּ֛י הֵמִ֥יר כְּבוֹד֖וֹ בְּל֥וֹא יוֹעִֽיל׃ (לא) שֹׁ֥מּוּ שָׁמַ֖יִם עַל־ זֹ֑את וְשַׂעֲר֛וּ חָרְב֥וּ מְאֹ֖ד נְאֻם־ יְהוָֽה׃ (לב) כִּֽי־ שְׁתַּ֥יִם רָע֖וֹת עָשָׂ֣ה עַמִּ֑י אֹתִ֨י עָזְב֜וּ מְק֣וֹר ׀ מַ֣יִם חַיִּ֗ים לַחְצֹ֤ב לָהֶם֙ בֹּאר֔וֹת בֹּארֹת֙ נִשְׁבָּרִ֔ים אֲשֶׁ֥ר לֹא־ יָכִ֖לוּ הַמָּֽיִם׃ (לג) הַעֶ֙בֶד֙ יִשְׂרָאֵ֔ל אִם־ יְלִ֥יד בַּ֖יִת ה֑וּא מַדּ֖וּעַ הָיָ֥ה לָבַֽז׃ (לד) עָלָיו֙ יִשְׁאֲג֣וּ כְפִרִ֔ים נָתְנ֖וּ קוֹלָ֑ם וַיָּשִׁ֤יתוּ אַרְצוֹ֙ לְשַׁמָּ֔ה עָרָ֥יו (נצתה) נִצְּת֖וּ מִבְּלִ֥י יֹשֵֽׁב׃ (לה) גַּם־ בְּנֵי־ נֹ֖ף (ותחפנס) וְתַחְפַּנְחֵ֑ס יִרְע֖וּךְ קָדְקֹֽד׃ (לו) הֲלוֹא־ זֹ֖את תַּעֲשֶׂה־ לָּ֑ךְ עָזְבֵךְ֙ אֶת־ יְהוָ֣ה אֱלֹהַ֔יִךְ בְּעֵ֖ת מוֹלִיכֵ֥ךְ בַּדָּֽרֶךְ׃ (לז) וְעַתָּ֗ה מַה־ לָּךְ֙ לְדֶ֣רֶךְ מִצְרַ֔יִם לִשְׁתּ֖וֹת מֵ֣י שִׁח֑וֹר וּמַה־ לָּךְ֙ לְדֶ֣רֶךְ אַשּׁ֔וּר לִשְׁתּ֖וֹת מֵ֥י נָהָֽר׃ (לח) תְּיַסְּרֵ֣ךְ רָעָתֵ֗ךְ וּמְשֻֽׁבוֹתַ֙יִךְ֙ תּוֹכִחֻ֔ךְ וּדְעִ֤י וּרְאִי֙ כִּי־ רַ֣ע וָמָ֔ר עָזְבֵ֖ךְ אֶת־ יְהוָ֣ה אֱלֹהָ֑יִךְ וְלֹ֤א פַחְדָּתִי֙ אֵלַ֔יִךְ נְאֻם־ אֲדֹנָ֥י יְהוִ֖ה צְבָאֽוֹת׃ (לט) כִּ֣י מֵעוֹלָ֞ם שָׁבַ֣רְתִּי עֻלֵּ֗ךְ נִתַּ֙קְתִּי֙ מוֹסְרֹתַ֔יִךְ וַתֹּאמְרִ֖י לֹ֣א (אעבד) אֶעֱב֑וֹר כִּ֣י עַֽל־ כָּל־ גִּבְעָ֞ה גְּבֹהָ֗ה וְתַ֙חַת֙ כָּל־ עֵ֣ץ רַעֲנָ֔ן אַ֖תְּ צֹעָ֥ה זֹנָֽה׃ (מ) וְאָֽנֹכִי֙ נְטַעְתִּ֣יךְ שֹׂרֵ֔ק כֻּלֹּ֖ה זֶ֣רַע אֱמֶ֑ת וְאֵיךְ֙ נֶהְפַּ֣כְתְּ לִ֔י סוּרֵ֖י הַגֶּ֥פֶן נָכְרִיָּֽה׃ (מא) כִּ֤י אִם־ תְּכַבְּסִי֙ בַּנֶּ֔תֶר וְתַרְבִּי־ לָ֖ךְ בֹּרִ֑ית נִכְתָּ֤ם עֲוֺנֵךְ֙ לְפָנַ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (מב) אֵ֣יךְ תֹּאמְרִ֞י לֹ֣א אַחֵ֗ר הֲיָשׁ֤וּב אֵלֶ֙יהָ֙ ע֔וֹד הֲל֛וֹא חָנ֥וֹף תֶּחֱנַ֖ף הָאָ֣רֶץ הַהִ֑יא וְאַ֗תְּ זָנִית֙ רֵעִ֣ים רַבִּ֔ים וְשׁ֥וֹב אֵלַ֖י נְאֻם־ יְהֹוָֽה׃ (נח) שְׂאִֽי־ עֵינַ֨יִךְ עַל־ שְׁפָיִ֜ם וּרְאִ֗י אֵיפֹה֙ שָֽׁם׃ (ד) וָאֹמַ֗ר אַחֲרֵ֨י עֲשׂוֹתָ֧הּ אֶת־ כָּל־ אֵ֛לֶּה אֵלַ֥י תָּשׁ֖וּב וְלֹא־ שָׁ֑בָה (ותראה) וַתֵּ֛רֶא בָּגוֹדָ֥ה אֲחוֹתָ֖הּ יְהוּדָֽה׃ (ה) וָאֵ֗רֶא כִּ֤י עַל־ כָּל־ אֹדוֹת֙ אֲשֶׁ֤ר נִֽאֲפָה֙ מְשֻׁבָ֣ה יִשְׂרָאֵ֔ל שִׁלַּחְתִּ֕יהָ וָאֶתֵּ֛ן אֶת־ סֵ֥פֶר כְּרִיתֻתֶ֖יהָ אֵלֶ֑יהָ וְלֹ֨א יָֽרְאָ֜ה בֹּֽגֵדָ֤ה יְהוּדָה֙ אֲחוֹתָ֔הּ וַתֵּ֖לֶךְ וַתִּ֥זֶן גַּם־ הִֽיא׃ (ו) וְהָיָה֙ מִקֹּ֣ל זְנוּתָ֔הּ וַתֶּחֱנַ֖ף אֶת־ הָאָ֑רֶץ וַתִּנְאַ֥ף אֶת־ הָאֶ֖בֶן וְאֶת־ הָעֵֽץ׃ (ז) וְגַם־ בְּכָל־ זֹ֗את לֹא־ שָׁ֨בָה אֵלַ֜י בָּגוֹדָ֧ה אֲחוֹתָ֛הּ יְהוּדָ֖ה בְּכָל־ לִבָּ֑הּ כִּ֥י אִם־ בְּשֶׁ֖קֶר נְאֻם־ יְהוָֽה׃ (ח) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י צִדְּקָ֥ה נַפְשָׁ֖הּ מְשֻׁבָ֣ה יִשְׂרָאֵ֑ל מִבֹּגֵדָ֖ה יְהוּדָֽה׃ (ט) הָלֹ֡ךְ וְקָֽרָאתָ֩ אֶת־ הַדְּבָרִ֨ים הָאֵ֜לֶּה צָפ֗וֹנָה וְ֠אָמַרְתָּ שׁ֣וּבָה מְשֻׁבָ֤ה אָמַ֗רְתִּי אֵ֚יךְ אֲשִׁיתֵ֣ךְ בַּבָּנִ֔ים וְאֶתֶּן־ לָךְ֙ אֶ֣רֶץ חֶמְדָּ֔ה נַחֲלַ֥ת צְבִ֖י צִבְא֣וֹת גּוֹיִ֑ם וָאֹמַ֗ר אָבִי֙ (תקראו)־ תִּקְרְאִי־ לִ֔י וּמֵאַחֲרַ֖י לֹ֥א (תשובו) תָשֽׁוּבִי׃ (יז) אָכֵ֛ן בָּגְדָ֥ה אִשָּׁ֖ה מֵרֵעָ֑הּ כֵּ֣ן בְּגַדְתֶּ֥ם בִּ֛י בֵּ֥ית יִשְׂרָאֵ֖ל נְאֻם־ יְהוָֽה׃ (יח) ק֚וֹל עַל־ שְׁפָיִ֣ים נִשְׁמָ֔ע בְּכִ֥י תַחֲנוּנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל כִּ֤י הֶעֱוּוּ֙ אֶת־ דַּרְכָּ֔ם שָׁכְח֖וּ אֶת־ יְהֹוָ֥ה אֱלֹהֵיהֶֽם׃ (יט) שׁ֚וּבוּ בָּנִ֣ים שׁוֹבָבִ֔ים אֶרְפָּ֖ה מְשׁוּבֹֽתֵיכֶ֑ם הִנְנוּ֙ אָתָ֣נוּ לָ֔ךְ כִּ֥י אַתָּ֖ה יְהֹוָ֥ה אֱלֹהֵֽינוּ׃ (כ) אָכֵ֥ן לַשֶּׁ֛קֶר מִגְּבָע֖וֹת הָמ֣וֹן הָרִ֑ים אָכֵן֙ בַּיהֹוָ֣ה אֱלֹהֵ֔ינוּ תְּשׁוּעַ֖ת יִשְׂרָאֵֽל׃ (כא) וְהַבֹּ֗שֶׁת אָֽכְלָ֛ה אֶת־ יְגִ֥יעַ אֲבוֹתֵ֖ינוּ מִנְּעוּרֵ֑ינוּ אֶת־ צֹאנָם֙ וְאֶת־ בְּקָרָ֔ם אֶת־ בְּנֵיהֶ֖ם וְאֶת־ בְּנוֹתֵיהֶֽם׃ (כב) נִשְׁכְּבָ֣ה בְּבָשְׁתֵּ֗נוּ וּֽתְכַסֵּנוּ֮ כְּלִמָּתֵנוּ֒ כִּי֩ לַיהוָ֨ה אֱלֹהֵ֜ינוּ חָטָ֗אנוּ אֲנַ֙חְנוּ֙ וַאֲבוֹתֵ֔ינוּ מִנְּעוּרֵ֖ינוּ וְעַד־ הַיּ֣וֹם הַזֶּ֑ה וְלֹ֣א שָׁמַ֔עְנוּ בְּק֖וֹל יְהֹוָ֥ה אֱלֹהֵֽינוּ׃ סוף פרק ג (כג) אִם־ תָּשׁ֨וּב יִשְׂרָאֵ֧ל ׀ נְאֻם־ יְהוָ֛ה אֵלַ֖י תָּשׁ֑וּב וְאִם־ תָּסִ֧יר שִׁקּוּצֶ֛יךָ מִפָּנַ֖י וְלֹ֥א תָנֽוּד׃ (כד) וְנִשְׁבַּ֙עְתָּ֙ חַי־ יְהוָ֔ה בֶּאֱמֶ֖ת בְּמִשְׁפָּ֣ט וּבִצְדָקָ֑ה וְהִתְבָּ֥רְכוּ ב֛וֹ גּוֹיִ֖ם וּב֥וֹ יִתְהַלָּֽלוּ׃ (כה) כִּי־ כֹ֣ה ׀ אָמַ֣ר יְהֹוָ֗ה לְאִ֤ישׁ יְהוּדָה֙ וְלִיר֣וּשָׁלִַ֔ם נִ֥ירוּ לָכֶ֖ם נִ֑יר וְאַֽל־ תִּזְרְע֖וּ אֶל־ קוֹצִֽים׃ (כו) הִמֹּ֣לוּ לַיהֹוָ֗ה וְהָסִ֙רוּ֙ עָרְל֣וֹת לְבַבְכֶ֔ם אִ֥ישׁ יְהוּדָ֖ה וְיֹשְׁבֵ֣י יְרוּשָׁלִָ֑ם פֶּן־ תֵּצֵ֨א כָאֵ֜שׁ חֲמָתִ֗י וּבָעֲרָה֙ וְאֵ֣ין מְכַבֶּ֔ה מִפְּנֵ֖י רֹ֥עַ מַעַלְלֵיכֶֽם׃ (כז) הַגִּ֣ידוּ בִֽיהוּדָ֗ה וּבִירוּשָׁלִַ֙ם֙ הַשְׁמִ֔יעוּ וְאִמְר֕וּ (ותקעו) תִּקְע֥וּ שׁוֹפָ֖ר בָּאָ֑רֶץ קִרְא֤וּ מַלְאוּ֙ וְאִמְר֔וּ הֵאָסְפ֥וּ וְנָב֖וֹאָה אֶל־ עָרֵ֥י הַמִּבְצָֽר׃ (כח) שְׂאוּ־ נֵ֣ס צִיּ֔וֹנָה הָעִ֖יזוּ אַֽל־ תַּעֲמֹ֑דוּ כִּ֣י רָעָ֗ה אָנֹכִ֛י מֵבִ֥יא מִצָּפ֖וֹן וְשֶׁ֥בֶר גָּדֽוֹל׃ (כט) עָלָ֤ה אַרְיֵה֙ מִֽסֻּבְּכ֔וֹ וּמַשְׁחִ֣ית גּוֹיִ֔ם נָסַ֖ע יָצָ֣א מִמְּקֹמ֑וֹ לָשׂ֤וּם אַרְצֵךְ֙ לְשַׁמָּ֔ה עָרַ֥יִךְ תִּצֶּ֖ינָה מֵאֵ֥ין יוֹשֵֽׁב׃ (ל) עַל־ זֹ֛את חִגְר֥וּ שַׂקִּ֖ים סִפְד֣וּ וְהֵילִ֑ילוּ כִּ֥י לֹא־ שָׁ֛ב חֲר֥וֹן אַף־ יְהֹוָ֖ה מִמֶּֽנּוּ׃ (לא) וְהָיָ֤ה בַיּוֹם־ הַהוּא֙ נְאֻם־ יְהוָ֔ה יֹאבַ֥ד לֵב־ הַמֶּ֖לֶךְ וְלֵ֣ב הַשָּׂרִ֑ים וְנָשַׁ֙מּוּ֙ הַכֹּ֣הֲנִ֔ים וְהַנְּבִיאִ֖ים יִתְמָֽהוּ׃ (לב) וָאֹמַ֞ר אֲהָ֣הּ ׀ אֲדֹנָ֣י יְהוִ֗ה אָכֵן֩ הַשֵּׁ֨א הִשֵּׁ֜אתָ לָעָ֤ם הַזֶּה֙ וְלִירוּשָׁלִַ֣ם לֵאמֹ֔ר שָׁל֖וֹם יִהְיֶ֣ה לָכֶ֑ם וְנָגְעָ֥ה חֶ֖רֶב עַד־ הַנָּֽפֶשׁ׃ (לג) בָּעֵ֣ת הַהִ֗יא יֵאָמֵ֤ר לָֽעָם־ הַזֶּה֙ וְלִיר֣וּשָׁלִַ֔ם ר֣וּחַ צַ֤ח שְׁפָיִים֙ בַּמִּדְבָּ֔ר דֶּ֖רֶךְ בַּת־ עַמִּ֑י ל֥וֹא לִזְר֖וֹת וְל֥וֹא לְהָבַֽר׃ (לד) ר֧וּחַ מָלֵ֛א מֵאֵ֖לֶּה יָ֣בוֹא לִ֑י עַתָּ֕ה גַּם־ אֲנִ֛י אֲדַבֵּ֥ר מִשְׁפָּטִ֖ים אוֹתָֽם׃ (לה) הִנֵּ֣ה ׀ כַּעֲנָנִ֣ים יַעֲלֶ֗ה וְכַסּוּפָה֙ מַרְכְּבוֹתָ֔יו קַלּ֥וּ מִנְּשָׁרִ֖ים סוּסָ֑יו א֥וֹי לָ֖נוּ כִּ֥י שֻׁדָּֽדְנוּ׃ (לו) כַּבְּסִ֨י מֵרָעָ֤ה לִבֵּךְ֙ יְר֣וּשָׁלִַ֔ם לְמַ֖עַן תִּוָּשֵׁ֑עִי עַד־ מָתַ֛י תָּלִ֥ין בְּקִרְבֵּ֖ךְ מַחְשְׁב֥וֹת אוֹנֵֽךְ׃ (לז) כִּ֛י ק֥וֹל מַגִּ֖יד מִדָּ֑ן וּמַשְׁמִ֥יעַ אָ֖וֶן מֵהַ֥ר אֶפְרָֽיִם׃ (לח) הַזְכִּ֣ירוּ לַגּוֹיִ֗ם הִנֵּה֙ הַשְׁמִ֣יעוּ עַל־ יְרוּשָׁלִַ֔ם נֹצְרִ֥ים בָּאִ֖ים מֵאֶ֣רֶץ הַמֶּרְחָ֑ק וַֽיִּתְּנ֛וּ עַל־ עָרֵ֥י יְהוּדָ֖ה קוֹלָֽם׃ (לט) כְּשֹׁמְרֵ֣י שָׂדַ֔י הָי֥וּ עָלֶ֖יהָ מִסָּבִ֑יב כִּי־ אֹתִ֥י מָרָ֖תָה נְאֻם־ יְהוָֽה׃ (מ) דַּרְכֵּךְ֙ וּמַ֣עֲלָלַ֔יִךְ עָשׂ֥וֹ אֵ֖לֶּה לָ֑ךְ זֹ֤את רָעָתֵךְ֙ כִּ֣י מָ֔ר כִּ֥י נָגַ֖ע עַד־ לִבֵּֽךְ׃ (מא) מֵעַ֣י ׀ מֵעַ֨י ׀ (אחולה) אוֹחִ֜ילָה קִיר֥וֹת לִבִּ֛י הֹֽמֶה־ לִּ֥י לִבִּ֖י לֹ֣א אַחֲרִ֑ישׁ כִּ֣י ק֤וֹל שׁוֹפָר֙ (שמעתי) שָׁמַ֣עַתְּ נַפְשִׁ֔י תְּרוּעַ֖ת מִלְחָמָֽה׃ (מב) שֶׁ֤בֶר עַל־ שֶׁ֙בֶר֙ נִקְרָ֔א כִּ֥י שֻׁדְּדָ֖ה כָּל־ הָאָ֑רֶץ פִּתְאֹם֙ שֻׁדְּד֣וּ אֹהָלַ֔י רֶ֖גַע יְרִיעֹתָֽי׃ (מג) עַד־ מָתַ֖י אֶרְאֶה־ נֵּ֑ס אֶשְׁמְעָ֖ה ק֥וֹל שׁוֹפָֽר׃ (מד) כִּ֣י ׀ אֱוִ֣יל עַמִּ֗י אוֹתִי֙ לֹ֣א יָדָ֔עוּ בָּנִ֤ים סְכָלִים֙ הֵ֔מָּה וְלֹ֥א נְבוֹנִ֖ים הֵ֑מָּה חֲכָמִ֥ים הֵ֙מָּה֙ לְהָרַ֔ע וּלְהֵיטִ֖יב לֹ֥א יָדָֽעוּ׃ (מה) רָאִ֙יתִי֙ אֶת־ הָאָ֔רֶץ וְהִנֵּה־ תֹ֖הוּ וָבֹ֑הוּ וְאֶל־ הַשָּׁמַ֖יִם וְאֵ֥ין אוֹרָֽם׃ (מו) רָאִ֙יתִי֙ הֶֽהָרִ֔ים וְהִנֵּ֖ה רֹעֲשִׁ֑ים וְכָל־ הַגְּבָע֖וֹת הִתְקַלְקָֽלוּ׃ (מז) רָאִ֕יתִי וְהִנֵּ֖ה אֵ֣ין הָאָדָ֑ם וְכָל־ ע֥וֹף הַשָּׁמַ֖יִם נָדָֽדוּ׃ (מח) רָאִ֕יתִי וְהִנֵּ֥ה הַכַּרְמֶ֖ל הַמִּדְבָּ֑ר וְכָל־ עָרָ֗יו נִתְּצוּ֙ מִפְּנֵ֣י יְהוָ֔ה מִפְּנֵ֖י חֲר֥וֹן אַפּֽוֹ׃ (מט) כִּי־ כֹה֙ אָמַ֣ר יְהוָ֔ה שְׁמָמָ֥ה תִהְיֶ֖ה כָּל־ הָאָ֑רֶץ וְכָלָ֖ה לֹ֥א אֶעֱשֶֽׂה׃ (נ) עַל־ זֹאת֙ תֶּאֱבַ֣ל הָאָ֔רֶץ וְקָדְר֥וּ הַשָּׁמַ֖יִם מִמָּ֑עַל עַ֤ל כִּי־ דִבַּ֙רְתִּי֙ זַמֹּ֔תִי וְלֹ֥א נִחַ֖מְתִּי וְלֹא־ אָשׁ֥וּב מִמֶּֽנָּה׃ (נא) מִקּ֨וֹל פָּרָ֜שׁ וְרֹ֣מֵה קֶ֗שֶׁת בֹּרַ֙חַת֙ כָּל־ הָעִ֔יר בָּ֚אוּ בֶּעָבִ֔ים וּבַכֵּפִ֖ים עָל֑וּ כָּל־ הָעִ֣יר עֲזוּבָ֔ה וְאֵין־ יוֹשֵׁ֥ב בָּהֵ֖ן אִֽישׁ׃ (נב) (ואתי) וְאַ֨תְּ שָׁד֜וּד מַֽה־ תַּעֲשִׂ֗י כִּֽי־ תִלְבְּשִׁ֨י שָׁנִ֜י כִּי־ תַעְדִּ֣י עֲדִי־ זָהָ֗ב כִּֽי־ תִקְרְעִ֤י בַפּוּךְ֙ עֵינַ֔יִךְ לַשָּׁ֖וְא תִּתְיַפִּ֑י מָאֲסוּ־ בָ֥ךְ עֹגְבִ֖ים נַפְשֵׁ֥ךְ יְבַקֵּֽשׁוּ׃ (נג) כִּי֩ ק֨וֹל כְּחוֹלָ֜ה שָׁמַ֗עְתִּי צָרָה֙ כְּמַבְכִּירָ֔ה ק֧וֹל בַּת־ צִיּ֛וֹן תִּתְיַפֵּ֖חַ תְּפָרֵ֣שׂ כַּפֶּ֑יהָ אֽוֹי־ נָ֣א לִ֔י כִּֽי־ עָיְפָ֥ה נַפְשִׁ֖י לְהֹרְגִֽים׃
