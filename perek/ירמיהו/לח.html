ירמיהו פרק לח לפי החלוקה לסדרים

פרק לח בספר ירמיהו (28 פסוקים) נמצא בירמיהו סדר כב (פסוקים כט–לה) ובירמיהו סדר כג (פסוקים א–כא).

ירמיהו סדר כב

(א) וַיְצַוֶּ֣ה הַ֠מֶּלֶךְ אֶת־ יְרַחְמְאֵ֨ל בֶּן־ הַמֶּ֜לֶךְ וְאֶת־ שְׂרָיָ֣הוּ בֶן־ עַזְרִיאֵ֗ל וְאֶת־ שֶֽׁלֶמְיָ֙הוּ֙ בֶּֽן־ עַבְדְּאֵ֔ל לָקַ֙חַת֙ אֶת־ בָּר֣וּךְ הַסֹּפֵ֔ר וְאֵ֖ת יִרְמְיָ֣הוּ הַנָּבִ֑יא וַיַּסְתִּרֵ֖ם יְהוָֽה׃ (ב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֶֽל־ יִרְמְיָ֑הוּ אַחֲרֵ֣י ׀ שְׂרֹ֣ף הַמֶּ֗לֶךְ אֶת־ הַמְּגִלָּה֙ וְאֶת־ הַדְּבָרִ֔ים אֲשֶׁ֨ר כָּתַ֥ב בָּר֛וּךְ מִפִּ֥י יִרְמְיָ֖הוּ לֵאמֹֽר׃ (ג) שׁ֥וּב קַח־ לְךָ֖ מְגִלָּ֣ה אַחֶ֑רֶת וּכְתֹ֣ב עָלֶ֗יהָ אֵ֤ת כָּל־ הַדְּבָרִים֙ הָרִ֣אשֹׁנִ֔ים אֲשֶׁ֣ר הָי֗וּ עַל־ הַמְּגִלָּה֙ הָרִ֣אשֹׁנָ֔ה אֲשֶׁ֥ר שָׂרַ֖ף יְהוֹיָקִ֥ים מֶֽלֶךְ־ יְהוּדָֽה׃ (ד) וְעַל־ יְהוֹיָקִ֤ים מֶֽלֶךְ־ יְהוּדָה֙ תֹּאמַ֔ר כֹּ֖ה אָמַ֣ר יְהוָ֑ה אַ֠תָּה שָׂרַ֜פְתָּ אֶת־ הַמְּגִלָּ֤ה הַזֹּאת֙ לֵאמֹ֔ר מַדּוּעַ֩ כָּתַ֨בְתָּ עָלֶ֜יהָ לֵאמֹ֗ר בֹּֽא־ יָב֤וֹא מֶֽלֶךְ־ בָּבֶל֙ וְהִשְׁחִית֙ אֶת־ הָאָ֣רֶץ הַזֹּ֔את וְהִשְׁבִּ֥ית מִמֶּ֖נָּה אָדָ֥ם וּבְהֵמָֽה׃ (ה) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה עַל־ יְהֽוֹיָקִים֙ מֶ֣לֶךְ יְהוּדָ֔ה לֹא־ יִֽהְיֶה־ לּ֥וֹ יוֹשֵׁ֖ב עַל־ כִּסֵּ֣א דָוִ֑ד וְנִבְלָתוֹ֙ תִּֽהְיֶ֣ה מֻשְׁלֶ֔כֶת לַחֹ֥רֶב בַּיּ֖וֹם וְלַקֶּ֥רַח בַּלָּֽיְלָה׃ (ו) וּפָקַדְתִּ֨י עָלָ֧יו וְעַל־ זַרְע֛וֹ וְעַל־ עֲבָדָ֖יו אֶת־ עֲוֺנָ֑ם וְהֵבֵאתִ֣י עֲ֠לֵיהֶם וְעַל־ יֹשְׁבֵ֨י יְרוּשָׁלִַ֜ם וְאֶל־ אִ֣ישׁ יְהוּדָ֗ה אֵ֧ת כָּל־ הָרָעָ֛ה אֲשֶׁר־ דִּבַּ֥רְתִּי אֲלֵיהֶ֖ם וְלֹ֥א שָׁמֵֽעוּ׃ (ז) וְיִרְמְיָ֜הוּ לָקַ֣ח ׀ מְגִלָּ֣ה אַחֶ֗רֶת וַֽיִּתְּנָהּ֮ אֶל־ בָּר֣וּךְ בֶּן־ נֵרִיָּהוּ֮ הַסֹּפֵר֒ וַיִּכְתֹּ֤ב עָלֶ֙יהָ֙ מִפִּ֣י יִרְמְיָ֔הוּ אֵ֚ת כָּל־ דִּבְרֵ֣י הַסֵּ֔פֶר אֲשֶׁ֥ר שָׂרַ֛ף יְהוֹיָקִ֥ים מֶֽלֶךְ־ יְהוּדָ֖ה בָּאֵ֑שׁ וְע֨וֹד נוֹסַ֧ף עֲלֵיהֶ֛ם דְּבָרִ֥ים רַבִּ֖ים כָּהֵֽמָּה׃ (ח) וַיִּ֨מְלָךְ־ מֶ֔לֶךְ צִדְקִיָּ֖הוּ בֶּן־ יֹֽאשִׁיָּ֑הוּ תַּ֗חַת כָּנְיָ֙הוּ֙ בֶּן־ יְה֣וֹיָקִ֔ים אֲשֶׁ֥ר הִמְלִ֛יךְ נְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל בְּאֶ֥רֶץ יְהוּדָֽה׃ (ט) וְלֹ֥א שָׁמַ֛ע ה֥וּא וַעֲבָדָ֖יו וְעַ֣ם הָאָ֑רֶץ אֶל־ דִּבְרֵ֣י יְהוָ֔ה אֲשֶׁ֣ר דִּבֶּ֔ר בְּיַ֖ד יִרְמְיָ֥הוּ הַנָּבִֽיא׃ (י) וַיִּשְׁלַח֩ הַמֶּ֨לֶךְ צִדְקִיָּ֜הוּ אֶת־ יְהוּכַ֣ל בֶּן־ שֶֽׁלֶמְיָ֗ה וְאֶת־ צְפַנְיָ֤הוּ בֶן־ מַֽעֲשֵׂיָה֙ הַכֹּהֵ֔ן אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא לֵאמֹ֑ר הִתְפַּלֶּל־ נָ֣א בַעֲדֵ֔נוּ אֶל־ יְהוָ֖ה אֱלֹהֵֽינוּ׃ (יא) וְיִרְמְיָ֕הוּ בָּ֥א וְיֹצֵ֖א בְּת֣וֹךְ הָעָ֑ם וְלֹֽא־ נָתְנ֥וּ אֹת֖וֹ בֵּ֥ית (הכליא) הַכְּלֽוּא׃ (יב) וְחֵ֥יל פַּרְעֹ֖ה יָצָ֣א מִמִּצְרָ֑יִם וַיִּשְׁמְע֨וּ הַכַּשְׂדִּ֜ים הַצָּרִ֤ים עַל־ יְרוּשָׁלִַ֙ם֙ אֶת־ שִׁמְעָ֔ם וַיֵּ֣עָל֔וּ מֵעַ֖ל יְרוּשָׁלִָֽם׃ (יג) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא לֵאמֹֽר׃ (יד) כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל כֹּ֤ה תֹֽאמְרוּ֙ אֶל־ מֶ֣לֶךְ יְהוּדָ֔ה הַשֹּׁלֵ֧חַ אֶתְכֶ֛ם אֵלַ֖י לְדָרְשֵׁ֑נִי הִנֵּ֣ה ׀ חֵ֣יל פַּרְעֹ֗ה הַיֹּצֵ֤א לָכֶם֙ לְעֶזְרָ֔ה שָׁ֥ב לְאַרְצ֖וֹ מִצְרָֽיִם׃ (טו) וְשָׁ֙בוּ֙ הַכַּשְׂדִּ֔ים וְנִלְחֲמ֖וּ עַל־ הָעִ֣יר הַזֹּ֑את וּלְכָדֻ֖הָ וּשְׂרָפֻ֥הָ בָאֵֽשׁ׃ (טז) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ תַּשִּׁ֤אוּ נַפְשֹֽׁתֵיכֶם֙ לֵאמֹ֔ר הָלֹ֛ךְ יֵלְכ֥וּ מֵעָלֵ֖ינוּ הַכַּשְׂדִּ֑ים כִּי־ לֹ֖א יֵלֵֽכוּ׃ (יז) כִּ֣י אִם־ הִכִּיתֶ֞ם כָּל־ חֵ֤יל כַּשְׂדִּים֙ הַנִּלְחָמִ֣ים אִתְּכֶ֔ם וְנִ֨שְׁאֲרוּ בָ֔ם אֲנָשִׁ֖ים מְדֻקָּרִ֑ים אִ֤ישׁ בְּאָהֳלוֹ֙ יָק֔וּמוּ וְשָֽׂרְפ֛וּ אֶת־ הָעִ֥יר הַזֹּ֖את בָּאֵֽשׁ׃ (יח) וְהָיָ֗ה בְּהֵֽעָלוֹת֙ חֵ֣יל הַכַּשְׂדִּ֔ים מֵעַ֖ל יְרֽוּשָׁלִָ֑ם מִפְּנֵ֖י חֵ֥יל פַּרְעֹֽה׃ (יט) וַיֵּצֵ֤א יִרְמְיָ֙הוּ֙ מִיר֣וּשָׁלִַ֔ם לָלֶ֖כֶת אֶ֣רֶץ בִּנְיָמִ֑ן לַחֲלִ֥ק מִשָּׁ֖ם בְּת֥וֹךְ הָעָֽם׃ (כ) וַיְהִי־ ה֞וּא בְּשַׁ֣עַר בִּנְיָמִ֗ן וְשָׁם֙ בַּ֣עַל פְּקִדֻ֔ת וּשְׁמוֹ֙ יִרְאִיָּ֔יה בֶּן־ שֶֽׁלֶמְיָ֖ה בֶּן־ חֲנַנְיָ֑ה וַיִּתְפֹּ֞שׂ אֶֽת־ יִרְמְיָ֤הוּ הַנָּבִיא֙ לֵאמֹ֔ר אֶל־ הַכַּשְׂדִּ֖ים אַתָּ֥ה נֹפֵֽל׃ (כא) וַיֹּ֨אמֶר יִרְמְיָ֜הוּ שֶׁ֗קֶר אֵינֶ֤נִּי נֹפֵל֙ עַל־ הַכַּשְׂדִּ֔ים וְלֹ֥א שָׁמַ֖ע אֵלָ֑יו וַיִּתְפֹּ֤שׂ יִרְאִיָּיה֙ בְּיִרְמְיָ֔הוּ וַיְבִאֵ֖הוּ אֶל־ הַשָּׂרִֽים׃ (כב) וַיִּקְצְפ֧וּ הַשָּׂרִ֛ים עַֽל־ יִרְמְיָ֖הוּ וְהִכּ֣וּ אֹת֑וֹ וְנָתְנ֨וּ אוֹת֜וֹ בֵּ֣ית הָאֵס֗וּר בֵּ֚ית יְהוֹנָתָ֣ן הַסֹּפֵ֔ר כִּֽי־ אֹת֥וֹ עָשׂ֖וּ לְבֵ֥ית הַכֶּֽלֶא׃ (כג) כִּ֣י בָ֧א יִרְמְיָ֛הוּ אֶל־ בֵּ֥ית הַבּ֖וֹר וְאֶל־ הַֽחֲנֻ֑יוֹת וַיֵּֽשֶׁב־ שָׁ֥ם יִרְמְיָ֖הוּ יָמִ֥ים רַבִּֽים׃ (כד) וַיִּשְׁלַח֩ הַמֶּ֨לֶךְ צִדְקִיָּ֜הוּ וַיִּקָּחֵ֗הוּ וַיִּשְׁאָלֵ֨הוּ הַמֶּ֤לֶךְ בְּבֵיתוֹ֙ בַּסֵּ֔תֶר וַיֹּ֕אמֶר הֲיֵ֥שׁ דָּבָ֖ר מֵאֵ֣ת יְהוָ֑ה וַיֹּ֤אמֶר יִרְמְיָ֙הוּ֙ יֵ֔שׁ וַיֹּ֕אמֶר בְּיַ֥ד מֶֽלֶךְ־ בָּבֶ֖ל תִּנָּתֵֽן׃ (כה) וַיֹּ֣אמֶר יִרְמְיָ֔הוּ אֶל־ הַמֶּ֖לֶךְ צִדְקִיָּ֑הוּ מֶה֩ חָטָ֨אתִֽי לְךָ֤ וְלַעֲבָדֶ֙יךָ֙ וְלָעָ֣ם הַזֶּ֔ה כִּֽי־ נְתַתֶּ֥ם אוֹתִ֖י אֶל־ בֵּ֥ית הַכֶּֽלֶא׃ (כו) (ואיו) וְאַיֵּה֙ נְבִ֣יאֵיכֶ֔ם אֲשֶׁר־ נִבְּא֥וּ לָכֶ֖ם לֵאמֹ֑ר לֹֽא־ יָבֹ֤א מֶֽלֶךְ־ בָּבֶל֙ עֲלֵיכֶ֔ם וְעַ֖ל הָאָ֥רֶץ הַזֹּֽאת׃ (כז) וְעַתָּ֕ה שְֽׁמַֽע־ נָ֖א אֲדֹנִ֣י הַמֶּ֑לֶךְ תִּפָּל־ נָ֤א תְחִנָּתִי֙ לְפָנֶ֔יךָ וְאַל־ תְּשִׁבֵ֗נִי בֵּ֚ית יְהוֹנָתָ֣ן הַסֹּפֵ֔ר וְלֹ֥א אָמ֖וּת שָֽׁם׃ (כח) וַיְצַוֶּ֞ה הַמֶּ֣לֶךְ צִדְקִיָּ֗הוּ וַיַּפְקִ֣דוּ אֶֽת־ יִרְמְיָהוּ֮ בַּחֲצַ֣ר הַמַּטָּרָה֒ וְנָתֹן֩ ל֨וֹ כִכַּר־ לֶ֤חֶם לַיּוֹם֙ מִח֣וּץ הָאֹפִ֔ים עַד־ תֹּ֥ם כָּל־ הַלֶּ֖חֶם מִן־ הָעִ֑יר וַיֵּ֣שֶׁב יִרְמְיָ֔הוּ בַּחֲצַ֖ר הַמַּטָּרָֽה׃ תחילת פרק לח (כט) וַיִּשְׁמַ֞ע שְׁפַטְיָ֣ה בֶן־ מַתָּ֗ן וּגְדַלְיָ֙הוּ֙ בֶּן־ פַּשְׁח֔וּר וְיוּכַל֙ בֶּן־ שֶׁ֣לֶמְיָ֔הוּ וּפַשְׁח֖וּר בֶּן־ מַלְכִּיָּ֑ה אֶ֨ת־ הַדְּבָרִ֔ים אֲשֶׁ֧ר יִרְמְיָ֛הוּ מְדַבֵּ֥ר אֶל־ כָּל־ הָעָ֖ם לֵאמֹֽר׃ (ל) כֹּה֮ אָמַ֣ר יְהוָה֒ הַיֹּשֵׁב֙ בָּעִ֣יר הַזֹּ֔את יָמ֕וּת בַּחֶ֖רֶב בָּרָעָ֣ב וּבַדָּ֑בֶר וְהַיֹּצֵ֤א אֶל־ הַכַּשְׂדִּים֙ (יחיה) וְחָיָ֔ה וְהָיְתָה־ לּ֥וֹ נַפְשׁ֛וֹ לְשָׁלָ֖ל וָחָֽי׃ (לא) כֹּ֖ה אָמַ֣ר יְהוָ֑ה הִנָּתֹ֨ן תִּנָּתֵ֜ן הָעִ֣יר הַזֹּ֗את בְּיַ֛ד חֵ֥יל מֶֽלֶךְ־ בָּבֶ֖ל וּלְכָדָֽהּ׃ (לב) וַיֹּאמְר֨וּ הַשָּׂרִ֜ים אֶל־ הַמֶּ֗לֶךְ י֣וּמַת נָא֮ אֶת־ הָאִ֣ישׁ הַזֶּה֒ כִּֽי־ עַל־ כֵּ֡ן הֽוּא־ מְרַפֵּ֡א אֶת־ יְדֵי֩ אַנְשֵׁ֨י הַמִּלְחָמָ֜ה הַֽנִּשְׁאָרִ֣ים ׀ בָּעִ֣יר הַזֹּ֗את וְאֵת֙ יְדֵ֣י כָל־ הָעָ֔ם לְדַבֵּ֣ר אֲלֵיהֶ֔ם כַּדְּבָרִ֖ים הָאֵ֑לֶּה כִּ֣י ׀ הָאִ֣ישׁ הַזֶּ֗ה אֵינֶ֨נּוּ דֹרֵ֧שׁ לְשָׁל֛וֹם לָעָ֥ם הַזֶּ֖ה כִּ֥י אִם־ לְרָעָֽה׃ (לג) וַיֹּ֙אמֶר֙ הַמֶּ֣לֶךְ צִדְקִיָּ֔הוּ הִנֵּה־ ה֖וּא בְּיֶדְכֶ֑ם כִּֽי־ אֵ֣ין הַמֶּ֔לֶךְ יוּכַ֥ל אֶתְכֶ֖ם דָּבָֽר׃ (לד) וַיִּקְח֣וּ אֶֽת־ יִרְמְיָ֗הוּ וַיַּשְׁלִ֨כוּ אֹת֜וֹ אֶל־ הַבּ֣וֹר ׀ מַלְכִּיָּ֣הוּ בֶן־ הַמֶּ֗לֶךְ אֲשֶׁר֙ בַּחֲצַ֣ר הַמַּטָּרָ֔ה וַיְשַׁלְּח֥וּ אֶֽת־ יִרְמְיָ֖הוּ בַּחֲבָלִ֑ים וּבַבּ֤וֹר אֵֽין־ מַ֙יִם֙ כִּ֣י אִם־ טִ֔יט וַיִּטְבַּ֥ע יִרְמְיָ֖הוּ בַּטִּֽיט׃ (לה) וַיִּשְׁמַ֡ע עֶֽבֶד־ מֶ֨לֶךְ הַכּוּשִׁ֜י אִ֣ישׁ סָרִ֗יס וְהוּא֙ בְּבֵ֣ית הַמֶּ֔לֶךְ כִּֽי־ נָתְנ֥וּ אֶֽת־ יִרְמְיָ֖הוּ אֶל־ הַבּ֑וֹר וְהַמֶּ֥לֶךְ יוֹשֵׁ֖ב בְּשַׁ֥עַר בִּנְיָמִֽן׃

ירמיהו סדר כג

(א) וַיֵּצֵ֥א עֶֽבֶד־ מֶ֖לֶךְ מִבֵּ֣ית הַמֶּ֑לֶךְ וַיְדַבֵּ֥ר אֶל־ הַמֶּ֖לֶךְ לֵאמֹֽר׃ (ב) אֲדֹנִ֣י הַמֶּ֗לֶךְ הֵרֵ֜עוּ הָאֲנָשִׁ֤ים הָאֵ֙לֶּה֙ אֵ֣ת כָּל־ אֲשֶׁ֤ר עָשׂוּ֙ לְיִרְמְיָ֣הוּ הַנָּבִ֔יא אֵ֥ת אֲשֶׁר־ הִשְׁלִ֖יכוּ אֶל־ הַבּ֑וֹר וַיָּ֤מָת תַּחְתָּיו֙ מִפְּנֵ֣י הָֽרָעָ֔ב כִּ֣י אֵ֥ין הַלֶּ֛חֶם ע֖וֹד בָּעִֽיר׃ (ג) וַיְצַוֶּ֣ה הַמֶּ֔לֶךְ אֵ֛ת עֶֽבֶד־ מֶ֥לֶךְ הַכּוּשִׁ֖י לֵאמֹ֑ר קַ֣ח בְּיָדְךָ֤ מִזֶּה֙ שְׁלֹשִׁ֣ים אֲנָשִׁ֔ים וְֽהַעֲלִ֜יתָ אֶֽת־ יִרְמְיָ֧הוּ הַנָּבִ֛יא מִן־ הַבּ֖וֹר בְּטֶ֥רֶם יָמֽוּת׃ (ד) וַיִּקַּ֣ח ׀ עֶֽבֶד־ מֶ֨לֶךְ אֶת־ הָאֲנָשִׁ֜ים בְּיָד֗וֹ וַיָּבֹ֤א בֵית־ הַמֶּ֙לֶךְ֙ אֶל־ תַּ֣חַת הָאוֹצָ֔ר וַיִּקַּ֤ח מִשָּׁם֙ בְּלוֹיֵ֣ (הסחבות) סְחָב֔וֹת וּבְלוֹיֵ֖ מְלָחִ֑ים וַיְשַׁלְּחֵ֧ם אֶֽל־ יִרְמְיָ֛הוּ אֶל־ הַבּ֖וֹר בַּחֲבָלִֽים׃ (ה) וַיֹּ֡אמֶר עֶבֶד־ מֶ֨לֶךְ הַכּוּשִׁ֜י אֶֽל־ יִרְמְיָ֗הוּ שִׂ֣ים נָ֠א בְּלוֹאֵ֨י הַסְּחָב֤וֹת וְהַמְּלָחִים֙ תַּ֚חַת אַצִּל֣וֹת יָדֶ֔יךָ מִתַּ֖חַת לַחֲבָלִ֑ים וַיַּ֥עַשׂ יִרְמְיָ֖הוּ כֵּֽן׃ (ו) וַיִּמְשְׁכ֤וּ אֶֽת־ יִרְמְיָ֙הוּ֙ בַּֽחֲבָלִ֔ים וַיַּעֲל֥וּ אֹת֖וֹ מִן־ הַבּ֑וֹר וַיֵּ֣שֶׁב יִרְמְיָ֔הוּ בַּחֲצַ֖ר הַמַּטָּרָֽה׃ (ז) וַיִּשְׁלַ֞ח הַמֶּ֣לֶךְ צִדְקִיָּ֗הוּ וַיִּקַּ֞ח אֶֽת־ יִרְמְיָ֤הוּ הַנָּבִיא֙ אֵלָ֔יו אֶל־ מָבוֹא֙ הַשְּׁלִישִׁ֔י אֲשֶׁ֖ר בְּבֵ֣ית יְהוָ֑ה וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֶֽל־ יִרְמְיָ֗הוּ שֹׁאֵ֨ל אֲנִ֤י אֹֽתְךָ֙ דָּבָ֔ר אַל־ תְּכַחֵ֥ד מִמֶּ֖נִּי דָּבָֽר׃ (ח) וַיֹּ֤אמֶר יִרְמְיָ֙הוּ֙ אֶל־ צִדְקִיָּ֔הוּ כִּ֚י אַגִּ֣יד לְךָ֔ הֲל֖וֹא הָמֵ֣ת תְּמִיתֵ֑נִי וְכִי֙ אִיעָ֣צְךָ֔ לֹ֥א תִשְׁמַ֖ע אֵלָֽי׃ (ט) וַיִּשָּׁבַ֞ע הַמֶּ֧לֶךְ צִדְקִיָּ֛הוּ אֶֽל־ יִרְמְיָ֖הוּ בַּסֵּ֣תֶר לֵאמֹ֑ר חַי־ יְהוָ֞ה (את) אֲשֶׁר֩ עָשָׂה־ לָ֨נוּ אֶת־ הַנֶּ֤פֶשׁ הַזֹּאת֙ אִם־ אֲמִיתֶ֔ךָ וְאִם־ אֶתֶּנְךָ֗ בְּיַד֙ הָאֲנָשִׁ֣ים הָאֵ֔לֶּה אֲשֶׁ֥ר מְבַקְשִׁ֖ים אֶת־ נַפְשֶֽׁךָ׃ (י) וַיֹּ֣אמֶר יִרְמְיָ֣הוּ אֶל־ צִדְקִיָּ֡הוּ כֹּֽה־ אָמַ֣ר יְהוָה֩ אֱלֹהֵ֨י צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל אִם־ יָצֹ֨א תֵצֵ֜א אֶל־ שָׂרֵ֤י מֶֽלֶךְ־ בָּבֶל֙ וְחָיְתָ֣ה נַפְשֶׁ֔ךָ וְהָעִ֣יר הַזֹּ֔את לֹ֥א תִשָּׂרֵ֖ף בָּאֵ֑שׁ וְחָיִ֖תָה אַתָּ֥ה וּבֵיתֶֽךָ׃ (יא) וְאִ֣ם לֹֽא־ תֵצֵ֗א אֶל־ שָׂרֵי֙ מֶ֣לֶךְ בָּבֶ֔ל וְנִתְּנָ֞ה הָעִ֤יר הַזֹּאת֙ בְּיַ֣ד הַכַּשְׂדִּ֔ים וּשְׂרָפ֖וּהָ בָּאֵ֑שׁ וְאַתָּ֖ה לֹֽא־ תִמָּלֵ֥ט מִיָּדָֽם׃ (יב) וַיֹּ֛אמֶר הַמֶּ֥לֶךְ צִדְקִיָּ֖הוּ אֶֽל־ יִרְמְיָ֑הוּ אֲנִ֧י דֹאֵ֣ג אֶת־ הַיְּהוּדִ֗ים אֲשֶׁ֤ר נָֽפְלוּ֙ אֶל־ הַכַּשְׂדִּ֔ים פֶּֽן־ יִתְּנ֥וּ אֹתִ֛י בְּיָדָ֖ם וְהִתְעַלְּלוּ־ בִֽי׃ (יג) וַיֹּ֥אמֶר יִרְמְיָ֖הוּ לֹ֣א יִתֵּ֑נוּ שְֽׁמַֽע־ נָ֣א ׀ בְּק֣וֹל יְהוָ֗ה לַאֲשֶׁ֤ר אֲנִי֙ דֹּבֵ֣ר אֵלֶ֔יךָ וְיִ֥יטַב לְךָ֖ וּתְחִ֥י נַפְשֶֽׁךָ׃ (יד) וְאִם־ מָאֵ֥ן אַתָּ֖ה לָצֵ֑את זֶ֣ה הַדָּבָ֔ר אֲשֶׁ֥ר הִרְאַ֖נִי יְהוָֽה׃ (טו) וְהִנֵּ֣ה כָל־ הַנָּשִׁ֗ים אֲשֶׁ֤ר נִשְׁאֲרוּ֙ בְּבֵ֣ית מֶֽלֶךְ־ יְהוּדָ֔ה מוּצָא֕וֹת אֶל־ שָׂרֵ֖י מֶ֣לֶךְ בָּבֶ֑ל וְהֵ֣נָּה אֹמְר֗וֹת הִסִּית֜וּךָ וְיָכְל֤וּ לְךָ֙ אַנְשֵׁ֣י שְׁלֹמֶ֔ךָ הָטְבְּע֥וּ בַבֹּ֛ץ רַגְלֶ֖ךָ נָסֹ֥גוּ אָחֽוֹר׃ (טז) וְאֶת־ כָּל־ נָשֶׁ֣יךָ וְאֶת־ בָּנֶ֗יךָ מֽוֹצִאִים֙ אֶל־ הַכַּשְׂדִּ֔ים וְאַתָּ֖ה לֹא־ תִמָּלֵ֣ט מִיָּדָ֑ם כִּ֣י בְיַ֤ד מֶֽלֶךְ־ בָּבֶל֙ תִּתָּפֵ֔שׂ וְאֶת־ הָעִ֥יר הַזֹּ֖את תִּשְׂרֹ֥ף בָּאֵֽשׁ׃ (יז) וַיֹּ֨אמֶר צִדְקִיָּ֜הוּ אֶֽל־ יִרְמְיָ֗הוּ אִ֛ישׁ אַל־ יֵדַ֥ע בַּדְּבָֽרִים־ הָאֵ֖לֶּה וְלֹ֥א תָמֽוּת׃ (יח) וְכִֽי־ יִשְׁמְע֣וּ הַשָּׂרִים֮ כִּֽי־ דִבַּ֣רְתִּי אִתָּךְ֒ וּבָ֣אוּ אֵלֶ֣יךָ וְֽאָמְר֪וּ אֵלֶ֟יךָ הַגִּֽידָה־ נָּ֨א לָ֜נוּ מַה־ דִּבַּ֧רְתָּ אֶל־ הַמֶּ֛לֶךְ אַל־ תְּכַחֵ֥ד מִמֶּ֖נּוּ וְלֹ֣א נְמִיתֶ֑ךָ וּמַה־ דִּבֶּ֥ר אֵלֶ֖יךָ הַמֶּֽלֶךְ׃ (יט) וְאָמַרְתָּ֣ אֲלֵיהֶ֔ם מַפִּיל־ אֲנִ֥י תְחִנָּתִ֖י לִפְנֵ֣י הַמֶּ֑לֶךְ לְבִלְתִּ֧י הֲשִׁיבֵ֛נִי בֵּ֥ית יְהוֹנָתָ֖ן לָמ֥וּת שָֽׁם׃ (כ) וַיָּבֹ֨אוּ כָל־ הַשָּׂרִ֤ים אֶֽל־ יִרְמְיָ֙הוּ֙ וַיִּשְׁאֲל֣וּ אֹת֔וֹ וַיַּגֵּ֤ד לָהֶם֙ כְּכָל־ הַדְּבָרִ֣ים הָאֵ֔לֶּה אֲשֶׁ֥ר צִוָּ֖ה הַמֶּ֑לֶךְ וַיַּחֲרִ֣שׁוּ מִמֶּ֔נּוּ כִּ֥י לֹֽא־ נִשְׁמַ֖ע הַדָּבָֽר׃ (כא) וַיֵּ֤שֶׁב יִרְמְיָ֙הוּ֙ בַּחֲצַ֣ר הַמַּטָּרָ֔ה עַד־ י֖וֹם אֲשֶׁר־ נִלְכְּדָ֣ה יְרוּשָׁלִָ֑ם וְהָיָ֕ה כַּאֲשֶׁ֥ר נִלְכְּדָ֖ה יְרוּשָׁלִָֽם׃ סוף פרק לח (כב) בַּשָּׁנָ֣ה הַ֠תְּשִׁעִית לְצִדְקִיָּ֨הוּ מֶלֶךְ־ יְהוּדָ֜ה בַּחֹ֣דֶשׁ הָעֲשִׂרִ֗י בָּ֠א נְבוּכַדְרֶאצַּ֨ר מֶֽלֶךְ־ בָּבֶ֤ל וְכָל־ חֵילוֹ֙ אֶל־ יְר֣וּשָׁלִַ֔ם וַיָּצֻ֖רוּ עָלֶֽיהָ׃ (כג) בְּעַשְׁתֵּֽי־ עֶשְׂרֵ֤ה שָׁנָה֙ לְצִדְקִיָּ֔הוּ בַּחֹ֥דֶשׁ הָרְבִיעִ֖י בְּתִשְׁעָ֣ה לַחֹ֑דֶשׁ הָבְקְעָ֖ה הָעִֽיר׃ (כד) וַיָּבֹ֗אוּ כֹּ֚ל שָׂרֵ֣י מֶֽלֶךְ־ בָּבֶ֔ל וַיֵּשְׁב֖וּ בְּשַׁ֣עַר הַתָּ֑וֶךְ נֵרְגַ֣ל שַׂר־ אֶ֠צֶר סַֽמְגַּר־ נְב֞וּ שַׂר־ סְכִ֣ים רַב־ סָרִ֗יס נֵרְגַ֤ל שַׂר־ אֶ֙צֶר֙ רַב־ מָ֔ג וְכָל־ שְׁאֵרִ֔ית שָׂרֵ֖י מֶ֥לֶךְ בָּבֶֽל׃ (כה) וַיְהִ֡י כַּאֲשֶׁ֣ר רָ֠אָם צִדְקִיָּ֨הוּ מֶֽלֶךְ־ יְהוּדָ֜ה וְכֹ֣ל ׀ אַנְשֵׁ֣י הַמִּלְחָמָ֗ה וַֽ֠יִּבְרְחוּ וַיֵּצְא֨וּ לַ֤יְלָה מִן־ הָעִיר֙ דֶּ֚רֶךְ גַּ֣ן הַמֶּ֔לֶךְ בְּשַׁ֖עַר בֵּ֣ין הַחֹמֹתָ֑יִם וַיֵּצֵ֖א דֶּ֥רֶךְ הָעֲרָבָֽה׃ (כו) וַיִּרְדְּפ֨וּ חֵיל־ כַּשְׂדִּ֜ים אַחֲרֵיהֶ֗ם וַיַּשִּׂ֣גוּ אֶת־ צִדְקִיָּהוּ֮ בְּעַֽרְב֣וֹת יְרֵחוֹ֒ וַיִּקְח֣וּ אֹת֗וֹ וַֽ֠יַּעֲלֻהוּ אֶל־ נְבוּכַדְרֶאצַּ֧ר מֶֽלֶךְ־ בָּבֶ֛ל רִבְלָ֖תָה בְּאֶ֣רֶץ חֲמָ֑ת וַיְדַבֵּ֥ר אִתּ֖וֹ מִשְׁפָּטִֽים׃ (כז) וַיִּשְׁחַט֩ מֶ֨לֶךְ בָּבֶ֜ל אֶת־ בְּנֵ֧י צִדְקִיָּ֛הוּ בְּרִבְלָ֖ה לְעֵינָ֑יו וְאֵת֙ כָּל־ חֹרֵ֣י יְהוּדָ֔ה שָׁחַ֖ט מֶ֥לֶךְ בָּבֶֽל׃ (כח) וְאֶת־ עֵינֵ֥י צִדְקִיָּ֖הוּ עִוֵּ֑ר וַיַּאַסְרֵ֙הוּ֙ בַּֽנְחֻשְׁתַּ֔יִם לָבִ֥יא אֹת֖וֹ בָּבֶֽלָה׃ (כט) וְאֶת־ בֵּ֤ית הַמֶּ֙לֶךְ֙ וְאֶת־ בֵּ֣ית הָעָ֔ם שָׂרְפ֥וּ הַכַּשְׂדִּ֖ים בָּאֵ֑שׁ וְאֶת־ חֹמ֥וֹת יְרוּשָׁלִַ֖ם נָתָֽצוּ׃ (ל) וְאֵת֩ יֶ֨תֶר הָעָ֜ם הַנִּשְׁאָרִ֣ים בָּעִ֗יר וְאֶת־ הַנֹּֽפְלִים֙ אֲשֶׁ֣ר נָפְל֣וּ עָלָ֔יו וְאֵ֛ת יֶ֥תֶר הָעָ֖ם הַנִּשְׁאָרִ֑ים הֶגְלָ֛ה נְבֽוּזַר־ אֲדָ֥ן רַב־ טַבָּחִ֖ים בָּבֶֽל׃ (לא) וּמִן־ הָעָ֣ם הַדַּלִּ֗ים אֲשֶׁ֤ר אֵין־ לָהֶם֙ מְא֔וּמָה הִשְׁאִ֛יר נְבוּזַרְאֲדָ֥ן רַב־ טַבָּחִ֖ים בְּאֶ֣רֶץ יְהוּדָ֑ה וַיִּתֵּ֥ן לָהֶ֛ם כְּרָמִ֥ים וִֽיגֵבִ֖ים בַּיּ֥וֹם הַהֽוּא׃ (לב) וַיְצַ֛ו נְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל עַֽל־ יִרְמְיָ֑הוּ בְּיַ֛ד נְבוּזַרְאֲדָ֥ן רַב־ טַבָּחִ֖ים לֵאמֹֽר׃ (לג) קָחֶ֗נּוּ וְעֵינֶ֙יךָ֙ שִׂ֣ים עָלָ֔יו וְאַל־ תַּ֥עַשׂ ל֖וֹ מְא֣וּמָה רָּ֑ע כִּ֗י (אם) כַּֽאֲשֶׁר֙ יְדַבֵּ֣ר אֵלֶ֔יךָ כֵּ֖ן עֲשֵׂ֥ה עִמּֽוֹ׃ (לד) וַיִּשְׁלַ֞ח נְבֽוּזַרְאֲדָ֣ן רַב־ טַבָּחִ֗ים וּנְבֽוּשַׁזְבָּ רַב־ סָרִ֔יס וְנֵרְגַ֥ל שַׂר־ אֶ֖צֶר רַב־ מָ֑ג וְכֹ֖ל רַבֵּ֥י מֶֽלֶךְ־ בָּבֶֽל׃ (לה) וַיִּשְׁלְחוּ֩ וַיִּקְח֨וּ אֶֽת־ יִרְמְיָ֜הוּ מֵחֲצַ֣ר הַמַּטָּרָ֗ה וַיִּתְּנ֤וּ אֹתוֹ֙ אֶל־ גְּדַלְיָ֙הוּ֙ בֶּן־ אֲחִיקָ֣ם בֶּן־ שָׁפָ֔ן לְהוֹצִאֵ֖הוּ אֶל־ הַבָּ֑יִת וַיֵּ֖שֶׁב בְּת֥וֹךְ הָעָֽם׃ (לו) וְאֶֽל־ יִרְמְיָ֖הוּ הָיָ֣ה דְבַר־ יְהוָ֑ה בִּֽהְיֹת֣וֹ עָצ֔וּר בַּחֲצַ֥ר הַמַּטָּרָ֖ה לֵאמֹֽר׃ (לז) הָל֣וֹךְ וְאָמַרְתָּ֡ לְעֶבֶד־ מֶ֨לֶךְ הַכּוּשִׁ֜י לֵאמֹ֗ר כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִי֩ (מבי) מֵבִ֨יא אֶת־ דְּבָרַ֜י אֶל־ הָעִ֥יר הַזֹּ֛את לְרָעָ֖ה וְלֹ֣א לְטוֹבָ֑ה וְהָי֥וּ לְפָנֶ֖יךָ בַּיּ֥וֹם הַהֽוּא׃ (לח) וְהִצַּלְתִּ֥יךָ בַיּוֹם־ הַה֖וּא נְאֻם־ יְהוָ֑ה וְלֹ֤א תִנָּתֵן֙ בְּיַ֣ד הָֽאֲנָשִׁ֔ים אֲשֶׁר־ אַתָּ֥ה יָג֖וֹר מִפְּנֵיהֶֽם׃
