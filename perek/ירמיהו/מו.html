ירמיהו פרק מו לפי החלוקה לסדרים

פרק מו בספר ירמיהו (28 פסוקים) נמצא בירמיהו סדר כו (פסוקים יז–מב) ובירמיהו סדר כז (פסוקים א–ב).

ירמיהו סדר כו

(א) וַיֹּ֥אמֶר יִרְמְיָ֖הוּ אֶל־ כָּל־ הָעָ֑ם עַל־ הַגְּבָרִ֤ים וְעַל־ הַנָּשִׁים֙ וְעַל־ כָּל־ הָעָ֔ם הָעֹנִ֥ים אֹת֛וֹ דָּבָ֖ר לֵאמֹֽר׃ (ב) הֲל֣וֹא אֶת־ הַקִּטֵּ֗ר אֲשֶׁ֨ר קִטַּרְתֶּ֜ם בְּעָרֵ֤י יְהוּדָה֙ וּבְחֻצ֣וֹת יְרוּשָׁלִַ֔ם אַתֶּ֧ם וַאֲבֽוֹתֵיכֶ֛ם מַלְכֵיכֶ֥ם וְשָׂרֵיכֶ֖ם וְעַ֣ם הָאָ֑רֶץ אֹתָם֙ זָכַ֣ר יְהוָ֔ה וַֽתַּעֲלֶ֖ה עַל־ לִבּֽוֹ׃ (ג) וְלֹֽא־ יוּכַל֩ יְהוָ֨ה ע֜וֹד לָשֵׂ֗את מִפְּנֵי֙ רֹ֣עַ מַעַלְלֵיכֶ֔ם מִפְּנֵ֥י הַתּוֹעֵבֹ֖ת אֲשֶׁ֣ר עֲשִׂיתֶ֑ם וַתְּהִ֣י אַ֠רְצְכֶם לְחָרְבָּ֨ה וּלְשַׁמָּ֧ה וְלִקְלָלָ֛ה מֵאֵ֥ין יוֹשֵׁ֖ב כְּהַיּ֥וֹם הַזֶּֽה׃ (ד) מִפְּנֵי֩ אֲשֶׁ֨ר קִטַּרְתֶּ֜ם וַאֲשֶׁ֧ר חֲטָאתֶ֣ם לַיהוָ֗ה וְלֹ֤א שְׁמַעְתֶּם֙ בְּק֣וֹל יְהוָ֔ה וּבְתֹרָת֧וֹ וּבְחֻקֹּתָ֛יו וּבְעֵדְוֺתָ֖יו לֹ֣א הֲלַכְתֶּ֑ם עַל־ כֵּ֞ן קָרָ֥את אֶתְכֶ֛ם הָרָעָ֥ה הַזֹּ֖את כַּיּ֥וֹם הַזֶּֽה׃ (ה) וַיֹּ֤אמֶר יִרְמְיָ֙הוּ֙ אֶל־ כָּל־ הָעָ֔ם וְאֶ֖ל כָּל־ הַנָּשִׁ֑ים שִׁמְעוּ֙ דְּבַר־ יְהוָ֔ה כָּל־ יְהוּדָ֕ה אֲשֶׁ֖ר בְּאֶ֥רֶץ מִצְרָֽיִם׃ (ו) כֹּֽה־ אָמַ֣ר יְהוָֽה־ צְבָאוֹת֩ אֱלֹהֵ֨י יִשְׂרָאֵ֜ל לֵאמֹ֗ר אַתֶּ֨ם וּנְשֵׁיכֶ֜ם וַתְּדַבֵּ֣רְנָה בְּפִיכֶם֮ וּבִידֵיכֶ֣ם מִלֵּאתֶ֣ם ׀ לֵאמֹר֒ עָשֹׂ֨ה נַעֲשֶׂ֜ה אֶת־ נְדָרֵ֗ינוּ אֲשֶׁ֤ר נָדַ֙רְנוּ֙ לְקַטֵּר֙ לִמְלֶ֣כֶת הַשָּׁמַ֔יִם וּלְהַסֵּ֥ךְ לָ֖הּ נְסָכִ֑ים הָקֵ֤ים תָּקִ֙ימְנָה֙ אֶת־ נִדְרֵיכֶ֔ם וְעָשֹׂ֥ה תַעֲשֶׂ֖ינָה אֶת־ נִדְרֵיכֶֽם׃ (ז) לָכֵן֙ שִׁמְע֣וּ דְבַר־ יְהוָ֔ה כָּל־ יְהוּדָ֕ה הַיֹּשְׁבִ֖ים בְּאֶ֣רֶץ מִצְרָ֑יִם הִנְנִ֨י נִשְׁבַּ֜עְתִּי בִּשְׁמִ֤י הַגָּדוֹל֙ אָמַ֣ר יְהוָ֔ה אִם־ יִהְיֶה֩ ע֨וֹד שְׁמִ֜י נִקְרָ֣א ׀ בְּפִ֣י ׀ כָּל־ אִ֣ישׁ יְהוּדָ֗ה אֹמֵ֛ר חַי־ אֲדֹנָ֥י יְהוִ֖ה בְּכָל־ אֶ֥רֶץ מִצְרָֽיִם׃ (ח) הִנְנִ֨י שֹׁקֵ֧ד עֲלֵיהֶ֛ם לְרָעָ֖ה וְלֹ֣א לְטוֹבָ֑ה וְתַמּוּ֩ כָל־ אִ֨ישׁ יְהוּדָ֜ה אֲשֶׁ֧ר בְּאֶֽרֶץ־ מִצְרַ֛יִם בַּחֶ֥רֶב וּבָרָעָ֖ב עַד־ כְּלוֹתָֽם׃ (ט) וּפְלִיטֵ֨י חֶ֜רֶב יְשֻׁב֨וּן מִן־ אֶ֧רֶץ מִצְרַ֛יִם אֶ֥רֶץ יְהוּדָ֖ה מְתֵ֣י מִסְפָּ֑ר וְֽיָדְע֞וּ כָּל־ שְׁאֵרִ֣ית יְהוּדָ֗ה הַבָּאִ֤ים לְאֶֽרֶץ־ מִצְרַ֙יִם֙ לָג֣וּר שָׁ֔ם דְּבַר־ מִ֥י יָק֖וּם מִמֶּ֥נִּי וּמֵהֶֽם׃ (י) וְזֹאת־ לָכֶ֤ם הָאוֹת֙ נְאֻם־ יְהוָ֔ה כִּֽי־ פֹקֵ֥ד אֲנִ֛י עֲלֵיכֶ֖ם בַּמָּק֣וֹם הַזֶּ֑ה לְמַ֙עַן֙ תֵּֽדְע֔וּ כִּי֩ ק֨וֹם יָק֧וּמוּ דְבָרַ֛י עֲלֵיכֶ֖ם לְרָעָֽה׃ (יא) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה הִנְנִ֣י נֹ֠תֵן אֶת־ פַּרְעֹ֨ה חָפְרַ֤ע מֶֽלֶךְ־ מִצְרַ֙יִם֙ בְּיַ֣ד אֹֽיְבָ֔יו וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשׁ֑וֹ כַּאֲשֶׁ֨ר נָתַ֜תִּי אֶת־ צִדְקִיָּ֣הוּ מֶֽלֶךְ־ יְהוּדָ֗ה בְּיַ֨ד נְבוּכַדְרֶאצַּ֧ר מֶֽלֶךְ־ בָּבֶ֛ל אֹיְב֖וֹ וּמְבַקֵּ֥שׁ נַפְשֽׁוֹ׃ (יב) הַדָּבָ֗ר אֲשֶׁ֤ר דִּבֶּר֙ יִרְמְיָ֣הוּ הַנָּבִ֔יא אֶל־ בָּר֖וּךְ בֶּֽן־ נֵֽרִיָּ֑ה בְּכָתְבוֹ֩ אֶת־ הַדְּבָרִ֨ים הָאֵ֤לֶּה עַל־ סֵ֙פֶר֙ מִפִּ֣י יִרְמְיָ֔הוּ בַּשָּׁנָה֙ הָֽרְבִעִ֔ית לִיהוֹיָקִ֧ים בֶּן־ יֹאשִׁיָּ֛הוּ מֶ֥לֶךְ יְהוּדָ֖ה לֵאמֹֽר׃ (יג) כֹּֽה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל עָלֶ֖יךָ בָּרֽוּךְ׃ (יד) אָמַ֙רְתָּ֙ אֽוֹי־ נָ֣א לִ֔י כִּֽי־ יָסַ֧ף יְהוָ֛ה יָג֖וֹן עַל־ מַכְאֹבִ֑י יָגַ֙עְתִּי֙ בְּאַנְחָתִ֔י וּמְנוּחָ֖ה לֹ֥א מָצָֽאתִי׃ (טו) כֹּ֣ה ׀ תֹּאמַ֣ר אֵלָ֗יו כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנֵּ֤ה אֲשֶׁר־ בָּנִ֙יתִי֙ אֲנִ֣י הֹרֵ֔ס וְאֵ֥ת אֲשֶׁר־ נָטַ֖עְתִּי אֲנִ֣י נֹתֵ֑שׁ וְאֶת־ כָּל־ הָאָ֖רֶץ הִֽיא׃ (טז) וְאַתָּ֛ה תְּבַקֶּשׁ־ לְךָ֥ גְדֹל֖וֹת אַל־ תְּבַקֵּ֑שׁ כִּ֡י הִנְנִי֩ מֵבִ֨יא רָעָ֤ה עַל־ כָּל־ בָּשָׂר֙ נְאֻם־ יְהוָ֔ה וְנָתַתִּ֨י לְךָ֤ אֶֽת־ נַפְשְׁךָ֙ לְשָׁלָ֔ל עַ֥ל כָּל־ הַמְּקֹמ֖וֹת אֲשֶׁ֥ר תֵּֽלֶךְ־ שָֽׁם׃ תחילת פרק מו (יז) אֲשֶׁ֨ר הָיָ֧ה דְבַר־ יְהוָ֛ה אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא עַל־ הַגּוֹיִֽם׃ (יח) לְמִצְרַ֗יִם עַל־ חֵ֨יל פַּרְעֹ֤ה נְכוֹ֙ מֶ֣לֶךְ מִצְרַ֔יִם אֲשֶׁר־ הָיָ֥ה עַל־ נְהַר־ פְּרָ֖ת בְּכַרְכְּמִ֑שׁ אֲשֶׁ֣ר הִכָּ֗ה נְבֽוּכַדְרֶאצַּר֙ מֶ֣לֶךְ בָּבֶ֔ל בִּשְׁנַת֙ הָֽרְבִיעִ֔ית לִיהוֹיָקִ֥ים בֶּן־ יֹאשִׁיָּ֖הוּ מֶ֥לֶךְ יְהוּדָֽה׃ (יט) עִרְכ֤וּ מָגֵן֙ וְצִנָּ֔ה וּגְשׁ֖וּ לַמִּלְחָמָֽה׃ (כ) אִסְר֣וּ הַסּוּסִ֗ים וַֽעֲלוּ֙ הַפָּ֣רָשִׁ֔ים וְהִֽתְיַצְּב֖וּ בְּכ֥וֹבָעִ֑ים מִרְקוּ֙ הָֽרְמָחִ֔ים לִבְשׁ֖וּ הַסִּרְיֹנֹֽת׃ (כא) מַדּ֣וּעַ רָאִ֗יתִי הֵ֣מָּה חַתִּים֮ נְסֹגִ֣ים אָחוֹר֒ וְגִבּוֹרֵיהֶ֣ם יֻכַּ֔תּוּ וּמָנ֥וֹס נָ֖סוּ וְלֹ֣א הִפְנ֑וּ מָג֥וֹר מִסָּבִ֖יב נְאֻם־ יְהוָֽה׃ (כב) אַל־ יָנ֣וּס הַקַּ֔ל וְאַל־ יִמָּלֵ֖ט הַגִּבּ֑וֹר צָפ֙וֹנָה֙ עַל־ יַ֣ד נְהַר־ פְּרָ֔ת כָּשְׁל֖וּ וְנָפָֽלוּ׃ (כג) מִי־ זֶ֖ה כַּיְאֹ֣ר יַֽעֲלֶ֑ה כַּנְּהָר֕וֹת יִֽתְגָּעֲשׁ֖וּ מֵימָֽיו׃ (כד) מִצְרַ֙יִם֙ כַּיְאֹ֣ר יַֽעֲלֶ֔ה וְכַנְּהָר֖וֹת יִתְגֹּ֣עֲשׁוּ מָ֑יִם וַיֹּ֗אמֶר אַֽעֲלֶה֙ אֲכַסֶּה־ אֶ֔רֶץ אֹבִ֥ידָה עִ֖יר וְיֹ֥שְׁבֵי בָֽהּ׃ (כה) עֲל֤וּ הַסּוּסִים֙ וְהִתְהֹלְל֣וּ הָרֶ֔כֶב וְיֵצְא֖וּ הַגִּבּוֹרִ֑ים כּ֤וּשׁ וּפוּט֙ תֹּפְשֵׂ֣י מָגֵ֔ן וְלוּדִ֕ים תֹּפְשֵׂ֖י דֹּ֥רְכֵי קָֽשֶׁת׃ (כו) וְֽהַיּ֨וֹם הַה֜וּא לַאדֹנָ֧י יְהוִ֣ה צְבָא֗וֹת י֤וֹם נְקָמָה֙ לְהִנָּקֵ֣ם מִצָּרָ֔יו וְאָכְלָ֥ה חֶ֙רֶב֙ וְשָׂ֣בְעָ֔ה וְרָוְתָ֖ה מִדָּמָ֑ם כִּ֣י זֶ֠בַח לַאדֹנָ֨י יְהוִ֧ה צְבָא֛וֹת בְּאֶ֥רֶץ צָפ֖וֹן אֶל־ נְהַר־ פְּרָֽת׃ (כז) עֲלִ֤י גִלְעָד֙ וּקְחִ֣י צֳרִ֔י בְּתוּלַ֖ת בַּת־ מִצְרָ֑יִם לַשָּׁוְא֙ (הרביתי) הִרְבֵּ֣ית רְפֻא֔וֹת תְּעָלָ֖ה אֵ֥ין לָֽךְ׃ (כח) שָׁמְע֤וּ גוֹיִם֙ קְלוֹנֵ֔ךְ וְצִוְחָתֵ֖ךְ מָלְאָ֣ה הָאָ֑רֶץ כִּֽי־ גִבּ֤וֹר בְּגִבּוֹר֙ כָּשָׁ֔לוּ יַחְדָּ֖יו נָפְל֥וּ שְׁנֵיהֶֽם׃ (כט) הַדָּבָר֙ אֲשֶׁ֣ר דִּבֶּ֣ר יְהוָ֔ה אֶֽל־ יִרְמְיָ֖הוּ הַנָּבִ֑יא לָב֗וֹא נְבֽוּכַדְרֶאצַּר֙ מֶ֣לֶךְ בָּבֶ֔ל לְהַכּ֖וֹת אֶת־ אֶ֥רֶץ מִצְרָֽיִם׃ (ל) הַגִּ֤ידוּ בְמִצְרַ֙יִם֙ וְהַשְׁמִ֣יעוּ בְמִגְדּ֔וֹל וְהַשְׁמִ֥יעוּ בְנֹ֖ף וּבְתַחְפַּנְחֵ֑ס אִמְר֗וּ הִתְיַצֵּב֙ וְהָכֵ֣ן לָ֔ךְ כִּֽי־ אָכְלָ֥ה חֶ֖רֶב סְבִיבֶֽיךָ׃ (לא) מַדּ֖וּעַ נִסְחַ֣ף אַבִּירֶ֑יךָ לֹ֣א עָמַ֔ד כִּ֥י יְהוָ֖ה הֲדָפֽוֹ׃ (לב) הִרְבָּ֖ה כּוֹשֵׁ֑ל גַּם־ נָפַ֞ל אִ֣ישׁ אֶל־ רֵעֵ֗הוּ וַיֹּֽאמְרוּ֙ ק֣וּמָה ׀ וְנָשֻׁ֣בָה אֶל־ עַמֵּ֗נוּ וְאֶל־ אֶ֙רֶץ֙ מֽוֹלַדְתֵּ֔נוּ מִפְּנֵ֖י חֶ֥רֶב הַיּוֹנָֽה׃ (לג) קָרְא֖וּ שָׁ֑ם פַּרְעֹ֤ה מֶֽלֶךְ־ מִצְרַ֙יִם֙ שָׁא֔וֹן (הילילי) הֵילִ֣ילוּ ׀ (וזעקי) וּֽזְעָ֑קוּ הַגִּ֣ידוּ בְאַרְנ֔וֹן כִּ֥י שֻׁדַּ֖ד מוֹאָֽב׃ (ל) וּמִשְׁפָּ֥ט בָּ֖א אֶל־ אֶ֣רֶץ הַמִּישֹׁ֑ר אֶל־ חֹל֥וֹן וְאֶל־ יַ֖הְצָה וְעַל־ (מופעת) מֵיפָֽעַת׃ (לא) וְעַל־ דִּיב֣וֹן וְעַל־ נְב֔וֹ וְעַל־ בֵּ֖ית דִּבְלָתָֽיִם׃ (לב) וְעַ֧ל קִרְיָתַ֛יִם וְעַל־ בֵּ֥ית גָּמ֖וּל וְעַל־ בֵּ֥ית מְעֽוֹן׃ (לג) וְעַל־ קְרִיּ֖וֹת וְעַל־ בָּצְרָ֑ה וְעַ֗ל כָּל־ עָרֵי֙ אֶ֣רֶץ מוֹאָ֔ב הָרְחֹק֖וֹת וְהַקְּרֹבֽוֹת׃ (לד) נִגְדְּעָה֙ קֶ֣רֶן מוֹאָ֔ב וּזְרֹע֖וֹ נִשְׁבָּ֑רָה נְאֻ֖ם יְהוָֽה׃ (לה) הַשְׁכִּירֻ֕הוּ כִּ֥י עַל־ יְהוָ֖ה הִגְדִּ֑יל וְסָפַ֤ק מוֹאָב֙ בְּקִיא֔וֹ וְהָיָ֥ה לִשְׂחֹ֖ק גַּם־ הֽוּא׃ (לו) וְאִ֣ם ׀ ל֣וֹא הַשְּׂחֹ֗ק הָיָ֤ה לְךָ֙ יִשְׂרָאֵ֔ל אִם־ בְּגַנָּבִ֖ים (נמצאה) נִמְצָ֑א כִּֽי־ מִדֵּ֧י דְבָרֶ֛יךָ בּ֖וֹ תִּתְנוֹדָֽד׃ (לז) עִזְב֤וּ עָרִים֙ וְשִׁכְנ֣וּ בַּסֶּ֔לַע יֹשְׁבֵ֖י מוֹאָ֑ב וִֽהְי֣וּ כְיוֹנָ֔ה תְּקַנֵּ֖ן בְּעֶבְרֵ֥י פִי־ פָֽחַת׃ (לח) שָׁמַ֥עְנוּ גְאוֹן־ מוֹאָ֖ב גֵּאֶ֣ה מְאֹ֑ד גָּבְה֧וֹ וּגְאוֹנ֛וֹ וְגַאֲוָת֖וֹ וְרֻ֥ם לִבּֽוֹ׃ (לט) אֲנִ֤י יָדַ֙עְתִּי֙ נְאֻם־ יְהוָ֔ה עֶבְרָת֖וֹ וְלֹא־ כֵ֑ן בַּדָּ֖יו לֹא־ כֵ֥ן עָשֽׂוּ׃ (מ) עַל־ כֵּן֙ עַל־ מוֹאָ֣ב אֲיֵלִ֔יל וּלְמוֹאָ֥ב כֻּלֹּ֖ה אֶזְעָ֑ק אֶל־ אַנְשֵׁ֥י קִֽיר־ חֶ֖רֶשׂ יֶהְגֶּֽה׃ (מא) מִבְּכִ֨י יַעְזֵ֤ר אֶבְכֶּה־ לָּךְ֙ הַגֶּ֣פֶן שִׂבְמָ֔ה נְטִֽישֹׁתַ֙יִךְ֙ עָ֣בְרוּ יָ֔ם עַ֛ד יָ֥ם יַעְזֵ֖ר נָגָ֑עוּ עַל־ קֵיצֵ֥ךְ וְעַל־ בְּצִירֵ֖ךְ שֹׁדֵ֥ד נָפָֽל׃ (מב) וְנֶאֶסְפָ֨ה שִׂמְחָ֥ה וָגִ֛יל מִכַּרְמֶ֖ל וּמֵאֶ֣רֶץ מוֹאָ֑ב וְיַ֙יִן֙ מִיקָבִ֣ים הִשְׁבַּ֔תִּי לֹֽא־ יִדְרֹ֣ךְ הֵידָ֔ד הֵידָ֖ד לֹ֥א הֵידָֽד׃ (מג) מִזַּעֲקַ֨ת חֶשְׁבּ֜וֹן עַד־ אֶלְעָלֵ֗ה עַד־ יַ֙הַץ֙ נָתְנ֣וּ קוֹלָ֔ם מִצֹּ֙עַר֙ עַד־ חֹ֣רֹנַ֔יִם עֶגְלַ֖ת שְׁלִֽשִׁיָּ֑ה כִּ֚י גַּם־ מֵ֣י נִמְרִ֔ים לִמְשַׁמּ֖וֹת יִהְיֽוּ׃ (מד) וְהִשְׁבַּתִּ֥י לְמוֹאָ֖ב נְאֻם־ יְהוָ֑ה מַעֲלֶ֣ה בָמָ֔ה וּמַקְטִ֖יר לֵאלֹהָֽיו׃ (מה) עַל־ כֵּ֞ן לִבִּ֤י לְמוֹאָב֙ כַּחֲלִלִ֣ים יֶהֱמֶ֔ה וְלִבִּי֙ אֶל־ אַנְשֵׁ֣י קִֽיר־ חֶ֔רֶשׂ כַּחֲלִילִ֖ים יֶהֱמֶ֑ה עַל־ כֵּ֛ן יִתְרַ֥ת עָשָׂ֖ה אָבָֽדוּ׃ (מו) כִּ֤י כָל־ רֹאשׁ֙ קָרְחָ֔ה וְכָל־ זָקָ֖ן גְּרֻעָ֑ה עַ֤ל כָּל־ יָדַ֙יִם֙ גְּדֻדֹ֔ת וְעַל־ מָתְנַ֖יִם שָֽׂק׃ (מז) עַ֣ל כָּל־ גַּגּ֥וֹת מוֹאָ֛ב וּבִרְחֹבֹתֶ֖יהָ כֻּלֹּ֣ה מִסְפֵּ֑ד כִּֽי־ שָׁבַ֣רְתִּי אֶת־ מוֹאָ֗ב כִּכְלִ֛י אֵֽין־ חֵ֥פֶץ בּ֖וֹ נְאֻם־ יְהוָֽה׃ (מח) אֵ֥יךְ חַ֙תָּה֙ הֵילִ֔ילוּ אֵ֛יךְ הִפְנָה־ עֹ֥רֶף מוֹאָ֖ב בּ֑וֹשׁ וְהָיָ֥ה מוֹאָ֛ב לִשְׂחֹ֥ק וְלִמְחִתָּ֖ה לְכָל־ סְבִיבָֽיו׃ (מט) כִּי־ כֹה֙ אָמַ֣ר יְהוָ֔ה הִנֵּ֥ה כַנֶּ֖שֶׁר יִדְאֶ֑ה וּפָרַ֥שׂ כְּנָפָ֖יו אֶל־ מוֹאָֽב׃ (נ) נִלְכְּדָה֙ הַקְּרִיּ֔וֹת וְהַמְּצָד֖וֹת נִתְפָּ֑שָׂה וְֽ֠הָיָה לֵ֞ב גִּבּוֹרֵ֤י מוֹאָב֙ בַּיּ֣וֹם הַה֔וּא כְּלֵ֖ב אִשָּׁ֥ה מְצֵרָֽה׃ (נא) וְנִשְׁמַ֥ד מוֹאָ֖ב מֵעָ֑ם כִּ֥י עַל־ יְהוָ֖ה הִגְדִּֽיל׃ (נב) פַּ֥חַד וָפַ֖חַת וָפָ֑ח עָלֶ֛יךָ יוֹשֵׁ֥ב מוֹאָ֖ב נְאֻם־ יְהוָֽה׃ (נג) (הניס) הַנָּ֞ס מִפְּנֵ֤י הַפַּ֙חַד֙ יִפֹּ֣ל אֶל־ הַפַּ֔חַת וְהָֽעֹלֶה֙ מִן־ הַפַּ֔חַת יִלָּכֵ֖ד בַּפָּ֑ח כִּֽי־ אָבִ֨יא אֵלֶ֧יהָ אֶל־ מוֹאָ֛ב שְׁנַ֥ת פְּקֻדָּתָ֖ם נְאֻם־ יְהוָֽה׃ (נד) בְּצֵ֥ל חֶשְׁבּ֛וֹן עָמְד֖וּ מִכֹּ֣חַ נָסִ֑ים כִּֽי־ אֵ֞שׁ יָצָ֣א מֵחֶשְׁבּ֗וֹן וְלֶֽהָבָה֙ מִבֵּ֣ין סִיח֔וֹן וַתֹּ֙אכַל֙ פְּאַ֣ת מוֹאָ֔ב וְקָדְקֹ֖ד בְּנֵ֥י שָׁאֽוֹן׃ (נה) אוֹי־ לְךָ֣ מוֹאָ֔ב אָבַ֖ד עַם־ כְּמ֑וֹשׁ כִּֽי־ לֻקְּח֤וּ בָנֶ֙יךָ֙ בַּשֶּׁ֔בִי וּבְנֹתֶ֖יךָ בַּשִּׁבְיָֽה׃ (נו) וְשַׁבְתִּ֧י שְׁבוּת־ מוֹאָ֛ב בְּאַחֲרִ֥ית הַיָּמִ֖ים נְאֻם־ יְהוָ֑ה עַד־ הֵ֖נָּה מִשְׁפַּ֥ט מוֹאָֽב׃ (נז) לִבְנֵ֣י עַמּ֗וֹן כֹּ֚ה אָמַ֣ר יְהוָ֔ה הֲבָנִ֥ים אֵין֙ לְיִשְׂרָאֵ֔ל אִם־ יוֹרֵ֖שׁ אֵ֣ין ל֑וֹ מַדּ֗וּעַ יָרַ֤שׁ מַלְכָּם֙ אֶת־ גָּ֔ד וְעַמּ֖וֹ בְּעָרָ֥יו יָשָֽׁב׃
