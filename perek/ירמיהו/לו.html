ירמיהו פרק לו לפי החלוקה לסדרים

פרק לו בספר ירמיהו (32 פסוקים) נמצא בירמיהו סדר כא (פסוקים יא–לה) ובירמיהו סדר כב (פסוקים א–ז).

ירמיהו סדר כא

(א) וַנֵּ֖שֶׁב בָּֽאֳהָלִ֑ים וַנִּשְׁמַ֣ע וַנַּ֔עַשׂ כְּכֹ֥ל אֲשֶׁר־ צִוָּ֖נוּ יוֹנָדָ֥ב אָבִֽינוּ׃ (ב) וַיְהִ֗י בַּעֲל֨וֹת נְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶל֮ אֶל־ הָאָרֶץ֒ וַנֹּ֗אמֶר בֹּ֚אוּ וְנָב֣וֹא יְרוּשָׁלִַ֔ם מִפְּנֵי֙ חֵ֣יל הַכַּשְׂדִּ֔ים וּמִפְּנֵ֖י חֵ֣יל אֲרָ֑ם וַנֵּ֖שֶׁב בִּירוּשָׁלִָֽם׃ (ג) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶֽל־ יִרְמְיָ֖הוּ לֵאמֹֽר׃ (ד) כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הָלֹ֤ךְ וְאָֽמַרְתָּ֙ לְאִ֣ישׁ יְהוּדָ֔ה וּלְיֽוֹשְׁבֵ֖י יְרֽוּשָׁלִָ֑ם הֲל֨וֹא תִקְח֥וּ מוּסָ֛ר לִשְׁמֹ֥עַ אֶל־ דְּבָרַ֖י נְאֻם־ יְהוָֽה׃ (ה) הוּקַ֡ם אֶת־ דִּבְרֵ֣י יְהוֹנָדָ֣ב בֶּן־ רֵ֠כָב אֲשֶׁר־ צִוָּ֨ה אֶת־ בָּנָ֜יו לְבִלְתִּ֣י שְׁתֽוֹת־ יַ֗יִן וְלֹ֤א שָׁתוּ֙ עַד־ הַיּ֣וֹם הַזֶּ֔ה כִּ֣י שָֽׁמְע֔וּ אֵ֖ת מִצְוַ֣ת אֲבִיהֶ֑ם וְאָ֨נֹכִ֜י דִּבַּ֤רְתִּי אֲלֵיכֶם֙ הַשְׁכֵּ֣ם וְדַבֵּ֔ר וְלֹ֥א שְׁמַעְתֶּ֖ם אֵלָֽי׃ (ו) וָאֶשְׁלַ֣ח אֲלֵיכֶ֣ם אֶת־ כָּל־ עֲבָדַ֣י הַנְּבִאִ֣ים ׀ הַשְׁכֵּ֣ים וְשָׁלֹ֣חַ ׀ לֵאמֹ֡ר שֻׁבוּ־ נָ֡א אִישׁ֩ מִדַּרְכּ֨וֹ הָרָעָ֜ה וְהֵיטִ֣יבוּ מַֽעַלְלֵיכֶ֗ם וְאַל־ תֵּ֨לְכ֜וּ אַחֲרֵ֨י אֱלֹהִ֤ים אֲחֵרִים֙ לְעָבְדָ֔ם וּשְׁבוּ֙ אֶל־ הָ֣אֲדָמָ֔ה אֲשֶׁר־ נָתַ֥תִּי לָכֶ֖ם וְלַאֲבֹֽתֵיכֶ֑ם וְלֹ֤א הִטִּיתֶם֙ אֶֽת־ אָזְנְכֶ֔ם וְלֹ֥א שְׁמַעְתֶּ֖ם אֵלָֽי׃ (ז) כִּ֣י הֵקִ֗ימוּ בְּנֵי֙ יְהוֹנָדָ֣ב בֶּן־ רֵכָ֔ב אֶת־ מִצְוַ֥ת אֲבִיהֶ֖ם אֲשֶׁ֣ר צִוָּ֑ם וְהָעָ֣ם הַזֶּ֔ה לֹ֥א שָׁמְע֖וּ אֵלָֽי׃ (ח) לָ֠כֵן כֹּֽה־ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֤י צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֧י מֵבִ֣יא אֶל־ יְהוּדָ֗ה וְאֶ֤ל כָּל־ יֽוֹשְׁבֵי֙ יְר֣וּשָׁלִַ֔ם אֵ֚ת כָּל־ הָ֣רָעָ֔ה אֲשֶׁ֥ר דִּבַּ֖רְתִּי עֲלֵיהֶ֑ם יַ֣עַן דִּבַּ֤רְתִּי אֲלֵיהֶם֙ וְלֹ֣א שָׁמֵ֔עוּ וָאֶקְרָ֥א לָהֶ֖ם וְלֹ֥א עָנֽוּ׃ (ט) וּלְבֵ֨ית הָרֵכָבִ֜ים אָמַ֣ר יִרְמְיָ֗הוּ כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל יַ֚עַן אֲשֶׁ֣ר שְׁמַעְתֶּ֔ם עַל־ מִצְוַ֖ת יְהוֹנָדָ֣ב אֲבִיכֶ֑ם וַֽתִּשְׁמְרוּ֙ אֶת־ כָּל־ מִצְוֺתָ֔יו וַֽתַּעֲשׂ֔וּ כְּכֹ֥ל אֲשֶׁר־ צִוָּ֖ה אֶתְכֶֽם׃ (י) לָכֵ֗ן כֹּ֥ה אָמַ֛ר יְהוָ֥ה צְבָא֖וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֑ל לֹֽא־ יִכָּרֵ֨ת אִ֜ישׁ לְיוֹנָדָ֧ב בֶּן־ רֵכָ֛ב עֹמֵ֥ד לְפָנַ֖י כָּל־ הַיָּמִֽים׃ תחילת פרק לו (יא) וַֽיְהִי֙ בַּשָּׁנָ֣ה הָרְבִיעִ֔ת לִיהוֹיָקִ֥ים בֶּן־ יֹאשִׁיָּ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה הָיָ֞ה הַדָּבָ֤ר הַזֶּה֙ אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (יב) קַח־ לְךָ֮ מְגִלַּת־ סֵפֶר֒ וְכָתַבְתָּ֣ אֵלֶ֗יהָ אֵ֣ת כָּל־ הַדְּבָרִ֞ים אֲשֶׁר־ דִּבַּ֧רְתִּי אֵלֶ֛יךָ עַל־ יִשְׂרָאֵ֥ל וְעַל־ יְהוּדָ֖ה וְעַל־ כָּל־ הַגּוֹיִ֑ם מִיּ֞וֹם דִּבַּ֤רְתִּי אֵלֶ֙יךָ֙ מִימֵ֣י יֹאשִׁיָּ֔הוּ וְעַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (יג) אוּלַ֤י יִשְׁמְעוּ֙ בֵּ֣ית יְהוּדָ֔ה אֵ֚ת כָּל־ הָ֣רָעָ֔ה אֲשֶׁ֛ר אָנֹכִ֥י חֹשֵׁ֖ב לַעֲשׂ֣וֹת לָהֶ֑ם לְמַ֣עַן יָשׁ֗וּבוּ אִ֚ישׁ מִדַּרְכּ֣וֹ הָרָעָ֔ה וְסָלַחְתִּ֥י לַעֲוֺנָ֖ם וּלְחַטָּאתָֽם׃ (יד) וַיִּקְרָ֣א יִרְמְיָ֔הוּ אֶת־ בָּר֖וּךְ בֶּן־ נֵֽרִיָּ֑ה וַיִּכְתֹּ֨ב בָּר֜וּךְ מִפִּ֣י יִרְמְיָ֗הוּ אֵ֣ת כָּל־ דִּבְרֵ֧י יְהוָ֛ה אֲשֶׁר־ דִּבֶּ֥ר אֵלָ֖יו עַל־ מְגִלַּת־ סֵֽפֶר׃ (טו) וַיְצַוֶּ֣ה יִרְמְיָ֔הוּ אֶת־ בָּר֖וּךְ לֵאמֹ֑ר אֲנִ֣י עָצ֔וּר לֹ֣א אוּכַ֔ל לָב֖וֹא בֵּ֥ית יְהוָֽה׃ (טז) וּבָאתָ֣ אַתָּ֡ה וְקָרָ֣אתָ בַמְּגִלָּ֣ה אֲשֶׁר־ כָּתַֽבְתָּ־ מִפִּי֩ אֶת־ דִּבְרֵ֨י יְהוָ֜ה בְּאָזְנֵ֥י הָעָ֛ם בֵּ֥ית יְהוָ֖ה בְּי֣וֹם צ֑וֹם וְגַ֨ם בְּאָזְנֵ֧י כָל־ יְהוּדָ֛ה הַבָּאִ֥ים מֵעָרֵיהֶ֖ם תִּקְרָאֵֽם׃ (יז) אוּלַ֞י תִּפֹּ֤ל תְּחִנָּתָם֙ לִפְנֵ֣י יְהוָ֔ה וְיָשֻׁ֕בוּ אִ֖ישׁ מִדַּרְכּ֣וֹ הָרָעָ֑ה כִּֽי־ גָד֤וֹל הָאַף֙ וְהַ֣חֵמָ֔ה אֲשֶׁר־ דִּבֶּ֥ר יְהוָ֖ה אֶל־ הָעָ֥ם הַזֶּֽה׃ (יח) וַיַּ֗עַשׂ בָּרוּךְ֙ בֶּן־ נֵ֣רִיָּ֔ה כְּכֹ֥ל אֲשֶׁר־ צִוָּ֖הוּ יִרְמְיָ֣הוּ הַנָּבִ֑יא לִקְרֹ֥א בַסֵּ֛פֶר דִּבְרֵ֥י יְהוָ֖ה בֵּ֥ית יְהֹוָֽה׃ (יט) וַיְהִ֣י בַשָּׁנָ֣ה הַ֠חֲמִשִׁית לִיהוֹיָקִ֨ים בֶּן־ יֹאשִׁיָּ֤הוּ מֶֽלֶךְ־ יְהוּדָה֙ בַּחֹ֣דֶשׁ הַתְּשִׁעִ֔י קָרְא֨וּ צ֜וֹם לִפְנֵ֧י יְהוָ֛ה כָּל־ הָעָ֖ם בִּירֽוּשָׁלִָ֑ם וְכָל־ הָעָ֗ם הַבָּאִ֛ים מֵעָרֵ֥י יְהוּדָ֖ה בִּירוּשָׁלִָֽם׃ (כ) וַיִּקְרָ֨א בָר֥וּךְ בַּסֵּ֛פֶר אֶת־ דִּבְרֵ֥י יִרְמְיָ֖הוּ בֵּ֣ית יְהוָ֑ה בְּלִשְׁכַּ֡ת גְּמַרְיָהוּ֩ בֶן־ שָׁפָ֨ן הַסֹּפֵ֜ר בֶּחָצֵ֣ר הָעֶלְי֗וֹן פֶּ֣תַח שַׁ֤עַר בֵּית־ יְהוָה֙ הֶֽחָדָ֔שׁ בְּאָזְנֵ֖י כָּל־ הָעָֽם׃ (כא) וַ֠יִּשְׁמַ֗ע מִכָ֨יְהוּ בֶן־ גְּמַרְיָ֧הוּ בֶן־ שָׁפָ֛ן אֶת־ כָּל־ דִּבְרֵ֥י יְהוָ֖ה מֵעַ֥ל הַסֵּֽפֶר׃ (כב) וַיֵּ֤רֶד בֵּית־ הַמֶּ֙לֶךְ֙ עַל־ לִשְׁכַּ֣ת הַסֹּפֵ֔ר וְהִ֨נֵּה־ שָׁ֔ם כָּל־ הַשָּׂרִ֖ים יֽוֹשְׁבִ֑ים אֱלִישָׁמָ֣ע הַסֹּפֵ֡ר וּדְלָיָ֣הוּ בֶן־ שְׁ֠מַעְיָהוּ וְאֶלְנָתָ֨ן בֶּן־ עַכְבּ֜וֹר וּגְמַרְיָ֧הוּ בֶן־ שָׁפָ֛ן וְצִדְקִיָּ֥הוּ בֶן־ חֲנַנְיָ֖הוּ וְכָל־ הַשָּׂרִֽים׃ (כג) וַיַּגֵּ֤ד לָהֶם֙ מִכָ֔יְהוּ אֵ֥ת כָּל־ הַדְּבָרִ֖ים אֲשֶׁ֣ר שָׁמֵ֑עַ בִּקְרֹ֥א בָר֛וּךְ בַּסֵּ֖פֶר בְּאָזְנֵ֥י הָעָֽם׃ (כד) וַיִּשְׁלְח֨וּ כָל־ הַשָּׂרִ֜ים אֶל־ בָּר֗וּךְ אֶת־ יְהוּדִ֡י בֶּן־ נְ֠תַנְיָהוּ בֶּן־ שֶׁלֶמְיָ֣הוּ בֶן־ כּוּשִׁי֮ לֵאמֹר֒ הַמְּגִלָּ֗ה אֲשֶׁ֨ר קָרָ֤אתָ בָּהּ֙ בְּאָזְנֵ֣י הָעָ֔ם קָחֶ֥נָּה בְיָדְךָ֖ וָלֵ֑ךְ וַ֠יִּקַּח בָּר֨וּךְ בֶּן־ נֵרִיָּ֤הוּ אֶת־ הַמְּגִלָּה֙ בְּיָד֔וֹ וַיָּבֹ֖א אֲלֵיהֶֽם׃ (כה) וַיֹּאמְר֣וּ אֵלָ֔יו שֵׁ֣ב נָ֔א וּקְרָאֶ֖נָּה בְּאָזְנֵ֑ינוּ וַיִּקְרָ֥א בָר֖וּךְ בְּאָזְנֵיהֶֽם׃ (כו) וַיְהִ֗י כְּשָׁמְעָם֙ אֶת־ כָּל־ הַדְּבָרִ֔ים פָּחֲד֖וּ אִ֣ישׁ אֶל־ רֵעֵ֑הוּ וַיֹּֽאמְרוּ֙ אֶל־ בָּר֔וּךְ הַגֵּ֤יד נַגִּיד֙ לַמֶּ֔לֶךְ אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (כז) וְאֶ֨ת־ בָּר֔וּךְ שָׁאֲל֖וּ לֵאמֹ֑ר הַגֶּד־ נָ֣א לָ֔נוּ אֵ֗יךְ כָּתַ֛בְתָּ אֶת־ כָּל־ הַדְּבָרִ֥ים הָאֵ֖לֶּה מִפִּֽיו׃ (כח) וַיֹּ֤אמֶר לָהֶם֙ בָּר֔וּךְ מִפִּיו֙ יִקְרָ֣א אֵלַ֔י אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַאֲנִ֛י כֹּתֵ֥ב עַל־ הַסֵּ֖פֶר בַּדְּיֽוֹ׃ (כט) וַיֹּאמְר֤וּ הַשָּׂרִים֙ אֶל־ בָּר֔וּךְ לֵ֥ךְ הִסָּתֵ֖ר אַתָּ֣ה וְיִרְמְיָ֑הוּ וְאִ֥ישׁ אַל־ יֵדַ֖ע אֵיפֹ֥ה אַתֶּֽם׃ (ל) וַיָּבֹ֤אוּ אֶל־ הַמֶּ֙לֶךְ֙ חָצֵ֔רָה וְאֶת־ הַמְּגִלָּ֣ה הִפְקִ֔דוּ בְּלִשְׁכַּ֖ת אֱלִישָׁמָ֣ע הַסֹּפֵ֑ר וַיַּגִּ֙ידוּ֙ בְּאָזְנֵ֣י הַמֶּ֔לֶךְ אֵ֖ת כָּל־ הַדְּבָרִֽים׃ (לא) וַיִּשְׁלַ֨ח הַמֶּ֜לֶךְ אֶת־ יְהוּדִ֗י לָקַ֙חַת֙ אֶת־ הַמְּגִלָּ֔ה וַיִּ֨קָּחֶ֔הָ מִלִּשְׁכַּ֖ת אֱלִישָׁמָ֣ע הַסֹּפֵ֑ר וַיִּקְרָאֶ֤הָ יְהוּדִי֙ בְּאָזְנֵ֣י הַמֶּ֔לֶךְ וּבְאָזְנֵי֙ כָּל־ הַשָּׂרִ֔ים הָעֹמְדִ֖ים מֵעַ֥ל הַמֶּֽלֶךְ׃ (לב) וְהַמֶּ֗לֶךְ יוֹשֵׁב֙ בֵּ֣ית הַחֹ֔רֶף בַּחֹ֖דֶשׁ הַתְּשִׁיעִ֑י וְאֶת־ הָאָ֖ח לְפָנָ֥יו מְבֹעָֽרֶת׃ (לג) וַיְהִ֣י ׀ כִּקְר֣וֹא יְהוּדִ֗י שָׁלֹ֣שׁ דְּלָתוֹת֮ וְאַרְבָּעָה֒ יִֽקְרָעֶ֙הָ֙ בְּתַ֣עַר הַסֹּפֵ֔ר וְהַשְׁלֵ֕ךְ אֶל־ הָאֵ֖שׁ אֲשֶׁ֣ר אֶל־ הָאָ֑ח עַד־ תֹּם֙ כָּל־ הַמְּגִלָּ֔ה עַל־ הָאֵ֖שׁ אֲשֶׁ֥ר עַל־ הָאָֽח׃ (לד) וְלֹ֣א פָחֲד֔וּ וְלֹ֥א קָרְע֖וּ אֶת־ בִּגְדֵיהֶ֑ם הַמֶּ֙לֶךְ֙ וְכָל־ עֲבָדָ֔יו הַשֹּׁ֣מְעִ֔ים אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (לה) וְגַם֩ אֶלְנָתָ֨ן וּדְלָיָ֤הוּ וּגְמַרְיָ֙הוּ֙ הִפְגִּ֣עוּ בַמֶּ֔לֶךְ לְבִלְתִּ֥י שְׂרֹ֖ף אֶת־ הַמְּגִלָּ֑ה וְלֹ֥א שָׁמַ֖ע אֲלֵיהֶֽם׃

(א) וַיְצַוֶּ֣ה הַ֠מֶּלֶךְ אֶת־ יְרַחְמְאֵ֨ל בֶּן־ הַמֶּ֜לֶךְ וְאֶת־ שְׂרָיָ֣הוּ בֶן־ עַזְרִיאֵ֗ל וְאֶת־ שֶֽׁלֶמְיָ֙הוּ֙ בֶּֽן־ עַבְדְּאֵ֔ל לָקַ֙חַת֙ אֶת־ בָּר֣וּךְ הַסֹּפֵ֔ר וְאֵ֖ת יִרְמְיָ֣הוּ הַנָּבִ֑יא וַיַּסְתִּרֵ֖ם יְהוָֽה׃ (ב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֶֽל־ יִרְמְיָ֑הוּ אַחֲרֵ֣י ׀ שְׂרֹ֣ף הַמֶּ֗לֶךְ אֶת־ הַמְּגִלָּה֙ וְאֶת־ הַדְּבָרִ֔ים אֲשֶׁ֨ר כָּתַ֥ב בָּר֛וּךְ מִפִּ֥י יִרְמְיָ֖הוּ לֵאמֹֽר׃ (ג) שׁ֥וּב קַח־ לְךָ֖ מְגִלָּ֣ה אַחֶ֑רֶת וּכְתֹ֣ב עָלֶ֗יהָ אֵ֤ת כָּל־ הַדְּבָרִים֙ הָרִ֣אשֹׁנִ֔ים אֲשֶׁ֣ר הָי֗וּ עַל־ הַמְּגִלָּה֙ הָרִ֣אשֹׁנָ֔ה אֲשֶׁ֥ר שָׂרַ֖ף יְהוֹיָקִ֥ים מֶֽלֶךְ־ יְהוּדָֽה׃ (ד) וְעַל־ יְהוֹיָקִ֤ים מֶֽלֶךְ־ יְהוּדָה֙ תֹּאמַ֔ר כֹּ֖ה אָמַ֣ר יְהוָ֑ה אַ֠תָּה שָׂרַ֜פְתָּ אֶת־ הַמְּגִלָּ֤ה הַזֹּאת֙ לֵאמֹ֔ר מַדּוּעַ֩ כָּתַ֨בְתָּ עָלֶ֜יהָ לֵאמֹ֗ר בֹּֽא־ יָב֤וֹא מֶֽלֶךְ־ בָּבֶל֙ וְהִשְׁחִית֙ אֶת־ הָאָ֣רֶץ הַזֹּ֔את וְהִשְׁבִּ֥ית מִמֶּ֖נָּה אָדָ֥ם וּבְהֵמָֽה׃ (ה) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה עַל־ יְהֽוֹיָקִים֙ מֶ֣לֶךְ יְהוּדָ֔ה לֹא־ יִֽהְיֶה־ לּ֥וֹ יוֹשֵׁ֖ב עַל־ כִּסֵּ֣א דָוִ֑ד וְנִבְלָתוֹ֙ תִּֽהְיֶ֣ה מֻשְׁלֶ֔כֶת לַחֹ֥רֶב בַּיּ֖וֹם וְלַקֶּ֥רַח בַּלָּֽיְלָה׃ (ו) וּפָקַדְתִּ֨י עָלָ֧יו וְעַל־ זַרְע֛וֹ וְעַל־ עֲבָדָ֖יו אֶת־ עֲוֺנָ֑ם וְהֵבֵאתִ֣י עֲ֠לֵיהֶם וְעַל־ יֹשְׁבֵ֨י יְרוּשָׁלִַ֜ם וְאֶל־ אִ֣ישׁ יְהוּדָ֗ה אֵ֧ת כָּל־ הָרָעָ֛ה אֲשֶׁר־ דִּבַּ֥רְתִּי אֲלֵיהֶ֖ם וְלֹ֥א שָׁמֵֽעוּ׃ (ז) וְיִרְמְיָ֜הוּ לָקַ֣ח ׀ מְגִלָּ֣ה אַחֶ֗רֶת וַֽיִּתְּנָהּ֮ אֶל־ בָּר֣וּךְ בֶּן־ נֵרִיָּהוּ֮ הַסֹּפֵר֒ וַיִּכְתֹּ֤ב עָלֶ֙יהָ֙ מִפִּ֣י יִרְמְיָ֔הוּ אֵ֚ת כָּל־ דִּבְרֵ֣י הַסֵּ֔פֶר אֲשֶׁ֥ר שָׂרַ֛ף יְהוֹיָקִ֥ים מֶֽלֶךְ־ יְהוּדָ֖ה בָּאֵ֑שׁ וְע֨וֹד נוֹסַ֧ף עֲלֵיהֶ֛ם דְּבָרִ֥ים רַבִּ֖ים כָּהֵֽמָּה׃ סוף פרק לו (ח) וַיִּ֨מְלָךְ־ מֶ֔לֶךְ צִדְקִיָּ֖הוּ בֶּן־ יֹֽאשִׁיָּ֑הוּ תַּ֗חַת כָּנְיָ֙הוּ֙ בֶּן־ יְה֣וֹיָקִ֔ים אֲשֶׁ֥ר הִמְלִ֛יךְ נְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל בְּאֶ֥רֶץ יְהוּדָֽה׃ (ט) וְלֹ֥א שָׁמַ֛ע ה֥וּא וַעֲבָדָ֖יו וְעַ֣ם הָאָ֑רֶץ אֶל־ דִּבְרֵ֣י יְהוָ֔ה אֲשֶׁ֣ר דִּבֶּ֔ר בְּיַ֖ד יִרְמְיָ֥הוּ הַנָּבִֽיא׃ (י) וַיִּשְׁלַח֩ הַמֶּ֨לֶךְ צִדְקִיָּ֜הוּ אֶת־ יְהוּכַ֣ל בֶּן־ שֶֽׁלֶמְיָ֗ה וְאֶת־ צְפַנְיָ֤הוּ בֶן־ מַֽעֲשֵׂיָה֙ הַכֹּהֵ֔ן אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא לֵאמֹ֑ר הִתְפַּלֶּל־ נָ֣א בַעֲדֵ֔נוּ אֶל־ יְהוָ֖ה אֱלֹהֵֽינוּ׃ (יא) וְיִרְמְיָ֕הוּ בָּ֥א וְיֹצֵ֖א בְּת֣וֹךְ הָעָ֑ם וְלֹֽא־ נָתְנ֥וּ אֹת֖וֹ בֵּ֥ית (הכליא) הַכְּלֽוּא׃ (יב) וְחֵ֥יל פַּרְעֹ֖ה יָצָ֣א מִמִּצְרָ֑יִם וַיִּשְׁמְע֨וּ הַכַּשְׂדִּ֜ים הַצָּרִ֤ים עַל־ יְרוּשָׁלִַ֙ם֙ אֶת־ שִׁמְעָ֔ם וַיֵּ֣עָל֔וּ מֵעַ֖ל יְרוּשָׁלִָֽם׃ (יג) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא לֵאמֹֽר׃ (יד) כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל כֹּ֤ה תֹֽאמְרוּ֙ אֶל־ מֶ֣לֶךְ יְהוּדָ֔ה הַשֹּׁלֵ֧חַ אֶתְכֶ֛ם אֵלַ֖י לְדָרְשֵׁ֑נִי הִנֵּ֣ה ׀ חֵ֣יל פַּרְעֹ֗ה הַיֹּצֵ֤א לָכֶם֙ לְעֶזְרָ֔ה שָׁ֥ב לְאַרְצ֖וֹ מִצְרָֽיִם׃ (טו) וְשָׁ֙בוּ֙ הַכַּשְׂדִּ֔ים וְנִלְחֲמ֖וּ עַל־ הָעִ֣יר הַזֹּ֑את וּלְכָדֻ֖הָ וּשְׂרָפֻ֥הָ בָאֵֽשׁ׃ (טז) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ תַּשִּׁ֤אוּ נַפְשֹֽׁתֵיכֶם֙ לֵאמֹ֔ר הָלֹ֛ךְ יֵלְכ֥וּ מֵעָלֵ֖ינוּ הַכַּשְׂדִּ֑ים כִּי־ לֹ֖א יֵלֵֽכוּ׃ (יז) כִּ֣י אִם־ הִכִּיתֶ֞ם כָּל־ חֵ֤יל כַּשְׂדִּים֙ הַנִּלְחָמִ֣ים אִתְּכֶ֔ם וְנִ֨שְׁאֲרוּ בָ֔ם אֲנָשִׁ֖ים מְדֻקָּרִ֑ים אִ֤ישׁ בְּאָהֳלוֹ֙ יָק֔וּמוּ וְשָֽׂרְפ֛וּ אֶת־ הָעִ֥יר הַזֹּ֖את בָּאֵֽשׁ׃ (יח) וְהָיָ֗ה בְּהֵֽעָלוֹת֙ חֵ֣יל הַכַּשְׂדִּ֔ים מֵעַ֖ל יְרֽוּשָׁלִָ֑ם מִפְּנֵ֖י חֵ֥יל פַּרְעֹֽה׃ (יט) וַיֵּצֵ֤א יִרְמְיָ֙הוּ֙ מִיר֣וּשָׁלִַ֔ם לָלֶ֖כֶת אֶ֣רֶץ בִּנְיָמִ֑ן לַחֲלִ֥ק מִשָּׁ֖ם בְּת֥וֹךְ הָעָֽם׃ (כ) וַיְהִי־ ה֞וּא בְּשַׁ֣עַר בִּנְיָמִ֗ן וְשָׁם֙ בַּ֣עַל פְּקִדֻ֔ת וּשְׁמוֹ֙ יִרְאִיָּ֔יה בֶּן־ שֶֽׁלֶמְיָ֖ה בֶּן־ חֲנַנְיָ֑ה וַיִּתְפֹּ֞שׂ אֶֽת־ יִרְמְיָ֤הוּ הַנָּבִיא֙ לֵאמֹ֔ר אֶל־ הַכַּשְׂדִּ֖ים אַתָּ֥ה נֹפֵֽל׃ (כא) וַיֹּ֨אמֶר יִרְמְיָ֜הוּ שֶׁ֗קֶר אֵינֶ֤נִּי נֹפֵל֙ עַל־ הַכַּשְׂדִּ֔ים וְלֹ֥א שָׁמַ֖ע אֵלָ֑יו וַיִּתְפֹּ֤שׂ יִרְאִיָּיה֙ בְּיִרְמְיָ֔הוּ וַיְבִאֵ֖הוּ אֶל־ הַשָּׂרִֽים׃ (כב) וַיִּקְצְפ֧וּ הַשָּׂרִ֛ים עַֽל־ יִרְמְיָ֖הוּ וְהִכּ֣וּ אֹת֑וֹ וְנָתְנ֨וּ אוֹת֜וֹ בֵּ֣ית הָאֵס֗וּר בֵּ֚ית יְהוֹנָתָ֣ן הַסֹּפֵ֔ר כִּֽי־ אֹת֥וֹ עָשׂ֖וּ לְבֵ֥ית הַכֶּֽלֶא׃ (כג) כִּ֣י בָ֧א יִרְמְיָ֛הוּ אֶל־ בֵּ֥ית הַבּ֖וֹר וְאֶל־ הַֽחֲנֻ֑יוֹת וַיֵּֽשֶׁב־ שָׁ֥ם יִרְמְיָ֖הוּ יָמִ֥ים רַבִּֽים׃ (כד) וַיִּשְׁלַח֩ הַמֶּ֨לֶךְ צִדְקִיָּ֜הוּ וַיִּקָּחֵ֗הוּ וַיִּשְׁאָלֵ֨הוּ הַמֶּ֤לֶךְ בְּבֵיתוֹ֙ בַּסֵּ֔תֶר וַיֹּ֕אמֶר הֲיֵ֥שׁ דָּבָ֖ר מֵאֵ֣ת יְהוָ֑ה וַיֹּ֤אמֶר יִרְמְיָ֙הוּ֙ יֵ֔שׁ וַיֹּ֕אמֶר בְּיַ֥ד מֶֽלֶךְ־ בָּבֶ֖ל תִּנָּתֵֽן׃ (כה) וַיֹּ֣אמֶר יִרְמְיָ֔הוּ אֶל־ הַמֶּ֖לֶךְ צִדְקִיָּ֑הוּ מֶה֩ חָטָ֨אתִֽי לְךָ֤ וְלַעֲבָדֶ֙יךָ֙ וְלָעָ֣ם הַזֶּ֔ה כִּֽי־ נְתַתֶּ֥ם אוֹתִ֖י אֶל־ בֵּ֥ית הַכֶּֽלֶא׃ (כו) (ואיו) וְאַיֵּה֙ נְבִ֣יאֵיכֶ֔ם אֲשֶׁר־ נִבְּא֥וּ לָכֶ֖ם לֵאמֹ֑ר לֹֽא־ יָבֹ֤א מֶֽלֶךְ־ בָּבֶל֙ עֲלֵיכֶ֔ם וְעַ֖ל הָאָ֥רֶץ הַזֹּֽאת׃ (כז) וְעַתָּ֕ה שְֽׁמַֽע־ נָ֖א אֲדֹנִ֣י הַמֶּ֑לֶךְ תִּפָּל־ נָ֤א תְחִנָּתִי֙ לְפָנֶ֔יךָ וְאַל־ תְּשִׁבֵ֗נִי בֵּ֚ית יְהוֹנָתָ֣ן הַסֹּפֵ֔ר וְלֹ֥א אָמ֖וּת שָֽׁם׃ (כח) וַיְצַוֶּ֞ה הַמֶּ֣לֶךְ צִדְקִיָּ֗הוּ וַיַּפְקִ֣דוּ אֶֽת־ יִרְמְיָהוּ֮ בַּחֲצַ֣ר הַמַּטָּרָה֒ וְנָתֹן֩ ל֨וֹ כִכַּר־ לֶ֤חֶם לַיּוֹם֙ מִח֣וּץ הָאֹפִ֔ים עַד־ תֹּ֥ם כָּל־ הַלֶּ֖חֶם מִן־ הָעִ֑יר וַיֵּ֣שֶׁב יִרְמְיָ֔הוּ בַּחֲצַ֖ר הַמַּטָּרָֽה׃ (כט) וַיִּשְׁמַ֞ע שְׁפַטְיָ֣ה בֶן־ מַתָּ֗ן וּגְדַלְיָ֙הוּ֙ בֶּן־ פַּשְׁח֔וּר וְיוּכַל֙ בֶּן־ שֶׁ֣לֶמְיָ֔הוּ וּפַשְׁח֖וּר בֶּן־ מַלְכִּיָּ֑ה אֶ֨ת־ הַדְּבָרִ֔ים אֲשֶׁ֧ר יִרְמְיָ֛הוּ מְדַבֵּ֥ר אֶל־ כָּל־ הָעָ֖ם לֵאמֹֽר׃ (ל) כֹּה֮ אָמַ֣ר יְהוָה֒ הַיֹּשֵׁב֙ בָּעִ֣יר הַזֹּ֔את יָמ֕וּת בַּחֶ֖רֶב בָּרָעָ֣ב וּבַדָּ֑בֶר וְהַיֹּצֵ֤א אֶל־ הַכַּשְׂדִּים֙ (יחיה) וְחָיָ֔ה וְהָיְתָה־ לּ֥וֹ נַפְשׁ֛וֹ לְשָׁלָ֖ל וָחָֽי׃ (לא) כֹּ֖ה אָמַ֣ר יְהוָ֑ה הִנָּתֹ֨ן תִּנָּתֵ֜ן הָעִ֣יר הַזֹּ֗את בְּיַ֛ד חֵ֥יל מֶֽלֶךְ־ בָּבֶ֖ל וּלְכָדָֽהּ׃ (לב) וַיֹּאמְר֨וּ הַשָּׂרִ֜ים אֶל־ הַמֶּ֗לֶךְ י֣וּמַת נָא֮ אֶת־ הָאִ֣ישׁ הַזֶּה֒ כִּֽי־ עַל־ כֵּ֡ן הֽוּא־ מְרַפֵּ֡א אֶת־ יְדֵי֩ אַנְשֵׁ֨י הַמִּלְחָמָ֜ה הַֽנִּשְׁאָרִ֣ים ׀ בָּעִ֣יר הַזֹּ֗את וְאֵת֙ יְדֵ֣י כָל־ הָעָ֔ם לְדַבֵּ֣ר אֲלֵיהֶ֔ם כַּדְּבָרִ֖ים הָאֵ֑לֶּה כִּ֣י ׀ הָאִ֣ישׁ הַזֶּ֗ה אֵינֶ֨נּוּ דֹרֵ֧שׁ לְשָׁל֛וֹם לָעָ֥ם הַזֶּ֖ה כִּ֥י אִם־ לְרָעָֽה׃ (לג) וַיֹּ֙אמֶר֙ הַמֶּ֣לֶךְ צִדְקִיָּ֔הוּ הִנֵּה־ ה֖וּא בְּיֶדְכֶ֑ם כִּֽי־ אֵ֣ין הַמֶּ֔לֶךְ יוּכַ֥ל אֶתְכֶ֖ם דָּבָֽר׃ (לד) וַיִּקְח֣וּ אֶֽת־ יִרְמְיָ֗הוּ וַיַּשְׁלִ֨כוּ אֹת֜וֹ אֶל־ הַבּ֣וֹר ׀ מַלְכִּיָּ֣הוּ בֶן־ הַמֶּ֗לֶךְ אֲשֶׁר֙ בַּחֲצַ֣ר הַמַּטָּרָ֔ה וַיְשַׁלְּח֥וּ אֶֽת־ יִרְמְיָ֖הוּ בַּחֲבָלִ֑ים וּבַבּ֤וֹר אֵֽין־ מַ֙יִם֙ כִּ֣י אִם־ טִ֔יט וַיִּטְבַּ֥ע יִרְמְיָ֖הוּ בַּטִּֽיט׃ (לה) וַיִּשְׁמַ֡ע עֶֽבֶד־ מֶ֨לֶךְ הַכּוּשִׁ֜י אִ֣ישׁ סָרִ֗יס וְהוּא֙ בְּבֵ֣ית הַמֶּ֔לֶךְ כִּֽי־ נָתְנ֥וּ אֶֽת־ יִרְמְיָ֖הוּ אֶל־ הַבּ֑וֹר וְהַמֶּ֥לֶךְ יוֹשֵׁ֖ב בְּשַׁ֥עַר בִּנְיָמִֽן׃
