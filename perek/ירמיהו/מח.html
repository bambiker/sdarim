ירמיהו פרק מח לפי החלוקה לסדרים

פרק מח בספר ירמיהו (47 פסוקים) נמצא בירמיהו סדר כז (פסוקים י–נו).

ירמיהו סדר כז

(א) וְ֠אַתָּה אַל־ תִּירָ֞א עַבְדִּ֤י יַֽעֲקֹב֙ וְאַל־ תֵּחַ֣ת יִשְׂרָאֵ֔ל כִּ֠י הִנְנִ֤י מוֹשִֽׁעֲךָ֙ מֵֽרָח֔וֹק וְאֶֽת־ זַרְעֲךָ֖ מֵאֶ֣רֶץ שִׁבְיָ֑ם וְשָׁ֧ב יַעֲק֛וֹב וְשָׁקַ֥ט וְשַׁאֲנַ֖ן וְאֵ֥ין מַחֲרִֽיד׃ (ב) אַ֠תָּה אַל־ תִּירָ֞א עַבְדִּ֤י יַֽעֲקֹב֙ נְאֻם־ יְהוָ֔ה כִּ֥י אִתְּךָ֖ אָ֑נִי כִּי֩ אֶעֱשֶׂ֨ה כָלָ֜ה בְּכָֽל־ הַגּוֹיִ֣ם ׀ אֲשֶׁ֧ר הִדַּחְתִּ֣יךָ שָׁ֗מָּה וְאֹֽתְךָ֙ לֹא־ אֶעֱשֶׂ֣ה כָלָ֔ה וְיִסַּרְתִּ֙יךָ֙ לַמִּשְׁפָּ֔ט וְנַקֵּ֖ה לֹ֥א אֲנַקֶּֽךָּ׃ (ג) אֲשֶׁ֨ר הָיָ֧ה דְבַר־ יְהוָ֛ה אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא אֶל־ פְּלִשְׁתִּ֑ים בְּטֶ֛רֶם יַכֶּ֥ה פַרְעֹ֖ה אֶת־ עַזָּֽה׃ (ד) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה הִנֵּה־ מַ֜יִם עֹלִ֤ים מִצָּפוֹן֙ וְהָיוּ֙ לְנַ֣חַל שׁוֹטֵ֔ף וְיִשְׁטְפוּ֙ אֶ֣רֶץ וּמְלוֹאָ֔הּ עִ֖יר וְיֹ֣שְׁבֵי בָ֑הּ וְזָֽעֲקוּ֙ הָֽאָדָ֔ם וְהֵילִ֕ל כֹּ֖ל יוֹשֵׁ֥ב הָאָֽרֶץ׃ (ה) מִקּ֗וֹל שַֽׁעֲטַת֙ פַּרְס֣וֹת אַבִּירָ֔יו מֵרַ֣עַשׁ לְרִכְבּ֔וֹ הֲמ֖וֹן גַּלְגִּלָּ֑יו לֹֽא־ הִפְנ֤וּ אָבוֹת֙ אֶל־ בָּנִ֔ים מֵֽרִפְי֖וֹן יָדָֽיִם׃ (ו) עַל־ הַיּ֗וֹם הַבָּא֙ לִשְׁד֣וֹד אֶת־ כָּל־ פְּלִשְׁתִּ֔ים לְהַכְרִ֤ית לְצֹר֙ וּלְצִיד֔וֹן כֹּ֖ל שָׂרִ֣יד עֹזֵ֑ר כִּֽי־ שֹׁדֵ֤ד יְהוָה֙ אֶת־ פְּלִשְׁתִּ֔ים שְׁאֵרִ֖ית אִ֥י כַפְתּֽוֹר׃ (ז) בָּ֤אָה קָרְחָה֙ אֶל־ עַזָּ֔ה נִדְמְתָ֥ה אַשְׁקְל֖וֹן שְׁאֵרִ֣ית עִמְקָ֑ם עַד־ מָתַ֖י תִּתְגּוֹדָֽדִי׃ (ח) ה֗וֹי חֶ֚רֶב לַֽיהוָ֔ה עַד־ אָ֖נָה לֹ֣א תִשְׁקֹ֑טִי הֵאָֽסְפִי֙ אַל־ תַּעְרֵ֔ךְ הֵרָגְעִ֖י וָדֹֽמִּי׃ (ט) אֵ֣יךְ תִּשְׁקֹ֔טִי וַֽיהוָ֖ה צִוָּה־ לָ֑הּ אֶֽל־ אַשְׁקְל֛וֹן וְאֶל־ ח֥וֹף הַיָּ֖ם שָׁ֥ם יְעָדָֽהּ׃ תחילת פרק מח (י) לְמוֹאָ֡ב כֹּֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל ה֤וֹי אֶל־ נְבוֹ֙ כִּ֣י שֻׁדָּ֔דָה הֹבִ֥ישָׁה נִלְכְּדָ֖ה קִרְיָתָ֑יִם הֹבִ֥ישָׁה הַמִּשְׂגָּ֖ב וָחָֽתָּה׃ (יא) אֵ֣ין עוֹד֮ תְּהִלַּ֣ת מוֹאָב֒ בְּחֶשְׁבּ֗וֹן חָשְׁב֤וּ עָלֶ֙יהָ֙ רָעָ֔ה לְכ֖וּ וְנַכְרִיתֶ֣נָּה מִגּ֑וֹי גַּם־ מַדְמֵ֣ן תִּדֹּ֔מִּי אַחֲרַ֖יִךְ תֵּ֥לֶךְ חָֽרֶב׃ (יב) ק֥וֹל צְעָקָ֖ה מֵחֹֽרוֹנָ֑יִם שֹׁ֖ד וָשֶׁ֥בֶר גָּדֽוֹל׃ (יג) נִשְׁבְּרָ֖ה מוֹאָ֑ב הִשְׁמִ֥יעוּ זְּעָקָ֖ה (צעוריה) צְעִירֶֽיהָ׃ (יד) כִּ֚י מַעֲלֵ֣ה (הלחות) הַלּוּחִ֔ית בִּבְכִ֖י יַֽעֲלֶה־ בֶּ֑כִי כִּ֚י בְּמוֹרַ֣ד חוֹרֹנַ֔יִם צָרֵ֥י צַֽעֲקַת־ שֶׁ֖בֶר שָׁמֵֽעוּ׃ (טו) נֻ֖סוּ מַלְּט֣וּ נַפְשְׁכֶ֑ם וְתִֽהְיֶ֕ינָה כַּעֲרוֹעֵ֖ר בַּמִּדְבָּֽר׃ (טז) כִּ֠י יַ֣עַן בִּטְחֵ֤ךְ בְּמַעֲשַׂ֙יִךְ֙ וּבְא֣וֹצְרוֹתַ֔יִךְ גַּם־ אַ֖תְּ תִּלָּכֵ֑דִי וְיָצָ֤א (כמיש) כְמוֹשׁ֙ בַּגּוֹלָ֔ה כֹּהֲנָ֥יו וְשָׂרָ֖יו (יחד) יַחְדָּֽיו׃ (יז) וְיָבֹ֨א שֹׁדֵ֜ד אֶל־ כָּל־ עִ֗יר וְעִיר֙ לֹ֣א תִמָּלֵ֔ט וְאָבַ֥ד הָעֵ֖מֶק וְנִשְׁמַ֣ד הַמִּישֹׁ֑ר אֲשֶׁ֖ר אָמַ֥ר יְהוָֽה׃ (יח) תְּנוּ־ צִ֣יץ לְמוֹאָ֔ב כִּ֥י נָצֹ֖א תֵּצֵ֑א וְעָרֶ֙יהָ֙ לְשַׁמָּ֣ה תִֽהְיֶ֔ינָה מֵאֵ֥ין יוֹשֵׁ֖ב בָּהֵֽן׃ (יט) אָר֗וּר עֹשֶׂ֛ה מְלֶ֥אכֶת יְהוָ֖ה רְמִיָּ֑ה וְאָר֕וּר מֹנֵ֥עַ חַרְבּ֖וֹ מִדָּֽם׃ (כ) שַׁאֲנַ֨ן מוֹאָ֜ב מִנְּעוּרָ֗יו וְשֹׁקֵ֥ט הוּא֙ אֶל־ שְׁמָרָ֔יו וְלֹֽא־ הוּרַ֤ק מִכְּלִי֙ אֶל־ כֶּ֔לִי וּבַגּוֹלָ֖ה לֹ֣א הָלָ֑ךְ עַל־ כֵּ֗ן עָמַ֤ד טַעְמוֹ֙ בּ֔וֹ וְרֵיח֖וֹ לֹ֥א נָמָֽר׃ (כא) לָכֵ֞ן הִנֵּ֖ה־ יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וְשִׁלַּחְתִּי־ ל֥וֹ צֹעִ֖ים וְצֵעֻ֑הוּ וְכֵלָ֣יו יָרִ֔יקוּ וְנִבְלֵיהֶ֖ם יְנַפֵּֽצוּ׃ (כב) וּבֹ֥שׁ מוֹאָ֖ב מִכְּמ֑וֹשׁ כַּאֲשֶׁר־ בֹּ֙שׁוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל מִבֵּ֥ית אֵ֖ל מִבְטֶחָֽם׃ (כג) אֵ֚יךְ תֹּֽאמְר֔וּ גִּבּוֹרִ֖ים אֲנָ֑חְנוּ וְאַנְשֵׁי־ חַ֖יִל לַמִּלְחָמָֽה׃ (כד) שֻׁדַּ֤ד מוֹאָב֙ וְעָרֶ֣יהָ עָלָ֔ה וּמִבְחַ֥ר בַּֽחוּרָ֖יו יָרְד֣וּ לַטָּ֑בַח נְאֻ֨ם־ הַמֶּ֔לֶךְ יְהוָ֥ה צְבָא֖וֹת שְׁמֽוֹ׃ (כה) קָר֥וֹב אֵיד־ מוֹאָ֖ב לָב֑וֹא וְרָ֣עָת֔וֹ מִהֲרָ֖ה מְאֹֽד׃ (כו) נֻ֤דוּ לוֹ֙ כָּל־ סְבִיבָ֔יו וְכֹ֖ל יֹדְעֵ֣י שְׁמ֑וֹ אִמְר֗וּ אֵיכָ֤ה נִשְׁבַּר֙ מַטֵּה־ עֹ֔ז מַקֵּ֖ל תִּפְאָרָֽה׃ (כז) רְדִ֤י מִכָּבוֹד֙ (ישבי) וּשְׁבִ֣י בַצָּמָ֔א יֹשֶׁ֖בֶת בַּת־ דִּיב֑וֹן כִּֽי־ שֹׁדֵ֤ד מוֹאָב֙ עָ֣לָה בָ֔ךְ שִׁחֵ֖ת מִבְצָרָֽיִךְ׃ (כח) אֶל־ דֶּ֛רֶךְ עִמְדִ֥י וְצַפִּ֖י יוֹשֶׁ֣בֶת עֲרוֹעֵ֑ר שַׁאֲלִי־ נָ֣ס וְנִמְלָ֔טָה אִמְרִ֖י מַה־ נִּֽהְיָֽתָה׃ (כט) הֹבִ֥ישׁ מוֹאָ֛ב כִּֽי־ חַ֖תָּה (הילילי) הֵילִ֣ילוּ ׀ (וזעקי) וּֽזְעָ֑קוּ הַגִּ֣ידוּ בְאַרְנ֔וֹן כִּ֥י שֻׁדַּ֖ד מוֹאָֽב׃ (ל) וּמִשְׁפָּ֥ט בָּ֖א אֶל־ אֶ֣רֶץ הַמִּישֹׁ֑ר אֶל־ חֹל֥וֹן וְאֶל־ יַ֖הְצָה וְעַל־ (מופעת) מֵיפָֽעַת׃ (לא) וְעַל־ דִּיב֣וֹן וְעַל־ נְב֔וֹ וְעַל־ בֵּ֖ית דִּבְלָתָֽיִם׃ (לב) וְעַ֧ל קִרְיָתַ֛יִם וְעַל־ בֵּ֥ית גָּמ֖וּל וְעַל־ בֵּ֥ית מְעֽוֹן׃ (לג) וְעַל־ קְרִיּ֖וֹת וְעַל־ בָּצְרָ֑ה וְעַ֗ל כָּל־ עָרֵי֙ אֶ֣רֶץ מוֹאָ֔ב הָרְחֹק֖וֹת וְהַקְּרֹבֽוֹת׃ (לד) נִגְדְּעָה֙ קֶ֣רֶן מוֹאָ֔ב וּזְרֹע֖וֹ נִשְׁבָּ֑רָה נְאֻ֖ם יְהוָֽה׃ (לה) הַשְׁכִּירֻ֕הוּ כִּ֥י עַל־ יְהוָ֖ה הִגְדִּ֑יל וְסָפַ֤ק מוֹאָב֙ בְּקִיא֔וֹ וְהָיָ֥ה לִשְׂחֹ֖ק גַּם־ הֽוּא׃ (לו) וְאִ֣ם ׀ ל֣וֹא הַשְּׂחֹ֗ק הָיָ֤ה לְךָ֙ יִשְׂרָאֵ֔ל אִם־ בְּגַנָּבִ֖ים (נמצאה) נִמְצָ֑א כִּֽי־ מִדֵּ֧י דְבָרֶ֛יךָ בּ֖וֹ תִּתְנוֹדָֽד׃ (לז) עִזְב֤וּ עָרִים֙ וְשִׁכְנ֣וּ בַּסֶּ֔לַע יֹשְׁבֵ֖י מוֹאָ֑ב וִֽהְי֣וּ כְיוֹנָ֔ה תְּקַנֵּ֖ן בְּעֶבְרֵ֥י פִי־ פָֽחַת׃ (לח) שָׁמַ֥עְנוּ גְאוֹן־ מוֹאָ֖ב גֵּאֶ֣ה מְאֹ֑ד גָּבְה֧וֹ וּגְאוֹנ֛וֹ וְגַאֲוָת֖וֹ וְרֻ֥ם לִבּֽוֹ׃ (לט) אֲנִ֤י יָדַ֙עְתִּי֙ נְאֻם־ יְהוָ֔ה עֶבְרָת֖וֹ וְלֹא־ כֵ֑ן בַּדָּ֖יו לֹא־ כֵ֥ן עָשֽׂוּ׃ (מ) עַל־ כֵּן֙ עַל־ מוֹאָ֣ב אֲיֵלִ֔יל וּלְמוֹאָ֥ב כֻּלֹּ֖ה אֶזְעָ֑ק אֶל־ אַנְשֵׁ֥י קִֽיר־ חֶ֖רֶשׂ יֶהְגֶּֽה׃ (מא) מִבְּכִ֨י יַעְזֵ֤ר אֶבְכֶּה־ לָּךְ֙ הַגֶּ֣פֶן שִׂבְמָ֔ה נְטִֽישֹׁתַ֙יִךְ֙ עָ֣בְרוּ יָ֔ם עַ֛ד יָ֥ם יַעְזֵ֖ר נָגָ֑עוּ עַל־ קֵיצֵ֥ךְ וְעַל־ בְּצִירֵ֖ךְ שֹׁדֵ֥ד נָפָֽל׃ (מב) וְנֶאֶסְפָ֨ה שִׂמְחָ֥ה וָגִ֛יל מִכַּרְמֶ֖ל וּמֵאֶ֣רֶץ מוֹאָ֑ב וְיַ֙יִן֙ מִיקָבִ֣ים הִשְׁבַּ֔תִּי לֹֽא־ יִדְרֹ֣ךְ הֵידָ֔ד הֵידָ֖ד לֹ֥א הֵידָֽד׃ (מג) מִזַּעֲקַ֨ת חֶשְׁבּ֜וֹן עַד־ אֶלְעָלֵ֗ה עַד־ יַ֙הַץ֙ נָתְנ֣וּ קוֹלָ֔ם מִצֹּ֙עַר֙ עַד־ חֹ֣רֹנַ֔יִם עֶגְלַ֖ת שְׁלִֽשִׁיָּ֑ה כִּ֚י גַּם־ מֵ֣י נִמְרִ֔ים לִמְשַׁמּ֖וֹת יִהְיֽוּ׃ (מד) וְהִשְׁבַּתִּ֥י לְמוֹאָ֖ב נְאֻם־ יְהוָ֑ה מַעֲלֶ֣ה בָמָ֔ה וּמַקְטִ֖יר לֵאלֹהָֽיו׃ (מה) עַל־ כֵּ֞ן לִבִּ֤י לְמוֹאָב֙ כַּחֲלִלִ֣ים יֶהֱמֶ֔ה וְלִבִּי֙ אֶל־ אַנְשֵׁ֣י קִֽיר־ חֶ֔רֶשׂ כַּחֲלִילִ֖ים יֶהֱמֶ֑ה עַל־ כֵּ֛ן יִתְרַ֥ת עָשָׂ֖ה אָבָֽדוּ׃ (מו) כִּ֤י כָל־ רֹאשׁ֙ קָרְחָ֔ה וְכָל־ זָקָ֖ן גְּרֻעָ֑ה עַ֤ל כָּל־ יָדַ֙יִם֙ גְּדֻדֹ֔ת וְעַל־ מָתְנַ֖יִם שָֽׂק׃ (מז) עַ֣ל כָּל־ גַּגּ֥וֹת מוֹאָ֛ב וּבִרְחֹבֹתֶ֖יהָ כֻּלֹּ֣ה מִסְפֵּ֑ד כִּֽי־ שָׁבַ֣רְתִּי אֶת־ מוֹאָ֗ב כִּכְלִ֛י אֵֽין־ חֵ֥פֶץ בּ֖וֹ נְאֻם־ יְהוָֽה׃ (מח) אֵ֥יךְ חַ֙תָּה֙ הֵילִ֔ילוּ אֵ֛יךְ הִפְנָה־ עֹ֥רֶף מוֹאָ֖ב בּ֑וֹשׁ וְהָיָ֥ה מוֹאָ֛ב לִשְׂחֹ֥ק וְלִמְחִתָּ֖ה לְכָל־ סְבִיבָֽיו׃ (מט) כִּי־ כֹה֙ אָמַ֣ר יְהוָ֔ה הִנֵּ֥ה כַנֶּ֖שֶׁר יִדְאֶ֑ה וּפָרַ֥שׂ כְּנָפָ֖יו אֶל־ מוֹאָֽב׃ (נ) נִלְכְּדָה֙ הַקְּרִיּ֔וֹת וְהַמְּצָד֖וֹת נִתְפָּ֑שָׂה וְֽ֠הָיָה לֵ֞ב גִּבּוֹרֵ֤י מוֹאָב֙ בַּיּ֣וֹם הַה֔וּא כְּלֵ֖ב אִשָּׁ֥ה מְצֵרָֽה׃ (נא) וְנִשְׁמַ֥ד מוֹאָ֖ב מֵעָ֑ם כִּ֥י עַל־ יְהוָ֖ה הִגְדִּֽיל׃ (נב) פַּ֥חַד וָפַ֖חַת וָפָ֑ח עָלֶ֛יךָ יוֹשֵׁ֥ב מוֹאָ֖ב נְאֻם־ יְהוָֽה׃ (נג) (הניס) הַנָּ֞ס מִפְּנֵ֤י הַפַּ֙חַד֙ יִפֹּ֣ל אֶל־ הַפַּ֔חַת וְהָֽעֹלֶה֙ מִן־ הַפַּ֔חַת יִלָּכֵ֖ד בַּפָּ֑ח כִּֽי־ אָבִ֨יא אֵלֶ֧יהָ אֶל־ מוֹאָ֛ב שְׁנַ֥ת פְּקֻדָּתָ֖ם נְאֻם־ יְהוָֽה׃ (נד) בְּצֵ֥ל חֶשְׁבּ֛וֹן עָמְד֖וּ מִכֹּ֣חַ נָסִ֑ים כִּֽי־ אֵ֞שׁ יָצָ֣א מֵחֶשְׁבּ֗וֹן וְלֶֽהָבָה֙ מִבֵּ֣ין סִיח֔וֹן וַתֹּ֙אכַל֙ פְּאַ֣ת מוֹאָ֔ב וְקָדְקֹ֖ד בְּנֵ֥י שָׁאֽוֹן׃ (נה) אוֹי־ לְךָ֣ מוֹאָ֔ב אָבַ֖ד עַם־ כְּמ֑וֹשׁ כִּֽי־ לֻקְּח֤וּ בָנֶ֙יךָ֙ בַּשֶּׁ֔בִי וּבְנֹתֶ֖יךָ בַּשִּׁבְיָֽה׃ (נו) וְשַׁבְתִּ֧י שְׁבוּת־ מוֹאָ֛ב בְּאַחֲרִ֥ית הַיָּמִ֖ים נְאֻם־ יְהוָ֑ה עַד־ הֵ֖נָּה מִשְׁפַּ֥ט מוֹאָֽב׃ סוף פרק מח (נז) לִבְנֵ֣י עַמּ֗וֹן כֹּ֚ה אָמַ֣ר יְהוָ֔ה הֲבָנִ֥ים אֵין֙ לְיִשְׂרָאֵ֔ל אִם־ יוֹרֵ֖שׁ אֵ֣ין ל֑וֹ מַדּ֗וּעַ יָרַ֤שׁ מַלְכָּם֙ אֶת־ גָּ֔ד וְעַמּ֖וֹ בְּעָרָ֥יו יָשָֽׁב׃
