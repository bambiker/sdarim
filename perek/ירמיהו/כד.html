ירמיהו פרק כד לפי החלוקה לסדרים

פרק כד בספר ירמיהו (10 פסוקים) נמצא בירמיהו סדר יב (פסוקים לו–מא) ובירמיהו סדר יג (פסוקים א–ד).

ירמיהו סדר יב

(א) בְּיָמָיו֙ תִּוָּשַׁ֣ע יְהוּדָ֔ה וְיִשְׂרָאֵ֖ל יִשְׁכֹּ֣ן לָבֶ֑טַח וְזֶה־ שְּׁמ֥וֹ אֲ‍ֽשֶׁר־ יִקְרְא֖וֹ יְהוָ֥ה ׀ צִדְקֵֽנוּ׃ (ב) לָכֵ֛ן הִנֵּֽה־ יָמִ֥ים בָּאִ֖ים נְאֻם־ יְהוָ֑ה וְלֹא־ יֹ֤אמְרוּ עוֹד֙ חַי־ יְהוָ֔ה אֲשֶׁ֧ר הֶעֱלָ֛ה אֶת־ בְּנֵ֥י יִשְׂרָאֵ֖ל מֵאֶ֥רֶץ מִצְרָֽיִם׃ (ג) כִּ֣י אִם־ חַי־ יְהוָ֗ה אֲשֶׁ֣ר הֶעֱלָה֩ וַאֲשֶׁ֨ר הֵבִ֜יא אֶת־ זֶ֨רַע בֵּ֤ית יִשְׂרָאֵל֙ מֵאֶ֣רֶץ צָפ֔וֹנָה וּמִכֹּל֙ הָֽאֲרָצ֔וֹת אֲשֶׁ֥ר הִדַּחְתִּ֖ים שָׁ֑ם וְיָשְׁב֖וּ עַל־ אַדְמָתָֽם׃ (ד) לַנְּבִאִ֞ים נִשְׁבַּ֧ר לִבִּ֣י בְקִרְבִּ֗י רָֽחֲפוּ֙ כָּל־ עַצְמוֹתַ֔י הָיִ֙יתִי֙ כְּאִ֣ישׁ שִׁכּ֔וֹר וּכְגֶ֖בֶר עֲבָ֣רוֹ יָ֑יִן מִפְּנֵ֣י יְהוָ֔ה וּמִפְּנֵ֖י דִּבְרֵ֥י קָדְשֽׁוֹ׃ (ה) כִּ֤י מְנָֽאֲפִים֙ מָלְאָ֣ה הָאָ֔רֶץ כִּֽי־ מִפְּנֵ֤י אָלָה֙ אָבְלָ֣ה הָאָ֔רֶץ יָבְשׁ֖וּ נְא֣וֹת מִדְבָּ֑ר וַתְּהִ֤י מְרֽוּצָתָם֙ רָעָ֔ה וּגְבוּרָתָ֖ם לֹא־ כֵֽן׃ (ו) כִּֽי־ גַם־ נָבִ֥יא גַם־ כֹּהֵ֖ן חָנֵ֑פוּ גַּם־ בְּבֵיתִ֛י מָצָ֥אתִי רָעָתָ֖ם נְאֻם־ יְהוָֽה׃ (ז) לָכֵן֩ יִֽהְיֶ֨ה דַרְכָּ֜ם לָהֶ֗ם כַּחֲלַקְלַקּוֹת֙ בָּֽאֲפֵלָ֔ה יִדַּ֖חוּ וְנָ֣פְלוּ בָ֑הּ כִּֽי־ אָבִ֨יא עֲלֵיהֶ֥ם רָעָ֛ה שְׁנַ֥ת פְּקֻדָּתָ֖ם נְאֻם־ יְהוָֽה׃ (ח) וּבִנְבִיאֵ֥י שֹׁמְר֖וֹן רָאִ֣יתִי תִפְלָ֑ה הִנַּבְּא֣וּ בַבַּ֔עַל וַיַּתְע֥וּ אֶת־ עַמִּ֖י אֶת־ יִשְׂרָאֵֽל׃ (ט) וּבִנְבִאֵ֨י יְרוּשָׁלִַ֜ם רָאִ֣יתִי שַׁעֲרוּרָ֗ה נָא֞וֹף וְהָלֹ֤ךְ בַּשֶּׁ֙קֶר֙ וְחִזְּקוּ֙ יְדֵ֣י מְרֵעִ֔ים לְבִ֨לְתִּי־ שָׁ֔בוּ אִ֖ישׁ מֵרָֽעָת֑וֹ הָֽיוּ־ לִ֤י כֻלָּם֙ כִּסְדֹ֔ם וְיֹשְׁבֶ֖יהָ כַּעֲמֹרָֽה׃ (י) לָכֵ֞ן כֹּֽה־ אָמַ֨ר יְהוָ֤ה צְבָאוֹת֙ עַל־ הַנְּבִאִ֔ים הִנְנִ֨י מַאֲכִ֤יל אוֹתָם֙ לַֽעֲנָ֔ה וְהִשְׁקִתִ֖ים מֵי־ רֹ֑אשׁ כִּ֗י מֵאֵת֙ נְבִיאֵ֣י יְרוּשָׁלִַ֔ם יָצְאָ֥ה חֲנֻפָּ֖ה לְכָל־ הָאָֽרֶץ׃ (יא) כֹּֽה־ אָמַ֞ר יְהוָ֣ה צְבָא֗וֹת אַֽל־ תִּשְׁמְע֞וּ עַל־ דִּבְרֵ֤י הַנְּבִאִים֙ הַנִּבְּאִ֣ים לָכֶ֔ם מַהְבִּלִ֥ים הֵ֖מָּה אֶתְכֶ֑ם חֲז֤וֹן לִבָּם֙ יְדַבֵּ֔רוּ לֹ֖א מִפִּ֥י יְהוָֽה׃ (יב) אֹמְרִ֤ים אָמוֹר֙ לִֽמְנַאֲצַ֔י דִּבֶּ֣ר יְהוָ֔ה שָׁל֖וֹם יִֽהְיֶ֣ה לָכֶ֑ם וְ֠כֹל הֹלֵ֞ךְ בִּשְׁרִר֤וּת לִבּוֹ֙ אָֽמְר֔וּ לֹֽא־ תָב֥וֹא עֲלֵיכֶ֖ם רָעָֽה׃ (יג) כִּ֣י מִ֤י עָמַד֙ בְּס֣וֹד יְהוָ֔ה וְיֵ֖רֶא וְיִשְׁמַ֣ע אֶת־ דְּבָר֑וֹ מִֽי־ הִקְשִׁ֥יב (דברי) דְּבָר֖וֹ וַיִּשְׁמָֽע׃ (יד) הִנֵּ֣ה ׀ סַעֲרַ֣ת יְהוָ֗ה חֵמָה֙ יָֽצְאָ֔ה וְסַ֖עַר מִתְחוֹלֵ֑ל עַ֛ל רֹ֥אשׁ רְשָׁעִ֖ים יָחֽוּל׃ (טו) לֹ֤א יָשׁוּב֙ אַף־ יְהוָ֔ה עַד־ עֲשֹׂת֥וֹ וְעַד־ הֲקִימ֖וֹ מְזִמּ֣וֹת לִבּ֑וֹ בְּאַֽחֲרִית֙ הַיָּמִ֔ים תִּתְבּ֥וֹנְנוּ בָ֖הּ בִּינָֽה׃ (טז) לֹא־ שָׁלַ֥חְתִּי אֶת־ הַנְּבִאִ֖ים וְהֵ֣ם רָ֑צוּ לֹא־ דִבַּ֥רְתִּי אֲלֵיהֶ֖ם וְהֵ֥ם נִבָּֽאוּ׃ (יז) וְאִֽם־ עָמְד֖וּ בְּסוֹדִ֑י וְיַשְׁמִ֤עוּ דְבָרַי֙ אֶת־ עַמִּ֔י וִֽישִׁבוּם֙ מִדַּרְכָּ֣ם הָרָ֔ע וּמֵרֹ֖עַ מַֽעַלְלֵיהֶֽם׃ (יח) הַאֱלֹהֵ֧י מִקָּרֹ֛ב אָ֖נִי נְאֻם־ יְהוָ֑ה וְלֹ֥א אֱלֹהֵ֖י מֵרָחֹֽק׃ (יט) אִם־ יִסָּתֵ֨ר אִ֧ישׁ בַּמִּסְתָּרִ֛ים וַאֲנִ֥י לֹֽא־ אֶרְאֶ֖נּוּ נְאֻם־ יְהוָ֑ה הֲל֨וֹא אֶת־ הַשָּׁמַ֧יִם וְאֶת־ הָאָ֛רֶץ אֲנִ֥י מָלֵ֖א נְאֻם־ יְהוָֽה׃ (כ) שָׁמַ֗עְתִּי אֵ֤ת אֲשֶׁר־ אָֽמְרוּ֙ הַנְּבִאִ֔ים הַֽנִּבְּאִ֥ים בִּשְׁמִ֛י שֶׁ֖קֶר לֵאמֹ֑ר חָלַ֖מְתִּי חָלָֽמְתִּי׃ (כא) עַד־ מָתַ֗י הֲיֵ֛שׁ בְּלֵ֥ב הַנְּבִאִ֖ים נִבְּאֵ֣י הַשָּׁ֑קֶר וּנְבִיאֵ֖י תַּרְמִ֥ת לִבָּֽם׃ (כב) הַחֹשְׁבִ֗ים לְהַשְׁכִּ֤יחַ אֶת־ עַמִּי֙ שְׁמִ֔י בַּחֲל֣וֹמֹתָ֔ם אֲשֶׁ֥ר יְסַפְּר֖וּ אִ֣ישׁ לְרֵעֵ֑הוּ כַּאֲשֶׁ֨ר שָׁכְח֧וּ אֲבוֹתָ֛ם אֶת־ שְׁמִ֖י בַּבָּֽעַל׃ (כג) הַנָּבִ֞יא אֲשֶׁר־ אִתּ֤וֹ חֲלוֹם֙ יְסַפֵּ֣ר חֲל֔וֹם וַאֲשֶׁ֤ר דְּבָרִי֙ אִתּ֔וֹ יְדַבֵּ֥ר דְּבָרִ֖י אֱמֶ֑ת מַה־ לַתֶּ֥בֶן אֶת־ הַבָּ֖ר נְאֻם־ יְהוָֽה׃ (כד) הֲל֨וֹא כֹ֧ה דְבָרִ֛י כָּאֵ֖שׁ נְאֻם־ יְהוָ֑ה וּכְפַטִּ֖ישׁ יְפֹ֥צֵֽץ סָֽלַע׃ (כה) לָכֵ֛ן הִנְנִ֥י עַל־ הַנְּבִאִ֖ים נְאֻם־ יְהוָ֑ה מְגַנְּבֵ֣י דְבָרַ֔י אִ֖ישׁ מֵאֵ֥ת רֵעֵֽהוּ׃ (כו) הִנְנִ֥י עַל־ הַנְּבִיאִ֖ם נְאֻם־ יְהוָ֑ה הַלֹּקְחִ֣ים לְשׁוֹנָ֔ם וַֽיִּנְאֲמ֖וּ נְאֻֽם׃ (כז) הִ֠נְנִי עַֽל־ נִבְּאֵ֞י חֲלֹמ֥וֹת שֶׁ֙קֶר֙ נְאֻם־ יְהוָ֔ה וַֽיְסַפְּרוּם֙ וַיַּתְע֣וּ אֶת־ עַמִּ֔י בְּשִׁקְרֵיהֶ֖ם וּבְפַחֲזוּתָ֑ם וְאָנֹכִ֨י לֹֽא־ שְׁלַחְתִּ֜ים וְלֹ֣א צִוִּיתִ֗ים וְהוֹעֵ֛יל לֹֽא־ יוֹעִ֥ילוּ לָֽעָם־ הַזֶּ֖ה נְאֻם־ יְהוָֽה׃ (כח) וְכִי־ יִשְׁאָלְךָ֩ הָעָ֨ם הַזֶּ֜ה אֽוֹ־ הַנָּבִ֤יא אֽוֹ־ כֹהֵן֙ לֵאמֹ֔ר מַה־ מַשָּׂ֖א יְהוָ֑ה וְאָמַרְתָּ֤ אֲלֵיהֶם֙ אֶת־ מַה־ מַשָּׂ֔א וְנָטַשְׁתִּ֥י אֶתְכֶ֖ם נְאֻם־ יְהוָֽה׃ (כט) וְהַנָּבִ֤יא וְהַכֹּהֵן֙ וְהָעָ֔ם אֲשֶׁ֥ר יֹאמַ֖ר מַשָּׂ֣א יְהוָ֑ה וּפָקַדְתִּ֛י עַל־ הָאִ֥ישׁ הַה֖וּא וְעַל־ בֵּיתֽוֹ׃ (ל) כֹּ֥ה תֹאמְר֛וּ אִ֥ישׁ עַל־ רֵעֵ֖הוּ וְאִ֣ישׁ אֶל־ אָחִ֑יו מֶה־ עָנָ֣ה יְהוָ֔ה וּמַה־ דִּבֶּ֖ר יְהוָֽה׃ (לא) וּמַשָּׂ֥א יְהוָ֖ה לֹ֣א תִזְכְּרוּ־ ע֑וֹד כִּ֣י הַמַּשָּׂ֗א יִֽהְיֶה֙ לְאִ֣ישׁ דְּבָר֔וֹ וַהֲפַכְתֶּ֗ם אֶת־ דִּבְרֵי֙ אֱלֹהִ֣ים חַיִּ֔ים יְהוָ֥ה צְבָא֖וֹת אֱלֹהֵֽינוּ׃ (לב) כֹּ֥ה תֹאמַ֖ר אֶל־ הַנָּבִ֑יא מֶה־ עָנָ֣ךְ יְהוָ֔ה וּמַה־ דִּבֶּ֖ר יְהוָֽה׃ (לג) וְאִם־ מַשָּׂ֣א יְהוָה֮ תֹּאמֵרוּ֒ לָכֵ֗ן כֹּ֚ה אָמַ֣ר יְהוָ֔ה יַ֧עַן אֲמָרְכֶ֛ם אֶת־ הַדָּבָ֥ר הַזֶּ֖ה מַשָּׂ֣א יְהוָ֑ה וָאֶשְׁלַ֤ח אֲלֵיכֶם֙ לֵאמֹ֔ר לֹ֥א תֹאמְר֖וּ מַשָּׂ֥א יְהוָֽה׃ (לד) לָכֵ֣ן הִנְנִ֔י וְנָשִׁ֥יתִי אֶתְכֶ֖ם נָשֹׁ֑א וְנָטַשְׁתִּ֣י אֶתְכֶ֗ם וְאֶת־ הָעִיר֙ אֲשֶׁ֨ר נָתַ֧תִּי לָכֶ֛ם וְלַאֲבוֹתֵיכֶ֖ם מֵעַ֥ל פָּנָֽי׃ (לה) וְנָתַתִּ֥י עֲלֵיכֶ֖ם חֶרְפַּ֣ת עוֹלָ֑ם וּכְלִמּ֣וּת עוֹלָ֔ם אֲשֶׁ֖ר לֹ֥א תִשָּׁכֵֽחַ׃ תחילת פרק כד (לו) הִרְאַנִי֮ יְהוָה֒ וְהִנֵּ֗ה שְׁנֵי֙ דּוּדָאֵ֣י תְאֵנִ֔ים מוּעָדִ֕ים לִפְנֵ֖י הֵיכַ֣ל יְהוָ֑ה אַחֲרֵ֣י הַגְל֣וֹת נְבוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֡ל אֶת־ יְכָנְיָ֣הוּ בֶן־ יְהוֹיָקִ֣ים מֶֽלֶךְ־ יְהוּדָה֩ וְאֶת־ שָׂרֵ֨י יְהוּדָ֜ה וְאֶת־ הֶחָרָ֤שׁ וְאֶת־ הַמַּסְגֵּר֙ מִיר֣וּשָׁלִַ֔ם וַיְבִאֵ֖ם בָּבֶֽל׃ (לז) הַדּ֣וּד אֶחָ֗ד תְּאֵנִים֙ טֹב֣וֹת מְאֹ֔ד כִּתְאֵנֵ֖י הַבַּכֻּר֑וֹת וְהַדּ֣וּד אֶחָ֗ד תְּאֵנִים֙ רָע֣וֹת מְאֹ֔ד אֲשֶׁ֥ר לֹא־ תֵֽאָכַ֖לְנָה מֵרֹֽעַ׃ (לח) וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י מָֽה־ אַתָּ֤ה רֹאֶה֙ יִרְמְיָ֔הוּ וָאֹמַ֖ר תְּאֵנִ֑ים הַתְּאֵנִ֤ים הַטֹּבוֹת֙ טֹב֣וֹת מְאֹ֔ד וְהָֽרָעוֹת֙ רָע֣וֹת מְאֹ֔ד אֲשֶׁ֥ר לֹא־ תֵאָכַ֖לְנָה מֵרֹֽעַ׃ (לט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (מ) כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל שְׁת֤וּ וְשִׁכְרוּ֙ וּקְי֔וּ וְנִפְל֖וּ וְלֹ֣א תָק֑וּמוּ מִפְּנֵ֣י הַחֶ֔רֶב אֲשֶׁ֛ר אָנֹכִ֥י שֹׁלֵ֖חַ בֵּינֵיכֶֽם׃ (לב) וְהָיָ֗ה כִּ֧י יְמָאֲנ֛וּ לָקַֽחַת־ הַכּ֥וֹס מִיָּדְךָ֖ לִשְׁתּ֑וֹת וְאָמַרְתָּ֣ אֲלֵיהֶ֗ם כֹּ֥ה אָמַ֛ר יְהוָ֥ה צְבָא֖וֹת שָׁת֥וֹ תִשְׁתּֽוּ׃ (לג) כִּי֩ הִנֵּ֨ה בָעִ֜יר אֲשֶׁ֧ר נִֽקְרָא־ שְׁמִ֣י עָלֶ֗יהָ אָֽנֹכִי֙ מֵחֵ֣ל לְהָרַ֔ע וְאַתֶּ֖ם הִנָּקֵ֣ה תִנָּק֑וּ לֹ֣א תִנָּק֔וּ כִּ֣י חֶ֗רֶב אֲנִ֤י קֹרֵא֙ עַל־ כָּל־ יֹשְׁבֵ֣י הָאָ֔רֶץ נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (לד) וְאַתָּה֙ תִּנָּבֵ֣א אֲלֵיהֶ֔ם אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וְאָמַרְתָּ֣ אֲלֵיהֶ֗ם יְהוָ֞ה מִמָּר֤וֹם יִשְׁאָג֙ וּמִמְּע֤וֹן קָדְשׁוֹ֙ יִתֵּ֣ן קוֹל֔וֹ שָׁאֹ֤ג יִשְׁאַג֙ עַל־ נָוֵ֔הוּ הֵידָד֙ כְּדֹרְכִ֣ים יַֽעֲנֶ֔ה אֶ֥ל כָּל־ יֹשְׁבֵ֖י הָאָֽרֶץ׃ (לה) בָּ֤א שָׁאוֹן֙ עַד־ קְצֵ֣ה הָאָ֔רֶץ כִּ֣י רִ֤יב לַֽיהוָה֙ בַּגּוֹיִ֔ם נִשְׁפָּ֥ט ה֖וּא לְכָל־ בָּשָׂ֑ר הָרְשָׁעִ֛ים נְתָנָ֥ם לַחֶ֖רֶב נְאֻם־ יְהוָֽה׃ (לו) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת הִנֵּ֥ה רָעָ֛ה יֹצֵ֖את מִגּ֣וֹי אֶל־ גּ֑וֹי וְסַ֣עַר גָּד֔וֹל יֵע֖וֹר מִיַּרְכְּתֵי־ אָֽרֶץ׃ (לז) וְהָי֞וּ חַֽלְלֵ֤י יְהוָה֙ בַּיּ֣וֹם הַה֔וּא מִקְצֵ֥ה הָאָ֖רֶץ וְעַד־ קְצֵ֣ה הָאָ֑רֶץ לֹ֣א יִסָּפְד֗וּ וְלֹ֤א יֵאָֽסְפוּ֙ וְלֹ֣א יִקָּבֵ֔רוּ לְדֹ֛מֶן עַל־ פְּנֵ֥י הָאֲדָמָ֖ה יִֽהְיֽוּ׃ (לח) הֵילִ֨ילוּ הָרֹעִ֜ים וְזַעֲק֗וּ וְהִֽתְפַּלְּשׁוּ֙ אַדִּירֵ֣י הַצֹּ֔אן כִּֽי־ מָלְא֥וּ יְמֵיכֶ֖ם לִטְב֑וֹחַ וּתְפוֹצ֣וֹתִיכֶ֔ם וּנְפַלְתֶּ֖ם כִּכְלִ֥י חֶמְדָּֽה׃ (לט) וְאָבַ֥ד מָנ֖וֹס מִן־ הָֽרֹעִ֑ים וּפְלֵיטָ֖ה מֵאַדִּירֵ֥י הַצֹּֽאן׃ (מ) ק֚וֹל צַעֲקַ֣ת הָֽרֹעִ֔ים וִֽילְלַ֖ת אַדִּירֵ֣י הַצֹּ֑אן כִּֽי־ שֹׁדֵ֥ד יְהוָ֖ה אֶת־ מַרְעִיתָֽם׃ (מא) וְנָדַ֖מּוּ נְא֣וֹת הַשָּׁל֑וֹם מִפְּנֵ֖י חֲר֥וֹן אַף־ יְהוָֽה׃ (מב) עָזַ֥ב כַּכְּפִ֖יר סֻכּ֑וֹ כִּֽי־ הָיְתָ֤ה אַרְצָם֙ לְשַׁמָּ֔ה מִפְּנֵי֙ חֲר֣וֹן הַיּוֹנָ֔ה וּמִפְּנֵ֖י חֲר֥וֹן אַפּֽוֹ׃
