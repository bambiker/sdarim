ירמיהו פרק כ לפי החלוקה לסדרים

פרק כ בספר ירמיהו (18 פסוקים) נמצא בירמיהו סדר י (פסוקים כא–לב) ובירמיהו סדר יא (פסוקים א–ו).

ירמיהו סדר י

(א) הַקְשִׁ֥יבָה יְהוָ֖ה אֵלָ֑י וּשְׁמַ֖ע לְק֥וֹל יְרִיבָֽי׃ (ב) הַיְשֻׁלַּ֤ם תַּֽחַת־ טוֹבָה֙ רָעָ֔ה כִּֽי־ כָר֥וּ שׁוּחָ֖ה לְנַפְשִׁ֑י זְכֹ֣ר ׀ עָמְדִ֣י לְפָנֶ֗יךָ לְדַבֵּ֤ר עֲלֵיהֶם֙ טוֹבָ֔ה לְהָשִׁ֥יב אֶת־ חֲמָתְךָ֖ מֵהֶֽם׃ (ג) לָכֵן֩ תֵּ֨ן אֶת־ בְּנֵיהֶ֜ם לָרָעָ֗ב וְהַגִּרֵם֮ עַל־ יְדֵי־ חֶרֶב֒ וְתִֽהְיֶ֨נָה נְשֵׁיהֶ֤ם שַׁכֻּלוֹת֙ וְאַלְמָנ֔וֹת וְאַ֨נְשֵׁיהֶ֔ם יִֽהְי֖וּ הֲרֻ֣גֵי מָ֑וֶת בַּח֣וּרֵיהֶ֔ם מֻכֵּי־ חֶ֖רֶב בַּמִּלְחָמָֽה׃ (ד) תִּשָּׁמַ֤ע זְעָקָה֙ מִבָּ֣תֵּיהֶ֔ם כִּֽי־ תָבִ֧יא עֲלֵיהֶ֛ם גְּד֖וּד פִּתְאֹ֑ם כִּֽי־ כָר֤וּ (שיחה) שׁוּחָה֙ לְלָכְדֵ֔נִי וּפַחִ֖ים טָמְנ֥וּ לְרַגְלָֽי׃ (ה) וְאַתָּ֣ה יְ֠הוָה יָדַ֜עְתָּ אֶֽת־ כָּל־ עֲצָתָ֤ם עָלַי֙ לַמָּ֔וֶת אַל־ תְּכַפֵּר֙ עַל־ עֲוֺנָ֔ם וְחַטָּאתָ֖ם מִלְּפָנֶ֣יךָ אַל־ תֶּ֑מְחִי (והיו) וְיִהְי֤וּ מֻכְשָׁלִים֙ לְפָנֶ֔יךָ בְּעֵ֥ת אַפְּךָ֖ עֲשֵׂ֥ה בָהֶֽם׃ (ו) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הָל֛וֹךְ וְקָנִ֥יתָ בַקְבֻּ֖ק יוֹצֵ֣ר חָ֑רֶשׂ וּמִזִּקְנֵ֣י הָעָ֔ם וּמִזִּקְנֵ֖י הַכֹּהֲנִֽים׃ (ז) וְיָצָ֙אתָ֙ אֶל־ גֵּ֣יא בֶן־ הִנֹּ֔ם אֲשֶׁ֕ר פֶּ֖תַח שַׁ֣עַר (החרסות) הַֽחַרְסִ֑ית וְקָרָ֣אתָ שָּׁ֔ם אֶת־ הַדְּבָרִ֖ים אֲשֶׁר־ אֲדַבֵּ֥ר אֵלֶֽיךָ׃ (ח) וְאָֽמַרְתָּ֙ שִׁמְע֣וּ דְבַר־ יְהוָ֔ה מַלְכֵ֣י יְהוּדָ֔ה וְיֹשְׁבֵ֖י יְרֽוּשָׁלִָ֑ם כֹּֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל הִנְנִ֨י מֵבִ֤יא רָעָה֙ עַל־ הַמָּק֣וֹם הַזֶּ֔ה אֲשֶׁ֥ר כָּל־ שֹׁמְעָ֖הּ תִּצַּ֥לְנָה אָזְנָֽיו׃ (ט) יַ֣עַן ׀ אֲשֶׁ֣ר עֲזָבֻ֗נִי וַֽיְנַכְּר֞וּ אֶת־ הַמָּק֤וֹם הַזֶּה֙ וַיְקַטְּרוּ־ בוֹ֙ לֵאלֹהִ֣ים אֲחֵרִ֔ים אֲשֶׁ֧ר לֹֽא־ יְדָע֛וּם הֵ֥מָּה וַאֲבֽוֹתֵיהֶ֖ם וּמַלְכֵ֣י יְהוּדָ֑ה וּמָֽלְא֛וּ אֶת־ הַמָּק֥וֹם הַזֶּ֖ה דַּ֥ם נְקִיִּֽם׃ (י) וּבָנ֞וּ אֶת־ בָּמ֣וֹת הַבַּ֗עַל לִשְׂרֹ֧ף אֶת־ בְּנֵיהֶ֛ם בָּאֵ֖שׁ עֹל֣וֹת לַבָּ֑עַל אֲשֶׁ֤ר לֹֽא־ צִוִּ֙יתִי֙ וְלֹ֣א דִבַּ֔רְתִּי וְלֹ֥א עָלְתָ֖ה עַל־ לִבִּֽי׃ (יא) לָכֵ֞ן הִנֵּֽה־ יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וְלֹא־ יִקָּרֵא֩ לַמָּק֨וֹם הַזֶּ֥ה ע֛וֹד הַתֹּ֖פֶת וְגֵ֣יא בֶן־ הִנֹּ֑ם כִּ֖י אִם־ גֵּ֥יא הַהֲרֵגָֽה׃ (יב) וּ֠בַקֹּתִי אֶת־ עֲצַ֨ת יְהוּדָ֤ה וִירוּשָׁלִַ֙ם֙ בַּמָּק֣וֹם הַזֶּ֔ה וְהִפַּלְתִּ֤ים בַּחֶ֙רֶב֙ לִפְנֵ֣י אֹֽיְבֵיהֶ֔ם וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשָׁ֑ם וְנָתַתִּ֤י אֶת־ נִבְלָתָם֙ לְמַֽאֲכָ֔ל לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֥ת הָאָֽרֶץ׃ (יג) וְשַׂמְתִּי֙ אֶת־ הָעִ֣יר הַזֹּ֔את לְשַׁמָּ֖ה וְלִשְׁרֵקָ֑ה כֹּ֚ל עֹבֵ֣ר עָלֶ֔יהָ יִשֹּׁ֥ם וְיִשְׁרֹ֖ק עַל־ כָּל־ מַכֹּתֶֽהָ׃ (יד) וְהַֽאֲכַלְתִּ֞ים אֶת־ בְּשַׂ֣ר בְּנֵיהֶ֗ם וְאֵת֙ בְּשַׂ֣ר בְּנֹתֵיהֶ֔ם וְאִ֥ישׁ בְּשַׂר־ רֵעֵ֖הוּ יֹאכֵ֑לוּ בְּמָצוֹר֙ וּבְמָצ֔וֹק אֲשֶׁ֨ר יָצִ֧יקוּ לָהֶ֛ם אֹיְבֵיהֶ֖ם וּמְבַקְשֵׁ֥י נַפְשָֽׁם׃ (טו) וְשָׁבַרְתָּ֖ הַבַּקְבֻּ֑ק לְעֵינֵי֙ הָֽאֲנָשִׁ֔ים הַהֹלְכִ֖ים אוֹתָֽךְ׃ (טז) וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם כֹּה־ אָמַ֣ר ׀ יְהוָ֣ה צְבָא֗וֹת כָּ֣כָה אֶשְׁבֹּ֞ר אֶת־ הָעָ֤ם הַזֶּה֙ וְאֶת־ הָעִ֣יר הַזֹּ֔את כַּאֲשֶׁ֤ר יִשְׁבֹּר֙ אֶת־ כְּלִ֣י הַיּוֹצֵ֔ר אֲשֶׁ֛ר לֹֽא־ יוּכַ֥ל לְהֵרָפֵ֖ה ע֑וֹד וּבְתֹ֣פֶת יִקְבְּר֔וּ מֵאֵ֥ין מָק֖וֹם לִקְבּֽוֹר׃ (יז) כֵּֽן־ אֶעֱשֶׂ֞ה לַמָּק֥וֹם הַזֶּ֛ה נְאֻם־ יְהוָ֖ה וּלְיֽוֹשְׁבָ֑יו וְלָתֵ֛ת אֶת־ הָעִ֥יר הַזֹּ֖את כְּתֹֽפֶת׃ (יח) וְהָי֞וּ בָּתֵּ֣י יְרוּשָׁלִַ֗ם וּבָתֵּי֙ מַלְכֵ֣י יְהוּדָ֔ה כִּמְק֥וֹם הַתֹּ֖פֶת הַטְּמֵאִ֑ים לְכֹ֣ל הַבָּתִּ֗ים אֲשֶׁ֨ר קִטְּר֜וּ עַל־ גַּגֹּֽתֵיהֶם֙ לְכֹל֙ צְבָ֣א הַשָּׁמַ֔יִם וְהַסֵּ֥ךְ נְסָכִ֖ים לֵאלֹהִ֥ים אֲחֵרִֽים׃ (יט) וַיָּבֹ֤א יִרְמְיָ֙הוּ֙ מֵֽהַתֹּ֔פֶת אֲשֶׁ֨ר שְׁלָח֧וֹ יְהוָ֛ה שָׁ֖ם לְהִנָּבֵ֑א וַֽיַּעֲמֹד֙ בַּחֲצַ֣ר בֵּית־ יְהוָ֔ה וַיֹּ֖אמֶר אֶל־ כָּל־ הָעָֽם׃ (כ) כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֨י (מבי) מֵבִ֜יא אֶל־ הָעִ֤יר הַזֹּאת֙ וְעַל־ כָּל־ עָרֶ֔יהָ אֵ֚ת כָּל־ הָ֣רָעָ֔ה אֲשֶׁ֥ר דִּבַּ֖רְתִּי עָלֶ֑יהָ כִּ֤י הִקְשׁוּ֙ אֶת־ עָרְפָּ֔ם לְבִלְתִּ֖י שְׁמ֥וֹעַ אֶת־ דְּבָרָֽי׃ תחילת פרק כ (כא) וַיִּשְׁמַ֤ע פַּשְׁחוּר֙ בֶּן־ אִמֵּ֣ר הַכֹּהֵ֔ן וְהֽוּא־ פָקִ֥יד נָגִ֖יד בְּבֵ֣ית יְהוָ֑ה אֶֽת־ יִרְמְיָ֔הוּ נִבָּ֖א אֶת־ הַדְּבָרִ֥ים הָאֵֽלֶּה׃ (כב) וַיַּכֶּ֣ה פַשְׁח֔וּר אֵ֖ת יִרְמְיָ֣הוּ הַנָּבִ֑יא וַיִּתֵּ֨ן אֹת֜וֹ עַל־ הַמַּהְפֶּ֗כֶת אֲשֶׁ֨ר בְּשַׁ֤עַר בִּנְיָמִן֙ הָֽעֶלְי֔וֹן אֲשֶׁ֖ר בְּבֵ֥ית יְהוָֽה׃ (כג) וַֽיְהִי֙ מִֽמָּחֳרָ֔ת וַיֹּצֵ֥א פַשְׁח֛וּר אֶֽת־ יִרְמְיָ֖הוּ מִן־ הַמַּהְפָּ֑כֶת וַיֹּ֨אמֶר אֵלָ֜יו יִרְמְיָ֗הוּ לֹ֤א פַשְׁחוּר֙ קָרָ֤א יְהוָה֙ שְׁמֶ֔ךָ כִּ֖י אִם־ מָג֥וֹר מִסָּבִֽיב׃ (כד) כִּ֣י כֹ֣ה אָמַ֣ר יְהוָ֡ה הִנְנִי֩ נֹתֶנְךָ֨ לְמָג֜וֹר לְךָ֣ וּלְכָל־ אֹהֲבֶ֗יךָ וְנָֽפְל֛וּ בְּחֶ֥רֶב אֹיְבֵיהֶ֖ם וְעֵינֶ֣יךָ רֹא֑וֹת וְאֶת־ כָּל־ יְהוּדָ֗ה אֶתֵּן֙ בְּיַ֣ד מֶֽלֶךְ־ בָּבֶ֔ל וְהִגְלָ֥ם בָּבֶ֖לָה וְהִכָּ֥ם בֶּחָֽרֶב׃ (כה) וְנָתַתִּ֗י אֶת־ כָּל־ חֹ֙סֶן֙ הָעִ֣יר הַזֹּ֔את וְאֶת־ כָּל־ יְגִיעָ֖הּ וְאֶת־ כָּל־ יְקָרָ֑הּ וְאֵ֨ת כָּל־ אוֹצְר֜וֹת מַלְכֵ֣י יְהוּדָ֗ה אֶתֵּן֙ בְּיַ֣ד אֹֽיְבֵיהֶ֔ם וּבְזָזוּם֙ וּלְקָח֔וּם וֶהֱבִיא֖וּם בָּבֶֽלָה׃ (כו) וְאַתָּ֣ה פַשְׁח֗וּר וְכֹל֙ יֹשְׁבֵ֣י בֵיתֶ֔ךָ תֵּלְכ֖וּ בַּשֶּׁ֑בִי וּבָבֶ֣ל תָּב֗וֹא וְשָׁ֤ם תָּמוּת֙ וְשָׁ֣ם תִּקָּבֵ֔ר אַתָּה֙ וְכָל־ אֹ֣הֲבֶ֔יךָ אֲשֶׁר־ נִבֵּ֥אתָ לָהֶ֖ם בַּשָּֽׁקֶר׃ (כז) פִּתִּיתַ֤נִי יְהוָה֙ וָֽאֶפָּ֔ת חֲזַקְתַּ֖נִי וַתּוּכָ֑ל הָיִ֤יתִי לִשְׂחוֹק֙ כָּל־ הַיּ֔וֹם כֻּלֹּ֖ה לֹעֵ֥ג לִֽי׃ (כח) כִּֽי־ מִדֵּ֤י אֲדַבֵּר֙ אֶזְעָ֔ק חָמָ֥ס וָשֹׁ֖ד אֶקְרָ֑א כִּֽי־ הָיָ֨ה דְבַר־ יְהוָ֥ה לִ֛י לְחֶרְפָּ֥ה וּלְקֶ֖לֶס כָּל־ הַיּֽוֹם׃ (כט) וְאָמַרְתִּ֣י לֹֽא־ אֶזְכְּרֶ֗נּוּ וְלֹֽא־ אֲדַבֵּ֥ר עוֹד֙ בִּשְׁמ֔וֹ וְהָיָ֤ה בְלִבִּי֙ כְּאֵ֣שׁ בֹּעֶ֔רֶת עָצֻ֖ר בְּעַצְמֹתָ֑י וְנִלְאֵ֥יתִי כַּֽלְכֵ֖ל וְלֹ֥א אוּכָֽל׃ (ל) כִּ֣י שָׁמַ֜עְתִּי דִּבַּ֣ת רַבִּים֮ מָג֣וֹר מִסָּבִיב֒ הַגִּ֙ידוּ֙ וְנַגִּידֶ֔נּוּ כֹּ֚ל אֱנ֣וֹשׁ שְׁלוֹמִ֔י שֹׁמְרֵ֖י צַלְעִ֑י אוּלַ֤י יְפֻתֶּה֙ וְנ֣וּכְלָה ל֔וֹ וְנִקְחָ֥ה נִקְמָתֵ֖נוּ מִמֶּֽנּוּ׃ (לא) וַֽיהוָ֤ה אוֹתִי֙ כְּגִבּ֣וֹר עָרִ֔יץ עַל־ כֵּ֛ן רֹדְפַ֥י יִכָּשְׁל֖וּ וְלֹ֣א יֻכָ֑לוּ בֹּ֤שׁוּ מְאֹד֙ כִּֽי־ לֹ֣א הִשְׂכִּ֔ילוּ כְּלִמַּ֥ת עוֹלָ֖ם לֹ֥א תִשָּׁכֵֽחַ׃ (לב) וַיהוָ֤ה צְבָאוֹת֙ בֹּחֵ֣ן צַדִּ֔יק רֹאֶ֥ה כְלָי֖וֹת וָלֵ֑ב אֶרְאֶ֤ה נִקְמָֽתְךָ֙ מֵהֶ֔ם כִּ֥י אֵלֶ֖יךָ גִּלִּ֥יתִי אֶת־ רִיבִֽי׃

וְסָפ֣וּן בָּאָ֔רֶז וּמָשׁ֖וֹחַ בַּשָּׁשַֽׁר׃ (לה) הֲתִֽמְלֹ֔ךְ כִּ֥י אַתָּ֖ה מְתַחֲרֶ֣ה בָאָ֑רֶז אָבִ֜יךָ הֲל֧וֹא אָכַ֣ל וְשָׁתָ֗ה וְעָשָׂ֤ה מִשְׁפָּט֙ וּצְדָקָ֔ה אָ֖ז ט֥וֹב לֽוֹ׃ (לו) דָּ֛ן דִּין־ עָנִ֥י וְאֶבְי֖וֹן אָ֣ז ט֑וֹב הֲלוֹא־ הִ֛יא הַדַּ֥עַת אֹתִ֖י נְאֻם־ יְהוָֽה׃ (לז) כִּ֣י אֵ֤ין עֵינֶ֙יךָ֙ וְלִבְּךָ֔ כִּ֖י אִם־ עַל־ בִּצְעֶ֑ךָ וְעַ֤ל דַּֽם־ הַנָּקִי֙ לִשְׁפּ֔וֹךְ וְעַל־ הָעֹ֥שֶׁק וְעַל־ הַמְּרוּצָ֖ה לַעֲשֽׂוֹת׃ (לח) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה אֶל־ יְהוֹיָקִ֤ים בֶּן־ יֹאשִׁיָּ֙הוּ֙ מֶ֣לֶךְ יְהוּדָ֔ה לֹא־ יִסְפְּד֣וּ ל֔וֹ ה֥וֹי אָחִ֖י וְה֣וֹי אָח֑וֹת לֹא־ יִסְפְּד֣וּ ל֔וֹ ה֥וֹי אָד֖וֹן וְה֥וֹי הֹדֹֽה׃ (לט) קְבוּרַ֥ת חֲמ֖וֹר יִקָּבֵ֑ר סָח֣וֹב וְהַשְׁלֵ֔ךְ מֵהָ֖לְאָה לְשַׁעֲרֵ֥י יְרוּשָׁלִָֽם׃ (מ) עֲלִ֤י הַלְּבָנוֹן֙ וּֽצְעָ֔קִי וּבַבָּשָׁ֖ן תְּנִ֣י קוֹלֵ֑ךְ וְצַֽעֲקִי֙ מֵֽעֲבָרִ֔ים כִּ֥י נִשְׁבְּר֖וּ כָּל־ מְאַהֲבָֽיִךְ׃ (מא) דִּבַּ֤רְתִּי אֵלַ֙יִךְ֙ בְּשַׁלְוֺתַ֔יִךְ אָמַ֖רְתְּ לֹ֣א אֶשְׁמָ֑ע זֶ֤ה דַרְכֵּךְ֙ מִנְּעוּרַ֔יִךְ כִּ֥י לֹֽא־ שָׁמַ֖עַתְּ בְּקוֹלִֽי׃ (מב) כָּל־ רֹעַ֙יִךְ֙ תִּרְעֶה־ ר֔וּחַ וּֽמְאַהֲבַ֖יִךְ בַּשְּׁבִ֣י יֵלֵ֑כוּ כִּ֣י אָ֤ז תֵּבֹ֙שִׁי֙ וְנִכְלַ֔מְתְּ מִכֹּ֖ל רָעָתֵֽךְ׃ (מג) (ישבתי) יֹשַׁבְתְּ֙ בַּלְּבָנ֔וֹן (מקננתי) מְקֻנַּ֖נְתְּ בָּֽאֲרָזִ֑ים מַה־ נֵּחַנְתְּ֙ בְּבֹא־ לָ֣ךְ חֲבָלִ֔ים חִ֖יל כַּיֹּלֵדָֽה׃ (מד) חַי־ אָנִי֮ נְאֻם־ יְהוָה֒ כִּ֣י אִם־ יִהְיֶ֞ה כָּנְיָ֤הוּ בֶן־ יְהֽוֹיָקִים֙ מֶ֣לֶךְ יְהוּדָ֔ה חוֹתָ֖ם עַל־ יַ֣ד יְמִינִ֑י כִּ֥י מִשָּׁ֖ם אֶתְּקֶֽנְךָּ׃ (מה) וּנְתַתִּ֗יךָ בְּיַד֙ מְבַקְשֵׁ֣י נַפְשֶׁ֔ךָ וּבְיַ֛ד אֲשֶׁר־ אַתָּ֥ה יָג֖וֹר מִפְּנֵיהֶ֑ם וּבְיַ֛ד נְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל וּבְיַ֥ד הַכַּשְׂדִּֽים׃ (מו) וְהֵֽטַלְתִּ֣י אֹתְךָ֗ וְאֶֽת־ אִמְּךָ֙ אֲשֶׁ֣ר יְלָדַ֔תְךָ עַ֚ל הָאָ֣רֶץ אַחֶ֔רֶת אֲשֶׁ֥ר לֹֽא־ יֻלַּדְתֶּ֖ם שָׁ֑ם וְשָׁ֖ם תָּמֽוּתוּ׃ (מז) וְעַל־ הָאָ֗רֶץ אֲשֶׁר־ הֵ֛ם מְנַשְּׂאִ֥ים אֶת־ נַפְשָׁ֖ם לָשׁ֣וּב שָׁ֑ם שָׁ֖מָּה לֹ֥א יָשֽׁוּבוּ׃ (מח) הַעֶ֨צֶב נִבְזֶ֜ה נָפ֗וּץ הָאִ֤ישׁ הַזֶּה֙ כָּנְיָ֔הוּ אִ֨ם־ כְּלִ֔י אֵ֥ין חֵ֖פֶץ בּ֑וֹ מַדּ֤וּעַ הֽוּטֲלוּ֙ ה֣וּא וְזַרְע֔וֹ וְהֻ֨שְׁלְכ֔וּ עַל־ הָאָ֖רֶץ אֲשֶׁ֥ר לֹא־ יָדָֽעוּ׃ (מט) אֶ֥רֶץ אֶ֖רֶץ אָ֑רֶץ שִׁמְעִ֖י דְּבַר־ יְהוָֽה׃ (נ) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה כִּתְב֞וּ אֶת־ הָאִ֤ישׁ הַזֶּה֙ עֲרִירִ֔י גֶּ֖בֶר לֹא־ יִצְלַ֣ח בְּיָמָ֑יו כִּי֩ לֹ֨א יִצְלַ֜ח מִזַּרְע֗וֹ אִ֚ישׁ יֹשֵׁב֙ עַל־ כִּסֵּ֣א דָוִ֔ד וּמֹשֵׁ֥ל ע֖וֹד בִּיהוּדָֽה׃ (נא) ה֣וֹי רֹעִ֗ים מְאַבְּדִ֧ים וּמְפִצִ֛ים אֶת־ צֹ֥אן מַרְעִיתִ֖י נְאֻם־ יְהוָֽה׃ (נב) לָ֠כֵן כֹּֽה־ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֣י יִשְׂרָאֵ֗ל עַֽל־ הָרֹעִים֮ הָרֹעִ֣ים אֶת־ עַמִּי֒ אַתֶּ֞ם הֲפִצֹתֶ֤ם אֶת־ צֹאנִי֙ וַתַּדִּח֔וּם וְלֹ֥א פְקַדְתֶּ֖ם אֹתָ֑ם הִנְנִ֨י פֹקֵ֧ד עֲלֵיכֶ֛ם אֶת־ רֹ֥עַ מַעַלְלֵיכֶ֖ם נְאֻם־ יְהוָֽה׃ (נג) וַאֲנִ֗י אֲקַבֵּץ֙ אֶת־ שְׁאֵרִ֣ית צֹאנִ֔י מִכֹּל֙ הָאֲרָצ֔וֹת אֲשֶׁר־ הִדַּ֥חְתִּי אֹתָ֖ם שָׁ֑ם וַהֲשִׁבֹתִ֥י אֶתְהֶ֛ן עַל־ נְוֵהֶ֖ן וּפָר֥וּ וְרָבֽוּ׃ (נד) וַהֲקִמֹתִ֧י עֲלֵיהֶ֛ם רֹעִ֖ים וְרָע֑וּם וְלֹא־ יִֽירְא֨וּ ע֧וֹד וְלֹא־ יֵחַ֛תּוּ וְלֹ֥א יִפָּקֵ֖דוּ נְאֻם־ יְהוָֽה׃ (נה) הִנֵּ֨ה יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וַהֲקִמֹתִ֥י לְדָוִ֖ד צֶ֣מַח צַדִּ֑יק וּמָ֤לַךְ מֶ֙לֶךְ֙ וְהִשְׂכִּ֔יל וְעָשָׂ֛ה מִשְׁפָּ֥ט וּצְדָקָ֖ה בָּאָֽרֶץ׃
