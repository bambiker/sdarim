ירמיהו פרק ז לפי החלוקה לסדרים

פרק ז בספר ירמיהו (34 פסוקים) נמצא בירמיהו סדר ד (פסוקים כט–נ) ובירמיהו סדר ה (פסוקים א–יב).

ירמיהו סדר ד

(א) אֵלֶ֛יהָ יָבֹ֥אוּ רֹעִ֖ים וְעֶדְרֵיהֶ֑ם תָּקְע֨וּ עָלֶ֤יהָ אֹהָלִים֙ סָבִ֔יב רָע֖וּ אִ֥ישׁ אֶת־ יָדֽוֹ׃ (ב) קַדְּשׁ֤וּ עָלֶ֙יהָ֙ מִלְחָמָ֔ה ק֖וּמוּ וְנַעֲלֶ֣ה בַֽצָּהֳרָ֑יִם א֥וֹי לָ֙נוּ֙ כִּי־ פָנָ֣ה הַיּ֔וֹם כִּ֥י יִנָּט֖וּ צִלְלֵי־ עָֽרֶב׃ (ג) ק֚וּמוּ וְנַעֲלֶ֣ה בַלָּ֔יְלָה וְנַשְׁחִ֖יתָה אַרְמְנוֹתֶֽיהָ׃ (ד) כִּ֣י כֹ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת כִּרְת֣וּ עֵצָ֔ה וְשִׁפְכ֥וּ עַל־ יְרוּשָׁלִַ֖ם סֹלְלָ֑ה הִ֚יא הָעִ֣יר הָפְקַ֔ד כֻּלָּ֖הּ עֹ֥שֶׁק בְּקִרְבָּֽהּ׃ (ה) כְּהָקִ֥יר (בור) בַּ֙יִר֙ מֵימֶ֔יהָ כֵּ֖ן הֵקֵ֣רָה רָעָתָ֑הּ חָמָ֣ס וָ֠שֹׁד יִשָּׁ֨מַע בָּ֧הּ עַל־ פָּנַ֛י תָּמִ֖יד חֳלִ֥י וּמַכָּֽה׃ (ו) הִוָּסְרִי֙ יְר֣וּשָׁלִַ֔ם פֶּן־ תֵּקַ֥ע נַפְשִׁ֖י מִמֵּ֑ךְ פֶּן־ אֲשִׂימֵ֣ךְ שְׁמָמָ֔ה אֶ֖רֶץ ל֥וֹא נוֹשָֽׁבָה׃ (ז) כֹּ֤ה אָמַר֙ יְהֹוָ֣ה צְבָא֔וֹת עוֹלֵ֛ל יְעוֹלְל֥וּ כַגֶּ֖פֶן שְׁאֵרִ֣ית יִשְׂרָאֵ֑ל הָשֵׁב֙ יָדְךָ֔ כְּבוֹצֵ֖ר עַל־ סַלְסִלּֽוֹת׃ (ח) עַל־ מִ֨י אֲדַבְּרָ֤ה וְאָעִ֙ידָה֙ וְיִשְׁמָ֔עוּ הִנֵּה֙ עֲרֵלָ֣ה אָזְנָ֔ם וְלֹ֥א יוּכְל֖וּ לְהַקְשִׁ֑יב הִנֵּ֣ה דְבַר־ יְהוָ֗ה הָיָ֥ה לָהֶ֛ם לְחֶרְפָּ֖ה לֹ֥א יַחְפְּצוּ־ בֽוֹ׃ (ט) וְאֵת֩ חֲמַ֨ת יְהוָ֤ה ׀ מָלֵ֙אתִי֙ נִלְאֵ֣יתִי הָכִ֔יל שְׁפֹ֤ךְ עַל־ עוֹלָל֙ בַּח֔וּץ וְעַ֛ל ס֥וֹד בַּחוּרִ֖ים יַחְדָּ֑ו כִּֽי־ גַם־ אִ֤ישׁ עִם־ אִשָּׁה֙ יִלָּכֵ֔דוּ זָקֵ֖ן עִם־ מְלֵ֥א יָמִֽים׃ (י) וְנָסַ֤בּוּ בָֽתֵּיהֶם֙ לַאֲחֵרִ֔ים שָׂד֥וֹת וְנָשִׁ֖ים יַחְדָּ֑ו כִּֽי־ אַטֶּ֧ה אֶת־ יָדִ֛י עַל־ יֹשְׁבֵ֥י הָאָ֖רֶץ נְאֻם־ יְהוָֽה׃ (יא) כִּ֤י מִקְּטַנָּם֙ וְעַד־ גְּדוֹלָ֔ם כֻּלּ֖וֹ בּוֹצֵ֣עַ בָּ֑צַע וּמִנָּבִיא֙ וְעַד־ כֹּהֵ֔ן כֻּלּ֖וֹ עֹ֥שֶׂה שָּֽׁקֶר׃ (יב) וַֽיְרַפְּא֞וּ אֶת־ שֶׁ֤בֶר עַמִּי֙ עַל־ נְקַלָּ֔ה לֵאמֹ֖ר שָׁל֣וֹם ׀ שָׁל֑וֹם וְאֵ֖ין שָׁלֽוֹם׃ (יג) הֹבִ֕ישׁוּ כִּ֥י תוֹעֵבָ֖ה עָשׂ֑וּ גַּם־ בּ֣וֹשׁ לֹֽא־ יֵב֗וֹשׁוּ גַּם־ הַכְלִים֙ לֹ֣א יָדָ֔עוּ לָכֵ֞ן יִפְּל֧וּ בַנֹּפְלִ֛ים בְּעֵת־ פְּקַדְתִּ֥ים יִכָּשְׁל֖וּ אָמַ֥ר יְהוָֽה׃ (יד) כֹּ֣ה אָמַ֣ר יְהוָ֡ה עִמְדוּ֩ עַל־ דְּרָכִ֨ים וּרְא֜וּ וְשַׁאֲל֣וּ ׀ לִנְתִב֣וֹת עוֹלָ֗ם אֵי־ זֶ֨ה דֶ֤רֶךְ הַטּוֹב֙ וּלְכוּ־ בָ֔הּ וּמִצְא֥וּ מַרְגּ֖וֹעַ לְנַפְשְׁכֶ֑ם וַיֹּאמְר֖וּ לֹ֥א נֵלֵֽךְ׃ (טו) וַהֲקִמֹתִ֤י עֲלֵיכֶם֙ צֹפִ֔ים הַקְשִׁ֖יבוּ לְק֣וֹל שׁוֹפָ֑ר וַיֹּאמְר֖וּ לֹ֥א נַקְשִֽׁיב׃ (טז) לָכֵ֖ן שִׁמְע֣וּ הַגּוֹיִ֑ם וּדְעִ֥י עֵדָ֖ה אֶת־ אֲשֶׁר־ בָּֽם׃ (יז) שִׁמְעִ֣י הָאָ֔רֶץ הִנֵּ֨ה אָנֹכִ֜י מֵבִ֥יא רָעָ֛ה אֶל־ הָעָ֥ם הַזֶּ֖ה פְּרִ֣י מַחְשְׁבוֹתָ֑ם כִּ֤י עַל־ דְּבָרַי֙ לֹ֣א הִקְשִׁ֔יבוּ וְתוֹרָתִ֖י וַיִּמְאֲסוּ־ בָֽהּ׃ (יח) לָמָּה־ זֶּ֨ה לִ֤י לְבוֹנָה֙ מִשְּׁבָ֣א תָב֔וֹא וְקָנֶ֥ה הַטּ֖וֹב מֵאֶ֣רֶץ מֶרְחָ֑ק עֹלֽוֹתֵיכֶם֙ לֹ֣א לְרָצ֔וֹן וְזִבְחֵיכֶ֖ם לֹא־ עָ֥רְבוּ לִֽי׃ (יט) לָכֵ֗ן כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֥י נֹתֵ֛ן אֶל־ הָעָ֥ם הַזֶּ֖ה מִכְשֹׁלִ֑ים וְכָ֣שְׁלוּ בָ֠ם אָב֨וֹת וּבָנִ֥ים יַחְדָּ֛ו שָׁכֵ֥ן וְרֵע֖וֹ (יאבדו) וְאָבָֽדוּ׃ (כ) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנֵּ֛ה עַ֥ם בָּ֖א מֵאֶ֣רֶץ צָפ֑וֹן וְג֣וֹי גָּד֔וֹל יֵע֖וֹר מִיַּרְכְּתֵי־ אָֽרֶץ׃ (כא) קֶ֣שֶׁת וְכִיד֞וֹן יַחֲזִ֗יקוּ אַכְזָרִ֥י הוּא֙ וְלֹ֣א יְרַחֵ֔מוּ קוֹלָם֙ כַּיָּ֣ם יֶהֱמֶ֔ה וְעַל־ סוּסִ֖ים יִרְכָּ֑בוּ עָר֗וּךְ כְּאִישׁ֙ לַמִּלְחָמָ֔ה עָלַ֖יִךְ בַּת־ צִיּֽוֹן׃ (כב) שָׁמַ֥עְנוּ אֶת־ שָׁמְע֖וֹ רָפ֣וּ יָדֵ֑ינוּ צָרָה֙ הֶחֱזִיקַ֔תְנוּ חִ֖יל כַּיּוֹלֵדָֽה׃ (כג) אַל־ (תצאי) תֵּֽצְאוּ֙ הַשָּׂדֶ֔ה וּבַדֶּ֖רֶךְ אַל־ (תלכי) תֵּלֵ֑כוּ כִּ֚י חֶ֣רֶב לְאֹיֵ֔ב מָג֖וֹר מִסָּבִֽיב׃ (כד) בַּת־ עַמִּ֤י חִגְרִי־ שָׂק֙ וְהִתְפַּלְּשִׁ֣י בָאֵ֔פֶר אֵ֤בֶל יָחִיד֙ עֲשִׂ֣י לָ֔ךְ מִסְפַּ֖ד תַּמְרוּרִ֑ים כִּ֣י פִתְאֹ֔ם יָבֹ֥א הַשֹּׁדֵ֖ד עָלֵֽינוּ׃ (כה) בָּח֛וֹן נְתַתִּ֥יךָ בְעַמִּ֖י מִבְצָ֑ר וְתֵדַ֕ע וּבָחַנְתָּ֖ אֶת־ דַּרְכָּֽם׃ (כו) כֻּלָּם֙ סָרֵ֣י סֽוֹרְרִ֔ים הֹלְכֵ֥י רָכִ֖יל נְחֹ֣שֶׁת וּבַרְזֶ֑ל כֻּלָּ֥ם מַשְׁחִיתִ֖ים הֵֽמָּה׃ (כז) נָחַ֣ר מַפֻּ֔חַ (מאשתם) מֵאֵ֖שׁ תַּ֣ם עֹפָ֑רֶת לַשָּׁוְא֙ צָרַ֣ף צָר֔וֹף וְרָעִ֖ים לֹ֥א נִתָּֽקוּ׃ (כח) כֶּ֣סֶף נִמְאָ֔ס קָרְא֖וּ לָהֶ֑ם כִּֽי־ מָאַ֥ס יְהוָ֖ה בָּהֶֽם׃ תחילת פרק ז (כט) הַדָּבָר֙ אֲשֶׁ֣ר הָיָ֣ה אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (ל) עֲמֹ֗ד בְּשַׁ֙עַר֙ בֵּ֣ית יְהוָ֔ה וְקָרָ֣אתָ שָּׁ֔ם אֶת־ הַדָּבָ֖ר הַזֶּ֑ה וְאָמַרְתָּ֞ שִׁמְע֣וּ דְבַר־ יְהוָ֗ה כָּל־ יְהוּדָה֙ הַבָּאִים֙ בַּשְּׁעָרִ֣ים הָאֵ֔לֶּה לְהִֽשְׁתַּחֲוֺ֖ת לַיהוָֽה׃ (לא) כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הֵיטִ֥יבוּ דַרְכֵיכֶ֖ם וּמַֽעַלְלֵיכֶ֑ם וַאֲשַׁכְּנָ֣ה אֶתְכֶ֔ם בַּמָּק֥וֹם הַזֶּֽה׃ (לב) אַל־ תִּבְטְח֣וּ לָכֶ֔ם אֶל־ דִּבְרֵ֥י הַשֶּׁ֖קֶר לֵאמֹ֑ר הֵיכַ֤ל יְהוָה֙ הֵיכַ֣ל יְהוָ֔ה הֵיכַ֥ל יְהוָ֖ה הֵֽמָּה׃ (לג) כִּ֤י אִם־ הֵיטֵיב֙ תֵּיטִ֔יבוּ אֶת־ דַּרְכֵיכֶ֖ם וְאֶת־ מַֽעַלְלֵיכֶ֑ם אִם־ עָשׂ֤וֹ תַֽעֲשׂוּ֙ מִשְׁפָּ֔ט בֵּ֥ין אִ֖ישׁ וּבֵ֥ין רֵעֵֽהוּ׃ (לד) גֵּ֣ר יָת֤וֹם וְאַלְמָנָה֙ לֹ֣א תַֽעֲשֹׁ֔קוּ וְדָ֣ם נָקִ֔י אַֽל־ תִּשְׁפְּכ֖וּ בַּמָּק֣וֹם הַזֶּ֑ה וְאַחֲרֵ֨י אֱלֹהִ֧ים אֲחֵרִ֛ים לֹ֥א תֵלְכ֖וּ לְרַ֥ע לָכֶֽם׃ (לה) וְשִׁכַּנְתִּ֤י אֶתְכֶם֙ בַּמָּק֣וֹם הַזֶּ֔ה בָּאָ֕רֶץ אֲשֶׁ֥ר נָתַ֖תִּי לַאֲבֽוֹתֵיכֶ֑ם לְמִן־ עוֹלָ֖ם וְעַד־ עוֹלָֽם׃ (לו) הִנֵּ֤ה אַתֶּם֙ בֹּטְחִ֣ים לָכֶ֔ם עַל־ דִּבְרֵ֖י הַשָּׁ֑קֶר לְבִלְתִּ֖י הוֹעִֽיל׃ (לז) הֲגָנֹ֤ב ׀ רָצֹ֙חַ֙ וְֽנָאֹ֔ף וְהִשָּׁבֵ֥עַ לַשֶּׁ֖קֶר וְקַטֵּ֣ר לַבָּ֑עַל וְהָלֹ֗ךְ אַחֲרֵ֛י אֱלֹהִ֥ים אֲחֵרִ֖ים אֲשֶׁ֥ר לֹֽא־ יְדַעְתֶּֽם׃ (לח) וּבָאתֶ֞ם וַעֲמַדְתֶּ֣ם לְפָנַ֗י בַּבַּ֤יִת הַזֶּה֙ אֲשֶׁ֣ר נִקְרָא־ שְׁמִ֣י עָלָ֔יו וַאֲמַרְתֶּ֖ם נִצַּ֑לְנוּ לְמַ֣עַן עֲשׂ֔וֹת אֵ֥ת כָּל־ הַתּוֹעֵב֖וֹת הָאֵֽלֶּה׃ (לט) הַמְעָרַ֣ת פָּרִצִ֗ים הָיָ֨ה הַבַּ֧יִת הַזֶּ֛ה אֲשֶׁר־ נִקְרָֽא־ שְׁמִ֥י עָלָ֖יו בְּעֵינֵיכֶ֑ם גַּ֧ם אָנֹכִ֛י הִנֵּ֥ה רָאִ֖יתִי נְאֻם־ יְהוָֽה׃ (מ) כִּ֣י לְכוּ־ נָ֗א אֶל־ מְקוֹמִי֙ אֲשֶׁ֣ר בְּשִׁיל֔וֹ אֲשֶׁ֨ר שִׁכַּ֧נְתִּֽי שְׁמִ֛י שָׁ֖ם בָּרִֽאשׁוֹנָ֑ה וּרְאוּ֙ אֵ֣ת אֲשֶׁר־ עָשִׂ֣יתִי ל֔וֹ מִפְּנֵ֕י רָעַ֖ת עַמִּ֥י יִשְׂרָאֵֽל׃ (מא) וְעַתָּ֗ה יַ֧עַן עֲשׂוֹתְכֶ֛ם אֶת־ כָּל־ הַמַּֽעֲשִׂ֥ים הָאֵ֖לֶּה נְאֻם־ יְהוָ֑ה וָאֲדַבֵּ֨ר אֲלֵיכֶ֜ם הַשְׁכֵּ֤ם וְדַבֵּר֙ וְלֹ֣א שְׁמַעְתֶּ֔ם וָאֶקְרָ֥א אֶתְכֶ֖ם וְלֹ֥א עֲנִיתֶֽם׃ (מב) וְעָשִׂ֜יתִי לַבַּ֣יִת ׀ אֲשֶׁ֧ר נִֽקְרָא־ שְׁמִ֣י עָלָ֗יו אֲשֶׁ֤ר אַתֶּם֙ בֹּטְחִ֣ים בּ֔וֹ וְלַ֨מָּק֔וֹם אֲשֶׁר־ נָתַ֥תִּי לָכֶ֖ם שָׁלֽוֹם׃ (כד) הֹבִ֕שׁוּ כִּ֥י תוֹעֵבָ֖ה עָשׂ֑וּ גַּם־ בּ֣וֹשׁ לֹֽא־ יֵבֹ֗שׁוּ וְהִכָּלֵם֙ לֹ֣א יָדָ֔עוּ לָכֵ֞ן יִפְּל֣וּ בַנֹּפְלִ֗ים בְּעֵ֧ת פְּקֻדָּתָ֛ם יִכָּשְׁל֖וּ אָמַ֥ר יְהוָֽה׃ (כה) אָסֹ֥ף אֲסִיפֵ֖ם נְאֻם־ יְהֹוָ֑ה אֵין֩ עֲנָבִ֨ים בַּגֶּ֜פֶן וְאֵ֧ין תְּאֵנִ֣ים בַּתְּאֵנָ֗ה וְהֶֽעָלֶה֙ נָבֵ֔ל וָאֶתֵּ֥ן לָהֶ֖ם יַעַבְרֽוּם׃ (כו) עַל־ מָה֙ אֲנַ֣חְנוּ יֹֽשְׁבִ֔ים הֵֽאָסְפ֗וּ וְנָב֛וֹא אֶל־ עָרֵ֥י הַמִּבְצָ֖ר וְנִדְּמָה־ שָּׁ֑ם כִּי֩ יְהוָ֨ה אֱלֹהֵ֤ינוּ הֲדִמָּ֙נוּ֙ וַיַּשְׁקֵ֣נוּ מֵי־ רֹ֔אשׁ כִּ֥י חָטָ֖אנוּ לַיהוָֽה׃ (כז) קַוֵּ֥ה לְשָׁל֖וֹם וְאֵ֣ין ט֑וֹב לְעֵ֥ת מַרְפֵּ֖ה וְהִנֵּ֥ה בְעָתָֽה׃ (כח) מִדָּ֤ן נִשְׁמַע֙ נַחְרַ֣ת סוּסָ֗יו מִקּוֹל֙ מִצְהֲל֣וֹת אַבִּירָ֔יו רָעֲשָׁ֖ה כָּל־ הָאָ֑רֶץ וַיָּב֗וֹאוּ וַיֹּֽאכְלוּ֙ אֶ֣רֶץ וּמְלוֹאָ֔הּ עִ֖יר וְיֹ֥שְׁבֵי בָֽהּ׃ (כט) כִּי֩ הִנְנִ֨י מְשַׁלֵּ֜חַ בָּכֶ֗ם נְחָשִׁים֙ צִפְעֹנִ֔ים אֲשֶׁ֥ר אֵין־ לָהֶ֖ם לָ֑חַשׁ וְנִשְּׁכ֥וּ אֶתְכֶ֖ם נְאֻם־ יְהוָֽה׃ (ל) מַבְלִ֥יגִיתִ֖י עֲלֵ֣י יָג֑וֹן עָלַ֖י לִבִּ֥י דַוָּֽי׃ (לא) הִנֵּה־ ק֞וֹל שַֽׁוְעַ֣ת בַּת־ עַמִּ֗י מֵאֶ֙רֶץ֙ מַרְחַקִּ֔ים הַֽיהוָה֙ אֵ֣ין בְּצִיּ֔וֹן אִם־ מַלְכָּ֖הּ אֵ֣ין בָּ֑הּ מַדּ֗וּעַ הִכְעִס֛וּנִי בִּפְסִלֵיהֶ֖ם בְּהַבְלֵ֥י נֵכָֽר׃ (לב) עָבַ֥ר קָצִ֖יר כָּ֣לָה קָ֑יִץ וַאֲנַ֖חְנוּ ל֥וֹא נוֹשָֽׁעְנוּ׃ (לג) עַל־ שֶׁ֥בֶר בַּת־ עַמִּ֖י הָשְׁבָּ֑רְתִּי קָדַ֕רְתִּי שַׁמָּ֖ה הֶחֱזִקָֽתְנִי׃ (לד) הַצֳרִי֙ אֵ֣ין בְּגִלְעָ֔ד אִם־ רֹפֵ֖א אֵ֣ין שָׁ֑ם כִּ֗י מַדּ֙וּעַ֙ לֹ֣א עָֽלְתָ֔ה אֲרֻכַ֖ת בַּת־ עַמִּֽי׃ (לה) מִֽי־ יִתֵּ֤ן רֹאשִׁי֙ מַ֔יִם וְעֵינִ֖י מְק֣וֹר דִּמְעָ֑ה וְאֶבְכֶּה֙ יוֹמָ֣ם וָלַ֔יְלָה אֵ֖ת חַֽלְלֵ֥י בַת־ עַמִּֽי׃ (לו) מִֽי־ יִתְּנֵ֣נִי בַמִּדְבָּ֗ר מְלוֹן֙ אֹֽרְחִ֔ים וְאֶֽעֶזְבָה֙ אֶת־ עַמִּ֔י וְאֵלְכָ֖ה מֵֽאִתָּ֑ם כִּ֤י כֻלָּם֙ מְנָ֣אֲפִ֔ים עֲצֶ֖רֶת בֹּגְדִֽים׃ (לז) וַֽיַּדְרְכ֤וּ אֶת־ לְשׁוֹנָם֙ קַשְׁתָּ֣ם שֶׁ֔קֶר וְלֹ֥א לֶאֱמוּנָ֖ה גָּבְר֣וּ בָאָ֑רֶץ כִּי֩ מֵרָעָ֨ה אֶל־ רָעָ֧ה ׀ יָצָ֛אוּ וְאֹתִ֥י לֹֽא־ יָדָ֖עוּ נְאֻם־ יְהוָֽה׃ (לח) אִ֤ישׁ מֵרֵעֵ֙הוּ֙ הִשָּׁמֵ֔רוּ וְעַל־ כָּל־ אָ֖ח אַל־ תִּבְטָ֑חוּ כִּ֤י כָל־ אָח֙ עָק֣וֹב יַעְקֹ֔ב וְכָל־ רֵ֖עַ רָכִ֥יל יַהֲלֹֽךְ׃ (לט) וְאִ֤ישׁ בְּרֵעֵ֙הוּ֙ יְהָתֵ֔לּוּ וֶאֱמֶ֖ת לֹ֣א יְדַבֵּ֑רוּ לִמְּד֧וּ לְשׁוֹנָ֛ם דַּבֶּר־ שֶׁ֖קֶר הַעֲוֵ֥ה נִלְאֽוּ׃ (מ) שִׁבְתְּךָ֖ בְּת֣וֹךְ מִרְמָ֑ה בְּמִרְמָ֛ה מֵאֲנ֥וּ דַֽעַת־ אוֹתִ֖י נְאֻם־ יְהוָֽה׃ (מא) לָכֵ֗ן כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת הִנְנִ֥י צוֹרְפָ֖ם וּבְחַנְתִּ֑ים כִּֽי־ אֵ֣יךְ אֶעֱשֶׂ֔ה מִפְּנֵ֖י בַּת־ עַמִּֽי׃ (מב) חֵ֥ץ (שוחט) שָׁח֛וּט לְשׁוֹנָ֖ם מִרְמָ֣ה דִבֵּ֑ר בְּפִ֗יו שָׁל֤וֹם אֶת־ רֵעֵ֙הוּ֙ יְדַבֵּ֔ר וּבְקִרְבּ֖וֹ יָשִׂ֥ים אָרְבּֽוֹ׃ (מג) הַעַל־ אֵ֥לֶּה לֹֽא־ אֶפְקָד־ בָּ֖ם נְאֻם־ יְהוָ֑ה אִ֚ם בְּג֣וֹי אֲשֶׁר־ כָּזֶ֔ה לֹ֥א תִתְנַקֵּ֖ם נַפְשִֽׁי׃ (מד) עַל־ הֶ֨הָרִ֜ים אֶשָּׂ֧א בְכִ֣י וָנֶ֗הִי וְעַל־ נְא֤וֹת מִדְבָּר֙ קִינָ֔ה כִּ֤י נִצְּתוּ֙ מִבְּלִי־ אִ֣ישׁ עֹבֵ֔ר וְלֹ֥א שָׁמְע֖וּ ק֣וֹל מִקְנֶ֑ה מֵע֤וֹף הַשָּׁמַ֙יִם֙ וְעַד־ בְּהֵמָ֔ה נָדְד֖וּ הָלָֽכוּ׃ (מה) וְנָתַתִּ֧י אֶת־ יְרוּשָׁלִַ֛ם לְגַלִּ֖ים מְע֣וֹן תַּנִּ֑ים וְאֶת־ עָרֵ֧י יְהוּדָ֛ה אֶתֵּ֥ן שְׁמָמָ֖ה מִבְּלִ֖י יוֹשֵֽׁב׃ (מו) מִֽי־ הָאִ֤ישׁ הֶֽחָכָם֙ וְיָבֵ֣ן אֶת־ זֹ֔את וַאֲשֶׁ֨ר דִּבֶּ֧ר פִּֽי־ יְהוָ֛ה אֵלָ֖יו וְיַגִּדָ֑הּ עַל־ מָה֙ אָבְדָ֣ה הָאָ֔רֶץ נִצְּתָ֥ה כַמִּדְבָּ֖ר מִבְּלִ֖י עֹבֵֽר׃ (מז) וַיֹּ֣אמֶר יְהוָ֔ה עַל־ עָזְבָם֙ אֶת־ תּ֣וֹרָתִ֔י אֲשֶׁ֥ר נָתַ֖תִּי לִפְנֵיהֶ֑ם וְלֹא־ שָׁמְע֥וּ בְקוֹלִ֖י וְלֹא־ הָ֥לְכוּ בָֽהּ׃ (מח) וַיֵּ֣לְכ֔וּ אַחֲרֵ֖י שְׁרִר֣וּת לִבָּ֑ם וְאַחֲרֵי֙ הַבְּעָלִ֔ים אֲשֶׁ֥ר לִמְּד֖וּם אֲבוֹתָֽם׃ (מט) לָכֵ֗ן כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֧י מַאֲכִילָ֛ם אֶת־ הָעָ֥ם הַזֶּ֖ה לַֽעֲנָ֑ה וְהִשְׁקִיתִ֖ים מֵי־ רֹֽאשׁ׃ (נ) וַהֲפִֽצוֹתִים֙ בַּגּוֹיִ֔ם אֲשֶׁר֙ לֹ֣א יָֽדְע֔וּ הֵ֖מָּה וַֽאֲבוֹתָ֑ם וְשִׁלַּחְתִּ֤י אַֽחֲרֵיהֶם֙ אֶת־ הַחֶ֔רֶב עַ֥ד כַּלּוֹתִ֖י אוֹתָֽם׃ (נא) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת הִתְבּֽוֹנְנ֛וּ וְקִרְא֥וּ לַמְקוֹנְנ֖וֹת וּתְבוֹאֶ֑ינָה וְאֶל־ הַחֲכָמ֥וֹת שִׁלְח֖וּ וְתָבֽוֹאנָה׃ (נב) וּתְמַהֵ֕רְנָה וְתִשֶּׂ֥נָה עָלֵ֖ינוּ נֶ֑הִי וְתֵרַ֤דְנָה עֵינֵ֙ינוּ֙ דִּמְעָ֔ה וְעַפְעַפֵּ֖ינוּ יִזְּלוּ־ מָֽיִם׃ (נג) כִּ֣י ק֥וֹל נְהִ֛י נִשְׁמַ֥ע מִצִּיּ֖וֹן אֵ֣יךְ שֻׁדָּ֑דְנוּ בֹּ֤שְׁנֽוּ מְאֹד֙ כִּֽי־ עָזַ֣בְנוּ אָ֔רֶץ כִּ֥י הִשְׁלִ֖יכוּ מִשְׁכְּנוֹתֵֽינוּ׃ (נד) כִּֽי־ שְׁמַ֤עְנָה נָשִׁים֙ דְּבַר־ יְהוָ֔ה וְתִקַּ֥ח אָזְנְכֶ֖ם דְּבַר־ פִּ֑יו וְלַמֵּ֤דְנָה בְנֽוֹתֵיכֶם֙ נֶ֔הִי וְאִשָּׁ֥ה רְעוּתָ֖הּ קִינָֽה׃ (נה) כִּֽי־ עָ֤לָה מָ֙וֶת֙ בְּחַלּוֹנֵ֔ינוּ בָּ֖א בְּאַרְמְנוֹתֵ֑ינוּ לְהַכְרִ֤ית עוֹלָל֙ מִח֔וּץ בַּחוּרִ֖ים מֵרְחֹבֽוֹת׃ (נו) דַּבֵּ֗ר כֹּ֚ה נְאֻם־ יְהוָ֔ה וְנָֽפְלָה֙ נִבְלַ֣ת הָֽאָדָ֔ם כְּדֹ֖מֶן עַל־ פְּנֵ֣י הַשָּׂדֶ֑ה וּכְעָמִ֛יר מֵאַחֲרֵ֥י הַקֹּצֵ֖ר וְאֵ֥ין מְאַסֵּֽף׃ (נז) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אַל־ יִתְהַלֵּ֤ל חָכָם֙ בְּחָכְמָת֔וֹ וְאַל־ יִתְהַלֵּ֥ל הַגִּבּ֖וֹר בִּגְבֽוּרָת֑וֹ אַל־ יִתְהַלֵּ֥ל עָשִׁ֖יר בְּעָשְׁרֽוֹ׃
