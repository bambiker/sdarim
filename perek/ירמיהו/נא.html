פרק נא בספר ירמיהו (64 פסוקים) נמצא בירמיהו סדר כט (פסוקים מג–נא) ובירמיהו סדר ל (פסוקים א–מט) ובירמיהו סדר לא (פסוקים א–ו).

ירמיהו סדר כט

(א) צִיּ֣וֹן יִשְׁאָ֔לוּ דֶּ֖רֶךְ הֵ֣נָּה פְנֵיהֶ֑ם בֹּ֚אוּ וְנִלְו֣וּ אֶל־ יְהוָ֔ה בְּרִ֥ית עוֹלָ֖ם לֹ֥א תִשָּׁכֵֽחַ׃ (ב) צֹ֤אן אֹֽבְדוֹת֙ (היה) הָי֣וּ עַמִּ֔י רֹעֵיהֶ֣ם הִתְע֔וּם הָרִ֖ים (שובבים) שֽׁוֹבְב֑וּם מֵהַ֤ר אֶל־ גִּבְעָה֙ הָלָ֔כוּ שָׁכְח֖וּ רִבְצָֽם׃ (ג) כָּל־ מוֹצְאֵיהֶ֣ם אֲכָל֔וּם וְצָרֵיהֶ֥ם אָמְר֖וּ לֹ֣א נֶאְשָׁ֑ם תַּ֗חַת אֲשֶׁ֨ר חָטְא֤וּ לַֽיהוָה֙ נְוֵה־ צֶ֔דֶק וּמִקְוֵ֥ה אֲבֽוֹתֵיהֶ֖ם יְהוָֽה׃ (ד) נֻ֚דוּ מִתּ֣וֹךְ בָּבֶ֔ל וּמֵאֶ֥רֶץ כַּשְׂדִּ֖ים (יצאו) צֵ֑אוּ וִהְי֕וּ כְּעַתּוּדִ֖ים לִפְנֵי־ צֹֽאן׃ (ה) כִּ֣י הִנֵּ֣ה אָנֹכִ֡י מֵעִיר֩ וּמַעֲלֶ֨ה עַל־ בָּבֶ֜ל קְהַל־ גּוֹיִ֤ם גְּדֹלִים֙ מֵאֶ֣רֶץ צָפ֔וֹן וְעָ֣רְכוּ לָ֔הּ מִשָּׁ֖ם תִּלָּכֵ֑ד חִצָּיו֙ כְּגִבּ֣וֹר מַשְׁכִּ֔יל לֹ֥א יָשׁ֖וּב רֵיקָֽם׃ (ו) וְהָיְתָ֥ה כַשְׂדִּ֖ים לְשָׁלָ֑ל כָּל־ שֹׁלְלֶ֥יהָ יִשְׂבָּ֖עוּ נְאֻם־ יְהוָֽה׃ (ז) כִּ֤י (תשמחי) תִשְׂמְחוּ֙ כִּ֣י (תעלזי) תַֽעֲלְז֔וּ שֹׁסֵ֖י נַחֲלָתִ֑י כִּ֤י (תפושי) תָפ֙וּשׁוּ֙ כְּעֶגְלָ֣ה דָשָׁ֔ה (ותצהלי) וְתִצְהֲל֖וּ כָּאֲבִּרִֽים׃ (ח) בּ֤וֹשָׁה אִמְּכֶם֙ מְאֹ֔ד חָפְרָ֖ה יֽוֹלַדְתְּכֶ֑ם הִנֵּה֙ אַחֲרִ֣ית גּוֹיִ֔ם מִדְבָּ֖ר צִיָּ֥ה וַעֲרָבָֽה׃ (ט) מִקֶּ֤צֶף יְהוָה֙ לֹ֣א תֵשֵׁ֔ב וְהָיְתָ֥ה שְׁמָמָ֖ה כֻּלָּ֑הּ כֹּ֚ל עֹבֵ֣ר עַל־ בָּבֶ֔ל יִשֹּׁ֥ם וְיִשְׁרֹ֖ק עַל־ כָּל־ מַכּוֹתֶֽיהָ׃ (י) עִרְכ֨וּ עַל־ בָּבֶ֤ל ׀ סָבִיב֙ כָּל־ דֹּ֣רְכֵי קֶ֔שֶׁת יְד֣וּ אֵלֶ֔יהָ אַֽל־ תַּחְמְל֖וּ אֶל־ חֵ֑ץ כִּ֥י לַֽיהוָ֖ה חָטָֽאָה׃ (יא) הָרִ֨יעוּ עָלֶ֤יהָ סָבִיב֙ נָתְנָ֣ה יָדָ֔הּ נָֽפְלוּ֙ (אשויתיה) אָשְׁיוֹתֶ֔יהָ נֶהֶרְס֖וּ חֽוֹמוֹתֶ֑יהָ כִּי֩ נִקְמַ֨ת יְהוָ֥ה הִיא֙ הִנָּ֣קְמוּ בָ֔הּ כַּאֲשֶׁ֥ר עָשְׂתָ֖ה עֲשׂוּ־ לָֽהּ׃ (יב) כִּרְת֤וּ זוֹרֵ֙עַ֙ מִבָּבֶ֔ל וְתֹפֵ֥שׂ מַגָּ֖ל בְּעֵ֣ת קָצִ֑יר מִפְּנֵי֙ חֶ֣רֶב הַיּוֹנָ֔ה אִ֤ישׁ אֶל־ עַמּוֹ֙ יִפְנ֔וּ וְאִ֥ישׁ לְאַרְצ֖וֹ יָנֻֽסוּ׃ (יג) שֶׂ֧ה פְזוּרָ֛ה יִשְׂרָאֵ֖ל אֲרָי֣וֹת הִדִּ֑יחוּ הָרִאשׁ֤וֹן אֲכָלוֹ֙ מֶ֣לֶךְ אַשּׁ֔וּר וְזֶ֤ה הָאַחֲרוֹן֙ עִצְּמ֔וֹ נְבוּכַדְרֶאצַּ֖ר מֶ֥לֶךְ בָּבֶֽל׃ (יד) לָכֵ֗ן כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֥י פֹקֵ֛ד אֶל־ מֶ֥לֶךְ בָּבֶ֖ל וְאֶל־ אַרְצ֑וֹ כַּאֲשֶׁ֥ר פָּקַ֖דְתִּי אֶל־ מֶ֥לֶךְ אַשּֽׁוּר׃ (טו) וְשֹׁבַבְתִּ֤י אֶת־ יִשְׂרָאֵל֙ אֶל־ נָוֵ֔הוּ וְרָעָ֥ה הַכַּרְמֶ֖ל וְהַבָּשָׁ֑ן וּבְהַ֥ר אֶפְרַ֛יִם וְהַגִּלְעָ֖ד תִּשְׂבַּ֥ע נַפְשֽׁוֹ׃ (טז) בַּיָּמִ֣ים הָהֵם֩ וּבָעֵ֨ת הַהִ֜יא נְאֻם־ יְהוָ֗ה יְבֻקַּ֞שׁ אֶת־ עֲוֺ֤ן יִשְׂרָאֵל֙ וְאֵינֶ֔נּוּ וְאֶת־ חַטֹּ֥את יְהוּדָ֖ה וְלֹ֣א תִמָּצֶ֑אינָה כִּ֥י אֶסְלַ֖ח לַאֲשֶׁ֥ר אַשְׁאִֽיר׃ (יז) עַל־ הָאָ֤רֶץ מְרָתַ֙יִם֙ עֲלֵ֣ה עָלֶ֔יהָ וְאֶל־ יוֹשְׁבֵ֖י פְּק֑וֹד חֲרֹ֨ב וְהַחֲרֵ֤ם אַֽחֲרֵיהֶם֙ נְאֻם־ יְהוָ֔ה וַעֲשֵׂ֕ה כְּכֹ֖ל אֲשֶׁ֥ר צִוִּיתִֽיךָ׃ (יח) ק֥וֹל מִלְחָמָ֖ה בָּאָ֑רֶץ וְשֶׁ֖בֶר גָּדֽוֹל׃ (יט) אֵ֤יךְ נִגְדַּע֙ וַיִּשָּׁבֵ֔ר פַּטִּ֖ישׁ כָּל־ הָאָ֑רֶץ אֵ֣יךְ הָיְתָ֧ה לְשַׁמָּ֛ה בָּבֶ֖ל בַּגּוֹיִֽם׃ (כ) יָקֹ֨שְׁתִּי לָ֤ךְ וְגַם־ נִלְכַּדְתְּ֙ בָּבֶ֔ל וְאַ֖תְּ לֹ֣א יָדָ֑עַתְּ נִמְצֵאת֙ וְגַם־ נִתְפַּ֔שְׂתְּ כִּ֥י בַֽיהוָ֖ה הִתְגָּרִֽית׃ (כא) פָּתַ֤ח יְהוָה֙ אֶת־ א֣וֹצָר֔וֹ וַיּוֹצֵ֖א אֶת־ כְּלֵ֣י זַעְמ֑וֹ כִּי־ מְלָאכָ֣ה הִ֗יא לַֽאדֹנָ֧י יְהוִ֛ה צְבָא֖וֹת בְּאֶ֥רֶץ כַּשְׂדִּֽים׃ (כב) בֹּֽאוּ־ לָ֤הּ מִקֵּץ֙ פִּתְח֣וּ מַאֲבֻסֶ֔יהָ סָלּ֥וּהָ כְמוֹ־ עֲרֵמִ֖ים וְהַחֲרִימ֑וּהָ אַל־ תְּהִי־ לָ֖הּ שְׁאֵרִֽית׃ (כג) חִרְבוּ֙ כָּל־ פָּרֶ֔יהָ יֵרְד֖וּ לַטָּ֑בַח ה֣וֹי עֲלֵיהֶ֔ם כִּֽי־ בָ֥א יוֹמָ֖ם עֵ֥ת פְּקֻדָּתָֽם׃ (כד) ק֥וֹל נָסִ֛ים וּפְלֵטִ֖ים מֵאֶ֣רֶץ בָּבֶ֑ל לְהַגִּ֣יד בְּצִיּ֗וֹן אֶת־ נִקְמַת֙ יְהוָ֣ה אֱלֹהֵ֔ינוּ נִקְמַ֖ת הֵיכָלֽוֹ׃ (כה) הַשְׁמִ֣יעוּ אֶל־ בָּבֶ֣ל ׀ רַ֠בִּים כָּל־ דֹּ֨רְכֵי קֶ֜שֶׁת חֲנ֧וּ עָלֶ֣יהָ סָבִ֗יב אַל־ יְהִי־ לָהּ֙ פְּלֵטָ֔ה שַׁלְּמוּ־ לָ֣הּ כְּפָעֳלָ֔הּ כְּכֹ֛ל אֲשֶׁ֥ר עָשְׂתָ֖ה עֲשׂוּ־ לָ֑הּ כִּ֧י אֶל־ יְהוָ֛ה זָ֖דָה אֶל־ קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (כו) לָכֵ֛ן יִפְּל֥וּ בַחוּרֶ֖יהָ בִּרְחֹבֹתֶ֑יהָ וְכָל־ אַנְשֵׁ֨י מִלְחַמְתָּ֥הּ יִדַּ֛מּוּ בַּיּ֥וֹם הַה֖וּא נְאֻם־ יְהוָֽה׃ (כז) הִנְנִ֤י אֵלֶ֙יךָ֙ זָד֔וֹן נְאֻם־ אֲדֹנָ֥י יְהוִ֖ה צְבָא֑וֹת כִּ֛י בָּ֥א יוֹמְךָ֖ עֵ֥ת פְּקַדְתִּֽיךָ׃ (כח) וְכָשַׁ֤ל זָדוֹן֙ וְנָפַ֔ל וְאֵ֥ין ל֖וֹ מֵקִ֑ים וְהִצַּ֤תִּי אֵשׁ֙ בְּעָרָ֔יו וְאָכְלָ֖ה כָּל־ סְבִיבֹתָֽיו׃ (כט) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת עֲשׁוּקִ֛ים בְּנֵי־ יִשְׂרָאֵ֥ל וּבְנֵי־ יְהוּדָ֖ה יַחְדָּ֑ו וְכָל־ שֹֽׁבֵיהֶם֙ הֶחֱזִ֣יקוּ בָ֔ם מֵאֲנ֖וּ שַׁלְּחָֽם׃ (ל) גֹּאֲלָ֣ם ׀ חָזָ֗ק יְהוָ֤ה צְבָאוֹת֙ שְׁמ֔וֹ רִ֥יב יָרִ֖יב אֶת־ רִיבָ֑ם לְמַ֙עַן֙ הִרְגִּ֣יעַ אֶת־ הָאָ֔רֶץ וְהִרְגִּ֖יז לְיֹשְׁבֵ֥י בָבֶֽל׃ (לא) חֶ֥רֶב עַל־ כַּשְׂדִּ֖ים נְאֻם־ יְהוָ֑ה וְאֶל־ יֹשְׁבֵ֣י בָבֶ֔ל וְאֶל־ שָׂרֶ֖יהָ וְאֶל־ חֲכָמֶֽיהָ׃ (לב) חֶ֥רֶב אֶל־ הַבַּדִּ֖ים וְנֹאָ֑לוּ חֶ֥רֶב אֶל־ גִּבּוֹרֶ֖יהָ וָחָֽתּוּ׃ (לג) חֶ֜רֶב אֶל־ סוּסָ֣יו וְאֶל־ רִכְבּ֗וֹ וְאֶל־ כָּל־ הָעֶ֛רֶב אֲשֶׁ֥ר בְּתוֹכָ֖הּ וְהָי֣וּ לְנָשִׁ֑ים חֶ֥רֶב אֶל־ אוֹצְרֹתֶ֖יהָ וּבֻזָּֽזוּ׃ (לד) חֹ֥רֶב אֶל־ מֵימֶ֖יהָ וְיָבֵ֑שׁוּ כִּ֣י אֶ֤רֶץ פְּסִלִים֙ הִ֔יא וּבָאֵימִ֖ים יִתְהֹלָֽלוּ׃ (לה) לָכֵ֗ן יֵשְׁב֤וּ צִיִּים֙ אֶת־ אִיִּ֔ים וְיָ֥שְׁבוּ בָ֖הּ בְּנ֣וֹת יַֽעֲנָ֑ה וְלֹֽא־ תֵשֵׁ֥ב עוֹד֙ לָנֶ֔צַח וְלֹ֥א תִשְׁכּ֖וֹן עַד־ דּ֥וֹר וָדֽוֹר׃ (לו) כְּמַהְפֵּכַ֨ת אֱלֹהִ֜ים אֶת־ סְדֹ֧ם וְאֶת־ עֲמֹרָ֛ה וְאֶת־ שְׁכֵנֶ֖יהָ נְאֻם־ יְהוָ֑ה לֹֽא־ יֵשֵׁ֥ב שָׁם֙ אִ֔ישׁ וְלֹֽא־ יָג֥וּר בָּ֖הּ בֶּן־ אָדָֽם׃ (לז) הִנֵּ֛ה עַ֥ם בָּ֖א מִצָּפ֑וֹן וְג֤וֹי גָּדוֹל֙ וּמְלָכִ֣ים רַבִּ֔ים יֵעֹ֖רוּ מִיַּרְכְּתֵי־ אָֽרֶץ׃ (לח) קֶ֣שֶׁת וְכִידֹ֞ן יַחֲזִ֗יקוּ אַכְזָרִ֥י הֵ֙מָּה֙ וְלֹ֣א יְרַחֵ֔מוּ קוֹלָם֙ כַּיָּ֣ם יֶהֱמֶ֔ה וְעַל־ סוּסִ֖ים יִרְכָּ֑בוּ עָר֗וּךְ כְּאִישׁ֙ לַמִּלְחָמָ֔ה עָלַ֖יִךְ בַּת־ בָּבֶֽל׃ (לט) שָׁמַ֧ע מֶֽלֶךְ־ בָּבֶ֛ל אֶת־ שִׁמְעָ֖ם וְרָפ֣וּ יָדָ֑יו צָרָה֙ הֶחֱזִיקַ֔תְהוּ וַיַּאַסְרֵ֣הוּ בַֽנְחֻשְׁתַּ֗יִם וַיְבִאֵ֤הוּ מֶֽלֶךְ־ בָּבֶל֙ בָּבֶ֔לָה וַיִּתְּנֵ֥הוּ (בבית)־ בֵֽית־ הַפְּקֻדֹּ֖ת עַד־ י֥וֹם מוֹתֽוֹ׃ (יח) וּבַחֹ֤דֶשׁ הַֽחֲמִישִׁי֙ בֶּעָשׂ֣וֹר לַחֹ֔דֶשׁ הִ֗יא שְׁנַת֙ תְּשַֽׁע־ עֶשְׂרֵ֣ה שָׁנָ֔ה לַמֶּ֖לֶךְ נְבוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֑ל בָּ֗א נְבֽוּזַרְאֲדָן֙ רַב־ טַבָּחִ֔ים עָמַ֛ד לִפְנֵ֥י מֶֽלֶךְ־ בָּבֶ֖ל בִּירוּשָׁלִָֽם׃ (יט) וַיִּשְׂרֹ֥ף אֶת־ בֵּית־ יְהוָ֖ה וְאֶת־ בֵּ֣ית הַמֶּ֑לֶךְ וְאֵ֨ת כָּל־ בָּתֵּ֧י יְרוּשָׁלִַ֛ם וְאֶת־ כָּל־ בֵּ֥ית הַגָּד֖וֹל שָׂרַ֥ף בָּאֵֽשׁ׃ (כ) וְאֶת־ כָּל־ חֹמ֥וֹת יְרוּשָׁלִַ֖ם סָבִ֑יב נָֽתְצוּ֙ כָּל־ חֵ֣יל כַּשְׂדִּ֔ים אֲשֶׁ֖ר אֶת־ רַב־ טַבָּחִֽים׃ (כא) וּמִדַּלּ֨וֹת הָעָ֜ם וְֽאֶת־ יֶ֥תֶר הָעָ֣ם ׀ הַנִּשְׁאָרִ֣ים בָּעִ֗יר וְאֶת־ הַנֹּֽפְלִים֙ אֲשֶׁ֤ר נָֽפְלוּ֙ אֶל־ מֶ֣לֶךְ בָּבֶ֔ל וְאֵ֖ת יֶ֣תֶר הָֽאָמ֑וֹן הֶגְלָ֕ה נְבוּזַרְאֲדָ֖ן רַב־ טַבָּחִֽים׃ (כב) וּמִדַּלּ֣וֹת הָאָ֔רֶץ הִשְׁאִ֕יר נְבוּזַרְאֲדָ֖ן רַב־ טַבָּחִ֑ים לְכֹרְמִ֖ים וּלְיֹגְבִֽים׃ (כג) וְאֶת־ עַמּוּדֵ֨י הַנְּחֹ֜שֶׁת אֲשֶׁ֣ר לְבֵית־ יְהוָ֗ה וְֽאֶת־ הַמְּכֹנ֞וֹת וְאֶת־ יָ֧ם הַנְּחֹ֛שֶׁת אֲשֶׁ֥ר בְּבֵית־ יְהוָ֖ה שִׁבְּר֣וּ כַשְׂדִּ֑ים וַיִּשְׂא֥וּ אֶת־ כָּל־ נְחֻשְׁתָּ֖ם בָּבֶֽלָה׃ (כד) וְאֶת־ הַ֠סִּרוֹת וְאֶת־ הַיָּעִ֨ים וְאֶת־ הַֽמְזַמְּר֜וֹת וְאֶת־ הַמִּזְרָקֹ֣ת וְאֶת־ הַכַּפּ֗וֹת וְאֵ֨ת כָּל־ כְּלֵ֧י הַנְּחֹ֛שֶׁת אֲשֶׁר־ יְשָׁרְת֥וּ בָהֶ֖ם לָקָֽחוּ׃ (כה) וְאֶת־ הַ֠סִּפִּים וְאֶת־ הַמַּחְתּ֨וֹת וְאֶת־ הַמִּזְרָק֜וֹת וְאֶת־ הַסִּיר֣וֹת וְאֶת־ הַמְּנֹר֗וֹת וְאֶת־ הַכַּפּוֹת֙ וְאֶת־ הַמְּנַקִי֔וֹת אֲשֶׁ֤ר זָהָב֙ זָהָ֔ב וַאֲשֶׁר־ כֶּ֖סֶף כָּ֑סֶף לָקַ֖ח רַב־ טַבָּחִֽים׃ (כו) הָעַמּוּדִ֣ים ׀ שְׁנַ֗יִם הַיָּ֤ם אֶחָד֙ וְהַבָּקָ֞ר שְׁנֵים־ עָשָׂ֤ר נְחֹ֙שֶׁת֙ אֲשֶׁר־ תַּ֣חַת הַמְּכֹנ֔וֹת אֲשֶׁ֥ר עָשָׂ֛ה הַמֶּ֥לֶךְ שְׁלֹמֹ֖ה לְבֵ֣ית יְהוָ֑ה לֹא־ הָיָ֣ה מִשְׁקָ֔ל לִנְחֻשְׁתָּ֖ם כָּל־ הַכֵּלִ֥ים הָאֵֽלֶּה׃ (כז) וְהָעַמּוּדִ֗ים שְׁמֹנֶ֨ה עֶשְׂרֵ֤ה אַמָּה֙ (קומה) קוֹמַת֙ הָעַמֻּ֣ד הָאֶחָ֔ד וְח֛וּט שְׁתֵּים־ עֶשְׂרֵ֥ה אַמָּ֖ה יְסֻבֶּ֑נּוּ וְעָבְי֛וֹ אַרְבַּ֥ע אַצְבָּע֖וֹת נָבֽוּב׃ (כח) וְכֹתֶ֨רֶת עָלָ֜יו נְחֹ֗שֶׁת וְקוֹמַ֨ת הַכֹּתֶ֥רֶת הָאַחַת֮ חָמֵ֣שׁ אַמּוֹת֒ וּשְׂבָכָ֨ה וְרִמּוֹנִ֧ים עַֽל־ הַכּוֹתֶ֛רֶת סָבִ֖יב הַכֹּ֣ל נְחֹ֑שֶׁת וְכָאֵ֛לֶּה לַֽעַמּ֥וּד הַשֵּׁנִ֖י וְרִמּוֹנִֽים׃ (כט) וַיִּֽהְיוּ֙ הָֽרִמֹּנִ֔ים תִּשְׁעִ֥ים וְשִׁשָּׁ֖ה ר֑וּחָה כָּל־ הָרִמּוֹנִ֥ים מֵאָ֛ה עַל־ הַשְּׂבָכָ֖ה סָבִֽיב׃ (ל) וַיִּקַּ֣ח רַב־ טַבָּחִ֗ים אֶת־ שְׂרָיָה֙ כֹּהֵ֣ן הָרֹ֔אשׁ וְאֶת־ צְפַנְיָ֖ה כֹּהֵ֣ן הַמִּשְׁנֶ֑ה וְאֶת־ שְׁלֹ֖שֶׁת שֹׁמְרֵ֥י הַסַּֽף׃ (לא) וּמִן־ הָעִ֡יר לָקַח֩ סָרִ֨יס אֶחָ֜ד אֲ‍ֽשֶׁר־ הָיָ֥ה פָקִ֣יד ׀ עַל־ אַנְשֵׁ֣י הַמִּלְחָמָ֗ה וְשִׁבְעָ֨ה אֲנָשִׁ֜ים מֵרֹאֵ֤י פְנֵי־ הַמֶּ֙לֶךְ֙ אֲשֶׁ֣ר נִמְצְא֣וּ בָעִ֔יר וְאֵ֗ת סֹפֵר֙ שַׂ֣ר הַצָּבָ֔א הַמַּצְבִּ֖א אֶת־ עַ֣ם הָאָ֑רֶץ וְשִׁשִּׁ֥ים אִישׁ֙ מֵעַ֣ם הָאָ֔רֶץ הַֽנִּמְצְאִ֖ים בְּת֥וֹךְ הָעִֽיר׃ (לב) וַיִּקַּ֣ח אוֹתָ֔ם נְבוּזַרְאֲדָ֖ן רַב־ טַבָּחִ֑ים וַיֹּ֧לֶךְ אוֹתָ֛ם אֶל־ מֶ֥לֶךְ בָּבֶ֖ל רִבְלָֽתָה׃ (לג) וַיַּכֶּ֣ה אוֹתָם֩ מֶ֨לֶךְ בָּבֶ֧ל וַיְמִתֵ֛ם בְּרִבְלָ֖ה בְּאֶ֣רֶץ חֲמָ֑ת וַיִּ֥גֶל יְהוּדָ֖ה מֵעַ֥ל אַדְמָתֽוֹ׃ (לד) זֶ֣ה הָעָ֔ם אֲשֶׁ֥ר הֶגְלָ֖ה נְבֽוּכַדְרֶאצַּ֑ר בִּשְׁנַת־ שֶׁ֕בַע יְהוּדִ֕ים שְׁלֹ֥שֶׁת אֲלָפִ֖ים וְעֶשְׂרִ֥ים וּשְׁלֹשָֽׁה׃ (לה) בִּשְׁנַ֛ת שְׁמוֹנֶ֥ה עֶשְׂרֵ֖ה לִנְבֽוּכַדְרֶאצַּ֑ר מִיר֣וּשָׁלִַ֔ם נֶ֕פֶשׁ שְׁמֹנֶ֥ה מֵא֖וֹת שְׁלֹשִׁ֥ים וּשְׁנָֽיִם׃ (לו) בִּשְׁנַ֨ת שָׁלֹ֣שׁ וְעֶשְׂרִים֮ לִנְבֽוּכַדְרֶאצַּר֒ הֶגְלָ֗ה נְבֽוּזַרְאֲדָן֙ רַב־ טַבָּחִ֔ים יְהוּדִ֕ים נֶ֕פֶשׁ שְׁבַ֥ע מֵא֖וֹת אַרְבָּעִ֣ים וַחֲמִשָּׁ֑ה כָּל־ נֶ֕פֶשׁ אַרְבַּ֥עַת אֲלָפִ֖ים וְשֵׁ֥שׁ מֵאֽוֹת׃ (לז) וַיְהִי֩ בִשְׁלֹשִׁ֨ים וָשֶׁ֜בַע שָׁנָ֗ה לְגָלוּת֙ יְהוֹיָכִ֣ן מֶֽלֶךְ־ יְהוּדָ֔ה בִּשְׁנֵ֤ים עָשָׂר֙ חֹ֔דֶשׁ בְּעֶשְׂרִ֥ים וַחֲמִשָּׁ֖ה לַחֹ֑דֶשׁ נָשָׂ֡א אֱוִ֣יל מְרֹדַךְ֩ מֶ֨לֶךְ בָּבֶ֜ל בִּשְׁנַ֣ת מַלְכֻת֗וֹ אֶת־ רֹאשׁ֙ יְהוֹיָכִ֣ין מֶֽלֶךְ־ יְהוּדָ֔ה וַיֹּצֵ֥א אוֹת֖וֹ מִבֵּ֥ית (הכליא) הַכְּלֽוּא׃ (לח) וַיְדַבֵּ֥ר אִתּ֖וֹ טֹב֑וֹת וַיִּתֵּן֙ אֶת־ כִּסְא֔וֹ מִמַּ֗עַל לְכִסֵּ֧א (מלכים) הַמְּלָכִ֛ים אֲשֶׁ֥ר אִתּ֖וֹ בְּבָבֶֽל׃ (לט) וְשִׁנָּ֕ה אֵ֖ת בִּגְדֵ֣י כִלְא֑וֹ וְאָכַ֨ל לֶ֧חֶם לְפָנָ֛יו תָּמִ֖יד כָּל־ יְמֵ֥י חַיָּֽו׃ (מ) וַאֲרֻחָת֗וֹ אֲרֻחַת֩ תָּמִ֨יד נִתְּנָה־ לּ֜וֹ מֵאֵ֧ת מֶֽלֶךְ־ בָּבֶ֛ל דְּבַר־ י֥וֹם בְּיוֹמ֖וֹ עַד־ י֣וֹם מוֹת֑וֹ כֹּ֖ל יְמֵ֥י חַיָּֽיו׃
