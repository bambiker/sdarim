ירמיהו פרק לב לפי החלוקה לסדרים

פרק לב בספר ירמיהו (44 פסוקים) נמצא בירמיהו סדר יח (פסוקים י–ל) ובירמיהו סדר יט (פסוקים א–כג).

ירמיהו סדר יח

(א) לֹ֣א כַבְּרִ֗ית אֲשֶׁ֤ר כָּרַ֙תִּי֙ אֶת־ אֲבוֹתָ֔ם בְּיוֹם֙ הֶחֱזִיקִ֣י בְיָדָ֔ם לְהוֹצִיאָ֖ם מֵאֶ֖רֶץ מִצְרָ֑יִם אֲשֶׁר־ הֵ֜מָּה הֵפֵ֣רוּ אֶת־ בְּרִיתִ֗י וְאָנֹכִ֛י בָּעַ֥לְתִּי בָ֖ם נְאֻם־ יְהוָֽה׃ (ב) כִּ֣י זֹ֣את הַבְּרִ֡ית אֲשֶׁ֣ר אֶכְרֹת֩ אֶת־ בֵּ֨ית יִשְׂרָאֵ֜ל אַחֲרֵ֨י הַיָּמִ֤ים הָהֵם֙ נְאֻם־ יְהוָ֔ה נָתַ֤תִּי אֶת־ תּֽוֹרָתִי֙ בְּקִרְבָּ֔ם וְעַל־ לִבָּ֖ם אֶכְתֲּבֶ֑נָּה וְהָיִ֤יתִי לָהֶם֙ לֵֽאלֹהִ֔ים וְהֵ֖מָּה יִֽהְיוּ־ לִ֥י לְעָֽם׃ (ג) וְלֹ֧א יְלַמְּד֣וּ ע֗וֹד אִ֣ישׁ אֶת־ רֵעֵ֜הוּ וְאִ֤ישׁ אֶת־ אָחִיו֙ לֵאמֹ֔ר דְּע֖וּ אֶת־ יְהוָ֑ה כִּֽי־ כוּלָּם֩ יֵדְע֨וּ אוֹתִ֜י לְמִקְטַנָּ֤ם וְעַד־ גְּדוֹלָם֙ נְאֻם־ יְהוָ֔ה כִּ֤י אֶסְלַח֙ לַֽעֲוֺנָ֔ם וּלְחַטָּאתָ֖ם לֹ֥א אֶזְכָּר־ עֽוֹד׃ (ד) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה נֹתֵ֥ן שֶׁ֙מֶשׁ֙ לְא֣וֹר יוֹמָ֔ם חֻקֹּ֛ת יָרֵ֥חַ וְכוֹכָבִ֖ים לְא֣וֹר לָ֑יְלָה רֹגַ֤ע הַיָּם֙ וַיֶּהֱמ֣וּ גַלָּ֔יו יְהוָ֥ה צְבָא֖וֹת שְׁמֽוֹ׃ (ה) אִם־ יָמֻ֜שׁוּ הַחֻקִּ֥ים הָאֵ֛לֶּה מִלְּפָנַ֖י נְאֻם־ יְהוָ֑ה גַּם֩ זֶ֨רַע יִשְׂרָאֵ֜ל יִשְׁבְּת֗וּ מִֽהְי֥וֹת גּ֛וֹי לְפָנַ֖י כָּל־ הַיָּמִֽים׃ (ו) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אִם־ יִמַּ֤דּוּ שָׁמַ֙יִם֙ מִלְמַ֔עְלָה וְיֵחָקְר֥וּ מֽוֹסְדֵי־ אֶ֖רֶץ לְמָ֑טָּה גַּם־ אֲנִ֞י אֶמְאַ֨ס בְּכָל־ זֶ֧רַע יִשְׂרָאֵ֛ל עַֽל־ כָּל־ אֲשֶׁ֥ר עָשׂ֖וּ נְאֻם־ יְהוָֽה׃ (ז) הִנֵּ֛ה יָמִ֥ים בָּאִ֖ים נְאֻם־ יְהוָ֑ה וְנִבְנְתָ֤ה הָעִיר֙ לַֽיהוָ֔ה מִמִּגְדַּ֥ל חֲנַנְאֵ֖ל שַׁ֥עַר הַפִּנָּֽה׃ (ח) וְיָצָ֨א ע֜וֹד (קוה) קָ֤ו הַמִּדָּה֙ נֶגְדּ֔וֹ עַ֖ל גִּבְעַ֣ת גָּרֵ֑ב וְנָסַ֖ב גֹּעָֽתָה׃ (ט) וְכָל־ הָעֵ֣מֶק הַפְּגָרִ֣ים ׀ וְהַדֶּ֡שֶׁן וְכָֽל־ (השרמות) הַשְּׁדֵמוֹת֩ עַד־ נַ֨חַל קִדְר֜וֹן עַד־ פִּנַּ֨ת שַׁ֤עַר הַסּוּסִים֙ מִזְרָ֔חָה קֹ֖דֶשׁ לַֽיהוָ֑ה לֹֽא־ יִנָּתֵ֧שׁ וְֽלֹא־ יֵהָרֵ֛ס ע֖וֹד לְעוֹלָֽם׃ תחילת פרק לב (י) הַדָּבָ֞ר אֲשֶׁר־ הָיָ֤ה אֶֽל־ יִרְמְיָ֙הוּ֙ מֵאֵ֣ת יְהוָ֔ה (בשנת) בַּשָּׁנָה֙ הָעֲשִׂרִ֔ית לְצִדְקִיָּ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה הִ֧יא הַשָּׁנָ֛ה שְׁמֹנֶֽה־ עֶשְׂרֵ֥ה שָׁנָ֖ה לִנְבֽוּכַדְרֶאצַּֽר׃ (יא) וְאָ֗ז חֵ֚יל מֶ֣לֶךְ בָּבֶ֔ל צָרִ֖ים עַל־ יְרוּשָׁלִָ֑ם וְיִרְמְיָ֣הוּ הַנָּבִ֗יא הָיָ֤ה כָלוּא֙ בַּחֲצַ֣ר הַמַּטָּרָ֔ה אֲשֶׁ֖ר בֵּֽית־ מֶ֥לֶךְ יְהוּדָֽה׃ (יב) אֲשֶׁ֣ר כְּלָא֔וֹ צִדְקִיָּ֥הוּ מֶֽלֶךְ־ יְהוּדָ֖ה לֵאמֹ֑ר מַדּוּעַ֩ אַתָּ֨ה נִבָּ֜א לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֨י נֹתֵ֜ן אֶת־ הָעִ֥יר הַזֹּ֛את בְּיַ֥ד מֶֽלֶךְ־ בָּבֶ֖ל וּלְכָדָֽהּ׃ (יג) וְצִדְקִיָּ֙הוּ֙ מֶ֣לֶךְ יְהוּדָ֔ה לֹ֥א יִמָּלֵ֖ט מִיַּ֣ד הַכַּשְׂדִּ֑ים כִּ֣י הִנָּתֹ֤ן יִנָּתֵן֙ בְּיַ֣ד מֶֽלֶךְ־ בָּבֶ֔ל וְדִבֶּר־ פִּ֣יו עִם־ פִּ֔יו וְעֵינָ֖יו אֶת־ (עינו) עֵינָ֥יו תִּרְאֶֽינָה׃ (יד) וּבָבֶ֞ל יוֹלִ֤ךְ אֶת־ צִדְקִיָּ֙הוּ֙ וְשָׁ֣ם יִֽהְיֶ֔ה עַד־ פָּקְדִ֥י אֹת֖וֹ נְאֻם־ יְהוָ֑ה כִּ֧י תִֽלָּחֲמ֛וּ אֶת־ הַכַּשְׂדִּ֖ים לֹ֥א תַצְלִֽיחוּ׃ (טו) וַיֹּ֖אמֶר יִרְמְיָ֑הוּ הָיָ֥ה דְּבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (טז) הִנֵּ֣ה חֲנַמְאֵ֗ל בֶּן־ שַׁלֻּם֙ דֹּֽדְךָ֔ בָּ֥א אֵלֶ֖יךָ לֵאמֹ֑ר קְנֵ֣ה לְךָ֗ אֶת־ שָׂדִי֙ אֲשֶׁ֣ר בַּעֲנָת֔וֹת כִּ֥י לְךָ֛ מִשְׁפַּ֥ט הַגְּאֻלָּ֖ה לִקְנֽוֹת׃ (יז) וַיָּבֹ֣א אֵ֠לַי חֲנַמְאֵ֨ל בֶּן־ דֹּדִ֜י כִּדְבַ֣ר יְהוָה֮ אֶל־ חֲצַ֣ר הַמַּטָּרָה֒ וַיֹּ֣אמֶר אֵלַ֡י קְנֵ֣ה נָ֠א אֶת־ שָׂדִ֨י אֲשֶׁר־ בַּעֲנָת֜וֹת אֲשֶׁ֣ר ׀ בְּאֶ֣רֶץ בִּנְיָמִ֗ין כִּֽי־ לְךָ֞ מִשְׁפַּ֧ט הַיְרֻשָּׁ֛ה וּלְךָ֥ הַגְּאֻלָּ֖ה קְנֵה־ לָ֑ךְ וָאֵדַ֕ע כִּ֥י דְבַר־ יְהוָ֖ה הֽוּא׃ (יח) וָֽאֶקְנֶה֙ אֶת־ הַשָּׂדֶ֔ה מֵאֵ֛ת חֲנַמְאֵ֥ל בֶּן־ דֹּדִ֖י אֲשֶׁ֣ר בַּעֲנָת֑וֹת וָֽאֶשְׁקֲלָה־ לּוֹ֙ אֶת־ הַכֶּ֔סֶף שִׁבְעָ֥ה שְׁקָלִ֖ים וַעֲשָׂרָ֥ה הַכָּֽסֶף׃ (יט) וָאֶכְתֹּ֤ב בַּסֵּ֙פֶר֙ וָֽאֶחְתֹּ֔ם וָאָעֵ֖ד עֵדִ֑ים וָאֶשְׁקֹ֥ל הַכֶּ֖סֶף בְּמֹאזְנָֽיִם׃ (כ) וָאֶקַּ֖ח אֶת־ סֵ֣פֶר הַמִּקְנָ֑ה אֶת־ הֶֽחָת֛וּם הַמִּצְוָ֥ה וְהַחֻקִּ֖ים וְאֶת־ הַגָּלֽוּי׃ (כא) וָאֶתֵּ֞ן אֶת־ הַסֵּ֣פֶר הַמִּקְנָ֗ה אֶל־ בָּר֣וּךְ בֶּן־ נֵרִיָּה֮ בֶּן־ מַחְסֵיָה֒ לְעֵינֵי֙ חֲנַמְאֵ֣ל דֹּדִ֔י וּלְעֵינֵי֙ הָֽעֵדִ֔ים הַכֹּתְבִ֖ים בְּסֵ֣פֶר הַמִּקְנָ֑ה לְעֵינֵי֙ כָּל־ הַיְּהוּדִ֔ים הַיֹּשְׁבִ֖ים בַּחֲצַ֥ר הַמַּטָּרָֽה׃ (כב) וָֽאֲצַוֶּה֙ אֶת בָּר֔וּךְ לְעֵינֵיהֶ֖ם לֵאמֹֽר׃ (כג) כֹּֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל לָק֣וֹחַ אֶת־ הַסְּפָרִ֣ים הָאֵ֡לֶּה אֵ֣ת סֵפֶר֩ הַמִּקְנָ֨ה הַזֶּ֜ה וְאֵ֣ת הֶחָת֗וּם וְאֵ֨ת סֵ֤פֶר הַגָּלוּי֙ הַזֶּ֔ה וּנְתַתָּ֖ם בִּכְלִי־ חָ֑רֶשׂ לְמַ֥עַן יַעַמְד֖וּ יָמִ֥ים רַבִּֽים׃ (כד) כִּ֣י כֹ֥ה אָמַ֛ר יְהוָ֥ה צְבָא֖וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֑ל ע֣וֹד יִקָּנ֥וּ בָתִּ֛ים וְשָׂד֥וֹת וּכְרָמִ֖ים בָּאָ֥רֶץ הַזֹּֽאת׃ (כה) וָאֶתְפַּלֵּ֖ל אֶל־ יְהוָ֑ה אַחֲרֵ֤י תִתִּי֙ אֶת־ סֵ֣פֶר הַמִּקְנָ֔ה אֶל־ בָּר֥וּךְ בֶּן־ נֵרִיָּ֖ה לֵאמֹֽר׃ (כו) אֲהָהּ֮ אֲדֹנָ֣י יְהוִה֒ הִנֵּ֣ה ׀ אַתָּ֣ה עָשִׂ֗יתָ אֶת־ הַשָּׁמַ֙יִם֙ וְאֶת־ הָאָ֔רֶץ בְּכֹֽחֲךָ֙ הַגָּד֔וֹל וּבִֽזְרֹעֲךָ֖ הַנְּטוּיָ֑ה לֹֽא־ יִפָּלֵ֥א מִמְּךָ֖ כָּל־ דָּבָֽר׃ (כז) עֹ֤שֶׂה חֶ֙סֶד֙ לַֽאֲלָפִ֔ים וּמְשַׁלֵּם֙ עֲוֺ֣ן אָב֔וֹת אֶל־ חֵ֥יק בְּנֵיהֶ֖ם אַחֲרֵיהֶ֑ם הָאֵ֤ל הַגָּדוֹל֙ הַגִּבּ֔וֹר יְהוָ֥ה צְבָא֖וֹת שְׁמֽוֹ׃ (כח) גְּדֹל֙ הָֽעֵצָ֔ה וְרַ֖ב הָעֲלִֽילִיָּ֑ה אֲשֶׁר־ עֵינֶ֣יךָ פְקֻח֗וֹת עַל־ כָּל־ דַּרְכֵי֙ בְּנֵ֣י אָדָ֔ם לָתֵ֤ת לְאִישׁ֙ כִּדְרָכָ֔יו וְכִפְרִ֖י מַעֲלָלָֽיו׃ (כט) אֲשֶׁר־ שַׂ֠מְתָּ אֹת֨וֹת וּמֹפְתִ֤ים בְּאֶֽרֶץ־ מִצְרַ֙יִם֙ עַד־ הַיּ֣וֹם הַזֶּ֔ה וּבְיִשְׂרָאֵ֖ל וּבָֽאָדָ֑ם וַתַּעֲשֶׂה־ לְּךָ֥ שֵׁ֖ם כַּיּ֥וֹם הַזֶּֽה׃ (ל) וַתֹּצֵ֛א אֶת־ עַמְּךָ֥ אֶת־ יִשְׂרָאֵ֖ל מֵאֶ֣רֶץ מִצְרָ֑יִם בְּאֹת֣וֹת וּבְמוֹפְתִ֗ים וּבְיָ֤ד חֲזָקָה֙ וּבְאֶזְר֣וֹעַ נְטוּיָ֔ה וּבְמוֹרָ֖א גָּדֽוֹל׃

ירמיהו סדר יט

(א) וַתִּתֵּ֤ן לָהֶם֙ אֶת־ הָאָ֣רֶץ הַזֹּ֔את אֲשֶׁר־ נִשְׁבַּ֥עְתָּ לַאֲבוֹתָ֖ם לָתֵ֣ת לָהֶ֑ם אֶ֛רֶץ זָבַ֥ת חָלָ֖ב וּדְבָֽשׁ׃ (ב) וַיָּבֹ֜אוּ וַיִּֽרְשׁ֣וּ אֹתָ֗הּ וְלֹֽא־ שָׁמְע֤וּ בְקוֹלֶ֙ךָ֙ (ובתרותך) וּבְתוֹרָתְךָ֣ לֹא־ הָלָ֔כוּ אֵת֩ כָּל־ אֲשֶׁ֨ר צִוִּ֧יתָה לָהֶ֛ם לַעֲשׂ֖וֹת לֹ֣א עָשׂ֑וּ וַתַּקְרֵ֣א אֹתָ֔ם אֵ֥ת כָּל־ הָרָעָ֖ה הַזֹּֽאת׃ (ג) הִנֵּ֣ה הַסֹּלְל֗וֹת בָּ֣אוּ הָעִיר֮ לְלָכְדָהּ֒ וְהָעִ֣יר נִתְּנָ֗ה בְּיַ֤ד הַכַּשְׂדִּים֙ הַנִּלְחָמִ֣ים עָלֶ֔יהָ מִפְּנֵ֛י הַחֶ֥רֶב וְהָרָעָ֖ב וְהַדָּ֑בֶר וַאֲשֶׁ֥ר דִּבַּ֛רְתָּ הָיָ֖ה וְהִנְּךָ֥ רֹאֶֽה׃ (ד) וְאַתָּ֞ה אָמַ֤רְתָּ יִֽהְיֶ֣ה ׀ בַּמָּק֣וֹם הַזֶּ֗ה הֶחָרֵ֛ב מֵֽאֵין־ אָדָ֥ם וְעַד־ בְּהֵמָ֖ה וּבְכָל־ עָרָ֑יו נְוֵ֣ה רֹעִ֔ים מַרְבִּצִ֖ים צֹֽאן׃ (לו) בְּעָרֵ֨י הָהָ֜ר בְּעָרֵ֤י הַשְּׁפֵלָה֙ וּבְעָרֵ֣י הַנֶּ֔גֶב וּבְאֶ֧רֶץ בִּנְיָמִ֛ן וּבִסְבִיבֵ֥י יְרוּשָׁלִַ֖ם וּבְעָרֵ֣י יְהוּדָ֑ה עֹ֣ד תַּעֲבֹ֧רְנָה הַצֹּ֛אן עַל־ יְדֵ֥י מוֹנֶ֖ה אָמַ֥ר יְהוָֽה׃ (לז) הִנֵּ֛ה יָמִ֥ים בָּאִ֖ים נְאֻם־ יְהוָ֑ה וַהֲקִֽמֹתִי֙ אֶת־ הַדָּבָ֣ר הַטּ֔וֹב אֲשֶׁ֥ר דִּבַּ֛רְתִּי אֶל־ בֵּ֥ית יִשְׂרָאֵ֖ל וְעַל־ בֵּ֥ית יְהוּדָֽה׃ (לח) בַּיָּמִ֤ים הָהֵם֙ וּבָעֵ֣ת הַהִ֔יא אַצְמִ֥יחַ לְדָוִ֖ד צֶ֣מַח צְדָקָ֑ה וְעָשָׂ֛ה מִשְׁפָּ֥ט וּצְדָקָ֖ה בָּאָֽרֶץ׃
