ירמיהו פרק יח לפי החלוקה לסדרים

פרק יח בספר ירמיהו (23 פסוקים) נמצא בירמיהו סדר ט (פסוקים כב–לט) ובירמיהו סדר י (פסוקים א–ה).

ירמיהו סדר ט

(א) בָּר֣וּךְ הַגֶּ֔בֶר אֲשֶׁ֥ר יִבְטַ֖ח בַּֽיהוָ֑ה וְהָיָ֥ה יְהוָ֖ה מִבְטַחֽוֹ׃ (ב) וְהָיָ֞ה כְּעֵ֣ץ ׀ שָׁת֣וּל עַל־ מַ֗יִם וְעַל־ יוּבַל֙ יְשַׁלַּ֣ח שָֽׁרָשָׁ֔יו וְלֹ֤א (ירא) יִרְאֶה֙ כִּֽי־ יָבֹ֣א חֹ֔ם וְהָיָ֥ה עָלֵ֖הוּ רַֽעֲנָ֑ן וּבִשְׁנַ֤ת בַּצֹּ֙רֶת֙ לֹ֣א יִדְאָ֔ג וְלֹ֥א יָמִ֖ישׁ מֵעֲשׂ֥וֹת פֶּֽרִי׃ (ג) עָקֹ֥ב הַלֵּ֛ב מִכֹּ֖ל וְאָנֻ֣שׁ ה֑וּא מִ֖י יֵדָעֶֽנּוּ׃ (ד) אֲנִ֧י יְהוָ֛ה חֹקֵ֥ר לֵ֖ב בֹּחֵ֣ן כְּלָי֑וֹת וְלָתֵ֤ת לְאִישׁ֙ (כדרכו) כִּדְרָכָ֔יו כִּפְרִ֖י מַעֲלָלָֽיו׃ (ה) קֹרֵ֤א דָגַר֙ וְלֹ֣א יָלָ֔ד עֹ֥שֶׂה עֹ֖שֶׁר וְלֹ֣א בְמִשְׁפָּ֑ט בַּחֲצִ֤י (ימו) יָמָיו֙ יַעַזְבֶ֔נּוּ וּבְאַחֲרִית֖וֹ יִהְיֶ֥ה נָבָֽל׃ (ו) כִּסֵּ֣א כָב֔וֹד מָר֖וֹם מֵֽרִאשׁ֑וֹן מְק֖וֹם מִקְדָּשֵֽׁנוּ׃ (ז) מִקְוֵ֤ה יִשְׂרָאֵל֙ יְהוָ֔ה כָּל־ עֹזְבֶ֖יךָ יֵבֹ֑שׁוּ (יסורי) וְסוּרַי֙ בָּאָ֣רֶץ יִכָּתֵ֔בוּ כִּ֥י עָזְב֛וּ מְק֥וֹר מַֽיִם־ חַיִּ֖ים אֶת־ יְהוָֽה׃ (ח) רְפָאֵ֤נִי יְהוָה֙ וְאֵ֣רָפֵ֔א הוֹשִׁיעֵ֖נִי וְאִוָּשֵׁ֑עָה כִּ֥י תְהִלָּתִ֖י אָֽתָּה׃ (ט) הִנֵּה־ הֵ֕מָּה אֹמְרִ֖ים אֵלָ֑י אַיֵּ֥ה דְבַר־ יְהוָ֖ה יָ֥בוֹא נָֽא׃ (י) וַאֲנִ֞י לֹא־ אַ֣צְתִּי ׀ מֵרֹעֶ֣ה אַחֲרֶ֗יךָ וְי֥וֹם אָנ֛וּשׁ לֹ֥א הִתְאַוֵּ֖יתִי אַתָּ֣ה יָדָ֑עְתָּ מוֹצָ֣א שְׂפָתַ֔י נֹ֥כַח פָּנֶ֖יךָ הָיָֽה׃ (יא) אַל־ תִּֽהְיֵה־ לִ֖י לִמְחִתָּ֑ה מַֽחֲסִי־ אַ֖תָּה בְּי֥וֹם רָעָֽה׃ (יב) יֵבֹ֤שׁוּ רֹדְפַי֙ וְאַל־ אֵבֹ֣שָׁה אָ֔נִי יֵחַ֣תּוּ הֵ֔מָּה וְאַל־ אֵחַ֖תָּה אָ֑נִי הָבִ֤יא עֲלֵיהֶם֙ י֣וֹם רָעָ֔ה וּמִשְׁנֶ֥ה שִׁבָּר֖וֹן שָׁבְרֵֽם׃ (יג) כֹּה־ אָמַ֨ר יְהוָ֜ה אֵלַ֗י הָלֹ֤ךְ וְעָֽמַדְתָּ֙ בְּשַׁ֣עַר בְּנֵֽי־ (עם) הָעָ֔ם אֲשֶׁ֨ר יָבֹ֤אוּ בוֹ֙ מַלְכֵ֣י יְהוּדָ֔ה וַאֲשֶׁ֖ר יֵ֣צְאוּ ב֑וֹ וּבְכֹ֖ל שַׁעֲרֵ֥י יְרוּשָׁלִָֽם׃ (יד) וְאָמַרְתָּ֣ אֲ֠לֵיהֶם שִׁמְע֨וּ דְבַר־ יְהוָ֜ה מַלְכֵ֤י יְהוּדָה֙ וְכָל־ יְהוּדָ֔ה וְכֹ֖ל יֹשְׁבֵ֣י יְרוּשָׁלִָ֑ם הַבָּאִ֖ים בַּשְּׁעָרִ֥ים הָאֵֽלֶּה׃ (טו) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִשָּׁמְר֖וּ בְּנַפְשֽׁוֹתֵיכֶ֑ם וְאַל־ תִּשְׂא֤וּ מַשָּׂא֙ בְּי֣וֹם הַשַּׁבָּ֔ת וַהֲבֵאתֶ֖ם בְּשַׁעֲרֵ֥י יְרוּשָׁלִָֽם׃ (טז) וְלֹא־ תוֹצִ֨יאוּ מַשָּׂ֤א מִבָּֽתֵּיכֶם֙ בְּי֣וֹם הַשַּׁבָּ֔ת וְכָל־ מְלָאכָ֖ה לֹ֣א תַֽעֲשׂ֑וּ וְקִדַּשְׁתֶּם֙ אֶת־ י֣וֹם הַשַּׁבָּ֔ת כַּאֲשֶׁ֥ר צִוִּ֖יתִי אֶת־ אֲבוֹתֵיכֶֽם׃ (יז) וְלֹ֣א שָֽׁמְע֔וּ וְלֹ֥א הִטּ֖וּ אֶת־ אָזְנָ֑ם וַיַּקְשׁוּ֙ אֶת־ עָרְפָּ֔ם לְבִלְתִּ֣י (שומע) שְׁמ֔וֹעַ וּלְבִלְתִּ֖י קַ֥חַת מוּסָֽר׃ (יח) וְ֠הָיָה אִם־ שָׁמֹ֨עַ תִּשְׁמְע֤וּן אֵלַי֙ נְאֻם־ יְהוָ֔ה לְבִלְתִּ֣י ׀ הָבִ֣יא מַשָּׂ֗א בְּשַׁעֲרֵ֛י הָעִ֥יר הַזֹּ֖את בְּי֣וֹם הַשַּׁבָּ֑ת וּלְקַדֵּשׁ֙ אֶת־ י֣וֹם הַשַּׁבָּ֔ת לְבִלְתִּ֥י עֲשֽׂוֹת־ (בה) בּ֖וֹ כָּל־ מְלָאכָֽה׃ (יט) וּבָ֣אוּ בְשַׁעֲרֵ֣י הָעִ֣יר הַזֹּ֡את מְלָכִ֣ים ׀ וְשָׂרִ֡ים יֹשְׁבִים֩ עַל־ כִּסֵּ֨א דָוִ֜ד רֹכְבִ֣ים ׀ בָּרֶ֣כֶב וּבַסּוּסִ֗ים הֵ֚מָּה וְשָׂ֣רֵיהֶ֔ם אִ֥ישׁ יְהוּדָ֖ה וְיֹשְׁבֵ֣י יְרוּשָׁלִָ֑ם וְיָשְׁבָ֥ה הָֽעִיר־ הַזֹּ֖את לְעוֹלָֽם׃ (כ) וּבָ֣אוּ מֵעָרֵֽי־ יְ֠הוּדָה וּמִסְּבִיב֨וֹת יְרוּשָׁלִַ֜ם וּמֵאֶ֣רֶץ בִּנְיָמִ֗ן וּמִן־ הַשְּׁפֵלָ֤ה וּמִן־ הָהָר֙ וּמִן־ הַנֶּ֔גֶב מְבִאִ֛ים עוֹלָ֥ה וְזֶ֖בַח וּמִנְחָ֣ה וּלְבוֹנָ֑ה וּמְבִאֵ֥י תוֹדָ֖ה בֵּ֥ית יְהוָֽה׃ (כא) וְאִם־ לֹ֨א תִשְׁמְע֜וּ אֵלַ֗י לְקַדֵּשׁ֙ אֶת־ י֣וֹם הַשַּׁבָּ֔ת וּלְבִלְתִּ֣י ׀ שְׂאֵ֣ת מַשָּׂ֗א וּבֹ֛א בְּשַׁעֲרֵ֥י יְרוּשָׁלִַ֖ם בְּי֣וֹם הַשַּׁבָּ֑ת וְהִצַּ֧תִּי אֵ֣שׁ בִּשְׁעָרֶ֗יהָ וְאָֽכְלָ֛ה אַרְמְנ֥וֹת יְרוּשָׁלִַ֖ם וְלֹ֥א תִכְבֶּֽה׃ תחילת פרק יח (כב) הַדָּבָר֙ אֲשֶׁ֣ר הָיָ֣ה אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (כג) ק֥וּם וְיָרַדְתָּ֖ בֵּ֣ית הַיּוֹצֵ֑ר וְשָׁ֖מָּה אַשְׁמִֽיעֲךָ֥ אֶת־ דְּבָרָֽי׃ (כד) וָאֵרֵ֖ד בֵּ֣ית הַיּוֹצֵ֑ר (והנהו) וְהִנֵּה־ ה֛וּא עֹשֶׂ֥ה מְלָאכָ֖ה עַל־ הָאָבְנָֽיִם׃ (כה) וְנִשְׁחַ֣ת הַכְּלִ֗י אֲשֶׁ֨ר ה֥וּא עֹשֶׂ֛ה בַּחֹ֖מֶר בְּיַ֣ד הַיּוֹצֵ֑ר וְשָׁ֗ב וַֽיַּעֲשֵׂ֙הוּ֙ כְּלִ֣י אַחֵ֔ר כַּאֲשֶׁ֥ר יָשַׁ֛ר בְּעֵינֵ֥י הַיּוֹצֵ֖ר לַעֲשֽׂוֹת׃ (כו) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֽוֹר׃ (כז) הֲכַיּוֹצֵ֨ר הַזֶּ֜ה לֹא־ אוּכַ֨ל לַעֲשׂ֥וֹת לָכֶ֛ם בֵּ֥ית יִשְׂרָאֵ֖ל נְאֻם־ יְהוָ֑ה הִנֵּ֤ה כַחֹ֙מֶר֙ בְּיַ֣ד הַיּוֹצֵ֔ר כֵּן־ אַתֶּ֥ם בְּיָדִ֖י בֵּ֥ית יִשְׂרָאֵֽל׃ (כח) רֶ֣גַע אֲדַבֵּ֔ר עַל־ גּ֖וֹי וְעַל־ מַמְלָכָ֑ה לִנְת֥וֹשׁ וְלִנְת֖וֹץ וּֽלְהַאֲבִֽיד׃ (כט) וְשָׁב֙ הַגּ֣וֹי הַה֔וּא מֵרָ֣עָת֔וֹ אֲשֶׁ֥ר דִּבַּ֖רְתִּי עָלָ֑יו וְנִֽחַמְתִּי֙ עַל־ הָ֣רָעָ֔ה אֲשֶׁ֥ר חָשַׁ֖בְתִּי לַעֲשׂ֥וֹת לֽוֹ׃ (ל) וְרֶ֣גַע אֲדַבֵּ֔ר עַל־ גּ֖וֹי וְעַל־ מַמְלָכָ֑ה לִבְנֹ֖ת וְלִנְטֹֽעַ׃ (לא) וְעָשָׂ֤ה (הרעה) הָרַע֙ בְּעֵינַ֔י לְבִלְתִּ֖י שְׁמֹ֣עַ בְּקוֹלִ֑י וְנִֽחַמְתִּי֙ עַל־ הַטּוֹבָ֔ה אֲשֶׁ֥ר אָמַ֖רְתִּי לְהֵיטִ֥יב אוֹתֽוֹ׃ (לב) וְעַתָּ֡ה אֱמָר־ נָ֣א אֶל־ אִישׁ־ יְהוּדָה֩ וְעַל־ יוֹשְׁבֵ֨י יְרוּשָׁלִַ֜ם לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנֵּ֨ה אָנֹכִ֜י יוֹצֵ֤ר עֲלֵיכֶם֙ רָעָ֔ה וְחֹשֵׁ֥ב עֲלֵיכֶ֖ם מַֽחֲשָׁבָ֑ה שׁ֣וּבוּ נָ֗א אִ֚ישׁ מִדַּרְכּ֣וֹ הָֽרָעָ֔ה וְהֵיטִ֥יבוּ דַרְכֵיכֶ֖ם וּמַעַלְלֵיכֶֽם׃ (לג) וְאָמְר֖וּ נוֹאָ֑שׁ כִּֽי־ אַחֲרֵ֤י מַחְשְׁבוֹתֵ֙ינוּ֙ נֵלֵ֔ךְ וְאִ֛ישׁ שְׁרִר֥וּת לִבּֽוֹ־ הָרָ֖ע נַעֲשֶֽׂה׃ (לד) לָכֵ֗ן כֹּ֚ה אָמַ֣ר יְהוָ֔ה שַֽׁאֲלוּ־ נָא֙ בַּגּוֹיִ֔ם מִ֥י שָׁמַ֖ע כָּאֵ֑לֶּה שַֽׁעֲרֻרִת֙ עָשְׂתָ֣ה מְאֹ֔ד בְּתוּלַ֖ת יִשְׂרָאֵֽל׃ (לה) הֲיַעֲזֹ֥ב מִצּ֛וּר שָׂדַ֖י שֶׁ֣לֶג לְבָנ֑וֹן אִם־ יִנָּתְשׁ֗וּ מַ֛יִם זָרִ֥ים קָרִ֖ים נוֹזְלִֽים׃ (לו) כִּֽי־ שְׁכֵחֻ֥נִי עַמִּ֖י לַשָּׁ֣וְא יְקַטֵּ֑רוּ וַיַּכְשִׁל֤וּם בְּדַרְכֵיהֶם֙ שְׁבִילֵ֣י עוֹלָ֔ם לָלֶ֣כֶת נְתִיב֔וֹת דֶּ֖רֶךְ לֹ֥א סְלוּלָֽה׃ (לז) לָשׂ֥וּם אַרְצָ֛ם לְשַׁמָּ֖ה (שרוקת) שְׁרִיק֣וֹת עוֹלָ֑ם כֹּ֚ל עוֹבֵ֣ר עָלֶ֔יהָ יִשֹּׁ֖ם וְיָנִ֥יד בְּרֹאשֽׁוֹ׃ (לח) כְּרֽוּחַ־ קָדִ֥ים אֲפִיצֵ֖ם לִפְנֵ֣י אוֹיֵ֑ב עֹ֧רֶף וְלֹֽא־ פָנִ֛ים אֶרְאֵ֖ם בְּי֥וֹם אֵידָֽם׃ (לט) וַיֹּאמְר֗וּ לְכ֨וּ וְנַחְשְׁבָ֣ה עַֽל־ יִרְמְיָהוּ֮ מַחֲשָׁבוֹת֒ כִּי֩ לֹא־ תֹאבַ֨ד תּוֹרָ֜ה מִכֹּהֵ֗ן וְעֵצָה֙ מֵֽחָכָ֔ם וְדָבָ֖ר מִנָּבִ֑יא לְכוּ֙ וְנַכֵּ֣הוּ בַלָּשׁ֔וֹן וְאַל־ נַקְשִׁ֖יבָה אֶל־ כָּל־ דְּבָרָֽיו׃

ירמיהו סדר י

(א) הַקְשִׁ֥יבָה יְהוָ֖ה אֵלָ֑י וּשְׁמַ֖ע לְק֥וֹל יְרִיבָֽי׃ (ב) הַיְשֻׁלַּ֤ם תַּֽחַת־ טוֹבָה֙ רָעָ֔ה כִּֽי־ כָר֥וּ שׁוּחָ֖ה לְנַפְשִׁ֑י זְכֹ֣ר ׀ עָמְדִ֣י לְפָנֶ֗יךָ לְדַבֵּ֤ר עֲלֵיהֶם֙ טוֹבָ֔ה לְהָשִׁ֥יב אֶת־ חֲמָתְךָ֖ מֵהֶֽם׃ (ג) לָכֵן֩ תֵּ֨ן אֶת־ בְּנֵיהֶ֜ם לָרָעָ֗ב וְהַגִּרֵם֮ עַל־ יְדֵי־ חֶרֶב֒ וְתִֽהְיֶ֨נָה נְשֵׁיהֶ֤ם שַׁכֻּלוֹת֙ וְאַלְמָנ֔וֹת וְאַ֨נְשֵׁיהֶ֔ם יִֽהְי֖וּ הֲרֻ֣גֵי מָ֑וֶת בַּח֣וּרֵיהֶ֔ם מֻכֵּי־ חֶ֖רֶב בַּמִּלְחָמָֽה׃ (ד) תִּשָּׁמַ֤ע זְעָקָה֙ מִבָּ֣תֵּיהֶ֔ם כִּֽי־ תָבִ֧יא עֲלֵיהֶ֛ם גְּד֖וּד פִּתְאֹ֑ם כִּֽי־ כָר֤וּ (שיחה) שׁוּחָה֙ לְלָכְדֵ֔נִי וּפַחִ֖ים טָמְנ֥וּ לְרַגְלָֽי׃ (ה) וְאַתָּ֣ה יְ֠הוָה יָדַ֜עְתָּ אֶֽת־ כָּל־ עֲצָתָ֤ם עָלַי֙ לַמָּ֔וֶת אַל־ תְּכַפֵּר֙ עַל־ עֲוֺנָ֔ם וְחַטָּאתָ֖ם מִלְּפָנֶ֣יךָ אַל־ תֶּ֑מְחִי (והיו) וְיִהְי֤וּ מֻכְשָׁלִים֙ לְפָנֶ֔יךָ בְּעֵ֥ת אַפְּךָ֖ עֲשֵׂ֥ה בָהֶֽם׃ סוף פרק יח (ו) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הָל֛וֹךְ וְקָנִ֥יתָ בַקְבֻּ֖ק יוֹצֵ֣ר חָ֑רֶשׂ וּמִזִּקְנֵ֣י הָעָ֔ם וּמִזִּקְנֵ֖י הַכֹּהֲנִֽים׃ (ז) וְיָצָ֙אתָ֙ אֶל־ גֵּ֣יא בֶן־ הִנֹּ֔ם אֲשֶׁ֕ר פֶּ֖תַח שַׁ֣עַר (החרסות) הַֽחַרְסִ֑ית וְקָרָ֣אתָ שָּׁ֔ם אֶת־ הַדְּבָרִ֖ים אֲשֶׁר־ אֲדַבֵּ֥ר אֵלֶֽיךָ׃ (ח) וְאָֽמַרְתָּ֙ שִׁמְע֣וּ דְבַר־ יְהוָ֔ה מַלְכֵ֣י יְהוּדָ֔ה וְיֹשְׁבֵ֖י יְרֽוּשָׁלִָ֑ם כֹּֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל הִנְנִ֨י מֵבִ֤יא רָעָה֙ עַל־ הַמָּק֣וֹם הַזֶּ֔ה אֲשֶׁ֥ר כָּל־ שֹׁמְעָ֖הּ תִּצַּ֥לְנָה אָזְנָֽיו׃ (ט) יַ֣עַן ׀ אֲשֶׁ֣ר עֲזָבֻ֗נִי וַֽיְנַכְּר֞וּ אֶת־ הַמָּק֤וֹם הַזֶּה֙ וַיְקַטְּרוּ־ בוֹ֙ לֵאלֹהִ֣ים אֲחֵרִ֔ים אֲשֶׁ֧ר לֹֽא־ יְדָע֛וּם הֵ֥מָּה וַאֲבֽוֹתֵיהֶ֖ם וּמַלְכֵ֣י יְהוּדָ֑ה וּמָֽלְא֛וּ אֶת־ הַמָּק֥וֹם הַזֶּ֖ה דַּ֥ם נְקִיִּֽם׃ (י) וּבָנ֞וּ אֶת־ בָּמ֣וֹת הַבַּ֗עַל לִשְׂרֹ֧ף אֶת־ בְּנֵיהֶ֛ם בָּאֵ֖שׁ עֹל֣וֹת לַבָּ֑עַל אֲשֶׁ֤ר לֹֽא־ צִוִּ֙יתִי֙ וְלֹ֣א דִבַּ֔רְתִּי וְלֹ֥א עָלְתָ֖ה עַל־ לִבִּֽי׃ (יא) לָכֵ֞ן הִנֵּֽה־ יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וְלֹא־ יִקָּרֵא֩ לַמָּק֨וֹם הַזֶּ֥ה ע֛וֹד הַתֹּ֖פֶת וְגֵ֣יא בֶן־ הִנֹּ֑ם כִּ֖י אִם־ גֵּ֥יא הַהֲרֵגָֽה׃ (יב) וּ֠בַקֹּתִי אֶת־ עֲצַ֨ת יְהוּדָ֤ה וִירוּשָׁלִַ֙ם֙ בַּמָּק֣וֹם הַזֶּ֔ה וְהִפַּלְתִּ֤ים בַּחֶ֙רֶב֙ לִפְנֵ֣י אֹֽיְבֵיהֶ֔ם וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשָׁ֑ם וְנָתַתִּ֤י אֶת־ נִבְלָתָם֙ לְמַֽאֲכָ֔ל לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֥ת הָאָֽרֶץ׃ (יג) וְשַׂמְתִּי֙ אֶת־ הָעִ֣יר הַזֹּ֔את לְשַׁמָּ֖ה וְלִשְׁרֵקָ֑ה כֹּ֚ל עֹבֵ֣ר עָלֶ֔יהָ יִשֹּׁ֥ם וְיִשְׁרֹ֖ק עַל־ כָּל־ מַכֹּתֶֽהָ׃ (יד) וְהַֽאֲכַלְתִּ֞ים אֶת־ בְּשַׂ֣ר בְּנֵיהֶ֗ם וְאֵת֙ בְּשַׂ֣ר בְּנֹתֵיהֶ֔ם וְאִ֥ישׁ בְּשַׂר־ רֵעֵ֖הוּ יֹאכֵ֑לוּ בְּמָצוֹר֙ וּבְמָצ֔וֹק אֲשֶׁ֨ר יָצִ֧יקוּ לָהֶ֛ם אֹיְבֵיהֶ֖ם וּמְבַקְשֵׁ֥י נַפְשָֽׁם׃ (טו) וְשָׁבַרְתָּ֖ הַבַּקְבֻּ֑ק לְעֵינֵי֙ הָֽאֲנָשִׁ֔ים הַהֹלְכִ֖ים אוֹתָֽךְ׃ (טז) וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם כֹּה־ אָמַ֣ר ׀ יְהוָ֣ה צְבָא֗וֹת כָּ֣כָה אֶשְׁבֹּ֞ר אֶת־ הָעָ֤ם הַזֶּה֙ וְאֶת־ הָעִ֣יר הַזֹּ֔את כַּאֲשֶׁ֤ר יִשְׁבֹּר֙ אֶת־ כְּלִ֣י הַיּוֹצֵ֔ר אֲשֶׁ֛ר לֹֽא־ יוּכַ֥ל לְהֵרָפֵ֖ה ע֑וֹד וּבְתֹ֣פֶת יִקְבְּר֔וּ מֵאֵ֥ין מָק֖וֹם לִקְבּֽוֹר׃ (יז) כֵּֽן־ אֶעֱשֶׂ֞ה לַמָּק֥וֹם הַזֶּ֛ה נְאֻם־ יְהוָ֖ה וּלְיֽוֹשְׁבָ֑יו וְלָתֵ֛ת אֶת־ הָעִ֥יר הַזֹּ֖את כְּתֹֽפֶת׃ (יח) וְהָי֞וּ בָּתֵּ֣י יְרוּשָׁלִַ֗ם וּבָתֵּי֙ מַלְכֵ֣י יְהוּדָ֔ה כִּמְק֥וֹם הַתֹּ֖פֶת הַטְּמֵאִ֑ים לְכֹ֣ל הַבָּתִּ֗ים אֲשֶׁ֨ר קִטְּר֜וּ עַל־ גַּגֹּֽתֵיהֶם֙ לְכֹל֙ צְבָ֣א הַשָּׁמַ֔יִם וְהַסֵּ֥ךְ נְסָכִ֖ים לֵאלֹהִ֥ים אֲחֵרִֽים׃ (יט) וַיָּבֹ֤א יִרְמְיָ֙הוּ֙ מֵֽהַתֹּ֔פֶת אֲשֶׁ֨ר שְׁלָח֧וֹ יְהוָ֛ה שָׁ֖ם לְהִנָּבֵ֑א וַֽיַּעֲמֹד֙ בַּחֲצַ֣ר בֵּית־ יְהוָ֔ה וַיֹּ֖אמֶר אֶל־ כָּל־ הָעָֽם׃ (כ) כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֨י (מבי) מֵבִ֜יא אֶל־ הָעִ֤יר הַזֹּאת֙ וְעַל־ כָּל־ עָרֶ֔יהָ אֵ֚ת כָּל־ הָ֣רָעָ֔ה אֲשֶׁ֥ר דִּבַּ֖רְתִּי עָלֶ֑יהָ כִּ֤י הִקְשׁוּ֙ אֶת־ עָרְפָּ֔ם לְבִלְתִּ֖י שְׁמ֥וֹעַ אֶת־ דְּבָרָֽי׃ (כא) וַיִּשְׁמַ֤ע פַּשְׁחוּר֙ בֶּן־ אִמֵּ֣ר הַכֹּהֵ֔ן וְהֽוּא־ פָקִ֥יד נָגִ֖יד בְּבֵ֣ית יְהוָ֑ה אֶֽת־ יִרְמְיָ֔הוּ נִבָּ֖א אֶת־ הַדְּבָרִ֥ים הָאֵֽלֶּה׃ (כב) וַיַּכֶּ֣ה פַשְׁח֔וּר אֵ֖ת יִרְמְיָ֣הוּ הַנָּבִ֑יא וַיִּתֵּ֨ן אֹת֜וֹ עַל־ הַמַּהְפֶּ֗כֶת אֲשֶׁ֨ר בְּשַׁ֤עַר בִּנְיָמִן֙ הָֽעֶלְי֔וֹן אֲשֶׁ֖ר בְּבֵ֥ית יְהוָֽה׃ (כג) וַֽיְהִי֙ מִֽמָּחֳרָ֔ת וַיֹּצֵ֥א פַשְׁח֛וּר אֶֽת־ יִרְמְיָ֖הוּ מִן־ הַמַּהְפָּ֑כֶת וַיֹּ֨אמֶר אֵלָ֜יו יִרְמְיָ֗הוּ לֹ֤א פַשְׁחוּר֙ קָרָ֤א יְהוָה֙ שְׁמֶ֔ךָ כִּ֖י אִם־ מָג֥וֹר מִסָּבִֽיב׃ (כד) כִּ֣י כֹ֣ה אָמַ֣ר יְהוָ֡ה הִנְנִי֩ נֹתֶנְךָ֨ לְמָג֜וֹר לְךָ֣ וּלְכָל־ אֹהֲבֶ֗יךָ וְנָֽפְל֛וּ בְּחֶ֥רֶב אֹיְבֵיהֶ֖ם וְעֵינֶ֣יךָ רֹא֑וֹת וְאֶת־ כָּל־ יְהוּדָ֗ה אֶתֵּן֙ בְּיַ֣ד מֶֽלֶךְ־ בָּבֶ֔ל וְהִגְלָ֥ם בָּבֶ֖לָה וְהִכָּ֥ם בֶּחָֽרֶב׃ (כה) וְנָתַתִּ֗י אֶת־ כָּל־ חֹ֙סֶן֙ הָעִ֣יר הַזֹּ֔את וְאֶת־ כָּל־ יְגִיעָ֖הּ וְאֶת־ כָּל־ יְקָרָ֑הּ וְאֵ֨ת כָּל־ אוֹצְר֜וֹת מַלְכֵ֣י יְהוּדָ֗ה אֶתֵּן֙ בְּיַ֣ד אֹֽיְבֵיהֶ֔ם וּבְזָזוּם֙ וּלְקָח֔וּם וֶהֱבִיא֖וּם בָּבֶֽלָה׃ (כו) וְאַתָּ֣ה פַשְׁח֗וּר וְכֹל֙ יֹשְׁבֵ֣י בֵיתֶ֔ךָ תֵּלְכ֖וּ בַּשֶּׁ֑בִי וּבָבֶ֣ל תָּב֗וֹא וְשָׁ֤ם תָּמוּת֙ וְשָׁ֣ם תִּקָּבֵ֔ר אַתָּה֙ וְכָל־ אֹ֣הֲבֶ֔יךָ אֲשֶׁר־ נִבֵּ֥אתָ לָהֶ֖ם בַּשָּֽׁקֶר׃ (כז) פִּתִּיתַ֤נִי יְהוָה֙ וָֽאֶפָּ֔ת חֲזַקְתַּ֖נִי וַתּוּכָ֑ל הָיִ֤יתִי לִשְׂחוֹק֙ כָּל־ הַיּ֔וֹם כֻּלֹּ֖ה לֹעֵ֥ג לִֽי׃ (כח) כִּֽי־ מִדֵּ֤י אֲדַבֵּר֙ אֶזְעָ֔ק חָמָ֥ס וָשֹׁ֖ד אֶקְרָ֑א כִּֽי־ הָיָ֨ה דְבַר־ יְהוָ֥ה לִ֛י לְחֶרְפָּ֥ה וּלְקֶ֖לֶס כָּל־ הַיּֽוֹם׃ (כט) וְאָמַרְתִּ֣י לֹֽא־ אֶזְכְּרֶ֗נּוּ וְלֹֽא־ אֲדַבֵּ֥ר עוֹד֙ בִּשְׁמ֔וֹ וְהָיָ֤ה בְלִבִּי֙ כְּאֵ֣שׁ בֹּעֶ֔רֶת עָצֻ֖ר בְּעַצְמֹתָ֑י וְנִלְאֵ֥יתִי כַּֽלְכֵ֖ל וְלֹ֥א אוּכָֽל׃ (ל) כִּ֣י שָׁמַ֜עְתִּי דִּבַּ֣ת רַבִּים֮ מָג֣וֹר מִסָּבִיב֒ הַגִּ֙ידוּ֙ וְנַגִּידֶ֔נּוּ כֹּ֚ל אֱנ֣וֹשׁ שְׁלוֹמִ֔י שֹׁמְרֵ֖י צַלְעִ֑י אוּלַ֤י יְפֻתֶּה֙ וְנ֣וּכְלָה ל֔וֹ וְנִקְחָ֥ה נִקְמָתֵ֖נוּ מִמֶּֽנּוּ׃ (לא) וַֽיהוָ֤ה אוֹתִי֙ כְּגִבּ֣וֹר עָרִ֔יץ עַל־ כֵּ֛ן רֹדְפַ֥י יִכָּשְׁל֖וּ וְלֹ֣א יֻכָ֑לוּ בֹּ֤שׁוּ מְאֹד֙ כִּֽי־ לֹ֣א הִשְׂכִּ֔ילוּ כְּלִמַּ֥ת עוֹלָ֖ם לֹ֥א תִשָּׁכֵֽחַ׃ (לב) וַיהוָ֤ה צְבָאוֹת֙ בֹּחֵ֣ן צַדִּ֔יק רֹאֶ֥ה כְלָי֖וֹת וָלֵ֑ב אֶרְאֶ֤ה נִקְמָֽתְךָ֙ מֵהֶ֔ם כִּ֥י אֵלֶ֖יךָ גִּלִּ֥יתִי אֶת־ רִיבִֽי׃
