ירמיהו פרק מד לפי החלוקה לסדרים

פרק מד בספר ירמיהו (30 פסוקים) נמצא בירמיהו סדר כה (פסוקים כה–מג) ובירמיהו סדר כו (פסוקים א–יא).

ירמיהו סדר כה

(א) וְאֶתֵּ֥ן לָכֶ֛ם רַחֲמִ֖ים וְרִחַ֣ם אֶתְכֶ֑ם וְהֵשִׁ֥יב אֶתְכֶ֖ם אֶל־ אַדְמַתְכֶֽם׃ (ב) וְאִם־ אֹמְרִ֣ים אַתֶּ֔ם לֹ֥א נֵשֵׁ֖ב בָּאָ֣רֶץ הַזֹּ֑את לְבִלְתִּ֣י שְׁמֹ֔עַ בְּק֖וֹל יְהוָ֥ה אֱלֹהֵיכֶֽם׃ (ג) לֵאמֹ֗ר לֹ֚א כִּ֣י אֶ֤רֶץ מִצְרַ֙יִם֙ נָב֔וֹא אֲשֶׁ֤ר לֹֽא־ נִרְאֶה֙ מִלְחָמָ֔ה וְק֥וֹל שׁוֹפָ֖ר לֹ֣א נִשְׁמָ֑ע וְלַלֶּ֥חֶם לֹֽא־ נִרְעָ֖ב וְשָׁ֥ם נֵשֵֽׁב׃ (ד) וְעַתָּ֕ה לָכֵ֛ן שִׁמְע֥וּ דְבַר־ יְהוָ֖ה שְׁאֵרִ֣ית יְהוּדָ֑ה כֹּֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל אִם־ אַ֠תֶּם שׂ֣וֹם תְּשִׂמ֤וּן פְּנֵיכֶם֙ לָבֹ֣א מִצְרַ֔יִם וּבָאתֶ֖ם לָג֥וּר שָֽׁם׃ (ה) וְהָיְתָ֣ה הַחֶ֗רֶב אֲשֶׁ֤ר אַתֶּם֙ יְרֵאִ֣ים מִמֶּ֔נָּה שָׁ֛ם תַּשִּׂ֥יג אֶתְכֶ֖ם בְּאֶ֣רֶץ מִצְרָ֑יִם וְהָרָעָ֞ב אֲשֶׁר־ אַתֶּ֣ם ׀ דֹּאֲגִ֣ים מִמֶּ֗נּוּ שָׁ֣ם יִדְבַּ֧ק אַחֲרֵיכֶ֛ם מִצְרַ֖יִם וְשָׁ֥ם תָּמֻֽתוּ׃ (ו) וְיִֽהְי֣וּ כָל־ הָאֲנָשִׁ֗ים אֲשֶׁר־ שָׂ֨מוּ אֶת־ פְּנֵיהֶ֜ם לָב֤וֹא מִצְרַ֙יִם֙ לָג֣וּר שָׁ֔ם יָמ֕וּתוּ בַּחֶ֖רֶב בָּרָעָ֣ב וּבַדָּ֑בֶר וְלֹֽא־ יִהְיֶ֤ה לָהֶם֙ שָׂרִ֣יד וּפָלִ֔יט מִפְּנֵי֙ הָֽרָעָ֔ה אֲשֶׁ֥ר אֲנִ֖י מֵבִ֥יא עֲלֵיהֶֽם׃ (ז) כִּי֩ כֹ֨ה אָמַ֜ר יְהוָ֣ה צְבָאוֹת֮ אֱלֹהֵ֣י יִשְׂרָאֵל֒ כַּאֲשֶׁר֩ נִתַּ֨ךְ אַפִּ֜י וַחֲמָתִ֗י עַל־ יֹֽשְׁבֵי֙ יְר֣וּשָׁלִַ֔ם כֵּ֣ן תִּתַּ֤ךְ חֲמָתִי֙ עֲלֵיכֶ֔ם בְּבֹאֲכֶ֖ם מִצְרָ֑יִם וִהְיִיתֶ֞ם לְאָלָ֤ה וּלְשַׁמָּה֙ וְלִקְלָלָ֣ה וּלְחֶרְפָּ֔ה וְלֹֽא־ תִרְא֣וּ ע֔וֹד אֶת־ הַמָּק֖וֹם הַזֶּֽה׃ (ח) דִּבֶּ֨ר יְהוָ֤ה עֲלֵיכֶם֙ שְׁאֵרִ֣ית יְהוּדָ֔ה אַל־ תָּבֹ֖אוּ מִצְרָ֑יִם יָדֹ֙עַ֙ תֵּֽדְע֔וּ כִּי־ הַעִידֹ֥תִי בָכֶ֖ם הַיּֽוֹם׃ (ט) כִּ֣י (התעתים) הִתְעֵיתֶם֮ בְּנַפְשֽׁוֹתֵיכֶם֒ כִּֽי־ אַתֶּ֞ם שְׁלַחְתֶּ֣ם אֹתִ֗י אֶל־ יְהוָ֤ה אֱלֹֽהֵיכֶם֙ לֵאמֹ֔ר הִתְפַּלֵּ֣ל בַּעֲדֵ֔נוּ אֶל־ יְהוָ֖ה אֱלֹהֵ֑ינוּ וּכְכֹל֩ אֲשֶׁ֨ר יֹאמַ֜ר יְהוָ֧ה אֱלֹהֵ֛ינוּ כֵּ֥ן הַגֶּד־ לָ֖נוּ וְעָשִֽׂינוּ׃ (י) וָאַגִּ֥ד לָכֶ֖ם הַיּ֑וֹם וְלֹ֣א שְׁמַעְתֶּ֗ם בְּקוֹל֙ יְהוָ֣ה אֱלֹֽהֵיכֶ֔ם וּלְכֹ֖ל אֲשֶׁר־ שְׁלָחַ֥נִי אֲלֵיכֶֽם׃ (יא) וְעַתָּה֙ יָדֹ֣עַ תֵּֽדְע֔וּ כִּ֗י בַּחֶ֛רֶב בָּרָעָ֥ב וּבַדֶּ֖בֶר תָּמ֑וּתוּ בַּמָּקוֹם֙ אֲשֶׁ֣ר חֲפַצְתֶּ֔ם לָב֖וֹא לָג֥וּר שָֽׁם׃ (יב) וַיְהִי֩ כְּכַלּ֨וֹת יִרְמְיָ֜הוּ לְדַבֵּ֣ר אֶל־ כָּל־ הָעָ֗ם אֶת־ כָּל־ דִּבְרֵי֙ יְהוָ֣ה אֱלֹהֵיהֶ֔ם אֲשֶׁ֧ר שְׁלָח֛וֹ יְהוָ֥ה אֱלֹהֵיהֶ֖ם אֲלֵיהֶ֑ם אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (יג) וַיֹּ֨אמֶר עֲזַרְיָ֤ה בֶן־ הוֹשַֽׁעְיָה֙ וְיוֹחָנָ֣ן בֶּן־ קָרֵ֔חַ וְכָל־ הָאֲנָשִׁ֖ים הַזֵּדִ֑ים אֹמְרִ֣ים אֶֽל־ יִרְמְיָ֗הוּ שֶׁ֚קֶר אַתָּ֣ה מְדַבֵּ֔ר לֹ֣א שְׁלָחֲךָ֞ יְהוָ֤ה אֱלֹהֵ֙ינוּ֙ לֵאמֹ֔ר לֹֽא־ תָבֹ֥אוּ מִצְרַ֖יִם לָג֥וּר שָֽׁם׃ (יד) כִּ֗י בָּרוּךְ֙ בֶּן־ נֵ֣רִיָּ֔ה מַסִּ֥ית אֹתְךָ֖ בָּ֑נוּ לְמַעַן֩ תֵּ֨ת אֹתָ֤נוּ בְיַֽד־ הַכַּשְׂדִּים֙ לְהָמִ֣ית אֹתָ֔נוּ וּלְהַגְל֥וֹת אֹתָ֖נוּ בָּבֶֽל׃ (טו) וְלֹֽא־ שָׁמַע֩ יוֹחָנָ֨ן בֶּן־ קָרֵ֜חַ וְכָל־ שָׂרֵ֧י הַחֲיָלִ֛ים וְכָל־ הָעָ֖ם בְּק֣וֹל יְהוָ֑ה לָשֶׁ֖בֶת בְּאֶ֥רֶץ יְהוּדָֽה׃ (טז) וַיִּקַּ֞ח יוֹחָנָ֤ן בֶּן־ קָרֵ֙חַ֙ וְכָל־ שָׂרֵ֣י הַחֲיָלִ֔ים אֵ֖ת כָּל־ שְׁאֵרִ֣ית יְהוּדָ֑ה אֲשֶׁר־ שָׁ֗בוּ מִכָּל־ הַגּוֹיִם֙ אֲשֶׁ֣ר נִדְּחוּ־ שָׁ֔ם לָג֖וּר בְּאֶ֥רֶץ יְהוּדָֽה׃ (יז) אֶֽת־ הַ֠גְּבָרִים וְאֶת־ הַנָּשִׁ֣ים וְאֶת־ הַטַּף֮ וְאֶת־ בְּנ֣וֹת הַמֶּלֶךְ֒ וְאֵ֣ת כָּל־ הַנֶּ֗פֶשׁ אֲשֶׁ֤ר הִנִּ֙יחַ֙ נְבוּזַרְאֲדָ֣ן רַב־ טַבָּחִ֔ים אֶת־ גְּדַלְיָ֖הוּ בֶּן־ אֲחִיקָ֣ם בֶּן־ שָׁפָ֑ן וְאֵת֙ יִרְמְיָ֣הוּ הַנָּבִ֔יא וְאֶת־ בָּר֖וּךְ בֶּן־ נֵרִיָּֽהוּ׃ (יח) וַיָּבֹ֙אוּ֙ אֶ֣רֶץ מִצְרַ֔יִם כִּ֛י לֹ֥א שָׁמְע֖וּ בְּק֣וֹל יְהוָ֑ה וַיָּבֹ֖אוּ עַד־ תַּחְפַּנְחֵֽס׃ (יט) וַיְהִ֤י דְבַר־ יְהוָה֙ אֶֽל־ יִרְמְיָ֔הוּ בְּתַחְפַּנְחֵ֖ס לֵאמֹֽר׃ (כ) קַ֣ח בְּיָדְךָ֞ אֲבָנִ֣ים גְּדֹל֗וֹת וּטְמַנְתָּ֤ם בַּמֶּ֙לֶט֙ בַּמַּלְבֵּ֔ן אֲשֶׁ֛ר בְּפֶ֥תַח בֵּית־ פַּרְעֹ֖ה בְּתַחְפַּנְחֵ֑ס לְעֵינֵ֖י אֲנָשִׁ֥ים יְהוּדִֽים׃ (כא) וְאָמַרְתָּ֣ אֲלֵיהֶ֡ם כֹּֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל הִנְנִ֤י שֹׁלֵ֙חַ֙ וְ֠לָקַחְתִּי אֶת־ נְבוּכַדְרֶאצַּ֤ר מֶֽלֶךְ־ בָּבֶל֙ עַבְדִּ֔י וְשַׂמְתִּ֣י כִסְא֔וֹ מִמַּ֛עַל לָאֲבָנִ֥ים הָאֵ֖לֶּה אֲשֶׁ֣ר טָמָ֑נְתִּי וְנָטָ֥ה אֶת־ (שפרורו) שַׁפְרִיר֖וֹ עֲלֵיהֶֽם׃ (כב) (ובאה) וּבָ֕א וְהִכָּ֖ה אֶת־ אֶ֣רֶץ מִצְרָ֑יִם אֲשֶׁ֧ר לַמָּ֣וֶת לַמָּ֗וֶת וַאֲשֶׁ֤ר לַשְּׁבִי֙ לַשֶּׁ֔בִי וַאֲשֶׁ֥ר לַחֶ֖רֶב לֶחָֽרֶב׃ (כג) וְהִצַּ֣תִּי אֵ֗שׁ בְּבָתֵּי֙ אֱלֹהֵ֣י מִצְרַ֔יִם וּשְׂרָפָ֖ם וְשָׁבָ֑ם וְעָטָה֩ אֶת־ אֶ֨רֶץ מִצְרַ֜יִם כַּאֲשֶׁר־ יַעְטֶ֤ה הָֽרֹעֶה֙ אֶת־ בִּגְד֔וֹ וְיָצָ֥א מִשָּׁ֖ם בְּשָׁלֽוֹם׃ (כד) וְשִׁבַּ֗ר אֶֽת־ מַצְּבוֹת֙ בֵּ֣ית שֶׁ֔מֶשׁ אֲשֶׁ֖ר בְּאֶ֣רֶץ מִצְרָ֑יִם וְאֶת־ בָּתֵּ֥י אֱלֹהֵֽי־ מִצְרַ֖יִם יִשְׂרֹ֥ף בָּאֵֽשׁ׃ תחילת פרק מד (כה) הַדָּבָר֙ אֲשֶׁ֣ר הָיָ֣ה אֶֽל־ יִרְמְיָ֔הוּ אֶ֚ל כָּל־ הַיְּהוּדִ֔ים הַיֹּשְׁבִ֖ים בְּאֶ֣רֶץ מִצְרָ֑יִם הַיֹּשְׁבִ֤ים בְּמִגְדֹּל֙ וּבְתַחְפַּנְחֵ֣ס וּבְנֹ֔ף וּבְאֶ֥רֶץ פַּתְר֖וֹס לֵאמֹֽר׃ (כו) כֹּה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אַתֶּ֣ם רְאִיתֶ֗ם אֵ֤ת כָּל־ הָֽרָעָה֙ אֲשֶׁ֤ר הֵבֵ֙אתִי֙ עַל־ יְר֣וּשָׁלִַ֔ם וְעַ֖ל כָּל־ עָרֵ֣י יְהוּדָ֑ה וְהִנָּ֤ם חָרְבָּה֙ הַיּ֣וֹם הַזֶּ֔ה וְאֵ֥ין בָּהֶ֖ם יוֹשֵֽׁב׃ (כז) מִפְּנֵ֣י רָעָתָ֗ם אֲשֶׁ֤ר עָשׂוּ֙ לְהַכְעִסֵ֔נִי לָלֶ֣כֶת לְקַטֵּ֔ר לַעֲבֹ֖ד לֵאלֹהִ֣ים אֲחֵרִ֑ים אֲשֶׁר֙ לֹ֣א יְדָע֔וּם הֵ֖מָּה אַתֶּ֥ם וַאֲבֹתֵיכֶֽם׃ (כח) וָאֶשְׁלַ֤ח אֲלֵיכֶם֙ אֶת־ כָּל־ עֲבָדַ֣י הַנְּבִיאִ֔ים הַשְׁכֵּ֥ים וְשָׁלֹ֖חַ לֵאמֹ֑ר אַל־ נָ֣א תַעֲשׂ֗וּ אֵ֛ת דְּבַֽר־ הַתֹּעֵבָ֥ה הַזֹּ֖את אֲשֶׁ֥ר שָׂנֵֽאתִי׃ (כט) וְלֹ֤א שָֽׁמְעוּ֙ וְלֹא־ הִטּ֣וּ אֶת־ אָזְנָ֔ם לָשׁ֖וּב מֵרָֽעָתָ֑ם לְבִלְתִּ֥י קַטֵּ֖ר לֵאלֹהִ֥ים אֲחֵרִֽים׃ (ל) וַתִּתַּ֤ךְ חֲמָתִי֙ וְאַפִּ֔י וַתִּבְעַר֙ בְּעָרֵ֣י יְהוּדָ֔ה וּבְחֻצ֖וֹת יְרֽוּשָׁלִָ֑ם וַתִּהְיֶ֛ינָה לְחָרְבָּ֥ה לִשְׁמָמָ֖ה כַּיּ֥וֹם הַזֶּֽה׃ (לא) וְעַתָּ֡ה כֹּֽה־ אָמַ֣ר יְהוָה֩ אֱלֹהֵ֨י צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל לָמָה֩ אַתֶּ֨ם עֹשִׂ֜ים רָעָ֤ה גְדוֹלָה֙ אֶל־ נַפְשֹׁ֣תֵכֶ֔ם לְהַכְרִ֨ית לָכֶ֧ם אִישׁ־ וְאִשָּׁ֛ה עוֹלֵ֥ל וְיוֹנֵ֖ק מִתּ֣וֹךְ יְהוּדָ֑ה לְבִלְתִּ֛י הוֹתִ֥יר לָכֶ֖ם שְׁאֵרִֽית׃ (לב) לְהַכְעִסֵ֙נִי֙ בְּמַעֲשֵׂ֣י יְדֵיכֶ֔ם לְקַטֵּ֞ר לֵאלֹהִ֤ים אֲחֵרִים֙ בְּאֶ֣רֶץ מִצְרַ֔יִם אֲשֶׁר־ אַתֶּ֥ם בָּאִ֖ים לָג֣וּר שָׁ֑ם לְמַ֙עַן֙ הַכְרִ֣ית לָכֶ֔ם וּלְמַ֤עַן יָסַ֧ף יְהוָ֛ה יָג֖וֹן עַל־ מַכְאֹבִ֑י יָגַ֙עְתִּי֙ בְּאַנְחָתִ֔י וּמְנוּחָ֖ה לֹ֥א מָצָֽאתִי׃ (טו) כֹּ֣ה ׀ תֹּאמַ֣ר אֵלָ֗יו כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנֵּ֤ה אֲשֶׁר־ בָּנִ֙יתִי֙ אֲנִ֣י הֹרֵ֔ס וְאֵ֥ת אֲשֶׁר־ נָטַ֖עְתִּי אֲנִ֣י נֹתֵ֑שׁ וְאֶת־ כָּל־ הָאָ֖רֶץ הִֽיא׃ (טז) וְאַתָּ֛ה תְּבַקֶּשׁ־ לְךָ֥ גְדֹל֖וֹת אַל־ תְּבַקֵּ֑שׁ כִּ֡י הִנְנִי֩ מֵבִ֨יא רָעָ֤ה עַל־ כָּל־ בָּשָׂר֙ נְאֻם־ יְהוָ֔ה וְנָתַתִּ֨י לְךָ֤ אֶֽת־ נַפְשְׁךָ֙ לְשָׁלָ֔ל עַ֥ל כָּל־ הַמְּקֹמ֖וֹת אֲשֶׁ֥ר תֵּֽלֶךְ־ שָֽׁם׃ (יז) אֲשֶׁ֨ר הָיָ֧ה דְבַר־ יְהוָ֛ה אֶל־ יִרְמְיָ֥הוּ הַנָּבִ֖יא עַל־ הַגּוֹיִֽם׃ (יח) לְמִצְרַ֗יִם עַל־ חֵ֨יל פַּרְעֹ֤ה נְכוֹ֙ מֶ֣לֶךְ מִצְרַ֔יִם אֲשֶׁר־ הָיָ֥ה עַל־ נְהַר־ פְּרָ֖ת בְּכַרְכְּמִ֑שׁ אֲשֶׁ֣ר הִכָּ֗ה נְבֽוּכַדְרֶאצַּר֙ מֶ֣לֶךְ בָּבֶ֔ל בִּשְׁנַת֙ הָֽרְבִיעִ֔ית לִיהוֹיָקִ֥ים בֶּן־ יֹאשִׁיָּ֖הוּ מֶ֥לֶךְ יְהוּדָֽה׃ (יט) עִרְכ֤וּ מָגֵן֙ וְצִנָּ֔ה וּגְשׁ֖וּ לַמִּלְחָמָֽה׃ (כ) אִסְר֣וּ הַסּוּסִ֗ים וַֽעֲלוּ֙ הַפָּ֣רָשִׁ֔ים וְהִֽתְיַצְּב֖וּ בְּכ֥וֹבָעִ֑ים מִרְקוּ֙ הָֽרְמָחִ֔ים לִבְשׁ֖וּ הַסִּרְיֹנֹֽת׃ (כא) מַדּ֣וּעַ רָאִ֗יתִי הֵ֣מָּה חַתִּים֮ נְסֹגִ֣ים אָחוֹר֒ וְגִבּוֹרֵיהֶ֣ם יֻכַּ֔תּוּ וּמָנ֥וֹס נָ֖סוּ וְלֹ֣א הִפְנ֑וּ מָג֥וֹר מִסָּבִ֖יב נְאֻם־ יְהוָֽה׃ (כב) אַל־ יָנ֣וּס הַקַּ֔ל וְאַל־ יִמָּלֵ֖ט הַגִּבּ֑וֹר צָפ֙וֹנָה֙ עַל־ יַ֣ד נְהַר־ פְּרָ֔ת כָּשְׁל֖וּ וְנָפָֽלוּ׃ (כג) מִי־ זֶ֖ה כַּיְאֹ֣ר יַֽעֲלֶ֑ה כַּנְּהָר֕וֹת יִֽתְגָּעֲשׁ֖וּ מֵימָֽיו׃ (כד) מִצְרַ֙יִם֙ כַּיְאֹ֣ר יַֽעֲלֶ֔ה וְכַנְּהָר֖וֹת יִתְגֹּ֣עֲשׁוּ מָ֑יִם וַיֹּ֗אמֶר אַֽעֲלֶה֙ אֲכַסֶּה־ אֶ֔רֶץ אֹבִ֥ידָה עִ֖יר וְיֹ֥שְׁבֵי בָֽהּ׃ (כה) עֲל֤וּ הַסּוּסִים֙ וְהִתְהֹלְל֣וּ הָרֶ֔כֶב וְיֵצְא֖וּ הַגִּבּוֹרִ֑ים כּ֤וּשׁ וּפוּט֙ תֹּפְשֵׂ֣י מָגֵ֔ן וְלוּדִ֕ים תֹּפְשֵׂ֖י דֹּ֥רְכֵי קָֽשֶׁת׃ (כו) וְֽהַיּ֨וֹם הַה֜וּא לַאדֹנָ֧י יְהוִ֣ה צְבָא֗וֹת י֤וֹם נְקָמָה֙ לְהִנָּקֵ֣ם מִצָּרָ֔יו וְאָכְלָ֥ה חֶ֙רֶב֙ וְשָׂ֣בְעָ֔ה וְרָוְתָ֖ה מִדָּמָ֑ם כִּ֣י זֶ֠בַח לַאדֹנָ֨י יְהוִ֧ה צְבָא֛וֹת בְּאֶ֥רֶץ צָפ֖וֹן אֶל־ נְהַר־ פְּרָֽת׃ (כז) עֲלִ֤י גִלְעָד֙ וּקְחִ֣י צֳרִ֔י בְּתוּלַ֖ת בַּת־ מִצְרָ֑יִם לַשָּׁוְא֙ (הרביתי) הִרְבֵּ֣ית רְפֻא֔וֹת תְּעָלָ֖ה אֵ֥ין לָֽךְ׃ (כח) שָׁמְע֤וּ גוֹיִם֙ קְלוֹנֵ֔ךְ וְצִוְחָתֵ֖ךְ מָלְאָ֣ה הָאָ֑רֶץ כִּֽי־ גִבּ֤וֹר בְּגִבּוֹר֙ כָּשָׁ֔לוּ יַחְדָּ֖יו נָפְל֥וּ שְׁנֵיהֶֽם׃ (כט) הַדָּבָר֙ אֲשֶׁ֣ר דִּבֶּ֣ר יְהוָ֔ה אֶֽל־ יִרְמְיָ֖הוּ הַנָּבִ֑יא לָב֗וֹא נְבֽוּכַדְרֶאצַּר֙ מֶ֣לֶךְ בָּבֶ֔ל לְהַכּ֖וֹת אֶת־ אֶ֥רֶץ מִצְרָֽיִם׃ (ל) הַגִּ֤ידוּ בְמִצְרַ֙יִם֙ וְהַשְׁמִ֣יעוּ בְמִגְדּ֔וֹל וְהַשְׁמִ֥יעוּ בְנֹ֖ף וּבְתַחְפַּנְחֵ֑ס אִמְר֗וּ הִתְיַצֵּב֙ וְהָכֵ֣ן לָ֔ךְ כִּֽי־ אָכְלָ֥ה חֶ֖רֶב סְבִיבֶֽיךָ׃ (לא) מַדּ֖וּעַ נִסְחַ֣ף אַבִּירֶ֑יךָ לֹ֣א עָמַ֔ד כִּ֥י יְהוָ֖ה הֲדָפֽוֹ׃ (לב) הִרְבָּ֖ה כּוֹשֵׁ֑ל גַּם־ נָפַ֞ל אִ֣ישׁ אֶל־ רֵעֵ֗הוּ וַיֹּֽאמְרוּ֙ ק֣וּמָה ׀ וְנָשֻׁ֣בָה אֶל־ עַמֵּ֗נוּ וְאֶל־ אֶ֙רֶץ֙ מֽוֹלַדְתֵּ֔נוּ מִפְּנֵ֖י חֶ֥רֶב הַיּוֹנָֽה׃ (לג) קָרְא֖וּ שָׁ֑ם פַּרְעֹ֤ה מֶֽלֶךְ־ מִצְרַ֙יִם֙ שָׁא֔וֹן הֶעֱבִ֖יר הַמּוֹעֵֽד׃ (לד) חַי־ אָ֙נִי֙ נְאֻם־ הַמֶּ֔לֶךְ יְהוָ֥ה צְבָא֖וֹת שְׁמ֑וֹ כִּ֚י כְּתָב֣וֹר בֶּֽהָרִ֔ים וּכְכַרְמֶ֖ל בַּיָּ֥ם יָבֽוֹא׃ (לה) כְּלֵ֤י גוֹלָה֙ עֲשִׂ֣י לָ֔ךְ יוֹשֶׁ֖בֶת בַּת־ מִצְרָ֑יִם כִּֽי־ נֹף֙ לְשַׁמָּ֣ה תִֽהְיֶ֔ה וְנִצְּתָ֖ה מֵאֵ֥ין יוֹשֵֽׁב׃ (לו) עֶגְלָ֥ה יְפֵֽה־ פִיָּ֖ה מִצְרָ֑יִם קֶ֥רֶץ מִצָּפ֖וֹן בָּ֥א בָֽא׃ (לז) גַּם־ שְׂכִרֶ֤יהָ בְקִרְבָּהּ֙ כְּעֶגְלֵ֣י מַרְבֵּ֔ק כִּֽי־ גַם־ הֵ֧מָּה הִפְנ֛וּ נָ֥סוּ יַחְדָּ֖יו לֹ֣א עָמָ֑דוּ כִּ֣י י֥וֹם אֵידָ֛ם בָּ֥א עֲלֵיהֶ֖ם עֵ֥ת פְּקֻדָּתָֽם׃ (לח) קוֹלָ֖הּ כַּנָּחָ֣שׁ יֵלֵ֑ךְ כִּֽי־ בְחַ֣יִל יֵלֵ֔כוּ וּבְקַרְדֻּמּוֹת֙ בָּ֣אוּ לָ֔הּ כְּחֹטְבֵ֖י עֵצִֽים׃ (לט) כָּרְת֤וּ יַעְרָהּ֙ נְאֻם־ יְהוָ֔ה כִּ֖י לֹ֣א יֵֽחָקֵ֑ר כִּ֤י רַבּוּ֙ מֵֽאַרְבֶּ֔ה וְאֵ֥ין לָהֶ֖ם מִסְפָּֽר׃ (מ) הֹבִ֖ישָׁה בַּת־ מִצְרָ֑יִם נִתְּנָ֖ה בְּיַ֥ד עַם־ צָפֽוֹן׃ (מא) אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל הִנְנִ֤י פוֹקֵד֙ אֶל־ אָמ֣וֹן מִנֹּ֔א וְעַל־ פַּרְעֹה֙ וְעַל־ מִצְרַ֔יִם וְעַל־ אֱלֹהֶ֖יהָ וְעַל־ מְלָכֶ֑יהָ וְעַ֨ל־ פַּרְעֹ֔ה וְעַ֥ל הַבֹּטְחִ֖ים בּֽוֹ׃ (מב) וּנְתַתִּ֗ים בְּיַד֙ מְבַקְשֵׁ֣י נַפְשָׁ֔ם וּבְיַ֛ד נְבֽוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל וּבְיַד־ עֲבָדָ֑יו וְאַחֲרֵי־ כֵ֛ן תִּשְׁכֹּ֥ן כִּֽימֵי־ קֶ֖דֶם נְאֻם־ יְהוָֽה׃
