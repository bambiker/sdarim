ירמיהו פרק לג לפי החלוקה לסדרים

פרק לג בספר ירמיהו (26 פסוקים) נמצא בירמיהו סדר יט (פסוקים כד–לח) ובירמיהו סדר כ (פסוקים א–יא).

ירמיהו סדר יט

(א) וַתִּתֵּ֤ן לָהֶם֙ אֶת־ הָאָ֣רֶץ הַזֹּ֔את אֲשֶׁר־ נִשְׁבַּ֥עְתָּ לַאֲבוֹתָ֖ם לָתֵ֣ת לָהֶ֑ם אֶ֛רֶץ זָבַ֥ת חָלָ֖ב וּדְבָֽשׁ׃ (ב) וַיָּבֹ֜אוּ וַיִּֽרְשׁ֣וּ אֹתָ֗הּ וְלֹֽא־ שָׁמְע֤וּ בְקוֹלֶ֙ךָ֙ (ובתרותך) וּבְתוֹרָתְךָ֣ לֹא־ הָלָ֔כוּ אֵת֩ כָּל־ אֲשֶׁ֨ר צִוִּ֧יתָה לָהֶ֛ם לַעֲשׂ֖וֹת לֹ֣א עָשׂ֑וּ וַתַּקְרֵ֣א אֹתָ֔ם אֵ֥ת כָּל־ הָרָעָ֖ה הַזֹּֽאת׃ (ג) הִנֵּ֣ה הַסֹּלְל֗וֹת בָּ֣אוּ הָעִיר֮ לְלָכְדָהּ֒ וְהָעִ֣יר נִתְּנָ֗ה בְּיַ֤ד הַכַּשְׂדִּים֙ הַנִּלְחָמִ֣ים עָלֶ֔יהָ מִפְּנֵ֛י הַחֶ֥רֶב וְהָרָעָ֖ב וְהַדָּ֑בֶר וַאֲשֶׁ֥ר דִּבַּ֛רְתָּ הָיָ֖ה וְהִנְּךָ֥ רֹאֶֽה׃ (ד) וְאַתָּ֞ה אָמַ֤רְתָּ אֵלַי֙ אֲדֹנָ֣י יְהוִ֔ה קְנֵֽה־ לְךָ֧ הַשָּׂדֶ֛ה בַּכֶּ֖סֶף וְהָעֵ֣ד עֵדִ֑ים וְהָעִ֥יר נִתְּנָ֖ה בְּיַ֥ד הַכַּשְׂדִּֽים׃ (ה) וַיְהִי֙ דְּבַר־ יְהוָ֔ה אֶֽל־ יִרְמְיָ֖הוּ לֵאמֹֽר׃ (ו) הִנֵּה֙ אֲנִ֣י יְהוָ֔ה אֱלֹהֵ֖י כָּל־ בָּשָׂ֑ר הֲ‍ֽמִמֶּ֔נִּי יִפָּלֵ֖א כָּל־ דָּבָֽר׃ (ז) לָכֵ֕ן כֹּ֖ה אָמַ֣ר יְהוָ֑ה הִנְנִ֣י נֹתֵן֩ אֶת־ הָעִ֨יר הַזֹּ֜את בְּיַ֣ד הַכַּשְׂדִּ֗ים וּבְיַ֛ד נְבֽוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל וּלְכָדָֽהּ׃ (ח) וּבָ֣אוּ הַכַּשְׂדִּ֗ים הַנִּלְחָמִים֙ עַל־ הָעִ֣יר הַזֹּ֔את וְהִצִּ֜יתוּ אֶת־ הָעִ֥יר הַזֹּ֛את בָּאֵ֖שׁ וּשְׂרָפ֑וּהָ וְאֵ֣ת הַבָּתִּ֡ים אֲשֶׁר֩ קִטְּר֨וּ עַל־ גַּגּֽוֹתֵיהֶ֜ם לַבַּ֗עַל וְהִסִּ֤כוּ נְסָכִים֙ לֵאלֹהִ֣ים אֲחֵרִ֔ים לְמַ֖עַן הַכְעִסֵֽנִי׃ (ט) כִּֽי־ הָי֨וּ בְנֵֽי־ יִשְׂרָאֵ֜ל וּבְנֵ֣י יְהוּדָ֗ה אַ֣ךְ עֹשִׂ֥ים הָרַ֛ע בְּעֵינַ֖י מִנְּעֻרֹֽתֵיהֶ֑ם כִּ֣י בְנֵֽי־ יִשְׂרָאֵ֗ל אַ֣ךְ מַכְעִסִ֥ים אֹתִ֛י בְּמַעֲשֵׂ֥ה יְדֵיהֶ֖ם נְאֻם־ יְהוָֽה׃ (י) כִּ֧י עַל־ אַפִּ֣י וְעַל־ חֲמָתִ֗י הָ֤יְתָה לִּי֙ הָעִ֣יר הַזֹּ֔את לְמִן־ הַיּוֹם֙ אֲשֶׁ֣ר בָּנ֣וּ אוֹתָ֔הּ וְעַ֖ד הַיּ֣וֹם הַזֶּ֑ה לַהֲסִירָ֖הּ מֵעַ֥ל פָּנָֽי׃ (יא) עַל֩ כָּל־ רָעַ֨ת בְּנֵֽי־ יִשְׂרָאֵ֜ל וּבְנֵ֣י יְהוּדָ֗ה אֲשֶׁ֤ר עָשׂוּ֙ לְהַכְעִסֵ֔נִי הֵ֤מָּה מַלְכֵיהֶם֙ שָֽׂרֵיהֶ֔ם כֹּהֲנֵיהֶ֖ם וּנְבִֽיאֵיהֶ֑ם וְאִ֣ישׁ יְהוּדָ֔ה וְיֹשְׁבֵ֖י יְרוּשָׁלִָֽם׃ (יב) וַיִּפְנ֥וּ אֵלַ֛י עֹ֖רֶף וְלֹ֣א פָנִ֑ים וְלַמֵּ֤ד אֹתָם֙ הַשְׁכֵּ֣ם וְלַמֵּ֔ד וְאֵינָ֥ם שֹׁמְעִ֖ים לָקַ֥חַת מוּסָֽר׃ (יג) וַיָּשִׂ֣ימוּ שִׁקּֽוּצֵיהֶ֗ם בַּבַּ֛יִת אֲשֶׁר־ נִקְרָֽא־ שְׁמִ֥י עָלָ֖יו לְטַמְּאֽוֹ׃ (יד) וַיִּבְנוּ֩ אֶת־ בָּמ֨וֹת הַבַּ֜עַל אֲשֶׁ֣ר ׀ בְּגֵ֣יא בֶן־ הִנֹּ֗ם לְ֠הַעֲבִיר אֶת־ בְּנֵיהֶ֣ם וְאֶת־ בְּנוֹתֵיהֶם֮ לַמֹּלֶךְ֒ אֲשֶׁ֣ר לֹֽא־ צִוִּיתִ֗ים וְלֹ֤א עָֽלְתָה֙ עַל־ לִבִּ֔י לַעֲשׂ֖וֹת הַתּוֹעֵבָ֣ה הַזֹּ֑את לְמַ֖עַן (החטי) הַחֲטִ֥יא אֶת־ יְהוּדָֽה׃ (טו) וְעַתָּ֕ה לָכֵ֛ן כֹּֽה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל אֶל־ הָעִ֨יר הַזֹּ֜את אֲשֶׁ֣ר ׀ אַתֶּ֣ם אֹמְרִ֗ים נִתְּנָה֙ בְּיַ֣ד מֶֽלֶךְ־ בָּבֶ֔ל בַּחֶ֖רֶב וּבָרָעָ֥ב וּבַדָּֽבֶר׃ (טז) הִנְנִ֤י מְקַבְּצָם֙ מִכָּל־ הָ֣אֲרָצ֔וֹת אֲשֶׁ֨ר הִדַּחְתִּ֥ים שָׁ֛ם בְּאַפִּ֥י וּבַחֲמָתִ֖י וּבְקֶ֣צֶף גָּד֑וֹל וַהֲשִֽׁבֹתִים֙ אֶל־ הַמָּק֣וֹם הַזֶּ֔ה וְהֹשַׁבְתִּ֖ים לָבֶֽטַח׃ (יז) וְהָ֥יוּ לִ֖י לְעָ֑ם וַאֲנִ֕י אֶהְיֶ֥ה לָהֶ֖ם לֵאלֹהִֽים׃ (יח) וְנָתַתִּ֨י לָהֶ֜ם לֵ֤ב אֶחָד֙ וְדֶ֣רֶךְ אֶחָ֔ד לְיִרְאָ֥ה אוֹתִ֖י כָּל־ הַיָּמִ֑ים לְט֣וֹב לָהֶ֔ם וְלִבְנֵיהֶ֖ם אַחֲרֵיהֶֽם׃ (יט) וְכָרַתִּ֤י לָהֶם֙ בְּרִ֣ית עוֹלָ֔ם אֲשֶׁ֤ר לֹֽא־ אָשׁוּב֙ מֵאַ֣חֲרֵיהֶ֔ם לְהֵיטִיבִ֖י אוֹתָ֑ם וְאֶת־ יִרְאָתִי֙ אֶתֵּ֣ן בִּלְבָבָ֔ם לְבִלְתִּ֖י ס֥וּר מֵעָלָֽי׃ (כ) וְשַׂשְׂתִּ֥י עֲלֵיהֶ֖ם לְהֵטִ֣יב אוֹתָ֑ם וּנְטַעְתִּ֞ים בָּאָ֤רֶץ הַזֹּאת֙ בֶּאֱמֶ֔ת בְּכָל־ לִבִּ֖י וּבְכָל־ נַפְשִֽׁי׃ (כא) כִּֽי־ כֹה֙ אָמַ֣ר יְהוָ֔ה כַּאֲשֶׁ֤ר הֵבֵ֙אתִי֙ אֶל־ הָעָ֣ם הַזֶּ֔ה אֵ֛ת כָּל־ הָרָעָ֥ה הַגְּדוֹלָ֖ה הַזֹּ֑את כֵּ֣ן אָנֹכִ֞י מֵבִ֤יא עֲלֵיהֶם֙ אֶת־ כָּל־ הַטּוֹבָ֔ה אֲשֶׁ֥ר אָנֹכִ֖י דֹּבֵ֥ר עֲלֵיהֶֽם׃ (כב) וְנִקְנָ֥ה הַשָּׂדֶ֖ה בָּאָ֣רֶץ הַזֹּ֑את אֲשֶׁ֣ר ׀ אַתֶּ֣ם אֹמְרִ֗ים שְׁמָמָ֥ה הִיא֙ מֵאֵ֤ין אָדָם֙ וּבְהֵמָ֔ה נִתְּנָ֖ה בְּיַ֥ד הַכַּשְׂדִּֽים׃ (כג) שָׂד֞וֹת בַּכֶּ֣סֶף יִקְנ֗וּ וְכָת֨וֹב בַּסֵּ֥פֶר ׀ וְחָתוֹם֮ וְהָעֵ֣ד עֵדִים֒ בְּאֶ֨רֶץ בִּנְיָמִ֜ן וּבִסְבִיבֵ֣י יְרוּשָׁלִַ֗ם וּבְעָרֵ֤י יְהוּדָה֙ וּבְעָרֵ֣י הָהָ֔ר וּבְעָרֵ֥י הַשְּׁפֵלָ֖ה וּבְעָרֵ֣י הַנֶּ֑גֶב כִּֽי־ אָשִׁ֥יב אֶת־ שְׁבוּתָ֖ם נְאֻם־ יְהוָֽה׃ תחילת פרק לג (כד) וַיְהִ֧י דְבַר־ יְהוָ֛ה אֶֽל־ יִרְמְיָ֖הוּ שֵׁנִ֑ית וְהוּא֙ עוֹדֶ֣נּוּ עָצ֔וּר בַּחֲצַ֥ר הַמַּטָּרָ֖ה לֵאמֹֽר׃ (כה) כֹּֽה־ אָמַ֥ר יְהוָ֖ה עֹשָׂ֑הּ יְהוָ֗ה יוֹצֵ֥ר אוֹתָ֛הּ לַהֲכִינָ֖הּ יְהוָ֥ה שְׁמֽוֹ׃ (כו) קְרָ֥א אֵלַ֖י וְאֶעֱנֶ֑ךָּ וְאַגִּ֧ידָה לְּךָ֛ גְּדֹל֥וֹת וּבְצֻר֖וֹת לֹ֥א יְדַעְתָּֽם׃ (כז) כִּי֩ כֹ֨ה אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל עַל־ בָּתֵּי֙ הָעִ֣יר הַזֹּ֔את וְעַל־ בָּתֵּ֖י מַלְכֵ֣י יְהוּדָ֑ה הַנְּתֻצִ֕ים אֶל־ הַסֹּלְל֖וֹת וְאֶל־ הֶחָֽרֶב׃ (כח) בָּאִ֗ים לְהִלָּחֵם֙ אֶת־ הַכַּשְׂדִּ֔ים וּלְמַלְאָם֙ אֶת־ פִּגְרֵ֣י הָאָדָ֔ם אֲשֶׁר־ הִכֵּ֥יתִי בְאַפִּ֖י וּבַחֲמָתִ֑י וַאֲשֶׁ֨ר הִסְתַּ֤רְתִּי פָנַי֙ מֵהָעִ֣יר הַזֹּ֔את עַ֖ל כָּל־ רָעָתָֽם׃ (כט) הִנְנִ֧י מַעֲלֶה־ לָּ֛הּ אֲרֻכָ֥ה וּמַרְפֵּ֖א וּרְפָאתִ֑ים וְגִלֵּיתִ֣י לָהֶ֔ם עֲתֶ֥רֶת שָׁל֖וֹם וֶאֱמֶֽת׃ (ל) וַהֲשִֽׁבֹתִי֙ אֶת־ שְׁב֣וּת יְהוּדָ֔ה וְאֵ֖ת שְׁב֣וּת יִשְׂרָאֵ֑ל וּבְנִתִ֖ים כְּבָרִֽאשֹׁנָֽה׃ (לא) וְטִ֣הַרְתִּ֔ים מִכָּל־ עֲוֺנָ֖ם אֲשֶׁ֣ר חָֽטְאוּ־ לִ֑י וְסָלַחְתִּ֗י (לכול)־ לְכָל־ עֲוֺנֽוֹתֵיהֶם֙ אֲשֶׁ֣ר חָֽטְאוּ־ לִ֔י וַאֲשֶׁ֖ר פָּ֥שְׁעוּ בִֽי׃ (לב) וְהָ֣יְתָה לִּ֗י לְשֵׁ֤ם שָׂשׂוֹן֙ לִתְהִלָּ֣ה וּלְתִפְאֶ֔רֶת לְכֹ֖ל גּוֹיֵ֣י הָאָ֑רֶץ אֲשֶׁ֨ר יִשְׁמְע֜וּ אֶת־ כָּל־ הַטּוֹבָ֗ה אֲשֶׁ֤ר אָֽנֹכִי֙ עֹשֶׂ֣ה אֹתָ֔ם וּפָחֲד֣וּ וְרָֽגְז֗וּ עַ֤ל כָּל־ הַטּוֹבָה֙ וְעַ֣ל כָּל־ הַשָּׁל֔וֹם אֲשֶׁ֥ר אָֽנֹכִ֖י עֹ֥שֶׂה לָּֽהּ׃ (לג) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה עוֹד֮ יִשָּׁמַ֣ע בַּמָּקוֹם־ הַזֶּה֒ אֲשֶׁר֙ אַתֶּ֣ם אֹֽמְרִ֔ים חָרֵ֣ב ה֔וּא מֵאֵ֥ין אָדָ֖ם וּמֵאֵ֣ין בְּהֵמָ֑ה בְּעָרֵ֤י יְהוּדָה֙ וּבְחֻצ֣וֹת יְרוּשָׁלִַ֔ם הַֽנְשַׁמּ֗וֹת מֵאֵ֥ין אָדָ֛ם וּמֵאֵ֥ין יוֹשֵׁ֖ב וּמֵאֵ֥ין בְּהֵמָֽה׃ (לד) ק֣וֹל שָׂשׂ֞וֹן וְק֣וֹל שִׂמְחָ֗ה ק֣וֹל חָתָן֮ וְק֣וֹל כַּלָּה֒ ק֣וֹל אֹמְרִ֡ים הוֹדוּ֩ אֶת־ יְהוָ֨ה צְבָא֜וֹת כִּֽי־ ט֤וֹב יְהוָה֙ כִּֽי־ לְעוֹלָ֣ם חַסְדּ֔וֹ מְבִאִ֥ים תּוֹדָ֖ה בֵּ֣ית יְהוָ֑ה כִּֽי־ אָשִׁ֧יב אֶת־ שְׁבוּת־ הָאָ֛רֶץ כְּבָרִאשֹׁנָ֖ה אָמַ֥ר יְהוָֽה׃ (לה) כֹּֽה־ אָמַר֮ יְהוָ֣ה צְבָאוֹת֒ ע֞וֹד שִׁלַּחְתֶּ֥ם חָפְשִׁ֖ים לְנַפְשָׁ֑ם וַתִּכְבְּשׁ֣וּ אֹתָ֔ם לִֽהְי֣וֹת לָכֶ֔ם לַעֲבָדִ֖ים וְלִשְׁפָחֽוֹת׃ (כח) לָכֵן֮ כֹּה־ אָמַ֣ר יְהוָה֒ אַתֶּם֙ לֹֽא־ שְׁמַעְתֶּ֣ם אֵלַ֔י לִקְרֹ֣א דְר֔וֹר אִ֥ישׁ לְאָחִ֖יו וְאִ֣ישׁ לְרֵעֵ֑הוּ הִנְנִ֣י קֹרֵא֩ לָכֶ֨ם דְּר֜וֹר נְאֻם־ יְהוָ֗ה אֶל־ הַחֶ֙רֶב֙ אֶל־ הַדֶּ֣בֶר וְאֶל־ הָרָעָ֔ב וְנָתַתִּ֤י אֶתְכֶם֙ (לזועה) לְזַעֲוָ֔ה לְכֹ֖ל מַמְלְכ֥וֹת הָאָֽרֶץ׃ (כט) וְנָתַתִּ֣י אֶת־ הָאֲנָשִׁ֗ים הָעֹֽבְרִים֙ אֶת־ בְּרִתִ֔י אֲשֶׁ֤ר לֹֽא־ הֵקִ֙ימוּ֙ אֶת־ דִּבְרֵ֣י הַבְּרִ֔ית אֲשֶׁ֥ר כָּרְת֖וּ לְפָנָ֑י הָעֵ֙גֶל֙ אֲשֶׁ֣ר כָּרְת֣וּ לִשְׁנַ֔יִם וַיַּעַבְר֖וּ בֵּ֥ין בְּתָרָֽיו׃ (ל) שָׂרֵ֨י יְהוּדָ֜ה וְשָׂרֵ֣י יְרוּשָׁלִַ֗ם הַסָּֽרִסִים֙ וְהַכֹּ֣הֲנִ֔ים וְכֹ֖ל עַ֣ם הָאָ֑רֶץ הָעֹ֣בְרִ֔ים בֵּ֖ין בִּתְרֵ֥י הָעֵֽגֶל׃ (לא) וְנָתַתִּ֤י אוֹתָם֙ בְּיַ֣ד אֹֽיְבֵיהֶ֔ם וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשָׁ֑ם וְהָיְתָ֤ה נִבְלָתָם֙ לְמַֽאֲכָ֔ל לְע֥וֹף הַשָּׁמַ֖יִם וּלְבֶהֱמַ֥ת הָאָֽרֶץ׃ (לב) וְאֶת־ צִדְקִיָּ֨הוּ מֶֽלֶךְ־ יְהוּדָ֜ה וְאֶת־ שָׂרָ֗יו אֶתֵּן֙ בְּיַ֣ד אֹֽיְבֵיהֶ֔ם וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשָׁ֑ם וּבְיַד חֵ֚יל מֶ֣לֶךְ בָּבֶ֔ל הָעֹלִ֖ים מֵעֲלֵיכֶֽם׃ (לג) הִנְנִ֨י מְצַוֶּ֜ה נְאֻם־ יְהוָ֗ה וַהֲשִׁ֨בֹתִ֜ים אֶל־ הָעִ֤יר הַזֹּאת֙ וְנִלְחֲמ֣וּ עָלֶ֔יהָ וּלְכָד֖וּהָ וּשְׂרָפֻ֣הָ בָאֵ֑שׁ וְאֶת־ עָרֵ֧י יְהוּדָ֛ה אֶתֵּ֥ן שְׁמָמָ֖ה מֵאֵ֥ין יֹשֵֽׁב׃ (לד) הַדָּבָ֛ר אֲשֶׁר־ הָיָ֥ה אֶֽל־ יִרְמְיָ֖הוּ מֵאֵ֣ת יְהוָ֑ה בִּימֵ֨י יְהוֹיָקִ֧ים בֶּן־ יֹאשִׁיָּ֛הוּ מֶ֥לֶךְ יְהוּדָ֖ה לֵאמֹֽר׃ (לה) הָלוֹךְ֮ אֶל־ בֵּ֣ית הָרֵכָבִים֒ וְדִבַּרְתָּ֣ אוֹתָ֔ם וַהֲבִֽאוֹתָם֙ בֵּ֣ית יְהוָ֔ה אֶל־ אַחַ֖ת הַלְּשָׁכ֑וֹת וְהִשְׁקִיתָ֥ אוֹתָ֖ם יָֽיִן׃ (לו) וָאֶקַּ֞ח אֶת־ יַאֲזַנְיָ֤ה בֶֽן־ יִרְמְיָ֙הוּ֙ בֶּן־ חֲבַצִּנְיָ֔ה וְאֶת־ אֶחָ֖יו וְאֶת־ כָּל־ בָּנָ֑יו וְאֵ֖ת כָּל־ בֵּ֥ית הָרֵכָבִֽים׃ (לז) וָאָבִ֤א אֹתָם֙ בֵּ֣ית יְהוָ֔ה אֶל־ לִשְׁכַּ֗ת בְּנֵ֛י חָנָ֥ן בֶּן־ יִגְדַּלְיָ֖הוּ אִ֣ישׁ הָאֱלֹהִ֑ים אֲשֶׁר־ אֵ֙צֶל֙ לִשְׁכַּ֣ת הַשָּׂרִ֔ים אֲשֶׁ֣ר מִמַּ֗עַל לְלִשְׁכַּ֛ת מַעֲשֵׂיָ֥הוּ בֶן־ שַׁלֻּ֖ם שֹׁמֵ֥ר הַסַּֽף׃ (לח) וָאֶתֵּ֞ן לִפְנֵ֣י ׀ בְּנֵ֣י בֵית־ הָרֵכָבִ֗ים גְּבִעִ֛ים מְלֵאִ֥ים יַ֖יִן וְכֹס֑וֹת וָאֹמַ֥ר אֲלֵיהֶ֖ם שְׁתוּ־ יָֽיִן׃ (לט) וַיֹּאמְר֖וּ לֹ֣א נִשְׁתֶּה־ יָּ֑יִן כִּי֩ יוֹנָדָ֨ב בֶּן־ רֵכָ֜ב אָבִ֗ינוּ צִוָּ֤ה עָלֵ֙ינוּ֙ לֵאמֹ֔ר לֹ֧א תִשְׁתּוּ־ יַ֛יִן אַתֶּ֥ם וּבְנֵיכֶ֖ם עַד־ עוֹלָֽם׃ (מ) וּבַ֣יִת לֹֽא־ תִבְנ֗וּ וְזֶ֤רַע לֹֽא־ תִזְרָ֙עוּ֙ וְכֶ֣רֶם לֹֽא־ תִטָּ֔עוּ וְלֹ֥א יִֽהְיֶ֖ה לָכֶ֑ם כִּ֠י בָּאֳהָלִ֤ים תֵּֽשְׁבוּ֙ כָּל־ יְמֵיכֶ֔ם לְמַ֨עַן תִּֽחְי֜וּ יָמִ֤ים רַבִּים֙ עַל־ פְּנֵ֣י הָאֲדָמָ֔ה אֲשֶׁ֥ר אַתֶּ֖ם גָּרִ֥ים שָֽׁם׃ (מא) וַנִּשְׁמַ֗ע בְּק֨וֹל יְהוֹנָדָ֤ב בֶּן־ רֵכָב֙ אָבִ֔ינוּ לְכֹ֖ל אֲשֶׁ֣ר צִוָּ֑נוּ לְבִלְתִּ֤י שְׁתֽוֹת־ יַ֙יִן֙ כָּל־ יָמֵ֔ינוּ אֲנַ֣חְנוּ נָשֵׁ֔ינוּ בָּנֵ֖ינוּ וּבְנֹתֵֽינוּ׃ (מב) וּלְבִלְתִּ֛י בְּנ֥וֹת בָּתִּ֖ים לְשִׁבְתֵּ֑נוּ וְכֶ֧רֶם וְשָׂדֶ֛ה וָזֶ֖רַע לֹ֥א יִֽהְיֶה־ לָּֽנוּ׃
