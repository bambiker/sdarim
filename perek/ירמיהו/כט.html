ירמיהו פרק כט לפי החלוקה לסדרים

פרק כט בספר ירמיהו (32 פסוקים) נמצא בירמיהו סדר טו (פסוקים לו–מא) ובירמיהו סדר טז (פסוקים א–כו).

ירמיהו סדר טו

(א) אָנֹכִ֞י עָשִׂ֣יתִי אֶת־ הָאָ֗רֶץ אֶת־ הָאָדָ֤ם וְאֶת־ הַבְּהֵמָה֙ אֲשֶׁר֙ עַל־ פְּנֵ֣י הָאָ֔רֶץ בְּכֹחִי֙ הַגָּד֔וֹל וּבִזְרוֹעִ֖י הַנְּטוּיָ֑ה וּנְתַתִּ֕יהָ לַאֲשֶׁ֖ר יָשַׁ֥ר בְּעֵינָֽי׃ (ב) וְעַתָּ֗ה אָֽנֹכִי֙ נָתַ֙תִּי֙ אֶת־ כָּל־ הָאֲרָצ֣וֹת הָאֵ֔לֶּה בְּיַ֛ד נְבוּכַדְנֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל עַבְדִּ֑י וְגַם֙ אֶת־ חַיַּ֣ת הַשָּׂדֶ֔ה נָתַ֥תִּי ל֖וֹ לְעָבְדֽוֹ׃ (ג) וְעָבְד֤וּ אֹתוֹ֙ כָּל־ הַגּוֹיִ֔ם וְאֶת־ בְּנ֖וֹ וְאֶֽת־ בֶּן־ בְּנ֑וֹ עַ֣ד בֹּא־ עֵ֤ת אַרְצוֹ֙ גַּם־ ה֔וּא וְעָ֤בְדוּ בוֹ֙ גּוֹיִ֣ם רַבִּ֔ים וּמְלָכִ֖ים גְּדֹלִֽים׃ (ד) וְהָיָ֨ה הַגּ֜וֹי וְהַמַּמְלָכָ֗ה אֲשֶׁ֨ר לֹֽא־ יַעַבְד֤וּ אֹתוֹ֙ אֶת־ נְבוּכַדְנֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֔ל וְאֵ֨ת אֲשֶׁ֤ר לֹֽא־ יִתֵּן֙ אֶת־ צַוָּאר֔וֹ בְּעֹ֖ל מֶ֣לֶךְ בָּבֶ֑ל בַּחֶרֶב֩ וּבָרָעָ֨ב וּבַדֶּ֜בֶר אֶפְקֹ֨ד עַל־ הַגּ֤וֹי הַהוּא֙ נְאֻם־ יְהוָ֔ה עַד־ תֻּמִּ֥י אֹתָ֖ם בְּיָדֽוֹ׃ (ה) וְ֠אַתֶּם אַל־ תִּשְׁמְע֨וּ אֶל־ נְבִיאֵיכֶ֜ם וְאֶל־ קֹֽסְמֵיכֶ֗ם וְאֶל֙ חֲלֹמֹ֣תֵיכֶ֔ם וְאֶל־ עֹֽנְנֵיכֶ֖ם וְאֶל־ כַּשָּׁפֵיכֶ֑ם אֲשֶׁר־ הֵ֞ם אֹמְרִ֤ים אֲלֵיכֶם֙ לֵאמֹ֔ר לֹ֥א תַעַבְד֖וּ אֶת־ מֶ֥לֶךְ בָּבֶֽל׃ (ו) כִּ֣י שֶׁ֔קֶר הֵ֖ם נִבְּאִ֣ים לָכֶ֑ם לְמַ֨עַן הַרְחִ֤יק אֶתְכֶם֙ מֵעַ֣ל אַדְמַתְכֶ֔ם וְהִדַּחְתִּ֥י אֶתְכֶ֖ם וַאֲבַדְתֶּֽם׃ (ז) וְהַגּ֗וֹי אֲשֶׁ֨ר יָבִ֧יא אֶת־ צַוָּאר֛וֹ בְּעֹ֥ל מֶֽלֶךְ־ בָּבֶ֖ל וַֽעֲבָד֑וֹ וְהִנַּחְתִּ֤יו עַל־ אַדְמָתוֹ֙ נְאֻם־ יְהוָ֔ה וַֽעֲבָדָ֖הּ וְיָ֥שַׁב בָּֽהּ׃ (ח) וְאֶל־ צִדְקִיָּ֤ה מֶֽלֶךְ־ יְהוּדָה֙ דִּבַּ֔רְתִּי כְּכָל־ הַדְּבָרִ֥ים הָאֵ֖לֶּה לֵאמֹ֑ר הָבִ֨יאוּ אֶת־ צַוְּארֵיכֶ֜ם בְּעֹ֣ל מֶֽלֶךְ־ בָּבֶ֗ל וְעִבְד֥וּ אֹת֛וֹ וְעַמּ֖וֹ וִֽחְיֽוּ׃ (ט) לָ֤מָּה תָמ֙וּתוּ֙ אַתָּ֣ה וְעַמֶּ֔ךָ בַּחֶ֖רֶב בָּרָעָ֣ב וּבַדָּ֑בֶר֙ כַּֽאֲשֶׁר֙ דִּבֶּ֣ר יְהוָ֔ה אֶל־ הַגּ֕וֹי אֲשֶׁ֥ר לֹֽא־ יַעֲבֹ֖ד אֶת־ מֶ֥לֶךְ בָּבֶֽל׃ (י) וְאַֽל־ תִּשְׁמְע֞וּ אֶל־ דִּבְרֵ֣י הַנְּבִאִ֗ים הָאֹמְרִ֤ים אֲלֵיכֶם֙ לֵאמֹ֔ר לֹ֥א תַעַבְד֖וּ אֶת־ מֶ֣לֶךְ בָּבֶ֑ל כִּ֣י שֶׁ֔קֶר הֵ֖ם נִבְּאִ֥ים לָכֶֽם׃ (יא) כִּ֣י לֹ֤א שְׁלַחְתִּים֙ נְאֻם־ יְהוָ֔ה וְהֵ֛ם נִבְּאִ֥ים בִּשְׁמִ֖י לַשָּׁ֑קֶר לְמַ֨עַן הַדִּיחִ֤י אֶתְכֶם֙ וַאֲבַדְתֶּ֔ם אַתֶּ֕ם וְהַנְּבִאִ֖ים הַֽנִּבְּאִ֥ים לָכֶֽם׃ (יב) וְאֶל־ הַכֹּהֲנִים֩ וְאֶל־ כָּל־ הָעָ֨ם הַזֶּ֜ה דִּבַּ֣רְתִּי לֵאמֹ֗ר כֹּה֮ אָמַ֣ר יְהוָה֒ אַֽל־ תִּשְׁמְע֞וּ אֶל־ דִּבְרֵ֣י נְבִֽיאֵיכֶ֗ם הַֽנִּבְּאִ֤ים לָכֶם֙ לֵאמֹ֔ר הִנֵּ֨ה כְלֵ֧י בֵית־ יְהוָ֛ה מוּשָׁבִ֥ים מִבָּבֶ֖לָה עַתָּ֣ה מְהֵרָ֑ה כִּ֣י שֶׁ֔קֶר הֵ֖מָּה נִבְּאִ֥ים לָכֶֽם׃ (יג) אַל־ תִּשְׁמְע֣וּ אֲלֵיהֶ֔ם עִבְד֥וּ אֶת־ מֶֽלֶךְ־ בָּבֶ֖ל וִֽחְי֑וּ לָ֧מָּה תִֽהְיֶ֛ה הָעִ֥יר הַזֹּ֖את חָרְבָּֽה׃ (יד) וְאִם־ נְבִאִ֣ים הֵ֔ם וְאִם־ יֵ֥שׁ דְּבַר־ יְהוָ֖ה אִתָּ֑ם יִפְגְּעוּ־ נָא֙ בַּֽיהוָ֣ה צְבָא֔וֹת לְבִלְתִּי־ בֹ֜אוּ הַכֵּלִ֣ים ׀ הַנּוֹתָרִ֣ים בְּבֵית־ יְהוָ֗ה וּבֵ֨ית מֶ֧לֶךְ יְהוּדָ֛ה וּבִירוּשָׁלִַ֖ם בָּבֶֽלָה׃ (טו) כִּ֣י כֹ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת אֶל־ הָֽעַמֻּדִ֔ים וְעַל־ הַיָּ֖ם וְעַל־ הַמְּכֹנ֑וֹת וְעַל֙ יֶ֣תֶר הַכֵּלִ֔ים הַנּוֹתָרִ֖ים בָּעִ֥יר הַזֹּֽאת׃ (טז) אֲשֶׁ֣ר לֹֽא־ לְקָחָ֗ם נְבֽוּכַדְנֶאצַּר֙ מֶ֣לֶךְ בָּבֶ֔ל בַּ֠גְלוֹתוֹ אֶת־ יְכָונְיָ֨ה בֶן־ יְהוֹיָקִ֧ים מֶֽלֶךְ־ יְהוּדָ֛ה מִירֽוּשָׁלִַ֖ם בָּבֶ֑לָה וְאֵ֛ת כָּל־ חֹרֵ֥י יְהוּדָ֖ה וִירוּשָׁלִָֽם׃ (יז) כִּ֣י כֹ֥ה אָמַ֛ר יְהוָ֥ה צְבָא֖וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֑ל עַל־ הַכֵּלִ֗ים הַנּֽוֹתָרִים֙ בֵּ֣ית יְהוָ֔ה וּבֵ֥ית מֶֽלֶךְ־ יְהוּדָ֖ה וִירוּשָׁלִָֽם׃ (יח) בָּבֶ֥לָה יוּבָ֖אוּ וְשָׁ֣מָּה יִֽהְי֑וּ עַ֠ד י֣וֹם פָּקְדִ֤י אֹתָם֙ נְאֻם־ יְהוָ֔ה וְהַֽעֲלִיתִים֙ וַהֲשִׁ֣יבֹתִ֔ים אֶל־ הַמָּק֖וֹם הַזֶּֽה׃ (יט) וַיְהִ֣י ׀ בַּשָּׁנָ֣ה הַהִ֗יא בְּרֵאשִׁית֙ מַמְלֶ֙כֶת֙ צִדְקִיָּ֣ה מֶֽלֶךְ־ יְהוּדָ֔ה (בשנת) בַּשָּׁנָה֙ הָֽרְבִעִ֔ית בַּחֹ֖דֶשׁ הַחֲמִישִׁ֑י אָמַ֣ר אֵלַ֡י חֲנַנְיָה֩ בֶן־ עַזּ֨וּר הַנָּבִ֜יא אֲשֶׁ֤ר מִגִּבְעוֹן֙ בְּבֵ֣ית יְהוָ֔ה לְעֵינֵ֧י הַכֹּהֲנִ֛ים וְכָל־ הָעָ֖ם לֵאמֹֽר׃ (כ) כֹּֽה־ אָמַ֞ר יְהוָ֧ה צְבָא֛וֹת אֱלֹהֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֑ר שָׁבַ֞רְתִּי אֶת־ עֹ֖ל מֶ֥לֶךְ בָּבֶֽל׃ (כא) בְּע֣וֹד ׀ שְׁנָתַ֣יִם יָמִ֗ים אֲנִ֤י מֵשִׁיב֙ אֶל־ הַמָּק֣וֹם הַזֶּ֔ה אֶֽת־ כָּל־ כְּלֵ֖י בֵּ֣ית יְהוָ֑ה אֲשֶׁ֨ר לָקַ֜ח נְבוּכַדנֶאצַּ֤ר מֶֽלֶךְ־ בָּבֶל֙ מִן־ הַמָּק֣וֹם הַזֶּ֔ה וַיְבִיאֵ֖ם בָּבֶֽל׃ (כב) וְאֶת־ יְכָנְיָ֣ה בֶן־ יְהוֹיָקִ֣ים מֶֽלֶךְ־ יְ֠הוּדָה וְאֶת־ כָּל־ גָּל֨וּת יְהוּדָ֜ה הַבָּאִ֣ים בָּבֶ֗לָה אֲנִ֥י מֵשִׁ֛יב אֶל־ הַמָּק֥וֹם הַזֶּ֖ה נְאֻם־ יְהוָ֑ה כִּ֣י אֶשְׁבֹּ֔ר אֶת־ עֹ֖ל מֶ֥לֶךְ בָּבֶֽל׃ (כג) וַיֹּ֙אמֶר֙ יִרְמְיָ֣ה הַנָּבִ֔יא אֶל־ חֲנַנְיָ֖ה הַנָּבִ֑יא לְעֵינֵ֤י הַכֹּֽהֲנִים֙ וּלְעֵינֵ֣י כָל־ הָעָ֔ם הָעֹמְדִ֖ים בְּבֵ֥ית יְהוָֽה׃ (כד) וַיֹּ֙אמֶר֙ יִרְמְיָ֣ה הַנָּבִ֔יא אָמֵ֕ן כֵּ֖ן יַעֲשֶׂ֣ה יְהוָ֑ה יָקֵ֤ם יְהוָה֙ אֶת־ דְּבָרֶ֔יךָ אֲשֶׁ֣ר נִבֵּ֗אתָ לְהָשִׁ֞יב כְּלֵ֤י בֵית־ יְהוָה֙ וְכָל־ הַגּוֹלָ֔ה מִבָּבֶ֖ל אֶל־ הַמָּק֥וֹם הַזֶּֽה׃ (כה) אַךְ־ שְׁמַֽע־ נָא֙ הַדָּבָ֣ר הַזֶּ֔ה אֲשֶׁ֥ר אָנֹכִ֖י דֹּבֵ֣ר בְּאָזְנֶ֑יךָ וּבְאָזְנֵ֖י כָּל־ הָעָֽם׃ (כו) הַנְּבִיאִ֗ים אֲשֶׁ֨ר הָי֧וּ לְפָנַ֛י וּלְפָנֶ֖יךָ מִן־ הָֽעוֹלָ֑ם וַיִּנָּ֨בְא֜וּ אֶל־ אֲרָצ֤וֹת רַבּוֹת֙ וְעַל־ מַמְלָכ֣וֹת גְּדֹל֔וֹת לְמִלְחָמָ֖ה וּלְרָעָ֥ה וּלְדָֽבֶר׃ (כז) הַנָּבִ֕יא אֲשֶׁ֥ר יִנָּבֵ֖א לְשָׁל֑וֹם בְּבֹא֙ דְּבַ֣ר הַנָּבִ֗יא יִוָּדַע֙ הַנָּבִ֔יא אֲשֶׁר־ שְׁלָח֥וֹ יְהוָ֖ה בֶּאֱמֶֽת׃ (כח) וַיִּקַּ֞ח חֲנַנְיָ֤ה הַנָּבִיא֙ אֶת־ הַמּוֹטָ֔ה מֵעַ֕ל צַוַּ֖אר יִרְמְיָ֣ה הַנָּבִ֑יא וַֽיִּשְׁבְּרֵֽהוּ׃ (כט) וַיֹּ֣אמֶר חֲנַנְיָה֩ לְעֵינֵ֨י כָל־ הָעָ֜ם לֵאמֹ֗ר כֹּה֮ אָמַ֣ר יְהוָה֒ כָּ֣כָה אֶשְׁבֹּ֞ר אֶת־ עֹ֣ל ׀ נְבֻֽכַדְנֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֗ל בְּעוֹד֙ שְׁנָתַ֣יִם יָמִ֔ים מֵעַ֕ל צַוַּ֖אר כָּל־ הַגּוֹיִ֑ם וַיֵּ֛לֶךְ יִרְמְיָ֥ה הַנָּבִ֖יא לְדַרְכּֽוֹ׃ (ל) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֶֽל־ יִרְמְיָ֑ה אַ֠חֲרֵי שְׁב֞וֹר חֲנַנְיָ֤ה הַנָּבִיא֙ אֶת־ הַמּוֹטָ֔ה מֵעַ֗ל צַוַּ֛אר יִרְמְיָ֥ה הַנָּבִ֖יא לֵאמֹֽר׃ (לא) הָלוֹךְ֩ וְאָמַרְתָּ֨ אֶל־ חֲנַנְיָ֜ה לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה מוֹטֹ֥ת עֵ֖ץ שָׁבָ֑רְתָּ וְעָשִׂ֥יתָ תַחְתֵּיהֶ֖ן מֹט֥וֹת בַּרְזֶֽל׃ (לב) כִּ֣י כֹֽה־ אָמַר֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל עֹ֣ל בַּרְזֶ֡ל נָתַ֜תִּי עַל־ צַוַּ֣אר ׀ כָּל־ הַגּוֹיִ֣ם הָאֵ֗לֶּה לַעֲבֹ֛ד אֶת־ נְבֻכַדְנֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אַל־ יַשִּׁ֧יאוּ לָכֶ֛ם נְבִֽיאֵיכֶ֥ם אֲשֶׁר־ בְּקִרְבְּכֶ֖ם וְקֹֽסְמֵיכֶ֑ם וְאַֽל־ תִּשְׁמְעוּ֙ אֶל־ חֲלֹמֹ֣תֵיכֶ֔ם אֲשֶׁ֥ר אַתֶּ֖ם מַחְלְמִֽים׃ (ג) כִּ֣י בְשֶׁ֔קֶר הֵ֛ם נִבְּאִ֥ים לָכֶ֖ם בִּשְׁמִ֑י לֹ֥א שְׁלַחְתִּ֖ים נְאֻם־ יְהוָֽה׃ (ד) כִּֽי־ כֹה֙ אָמַ֣ר יְהוָ֔ה כִּ֠י לְפִ֞י מְלֹ֧את לְבָבֶ֛ל שִׁבְעִ֥ים שָׁנָ֖ה אֶפְקֹ֣ד אֶתְכֶ֑ם וַהֲקִמֹתִ֤י עֲלֵיכֶם֙ אֶת־ בְּכָל־ לְבַבְכֶֽם׃ (ח) וְנִמְצֵ֣אתִי לָכֶם֮ נְאֻם־ יְהוָה֒ וְשַׁבְתִּ֣י אֶת־ (שביתכם) שְׁבוּתְכֶ֗ם וְקִבַּצְתִּ֣י אֶ֠תְכֶם מִֽכָּל־ הַגּוֹיִ֞ם וּמִכָּל־ הַמְּקוֹמ֗וֹת אֲשֶׁ֨ר הִדַּ֧חְתִּי אֶתְכֶ֛ם בְּנ֤וּ בָתִּים֙ וְשֵׁ֔בוּ וְנִטְע֣וּ גַנּ֔וֹת וְאִכְל֖וּ אֶת־ פְּרִיהֶֽן׃ (כג) וַיִּקְרָ֛א צְפַנְיָ֥ה הַכֹּהֵ֖ן אֶת־ הַסֵּ֣פֶר הַזֶּ֑ה בְּאָזְנֵ֖י יִרְמְיָ֥הוּ הַנָּבִֽיא׃ (כד) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶֽל־ יִרְמְיָ֖הוּ לֵאמֹֽר׃ (כה) שְׁלַ֤ח עַל־ כָּל־ הַגּוֹלָה֙ לֵאמֹ֔ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה אֶל־ שְׁמַעְיָ֖ה הַנֶּחֱלָמִ֑י יַ֡עַן אֲשֶׁר֩ נִבָּ֨א לָכֶ֜ם שְׁמַעְיָ֗ה וַֽאֲנִי֙ לֹ֣א שְׁלַחְתִּ֔יו וַיַּבְטַ֥ח אֶתְכֶ֖ם עַל־ שָֽׁקֶר׃ (כו) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה הִנְנִ֨י פֹקֵ֜ד עַל־ שְׁמַעְיָ֣ה הַנֶּחֱלָמִי֮ וְעַל־ זַרְעוֹ֒ לֹא־ יִהְיֶ֨ה ל֜וֹ אִ֣ישׁ ׀ יוֹשֵׁ֣ב ׀ בְּתוֹךְ־ הָעָ֣ם הַזֶּ֗ה וְלֹֽא־ יִרְאֶ֥ה בַטּ֛וֹב אֲשֶׁר־ אֲנִ֥י עֹשֶֽׂה־ לְעַמִּ֖י נְאֻם־ יְהוָ֑ה כִּֽי־ סָרָ֥ה דִבֶּ֖ר עַל־ יְהוָֽה׃ סוף פרק כט (כז) הַדָּבָר֙ אֲשֶׁ֣ר הָיָ֣ה אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (כח) כֹּֽה־ אָמַ֧ר יְהוָ֛ה אֱלֹהֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֑ר כְּתָב־ לְךָ֗ אֵ֧ת כָּל־ הַדְּבָרִ֛ים אֲשֶׁר־ דִּבַּ֥רְתִּי אֵלֶ֖יךָ אֶל־ סֵֽפֶר׃ (כט) כִּ֠י הִנֵּ֨ה יָמִ֤ים בָּאִים֙ נְאֻם־ יְהוָ֔ה וְ֠שַׁבְתִּי אֶת־ שְׁב֨וּת עַמִּ֧י יִשְׂרָאֵ֛ל וִֽיהוּדָ֖ה אָמַ֣ר יְהוָ֑ה וַהֲשִׁבֹתִ֗ים אֶל־ הָאָ֛רֶץ אֲשֶׁר־ נָתַ֥תִּי לַאֲבוֹתָ֖ם וִֽירֵשֽׁוּהָ׃ (ל) וְאֵ֣לֶּה הַדְּבָרִ֗ים אֲשֶׁ֨ר דִּבֶּ֧ר יְהוָ֛ה אֶל־ יִשְׂרָאֵ֖ל וְאֶל־ יְהוּדָֽה׃ (לא) כִּי־ כֹה֙ אָמַ֣ר יְהוָ֔ה ק֥וֹל חֲרָדָ֖ה שָׁמָ֑עְנוּ פַּ֖חַד וְאֵ֥ין שָׁלֽוֹם׃ (לב) שַׁאֲלוּ־ נָ֣א וּרְא֔וּ אִם־ יֹלֵ֖ד זָכָ֑ר מַדּוּעַ֩ רָאִ֨יתִי כָל־ גֶּ֜בֶר יָדָ֤יו עַל־ חֲלָצָיו֙ כַּיּ֣וֹלֵדָ֔ה וְנֶהֶפְכ֥וּ כָל־ פָּנִ֖ים לְיֵרָקֽוֹן׃ (לג) ה֗וֹי כִּ֥י גָד֛וֹל הַיּ֥וֹם הַה֖וּא מֵאַ֣יִן כָּמֹ֑הוּ וְעֵֽת־ צָרָ֥ה הִיא֙ לְיַֽעֲקֹ֔ב וּמִמֶּ֖נָּה יִוָּשֵֽׁעַ׃ (לד) וְהָיָה֩ בַיּ֨וֹם הַה֜וּא נְאֻ֣ם ׀ יְהוָ֣ה צְבָא֗וֹת אֶשְׁבֹּ֤ר עֻלּוֹ֙ מֵעַ֣ל צַוָּארֶ֔ךָ וּמוֹסְרוֹתֶ֖יךָ אֲנַתֵּ֑ק וְלֹא־ יַעַבְדוּ־ ב֥וֹ ע֖וֹד זָרִֽים׃
