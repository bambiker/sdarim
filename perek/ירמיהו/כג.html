ירמיהו פרק כג לפי החלוקה לסדרים

פרק כג בספר ירמיהו (40 פסוקים) נמצא בירמיהו סדר יא (פסוקים נא–נה) ובירמיהו סדר יב (פסוקים א–לה).

ירמיהו סדר יא

(א) שִׁ֚ירוּ לַֽיהוָ֔ה הַֽלְל֖וּ אֶת־ יְהוָ֑ה כִּ֥י הִצִּ֛יל אֶת־ נֶ֥פֶשׁ אֶבְי֖וֹן מִיַּ֥ד מְרֵעִֽים׃ (ב) אָר֣וּר הַיּ֔וֹם אֲשֶׁ֥ר יֻלַּ֖דְתִּי בּ֑וֹ י֛וֹם אֲשֶׁר־ יְלָדַ֥תְנִי אִמִּ֖י אַל־ יְהִ֥י בָרֽוּךְ׃ (ג) אָר֣וּר הָאִ֗ישׁ אֲשֶׁ֨ר בִּשַּׂ֤ר אֶת־ אָבִי֙ לֵאמֹ֔ר יֻֽלַּד־ לְךָ֖ בֵּ֣ן זָכָ֑ר שַׂמֵּ֖חַ שִׂמֳּחָֽהוּ׃ (ד) וְהָיָה֙ הָאִ֣ישׁ הַה֔וּא כֶּֽעָרִ֛ים אֲשֶׁר־ הָפַ֥ךְ יְהוָ֖ה וְלֹ֣א נִחָ֑ם וְשָׁמַ֤ע זְעָקָה֙ בַּבֹּ֔קֶר וּתְרוּעָ֖ה בְּעֵ֥ת צָהֳרָֽיִם׃ (ה) אֲשֶׁ֥ר לֹא־ מוֹתְתַ֖נִי מֵרָ֑חֶם וַתְּהִי־ לִ֤י אִמִּי֙ קִבְרִ֔י וְרַחְמָ֖הֿ הֲרַ֥ת עוֹלָֽם׃ (ו) לָ֤מָּה זֶּה֙ מֵרֶ֣חֶם יָצָ֔אתִי לִרְא֥וֹת עָמָ֖ל וְיָג֑וֹן וַיִּכְל֥וּ בְּבֹ֖שֶׁת יָמָֽי׃ (ז) הַדָּבָ֛ר אֲשֶׁר־ הָיָ֥ה אֶֽל־ יִרְמְיָ֖הוּ מֵאֵ֣ת יְהוָ֑ה בִּשְׁלֹ֨חַ אֵלָ֜יו הַמֶּ֣לֶךְ צִדְקִיָּ֗הוּ אֶת־ פַּשְׁחוּר֙ בֶּן־ מַלְכִּיָּ֔ה וְאֶת־ צְפַנְיָ֧ה בֶן־ מַעֲשֵׂיָ֛ה הַכֹּהֵ֖ן לֵאמֹֽר׃ (ח) דְּרָשׁ־ נָ֤א בַעֲדֵ֙נוּ֙ אֶת־ יְהוָ֔ה כִּ֛י נְבוּכַדְרֶאצַּ֥ר מֶֽלֶךְ־ בָּבֶ֖ל נִלְחָ֣ם עָלֵ֑ינוּ אוּלַי֩ יַעֲשֶׂ֨ה יְהוָ֤ה אוֹתָ֙נוּ֙ כְּכָל־ נִפְלְאֹתָ֔יו וְיַעֲלֶ֖ה מֵעָלֵֽינוּ׃ (ט) וַיֹּ֥אמֶר יִרְמְיָ֖הוּ אֲלֵיהֶ֑ם כֹּ֥ה תֹאמְרֻ֖ן אֶל־ צִדְקִיָּֽהוּ׃ (י) כֹּֽה־ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֣י יִשְׂרָאֵ֗ל הִנְנִ֣י מֵסֵב֮ אֶת־ כְּלֵ֣י הַמִּלְחָמָה֮ אֲשֶׁ֣ר בְּיֶדְכֶם֒ אֲשֶׁ֨ר אַתֶּ֜ם נִלְחָמִ֣ים בָּ֗ם אֶת־ מֶ֤לֶךְ בָּבֶל֙ וְאֶת־ הַכַּשְׂדִּ֔ים הַצָּרִ֣ים עֲלֵיכֶ֔ם מִח֖וּץ לַֽחוֹמָ֑ה וְאָסַפְתִּ֣י אוֹתָ֔ם אֶל־ תּ֖וֹךְ הָעִ֥יר הַזֹּֽאת׃ (יא) וְנִלְחַמְתִּ֤י אֲנִי֙ אִתְּכֶ֔ם בְּיָ֥ד נְטוּיָ֖ה וּבִזְר֣וֹעַ חֲזָקָ֑ה וּבְאַ֥ף וּבְחֵמָ֖ה וּבְקֶ֥צֶף גָּדֽוֹל׃ (יב) וְהִכֵּיתִ֗י אֶת־ יֽוֹשְׁבֵי֙ הָעִ֣יר הַזֹּ֔את וְאֶת־ הָאָדָ֖ם וְאֶת־ הַבְּהֵמָ֑ה בְּדֶ֥בֶר גָּד֖וֹל יָמֻֽתוּ׃ (יג) וְאַחֲרֵי־ כֵ֣ן נְאֻם־ יְהוָ֡ה אֶתֵּ֣ן אֶת־ צִדְקִיָּ֣הוּ מֶֽלֶךְ־ יְהוּדָ֣ה וְאֶת־ עֲבָדָ֣יו ׀ וְאֶת־ הָעָ֡ם וְאֶת־ הַנִּשְׁאָרִים֩ בָּעִ֨יר הַזֹּ֜את מִן־ הַדֶּ֣בֶר ׀ מִן־ הַחֶ֣רֶב וּמִן־ הָרָעָ֗ב בְּיַד֙ נְבוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֔ל וּבְיַד֙ אֹֽיְבֵיהֶ֔ם וּבְיַ֖ד מְבַקְשֵׁ֣י נַפְשָׁ֑ם וְהִכָּ֣ם לְפִי־ חֶ֔רֶב לֹֽא־ יָח֣וּס עֲלֵיהֶ֔ם וְלֹ֥א יַחְמֹ֖ל וְלֹ֥א יְרַחֵֽם׃ (יד) וְאֶל־ הָעָ֤ם הַזֶּה֙ תֹּאמַ֔ר כֹּ֖ה אָמַ֣ר יְהוָ֑ה הִנְנִ֤י נֹתֵן֙ לִפְנֵיכֶ֔ם אֶת־ דֶּ֥רֶךְ הַחַיִּ֖ים וְאֶת־ דֶּ֥רֶךְ הַמָּֽוֶת׃ (טו) הַיֹּשֵׁב֙ בָּעִ֣יר הַזֹּ֔את יָמ֕וּת בַּחֶ֖רֶב וּבָרָעָ֣ב וּבַדָּ֑בֶר וְהַיּוֹצֵא֩ וְנָפַ֨ל עַל־ הַכַּשְׂדִּ֜ים הַצָּרִ֤ים עֲלֵיכֶם֙ (יחיה) וְחָיָ֔ה וְהָֽיְתָה־ לּ֥וֹ נַפְשׁ֖וֹ לְשָׁלָֽל׃ (טז) כִּ֣י שַׂ֣מְתִּי פָ֠נַי בָּעִ֨יר הַזֹּ֧את לְרָעָ֛ה וְלֹ֥א לְטוֹבָ֖ה נְאֻם־ יְהוָ֑ה בְּיַד־ מֶ֤לֶךְ בָּבֶל֙ תִּנָּתֵ֔ן וּשְׂרָפָ֖הּ בָּאֵֽשׁ׃ (יז) וּלְבֵית֙ מֶ֣לֶךְ יְהוּדָ֔ה שִׁמְע֖וּ דְּבַר־ יְהוָֽה׃ (יח) בֵּ֣ית דָּוִ֗ד כֹּ֚ה אָמַ֣ר יְהוָ֔ה דִּ֤ינוּ לַבֹּ֙קֶר֙ מִשְׁפָּ֔ט וְהַצִּ֥ילוּ גָז֖וּל מִיַּ֣ד עוֹשֵׁ֑ק פֶּן־ תֵּצֵ֨א כָאֵ֜שׁ חֲמָתִ֗י וּבָעֲרָה֙ וְאֵ֣ין מְכַבֶּ֔ה מִפְּנֵ֖י רֹ֥עַ (מעלליהם) מַעַלְלֵיכֶֽם׃ (יט) הִנְנִ֨י אֵלַ֜יִךְ יֹשֶׁ֧בֶת הָעֵ֛מֶק צ֥וּר הַמִּישֹׁ֖ר נְאֻם־ יְהוָ֑ה הָאֹֽמְרִים֙ מִֽי־ יֵחַ֣ת עָלֵ֔ינוּ וּמִ֥י יָב֖וֹא בִּמְעוֹנוֹתֵֽינוּ׃ (כ) וּפָקַדְתִּ֧י עֲלֵיכֶ֛ם כִּפְרִ֥י מַעַלְלֵיכֶ֖ם נְאֻם־ יְהוָ֑ה וְהִצַּ֤תִּי אֵשׁ֙ בְּיַעְרָ֔הּ וְאָכְלָ֖ה כָּל־ סְבִיבֶֽיהָ׃ (כא) כֹּ֚ה אָמַ֣ר יְהוָ֔ה רֵ֖ד בֵּֽית־ מֶ֣לֶךְ יְהוּדָ֑ה וְדִבַּרְתָּ֣ שָׁ֔ם אֶת־ הַדָּבָ֖ר הַזֶּֽה׃ (כב) וְאָֽמַרְתָּ֙ שְׁמַ֣ע דְּבַר־ יְהוָ֔ה מֶ֣לֶךְ יְהוּדָ֔ה הַיֹּשֵׁ֖ב עַל־ כִּסֵּ֣א דָוִ֑ד אַתָּ֤ה וַעֲבָדֶ֙יךָ֙ וְעַמְּךָ֔ הַבָּאִ֖ים בַּשְּׁעָרִ֥ים הָאֵֽלֶּה׃ (כג) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה עֲשׂ֤וּ מִשְׁפָּט֙ וּצְדָקָ֔ה וְהַצִּ֥ילוּ גָז֖וּל מִיַּ֣ד עָשׁ֑וֹק וְגֵר֩ יָת֨וֹם וְאַלְמָנָ֤ה אַל־ תֹּנוּ֙ אַל־ תַּחְמֹ֔סוּ וְדָ֣ם נָקִ֔י אַֽל־ תִּשְׁפְּכ֖וּ בַּמָּק֥וֹם הַזֶּֽה׃ (כד) כִּ֤י אִם־ עָשׂוֹ֙ תַּֽעֲשׂ֔וּ אֶת־ הַדָּבָ֖ר הַזֶּ֑ה וּבָ֣אוּ בְשַׁעֲרֵ֣י הַבַּ֣יִת הַזֶּ֡ה מְלָכִים֩ יֹשְׁבִ֨ים לְדָוִ֜ד עַל־ כִּסְא֗וֹ רֹֽכְבִים֙ בָּרֶ֣כֶב וּבַסּוּסִ֔ים ה֥וּא (ועבדו) וַעֲבָדָ֖יו וְעַמּֽוֹ׃ (כה) וְאִם֙ לֹ֣א תִשְׁמְע֔וּ אֶת־ הַדְּבָרִ֖ים הָאֵ֑לֶּה בִּ֤י נִשְׁבַּ֙עְתִּי֙ נְאֻם־ יְהוָ֔ה כִּי־ לְחָרְבָּ֥ה יִֽהְיֶ֖ה הַבַּ֥יִת הַזֶּֽה׃ (כו) כִּֽי־ כֹ֣ה ׀ אָמַ֣ר יְהוָ֗ה עַל־ בֵּית֙ מֶ֣לֶךְ יְהוּדָ֔ה גִּלְעָ֥ד אַתָּ֛ה לִ֖י רֹ֣אשׁ הַלְּבָנ֑וֹן אִם־ לֹ֤א אֲשִֽׁיתְךָ֙ מִדְבָּ֔ר עָרִ֖ים לֹ֥א (נושבה) נוֹשָֽׁבוּ׃ (כז) וְקִדַּשְׁתִּ֥י עָלֶ֛יךָ מַשְׁחִתִ֖ים אִ֣ישׁ וְכֵלָ֑יו וְכָֽרְתוּ֙ מִבְחַ֣ר אֲרָזֶ֔יךָ וְהִפִּ֖ילוּ עַל־ הָאֵֽשׁ׃ (כח) וְעָֽבְרוּ֙ גּוֹיִ֣ם רַבִּ֔ים עַ֖ל הָעִ֣יר הַזֹּ֑את וְאָֽמְרוּ֙ אִ֣ישׁ אֶל־ רֵעֵ֔הוּ עַל־ מֶ֨ה עָשָׂ֤ה יְהוָה֙ כָּ֔כָה לָעִ֥יר הַגְּדוֹלָ֖ה הַזֹּֽאת׃ (כט) וְאָ֣מְר֔וּ עַ֚ל אֲשֶׁ֣ר עָֽזְב֔וּ אֶת־ בְּרִ֥ית יְהוָ֖ה אֱלֹֽהֵיהֶ֑ם וַיִּֽשְׁתַּחֲו֛וּ לֵאלֹהִ֥ים אֲחֵרִ֖ים וַיַּעַבְדֽוּם׃ (ל) אַל־ תִּבְכּ֣וּ לְמֵ֔ת וְאַל־ תָּנֻ֖דוּ ל֑וֹ בְּכ֤וּ בָכוֹ֙ לַֽהֹלֵ֔ךְ כִּ֣י לֹ֤א יָשׁוּב֙ ע֔וֹד וְרָאָ֖ה אֶת־ אֶ֥רֶץ מוֹלַדְתּֽוֹ׃ (לא) כִּ֣י כֹ֣ה אָֽמַר־ יְ֠הוָה אֶל־ שַׁלֻּ֨ם בֶּן־ יֹאשִׁיָּ֜הוּ מֶ֣לֶךְ יְהוּדָ֗ה הַמֹּלֵךְ֙ תַּ֚חַת יֹאשִׁיָּ֣הוּ אָבִ֔יו אֲשֶׁ֥ר יָצָ֖א מִן־ הַמָּק֣וֹם הַזֶּ֑ה לֹֽא־ יָשׁ֥וּב שָׁ֖ם עֽוֹד׃ (לב) כִּ֗י בִּמְק֛וֹם אֲשֶׁר־ הִגְל֥וּ אֹת֖וֹ שָׁ֣ם יָמ֑וּת וְאֶת־ הָאָ֥רֶץ הַזֹּ֖את לֹֽא־ יִרְאֶ֥ה עֽוֹד׃ (לג) ה֣וֹי בֹּנֶ֤ה בֵיתוֹ֙ בְּֽלֹא־ צֶ֔דֶק וַעֲלִיּוֹתָ֖יו בְּלֹ֣א מִשְׁפָּ֑ט בְּרֵעֵ֙הוּ֙ יַעֲבֹ֣ד חִנָּ֔ם וּפֹעֲל֖וֹ לֹ֥א יִתֶּן־ לֽוֹ׃ (לד) הָאֹמֵ֗ר אֶבְנֶה־ לִּי֙ בֵּ֣ית מִדּ֔וֹת וַעֲלִיּ֖וֹת מְרֻוָּחִ֑ים וְקָ֤רַֽע לוֹ֙ חַלּוֹנָ֔י וְסָפ֣וּן בָּאָ֔רֶז וּמָשׁ֖וֹחַ בַּשָּׁשַֽׁר׃ (לה) הֲתִֽמְלֹ֔ךְ כִּ֥י אַתָּ֖ה מְתַחֲרֶ֣ה בָאָ֑רֶז אָבִ֜יךָ הֲל֧וֹא אָכַ֣ל וְשָׁתָ֗ה וְעָשָׂ֤ה מִשְׁפָּט֙ וּצְדָקָ֔ה אָ֖ז ט֥וֹב לֽוֹ׃ (לו) דָּ֛ן דִּין־ עָנִ֥י וְאֶבְי֖וֹן אָ֣ז ט֑וֹב הֲלוֹא־ הִ֛יא הַדַּ֥עַת אֹתִ֖י נְאֻם־ יְהוָֽה׃ (לז) כִּ֣י אֵ֤ין עֵינֶ֙יךָ֙ וְלִבְּךָ֔ כִּ֖י אִם־ עַל־ בִּצְעֶ֑ךָ וְעַ֤ל דַּֽם־ הַנָּקִי֙ לִשְׁפּ֔וֹךְ וְעַל־ הָעֹ֥שֶׁק וְעַל־ הַמְּרוּצָ֖ה לַעֲשֽׂוֹת׃ (לח) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה אֶל־ יְהוֹיָקִ֤ים בֶּן־ יֹאשִׁיָּ֙הוּ֙ מֶ֣לֶךְ יְהוּדָ֔ה לֹא־ יִסְפְּד֣וּ ל֔וֹ ה֥וֹי אָחִ֖י וְה֣וֹי אָח֑וֹת לֹא־ יִסְפְּד֣וּ ל֔וֹ מִקָּרֹ֛ב אָ֖נִי נְאֻם־ יְהוָ֑ה וְלֹ֥א אֱלֹהֵ֖י מֵרָחֹֽק׃ (יט) אִם־ יִסָּתֵ֨ר אִ֧ישׁ בַּמִּסְתָּרִ֛ים וַאֲנִ֥י לֹֽא־ אֶרְאֶ֖נּוּ נְאֻם־ יְהוָ֑ה הֲל֨וֹא אֶת־ הַשָּׁמַ֧יִם וְאֶת־ הָאָ֛רֶץ אֲנִ֥י מָלֵ֖א נְאֻם־ יְהוָֽה׃ (כ) שָׁמַ֗עְתִּי אֵ֤ת אֲשֶׁר־ אָֽמְרוּ֙ הַנְּבִאִ֔ים הַֽנִּבְּאִ֥ים בִּשְׁמִ֛י שֶׁ֖קֶר לֵאמֹ֑ר חָלַ֖מְתִּי חָלָֽמְתִּי׃ (כא) עַד־ מָתַ֗י הֲיֵ֛שׁ בְּלֵ֥ב הַנְּבִאִ֖ים נִבְּאֵ֣י הַשָּׁ֑קֶר וּנְבִיאֵ֖י תַּרְמִ֥ת לִבָּֽם׃ (כב) הַחֹשְׁבִ֗ים לְהַשְׁכִּ֤יחַ אֶת־ עַמִּי֙ שְׁמִ֔י בַּחֲל֣וֹמֹתָ֔ם אֲשֶׁ֥ר יְסַפְּר֖וּ אִ֣ישׁ לְרֵעֵ֑הוּ כַּאֲשֶׁ֨ר שָׁכְח֧וּ אֲבוֹתָ֛ם אֶת־ שְׁמִ֖י בַּבָּֽעַל׃ (כג) הַנָּבִ֞יא אֲשֶׁר־ אִתּ֤וֹ חֲלוֹם֙ יְסַפֵּ֣ר חֲל֔וֹם וַאֲשֶׁ֤ר דְּבָרִי֙ אִתּ֔וֹ יְדַבֵּ֥ר דְּבָרִ֖י אֱמֶ֑ת מַה־ לַתֶּ֥בֶן אֶת־ הַבָּ֖ר נְאֻם־ יְהוָֽה׃ (כד) הֲל֨וֹא כֹ֧ה דְבָרִ֛י כָּאֵ֖שׁ נְאֻם־ יְהוָ֑ה וּכְפַטִּ֖ישׁ יְפֹ֥צֵֽץ סָֽלַע׃ (כה) לָכֵ֛ן הִנְנִ֥י עַל־ הַנְּבִאִ֖ים נְאֻם־ יְהוָ֑ה מְגַנְּבֵ֣י דְבָרַ֔י אִ֖ישׁ מֵאֵ֥ת רֵעֵֽהוּ׃ (כו) הִנְנִ֥י עַל־ הַנְּבִיאִ֖ם נְאֻם־ יְהוָ֑ה הַלֹּקְחִ֣ים לְשׁוֹנָ֔ם וַֽיִּנְאֲמ֖וּ נְאֻֽם׃ (כז) הִ֠נְנִי עַֽל־ נִבְּאֵ֞י חֲלֹמ֥וֹת שֶׁ֙קֶר֙ נְאֻם־ יְהוָ֔ה וַֽיְסַפְּרוּם֙ וַיַּתְע֣וּ אֶת־ עַמִּ֔י בְּשִׁקְרֵיהֶ֖ם וּבְפַחֲזוּתָ֑ם וְאָנֹכִ֨י לֹֽא־ שְׁלַחְתִּ֜ים וְלֹ֣א צִוִּיתִ֗ים וְהוֹעֵ֛יל לֹֽא־ יוֹעִ֥ילוּ לָֽעָם־ הַזֶּ֖ה נְאֻם־ יְהוָֽה׃ (כח) וְכִי־ יִשְׁאָלְךָ֩ הָעָ֨ם הַזֶּ֜ה אֽוֹ־ הַנָּבִ֤יא אֽוֹ־ כֹהֵן֙ לֵאמֹ֔ר מַה־ מַשָּׂ֖א יְהוָ֑ה וְאָמַרְתָּ֤ אֲלֵיהֶם֙ אֶת־ מַה־ מַשָּׂ֔א וְנָטַשְׁתִּ֥י אֶתְכֶ֖ם נְאֻם־ יְהוָֽה׃ (כט) וְהַנָּבִ֤יא וְהַכֹּהֵן֙ וְהָעָ֔ם אֲשֶׁ֥ר יֹאמַ֖ר מַשָּׂ֣א יְהוָ֑ה וּפָקַדְתִּ֛י עַל־ הָאִ֥ישׁ הַה֖וּא וְעַל־ בֵּיתֽוֹ׃ (ל) כֹּ֥ה תֹאמְר֛וּ אִ֥ישׁ עַל־ רֵעֵ֖הוּ וְאִ֣ישׁ אֶל־ אָחִ֑יו מֶה־ עָנָ֣ה יְהוָ֔ה וּמַה־ דִּבֶּ֖ר יְהוָֽה׃ (לא) וּמַשָּׂ֥א יְהוָ֖ה לֹ֣א תִזְכְּרוּ־ ע֑וֹד כִּ֣י הַמַּשָּׂ֗א יִֽהְיֶה֙ לְאִ֣ישׁ דְּבָר֔וֹ וַהֲפַכְתֶּ֗ם אֶת־ דִּבְרֵי֙ אֱלֹהִ֣ים חַיִּ֔ים יְהוָ֥ה צְבָא֖וֹת אֱלֹהֵֽינוּ׃ (לב) כֹּ֥ה תֹאמַ֖ר אֶל־ הַנָּבִ֑יא מֶה־ עָנָ֣ךְ יְהוָ֔ה וּמַה־ דִּבֶּ֖ר יְהוָֽה׃ (לג) וְאִם־ מַשָּׂ֣א יְהוָה֮ תֹּאמֵרוּ֒ לָכֵ֗ן כֹּ֚ה אָמַ֣ר יְהוָ֔ה יַ֧עַן אֲמָרְכֶ֛ם אֶת־ הַדָּבָ֥ר הַזֶּ֖ה מַשָּׂ֣א יְהוָ֑ה וָאֶשְׁלַ֤ח אֲלֵיכֶם֙ לֵאמֹ֔ר לֹ֥א תֹאמְר֖וּ מַשָּׂ֥א יְהוָֽה׃ (לד) לָכֵ֣ן הִנְנִ֔י וְנָשִׁ֥יתִי אֶתְכֶ֖ם נָשֹׁ֑א וְנָטַשְׁתִּ֣י אֶתְכֶ֗ם וְאֶת־ הָעִיר֙ אֲשֶׁ֨ר נָתַ֧תִּי לָכֶ֛ם וְלַאֲבוֹתֵיכֶ֖ם מֵעַ֥ל פָּנָֽי׃ (לה) וְנָתַתִּ֥י עֲלֵיכֶ֖ם חֶרְפַּ֣ת עוֹלָ֑ם וּכְלִמּ֣וּת עוֹלָ֔ם אֲשֶׁ֖ר לֹ֥א תִשָּׁכֵֽחַ׃ סוף פרק כג (לו) הִרְאַנִי֮ יְהוָה֒ וְהִנֵּ֗ה שְׁנֵי֙ דּוּדָאֵ֣י תְאֵנִ֔ים מוּעָדִ֕ים לִפְנֵ֖י הֵיכַ֣ל יְהוָ֑ה אַחֲרֵ֣י הַגְל֣וֹת נְבוּכַדְרֶאצַּ֣ר מֶֽלֶךְ־ בָּבֶ֡ל אֶת־ יְכָנְיָ֣הוּ בֶן־ יְהוֹיָקִ֣ים מֶֽלֶךְ־ יְהוּדָה֩ וְאֶת־ שָׂרֵ֨י יְהוּדָ֜ה וְאֶת־ הֶחָרָ֤שׁ וְאֶת־ הַמַּסְגֵּר֙ מִיר֣וּשָׁלִַ֔ם וַיְבִאֵ֖ם בָּבֶֽל׃ (לז) הַדּ֣וּד אֶחָ֗ד תְּאֵנִים֙ טֹב֣וֹת מְאֹ֔ד כִּתְאֵנֵ֖י הַבַּכֻּר֑וֹת וְהַדּ֣וּד אֶחָ֗ד תְּאֵנִים֙ רָע֣וֹת מְאֹ֔ד אֲשֶׁ֥ר לֹא־ תֵֽאָכַ֖לְנָה מֵרֹֽעַ׃ (לח) וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י מָֽה־ אַתָּ֤ה רֹאֶה֙ יִרְמְיָ֔הוּ וָאֹמַ֖ר תְּאֵנִ֑ים הַתְּאֵנִ֤ים הַטֹּבוֹת֙ טֹב֣וֹת מְאֹ֔ד וְהָֽרָעוֹת֙ רָע֣וֹת מְאֹ֔ד אֲשֶׁ֥ר לֹא־ תֵאָכַ֖לְנָה מֵרֹֽעַ׃ (לט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (מ) כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל כַּתְּאֵנִ֥ים הַטֹּב֖וֹת הָאֵ֑לֶּה כֵּֽן־ אַכִּ֞יר אֶת־ גָּל֣וּת יְהוּדָ֗ה אֲשֶׁ֨ר שִׁלַּ֜חְתִּי מִן־ הַמָּק֥וֹם הַזֶּ֛ה אֶ֥רֶץ כַּשְׂדִּ֖ים לְטוֹבָֽה׃ (מא) וְשַׂמְתִּ֨י עֵינִ֤י עֲלֵיהֶם֙ לְטוֹבָ֔ה וַהֲשִׁבֹתִ֖ים עַל־ הָאָ֣רֶץ הַזֹּ֑את וּבְנִיתִים֙ וְלֹ֣א אֶהֱרֹ֔ס וּנְטַעְתִּ֖ים וְלֹ֥א אֶתּֽוֹשׁ׃
