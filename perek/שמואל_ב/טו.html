שמואל ב פרק טו לפי החלוקה לסדרים

פרק טו בספר שמואל ב (37 פסוקים) נמצא בשמואל סדר כח (פסוקים ב–לז) ובשמואל סדר כט (פסוקים א–א).

שמואל סדר כח

(א) וַיָּבֹ֨א יוֹאָ֣ב אֶל־ הַמֶּלֶךְ֮ וַיַּגֶּד־ לוֹ֒ וַיִּקְרָ֤א אֶל־ אַבְשָׁלוֹם֙ וַיָּבֹ֣א אֶל־ הַמֶּ֔לֶךְ וַיִּשְׁתַּ֨חוּ ל֧וֹ עַל־ אַפָּ֛יו אַ֖רְצָה לִפְנֵ֣י הַמֶּ֑לֶךְ וַיִּשַּׁ֥ק הַמֶּ֖לֶךְ לְאַבְשָׁלֽוֹם׃ תחילת פרק טו (ב) וַֽיְהִי֙ מֵאַ֣חֲרֵי כֵ֔ן וַיַּ֤עַשׂ לוֹ֙ אַבְשָׁל֔וֹם מֶרְכָּבָ֖ה וְסֻסִ֑ים וַחֲמִשִּׁ֥ים אִ֖ישׁ רָצִ֥ים לְפָנָֽיו׃ (ג) וְהִשְׁכִּים֙ אַבְשָׁל֔וֹם וְעָמַ֕ד עַל־ יַ֖ד דֶּ֣רֶךְ הַשָּׁ֑עַר וַיְהִ֡י כָּל־ הָאִ֣ישׁ אֲשֶֽׁר־ יִהְיֶה־ לּוֹ־ רִיב֩ לָב֨וֹא אֶל־ הַמֶּ֜לֶךְ לַמִּשְׁפָּ֗ט וַיִּקְרָ֨א אַבְשָׁל֤וֹם אֵלָיו֙ וַיֹּ֗אמֶר אֵֽי־ מִזֶּ֥ה עִיר֙ אַ֔תָּה וַיֹּ֕אמֶר מֵאַחַ֥ד שִׁבְטֵֽי־ יִשְׂרָאֵ֖ל עַבְדֶּֽךָ׃ (ד) וַיֹּ֤אמֶר אֵלָיו֙ אַבְשָׁל֔וֹם רְאֵ֥ה דְבָרֶ֖ךָ טוֹבִ֣ים וּנְכֹחִ֑ים וְשֹׁמֵ֥עַ אֵין־ לְךָ֖ מֵאֵ֥ת הַמֶּֽלֶךְ׃ (ה) וַיֹּ֙אמֶר֙ אַבְשָׁל֔וֹם מִי־ יְשִׂמֵ֥נִי שֹׁפֵ֖ט בָּאָ֑רֶץ וְעָלַ֗י יָב֥וֹא כָּל־ אִ֛ישׁ אֲשֶֽׁר־ יִהְיֶה־ לּוֹ־ רִ֥יב וּמִשְׁפָּ֖ט וְהִצְדַּקְתִּֽיו׃ (ו) וְהָיָה֙ בִּקְרָב־ אִ֔ישׁ לְהִשְׁתַּחֲוֺ֖ת ל֑וֹ וְשָׁלַ֧ח אֶת־ יָד֛וֹ וְהֶחֱזִ֥יק ל֖וֹ וְנָ֥שַׁק לֽוֹ׃ (ז) וַיַּ֨עַשׂ אַבְשָׁל֜וֹם כַּדָּבָ֤ר הַזֶּה֙ לְכָל־ יִשְׂרָאֵ֔ל אֲשֶׁר־ יָבֹ֥אוּ לַמִּשְׁפָּ֖ט אֶל־ הַמֶּ֑לֶךְ וַיְגַנֵּב֙ אַבְשָׁל֔וֹם אֶת־ לֵ֖ב אַנְשֵׁ֥י יִשְׂרָאֵֽל׃ (ח) וַיְהִ֕י מִקֵּ֖ץ אַרְבָּעִ֣ים שָׁנָ֑ה וַיֹּ֤אמֶר אַבְשָׁלוֹם֙ אֶל־ הַמֶּ֔לֶךְ אֵ֣לֲכָה נָּ֗א וַאֲשַׁלֵּ֛ם אֶת־ נִדְרִ֛י אֲשֶׁר־ נָדַ֥רְתִּי לַֽיהוָ֖ה בְּחֶבְרֽוֹן׃ (ט) כִּי־ נֵ֙דֶר֙ נָדַ֣ר עַבְדְּךָ֔ בְּשִׁבְתִּ֥י בִגְשׁ֛וּר בַּאֲרָ֖ם לֵאמֹ֑ר אִם־ (ישיב) יָשׁ֨וֹב יְשִׁיבֵ֤נִי יְהוָה֙ יְר֣וּשָׁלִַ֔ם וְעָבַדְתִּ֖י אֶת־ יְהוָֽה׃ (י) וַיֹּֽאמֶר־ ל֥וֹ הַמֶּ֖לֶךְ לֵ֣ךְ בְּשָׁל֑וֹם וַיָּ֖קָם וַיֵּ֥לֶךְ חֶבְרֽוֹנָה׃ (יא) וַיִּשְׁלַ֤ח אַבְשָׁלוֹם֙ מְרַגְּלִ֔ים בְּכָל־ שִׁבְטֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֑ר כְּשָׁמְעֲכֶם֙ אֶת־ ק֣וֹל הַשֹּׁפָ֔ר וַאֲמַרְתֶּ֕ם מָלַ֥ךְ אַבְשָׁל֖וֹם בְּחֶבְרֽוֹן׃ (יב) וְאֶת־ אַבְשָׁל֗וֹם הָלְכ֞וּ מָאתַ֤יִם אִישׁ֙ מִיר֣וּשָׁלִַ֔ם קְרֻאִ֖ים וְהֹלְכִ֣ים לְתֻמָּ֑ם וְלֹ֥א יָדְע֖וּ כָּל־ דָּבָֽר׃ (יג) וַיִּשְׁלַ֣ח אַ֠בְשָׁלוֹם אֶת־ אֲחִיתֹ֨פֶל הַגִּֽילֹנִ֜י יוֹעֵ֣ץ דָּוִ֗ד מֵֽעִירוֹ֙ מִגִּלֹ֔ה בְּזָבְח֖וֹ אֶת־ הַזְּבָחִ֑ים וַיְהִ֤י הַקֶּ֙שֶׁר֙ אַמִּ֔ץ וְהָעָ֛ם הוֹלֵ֥ךְ וָרָ֖ב אֶת־ אַבְשָׁלֽוֹם׃ (יד) וַיָּבֹא֙ הַמַּגִּ֔יד אֶל־ דָּוִ֖ד לֵאמֹ֑ר הָיָ֛ה לֶב־ אִ֥ישׁ יִשְׂרָאֵ֖ל אַחֲרֵ֥י אַבְשָׁלֽוֹם׃ (טו) וַיֹּ֣אמֶר דָּ֠וִד לְכָל־ עֲבָדָ֨יו אֲשֶׁר־ אִתּ֤וֹ בִירוּשָׁלִַ֙ם֙ ק֣וּמוּ וְנִבְרָ֔חָה כִּ֛י לֹא־ תִֽהְיֶה־ לָּ֥נוּ פְלֵיטָ֖ה מִפְּנֵ֣י אַבְשָׁל֑וֹם מַהֲר֣וּ לָלֶ֗כֶת פֶּן־ יְמַהֵ֤ר וְהִשִּׂגָ֙נוּ֙ וְהִדִּ֤יחַ עָלֵ֙ינוּ֙ אֶת־ הָ֣רָעָ֔ה וְהִכָּ֥ה הָעִ֖יר לְפִי־ חָֽרֶב׃ (טז) וַיֹּאמְר֥וּ עַבְדֵֽי־ הַמֶּ֖לֶךְ אֶל־ הַמֶּ֑לֶךְ כְּכֹ֧ל אֲשֶׁר־ יִבְחַ֛ר אֲדֹנִ֥י הַמֶּ֖לֶךְ הִנֵּ֥ה עֲבָדֶֽיךָ׃ (יז) וַיֵּצֵ֥א הַמֶּ֛לֶךְ וְכָל־ בֵּית֖וֹ בְּרַגְלָ֑יו וַיַּעֲזֹ֣ב הַמֶּ֗לֶךְ אֵ֣ת עֶ֧שֶׂר נָשִׁ֛ים פִּֽלַגְשִׁ֖ים לִשְׁמֹ֥ר הַבָּֽיִת׃ (יח) וַיֵּצֵ֥א הַמֶּ֛לֶךְ וְכָל־ הָעָ֖ם בְּרַגְלָ֑יו וַיַּעַמְד֖וּ בֵּ֥ית הַמֶּרְחָֽק׃ (יט) וְכָל־ עֲבָדָיו֙ עֹבְרִ֣ים עַל־ יָד֔וֹ וְכָל־ הַכְּרֵתִ֖י וְכָל־ הַפְּלֵתִ֑י וְכָֽל־ הַגִּתִּ֞ים שֵׁשׁ־ מֵא֣וֹת אִ֗ישׁ אֲשֶׁר־ בָּ֤אוּ בְרַגְלוֹ֙ מִגַּ֔ת עֹבְרִ֖ים עַל־ פְּנֵ֥י הַמֶּֽלֶךְ׃ (כ) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ אִתַּ֣י הַגִּתִּ֔י לָ֧מָּה תֵלֵ֛ךְ גַּם־ אַתָּ֖ה אִתָּ֑נוּ שׁ֣וּב וְשֵׁ֤ב עִם־ הַמֶּ֙לֶךְ֙ כִּֽי־ נָכְרִ֣י אַ֔תָּה וְגַם־ גֹּלֶ֥ה אַתָּ֖ה לִמְקוֹמֶֽךָ׃ (כא) תְּמ֣וֹל ׀ בּוֹאֶ֗ךָ וְהַיּ֞וֹם (אנועך) אֲנִֽיעֲךָ֤ עִמָּ֙נוּ֙ לָלֶ֔כֶת וַאֲנִ֣י הוֹלֵ֔ךְ עַ֥ל אֲשֶׁר־ אֲנִ֖י הוֹלֵ֑ךְ שׁ֣וּב וְהָשֵׁ֧ב אֶת־ אַחֶ֛יךָ עִמָּ֖ךְ חֶ֥סֶד וֶאֱמֶֽת׃ (כב) וַיַּ֧עַן אִתַּ֛י אֶת־ הַמֶּ֖לֶךְ וַיֹּאמַ֑ר חַי־ יְהוָ֗ה וְחֵי֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ כִּ֠י אִם־ בִּמְק֞וֹם אֲשֶׁ֥ר יִֽהְיֶה־ שָּׁ֣ם ׀ אֲדֹנִ֣י הַמֶּ֗לֶךְ אִם־ לְמָ֙וֶת֙ אִם־ לְחַיִּ֔ים כִּי־ שָׁ֖ם יִהְיֶ֥ה עַבְדֶּֽךָ׃ (כג) וַיֹּ֧אמֶר דָּוִ֛ד אֶל־ אִתַּ֖י לֵ֣ךְ וַעֲבֹ֑ר וַֽיַּעֲבֹ֞ר אִתַּ֤י הַגִּתִּי֙ וְכָל־ אֲנָשָׁ֔יו וְכָל־ הַטַּ֖ף אֲשֶׁ֥ר אִתּֽוֹ׃ (כד) וְכָל־ הָאָ֗רֶץ בּוֹכִים֙ ק֣וֹל גָּד֔וֹל וְכָל־ הָעָ֖ם עֹֽבְרִ֑ים וְהַמֶּ֗לֶךְ עֹבֵר֙ בְּנַ֣חַל קִדְר֔וֹן וְכָל־ הָעָם֙ עֹבְרִ֔ים עַל־ פְּנֵי־ דֶ֖רֶךְ אֶת־ הַמִּדְבָּֽר׃ (כה) וְהִנֵּ֨ה גַם־ צָד֜וֹק וְכָֽל־ הַלְוִיִּ֣ם אִתּ֗וֹ נֹֽשְׂאִים֙ אֶת־ אֲרוֹן֙ בְּרִ֣ית הָאֱלֹהִ֔ים וַיַּצִּ֙קוּ֙ אֶת־ אֲר֣וֹן הָאֱלֹהִ֔ים וַיַּ֖עַל אֶבְיָתָ֑ר עַד־ תֹּ֥ם כָּל־ הָעָ֖ם לַעֲב֥וֹר מִן־ הָעִֽיר׃ (כו) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ לְצָד֔וֹק הָשֵׁ֛ב אֶת־ אֲר֥וֹן הָאֱלֹהִ֖ים הָעִ֑יר אִם־ אֶמְצָ֥א חֵן֙ בְּעֵינֵ֣י יְהוָ֔ה וֶהֱשִׁבַ֕נִי וְהִרְאַ֥נִי אֹת֖וֹ וְאֶת־ נָוֵֽהוּ׃ (כז) וְאִם֙ כֹּ֣ה יֹאמַ֔ר לֹ֥א חָפַ֖צְתִּי בָּ֑ךְ הִנְנִ֕י יַֽעֲשֶׂה־ לִּ֕י כַּאֲשֶׁ֥ר ט֖וֹב בְּעֵינָֽיו׃ (כח) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ צָד֣וֹק הַכֹּהֵ֔ן הֲרוֹאֶ֣ה אַתָּ֔ה שֻׁ֥בָה הָעִ֖יר בְּשָׁל֑וֹם וַאֲחִימַ֨עַץ בִּנְךָ֜ וִיהוֹנָתָ֧ן בֶּן־ אֶבְיָתָ֛ר שְׁנֵ֥י בְנֵיכֶ֖ם אִתְּכֶֽם׃ (כט) רְאוּ֙ אָנֹכִ֣י מִתְמַהְמֵ֔הַּ (בעברות) בְּעַֽרְב֖וֹת הַמִּדְבָּ֑ר עַ֣ד בּ֥וֹא דָבָ֛ר מֵעִמָּכֶ֖ם לְהַגִּ֥יד לִֽי׃ (ל) וַיָּ֨שֶׁב צָד֧וֹק וְאֶבְיָתָ֛ר אֶת־ אֲר֥וֹן הָאֱלֹהִ֖ים יְרוּשָׁלִָ֑ם וַיֵּשְׁב֖וּ שָֽׁם׃ (לא) וְדָוִ֡ד עֹלֶה֩ בְמַעֲלֵ֨ה הַזֵּיתִ֜ים עֹלֶ֣ה ׀ וּבוֹכֶ֗ה וְרֹ֥אשׁ לוֹ֙ חָפ֔וּי וְה֖וּא הֹלֵ֣ךְ יָחֵ֑ף וְכָל־ הָעָ֣ם אֲשֶׁר־ אִתּ֗וֹ חָפוּ֙ אִ֣ישׁ רֹאשׁ֔וֹ וְעָל֥וּ עָלֹ֖ה וּבָכֹֽה׃ (לב) וְדָוִד֙ הִגִּ֣יד לֵאמֹ֔ר אֲחִיתֹ֥פֶל בַּקֹּשְׁרִ֖ים עִם־ אַבְשָׁל֑וֹם וַיֹּ֣אמֶר דָּוִ֔ד סַכֶּל־ נָ֛א אֶת־ עֲצַ֥ת אֲחִיתֹ֖פֶל יְהוָֽה׃ (לג) וַיְהִ֤י דָוִד֙ בָּ֣א עַד־ הָרֹ֔אשׁ אֲשֶֽׁר־ יִשְׁתַּחֲוֶ֥ה שָׁ֖ם לֵאלֹהִ֑ים וְהִנֵּ֤ה לִקְרָאתוֹ֙ חוּשַׁ֣י הָאַרְכִּ֔י קָר֙וּעַ֙ כֻּתָּנְתּ֔וֹ וַאֲדָמָ֖ה עַל־ רֹאשֽׁוֹ׃ (לד) וַיֹּ֥אמֶר ל֖וֹ דָּוִ֑ד אִ֚ם עָבַ֣רְתָּ אִתִּ֔י וְהָיִ֥תָ עָלַ֖י לְמַשָּֽׂא׃ (לה) וְאִם־ הָעִ֣יר תָּשׁ֗וּב וְאָמַרְתָּ֤ לְאַבְשָׁלוֹם֙ עַבְדְּךָ֨ אֲנִ֤י הַמֶּ֙לֶךְ֙ אֶֽהְיֶ֔ה עֶ֣בֶד אָבִ֤יךָ וַֽאֲנִי֙ מֵאָ֔ז וְעַתָּ֖ה וַאֲנִ֣י עַבְדֶּ֑ךָ וְהֵפַרְתָּ֣ה לִ֔י אֵ֖ת עֲצַ֥ת אֲחִיתֹֽפֶל׃ (לו) וַהֲל֤וֹא עִמְּךָ֙ שָׁ֔ם צָד֥וֹק וְאֶבְיָתָ֖ר הַכֹּהֲנִ֑ים וְהָיָ֗ה כָּל־ הַדָּבָר֙ אֲשֶׁ֤ר תִּשְׁמַע֙ מִבֵּ֣ית (א) וַיָּבֹ֥א חוּשַׁ֛י רֵעֶ֥ה דָוִ֖ד הָעִ֑יר וְאַבְשָׁלֹ֔ם יָבֹ֖א יְרוּשָׁלִָֽם׃ סוף פרק טו (ב) וְדָוִ֗ד עָבַ֤ר מְעַט֙ מֵֽהָרֹ֔אשׁ וְהִנֵּ֥ה צִיבָ֛א נַ֥עַר מְפִי־ בֹ֖שֶׁת לִקְרָאת֑וֹ וְצֶ֨מֶד חֲמֹרִ֜ים חֲבֻשִׁ֗ים וַעֲלֵיהֶם֩ מָאתַ֨יִם לֶ֜חֶם וּמֵאָ֧ה צִמּוּקִ֛ים וּמֵ֥אָה קַ֖יִץ וְנֵ֥בֶל יָֽיִן׃ (ג) וַיֹּ֧אמֶר הַמֶּ֛לֶךְ אֶל־ צִיבָ֖א מָה־ אֵ֣לֶּה לָּ֑ךְ וַיֹּ֣אמֶר צִ֠יבָא הַחֲמוֹרִ֨ים לְבֵית־ הַמֶּ֜לֶךְ לִרְכֹּ֗ב (ולהלחם) וְהַלֶּ֤חֶם וְהַקַּ֙יִץ֙ לֶאֱכ֣וֹל הַנְּעָרִ֔ים וְהַיַּ֕יִן לִשְׁתּ֥וֹת הַיָּעֵ֖ף בַּמִּדְבָּֽר׃ (ד) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ וְאַיֵּ֖ה בֶּן־ אֲדֹנֶ֑יךָ וַיֹּ֨אמֶר צִיבָ֜א אֶל־ הַמֶּ֗לֶךְ הִנֵּה֙ יוֹשֵׁ֣ב בִּירוּשָׁלִַ֔ם כִּ֣י אָמַ֔ר הַיּ֗וֹם יָשִׁ֤יבוּ לִי֙ בֵּ֣ית יִשְׂרָאֵ֔ל אֵ֖ת מַמְלְכ֥וּת אָבִֽי׃ (ה) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ לְצִבָ֔א הִנֵּ֣ה לְךָ֔ כֹּ֖ל אֲשֶׁ֣ר לִמְפִי־ בֹ֑שֶׁת וַיֹּ֤אמֶר צִיבָא֙ הִֽשְׁתַּחֲוֵ֔יתִי אֶמְצָא־ חֵ֥ן בְּעֵינֶ֖יךָ אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (ו) וּבָ֛א הַמֶּ֥לֶךְ דָּוִ֖ד עַד־ בַּֽחוּרִ֑ים וְהִנֵּ֣ה מִשָּׁם֩ אִ֨ישׁ יוֹצֵ֜א מִמִּשְׁפַּ֣חַת בֵּית־ שָׁא֗וּל וּשְׁמוֹ֙ שִׁמְעִ֣י בֶן־ גֵּרָ֔א יֹצֵ֥א יָצ֖וֹא וּמְקַלֵּֽל׃ (ז) וַיְסַקֵּ֤ל בָּֽאֲבָנִים֙ אֶת־ דָּוִ֔ד וְאֶת־ כָּל־ עַבְדֵ֖י הַמֶּ֣לֶךְ דָּוִ֑ד וְכָל־ הָעָם֙ וְכָל־ הַגִּבֹּרִ֔ים מִימִינ֖וֹ וּמִשְּׂמֹאלֽוֹ׃ (ח) וְכֹֽה־ אָמַ֥ר שִׁמְעִ֖י בְּקַֽלְל֑וֹ צֵ֥א צֵ֛א אִ֥ישׁ הַדָּמִ֖ים וְאִ֥ישׁ הַבְּלִיָּֽעַל׃ (ט) הֵשִׁיב֩ עָלֶ֨יךָ יְהוָ֜ה כֹּ֣ל ׀ דְּמֵ֣י בֵית־ שָׁא֗וּל אֲשֶׁ֤ר מָלַ֙כְתָּ֙ (תחתו) תַּחְתָּ֔יו וַיִּתֵּ֤ן יְהוָה֙ אֶת־ הַמְּלוּכָ֔ה בְּיַ֖ד אַבְשָׁל֣וֹם בְּנֶ֑ךָ וְהִנְּךָ֙ בְּרָ֣עָתֶ֔ךָ כִּ֛י אִ֥ישׁ דָּמִ֖ים אָֽתָּה׃ (י) וַיֹּ֨אמֶר אֲבִישַׁ֤י בֶּן־ צְרוּיָה֙ אֶל־ הַמֶּ֔לֶךְ לָ֣מָּה יְקַלֵּ֞ל הַכֶּ֤לֶב הַמֵּת֙ הַזֶּ֔ה אֶת־ אֲדֹנִ֖י הַמֶּ֑לֶךְ אֶעְבְּרָה־ נָּ֖א וְאָסִ֥ירָה אֶת־ רֹאשֽׁוֹ׃ (יא) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ מַה־ לִּ֥י וְלָכֶ֖ם בְּנֵ֣י וַיֹּ֣אמֶר חוּשַׁ֗י אַתָּ֣ה יָ֠דַעְתָּ אֶת־ אָבִ֨יךָ וְאֶת־ אֲנָשָׁ֜יו כִּ֧י גִבֹּרִ֣ים הֵ֗מָּה וּמָרֵ֥י נֶ֙פֶשׁ֙ הֵ֔מָּה כְּדֹ֥ב שַׁכּ֖וּל בַּשָּׂדֶ֑ה וְאָבִ֙יךָ֙ אִ֣ישׁ מִלְחָמָ֔ה וְלֹ֥א יָלִ֖ין אֶת־ הָעָֽם׃ (לג) הִנֵּ֨ה עַתָּ֤ה הֽוּא־ נֶחְבָּא֙ בְּאַחַ֣ת הַפְּחָתִ֔ים א֖וֹ בְּאַחַ֣ד הַמְּקוֹמֹ֑ת וְהָיָ֗ה כִּנְפֹ֤ל בָּהֶם֙ בַּתְּחִלָּ֔ה וְשָׁמַ֤ע הַשֹּׁמֵ֙עַ֙ וְאָמַ֔ר הָֽיְתָה֙ מַגֵּפָ֔ה בָּעָ֕ם אֲשֶׁ֖ר אַחֲרֵ֥י אַבְשָׁלֹֽם׃ (לד) וְה֣וּא גַם־ בֶּן־ חַ֗יִל אֲשֶׁ֥ר לִבּ֛וֹ כְּלֵ֥ב הָאַרְיֵ֖ה הִמֵּ֣ס יִמָּ֑ס כִּֽי־ יֹדֵ֤עַ כָּל־ יִשְׂרָאֵל֙ כִּי־ גִבּ֣וֹר אָבִ֔יךָ וּבְנֵי־ חַ֖יִל אֲשֶׁ֥ר אִתּֽוֹ׃ (לה) כִּ֣י יָעַ֗צְתִּי הֵ֠אָסֹף יֵאָסֵ֨ף עָלֶ֤יךָ כָל־ יִשְׂרָאֵל֙ מִדָּן֙ וְעַד־ בְּאֵ֣ר שֶׁ֔בַע כַּח֥וֹל אֲשֶׁר־ עַל־ הַיָּ֖ם לָרֹ֑ב וּפָנֶ֥יךָ הֹלְכִ֖ים בַּקְרָֽב׃ (לו) וּבָ֣אנוּ אֵלָ֗יו (באחת) בְּאַחַ֤ד הַמְּקוֹמֹת֙ אֲשֶׁ֣ר נִמְצָ֣א שָׁ֔ם וְנַ֣חְנוּ עָלָ֔יו כַּאֲשֶׁ֛ר יִפֹּ֥ל הַטַּ֖ל עַל־ הָאֲדָמָ֑ה וְלֹֽא־ נ֥וֹתַר בּ֛וֹ וּבְכָל־ הָאֲנָשִׁ֥ים אֲשֶׁר־ אִתּ֖וֹ גַּם־ אֶחָֽד׃ (לז) וְאִם־ אֶל־ עִיר֙ יֵֽאָסֵ֔ף וְהִשִּׂ֧יאוּ כָֽל־ יִשְׂרָאֵ֛ל אֶל־ הָעִ֥יר הַהִ֖יא חֲבָלִ֑ים וְסָחַ֤בְנוּ אֹתוֹ֙ עַד־ הַנַּ֔חַל עַ֛ד אֲשֶֽׁר־ לֹא־ נִמְצָ֥א שָׁ֖ם גַּם־ צְרֽוֹר׃ (לח) וַיֹּ֤אמֶר אַבְשָׁלוֹם֙ וְכָל־ אִ֣ישׁ יִשְׂרָאֵ֔ל טוֹבָ֗ה עֲצַת֙ חוּשַׁ֣י הָאַרְכִּ֔י מֵעֲצַ֖ת אֲחִיתֹ֑פֶל וַיהוָ֣ה צִוָּ֗ה לְהָפֵ֞ר אֶת־ עֲצַ֤ת אֲחִיתֹ֙פֶל֙ הַטּוֹבָ֔ה לְבַעֲב֗וּר הָבִ֧יא יְהוָ֛ה אֶל־ אַבְשָׁל֖וֹם אֶת־ הָרָעָֽה׃ (לט) וַיֹּ֣אמֶר חוּשַׁ֗י אֶל־ צָד֤וֹק וְאֶל־ אֶבְיָתָר֙ הַכֹּ֣הֲנִ֔ים כָּזֹ֣את וְכָזֹ֗את יָעַ֤ץ אֲחִיתֹ֙פֶל֙ אֶת־ אַבְשָׁלֹ֔ם וְאֵ֖ת זִקְנֵ֣י יִשְׂרָאֵ֑ל וְכָזֹ֥את וְכָזֹ֖את יָעַ֥צְתִּי אָֽנִי׃ (מ) וְעַתָּ֡ה שִׁלְח֣וּ מְהֵרָה֩ וְהַגִּ֨ידוּ לְדָוִ֜ד לֵאמֹ֗ר אַל־ תָּ֤לֶן הַלַּ֙יְלָה֙ בְּעַֽרְב֣וֹת הַמִּדְבָּ֔ר וְגַ֖ם עָב֣וֹר תַּעֲב֑וֹר פֶּ֚ן יְבֻלַּ֣ע לַמֶּ֔לֶךְ וּלְכָל־ הָעָ֖ם אֲשֶׁ֥ר אִתּֽוֹ׃ (מא) וִיהוֹנָתָ֨ן וַאֲחִימַ֜עַץ עֹמְדִ֣ים בְּעֵין־ רֹגֵ֗ל וְהָלְכָ֤ה הַשִּׁפְחָה֙ וְהִגִּ֣ידָה לָהֶ֔ם וְהֵם֙ יֵֽלְכ֔וּ וְהִגִּ֖ידוּ לַמֶּ֣לֶךְ דָּוִ֑ד כִּ֣י לֹ֥א יוּכְל֛וּ לְהֵרָא֖וֹת לָב֥וֹא הָעִֽירָה׃ (מב) וַיַּ֤רְא אֹתָם֙ נַ֔עַר וַיַּגֵּ֖ד לְאַבְשָׁלֹ֑ם וַיֵּלְכוּ֩ שְׁנֵיהֶ֨ם מְהֵרָ֜ה וַיָּבֹ֣אוּ ׀ אֶל־ בֵּֽית־ אִ֣ישׁ בְּבַחוּרִ֗ים וְל֥וֹ בְאֵ֛ר בַּחֲצֵר֖וֹ וַיֵּ֥רְדוּ שָֽׁם׃ (מג) וַתִּקַּ֣ח הָאִשָּׁ֗ה וַתִּפְרֹ֤שׂ אֶת־ הַמָּסָךְ֙ עַל־ פְּנֵ֣י הַבְּאֵ֔ר וַתִּשְׁטַ֥ח עָלָ֖יו הָֽרִפ֑וֹת וְלֹ֥א נוֹדַ֖ע דָּבָֽר׃
