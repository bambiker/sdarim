שמואל ב פרק יד לפי החלוקה לסדרים

פרק יד בספר שמואל ב (33 פסוקים) נמצא בשמואל סדר כז (פסוקים טז–מז) ובשמואל סדר כח (פסוקים א–א).

שמואל סדר כז

(א) וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֶל־ אַבְשָׁל֗וֹם אַל־ בְּנִי֙ אַל־ נָ֤א נֵלֵךְ֙ כֻּלָּ֔נוּ וְלֹ֥א נִכְבַּ֖ד עָלֶ֑יךָ וַיִּפְרָץ־ בּ֛וֹ וְלֹֽא־ אָבָ֥ה לָלֶ֖כֶת וַֽיְבָרֲכֵֽהוּ׃ (ב) וַיֹּ֙אמֶר֙ אַבְשָׁל֔וֹם וָלֹ֕א יֵֽלֶךְ־ נָ֥א אִתָּ֖נוּ אַמְנ֣וֹן אָחִ֑י וַיֹּ֤אמֶר לוֹ֙ הַמֶּ֔לֶךְ לָ֥מָּה יֵלֵ֖ךְ עִמָּֽךְ׃ (ג) וַיִּפְרָץ־ בּ֖וֹ אַבְשָׁל֑וֹם וַיִּשְׁלַ֤ח אִתּוֹ֙ אֶת־ אַמְנ֔וֹן וְאֵ֖ת כָּל־ בְּנֵ֥י הַמֶּֽלֶךְ׃ (ד) וַיְצַו֩ אַבְשָׁל֨וֹם אֶת־ נְעָרָ֜יו לֵאמֹ֗ר רְא֣וּ נָ֠א כְּט֨וֹב לֵב־ אַמְנ֤וֹן בַּיַּ֙יִן֙ וְאָמַרְתִּ֣י אֲלֵיכֶ֔ם הַכּ֧וּ אֶת־ אַמְנ֛וֹן וַהֲמִתֶּ֥ם אֹת֖וֹ אַל־ תִּירָ֑אוּ הֲל֗וֹא כִּ֤י אָֽנֹכִי֙ צִוִּ֣יתִי אֶתְכֶ֔ם חִזְק֖וּ וִהְי֥וּ לִבְנֵי־ חָֽיִל׃ (ה) וַֽיַּעֲשׂ֞וּ נַעֲרֵ֤י אַבְשָׁלוֹם֙ לְאַמְנ֔וֹן כַּאֲשֶׁ֥ר צִוָּ֖ה אַבְשָׁל֑וֹם וַיָּקֻ֣מוּ ׀ כָּל־ בְּנֵ֣י הַמֶּ֗לֶךְ וַֽיִּרְכְּב֛וּ אִ֥ישׁ עַל־ פִּרְדּ֖וֹ וַיָּנֻֽסוּ׃ (ו) וַֽיְהִי֙ הֵ֣מָּה בַדֶּ֔רֶךְ וְהַשְּׁמֻעָ֣ה בָ֔אָה אֶל־ דָּוִ֖ד לֵאמֹ֑ר הִכָּ֤ה אַבְשָׁלוֹם֙ אֶת־ כָּל־ בְּנֵ֣י הַמֶּ֔לֶךְ וְלֹֽא־ נוֹתַ֥ר מֵהֶ֖ם אֶחָֽד׃ (ז) וַיָּ֧קָם הַמֶּ֛לֶךְ וַיִּקְרַ֥ע אֶת־ בְּגָדָ֖יו וַיִּשְׁכַּ֣ב אָ֑רְצָה וְכָל־ עֲבָדָ֥יו נִצָּבִ֖ים קְרֻעֵ֥י בְגָדִֽים׃ (ח) וַיַּ֡עַן יוֹנָדָ֣ב ׀ בֶּן־ שִׁמְעָ֨ה אֲחִֽי־ דָוִ֜ד וַיֹּ֗אמֶר אַל־ יֹאמַ֤ר אֲדֹנִי֙ אֵ֣ת כָּל־ הַנְּעָרִ֤ים בְּנֵֽי־ הַמֶּ֙לֶךְ֙ הֵמִ֔יתוּ כִּֽי־ אַמְנ֥וֹן לְבַדּ֖וֹ מֵ֑ת כִּֽי־ עַל־ פִּ֤י אַבְשָׁלוֹם֙ הָיְתָ֣ה שׂוּמָ֔ה מִיּוֹם֙ עַנֹּת֔וֹ אֵ֖ת תָּמָ֥ר אֲחֹתֽוֹ׃ (ט) וְעַתָּ֡ה אַל־ יָשֵׂם֩ אֲדֹנִ֨י הַמֶּ֤לֶךְ אֶל־ לִבּוֹ֙ דָּבָ֣ר לֵאמֹ֔ר כָּל־ בְּנֵ֥י הַמֶּ֖לֶךְ מֵ֑תוּ כִּֽי־ אִם־ אַמְנ֥וֹן לְבַדּ֖וֹ מֵֽת׃ (י) וַיִּבְרַ֖ח אַבְשָׁל֑וֹם וַיִּשָּׂ֞א הַנַּ֤עַר הַצֹּפֶה֙ אֶת־ (עינו) עֵינָ֔יו וַיַּ֗רְא וְהִנֵּ֨ה עַם־ רַ֜ב הֹלְכִ֥ים מִדֶּ֛רֶךְ אַחֲרָ֖יו מִצַּ֥ד הָהָֽר׃ (יא) וַיֹּ֤אמֶר יֽוֹנָדָב֙ אֶל־ הַמֶּ֔לֶךְ הִנֵּ֥ה בְנֵֽי־ הַמֶּ֖לֶךְ בָּ֑אוּ כִּדְבַ֥ר עַבְדְּךָ֖ כֵּ֥ן הָיָֽה׃ (יב) וַיְהִ֣י ׀ כְּכַלֹּת֣וֹ לְדַבֵּ֗ר וְהִנֵּ֤ה בְנֵֽי־ הַמֶּ֙לֶךְ֙ בָּ֔אוּ וַיִּשְׂא֥וּ קוֹלָ֖ם וַיִּבְכּ֑וּ וְגַם־ הַמֶּ֙לֶךְ֙ וְכָל־ עֲבָדָ֔יו בָּכ֕וּ בְּכִ֖י גָּד֥וֹל מְאֹֽד׃ (יג) וְאַבְשָׁל֣וֹם בָּרַ֔ח וַיֵּ֛לֶךְ אֶל־ תַּלְמַ֥י בֶּן־ (עמיחור) עַמִּיה֖וּד מֶ֣לֶךְ גְּשׁ֑וּר וַיִּתְאַבֵּ֥ל עַל־ בְּנ֖וֹ כָּל־ הַיָּמִֽים׃ (יד) וְאַבְשָׁל֥וֹם בָּרַ֖ח וַיֵּ֣לֶךְ גְּשׁ֑וּר וַיְהִי־ שָׁ֖ם שָׁלֹ֥שׁ שָׁנִֽים׃ (טו) וַתְּכַל֙ דָּוִ֣ד הַמֶּ֔לֶךְ לָצֵ֖את אֶל־ אַבְשָׁל֑וֹם כִּֽי־ נִחַ֥ם עַל־ אַמְנ֖וֹן כִּֽי־ מֵֽת׃ תחילת פרק יד (טז) וַיֵּ֖דַע יוֹאָ֣ב בֶּן־ צְרֻיָ֑ה כִּֽי־ לֵ֥ב הַמֶּ֖לֶךְ עַל־ אַבְשָׁלֽוֹם׃ (יז) וַיִּשְׁלַ֤ח יוֹאָב֙ תְּק֔וֹעָה וַיִּקַּ֥ח מִשָּׁ֖ם אִשָּׁ֣ה חֲכָמָ֑ה וַיֹּ֣אמֶר אֵ֠לֶיהָ הִֽתְאַבְּלִי־ נָ֞א וְלִבְשִׁי־ נָ֣א בִגְדֵי־ אֵ֗בֶל וְאַל־ תָּס֙וּכִי֙ שֶׁ֔מֶן וְהָיִ֕ית כְּאִשָּׁ֗ה זֶ֚ה יָמִ֣ים רַבִּ֔ים מִתְאַבֶּ֖לֶת עַל־ מֵֽת׃ (יח) וּבָאת֙ אֶל־ הַמֶּ֔לֶךְ וְדִבַּ֥רְתְּ אֵלָ֖יו כַּדָּבָ֣ר הַזֶּ֑ה וַיָּ֧שֶׂם יוֹאָ֛ב אֶת־ הַדְּבָרִ֖ים בְּפִֽיהָ׃ (יט) וַ֠תֹּאמֶר הָאִשָּׁ֤ה הַתְּקֹעִית֙ אֶל־ הַמֶּ֔לֶךְ וַתִּפֹּ֧ל עַל־ אַפֶּ֛יהָ אַ֖רְצָה וַתִּשְׁתָּ֑חוּ וַתֹּ֖אמֶר הוֹשִׁ֥עָה הַמֶּֽלֶךְ׃ (כ) וַיֹּֽאמֶר־ לָ֥הּ הַמֶּ֖לֶךְ מַה־ לָּ֑ךְ וַתֹּ֗אמֶר אֲבָ֛ל אִשָּֽׁה־ אַלְמָנָ֥ה אָ֖נִי וַיָּ֥מָת אִישִֽׁי׃ (כא) וּלְשִׁפְחָֽתְךָ֙ שְׁנֵ֣י בָנִ֔ים וַיִּנָּצ֤וּ שְׁנֵיהֶם֙ בַּשָּׂדֶ֔ה וְאֵ֥ין מַצִּ֖יל בֵּֽינֵיהֶ֑ם וַיַּכּ֧וֹ הָאֶחָ֛ד אֶת־ הָאֶחָ֖ד וַיָּ֥מֶת אֹתֽוֹ׃ (כב) וְהִנֵּה֩ קָ֨מָה כָֽל־ הַמִּשְׁפָּחָ֜ה עַל־ שִׁפְחָתֶ֗ךָ וַיֹּֽאמְרוּ֙ תְּנִ֣י ׀ אֶת־ מַכֵּ֣ה אָחִ֗יו וּנְמִתֵ֙הוּ֙ בְּנֶ֤פֶשׁ אָחִיו֙ אֲשֶׁ֣ר הָרָ֔ג וְנַשְׁמִ֖ידָה גַּ֣ם אֶת־ הַיּוֹרֵ֑שׁ וְכִבּ֗וּ אֶת־ גַּֽחַלְתִּי֙ אֲשֶׁ֣ר נִשְׁאָ֔רָה לְבִלְתִּ֧י (שום)־ שִׂים־ לְאִישִׁ֛י שֵׁ֥ם וּשְׁאֵרִ֖ית עַל־ פְּנֵ֥י הָאֲדָמָֽה׃ (כג) וַיֹּ֧אמֶר הַמֶּ֛לֶךְ אֶל־ הָאִשָּׁ֖ה לְכִ֣י לְבֵיתֵ֑ךְ וַאֲנִ֖י אֲצַוֶּ֥ה עָלָֽיִךְ׃ (כד) וַתֹּ֜אמֶר הָאִשָּׁ֤ה הַתְּקוֹעִית֙ אֶל־ הַמֶּ֔לֶךְ עָלַ֞י אֲדֹנִ֥י הַמֶּ֛לֶךְ הֶעָוֺ֖ן וְעַל־ בֵּ֣ית אָבִ֑י וְהַמֶּ֥לֶךְ וְכִסְא֖וֹ נָקִֽי׃ (כה) וַיֹּ֖אמֶר הַמֶּ֑לֶךְ הַֽמְדַבֵּ֤ר אֵלַ֙יִךְ֙ וַֽהֲבֵאת֣וֹ אֵלַ֔י וְלֹֽא־ יֹסִ֥יף ע֖וֹד לָגַ֥עַת בָּֽךְ׃ (כו) וַתֹּאמֶר֩ יִזְכָּר־ נָ֨א הַמֶּ֜לֶךְ אֶת־ יְהוָ֣ה אֱלֹהֶ֗יךָ (מהרבית) מֵהַרְבַּ֞ת גֹּאֵ֤ל הַדָּם֙ לְשַׁחֵ֔ת וְלֹ֥א יַשְׁמִ֖ידוּ אֶת־ בְּנִ֑י וַיֹּ֙אמֶר֙ חַי־ יְהוָ֔ה אִם־ יִפֹּ֛ל מִשַּׂעֲרַ֥ת בְּנֵ֖ךְ אָֽרְצָה׃ (כז) וַתֹּ֙אמֶר֙ הָֽאִשָּׁ֔ה תְּדַבֶּר־ נָ֧א שִׁפְחָתְךָ֛ אֶל־ אֲדֹנִ֥י הַמֶּ֖לֶךְ דָּבָ֑ר וַיֹּ֖אמֶר דַּבֵּֽרִי׃ (כח) וַתֹּ֙אמֶר֙ הָֽאִשָּׁ֔ה וְלָ֧מָּה חָשַׁ֛בְתָּה כָּזֹ֖את עַל־ עַ֣ם אֱלֹהִ֑ים וּמִדַּבֵּ֨ר הַמֶּ֜לֶךְ הַדָּבָ֤ר הַזֶּה֙ כְּאָשֵׁ֔ם לְבִלְתִּ֛י הָשִׁ֥יב הַמֶּ֖לֶךְ אֶֽת־ נִדְּחֽוֹ׃ (כט) כִּי־ מ֣וֹת נָמ֔וּת וְכַמַּ֙יִם֙ הַנִּגָּרִ֣ים אַ֔רְצָה אֲשֶׁ֖ר לֹ֣א יֵאָסֵ֑פוּ וְלֹֽא־ יִשָּׂ֤א אֱלֹהִים֙ נֶ֔פֶשׁ וְחָשַׁב֙ מַֽחֲשָׁב֔וֹת לְבִלְתִּ֛י יִדַּ֥ח מִמֶּ֖נּוּ נִדָּֽח׃ (ל) וְ֠עַתָּה אֲשֶׁר־ בָּ֜אתִי לְדַבֵּ֨ר אֶל־ הַמֶּ֤לֶךְ אֲדֹנִי֙ אֶת־ הַדָּבָ֣ר הַזֶּ֔ה כִּ֥י יֵֽרְאֻ֖נִי הָעָ֑ם וַתֹּ֤אמֶר שִׁפְחָֽתְךָ֙ אֲדַבְּרָה־ נָּ֣א אֶל־ הַמֶּ֔לֶךְ אוּלַ֛י יַעֲשֶׂ֥ה הַמֶּ֖לֶךְ אֶת־ דְּבַ֥ר אֲמָתֽוֹ׃ (לא) כִּ֚י יִשְׁמַ֣ע הַמֶּ֔לֶךְ לְהַצִּ֥יל אֶת־ אֲמָת֖וֹ מִכַּ֣ף הָאִ֑ישׁ לְהַשְׁמִ֨יד אֹתִ֤י וְאֶת־ בְּנִי֙ יַ֔חַד מִֽנַּחֲלַ֖ת אֱלֹהִֽים׃ (לב) וַתֹּ֙אמֶר֙ שִׁפְחָ֣תְךָ֔ יִֽהְיֶה־ נָּ֛א דְּבַר־ אֲדֹנִ֥י הַמֶּ֖לֶךְ לִמְנוּחָ֑ה כִּ֣י ׀ כְּמַלְאַ֣ךְ הָאֱלֹהִ֗ים כֵּ֣ן אֲדֹנִ֤י הַמֶּ֙לֶךְ֙ לִשְׁמֹ֙עַ֙ הַטּ֣וֹב וְהָרָ֔ע וַֽיהוָ֥ה אֱלֹהֶ֖יךָ יְהִ֥י עִמָּֽךְ׃ (יח) וַיֵּצֵ֥א הַמֶּ֛לֶךְ וְכָל־ הָעָ֖ם בְּרַגְלָ֑יו וַיַּעַמְד֖וּ בֵּ֥ית הַמֶּרְחָֽק׃ (יט) וְכָל־ עֲבָדָיו֙ עֹבְרִ֣ים עַל־ יָד֔וֹ וְכָל־ הַכְּרֵתִ֖י וְכָל־ הַפְּלֵתִ֑י וְכָֽל־ הַגִּתִּ֞ים שֵׁשׁ־ מֵא֣וֹת אִ֗ישׁ אֲשֶׁר־ בָּ֤אוּ בְרַגְלוֹ֙ מִגַּ֔ת עֹבְרִ֖ים עַל־ פְּנֵ֥י הַמֶּֽלֶךְ׃ (כ) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ אִתַּ֣י הַגִּתִּ֔י לָ֧מָּה תֵלֵ֛ךְ גַּם־ אַתָּ֖ה אִתָּ֑נוּ שׁ֣וּב וְשֵׁ֤ב עִם־ הַמֶּ֙לֶךְ֙ כִּֽי־ נָכְרִ֣י אַ֔תָּה וְגַם־ גֹּלֶ֥ה אַתָּ֖ה לִמְקוֹמֶֽךָ׃ (כא) תְּמ֣וֹל ׀ בּוֹאֶ֗ךָ וְהַיּ֞וֹם (אנועך) אֲנִֽיעֲךָ֤ עִמָּ֙נוּ֙ לָלֶ֔כֶת וַאֲנִ֣י הוֹלֵ֔ךְ עַ֥ל אֲשֶׁר־ אֲנִ֖י הוֹלֵ֑ךְ שׁ֣וּב וְהָשֵׁ֧ב אֶת־ אַחֶ֛יךָ עִמָּ֖ךְ חֶ֥סֶד וֶאֱמֶֽת׃ (כב) וַיַּ֧עַן אִתַּ֛י אֶת־ הַמֶּ֖לֶךְ וַיֹּאמַ֑ר חַי־ יְהוָ֗ה וְחֵי֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ כִּ֠י אִם־ בִּמְק֞וֹם אֲשֶׁ֥ר יִֽהְיֶה־ שָּׁ֣ם ׀ אֲדֹנִ֣י הַמֶּ֗לֶךְ אִם־ לְמָ֙וֶת֙ אִם־ לְחַיִּ֔ים כִּי־ שָׁ֖ם יִהְיֶ֥ה עַבְדֶּֽךָ׃ (כג) וַיֹּ֧אמֶר דָּוִ֛ד אֶל־ אִתַּ֖י לֵ֣ךְ וַעֲבֹ֑ר וַֽיַּעֲבֹ֞ר אִתַּ֤י הַגִּתִּי֙ וְכָל־ אֲנָשָׁ֔יו וְכָל־ הַטַּ֖ף אֲשֶׁ֥ר אִתּֽוֹ׃ (כד) וְכָל־ הָאָ֗רֶץ בּוֹכִים֙ ק֣וֹל גָּד֔וֹל וְכָל־ הָעָ֖ם עֹֽבְרִ֑ים וְהַמֶּ֗לֶךְ עֹבֵר֙ בְּנַ֣חַל קִדְר֔וֹן וְכָל־ הָעָם֙ עֹבְרִ֔ים עַל־ פְּנֵי־ דֶ֖רֶךְ אֶת־ הַמִּדְבָּֽר׃ (כה) וְהִנֵּ֨ה גַם־ צָד֜וֹק וְכָֽל־ הַלְוִיִּ֣ם אִתּ֗וֹ נֹֽשְׂאִים֙ אֶת־ אֲרוֹן֙ בְּרִ֣ית הָאֱלֹהִ֔ים וַיַּצִּ֙קוּ֙ אֶת־ אֲר֣וֹן הָאֱלֹהִ֔ים וַיַּ֖עַל אֶבְיָתָ֑ר עַד־ תֹּ֥ם כָּל־ הָעָ֖ם לַעֲב֥וֹר מִן־ הָעִֽיר׃ (כו) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ לְצָד֔וֹק הָשֵׁ֛ב אֶת־ אֲר֥וֹן הָאֱלֹהִ֖ים הָעִ֑יר אִם־ אֶמְצָ֥א חֵן֙ בְּעֵינֵ֣י יְהוָ֔ה וֶהֱשִׁבַ֕נִי וְהִרְאַ֥נִי אֹת֖וֹ וְאֶת־ נָוֵֽהוּ׃ (כז) וְאִם֙ כֹּ֣ה יֹאמַ֔ר לֹ֥א חָפַ֖צְתִּי בָּ֑ךְ הִנְנִ֕י יַֽעֲשֶׂה־ לִּ֕י כַּאֲשֶׁ֥ר ט֖וֹב בְּעֵינָֽיו׃ (כח) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ צָד֣וֹק הַכֹּהֵ֔ן הֲרוֹאֶ֣ה אַתָּ֔ה שֻׁ֥בָה הָעִ֖יר בְּשָׁל֑וֹם וַאֲחִימַ֨עַץ בִּנְךָ֜ וִיהוֹנָתָ֧ן בֶּן־ אֶבְיָתָ֛ר שְׁנֵ֥י בְנֵיכֶ֖ם אִתְּכֶֽם׃ (כט) רְאוּ֙ אָנֹכִ֣י מִתְמַהְמֵ֔הַּ (בעברות) בְּעַֽרְב֖וֹת הַמִּדְבָּ֑ר עַ֣ד בּ֥וֹא דָבָ֛ר מֵעִמָּכֶ֖ם לְהַגִּ֥יד לִֽי׃ (ל) וַיָּ֨שֶׁב צָד֧וֹק וְאֶבְיָתָ֛ר אֶת־ אֲר֥וֹן הָאֱלֹהִ֖ים יְרוּשָׁלִָ֑ם וַיֵּשְׁב֖וּ שָֽׁם׃ (לא) וְדָוִ֡ד עֹלֶה֩ בְמַעֲלֵ֨ה הַזֵּיתִ֜ים עֹלֶ֣ה ׀ וּבוֹכֶ֗ה וְרֹ֥אשׁ לוֹ֙ חָפ֔וּי וְה֖וּא הֹלֵ֣ךְ יָחֵ֑ף וְכָל־ הָעָ֣ם אֲשֶׁר־ אִתּ֗וֹ חָפוּ֙ אִ֣ישׁ רֹאשׁ֔וֹ וְעָל֥וּ עָלֹ֖ה וּבָכֹֽה׃ (לב) וְדָוִד֙ הִגִּ֣יד לֵאמֹ֔ר אֲחִיתֹ֥פֶל בַּקֹּשְׁרִ֖ים עִם־ אַבְשָׁל֑וֹם וַיֹּ֣אמֶר דָּוִ֔ד סַכֶּל־ נָ֛א אֶת־ עֲצַ֥ת אֲחִיתֹ֖פֶל יְהוָֽה׃ (לג) וַיְהִ֤י דָוִד֙ בָּ֣א עַד־ הָרֹ֔אשׁ אֲשֶֽׁר־ יִשְׁתַּחֲוֶ֥ה שָׁ֖ם לֵאלֹהִ֑ים וְהִנֵּ֤ה לִקְרָאתוֹ֙ חוּשַׁ֣י הָאַרְכִּ֔י קָר֙וּעַ֙ כֻּתָּנְתּ֔וֹ וַאֲדָמָ֖ה עַל־ רֹאשֽׁוֹ׃ (לד) וַיֹּ֥אמֶר ל֖וֹ דָּוִ֑ד אִ֚ם עָבַ֣רְתָּ אִתִּ֔י וְהָיִ֥תָ עָלַ֖י לְמַשָּֽׂא׃ (לה) וְאִם־ הָעִ֣יר תָּשׁ֗וּב וְאָמַרְתָּ֤ לְאַבְשָׁלוֹם֙ עַבְדְּךָ֨ אֲנִ֤י הַמֶּ֙לֶךְ֙ אֶֽהְיֶ֔ה עֶ֣בֶד אָבִ֤יךָ וַֽאֲנִי֙ מֵאָ֔ז וְעַתָּ֖ה וַאֲנִ֣י עַבְדֶּ֑ךָ וְהֵפַרְתָּ֣ה לִ֔י אֵ֖ת עֲצַ֥ת אֲחִיתֹֽפֶל׃ (לו) וַהֲל֤וֹא עִמְּךָ֙ שָׁ֔ם צָד֥וֹק וְאֶבְיָתָ֖ר הַכֹּהֲנִ֑ים וְהָיָ֗ה כָּל־ הַדָּבָר֙ אֲשֶׁ֤ר תִּשְׁמַע֙ מִבֵּ֣ית הַמֶּ֔לֶךְ תַּגִּ֕יד לְצָד֥וֹק וּלְאֶבְיָתָ֖ר הַכֹּהֲנִֽים׃ (לז) הִנֵּה־ שָׁ֤ם עִמָּם֙ שְׁנֵ֣י בְנֵיהֶ֔ם אֲחִימַ֣עַץ לְצָד֔וֹק וִיהוֹנָתָ֖ן לְאֶבְיָתָ֑ר וּשְׁלַחְתֶּ֤ם בְּיָדָם֙ אֵלַ֔י כָּל־ דָּבָ֖ר אֲשֶׁ֥ר תִּשְׁמָֽעוּ׃
