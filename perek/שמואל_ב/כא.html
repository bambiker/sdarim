שמואל ב פרק כא לפי החלוקה לסדרים

פרק כא בספר שמואל ב (22 פסוקים) נמצא בשמואל סדר לב (פסוקים לב–לז) ובשמואל סדר לג (פסוקים א–טז).

שמואל סדר לב

(א) וַיַּעֲבֹ֧ר כָּל־ הָעָ֛ם אֶת־ הַיַּרְדֵּ֖ן וְהַמֶּ֣לֶךְ עָבָ֑ר וַיִּשַּׁ֨ק הַמֶּ֤לֶךְ לְבַרְזִלַּי֙ וַיְבָ֣רֲכֵ֔הוּ וַיָּ֖שָׁב לִמְקֹמֽוֹ׃ (ב) וַיַּעֲבֹ֤ר הַמֶּ֙לֶךְ֙ הַגִּלְגָּ֔לָה וְכִמְהָ֖ן עָבַ֣ר עִמּ֑וֹ וְכָל־ עַ֤ם יְהוּדָה֙ (ויעברו) הֶעֱבִ֣ירוּ אֶת־ הַמֶּ֔לֶךְ וְגַ֕ם חֲצִ֖י עַ֥ם יִשְׂרָאֵֽל׃ (ג) וְהִנֵּ֛ה כָּל־ אִ֥ישׁ יִשְׂרָאֵ֖ל בָּאִ֣ים אֶל־ הַמֶּ֑לֶךְ וַיֹּאמְר֣וּ אֶל־ הַמֶּ֡לֶךְ מַדּוּעַ֩ גְּנָב֨וּךָ אַחֵ֜ינוּ אִ֣ישׁ יְהוּדָ֗ה וַיַּעֲבִ֨רוּ אֶת־ הַמֶּ֤לֶךְ וְאֶת־ בֵּיתוֹ֙ אֶת־ הַיַּרְדֵּ֔ן וְכָל־ אַנְשֵׁ֥י דָוִ֖ד עִמּֽוֹ׃ (ד) וַיַּעַן֩ כָּל־ אִ֨ישׁ יְהוּדָ֜ה עַל־ אִ֣ישׁ יִשְׂרָאֵ֗ל כִּֽי־ קָר֤וֹב הַמֶּ֙לֶךְ֙ אֵלַ֔י וְלָ֤מָּה זֶּה֙ חָרָ֣ה לְךָ֔ עַל־ הַדָּבָ֖ר הַזֶּ֑ה הֶאָכ֤וֹל אָכַ֙לְנוּ֙ מִן־ הַמֶּ֔לֶךְ אִם־ נִשֵּׂ֥את נִשָּׂ֖א לָֽנוּ׃ (ה) וַיַּ֣עַן אִֽישׁ־ יִשְׂרָאֵל֩ אֶת־ אִ֨ישׁ יְהוּדָ֜ה וַיֹּ֗אמֶר עֶֽשֶׂר־ יָד֨וֹת לִ֣י בַמֶּלֶךְ֮ וְגַם־ בְּדָוִד֮ אֲנִ֣י מִמְּךָ֒ וּמַדּ֙וּעַ֙ הֱקִלֹּתַ֔נִי וְלֹא־ הָיָ֨ה דְבָרִ֥י רִאשׁ֛וֹן לִ֖י לְהָשִׁ֣יב אֶת־ מַלְכִּ֑י וַיִּ֙קֶשׁ֙ דְּבַר־ אִ֣ישׁ יְהוּדָ֔ה מִדְּבַ֖ר אִ֥ישׁ יִשְׂרָאֵֽל׃ (ו) וְשָׁ֨ם נִקְרָ֜א אִ֣ישׁ בְּלִיַּ֗עַל וּשְׁמ֛וֹ שֶׁ֥בַע בֶּן־ בִּכְרִ֖י אִ֣ישׁ יְמִינִ֑י וַיִּתְקַ֣ע בַּשֹּׁפָ֗ר וַ֠יֹּאמֶר אֵֽין־ לָ֨נוּ חֵ֜לֶק בְּדָוִ֗ד וְלֹ֤א נַֽחֲלָה־ לָ֙נוּ֙ בְּבֶן־ יִשַׁ֔י אִ֥ישׁ לְאֹהָלָ֖יו יִשְׂרָאֵֽל׃ (ז) וַיַּ֜עַל כָּל־ אִ֤ישׁ יִשְׂרָאֵל֙ מֵאַחֲרֵ֣י דָוִ֔ד אַחֲרֵ֖י שֶׁ֣בַע בֶּן־ בִּכְרִ֑י וְאִ֤ישׁ יְהוּדָה֙ דָּבְק֣וּ בְמַלְכָּ֔ם מִן־ הַיַּרְדֵּ֖ן וְעַד־ יְרוּשָׁלִָֽם׃ (ח) וַיָּבֹ֨א דָוִ֣ד אֶל־ בֵּיתוֹ֮ יְרֽוּשָׁלִַם֒ וַיִּקַּ֣ח הַמֶּ֡לֶךְ אֵ֣ת עֶֽשֶׂר־ נָשִׁ֣ים ׀ פִּלַגְשִׁ֡ים אֲשֶׁ֣ר הִנִּיחַ֩ לִשְׁמֹ֨ר הַבַּ֜יִת וַֽיִּתְּנֵ֤ם בֵּית־ מִשְׁמֶ֙רֶת֙ וַֽיְכַלְכְּלֵ֔ם וַאֲלֵיהֶ֖ם לֹא־ בָ֑א וַתִּהְיֶ֧ינָה צְרֻר֛וֹת עַד־ י֥וֹם מֻתָ֖ן אַלְמְנ֥וּת חַיּֽוּת׃ (ט) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ עֲמָשָׂ֔א הַזְעֶק־ לִ֥י אֶת־ אִישׁ־ יְהוּדָ֖ה שְׁלֹ֣שֶׁת יָמִ֑ים וְאַתָּ֖ה פֹּ֥ה עֲמֹֽד׃ (י) וַיֵּ֥לֶךְ עֲמָשָׂ֖א לְהַזְעִ֣יק אֶת־ יְהוּדָ֑ה (וייחר) וַיּ֕וֹחֶר מִן־ הַמּוֹעֵ֖ד אֲשֶׁ֥ר יְעָדֽוֹ׃ (יא) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ אֲבִישַׁ֔י עַתָּ֗ה יֵ֧רַֽע לָ֛נוּ שֶׁ֥בַע בֶּן־ בִּכְרִ֖י מִן־ אַבְשָׁל֑וֹם אַ֠תָּה קַ֞ח אֶת־ עַבְדֵ֤י אֲדֹנֶ֙יךָ֙ וּרְדֹ֣ף אַחֲרָ֔יו פֶּן־ מָ֥צָא ל֛וֹ עָרִ֥ים בְּצֻר֖וֹת וְהִצִּ֥יל עֵינֵֽנוּ׃ (יב) וַיֵּצְא֤וּ אַֽחֲרָיו֙ אַנְשֵׁ֣י יוֹאָ֔ב וְהַכְּרֵתִ֥י וְהַפְּלֵתִ֖י וְכָל־ הַגִּבֹּרִ֑ים וַיֵּֽצְאוּ֙ מִיר֣וּשָׁלִַ֔ם לִרְדֹּ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (יג) הֵ֗ם עִם־ הָאֶ֤בֶן הַגְּדוֹלָה֙ אֲשֶׁ֣ר בְּגִבְע֔וֹן וַעֲמָשָׂ֖א בָּ֣א לִפְנֵיהֶ֑ם וְיוֹאָ֞ב חָג֣וּר ׀ מִדּ֣וֹ לְבֻשׁ֗וּ (ועלו) וְעָלָ֞יו חֲג֥וֹר חֶ֙רֶב֙ מְצֻמֶּ֤דֶת עַל־ מָתְנָיו֙ בְּתַעְרָ֔הּ וְה֥וּא יָצָ֖א וַתִּפֹּֽל׃ (יד) וַיֹּ֤אמֶר יוֹאָב֙ לַעֲמָשָׂ֔א הֲשָׁל֥וֹם אַתָּ֖ה אָחִ֑י וַתֹּ֜חֶז יַד־ יְמִ֥ין יוֹאָ֛ב בִּזְקַ֥ן עֲמָשָׂ֖א לִנְשָׁק־ לֽוֹ׃ (טו) וַעֲמָשָׂ֨א לֹֽא־ נִשְׁמַ֜ר בַּחֶ֣רֶב ׀ אֲשֶׁ֣ר בְּיַד־ יוֹאָ֗ב וַיַּכֵּהוּ֩ בָ֨הּ אֶל־ הַחֹ֜מֶשׁ וַיִּשְׁפֹּ֨ךְ מֵעָ֥יו אַ֛רְצָה וְלֹא־ שָׁ֥נָה ל֖וֹ וַיָּמֹ֑ת וְיוֹאָב֙ וַאֲבִישַׁ֣י אָחִ֔יו רָדַ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (טז) וְאִישׁ֙ עָמַ֣ד עָלָ֔יו מִֽנַּעֲרֵ֖י יוֹאָ֑ב וַיֹּ֗אמֶר מִי֩ אֲשֶׁ֨ר חָפֵ֧ץ בְּיוֹאָ֛ב וּמִ֥י אֲשֶׁר־ לְדָוִ֖ד אַחֲרֵ֥י יוֹאָֽב׃ (יז) וַעֲמָשָׂ֛א מִתְגֹּלֵ֥ל בַּדָּ֖ם בְּת֣וֹךְ הַֽמְסִּלָּ֑ה וַיַּ֨רְא הָאִ֜ישׁ כִּֽי־ עָמַ֣ד כָּל־ הָעָ֗ם וַיַּסֵּב֩ אֶת־ עֲמָשָׂ֨א מִן־ הַֽמְסִלָּ֤ה הַשָּׂדֶה֙ וַיַּשְׁלֵ֤ךְ עָלָיו֙ בֶּ֔גֶד כַּאֲשֶׁ֣ר רָאָ֔ה כָּל־ הַבָּ֥א עָלָ֖יו וְעָמָֽד׃ (יח) כַּאֲשֶׁ֥ר הֹגָ֖ה מִן־ הַֽמְסִלָּ֑ה עָבַ֤ר כָּל־ אִישׁ֙ אַחֲרֵ֣י יוֹאָ֔ב לִרְדֹּ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (יט) וַֽיַּעֲבֹ֞ר בְּכָל־ שִׁבְטֵ֣י יִשְׂרָאֵ֗ל אָבֵ֛לָה וּבֵ֥ית מַעֲכָ֖ה וְכָל־ הַבֵּרִ֑ים (ויקלהו) וַיִּקָּ֣הֲל֔וּ וַיָּבֹ֖אוּ אַף־ אַחֲרָֽיו׃ (כ) וַיָּבֹ֜אוּ וַיָּצֻ֣רוּ עָלָ֗יו בְּאָבֵ֙לָה֙ בֵּ֣ית הַֽמַּעֲכָ֔ה וַיִּשְׁפְּכ֤וּ סֹֽלְלָה֙ אֶל־ הָעִ֔יר וַֽתַּעֲמֹ֖ד בַּחֵ֑ל וְכָל־ הָעָם֙ אֲשֶׁ֣ר אֶת־ יוֹאָ֔ב מַשְׁחִיתִ֖ם לְהַפִּ֥יל הַחוֹמָֽה׃ (כא) וַתִּקְרָ֛א אִשָּׁ֥ה חֲכָמָ֖ה מִן־ הָעִ֑יר שִׁמְע֤וּ שִׁמְעוּ֙ אִמְרוּ־ נָ֣א אֶל־ יוֹאָ֔ב קְרַ֣ב עַד־ הֵ֔נָּה וַאֲדַבְּרָ֖ה אֵלֶֽיךָ׃ (כב) וַיִּקְרַ֣ב אֵלֶ֔יהָ וַתֹּ֧אמֶר הָאִשָּׁ֛ה הַאַתָּ֥ה יוֹאָ֖ב וַיֹּ֣אמֶר אָ֑נִי וַתֹּ֣אמֶר ל֗וֹ שְׁמַע֙ דִּבְרֵ֣י אֲמָתֶ֔ךָ וַיֹּ֖אמֶר שֹׁמֵ֥עַ אָנֹֽכִי׃ (כג) וַתֹּ֖אמֶר לֵאמֹ֑ר דַּבֵּ֨ר יְדַבְּר֤וּ בָרִֽאשֹׁנָה֙ לֵאמֹ֔ר שָׁאֹ֧ל יְשָׁאֲל֛וּ בְּאָבֵ֖ל וְכֵ֥ן הֵתַֽמּוּ׃ (כד) אָנֹכִ֕י שְׁלֻמֵ֖י אֱמוּנֵ֣י יִשְׂרָאֵ֑ל אַתָּ֣ה מְבַקֵּ֗שׁ לְהָמִ֨ית עִ֤יר וְאֵם֙ בְּיִשְׂרָאֵ֔ל לָ֥מָּה תְבַלַּ֖ע נַחֲלַ֥ת יְהוָֽה׃ (כה) וַיַּ֥עַן יוֹאָ֖ב וַיֹּאמַ֑ר חָלִ֤ילָה חָלִ֙ילָה֙ לִ֔י אִם־ אֲבַלַּ֖ע וְאִם־ אַשְׁחִֽית׃ (כו) לֹא־ כֵ֣ן הַדָּבָ֗ר כִּ֡י אִישׁ֩ מֵהַ֨ר אֶפְרַ֜יִם שֶׁ֧בַע בֶּן־ בִּכְרִ֣י שְׁמ֗וֹ נָשָׂ֤א יָדוֹ֙ בַּמֶּ֣לֶךְ בְּדָוִ֔ד תְּנֽוּ־ אֹת֣וֹ לְבַדּ֔וֹ וְאֵלְכָ֖ה מֵעַ֣ל הָעִ֑יר וַתֹּ֤אמֶר הָֽאִשָּׁה֙ אֶל־ יוֹאָ֔ב הִנֵּ֥ה רֹאשׁ֛וֹ מֻשְׁלָ֥ךְ אֵלֶ֖יךָ בְּעַ֥ד הַחוֹמָֽה׃ (כז) וַתָּבוֹא֩ הָאִשָּׁ֨ה אֶל־ כָּל־ הָעָ֜ם בְּחָכְמָתָ֗הּ וַֽיִּכְרְת֞וּ אֶת־ רֹ֨אשׁ שֶׁ֤בַע בֶּן־ בִּכְרִי֙ וַיַּשְׁלִ֣כוּ אֶל־ יוֹאָ֔ב וַיִּתְקַע֙ בַּשּׁוֹפָ֔ר וַיָּפֻ֥צוּ מֵֽעַל־ הָעִ֖יר אִ֣ישׁ לְאֹהָלָ֑יו וְיוֹאָ֛ב שָׁ֥ב יְרוּשָׁלִַ֖ם אֶל־ הַמֶּֽלֶךְ׃ (כח) וְיוֹאָ֕ב אֶ֥ל כָּל־ הַצָּבָ֖א יִשְׂרָאֵ֑ל וּבְנָיָה֙ בֶּן־ יְה֣וֹיָדָ֔ע עַל־ (הכרי) הַכְּרֵתִ֖י וְעַל־ הַפְּלֵתִֽי׃ (כט) וַאֲדֹרָ֖ם עַל־ הַמַּ֑ס וִיהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד הַמַּזְכִּֽיר׃ (ל) (ושיא) וּשְׁוָ֖א סֹפֵ֑ר וְצָד֥וֹק וְאֶבְיָתָ֖ר כֹּהֲנִֽים׃ (לא) וְגַ֗ם עִירָא֙ הַיָּ֣אִרִ֔י הָיָ֥ה כֹהֵ֖ן לְדָוִֽד׃ תחילת פרק כא (לב) וַיְהִ֣י רָעָב֩ בִּימֵ֨י דָוִ֜ד שָׁלֹ֣שׁ שָׁנִ֗ים שָׁנָה֙ אַחֲרֵ֣י שָׁנָ֔ה וַיְבַקֵּ֥שׁ דָּוִ֖ד אֶת־ פְּנֵ֣י יְהוָ֑ה וַיֹּ֣אמֶר יְהוָ֗ה אֶל־ שָׁאוּל֙ וְאֶל־ בֵּ֣ית הַדָּמִ֔ים עַל־ אֲשֶׁר־ הֵמִ֖ית אֶת־ הַגִּבְעֹנִֽים׃ (לג) וַיִּקְרָ֥א הַמֶּ֛לֶךְ לַגִּבְעֹנִ֖ים וַיֹּ֣אמֶר אֲלֵיהֶ֑ם וְהַגִּבְעֹנִ֞ים לֹ֣א מִבְּנֵ֧י יִשְׂרָאֵ֣ל הֵ֗מָּה כִּ֚י אִם־ מִיֶּ֣תֶר הָאֱמֹרִ֔י (א) וַיַּחְמֹ֣ל הַמֶּ֔לֶךְ עַל־ מְפִי־ בֹ֖שֶׁת בֶּן־ יְהוֹנָתָ֣ן בֶּן־ שָׁא֑וּל עַל־ שְׁבֻעַ֤ת יְהוָה֙ אֲשֶׁ֣ר בֵּֽינֹתָ֔ם בֵּ֣ין דָּוִ֔ד וּבֵ֖ין יְהוֹנָתָ֥ן בֶּן־ שָׁאֽוּל׃ (ב) וַיִּקַּ֣ח הַמֶּ֡לֶךְ אֶת־ שְׁ֠נֵי בְּנֵ֨י רִצְפָּ֤ה בַת־ אַיָּה֙ אֲשֶׁ֣ר יָלְדָ֣ה לְשָׁא֔וּל אֶת־ אַרְמֹנִ֖י וְאֶת־ מְפִבֹ֑שֶׁת וְאֶת־ חֲמֵ֗שֶׁת בְּנֵי֙ מִיכַ֣ל בַּת־ שָׁא֔וּל אֲשֶׁ֥ר יָלְדָ֛ה לְעַדְרִיאֵ֥ל בֶּן־ בַּרְזִלַּ֖י הַמְּחֹלָתִֽי׃ (ג) וַֽיִּתְּנֵ֞ם בְּיַ֣ד הַגִּבְעֹנִ֗ים וַיֹּקִיעֻ֤ם בָּהָר֙ לִפְנֵ֣י יְהוָ֔ה וַיִּפְּל֥וּ (שבעתים) שְׁבַעְתָּ֖ם יָ֑חַד (והם) וְהֵ֨מָּה הֻמְת֜וּ בִּימֵ֤י קָצִיר֙ בָּרִ֣אשֹׁנִ֔ים (תחלת) בִּתְחִלַּ֖ת קְצִ֥יר שְׂעֹרִֽים׃ (ד) וַתִּקַּ֣ח רִצְפָּה֩ בַת־ אַיָּ֨ה אֶת־ הַשַּׂ֜ק וַתַּטֵּ֨הוּ לָ֤הּ אֶל־ הַצּוּר֙ מִתְּחִלַּ֣ת קָצִ֔יר עַ֛ד נִתַּךְ־ מַ֥יִם עֲלֵיהֶ֖ם מִן־ הַשָּׁמָ֑יִם וְלֹֽא־ נָתְנָה֩ ע֨וֹף הַשָּׁמַ֜יִם לָנ֤וּחַ עֲלֵיהֶם֙ יוֹמָ֔ם וְאֶת־ חַיַּ֥ת הַשָּׂדֶ֖ה לָֽיְלָה׃ (ה) וַיֻּגַּ֖ד לְדָוִ֑ד אֵ֧ת אֲשֶׁר־ עָשְׂתָ֛ה רִצְפָּ֥ה בַת־ אַיָּ֖ה פִּלֶ֥גֶשׁ שָׁאֽוּל׃ (ו) וַיֵּ֣לֶךְ דָּוִ֗ד וַיִּקַּ֞ח אֶת־ עַצְמ֤וֹת שָׁאוּל֙ וְאֶת־ עַצְמוֹת֙ יְהוֹנָתָ֣ן בְּנ֔וֹ מֵאֵ֕ת בַּעֲלֵ֖י יָבֵ֣ישׁ גִּלְעָ֑ד אֲשֶׁר֩ גָּנְב֨וּ אֹתָ֜ם מֵרְחֹ֣ב בֵּֽית־ שַׁ֗ן אֲשֶׁ֨ר (תלום) תְּלָא֥וּם (שם הפלשתים) שָׁ֙מָּה֙ פְּלִשְׁתִּ֔ים בְּי֨וֹם הַכּ֧וֹת פְּלִשְׁתִּ֛ים אֶת־ שָׁא֖וּל בַּגִּלְבֹּֽעַ׃ (ז) וַיַּ֤עַל מִשָּׁם֙ אֶת־ עַצְמ֣וֹת שָׁא֔וּל וְאֶת־ עַצְמ֖וֹת יְהוֹנָתָ֣ן בְּנ֑וֹ וַיַּ֣אַסְפ֔וּ אֶת־ עַצְמ֖וֹת הַמּוּקָעִֽים׃ (ח) וַיִּקְבְּר֣וּ אֶת־ עַצְמוֹת־ שָׁא֣וּל וִיהוֹנָֽתָן־ בְּ֠נוֹ בְּאֶ֨רֶץ בִּנְיָמִ֜ן בְּצֵלָ֗ע בְּקֶ֙בֶר֙ קִ֣ישׁ אָבִ֔יו וַֽיַּעֲשׂ֔וּ כֹּ֥ל לְהַכּ֥וֹת אֶת־ דָּוִֽד׃ (יא) וַיַּֽעֲזָר־ לוֹ֙ אֲבִישַׁ֣י בֶּן־ צְרוּיָ֔ה וַיַּ֥ךְ אֶת־ הַפְּלִשְׁתִּ֖י וַיְמִיתֵ֑הוּ אָ֣ז נִשְׁבְּעוּ֩ אַנְשֵׁי־ דָוִ֨ד ל֜וֹ לֵאמֹ֗ר לֹא־ תֵצֵ֨א ע֤וֹד אִתָּ֙נוּ֙ לַמִּלְחָמָ֔ה וְלֹ֥א תְכַבֶּ֖ה אֶת־ נֵ֥ר יִשְׂרָאֵֽל׃ (יב) וַֽיְהִי֙ אַֽחֲרֵי־ כֵ֔ן וַתְּהִי־ ע֧וֹד הַמִּלְחָמָ֛ה בְּג֖וֹב עִם־ פְּלִשְׁתִּ֑ים אָ֣ז הִכָּ֗ה סִבְּכַי֙ הַחֻ֣שָׁתִ֔י אֶת־ סַ֕ף אֲשֶׁ֖ר ה֖וּא יֻלַּ֥ד לְהָרָפָֽה׃ (טו) וַיְחָרֵ֖ף אֶת־ יִשְׂרָאֵ֑ל וַיַּכֵּ֙הוּ֙ יְה֣וֹנָתָ֔ן בֶּן־ (שמעי) שִׁמְעָ֖ה אֲחִ֥י דָוִֽד׃ (טז) אֶת־ אַרְבַּ֥עַת אֵ֛לֶּה יֻלְּד֥וּ לְהָרָפָ֖ה בְּגַ֑ת וַיִּפְּל֥וּ בְיַד־ דָּוִ֖ד וּבְיַ֥ד עֲבָדָֽיו׃ סוף פרק כא (יז) וַיְדַבֵּ֤ר דָּוִד֙ לַֽיהוָ֔ה אֶת־ דִּבְרֵ֖י הַשִּׁירָ֣ה הַזֹּ֑את בְּיוֹם֩ הִצִּ֨יל יְהוָ֥ה אֹת֛וֹ מִכַּ֥ף כָּל־ אֹיְבָ֖יו וּמִכַּ֥ף שָׁאֽוּל׃ (יח) אֵֽשׁ׃ (ל) יַרְעֵ֥ם מִן־ שָׁמַ֖יִם יְהוָ֑ה וְעֶלְי֖וֹן יִתֵּ֥ן קוֹלֽוֹ׃ (לא) וַיִּשְׁלַ֥ח חִצִּ֖ים וַיְפִיצֵ֑ם בָּרָ֖ק (ויהמם) וַיָּהֹֽם׃ (לב) וַיֵּֽרָאוּ֙ אֲפִ֣קֵי יָ֔ם יִגָּל֖וּ מֹסְד֣וֹת תֵּבֵ֑ל בְּגַעֲרַ֣ת יְהוָ֔ה מִנִּשְׁמַ֖ת ר֥וּחַ אַפּֽוֹ׃ (לג) יִשְׁלַ֥ח מִמָּר֖וֹם יִקָּחֵ֑נִי יַֽמְשֵׁ֖נִי מִמַּ֥יִם רַבִּֽים׃ (לד) יַצִּילֵ֕נִי מֵאֹיְבִ֖י עָ֑ז מִשֹּׂ֣נְאַ֔י כִּ֥י אָמְצ֖וּ מִמֶּֽנִּי׃ (לה) יְקַדְּמֻ֖נִי בְּי֣וֹם אֵידִ֑י וַיְהִ֧י יְהוָ֛ה מִשְׁעָ֖ן לִֽי׃ (לו) וַיֹּצֵ֥א לַמֶּרְחָ֖ב אֹתִ֑י יְחַלְּצֵ֖נִי כִּי־ חָ֥פֵֽץ בִּֽי׃ (לז) יִגְמְלֵ֥נִי יְהוָ֖ה כְּצִדְקָתִ֑י כְּבֹ֥ר יָדַ֖י יָשִׁ֥יב לִֽי׃ (לח) כִּ֥י שָׁמַ֖רְתִּי דַּרְכֵ֣י יְהוָ֑ה וְלֹ֥א רָשַׁ֖עְתִּי מֵאֱלֹהָֽי׃ (לט) כִּ֥י כָל־ (משפטו) מִשְׁפָּטָ֖יו לְנֶגְדִּ֑י וְחֻקֹּתָ֖יו לֹא־ אָס֥וּר מִמֶּֽנָּה׃ (מ) וָאֶהְיֶ֥ה תָמִ֖ים ל֑וֹ וָאֶשְׁתַּמְּרָ֖ה מֵעֲוֺנִֽי׃ (מא) וַיָּ֧שֶׁב יְהוָ֛ה לִ֖י כְּצִדְקָתִ֑י כְּבֹרִ֖י לְנֶ֥גֶד עֵינָֽיו׃ (מב) עִם־ חָסִ֖יד תִּתְחַסָּ֑ד עִם־ גִּבּ֥וֹר תָּמִ֖ים תִּתַּמָּֽם׃ (מג) עִם־ נָבָ֖ר תִּתָּבָ֑ר וְעִם־ עִקֵּ֖שׁ תִּתַּפָּֽל׃ (מד) וְאֶת־ עַ֥ם עָנִ֖י תּוֹשִׁ֑יעַ וְעֵינֶ֖יךָ עַל־ רָמִ֥ים תַּשְׁפִּֽיל׃ (מה) כִּֽי־ אַתָּ֥ה נֵירִ֖י יְהוָ֑ה וַיהוָ֖ה יַגִּ֥יהַּ חָשְׁכִּֽי׃ (מו) כִּ֥י בְכָ֖ה אָר֣וּץ גְּד֑וּד בֵּאלֹהַ֖י אֲדַלֶּג־ שֽׁוּר׃ (מז) הָאֵ֖ל תָּמִ֣ים דַּרְכּ֑וֹ אִמְרַ֤ת יְהוָה֙ צְרוּפָ֔ה מָגֵ֣ן ה֔וּא לְכֹ֖ל הַחֹסִ֥ים בּֽוֹ׃ (מח) כִּ֥י מִי־ אֵ֖ל מִבַּלְעֲדֵ֣י יְהוָ֑ה וּמִ֥י צ֖וּר מִֽבַּלְעֲדֵ֥י אֱלֹהֵֽינוּ׃ (מט) הָאֵ֥ל מָעוּזִּ֖י חָ֑יִל וַיַּתֵּ֥ר תָּמִ֖ים (דרכו) דַּרְכִּֽי׃ (נ) מְשַׁוֶּ֥ה (רגליו) רַגְלַ֖י כָּאַיָּל֑וֹת וְעַ֥ל בָּמוֹתַ֖י יַעֲמִדֵֽנִי׃ (נא) מְלַמֵּ֥ד יָדַ֖י לַמִּלְחָמָ֑ה וְנִחַ֥ת קֶֽשֶׁת־ נְחוּשָׁ֖ה זְרֹעֹתָֽי׃ (נב) וַתִּתֶּן־ לִ֖י מָגֵ֣ן יִשְׁעֶ֑ךָ וַעֲנֹתְךָ֖ תַּרְבֵּֽנִי׃ (נג) תַּרְחִ֥יב צַעֲדִ֖י תַּחְתֵּ֑נִי וְלֹ֥א מָעֲד֖וּ קַרְסֻלָּֽי׃ (נד) אֶרְדְּפָ֥ה אֹיְבַ֖י וָאַשְׁמִידֵ֑ם וְלֹ֥א אָשׁ֖וּב עַד־ כַּלּוֹתָֽם׃ (נה) וָאֲכַלֵּ֥ם וָאֶמְחָצֵ֖ם וְלֹ֣א יְקוּמ֑וּן וַֽיִּפְּל֖וּ תַּ֥חַת רַגְלָֽי׃ (נו) וַתַּזְרֵ֥נִי חַ֖יִל לַמִּלְחָמָ֑ה תַּכְרִ֥יעַ קָמַ֖י תַּחְתֵּֽנִי׃ (נז) וְאֹ֣יְבַ֔י תַּ֥תָּה לִּ֖י עֹ֑רֶף מְשַׂנְאַ֖י וָאַצְמִיתֵֽם׃ (נח) יִשְׁע֖וּ וְאֵ֣ין מֹשִׁ֑יעַ אֶל־ יְהוָ֖ה וְלֹ֥א עָנָֽם׃ (נט) וְאֶשְׁחָקֵ֖ם כַּעֲפַר־ אָ֑רֶץ כְּטִיט־ חוּצ֥וֹת אֲדִקֵּ֖ם אֶרְקָעֵֽם׃ (ס) וַֽתְּפַלְּטֵ֔נִי מֵרִיבֵ֖י עַמִּ֑י תִּשְׁמְרֵ֙נִי֙ לְרֹ֣אשׁ גּוֹיִ֔ם עַ֥ם לֹא־ יָדַ֖עְתִּי יַעַבְדֻֽנִי׃ (סא) בְּנֵ֥י נֵכָ֖ר יִתְכַּֽחֲשׁוּ־ לִ֑י לִשְׁמ֥וֹעַ אֹ֖זֶן יִשָּׁ֥מְעוּ לִֽי׃ (סב) בְּנֵ֥י נֵכָ֖ר יִבֹּ֑לוּ וְיַחְגְּר֖וּ מִמִּסְגְּרוֹתָֽם׃ (סג) חַי־ יְהוָ֖ה וּבָר֣וּךְ צוּרִ֑י וְיָרֻ֕ם אֱלֹהֵ֖י צ֥וּר יִשְׁעִֽי׃ (סד) הָאֵ֕ל הַנֹּתֵ֥ן נְקָמֹ֖ת לִ֑י וּמוֹרִ֥יד עַמִּ֖ים תַּחְתֵּֽנִי׃ (סה) וּמוֹצִיאִ֖י מֵאֹֽיְבָ֑י וּמִקָּמַי֙ תְּר֣וֹמְמֵ֔נִי מֵאִ֥ישׁ חֲמָסִ֖ים תַּצִּילֵֽנִי׃ (סו) עַל־ כֵּ֛ן אוֹדְךָ֥ יְהוָ֖ה בַּגּוֹיִ֑ם וּלְשִׁמְךָ֖ אֲזַמֵּֽר׃
