שמואל ב פרק ה לפי החלוקה לסדרים

פרק ה בספר שמואל ב (25 פסוקים) נמצא בשמואל סדר כב (פסוקים לב–מ) ובשמואל סדר כג (פסוקים א–טז).

שמואל סדר כב

(א) וַיֹּ֣אמֶר אַבְנֵ֣ר אֶל־ דָּוִ֡ד אָק֣וּמָה ׀ וְֽאֵלֵ֡כָה וְאֶקְבְּצָה֩ אֶל־ אֲדֹנִ֨י הַמֶּ֜לֶךְ אֶת־ כָּל־ יִשְׂרָאֵ֗ל וְיִכְרְת֤וּ אִתְּךָ֙ בְּרִ֔ית וּמָ֣לַכְתָּ֔ בְּכֹ֥ל אֲשֶׁר־ תְּאַוֶּ֖ה נַפְשֶׁ֑ךָ וַיְּשַׁלַּ֥ח דָּוִ֛ד אֶת־ אַבְנֵ֖ר וַיֵּ֥לֶךְ בְּשָׁלֽוֹם׃ (ב) וְהִנֵּה֩ עַבְדֵ֨י דָוִ֤ד וְיוֹאָב֙ בָּ֣א מֵֽהַגְּד֔וּד וְשָׁלָ֥ל רָ֖ב עִמָּ֣ם הֵבִ֑יאוּ וְאַבְנֵ֗ר אֵינֶ֤נּוּ עִם־ דָּוִד֙ בְּחֶבְר֔וֹן כִּ֥י שִׁלְּח֖וֹ וַיֵּ֥לֶךְ בְּשָׁלֽוֹם׃ (ג) וְיוֹאָ֛ב וְכָל־ הַצָּבָ֥א אֲשֶׁר־ אִתּ֖וֹ בָּ֑אוּ וַיַּגִּ֤דוּ לְיוֹאָב֙ לֵאמֹ֔ר בָּֽא־ אַבְנֵ֤ר בֶּן־ נֵר֙ אֶל־ הַמֶּ֔לֶךְ וַֽיְשַׁלְּחֵ֖הוּ וַיֵּ֥לֶךְ בְּשָׁלֽוֹם׃ (ד) וַיָּבֹ֤א יוֹאָב֙ אֶל־ הַמֶּ֔לֶךְ וַיֹּ֖אמֶר מֶ֣ה עָשִׂ֑יתָה הִנֵּה־ בָ֤א אַבְנֵר֙ אֵלֶ֔יךָ לָמָּה־ זֶּ֥ה שִׁלַּחְתּ֖וֹ וַיֵּ֥לֶךְ הָלֽוֹךְ׃ (ה) יָדַ֙עְתָּ֙ אֶת־ אַבְנֵ֣ר בֶּן־ נֵ֔ר כִּ֥י לְפַתֹּתְךָ֖ בָּ֑א וְלָדַ֜עַת אֶת־ מוֹצָֽאֲךָ֙ וְאֶת־ (מבואך) מ֣וֹבָאֶ֔ךָ וְלָדַ֕עַת אֵ֛ת כָּל־ אֲשֶׁ֥ר אַתָּ֖ה עֹשֶֽׂה׃ (ו) וַיֵּצֵ֤א יוֹאָב֙ מֵעִ֣ם דָּוִ֔ד וַיִּשְׁלַ֤ח מַלְאָכִים֙ אַחֲרֵ֣י אַבְנֵ֔ר וַיָּשִׁ֥בוּ אֹת֖וֹ מִבּ֣וֹר הַסִּרָ֑ה וְדָוִ֖ד לֹ֥א יָדָֽע׃ (ז) וַיָּ֤שָׁב אַבְנֵר֙ חֶבְר֔וֹן וַיַּטֵּ֤הוּ יוֹאָב֙ אֶל־ תּ֣וֹךְ הַשַּׁ֔עַר לְדַבֵּ֥ר אִתּ֖וֹ בַּשֶּׁ֑לִי וַיַּכֵּ֤הוּ שָׁם֙ הַחֹ֔מֶשׁ וַיָּ֕מָת בְּדַ֖ם עֲשָׂה־ אֵ֥ל אָחִֽיו׃ (ח) וַיִּשְׁמַ֤ע דָּוִד֙ מֵאַ֣חֲרֵי כֵ֔ן וַיֹּ֗אמֶר נָקִ֨י אָנֹכִ֧י וּמַמְלַכְתִּ֛י מֵעִ֥ם יְהוָ֖ה עַד־ עוֹלָ֑ם מִדְּמֵ֖י אַבְנֵ֥ר בֶּן־ נֵֽר׃ (ט) יָחֻ֙לוּ֙ עַל־ רֹ֣אשׁ יוֹאָ֔ב וְאֶ֖ל כָּל־ בֵּ֣ית אָבִ֑יו וְֽאַל־ יִכָּרֵ֣ת מִבֵּ֣ית יוֹאָ֡ב זָ֠ב וּמְצֹרָ֞ע וּמַחֲזִ֥יק בַּפֶּ֛לֶךְ וְנֹפֵ֥ל בַּחֶ֖רֶב וַחֲסַר־ לָֽחֶם׃ (י) וְיוֹאָב֙ וַאֲבִישַׁ֣י אָחִ֔יו הָרְג֖וּ לְאַבְנֵ֑ר עַל֩ אֲשֶׁ֨ר הֵמִ֜ית אֶת־ עֲשָׂהאֵ֧ל אֲחִיהֶ֛ם בְּגִבְע֖וֹן בַּמִּלְחָמָֽה׃ (יא) וַיֹּאמֶר֩ דָּוִ֨ד אֶל־ יוֹאָ֜ב וְאֶל־ כָּל־ הָעָ֣ם אֲשֶׁר־ אִתּ֗וֹ קִרְע֤וּ בִגְדֵיכֶם֙ וְחִגְר֣וּ שַׂקִּ֔ים וְסִפְד֖וּ לִפְנֵ֣י אַבְנֵ֑ר וְהַמֶּ֣לֶךְ דָּוִ֔ד הֹלֵ֖ךְ אַחֲרֵ֥י הַמִּטָּֽה׃ (יב) וַיִּקְבְּר֥וּ אֶת־ אַבְנֵ֖ר בְּחֶבְר֑וֹן וַיִשָּׂ֧א הַמֶּ֣לֶךְ אֶת־ קוֹל֗וֹ וַיֵּבְךְּ֙ אֶל־ קֶ֣בֶר אַבְנֵ֔ר וַיִּבְכּ֖וּ כָּל־ הָעָֽם׃ (יג) וַיְקֹנֵ֥ן הַמֶּ֛לֶךְ אֶל־ אַבְנֵ֖ר וַיֹּאמַ֑ר הַכְּמ֥וֹת נָבָ֖ל יָמ֥וּת אַבְנֵֽר׃ (יד) יָדֶ֣ךָ לֹֽא־ אֲסֻר֗וֹת וְרַגְלֶ֙יךָ֙ לֹא־ לִנְחֻשְׁתַּ֣יִם הֻגָּ֔שׁוּ כִּנְפ֛וֹל לִפְנֵ֥י בְנֵֽי־ עַוְלָ֖ה נָפָ֑לְתָּ וַיֹּסִ֥פוּ כָל־ הָעָ֖ם לִבְכּ֥וֹת עָלָֽיו׃ (טו) וַיָּבֹ֣א כָל־ הָעָ֗ם לְהַבְר֧וֹת אֶת־ דָּוִ֛ד לֶ֖חֶם בְּע֣וֹד הַיּ֑וֹם וַיִּשָּׁבַ֨ע דָּוִ֜ד לֵאמֹ֗ר כֹּ֣ה יַעֲשֶׂה־ לִּ֤י אֱלֹהִים֙ וְכֹ֣ה יֹסִ֔יף כִּ֣י אִם־ לִפְנֵ֧י בֽוֹא־ הַשֶּׁ֛מֶשׁ אֶטְעַם־ לֶ֖חֶם א֥וֹ כָל־ מְאֽוּמָה׃ (טז) וְכָל־ הָעָ֣ם הִכִּ֔ירוּ וַיִּיטַ֖ב בְּעֵֽינֵיהֶ֑ם כְּכֹל֙ אֲשֶׁ֣ר עָשָׂ֣ה הַמֶּ֔לֶךְ בְּעֵינֵ֥י כָל־ הָעָ֖ם טֽוֹב׃ (יז) וַיֵּדְע֧וּ כָל־ הָעָ֛ם וְכָל־ יִשְׂרָאֵ֖ל בַּיּ֣וֹם הַה֑וּא כִּ֣י לֹ֤א הָיְתָה֙ מֵֽהַמֶּ֔לֶךְ לְהָמִ֖ית אֶת־ אַבְנֵ֥ר בֶּן־ נֵֽר׃ (יח) וַיֹּ֥אמֶר הַמֶּ֖לֶךְ אֶל־ עֲבָדָ֑יו הֲל֣וֹא תֵדְע֔וּ כִּי־ שַׂ֣ר וְגָד֗וֹל נָפַ֛ל הַיּ֥וֹם הַזֶּ֖ה בְּיִשְׂרָאֵֽל׃ (יט) וְאָנֹכִ֨י הַיּ֥וֹם רַךְ֙ וּמָשׁ֣וּחַ מֶ֔לֶךְ וְהָאֲנָשִׁ֥ים הָאֵ֛לֶּה בְּנֵ֥י צְרוּיָ֖ה קָשִׁ֣ים מִמֶּ֑נִּי יְשַׁלֵּ֧ם יְהוָ֛ה לְעֹשֵׂ֥ה הָרָעָ֖ה כְּרָעָתֽוֹ׃ (כ) וַיִּשְׁמַ֣ע בֶּן־ שָׁא֗וּל כִּ֣י מֵ֤ת אַבְנֵר֙ בְּחֶבְר֔וֹן וַיִּרְפּ֖וּ יָדָ֑יו וְכָל־ יִשְׂרָאֵ֖ל נִבְהָֽלוּ׃ (כא) וּשְׁנֵ֣י אֲנָשִׁ֣ים שָׂרֵֽי־ גְדוּדִ֣ים הָי֪וּ בֶן־ שָׁא֟וּל שֵׁם֩ הָאֶחָ֨ד בַּֽעֲנָ֜ה וְשֵׁ֧ם הַשֵּׁנִ֣י רֵכָ֗ב בְּנֵ֛י רִמּ֥וֹן הַבְּאֶֽרֹתִ֖י מִבְּנֵ֣י בִנְיָמִ֑ן כִּ֚י גַּם־ בְּאֵר֔וֹת תֵּחָשֵׁ֖ב עַל־ בִּנְיָמִֽן׃ (כב) וַיִּבְרְח֥וּ הַבְּאֵרֹתִ֖ים גִּתָּ֑יְמָה וַֽיִּהְיוּ־ שָׁ֣ם גָּרִ֔ים עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כג) וְלִיהֽוֹנָתָן֙ בֶּן־ שָׁא֔וּל בֵּ֖ן נְכֵ֣ה רַגְלָ֑יִם בֶּן־ חָמֵ֣שׁ שָׁנִ֣ים הָיָ֡ה בְּבֹ֣א שְׁמֻעַת֩ שָׁא֨וּל וִיהֽוֹנָתָ֜ן מִֽיִּזְרְעֶ֗אל וַתִּשָּׂאֵ֤הוּ אֹֽמַנְתּוֹ֙ וַתָּנֹ֔ס וַיְהִ֞י בְּחָפְזָ֥הּ לָנ֛וּס וַיִּפֹּ֥ל וַיִּפָּסֵ֖חַ וּשְׁמ֥וֹ מְפִיבֹֽשֶׁת׃ (כד) וַיֵּ֨לְכ֜וּ בְּנֵֽי־ רִמּ֤וֹן הַבְּאֵֽרֹתִי֙ רֵכָ֣ב וּבַעֲנָ֔ה וַיָּבֹ֙אוּ֙ כְּחֹ֣ם הַיּ֔וֹם אֶל־ בֵּ֖ית אִ֣ישׁ בֹּ֑שֶׁת וְה֣וּא שֹׁכֵ֔ב אֵ֖ת מִשְׁכַּ֥ב הַֽצָּהֳרָֽיִם׃ (כה) וְ֠הֵנָּה בָּ֜אוּ עַד־ תּ֤וֹךְ הַבַּ֙יִת֙ לֹקְחֵ֣י חִטִּ֔ים וַיַּכֻּ֖הוּ אֶל־ הַחֹ֑מֶשׁ וְרֵכָ֛ב וּבַעֲנָ֥ה אָחִ֖יו נִמְלָֽטוּ׃ (כו) וַיָּבֹ֣אוּ הַבַּ֗יִת וְהֽוּא־ שֹׁכֵ֤ב עַל־ מִטָּתוֹ֙ בַּחֲדַ֣ר מִשְׁכָּב֔וֹ וַיַּכֻּ֙הוּ֙ וַיְמִתֻ֔הוּ וַיָּסִ֖ירוּ אֶת־ רֹאשׁ֑וֹ וַיִּקְחוּ֙ אֶת־ רֹאשׁ֔וֹ וַיֵּֽלְכ֛וּ דֶּ֥רֶךְ הָעֲרָבָ֖ה כָּל־ הַלָּֽיְלָה׃ (כז) וַ֠יָּבִאוּ אֶת־ רֹ֨אשׁ אִֽישׁ־ בֹּ֥שֶׁת אֶל־ דָּוִד֮ חֶבְרוֹן֒ וַיֹּֽאמְרוּ֙ אֶל־ הַמֶּ֔לֶךְ הִנֵּֽה־ רֹ֣אשׁ אִֽישׁ־ בֹּ֗שֶׁת בֶּן־ שָׁאוּל֙ אֹֽיִבְךָ֔ אֲשֶׁ֥ר בִּקֵּ֖שׁ אֶת־ נַפְשֶׁ֑ךָ וַיִּתֵּ֣ן יְ֠הוָה לַֽאדֹנִ֨י הַמֶּ֤לֶךְ נְקָמוֹת֙ הַיּ֣וֹם הַזֶּ֔ה מִשָּׁא֖וּל וּמִזַּרְעֽוֹ׃ (כח) וַיַּ֨עַן דָּוִ֜ד אֶת־ רֵכָ֣ב ׀ וְאֶת־ בַּעֲנָ֣ה אָחִ֗יו בְּנֵ֛י רִמּ֥וֹן הַבְּאֵֽרֹתִ֖י וַיֹּ֣אמֶר לָהֶ֑ם חַי־ יְהוָ֕ה אֲשֶׁר־ פָּדָ֥ה אֶת־ נַפְשִׁ֖י מִכָּל־ צָרָֽה׃ (כט) כִּ֣י הַמַּגִּיד֩ לִ֨י לֵאמֹ֜ר הִנֵּה־ מֵ֣ת שָׁא֗וּל וְהֽוּא־ הָיָ֤ה כִמְבַשֵּׂר֙ בְּעֵינָ֔יו וָאֹחֲזָ֣ה ב֔וֹ וָאֶהְרְגֵ֖הוּ בְּצִֽקְלָ֑ג אֲשֶׁ֥ר לְתִתִּי־ ל֖וֹ בְּשֹׂרָֽה׃ (ל) אַ֞ף כִּֽי־ אֲנָשִׁ֣ים רְשָׁעִ֗ים הָרְג֧וּ אֶת־ אִישׁ־ צַדִּ֛יק בְּבֵית֖וֹ עַל־ מִשְׁכָּב֑וֹ וְעַתָּ֗ה הֲל֨וֹא אֲבַקֵּ֤שׁ אֶת־ דָּמוֹ֙ מִיֶּדְכֶ֔ם וּבִעַרְתִּ֥י אֶתְכֶ֖ם מִן־ הָאָֽרֶץ׃ (לא) וַיְצַו֩ דָּוִ֨ד אֶת־ הַנְּעָרִ֜ים וַיַּהַרְג֗וּם וַֽיְקַצְּצ֤וּ אֶת־ יְדֵיהֶם֙ וְאֶת־ רַגְלֵיהֶ֔ם וַיִּתְל֥וּ עַל־ הַבְּרֵכָ֖ה בְּחֶבְר֑וֹן וְאֵ֨ת רֹ֤אשׁ אִֽישׁ־ בֹּ֙שֶׁת֙ לָקָ֔חוּ וַיִּקְבְּר֥וּ בְקֶֽבֶר־ אַבְנֵ֖ר בְּחֶבְרֽוֹן׃ תחילת פרק ה (לב) וַיָּבֹ֜אוּ כָּל־ שִׁבְטֵ֧י יִשְׂרָאֵ֛ל אֶל־ דָּוִ֖ד חֶבְר֑וֹנָה וַיֹּאמְר֣וּ לֵאמֹ֔ר הִנְנ֛וּ עַצְמְךָ֥ וּֽבְשָׂרְךָ֖ אֲנָֽחְנוּ׃ (לג) גַּם־ אֶתְמ֣וֹל גַּם־ בֵּיתֽוֹ׃ (כח) וַיֻּגַּ֗ד לַמֶּ֣לֶךְ דָּוִד֮ לֵאמֹר֒ בֵּרַ֣ךְ יְהוָ֗ה אֶת־ בֵּ֨ית עֹבֵ֤ד אֱדֹם֙ וְאֶת־ כָּל־ אֲשֶׁר־ ל֔וֹ בַּעֲב֖וּר אֲר֣וֹן הָאֱלֹהִ֑ים וַיֵּ֣לֶךְ דָּוִ֗ד וַיַּעַל֩ אֶת־ אֲר֨וֹן הָאֱלֹהִ֜ים מִבֵּ֨ית עֹבֵ֥ד אֱדֹ֛ם עִ֥יר דָּוִ֖ד בְּשִׂמְחָֽה׃ (כט) וַיְהִ֗י כִּ֧י צָעֲד֛וּ נֹשְׂאֵ֥י אֲרוֹן־ יְהוָ֖ה שִׁשָּׁ֣ה צְעָדִ֑ים וַיִּזְבַּ֥ח שׁ֖וֹר וּמְרִֽיא׃ (ל) וְדָוִ֛ד מְכַרְכֵּ֥ר בְּכָל־ עֹ֖ז לִפְנֵ֣י יְהוָ֑ה וְדָוִ֕ד חָג֖וּר אֵפ֥וֹד בָּֽד׃ (לא) וְדָוִד֙ וְכָל־ בֵּ֣ית יִשְׂרָאֵ֔ל מַעֲלִ֖ים אֶת־ אֲר֣וֹן יְהוָ֑ה בִּתְרוּעָ֖ה וּבְק֥וֹל שׁוֹפָֽר׃ (לב) וְהָיָה֙ אֲר֣וֹן יְהוָ֔ה בָּ֖א עִ֣יר דָּוִ֑ד וּמִיכַ֨ל בַּת־ שָׁא֜וּל נִשְׁקְפָ֣ה ׀ בְּעַ֣ד הַחַלּ֗וֹן וַתֵּ֨רֶא אֶת־ הַמֶּ֤לֶךְ דָּוִד֙ מְפַזֵּ֤ז וּמְכַרְכֵּר֙ לִפְנֵ֣י יְהוָ֔ה וַתִּ֥בֶז ל֖וֹ בְּלִבָּֽהּ׃ (לג) וַיָּבִ֜אוּ אֶת־ אֲר֣וֹן יְהוָ֗ה וַיַּצִּ֤גוּ אֹתוֹ֙ בִּמְקוֹמ֔וֹ בְּת֣וֹךְ הָאֹ֔הֶל אֲשֶׁ֥ר נָטָה־ ל֖וֹ דָּוִ֑ד וַיַּ֨עַל דָּוִ֥ד עֹל֛וֹת לִפְנֵ֥י יְהוָ֖ה וּשְׁלָמִֽים׃ (לד) וַיְכַ֣ל דָּוִ֔ד מֵהַעֲל֥וֹת הָעוֹלָ֖ה וְהַשְּׁלָמִ֑ים וַיְבָ֣רֶךְ אֶת־ הָעָ֔ם בְּשֵׁ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (לה) וַיְחַלֵּ֨ק לְכָל־ הָעָ֜ם לְכָל־ הֲמ֣וֹן יִשְׂרָאֵל֮ לְמֵאִ֣ישׁ וְעַד־ אִשָּׁה֒ לְאִ֗ישׁ חַלַּ֥ת לֶ֙חֶם֙ אַחַ֔ת וְאֶשְׁפָּ֣ר אֶחָ֔ד וַאֲשִׁישָׁ֖ה אֶחָ֑ת וַיֵּ֥לֶךְ כָּל־ הָעָ֖ם אִ֥ישׁ לְבֵיתֽוֹ׃ (לו) וַיָּ֥שָׁב דָּוִ֖ד לְבָרֵ֣ךְ אֶת־ בֵּית֑וֹ וַתֵּצֵ֞א מִיכַ֤ל בַּת־ שָׁאוּל֙ לִקְרַ֣את דָּוִ֔ד וַתֹּ֗אמֶר מַה־ נִּכְבַּ֨ד הַיּ֜וֹם מֶ֣לֶךְ יִשְׂרָאֵ֗ל אֲשֶׁ֨ר נִגְלָ֤ה הַיּוֹם֙ לְעֵינֵ֨י אַמְה֣וֹת עֲבָדָ֔יו כְּהִגָּל֥וֹת נִגְל֖וֹת אַחַ֥ד הָרֵקִֽים׃ (לז) וַיֹּ֣אמֶר דָּוִד֮ אֶל־ מִיכַל֒ לִפְנֵ֣י יְהוָ֗ה אֲשֶׁ֨ר בָּֽחַר־ בִּ֤י מֵֽאָבִיךְ֙ וּמִכָּל־ בֵּית֔וֹ לְצַוֺּ֨ת אֹתִ֥י נָגִ֛יד עַל־ עַ֥ם יְהוָ֖ה עַל־ יִשְׂרָאֵ֑ל וְשִׂחַקְתִּ֖י לִפְנֵ֥י יְהוָֽה׃ (לח) וּנְקַלֹּ֤תִי עוֹד֙ מִזֹּ֔את וְהָיִ֥יתִי שָׁפָ֖ל בְּעֵינָ֑י וְעִם־ הָֽאֲמָהוֹת֙ אֲשֶׁ֣ר אָמַ֔רְתְּ עִמָּ֖ם אִכָּבֵֽדָה׃ (לט) וּלְמִיכַל֙ בַּת־ שָׁא֔וּל לֹֽא־ הָ֥יָה לָ֖הּ יָ֑לֶד עַ֖ד י֥וֹם מוֹתָֽהּ׃ (מ) וַיְהִ֕י כִּי־ יָשַׁ֥ב הַמֶּ֖לֶךְ בְּבֵית֑וֹ וַיהוָ֛ה הֵנִֽיחַ־ ל֥וֹ מִסָּבִ֖יב מִכָּל־ אֹיְבָֽיו׃ (מא) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ נָתָ֣ן הַנָּבִ֔יא רְאֵ֣ה נָ֔א אָנֹכִ֥י יוֹשֵׁ֖ב בְּבֵ֣ית אֲרָזִ֑ים וַֽאֲרוֹן֙ הָֽאֱלֹהִ֔ים יֹשֵׁ֖ב בְּת֥וֹךְ הַיְרִיעָֽה׃ (מב) וַיֹּ֤אמֶר נָתָן֙ אֶל־ הַמֶּ֔לֶךְ כֹּ֛ל אֲשֶׁ֥ר בִּֽלְבָבְךָ֖ לֵ֣ךְ עֲשֵׂ֑ה כִּ֥י יְהוָ֖ה עִמָּֽךְ׃ (מג) וַיְהִ֖י בַּלַּ֣יְלָה הַה֑וּא וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ נָתָ֖ן לֵאמֹֽר׃ (מד) לֵ֤ךְ וְאָֽמַרְתָּ֙ אֶל־ עַבְדִּ֣י אֶל־ דָּוִ֔ד כֹּ֖ה אָמַ֣ר יְהוָ֑ה הַאַתָּ֛ה תִּבְנֶה־ לִּ֥י בַ֖יִת לְשִׁבְתִּֽי׃ (מה) כִּ֣י לֹ֤א יָשַׁ֙בְתִּי֙ בְּבַ֔יִת לְ֠מִיּוֹם הַעֲלֹתִ֞י אֶת־ בְּנֵ֤י יִשְׂרָאֵל֙ מִמִּצְרַ֔יִם וְעַ֖ד הַיּ֣וֹם הַזֶּ֑ה וָאֶֽהְיֶה֙ מִתְהַלֵּ֔ךְ בְּאֹ֖הֶל וּבְמִשְׁכָּֽן׃ (מו) בְּכֹ֥ל אֲשֶֽׁר־ הִתְהַלַּכְתִּי֮ בְּכָל־ בְּנֵ֣י יִשְׂרָאֵל֒ הֲדָבָ֣ר דִּבַּ֗רְתִּי אֶת־ אַחַד֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֣ר צִוִּ֗יתִי לִרְע֛וֹת אֶת־ עַמִּ֥י אֶת־ יִשְׂרָאֵ֖ל לֵאמֹ֑ר לָ֛מָּה לֹֽא־ בְנִיתֶ֥ם לִ֖י בֵּ֥ית אֲרָזִֽים׃ (מז) וְ֠עַתָּה כֹּֽה־ תֹאמַ֞ר לְעַבְדִּ֣י לְדָוִ֗ד כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת אֲנִ֤י לְקַחְתִּ֙יךָ֙ מִן־ הַנָּוֶ֔ה מֵאַחַ֖ר הַצֹּ֑אן לִֽהְי֣וֹת נָגִ֔יד עַל־ עַמִּ֖י עַל־ יִשְׂרָאֵֽל׃ (מח) וָאֶהְיֶ֣ה עִמְּךָ֗ בְּכֹל֙ אֲשֶׁ֣ר הָלַ֔כְתָּ וָאַכְרִ֥תָה אֶת־ כָּל־ אֹיְבֶ֖יךָ מִפָּנֶ֑יךָ וְעָשִׂ֤תִֽי לְךָ֙ שֵׁ֣ם גָּד֔וֹל כְּשֵׁ֥ם הַגְּדֹלִ֖ים אֲשֶׁ֥ר בָּאָֽרֶץ׃ (מט) וְשַׂמְתִּ֣י מָ֠קוֹם לְעַמִּ֨י לְיִשְׂרָאֵ֤ל וּנְטַעְתִּיו֙ וְשָׁכַ֣ן תַּחְתָּ֔יו וְלֹ֥א יִרְגַּ֖ז ע֑וֹד וְלֹֽא־ יֹסִ֤יפוּ בְנֵֽי־ עַוְלָה֙ לְעַנּוֹת֔וֹ כַּאֲשֶׁ֖ר בָּרִאשׁוֹנָֽה׃ (נ) וּלְמִן־ הַיּ֗וֹם אֲשֶׁ֨ר צִוִּ֤יתִי שֹֽׁפְטִים֙ עַל־ עַמִּ֣י יִשְׂרָאֵ֔ל וַהֲנִיחֹ֥תִי לְךָ֖ מִכָּל־ אֹיְבֶ֑יךָ וְהִגִּ֤יד לְךָ֙ יְהוָ֔ה כִּי־ בַ֖יִת יַעֲשֶׂה־ לְּךָ֥ יְהוָֽה׃ (נא) כִּ֣י ׀ יִמְלְא֣וּ יָמֶ֗יךָ וְשָֽׁכַבְתָּ֙ אֶת־ אֲבֹתֶ֔יךָ וַהֲקִימֹתִ֤י אֶֽת־ זַרְעֲךָ֙ אַחֲרֶ֔יךָ אֲשֶׁ֥ר יֵצֵ֖א מִמֵּעֶ֑יךָ וַהֲכִינֹתִ֖י אֶת־ מַמְלַכְתּֽוֹ׃ (נב) ה֥וּא יִבְנֶה־ בַּ֖יִת לִשְׁמִ֑י וְכֹנַנְתִּ֛י אֶת־ כִּסֵּ֥א מַמְלַכְתּ֖וֹ עַד־ עוֹלָֽם׃ (נג) אֲנִי֙ אֶהְיֶה־ לּ֣וֹ לְאָ֔ב וְה֖וּא יִהְיֶה־ לִּ֣י לְבֵ֑ן אֲשֶׁר֙ בְּהַ֣עֲוֺת֔וֹ וְהֹֽכַחְתִּיו֙ בְּשֵׁ֣בֶט אֲנָשִׁ֔ים וּבְנִגְעֵ֖י בְּנֵ֥י אָדָֽם׃ (נד) וְחַסְדִּ֖י לֹא־ יָס֣וּר מִמֶּ֑נּוּ כַּאֲשֶׁ֤ר הֲסִרֹ֙תִי֙ מֵעִ֣ם שָׁא֔וּל אֲשֶׁ֥ר הֲסִרֹ֖תִי מִלְּפָנֶֽיךָ׃
