שמואל ב פרק יז לפי החלוקה לסדרים

פרק יז בספר שמואל ב (29 פסוקים) נמצא בשמואל סדר כט (פסוקים כה–מג) ובשמואל סדר ל (פסוקים א–י).

שמואל סדר כט

(א) וַיָּבֹ֥א חוּשַׁ֛י רֵעֶ֥ה דָוִ֖ד הָעִ֑יר וְאַבְשָׁלֹ֔ם יָבֹ֖א יְרוּשָׁלִָֽם׃ (ב) וְדָוִ֗ד עָבַ֤ר מְעַט֙ מֵֽהָרֹ֔אשׁ וְהִנֵּ֥ה צִיבָ֛א נַ֥עַר מְפִי־ בֹ֖שֶׁת לִקְרָאת֑וֹ וְצֶ֨מֶד חֲמֹרִ֜ים חֲבֻשִׁ֗ים וַעֲלֵיהֶם֩ מָאתַ֨יִם לֶ֜חֶם וּמֵאָ֧ה צִמּוּקִ֛ים וּמֵ֥אָה קַ֖יִץ וְנֵ֥בֶל יָֽיִן׃ (ג) וַיֹּ֧אמֶר הַמֶּ֛לֶךְ אֶל־ צִיבָ֖א מָה־ אֵ֣לֶּה לָּ֑ךְ וַיֹּ֣אמֶר צִ֠יבָא הַחֲמוֹרִ֨ים לְבֵית־ הַמֶּ֜לֶךְ לִרְכֹּ֗ב (ולהלחם) וְהַלֶּ֤חֶם וְהַקַּ֙יִץ֙ לֶאֱכ֣וֹל הַנְּעָרִ֔ים וְהַיַּ֕יִן לִשְׁתּ֥וֹת הַיָּעֵ֖ף בַּמִּדְבָּֽר׃ (ד) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ וְאַיֵּ֖ה בֶּן־ אֲדֹנֶ֑יךָ וַיֹּ֨אמֶר צִיבָ֜א אֶל־ הַמֶּ֗לֶךְ הִנֵּה֙ יוֹשֵׁ֣ב בִּירוּשָׁלִַ֔ם כִּ֣י אָמַ֔ר הַיּ֗וֹם יָשִׁ֤יבוּ לִי֙ בֵּ֣ית יִשְׂרָאֵ֔ל אֵ֖ת מַמְלְכ֥וּת אָבִֽי׃ (ה) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ לְצִבָ֔א הִנֵּ֣ה לְךָ֔ כֹּ֖ל אֲשֶׁ֣ר לִמְפִי־ בֹ֑שֶׁת וַיֹּ֤אמֶר צִיבָא֙ הִֽשְׁתַּחֲוֵ֔יתִי אֶמְצָא־ חֵ֥ן בְּעֵינֶ֖יךָ אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (ו) וּבָ֛א הַמֶּ֥לֶךְ דָּוִ֖ד עַד־ בַּֽחוּרִ֑ים וְהִנֵּ֣ה מִשָּׁם֩ אִ֨ישׁ יוֹצֵ֜א מִמִּשְׁפַּ֣חַת בֵּית־ שָׁא֗וּל וּשְׁמוֹ֙ שִׁמְעִ֣י בֶן־ גֵּרָ֔א יֹצֵ֥א יָצ֖וֹא וּמְקַלֵּֽל׃ (ז) וַיְסַקֵּ֤ל בָּֽאֲבָנִים֙ אֶת־ דָּוִ֔ד וְאֶת־ כָּל־ עַבְדֵ֖י הַמֶּ֣לֶךְ דָּוִ֑ד וְכָל־ הָעָם֙ וְכָל־ הַגִּבֹּרִ֔ים מִימִינ֖וֹ וּמִשְּׂמֹאלֽוֹ׃ (ח) וְכֹֽה־ אָמַ֥ר שִׁמְעִ֖י בְּקַֽלְל֑וֹ צֵ֥א צֵ֛א אִ֥ישׁ הַדָּמִ֖ים וְאִ֥ישׁ הַבְּלִיָּֽעַל׃ (ט) הֵשִׁיב֩ עָלֶ֨יךָ יְהוָ֜ה כֹּ֣ל ׀ דְּמֵ֣י בֵית־ שָׁא֗וּל אֲשֶׁ֤ר מָלַ֙כְתָּ֙ (תחתו) תַּחְתָּ֔יו וַיִּתֵּ֤ן יְהוָה֙ אֶת־ הַמְּלוּכָ֔ה בְּיַ֖ד אַבְשָׁל֣וֹם בְּנֶ֑ךָ וְהִנְּךָ֙ בְּרָ֣עָתֶ֔ךָ כִּ֛י אִ֥ישׁ דָּמִ֖ים אָֽתָּה׃ (י) וַיֹּ֨אמֶר אֲבִישַׁ֤י בֶּן־ צְרוּיָה֙ אֶל־ הַמֶּ֔לֶךְ לָ֣מָּה יְקַלֵּ֞ל הַכֶּ֤לֶב הַמֵּת֙ הַזֶּ֔ה אֶת־ אֲדֹנִ֖י הַמֶּ֑לֶךְ אֶעְבְּרָה־ נָּ֖א וְאָסִ֥ירָה אֶת־ רֹאשֽׁוֹ׃ (יא) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ מַה־ לִּ֥י וְלָכֶ֖ם בְּנֵ֣י צְרֻיָ֑ה (כי) כֹּ֣ה יְקַלֵּ֗ל (וכי) כִּ֤י יְהוָה֙ אָ֤מַר לוֹ֙ קַלֵּ֣ל אֶת־ דָּוִ֔ד וּמִ֣י יֹאמַ֔ר מַדּ֖וּעַ עָשִׂ֥יתָה כֵּֽן׃ (יב) וַיֹּ֨אמֶר דָּוִ֤ד אֶל־ אֲבִישַׁי֙ וְאֶל־ כָּל־ עֲבָדָ֔יו הִנֵּ֥ה בְנִ֛י אֲשֶׁר־ יָצָ֥א מִמֵּעַ֖י מְבַקֵּ֣שׁ אֶת־ נַפְשִׁ֑י וְאַ֨ף כִּֽי־ עַתָּ֜ה בֶּן־ הַיְמִינִ֗י הַנִּ֤חוּ לוֹ֙ וִֽיקַלֵּ֔ל כִּ֥י אָֽמַר־ ל֖וֹ יְהוָֽה׃ (יג) אוּלַ֛י יִרְאֶ֥ה יְהוָ֖ה (בעוני) בְּעֵינִ֑י וְהֵשִׁ֨יב יְהוָ֥ה לִי֙ טוֹבָ֔ה תַּ֥חַת קִלְלָת֖וֹ הַיּ֥וֹם הַזֶּֽה׃ (יד) וַיֵּ֧לֶךְ דָּוִ֛ד וַאֲנָשָׁ֖יו בַּדָּ֑רֶךְ וְשִׁמְעִ֡י הֹלֵךְ֩ בְּצֵ֨לַע הָהָ֜ר לְעֻמָּת֗וֹ הָלוֹךְ֙ וַיְקַלֵּ֔ל וַיְסַקֵּ֤ל בָּֽאֲבָנִים֙ לְעֻמָּת֔וֹ וְעִפַּ֖ר בֶּעָפָֽר׃ (טו) וַיָּבֹ֥א הַמֶּ֛לֶךְ וְכָל־ הָעָ֥ם אֲשֶׁר־ אִתּ֖וֹ עֲיֵפִ֑ים וַיִּנָּפֵ֖שׁ שָֽׁם׃ (טז) וְאַבְשָׁל֗וֹם וְכָל־ הָעָם֙ אִ֣ישׁ יִשְׂרָאֵ֔ל בָּ֖אוּ יְרוּשָׁלִָ֑ם וַאֲחִיתֹ֖פֶל אִתּֽוֹ׃ (יז) וַיְהִ֗י כַּֽאֲשֶׁר־ בָּ֞א חוּשַׁ֧י הָאַרְכִּ֛י רֵעֶ֥ה דָוִ֖ד אֶל־ אַבְשָׁל֑וֹם וַיֹּ֤אמֶר חוּשַׁי֙ אֶל־ אַבְשָׁלֹ֔ם יְחִ֥י הַמֶּ֖לֶךְ יְחִ֥י הַמֶּֽלֶךְ׃ (יח) וַיֹּ֤אמֶר אַבְשָׁלוֹם֙ אֶל־ חוּשַׁ֔י זֶ֥ה חַסְדְּךָ֖ אֶת־ רֵעֶ֑ךָ לָ֥מָּה לֹֽא־ הָלַ֖כְתָּ אֶת־ רֵעֶֽךָ׃ (יט) וַיֹּ֣אמֶר חוּשַׁי֮ אֶל־ אַבְשָׁלֹם֒ לֹ֕א כִּי֩ אֲשֶׁ֨ר בָּחַ֧ר יְהוָ֛ה וְהָעָ֥ם הַזֶּ֖ה וְכָל־ אִ֣ישׁ יִשְׂרָאֵ֑ל (לא) ל֥וֹ אֶהְיֶ֖ה וְאִתּ֥וֹ אֵשֵֽׁב׃ (כ) וְהַשֵּׁנִ֗ית לְמִי֙ אֲנִ֣י אֶֽעֱבֹ֔ד הֲל֖וֹא לִפְנֵ֣י בְנ֑וֹ כַּאֲשֶׁ֤ר עָבַ֙דְתִּי֙ לִפְנֵ֣י אָבִ֔יךָ כֵּ֖ן אֶהְיֶ֥ה לְפָנֶֽיךָ׃ (כא) וַיֹּ֥אמֶר אַבְשָׁל֖וֹם אֶל־ אֲחִיתֹ֑פֶל הָב֥וּ לָכֶ֛ם עֵצָ֖ה מַֽה־ נַּעֲשֶֽׂה׃ (כב) וַיֹּ֤אמֶר אֲחִיתֹ֙פֶל֙ אֶל־ אַבְשָׁלֹ֔ם בּ֚וֹא אֶל־ פִּלַגְשֵׁ֣י אָבִ֔יךָ אֲשֶׁ֥ר הִנִּ֖יחַ לִשְׁמ֣וֹר הַבָּ֑יִת וְשָׁמַ֤ע כָּל־ יִשְׂרָאֵל֙ כִּֽי־ נִבְאַ֣שְׁתָּ אֶת־ אָבִ֔יךָ וְחָ֣זְק֔וּ יְדֵ֖י כָּל־ אֲשֶׁ֥ר אִתָּֽךְ׃ (כג) וַיַּטּ֧וּ לְאַבְשָׁל֛וֹם הָאֹ֖הֶל עַל־ הַגָּ֑ג וַיָּבֹ֤א אַבְשָׁלוֹם֙ אֶל־ פִּֽלַגְשֵׁ֣י אָבִ֔יו לְעֵינֵ֖י כָּל־ יִשְׂרָאֵֽל׃ (כד) וַעֲצַ֣ת אֲחִיתֹ֗פֶל אֲשֶׁ֤ר יָעַץ֙ בַּיָּמִ֣ים הָהֵ֔ם כַּאֲשֶׁ֥ר יִשְׁאַל־ אִ֖ישׁ בִּדְבַ֣ר הָאֱלֹהִ֑ים כֵּ֚ן כָּל־ עֲצַ֣ת אֲחִיתֹ֔פֶל גַּם־ לְדָוִ֖ד גַּ֥ם לְאַבְשָׁלֹֽם׃ תחילת פרק יז (כה) וַיֹּ֥אמֶר אֲחִיתֹ֖פֶל אֶל־ אַבְשָׁלֹ֑ם אֶבְחֲרָ֣ה נָּ֗א שְׁנֵים־ עָשָׂ֥ר אֶ֙לֶף֙ אִ֔ישׁ וְאָק֛וּמָה וְאֶרְדְּפָ֥ה אַחֲרֵי־ דָוִ֖ד הַלָּֽיְלָה׃ (כו) וְאָב֣וֹא עָלָ֗יו וְה֤וּא יָגֵ֙עַ֙ וּרְפֵ֣ה יָדַ֔יִם וְהַֽחֲרַדְתִּ֣י אֹת֔וֹ וְנָ֖ס כָּל־ הָעָ֣ם אֲשֶׁר־ אִתּ֑וֹ וְהִכֵּיתִ֥י אֶת־ הַמֶּ֖לֶךְ לְבַדּֽוֹ׃ (כז) וְאָשִׁ֥יבָה כָל־ הָעָ֖ם אֵלֶ֑יךָ כְּשׁ֣וּב הַכֹּ֔ל הָאִישׁ֙ אֲשֶׁ֣ר אַתָּ֣ה מְבַקֵּ֔שׁ כָּל־ הָעָ֖ם יִהְיֶ֥ה שָׁלֽוֹם׃ (כח) וַיִּישַׁ֥ר הַדָּבָ֖ר בְּעֵינֵ֣י אַבְשָׁלֹ֑ם וּבְעֵינֵ֖י כָּל־ זִקְנֵ֥י יִשְׂרָאֵֽל׃ (כט) וַיֹּ֙אמֶר֙ אַבְשָׁל֔וֹם קְרָ֣א נָ֔א גַּ֖ם לְחוּשַׁ֣י הָאַרְכִּ֑י וְנִשְׁמְעָ֥ה מַה־ בְּפִ֖יו גַּם־ הֽוּא׃ (ל) וַיָּבֹ֣א חוּשַׁי֮ אֶל־ אַבְשָׁלוֹם֒ וַיֹּאמֶר֩ אַבְשָׁל֨וֹם אֵלָ֜יו לֵאמֹ֗ר כַּדָּבָ֤ר הַזֶּה֙ דִּבֶּ֣ר אֲחִיתֹ֔פֶל הֲנַעֲשֶׂ֖ה אֶת־ דְּבָר֑וֹ אִם־ אַ֖יִן אַתָּ֥ה דַבֵּֽר׃ (לא) וַיֹּ֥אמֶר חוּשַׁ֖י אֶל־ אַבְשָׁל֑וֹם לֹֽא־ טוֹבָ֧ה הָעֵצָ֛ה אֲשֶׁר־ יָעַ֥ץ אֲחִיתֹ֖פֶל בַּפַּ֥עַם הַזֹּֽאת׃ (לב) וַיֹּ֣אמֶר חוּשַׁ֗י אַתָּ֣ה יָ֠דַעְתָּ אֶת־ אָבִ֨יךָ וְאֶת־ אֲנָשָׁ֜יו כִּ֧י גִבֹּרִ֣ים הֵ֗מָּה וּמָרֵ֥י נֶ֙פֶשׁ֙ הֵ֔מָּה כְּדֹ֥ב שַׁכּ֖וּל בַּשָּׂדֶ֑ה וְאָבִ֙יךָ֙ אִ֣ישׁ מִלְחָמָ֔ה וְלֹ֥א יָלִ֖ין אֶת־ הָעָֽם׃ (לג) הִנֵּ֨ה עַתָּ֤ה הֽוּא־ נֶחְבָּא֙ בְּאַחַ֣ת הַפְּחָתִ֔ים א֖וֹ בְּאַחַ֣ד הַמְּקוֹמֹ֑ת וְהָיָ֗ה כִּנְפֹ֤ל בָּהֶם֙ בַּתְּחִלָּ֔ה וְשָׁמַ֤ע הַשֹּׁמֵ֙עַ֙ וְאָמַ֔ר הָֽיְתָה֙ מַגֵּפָ֔ה בָּעָ֕ם אֲשֶׁ֖ר אַחֲרֵ֥י אַבְשָׁלֹֽם׃ (לד) וְה֣וּא גַם־ בֶּן־ חַ֗יִל אֲשֶׁ֥ר לִבּ֛וֹ כְּלֵ֥ב הָאַרְיֵ֖ה הִמֵּ֣ס יִמָּ֑ס כִּֽי־ יֹדֵ֤עַ כָּל־ יִשְׂרָאֵל֙ כִּי־ גִבּ֣וֹר אָבִ֔יךָ וּבְנֵי־ חַ֖יִל אֲשֶׁ֥ר אִתּֽוֹ׃ (לה) כִּ֣י יָעַ֗צְתִּי הֵ֠אָסֹף יֵאָסֵ֨ף עָלֶ֤יךָ כָל־ יִשְׂרָאֵל֙ מִדָּן֙ וְעַד־ בְּאֵ֣ר שֶׁ֔בַע כַּח֥וֹל אֲשֶׁר־ עַל־ הַיָּ֖ם לָרֹ֑ב וּפָנֶ֥יךָ הֹלְכִ֖ים בַּקְרָֽב׃ (לו) וּבָ֣אנוּ אֵלָ֗יו (באחת) בְּאַחַ֤ד הַמְּקוֹמֹת֙ אֲשֶׁ֣ר נִמְצָ֣א שָׁ֔ם וְנַ֣חְנוּ עָלָ֔יו כַּאֲשֶׁ֛ר יִפֹּ֥ל הַטַּ֖ל עַל־ הָאֲדָמָ֑ה וְלֹֽא־ נ֥וֹתַר בּ֛וֹ וּבְכָל־ הָאֲנָשִׁ֥ים אֲשֶׁר־ אִתּ֖וֹ גַּם־ אֶחָֽד׃ (לז) וְאִם־ אֶל־ עִיר֙ לֹֽא־ הִכִּית֥וֹ שָׁ֖ם אָ֑רְצָה וְעָלַ֗י לָ֤תֶת לְךָ֙ עֲשָׂ֣רָה כֶ֔סֶף וַחֲגֹרָ֖ה אֶחָֽת׃ (כב) וַיֹּ֤אמֶר הָאִישׁ֙ אֶל־ יוֹאָ֔ב (ולא) וְל֨וּא אָנֹכִ֜י שֹׁקֵ֤ל עַל־ כַּפַּי֙ אֶ֣לֶף כֶּ֔סֶף לֹֽא־ אֶשְׁלַ֥ח יָדִ֖י אֶל־ בֶּן־ הַמֶּ֑לֶךְ כִּ֣י בְאָזְנֵ֜ינוּ צִוָּ֣ה הַמֶּ֗לֶךְ אֹ֠תְךָ וְאֶת־ אֲבִישַׁ֤י וְאֶת־ אִתַּי֙ לֵאמֹ֔ר שִׁמְרוּ־ מִ֕י בַּנַּ֖עַר בְּאַבְשָׁלֽוֹם׃ (כג) אֽוֹ־ עָשִׂ֤יתִי (בנפשו) בְנַפְשִׁי֙ שֶׁ֔קֶר וְכָל־ דָּבָ֖ר לֹא־ יִכָּחֵ֣ד מִן־ הַמֶּ֑לֶךְ וְאַתָּ֖ה תִּתְיַצֵּ֥ב מִנֶּֽגֶד׃ (כד) וַיֹּ֣אמֶר יוֹאָ֔ב לֹא־ כֵ֖ן אֹחִ֣ילָה לְפָנֶ֑יךָ וַיִּקַּח֩ שְׁלֹשָׁ֨ה שְׁבָטִ֜ים בְּכַפּ֗וֹ וַיִּתְקָעֵם֙ בְּלֵ֣ב אַבְשָׁל֔וֹם עוֹדֶ֥נּוּ חַ֖י בְּלֵ֥ב הָאֵלָֽה׃ (כה) וַיָּסֹ֙בּוּ֙ עֲשָׂרָ֣ה נְעָרִ֔ים נֹשְׂאֵ֖י כְּלֵ֣י יוֹאָ֑ב וַיַּכּ֥וּ אֶת־ אַבְשָׁל֖וֹם וַיְמִיתֻֽהוּ׃ (כו) וַיִּתְקַ֤ע יוֹאָב֙ בַּשֹּׁפָ֔ר וַיָּ֣שָׁב הָעָ֔ם מִרְדֹ֖ף אַחֲרֵ֣י יִשְׂרָאֵ֑ל כִּֽי־ חָשַׂ֥ךְ יוֹאָ֖ב אֶת־ הָעָֽם׃ (כז) וַיִּקְח֣וּ אֶת־ אַבְשָׁל֗וֹם וַיַּשְׁלִ֨יכוּ אֹת֤וֹ בַיַּ֙עַר֙ אֶל־ הַפַּ֣חַת הַגָּד֔וֹל וַיַּצִּ֧בוּ עָלָ֛יו גַּל־ אֲבָנִ֖ים גָּד֣וֹל מְאֹ֑ד וְכָל־ יִשְׂרָאֵ֔ל נָ֖סוּ אִ֥ישׁ (לאהלו) לְאֹהָלָֽיו׃ (כח) וְאַבְשָׁלֹ֣ם לָקַ֗ח וַיַּצֶּב־ ל֤וֹ (בחיו) בְחַיָּיו֙ אֶת־ מַצֶּ֙בֶת֙ אֲשֶׁ֣ר בְּעֵֽמֶק־ הַמֶּ֔לֶךְ כִּ֤י אָמַר֙ אֵֽין־ לִ֣י בֵ֔ן בַּעֲב֖וּר הַזְכִּ֣יר שְׁמִ֑י וַיִּקְרָ֤א לַמַּצֶּ֙בֶת֙ עַל־ שְׁמ֔וֹ וַיִּקָּ֤רֵא לָהּ֙ יַ֣ד אַבְשָׁלֹ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כט) וַאֲחִימַ֤עַץ בֶּן־ צָדוֹק֙ אָמַ֔ר אָר֣וּצָה נָּ֔א וַאֲבַשְּׂרָ֖ה אֶת־ הַמֶּ֑לֶךְ כִּי־ שְׁפָט֥וֹ יְהוָ֖ה מִיַּ֥ד אֹיְבָֽיו׃ (ל) וַיֹּ֧אמֶר ל֣וֹ יוֹאָ֗ב לֹא֩ אִ֨ישׁ בְּשֹׂרָ֤ה אַתָּה֙ הַיּ֣וֹם הַזֶּ֔ה וּבִשַּׂרְתָּ֖ בְּי֣וֹם אַחֵ֑ר וְהַיּ֤וֹם הַזֶּה֙ לֹ֣א תְבַשֵּׂ֔ר כִּֽי־ (על) עַל־ כֵּ֥ן בֶּן־ הַמֶּ֖לֶךְ מֵֽת׃ (לא) וַיֹּ֤אמֶר יוֹאָב֙ לַכּוּשִׁ֔י לֵ֛ךְ הַגֵּ֥ד לַמֶּ֖לֶךְ אֲשֶׁ֣ר רָאִ֑יתָה וַיִּשְׁתַּ֧חוּ כוּשִׁ֛י לְיוֹאָ֖ב וַיָּרֹֽץ׃ (לב) וַיֹּ֨סֶף ע֜וֹד אֲחִימַ֤עַץ בֶּן־ צָדוֹק֙ וַיֹּ֣אמֶר אֶל־ יוֹאָ֔ב וִ֣יהִי מָ֔ה אָרֻֽצָה־ נָּ֥א גַם־ אָ֖נִי אַחֲרֵ֣י הַכּוּשִׁ֑י וַיֹּ֣אמֶר יוֹאָ֗ב לָֽמָּה־ זֶּ֞ה אַתָּ֥ה רָץ֙ בְּנִ֔י וּלְכָ֖ה אֵין־ בְּשׂוֹרָ֥ה מֹצֵֽאת׃ (לג) וִיהִי־ מָ֣ה אָר֔וּץ וַיֹּ֥אמֶר ל֖וֹ ר֑וּץ וַיָּ֤רָץ אֲחִימַ֙עַץ֙ דֶּ֣רֶךְ הַכִּכָּ֔ר וַֽיַּעֲבֹ֖ר אֶת־ הַכּוּשִֽׁי׃ (לד) וְדָוִ֥ד יוֹשֵׁ֖ב בֵּין־ שְׁנֵ֣י הַשְּׁעָרִ֑ים וַיֵּ֨לֶךְ הַצֹּפֶ֜ה אֶל־ גַּ֤ג הַשַּׁ֙עַר֙ אֶל־ הַ֣חוֹמָ֔ה וַיִּשָּׂ֤א אֶת־ עֵינָיו֙ וַיַּ֔רְא וְהִנֵּה־ אִ֖ישׁ רָ֥ץ לְבַדּֽוֹ׃ (לה) וַיִּקְרָ֤א הַצֹּפֶה֙ וַיַּגֵּ֣ד לַמֶּ֔לֶךְ וַיֹּ֣אמֶר הַמֶּ֔לֶךְ אִם־ לְבַדּ֖וֹ בְּשׂוֹרָ֣ה בְּפִ֑יו וַיֵּ֥לֶךְ הָל֖וֹךְ וְקָרֵֽב׃ (לו) וַיַּ֣רְא הַצֹּפֶה֮ אִישׁ־ אַחֵ֣ר רָץ֒ וַיִּקְרָ֤א הַצֹּפֶה֙ אֶל־ הַשֹּׁעֵ֔ר וַיֹּ֕אמֶר הִנֵּה־ אִ֖ישׁ רָ֣ץ לְבַדּ֑וֹ וַיֹּ֥אמֶר הַמֶּ֖לֶךְ גַּם־ זֶ֥ה מְבַשֵּֽׂר׃
