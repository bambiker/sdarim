שמואל ב פרק יט לפי החלוקה לסדרים

פרק יט בספר שמואל ב (44 פסוקים) נמצא בשמואל סדר לא (פסוקים ז–מה) ובשמואל סדר לב (פסוקים א–ה).

שמואל סדר לא

(א) וַיֹּ֙אמֶר֙ הַצֹּפֶ֔ה אֲנִ֤י רֹאֶה֙ אֶת־ מְרוּצַ֣ת הָרִאשׁ֔וֹן כִּמְרֻצַ֖ת אֲחִימַ֣עַץ בֶּן־ צָד֑וֹק וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אִֽישׁ־ ט֣וֹב זֶ֔ה וְאֶל־ בְּשׂוֹרָ֥ה טוֹבָ֖ה יָבֽוֹא׃ (ב) וַיִּקְרָ֣א אֲחִימַ֗עַץ וַיֹּ֤אמֶר אֶל־ הַמֶּ֙לֶךְ֙ שָׁל֔וֹם וַיִּשְׁתַּ֧חוּ לַמֶּ֛לֶךְ לְאַפָּ֖יו אָ֑רְצָה וַיֹּ֗אמֶר בָּרוּךְ֙ יְהוָ֣ה אֱלֹהֶ֔יךָ אֲשֶׁ֤ר סִגַּר֙ אֶת־ הָ֣אֲנָשִׁ֔ים אֲשֶׁר־ נָשְׂא֥וּ אֶת־ יָדָ֖ם בַּֽאדֹנִ֥י הַמֶּֽלֶךְ׃ (ג) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ שָׁל֥וֹם לַנַּ֖עַר לְאַבְשָׁל֑וֹם וַיֹּ֣אמֶר אֲחִימַ֡עַץ רָאִיתִי֩ הֶהָמ֨וֹן הַגָּד֜וֹל לִ֠שְׁלֹחַ אֶת־ עֶ֨בֶד הַמֶּ֤לֶךְ יוֹאָב֙ וְאֶת־ עַבְדֶּ֔ךָ וְלֹ֥א יָדַ֖עְתִּי מָֽה׃ (ד) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ סֹ֖ב הִתְיַצֵּ֣ב כֹּ֑ה וַיִּסֹּ֖ב וַֽיַּעֲמֹֽד׃ (ה) וְהִנֵּ֥ה הַכּוּשִׁ֖י בָּ֑א וַיֹּ֣אמֶר הַכּוּשִׁ֗י יִתְבַּשֵּׂר֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ כִּֽי־ שְׁפָטְךָ֤ יְהוָה֙ הַיּ֔וֹם מִיַּ֖ד כָּל־ הַקָּמִ֥ים עָלֶֽיךָ׃ (ו) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ הַכּוּשִׁ֔י הֲשָׁל֥וֹם לַנַּ֖עַר לְאַבְשָׁל֑וֹם וַיֹּ֣אמֶר הַכּוּשִׁ֗י יִהְי֤וּ כַנַּ֙עַר֙ אֹֽיְבֵי֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ וְכֹ֛ל אֲשֶׁר־ קָ֥מוּ עָלֶ֖יךָ לְרָעָֽה׃ תחילת פרק יט (ז) וַיִּרְגַּ֣ז הַמֶּ֗לֶךְ וַיַּ֛עַל עַל־ עֲלִיַּ֥ת הַשַּׁ֖עַר וַיֵּ֑בְךְּ וְכֹ֣ה ׀ אָמַ֣ר בְּלֶכְתּ֗וֹ בְּנִ֤י אַבְשָׁלוֹם֙ בְּנִ֣י בְנִ֣י אַבְשָׁל֔וֹם מִֽי־ יִתֵּ֤ן מוּתִי֙ אֲנִ֣י תַחְתֶּ֔יךָ אַבְשָׁל֖וֹם בְּנִ֥י בְנִֽי׃ (ח) וַיֻּגַּ֖ד לְיוֹאָ֑ב הִנֵּ֨ה הַמֶּ֧לֶךְ בֹּכֶ֛ה וַיִּתְאַבֵּ֖ל עַל־ אַבְשָׁלֹֽם׃ (ט) וַתְּהִ֨י הַתְּשֻׁעָ֜ה בַּיּ֥וֹם הַה֛וּא לְאֵ֖בֶל לְכָל־ הָעָ֑ם כִּֽי־ שָׁמַ֣ע הָעָ֗ם בַּיּ֤וֹם הַהוּא֙ לֵאמֹ֔ר נֶעֱצַ֥ב הַמֶּ֖לֶךְ עַל־ בְּנֽוֹ׃ (י) וַיִּתְגַּנֵּ֥ב הָעָ֛ם בַּיּ֥וֹם הַה֖וּא לָב֣וֹא הָעִ֑יר כַּאֲשֶׁ֣ר יִתְגַּנֵּ֗ב הָעָם֙ הַנִּכְלָמִ֔ים בְּנוּסָ֖ם בַּמִּלְחָמָֽה׃ (יא) וְהַמֶּ֙לֶךְ֙ לָאַ֣ט אֶת־ פָּנָ֔יו וַיִּזְעַ֥ק הַמֶּ֖לֶךְ ק֣וֹל גָּד֑וֹל בְּנִי֙ אַבְשָׁל֔וֹם אַבְשָׁל֖וֹם בְּנִ֥י בְנִֽי׃ (יב) וַיָּבֹ֥א יוֹאָ֛ב אֶל־ הַמֶּ֖לֶךְ הַבָּ֑יִת וַיֹּאמֶר֩ הֹבַ֨שְׁתָּ הַיּ֜וֹם אֶת־ פְּנֵ֣י כָל־ עֲבָדֶ֗יךָ הַֽמְמַלְּטִ֤ים אֶֽת־ נַפְשְׁךָ֙ הַיּ֔וֹם וְאֵ֨ת נֶ֤פֶשׁ בָּנֶ֙יךָ֙ וּבְנֹתֶ֔יךָ וְנֶ֣פֶשׁ נָשֶׁ֔יךָ וְנֶ֖פֶשׁ פִּלַגְשֶֽׁיךָ׃ (יג) לְאַֽהֲבָה֙ אֶת־ שֹׂ֣נְאֶ֔יךָ וְלִשְׂנֹ֖א אֶת־ אֹהֲבֶ֑יךָ כִּ֣י ׀ הִגַּ֣דְתָּ הַיּ֗וֹם כִּ֣י אֵ֤ין לְךָ֙ שָׂרִ֣ים וַעֲבָדִ֔ים כִּ֣י ׀ יָדַ֣עְתִּי הַיּ֗וֹם כִּ֠י (לא) ל֣וּ אַבְשָׁל֥וֹם חַי֙ וְכֻלָּ֤נוּ הַיּוֹם֙ מֵתִ֔ים כִּי־ אָ֖ז יָשָׁ֥ר בְּעֵינֶֽיךָ׃ (יד) וְעַתָּה֙ ק֣וּם צֵ֔א וְדַבֵּ֖ר עַל־ לֵ֣ב עֲבָדֶ֑יךָ כִּי֩ בַיהוָ֨ה נִשְׁבַּ֜עְתִּי כִּי־ אֵינְךָ֣ יוֹצֵ֗א אִם־ יָלִ֨ין אִ֤ישׁ אִתְּךָ֙ הַלַּ֔יְלָה וְרָעָ֧ה לְךָ֣ זֹ֗את מִכָּל־ הָרָעָה֙ אֲשֶׁר־ בָּ֣אָה עָלֶ֔יךָ מִנְּעֻרֶ֖יךָ עַד־ עָֽתָּה׃ (טו) וַיָּ֥קָם הַמֶּ֖לֶךְ וַיֵּ֣שֶׁב בַּשָּׁ֑עַר וּֽלְכָל־ הָעָ֞ם הִגִּ֣ידוּ לֵאמֹ֗ר הִנֵּ֤ה הַמֶּ֙לֶךְ֙ יוֹשֵׁ֣ב בַּשַּׁ֔עַר וַיָּבֹ֤א כָל־ הָעָם֙ לִפְנֵ֣י הַמֶּ֔לֶךְ וְיִשְׂרָאֵ֔ל נָ֖ס אִ֥ישׁ לְאֹהָלָֽיו׃ (טז) וַיְהִ֤י כָל־ הָעָם֙ נָד֔וֹן בְּכָל־ שִׁבְטֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֑ר הַמֶּ֜לֶךְ הִצִּילָ֣נוּ ׀ מִכַּ֣ף אֹיְבֵ֗ינוּ וְה֤וּא מִלְּטָ֙נוּ֙ מִכַּ֣ף פְּלִשְׁתִּ֔ים וְעַתָּ֛ה בָּרַ֥ח מִן־ הָאָ֖רֶץ מֵעַ֥ל אַבְשָׁלֽוֹם׃ (יז) וְאַבְשָׁלוֹם֙ אֲשֶׁ֣ר מָשַׁ֣חְנוּ עָלֵ֔ינוּ מֵ֖ת בַּמִּלְחָמָ֑ה וְעַתָּ֗ה לָמָ֥ה אַתֶּ֛ם מַחֲרִשִׁ֖ים לְהָשִׁ֥יב אֶת־ הַמֶּֽלֶךְ׃ (יח) וְהַמֶּ֣לֶךְ דָּוִ֗ד שָׁ֠לַח אֶל־ צָד֨וֹק וְאֶל־ אֶבְיָתָ֥ר הַכֹּהֲנִים֮ לֵאמֹר֒ דַּבְּר֞וּ אֶל־ זִקְנֵ֤י יְהוּדָה֙ לֵאמֹ֔ר לָ֤מָּה תִֽהְיוּ֙ אַֽחֲרֹנִ֔ים לְהָשִׁ֥יב אֶת־ הַמֶּ֖לֶךְ אֶל־ בֵּית֑וֹ וּדְבַר֙ כָּל־ יִשְׂרָאֵ֔ל בָּ֥א אֶל־ הַמֶּ֖לֶךְ אֶל־ בֵּיתֽוֹ׃ (יט) אַחַ֣י אַתֶּ֔ם עַצְמִ֥י וּבְשָׂרִ֖י אַתֶּ֑ם וְלָ֧מָּה תִהְי֛וּ אַחֲרֹנִ֖ים לְהָשִׁ֥יב אֶת־ הַמֶּֽלֶךְ׃ (כ) וְלַֽעֲמָשָׂא֙ תֹּֽמְר֔וּ הֲל֛וֹא עַצְמִ֥י וּבְשָׂרִ֖י אָ֑תָּה כֹּ֣ה יַֽעֲשֶׂה־ לִּ֤י אֱלֹהִים֙ וְכֹ֣ה יוֹסִ֔יף אִם־ לֹ֠א שַׂר־ צָבָ֞א תִּהְיֶ֧ה לְפָנַ֛י כָּל־ הַיָּמִ֖ים תַּ֥חַת יוֹאָֽב׃ (כא) וַיַּ֛ט אֶת־ לְבַ֥ב כָּל־ אִישׁ־ יְהוּדָ֖ה כְּאִ֣ישׁ אֶחָ֑ד וַֽיִּשְׁלְחוּ֙ אֶל־ הַמֶּ֔לֶךְ שׁ֥וּב אַתָּ֖ה וְכָל־ עֲבָדֶֽיךָ׃ (כב) וַיָּ֣שָׁב הַמֶּ֔לֶךְ וַיָּבֹ֖א עַד־ הַיַּרְדֵּ֑ן וִיהוּדָ֞ה בָּ֣א הַגִּלְגָּ֗לָה לָלֶ֙כֶת֙ לִקְרַ֣את הַמֶּ֔לֶךְ לְהַעֲבִ֥יר אֶת־ הַמֶּ֖לֶךְ אֶת־ הַיַּרְדֵּֽן׃ (כג) וַיְמַהֵ֗ר שִׁמְעִ֤י בֶן־ גֵּרָא֙ בֶּן־ הַיְמִינִ֔י אֲשֶׁ֖ר מִבַּֽחוּרִ֑ים וַיֵּ֙רֶד֙ עִם־ אִ֣ישׁ יְהוּדָ֔ה לִקְרַ֖את הַמֶּ֥לֶךְ דָּוִֽד׃ (כד) וְאֶ֨לֶף אִ֣ישׁ עִמּוֹ֮ מִבִּנְיָמִן֒ וְצִיבָ֗א נַ֚עַר בֵּ֣ית שָׁא֔וּל וַחֲמֵ֨שֶׁת עָשָׂ֥ר בָּנָ֛יו וְעֶשְׂרִ֥ים עֲבָדָ֖יו אִתּ֑וֹ וְצָלְח֥וּ הַיַּרְדֵּ֖ן לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כה) וְעָבְרָ֣ה הָעֲבָרָ֗ה לַֽעֲבִיר֙ אֶת־ בֵּ֣ית הַמֶּ֔לֶךְ וְלַעֲשׂ֥וֹת הַטּ֖וֹב (בעינו) בְּעֵינָ֑יו וְשִׁמְעִ֣י בֶן־ גֵּרָ֗א נָפַל֙ לִפְנֵ֣י הַמֶּ֔לֶךְ בְּעָבְר֖וֹ בַּיַּרְדֵּֽן׃ (כו) וַיֹּ֣אמֶר אֶל־ הַמֶּ֗לֶךְ אַל־ יַחֲשָׁב־ לִ֣י אֲדֹנִי֮ עָוֺן֒ וְאַל־ תִּזְכֹּ֗ר אֵ֚ת אֲשֶׁ֣ר הֶעֱוָ֣ה עַבְדְּךָ֔ בַּיּ֕וֹם אֲשֶׁר־ יָׄצָ֥ׄאׄ אֲדֹנִֽי־ הַמֶּ֖לֶךְ מִירֽוּשָׁלִָ֑ם לָשׂ֥וּם הַמֶּ֖לֶךְ אֶל־ לִבּֽוֹ׃ (כז) כִּ֚י יָדַ֣ע עַבְדְּךָ֔ כִּ֖י אֲנִ֣י חָטָ֑אתִי וְהִנֵּֽה־ בָ֣אתִי הַיּ֗וֹם רִאשׁוֹן֙ לְכָל־ בֵּ֣ית יוֹסֵ֔ף לָרֶ֕דֶת לִקְרַ֖את אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (כח) וַיַּ֨עַן אֲבִישַׁ֤י בֶּן־ צְרוּיָה֙ וַיֹּ֔אמֶר הֲתַ֣חַת זֹ֔את לֹ֥א יוּמַ֖ת שִׁמְעִ֑י כִּ֥י קִלֵּ֖ל אֶת־ מְשִׁ֥יחַ יְהוָֽה׃ (כט) וַיֹּ֣אמֶר דָּוִ֗ד מַה־ לִּ֤י וְלָכֶם֙ בְּנֵ֣י צְרוּיָ֔ה כִּי־ תִֽהְיוּ־ לִ֥י הַיּ֖וֹם לְשָׂטָ֑ן הַיּ֗וֹם י֤וּמַת אִישׁ֙ בְּיִשְׂרָאֵ֔ל כִּ֚י הֲל֣וֹא יָדַ֔עְתִּי כִּ֥י הַיּ֖וֹם אֲנִי־ מֶ֥לֶךְ עַל־ יִשְׂרָאֵֽל׃ (ל) וַיֹּ֧אמֶר הַמֶּ֛לֶךְ אֶל־ שִׁמְעִ֖י לֹ֣א תָמ֑וּת וַיִּשָּׁ֥בַֽע ל֖וֹ הַמֶּֽלֶךְ׃ (לא) וּמְפִבֹ֙שֶׁת֙ בֶּן־ שָׁא֔וּל יָרַ֖ד לִקְרַ֣את הַמֶּ֑לֶךְ וְלֹא־ עָשָׂ֨ה רַגְלָ֜יו וְלֹא־ עָשָׂ֣ה שְׂפָמ֗וֹ וְאֶת־ בְּגָדָיו֙ לֹ֣א כִבֵּ֔ס לְמִן־ הַיּוֹם֙ לֶ֣כֶת הַמֶּ֔לֶךְ עַד־ הַיּ֖וֹם אֲשֶׁר־ בָּ֥א בְשָׁלֽוֹם׃ (לב) וַיְהִ֛י כִּי־ בָ֥א יְרוּשָׁלִַ֖ם לִקְרַ֣את הַמֶּ֑לֶךְ וַיֹּ֤אמֶר לוֹ֙ הַמֶּ֔לֶךְ לָ֛מָּה לֹא־ הָלַ֥כְתָּ עִמִּ֖י מְפִיבֹֽשֶׁת׃ (לג) וַיֹּאמַ֕ר אֲדֹנִ֥י הַמֶּ֖לֶךְ עַבְדִּ֣י רִמָּ֑נִי כִּֽי־ אָמַ֨ר עַבְדְּךָ֜ אֶחְבְּשָׁה־ לִּי֩ הַחֲמ֨וֹר וְאֶרְכַּ֤ב עָלֶ֙יהָ֙ וְאֵלֵ֣ךְ אֶת־ הַמֶּ֔לֶךְ כִּ֥י פִסֵּ֖חַ עַבְדֶּֽךָ׃ (לד) וַיְרַגֵּ֣ל בְּעַבְדְּךָ֔ אֶל־ אֲדֹנִ֖י הַמֶּ֑לֶךְ וַאדֹנִ֤י הַמֶּ֙לֶךְ֙ כְּמַלְאַ֣ךְ הָאֱלֹהִ֔ים וַעֲשֵׂ֥ה הַטּ֖וֹב בְּעֵינֶֽיךָ׃ (לה) כִּי֩ לֹ֨א הָיָ֜ה כָּל־ בֵּ֣ית אָבִ֗י כִּ֤י אִם־ אַנְשֵׁי־ מָ֙וֶת֙ לַאדֹנִ֣י הַמֶּ֔לֶךְ וַתָּ֙שֶׁת֙ אֶֽת־ עַבְדְּךָ֔ בְּאֹכְלֵ֖י שֻׁלְחָנֶ֑ךָ וּמַה־ יֶּשׁ־ לִ֥י עוֹד֙ צְדָקָ֔ה וְלִזְעֹ֥ק ע֖וֹד אֶל־ הַמֶּֽלֶךְ׃ (לו) וַיֹּ֤אמֶר לוֹ֙ הַמֶּ֔לֶךְ לָ֛מָּה תְּדַבֵּ֥ר ע֖וֹד דְּבָרֶ֑יךָ אָמַ֕רְתִּי אַתָּ֣ה וְצִיבָ֔א תַּחְלְק֖וּ אֶת־ הַשָּׂדֶֽה׃ (לז) וַיֹּ֤אמֶר מְפִיבֹ֙שֶׁת֙ אֶל־ הַמֶּ֔לֶךְ גַּ֥ם אֶת־ הַכֹּ֖ל יִקָּ֑ח אַ֠חֲרֵי אֲשֶׁר־ בָּ֞א אֲדֹנִ֥י הַמֶּ֛לֶך בְּשָׁל֖וֹם אֶל־ בֵּיתֽוֹ׃ (לח) וּבַרְזִלַּי֙ הַגִּלְעָדִ֔י יָרַ֖ד מֵרֹגְלִ֑ים וַיַּעֲבֹ֤ר אֶת־ הַמֶּ֙לֶךְ֙ הַיַּרְדֵּ֔ן לְשַׁלְּח֖וֹ אֶת־ (בירדן) הַיַּרְדֵּֽן׃ (לט) וּבַרְזִלַּי֙ זָקֵ֣ן מְאֹ֔ד בֶּן־ שְׁמֹנִ֖ים שָׁנָ֑ה וְהֽוּא־ כִלְכַּ֤ל אֶת־ הַמֶּ֙לֶךְ֙ בְּשִׁיבָת֣וֹ בְמַחֲנַ֔יִם כִּֽי־ אִ֛ישׁ גָּד֥וֹל ה֖וּא מְאֹֽד׃ (מ) וַיֹּ֥אמֶר הַמֶּ֖לֶךְ אֶל־ בַּרְזִלָּ֑י אַתָּה֙ עֲבֹ֣ר אִתִּ֔י וְכִלְכַּלְתִּ֥י אֹתְךָ֛ עִמָּדִ֖י בִּירוּשָׁלִָֽם׃ (מא) וַיֹּ֥אמֶר בַּרְזִלַּ֖י אֶל־ הַמֶּ֑לֶךְ כַּמָּ֗ה יְמֵי֙ שְׁנֵ֣י חַיַּ֔י כִּי־ אֶעֱלֶ֥ה אֶת־ הַמֶּ֖לֶךְ יְרוּשָׁלִָֽם׃ (מב) בֶּן־ שְׁמֹנִ֣ים שָׁנָה֩ אָנֹכִ֨י הַיּ֜וֹם הַאֵדַ֣ע ׀ בֵּין־ ט֣וֹב לְרָ֗ע אִם־ יִטְעַ֤ם עַבְדְּךָ֙ אֶת־ אֲשֶׁ֤ר אֹכַל֙ וְאֶת־ אֲשֶׁ֣ר אֶשְׁתֶּ֔ה אִם־ אֶשְׁמַ֣ע ע֔וֹד בְּק֖וֹל שָׁרִ֣ים וְשָׁר֑וֹת וְלָמָּה֩ יִֽהְיֶ֨ה עַבְדְּךָ֥ עוֹד֙ לְמַשָּׂ֔א אֶל־ אֲדֹנִ֖י הַמֶּֽלֶךְ׃ (מג) כִּמְעַ֞ט יַעֲבֹ֧ר עַבְדְּךָ֛ אֶת־ הַיַּרְדֵּ֖ן אֶת־ הַמֶּ֑לֶךְ וְלָ֙מָּה֙ יִגְמְלֵ֣נִי הַמֶּ֔לֶךְ הַגְּמוּלָ֖ה הַזֹּֽאת׃ (מד) יָֽשָׁב־ נָ֤א עַבְדְּךָ֙ וְאָמֻ֣ת בְּעִירִ֔י עִ֛ם קֶ֥בֶר אָבִ֖י וְאִמִּ֑י וְהִנֵּ֣ה ׀ עַבְדְּךָ֣ כִמְהָ֗ם יַֽעֲבֹר֙ עִם־ אֲדֹנִ֣י הַמֶּ֔לֶךְ וַעֲשֵׂה־ ל֕וֹ אֵ֥ת אֲשֶׁר־ ט֖וֹב בְּעֵינֶֽיךָ׃ (מה) וַיֹּ֣אמֶר הַמֶּ֗לֶךְ אִתִּי֙ יַעֲבֹ֣ר כִּמְהָ֔ם וַאֲנִי֙ אֶעֱשֶׂה־ לּ֔וֹ אֶת־ הַטּ֖וֹב בְּעֵינֶ֑יךָ וְכֹ֛ל אֲשֶׁר־ תִּבְחַ֥ר עָלַ֖י אֶֽעֱשֶׂה־ לָּֽךְ׃

(א) וַיַּעֲבֹ֧ר כָּל־ הָעָ֛ם אֶת־ הַיַּרְדֵּ֖ן וְהַמֶּ֣לֶךְ עָבָ֑ר וַיִּשַּׁ֨ק הַמֶּ֤לֶךְ לְבַרְזִלַּי֙ וַיְבָ֣רֲכֵ֔הוּ וַיָּ֖שָׁב לִמְקֹמֽוֹ׃ (ב) וַיַּעֲבֹ֤ר הַמֶּ֙לֶךְ֙ הַגִּלְגָּ֔לָה וְכִמְהָ֖ן עָבַ֣ר עִמּ֑וֹ וְכָל־ עַ֤ם יְהוּדָה֙ (ויעברו) הֶעֱבִ֣ירוּ אֶת־ הַמֶּ֔לֶךְ וְגַ֕ם חֲצִ֖י עַ֥ם יִשְׂרָאֵֽל׃ (ג) וְהִנֵּ֛ה כָּל־ אִ֥ישׁ יִשְׂרָאֵ֖ל בָּאִ֣ים אֶל־ הַמֶּ֑לֶךְ וַיֹּאמְר֣וּ אֶל־ הַמֶּ֡לֶךְ מַדּוּעַ֩ גְּנָב֨וּךָ אַחֵ֜ינוּ אִ֣ישׁ יְהוּדָ֗ה וַיַּעֲבִ֨רוּ אֶת־ הַמֶּ֤לֶךְ וְאֶת־ בֵּיתוֹ֙ אֶת־ הַיַּרְדֵּ֔ן וְכָל־ אַנְשֵׁ֥י דָוִ֖ד עִמּֽוֹ׃ (ד) וַיַּעַן֩ כָּל־ אִ֨ישׁ יְהוּדָ֜ה עַל־ אִ֣ישׁ יִשְׂרָאֵ֗ל כִּֽי־ קָר֤וֹב הַמֶּ֙לֶךְ֙ אֵלַ֔י וְלָ֤מָּה זֶּה֙ חָרָ֣ה לְךָ֔ עַל־ הַדָּבָ֖ר הַזֶּ֑ה הֶאָכ֤וֹל אָכַ֙לְנוּ֙ מִן־ הַמֶּ֔לֶךְ אִם־ נִשֵּׂ֥את נִשָּׂ֖א לָֽנוּ׃ (ה) וַיַּ֣עַן אִֽישׁ־ יִשְׂרָאֵל֩ אֶת־ אִ֨ישׁ יְהוּדָ֜ה וַיֹּ֗אמֶר עֶֽשֶׂר־ יָד֨וֹת לִ֣י בַמֶּלֶךְ֮ וְגַם־ בְּדָוִד֮ אֲנִ֣י מִמְּךָ֒ וּמַדּ֙וּעַ֙ הֱקִלֹּתַ֔נִי וְלֹא־ הָיָ֨ה דְבָרִ֥י רִאשׁ֛וֹן לִ֖י לְהָשִׁ֣יב אֶת־ מַלְכִּ֑י וַיִּ֙קֶשׁ֙ דְּבַר־ אִ֣ישׁ יְהוּדָ֔ה מִדְּבַ֖ר אִ֥ישׁ יִשְׂרָאֵֽל׃ סוף פרק יט (ו) וְשָׁ֨ם נִקְרָ֜א אִ֣ישׁ בְּלִיַּ֗עַל וּשְׁמ֛וֹ שֶׁ֥בַע בֶּן־ בִּכְרִ֖י אִ֣ישׁ יְמִינִ֑י וַיִּתְקַ֣ע הַמּוֹעֵ֖ד אֲשֶׁ֥ר יְעָדֽוֹ׃ (יא) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ אֲבִישַׁ֔י עַתָּ֗ה יֵ֧רַֽע לָ֛נוּ שֶׁ֥בַע בֶּן־ בִּכְרִ֖י מִן־ אַבְשָׁל֑וֹם אַ֠תָּה קַ֞ח אֶת־ עַבְדֵ֤י אֲדֹנֶ֙יךָ֙ וּרְדֹ֣ף אַחֲרָ֔יו פֶּן־ מָ֥צָא ל֛וֹ עָרִ֥ים בְּצֻר֖וֹת וְהִצִּ֥יל עֵינֵֽנוּ׃ (יב) וַיֵּצְא֤וּ אַֽחֲרָיו֙ אַנְשֵׁ֣י יוֹאָ֔ב וְהַכְּרֵתִ֥י וְהַפְּלֵתִ֖י וְכָל־ הַגִּבֹּרִ֑ים וַיֵּֽצְאוּ֙ מִיר֣וּשָׁלִַ֔ם לִרְדֹּ֕ף אַחֲרֵ֖י מָתְנָיו֙ בְּתַעְרָ֔הּ וְה֥וּא יָצָ֖א וַתִּפֹּֽל׃ (יד) וַיֹּ֤אמֶר יוֹאָב֙ לַעֲמָשָׂ֔א הֲשָׁל֥וֹם אַתָּ֖ה אָחִ֑י וַתֹּ֜חֶז יַד־ יְמִ֥ין יוֹאָ֛ב בִּזְקַ֥ן עֲמָשָׂ֖א לִנְשָׁק־ לֽוֹ׃ (טו) וַעֲמָשָׂ֨א לֹֽא־ נִשְׁמַ֜ר בַּחֶ֣רֶב ׀ אֲשֶׁ֣ר בְּיַד־ יוֹאָ֗ב וַיַּכֵּהוּ֩ בָ֨הּ אֶל־ הַחֹ֜מֶשׁ וַיִּשְׁפֹּ֨ךְ מֵעָ֥יו אַ֛רְצָה וְלֹא־ שָׁ֥נָה ל֖וֹ וַיָּמֹ֑ת וְיוֹאָב֙ וַאֲבִישַׁ֣י אָחִ֔יו רָדַ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (טז) וְאִישׁ֙ עָמַ֣ד עָלָ֔יו מִֽנַּעֲרֵ֖י יוֹאָ֑ב וַיֹּ֗אמֶר מִי֩ אֲשֶׁ֨ר חָפֵ֧ץ בְּיוֹאָ֛ב וּמִ֥י אֲשֶׁר־ לְדָוִ֖ד אַחֲרֵ֥י יוֹאָֽב׃ (יז) וַעֲמָשָׂ֛א מִתְגֹּלֵ֥ל בַּדָּ֖ם בְּת֣וֹךְ הַֽמְסִּלָּ֑ה וַיַּ֨רְא הָאִ֜ישׁ כִּֽי־ עָמַ֣ד כָּל־ הָעָ֗ם וַיַּסֵּב֩ אֶת־ עֲמָשָׂ֨א מִן־ הַֽמְסִלָּ֤ה הַשָּׂדֶה֙ וַיַּשְׁלֵ֤ךְ עָלָיו֙ בֶּ֔גֶד כַּאֲשֶׁ֣ר רָאָ֔ה כָּל־ הַבָּ֥א עָלָ֖יו וְעָמָֽד׃ (יח) כַּאֲשֶׁ֥ר הֹגָ֖ה מִן־ הַֽמְסִלָּ֑ה עָבַ֤ר כָּל־ אִישׁ֙ אַחֲרֵ֣י יוֹאָ֔ב לִרְדֹּ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (יט) וַֽיַּעֲבֹ֞ר בְּכָל־ שִׁבְטֵ֣י יִשְׂרָאֵ֗ל אָבֵ֛לָה וּבֵ֥ית מַעֲכָ֖ה וְכָל־ הַבֵּרִ֑ים (ויקלהו) וַיִּקָּ֣הֲל֔וּ וַיָּבֹ֖אוּ אַף־ אַחֲרָֽיו׃ (כ) וַיָּבֹ֜אוּ וַיָּצֻ֣רוּ עָלָ֗יו בְּאָבֵ֙לָה֙ בֵּ֣ית הַֽמַּעֲכָ֔ה וַיִּשְׁפְּכ֤וּ סֹֽלְלָה֙ אֶל־ הָעִ֔יר וַֽתַּעֲמֹ֖ד בַּחֵ֑ל וְכָל־ הָעָם֙ אֲשֶׁ֣ר אֶת־ יוֹאָ֔ב מַשְׁחִיתִ֖ם לְהַפִּ֥יל הַחוֹמָֽה׃ (כא) וַתִּקְרָ֛א אִשָּׁ֥ה חֲכָמָ֖ה מִן־ הָעִ֑יר שִׁמְע֤וּ שִׁמְעוּ֙ אִמְרוּ־ נָ֣א אֶל־ יוֹאָ֔ב קְרַ֣ב עַד־ הֵ֔נָּה וַאֲדַבְּרָ֖ה אֵלֶֽיךָ׃ (כב) וַיִּקְרַ֣ב אֵלֶ֔יהָ וַתֹּ֧אמֶר הָאִשָּׁ֛ה הַאַתָּ֥ה יוֹאָ֖ב וַיֹּ֣אמֶר אָ֑נִי וַתֹּ֣אמֶר ל֗וֹ שְׁמַע֙ דִּבְרֵ֣י אֲמָתֶ֔ךָ וַיֹּ֖אמֶר שֹׁמֵ֥עַ אָנֹֽכִי׃ (כג) וַתֹּ֖אמֶר לֵאמֹ֑ר דַּבֵּ֨ר יְדַבְּר֤וּ בָרִֽאשֹׁנָה֙ לֵאמֹ֔ר שָׁאֹ֧ל יְשָׁאֲל֛וּ בְּאָבֵ֖ל וְכֵ֥ן הֵתַֽמּוּ׃ (כד) אָנֹכִ֕י שְׁלֻמֵ֖י אֱמוּנֵ֣י יִשְׂרָאֵ֑ל אַתָּ֣ה מְבַקֵּ֗שׁ לְהָמִ֨ית עִ֤יר וְאֵם֙ בְּיִשְׂרָאֵ֔ל לָ֥מָּה תְבַלַּ֖ע נַחֲלַ֥ת יְהוָֽה׃ (כה) וַיַּ֥עַן יוֹאָ֖ב וַיֹּאמַ֑ר חָלִ֤ילָה חָלִ֙ילָה֙ לִ֔י אִם־ אֲבַלַּ֖ע וְאִם־ אַשְׁחִֽית׃ (כו) לֹא־ כֵ֣ן הַדָּבָ֗ר כִּ֡י אִישׁ֩ מֵהַ֨ר אֶפְרַ֜יִם שֶׁ֧בַע בֶּן־ בִּכְרִ֣י שְׁמ֗וֹ נָשָׂ֤א יָדוֹ֙ בַּמֶּ֣לֶךְ בְּדָוִ֔ד תְּנֽוּ־ אֹת֣וֹ לְבַדּ֔וֹ וְאֵלְכָ֖ה מֵעַ֣ל הָעִ֑יר וַתֹּ֤אמֶר הָֽאִשָּׁה֙ אֶל־ יוֹאָ֔ב הִנֵּ֥ה רֹאשׁ֛וֹ מֻשְׁלָ֥ךְ אֵלֶ֖יךָ בְּעַ֥ד הַחוֹמָֽה׃ (כז) וַתָּבוֹא֩ הָאִשָּׁ֨ה אֶל־ כָּל־ הָעָ֜ם בְּחָכְמָתָ֗הּ וַֽיִּכְרְת֞וּ אֶת־ רֹ֨אשׁ שֶׁ֤בַע בֶּן־ בִּכְרִי֙ וַיַּשְׁלִ֣כוּ אֶל־ יוֹאָ֔ב וַיִּתְקַע֙ בַּשּׁוֹפָ֔ר וַיָּפֻ֥צוּ מֵֽעַל־ הָעִ֖יר אִ֣ישׁ לְאֹהָלָ֑יו וְיוֹאָ֛ב שָׁ֥ב יְרוּשָׁלִַ֖ם אֶל־ הַמֶּֽלֶךְ׃ (כח) וְיוֹאָ֕ב אֶ֥ל כָּל־ הַצָּבָ֖א יִשְׂרָאֵ֑ל וּבְנָיָה֙ בֶּן־ יְה֣וֹיָדָ֔ע עַל־ (הכרי) הַכְּרֵתִ֖י וְעַל־ הַפְּלֵתִֽי׃ (כט) וַאֲדֹרָ֖ם עַל־ הַמַּ֑ס וִיהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד הַמַּזְכִּֽיר׃ (ל) (ושיא) וּשְׁוָ֖א סֹפֵ֑ר וְצָד֥וֹק וְאֶבְיָתָ֖ר כֹּהֲנִֽים׃ (לא) וְגַ֗ם עִירָא֙ הַיָּ֣אִרִ֔י הָיָ֥ה כֹהֵ֖ן לְדָוִֽד׃ (לב) וַיְהִ֣י רָעָב֩ בִּימֵ֨י דָוִ֜ד שָׁלֹ֣שׁ שָׁנִ֗ים שָׁנָה֙ אַחֲרֵ֣י שָׁנָ֔ה וַיְבַקֵּ֥שׁ דָּוִ֖ד אֶת־ פְּנֵ֣י יְהוָ֑ה וַיֹּ֣אמֶר יְהוָ֗ה אֶל־ שָׁאוּל֙ וְאֶל־ בֵּ֣ית הַדָּמִ֔ים עַל־ אֲשֶׁר־ הֵמִ֖ית אֶת־ הַגִּבְעֹנִֽים׃ (לג) וַיִּקְרָ֥א הַמֶּ֛לֶךְ לַגִּבְעֹנִ֖ים וַיֹּ֣אמֶר אֲלֵיהֶ֑ם וְהַגִּבְעֹנִ֞ים לֹ֣א מִבְּנֵ֧י יִשְׂרָאֵ֣ל הֵ֗מָּה כִּ֚י אִם־ מִיֶּ֣תֶר הָאֱמֹרִ֔י וּבְנֵ֤י יִשְׂרָאֵל֙ נִשְׁבְּע֣וּ לָהֶ֔ם וַיְבַקֵּ֤שׁ שָׁאוּל֙ לְהַכֹּתָ֔ם בְּקַנֹּאת֥וֹ לִבְנֵֽי־ יִשְׂרָאֵ֖ל וִיהוּדָֽה׃ (לד) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ הַגִּבְעֹנִ֔ים מָ֥ה אֶעֱשֶׂ֖ה לָכֶ֑ם וּבַמָּ֣ה אֲכַפֵּ֔ר וּבָרְכ֖וּ אֶת־ נַחֲלַ֥ת יְהוָֽה׃ (לה) וַיֹּ֧אמְרוּ ל֣וֹ הַגִּבְעֹנִ֗ים אֵֽין־ (לי) לָ֜נוּ כֶּ֤סֶף וְזָהָב֙ עִם־ שָׁא֣וּל וְעִם־ בֵּית֔וֹ וְאֵֽין־ לָ֥נוּ אִ֖ישׁ לְהָמִ֣ית בְּיִשְׂרָאֵ֑ל וַיֹּ֛אמֶר מָֽה־ אַתֶּ֥ם אֹמְרִ֖ים אֶעֱשֶׂ֥ה לָכֶֽם׃ (לו) וַיֹּֽאמְרוּ֙ אֶל־ הַמֶּ֔לֶךְ הָאִישׁ֙ אֲשֶׁ֣ר כִּלָּ֔נוּ וַאֲשֶׁ֖ר דִּמָּה־ לָ֑נוּ נִשְׁמַ֕דְנוּ מֵֽהִתְיַצֵּ֖ב בְּכָל־ גְּבֻ֥ל יִשְׂרָאֵֽל׃ (לז) (ינתן)־ יֻתַּן־ לָ֜נוּ שִׁבְעָ֤ה אֲנָשִׁים֙ מִבָּנָ֔יו וְהוֹקַֽעֲנוּם֙ לַֽיהוָ֔ה בְּגִבְעַ֥ת שָׁא֖וּל בְּחִ֣יר יְהוָ֑ה וַיֹּ֥אמֶר הַמֶּ֖לֶךְ אֲנִ֥י אֶתֵּֽן׃
