שמואל ב פרק י לפי החלוקה לסדרים

פרק י בספר שמואל ב (19 פסוקים) נמצא בשמואל סדר כד (פסוקים מו–נז) ובשמואל סדר כה (פסוקים א–ז).

שמואל סדר כד

(א) וְנֶאְמַ֨ן בֵּיתְךָ֧ וּמַֽמְלַכְתְּךָ֛ עַד־ עוֹלָ֖ם לְפָנֶ֑יךָ כִּֽסְאֲךָ֔ יִהְיֶ֥ה נָכ֖וֹן עַד־ עוֹלָֽם׃ (ב) כְּכֹל֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה וּכְכֹ֖ל הַחִזָּי֣וֹן הַזֶּ֑ה כֵּ֛ן דִּבֶּ֥ר נָתָ֖ן אֶל־ דָּוִֽד׃ (ג) וַיָּבֹא֙ הַמֶּ֣לֶךְ דָּוִ֔ד וַיֵּ֖שֶׁב לִפְנֵ֣י יְהוָ֑ה וַיֹּ֗אמֶר מִ֣י אָנֹכִ֞י אֲדֹנָ֤י יְהוִה֙ וּמִ֣י בֵיתִ֔י כִּ֥י הֲבִיאֹתַ֖נִי עַד־ הֲלֹֽם׃ (ד) וַתִּקְטַן֩ ע֨וֹד זֹ֤את בְּעֵינֶ֙יךָ֙ אֲדֹנָ֣י יְהוִ֔ה וַתְּדַבֵּ֛ר גַּ֥ם אֶל־ בֵּֽית־ עַבְדְּךָ֖ לְמֵֽרָח֑וֹק וְזֹ֛את תּוֹרַ֥ת הָאָדָ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ה) וּמַה־ יּוֹסִ֥יף דָּוִ֛ד ע֖וֹד לְדַבֵּ֣ר אֵלֶ֑יךָ וְאַתָּ֛ה יָדַ֥עְתָּ אֶֽת־ עַבְדְּךָ֖ אֲדֹנָ֥י יְהוִֽה׃ (ו) בַּעֲב֤וּר דְּבָֽרְךָ֙ וּֽכְלִבְּךָ֔ עָשִׂ֕יתָ אֵ֥ת כָּל־ הַגְּדוּלָּ֖ה הַזֹּ֑את לְהוֹדִ֖יעַ אֶת־ עַבְדֶּֽךָ׃ (ז) עַל־ כֵּ֥ן גָּדַ֖לְתָּ אֲדֹנָ֣י יְהוִ֑ה כִּֽי־ אֵ֣ין כָּמ֗וֹךָ וְאֵ֤ין אֱלֹהִים֙ זֽוּלָתֶ֔ךָ בְּכֹ֥ל אֲשֶׁר־ שָׁמַ֖עְנוּ בְּאָזְנֵֽינוּ׃ (ח) וּמִ֤י כְעַמְּךָ֙ כְּיִשְׂרָאֵ֔ל גּ֥וֹי אֶחָ֖ד בָּאָ֑רֶץ אֲשֶׁ֣ר הָלְכֽוּ־ אֱ֠לֹהִים לִפְדּֽוֹת־ ל֨וֹ לְעָ֜ם וְלָשׂ֧וּם ל֣וֹ שֵׁ֗ם וְלַעֲשׂ֨וֹת לָכֶ֜ם הַגְּדוּלָּ֤ה וְנֹֽרָאוֹת֙ לְאַרְצֶ֔ךָ מִפְּנֵ֣י עַמְּךָ֗ אֲשֶׁ֨ר פָּדִ֤יתָ לְּךָ֙ מִמִּצְרַ֔יִם גּוֹיִ֖ם וֵאלֹהָֽיו׃ (ט) וַתְּכ֣וֹנֵֽן לְ֠ךָ אֶת־ עַמְּךָ֨ יִשְׂרָאֵ֧ל ׀ לְךָ֛ לְעָ֖ם עַד־ עוֹלָ֑ם וְאַתָּ֣ה יְהוָ֔ה הָיִ֥יתָ לָהֶ֖ם לֵאלֹהִֽים׃ (י) וְעַתָּה֙ יְהוָ֣ה אֱלֹהִ֔ים הַדָּבָ֗ר אֲשֶׁ֨ר דִּבַּ֤רְתָּ עַֽל־ עַבְדְּךָ֙ וְעַל־ בֵּית֔וֹ הָקֵ֖ם עַד־ עוֹלָ֑ם וַעֲשֵׂ֖ה כַּאֲשֶׁ֥ר דִּבַּֽרְתָּ׃ (יא) וְיִגְדַּ֨ל שִׁמְךָ֤ עַד־ עוֹלָם֙ לֵאמֹ֔ר יְהוָ֣ה צְבָא֔וֹת אֱלֹהִ֖ים עַל־ יִשְׂרָאֵ֑ל וּבֵית֙ עַבְדְּךָ֣ דָוִ֔ד יִהְיֶ֥ה נָכ֖וֹן לְפָנֶֽיךָ׃ (יב) כִּֽי־ אַתָּה֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל גָּלִ֜יתָה אֶת־ אֹ֤זֶן עַבְדְּךָ֙ לֵאמֹ֔ר בַּ֖יִת אֶבְנֶה־ לָּ֑ךְ עַל־ כֵּ֗ן מָצָ֤א עַבְדְּךָ֙ אֶת־ לִבּ֔וֹ לְהִתְפַּלֵּ֣ל אֵלֶ֔יךָ אֶת־ הַתְּפִלָּ֖ה הַזֹּֽאת׃ (יג) וְעַתָּ֣ה ׀ אֲדֹנָ֣י יְהוִ֗ה אַתָּה־ הוּא֙ הָֽאֱלֹהִ֔ים וּדְבָרֶ֖יךָ יִהְי֣וּ אֱמֶ֑ת וַתְּדַבֵּר֙ אֶֽל־ עַבְדְּךָ֔ אֶת־ הַטּוֹבָ֖ה הַזֹּֽאת׃ (יד) וְעַתָּ֗ה הוֹאֵל֙ וּבָרֵךְ֙ אֶת־ בֵּ֣ית עַבְדְּךָ֔ לִהְי֥וֹת לְעוֹלָ֖ם לְפָנֶ֑יךָ כִּֽי־ אַתָּ֞ה אֲדֹנָ֤י יְהוִה֙ דִּבַּ֔רְתָּ וּמִבִּרְכָ֣תְךָ֔ יְבֹרַ֥ךְ בֵּֽית־ עַבְדְּךָ֖ לְעוֹלָֽם׃ (טו) וַֽיְהִי֙ אַֽחֲרֵי־ כֵ֔ן וַיַּ֥ךְ דָּוִ֛ד אֶת־ פְּלִשְׁתִּ֖ים וַיַּכְנִיעֵ֑ם וַיִּקַּ֥ח דָּוִ֛ד אֶת־ מֶ֥תֶג הָאַמָּ֖ה מִיַּ֥ד פְּלִשְׁתִּֽים׃ (טז) וַיַּ֣ךְ אֶת־ מוֹאָ֗ב וַֽיְמַדְּדֵ֤ם בַּחֶ֙בֶל֙ הַשְׁכֵּ֣ב אוֹתָ֣ם אַ֔רְצָה וַיְמַדֵּ֤ד שְׁנֵֽי־ חֲבָלִים֙ לְהָמִ֔ית וּמְלֹ֥א הַחֶ֖בֶל לְהַחֲי֑וֹת וַתְּהִ֤י מוֹאָב֙ לְדָוִ֔ד לַעֲבָדִ֖ים נֹשְׂאֵ֥י מִנְחָֽה׃ (יז) וַיַּ֣ךְ דָּוִ֔ד אֶת־ הֲדַדְעֶ֥זֶר בֶּן־ רְחֹ֖ב מֶ֣לֶךְ צוֹבָ֑ה בְּלֶכְתּ֕וֹ לְהָשִׁ֥יב יָד֖וֹ בִּֽנְהַר־ פְּרָֽת׃ (יח) וַיִּלְכֹּ֨ד דָּוִ֜ד מִמֶּ֗נּוּ אֶ֤לֶף וּשְׁבַע־ מֵאוֹת֙ פָּרָשִׁ֔ים וְעֶשְׂרִ֥ים אֶ֖לֶף אִ֣ישׁ רַגְלִ֑י וַיְעַקֵּ֤ר דָּוִד֙ אֶת־ כָּל־ הָרֶ֔כֶב וַיּוֹתֵ֥ר מִמֶּ֖נּוּ מֵ֥אָה רָֽכֶב׃ (יט) וַתָּבֹא֙ אֲרַ֣ם דַּמֶּ֔שֶׂק לַעְזֹ֕ר לַהֲדַדְעֶ֖זֶר מֶ֣לֶךְ צוֹבָ֑ה וַיַּ֤ךְ דָּוִד֙ בַּֽאֲרָ֔ם עֶשְׂרִֽים־ וּשְׁנַ֥יִם אֶ֖לֶף אִֽישׁ׃ (כ) וַיָּ֨שֶׂם דָּוִ֤ד נְצִבִים֙ בַּאֲרַ֣ם דַּמֶּ֔שֶׂק וַתְּהִ֤י אֲרָם֙ לְדָוִ֔ד לַעֲבָדִ֖ים נוֹשְׂאֵ֣י מִנְחָ֑ה וַיֹּ֤שַׁע יְהוָה֙ אֶת־ דָּוִ֔ד בְּכֹ֖ל אֲשֶׁ֥ר הָלָֽךְ׃ (כא) וַיִּקַּ֣ח דָּוִ֗ד אֵ֚ת שִׁלְטֵ֣י הַזָּהָ֔ב אֲשֶׁ֣ר הָי֔וּ אֶ֖ל עַבְדֵ֣י הֲדַדְעָ֑זֶר וַיְבִיאֵ֖ם יְרוּשָׁלִָֽם׃ (כב) וּמִבֶּ֥טַח וּמִבֵּֽרֹתַ֖י עָרֵ֣י הֲדַדְעָ֑זֶר לָקַ֞ח הַמֶּ֧לֶךְ דָּוִ֛ד נְחֹ֖שֶׁת הַרְבֵּ֥ה מְאֹֽד׃ (כג) וַיִּשְׁמַ֕ע תֹּ֖עִי מֶ֣לֶךְ חֲמָ֑ת כִּ֚י הִכָּ֣ה דָוִ֔ד אֵ֖ת כָּל־ חֵ֥יל הֲדַדְעָֽזֶר׃ (כד) וַיִּשְׁלַ֣ח תֹּ֣עִי אֶת־ יֽוֹרָם־ בְּנ֣וֹ אֶל־ הַמֶּֽלֶךְ־ דָּ֠וִד לִשְׁאָל־ ל֨וֹ לְשָׁל֜וֹם וּֽלְבָרֲכ֗וֹ עַל֩ אֲשֶׁ֨ר נִלְחַ֤ם בַּהֲדַדְעֶ֙זֶר֙ וַיַּכֵּ֔הוּ כִּי־ אִ֛ישׁ מִלְחֲמ֥וֹת תֹּ֖עִי הָיָ֣ה הֲדַדְעָ֑זֶר וּבְיָד֗וֹ הָי֛וּ כְּלֵֽי־ כֶ֥סֶף וּכְלֵֽי־ זָהָ֖ב וּכְלֵ֥י נְחֹֽשֶׁת׃ (כה) גַּם־ אֹתָ֕ם הִקְדִּ֛ישׁ הַמֶּ֥לֶךְ דָּוִ֖ד לַֽיהוָ֑ה עִם־ הַכֶּ֤סֶף וְהַזָּהָב֙ אֲשֶׁ֣ר הִקְדִּ֔ישׁ מִכָּל־ הַגּוֹיִ֖ם אֲשֶׁ֥ר כִּבֵּֽשׁ׃ (כו) מֵאֲרָ֤ם וּמִמּוֹאָב֙ וּמִבְּנֵ֣י עַמּ֔וֹן וּמִפְּלִשְׁתִּ֖ים וּמֵֽעֲמָלֵ֑ק וּמִשְּׁלַ֛ל הֲדַדְעֶ֥זֶר בֶּן־ רְחֹ֖ב מֶ֥לֶךְ צוֹבָֽה׃ (כז) וַיַּ֤עַשׂ דָּוִד֙ שֵׁ֔ם בְּשֻׁב֕וֹ מֵהַכּוֹת֥וֹ אֶת־ אֲרָ֖ם בְּגֵיא־ מֶ֑לַח שְׁמוֹנָ֥ה עָשָׂ֖ר אָֽלֶף׃ (כח) וַיָּ֨שֶׂם בֶּאֱד֜וֹם נְצִבִ֗ים בְּכָל־ אֱדוֹם֙ שָׂ֣ם נְצִבִ֔ים וַיְהִ֥י כָל־ אֱד֖וֹם עֲבָדִ֣ים לְדָוִ֑ד וַיּ֤וֹשַׁע יְהוָה֙ אֶת־ דָּוִ֔ד בְּכֹ֖ל אֲשֶׁ֥ר הָלָֽךְ׃ (כט) וַיִּמְלֹ֥ךְ דָּוִ֖ד עַל־ כָּל־ יִשְׂרָאֵ֑ל וַיְהִ֣י דָוִ֗ד עֹשֶׂ֛ה מִשְׁפָּ֥ט וּצְדָקָ֖ה לְכָל־ עַמּֽוֹ׃ (ל) וְיוֹאָ֥ב בֶּן־ צְרוּיָ֖ה עַל־ הַצָּבָ֑א וִיהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד מַזְכִּֽיר׃ (לא) וְצָד֧וֹק בֶּן־ אֲחִיט֛וּב וַאֲחִימֶ֥לֶךְ בֶּן־ אֶבְיָתָ֖ר כֹּהֲנִ֑ים וּשְׂרָיָ֖ה סוֹפֵֽר׃ (לב) וּבְנָיָ֙הוּ֙ בֶּן־ יְה֣וֹיָדָ֔ע וְהַכְּרֵתִ֖י וְהַפְּלֵתִ֑י וּבְנֵ֥י דָוִ֖ד כֹּהֲנִ֥ים הָיֽוּ׃ (לג) וַיֹּ֣אמֶר דָּוִ֔ד הֲכִ֣י יֶשׁ־ ע֔וֹד אֲשֶׁ֥ר נוֹתַ֖ר לְבֵ֣ית שָׁא֑וּל וְאֶעֱשֶׂ֤ה עִמּוֹ֙ חֶ֔סֶד בַּעֲב֖וּר יְהוֹנָתָֽן׃ (לד) וּלְבֵ֨ית שָׁא֥וּל עֶ֙בֶד֙ וּשְׁמ֣וֹ צִיבָ֔א וַיִּקְרְאוּ־ ל֖וֹ אֶל־ דָּוִ֑ד וַיֹּ֨אמֶר הַמֶּ֧לֶךְ אֵלָ֛יו הַאַתָּ֥ה צִיבָ֖א וַיֹּ֥אמֶר עַבְדֶּֽךָ׃ (לה) וַיֹּ֣אמֶר הַמֶּ֗לֶךְ הַאֶ֨פֶס ע֥וֹד אִישׁ֙ לְבֵ֣ית שָׁא֔וּל וְאֶעֱשֶׂ֥ה עִמּ֖וֹ חֶ֣סֶד אֱלֹהִ֑ים וַיֹּ֤אמֶר צִיבָא֙ אֶל־ הַמֶּ֔לֶךְ ע֛וֹד בֵּ֥ן לִיהוֹנָתָ֖ן נְכֵ֥ה רַגְלָֽיִם׃ (לו) וַיֹּֽאמֶר־ ל֥וֹ הַמֶּ֖לֶךְ אֵיפֹ֣ה ה֑וּא וַיֹּ֤אמֶר צִיבָא֙ אֶל־ הַמֶּ֔לֶךְ הִנֵּה־ ה֗וּא בֵּ֛ית מָכִ֥יר בֶּן־ עַמִּיאֵ֖ל בְּל֥וֹ דְבָֽר׃ (לז) וַיִּשְׁלַ֖ח הַמֶּ֣לֶךְ דָּוִ֑ד וַיִּקָּחֵ֗הוּ מִבֵּ֛ית מָכִ֥יר בֶּן־ עַמִּיאֵ֖ל מִלּ֥וֹ דְבָֽר׃ (לח) וַ֠יָּבֹא מְפִיבֹ֨שֶׁת בֶּן־ יְהוֹנָתָ֤ן בֶּן־ שָׁאוּל֙ אֶל־ דָּוִ֔ד וַיִּפֹּ֥ל עַל־ מִטֻּמְאָתָ֑הּ וַתָּ֖שָׁב אֶל־ בֵּיתָֽהּ׃ (יב) וַתַּ֖הַר הָֽאִשָּׁ֑ה וַתִּשְׁלַח֙ וַתַּגֵּ֣ד לְדָוִ֔ד וַתֹּ֖אמֶר הָרָ֥ה אָנֹֽכִי׃ (יג) וַיִּשְׁלַ֤ח דָּוִד֙ אֶל־ יוֹאָ֔ב שְׁלַ֣ח אֵלַ֔י אֶת־ אֽוּרִיָּ֖ה הַחִתִּ֑י וַיִּשְׁלַ֥ח יוֹאָ֛ב אֶת־ אֽוּרִיָּ֖ה אֶל־ דָּוִֽד׃ (יד) וַיָּבֹ֥א אוּרִיָּ֖ה אֵלָ֑יו וַיִּשְׁאַ֣ל דָּוִ֗ד לִשְׁל֤וֹם יוֹאָב֙ וְלִשְׁל֣וֹם הָעָ֔ם וְלִשְׁל֖וֹם הַמִּלְחָמָֽה׃ (טו) וַיֹּ֤אמֶר דָּוִד֙ לְא֣וּרִיָּ֔ה רֵ֥ד לְבֵיתְךָ֖ וּרְחַ֣ץ רַגְלֶ֑יךָ וַיֵּצֵ֤א אֽוּרִיָּה֙ מִבֵּ֣ית הַמֶּ֔לֶךְ וַתֵּצֵ֥א אַחֲרָ֖יו מַשְׂאַ֥ת הַמֶּֽלֶךְ׃ (טז) וַיִּשְׁכַּ֣ב אוּרִיָּ֗ה פֶּ֚תַח בֵּ֣ית הַמֶּ֔לֶךְ אֵ֖ת כָּל־ עַבְדֵ֣י אֲדֹנָ֑יו וְלֹ֥א יָרַ֖ד אֶל־ בֵּיתֽוֹ׃ (יז) וַיַּגִּ֤דוּ לְדָוִד֙ לֵאמֹ֔ר לֹֽא־ יָרַ֥ד אוּרִיָּ֖ה אֶל־ בֵּית֑וֹ וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ אוּרִיָּ֗ה הֲל֤וֹא מִדֶּ֙רֶךְ֙ אַתָּ֣ה בָ֔א מַדּ֖וּעַ לֹֽא־ יָרַ֥דְתָּ אֶל־ בֵּיתֶֽךָ׃ (יח) וַיֹּ֨אמֶר אוּרִיָּ֜ה אֶל־ דָּוִ֗ד הָ֠אָרוֹן וְיִשְׂרָאֵ֨ל וִֽיהוּדָ֜ה יֹשְׁבִ֣ים בַּסֻּכּ֗וֹת וַאדֹנִ֨י יוֹאָ֜ב וְעַבְדֵ֤י אֲדֹנִ֨י עַל־ פְּנֵ֤י הַשָּׂדֶה֙ חֹנִ֔ים וַאֲנִ֞י אָב֧וֹא אֶל־ בֵּיתִ֛י לֶאֱכֹ֥ל וְלִשְׁתּ֖וֹת וְלִשְׁכַּ֣ב עִם־ אִשְׁתִּ֑י חַיֶּ֙ךָ֙ וְחֵ֣י נַפְשֶׁ֔ךָ אִֽם־ אֶעֱשֶׂ֖ה אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (יט) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ אוּרִיָּ֗ה שֵׁ֥ב בָּזֶ֛ה גַּם־ הַיּ֖וֹם וּמָחָ֣ר אֲשַׁלְּחֶ֑ךָּ וַיֵּ֨שֶׁב אוּרִיָּ֧ה בִירוּשָׁלִַ֛ם בַּיּ֥וֹם הַה֖וּא וּמִֽמָּחֳרָֽת׃ (כ) וַיִּקְרָא־ ל֣וֹ דָוִ֗ד וַיֹּ֧אכַל לְפָנָ֛יו וַיֵּ֖שְׁתְּ וַֽיְשַׁכְּרֵ֑הוּ וַיֵּצֵ֣א בָעֶ֗רֶב לִשְׁכַּ֤ב בְּמִשְׁכָּבוֹ֙ עִם־ עַבְדֵ֣י אֲדֹנָ֔יו וְאֶל־ בֵּית֖וֹ לֹ֥א יָרָֽד׃ (כא) וַיְהִ֣י בַבֹּ֔קֶר וַיִּכְתֹּ֥ב דָּוִ֛ד סֵ֖פֶר אֶל־ יוֹאָ֑ב וַיִּשְׁלַ֖ח בְּיַ֥ד אוּרִיָּֽה׃ (כב) וַיִּכְתֹּ֥ב בַּסֵּ֖פֶר לֵאמֹ֑ר הָב֣וּ אֶת־ אֽוּרִיָּ֗ה אֶל־ מוּל֙ פְּנֵ֤י הַמִּלְחָמָה֙ הַֽחֲזָקָ֔ה וְשַׁבְתֶּ֥ם מֵאַחֲרָ֖יו וְנִכָּ֥ה וָמֵֽת׃ (כג) וַיְהִ֕י בִּשְׁמ֥וֹר יוֹאָ֖ב אֶל־ הָעִ֑יר וַיִּתֵּן֙ אֶת־ א֣וּרִיָּ֔ה אֶל־ הַמָּקוֹם֙ אֲשֶׁ֣ר יָדַ֔ע כִּ֥י אַנְשֵׁי־ חַ֖יִל שָֽׁם׃ (כד) וַיֵּ֨צְא֜וּ אַנְשֵׁ֤י הָעִיר֙ וַיִּלָּחֲמ֣וּ אֶת־ יוֹאָ֔ב וַיִּפֹּ֥ל מִן־ הָעָ֖ם מֵעַבְדֵ֣י דָוִ֑ד וַיָּ֕מָת גַּ֖ם אוּרִיָּ֥ה הַחִתִּֽי׃ (כה) וַיִּשְׁלַ֖ח יוֹאָ֑ב וַיַּגֵּ֣ד לְדָוִ֔ד אֶת־ כָּל־ דִּבְרֵ֖י הַמִּלְחָמָֽה׃ (כו) וַיְצַ֥ו אֶת־ הַמַּלְאָ֖ךְ לֵאמֹ֑ר כְּכַלּוֹתְךָ֗ אֵ֛ת כָּל־ דִּבְרֵ֥י הַמִּלְחָמָ֖ה לְדַבֵּ֥ר אֶל־ הַמֶּֽלֶךְ׃ (כז) וְהָיָ֗ה אִֽם־ תַּעֲלֶה֙ חֲמַ֣ת הַמֶּ֔לֶךְ וְאָמַ֣ר לְךָ֔ מַדּ֛וּעַ נִגַּשְׁתֶּ֥ם אֶל־ הָעִ֖יר לְהִלָּחֵ֑ם הֲל֣וֹא יְדַעְתֶּ֔ם אֵ֥ת אֲשֶׁר־ יֹר֖וּ מֵעַ֥ל הַחוֹמָֽה׃ (כח) מִֽי־ הִכָּ֞ה אֶת־ אֲבִימֶ֣לֶךְ בֶּן־ יְרֻבֶּ֗שֶׁת הֲלֽוֹא־ אִשָּׁ֡ה הִשְׁלִ֣יכָה עָלָיו֩ פֶּ֨לַח רֶ֜כֶב מֵעַ֤ל הַֽחוֹמָה֙ וַיָּ֣מָת בְּתֵבֵ֔ץ לָ֥מָּה נִגַּשְׁתֶּ֖ם אֶל־ הַֽחוֹמָ֑ה וְאָ֣מַרְתָּ֔ גַּ֗ם עַבְדְּךָ֛ אוּרִיָּ֥ה הַחִתִּ֖י מֵֽת׃ (כט) וַיֵּ֖לֶךְ הַמַּלְאָ֑ךְ וַיָּבֹא֙ וַיַּגֵּ֣ד לְדָוִ֔ד אֵ֛ת כָּל־ אֲשֶׁ֥ר שְׁלָח֖וֹ יוֹאָֽב׃ (ל) וַיֹּ֤אמֶר הַמַּלְאָךְ֙ אֶל־ דָּוִ֔ד כִּֽי־ גָבְר֤וּ עָלֵ֙ינוּ֙ הָֽאֲנָשִׁ֔ים וַיֵּצְא֥וּ אֵלֵ֖ינוּ הַשָּׂדֶ֑ה וַנִּהְיֶ֥ה עֲלֵיהֶ֖ם עַד־ פֶּ֥תַח הַשָּֽׁעַר׃ (לא) (ויראו) וַיֹּר֨וּ (המוראים) הַמּוֹרִ֤ים אֶל־ עֲבָדֶ֙ךָ֙ מֵעַ֣ל הַחוֹמָ֔ה וַיָּמ֖וּתוּ מֵעַבְדֵ֣י הַמֶּ֑לֶךְ וְגַ֗ם עַבְדְּךָ֛ אוּרִיָּ֥ה הַחִתִּ֖י מֵֽת׃ (לב) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ הַמַּלְאָ֗ךְ כֹּֽה־ תֹאמַ֤ר אֶל־ יוֹאָב֙ אַל־ יֵרַ֤ע בְּעֵינֶ֙יךָ֙ אֶת־ הַדָּבָ֣ר הַזֶּ֔ה כִּֽי־ כָזֹ֥ה וְכָזֶ֖ה תֹּאכַ֣ל הֶחָ֑רֶב הַחֲזֵ֨ק מִלְחַמְתְּךָ֧ אֶל־ הָעִ֛יר וְהָרְסָ֖הּ וְחַזְּקֵֽהוּ׃ (לג) וַתִּשְׁמַע֙ אֵ֣שֶׁת אֽוּרִיָּ֔ה כִּי־ מֵ֖ת אוּרִיָּ֣ה אִישָׁ֑הּ וַתִּסְפֹּ֖ד עַל־ בַּעְלָֽהּ׃ (לד) וַיַּעֲבֹ֣ר הָאֵ֗בֶל וַיִּשְׁלַ֨ח דָּוִ֜ד וַיַּאַסְפָ֤הּ אֶל־ בֵּיתוֹ֙ וַתְּהִי־ ל֣וֹ לְאִשָּׁ֔ה וַתֵּ֥לֶד ל֖וֹ בֵּ֑ן וַיֵּ֧רַע הַדָּבָ֛ר אֲשֶׁר־ עָשָׂ֥ה דָוִ֖ד בְּעֵינֵ֥י יְהוָֽה׃ (לה) וַיִּשְׁלַ֧ח יְהוָ֛ה אֶת־ נָתָ֖ן אֶל־ דָּוִ֑ד וַיָּבֹ֣א אֵלָ֗יו וַיֹּ֤אמֶר לוֹ֙ שְׁנֵ֣י אֲנָשִׁ֗ים הָיוּ֙ בְּעִ֣יר אֶחָ֔ת אֶחָ֥ד עָשִׁ֖יר וְאֶחָ֥ד רָֽאשׁ׃ (לו) לְעָשִׁ֗יר הָיָ֛ה צֹ֥אן וּבָקָ֖ר הַרְבֵּ֥ה מְאֹֽד׃ (לז) וְלָרָ֣שׁ אֵֽין־ כֹּ֗ל כִּי֩ אִם־ כִּבְשָׂ֨ה אַחַ֤ת קְטַנָּה֙ אֲשֶׁ֣ר קָנָ֔ה וַיְחַיֶּ֕הָ וַתִּגְדַּ֥ל עִמּ֛וֹ וְעִם־ בָּנָ֖יו יַחְדָּ֑ו מִפִּתּ֨וֹ תֹאכַ֜ל וּמִכֹּס֤וֹ תִשְׁתֶּה֙ וּבְחֵיק֣וֹ תִשְׁכָּ֔ב וַתְּהִי־ ל֖וֹ כְּבַֽת׃ (לח) וַיָּ֣בֹא הֵלֶךְ֮ לְאִ֣ישׁ הֶֽעָשִׁיר֒ וַיַּחְמֹ֗ל לָקַ֤חַת מִצֹּאנוֹ֙ וּמִבְּקָר֔וֹ לַעֲשׂ֕וֹת לָאֹרֵ֖חַ הַבָּא־ ל֑וֹ וַיִּקַּ֗ח אֶת־ כִּבְשַׂת֙ הָאִ֣ישׁ הָרָ֔אשׁ וַֽיַּעֲשֶׂ֔הָ לָאִ֖ישׁ הַבָּ֥א אֵלָֽיו׃ (לט) וַיִּֽחַר־ אַ֥ף דָּוִ֛ד בָּאִ֖ישׁ מְאֹ֑ד וַיֹּ֙אמֶר֙ אֶל־ נָתָ֔ן חַי־ יְהוָ֕ה כִּ֣י בֶן־ מָ֔וֶת הָאִ֖ישׁ הָעֹשֶׂ֥ה זֹֽאת׃ (מ) וְאֶת־ הַכִּבְשָׂ֖ה יְשַׁלֵּ֣ם אַרְבַּעְתָּ֑יִם עֵ֗קֶב אֲשֶׁ֤ר עָשָׂה֙ אֶת־ הַדָּבָ֣ר הַזֶּ֔ה וְעַ֖ל אֲשֶׁ֥ר לֹֽא־ חָמָֽל׃ (מא) וַיֹּ֧אמֶר נָתָ֛ן אֶל־ דָּוִ֖ד אַתָּ֣ה הָאִ֑ישׁ כֹּה־ אָמַ֨ר יְהוָ֜ה אֱלֹהֵ֣י יִשְׂרָאֵ֗ל אָנֹכִ֞י מְשַׁחְתִּ֤יךָֽ לְמֶ֙לֶךְ֙ עַל־ יִשְׂרָאֵ֔ל וְאָנֹכִ֥י הִצַּלְתִּ֖יךָ מִיַּ֥ד שָׁאֽוּל׃ (מב) וָאֶתְּנָ֨ה לְךָ֜ אֶת־ בֵּ֣ית אֲדֹנֶ֗יךָ וְאֶת־ נְשֵׁ֤י אֲדֹנֶ֙יךָ֙ בְּחֵיקֶ֔ךָ וָאֶתְּנָ֣ה לְךָ֔ אֶת־ בֵּ֥ית יִשְׂרָאֵ֖ל וִֽיהוּדָ֑ה וְאִ֨ם־ מְעָ֔ט וְאֹסִ֥פָה לְּךָ֖ כָּהֵ֥נָּה וְכָהֵֽנָּה׃ (מג) מַדּ֜וּעַ בָּזִ֣יתָ ׀ אֶת־ דְּבַ֣ר יְהוָ֗ה לַעֲשׂ֣וֹת הָרַע֮ (בעינו) בְּעֵינַי֒ אֵ֣ת אוּרִיָּ֤ה הַֽחִתִּי֙ הִכִּ֣יתָ בַחֶ֔רֶב וְאֶ֨ת־ אִשְׁתּ֔וֹ לָקַ֥חְתָּ לְּךָ֖ לְאִשָּׁ֑ה וְאֹת֣וֹ הָרַ֔גְתָּ בְּחֶ֖רֶב בְּנֵ֥י עַמּֽוֹן׃ (מד) וְעַתָּ֗ה לֹא־ תָס֥וּר חֶ֛רֶב מִבֵּיתְךָ֖ עַד־ עוֹלָ֑ם עֵ֚קֶב כִּ֣י בְזִתָ֔נִי וַתִּקַּ֗ח אֶת־ אֵ֙שֶׁת֙ אוּרִיָּ֣ה הַחִתִּ֔י לִהְי֥וֹת לְךָ֖ לְאִשָּֽׁה׃ (מה) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה הִנְנִי֩ מֵקִ֨ים עָלֶ֤יךָ רָעָה֙ מִבֵּיתֶ֔ךָ וְלָקַחְתִּ֤י אֶת־ נָשֶׁ֙יךָ֙ לְעֵינֶ֔יךָ וְנָתַתִּ֖י לְרֵעֶ֑יךָ וְשָׁכַב֙ עִם־ נָשֶׁ֔יךָ לְעֵינֵ֖י הַשֶּׁ֥מֶשׁ הַזֹּֽאת׃ (מו) כִּ֥י אַתָּ֖ה עָשִׂ֣יתָ בַסָּ֑תֶר וַאֲנִ֗י
