שמואל ב פרק יב לפי החלוקה לסדרים

פרק יב בספר שמואל ב (31 פסוקים) נמצא בשמואל סדר כה (פסוקים לה–מו) ובשמואל סדר כו (פסוקים א–יט).

שמואל סדר כה

(א) וַיִּגַּ֣שׁ יוֹאָ֗ב וְהָעָם֙ אֲשֶׁ֣ר עִמּ֔וֹ לַמִּלְחָמָ֖ה בַּֽאֲרָ֑ם וַיָּנֻ֖סוּ מִפָּנָֽיו׃ (ב) וּבְנֵ֨י עַמּ֤וֹן רָאוּ֙ כִּי־ נָ֣ס אֲרָ֔ם וַיָּנֻ֙סוּ֙ מִפְּנֵ֣י אֲבִישַׁ֔י וַיָּבֹ֖אוּ הָעִ֑יר וַיָּ֣שָׁב יוֹאָ֗ב מֵעַל֙ בְּנֵ֣י עַמּ֔וֹן וַיָּבֹ֖א יְרוּשָׁלִָֽם׃ (ג) וַיַּ֣רְא אֲרָ֔ם כִּ֥י נִגַּ֖ף לִפְנֵ֣י יִשְׂרָאֵ֑ל וַיֵּאָסְפ֖וּ יָֽחַד׃ (ד) וַיִּשְׁלַ֣ח הֲדַדְעֶ֗זֶר וַיֹּצֵ֤א אֶת־ אֲרָם֙ אֲשֶׁר֙ מֵעֵ֣בֶר הַנָּהָ֔ר וַיָּבֹ֖אוּ חֵילָ֑ם וְשׁוֹבַ֛ךְ שַׂר־ צְבָ֥א הֲדַדְעֶ֖זֶר לִפְנֵיהֶֽם׃ (ה) וַיֻּגַּ֣ד לְדָוִ֗ד וַיֶּאֱסֹ֤ף אֶת־ כָּל־ יִשְׂרָאֵל֙ וַיַּעֲבֹ֣ר אֶת־ הַיַּרְדֵּ֔ן וַיָּבֹ֖א חֵלָ֑אמָה וַיַּעַרְכ֤וּ אֲרָם֙ לִקְרַ֣את דָּוִ֔ד וַיִּלָּחֲמ֖וּ עִמּֽוֹ׃ (ו) וַיָּ֣נָס אֲרָם֮ מִפְּנֵ֣י יִשְׂרָאֵל֒ וַיַּהֲרֹ֨ג דָּוִ֜ד מֵאֲרָ֗ם שְׁבַ֤ע מֵאוֹת֙ רֶ֔כֶב וְאַרְבָּעִ֥ים אֶ֖לֶף פָּרָשִׁ֑ים וְאֵ֨ת שׁוֹבַ֧ךְ שַׂר־ צְבָא֛וֹ הִכָּ֖ה וַיָּ֥מָת שָֽׁם׃ (ז) וַיִּרְא֨וּ כָֽל־ הַמְּלָכִ֜ים עַבְדֵ֣י הֲדַדְעֶ֗זֶר כִּ֤י נִגְּפוּ֙ לִפְנֵ֣י יִשְׂרָאֵ֔ל וַיַּשְׁלִ֥מוּ אֶת־ יִשְׂרָאֵ֖ל וַיַּֽעַבְד֑וּם וַיִּֽרְא֣וּ אֲרָ֔ם לְהוֹשִׁ֥יעַ ע֖וֹד אֶת־ בְּנֵ֥י עַמּֽוֹן׃ (ח) וַיְהִי֩ לִתְשׁוּבַ֨ת הַשָּׁנָ֜ה לְעֵ֣ת ׀ צֵ֣את הַמַּלְאֿכִ֗ים וַיִּשְׁלַ֣ח דָּוִ֡ד אֶת־ יוֹאָב֩ וְאֶת־ עֲבָדָ֨יו עִמּ֜וֹ וְאֶת־ כָּל־ יִשְׂרָאֵ֗ל וַיַּשְׁחִ֙תוּ֙ אֶת־ בְּנֵ֣י עַמּ֔וֹן וַיָּצֻ֖רוּ עַל־ רַבָּ֑ה וְדָוִ֖ד יוֹשֵׁ֥ב בִּירוּשָׁלִָֽם׃ (ט) וַיְהִ֣י ׀ לְעֵ֣ת הָעֶ֗רֶב וַיָּ֨קָם דָּוִ֜ד מֵעַ֤ל מִשְׁכָּבוֹ֙ וַיִּתְהַלֵּךְ֙ עַל־ גַּ֣ג בֵּית־ הַמֶּ֔לֶךְ וַיַּ֥רְא אִשָּׁ֛ה רֹחֶ֖צֶת מֵעַ֣ל הַגָּ֑ג וְהָ֣אִשָּׁ֔ה טוֹבַ֥ת מַרְאֶ֖ה מְאֹֽד׃ (י) וַיִּשְׁלַ֣ח דָּוִ֔ד וַיִּדְרֹ֖שׁ לָֽאִשָּׁ֑ה וַיֹּ֗אמֶר הֲלוֹא־ זֹאת֙ בַּת־ שֶׁ֣בַע בַּת־ אֱלִיעָ֔ם אֵ֖שֶׁת אוּרִיָּ֥ה הַחִתִּֽי׃ (יא) וַיִּשְׁלַח֩ דָּוִ֨ד מַלְאָכִ֜ים וַיִּקָּחֶ֗הָ וַתָּב֤וֹא אֵלָיו֙ וַיִּשְׁכַּ֣ב עִמָּ֔הּ וְהִ֥יא מִתְקַדֶּ֖שֶׁת מִטֻּמְאָתָ֑הּ וַתָּ֖שָׁב אֶל־ בֵּיתָֽהּ׃ (יב) וַתַּ֖הַר הָֽאִשָּׁ֑ה וַתִּשְׁלַח֙ וַתַּגֵּ֣ד לְדָוִ֔ד וַתֹּ֖אמֶר הָרָ֥ה אָנֹֽכִי׃ (יג) וַיִּשְׁלַ֤ח דָּוִד֙ אֶל־ יוֹאָ֔ב שְׁלַ֣ח אֵלַ֔י אֶת־ אֽוּרִיָּ֖ה הַחִתִּ֑י וַיִּשְׁלַ֥ח יוֹאָ֛ב אֶת־ אֽוּרִיָּ֖ה אֶל־ דָּוִֽד׃ (יד) וַיָּבֹ֥א אוּרִיָּ֖ה אֵלָ֑יו וַיִּשְׁאַ֣ל דָּוִ֗ד לִשְׁל֤וֹם יוֹאָב֙ וְלִשְׁל֣וֹם הָעָ֔ם וְלִשְׁל֖וֹם הַמִּלְחָמָֽה׃ (טו) וַיֹּ֤אמֶר דָּוִד֙ לְא֣וּרִיָּ֔ה רֵ֥ד לְבֵיתְךָ֖ וּרְחַ֣ץ רַגְלֶ֑יךָ וַיֵּצֵ֤א אֽוּרִיָּה֙ מִבֵּ֣ית הַמֶּ֔לֶךְ וַתֵּצֵ֥א אַחֲרָ֖יו מַשְׂאַ֥ת הַמֶּֽלֶךְ׃ (טז) וַיִּשְׁכַּ֣ב אוּרִיָּ֗ה פֶּ֚תַח בֵּ֣ית הַמֶּ֔לֶךְ אֵ֖ת כָּל־ עַבְדֵ֣י אֲדֹנָ֑יו וְלֹ֥א יָרַ֖ד אֶל־ בֵּיתֽוֹ׃ (יז) וַיַּגִּ֤דוּ לְדָוִד֙ לֵאמֹ֔ר לֹֽא־ יָרַ֥ד אוּרִיָּ֖ה אֶל־ בֵּית֑וֹ וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ אוּרִיָּ֗ה הֲל֤וֹא מִדֶּ֙רֶךְ֙ אַתָּ֣ה בָ֔א מַדּ֖וּעַ לֹֽא־ יָרַ֥דְתָּ אֶל־ בֵּיתֶֽךָ׃ (יח) וַיֹּ֨אמֶר אוּרִיָּ֜ה אֶל־ דָּוִ֗ד הָ֠אָרוֹן וְיִשְׂרָאֵ֨ל וִֽיהוּדָ֜ה יֹשְׁבִ֣ים בַּסֻּכּ֗וֹת וַאדֹנִ֨י יוֹאָ֜ב וְעַבְדֵ֤י אֲדֹנִ֨י עַל־ פְּנֵ֤י הַשָּׂדֶה֙ חֹנִ֔ים וַאֲנִ֞י אָב֧וֹא אֶל־ בֵּיתִ֛י לֶאֱכֹ֥ל וְלִשְׁתּ֖וֹת וְלִשְׁכַּ֣ב עִם־ אִשְׁתִּ֑י חַיֶּ֙ךָ֙ וְחֵ֣י נַפְשֶׁ֔ךָ אִֽם־ אֶעֱשֶׂ֖ה אֶת־ הַדָּבָ֥ר הַזֶּֽה׃ (יט) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ אוּרִיָּ֗ה שֵׁ֥ב בָּזֶ֛ה גַּם־ הַיּ֖וֹם וּמָחָ֣ר אֲשַׁלְּחֶ֑ךָּ וַיֵּ֨שֶׁב אוּרִיָּ֧ה בִירוּשָׁלִַ֛ם בַּיּ֥וֹם הַה֖וּא וּמִֽמָּחֳרָֽת׃ (כ) וַיִּקְרָא־ ל֣וֹ דָוִ֗ד וַיֹּ֧אכַל לְפָנָ֛יו וַיֵּ֖שְׁתְּ וַֽיְשַׁכְּרֵ֑הוּ וַיֵּצֵ֣א בָעֶ֗רֶב לִשְׁכַּ֤ב בְּמִשְׁכָּבוֹ֙ עִם־ עַבְדֵ֣י אֲדֹנָ֔יו וְאֶל־ בֵּית֖וֹ לֹ֥א יָרָֽד׃ (כא) וַיְהִ֣י בַבֹּ֔קֶר וַיִּכְתֹּ֥ב דָּוִ֛ד סֵ֖פֶר אֶל־ יוֹאָ֑ב וַיִּשְׁלַ֖ח בְּיַ֥ד אוּרִיָּֽה׃ (כב) וַיִּכְתֹּ֥ב בַּסֵּ֖פֶר לֵאמֹ֑ר הָב֣וּ אֶת־ אֽוּרִיָּ֗ה אֶל־ מוּל֙ פְּנֵ֤י הַמִּלְחָמָה֙ הַֽחֲזָקָ֔ה וְשַׁבְתֶּ֥ם מֵאַחֲרָ֖יו וְנִכָּ֥ה וָמֵֽת׃ (כג) וַיְהִ֕י בִּשְׁמ֥וֹר יוֹאָ֖ב אֶל־ הָעִ֑יר וַיִּתֵּן֙ אֶת־ א֣וּרִיָּ֔ה אֶל־ הַמָּקוֹם֙ אֲשֶׁ֣ר יָדַ֔ע כִּ֥י אַנְשֵׁי־ חַ֖יִל שָֽׁם׃ (כד) וַיֵּ֨צְא֜וּ אַנְשֵׁ֤י הָעִיר֙ וַיִּלָּחֲמ֣וּ אֶת־ יוֹאָ֔ב וַיִּפֹּ֥ל מִן־ הָעָ֖ם מֵעַבְדֵ֣י דָוִ֑ד וַיָּ֕מָת גַּ֖ם אוּרִיָּ֥ה הַחִתִּֽי׃ (כה) וַיִּשְׁלַ֖ח יוֹאָ֑ב וַיַּגֵּ֣ד לְדָוִ֔ד אֶת־ כָּל־ דִּבְרֵ֖י הַמִּלְחָמָֽה׃ (כו) וַיְצַ֥ו אֶת־ הַמַּלְאָ֖ךְ לֵאמֹ֑ר כְּכַלּוֹתְךָ֗ אֵ֛ת כָּל־ דִּבְרֵ֥י הַמִּלְחָמָ֖ה לְדַבֵּ֥ר אֶל־ הַמֶּֽלֶךְ׃ (כז) וְהָיָ֗ה אִֽם־ תַּעֲלֶה֙ חֲמַ֣ת הַמֶּ֔לֶךְ וְאָמַ֣ר לְךָ֔ מַדּ֛וּעַ נִגַּשְׁתֶּ֥ם אֶל־ הָעִ֖יר לְהִלָּחֵ֑ם הֲל֣וֹא יְדַעְתֶּ֔ם אֵ֥ת אֲשֶׁר־ יֹר֖וּ מֵעַ֥ל הַחוֹמָֽה׃ (כח) מִֽי־ הִכָּ֞ה אֶת־ אֲבִימֶ֣לֶךְ בֶּן־ יְרֻבֶּ֗שֶׁת הֲלֽוֹא־ אִשָּׁ֡ה הִשְׁלִ֣יכָה עָלָיו֩ פֶּ֨לַח רֶ֜כֶב מֵעַ֤ל הַֽחוֹמָה֙ וַיָּ֣מָת בְּתֵבֵ֔ץ לָ֥מָּה נִגַּשְׁתֶּ֖ם אֶל־ הַֽחוֹמָ֑ה וְאָ֣מַרְתָּ֔ גַּ֗ם עַבְדְּךָ֛ אוּרִיָּ֥ה הַחִתִּ֖י מֵֽת׃ (כט) וַיֵּ֖לֶךְ הַמַּלְאָ֑ךְ וַיָּבֹא֙ וַיַּגֵּ֣ד לְדָוִ֔ד אֵ֛ת כָּל־ אֲשֶׁ֥ר שְׁלָח֖וֹ יוֹאָֽב׃ (ל) וַיֹּ֤אמֶר הַמַּלְאָךְ֙ אֶל־ דָּוִ֔ד כִּֽי־ גָבְר֤וּ עָלֵ֙ינוּ֙ הָֽאֲנָשִׁ֔ים וַיֵּצְא֥וּ אֵלֵ֖ינוּ הַשָּׂדֶ֑ה וַנִּהְיֶ֥ה עֲלֵיהֶ֖ם עַד־ פֶּ֥תַח הַשָּֽׁעַר׃ (לא) (ויראו) וַיֹּר֨וּ (המוראים) הַמּוֹרִ֤ים אֶל־ עֲבָדֶ֙ךָ֙ מֵעַ֣ל הַחוֹמָ֔ה וַיָּמ֖וּתוּ מֵעַבְדֵ֣י הַמֶּ֑לֶךְ וְגַ֗ם עַבְדְּךָ֛ אוּרִיָּ֥ה הַחִתִּ֖י מֵֽת׃ (לב) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ הַמַּלְאָ֗ךְ כֹּֽה־ תֹאמַ֤ר אֶל־ יוֹאָב֙ אַל־ יֵרַ֤ע בְּעֵינֶ֙יךָ֙ אֶת־ הַדָּבָ֣ר הַזֶּ֔ה כִּֽי־ כָזֹ֥ה וְכָזֶ֖ה תֹּאכַ֣ל הֶחָ֑רֶב הַחֲזֵ֨ק מִלְחַמְתְּךָ֧ אֶל־ הָעִ֛יר וְהָרְסָ֖הּ וְחַזְּקֵֽהוּ׃ (לג) וַתִּשְׁמַע֙ אֵ֣שֶׁת אֽוּרִיָּ֔ה כִּי־ מֵ֖ת אוּרִיָּ֣ה אִישָׁ֑הּ וַתִּסְפֹּ֖ד עַל־ בַּעְלָֽהּ׃ (לד) וַיַּעֲבֹ֣ר הָאֵ֗בֶל וַיִּשְׁלַ֨ח דָּוִ֜ד וַיַּאַסְפָ֤הּ אֶל־ בֵּיתוֹ֙ וַתְּהִי־ ל֣וֹ לְאִשָּׁ֔ה וַתֵּ֥לֶד ל֖וֹ בֵּ֑ן וַיֵּ֧רַע הַדָּבָ֛ר אֲשֶׁר־ עָשָׂ֥ה דָוִ֖ד בְּעֵינֵ֥י יְהוָֽה׃ תחילת פרק יב (לה) וַיִּשְׁלַ֧ח יְהוָ֛ה אֶת־ נָתָ֖ן אֶל־ דָּוִ֑ד וַיָּבֹ֣א אֵלָ֗יו וַיֹּ֤אמֶר לוֹ֙ שְׁנֵ֣י אֲנָשִׁ֗ים הָיוּ֙ בְּעִ֣יר אֶחָ֔ת אֶחָ֥ד עָשִׁ֖יר וְאֶחָ֥ד רָֽאשׁ׃ (לו) לְעָשִׁ֗יר הָיָ֛ה צֹ֥אן וּבָקָ֖ר הַרְבֵּ֥ה מְאֹֽד׃ (לז) וְלָרָ֣שׁ אֵֽין־ כֹּ֗ל כִּי֩ אִם־ כִּבְשָׂ֨ה אַחַ֤ת קְטַנָּה֙ אֲשֶׁ֣ר קָנָ֔ה וַיְחַיֶּ֕הָ וַתִּגְדַּ֥ל עִמּ֛וֹ וְעִם־ (א) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ נָתָ֔ן חָטָ֖אתִי לַֽיהוָ֑ה וַיֹּ֨אמֶר נָתָ֜ן אֶל־ דָּוִ֗ד גַּם־ יְהוָ֛ה הֶעֱבִ֥יר חַטָּאתְךָ֖ לֹ֥א תָמֽוּת׃ (ב) אֶ֗פֶס כִּֽי־ נִאֵ֤ץ נִאַ֙צְתָּ֙ אֶת־ אֹיְבֵ֣י יְהוָ֔ה בַּדָּבָ֖ר הַזֶּ֑ה גַּ֗ם הַבֵּ֛ן הַיִּלּ֥וֹד לְךָ֖ מ֥וֹת יָמֽוּת׃ (ג) וַיֵּ֥לֶךְ נָתָ֖ן אֶל־ בֵּית֑וֹ וַיִּגֹּ֣ף יְהוָ֗ה אֶת־ הַיֶּ֜לֶד אֲשֶׁ֨ר יָלְדָ֧ה אֵֽשֶׁת־ אוּרִיָּ֛ה לְדָוִ֖ד וַיֵּאָנַֽשׁ׃ (ד) וַיְבַקֵּ֥שׁ דָּוִ֛ד אֶת־ הָאֱלֹהִ֖ים בְּעַ֣ד הַנָּ֑עַר וַיָּ֤צָם דָּוִד֙ צ֔וֹם וּבָ֥א וְלָ֖ן וְשָׁכַ֥ב אָֽרְצָה׃ (ה) וַיָּקֻ֜מוּ זִקְנֵ֤י בֵיתוֹ֙ עָלָ֔יו לַהֲקִימ֖וֹ מִן־ הָאָ֑רֶץ וְלֹ֣א אָבָ֔ה וְלֹֽא־ בָרָ֥א אִתָּ֖ם לָֽחֶם׃ (ו) וַיְהִ֛י בַּיּ֥וֹם הַשְּׁבִיעִ֖י וַיָּ֣מָת הַיָּ֑לֶד וַיִּֽרְאוּ֩ עַבְדֵ֨י דָוִ֜ד לְהַגִּ֥יד ל֣וֹ ׀ כִּי־ מֵ֣ת הַיֶּ֗לֶד כִּ֤י אָֽמְרוּ֙ הִנֵּה֩ בִהְי֨וֹת הַיֶּ֜לֶד מַדּ֣וּעַ אַ֠תָּה כָּ֣כָה דַּ֤ל בֶּן־ הַמֶּ֙לֶךְ֙ בַּבֹּ֣קֶר בַּבֹּ֔קֶר הֲל֖וֹא תַּגִּ֣יד לִ֑י וַיֹּ֤אמֶר לוֹ֙ אַמְנ֔וֹן אֶת־ תָּמָ֗ר אֲח֛וֹת אַבְשָׁלֹ֥ם אָחִ֖י אֲנִ֥י אֹהֵֽב׃ (כד) וַיֹּ֤אמֶר לוֹ֙ יְה֣וֹנָדָ֔ב שְׁכַ֥ב עַל־ מִשְׁכָּבְךָ֖ וְהִתְחָ֑ל וּבָ֧א אָבִ֣יךָ לִרְאוֹתֶ֗ךָ וְאָמַרְתָּ֣ אֵלָ֡יו תָּ֣בֹא נָא֩ תָמָ֨ר אֲחוֹתִ֜י וְתַבְרֵ֣נִי לֶ֗חֶם וְעָשְׂתָ֤ה לְעֵינַי֙ אֶת־ הַבִּרְיָ֔ה לְמַ֙עַן֙ אֲשֶׁ֣ר אֶרְאֶ֔ה וְאָכַלְתִּ֖י מִיָּדָֽהּ׃ (כה) וַיִּשְׁכַּ֥ב אַמְנ֖וֹן וַיִּתְחָ֑ל וַיָּבֹ֨א הַמֶּ֜לֶךְ לִרְאֹת֗וֹ וַיֹּ֨אמֶר אַמְנ֤וֹן אֶל־ הַמֶּ֙לֶךְ֙ תָּֽבוֹא־ נָ֞א תָּמָ֣ר אֲחֹתִ֗י וּתְלַבֵּ֤ב לְעֵינַי֙ שְׁתֵּ֣י לְבִב֔וֹת וְאֶבְרֶ֖ה מִיָּדָֽהּ׃ (כו) וַיִּשְׁלַ֥ח דָּוִ֛ד אֶל־ תָּמָ֖ר הַבַּ֣יְתָה לֵאמֹ֑ר לְכִ֣י נָ֗א בֵּ֚ית אַמְנ֣וֹן אָחִ֔יךְ וַעֲשִׂי־ ל֖וֹ הַבִּרְיָֽה׃ (כז) וַתֵּ֣לֶךְ תָּמָ֗ר בֵּ֛ית אַמְנ֥וֹן אָחִ֖יהָ וְה֣וּא שֹׁכֵ֑ב וַתִּקַּ֨ח אֶת־ הַבָּצֵ֤ק (ותלוש) וַתָּ֙לָשׁ֙ וַתְּלַבֵּ֣ב לְעֵינָ֔יו וַתְּבַשֵּׁ֖ל אֶת־ הַלְּבִבֽוֹת׃ (כח) וַתִּקַּ֤ח אֶת־ הַמַּשְׂרֵת֙ וַתִּצֹ֣ק לְפָנָ֔יו וַיְמָאֵ֖ן לֶאֱכ֑וֹל וַיֹּ֣אמֶר אַמְנ֗וֹן הוֹצִ֤יאוּ כָל־ אִישׁ֙ מֵֽעָלַ֔י וַיֵּצְא֥וּ כָל־ אִ֖ישׁ מֵעָלָֽיו׃ (כט) וַיֹּ֨אמֶר אַמְנ֜וֹן אֶל־ תָּמָ֗ר הָבִ֤יאִי הַבִּרְיָה֙ הַחֶ֔דֶר וְאֶבְרֶ֖ה מִיָּדֵ֑ךְ וַתִּקַּ֣ח תָּמָ֗ר אֶת־ הַלְּבִבוֹת֙ אֲשֶׁ֣ר עָשָׂ֔תָה וַתָּבֵ֛א לְאַמְנ֥וֹן אָחִ֖יהָ הֶחָֽדְרָה׃ (ל) וַתַּגֵּ֥שׁ אֵלָ֖יו לֶֽאֱכֹ֑ל וַיַּֽחֲזֶק־ בָּהּ֙ וַיֹּ֣אמֶר לָ֔הּ בּ֛וֹאִי שִׁכְבִ֥י עִמִּ֖י אֲחוֹתִֽי׃ (לא) וַתֹּ֣אמֶר ל֗וֹ אַל־ אָחִי֙ אַל־ תְּעַנֵּ֔נִי כִּ֛י לֹא־ יֵֽעָשֶׂ֥ה כֵ֖ן בְּיִשְׂרָאֵ֑ל אַֽל־ תַּעֲשֵׂ֖ה אֶת־ הַנְּבָלָ֥ה הַזֹּֽאת׃ (לב) וַאֲנִ֗י אָ֤נָה אוֹלִיךְ֙ אֶת־ חֶרְפָּתִ֔י וְאַתָּ֗ה תִּהְיֶ֛ה כְּאַחַ֥ד הַנְּבָלִ֖ים בְּיִשְׂרָאֵ֑ל וְעַתָּה֙ דַּבֶּר־ נָ֣א אֶל־ הַמֶּ֔לֶךְ כִּ֛י לֹ֥א יִמְנָעֵ֖נִי מִמֶּֽךָּ׃ (לג) וְלֹ֥א אָבָ֖ה לִשְׁמֹ֣עַ בְּקוֹלָ֑הּ וַיֶּחֱזַ֤ק מִמֶּ֙נָּה֙ וַיְעַנֶּ֔הָ וַיִּשְׁכַּ֖ב אֹתָֽהּ׃ (לד) וַיִּשְׂנָאֶ֣הָ אַמְנ֗וֹן שִׂנְאָה֙ גְּדוֹלָ֣ה מְאֹ֔ד כִּ֣י גְדוֹלָ֗ה הַשִּׂנְאָה֙ אֲשֶׁ֣ר שְׂנֵאָ֔הּ מֵאַהֲבָ֖ה אֲשֶׁ֣ר אֲהֵבָ֑הּ וַֽיֹּאמֶר־ לָ֥הּ אַמְנ֖וֹן ק֥וּמִי לֵֽכִי׃ (לה) וַתֹּ֣אמֶר ל֗וֹ אַל־ אוֹדֹ֞ת הָרָעָ֤ה הַגְּדוֹלָה֙ הַזֹּ֔את מֵאַחֶ֛רֶת אֲשֶׁר־ עָשִׂ֥יתָ עִמִּ֖י לְשַׁלְּחֵ֑נִי וְלֹ֥א אָבָ֖ה לִשְׁמֹ֥עַֽ לָֽהּ׃ (לו) וַיִּקְרָ֗א אֶֽת־ נַעֲרוֹ֙ מְשָׁ֣רְת֔וֹ וַיֹּ֕אמֶר שִׁלְחוּ־ נָ֥א אֶת־ זֹ֛את מֵעָלַ֖י הַח֑וּצָה וּנְעֹ֥ל הַדֶּ֖לֶת אַחֲרֶֽיהָ׃ (לז) וְעָלֶ֙יהָ֙ כְּתֹ֣נֶת פַּסִּ֔ים כִּי֩ כֵ֨ן תִּלְבַּ֧שְׁןָ בְנוֹת־ הַמֶּ֛לֶךְ הַבְּתוּלֹ֖ת מְעִילִ֑ים וַיֹּצֵ֨א אוֹתָ֤הּ מְשָֽׁרְתוֹ֙ הַח֔וּץ וְנָעַ֥ל הַדֶּ֖לֶת אַחֲרֶֽיהָ׃ (לח) וַתִּקַּ֨ח תָּמָ֥ר אֵ֙פֶר֙ עַל־ רֹאשָׁ֔הּ וּכְתֹ֧נֶת הַפַּסִּ֛ים אֲשֶׁ֥ר עָלֶ֖יהָ קָרָ֑עָה וַתָּ֤שֶׂם יָדָהּ֙ עַל־ רֹאשָׁ֔הּ וַתֵּ֥לֶךְ הָל֖וֹךְ וְזָעָֽקָה׃ (לט) וַיֹּ֨אמֶר אֵלֶ֜יהָ אַבְשָׁל֣וֹם אָחִ֗יהָ הַאֲמִינ֣וֹן אָחִיךְ֮ הָיָ֣ה עִמָּךְ֒ וְעַתָּ֞ה אֲחוֹתִ֤י הַחֲרִ֙ישִׁי֙ אָחִ֣יךְ ה֔וּא אַל־ תָּשִׁ֥יתִי אֶת־ לִבֵּ֖ךְ לַדָּבָ֣ר הַזֶּ֑ה וַתֵּ֤שֶׁב תָּמָר֙ וְשֹׁ֣מֵמָ֔ה בֵּ֖ית אַבְשָׁל֥וֹם אָחִֽיהָ׃ (מ) וְהַמֶּ֣לֶךְ דָּוִ֔ד שָׁמַ֕ע אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיִּ֥חַר ל֖וֹ מְאֹֽד׃ (מא) וְלֹֽא־ דִבֶּ֧ר אַבְשָׁל֛וֹם עִם־ אַמְנ֖וֹן לְמֵרָ֣ע וְעַד־ ט֑וֹב כִּֽי־ שָׂנֵ֤א אַבְשָׁלוֹם֙ אֶת־ אַמְנ֔וֹן עַל־ דְּבַר֙ אֲשֶׁ֣ר עִנָּ֔ה אֵ֖ת תָּמָ֥ר אֲחֹתֽוֹ׃ (מב) וַֽיְהִי֙ לִשְׁנָתַ֣יִם יָמִ֔ים וַיִּהְי֤וּ גֹֽזְזִים֙ לְאַבְשָׁל֔וֹם בְּבַ֥עַל חָצ֖וֹר אֲשֶׁ֣ר עִם־ אֶפְרָ֑יִם וַיִּקְרָ֥א אַבְשָׁל֖וֹם לְכָל־ בְּנֵ֥י הַמֶּֽלֶךְ׃ (מג) וַיָּבֹ֤א אַבְשָׁלוֹם֙ אֶל־ הַמֶּ֔לֶךְ וַיֹּ֕אמֶר הִנֵּה־ נָ֥א גֹזְזִ֖ים לְעַבְדֶּ֑ךָ יֵֽלֶךְ־ נָ֥א הַמֶּ֛לֶךְ וַעֲבָדָ֖יו עִם־ עַבְדֶּֽךָ׃
