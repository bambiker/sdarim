שמואל ב פרק יח לפי החלוקה לסדרים

פרק יח בספר שמואל ב (32 פסוקים) נמצא בשמואל סדר ל (פסוקים יא–לו) ובשמואל סדר לא (פסוקים א–ו).

שמואל סדר ל

(א) וַיָּבֹ֣אוּ עַבְדֵי֩ אַבְשָׁל֨וֹם אֶֽל־ הָאִשָּׁ֜ה הַבַּ֗יְתָה וַיֹּֽאמְרוּ֙ אַיֵּ֗ה אֲחִימַ֙עַץ֙ וִיה֣וֹנָתָ֔ן וַתֹּ֤אמֶר לָהֶם֙ הָֽאִשָּׁ֔ה עָבְר֖וּ מִיכַ֣ל הַמָּ֑יִם וַיְבַקְשׁוּ֙ וְלֹ֣א מָצָ֔אוּ וַיָּשֻׁ֖בוּ יְרוּשָׁלִָֽם׃ (ב) וַיְהִ֣י ׀ אַחֲרֵ֣י לֶכְתָּ֗ם וַֽיַּעֲלוּ֙ מֵֽהַבְּאֵ֔ר וַיֵּ֣לְכ֔וּ וַיַּגִּ֖דוּ לַמֶּ֣לֶךְ דָּוִ֑ד וַיֹּאמְר֣וּ אֶל־ דָּוִ֗ד ק֣וּמוּ וְעִבְר֤וּ מְהֵרָה֙ אֶת־ הַמַּ֔יִם כִּי־ כָ֛כָה יָעַ֥ץ עֲלֵיכֶ֖ם אֲחִיתֹֽפֶל׃ (ג) וַיָּ֣קָם דָּוִ֗ד וְכָל־ הָעָם֙ אֲשֶׁ֣ר אִתּ֔וֹ וַיַּעַבְר֖וּ אֶת־ הַיַּרְדֵּ֑ן עַד־ א֣וֹר הַבֹּ֗קֶר עַד־ אַחַד֙ לֹ֣א נֶעְדָּ֔ר אֲשֶׁ֥ר לֹא־ עָבַ֖ר אֶת־ הַיַּרְדֵּֽן׃ (ד) וַאֲחִיתֹ֣פֶל רָאָ֗ה כִּ֣י לֹ֣א נֶעֶשְׂתָה֮ עֲצָתוֹ֒ וַיַּחֲבֹ֣שׁ אֶֽת־ הַחֲמ֗וֹר וַיָּ֜קָם וַיֵּ֤לֶךְ אֶל־ בֵּיתוֹ֙ אֶל־ עִיר֔וֹ וַיְצַ֥ו אֶל־ בֵּית֖וֹ וַיֵּחָנַ֑ק וַיָּ֕מָת וַיִּקָּבֵ֖ר בְּקֶ֥בֶר אָבִֽיו׃ (ה) וְדָוִ֖ד בָּ֣א מַחֲנָ֑יְמָה וְאַבְשָׁלֹ֗ם עָבַר֙ אֶת־ הַיַּרְדֵּ֔ן ה֕וּא וְכָל־ אִ֥ישׁ יִשְׂרָאֵ֖ל עִמּֽוֹ׃ (ו) וְאֶת־ עֲמָשָׂ֗א שָׂ֧ם אַבְשָׁלֹ֛ם תַּ֥חַת יוֹאָ֖ב עַל־ הַצָּבָ֑א וַעֲמָשָׂ֣א בֶן־ אִ֗ישׁ וּשְׁמוֹ֙ יִתְרָ֣א הַיִּשְׂרְאֵלִ֔י אֲשֶׁר־ בָּא֙ אֶל־ אֲבִיגַ֣ל בַּת־ נָחָ֔שׁ אֲח֥וֹת צְרוּיָ֖ה אֵ֥ם יוֹאָֽב׃ (ז) וַיִּ֤חַן יִשְׂרָאֵל֙ וְאַבְשָׁלֹ֔ם אֶ֖רֶץ הַגִּלְעָֽד׃ (ח) וַיְהִ֕י כְּב֥וֹא דָוִ֖ד מַחֲנָ֑יְמָה וְשֹׁבִ֨י בֶן־ נָחָ֜שׁ מֵרַבַּ֣ת בְּנֵֽי־ עַמּ֗וֹן וּמָכִ֤יר בֶּן־ עַמִּיאֵל֙ מִלֹּ֣א דְבָ֔ר וּבַרְזִלַּ֥י הַגִּלְעָדִ֖י מֵרֹגְלִֽים׃ (ט) מִשְׁכָּ֤ב וְסַפּוֹת֙ וּכְלִ֣י יוֹצֵ֔ר וְחִטִּ֥ים וּשְׂעֹרִ֖ים וְקֶ֣מַח וְקָלִ֑י וּפ֥וֹל וַעֲדָשִׁ֖ים וְקָלִֽי׃ (י) וּדְבַ֣שׁ וְחֶמְאָ֗ה וְצֹאן֙ וּשְׁפ֣וֹת בָּקָ֔ר הִגִּ֧ישׁוּ לְדָוִ֛ד וְלָעָ֥ם אֲשֶׁר־ אִתּ֖וֹ לֶאֱכ֑וֹל כִּ֣י אָמְר֔וּ הָעָ֗ם רָעֵ֛ב וְעָיֵ֥ף וְצָמֵ֖א בַּמִּדְבָּֽר׃ תחילת פרק יח (יא) וַיִּפְקֹ֣ד דָּוִ֔ד אֶת־ הָעָ֖ם אֲשֶׁ֣ר אִתּ֑וֹ וַיָּ֣שֶׂם עֲלֵיהֶ֔ם שָׂרֵ֥י אֲלָפִ֖ים וְשָׂרֵ֥י מֵאֽוֹת׃ (יב) וַיְשַׁלַּ֨ח דָּוִ֜ד אֶת־ הָעָ֗ם הַשְּׁלִשִׁ֤ית בְּיַד־ יוֹאָב֙ וְ֠הַשְּׁלִשִׁית בְּיַ֨ד אֲבִישַׁ֤י בֶּן־ צְרוּיָה֙ אֲחִ֣י יוֹאָ֔ב וְהַ֨שְּׁלִשִׁ֔ת בְּיַ֖ד אִתַּ֣י הַגִּתִּ֑י וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ הָעָ֔ם יָצֹ֥א אֵצֵ֛א גַּם־ אֲנִ֖י עִמָּכֶֽם׃ (יג) וַיֹּ֨אמֶר הָעָ֜ם לֹ֣א תֵצֵ֗א כִּי֩ אִם־ נֹ֨ס נָנ֜וּס לֹא־ יָשִׂ֧ימוּ אֵלֵ֣ינוּ לֵ֗ב וְאִם־ יָמֻ֤תוּ חֶצְיֵ֙נוּ֙ לֹֽא־ יָשִׂ֤ימוּ אֵלֵ֙ינוּ֙ לֵ֔ב כִּֽי־ עַתָּ֥ה כָמֹ֖נוּ עֲשָׂרָ֣ה אֲלָפִ֑ים וְעַתָּ֣ה ט֔וֹב כִּי־ תִֽהְיֶה־ לָּ֥נוּ מֵעִ֖יר (לעזיר) לַעְזֽוֹר׃ (יד) וַיֹּ֤אמֶר אֲלֵיהֶם֙ הַמֶּ֔לֶךְ אֲשֶׁר־ יִיטַ֥ב בְּעֵינֵיכֶ֖ם אֶעֱשֶׂ֑ה וַיַּעֲמֹ֤ד הַמֶּ֙לֶךְ֙ אֶל־ יַ֣ד הַשַּׁ֔עַר וְכָל־ הָעָם֙ יָֽצְא֔וּ לְמֵא֖וֹת וְלַאֲלָפִֽים׃ (טו) וַיְצַ֣ו הַמֶּ֡לֶךְ אֶת־ י֠וֹאָב וְאֶת־ אֲבִישַׁ֤י וְאֶת־ אִתַּי֙ לֵאמֹ֔ר לְאַט־ לִ֖י לַנַּ֣עַר לְאַבְשָׁל֑וֹם וְכָל־ הָעָ֣ם שָׁמְע֗וּ בְּצַוֺּ֥ת הַמֶּ֛לֶךְ אֶת־ כָּל־ הַשָּׂרִ֖ים עַל־ דְּבַ֥ר אַבְשָׁלֽוֹם׃ (טז) וַיֵּצֵ֥א הָעָ֛ם הַשָּׂדֶ֖ה לִקְרַ֣את יִשְׂרָאֵ֑ל וַתְּהִ֥י הַמִּלְחָמָ֖ה בְּיַ֥עַר אֶפְרָֽיִם׃ (יז) וַיִּנָּ֤גְפוּ שָׁם֙ עַ֣ם יִשְׂרָאֵ֔ל לִפְנֵ֖י עַבְדֵ֣י דָוִ֑ד וַתְּהִי־ שָׁ֞ם הַמַּגֵּפָ֧ה גְדוֹלָ֛ה בַּיּ֥וֹם הַה֖וּא עֶשְׂרִ֥ים אָֽלֶף׃ (יח) וַתְּהִי־ שָׁ֧ם הַמִּלְחָמָ֛ה (נפצית) נָפֹ֖צֶת עַל־ פְּנֵ֣י כָל־ הָאָ֑רֶץ וַיֶּ֤רֶב הַיַּ֙עַר֙ לֶאֱכֹ֣ל בָּעָ֔ם מֵאֲשֶׁ֥ר אָכְלָ֛ה הַחֶ֖רֶב בַּיּ֥וֹם הַהֽוּא׃ (יט) וַיִּקָּרֵא֙ אַבְשָׁל֔וֹם לִפְנֵ֖י עַבְדֵ֣י דָוִ֑ד וְאַבְשָׁל֞וֹם רֹכֵ֣ב עַל־ הַפֶּ֗רֶד וַיָּבֹ֣א הַפֶּ֡רֶד תַּ֣חַת שׂוֹבֶךְ֩ הָאֵלָ֨ה הַגְּדוֹלָ֜ה וַיֶּחֱזַ֧ק רֹאשׁ֣וֹ בָאֵלָ֗ה וַיֻּתַּן֙ בֵּ֤ין הַשָּׁמַ֙יִם֙ וּבֵ֣ין הָאָ֔רֶץ וְהַפֶּ֥רֶד אֲשֶׁר־ תַּחְתָּ֖יו עָבָֽר׃ (כ) וַיַּרְא֙ אִ֣ישׁ אֶחָ֔ד וַיַּגֵּ֖ד לְיוֹאָ֑ב וַיֹּ֗אמֶר הִנֵּה֙ רָאִ֣יתִי אֶת־ אַבְשָׁלֹ֔ם תָּל֖וּי בָּאֵלָֽה׃ (כא) וַיֹּ֣אמֶר יוֹאָ֗ב לָאִישׁ֙ הַמַּגִּ֣יד ל֔וֹ וְהִנֵּ֣ה רָאִ֔יתָ וּמַדּ֛וּעַ לֹֽא־ הִכִּית֥וֹ שָׁ֖ם אָ֑רְצָה וְעָלַ֗י לָ֤תֶת לְךָ֙ עֲשָׂ֣רָה כֶ֔סֶף וַחֲגֹרָ֖ה אֶחָֽת׃ (כב) וַיֹּ֤אמֶר הָאִישׁ֙ אֶל־ יוֹאָ֔ב (ולא) וְל֨וּא אָנֹכִ֜י שֹׁקֵ֤ל עַל־ כַּפַּי֙ אֶ֣לֶף כֶּ֔סֶף לֹֽא־ אֶשְׁלַ֥ח יָדִ֖י אֶל־ בֶּן־ הַמֶּ֑לֶךְ כִּ֣י בְאָזְנֵ֜ינוּ צִוָּ֣ה הַמֶּ֗לֶךְ אֹ֠תְךָ וְאֶת־ אֲבִישַׁ֤י וְאֶת־ אִתַּי֙ לֵאמֹ֔ר שִׁמְרוּ־ מִ֕י בַּנַּ֖עַר בְּאַבְשָׁלֽוֹם׃ (כג) אֽוֹ־ עָשִׂ֤יתִי (בנפשו) בְנַפְשִׁי֙ שֶׁ֔קֶר וְכָל־ דָּבָ֖ר לֹא־ יִכָּחֵ֣ד מִן־ הַמֶּ֑לֶךְ וְאַתָּ֖ה תִּתְיַצֵּ֥ב מִנֶּֽגֶד׃ (כד) וַיֹּ֣אמֶר יוֹאָ֔ב לֹא־ כֵ֖ן אֹחִ֣ילָה לְפָנֶ֑יךָ וַיִּקַּח֩ שְׁלֹשָׁ֨ה שְׁבָטִ֜ים בְּכַפּ֗וֹ וַיִּתְקָעֵם֙ בְּלֵ֣ב אַבְשָׁל֔וֹם עוֹדֶ֥נּוּ חַ֖י בְּלֵ֥ב הָאֵלָֽה׃ (כה) וַיָּסֹ֙בּוּ֙ עֲשָׂרָ֣ה נְעָרִ֔ים נֹשְׂאֵ֖י כְּלֵ֣י יוֹאָ֑ב וַיַּכּ֥וּ אֶת־ אַבְשָׁל֖וֹם וַיְמִיתֻֽהוּ׃ (כו) וַיִּתְקַ֤ע יוֹאָב֙ בַּשֹּׁפָ֔ר וַיָּ֣שָׁב הָעָ֔ם מִרְדֹ֖ף אַחֲרֵ֣י יִשְׂרָאֵ֑ל כִּֽי־ חָשַׂ֥ךְ יוֹאָ֖ב אֶת־ הָעָֽם׃ (כז) וַיִּקְח֣וּ אֶת־ אַבְשָׁל֗וֹם וַיַּשְׁלִ֨יכוּ אֹת֤וֹ בַיַּ֙עַר֙ אֶל־ הַפַּ֣חַת הַגָּד֔וֹל וַיַּצִּ֧בוּ עָלָ֛יו גַּל־ אֲבָנִ֖ים גָּד֣וֹל מְאֹ֑ד וְכָל־ יִשְׂרָאֵ֔ל נָ֖סוּ אִ֥ישׁ (לאהלו) לְאֹהָלָֽיו׃ (כח) וְאַבְשָׁלֹ֣ם לָקַ֗ח וַיַּצֶּב־ ל֤וֹ (בחיו) בְחַיָּיו֙ אֶת־ מַצֶּ֙בֶת֙ אֲשֶׁ֣ר בְּעֵֽמֶק־ הַמֶּ֔לֶךְ כִּ֤י אָמַר֙ אֵֽין־ לִ֣י בֵ֔ן בַּעֲב֖וּר הַזְכִּ֣יר שְׁמִ֑י וַיִּקְרָ֤א לַמַּצֶּ֙בֶת֙ עַל־ שְׁמ֔וֹ וַיִּקָּ֤רֵא לָהּ֙ יַ֣ד אַבְשָׁלֹ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כט) וַאֲחִימַ֤עַץ בֶּן־ צָדוֹק֙ אָמַ֔ר אָר֣וּצָה נָּ֔א וַאֲבַשְּׂרָ֖ה אֶת־ הַמֶּ֑לֶךְ כִּי־ שְׁפָט֥וֹ יְהוָ֖ה מִיַּ֥ד אֹיְבָֽיו׃ (ל) וַיֹּ֧אמֶר ל֣וֹ יוֹאָ֗ב לֹא֩ אִ֨ישׁ בְּשֹׂרָ֤ה אַתָּה֙ הַיּ֣וֹם הַזֶּ֔ה וּבִשַּׂרְתָּ֖ בְּי֣וֹם אַחֵ֑ר וְהַיּ֤וֹם הַזֶּה֙ לֹ֣א תְבַשֵּׂ֔ר כִּֽי־ (על) עַל־ כֵּ֥ן בֶּן־ הַמֶּ֖לֶךְ מֵֽת׃ (לא) וַיֹּ֤אמֶר יוֹאָב֙ לַכּוּשִׁ֔י לֵ֛ךְ הַגֵּ֥ד לַמֶּ֖לֶךְ אֲשֶׁ֣ר רָאִ֑יתָה וַיִּשְׁתַּ֧חוּ כוּשִׁ֛י לְיוֹאָ֖ב וַיָּרֹֽץ׃ (לב) וַיֹּ֨סֶף ע֜וֹד אֲחִימַ֤עַץ בֶּן־ צָדוֹק֙ וַיֹּ֣אמֶר אֶל־ יוֹאָ֔ב וִ֣יהִי מָ֔ה אָרֻֽצָה־ נָּ֥א גַם־ אָ֖נִי אַחֲרֵ֣י הַכּוּשִׁ֑י וַיֹּ֣אמֶר יוֹאָ֗ב שָׁא֔וּל וַחֲמֵ֨שֶׁת עָשָׂ֥ר בָּנָ֛יו וְעֶשְׂרִ֥ים עֲבָדָ֖יו אִתּ֑וֹ וְצָלְח֥וּ הַיַּרְדֵּ֖ן לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כה) וְעָבְרָ֣ה הָעֲבָרָ֗ה לַֽעֲבִיר֙ אֶת־ בֵּ֣ית הַמֶּ֔לֶךְ וְלַעֲשׂ֥וֹת הַטּ֖וֹב (בעינו) בְּעֵינָ֑יו וְשִׁמְעִ֣י בֶן־ גֵּרָ֗א נָפַל֙ לִפְנֵ֣י הַמֶּ֔לֶךְ בְּעָבְר֖וֹ בַּיַּרְדֵּֽן׃ (כו) וַיֹּ֣אמֶר אֶל־ הַמֶּ֗לֶךְ אַל־ יַחֲשָׁב־ לִ֣י אֲדֹנִי֮ עָוֺן֒ וְאַל־ תִּזְכֹּ֗ר אֵ֚ת אֲשֶׁ֣ר הֶעֱוָ֣ה עַבְדְּךָ֔ בַּיּ֕וֹם אֲשֶׁר־ יָׄצָ֥ׄאׄ אֲדֹנִֽי־ הַמֶּ֖לֶךְ מִירֽוּשָׁלִָ֑ם לָשׂ֥וּם הַמֶּ֖לֶךְ אֶל־ לִבּֽוֹ׃ (כז) כִּ֚י יָדַ֣ע עַבְדְּךָ֔ כִּ֖י אֲנִ֣י חָטָ֑אתִי וְהִנֵּֽה־ בָ֣אתִי הַיּ֗וֹם רִאשׁוֹן֙ לְכָל־ בֵּ֣ית יוֹסֵ֔ף לָרֶ֕דֶת לִקְרַ֖את אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (כח) וַיַּ֨עַן אֲבִישַׁ֤י בֶּן־ צְרוּיָה֙ וַיֹּ֔אמֶר הֲתַ֣חַת זֹ֔את לֹ֥א יוּמַ֖ת שִׁמְעִ֑י כִּ֥י קִלֵּ֖ל אֶת־ מְשִׁ֥יחַ יְהוָֽה׃ (כט) וַיֹּ֣אמֶר דָּוִ֗ד מַה־ לִּ֤י וְלָכֶם֙ בְּנֵ֣י צְרוּיָ֔ה כִּי־ תִֽהְיוּ־ לִ֥י הַיּ֖וֹם לְשָׂטָ֑ן הַיּ֗וֹם י֤וּמַת אִישׁ֙ בְּיִשְׂרָאֵ֔ל כִּ֚י הֲל֣וֹא יָדַ֔עְתִּי כִּ֥י הַיּ֖וֹם אֲנִי־ מֶ֥לֶךְ עַל־ יִשְׂרָאֵֽל׃ (ל) וַיֹּ֧אמֶר הַמֶּ֛לֶךְ אֶל־ שִׁמְעִ֖י לֹ֣א תָמ֑וּת וַיִּשָּׁ֥בַֽע ל֖וֹ הַמֶּֽלֶךְ׃ (לא) וּמְפִבֹ֙שֶׁת֙ בֶּן־ שָׁא֔וּל יָרַ֖ד לִקְרַ֣את הַמֶּ֑לֶךְ וְלֹא־ עָשָׂ֨ה רַגְלָ֜יו וְלֹא־ עָשָׂ֣ה שְׂפָמ֗וֹ וְאֶת־ בְּגָדָיו֙ לֹ֣א כִבֵּ֔ס לְמִן־ הַיּוֹם֙ לֶ֣כֶת הַמֶּ֔לֶךְ עַד־ הַיּ֖וֹם אֲשֶׁר־ בָּ֥א בְשָׁלֽוֹם׃ (לב) וַיְהִ֛י כִּי־ בָ֥א יְרוּשָׁלִַ֖ם לִקְרַ֣את הַמֶּ֑לֶךְ וַיֹּ֤אמֶר לוֹ֙ הַמֶּ֔לֶךְ לָ֛מָּה לֹא־ הָלַ֥כְתָּ עִמִּ֖י מְפִיבֹֽשֶׁת׃ (לג) וַיֹּאמַ֕ר אֲדֹנִ֥י הַמֶּ֖לֶךְ עַבְדִּ֣י רִמָּ֑נִי כִּֽי־ אָמַ֨ר עַבְדְּךָ֜ אֶחְבְּשָׁה־ לִּי֩ הַחֲמ֨וֹר וְאֶרְכַּ֤ב עָלֶ֙יהָ֙ וְאֵלֵ֣ךְ אֶת־ הַמֶּ֔לֶךְ כִּ֥י פִסֵּ֖חַ עַבְדֶּֽךָ׃ (לד) וַיְרַגֵּ֣ל בְּעַבְדְּךָ֔ אֶל־ אֲדֹנִ֖י הַמֶּ֑לֶךְ וַאדֹנִ֤י הַמֶּ֙לֶךְ֙ כְּמַלְאַ֣ךְ הָאֱלֹהִ֔ים וַעֲשֵׂ֥ה הַטּ֖וֹב בְּעֵינֶֽיךָ׃ (לה) כִּי֩ לֹ֨א הָיָ֜ה כָּל־ בֵּ֣ית אָבִ֗י כִּ֤י אִם־ אַנְשֵׁי־ מָ֙וֶת֙ לַאדֹנִ֣י הַמֶּ֔לֶךְ וַתָּ֙שֶׁת֙ אֶֽת־ עַבְדְּךָ֔ בְּאֹכְלֵ֖י שֻׁלְחָנֶ֑ךָ וּמַה־ יֶּשׁ־ לִ֥י עוֹד֙ צְדָקָ֔ה וְלִזְעֹ֥ק ע֖וֹד אֶל־ הַמֶּֽלֶךְ׃ (לו) וַיֹּ֤אמֶר לוֹ֙ הַמֶּ֔לֶךְ לָ֛מָּה תְּדַבֵּ֥ר ע֖וֹד דְּבָרֶ֑יךָ אָמַ֕רְתִּי אַתָּ֣ה וְצִיבָ֔א תַּחְלְק֖וּ אֶת־ הַשָּׂדֶֽה׃ (לז) וַיֹּ֤אמֶר מְפִיבֹ֙שֶׁת֙ אֶל־ הַמֶּ֔לֶךְ גַּ֥ם אֶת־ הַכֹּ֖ל יִקָּ֑ח אַ֠חֲרֵי אֲשֶׁר־ בָּ֞א אֲדֹנִ֥י הַמֶּ֛לֶך בְּשָׁל֖וֹם אֶל־ בֵּיתֽוֹ׃ (לח) וּבַרְזִלַּי֙ הַגִּלְעָדִ֔י יָרַ֖ד מֵרֹגְלִ֑ים וַיַּעֲבֹ֤ר אֶת־ הַמֶּ֙לֶךְ֙ הַיַּרְדֵּ֔ן לְשַׁלְּח֖וֹ אֶת־ (בירדן) הַיַּרְדֵּֽן׃ (לט) וּבַרְזִלַּי֙ זָקֵ֣ן מְאֹ֔ד בֶּן־ שְׁמֹנִ֖ים שָׁנָ֑ה וְהֽוּא־ כִלְכַּ֤ל אֶת־ הַמֶּ֙לֶךְ֙ בְּשִׁיבָת֣וֹ בְמַחֲנַ֔יִם כִּֽי־ אִ֛ישׁ גָּד֥וֹל ה֖וּא מְאֹֽד׃ (מ) וַיֹּ֥אמֶר הַמֶּ֖לֶךְ אֶל־ בַּרְזִלָּ֑י אַתָּה֙ עֲבֹ֣ר אִתִּ֔י וְכִלְכַּלְתִּ֥י אֹתְךָ֛ עִמָּדִ֖י בִּירוּשָׁלִָֽם׃ (מא) וַיֹּ֥אמֶר בַּרְזִלַּ֖י אֶל־ הַמֶּ֑לֶךְ כַּמָּ֗ה יְמֵי֙ שְׁנֵ֣י חַיַּ֔י כִּי־ אֶעֱלֶ֥ה אֶת־ הַמֶּ֖לֶךְ יְרוּשָׁלִָֽם׃ (מב) בֶּן־ שְׁמֹנִ֣ים שָׁנָה֩ אָנֹכִ֨י הַיּ֜וֹם הַאֵדַ֣ע ׀ בֵּין־ ט֣וֹב לְרָ֗ע אִם־ יִטְעַ֤ם עַבְדְּךָ֙ אֶת־ אֲשֶׁ֤ר אֹכַל֙ וְאֶת־ אֲשֶׁ֣ר אֶשְׁתֶּ֔ה אִם־ אֶשְׁמַ֣ע ע֔וֹד בְּק֖וֹל שָׁרִ֣ים וְשָׁר֑וֹת וְלָמָּה֩ יִֽהְיֶ֨ה עַבְדְּךָ֥ עוֹד֙ לְמַשָּׂ֔א אֶל־ אֲדֹנִ֖י הַמֶּֽלֶךְ׃ (מג) כִּמְעַ֞ט יַעֲבֹ֧ר עַבְדְּךָ֛ אֶת־ הַיַּרְדֵּ֖ן אֶת־ הַמֶּ֑לֶךְ וְלָ֙מָּה֙ יִגְמְלֵ֣נִי הַמֶּ֔לֶךְ הַגְּמוּלָ֖ה הַזֹּֽאת׃ (מד) יָֽשָׁב־ נָ֤א עַבְדְּךָ֙ וְאָמֻ֣ת בְּעִירִ֔י עִ֛ם קֶ֥בֶר אָבִ֖י וְאִמִּ֑י וְהִנֵּ֣ה ׀ עַבְדְּךָ֣ כִמְהָ֗ם יַֽעֲבֹר֙ עִם־ אֲדֹנִ֣י הַמֶּ֔לֶךְ וַעֲשֵׂה־ ל֕וֹ אֵ֥ת אֲשֶׁר־ ט֖וֹב בְּעֵינֶֽיךָ׃ (מה) וַיֹּ֣אמֶר הַמֶּ֗לֶךְ אִתִּי֙ יַעֲבֹ֣ר כִּמְהָ֔ם וַאֲנִי֙ אֶעֱשֶׂה־ לּ֔וֹ אֶת־ הַטּ֖וֹב בְּעֵינֶ֑יךָ וְכֹ֛ל אֲשֶׁר־ תִּבְחַ֥ר עָלַ֖י אֶֽעֱשֶׂה־ לָּֽךְ׃
