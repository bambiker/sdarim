שמואל ב פרק יג לפי החלוקה לסדרים

פרק יג בספר שמואל ב (39 פסוקים) נמצא בשמואל סדר כו (פסוקים כ–מג) ובשמואל סדר כז (פסוקים א–טו).

שמואל סדר כו

(א) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ נָתָ֔ן חָטָ֖אתִי לַֽיהוָ֑ה וַיֹּ֨אמֶר נָתָ֜ן אֶל־ דָּוִ֗ד גַּם־ יְהוָ֛ה הֶעֱבִ֥יר חַטָּאתְךָ֖ לֹ֥א תָמֽוּת׃ (ב) אֶ֗פֶס כִּֽי־ נִאֵ֤ץ נִאַ֙צְתָּ֙ אֶת־ אֹיְבֵ֣י יְהוָ֔ה בַּדָּבָ֖ר הַזֶּ֑ה גַּ֗ם הַבֵּ֛ן הַיִּלּ֥וֹד לְךָ֖ מ֥וֹת יָמֽוּת׃ (ג) וַיֵּ֥לֶךְ נָתָ֖ן אֶל־ בֵּית֑וֹ וַיִּגֹּ֣ף יְהוָ֗ה אֶת־ הַיֶּ֜לֶד אֲשֶׁ֨ר יָלְדָ֧ה אֵֽשֶׁת־ אוּרִיָּ֛ה לְדָוִ֖ד וַיֵּאָנַֽשׁ׃ (ד) וַיְבַקֵּ֥שׁ דָּוִ֛ד אֶת־ הָאֱלֹהִ֖ים בְּעַ֣ד הַנָּ֑עַר וַיָּ֤צָם דָּוִד֙ צ֔וֹם וּבָ֥א וְלָ֖ן וְשָׁכַ֥ב אָֽרְצָה׃ (ה) וַיָּקֻ֜מוּ זִקְנֵ֤י בֵיתוֹ֙ עָלָ֔יו לַהֲקִימ֖וֹ מִן־ הָאָ֑רֶץ וְלֹ֣א אָבָ֔ה וְלֹֽא־ בָרָ֥א אִתָּ֖ם לָֽחֶם׃ (ו) וַיְהִ֛י בַּיּ֥וֹם הַשְּׁבִיעִ֖י וַיָּ֣מָת הַיָּ֑לֶד וַיִּֽרְאוּ֩ עַבְדֵ֨י דָוִ֜ד לְהַגִּ֥יד ל֣וֹ ׀ כִּי־ מֵ֣ת הַיֶּ֗לֶד כִּ֤י אָֽמְרוּ֙ הִנֵּה֩ בִהְי֨וֹת הַיֶּ֜לֶד חַ֗י דִּבַּ֤רְנוּ אֵלָיו֙ וְלֹא־ שָׁמַ֣ע בְּקוֹלֵ֔נוּ וְאֵ֨יךְ נֹאמַ֥ר אֵלָ֛יו מֵ֥ת הַיֶּ֖לֶד וְעָשָׂ֥ה רָעָֽה׃ (ז) וַיַּ֣רְא דָּוִ֗ד כִּ֤י עֲבָדָיו֙ מִֽתְלַחֲשִׁ֔ים וַיָּ֥בֶן דָּוִ֖ד כִּ֣י מֵ֣ת הַיָּ֑לֶד וַיֹּ֨אמֶר דָּוִ֧ד אֶל־ עֲבָדָ֛יו הֲמֵ֥ת הַיֶּ֖לֶד וַיֹּ֥אמְרוּ מֵֽת׃ (ח) וַיָּקָם֩ דָּוִ֨ד מֵהָאָ֜רֶץ וַיִּרְחַ֣ץ וַיָּ֗סֶךְ וַיְחַלֵּף֙ (שמלתו) שִׂמְלֹתָ֔יו וַיָּבֹ֥א בֵית־ יְהוָ֖ה וַיִּשְׁתָּ֑חוּ וַיָּבֹא֙ אֶל־ בֵּית֔וֹ וַיִּשְׁאַ֕ל וַיָּשִׂ֥ימוּ ל֛וֹ לֶ֖חֶם וַיֹּאכַֽל׃ (ט) וַיֹּאמְר֤וּ עֲבָדָיו֙ אֵלָ֔יו מָֽה־ הַדָּבָ֥ר הַזֶּ֖ה אֲשֶׁ֣ר עָשִׂ֑יתָה בַּעֲב֞וּר הַיֶּ֤לֶד חַי֙ צַ֣מְתָּ וַתֵּ֔בְךְּ וְכַֽאֲשֶׁר֙ מֵ֣ת הַיֶּ֔לֶד קַ֖מְתָּ וַתֹּ֥אכַל לָֽחֶם׃ (י) וַיֹּ֕אמֶר בְּעוֹד֙ הַיֶּ֣לֶד חַ֔י צַ֖מְתִּי וָֽאֶבְכֶּ֑ה כִּ֤י אָמַ֙רְתִּי֙ מִ֣י יוֹדֵ֔עַ (יחנני) וְחַנַּ֥נִי יְהוָ֖ה וְחַ֥י הַיָּֽלֶד׃ (יא) וְעַתָּ֣ה ׀ מֵ֗ת לָ֤מָּה זֶּה֙ אֲנִ֣י צָ֔ם הַאוּכַ֥ל לַהֲשִׁיב֖וֹ ע֑וֹד אֲנִי֙ הֹלֵ֣ךְ אֵלָ֔יו וְה֖וּא לֹֽא־ יָשׁ֥וּב אֵלָֽי׃ (יב) וַיְנַחֵ֣ם דָּוִ֗ד אֵ֚ת בַּת־ שֶׁ֣בַע אִשְׁתּ֔וֹ וַיָּבֹ֥א אֵלֶ֖יהָ וַיִּשְׁכַּ֣ב עִמָּ֑הּ וַתֵּ֣לֶד בֵּ֗ן (ויקרא) וַתִּקְרָ֤א אֶת־ שְׁמוֹ֙ שְׁלֹמֹ֔ה וַיהוָ֖ה אֲהֵבֽוֹ׃ (יג) וַיִּשְׁלַ֗ח בְּיַד֙ נָתָ֣ן הַנָּבִ֔יא וַיִּקְרָ֥א אֶת־ שְׁמ֖וֹ יְדִ֣ידְיָ֑הּ בַּעֲב֖וּר יְהוָֽה׃ (יד) וַיִּלָּ֣חֶם יוֹאָ֔ב בְּרַבַּ֖ת בְּנֵ֣י עַמּ֑וֹן וַיִּלְכֹּ֖ד אֶת־ עִ֥יר הַמְּלוּכָֽה׃ (טו) וַיִּשְׁלַ֥ח יוֹאָ֛ב מַלְאָכִ֖ים אֶל־ דָּוִ֑ד וַיֹּ֙אמֶר֙ נִלְחַ֣מְתִּי בְרַבָּ֔ה גַּם־ לָכַ֖דְתִּי אֶת־ עִ֥יר הַמָּֽיִם׃ (טז) וְעַתָּ֗ה אֱסֹף֙ אֶת־ יֶ֣תֶר הָעָ֔ם וַחֲנֵ֥ה עַל־ הָעִ֖יר וְלָכְדָ֑הּ פֶּן־ אֶלְכֹּ֤ד אֲנִי֙ אֶת־ הָעִ֔יר וְנִקְרָ֥א שְׁמִ֖י עָלֶֽיהָ׃ (יז) וַיֶּאֱסֹ֥ף דָּוִ֛ד אֶת־ כָּל־ הָעָ֖ם וַיֵּ֣לֶךְ רַבָּ֑תָה וַיִּלָּ֥חֶם בָּ֖הּ וַֽיִּלְכְּדָֽהּ׃ (יח) וַיִּקַּ֣ח אֶת־ עֲטֶֽרֶת־ מַלְכָּם֩ מֵעַ֨ל רֹאשׁ֜וֹ וּמִשְׁקָלָ֨הּ כִּכַּ֤ר זָהָב֙ וְאֶ֣בֶן יְקָרָ֔ה וַתְּהִ֖י עַל־ רֹ֣אשׁ דָּוִ֑ד וּשְׁלַ֥ל הָעִ֛יר הוֹצִ֖יא הַרְבֵּ֥ה מְאֹֽד׃ (יט) וְאֶת־ הָעָ֨ם אֲשֶׁר־ בָּ֜הּ הוֹצִ֗יא וַיָּ֣שֶׂם בַּ֠מְּגֵרָה וּבַחֲרִצֵ֨י הַבַּרְזֶ֜ל וּֽבְמַגְזְרֹ֣ת הַבַּרְזֶ֗ל וְהֶעֱבִ֤יר אוֹתָם֙ (במלכן) בַּמַּלְבֵּ֔ן וְכֵ֣ן יַעֲשֶׂ֔ה לְכֹ֖ל עָרֵ֣י בְנֵֽי־ עַמּ֑וֹן וַיָּ֧שָׁב דָּוִ֛ד וְכָל־ הָעָ֖ם יְרוּשָׁלִָֽם׃ תחילת פרק יג (כ) וַיְהִ֣י אַֽחֲרֵי־ כֵ֗ן וּלְאַבְשָׁל֧וֹם בֶּן־ דָּוִ֛ד אָח֥וֹת יָפָ֖ה וּשְׁמָ֣הּ תָּמָ֑ר וַיֶּאֱהָבֶ֖הָ אַמְנ֥וֹן בֶּן־ דָּוִֽד׃ (כא) וַיֵּ֨צֶר לְאַמְנ֜וֹן לְהִתְחַלּ֗וֹת בַּֽעֲבוּר֙ תָּמָ֣ר אֲחֹת֔וֹ כִּ֥י בְתוּלָ֖ה הִ֑יא וַיִּפָּלֵא֙ בְּעֵינֵ֣י אַמְנ֔וֹן לַעֲשׂ֥וֹת לָ֖הּ מְאֽוּמָה׃ (כב) וּלְאַמְנ֣וֹן רֵ֗עַ וּשְׁמוֹ֙ יֽוֹנָדָ֔ב בֶּן־ שִׁמְעָ֖ה אֲחִ֣י דָוִ֑ד וְי֣וֹנָדָ֔ב אִ֥ישׁ חָכָ֖ם מְאֹֽד׃ (כג) וַיֹּ֣אמֶר ל֗וֹ מַדּ֣וּעַ אַ֠תָּה כָּ֣כָה דַּ֤ל בֶּן־ הַמֶּ֙לֶךְ֙ בַּבֹּ֣קֶר בַּבֹּ֔קֶר הֲל֖וֹא תַּגִּ֣יד לִ֑י וַיֹּ֤אמֶר לוֹ֙ אַמְנ֔וֹן אֶת־ תָּמָ֗ר אֲח֛וֹת אַבְשָׁלֹ֥ם אָחִ֖י אֲנִ֥י אֹהֵֽב׃ (כד) וַיֹּ֤אמֶר לוֹ֙ יְה֣וֹנָדָ֔ב שְׁכַ֥ב עַל־ מִשְׁכָּבְךָ֖ וְהִתְחָ֑ל וּבָ֧א אָבִ֣יךָ לִרְאוֹתֶ֗ךָ וְאָמַרְתָּ֣ אֵלָ֡יו תָּ֣בֹא נָא֩ תָמָ֨ר אֲחוֹתִ֜י וְתַבְרֵ֣נִי לֶ֗חֶם וְעָשְׂתָ֤ה לְעֵינַי֙ אֶת־ הַבִּרְיָ֔ה לְמַ֙עַן֙ אֲשֶׁ֣ר אֶרְאֶ֔ה וְאָכַלְתִּ֖י מִיָּדָֽהּ׃ (כה) וַיִּשְׁכַּ֥ב אַמְנ֖וֹן וַיִּתְחָ֑ל וַיָּבֹ֨א הַמֶּ֜לֶךְ לִרְאֹת֗וֹ וַיֹּ֨אמֶר אַמְנ֤וֹן אֶל־ הַמֶּ֙לֶךְ֙ תָּֽבוֹא־ נָ֞א תָּמָ֣ר אֲחֹתִ֗י וּתְלַבֵּ֤ב לְעֵינַי֙ שְׁתֵּ֣י לְבִב֔וֹת וְאֶבְרֶ֖ה מִיָּדָֽהּ׃ (כו) וַיִּשְׁלַ֥ח דָּוִ֛ד אֶל־ תָּמָ֖ר הַבַּ֣יְתָה לֵאמֹ֑ר לְכִ֣י נָ֗א בֵּ֚ית אַמְנ֣וֹן אָחִ֔יךְ וַעֲשִׂי־ ל֖וֹ הַבִּרְיָֽה׃ (כז) וַתֵּ֣לֶךְ תָּמָ֗ר בֵּ֛ית אַמְנ֥וֹן אָחִ֖יהָ וְה֣וּא שֹׁכֵ֑ב וַתִּקַּ֨ח אֶת־ הַבָּצֵ֤ק (ותלוש) וַתָּ֙לָשׁ֙ וַתְּלַבֵּ֣ב לְעֵינָ֔יו וַתְּבַשֵּׁ֖ל אֶת־ הַלְּבִבֽוֹת׃ (כח) וַתִּקַּ֤ח אֶת־ הַמַּשְׂרֵת֙ וַתִּצֹ֣ק לְפָנָ֔יו וַיְמָאֵ֖ן לֶאֱכ֑וֹל וַיֹּ֣אמֶר אַמְנ֗וֹן הוֹצִ֤יאוּ כָל־ אִישׁ֙ מֵֽעָלַ֔י וַיֵּצְא֥וּ כָל־ אִ֖ישׁ מֵעָלָֽיו׃ (כט) וַיֹּ֨אמֶר אַמְנ֜וֹן אֶל־ תָּמָ֗ר הָבִ֤יאִי הַבִּרְיָה֙ הַחֶ֔דֶר וְאֶבְרֶ֖ה מִיָּדֵ֑ךְ וַתִּקַּ֣ח תָּמָ֗ר אֶת־ הַלְּבִבוֹת֙ אֲשֶׁ֣ר עָשָׂ֔תָה וַתָּבֵ֛א לְאַמְנ֥וֹן אָחִ֖יהָ הֶחָֽדְרָה׃ (ל) וַתַּגֵּ֥שׁ אֵלָ֖יו לֶֽאֱכֹ֑ל וַיַּֽחֲזֶק־ בָּהּ֙ וַיֹּ֣אמֶר לָ֔הּ בּ֛וֹאִי שִׁכְבִ֥י עִמִּ֖י אֲחוֹתִֽי׃ (לא) וַתֹּ֣אמֶר ל֗וֹ אַל־ אָחִי֙ אַל־ תְּעַנֵּ֔נִי כִּ֛י לֹא־ יֵֽעָשֶׂ֥ה כֵ֖ן בְּיִשְׂרָאֵ֑ל אַֽל־ תַּעֲשֵׂ֖ה אֶת־ הַנְּבָלָ֥ה הַזֹּֽאת׃ (לב) וַאֲנִ֗י אָ֤נָה אוֹלִיךְ֙ אֶת־ חֶרְפָּתִ֔י וְאַתָּ֗ה תִּהְיֶ֛ה כְּאַחַ֥ד הַנְּבָלִ֖ים בְּיִשְׂרָאֵ֑ל וְעַתָּה֙ דַּבֶּר־ נָ֣א אֶל־ הַמֶּ֔לֶךְ כִּ֛י לֹ֥א יִמְנָעֵ֖נִי מִמֶּֽךָּ׃ (לג) וְלֹ֥א אָבָ֖ה לִשְׁמֹ֣עַ בְּקוֹלָ֑הּ וַיֶּחֱזַ֤ק מִמֶּ֙נָּה֙ וַיְעַנֶּ֔הָ וַיִּשְׁכַּ֖ב אֹתָֽהּ׃ (לד) וַיִּשְׂנָאֶ֣הָ אַמְנ֗וֹן שִׂנְאָה֙ גְּדוֹלָ֣ה מְאֹ֔ד כִּ֣י גְדוֹלָ֗ה הַשִּׂנְאָה֙ אֲשֶׁ֣ר שְׂנֵאָ֔הּ מֵאַהֲבָ֖ה אֲשֶׁ֣ר אֲהֵבָ֑הּ וַֽיֹּאמֶר־ לָ֥הּ אַמְנ֖וֹן ק֥וּמִי לֵֽכִי׃ (לה) וַתֹּ֣אמֶר ל֗וֹ אַל־ אוֹדֹ֞ת הָרָעָ֤ה הַגְּדוֹלָה֙ הַזֹּ֔את מֵאַחֶ֛רֶת אֲשֶׁר־ עָשִׂ֥יתָ עִמִּ֖י לְשַׁלְּחֵ֑נִי וְלֹ֥א אָבָ֖ה לִשְׁמֹ֥עַֽ לָֽהּ׃ (לו) וַיִּקְרָ֗א אֶֽת־ נַעֲרוֹ֙ מְשָׁ֣רְת֔וֹ וַיֹּ֕אמֶר שִׁלְחוּ־ נָ֥א אֶת־ זֹ֛את מֵעָלַ֖י הַח֑וּצָה וּנְעֹ֥ל הַדֶּ֖לֶת אַחֲרֶֽיהָ׃ (לז) וְעָלֶ֙יהָ֙ כְּתֹ֣נֶת פַּסִּ֔ים כִּי֩ כֵ֨ן תִּלְבַּ֧שְׁןָ בְנוֹת־ הַמֶּ֛לֶךְ הַדָּם֙ לְשַׁחֵ֔ת וְלֹ֥א יַשְׁמִ֖ידוּ אֶת־ בְּנִ֑י וַיֹּ֙אמֶר֙ חַי־ יְהוָ֔ה אִם־ יִפֹּ֛ל מִשַּׂעֲרַ֥ת בְּנֵ֖ךְ אָֽרְצָה׃ (כז) וַתֹּ֙אמֶר֙ הָֽאִשָּׁ֔ה תְּדַבֶּר־ נָ֧א שִׁפְחָתְךָ֛ אֶל־ אֲדֹנִ֥י הַמֶּ֖לֶךְ דָּבָ֑ר וַיֹּ֖אמֶר דַּבֵּֽרִי׃ (כח) וַתֹּ֙אמֶר֙ הָֽאִשָּׁ֔ה וְלָ֧מָּה חָשַׁ֛בְתָּה כָּזֹ֖את עַל־ עַ֣ם אֱלֹהִ֑ים וּמִדַּבֵּ֨ר הַמֶּ֜לֶךְ הַדָּבָ֤ר הַזֶּה֙ כְּאָשֵׁ֔ם לְבִלְתִּ֛י הָשִׁ֥יב הַמֶּ֖לֶךְ אֶֽת־ נִדְּחֽוֹ׃ (כט) כִּי־ מ֣וֹת נָמ֔וּת וְכַמַּ֙יִם֙ הַנִּגָּרִ֣ים אַ֔רְצָה אֲשֶׁ֖ר לֹ֣א יֵאָסֵ֑פוּ וְלֹֽא־ יִשָּׂ֤א אֱלֹהִים֙ נֶ֔פֶשׁ וְחָשַׁב֙ מַֽחֲשָׁב֔וֹת לְבִלְתִּ֛י יִדַּ֥ח מִמֶּ֖נּוּ נִדָּֽח׃ (ל) וְ֠עַתָּה אֲשֶׁר־ בָּ֜אתִי לְדַבֵּ֨ר אֶל־ הַמֶּ֤לֶךְ אֲדֹנִי֙ אֶת־ הַדָּבָ֣ר הַזֶּ֔ה כִּ֥י יֵֽרְאֻ֖נִי הָעָ֑ם וַתֹּ֤אמֶר שִׁפְחָֽתְךָ֙ אֲדַבְּרָה־ נָּ֣א אֶל־ הַמֶּ֔לֶךְ אוּלַ֛י יַעֲשֶׂ֥ה הַמֶּ֖לֶךְ אֶת־ דְּבַ֥ר אֲמָתֽוֹ׃ (לא) כִּ֚י יִשְׁמַ֣ע הַמֶּ֔לֶךְ לְהַצִּ֥יל אֶת־ אֲמָת֖וֹ מִכַּ֣ף הָאִ֑ישׁ לְהַשְׁמִ֨יד אֹתִ֤י וְאֶת־ בְּנִי֙ יַ֔חַד מִֽנַּחֲלַ֖ת אֱלֹהִֽים׃ (לב) וַתֹּ֙אמֶר֙ שִׁפְחָ֣תְךָ֔ יִֽהְיֶה־ נָּ֛א דְּבַר־ אֲדֹנִ֥י הַמֶּ֖לֶךְ לִמְנוּחָ֑ה כִּ֣י ׀ כְּמַלְאַ֣ךְ הָאֱלֹהִ֗ים כֵּ֣ן אֲדֹנִ֤י הַמֶּ֙לֶךְ֙ לִשְׁמֹ֙עַ֙ הַטּ֣וֹב וְהָרָ֔ע וַֽיהוָ֥ה אֱלֹהֶ֖יךָ יְהִ֥י עִמָּֽךְ׃ (לג) וַיַּ֣עַן הַמֶּ֗לֶךְ וַיֹּ֙אמֶר֙ אֶל־ הָ֣אִשָּׁ֔ה אַל־ נָ֨א תְכַחֲדִ֤י מִמֶּ֙נִּי֙ דָּבָ֔ר אֲשֶׁ֥ר אָנֹכִ֖י שֹׁאֵ֣ל אֹתָ֑ךְ וַתֹּ֙אמֶר֙ הָֽאִשָּׁ֔ה יְדַבֶּר־ נָ֖א אֲדֹנִ֥י הַמֶּֽלֶךְ׃ (לד) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ הֲיַ֥ד יוֹאָ֛ב אִתָּ֖ךְ בְּכָל־ זֹ֑את וַתַּ֣עַן הָאִשָּׁ֣ה וַתֹּ֡אמֶר חֵֽי־ נַפְשְׁךָ֩ אֲדֹנִ֨י הַמֶּ֜לֶךְ אִם־ אִ֣שׁ ׀ לְהֵמִ֣ין וּלְהַשְׂמִ֗יל מִכֹּ֤ל אֲשֶׁר־ דִּבֶּר֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ כִּֽי־ עַבְדְּךָ֤ יוֹאָב֙ ה֣וּא צִוָּ֔נִי וְה֗וּא שָׂ֚ם בְּפִ֣י שִׁפְחָֽתְךָ֔ אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (לה) לְבַעֲב֤וּר סַבֵּב֙ אֶת־ פְּנֵ֣י הַדָּבָ֔ר עָשָׂ֛ה עַבְדְּךָ֥ יוֹאָ֖ב אֶת־ הַדָּבָ֣ר הַזֶּ֑ה וַאדֹנִ֣י חָכָ֗ם כְּחָכְמַת֙ מַלְאַ֣ךְ הָאֱלֹהִ֔ים לָדַ֖עַת אֶֽת־ כָּל־ אֲשֶׁ֥ר בָּאָֽרֶץ׃ (לו) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ יוֹאָ֔ב הִנֵּה־ נָ֥א עָשִׂ֖יתִי אֶת־ הַדָּבָ֣ר הַזֶּ֑ה וְלֵ֛ךְ הָשֵׁ֥ב אֶת־ הַנַּ֖עַר אֶת־ אַבְשָׁלֽוֹם׃ (לז) וַיִּפֹּל֩ יוֹאָ֨ב אֶל־ פָּנָ֥יו אַ֛רְצָה וַיִּשְׁתַּ֖חוּ וַיְבָ֣רֶךְ אֶת־ הַמֶּ֑לֶךְ וַיֹּ֣אמֶר יוֹאָ֡ב הַיּוֹם֩ יָדַ֨ע עַבְדְּךָ֜ כִּי־ מָצָ֨אתִי חֵ֤ן בְּעֵינֶ֙יךָ֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ אֲשֶׁר־ עָשָׂ֥ה הַמֶּ֖לֶךְ אֶת־ דְּבַ֥ר (עבדו) עַבְדֶּֽךָ׃ (לח) וַיָּ֥קָם יוֹאָ֖ב וַיֵּ֣לֶךְ גְּשׁ֑וּרָה וַיָּבֵ֥א אֶת־ אַבְשָׁל֖וֹם יְרוּשָׁלִָֽם׃ (לט) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ יִסֹּ֣ב אֶל־ בֵּית֔וֹ וּפָנַ֖י לֹ֣א יִרְאֶ֑ה וַיִּסֹּ֤ב אַבְשָׁלוֹם֙ אֶל־ בֵּית֔וֹ וּפְנֵ֥י הַמֶּ֖לֶךְ לֹ֥א רָאָֽה׃ (מ) וּכְאַבְשָׁל֗וֹם לֹא־ הָיָ֧ה אִישׁ־ יָפֶ֛ה בְּכָל־ יִשְׂרָאֵ֖ל לְהַלֵּ֣ל מְאֹ֑ד מִכַּ֤ף רַגְלוֹ֙ וְעַ֣ד קָדְקֳד֔וֹ לֹא־ הָ֥יָה ב֖וֹ מֽוּם׃ (מא) וּֽבְגַלְּחוֹ֮ אֶת־ רֹאשׁוֹ֒ וְֽ֠הָיָה מִקֵּ֨ץ יָמִ֤ים ׀ לַיָּמִים֙ אֲשֶׁ֣ר יְגַלֵּ֔חַ כִּֽי־ כָבֵ֥ד עָלָ֖יו וְגִלְּח֑וֹ וְשָׁקַל֙ אֶת־ שְׂעַ֣ר רֹאשׁ֔וֹ מָאתַ֥יִם שְׁקָלִ֖ים בְּאֶ֥בֶן הַמֶּֽלֶךְ׃ (מב) וַיִּֽוָּלְד֤וּ לְאַבְשָׁלוֹם֙ שְׁלוֹשָׁ֣ה בָנִ֔ים וּבַ֥ת אַחַ֖ת וּשְׁמָ֣הּ תָּמָ֑ר הִ֣יא הָיְתָ֔ה אִשָּׁ֖ה יְפַ֥ת מַרְאֶֽה׃ (מג) וַיֵּ֧שֶׁב אַבְשָׁל֛וֹם בִּירוּשָׁלִַ֖ם שְׁנָתַ֣יִם יָמִ֑ים וּפְנֵ֥י הַמֶּ֖לֶךְ לֹ֥א רָאָֽה׃ (מד) וַיִּשְׁלַ֨ח אַבְשָׁל֜וֹם אֶל־ יוֹאָ֗ב לִשְׁלֹ֤חַ אֹתוֹ֙ אֶל־ הַמֶּ֔לֶךְ וְלֹ֥א אָבָ֖ה לָב֣וֹא אֵלָ֑יו וַיִּשְׁלַ֥ח עוֹד֙ שֵׁנִ֔ית וְלֹ֥א אָבָ֖ה לָבֽוֹא׃ (מה) וַיֹּ֨אמֶר אֶל־ עֲבָדָ֜יו רְאוּ֩ חֶלְקַ֨ת יוֹאָ֤ב אֶל־ יָדִי֙ וְלוֹ־ שָׁ֣ם שְׂעֹרִ֔ים לְכ֖וּ (והוצתיה) וְהַצִּית֣וּהָ בָאֵ֑שׁ וַיַּצִּ֜תוּ עַבְדֵ֧י אַבְשָׁל֛וֹם אֶת־ הַחֶלְקָ֖ה בָּאֵֽשׁ׃ (מו) וַיָּ֣קָם יוֹאָ֔ב וַיָּבֹ֥א אֶל־ אַבְשָׁל֖וֹם הַבָּ֑יְתָה וַיֹּ֣אמֶר אֵלָ֔יו לָ֣מָּה הִצִּ֧יתוּ עֲבָדֶ֛ךָ אֶת־ הַחֶלְקָ֥ה אֲשֶׁר־ לִ֖י בָּאֵֽשׁ׃ (מז) וַיֹּ֣אמֶר אַבְשָׁל֣וֹם אֶל־ יוֹאָ֡ב הִנֵּ֣ה שָׁלַ֣חְתִּי אֵלֶ֣יךָ ׀ לֵאמֹ֡ר בֹּ֣א הֵ֠נָּה וְאֶשְׁלְחָה֩ אֹתְךָ֨ אֶל־ הַמֶּ֜לֶךְ לֵאמֹ֗ר לָ֤מָּה בָּ֙אתִי֙ מִגְּשׁ֔וּר ט֥וֹב לִ֖י עֹ֣ד אֲנִי־ שָׁ֑ם וְעַתָּ֗ה אֶרְאֶה֙ פְּנֵ֣י הַמֶּ֔לֶךְ וְאִם־ יֶשׁ־ בִּ֥י עָוֺ֖ן וֶהֱמִתָֽנִי׃
