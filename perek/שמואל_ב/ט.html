שמואל סדר כד

(א) וְנֶאְמַ֨ן בֵּיתְךָ֧ וּמַֽמְלַכְתְּךָ֛ עַד־ עוֹלָ֖ם לְפָנֶ֑יךָ כִּֽסְאֲךָ֔ יִהְיֶ֥ה נָכ֖וֹן עַד־ עוֹלָֽם׃ (ב) כְּכֹל֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה וּכְכֹ֖ל הַחִזָּי֣וֹן הַזֶּ֑ה כֵּ֛ן דִּבֶּ֥ר נָתָ֖ן אֶל־ דָּוִֽד׃ (ג) וַיָּבֹא֙ הַמֶּ֣לֶךְ דָּוִ֔ד וַיֵּ֖שֶׁב לִפְנֵ֣י יְהוָ֑ה וַיֹּ֗אמֶר מִ֣י אָנֹכִ֞י אֲדֹנָ֤י יְהוִה֙ וּמִ֣י בֵיתִ֔י כִּ֥י הֲבִיאֹתַ֖נִי עַד־ הֲלֹֽם׃ (ד) וַתִּקְטַן֩ ע֨וֹד זֹ֤את בְּעֵינֶ֙יךָ֙ אֲדֹנָ֣י יְהוִ֔ה וַתְּדַבֵּ֛ר גַּ֥ם אֶל־ בֵּֽית־ עַבְדְּךָ֖ לְמֵֽרָח֑וֹק וְזֹ֛את תּוֹרַ֥ת הָאָדָ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ה) וּמַה־ יּוֹסִ֥יף דָּוִ֛ד ע֖וֹד לְדַבֵּ֣ר אֵלֶ֑יךָ וְאַתָּ֛ה יָדַ֥עְתָּ אֶֽת־ עַבְדְּךָ֖ אֲדֹנָ֥י יְהוִֽה׃ (ו) בַּעֲב֤וּר דְּבָֽרְךָ֙ וּֽכְלִבְּךָ֔ עָשִׂ֕יתָ אֵ֥ת כָּל־ הַגְּדוּלָּ֖ה הַזֹּ֑את לְהוֹדִ֖יעַ אֶת־ עַבְדֶּֽךָ׃ (ז) עַל־ כֵּ֥ן גָּדַ֖לְתָּ אֲדֹנָ֣י יְהוִ֑ה כִּֽי־ אֵ֣ין כָּמ֗וֹךָ וְאֵ֤ין אֱלֹהִים֙ זֽוּלָתֶ֔ךָ בְּכֹ֥ל אֲשֶׁר־ שָׁמַ֖עְנוּ בְּאָזְנֵֽינוּ׃ (ח) וּמִ֤י כְעַמְּךָ֙ כְּיִשְׂרָאֵ֔ל גּ֥וֹי אֶחָ֖ד בָּאָ֑רֶץ אֲשֶׁ֣ר הָלְכֽוּ־ אֱ֠לֹהִים לִפְדּֽוֹת־ ל֨וֹ לְעָ֜ם וְלָשׂ֧וּם ל֣וֹ שֵׁ֗ם וְלַעֲשׂ֨וֹת לָכֶ֜ם הַגְּדוּלָּ֤ה וְנֹֽרָאוֹת֙ לְאַרְצֶ֔ךָ מִפְּנֵ֣י עַמְּךָ֗ אֲשֶׁ֨ר פָּדִ֤יתָ לְּךָ֙ מִמִּצְרַ֔יִם גּוֹיִ֖ם וֵאלֹהָֽיו׃ (ט) וַתְּכ֣וֹנֵֽן לְ֠ךָ אֶת־ עַמְּךָ֨ יִשְׂרָאֵ֧ל ׀ לְךָ֛ לְעָ֖ם עַד־ עוֹלָ֑ם וְאַתָּ֣ה יְהוָ֔ה הָיִ֥יתָ לָהֶ֖ם לֵאלֹהִֽים׃ (י) וְעַתָּה֙ יְהוָ֣ה אֱלֹהִ֔ים הַדָּבָ֗ר אֲשֶׁ֨ר דִּבַּ֤רְתָּ עַֽל־ עַבְדְּךָ֙ וְעַל־ בֵּית֔וֹ הָקֵ֖ם עַד־ עוֹלָ֑ם וַעֲשֵׂ֖ה כַּאֲשֶׁ֥ר דִּבַּֽרְתָּ׃ (יא) וְיִגְדַּ֨ל שִׁמְךָ֤ עַד־ עוֹלָם֙ לֵאמֹ֔ר יְהוָ֣ה צְבָא֔וֹת אֱלֹהִ֖ים עַל־ יִשְׂרָאֵ֑ל וּבֵית֙ עַבְדְּךָ֣ דָוִ֔ד יִהְיֶ֥ה נָכ֖וֹן לְפָנֶֽיךָ׃ (יב) כִּֽי־ אַתָּה֩ יְהוָ֨ה צְבָא֜וֹת אֱלֹהֵ֣י יִשְׂרָאֵ֗ל גָּלִ֜יתָה אֶת־ אֹ֤זֶן עַבְדְּךָ֙ לֵאמֹ֔ר בַּ֖יִת אֶבְנֶה־ לָּ֑ךְ עַל־ כֵּ֗ן מָצָ֤א עַבְדְּךָ֙ אֶת־ לִבּ֔וֹ לְהִתְפַּלֵּ֣ל אֵלֶ֔יךָ אֶת־ הַתְּפִלָּ֖ה הַזֹּֽאת׃ (יג) וְעַתָּ֣ה ׀ אֲדֹנָ֣י יְהוִ֗ה אַתָּה־ הוּא֙ הָֽאֱלֹהִ֔ים וּדְבָרֶ֖יךָ יִהְי֣וּ אֱמֶ֑ת וַתְּדַבֵּר֙ אֶֽל־ עַבְדְּךָ֔ אֶת־ הַטּוֹבָ֖ה הַזֹּֽאת׃ (יד) וְעַתָּ֗ה הוֹאֵל֙ וּבָרֵךְ֙ אֶת־ בֵּ֣ית עַבְדְּךָ֔ לִהְי֥וֹת לְעוֹלָ֖ם לְפָנֶ֑יךָ כִּֽי־ אַתָּ֞ה אֲדֹנָ֤י יְהוִה֙ דִּבַּ֔רְתָּ וּמִבִּרְכָ֣תְךָ֔ יְבֹרַ֥ךְ בֵּֽית־ עַבְדְּךָ֖ לְעוֹלָֽם׃ (טו) וַֽיְהִי֙ אַֽחֲרֵי־ כֵ֔ן וַיַּ֥ךְ דָּוִ֛ד אֶת־ פְּלִשְׁתִּ֖ים וַיַּכְנִיעֵ֑ם וַיִּקַּ֥ח דָּוִ֛ד אֶת־ מֶ֥תֶג הָאַמָּ֖ה מִיַּ֥ד פְּלִשְׁתִּֽים׃ (טז) וַיַּ֣ךְ אֶת־ מוֹאָ֗ב וַֽיְמַדְּדֵ֤ם בַּחֶ֙בֶל֙ הַשְׁכֵּ֣ב אוֹתָ֣ם אַ֔רְצָה וַיְמַדֵּ֤ד שְׁנֵֽי־ חֲבָלִים֙ לְהָמִ֔ית וּמְלֹ֥א הַחֶ֖בֶל לְהַחֲי֑וֹת וַתְּהִ֤י מוֹאָב֙ לְדָוִ֔ד לַעֲבָדִ֖ים נֹשְׂאֵ֥י מִנְחָֽה׃ (יז) וַיַּ֣ךְ דָּוִ֔ד אֶת־ הֲדַדְעֶ֥זֶר בֶּן־ רְחֹ֖ב מֶ֣לֶךְ צוֹבָ֑ה בְּלֶכְתּ֕וֹ לְהָשִׁ֥יב יָד֖וֹ בִּֽנְהַר־ פְּרָֽת׃ (יח) וַיִּלְכֹּ֨ד דָּוִ֜ד מִמֶּ֗נּוּ אֶ֤לֶף וּשְׁבַע־ מֵאוֹת֙ פָּרָשִׁ֔ים וְעֶשְׂרִ֥ים אֶ֖לֶף אִ֣ישׁ רַגְלִ֑י וַיְעַקֵּ֤ר דָּוִד֙ אֶת־ כָּל־ הָרֶ֔כֶב וַיּוֹתֵ֥ר מִמֶּ֖נּוּ מֵ֥אָה רָֽכֶב׃ (יט) וַתָּבֹא֙ אֲרַ֣ם דַּמֶּ֔שֶׂק לַעְזֹ֕ר לַהֲדַדְעֶ֖זֶר מֶ֣לֶךְ צוֹבָ֑ה וַיַּ֤ךְ דָּוִד֙ בַּֽאֲרָ֔ם עֶשְׂרִֽים־ וּשְׁנַ֥יִם אֶ֖לֶף אִֽישׁ׃ (כ) וַיָּ֨שֶׂם דָּוִ֤ד נְצִבִים֙ בַּאֲרַ֣ם דַּמֶּ֔שֶׂק וַתְּהִ֤י אֲרָם֙ לְדָוִ֔ד לַעֲבָדִ֖ים נוֹשְׂאֵ֣י מִנְחָ֑ה וַיֹּ֤שַׁע יְהוָה֙ אֶת־ דָּוִ֔ד בְּכֹ֖ל אֲשֶׁ֥ר הָלָֽךְ׃ (כא) וַיִּקַּ֣ח דָּוִ֗ד אֵ֚ת שִׁלְטֵ֣י הַזָּהָ֔ב אֲשֶׁ֣ר הָי֔וּ אֶ֖ל עַבְדֵ֣י הֲדַדְעָ֑זֶר וַיְבִיאֵ֖ם יְרוּשָׁלִָֽם׃ (כב) וּמִבֶּ֥טַח וּמִבֵּֽרֹתַ֖י עָרֵ֣י הֲדַדְעָ֑זֶר לָקַ֞ח הַמֶּ֧לֶךְ דָּוִ֛ד נְחֹ֖שֶׁת הַרְבֵּ֥ה מְאֹֽד׃ (כג) וַיִּשְׁמַ֕ע תֹּ֖עִי מֶ֣לֶךְ חֲמָ֑ת כִּ֚י הִכָּ֣ה דָוִ֔ד אֵ֖ת כָּל־ חֵ֥יל הֲדַדְעָֽזֶר׃ (כד) וַיִּשְׁלַ֣ח תֹּ֣עִי אֶת־ יֽוֹרָם־ בְּנ֣וֹ אֶל־ הַמֶּֽלֶךְ־ דָּ֠וִד לִשְׁאָל־ ל֨וֹ לְשָׁל֜וֹם וּֽלְבָרֲכ֗וֹ עַל֩ אֲשֶׁ֨ר נִלְחַ֤ם בַּהֲדַדְעֶ֙זֶר֙ וַיַּכֵּ֔הוּ כִּי־ אִ֛ישׁ מִלְחֲמ֥וֹת תֹּ֖עִי הָיָ֣ה הֲדַדְעָ֑זֶר וּבְיָד֗וֹ הָי֛וּ כְּלֵֽי־ כֶ֥סֶף וּכְלֵֽי־ זָהָ֖ב וּכְלֵ֥י נְחֹֽשֶׁת׃ (כה) גַּם־ אֹתָ֕ם הִקְדִּ֛ישׁ הַמֶּ֥לֶךְ דָּוִ֖ד לַֽיהוָ֑ה עִם־ הַכֶּ֤סֶף וְהַזָּהָב֙ אֲשֶׁ֣ר הִקְדִּ֔ישׁ מִכָּל־ הַגּוֹיִ֖ם אֲשֶׁ֥ר כִּבֵּֽשׁ׃ (כו) מֵאֲרָ֤ם וּמִמּוֹאָב֙ וּמִבְּנֵ֣י עַמּ֔וֹן וּמִפְּלִשְׁתִּ֖ים וּמֵֽעֲמָלֵ֑ק וּמִשְּׁלַ֛ל הֲדַדְעֶ֥זֶר בֶּן־ רְחֹ֖ב מֶ֥לֶךְ צוֹבָֽה׃ (כז) וַיַּ֤עַשׂ דָּוִד֙ שֵׁ֔ם בְּשֻׁב֕וֹ מֵהַכּוֹת֥וֹ אֶת־ אֲרָ֖ם בְּגֵיא־ מֶ֑לַח שְׁמוֹנָ֥ה עָשָׂ֖ר אָֽלֶף׃ (כח) וַיָּ֨שֶׂם בֶּאֱד֜וֹם נְצִבִ֗ים בְּכָל־ אֱדוֹם֙ שָׂ֣ם נְצִבִ֔ים וַיְהִ֥י כָל־ אֱד֖וֹם עֲבָדִ֣ים לְדָוִ֑ד וַיּ֤וֹשַׁע יְהוָה֙ אֶת־ דָּוִ֔ד בְּכֹ֖ל אֲשֶׁ֥ר הָלָֽךְ׃ (כט) וַיִּמְלֹ֥ךְ דָּוִ֖ד עַל־ כָּל־ יִשְׂרָאֵ֑ל וַיְהִ֣י דָוִ֗ד עֹשֶׂ֛ה מִשְׁפָּ֥ט וּצְדָקָ֖ה לְכָל־ עַמּֽוֹ׃ (ל) וְיוֹאָ֥ב בֶּן־ צְרוּיָ֖ה עַל־ הַצָּבָ֑א וִיהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד מַזְכִּֽיר׃ (לא) וְצָד֧וֹק בֶּן־ אֲחִיט֛וּב וַאֲחִימֶ֥לֶךְ בֶּן־ אֶבְיָתָ֖ר כֹּהֲנִ֑ים וּשְׂרָיָ֖ה סוֹפֵֽר׃ (לב) וּבְנָיָ֙הוּ֙ בֶּן־ יְה֣וֹיָדָ֔ע וְהַכְּרֵתִ֖י וְהַפְּלֵתִ֑י וּבְנֵ֥י דָוִ֖ד כֹּהֲנִ֥ים הָיֽוּ׃ תחילת פרק ט (לג) וַיֹּ֣אמֶר דָּוִ֔ד הֲכִ֣י יֶשׁ־ ע֔וֹד אֲשֶׁ֥ר נוֹתַ֖ר לְבֵ֣ית שָׁא֑וּל וְאֶעֱשֶׂ֤ה עִמּוֹ֙ חֶ֔סֶד בַּעֲב֖וּר יְהוֹנָתָֽן׃ (לד) וּלְבֵ֨ית שָׁא֥וּל עֶ֙בֶד֙ וּשְׁמ֣וֹ צִיבָ֔א וַיִּקְרְאוּ־ ל֖וֹ אֶל־ דָּוִ֑ד וַיֹּ֨אמֶר הַמֶּ֧לֶךְ אֵלָ֛יו הַאַתָּ֥ה צִיבָ֖א וַיֹּ֥אמֶר עַבְדֶּֽךָ׃ (לה) וַיֹּ֣אמֶר הַמֶּ֗לֶךְ הַאֶ֨פֶס ע֥וֹד אִישׁ֙ לְבֵ֣ית שָׁא֔וּל וְאֶעֱשֶׂ֥ה עִמּ֖וֹ חֶ֣סֶד אֱלֹהִ֑ים וַיֹּ֤אמֶר צִיבָא֙ אֶל־ הַמֶּ֔לֶךְ ע֛וֹד בֵּ֥ן לִיהוֹנָתָ֖ן נְכֵ֥ה רַגְלָֽיִם׃ (לו) וַיֹּֽאמֶר־ ל֥וֹ הַמֶּ֖לֶךְ אֵיפֹ֣ה ה֑וּא וַיֹּ֤אמֶר צִיבָא֙ אֶל־ הַמֶּ֔לֶךְ הִנֵּה־ ה֗וּא בֵּ֛ית מָכִ֥יר בֶּן־ עַמִּיאֵ֖ל בְּל֥וֹ דְבָֽר׃ (לז) וַיִּשְׁלַ֖ח הַמֶּ֣לֶךְ דָּוִ֑ד וַיִּקָּחֵ֗הוּ מִבֵּ֛ית מָכִ֥יר בֶּן־ עַמִּיאֵ֖ל מִלּ֥וֹ דְבָֽר׃ (לח) וַ֠יָּבֹא מְפִיבֹ֨שֶׁת בֶּן־ יְהוֹנָתָ֤ן בֶּן־ שָׁאוּל֙ אֶל־ דָּוִ֔ד וַיִּפֹּ֥ל עַל־ פָּנָ֖יו וַיִּשְׁתָּ֑חוּ וַיֹּ֤אמֶר דָּוִד֙ מְפִיבֹ֔שֶׁת וַיֹּ֖אמֶר הִנֵּ֥ה עַבְדֶּֽךָ׃ (לט) וַיֹּאמֶר֩ ל֨וֹ דָוִ֜ד אַל־ תִּירָ֗א כִּ֣י עָשֹׂה֩ אֶעֱשֶׂ֨ה עִמְּךָ֥ חֶ֙סֶד֙ בַּֽעֲבוּר֙ יְהוֹנָתָ֣ן אָבִ֔יךָ וַהֲשִׁבֹתִ֣י לְךָ֔ אֶֽת־ כָּל־ שְׂדֵ֖ה שָׁא֣וּל אָבִ֑יךָ וְאַתָּ֗ה תֹּ֥אכַל לֶ֛חֶם עַל־ שֻׁלְחָנִ֖י תָּמִֽיד׃ (מ) וַיִּשְׁתַּ֕חוּ וַיֹּ֖אמֶר מֶ֣ה עַבְדֶּ֑ךָ כִּ֣י פָנִ֔יתָ אֶל־ הַכֶּ֥לֶב הַמֵּ֖ת אֲשֶׁ֥ר כָּמֽוֹנִי׃ (מא) וַיִּקְרָ֣א הַמֶּ֗לֶךְ אֶל־ צִיבָ֛א נַ֥עַר שָׁא֖וּל וַיֹּ֣אמֶר אֵלָ֑יו כֹּל֩ אֲשֶׁ֨ר הָיָ֤ה לְשָׁאוּל֙ וּלְכָל־ בֵּית֔וֹ נָתַ֖תִּי לְבֶן־ אֲדֹנֶֽיךָ׃ (מב) וְעָבַ֣דְתָּ לּ֣וֹ אֶֽת־ הָאֲדָמָ֡ה אַתָּה֩ וּבָנֶ֨יךָ וַעֲבָדֶ֜יךָ וְהֵבֵ֗אתָ וְהָיָ֨ה לְבֶן־ אֲדֹנֶ֤יךָ לֶּ֙חֶם֙ וַאֲכָל֔וֹ וּמְפִיבֹ֙שֶׁת֙ בֶּן־ אֲדֹנֶ֔יךָ יֹאכַ֥ל תָּמִ֛יד לֶ֖חֶם עַל־ שֻׁלְחָנִ֑י וּלְצִיבָ֗א חֲמִשָּׁ֥ה עָשָׂ֛ר בָּנִ֖ים וְעֶשְׂרִ֥ים עֲבָדִֽים׃ (מג) וַיֹּ֤אמֶר צִיבָא֙ אֶל־ הַמֶּ֔לֶךְ כְּכֹל֩ אֲשֶׁ֨ר יְצַוֶּ֜ה אֲדֹנִ֤י הַמֶּ֙לֶךְ֙ אֶת־ עַבְדּ֔וֹ כֵּ֖ן יַעֲשֶׂ֣ה עַבְדֶּ֑ךָ וּמְפִיבֹ֗שֶׁת אֹכֵל֙ עַל־ שֻׁלְחָנִ֔י כְּאַחַ֖ד מִבְּנֵ֥י הַמֶּֽלֶךְ׃ (מד) וְלִמְפִיבֹ֥שֶׁת בֵּן־ קָטָ֖ן וּשְׁמ֣וֹ מִיכָ֑א וְכֹל֙ מוֹשַׁ֣ב בֵּית־ צִיבָ֔א עֲבָדִ֖ים לִמְפִיבֹֽשֶׁת׃ (מה) וּמְפִיבֹ֗שֶׁת יֹשֵׁב֙ בִּיר֣וּשָׁלִַ֔ם כִּ֣י עַל־ שֻׁלְחַ֥ן הַמֶּ֛לֶךְ תָּמִ֖יד ה֣וּא אֹכֵ֑ל וְה֥וּא פִסֵּ֖חַ שְׁתֵּ֥י רַגְלָֽיו׃ סוף פרק ט (מו) וַֽיְהִי֙ אַֽחֲרֵי־ כֵ֔ן וַיָּ֕מָת מֶ֖לֶךְ בְּנֵ֣י עַמּ֑וֹן וַיִּמְלֹ֛ךְ חָנ֥וּן בְּנ֖וֹ תַּחְתָּֽיו׃ (מז) וַיֹּ֨אמֶר דָּוִ֜ד אֶעֱשֶׂה־ חֶ֣סֶד ׀ עִם־ חָנ֣וּן בֶּן־ נָחָ֗שׁ כַּאֲשֶׁר֩ עָשָׂ֨ה אָבִ֤יו עִמָּדִי֙ חֶ֔סֶד וַיִּשְׁלַ֨ח דָּוִ֧ד לְנַחֲמ֛וֹ בְּיַד־ עֲבָדָ֖יו אֶל־ אָבִ֑יו וַיָּבֹ֙אוּ֙ עַבְדֵ֣י דָוִ֔ד אֶ֖רֶץ בְּנֵ֥י עַמּֽוֹן׃ (מח) וַיֹּאמְרוּ֩ שָׂרֵ֨י בְנֵֽי־ עַמּ֜וֹן אֶל־ חָנ֣וּן אֲדֹֽנֵיהֶ֗ם הַֽמְכַבֵּ֨ד דָּוִ֤ד אֶת־ אָבִ֙יךָ֙ בְּעֵינֶ֔יךָ כִּֽי־ שָׁלַ֥ח לְךָ֖ מְנַֽחֲמִ֑ים הֲ֠לוֹא בַּעֲב֞וּר חֲק֤וֹר אֶת־ הָעִיר֙ וּלְרַגְּלָ֣הּ וּלְהָפְכָ֔הּ שָׁלַ֥ח דָּוִ֛ד אֶת־ עֲבָדָ֖יו אֵלֶֽיךָ׃ (מט) וַיִּקַּ֨ח חָנ֜וּן אֶת־ עַבְדֵ֣י דָוִ֗ד וַיְגַלַּח֙ אֶת־ חֲצִ֣י זְקָנָ֔ם וַיִּכְרֹ֧ת אֶת־ מַדְוֵיהֶ֛ם בַּחֵ֖צִי עַ֣ד שְׁתֽוֹתֵיהֶ֑ם וַֽיְשַׁלְּחֵֽם׃ (נ) וַיַּגִּ֤דוּ לְדָוִד֙ וַיִּשְׁלַ֣ח לִקְרָאתָ֔ם כִּֽי־ הָי֥וּ הָאֲנָשִׁ֖ים נִכְלָמִ֣ים מְאֹ֑ד וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ שְׁב֣וּ בִֽירֵח֔וֹ עַד־ יְצַמַּ֥ח זְקַנְכֶ֖ם וְשַׁבְתֶּֽם׃ (נא) וַיִּרְאוּ֙ בְּנֵ֣י עַמּ֔וֹן כִּ֥י נִבְאֲשׁ֖וּ בְּדָוִ֑ד וַיִּשְׁלְח֣וּ בְנֵֽי־ עַמּ֡וֹן וַיִּשְׂכְּרוּ֩ אֶת־ אֲרַ֨ם בֵּית־ רְח֜וֹב וְאֶת־ אֲרַ֣ם צוֹבָ֗א עֶשְׂרִ֥ים אֶ֙לֶף֙ רַגְלִ֔י וְאֶת־ מֶ֤לֶךְ מַֽעֲכָה֙ אֶ֣לֶף אִ֔ישׁ וְאִ֣ישׁ ט֔וֹב שְׁנֵים־ עָשָׂ֥ר אֶ֖לֶף אִֽישׁ׃ (נב) וַיִּשְׁמַ֖ע דָּוִ֑ד וַיִּשְׁלַח֙ אֶת־ יוֹאָ֔ב וְאֵ֥ת כָּל־ הַצָּבָ֖א הַגִּבֹּרִֽים׃ (נג) וַיֵּֽצְאוּ֙ בְּנֵ֣י עַמּ֔וֹן וַיַּעַרְכ֥וּ מִלְחָמָ֖ה פֶּ֣תַח הַשָּׁ֑עַר וַאֲרַ֨ם צוֹבָ֤א וּרְחוֹב֙ וְאִֽישׁ־ ט֣וֹב וּמַֽעֲכָ֔ה לְבַדָּ֖ם בַּשָּׂדֶֽה׃ (נד) וַיַּ֣רְא יוֹאָ֗ב כִּֽי־ הָיְתָ֤ה אֵלָיו֙ פְּנֵ֣י הַמִּלְחָמָ֔ה מִפָּנִ֖ים וּמֵֽאָח֑וֹר וַיִּבְחַ֗ר מִכֹּל֙ בְּחוּרֵ֣י (בישראל) יִשְׂרָאֵ֔ל וַֽיַּעֲרֹ֖ךְ לִקְרַ֥את אֲרָֽם׃ (נה) וְאֵת֙ יֶ֣תֶר הָעָ֔ם נָתַ֕ן בְּיַ֖ד אַבְשַׁ֣י אָחִ֑יו וַֽיַּעֲרֹ֕ךְ לִקְרַ֖את בְּנֵ֥י עַמּֽוֹן׃ (נו) וַיֹּ֗אמֶר אִם־ תֶּחֱזַ֤ק אֲרָם֙ מִמֶּ֔נִּי וְהָיִ֥תָה לִּ֖י לִֽישׁוּעָ֑ה וְאִם־ בְּנֵ֤י עַמּוֹן֙ יֶחֱזְק֣וּ מִמְּךָ֔ וְהָלַכְתִּ֖י לְהוֹשִׁ֥יעַֽ לָֽךְ׃ (נז) חֲזַ֤ק וְנִתְחַזַּק֙ בְּעַד־ עַמֵּ֔נוּ וּבְעַ֖ד עָרֵ֣י אֱלֹהֵ֑ינוּ וַֽיהוָ֔ה יַעֲשֶׂ֥ה הַטּ֖וֹב בְּעֵינָֽיו׃
