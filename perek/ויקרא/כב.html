ויקרא פרק כב לפי החלוקה לסדרים

פרק כב בספר ויקרא (33 פסוקים) נמצא בויקרא סדר יז (פסוקים כה–מ) ובויקרא סדר יח (פסוקים א–יז).

ויקרא סדר יז

(א) וַיֹּ֤אמֶר יְהוָה֙ אֶל־ מֹשֶׁ֔ה אֱמֹ֥ר אֶל־ הַכֹּהֲנִ֖ים בְּנֵ֣י אַהֲרֹ֑ן וְאָמַרְתָּ֣ אֲלֵהֶ֔ם לְנֶ֥פֶשׁ לֹֽא־ יִטַּמָּ֖א בְּעַמָּֽיו׃ (ב) כִּ֚י אִם־ לִשְׁאֵר֔וֹ הַקָּרֹ֖ב אֵלָ֑יו לְאִמּ֣וֹ וּלְאָבִ֔יו וְלִבְנ֥וֹ וּלְבִתּ֖וֹ וּלְאָחִֽיו׃ (ג) וְלַאֲחֹת֤וֹ הַבְּתוּלָה֙ הַקְּרוֹבָ֣ה אֵלָ֔יו אֲשֶׁ֥ר לֹֽא־ הָיְתָ֖ה לְאִ֑ישׁ לָ֖הּ יִטַּמָּֽא׃ (ד) לֹ֥א יִטַּמָּ֖א בַּ֣עַל בְּעַמָּ֑יו לְהֵ֖חַלּֽוֹ׃ (ה) לֹֽא־ (יקרחה) יִקְרְח֤וּ קָרְחָה֙ בְּרֹאשָׁ֔ם וּפְאַ֥ת זְקָנָ֖ם לֹ֣א יְגַלֵּ֑חוּ וּבִ֨בְשָׂרָ֔ם לֹ֥א יִשְׂרְט֖וּ שָׂרָֽטֶת׃ (ו) קְדֹשִׁ֤ים יִהְיוּ֙ לֵאלֹ֣הֵיהֶ֔ם וְלֹ֣א יְחַלְּל֔וּ שֵׁ֖ם אֱלֹהֵיהֶ֑ם כִּי֩ אֶת־ אִשֵּׁ֨י יְהוָ֜ה לֶ֧חֶם אֱלֹהֵיהֶ֛ם הֵ֥ם מַקְרִיבִ֖ם וְהָ֥יוּ קֹֽדֶשׁ׃ (ז) אִשָּׁ֨ה זֹנָ֤ה וַחֲלָלָה֙ לֹ֣א יִקָּ֔חוּ וְאִשָּׁ֛ה גְּרוּשָׁ֥ה מֵאִישָׁ֖הּ לֹ֣א יִקָּ֑חוּ כִּֽי־ קָדֹ֥שׁ ה֖וּא לֵאלֹהָֽיו׃ (ח) וְקִדַּשְׁתּ֔וֹ כִּֽי־ אֶת־ לֶ֥חֶם אֱלֹהֶ֖יךָ ה֣וּא מַקְרִ֑יב קָדֹשׁ֙ יִֽהְיֶה־ לָּ֔ךְ כִּ֣י קָד֔וֹשׁ אֲנִ֥י יְהוָ֖ה מְקַדִּשְׁכֶֽם׃ (ט) וּבַת֙ אִ֣ישׁ כֹּהֵ֔ן כִּ֥י תֵחֵ֖ל לִזְנ֑וֹת אֶת־ אָבִ֙יהָ֙ הִ֣יא מְחַלֶּ֔לֶת בָּאֵ֖שׁ תִּשָּׂרֵֽף׃ (י) וְהַכֹּהֵן֩ הַגָּד֨וֹל מֵאֶחָ֜יו אֲ‍ֽשֶׁר־ יוּצַ֥ק עַל־ רֹאשׁ֣וֹ ׀ שֶׁ֤מֶן הַמִּשְׁחָה֙ וּמִלֵּ֣א אֶת־ יָד֔וֹ לִלְבֹּ֖שׁ אֶת־ הַבְּגָדִ֑ים אֶת־ רֹאשׁוֹ֙ לֹ֣א יִפְרָ֔ע וּבְגָדָ֖יו לֹ֥א יִפְרֹֽם׃ (יא) וְעַ֛ל כָּל־ נַפְשֹׁ֥ת מֵ֖ת לֹ֣א יָבֹ֑א לְאָבִ֥יו וּלְאִמּ֖וֹ לֹ֥א יִטַּמָּֽא׃ (יב) וּמִן־ הַמִּקְדָּשׁ֙ לֹ֣א יֵצֵ֔א וְלֹ֣א יְחַלֵּ֔ל אֵ֖ת מִקְדַּ֣שׁ אֱלֹהָ֑יו כִּ֡י נֵ֠זֶר שֶׁ֣מֶן מִשְׁחַ֧ת אֱלֹהָ֛יו עָלָ֖יו אֲנִ֥י יְהוָֽה׃ (יג) וְה֕וּא אִשָּׁ֥ה בִבְתוּלֶ֖יהָ יִקָּֽח׃ (יד) אַלְמָנָ֤ה וּגְרוּשָׁה֙ וַחֲלָלָ֣ה זֹנָ֔ה אֶת־ אֵ֖לֶּה לֹ֣א יִקָּ֑ח כִּ֛י אִם־ בְּתוּלָ֥ה מֵעַמָּ֖יו יִקַּ֥ח אִשָּֽׁה׃ (טו) וְלֹֽא־ יְחַלֵּ֥ל זַרְע֖וֹ בְּעַמָּ֑יו כִּ֛י אֲנִ֥י יְהוָ֖ה מְקַדְּשֽׁוֹ׃ (טז) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (יז) דַּבֵּ֥ר אֶֽל־ אַהֲרֹ֖ן לֵאמֹ֑ר אִ֣ישׁ מִֽזַּרְעֲךָ֞ לְדֹרֹתָ֗ם אֲשֶׁ֨ר יִהְיֶ֥ה בוֹ֙ מ֔וּם לֹ֣א יִקְרַ֔ב לְהַקְרִ֖יב לֶ֥חֶם אֱלֹהָֽיו׃ (יח) כִּ֥י כָל־ אִ֛ישׁ אֲשֶׁר־ בּ֥וֹ מ֖וּם לֹ֣א יִקְרָ֑ב אִ֤ישׁ עִוֵּר֙ א֣וֹ פִסֵּ֔חַ א֥וֹ חָרֻ֖ם א֥וֹ שָׂרֽוּעַ׃ (יט) א֣וֹ אִ֔ישׁ אֲשֶׁר־ יִהְיֶ֥ה ב֖וֹ שֶׁ֣בֶר רָ֑גֶל א֖וֹ שֶׁ֥בֶר יָֽד׃ (כ) אֽוֹ־ גִבֵּ֣ן אוֹ־ דַ֔ק א֖וֹ תְּבַלֻּ֣ל בְּעֵינ֑וֹ א֤וֹ גָרָב֙ א֣וֹ יַלֶּ֔פֶת א֖וֹ מְר֥וֹחַ אָֽשֶׁךְ׃ (כא) כָּל־ אִ֞ישׁ אֲשֶׁר־ בּ֣וֹ מ֗וּם מִזֶּ֙רַע֙ אַהֲרֹ֣ן הַכֹּהֵ֔ן לֹ֣א יִגַּ֔שׁ לְהַקְרִ֖יב אֶת־ אִשֵּׁ֣י יְהוָ֑ה מ֣וּם בּ֔וֹ אֵ֚ת לֶ֣חֶם אֱלֹהָ֔יו לֹ֥א יִגַּ֖שׁ לְהַקְרִֽיב׃ (כב) לֶ֣חֶם אֱלֹהָ֔יו מִקָּדְשֵׁ֖י הַקֳּדָשִׁ֑ים וּמִן־ הַקֳּדָשִׁ֖ים יֹאכֵֽל׃ (כג) אַ֣ךְ אֶל־ הַפָּרֹ֜כֶת לֹ֣א יָבֹ֗א וְאֶל־ הַמִּזְבֵּ֛חַ לֹ֥א יִגַּ֖שׁ כִּֽי־ מ֣וּם בּ֑וֹ וְלֹ֤א יְחַלֵּל֙ אֶת־ מִקְדָּשַׁ֔י כִּ֛י אֲנִ֥י יְהוָ֖ה מְקַדְּשָֽׁם׃ (כד) וַיְדַבֵּ֣ר מֹשֶׁ֔ה אֶֽל־ אַהֲרֹ֖ן וְאֶל־ בָּנָ֑יו וְאֶֽל־ כָּל־ בְּנֵ֖י יִשְׂרָאֵֽל׃ תחילת פרק כב (כה) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (כו) דַּבֵּ֨ר אֶֽל־ אַהֲרֹ֜ן וְאֶל־ בָּנָ֗יו וְיִנָּֽזְרוּ֙ מִקָּדְשֵׁ֣י בְנֵֽי־ יִשְׂרָאֵ֔ל וְלֹ֥א יְחַלְּל֖וּ אֶת־ שֵׁ֣ם קָדְשִׁ֑י אֲשֶׁ֨ר הֵ֧ם מַקְדִּשִׁ֛ים לִ֖י אֲנִ֥י יְהוָֽה׃ (כז) אֱמֹ֣ר אֲלֵהֶ֗ם לְדֹרֹ֨תֵיכֶ֜ם כָּל־ אִ֣ישׁ ׀ אֲשֶׁר־ יִקְרַ֣ב מִכָּל־ זַרְעֲכֶ֗ם אֶל־ הַקֳּדָשִׁים֙ אֲשֶׁ֨ר יַקְדִּ֤ישׁוּ בְנֵֽי־ יִשְׂרָאֵל֙ לַֽיהוָ֔ה וְטֻמְאָת֖וֹ עָלָ֑יו וְנִכְרְתָ֞ה הַנֶּ֧פֶשׁ הַהִ֛וא מִלְּפָנַ֖י אֲנִ֥י יְהוָֽה׃ (כח) אִ֣ישׁ אִ֞ישׁ מִזֶּ֣רַע אַהֲרֹ֗ן וְה֤וּא צָר֙וּעַ֙ א֣וֹ זָ֔ב בַּקֳּדָשִׁים֙ לֹ֣א יֹאכַ֔ל עַ֖ד אֲשֶׁ֣ר יִטְהָ֑ר וְהַנֹּגֵ֙עַ֙ בְּכָל־ טְמֵא־ נֶ֔פֶשׁ א֣וֹ אִ֔ישׁ אֲשֶׁר־ תֵּצֵ֥א מִמֶּ֖נּוּ שִׁכְבַת־ זָֽרַע׃ (כט) אוֹ־ אִישׁ֙ אֲשֶׁ֣ר יִגַּ֔ע בְּכָל־ שֶׁ֖רֶץ אֲשֶׁ֣ר יִטְמָא־ ל֑וֹ א֤וֹ בְאָדָם֙ אֲשֶׁ֣ר יִטְמָא־ ל֔וֹ לְכֹ֖ל טֻמְאָתֽוֹ׃ (ל) נֶ֚פֶשׁ אֲשֶׁ֣ר תִּגַּע־ בּ֔וֹ וְטָמְאָ֖ה עַד־ הָעָ֑רֶב וְלֹ֤א יֹאכַל֙ מִן־ הַקֳּדָשִׁ֔ים כִּ֛י אִם־ רָחַ֥ץ בְּשָׂר֖וֹ בַּמָּֽיִם׃ (לא) וּבָ֥א הַשֶּׁ֖מֶשׁ וְטָהֵ֑ר וְאַחַר֙ יֹאכַ֣ל מִן־ הַקֳּדָשִׁ֔ים כִּ֥י לַחְמ֖וֹ הֽוּא׃ (לב) נְבֵלָ֧ה וּטְרֵפָ֛ה לֹ֥א יֹאכַ֖ל לְטָמְאָה־ בָ֑הּ אֲנִ֖י יְהוָֽה׃ (לג) וְשָׁמְר֣וּ אֶת־ מִשְׁמַרְתִּ֗י וְלֹֽא־ יִשְׂא֤וּ עָלָיו֙ חֵ֔טְא וּמֵ֥תוּ ב֖וֹ כִּ֣י יְחַלְּלֻ֑הוּ אֲנִ֥י יְהוָ֖ה מְקַדְּשָֽׁם׃ (לד) וְכָל־ זָ֖ר לֹא־ יֹ֣אכַל קֹ֑דֶשׁ תּוֹשַׁ֥ב כֹּהֵ֛ן וְשָׂכִ֖יר לֹא־ יֹ֥אכַל קֹֽדֶשׁ׃ (לה) וְכֹהֵ֗ן כִּֽי־ יִקְנֶ֥ה נֶ֙פֶשׁ֙ קִנְיַ֣ן כַּסְפּ֔וֹ ה֖וּא יֹ֣אכַל בּ֑וֹ וִילִ֣יד בֵּית֔וֹ הֵ֖ם יֹאכְל֥וּ בְלַחְמֽוֹ׃ (לו) וּבַת־ כֹּהֵ֔ן כִּ֥י תִהְיֶ֖ה לְאִ֣ישׁ זָ֑ר הִ֕וא בִּתְרוּמַ֥ת הַקֳּדָשִׁ֖ים לֹ֥א תֹאכֵֽל׃ (לז) וּבַת־ כֹּהֵן֩ כִּ֨י תִהְיֶ֜ה אַלְמָנָ֣ה וּגְרוּשָׁ֗ה וְזֶרַע֮ אֵ֣ין לָהּ֒ וְשָׁבָ֞ה אֶל־ בֵּ֤ית אָבִ֙יהָ֙ כִּנְעוּרֶ֔יהָ מִלֶּ֥חֶם אָבִ֖יהָ תֹּאכֵ֑ל וְכָל־ זָ֖ר לֹא־ יֹ֥אכַל בּֽוֹ׃ (לח) וְאִ֕ישׁ כִּֽי־ יֹאכַ֥ל קֹ֖דֶשׁ בִּשְׁגָגָ֑ה וְיָסַ֤ף חֲמִֽשִׁיתוֹ֙ עָלָ֔יו וְנָתַ֥ן לַכֹּהֵ֖ן אֶת־ הַקֹּֽדֶשׁ׃ (לט) וְלֹ֣א יְחַלְּל֔וּ אֶת־ קָדְשֵׁ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל אֵ֥ת אֲשֶׁר־ יָרִ֖ימוּ לַיהוָֽה׃ (מ) וְהִשִּׂ֤יאוּ אוֹתָם֙ עֲוֺ֣ן אַשְׁמָ֔ה בְּאָכְלָ֖ם אֶת־ קָדְשֵׁיהֶ֑ם כִּ֛י אֲנִ֥י יְהוָ֖ה מְקַדְּשָֽׁם׃

ויקרא סדר יח

(א) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (ב) דַּבֵּ֨ר אֶֽל־ אַהֲרֹ֜ן וְאֶל־ בָּנָ֗יו וְאֶל֙ כָּל־ בְּנֵ֣י יִשְׂרָאֵ֔ל וְאָמַרְתָּ֖ אֲלֵהֶ֑ם אִ֣ישׁ אִישׁ֩ מִבֵּ֨ית יִשְׂרָאֵ֜ל וּמִן־ הַגֵּ֣ר בְּיִשְׂרָאֵ֗ל אֲשֶׁ֨ר יַקְרִ֤יב קָרְבָּנוֹ֙ לְכָל־ נִדְרֵיהֶם֙ וּלְכָל־ נִדְבוֹתָ֔ם אֲשֶׁר־ יַקְרִ֥יבוּ לַיהוָ֖ה לְעֹלָֽה׃ (ג) לִֽרְצֹנְכֶ֑ם תָּמִ֣ים זָכָ֔ר בַּבָּקָ֕ר בַּכְּשָׂבִ֖ים וּבָֽעִזִּֽים׃ (ד) כֹּ֛ל אֲשֶׁר־ בּ֥וֹ מ֖וּם לֹ֣א תַקְרִ֑יבוּ כִּי־ לֹ֥א לְרָצ֖וֹן יִהְיֶ֥ה לָכֶֽם׃ (ה) וְאִ֗ישׁ כִּֽי־ יַקְרִ֤יב זֶֽבַח־ שְׁלָמִים֙ לַיהוָ֔ה לְפַלֵּא־ נֶ֙דֶר֙ א֣וֹ לִנְדָבָ֔ה בַּבָּקָ֖ר א֣וֹ בַצֹּ֑אן תָּמִ֤ים יִֽהְיֶה֙ לְרָצ֔וֹן כָּל־ מ֖וּם לֹ֥א יִהְיֶה־ בּֽוֹ׃ (ו) עַוֶּרֶת֩ א֨וֹ שָׁב֜וּר אוֹ־ חָר֣וּץ אֽוֹ־ יַבֶּ֗לֶת א֤וֹ גָרָב֙ א֣וֹ יַלֶּ֔פֶת לֹא־ תַקְרִ֥יבוּ אֵ֖לֶּה לַיהוָ֑ה וְאִשֶּׁ֗ה לֹא־ תִתְּנ֥וּ מֵהֶ֛ם עַל־ הַמִּזְבֵּ֖חַ לַיהוָֽה׃ (ז) וְשׁ֥וֹר וָשֶׂ֖ה שָׂר֣וּעַ וְקָל֑וּט נְדָבָה֙ תַּעֲשֶׂ֣ה אֹת֔וֹ וּלְנֵ֖דֶר לֹ֥א יֵרָצֶֽה׃ (ח) וּמָע֤וּךְ וְכָתוּת֙ וְנָת֣וּק וְכָר֔וּת לֹ֥א תַקְרִ֖יבוּ לַֽיהוָ֑ה וּֽבְאַרְצְכֶ֖ם לֹ֥א תַעֲשֽׂוּ׃ (ט) וּמִיַּ֣ד בֶּן־ נֵכָ֗ר לֹ֥א תַקְרִ֛יבוּ אֶת־ לֶ֥חֶם אֱלֹהֵיכֶ֖ם מִכָּל־ אֵ֑לֶּה כִּ֣י מָשְׁחָתָ֤ם בָּהֶם֙ מ֣וּם בָּ֔ם לֹ֥א יֵרָצ֖וּ לָכֶֽם׃ (י) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (יא) שׁ֣וֹר אוֹ־ כֶ֤שֶׂב אוֹ־ עֵז֙ כִּ֣י יִוָּלֵ֔ד וְהָיָ֛ה שִׁבְעַ֥ת יָמִ֖ים תַּ֣חַת אִמּ֑וֹ וּמִיּ֤וֹם הַשְּׁמִינִי֙ וָהָ֔לְאָה יֵרָצֶ֕ה לְקָרְבַּ֥ן אִשֶּׁ֖ה לַיהוָֽה׃ (יב) וְשׁ֖וֹר אוֹ־ שֶׂ֑ה אֹת֣וֹ וְאֶת־ בְּנ֔וֹ לֹ֥א תִשְׁחֲט֖וּ בְּי֥וֹם אֶחָֽד׃ (יג) וְכִֽי־ תִזְבְּח֥וּ זֶֽבַח־ תּוֹדָ֖ה לַיהוָ֑ה לִֽרְצֹנְכֶ֖ם תִּזְבָּֽחוּ׃ (יד) בַּיּ֤וֹם הַהוּא֙ יֵאָכֵ֔ל לֹֽא־ תוֹתִ֥ירוּ מִמֶּ֖נּוּ עַד־ בֹּ֑קֶר אֲנִ֖י יְהוָֽה׃ (טו) וּשְׁמַרְתֶּם֙ מִצְוֺתַ֔י וַעֲשִׂיתֶ֖ם אֹתָ֑ם אֲנִ֖י יְהוָֽה׃ (טז) וְלֹ֤א תְחַלְּלוּ֙ אֶת־ שֵׁ֣ם קָדְשִׁ֔י וְנִ֨קְדַּשְׁתִּ֔י בְּת֖וֹךְ בְּנֵ֣י יִשְׂרָאֵ֑ל אֲנִ֥י יְהוָ֖ה מְקַדִּשְׁכֶֽם׃ (יז) הַמּוֹצִ֤יא אֶתְכֶם֙ מֵאֶ֣רֶץ מִצְרַ֔יִם לִהְי֥וֹת לָכֶ֖ם לֵאלֹהִ֑ים אֲנִ֖י יְהוָֽה׃ סוף פרק כב (יח) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (יט) דַּבֵּ֞ר אֶל־ בְּנֵ֤י יִשְׂרָאֵל֙ וְאָמַרְתָּ֣ אֲלֵהֶ֔ם מוֹעֲדֵ֣י יְהוָ֔ה אֲשֶׁר־ תִּקְרְא֥וּ אֹתָ֖ם מִקְרָאֵ֣י קֹ֑דֶשׁ אֵ֥לֶּה הֵ֖ם מוֹעֲדָֽי׃ (כ) שֵׁ֣שֶׁת יָמִים֮ תֵּעָשֶׂ֣ה מְלָאכָה֒ וּבַיּ֣וֹם הַשְּׁבִיעִ֗י שַׁבַּ֤ת שַׁבָּתוֹן֙ מִקְרָא־ קֹ֔דֶשׁ כָּל־ מְלָאכָ֖ה לֹ֣א תַעֲשׂ֑וּ שַׁבָּ֥ת הִוא֙ לַֽיהוָ֔ה בְּכֹ֖ל מֽוֹשְׁבֹתֵיכֶֽם׃ (כא) אֵ֚לֶּה מוֹעֲדֵ֣י יְהוָ֔ה מִקְרָאֵ֖י קֹ֑דֶשׁ אֲשֶׁר־ תִּקְרְא֥וּ אֹתָ֖ם בְּמוֹעֲדָֽם׃ (כב) בַּחֹ֣דֶשׁ הָרִאשׁ֗וֹן בְּאַרְבָּעָ֥ה עָשָׂ֛ר לַחֹ֖דֶשׁ בֵּ֣ין הָעַרְבָּ֑יִם פֶּ֖סַח לַיהוָֽה׃ (כג) וּבַחֲמִשָּׁ֨ה עָשָׂ֥ר יוֹם֙ לַחֹ֣דֶשׁ הַזֶּ֔ה חַ֥ג הַמַּצּ֖וֹת לַיהוָ֑ה שִׁבְעַ֥ת יָמִ֖ים מַצּ֥וֹת תֹּאכֵֽלוּ׃ (כד) בַּיּוֹם֙ הָֽרִאשׁ֔וֹן מִקְרָא־ קֹ֖דֶשׁ יִהְיֶ֣ה לָכֶ֑ם כָּל־ מְלֶ֥אכֶת עֲבֹדָ֖ה לֹ֥א תַעֲשֽׂוּ׃ (כה) וְהִקְרַבְתֶּ֥ם אִשֶּׁ֛ה לַיהוָ֖ה שִׁבְעַ֣ת יָמִ֑ים בַּיּ֤וֹם הַשְּׁבִיעִי֙ מִקְרָא־ קֹ֔דֶשׁ כָּל־ מְלֶ֥אכֶת עֲבֹדָ֖ה לֹ֥א תַעֲשֽׂוּ׃ (כו) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (כז) דַּבֵּ֞ר אֶל־ בְּנֵ֤י יִשְׂרָאֵל֙ וְאָמַרְתָּ֣ אֲלֵהֶ֔ם כִּֽי־ תָבֹ֣אוּ אֶל־ הָאָ֗רֶץ אֲשֶׁ֤ר אֲנִי֙ נֹתֵ֣ן לָכֶ֔ם וּקְצַרְתֶּ֖ם אֶת־ קְצִירָ֑הּ וַהֲבֵאתֶ֥ם אֶת־ עֹ֛מֶר רֵאשִׁ֥ית קְצִירְכֶ֖ם אֶל־ הַכֹּהֵֽן׃ (כח) וְהֵנִ֧יף אֶת־ הָעֹ֛מֶר לִפְנֵ֥י יְהוָ֖ה לִֽרְצֹנְכֶ֑ם מִֽמָּחֳרַת֙ הַשַּׁבָּ֔ת יְנִיפֶ֖נּוּ הַכֹּהֵֽן׃ (כט) וַעֲשִׂיתֶ֕ם בְּי֥וֹם הֲנִֽיפְכֶ֖ם אֶת־ הָעֹ֑מֶר כֶּ֣בֶשׂ תָּמִ֧ים בֶּן־ שְׁנָת֛וֹ לְעֹלָ֖ה לַיהוָֽה׃ (ל) וּמִנְחָתוֹ֩ שְׁנֵ֨י עֶשְׂרֹנִ֜ים סֹ֣לֶת בְּלוּלָ֥ה בַשֶּׁ֛מֶן אִשֶּׁ֥ה לַיהוָ֖ה רֵ֣יחַ נִיחֹ֑חַ וְנִסְכֹּ֥ה יַ֖יִן רְבִיעִ֥ת הַהִֽין׃ (לא) וְלֶחֶם֩ וְקָלִ֨י וְכַרְמֶ֜ל לֹ֣א תֹֽאכְל֗וּ עַד־ עֶ֙צֶם֙ הַיּ֣וֹם הַזֶּ֔ה עַ֚ד הֲבִ֣יאֲכֶ֔ם אֶת־ קָרְבַּ֖ן אֱלֹהֵיכֶ֑ם חֻקַּ֤ת עוֹלָם֙ לְדֹרֹ֣תֵיכֶ֔ם בְּכֹ֖ל מֹשְׁבֹֽתֵיכֶֽם׃
