ישעיהו סדר יא

פתיחה באתר · ספר ישעיהו כולו ·

(א) וְהָיָה֙ בַּיּ֣וֹם הַה֔וּא יַחְבֹּ֧ט יְהוָ֛ה מִשִּׁבֹּ֥לֶת הַנָּהָ֖ר עַד־ נַ֣חַל מִצְרָ֑יִם וְאַתֶּ֧ם תְּלֻקְּט֛וּ לְאַחַ֥ד אֶחָ֖ד בְּנֵ֥י יִשְׂרָאֵֽל׃ (ב) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא יִתָּקַע֮ בְּשׁוֹפָ֣ר גָּדוֹל֒ וּבָ֗אוּ הָאֹֽבְדִים֙ בְּאֶ֣רֶץ אַשּׁ֔וּר וְהַנִּדָּחִ֖ים בְּאֶ֣רֶץ מִצְרָ֑יִם וְהִשְׁתַּחֲו֧וּ לַיהוָ֛ה בְּהַ֥ר הַקֹּ֖דֶשׁ בִּירוּשָׁלִָֽם׃ (ג) ה֗וֹי עֲטֶ֤רֶת גֵּאוּת֙ שִׁכֹּרֵ֣י אֶפְרַ֔יִם וְצִ֥יץ נֹבֵ֖ל צְבִ֣י תִפְאַרְתּ֑וֹ אֲשֶׁ֛ר עַל־ רֹ֥אשׁ גֵּֽיא־ שְׁמָנִ֖ים הֲל֥וּמֵי יָֽיִן׃ (ד) הִנֵּ֨ה חָזָ֤ק וְאַמִּץ֙ לַֽאדֹנָ֔י כְּזֶ֥רֶם בָּרָ֖ד שַׂ֣עַר קָ֑טֶב כְּ֠זֶרֶם מַ֣יִם כַּבִּירִ֥ים שֹׁטְפִ֛ים הִנִּ֥יחַ לָאָ֖רֶץ בְּיָֽד׃ (ה) בְּרַגְלַ֖יִם תֵּֽרָמַ֑סְנָה עֲטֶ֥רֶת גֵּא֖וּת שִׁכּוֹרֵ֥י אֶפְרָֽיִם׃ (ו) וְֽהָ֨יְתָ֜ה צִיצַ֤ת נֹבֵל֙ צְבִ֣י תִפְאַרְתּ֔וֹ אֲשֶׁ֥ר עַל־ רֹ֖אשׁ גֵּ֣יא שְׁמָנִ֑ים כְּבִכּוּרָהּ֙ בְּטֶ֣רֶם קַ֔יִץ אֲשֶׁ֨ר יִרְאֶ֤ה הָֽרֹאֶה֙ אוֹתָ֔הּ בְּעוֹדָ֥הּ בְּכַפּ֖וֹ יִבְלָעֶֽנָּה׃ (ז) בַּיּ֣וֹם הַה֗וּא יִֽהְיֶה֙ יְהוָ֣ה צְבָא֔וֹת לַעֲטֶ֣רֶת צְבִ֔י וְלִצְפִירַ֖ת תִּפְאָרָ֑ה לִשְׁאָ֖ר עַמּֽוֹ׃ (ח) וּלְר֖וּחַ מִשְׁפָּ֑ט לַיּוֹשֵׁב֙ עַל־ הַמִּשְׁפָּ֔ט וְלִ֨גְבוּרָ֔ה מְשִׁיבֵ֥י מִלְחָמָ֖ה שָֽׁעְרָה׃ (ט) וְגַם־ אֵ֙לֶּה֙ בַּיַּ֣יִן שָׁג֔וּ וּבַשֵּׁכָ֖ר תָּע֑וּ כֹּהֵ֣ן וְנָבִיא֩ שָׁג֨וּ בַשֵּׁכָ֜ר נִבְלְע֣וּ מִן־ הַיַּ֗יִן תָּעוּ֙ מִן־ הַשֵּׁכָ֔ר שָׁגוּ֙ בָּֽרֹאֶ֔ה פָּק֖וּ פְּלִילִיָּֽה׃ (י) כִּ֚י כָּל־ שֻׁלְחָנ֔וֹת מָלְא֖וּ קִ֣יא צֹאָ֑ה בְּלִ֖י מָקֽוֹם׃ (יא) אֶת־ מִי֙ יוֹרֶ֣ה דֵעָ֔ה וְאֶת־ מִ֖י יָבִ֣ין שְׁמוּעָ֑ה גְּמוּלֵי֙ מֵֽחָלָ֔ב עַתִּיקֵ֖י מִשָּׁדָֽיִם׃ (יב) כִּ֣י צַ֤ו לָצָו֙ צַ֣ו לָצָ֔ו קַ֥ו לָקָ֖ו קַ֣ו לָקָ֑ו זְעֵ֥יר שָׁ֖ם זְעֵ֥יר שָֽׁם׃ (יג) כִּ֚י בְּלַעֲגֵ֣י שָׂפָ֔ה וּבְלָשׁ֖וֹן אַחֶ֑רֶת יְדַבֵּ֖ר אֶל־ הָעָ֥ם הַזֶּֽה׃ (יד) אֲשֶׁ֣ר ׀ אָמַ֣ר אֲלֵיהֶ֗ם זֹ֤את הַמְּנוּחָה֙ הָנִ֣יחוּ לֶֽעָיֵ֔ף וְזֹ֖את הַמַּרְגֵּעָ֑ה וְלֹ֥א אָב֖וּא שְׁמֽוֹעַ׃ (טו) וְהָיָ֨ה לָהֶ֜ם דְּבַר־ יְהוָ֗ה צַ֣ו לָצָ֞ו צַ֤ו לָצָו֙ קַ֤ו לָקָו֙ קַ֣ו לָקָ֔ו זְעֵ֥יר שָׁ֖ם זְעֵ֣יר שָׁ֑ם לְמַ֨עַן יֵלְכ֜וּ וְכָשְׁל֤וּ אָחוֹר֙ וְנִשְׁבָּ֔רוּ וְנוֹקְשׁ֖וּ וְנִלְכָּֽדוּ׃ (טז) לָכֵ֛ן שִׁמְע֥וּ דְבַר־ יְהוָ֖ה אַנְשֵׁ֣י לָצ֑וֹן מֹֽשְׁלֵי֙ הָעָ֣ם הַזֶּ֔ה אֲשֶׁ֖ר בִּירוּשָׁלִָֽם׃ (יז) כִּ֣י אֲמַרְתֶּ֗ם כָּרַ֤תְנֽוּ בְרִית֙ אֶת־ מָ֔וֶת וְעִם־ שְׁא֖וֹל עָשִׂ֣ינוּ חֹזֶ֑ה (שיט) שׁ֣וֹט שׁוֹטֵ֤ף כִּֽי־ (עבר) יַֽעֲבֹר֙ לֹ֣א יְבוֹאֵ֔נוּ כִּ֣י שַׂ֧מְנוּ כָזָ֛ב מַחְסֵ֖נוּ וּבַשֶּׁ֥קֶר נִסְתָּֽרְנוּ׃ (יח) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֛י יִסַּ֥ד בְּצִיּ֖וֹן אָ֑בֶן אֶ֣בֶן בֹּ֜חַן פִּנַּ֤ת יִקְרַת֙ מוּסָ֣ד מוּסָּ֔ד הַֽמַּאֲמִ֖ין לֹ֥א יָחִֽישׁ׃ (יט) וְשַׂמְתִּ֤י מִשְׁפָּט֙ לְקָ֔ו וּצְדָקָ֖ה לְמִשְׁקָ֑לֶת וְיָעָ֤ה בָרָד֙ מַחְסֵ֣ה כָזָ֔ב וְסֵ֥תֶר מַ֖יִם יִשְׁטֹֽפוּ׃ (כ) וְכֻפַּ֤ר בְּרִֽיתְכֶם֙ אֶת־ מָ֔וֶת וְחָזוּתְכֶ֥ם אֶת־ שְׁא֖וֹל לֹ֣א תָק֑וּם שׁ֤וֹט שׁוֹטֵף֙ כִּ֣י יַֽעֲבֹ֔ר וִהְיִ֥יתֶם ל֖וֹ לְמִרְמָֽס׃ (כא) מִדֵּ֤י עָבְרוֹ֙ יִקַּ֣ח אֶתְכֶ֔ם כִּֽי־ בַבֹּ֧קֶר בַּבֹּ֛קֶר יַעֲבֹ֖ר בַּיּ֣וֹם וּבַלָּ֑יְלָה וְהָיָ֥ה רַק־ זְוָעָ֖ה הָבִ֥ין שְׁמוּעָֽה׃ (כב) כִּֽי־ קָצַ֥ר הַמַּצָּ֖ע מֵֽהִשְׂתָּרֵ֑עַ וְהַמַּסֵּכָ֥ה צָ֖רָה כְּהִתְכַּנֵּֽס׃ (כג) כִּ֤י כְהַר־ פְּרָצִים֙ יָק֣וּם יְהוָ֔ה כְּעֵ֖מֶק בְּגִבְע֣וֹן יִרְגָּ֑ז לַעֲשׂ֤וֹת מַעֲשֵׂ֙הוּ֙ זָ֣ר מַעֲשֵׂ֔הוּ וְלַֽעֲבֹד֙ עֲבֹ֣דָת֔וֹ נָכְרִיָּ֖ה עֲבֹדָתֽוֹ׃ (כד) וְעַתָּה֙ אַל־ תִּתְלוֹצָ֔צוּ פֶּֽן־ יֶחְזְק֖וּ מֽוֹסְרֵיכֶ֑ם כִּֽי־ כָלָ֨ה וְנֶחֱרָצָ֜ה שָׁמַ֗עְתִּי מֵאֵ֨ת אֲדֹנָ֧י יְהוִ֛ה צְבָא֖וֹת עַל־ כָּל־ הָאָֽרֶץ׃ (כה) הַאֲזִ֥ינוּ וְשִׁמְע֖וּ קוֹלִ֑י הַקְשִׁ֥יבוּ וְשִׁמְע֖וּ אִמְרָתִֽי׃ (כו) הֲכֹ֣ל הַיּ֔וֹם יַחֲרֹ֥שׁ הַחֹרֵ֖שׁ לִזְרֹ֑עַ יְפַתַּ֥ח וִֽישַׂדֵּ֖ד אַדְמָתֽוֹ׃ (כז) הֲלוֹא֙ אִם־ שִׁוָּ֣ה פָנֶ֔יהָ וְהֵפִ֥יץ קֶ֖צַח וְכַמֹּ֣ן יִזְרֹ֑ק וְשָׂ֨ם חִטָּ֤ה שׂוֹרָה֙ וּשְׂעֹרָ֣ה נִסְמָ֔ן וְכֻסֶּ֖מֶת גְּבֻלָתֽוֹ׃ (כח) וְיִסְּר֥וֹ לַמִּשְׁפָּ֖ט אֱלֹהָ֥יו יוֹרֶֽנּוּ׃ (כט) כִּ֣י לֹ֤א בֶֽחָרוּץ֙ י֣וּדַשׁ קֶ֔צַח וְאוֹפַ֣ן עֲגָלָ֔ה עַל־ כַּמֹּ֖ן יוּסָּ֑ב כִּ֧י בַמַּטֶּ֛ה יֵחָ֥בֶט קֶ֖צַח וְכַמֹּ֥ן בַּשָּֽׁבֶט׃ (ל) לֶ֣חֶם יוּדָ֔ק כִּ֛י לֹ֥א לָנֶ֖צַח אָד֣וֹשׁ יְדוּשֶׁ֑נּוּ וְ֠הָמַם גִּלְגַּ֧ל עֶגְלָת֛וֹ וּפָרָשָׁ֖יו לֹֽא־ יְדֻקֶּֽנּוּ׃ (לא) גַּם־ זֹ֕את מֵעִ֛ם יְהוָ֥ה צְבָא֖וֹת יָצָ֑אָה הִפְלִ֣יא עֵצָ֔ה הִגְדִּ֖יל תּוּשִׁיָּֽה׃ (לב) ה֚וֹי אֲרִיאֵ֣ל אֲרִיאֵ֔ל קִרְיַ֖ת חָנָ֣ה דָוִ֑ד סְפ֥וּ שָׁנָ֛ה עַל־ שָׁנָ֖ה חַגִּ֥ים יִנְקֹֽפוּ׃ (לג) וַהֲצִיק֖וֹתִי לַֽאֲרִיאֵ֑ל וְהָיְתָ֤ה תַֽאֲנִיָּה֙ וַֽאֲנִיָּ֔ה וְהָ֥יְתָה לִּ֖י כַּאֲרִיאֵֽל׃ (לד) וְחָנִ֥יתִי כַדּ֖וּר עָלָ֑יִךְ וְצַרְתִּ֤י עָלַ֙יִךְ֙ מֻצָּ֔ב וַהֲקִֽימֹתִ֥י עָלַ֖יִךְ מְצֻרֹֽת׃ (לה) וְשָׁפַלְתְּ֙ מֵאֶ֣רֶץ תְּדַבֵּ֔רִי וּמֵֽעָפָ֖ר תִּשַּׁ֣ח אִמְרָתֵ֑ךְ וְֽ֠הָיָה כְּא֤וֹב מֵאֶ֙רֶץ֙ קוֹלֵ֔ךְ וּמֵעָפָ֖ר אִמְרָתֵ֥ךְ תְּצַפְצֵֽף׃ (לו) וְהָיָ֛ה כְּאָבָ֥ק דַּ֖ק הֲמ֣וֹן זָרָ֑יִךְ וּכְמֹ֤ץ עֹבֵר֙ הֲמ֣וֹן עָֽרִיצִ֔ים וְהָיָ֖ה לְפֶ֥תַע פִּתְאֹֽם׃ (לז) מֵעִ֨ם יְהוָ֤ה צְבָאוֹת֙ תִּפָּקֵ֔ד בְּרַ֥עַם וּבְרַ֖עַשׁ וְק֣וֹל גָּד֑וֹל סוּפָה֙ וּסְעָרָ֔ה וְלַ֖הַב אֵ֥שׁ אוֹכֵלָֽה׃ (לח) וְהָיָ֗ה כַּֽחֲלוֹם֙ חֲז֣וֹן לַ֔יְלָה הֲמוֹן֙ כָּל־ הַגּוֹיִ֔ם הַצֹּבְאִ֖ים עַל־ אֲרִיאֵ֑ל וְכָל־ צֹבֶ֙יהָ֙ וּמְצֹ֣דָתָ֔הּ וְהַמְּצִיקִ֖ים לָֽהּ׃ (לט) וְהָיָ֡ה כַּאֲשֶׁר֩ יַחֲלֹ֨ם הָרָעֵ֜ב וְהִנֵּ֣ה אוֹכֵ֗ל וְהֵקִיץ֮ וְרֵיקָ֣ה נַפְשׁוֹ֒ וְכַאֲשֶׁ֨ר יַחֲלֹ֤ם הַצָּמֵא֙ וְהִנֵּ֣ה שֹׁתֶ֔ה וְהֵקִיץ֙ וְהִנֵּ֣ה עָיֵ֔ף וְנַפְשׁ֖וֹ שׁוֹקֵקָ֑ה כֵּ֣ן יִֽהְיֶ֗ה הֲמוֹן֙ כָּל־ הַגּוֹיִ֔ם הַצֹּבְאִ֖ים עַל־ הַ֥ר צִיּֽוֹן׃ (מ) הִתְמַהְמְה֣וּ וּתְמָ֔הוּ הִשְׁתַּֽעַשְׁע֖וּ וָשֹׁ֑עוּ שָֽׁכְר֣וּ וְלֹא־ יַ֔יִן נָע֖וּ וְלֹ֥א שֵׁכָֽר׃ (מא) כִּֽי־ נָסַ֨ךְ עֲלֵיכֶ֤ם יְהוָה֙ ר֣וּחַ תַּרְדֵּמָ֔ה וַיְעַצֵּ֖ם אֶת־ עֵֽינֵיכֶ֑ם אֶת־ הַנְּבִיאִ֛ים וְאֶת־ רָאשֵׁיכֶ֥ם הַחֹזִ֖ים כִּסָּֽה׃ (מב) וַתְּהִ֨י לָכֶ֜ם חָז֣וּת הַכֹּ֗ל כְּדִבְרֵי֮ הַסֵּ֣פֶר הֶֽחָתוּם֒ אֲשֶֽׁר־ יִתְּנ֣וּ אֹת֗וֹ אֶל־ יוֹדֵ֥עַ (הספר) סֵ֛פֶר לֵאמֹ֖ר קְרָ֣א נָא־ זֶ֑ה וְאָמַר֙ לֹ֣א אוּכַ֔ל כִּ֥י חָת֖וּם הֽוּא׃ (מג) וְנִתַּ֣ן הַסֵּ֗פֶר עַל֩ אֲשֶׁ֨ר לֹֽא־ יָדַ֥ע סֵ֛פֶר לֵאמֹ֖ר קְרָ֣א נָא־ זֶ֑ה וְאָמַ֕ר לֹ֥א יָדַ֖עְתִּי סֵֽפֶר׃ (מד) וַיֹּ֣אמֶר אֲדֹנָ֗י יַ֚עַן כִּ֤י נִגַּשׁ֙ הָעָ֣ם הַזֶּ֔ה בְּפִ֤יו וּבִשְׂפָתָיו֙ כִּבְּד֔וּנִי וְלִבּ֖וֹ רִחַ֣ק מִמֶּ֑נִּי וַתְּהִ֤י יִרְאָתָם֙ אֹתִ֔י מִצְוַ֥ת אֲנָשִׁ֖ים מְלֻמָּדָֽה׃ (מה) לָכֵ֗ן הִנְנִ֥י יוֹסִ֛ף לְהַפְלִ֥יא אֶת־ הָֽעָם־ הַזֶּ֖ה הַפְלֵ֣א וָפֶ֑לֶא וְאָֽבְדָה֙ חָכְמַ֣ת חֲכָמָ֔יו וּבִינַ֥ת נְבֹנָ֖יו תִּסְתַּתָּֽר׃ (מו) ה֛וֹי הַמַּעֲמִיקִ֥ים מֵֽיהוָ֖ה לַסְתִּ֣ר עֵצָ֑ה וְהָיָ֤ה בְמַחְשָׁךְ֙ מַֽעֲשֵׂיהֶ֔ם וַיֹּ֣אמְר֔וּ מִ֥י רֹאֵ֖נוּ וּמִ֥י יוֹדְעֵֽנוּ׃ (מז) הַ֨פְכְּכֶ֔ם אִם־ כְּחֹ֥מֶר הַיֹּצֵ֖ר יֵֽחָשֵׁ֑ב כִּֽי־ יֹאמַ֨ר מַעֲשֶׂ֤ה לְעֹשֵׂ֙הוּ֙ לֹ֣א עָשָׂ֔נִי וְיֵ֛צֶר אָמַ֥ר לְיוֹצְר֖וֹ לֹ֥א הֵבִֽין׃ (מח) הֲלוֹא־ עוֹד֙ מְעַ֣ט מִזְעָ֔ר וְשָׁ֥ב לְבָנ֖וֹן לַכַּרְמֶ֑ל וְהַכַּרְמֶ֖ל לַיַּ֥עַר יֵחָשֵֽׁב׃ (מט) וְשָׁמְע֧וּ בַיּוֹם־ הַה֛וּא הַחֵרְשִׁ֖ים דִּבְרֵי־ סֵ֑פֶר וּמֵאֹ֣פֶל וּמֵחֹ֔שֶׁךְ עֵינֵ֥י עִוְרִ֖ים תִּרְאֶֽינָה׃ (נ) וְיָסְפ֧וּ עֲנָוִ֛ים בַּֽיהוָ֖ה שִׂמְחָ֑ה וְאֶבְיוֹנֵ֣י אָדָ֔ם בִּקְד֥וֹשׁ יִשְׂרָאֵ֖ל יָגִֽילוּ׃ (נא) כִּֽי־ אָפֵ֥ס עָרִ֖יץ וְכָ֣לָה לֵ֑ץ וְנִכְרְת֖וּ כָּל־ שֹׁ֥קְדֵי אָֽוֶן׃ (נב) מַחֲטִיאֵ֤י אָדָם֙ בְּדָבָ֔ר וְלַמּוֹכִ֥יחַ בַּשַּׁ֖עַר יְקֹשׁ֑וּן וַיַּטּ֥וּ בַתֹּ֖הוּ צַדִּֽיק׃ (נג) לָכֵ֗ן כֹּֽה־ אָמַ֤ר יְהוָה֙ אֶל־ בֵּ֣ית יַֽעֲקֹ֔ב אֲשֶׁ֥ר פָּדָ֖ה אֶת־ אַבְרָהָ֑ם לֹֽא־ עַתָּ֤ה יֵבוֹשׁ֙ יַֽעֲקֹ֔ב וְלֹ֥א עַתָּ֖ה פָּנָ֥יו יֶחֱוָֽרוּ׃
