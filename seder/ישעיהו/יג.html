ישעיהו סדר יג

פתיחה באתר · ספר ישעיהו כולו ·

(א) וְיָשַׁ֥ב עַמִּ֖י בִּנְוֵ֣ה שָׁל֑וֹם וּֽבְמִשְׁכְּנוֹת֙ מִבְטַחִ֔ים וּבִמְנוּחֹ֖ת שַׁאֲנַנּֽוֹת׃ (ב) וּבָרַ֖ד בְּרֶ֣דֶת הַיָּ֑עַר וּבַשִּׁפְלָ֖ה תִּשְׁפַּ֥ל הָעִֽיר׃ (ג) אַשְׁרֵיכֶ֕ם זֹרְעֵ֖י עַל־ כָּל־ מָ֑יִם מְשַׁלְּחֵ֥י רֶֽגֶל־ הַשּׁ֖וֹר וְהַחֲמֽוֹר׃ (ד) ה֣וֹי שׁוֹדֵ֗ד וְאַתָּה֙ לֹ֣א שָׁד֔וּד וּבוֹגֵ֖ד וְלֹא־ בָ֣גְדוּ ב֑וֹ כַּהֲתִֽמְךָ֤ שׁוֹדֵד֙ תּוּשַּׁ֔ד כַּנְּלֹתְךָ֥ לִבְגֹּ֖ד יִבְגְּדוּ־ בָֽךְ׃ (ה) יְהוָ֥ה חָנֵּ֖נוּ לְךָ֣ קִוִּ֑ינוּ הֱיֵ֤ה זְרֹעָם֙ לַבְּקָרִ֔ים אַף־ יְשׁוּעָתֵ֖נוּ בְּעֵ֥ת צָרָֽה׃ (ו) מִקּ֣וֹל הָמ֔וֹן נָדְד֖וּ עַמִּ֑ים מֵר֣וֹמְמֻתֶ֔ךָ נָפְצ֖וּ גּוֹיִֽם׃ (ז) וְאֻסַּ֣ף שְׁלַלְכֶ֔ם אֹ֖סֶף הֶֽחָסִ֑יל כְּמַשַּׁ֥ק גֵּבִ֖ים שׁוֹקֵ֥ק בּֽוֹ׃ (ח) נִשְׂגָּ֣ב יְהוָ֔ה כִּ֥י שֹׁכֵ֖ן מָר֑וֹם מִלֵּ֣א צִיּ֔וֹן מִשְׁפָּ֖ט וּצְדָקָֽה׃ (ט) וְהָיָה֙ אֱמוּנַ֣ת עִתֶּ֔יךָ חֹ֥סֶן יְשׁוּעֹ֖ת חָכְמַ֣ת וָדָ֑עַת יִרְאַ֥ת יְהוָ֖ה הִ֥יא אוֹצָרֽוֹ׃ (י) הֵ֚ן אֶרְאֶלָּ֔ם צָעֲק֖וּ חֻ֑צָה מַלְאֲכֵ֣י שָׁל֔וֹם מַ֖ר יִבְכָּיֽוּן׃ (יא) נָשַׁ֣מּוּ מְסִלּ֔וֹת שָׁבַ֖ת עֹבֵ֣ר אֹ֑רַח הֵפֵ֤ר בְּרִית֙ מָאַ֣ס עָרִ֔ים לֹ֥א חָשַׁ֖ב אֱנֽוֹשׁ׃ (יב) אָבַ֤ל אֻמְלְלָה֙ אָ֔רֶץ הֶחְפִּ֥יר לְבָנ֖וֹן קָמַ֑ל הָיָ֤ה הַשָּׁרוֹן֙ כָּֽעֲרָבָ֔ה וְנֹעֵ֥ר בָּשָׁ֖ן וְכַרְמֶֽל׃ (יג) עַתָּ֥ה אָק֖וּם יֹאמַ֣ר יְהוָ֑ה עַתָּה֙ אֵֽרוֹמָ֔ם עַתָּ֖ה אֶנָּשֵֽׂא׃ (יד) תַּהֲר֥וּ חֲשַׁ֖שׁ תֵּ֣לְדוּ קַ֑שׁ רוּחֲכֶ֕ם אֵ֖שׁ תֹּאכַלְכֶֽם׃ (טו) וְהָי֥וּ עַמִּ֖ים מִשְׂרְפ֣וֹת שִׂ֑יד קוֹצִ֥ים כְּסוּחִ֖ים בָּאֵ֥שׁ יִצַּֽתּוּ׃ (טז) שִׁמְע֥וּ רְחוֹקִ֖ים אֲשֶׁ֣ר עָשִׂ֑יתִי וּדְע֥וּ קְרוֹבִ֖ים גְּבֻרָתִֽי׃ (יז) פָּחֲד֤וּ בְצִיּוֹן֙ חַטָּאִ֔ים אָחֲזָ֥ה רְעָדָ֖ה חֲנֵפִ֑ים מִ֣י ׀ יָג֣וּר לָ֗נוּ אֵ֚שׁ אוֹכֵלָ֔ה מִי־ יָג֥וּר לָ֖נוּ מוֹקְדֵ֥י עוֹלָֽם׃ (יח) הֹלֵ֣ךְ צְדָק֔וֹת וְדֹבֵ֖ר מֵֽישָׁרִ֑ים מֹאֵ֞ס בְּבֶ֣צַע מַעֲשַׁקּ֗וֹת נֹעֵ֤ר כַּפָּיו֙ מִתְּמֹ֣ךְ בַּשֹּׁ֔חַד אֹטֵ֤ם אָזְנוֹ֙ מִשְּׁמֹ֣עַ דָּמִ֔ים וְעֹצֵ֥ם עֵינָ֖יו מֵרְא֥וֹת בְּרָֽע׃ (יט) ה֚וּא מְרוֹמִ֣ים יִשְׁכֹּ֔ן מְצָד֥וֹת סְלָעִ֖ים מִשְׂגַּבּ֑וֹ לַחְמ֣וֹ נִתָּ֔ן מֵימָ֖יו נֶאֱמָנִֽים׃ (כ) מֶ֥לֶךְ בְּיָפְי֖וֹ תֶּחֱזֶ֣ינָה עֵינֶ֑יךָ תִּרְאֶ֖ינָה אֶ֥רֶץ מַרְחַקִּֽים׃ (כא) לִבְּךָ֖ יֶהְגֶּ֣ה אֵימָ֑ה אַיֵּ֤ה סֹפֵר֙ אַיֵּ֣ה שֹׁקֵ֔ל אַיֵּ֖ה סֹפֵ֥ר אֶת־ הַמִּגְדָּלִֽים׃ (כב) אֶת־ עַ֥ם נוֹעָ֖ז לֹ֣א תִרְאֶ֑ה עַ֣ם עִמְקֵ֤י שָׂפָה֙ מִשְּׁמ֔וֹעַ נִלְעַ֥ג לָשׁ֖וֹן אֵ֥ין בִּינָֽה׃ (כג) חֲזֵ֣ה צִיּ֔וֹן קִרְיַ֖ת מֽוֹעֲדֵ֑נוּ עֵינֶיךָ֩ תִרְאֶ֨ינָה יְרוּשָׁלִַ֜ם נָוֶ֣ה שַׁאֲנָ֗ן אֹ֤הֶל בַּל־ יִצְעָן֙ בַּל־ יִסַּ֤ע יְתֵֽדֹתָיו֙ לָנֶ֔צַח וְכָל־ חֲבָלָ֖יו בַּל־ יִנָּתֵֽקוּ׃ (כד) כִּ֣י אִם־ שָׁ֞ם אַדִּ֤יר יְהוָה֙ לָ֔נוּ מְקוֹם־ נְהָרִ֥ים יְאֹרִ֖ים רַחֲבֵ֣י יָדָ֑יִם בַּל־ תֵּ֤לֶךְ בּוֹ֙ אֳנִי־ שַׁ֔יִט וְצִ֥י אַדִּ֖יר לֹ֥א יַעַבְרֶֽנּוּ׃ (כה) כִּ֤י יְהוָה֙ שֹׁפְטֵ֔נוּ יְהוָ֖ה מְחֹקְקֵ֑נוּ יְהוָ֥ה מַלְכֵּ֖נוּ ה֥וּא יוֹשִׁיעֵֽנוּ׃ (כו) נִטְּשׁ֖וּ חֲבָלָ֑יִךְ בַּל־ יְחַזְּק֤וּ כֵן־ תָּרְנָם֙ בַּל־ פָּ֣רְשׂוּ נֵ֔ס אָ֣ז חֻלַּ֤ק עַֽד־ שָׁלָל֙ מַרְבֶּ֔ה פִּסְחִ֖ים בָּ֥זְזוּ בַֽז׃ (כז) וּבַל־ יֹאמַ֥ר שָׁכֵ֖ן חָלִ֑יתִי הָעָ֛ם הַיֹּשֵׁ֥ב בָּ֖הּ נְשֻׂ֥א עָוֺֽן׃ (כח) קִרְב֤וּ גוֹיִם֙ לִשְׁמֹ֔עַ וּלְאֻמִּ֖ים הַקְשִׁ֑יבוּ תִּשְׁמַ֤ע הָאָ֙רֶץ֙ וּמְלֹאָ֔הּ תֵּבֵ֖ל וְכָל־ צֶאֱצָאֶֽיהָ׃ (כט) כִּ֣י קֶ֤צֶף לַֽיהוָה֙ עַל־ כָּל־ הַגּוֹיִ֔ם וְחֵמָ֖ה עַל־ כָּל־ צְבָאָ֑ם הֶחֱרִימָ֖ם נְתָנָ֥ם לַטָּֽבַח׃ (ל) וְחַלְלֵיהֶ֣ם יֻשְׁלָ֔כוּ וּפִגְרֵיהֶ֖ם יַעֲלֶ֣ה בָאְשָׁ֑ם וְנָמַ֥סּוּ הָרִ֖ים מִדָּמָֽם׃ (לא) וְנָמַ֙קּוּ֙ כָּל־ צְבָ֣א הַשָּׁמַ֔יִם וְנָגֹ֥לּוּ כַסֵּ֖פֶר הַשָּׁמָ֑יִם וְכָל־ צְבָאָ֣ם יִבּ֔וֹל כִּנְבֹ֤ל עָלֶה֙ מִגֶּ֔פֶן וּכְנֹבֶ֖לֶת מִתְּאֵנָֽה׃ (לב) כִּֽי־ רִוְּתָ֥ה בַשָּׁמַ֖יִם חַרְבִּ֑י הִנֵּה֙ עַל־ אֱד֣וֹם תֵּרֵ֔ד וְעַל־ עַ֥ם חֶרְמִ֖י לְמִשְׁפָּֽט׃ (לג) חֶ֣רֶב לַיהוָ֞ה מָלְאָ֥ה דָם֙ הֻדַּ֣שְׁנָה מֵחֵ֔לֶב מִדַּ֤ם כָּרִים֙ וְעַתּוּדִ֔ים מֵחֵ֖לֶב כִּלְי֣וֹת אֵילִ֑ים כִּ֣י זֶ֤בַח לַֽיהוָה֙ בְּבָצְרָ֔ה וְטֶ֥בַח גָּד֖וֹל בְּאֶ֥רֶץ אֱדֽוֹם׃ (לד) וְיָרְד֤וּ רְאֵמִים֙ עִמָּ֔ם וּפָרִ֖ים עִם־ אַבִּירִ֑ים וְרִוְּתָ֤ה אַרְצָם֙ מִדָּ֔ם וַעֲפָרָ֖ם מֵחֵ֥לֶב יְדֻשָּֽׁן׃ (לה) כִּ֛י י֥וֹם נָקָ֖ם לַֽיהוָ֑ה שְׁנַ֥ת שִׁלּוּמִ֖ים לְרִ֥יב צִיּֽוֹן׃ (לו) וְנֶהֶפְכ֤וּ נְחָלֶ֙יהָ֙ לְזֶ֔פֶת וַעֲפָרָ֖הּ לְגָפְרִ֑ית וְהָיְתָ֣ה אַרְצָ֔הּ לְזֶ֖פֶת בֹּעֵרָֽה׃ (לז) לַ֤יְלָה וְיוֹמָם֙ לֹ֣א תִכְבֶּ֔ה לְעוֹלָ֖ם יַעֲלֶ֣ה עֲשָׁנָ֑הּ מִדּ֤וֹר לָדוֹר֙ תֶּחֱרָ֔ב לְנֵ֣צַח נְצָחִ֔ים אֵ֥ין עֹבֵ֖ר בָּֽהּ׃ (לח) וִירֵשׁ֙וּהָ֙ קָאַ֣ת וְקִפּ֔וֹד וְיַנְשׁ֥וֹף וְעֹרֵ֖ב יִשְׁכְּנוּ־ בָ֑הּ וְנָטָ֥ה עָלֶ֛יהָ קַֽו־ תֹ֖הוּ וְאַבְנֵי־ בֹֽהוּ׃ (לט) חֹרֶ֥יהָ וְאֵֽין־ שָׁ֖ם מְלוּכָ֣ה יִקְרָ֑אוּ וְכָל־ שָׂרֶ֖יהָ יִ֥הְיוּ אָֽפֶס׃ (מ) וְעָלְתָ֤ה אַרְמְנֹתֶ֙יהָ֙ סִירִ֔ים קִמּ֥וֹשׂ וָח֖וֹחַ בְּמִבְצָרֶ֑יהָ וְהָיְתָה֙ נְוֵ֣ה תַנִּ֔ים חָצִ֖יר לִבְנ֥וֹת יַעֲנָֽה׃ (מא) וּפָגְשׁ֤וּ צִיִּים֙ אֶת־ אִיִּ֔ים וְשָׂעִ֖יר עַל־ רֵעֵ֣הוּ יִקְרָ֑א אַךְ־ שָׁם֙ הִרְגִּ֣יעָה לִּילִ֔ית וּמָצְאָ֥ה לָ֖הּ מָנֽוֹחַ׃ (מב) שָׁ֣מָּה קִנְּנָ֤ה קִפּוֹז֙ וַתְּמַלֵּ֔ט וּבָקְעָ֖ה וְדָגְרָ֣ה בְצִלָּ֑הּ אַךְ־ שָׁ֛ם נִקְבְּצ֥וּ דַיּ֖וֹת אִשָּׁ֥ה רְעוּתָֽהּ׃ (מג) דִּרְשׁ֨וּ מֵֽעַל־ סֵ֤פֶר יְהוָה֙ וּֽקְרָ֔אוּ אַחַ֤ת מֵהֵ֙נָּה֙ לֹ֣א נֶעְדָּ֔רָה אִשָּׁ֥ה רְעוּתָ֖הּ לֹ֣א פָקָ֑דוּ כִּֽי־ פִי֙ ה֣וּא צִוָּ֔ה וְרוּח֖וֹ ה֥וּא קִבְּצָֽן׃ (מד) וְהֽוּא־ הִפִּ֤יל לָהֶן֙ גּוֹרָ֔ל וְיָד֛וֹ חִלְּקַ֥תָּה לָהֶ֖ם בַּקָּ֑ו עַד־ עוֹלָם֙ יִֽירָשׁ֔וּהָ לְד֥וֹר וָד֖וֹר יִשְׁכְּנוּ־ בָֽהּ׃ (מה) יְשֻׂשׂ֥וּם מִדְבָּ֖ר וְצִיָּ֑ה וְתָגֵ֧ל עֲרָבָ֛ה וְתִפְרַ֖ח כַּחֲבַצָּֽלֶת׃ (מו) פָּרֹ֨חַ תִּפְרַ֜ח וְתָגֵ֗ל אַ֚ף גִּילַ֣ת וְרַנֵּ֔ן כְּב֤וֹד הַלְּבָנוֹן֙ נִתַּן־ לָ֔הּ הֲדַ֥ר הַכַּרְמֶ֖ל וְהַשָּׁר֑וֹן הֵ֛מָּה יִרְא֥וּ כְבוֹד־ יְהוָ֖ה הֲדַ֥ר אֱלֹהֵֽינוּ׃ (מז) חַזְּק֖וּ יָדַ֣יִם רָפ֑וֹת וּבִרְכַּ֥יִם כֹּשְׁל֖וֹת אַמֵּֽצוּ׃ (מח) אִמְרוּ֙ לְנִמְהֲרֵי־ לֵ֔ב חִזְק֖וּ אַל־ תִּירָ֑אוּ הִנֵּ֤ה אֱלֹֽהֵיכֶם֙ נָקָ֣ם יָב֔וֹא גְּמ֣וּל אֱלֹהִ֔ים ה֥וּא יָב֖וֹא וְיֹשַׁעֲכֶֽם׃ (מט) אָ֥ז תִּפָּקַ֖חְנָה עֵינֵ֣י עִוְרִ֑ים וְאָזְנֵ֥י חֵרְשִׁ֖ים תִּפָּתַֽחְנָה׃ (נ) אָ֣ז יְדַלֵּ֤ג כָּֽאַיָּל֙ פִּסֵּ֔חַ וְתָרֹ֖ן לְשׁ֣וֹן אִלֵּ֑ם כִּֽי־ נִבְקְע֤וּ בַמִּדְבָּר֙ מַ֔יִם וּנְחָלִ֖ים בָּעֲרָבָֽה׃ (נא) וְהָיָ֤ה הַשָּׁרָב֙ לַאֲגַ֔ם וְצִמָּא֖וֹן לְמַבּ֣וּעֵי מָ֑יִם בִּנְוֵ֤ה תַנִּים֙ רִבְצָ֔הּ חָצִ֖יר לְקָנֶ֥ה וָגֹֽמֶא׃ (נב) וְהָיָה־ שָׁ֞ם מַסְל֣וּל וָדֶ֗רֶךְ וְדֶ֤רֶךְ הַקֹּ֙דֶשׁ֙ יִקָּ֣רֵא לָ֔הּ לֹֽא־ יַעַבְרֶ֥נּוּ טָמֵ֖א וְהוּא־ לָ֑מוֹ הֹלֵ֥ךְ דֶּ֛רֶךְ וֶאֱוִילִ֖ים לֹ֥א יִתְעֽוּ׃ (נג) לֹא־ יִהְיֶ֨ה שָׁ֜ם אַרְיֵ֗ה וּפְרִ֤יץ חַיּוֹת֙ בַּֽל־ יַעֲלֶ֔נָּה לֹ֥א תִמָּצֵ֖א שָׁ֑ם וְהָלְכ֖וּ גְּאוּלִֽים׃
