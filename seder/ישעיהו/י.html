ישעיהו סדר י

פתיחה באתר · ספר ישעיהו כולו ·

(א) וְחָֽפְרָה֙ הַלְּבָנָ֔ה וּבוֹשָׁ֖ה הַֽחַמָּ֑ה כִּֽי־ מָלַ֞ךְ יְהוָ֣ה צְבָא֗וֹת בְּהַ֤ר צִיּוֹן֙ וּבִיר֣וּשָׁלִַ֔ם וְנֶ֥גֶד זְקֵנָ֖יו כָּבֽוֹד׃ (ב) יְהוָ֤ה אֱלֹהַי֙ אַתָּ֔ה אֲרֽוֹמִמְךָ֙ אוֹדֶ֣ה שִׁמְךָ֔ כִּ֥י עָשִׂ֖יתָ פֶּ֑לֶא עֵצ֥וֹת מֵֽרָח֖וֹק אֱמ֥וּנָה אֹֽמֶן׃ (ג) כִּ֣י שַׂ֤מְתָּ מֵעִיר֙ לַגָּ֔ל קִרְיָ֥ה בְצוּרָ֖ה לְמַפֵּלָ֑ה אַרְמ֤וֹן זָרִים֙ מֵעִ֔יר לְעוֹלָ֖ם לֹ֥א יִבָּנֶֽה׃ (ד) עַל־ כֵּ֖ן יְכַבְּד֣וּךָ עַם־ עָ֑ז קִרְיַ֛ת גּוֹיִ֥ם עָרִיצִ֖ים יִירָאֽוּךָ׃ (ה) כִּֽי־ הָיִ֨יתָ מָע֥וֹז לַדָּ֛ל מָע֥וֹז לָאֶבְי֖וֹן בַּצַּר־ ל֑וֹ מַחְסֶ֤ה מִזֶּ֙רֶם֙ צֵ֣ל מֵחֹ֔רֶב כִּ֛י ר֥וּחַ עָרִיצִ֖ים כְּזֶ֥רֶם קִֽיר׃ (ו) כְּחֹ֣רֶב בְּצָי֔וֹן שְׁא֥וֹן זָרִ֖ים תַּכְנִ֑יעַ חֹ֚רֶב בְּצֵ֣ל עָ֔ב זְמִ֥יר עָֽרִיצִ֖ים יַעֲנֶֽה׃ (ז) וְעָשָׂה֩ יְהוָ֨ה צְבָא֜וֹת לְכָל־ הָֽעַמִּים֙ בָּהָ֣ר הַזֶּ֔ה מִשְׁתֵּ֥ה שְׁמָנִ֖ים מִשְׁתֵּ֣ה שְׁמָרִ֑ים שְׁמָנִים֙ מְמֻ֣חָיִ֔ם שְׁמָרִ֖ים מְזֻקָּקִֽים׃ (ח) וּבִלַּע֙ בָּהָ֣ר הַזֶּ֔ה פְּנֵֽי־ הַלּ֥וֹט ׀ הַלּ֖וֹט עַל־ כָּל־ הָֽעַמִּ֑ים וְהַמַּסֵּכָ֥ה הַנְּסוּכָ֖ה עַל־ כָּל־ הַגּוֹיִֽם׃ (ט) בִּלַּ֤ע הַמָּ֙וֶת֙ לָנֶ֔צַח וּמָחָ֨ה אֲדֹנָ֧י יְהוִ֛ה דִּמְעָ֖ה מֵעַ֣ל כָּל־ פָּנִ֑ים וְחֶרְפַּ֣ת עַמּ֗וֹ יָסִיר֙ מֵעַ֣ל כָּל־ הָאָ֔רֶץ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (י) וְאָמַר֙ בַּיּ֣וֹם הַה֔וּא הִנֵּ֨ה אֱלֹהֵ֥ינוּ זֶ֛ה קִוִּ֥ינוּ ל֖וֹ וְיֽוֹשִׁיעֵ֑נוּ זֶ֤ה יְהוָה֙ קִוִּ֣ינוּ ל֔וֹ נָגִ֥ילָה וְנִשְׂמְחָ֖ה בִּישׁוּעָתֽוֹ׃ (יא) כִּֽי־ תָנ֥וּחַ יַד־ יְהוָ֖ה בָּהָ֣ר הַזֶּ֑ה וְנָ֤דוֹשׁ מוֹאָב֙ תַּחְתָּ֔יו כְּהִדּ֥וּשׁ מַתְבֵּ֖ן (במי) בְּמ֥וֹ מַדְמֵנָֽה׃ (יב) וּפֵרַ֤שׂ יָדָיו֙ בְּקִרְבּ֔וֹ כַּאֲשֶׁ֛ר יְפָרֵ֥שׂ הַשֹּׂחֶ֖ה לִשְׂח֑וֹת וְהִשְׁפִּיל֙ גַּֽאֲוָת֔וֹ עִ֖ם אָרְבּ֥וֹת יָדָֽיו׃ (יג) וּמִבְצַ֞ר מִשְׂגַּ֣ב חוֹמֹתֶ֗יךָ הֵשַׁ֥ח הִשְׁפִּ֛יל הִגִּ֥יעַ לָאָ֖רֶץ עַד־ עָפָֽר׃ (יד) בַּיּ֣וֹם הַה֔וּא יוּשַׁ֥ר הַשִּׁיר־ הַזֶּ֖ה בְּאֶ֣רֶץ יְהוּדָ֑ה עִ֣יר עָז־ לָ֔נוּ יְשׁוּעָ֥ה יָשִׁ֖ית חוֹמ֥וֹת וָחֵֽל׃ (טו) פִּתְח֖וּ שְׁעָרִ֑ים וְיָבֹ֥א גוֹי־ צַדִּ֖יק שֹׁמֵ֥ר אֱמֻנִֽים׃ (טז) יֵ֣צֶר סָמ֔וּךְ תִּצֹּ֖ר שָׁל֣וֹם ׀ שָׁל֑וֹם כִּ֥י בְךָ֖ בָּטֽוּחַ׃ (יז) בִּטְח֥וּ בַֽיהוָ֖ה עֲדֵי־ עַ֑ד כִּ֚י בְּיָ֣הּ יְהוָ֔ה צ֖וּר עוֹלָמִֽים׃ (יח) כִּ֤י הֵשַׁח֙ יֹשְׁבֵ֣י מָר֔וֹם קִרְיָ֖ה נִשְׂגָּבָ֑ה יַשְׁפִּילֶ֤נָּה יַשְׁפִּילָהּ֙ עַד־ אֶ֔רֶץ יַגִּיעֶ֖נָּה עַד־ עָפָֽר׃ (יט) תִּרְמְסֶ֖נָּה רָ֑גֶל רַגְלֵ֥י עָנִ֖י פַּעֲמֵ֥י דַלִּֽים׃ (כ) אֹ֥רַח לַצַּדִּ֖יק מֵֽישָׁרִ֑ים יָשָׁ֕ר מַעְגַּ֥ל צַדִּ֖יק תְּפַלֵּֽס׃ (כא) אַ֣ף אֹ֧רַח מִשְׁפָּטֶ֛יךָ יְהוָ֖ה קִוִּינ֑וּךָ לְשִׁמְךָ֥ וּֽלְזִכְרְךָ֖ תַּאֲוַת־ נָֽפֶשׁ׃ (כב) נַפְשִׁ֤י אִוִּיתִ֙יךָ֙ בַּלַּ֔יְלָה אַף־ רוּחִ֥י בְקִרְבִּ֖י אֲשַֽׁחֲרֶ֑ךָּ כִּ֞י כַּאֲשֶׁ֤ר מִשְׁפָּטֶ֙יךָ֙ לָאָ֔רֶץ צֶ֥דֶק לָמְד֖וּ יֹשְׁבֵ֥י תֵבֵֽל׃ (כג) יֻחַ֤ן רָשָׁע֙ בַּל־ לָמַ֣ד צֶ֔דֶק בְּאֶ֥רֶץ נְכֹח֖וֹת יְעַוֵּ֑ל וּבַל־ יִרְאֶ֖ה גֵּא֥וּת יְהוָֽה׃ (כד) יְהוָ֛ה רָ֥מָה יָדְךָ֖ בַּל־ יֶחֱזָי֑וּן יֶחֱז֤וּ וְיֵבֹ֙שׁוּ֙ קִנְאַת־ עָ֔ם אַף־ אֵ֖שׁ צָרֶ֥יךָ תֹאכְלֵֽם׃ (כה) יְהוָ֕ה תִּשְׁפֹּ֥ת שָׁל֖וֹם לָ֑נוּ כִּ֛י גַּ֥ם כָּֽל־ מַעֲשֵׂ֖ינוּ פָּעַ֥לְתָּ לָּֽנוּ׃ (כו) יְהוָ֣ה אֱלֹהֵ֔ינוּ בְּעָל֥וּנוּ אֲדֹנִ֖ים זֽוּלָתֶ֑ךָ לְבַד־ בְּךָ֖ נַזְכִּ֥יר שְׁמֶֽךָ׃ (כז) מֵתִים֙ בַּל־ יִחְי֔וּ רְפָאִ֖ים בַּל־ יָקֻ֑מוּ לָכֵ֤ן פָּקַ֙דְתָּ֙ וַתַּשְׁמִידֵ֔ם וַתְּאַבֵּ֥ד כָּל־ זֵ֖כֶר לָֽמוֹ׃ (כח) יָסַ֤פְתָּ לַגּוֹי֙ יְהוָ֔ה יָסַ֥פְתָּ לַגּ֖וֹי נִכְבָּ֑דְתָּ רִחַ֖קְתָּ כָּל־ קַצְוֵי־ אָֽרֶץ׃ (כט) יְהוָ֖ה בַּצַּ֣ר פְּקָד֑וּךָ צָק֣וּן לַ֔חַשׁ מוּסָרְךָ֖ לָֽמוֹ׃ (ל) כְּמ֤וֹ הָרָה֙ תַּקְרִ֣יב לָלֶ֔דֶת תָּחִ֥יל תִּזְעַ֖ק בַּחֲבָלֶ֑יהָ כֵּ֛ן הָיִ֥ינוּ מִפָּנֶ֖יךָ יְהוָֽה׃ (לא) הָרִ֣ינוּ חַ֔לְנוּ כְּמ֖וֹ יָלַ֣דְנוּ ר֑וּחַ יְשׁוּעֹת֙ בַּל־ נַ֣עֲשֶׂה אֶ֔רֶץ וּבַֽל־ יִפְּל֖וּ יֹשְׁבֵ֥י תֵבֵֽל׃ (לב) יִֽחְי֣וּ מֵתֶ֔יךָ נְבֵלָתִ֖י יְקוּמ֑וּן הָקִ֨יצוּ וְרַנְּנ֜וּ שֹׁכְנֵ֣י עָפָ֗ר כִּ֣י טַ֤ל אוֹרֹת֙ טַלֶּ֔ךָ וָאָ֖רֶץ רְפָאִ֥ים תַּפִּֽיל׃ (לג) לֵ֤ךְ עַמִּי֙ בֹּ֣א בַחֲדָרֶ֔יךָ וּֽסְגֹ֥ר (דלתיך) דְּלָתְךָ֖ בַּעֲדֶ֑ךָ חֲבִ֥י כִמְעַט־ רֶ֖גַע עַד־ (יעבור)־ יַעֲבָר־ זָֽעַם׃ (לד) כִּֽי־ הִנֵּ֤ה יְהוָה֙ יֹצֵ֣א מִמְּקוֹמ֔וֹ לִפְקֹ֛ד עֲוֺ֥ן יֹֽשֵׁב־ הָאָ֖רֶץ עָלָ֑יו וְגִלְּתָ֤ה הָאָ֙רֶץ֙ אֶת־ דָּמֶ֔יהָ וְלֹֽא־ תְכַסֶּ֥ה ע֖וֹד עַל־ הֲרוּגֶֽיהָ׃ (לה) בַּיּ֣וֹם הַה֡וּא יִפְקֹ֣ד יְהוָה֩ בְּחַרְב֨וֹ הַקָּשָׁ֜ה וְהַגְּדוֹלָ֣ה וְהַֽחֲזָקָ֗ה עַ֤ל לִוְיָתָן֙ נָחָ֣שׁ בָּרִ֔חַ וְעַל֙ לִוְיָתָ֔ן נָחָ֖שׁ עֲקַלָּת֑וֹן וְהָרַ֥ג אֶת־ הַתַּנִּ֖ין אֲשֶׁ֥ר בַּיָּֽם׃ (לו) בַּיּ֖וֹם הַה֑וּא כֶּ֥רֶם חֶ֖מֶד עַנּוּ־ לָֽהּ׃ (לז) אֲנִ֤י יְהוָה֙ נֹֽצְרָ֔הּ לִרְגָעִ֖ים אַשְׁקֶ֑נָּה פֶּ֚ן יִפְקֹ֣ד עָלֶ֔יהָ לַ֥יְלָה וָי֖וֹם אֶצֳּרֶֽנָּה׃ (לח) חֵמָ֖ה אֵ֣ין לִ֑י מִֽי־ יִתְּנֵ֜נִי שָׁמִ֥יר שַׁ֙יִת֙ בַּמִּלְחָמָ֔ה אֶפְשְׂעָ֥ה בָ֖הּ אֲצִיתֶ֥נָּה יָּֽחַד׃ (לט) א֚וֹ יַחֲזֵ֣ק בְּמָעוּזִּ֔י יַעֲשֶׂ֥ה שָׁל֖וֹם לִ֑י שָׁל֖וֹם יַֽעֲשֶׂה־ לִּֽי׃ (מ) הַבָּאִים֙ יַשְׁרֵ֣שׁ יַֽעֲקֹ֔ב יָצִ֥יץ וּפָרַ֖ח יִשְׂרָאֵ֑ל וּמָלְא֥וּ פְנֵי־ תֵבֵ֖ל תְּנוּבָֽה׃ (מא) הַכְּמַכַּ֥ת מַכֵּ֖הוּ הִכָּ֑הוּ אִם־ כְּהֶ֥רֶג הֲרֻגָ֖יו הֹרָֽג׃ (מב) בְּסַּאסְּאָ֖ה בְּשַׁלְחָ֣הּ תְּרִיבֶ֑נָּה הָגָ֛ה בְּרוּח֥וֹ הַקָּשָׁ֖ה בְּי֥וֹם קָדִֽים׃ (מג) לָכֵ֗ן בְּזֹאת֙ יְכֻפַּ֣ר עֲוֺֽן־ יַעֲקֹ֔ב וְזֶ֕ה כָּל־ פְּרִ֖י הָסִ֣ר חַטָּאת֑וֹ בְּשׂוּמ֣וֹ ׀ כָּל־ אַבְנֵ֣י מִזְבֵּ֗חַ כְּאַבְנֵי־ גִר֙ מְנֻפָּצ֔וֹת לֹֽא־ יָקֻ֥מוּ אֲשֵׁרִ֖ים וְחַמָּנִֽים׃ (מד) כִּ֣י עִ֤יר בְּצוּרָה֙ בָּדָ֔ד נָוֶ֕ה מְשֻׁלָּ֥ח וְנֶעֱזָ֖ב כַּמִּדְבָּ֑ר שָׁ֣ם יִרְעֶ֥ה עֵ֛גֶל וְשָׁ֥ם יִרְבָּ֖ץ וְכִלָּ֥ה סְעִפֶֽיהָ׃ (מה) בִּיבֹ֤שׁ קְצִירָהּ֙ תִּשָּׁבַ֔רְנָה נָשִׁ֕ים בָּא֖וֹת מְאִיר֣וֹת אוֹתָ֑הּ כִּ֣י לֹ֤א עַם־ בִּינוֹת֙ ה֔וּא עַל־ כֵּן֙ לֹֽא־ יְרַחֲמֶ֣נּוּ עֹשֵׂ֔הוּ וְיֹצְר֖וֹ לֹ֥א יְחֻנֶּֽנּוּ׃
