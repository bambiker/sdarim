ישעיהו סדר ו

פתיחה באתר · ספר ישעיהו כולו ·

(א) וּלְקָח֣וּם עַמִּים֮ וֶהֱבִיא֣וּם אֶל־ מְקוֹמָם֒ וְהִֽתְנַחֲל֣וּם בֵּֽית־ יִשְׂרָאֵ֗ל עַ֚ל אַדְמַ֣ת יְהוָ֔ה לַעֲבָדִ֖ים וְלִשְׁפָח֑וֹת וְהָיוּ֙ שֹׁבִ֣ים לְשֹֽׁבֵיהֶ֔ם וְרָד֖וּ בְּנֹגְשֵׂיהֶֽם׃ (ב) וְהָיָ֗ה בְּי֨וֹם הָנִ֤יחַ יְהוָה֙ לְךָ֔ מֵֽעָצְבְּךָ֖ וּמֵרָגְזֶ֑ךָ וּמִן־ הָעֲבֹדָ֥ה הַקָּשָׁ֖ה אֲשֶׁ֥ר עֻבַּד־ בָּֽךְ׃ (ג) וְנָשָׂ֜אתָ הַמָּשָׁ֥ל הַזֶּ֛ה עַל־ מֶ֥לֶךְ בָּבֶ֖ל וְאָמָ֑רְתָּ אֵ֚יךְ שָׁבַ֣ת נֹגֵ֔שׂ שָׁבְתָ֖ה מַדְהֵבָֽה׃ (ד) שָׁבַ֥ר יְהוָ֖ה מַטֵּ֣ה רְשָׁעִ֑ים שֵׁ֖בֶט מֹשְׁלִֽים׃ (ה) מַכֶּ֤ה עַמִּים֙ בְּעֶבְרָ֔ה מַכַּ֖ת בִּלְתִּ֣י סָרָ֑ה רֹדֶ֤ה בָאַף֙ גּוֹיִ֔ם מֻרְדָּ֖ף בְּלִ֥י חָשָֽׂךְ׃ (ו) נָ֥חָה שָׁקְטָ֖ה כָּל־ הָאָ֑רֶץ פָּצְח֖וּ רִנָּֽה׃ (ז) גַּם־ בְּרוֹשִׁ֛ים שָׂמְח֥וּ לְךָ֖ אַרְזֵ֣י לְבָנ֑וֹן מֵאָ֣ז שָׁכַ֔בְתָּ לֹֽא־ יַעֲלֶ֥ה הַכֹּרֵ֖ת עָלֵֽינוּ׃ (ח) שְׁא֗וֹל מִתַּ֛חַת רָגְזָ֥ה לְךָ֖ לִקְרַ֣את בּוֹאֶ֑ךָ עוֹרֵ֨ר לְךָ֤ רְפָאִים֙ כָּל־ עַתּ֣וּדֵי אָ֔רֶץ הֵקִים֙ מִכִּסְאוֹתָ֔ם כֹּ֖ל מַלְכֵ֥י גוֹיִֽם׃ (ט) כֻּלָּ֣ם יַֽעֲנ֔וּ וְיֹאמְר֖וּ אֵלֶ֑יךָ גַּם־ אַתָּ֛ה חֻלֵּ֥יתָ כָמ֖וֹנוּ אֵלֵ֥ינוּ נִמְשָֽׁלְתָּ׃ (י) הוּרַ֥ד שְׁא֛וֹל גְּאוֹנֶ֖ךָ הֶמְיַ֣ת נְבָלֶ֑יךָ תַּחְתֶּ֙יךָ֙ יֻצַּ֣ע רִמָּ֔ה וּמְכַסֶּ֖יךָ תּוֹלֵעָֽה׃ (יא) אֵ֛יךְ נָפַ֥לְתָּ מִשָּׁמַ֖יִם הֵילֵ֣ל בֶּן־ שָׁ֑חַר נִגְדַּ֣עְתָּ לָאָ֔רֶץ חוֹלֵ֖שׁ עַל־ גּוֹיִֽם׃ (יב) וְאַתָּ֞ה אָמַ֤רְתָּ בִֽלְבָבְךָ֙ הַשָּׁמַ֣יִם אֶֽעֱלֶ֔ה מִמַּ֥עַל לְכֽוֹכְבֵי־ אֵ֖ל אָרִ֣ים כִּסְאִ֑י וְאֵשֵׁ֥ב בְּהַר־ מוֹעֵ֖ד בְּיַרְכְּתֵ֥י צָפֽוֹן׃ (יג) אֶעֱלֶ֖ה עַל־ בָּ֣מֳתֵי עָ֑ב אֶדַּמֶּ֖ה לְעֶלְיֽוֹן׃ (יד) אַ֧ךְ אֶל־ שְׁא֛וֹל תּוּרָ֖ד אֶל־ יַרְכְּתֵי־ בֽוֹר׃ (טו) רֹאֶ֙יךָ֙ אֵלֶ֣יךָ יַשְׁגִּ֔יחוּ אֵלֶ֖יךָ יִתְבּוֹנָ֑נוּ הֲזֶ֤ה הָאִישׁ֙ מַרְגִּ֣יז הָאָ֔רֶץ מַרְעִ֖ישׁ מַמְלָכֽוֹת׃ (טז) שָׂ֥ם תֵּבֵ֛ל כַּמִּדְבָּ֖ר וְעָרָ֣יו הָרָ֑ס אֲסִירָ֖יו לֹא־ פָ֥תַח בָּֽיְתָה׃ (יז) כָּל־ מַלְכֵ֥י גוֹיִ֖ם כֻּלָּ֑ם שָׁכְב֥וּ בְכָב֖וֹד אִ֥ישׁ בְּבֵיתֽוֹ׃ (יח) וְאַתָּ֞ה הָשְׁלַ֤כְתָּ מִֽקִּבְרְךָ֙ כְּנֵ֣צֶר נִתְעָ֔ב לְב֥וּשׁ הֲרֻגִ֖ים מְטֹ֣עֲנֵי חָ֑רֶב יוֹרְדֵ֥י אֶל־ אַבְנֵי־ ב֖וֹר כְּפֶ֥גֶר מוּבָֽס׃ (יט) לֹֽא־ תֵחַ֤ד אִתָּם֙ בִּקְבוּרָ֔ה כִּֽי־ אַרְצְךָ֥ שִׁחַ֖תָּ עַמְּךָ֣ הָרָ֑גְתָּ לֹֽא־ יִקָּרֵ֥א לְעוֹלָ֖ם זֶ֥רַע מְרֵעִֽים׃ (כ) הָכִ֧ינוּ לְבָנָ֛יו מַטְבֵּ֖חַ בַּעֲוֺ֣ן אֲבוֹתָ֑ם בַּל־ יָקֻ֙מוּ֙ וְיָ֣רְשׁוּ אָ֔רֶץ וּמָלְא֥וּ פְנֵֽי־ תֵבֵ֖ל עָרִֽים׃ (כא) וְקַמְתִּ֣י עֲלֵיהֶ֔ם נְאֻ֖ם יְהוָ֣ה צְבָא֑וֹת וְהִכְרַתִּ֨י לְבָבֶ֜ל שֵׁ֥ם וּשְׁאָ֛ר וְנִ֥ין וָנֶ֖כֶד נְאֻם־ יְהוָֽה׃ (כב) וְשַׂמְתִּ֛יהָ לְמוֹרַ֥שׁ קִפֹּ֖ד וְאַגְמֵי־ מָ֑יִם וְטֵֽאטֵאתִ֙יהָ֙ בְּמַטְאֲטֵ֣א הַשְׁמֵ֔ד נְאֻ֖ם יְהוָ֥ה צְבָאֽוֹת׃ (כג) נִשְׁבַּ֛ע יְהוָ֥ה צְבָא֖וֹת לֵאמֹ֑ר אִם־ לֹ֞א כַּאֲשֶׁ֤ר דִּמִּ֙יתִי֙ כֵּ֣ן הָיָ֔תָה וְכַאֲשֶׁ֥ר יָעַ֖צְתִּי הִ֥יא תָקֽוּם׃ (כד) לִשְׁבֹּ֤ר אַשּׁוּר֙ בְּאַרְצִ֔י וְעַל־ הָרַ֖י אֲבוּסֶ֑נּוּ וְסָ֤ר מֵֽעֲלֵיהֶם֙ עֻלּ֔וֹ וְסֻ֨בֳּל֔וֹ מֵעַ֥ל שִׁכְמ֖וֹ יָסֽוּר׃ (כה) זֹ֛את הָעֵצָ֥ה הַיְּעוּצָ֖ה עַל־ כָּל־ הָאָ֑רֶץ וְזֹ֛את הַיָּ֥ד הַנְּטוּיָ֖ה עַל־ כָּל־ הַגּוֹיִֽם׃ (כו) כִּֽי־ יְהוָ֧ה צְבָא֛וֹת יָעָ֖ץ וּמִ֣י יָפֵ֑ר וְיָד֥וֹ הַנְּטוּיָ֖ה וּֽמִ֥י יְשִׁיבֶֽנָּה׃ (כז) בִּשְׁנַת־ מ֖וֹת הַמֶּ֣לֶךְ אָחָ֑ז הָיָ֖ה הַמַּשָּׂ֥א הַזֶּֽה׃ (כח) אַֽל־ תִּשְׂמְחִ֤י פְלֶ֙שֶׁת֙ כֻּלֵּ֔ךְ כִּ֥י נִשְׁבַּ֖ר שֵׁ֣בֶט מַכֵּ֑ךְ כִּֽי־ מִשֹּׁ֤רֶשׁ נָחָשׁ֙ יֵ֣צֵא צֶ֔פַע וּפִרְי֖וֹ שָׂרָ֥ף מְעוֹפֵֽף׃ (כט) וְרָעוּ֙ בְּכוֹרֵ֣י דַלִּ֔ים וְאֶבְיוֹנִ֖ים לָבֶ֣טַח יִרְבָּ֑צוּ וְהֵמַתִּ֤י בָֽרָעָב֙ שָׁרְשֵׁ֔ךְ וּשְׁאֵרִיתֵ֖ךְ יַהֲרֹֽג׃ (ל) הֵילִ֤ילִֽי שַׁ֙עַר֙ זַֽעֲקִי־ עִ֔יר נָמ֖וֹג פְּלֶ֣שֶׁת כֻּלֵּ֑ךְ כִּ֤י מִצָּפוֹן֙ עָשָׁ֣ן בָּ֔א וְאֵ֥ין בּוֹדֵ֖ד בְּמוֹעָדָֽיו׃ (לא) וּמַֽה־ יַּעֲנֶ֖ה מַלְאֲכֵי־ ג֑וֹי כִּ֤י יְהוָה֙ יִסַּ֣ד צִיּ֔וֹן וּבָ֥הּ יֶחֱס֖וּ עֲנִיֵּ֥י עַמּֽוֹ׃ (לב) מַשָּׂ֖א מוֹאָ֑ב כִּ֠י בְּלֵ֞יל שֻׁדַּ֨ד עָ֤ר מוֹאָב֙ נִדְמָ֔ה כִּ֗י בְּלֵ֛יל שֻׁדַּ֥ד קִיר־ מוֹאָ֖ב נִדְמָֽה׃ (לג) עָלָ֨ה הַבַּ֧יִת וְדִיבֹ֛ן הַבָּמ֖וֹת לְבֶ֑כִי עַל־ נְב֞וֹ וְעַ֤ל מֵֽידְבָא֙ מוֹאָ֣ב יְיֵלִ֔יל בְּכָל־ רֹאשָׁ֣יו קָרְחָ֔ה כָּל־ זָקָ֖ן גְּרוּעָֽה׃ (לד) בְּחוּצֹתָ֖יו חָ֣גְרוּ שָׂ֑ק עַ֣ל גַּגּוֹתֶ֧יהָ וּבִרְחֹבֹתֶ֛יהָ כֻּלֹּ֥ה יְיֵלִ֖יל יֹרֵ֥ד בַּבֶּֽכִי׃ (לה) וַתִּזְעַ֤ק חֶשְׁבּוֹן֙ וְאֶלְעָלֵ֔ה עַד־ יַ֖הַץ נִשְׁמַ֣ע קוֹלָ֑ם עַל־ כֵּ֗ן חֲלֻצֵ֤י מוֹאָב֙ יָרִ֔יעוּ נַפְשׁ֖וֹ יָ֥רְעָה לּֽוֹ׃ (לו) לִבִּי֙ לְמוֹאָ֣ב יִזְעָ֔ק בְּרִיחֶ֕הָ עַד־ צֹ֖עַר עֶגְלַ֣ת שְׁלִשִׁיָּ֑ה כִּ֣י ׀ מַעֲלֵ֣ה הַלּוּחִ֗ית בִּבְכִי֙ יַֽעֲלֶה־ בּ֔וֹ כִּ֚י דֶּ֣רֶךְ חוֹרֹנַ֔יִם זַעֲקַת־ שֶׁ֖בֶר יְעֹעֵֽרוּ׃ (לז) כִּֽי־ מֵ֥י נִמְרִ֖ים מְשַׁמּ֣וֹת יִֽהְי֑וּ כִּֽי־ יָבֵ֤שׁ חָצִיר֙ כָּ֣לָה דֶ֔שֶׁא יֶ֖רֶק לֹ֥א הָיָֽה׃ (לח) עַל־ כֵּ֖ן יִתְרָ֣ה עָשָׂ֑ה וּפְקֻדָּתָ֔ם עַ֛ל נַ֥חַל הָעֲרָבִ֖ים יִשָּׂאֽוּם׃ (לט) כִּֽי־ הִקִּ֥יפָה הַזְּעָקָ֖ה אֶת־ גְּב֣וּל מוֹאָ֑ב עַד־ אֶגְלַ֙יִם֙ יִלְלָתָ֔הּ וּבְאֵ֥ר אֵילִ֖ים יִלְלָתָֽהּ׃ (מ) כִּ֣י מֵ֤י דִימוֹן֙ מָ֣לְאוּ דָ֔ם כִּֽי־ אָשִׁ֥ית עַל־ דִּימ֖וֹן נוֹסָפ֑וֹת לִפְלֵיטַ֤ת מוֹאָב֙ אַרְיֵ֔ה וְלִשְׁאֵרִ֖ית אֲדָמָֽה׃ (מא) שִׁלְחוּ־ כַ֥ר מֹשֵֽׁל־ אֶ֖רֶץ מִסֶּ֣לַע מִדְבָּ֑רָה אֶל־ הַ֖ר בַּת־ צִיּֽוֹן׃ (מב) וְהָיָ֥ה כְעוֹף־ נוֹדֵ֖ד קֵ֣ן מְשֻׁלָּ֑ח תִּֽהְיֶ֙ינָה֙ בְּנ֣וֹת מוֹאָ֔ב מַעְבָּרֹ֖ת לְאַרְנֽוֹן׃ (מג) (הביאו) הָבִ֤יאִי עֵצָה֙ עֲשׂ֣וּ פְלִילָ֔ה שִׁ֧יתִי כַלַּ֛יִל צִלֵּ֖ךְ בְּת֣וֹךְ צָהֳרָ֑יִם סַתְּרִי֙ נִדָּחִ֔ים נֹדֵ֖ד אַל־ תְּגַלִּֽי׃ (מד) יָג֤וּרוּ בָךְ֙ נִדָּחַ֔י מוֹאָ֛ב הֱוִי־ סֵ֥תֶר לָ֖מוֹ מִפְּנֵ֣י שׁוֹדֵ֑ד כִּֽי־ אָפֵ֤ס הַמֵּץ֙ כָּ֣לָה שֹׁ֔ד תַּ֥מּוּ רֹמֵ֖ס מִן־ הָאָֽרֶץ׃
