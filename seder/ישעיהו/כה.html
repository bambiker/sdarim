ישעיהו סדר כה

פתיחה באתר · ספר ישעיהו כולו ·

(א) וְנוֹדַ֤ע בַּגּוֹיִם֙ זַרְעָ֔ם וְצֶאֱצָאֵיהֶ֖ם בְּת֣וֹךְ הָעַמִּ֑ים כָּל־ רֹֽאֵיהֶם֙ יַכִּיר֔וּם כִּ֛י הֵ֥ם זֶ֖רַע בֵּרַ֥ךְ יְהוָֽה׃ (ב) שׂ֧וֹשׂ אָשִׂ֣ישׂ בַּֽיהוָ֗ה תָּגֵ֤ל נַפְשִׁי֙ בֵּֽאלֹהַ֔י כִּ֤י הִלְבִּישַׁ֙נִי֙ בִּגְדֵי־ יֶ֔שַׁע מְעִ֥יל צְדָקָ֖ה יְעָטָ֑נִי כֶּֽחָתָן֙ יְכַהֵ֣ן פְּאֵ֔ר וְכַכַּלָּ֖ה תַּעְדֶּ֥ה כֵלֶֽיהָ׃ (ג) כִּ֤י כָאָ֙רֶץ֙ תּוֹצִ֣יא צִמְחָ֔הּ וּכְגַנָּ֖ה זֵרוּעֶ֣יהָ תַצְמִ֑יחַ כֵּ֣ן ׀ אֲדֹנָ֣י יְהוִ֗ה יַצְמִ֤יחַ צְדָקָה֙ וּתְהִלָּ֔ה נֶ֖גֶד כָּל־ הַגּוֹיִֽם׃ (ד) לְמַ֤עַן צִיּוֹן֙ לֹ֣א אֶחֱשֶׁ֔ה וּלְמַ֥עַן יְרוּשָׁלִַ֖ם לֹ֣א אֶשְׁק֑וֹט עַד־ יֵצֵ֤א כַנֹּ֙גַהּ֙ צִדְקָ֔הּ וִישׁוּעָתָ֖הּ כְּלַפִּ֥יד יִבְעָֽר׃ (ה) וְרָא֤וּ גוֹיִם֙ צִדְקֵ֔ךְ וְכָל־ מְלָכִ֖ים כְּבוֹדֵ֑ךְ וְקֹ֤רָא לָךְ֙ שֵׁ֣ם חָדָ֔שׁ אֲשֶׁ֛ר פִּ֥י יְהוָ֖ה יִקֳּבֶֽנּוּ׃ (ו) וְהָיִ֛יתְ עֲטֶ֥רֶת תִּפְאֶ֖רֶת בְּיַד־ יְהוָ֑ה (וצנוף) וּצְנִ֥יף מְלוּכָ֖ה בְּכַף־ אֱלֹהָֽיִךְ׃ (ז) לֹֽא־ יֵאָמֵר֩ לָ֨ךְ ע֜וֹד עֲזוּבָ֗ה וּלְאַרְצֵךְ֙ לֹא־ יֵאָמֵ֥ר עוֹד֙ שְׁמָמָ֔ה כִּ֣י לָ֗ךְ יִקָּרֵא֙ חֶפְצִי־ בָ֔הּ וּלְאַרְצֵ֖ךְ בְּעוּלָ֑ה כִּֽי־ חָפֵ֤ץ יְהוָה֙ בָּ֔ךְ וְאַרְצֵ֖ךְ תִּבָּעֵֽל׃ (ח) כִּֽי־ יִבְעַ֤ל בָּחוּר֙ בְּתוּלָ֔ה יִבְעָל֖וּךְ בָּנָ֑יִךְ וּמְשׂ֤וֹשׂ חָתָן֙ עַל־ כַּלָּ֔ה יָשִׂ֥ישׂ עָלַ֖יִךְ אֱלֹהָֽיִךְ׃ (ט) עַל־ חוֹמֹתַ֣יִךְ יְרוּשָׁלִַ֗ם הִפְקַ֙דְתִּי֙ שֹֽׁמְרִ֔ים כָּל־ הַיּ֧וֹם וְכָל־ הַלַּ֛יְלָה תָּמִ֖יד לֹ֣א יֶחֱשׁ֑וּ הַמַּזְכִּרִים֙ אֶת־ יְהוָ֔ה אַל־ דֳּמִ֖י לָכֶֽם׃ (י) וְאַֽל־ תִּתְּנ֥וּ דֳמִ֖י ל֑וֹ עַד־ יְכוֹנֵ֞ן וְעַד־ יָשִׂ֧ים אֶת־ יְרֽוּשָׁלִַ֛ם תְּהִלָּ֖ה בָּאָֽרֶץ׃ (יא) נִשְׁבַּ֧ע יְהוָ֛ה בִּֽימִינ֖וֹ וּבִזְר֣וֹעַ עֻזּ֑וֹ אִם־ אֶתֵּן֩ אֶת־ דְּגָנֵ֨ךְ ע֤וֹד מַֽאֲכָל֙ לְאֹ֣יְבַ֔יִךְ וְאִם־ יִשְׁתּ֤וּ בְנֵֽי־ נֵכָר֙ תִּֽירוֹשֵׁ֔ךְ אֲשֶׁ֥ר יָגַ֖עַתְּ בּֽוֹ׃ (יב) כִּ֤י מְאַסְפָיו֙ יֹאכְלֻ֔הוּ וְהִֽלְל֖וּ אֶת־ יְהוָ֑ה וּֽמְקַבְּצָ֥יו יִשְׁתֻּ֖הוּ בְּחַצְר֥וֹת קָדְשִֽׁי׃ (יג) עִבְר֤וּ עִבְרוּ֙ בַּשְּׁעָרִ֔ים פַּנּ֖וּ דֶּ֣רֶךְ הָעָ֑ם סֹ֣לּוּ סֹ֤לּוּ הַֽמְסִלָּה֙ סַקְּל֣וּ מֵאֶ֔בֶן הָרִ֥ימוּ נֵ֖ס עַל־ הָעַמִּֽים׃ (יד) הִנֵּ֣ה יְהוָ֗ה הִשְׁמִ֙יעַ֙ אֶל־ קְצֵ֣ה הָאָ֔רֶץ אִמְרוּ֙ לְבַת־ צִיּ֔וֹן הִנֵּ֥ה יִשְׁעֵ֖ךְ בָּ֑א הִנֵּ֤ה שְׂכָרוֹ֙ אִתּ֔וֹ וּפְעֻלָּת֖וֹ לְפָנָֽיו׃ (טו) וְקָרְא֥וּ לָהֶ֛ם עַם־ הַקֹּ֖דֶשׁ גְּאוּלֵ֣י יְהוָ֑ה וְלָךְ֙ יִקָּרֵ֣א דְרוּשָׁ֔ה עִ֖יר לֹ֥א נֶעֱזָֽבָה׃ (טז) מִי־ זֶ֣ה ׀ בָּ֣א מֵאֱד֗וֹם חֲמ֤וּץ בְּגָדִים֙ מִבָּצְרָ֔ה זֶ֚ה הָד֣וּר בִּלְבוּשׁ֔וֹ צֹעֶ֖ה בְּרֹ֣ב כֹּח֑וֹ אֲנִ֛י מְדַבֵּ֥ר בִּצְדָקָ֖ה רַ֥ב לְהוֹשִֽׁיעַ׃ (יז) מַדּ֥וּעַ אָדֹ֖ם לִלְבוּשֶׁ֑ךָ וּבְגָדֶ֖יךָ כְּדֹרֵ֥ךְ בְּגַֽת׃ (יח) פּוּרָ֣ה ׀ דָּרַ֣כְתִּי לְבַדִּ֗י וּמֵֽעַמִּים֙ אֵֽין־ אִ֣ישׁ אִתִּ֔י וְאֶדְרְכֵ֣ם בְּאַפִּ֔י וְאֶרְמְסֵ֖ם בַּחֲמָתִ֑י וְיֵ֤ז נִצְחָם֙ עַל־ בְּגָדַ֔י וְכָל־ מַלְבּוּשַׁ֖י אֶגְאָֽלְתִּי׃ (יט) כִּ֛י י֥וֹם נָקָ֖ם בְּלִבִּ֑י וּשְׁנַ֥ת גְּאוּלַ֖י בָּֽאָה׃ (כ) וְאַבִּיט֙ וְאֵ֣ין עֹזֵ֔ר וְאֶשְׁתּוֹמֵ֖ם וְאֵ֣ין סוֹמֵ֑ךְ וַתּ֤וֹשַֽׁע לִי֙ זְרֹעִ֔י וַחֲמָתִ֖י הִ֥יא סְמָכָֽתְנִי׃ (כא) וְאָב֤וּס עַמִּים֙ בְּאַפִּ֔י וַאֲשַׁכְּרֵ֖ם בַּחֲמָתִ֑י וְאוֹרִ֥יד לָאָ֖רֶץ נִצְחָֽם׃ (כב) חַֽסְדֵ֨י יְהוָ֤ה ׀ אַזְכִּיר֙ תְּהִלֹּ֣ת יְהוָ֔ה כְּעַ֕ל כֹּ֥ל אֲשֶׁר־ גְּמָלָ֖נוּ יְהוָ֑ה וְרַב־ טוּב֙ לְבֵ֣ית יִשְׂרָאֵ֔ל אֲשֶׁר־ גְּמָלָ֥ם כְּֽרַחֲמָ֖יו וּכְרֹ֥ב חֲסָדָֽיו׃ (כג) וַיֹּ֙אמֶר֙ אַךְ־ עַמִּ֣י הֵ֔מָּה בָּנִ֖ים לֹ֣א יְשַׁקֵּ֑רוּ וַיְהִ֥י לָהֶ֖ם לְמוֹשִֽׁיעַ׃ (כד) בְּֽכָל־ צָרָתָ֣ם ׀ (לא) ל֣וֹ צָ֗ר וּמַלְאַ֤ךְ פָּנָיו֙ הֽוֹשִׁיעָ֔ם בְּאַהֲבָת֥וֹ וּבְחֶמְלָת֖וֹ ה֣וּא גְאָלָ֑ם וַֽיְנַטְּלֵ֥ם וַֽיְנַשְּׂאֵ֖ם כָּל־ יְמֵ֥י עוֹלָֽם׃ (כה) וְהֵ֛מָּה מָר֥וּ וְעִצְּב֖וּ אֶת־ ר֣וּחַ קָדְשׁ֑וֹ וַיֵּהָפֵ֥ךְ לָהֶ֛ם לְאוֹיֵ֖ב ה֥וּא נִלְחַם־ בָּֽם׃ (כו) וַיִּזְכֹּ֥ר יְמֵֽי־ עוֹלָ֖ם מֹשֶׁ֣ה עַמּ֑וֹ אַיֵּ֣ה ׀ הַֽמַּעֲלֵ֣ם מִיָּ֗ם אֵ֚ת רֹעֵ֣י צֹאנ֔וֹ אַיֵּ֛ה הַשָּׂ֥ם בְּקִרְבּ֖וֹ אֶת־ ר֥וּחַ קָדְשֽׁוֹ׃ (כז) מוֹלִיךְ֙ לִימִ֣ין מֹשֶׁ֔ה זְר֖וֹעַ תִּפְאַרְתּ֑וֹ בּ֤וֹקֵֽעַ מַ֙יִם֙ מִפְּנֵיהֶ֔ם לַעֲשׂ֥וֹת ל֖וֹ שֵׁ֥ם עוֹלָֽם׃ (כח) מוֹלִיכָ֖ם בַּתְּהֹמ֑וֹת כַּסּ֥וּס בַּמִּדְבָּ֖ר לֹ֥א יִכָּשֵֽׁלוּ׃ (כט) כַּבְּהֵמָה֙ בַּבִּקְעָ֣ה תֵרֵ֔ד ר֥וּחַ יְהוָ֖ה תְּנִיחֶ֑נּוּ כֵּ֚ן נִהַ֣גְתָּ עַמְּךָ֔ לַעֲשׂ֥וֹת לְךָ֖ שֵׁ֥ם תִּפְאָֽרֶת׃ (ל) הַבֵּ֤ט מִשָּׁמַ֙יִם֙ וּרְאֵ֔ה מִזְּבֻ֥ל קָדְשְׁךָ֖ וְתִפְאַרְתֶּ֑ךָ אַיֵּ֤ה קִנְאָֽתְךָ֙ וּגְב֣וּרֹתֶ֔ךָ הֲמ֥וֹן מֵעֶ֛יךָ וְֽרַחֲמֶ֖יךָ אֵלַ֥י הִתְאַפָּֽקוּ׃ (לא) כִּֽי־ אַתָּ֣ה אָבִ֔ינוּ כִּ֤י אַבְרָהָם֙ לֹ֣א יְדָעָ֔נוּ וְיִשְׂרָאֵ֖ל לֹ֣א יַכִּירָ֑נוּ אַתָּ֤ה יְהוָה֙ אָבִ֔ינוּ גֹּאֲלֵ֥נוּ מֵֽעוֹלָ֖ם שְׁמֶֽךָ׃ (לב) לָ֣מָּה תַתְעֵ֤נוּ יְהוָה֙ מִדְּרָכֶ֔יךָ תַּקְשִׁ֥יחַ לִבֵּ֖נוּ מִיִּרְאָתֶ֑ךָ שׁ֚וּב לְמַ֣עַן עֲבָדֶ֔יךָ שִׁבְטֵ֖י נַחֲלָתֶֽךָ׃ (לג) לַמִּצְעָ֕ר יָרְשׁ֖וּ עַם־ קָדְשֶׁ֑ךָ צָרֵ֕ינוּ בּוֹסְס֖וּ מִקְדָּשֶֽׁךָ׃ (לד) הָיִ֗ינוּ מֵֽעוֹלָם֙ לֹֽא־ מָשַׁ֣לְתָּ בָּ֔ם לֹֽא־ נִקְרָ֥א שִׁמְךָ֖ עֲלֵיהֶ֑ם לוּא־ קָרַ֤עְתָּ שָׁמַ֙יִם֙ יָרַ֔דְתָּ מִפָּנֶ֖יךָ הָרִ֥ים נָזֹֽלּוּ׃ (לה) כִּקְדֹ֧חַ אֵ֣שׁ הֲמָסִ֗ים מַ֚יִם תִּבְעֶה־ אֵ֔שׁ לְהוֹדִ֥יעַ שִׁמְךָ֖ לְצָרֶ֑יךָ מִפָּנֶ֖יךָ גּוֹיִ֥ם יִרְגָּֽזוּ׃ (לו) בַּעֲשׂוֹתְךָ֥ נוֹרָא֖וֹת לֹ֣א נְקַוֶּ֑ה יָרַ֕דְתָּ מִפָּנֶ֖יךָ הָרִ֥ים נָזֹֽלּוּ׃ (לז) וּמֵעוֹלָ֥ם לֹא־ שָׁמְע֖וּ לֹ֣א הֶאֱזִ֑ינוּ עַ֣יִן לֹֽא־ רָאָ֗תָה אֱלֹהִים֙ זוּלָ֣תְךָ֔ יַעֲשֶׂ֖ה לִמְחַכֵּה־ לֽוֹ׃ (לח) פָּגַ֤עְתָּ אֶת־ שָׂשׂ֙ וְעֹ֣שֵׂה צֶ֔דֶק בִּדְרָכֶ֖יךָ יִזְכְּר֑וּךָ הֵן־ אַתָּ֤ה קָצַ֙פְתָּ֙ וַֽנֶּחֱטָ֔א בָּהֶ֥ם עוֹלָ֖ם וְנִוָּשֵֽׁעַ׃ (לט) וַנְּהִ֤י כַטָּמֵא֙ כֻּלָּ֔נוּ וּכְבֶ֥גֶד עִדִּ֖ים כָּל־ צִדְקֹתֵ֑ינוּ וַנָּ֤בֶל כֶּֽעָלֶה֙ כֻּלָּ֔נוּ וַעֲוֺנֵ֖נוּ כָּר֥וּחַ יִשָּׂאֻֽנוּ׃ (מ) וְאֵין־ קוֹרֵ֣א בְשִׁמְךָ֔ מִתְעוֹרֵ֖ר לְהַחֲזִ֣יק בָּ֑ךְ כִּֽי־ הִסְתַּ֤רְתָּ פָנֶ֙יךָ֙ מִמֶּ֔נּוּ וַתְּמוּגֵ֖נוּ בְּיַד־ עֲוֺנֵֽנוּ׃ (מא) וְעַתָּ֥ה יְהוָ֖ה אָבִ֣ינוּ אָ֑תָּה אֲנַ֤חְנוּ הַחֹ֙מֶר֙ וְאַתָּ֣ה יֹצְרֵ֔נוּ וּמַעֲשֵׂ֥ה יָדְךָ֖ כֻּלָּֽנוּ׃ (מב) אַל־ תִּקְצֹ֤ף יְהוָה֙ עַד־ מְאֹ֔ד וְאַל־ לָעַ֖ד תִּזְכֹּ֣ר עָוֺ֑ן הֵ֥ן הַבֶּט־ נָ֖א עַמְּךָ֥ כֻלָּֽנוּ׃ (מג) עָרֵ֥י קָדְשְׁךָ֖ הָי֣וּ מִדְבָּ֑ר צִיּוֹן֙ מִדְבָּ֣ר הָיָ֔תָה יְרוּשָׁלִַ֖ם שְׁמָמָֽה׃ (מד) בֵּ֧ית קָדְשֵׁ֣נוּ וְתִפְאַרְתֵּ֗נוּ אֲשֶׁ֤ר הִֽלְל֙וּךָ֙ אֲבֹתֵ֔ינוּ הָיָ֖ה לִשְׂרֵ֣פַת אֵ֑שׁ וְכָל־ מַחֲמַדֵּ֖ינוּ הָיָ֥ה לְחָרְבָּֽה׃ (מה) הַעַל־ אֵ֥לֶּה תִתְאַפַּ֖ק יְהוָ֑ה תֶּחֱשֶׁ֥ה וּתְעַנֵּ֖נוּ עַד־ מְאֹֽד׃ (מו) נִדְרַ֙שְׁתִּי֙ לְל֣וֹא שָׁאָ֔לוּ נִמְצֵ֖אתִי לְלֹ֣א בִקְשֻׁ֑נִי אָמַ֙רְתִּי֙ הִנֵּ֣נִי הִנֵּ֔נִי אֶל־ גּ֖וֹי לֹֽא־ קֹרָ֥א בִשְׁמִֽי׃ (מז) פֵּרַ֧שְׂתִּי יָדַ֛י כָּל־ הַיּ֖וֹם אֶל־ עַ֣ם סוֹרֵ֑ר הַהֹלְכִים֙ הַדֶּ֣רֶךְ לֹא־ ט֔וֹב אַחַ֖ר מַחְשְׁבֹתֵיהֶֽם׃ (מח) הָעָ֗ם הַמַּכְעִיסִ֥ים אוֹתִ֛י עַל־ פָּנַ֖י תָּמִ֑יד זֹֽבְחִים֙ בַּגַּנּ֔וֹת וּֽמְקַטְּרִ֖ים עַל־ הַלְּבֵנִֽים׃ (מט) הַיֹּֽשְׁבִים֙ בַּקְּבָרִ֔ים וּבַנְּצוּרִ֖ים יָלִ֑ינוּ הָאֹֽכְלִים֙ בְּשַׂ֣ר הַחֲזִ֔יר (ופרק) וּמְרַ֥ק פִּגֻּלִ֖ים כְּלֵיהֶֽם׃ (נ) הָאֹֽמְרִים֙ קְרַ֣ב אֵלֶ֔יךָ אַל־ תִּגַּשׁ־ בִּ֖י כִּ֣י קְדַשְׁתִּ֑יךָ אֵ֚לֶּה עָשָׁ֣ן בְּאַפִּ֔י אֵ֥שׁ יֹקֶ֖דֶת כָּל־ הַיּֽוֹם׃ (נא) הִנֵּ֥ה כְתוּבָ֖ה לְפָנָ֑י לֹ֤א אֶחֱשֶׂה֙ כִּ֣י אִם־ שִׁלַּ֔מְתִּי וְשִׁלַּמְתִּ֖י עַל־ חֵיקָֽם׃ (נב) עֲ֠וֺנֹתֵיכֶם וַעֲוֺנֹ֨ת אֲבוֹתֵיכֶ֤ם יַחְדָּו֙ אָמַ֣ר יְהוָ֔ה אֲשֶׁ֤ר קִטְּרוּ֙ עַל־ הֶ֣הָרִ֔ים וְעַל־ הַגְּבָע֖וֹת חֵרְפ֑וּנִי וּמַדֹּתִ֧י פְעֻלָּתָ֛ם רִֽאשֹׁנָ֖ה (על)־ אֶל־ חֵיקָֽם׃ (נג) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה כַּאֲשֶׁ֨ר יִמָּצֵ֤א הַתִּירוֹשׁ֙ בָּֽאֶשְׁכּ֔וֹל וְאָמַר֙ אַל־ תַּשְׁחִיתֵ֔הוּ כִּ֥י בְרָכָ֖ה בּ֑וֹ כֵּ֤ן אֶֽעֱשֶׂה֙ לְמַ֣עַן עֲבָדַ֔י לְבִלְתִּ֖י הַֽשְׁחִ֥ית הַכֹּֽל׃
