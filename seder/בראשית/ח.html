בראשית סדר ח

פרשת נח

פתיחה באתר · ספר בראשית כולו ·

(א) וַיִּֽהְי֣וּ בְנֵי־ נֹ֗חַ הַיֹּֽצְאִים֙ מִן־ הַתֵּבָ֔ה שֵׁ֖ם וְחָ֣ם וָיָ֑פֶת וְחָ֕ם ה֖וּא אֲבִ֥י כְנָֽעַן׃ (ב) שְׁלֹשָׁ֥ה אֵ֖לֶּה בְּנֵי־ נֹ֑חַ וּמֵאֵ֖לֶּה נָֽפְצָ֥ה כָל־ הָאָֽרֶץ׃ (ג) וַיָּ֥חֶל נֹ֖חַ אִ֣ישׁ הָֽאֲדָמָ֑ה וַיִּטַּ֖ע כָּֽרֶם׃ (ד) וַיֵּ֥שְׁתְּ מִן־ הַיַּ֖יִן וַיִּשְׁכָּ֑ר וַיִּתְגַּ֖ל בְּת֥וֹךְ אָהֳלֹֽה׃ (ה) וַיַּ֗רְא חָ֚ם אֲבִ֣י כְנַ֔עַן אֵ֖ת עֶרְוַ֣ת אָבִ֑יו וַיַּגֵּ֥ד לִשְׁנֵֽי־ אֶחָ֖יו בַּחֽוּץ׃ (ו) וַיִּקַּח֩ שֵׁ֨ם וָיֶ֜פֶת אֶת־ הַשִּׂמְלָ֗ה וַיָּשִׂ֙ימוּ֙ עַל־ שְׁכֶ֣ם שְׁנֵיהֶ֔ם וַיֵּֽלְכוּ֙ אֲחֹ֣רַנִּ֔ית וַיְכַסּ֕וּ אֵ֖ת עֶרְוַ֣ת אֲבִיהֶ֑ם וּפְנֵיהֶם֙ אֲחֹ֣רַנִּ֔ית וְעֶרְוַ֥ת אֲבִיהֶ֖ם לֹ֥א רָאֽוּ׃ (ז) וַיִּ֥יקֶץ נֹ֖חַ מִיֵּינ֑וֹ וַיֵּ֕דַע אֵ֛ת אֲשֶׁר־ עָ֥שָׂה־ ל֖וֹ בְּנ֥וֹ הַקָּטָֽן׃ (ח) וַיֹּ֖אמֶר אָר֣וּר כְּנָ֑עַן עֶ֥בֶד עֲבָדִ֖ים יִֽהְיֶ֥ה לְאֶחָֽיו׃ (ט) וַיֹּ֕אמֶר בָּר֥וּךְ יְהֹוָ֖ה אֱלֹ֣הֵי שֵׁ֑ם וִיהִ֥י כְנַ֖עַן עֶ֥בֶד לָֽמוֹ׃ (י) יַ֤פְתְּ אֱלֹהִים֙ לְיֶ֔פֶת וְיִשְׁכֹּ֖ן בְּאָֽהֳלֵי־ שֵׁ֑ם וִיהִ֥י כְנַ֖עַן עֶ֥בֶד לָֽמוֹ׃ (יא) וַֽיְחִי־ נֹ֖חַ אַחַ֣ר הַמַּבּ֑וּל שְׁלֹ֤שׁ מֵאוֹת֙ שָׁנָ֔ה וַֽחֲמִשִּׁ֖ים שָׁנָֽה׃ (יב) וַיִּֽהְיוּ֙ כָּל־ יְמֵי־ נֹ֔חַ תְּשַׁ֤ע מֵאוֹת֙ שָׁנָ֔ה וַחֲמִשִּׁ֖ים שָׁנָ֑ה וַיָּמֹֽת׃
(יג) וְאֵ֙לֶּה֙ תּוֹלְדֹ֣ת בְּנֵי־ נֹ֔חַ שֵׁ֖ם חָ֣ם וָיָ֑פֶת וַיִּוָּלְד֥וּ לָהֶ֛ם בָּנִ֖ים אַחַ֥ר הַמַּבּֽוּל׃ (יד) בְּנֵ֣י יֶ֔פֶת גֹּ֣מֶר וּמָג֔וֹג וּמָדַ֖י וְיָוָ֣ן וְתֻבָ֑ל וּמֶ֖שֶׁךְ וְתִירָֽס׃ (טו) וּבְנֵ֖י גֹּ֑מֶר אַשְׁכֲּנַ֥ז וְרִיפַ֖ת וְתֹגַרְמָֽה׃ (טז) וּבְנֵ֥י יָוָ֖ן אֱלִישָׁ֣ה וְתַרְשִׁ֑ישׁ כִּתִּ֖ים וְדֹדָנִֽים׃ (יז) מֵ֠אֵלֶּה נִפְרְד֞וּ אִיֵּ֤י הַגּוֹיִם֙ בְּאַרְצֹתָ֔ם אִ֖ישׁ לִלְשֹׁנ֑וֹ לְמִשְׁפְּחֹתָ֖ם בְּגוֹיֵהֶֽם׃ (יח) וּבְנֵ֖י חָ֑ם כּ֥וּשׁ וּמִצְרַ֖יִם וּפ֥וּט וּכְנָֽעַן׃ (יט) וּבְנֵ֣י כ֔וּשׁ סְבָא֙ וַֽחֲוִילָ֔ה וְסַבְתָּ֥ה וְרַעְמָ֖ה וְסַבְתְּכָ֑א וּבְנֵ֥י רַעְמָ֖ה שְׁבָ֥א וּדְדָֽן׃ (כ) וְכ֖וּשׁ יָלַ֣ד אֶת־ נִמְרֹ֑ד ה֣וּא הֵחֵ֔ל לִֽהְי֥וֹת גִּבֹּ֖ר בָּאָֽרֶץ׃ (כא) הֽוּא־ הָיָ֥ה גִבֹּֽר־ צַ֖יִד לִפְנֵ֣י יְהוָ֑ה עַל־ כֵּן֙ יֵֽאָמַ֔ר כְּנִמְרֹ֛ד גִּבּ֥וֹר צַ֖יִד לִפְנֵ֥י יְהוָֽה׃ (כב) וַתְּהִ֨י רֵאשִׁ֤ית מַמְלַכְתּוֹ֙ בָּבֶ֔ל וְאֶ֖רֶךְ וְאַכַּ֣ד וְכַלְנֵ֑ה בְּאֶ֖רֶץ שִׁנְעָֽר׃ (כג) מִן־ הָאָ֥רֶץ הַהִ֖וא יָצָ֣א אַשּׁ֑וּר וַיִּ֙בֶן֙ אֶת־ נִ֣ינְוֵ֔ה וְאֶת־ רְחֹבֹ֥ת עִ֖יר וְאֶת־ כָּֽלַח׃ (כד) וְֽאֶת־ רֶ֔סֶן בֵּ֥ין נִֽינְוֵ֖ה וּבֵ֣ין כָּ֑לַח הִ֖וא הָעִ֥יר הַגְּדֹלָֽה׃ (כה) וּמִצְרַ֡יִם יָלַ֞ד אֶת־ לוּדִ֧ים וְאֶת־ עֲנָמִ֛ים וְאֶת־ לְהָבִ֖ים וְאֶת־ נַפְתֻּחִֽים׃ (כו) וְֽאֶת־ פַּתְרֻסִ֞ים וְאֶת־ כַּסְלֻחִ֗ים אֲשֶׁ֨ר יָצְא֥וּ מִשָּׁ֛ם פְּלִשְׁתִּ֖ים וְאֶת־ כַּפְתֹּרִֽים׃      (כז) וּכְנַ֗עַן יָלַ֛ד אֶת־ צִידֹ֥ן בְּכֹר֖וֹ וְאֶת־ חֵֽת׃ (כח) וְאֶת־ הַיְבוּסִי֙ וְאֶת־ הָ֣אֱמֹרִ֔י וְאֵ֖ת הַגִּרְגָּשִֽׁי׃ (כט) וְאֶת־ הַֽחִוִּ֥י וְאֶת־ הַֽעַרְקִ֖י וְאֶת־ הַסִּינִֽי׃ (ל) וְאֶת־ הָֽאַרְוָדִ֥י וְאֶת־ הַצְּמָרִ֖י וְאֶת־ הַֽחֲמָתִ֑י וְאַחַ֣ר נָפֹ֔צוּ מִשְׁפְּח֖וֹת הַֽכְּנַעֲנִֽי׃ (לא) וַֽיְהִ֞י גְּב֤וּל הַֽכְּנַעֲנִי֙ מִצִּידֹ֔ן בֹּאֲכָ֥ה גְרָ֖רָה עַד־ עַזָּ֑ה בֹּאֲכָ֞ה סְדֹ֧מָה וַעֲמֹרָ֛ה וְאַדְמָ֥ה וּצְבֹיִ֖ם עַד־ לָֽשַׁע׃ (לב) אֵ֣לֶּה בְנֵי־ חָ֔ם לְמִשְׁפְּחֹתָ֖ם לִלְשֹֽׁנֹתָ֑ם בְּאַרְצֹתָ֖ם בְּגוֹיֵהֶֽם׃      (לג) וּלְשֵׁ֥ם יֻלַּ֖ד גַּם־ ה֑וּא אֲבִי֙ כָּל־ בְּנֵי־ עֵ֔בֶר אֲחִ֖י יֶ֥פֶת הַגָּדֽוֹל׃ (לד) בְּנֵ֥י שֵׁ֖ם עֵילָ֣ם וְאַשּׁ֑וּר וְאַרְפַּכְשַׁ֖ד וְל֥וּד וַֽאֲרָֽם׃ (לה) וּבְנֵ֖י אֲרָ֑ם ע֥וּץ וְח֖וּל וְגֶ֥תֶר וָמַֽשׁ׃ (לו) וְאַרְפַּכְשַׁ֖ד יָלַ֣ד אֶת־ שָׁ֑לַח וְשֶׁ֖לַח יָלַ֥ד אֶת־ עֵֽבֶר׃ (לז) וּלְעֵ֥בֶר יֻלַּ֖ד שְׁנֵ֣י בָנִ֑ים שֵׁ֣ם הָֽאֶחָ֞ד פֶּ֗לֶג כִּ֤י בְיָמָיו֙ נִפְלְגָ֣ה הָאָ֔רֶץ וְשֵׁ֥ם אָחִ֖יו יָקְטָֽן׃ (לח) וְיָקְטָ֣ן יָלַ֔ד אֶת־ אַלְמוֹדָ֖ד וְאֶת־ שָׁ֑לֶף וְאֶת־ חֲצַרְמָ֖וֶת וְאֶת־ יָֽרַח׃ (לט) וְאֶת־ הֲדוֹרָ֥ם וְאֶת־ אוּזָ֖ל וְאֶת־ דִּקְלָֽה׃ (מ) וְאֶת־ עוֹבָ֥ל וְאֶת־ אֲבִֽימָאֵ֖ל וְאֶת־ שְׁבָֽא׃ (מא) וְאֶת־ אוֹפִ֥ר וְאֶת־ חֲוִילָ֖ה וְאֶת־ יוֹבָ֑ב כָּל־ אֵ֖לֶּה בְּנֵ֥י יָקְטָֽן׃ (מב) וַֽיְהִ֥י מוֹשָׁבָ֖ם מִמֵּשָׁ֑א בֹּאֲכָ֥ה סְפָ֖רָה הַ֥ר הַקֶּֽדֶם׃ (מג) אֵ֣לֶּה בְנֵי־ שֵׁ֔ם לְמִשְׁפְּחֹתָ֖ם לִלְשֹׁנֹתָ֑ם בְּאַרְצֹתָ֖ם לְגוֹיֵהֶֽם׃ (מד) אֵ֣לֶּה מִשְׁפְּחֹ֧ת בְּנֵי־ נֹ֛חַ לְתוֹלְדֹתָ֖ם בְּגוֹיֵהֶ֑ם וּמֵאֵ֜לֶּה נִפְרְד֧וּ הַגּוֹיִ֛ם בָּאָ֖רֶץ אַחַ֥ר הַמַּבּֽוּל׃
