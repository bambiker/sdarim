בראשית סדר לא

לפי החלוקה לפרקים: בראשית לג, יח – לה, ח · 42 פסוקים · פרשת וישלח

פתיחה באתר · ספר בראשית כולו ·

(א) וַיָּבֹא֩ יַעֲקֹ֨ב שָׁלֵ֜ם עִ֣יר שְׁכֶ֗ם אֲשֶׁר֙ בְּאֶ֣רֶץ כְּנַ֔עַן בְּבֹא֖וֹ מִפַּדַּ֣ן אֲרָ֑ם וַיִּ֖חַן אֶת־ פְּנֵ֥י הָעִֽיר׃ (ב) וַיִּ֜קֶן אֶת־ חֶלְקַ֣ת הַשָּׂדֶ֗ה אֲשֶׁ֤ר נָֽטָה־ שָׁם֙ אָהֳל֔וֹ מִיַּ֥ד בְּנֵֽי־ חֲמ֖וֹר אֲבִ֣י שְׁכֶ֑ם בְּמֵאָ֖ה קְשִׂיטָֽה׃ (ג) וַיַּצֶּב־ שָׁ֖ם מִזְבֵּ֑חַ וַיִּ֨קְרָא־ ל֔וֹ אֵ֖ל אֱלֹהֵ֥י יִשְׂרָאֵֽל׃      (ד) וַתֵּצֵ֤א דִינָה֙ בַּת־ לֵאָ֔ה אֲשֶׁ֥ר יָלְדָ֖ה לְיַעֲקֹ֑ב לִרְא֖וֹת בִּבְנ֥וֹת הָאָֽרֶץ׃ (ה) וַיַּ֨רְא אֹתָ֜הּ שְׁכֶ֧ם בֶּן־ חֲמ֛וֹר הַֽחִוִּ֖י נְשִׂ֣יא הָאָ֑רֶץ וַיִּקַּ֥ח אֹתָ֛הּ וַיִּשְׁכַּ֥ב אֹתָ֖הּ וַיְעַנֶּֽהָ׃ (ו) וַתִּדְבַּ֣ק נַפְשׁ֔וֹ בְּדִינָ֖ה בַּֽת־ יַעֲקֹ֑ב וַיֶּֽאֱהַב֙ אֶת־ הַֽנַּעֲרָ֔ וַיְדַבֵּ֖ר עַל־ לֵ֥ב הַֽנַּעֲרָֽ׃ (ז) וַיֹּ֣אמֶר שְׁכֶ֔ם אֶל־ חֲמ֥וֹר אָבִ֖יו לֵאמֹ֑ר קַֽח־ לִ֛י אֶת־ הַיַּלְדָּ֥ה הַזֹּ֖את לְאִשָּֽׁה׃ (ח) וְיַעֲקֹ֣ב שָׁמַ֗ע כִּ֤י טִמֵּא֙ אֶת־ דִּינָ֣ה בִתּ֔וֹ וּבָנָ֛יו הָי֥וּ אֶת־ מִקְנֵ֖הוּ בַּשָּׂדֶ֑ה וְהֶחֱרִ֥שׁ יַעֲקֹ֖ב עַד־ בֹּאָֽם׃ (ט) וַיֵּצֵ֛א חֲמ֥וֹר אֲבִֽי־ שְׁכֶ֖ם אֶֽל־ יַעֲקֹ֑ב לְדַבֵּ֖ר אִתּֽוֹ׃ (י) וּבְנֵ֨י יַעֲקֹ֜ב בָּ֤אוּ מִן־ הַשָּׂדֶה֙ כְּשָׁמְעָ֔ם וַיִּֽתְעַצְּבוּ֙ הָֽאֲנָשִׁ֔ים וַיִּ֥חַר לָהֶ֖ם מְאֹ֑ד כִּֽי־ נְבָלָ֞ה עָשָׂ֣ה בְיִשְׂרָאֵ֗ל לִשְׁכַּב֙ אֶת־ בַּֽת־ יַעֲקֹ֔ב וְכֵ֖ן לֹ֥א יֵעָשֶֽׂה׃ (יא) וַיְדַבֵּ֥ר חֲמ֖וֹר אִתָּ֣ם לֵאמֹ֑ר שְׁכֶ֣ם בְּנִ֗י חָֽשְׁקָ֤ה נַפְשׁוֹ֙ בְּבִתְּכֶ֔ם תְּנ֨וּ נָ֥א אֹתָ֛הּ ל֖וֹ לְאִשָּֽׁה׃ (יב) וְהִֽתְחַתְּנ֖וּ אֹתָ֑נוּ בְּנֹֽתֵיכֶם֙ תִּתְּנוּ־ לָ֔נוּ וְאֶת־ בְּנֹתֵ֖ינוּ תִּקְח֥וּ לָכֶֽם׃ (יג) וְאִתָּ֖נוּ תֵּשֵׁ֑בוּ וְהָאָ֙רֶץ֙ תִּהְיֶ֣ה לִפְנֵיכֶ֔ם שְׁבוּ֙ וּסְחָר֔וּהָ וְהֵֽאָחֲז֖וּ בָּֽהּ׃ (יד) וַיֹּ֤אמֶר שְׁכֶם֙ אֶל־ אָבִ֣יה וְאֶל־ אַחֶ֔יהָ אֶמְצָא־ חֵ֖ן בְּעֵינֵיכֶ֑ם וַאֲשֶׁ֥ר תֹּאמְר֛וּ אֵלַ֖י אֶתֵּֽן׃ (טו) הַרְבּ֨וּ עָלַ֤י מְאֹד֙ מֹ֣הַר וּמַתָּ֔ן וְאֶ֨תְּנָ֔ה כַּאֲשֶׁ֥ר תֹּאמְר֖וּ אֵלָ֑י וּתְנוּ־ לִ֥י אֶת־ הַֽנַּעֲרָ֖ לְאִשָּֽׁה׃ (טז) וַיַּעֲנ֨וּ בְנֵֽי־ יַעֲקֹ֜ב אֶת־ שְׁכֶ֨ם וְאֶת־ חֲמ֥וֹר אָבִ֛יו בְּמִרְמָ֖ה וַיְדַבֵּ֑רוּ אֲשֶׁ֣ר טִמֵּ֔א אֵ֖ת דִּינָ֥ה אֲחֹתָֽם׃ (יז) וַיֹּאמְר֣וּ אֲלֵיהֶ֗ם לֹ֤א נוּכַל֙ לַעֲשׂוֹת֙ הַדָּבָ֣ר הַזֶּ֔ה לָתֵת֙ אֶת־ אֲחֹתֵ֔נוּ לְאִ֖ישׁ אֲשֶׁר־ ל֣וֹ עָרְלָ֑ה כִּֽי־ חֶרְפָּ֥ה הִ֖וא לָֽנוּ׃ (יח) אַךְ־ בְּזֹ֖את נֵא֣וֹת לָכֶ֑ם אִ֚ם תִּהְי֣וּ כָמֹ֔נוּ לְהִמֹּ֥ל לָכֶ֖ם כָּל־ זָכָֽר׃ (יט) וְנָתַ֤נּוּ אֶת־ בְּנֹתֵ֙ינוּ֙ לָכֶ֔ם וְאֶת־ בְּנֹתֵיכֶ֖ם נִֽקַּֽח־ לָ֑נוּ וְיָשַׁ֣בְנוּ אִתְּכֶ֔ם וְהָיִ֖ינוּ לְעַ֥ם אֶחָֽד׃ (כ) וְאִם־ לֹ֧א תִשְׁמְע֛וּ אֵלֵ֖ינוּ לְהִמּ֑וֹל וְלָקַ֥חְנוּ אֶת־ בִּתֵּ֖נוּ וְהָלָֽכְנוּ׃ (כא) וַיִּֽיטְב֥וּ דִבְרֵיהֶ֖ם בְּעֵינֵ֣י חֲמ֑וֹר וּבְעֵינֵ֖י שְׁכֶ֥ם בֶּן־ חֲמֽוֹר׃ (כב) וְלֹֽא־ אֵחַ֤ר הַנַּ֙עַר֙ לַעֲשׂ֣וֹת הַדָּבָ֔ר כִּ֥י חָפֵ֖ץ בְּבַֽת־ יַעֲקֹ֑ב וְה֣וּא נִכְבָּ֔ד מִכֹּ֖ל בֵּ֥ית אָבִֽיו׃ (כג) וַיָּבֹ֥א חֲמ֛וֹר וּשְׁכֶ֥ם בְּנ֖וֹ אֶל־ שַׁ֣עַר עִירָ֑ם וַֽיְדַבְּר֛וּ אֶל־ אַנְשֵׁ֥י עִירָ֖ם לֵאמֹֽר׃ (כד) הָאֲנָשִׁ֨ים הָאֵ֜לֶּה שְֽׁלֵמִ֧ים הֵ֣ם אִתָּ֗נוּ וְיֵשְׁב֤וּ בָאָ֙רֶץ֙ וְיִסְחֲר֣וּ אֹתָ֔הּ וְהָאָ֛רֶץ הִנֵּ֥ה רַֽחֲבַת־ יָדַ֖יִם לִפְנֵיהֶ֑ם אֶת־ בְּנֹתָם֙ נִקַּֽח־ לָ֣נוּ לְנָשִׁ֔ים וְאֶת־ בְּנֹתֵ֖ינוּ נִתֵּ֥ן לָהֶֽם׃ (כה) אַךְ־ בְּ֠זֹאת יֵאֹ֨תוּ לָ֤נוּ הָאֲנָשִׁים֙ לָשֶׁ֣בֶת אִתָּ֔נוּ לִהְי֖וֹת לְעַ֣ם אֶחָ֑ד בְּהִמּ֥וֹל לָ֙נוּ֙ כָּל־ זָכָ֔ר כַּאֲשֶׁ֖ר הֵ֥ם נִמֹּלִֽים׃ (כו) מִקְנֵהֶ֤ם וְקִנְיָנָם֙ וְכָל־ בְּהֶמְתָּ֔ם הֲל֥וֹא לָ֖נוּ הֵ֑ם אַ֚ךְ נֵא֣וֹתָה לָהֶ֔ם וְיֵשְׁב֖וּ אִתָּֽנוּ׃ (כז) וַיִּשְׁמְע֤וּ אֶל־ חֲמוֹר֙ וְאֶל־ שְׁכֶ֣ם בְּנ֔וֹ כָּל־ יֹצְאֵ֖י שַׁ֣עַר עִיר֑וֹ וַיִּמֹּ֙לוּ֙ כָּל־ זָכָ֔ר כָּל־ יֹצְאֵ֖י שַׁ֥עַר עִירֽוֹ׃ (כח) וַיְהִי֩ בַיּ֨וֹם הַשְּׁלִישִׁ֜י בִּֽהְיוֹתָ֣ם כֹּֽאֲבִ֗ים וַיִּקְח֣וּ שְׁנֵֽי־ בְנֵי־ יַ֠עֲקֹב שִׁמְע֨וֹן וְלֵוִ֜י אֲחֵ֤י דִינָה֙ אִ֣ישׁ חַרְבּ֔וֹ וַיָּבֹ֥אוּ עַל־ הָעִ֖יר בֶּ֑טַח וַיַּֽהַרְג֖וּ כָּל־ זָכָֽר׃ (כט) וְאֶת־ חֲמוֹר֙ וְאֶת־ שְׁכֶ֣ם בְּנ֔וֹ הָרְג֖וּ לְפִי־ חָ֑רֶב וַיִּקְח֧וּ אֶת־ דִּינָ֛ה מִבֵּ֥ית שְׁכֶ֖ם וַיֵּצֵֽאוּ׃ (ל) בְּנֵ֣י יַעֲקֹ֗ב בָּ֚אוּ עַל־ הַ֣חֲלָלִ֔ים וַיָּבֹ֖זּוּ הָעִ֑יר אֲשֶׁ֥ר טִמְּא֖וּ אֲחוֹתָֽם׃ (לא) אֶת־ צֹאנָ֥ם וְאֶת־ בְּקָרָ֖ם וְאֶת־ חֲמֹרֵיהֶּ֑ם וְאֵ֧ת אֲשֶׁר־ בָּעִ֛יר וְאֶת־ אֲשֶׁ֥ר בַּשָּׂדֶ֖ה לָקָֽחוּ׃ (לב) וְאֶת־ כָּל־ חֵילָ֤ם וְאֶת־ כָּל־ טַפָּם֙ וְאֶת־ נְשֵׁיהֶ֔ם שָׁב֖וּ וַיָּבֹ֑זּוּ וְאֵ֖ת כָּל־ אֲשֶׁ֥ר בַּבָּֽיִת׃ (לג) וַיֹּ֨אמֶר יַעֲקֹ֜ב אֶל־ שִׁמְע֣וֹן וְאֶל־ לֵוִי֮ עֲכַרְתֶּ֣ם אֹתִי֒ לְהַבְאִישֵׁ֙נִי֙ בְּיֹשֵׁ֣ב הָאָ֔רֶץ בַּֽכְּנַעֲנִ֖י וּבַפְּרִזִּ֑י וַאֲנִי֙ מְתֵ֣י מִסְפָּ֔ר וְנֶאֶסְפ֤וּ עָלַי֙ וְהִכּ֔וּנִי וְנִשְׁמַדְתִּ֖י אֲנִ֥י וּבֵיתִֽי׃ (לד) וַיֹּאמְר֑וּ הַכְזוֹנָ֕ה יַעֲשֶׂ֖ה אֶת־ אֲחוֹתֵֽנוּ׃
(לה) וַיֹּ֤אמֶר אֱלֹהִים֙ אֶֽל־ יַעֲקֹ֔ב ק֛וּם עֲלֵ֥ה בֵֽית־ אֵ֖ל וְשֶׁב־ שָׁ֑ם וַעֲשֵׂה־ שָׁ֣ם מִזְבֵּ֔חַ לָאֵל֙ הַנִּרְאֶ֣ה אֵלֶ֔יךָ בְּבָרְחֲךָ֔ מִפְּנֵ֖י עֵשָׂ֥ו אָחִֽיךָ׃ (לו) וַיֹּ֤אמֶר יַעֲקֹב֙ אֶל־ בֵּית֔וֹ וְאֶ֖ל כָּל־ אֲשֶׁ֣ר עִמּ֑וֹ הָסִ֜רוּ אֶת־ אֱלֹהֵ֤י הַנֵּכָר֙ אֲשֶׁ֣ר בְּתֹכְכֶ֔ם וְהִֽטַּהֲר֔וּ וְהַחֲלִ֖יפוּ שִׂמְלֹתֵיכֶֽם׃ (לז) וְנָק֥וּמָה וְנַעֲלֶ֖ה בֵּֽית־ אֵ֑ל וְאֶֽעֱשֶׂה־ שָּׁ֣ם מִזְבֵּ֗חַ לָאֵ֞ל הָעֹנֶ֤ה אֹתִי֙ בְּי֣וֹם צָֽרָתִ֔י וַיְהִי֙ עִמָּדִ֔י בַּדֶּ֖רֶךְ אֲשֶׁ֥ר הָלָֽכְתִּי׃ (לח) וַיִּתְּנ֣וּ אֶֽל־ יַעֲקֹ֗ב אֵ֣ת כָּל־ אֱלֹהֵ֤י הַנֵּכָר֙ אֲשֶׁ֣ר בְּיָדָ֔ם וְאֶת־ הַנְּזָמִ֖ים אֲשֶׁ֣ר בְּאָזְנֵיהֶ֑ם וַיִּטְמֹ֤ן אֹתָם֙ יַעֲקֹ֔ב תַּ֥חַת הָאֵלָ֖ה אֲשֶׁ֥ר עִם־ שְׁכֶֽם׃ (לט) וַיִּסָּ֑עוּ וַיְהִ֣י ׀ חִתַּ֣ת אֱלֹהִ֗ים עַל־ הֶֽעָרִים֙ אֲשֶׁר֙ סְבִיבֹ֣תֵיהֶ֔ם וְלֹ֣א רָֽדְפ֔וּ אַחֲרֵ֖י בְּנֵ֥י יַעֲקֹֽב׃ (מ) וַיָּבֹ֨א יַעֲקֹ֜ב ל֗וּזָה אֲשֶׁר֙ בְּאֶ֣רֶץ כְּנַ֔עַן הִ֖וא בֵּֽית־ אֵ֑ל ה֖וּא וְכָל־ הָעָ֥ם אֲשֶׁר־ עִמּֽוֹ׃ (מא) וַיִּ֤בֶן שָׁם֙ מִזְבֵּ֔חַ וַיִּקְרָא֙ לַמָּק֔וֹם אֵ֖ל בֵּֽית־ אֵ֑ל כִּ֣י שָׁ֗ם נִגְל֤וּ אֵלָיו֙ הָֽאֱלֹהִ֔ים בְּבָרְח֖וֹ מִפְּנֵ֥י אָחִֽיו׃ (מב) וַתָּ֤מָת דְּבֹרָה֙ מֵינֶ֣קֶת רִבְקָ֔ה וַתִּקָּבֵ֛ר מִתַּ֥חַת לְבֵֽית־ אֵ֖ל תַּ֣חַת הָֽאַלּ֑וֹן וַיִּקְרָ֥א שְׁמ֖וֹ אַלּ֥וֹן בָּכֽוּת׃
