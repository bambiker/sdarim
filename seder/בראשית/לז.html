בראשית סדר לז

37 פסוקים · פרשת מקץ

פתיחה באתר · ספר בראשית כולו ·

(א) וַיֹּ֥אמֶר פַּרְעֹ֖ה אֶל־ עֲבָדָ֑יו הֲנִמְצָ֣א כָזֶ֔ה אִ֕ישׁ אֲשֶׁ֛ר ר֥וּחַ אֱלֹהִ֖ים בּֽוֹ׃ (ב) וַיֹּ֤אמֶר פַּרְעֹה֙ אֶל־ יוֹסֵ֔ף אַחֲרֵ֨י הוֹדִ֧יעַ אֱלֹהִ֛ים אוֹתְךָ֖ אֶת־ כָּל־ זֹ֑את אֵין־ נָב֥וֹן וְחָכָ֖ם כָּמֽוֹךָ׃ (ג) אַתָּה֙ תִּהְיֶ֣ה עַל־ בֵּיתִ֔י וְעַל־ פִּ֖יךָ יִשַּׁ֣ק כָּל־ עַמִּ֑י רַ֥ק הַכִּסֵּ֖א אֶגְדַּ֥ל מִמֶּֽךָּ׃ (ד) וַיֹּ֥אמֶר פַּרְעֹ֖ה אֶל־ יוֹסֵ֑ף רְאֵה֙ נָתַ֣תִּי אֹֽתְךָ֔ עַ֖ל כָּל־ אֶ֥רֶץ מִצְרָֽיִם׃ (ה) וַיָּ֨סַר פַּרְעֹ֤ה אֶת־ טַבַּעְתּוֹ֙ מֵעַ֣ל יָד֔וֹ וַיִּתֵּ֥ן אֹתָ֖הּ עַל־ יַ֣ד יוֹסֵ֑ף וַיַּלְבֵּ֤שׁ אֹתוֹ֙ בִּגְדֵי־ שֵׁ֔שׁ וַיָּ֛שֶׂם רְבִ֥ד הַזָּהָ֖ב עַל־ צַוָּארֽוֹ׃ (ו) וַיַּרְכֵּ֣ב אֹת֗וֹ בְּמִרְכֶּ֤בֶת הַמִּשְׁנֶה֙ אֲשֶׁר־ ל֔וֹ וַיִּקְרְא֥וּ לְפָנָ֖יו אַבְרֵ֑ךְ וְנָת֣וֹן אֹת֔וֹ עַ֖ל כָּל־ אֶ֥רֶץ מִצְרָֽיִם׃ (ז) וַיֹּ֧אמֶר פַּרְעֹ֛ה אֶל־ יוֹסֵ֖ף אֲנִ֣י פַרְעֹ֑ה וּבִלְעָדֶ֗יךָ לֹֽא־ יָרִ֨ים אִ֧ישׁ אֶת־ יָד֛וֹ וְאֶת־ רַגְל֖וֹ בְּכָל־ אֶ֥רֶץ מִצְרָֽיִם׃ (ח) וַיִּקְרָ֨א פַרְעֹ֣ה שֵׁם־ יוֹסֵף֮ צָֽפְנַ֣ת פַּעְנֵחַ֒ וַיִּתֶּן־ ל֣וֹ אֶת־ אָֽסְנַ֗ת בַּת־ פּ֥וֹטִי פֶ֛רַע כֹּהֵ֥ן אֹ֖ן לְאִשָּׁ֑ה וַיֵּצֵ֥א יוֹסֵ֖ף עַל־ אֶ֥רֶץ מִצְרָֽיִם׃ (ט) וְיוֹסֵף֙ בֶּן־ שְׁלֹשִׁ֣ים שָׁנָ֔ה בְּעָמְד֕וֹ לִפְנֵ֖י פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרָ֑יִם וַיֵּצֵ֤א יוֹסֵף֙ מִלִּפְנֵ֣י פַרְעֹ֔ה וַֽיַּעְבֹ֖ר בְּכָל־ אֶ֥רֶץ מִצְרָֽיִם׃ (י) וַתַּ֣עַשׂ הָאָ֔רֶץ בְּשֶׁ֖בַע שְׁנֵ֣י הַשָּׂבָ֑ע לִקְמָצִֽים׃ (יא) וַיִּקְבֹּ֞ץ אֶת־ כָּל־ אֹ֣כֶל ׀ שֶׁ֣בַע שָׁנִ֗ים אֲשֶׁ֤ר הָיוּ֙ בְּאֶ֣רֶץ מִצְרַ֔יִם וַיִּתֶּן־ אֹ֖כֶל בֶּעָרִ֑ים אֹ֧כֶל שְׂדֵה־ הָעִ֛יר אֲשֶׁ֥ר סְבִיבֹתֶ֖יהָ נָתַ֥ן בְּתוֹכָֽהּ׃ (יב) וַיִּצְבֹּ֨ר יוֹסֵ֥ף בָּ֛ר כְּח֥וֹל הַיָּ֖ם הַרְבֵּ֣ה מְאֹ֑ד עַ֛ד כִּי־ חָדַ֥ל לִסְפֹּ֖ר כִּי־ אֵ֥ין מִסְפָּֽר׃ (יג) וּלְיוֹסֵ֤ף יֻלַּד֙ שְׁנֵ֣י בָנִ֔ים בְּטֶ֥רֶם תָּב֖וֹא שְׁנַ֣ת הָרָעָ֑ב אֲשֶׁ֤ר יָֽלְדָה־ לּוֹ֙ אָֽסְנַ֔ת בַּת־ פּ֥וֹטִי פֶ֖רַע כֹּהֵ֥ן אֽוֹן׃ (יד) וַיִּקְרָ֥א יוֹסֵ֛ף אֶת־ שֵׁ֥ם הַבְּכ֖וֹר מְנַשֶּׁ֑ה כִּֽי־ נַשַּׁ֤נִי אֱלֹהִים֙ אֶת־ כָּל־ עֲמָלִ֔י וְאֵ֖ת כָּל־ בֵּ֥ית אָבִֽי׃ (טו) וְאֵ֛ת שֵׁ֥ם הַשֵּׁנִ֖י קָרָ֣א אֶפְרָ֑יִם כִּֽי־ הִפְרַ֥נִי אֱלֹהִ֖ים בְּאֶ֥רֶץ עָנְיִֽי׃ (טז) וַתִּכְלֶ֕ינָה שֶׁ֖בַע שְׁנֵ֣י הַשָּׂבָ֑ע אֲשֶׁ֥ר הָיָ֖ה בְּאֶ֥רֶץ מִצְרָֽיִם׃ (יז) וַתְּחִלֶּ֜ינָה שֶׁ֣בַע שְׁנֵ֤י הָרָעָב֙ לָב֔וֹא כַּאֲשֶׁ֖ר אָמַ֣ר יוֹסֵ֑ף וַיְהִ֤י רָעָב֙ בְּכָל־ הָ֣אֲרָצ֔וֹת וּבְכָל־ אֶ֥רֶץ מִצְרַ֖יִם הָ֥יָה לָֽחֶם׃ (יח) וַתִּרְעַב֙ כָּל־ אֶ֣רֶץ מִצְרַ֔יִם וַיִּצְעַ֥ק הָעָ֛ם אֶל־ פַּרְעֹ֖ה לַלָּ֑חֶם וַיֹּ֨אמֶר פַּרְעֹ֤ה לְכָל־ מִצְרַ֙יִם֙ לְכ֣וּ אֶל־ יוֹסֵ֔ף אֲשֶׁר־ יֹאמַ֥ר לָכֶ֖ם תַּעֲשֽׂוּ׃ (יט) וְהָרָעָ֣ב הָיָ֔ה עַ֖ל כָּל־ פְּנֵ֣י הָאָ֑רֶץ וַיִּפְתַּ֨ח יוֹסֵ֜ף אֶֽת־ כָּל־ אֲשֶׁ֤ר בָּהֶם֙ וַיִּשְׁבֹּ֣ר לְמִצְרַ֔יִם וַיֶּחֱזַ֥ק הָֽרָעָ֖ב בְּאֶ֥רֶץ מִצְרָֽיִם׃ (כ) וְכָל־ הָאָ֙רֶץ֙ בָּ֣אוּ מִצְרַ֔יְמָה לִשְׁבֹּ֖ר אֶל־ יוֹסֵ֑ף כִּֽי־ חָזַ֥ק הָרָעָ֖ב בְּכָל־ הָאָֽרֶץ׃ (כא) וַיַּ֣רְא יַעֲקֹ֔ב כִּ֥י יֶשׁ־ שֶׁ֖בֶר בְּמִצְרָ֑יִם וַיֹּ֤אמֶר יַעֲקֹב֙ לְבָנָ֔יו לָ֖מָּה תִּתְרָאֽוּ׃ (כב) וַיֹּ֕אמֶר הִנֵּ֣ה שָׁמַ֔עְתִּי כִּ֥י יֶשׁ־ שֶׁ֖בֶר בְּמִצְרָ֑יִם רְדוּ־ שָׁ֙מָּה֙ וְשִׁבְרוּ־ לָ֣נוּ מִשָּׁ֔ם וְנִחְיֶ֖ה וְלֹ֥א נָמֽוּת׃ (כג) וַיֵּרְד֥וּ אֲחֵֽי־ יוֹסֵ֖ף עֲשָׂרָ֑ה לִשְׁבֹּ֥ר בָּ֖ר מִמִּצְרָֽיִם׃ (כד) וְאֶת־ בִּנְיָמִין֙ אֲחִ֣י יוֹסֵ֔ף לֹא־ שָׁלַ֥ח יַעֲקֹ֖ב אֶת־ אֶחָ֑יו כִּ֣י אָמַ֔ר פֶּן־ יִקְרָאֶ֖נּוּ אָסֽוֹן׃ (כה) וַיָּבֹ֙אוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לִשְׁבֹּ֖ר בְּת֣וֹךְ הַבָּאִ֑ים כִּֽי־ הָיָ֥ה הָרָעָ֖ב בְּאֶ֥רֶץ כְּנָֽעַן׃ (כו) וְיוֹסֵ֗ף ה֚וּא הַשַּׁלִּ֣יט עַל־ הָאָ֔רֶץ ה֥וּא הַמַּשְׁבִּ֖יר לְכָל־ עַ֣ם הָאָ֑רֶץ וַיָּבֹ֙אוּ֙ אֲחֵ֣י יוֹסֵ֔ף וַיִּשְׁתַּֽחֲווּ־ ל֥וֹ אַפַּ֖יִם אָֽרְצָה׃ (כז) וַיַּ֥רְא יוֹסֵ֛ף אֶת־ אֶחָ֖יו וַיַּכִּרֵ֑ם וַיִּתְנַכֵּ֨ר אֲלֵיהֶ֜ם וַיְדַבֵּ֧ר אִתָּ֣ם קָשׁ֗וֹת וַיֹּ֤אמֶר אֲלֵהֶם֙ מֵאַ֣יִן בָּאתֶ֔ם וַיֹּ֣אמְר֔וּ מֵאֶ֥רֶץ כְּנַ֖עַן לִשְׁבָּר־ אֹֽכֶל׃ (כח) וַיַּכֵּ֥ר יוֹסֵ֖ף אֶת־ אֶחָ֑יו וְהֵ֖ם לֹ֥א הִכִּרֻֽהוּ׃ (כט) וַיִּזְכֹּ֣ר יוֹסֵ֔ף אֵ֚ת הַחֲלֹמ֔וֹת אֲשֶׁ֥ר חָלַ֖ם לָהֶ֑ם וַיֹּ֤אמֶר אֲלֵהֶם֙ מְרַגְּלִ֣ים אַתֶּ֔ם לִרְא֛וֹת אֶת־ עֶרְוַ֥ת הָאָ֖רֶץ בָּאתֶֽם׃ (ל) וַיֹּאמְר֥וּ אֵלָ֖יו לֹ֣א אֲדֹנִ֑י וַעֲבָדֶ֥יךָ בָּ֖אוּ לִשְׁבָּר־ אֹֽכֶל׃ (לא) כֻּלָּ֕נוּ בְּנֵ֥י אִישׁ־ אֶחָ֖ד נָ֑חְנוּ כֵּנִ֣ים אֲנַ֔חְנוּ לֹא־ הָי֥וּ עֲבָדֶ֖יךָ מְרַגְּלִֽים׃ (לב) וַיֹּ֖אמֶר אֲלֵהֶ֑ם לֹ֕א כִּֽי־ עֶרְוַ֥ת הָאָ֖רֶץ בָּאתֶ֥ם לִרְאֽוֹת׃ (לג) וַיֹּאמְר֗וּ שְׁנֵ֣ים עָשָׂר֩ עֲבָדֶ֨יךָ אַחִ֧ים ׀ אֲנַ֛חְנוּ בְּנֵ֥י אִישׁ־ אֶחָ֖ד בְּאֶ֣רֶץ כְּנָ֑עַן וְהִנֵּ֨ה הַקָּטֹ֤ן אֶת־ אָבִ֙ינוּ֙ הַיּ֔וֹם וְהָאֶחָ֖ד אֵינֶֽנּוּ׃ (לד) וַיֹּ֥אמֶר אֲלֵהֶ֖ם יוֹסֵ֑ף ה֗וּא אֲשֶׁ֨ר דִּבַּ֧רְתִּי אֲלֵכֶ֛ם לֵאמֹ֖ר מְרַגְּלִ֥ים אַתֶּֽם׃ (לה) בְּזֹ֖את תִּבָּחֵ֑נוּ חֵ֤י פַרְעֹה֙ אִם־ תֵּצְא֣וּ מִזֶּ֔ה כִּ֧י אִם־ בְּב֛וֹא אֲחִיכֶ֥ם הַקָּטֹ֖ן הֵֽנָּה׃ (לו) שִׁלְח֨וּ מִכֶּ֣ם אֶחָד֮ וְיִקַּ֣ח אֶת־ אֲחִיכֶם֒ וְאַתֶּם֙ הֵאָ֣סְר֔וּ וְיִבָּֽחֲנוּ֙ דִּבְרֵיכֶ֔ם הַֽאֱמֶ֖ת אִתְּכֶ֑ם וְאִם־ לֹ֕א חֵ֣י פַרְעֹ֔ה כִּ֥י מְרַגְּלִ֖ים אַתֶּֽם׃ (לז) וַיֶּאֱסֹ֥ף אֹתָ֛ם אֶל־ מִשְׁמָ֖ר שְׁלֹ֥שֶׁת יָמִֽים׃
