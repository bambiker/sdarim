שמואל סדר כו

לפי החלוקה לפרקים: שמואל ב יב, יג – יג, כד · 43 פסוקים

פתיחה באתר · ספר שמואל כולו ·

(א) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ נָתָ֔ן חָטָ֖אתִי לַֽיהוָ֑ה וַיֹּ֨אמֶר נָתָ֜ן אֶל־ דָּוִ֗ד גַּם־ יְהוָ֛ה הֶעֱבִ֥יר חַטָּאתְךָ֖ לֹ֥א תָמֽוּת׃ (ב) אֶ֗פֶס כִּֽי־ נִאֵ֤ץ נִאַ֙צְתָּ֙ אֶת־ אֹיְבֵ֣י יְהוָ֔ה בַּדָּבָ֖ר הַזֶּ֑ה גַּ֗ם הַבֵּ֛ן הַיִּלּ֥וֹד לְךָ֖ מ֥וֹת יָמֽוּת׃ (ג) וַיֵּ֥לֶךְ נָתָ֖ן אֶל־ בֵּית֑וֹ וַיִּגֹּ֣ף יְהוָ֗ה אֶת־ הַיֶּ֜לֶד אֲשֶׁ֨ר יָלְדָ֧ה אֵֽשֶׁת־ אוּרִיָּ֛ה לְדָוִ֖ד וַיֵּאָנַֽשׁ׃ (ד) וַיְבַקֵּ֥שׁ דָּוִ֛ד אֶת־ הָאֱלֹהִ֖ים בְּעַ֣ד הַנָּ֑עַר וַיָּ֤צָם דָּוִד֙ צ֔וֹם וּבָ֥א וְלָ֖ן וְשָׁכַ֥ב אָֽרְצָה׃ (ה) וַיָּקֻ֜מוּ זִקְנֵ֤י בֵיתוֹ֙ עָלָ֔יו לַהֲקִימ֖וֹ מִן־ הָאָ֑רֶץ וְלֹ֣א אָבָ֔ה וְלֹֽא־ בָרָ֥א אִתָּ֖ם לָֽחֶם׃ (ו) וַיְהִ֛י בַּיּ֥וֹם הַשְּׁבִיעִ֖י וַיָּ֣מָת הַיָּ֑לֶד וַיִּֽרְאוּ֩ עַבְדֵ֨י דָוִ֜ד לְהַגִּ֥יד ל֣וֹ ׀ כִּי־ מֵ֣ת הַיֶּ֗לֶד כִּ֤י אָֽמְרוּ֙ הִנֵּה֩ בִהְי֨וֹת הַיֶּ֜לֶד חַ֗י דִּבַּ֤רְנוּ אֵלָיו֙ וְלֹא־ שָׁמַ֣ע בְּקוֹלֵ֔נוּ וְאֵ֨יךְ נֹאמַ֥ר אֵלָ֛יו מֵ֥ת הַיֶּ֖לֶד וְעָשָׂ֥ה רָעָֽה׃ (ז) וַיַּ֣רְא דָּוִ֗ד כִּ֤י עֲבָדָיו֙ מִֽתְלַחֲשִׁ֔ים וַיָּ֥בֶן דָּוִ֖ד כִּ֣י מֵ֣ת הַיָּ֑לֶד וַיֹּ֨אמֶר דָּוִ֧ד אֶל־ עֲבָדָ֛יו הֲמֵ֥ת הַיֶּ֖לֶד וַיֹּ֥אמְרוּ מֵֽת׃ (ח) וַיָּקָם֩ דָּוִ֨ד מֵהָאָ֜רֶץ וַיִּרְחַ֣ץ וַיָּ֗סֶךְ וַיְחַלֵּף֙ (שמלתו) שִׂמְלֹתָ֔יו וַיָּבֹ֥א בֵית־ יְהוָ֖ה וַיִּשְׁתָּ֑חוּ וַיָּבֹא֙ אֶל־ בֵּית֔וֹ וַיִּשְׁאַ֕ל וַיָּשִׂ֥ימוּ ל֛וֹ לֶ֖חֶם וַיֹּאכַֽל׃ (ט) וַיֹּאמְר֤וּ עֲבָדָיו֙ אֵלָ֔יו מָֽה־ הַדָּבָ֥ר הַזֶּ֖ה אֲשֶׁ֣ר עָשִׂ֑יתָה בַּעֲב֞וּר הַיֶּ֤לֶד חַי֙ צַ֣מְתָּ וַתֵּ֔בְךְּ וְכַֽאֲשֶׁר֙ מֵ֣ת הַיֶּ֔לֶד קַ֖מְתָּ וַתֹּ֥אכַל לָֽחֶם׃ (י) וַיֹּ֕אמֶר בְּעוֹד֙ הַיֶּ֣לֶד חַ֔י צַ֖מְתִּי וָֽאֶבְכֶּ֑ה כִּ֤י אָמַ֙רְתִּי֙ מִ֣י יוֹדֵ֔עַ (יחנני) וְחַנַּ֥נִי יְהוָ֖ה וְחַ֥י הַיָּֽלֶד׃ (יא) וְעַתָּ֣ה ׀ מֵ֗ת לָ֤מָּה זֶּה֙ אֲנִ֣י צָ֔ם הַאוּכַ֥ל לַהֲשִׁיב֖וֹ ע֑וֹד אֲנִי֙ הֹלֵ֣ךְ אֵלָ֔יו וְה֖וּא לֹֽא־ יָשׁ֥וּב אֵלָֽי׃ (יב) וַיְנַחֵ֣ם דָּוִ֗ד אֵ֚ת בַּת־ שֶׁ֣בַע אִשְׁתּ֔וֹ וַיָּבֹ֥א אֵלֶ֖יהָ וַיִּשְׁכַּ֣ב עִמָּ֑הּ וַתֵּ֣לֶד בֵּ֗ן (ויקרא) וַתִּקְרָ֤א אֶת־ שְׁמוֹ֙ שְׁלֹמֹ֔ה וַיהוָ֖ה אֲהֵבֽוֹ׃ (יג) וַיִּשְׁלַ֗ח בְּיַד֙ נָתָ֣ן הַנָּבִ֔יא וַיִּקְרָ֥א אֶת־ שְׁמ֖וֹ יְדִ֣ידְיָ֑הּ בַּעֲב֖וּר יְהוָֽה׃ (יד) וַיִּלָּ֣חֶם יוֹאָ֔ב בְּרַבַּ֖ת בְּנֵ֣י עַמּ֑וֹן וַיִּלְכֹּ֖ד אֶת־ עִ֥יר הַמְּלוּכָֽה׃ (טו) וַיִּשְׁלַ֥ח יוֹאָ֛ב מַלְאָכִ֖ים אֶל־ דָּוִ֑ד וַיֹּ֙אמֶר֙ נִלְחַ֣מְתִּי בְרַבָּ֔ה גַּם־ לָכַ֖דְתִּי אֶת־ עִ֥יר הַמָּֽיִם׃ (טז) וְעַתָּ֗ה אֱסֹף֙ אֶת־ יֶ֣תֶר הָעָ֔ם וַחֲנֵ֥ה עַל־ הָעִ֖יר וְלָכְדָ֑הּ פֶּן־ אֶלְכֹּ֤ד אֲנִי֙ אֶת־ הָעִ֔יר וְנִקְרָ֥א שְׁמִ֖י עָלֶֽיהָ׃ (יז) וַיֶּאֱסֹ֥ף דָּוִ֛ד אֶת־ כָּל־ הָעָ֖ם וַיֵּ֣לֶךְ רַבָּ֑תָה וַיִּלָּ֥חֶם בָּ֖הּ וַֽיִּלְכְּדָֽהּ׃ (יח) וַיִּקַּ֣ח אֶת־ עֲטֶֽרֶת־ מַלְכָּם֩ מֵעַ֨ל רֹאשׁ֜וֹ וּמִשְׁקָלָ֨הּ כִּכַּ֤ר זָהָב֙ וְאֶ֣בֶן יְקָרָ֔ה וַתְּהִ֖י עַל־ רֹ֣אשׁ דָּוִ֑ד וּשְׁלַ֥ל הָעִ֛יר הוֹצִ֖יא הַרְבֵּ֥ה מְאֹֽד׃ (יט) וְאֶת־ הָעָ֨ם אֲשֶׁר־ בָּ֜הּ הוֹצִ֗יא וַיָּ֣שֶׂם בַּ֠מְּגֵרָה וּבַחֲרִצֵ֨י הַבַּרְזֶ֜ל וּֽבְמַגְזְרֹ֣ת הַבַּרְזֶ֗ל וְהֶעֱבִ֤יר אוֹתָם֙ (במלכן) בַּמַּלְבֵּ֔ן וְכֵ֣ן יַעֲשֶׂ֔ה לְכֹ֖ל עָרֵ֣י בְנֵֽי־ עַמּ֑וֹן וַיָּ֧שָׁב דָּוִ֛ד וְכָל־ הָעָ֖ם יְרוּשָׁלִָֽם׃ (כ) וַיְהִ֣י אַֽחֲרֵי־ כֵ֗ן וּלְאַבְשָׁל֧וֹם בֶּן־ דָּוִ֛ד אָח֥וֹת יָפָ֖ה וּשְׁמָ֣הּ תָּמָ֑ר וַיֶּאֱהָבֶ֖הָ אַמְנ֥וֹן בֶּן־ דָּוִֽד׃ (כא) וַיֵּ֨צֶר לְאַמְנ֜וֹן לְהִתְחַלּ֗וֹת בַּֽעֲבוּר֙ תָּמָ֣ר אֲחֹת֔וֹ כִּ֥י בְתוּלָ֖ה הִ֑יא וַיִּפָּלֵא֙ בְּעֵינֵ֣י אַמְנ֔וֹן לַעֲשׂ֥וֹת לָ֖הּ מְאֽוּמָה׃ (כב) וּלְאַמְנ֣וֹן רֵ֗עַ וּשְׁמוֹ֙ יֽוֹנָדָ֔ב בֶּן־ שִׁמְעָ֖ה אֲחִ֣י דָוִ֑ד וְי֣וֹנָדָ֔ב אִ֥ישׁ חָכָ֖ם מְאֹֽד׃ (כג) וַיֹּ֣אמֶר ל֗וֹ מַדּ֣וּעַ אַ֠תָּה כָּ֣כָה דַּ֤ל בֶּן־ הַמֶּ֙לֶךְ֙ בַּבֹּ֣קֶר בַּבֹּ֔קֶר הֲל֖וֹא תַּגִּ֣יד לִ֑י וַיֹּ֤אמֶר לוֹ֙ אַמְנ֔וֹן אֶת־ תָּמָ֗ר אֲח֛וֹת אַבְשָׁלֹ֥ם אָחִ֖י אֲנִ֥י אֹהֵֽב׃ (כד) וַיֹּ֤אמֶר לוֹ֙ יְה֣וֹנָדָ֔ב שְׁכַ֥ב עַל־ מִשְׁכָּבְךָ֖ וְהִתְחָ֑ל וּבָ֧א אָבִ֣יךָ לִרְאוֹתֶ֗ךָ וְאָמַרְתָּ֣ אֵלָ֡יו תָּ֣בֹא נָא֩ תָמָ֨ר אֲחוֹתִ֜י וְתַבְרֵ֣נִי לֶ֗חֶם וְעָשְׂתָ֤ה לְעֵינַי֙ אֶת־ הַבִּרְיָ֔ה לְמַ֙עַן֙ אֲשֶׁ֣ר אֶרְאֶ֔ה וְאָכַלְתִּ֖י מִיָּדָֽהּ׃ (כה) וַיִּשְׁכַּ֥ב אַמְנ֖וֹן וַיִּתְחָ֑ל וַיָּבֹ֨א הַמֶּ֜לֶךְ לִרְאֹת֗וֹ וַיֹּ֨אמֶר אַמְנ֤וֹן אֶל־ הַמֶּ֙לֶךְ֙ תָּֽבוֹא־ נָ֞א תָּמָ֣ר אֲחֹתִ֗י וּתְלַבֵּ֤ב לְעֵינַי֙ שְׁתֵּ֣י לְבִב֔וֹת וְאֶבְרֶ֖ה מִיָּדָֽהּ׃ (כו) וַיִּשְׁלַ֥ח דָּוִ֛ד אֶל־ תָּמָ֖ר הַבַּ֣יְתָה לֵאמֹ֑ר לְכִ֣י נָ֗א בֵּ֚ית אַמְנ֣וֹן אָחִ֔יךְ וַעֲשִׂי־ ל֖וֹ הַבִּרְיָֽה׃ (כז) וַתֵּ֣לֶךְ תָּמָ֗ר בֵּ֛ית אַמְנ֥וֹן אָחִ֖יהָ וְה֣וּא שֹׁכֵ֑ב וַתִּקַּ֨ח אֶת־ הַבָּצֵ֤ק (ותלוש) וַתָּ֙לָשׁ֙ וַתְּלַבֵּ֣ב לְעֵינָ֔יו וַתְּבַשֵּׁ֖ל אֶת־ הַלְּבִבֽוֹת׃ (כח) וַתִּקַּ֤ח אֶת־ הַמַּשְׂרֵת֙ וַתִּצֹ֣ק לְפָנָ֔יו וַיְמָאֵ֖ן לֶאֱכ֑וֹל וַיֹּ֣אמֶר אַמְנ֗וֹן הוֹצִ֤יאוּ כָל־ אִישׁ֙ מֵֽעָלַ֔י וַיֵּצְא֥וּ כָל־ אִ֖ישׁ מֵעָלָֽיו׃ (כט) וַיֹּ֨אמֶר אַמְנ֜וֹן אֶל־ תָּמָ֗ר הָבִ֤יאִי הַבִּרְיָה֙ הַחֶ֔דֶר וְאֶבְרֶ֖ה מִיָּדֵ֑ךְ וַתִּקַּ֣ח תָּמָ֗ר אֶת־ הַלְּבִבוֹת֙ אֲשֶׁ֣ר עָשָׂ֔תָה וַתָּבֵ֛א לְאַמְנ֥וֹן אָחִ֖יהָ הֶחָֽדְרָה׃ (ל) וַתַּגֵּ֥שׁ אֵלָ֖יו לֶֽאֱכֹ֑ל וַיַּֽחֲזֶק־ בָּהּ֙ וַיֹּ֣אמֶר לָ֔הּ בּ֛וֹאִי שִׁכְבִ֥י עִמִּ֖י אֲחוֹתִֽי׃ (לא) וַתֹּ֣אמֶר ל֗וֹ אַל־ אָחִי֙ אַל־ תְּעַנֵּ֔נִי כִּ֛י לֹא־ יֵֽעָשֶׂ֥ה כֵ֖ן בְּיִשְׂרָאֵ֑ל אַֽל־ תַּעֲשֵׂ֖ה אֶת־ הַנְּבָלָ֥ה הַזֹּֽאת׃ (לב) וַאֲנִ֗י אָ֤נָה אוֹלִיךְ֙ אֶת־ חֶרְפָּתִ֔י וְאַתָּ֗ה תִּהְיֶ֛ה כְּאַחַ֥ד הַנְּבָלִ֖ים בְּיִשְׂרָאֵ֑ל וְעַתָּה֙ דַּבֶּר־ נָ֣א אֶל־ הַמֶּ֔לֶךְ כִּ֛י לֹ֥א יִמְנָעֵ֖נִי מִמֶּֽךָּ׃ (לג) וְלֹ֥א אָבָ֖ה לִשְׁמֹ֣עַ בְּקוֹלָ֑הּ וַיֶּחֱזַ֤ק מִמֶּ֙נָּה֙ וַיְעַנֶּ֔הָ וַיִּשְׁכַּ֖ב אֹתָֽהּ׃ (לד) וַיִּשְׂנָאֶ֣הָ אַמְנ֗וֹן שִׂנְאָה֙ גְּדוֹלָ֣ה מְאֹ֔ד כִּ֣י גְדוֹלָ֗ה הַשִּׂנְאָה֙ אֲשֶׁ֣ר שְׂנֵאָ֔הּ מֵאַהֲבָ֖ה אֲשֶׁ֣ר אֲהֵבָ֑הּ וַֽיֹּאמֶר־ לָ֥הּ אַמְנ֖וֹן ק֥וּמִי לֵֽכִי׃ (לה) וַתֹּ֣אמֶר ל֗וֹ אַל־ אוֹדֹ֞ת הָרָעָ֤ה הַגְּדוֹלָה֙ הַזֹּ֔את מֵאַחֶ֛רֶת אֲשֶׁר־ עָשִׂ֥יתָ עִמִּ֖י לְשַׁלְּחֵ֑נִי וְלֹ֥א אָבָ֖ה לִשְׁמֹ֥עַֽ לָֽהּ׃ (לו) וַיִּקְרָ֗א אֶֽת־ נַעֲרוֹ֙ מְשָׁ֣רְת֔וֹ וַיֹּ֕אמֶר שִׁלְחוּ־ נָ֥א אֶת־ זֹ֛את מֵעָלַ֖י הַח֑וּצָה וּנְעֹ֥ל הַדֶּ֖לֶת אַחֲרֶֽיהָ׃ (לז) וְעָלֶ֙יהָ֙ כְּתֹ֣נֶת פַּסִּ֔ים כִּי֩ כֵ֨ן תִּלְבַּ֧שְׁןָ בְנוֹת־ הַמֶּ֛לֶךְ הַבְּתוּלֹ֖ת מְעִילִ֑ים וַיֹּצֵ֨א אוֹתָ֤הּ מְשָֽׁרְתוֹ֙ הַח֔וּץ וְנָעַ֥ל הַדֶּ֖לֶת אַחֲרֶֽיהָ׃ (לח) וַתִּקַּ֨ח תָּמָ֥ר אֵ֙פֶר֙ עַל־ רֹאשָׁ֔הּ וּכְתֹ֧נֶת הַפַּסִּ֛ים אֲשֶׁ֥ר עָלֶ֖יהָ קָרָ֑עָה וַתָּ֤שֶׂם יָדָהּ֙ עַל־ רֹאשָׁ֔הּ וַתֵּ֥לֶךְ הָל֖וֹךְ וְזָעָֽקָה׃ (לט) וַיֹּ֨אמֶר אֵלֶ֜יהָ אַבְשָׁל֣וֹם אָחִ֗יהָ הַאֲמִינ֣וֹן אָחִיךְ֮ הָיָ֣ה עִמָּךְ֒ וְעַתָּ֞ה אֲחוֹתִ֤י הַחֲרִ֙ישִׁי֙ אָחִ֣יךְ ה֔וּא אַל־ תָּשִׁ֥יתִי אֶת־ לִבֵּ֖ךְ לַדָּבָ֣ר הַזֶּ֑ה וַתֵּ֤שֶׁב תָּמָר֙ וְשֹׁ֣מֵמָ֔ה בֵּ֖ית אַבְשָׁל֥וֹם אָחִֽיהָ׃ (מ) וְהַמֶּ֣לֶךְ דָּוִ֔ד שָׁמַ֕ע אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיִּ֥חַר ל֖וֹ מְאֹֽד׃ (מא) וְלֹֽא־ דִבֶּ֧ר אַבְשָׁל֛וֹם עִם־ אַמְנ֖וֹן לְמֵרָ֣ע וְעַד־ ט֑וֹב כִּֽי־ שָׂנֵ֤א אַבְשָׁלוֹם֙ אֶת־ אַמְנ֔וֹן עַל־ דְּבַר֙ אֲשֶׁ֣ר עִנָּ֔ה אֵ֖ת תָּמָ֥ר אֲחֹתֽוֹ׃ (מב) וַֽיְהִי֙ לִשְׁנָתַ֣יִם יָמִ֔ים וַיִּהְי֤וּ גֹֽזְזִים֙ לְאַבְשָׁל֔וֹם בְּבַ֥עַל חָצ֖וֹר אֲשֶׁ֣ר עִם־ אֶפְרָ֑יִם וַיִּקְרָ֥א אַבְשָׁל֖וֹם לְכָל־ בְּנֵ֥י הַמֶּֽלֶךְ׃ (מג) וַיָּבֹ֤א אַבְשָׁלוֹם֙ אֶל־ הַמֶּ֔לֶךְ וַיֹּ֕אמֶר הִנֵּה־ נָ֥א גֹזְזִ֖ים לְעַבְדֶּ֑ךָ יֵֽלֶךְ־ נָ֥א הַמֶּ֛לֶךְ וַעֲבָדָ֖יו עִם־ עַבְדֶּֽךָ׃
