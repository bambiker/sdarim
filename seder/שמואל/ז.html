שמואל סדר ז

פתיחה באתר · ספר שמואל כולו ·

(א) כִּ֠י לֹֽא־ יִטֹּ֤שׁ יְהוָה֙ אֶת־ עַמּ֔וֹ בַּעֲב֖וּר שְׁמ֣וֹ הַגָּד֑וֹל כִּ֚י הוֹאִ֣יל יְהוָ֔ה לַעֲשׂ֥וֹת אֶתְכֶ֛ם ל֖וֹ לְעָֽם׃ (ב) גַּ֣ם אָנֹכִ֗י חָלִ֤ילָה לִּי֙ מֵחֲטֹ֣א לַֽיהוָ֔ה מֵחֲדֹ֖ל לְהִתְפַּלֵּ֣ל בַּעַדְכֶ֑ם וְהוֹרֵיתִ֣י אֶתְכֶ֔ם בְּדֶ֥רֶךְ הַטּוֹבָ֖ה וְהַיְשָׁרָֽה׃ (ג) אַ֣ךְ ׀ יְר֣אוּ אֶת־ יְהוָ֗ה וַעֲבַדְתֶּ֥ם אֹת֛וֹ בֶּאֱמֶ֖ת בְּכָל־ לְבַבְכֶ֑ם כִּ֣י רְא֔וּ אֵ֥ת אֲשֶׁר־ הִגְדִּ֖ל עִמָּכֶֽם׃ (ד) וְאִם־ הָרֵ֖עַ תָּרֵ֑עוּ גַּם־ אַתֶּ֥ם גַּֽם־ מַלְכְּכֶ֖ם תִּסָּפֽוּ׃ (ה) בֶּן־ שָׁנָ֖ה שָׁא֣וּל בְּמָלְכ֑וֹ וּשְׁתֵּ֣י שָׁנִ֔ים מָלַ֖ךְ עַל־ יִשְׂרָאֵֽל׃ (ו) וַיִּבְחַר־ ל֨וֹ שָׁא֜וּל שְׁלֹ֣שֶׁת אֲלָפִים֮ מִיִּשְׂרָאֵל֒ וַיִּהְי֨וּ עִם־ שָׁא֜וּל אַלְפַּ֗יִם בְּמִכְמָשׂ֙ וּבְהַ֣ר בֵּֽית־ אֵ֔ל וְאֶ֗לֶף הָיוּ֙ עִם־ י֣וֹנָתָ֔ן בְּגִבְעַ֖ת בִּנְיָמִ֑ין וְיֶ֣תֶר הָעָ֔ם שִׁלַּ֖ח אִ֥ישׁ לְאֹהָלָֽיו׃ (ז) וַיַּ֣ךְ יוֹנָתָ֗ן אֵ֣ת נְצִ֤יב פְּלִשְׁתִּים֙ אֲשֶׁ֣ר בְּגֶ֔בַע וַֽיִּשְׁמְע֖וּ פְּלִשְׁתִּ֑ים וְשָׁאוּל֩ תָּקַ֨ע בַּשּׁוֹפָ֤ר בְּכָל־ הָאָ֙רֶץ֙ לֵאמֹ֔ר יִשְׁמְע֖וּ הָעִבְרִֽים׃ (ח) וְכָל־ יִשְׂרָאֵ֞ל שָׁמְע֣וּ לֵאמֹ֗ר הִכָּ֤ה שָׁאוּל֙ אֶת־ נְצִ֣יב פְּלִשְׁתִּ֔ים וְגַם־ נִבְאַשׁ יִשְׂרָאֵ֖ל בַּפְּלִשְׁתִּ֑ים וַיִּצָּעֲק֥וּ הָעָ֛ם אַחֲרֵ֥י שָׁא֖וּל הַגִּלְגָּֽל׃ (ט) וּפְלִשְׁתִּ֞ים נֶאֶסְפ֣וּ ׀ לְהִלָּחֵ֣ם עִם־ יִשְׂרָאֵ֗ל שְׁלֹשִׁ֨ים אֶ֤לֶף רֶ֙כֶב֙ וְשֵׁ֤שֶׁת אֲלָפִים֙ פָּרָשִׁ֔ים וְעָ֕ם כַּח֛וֹל אֲשֶׁ֥ר עַל־ שְׂפַֽת־ הַיָּ֖ם לָרֹ֑ב וַֽיַּעֲלוּ֙ וַיַּחֲנ֣וּ בְמִכְמָ֔שׂ קִדְמַ֖ת בֵּ֥ית אָֽוֶן׃ (י) וְאִ֨ישׁ יִשְׂרָאֵ֤ל רָאוּ֙ כִּ֣י צַר־ ל֔וֹ כִּ֥י נִגַּ֖שׂ הָעָ֑ם וַיִּֽתְחַבְּא֣וּ הָעָ֗ם בַּמְּעָר֤וֹת וּבַֽחֲוָחִים֙ וּבַסְּלָעִ֔ים וּבַצְּרִחִ֖ים וּבַבֹּרֽוֹת׃ (יא) וְעִבְרִ֗ים עָֽבְרוּ֙ אֶת־ הַיַּרְדֵּ֔ן אֶ֥רֶץ גָּ֖ד וְגִלְעָ֑ד וְשָׁאוּל֙ עוֹדֶ֣נּוּ בַגִּלְגָּ֔ל וְכָל־ הָעָ֖ם חָרְד֥וּ אַחֲרָֽיו׃ (יב) (וייחל) וַיּ֣וֹחֶל ׀ שִׁבְעַ֣ת יָמִ֗ים לַמּוֹעֵד֙ אֲשֶׁ֣ר שְׁמוּאֵ֔ל וְלֹא־ בָ֥א שְׁמוּאֵ֖ל הַגִּלְגָּ֑ל וַיָּ֥פֶץ הָעָ֖ם מֵעָלָֽיו׃ (יג) וַיֹּ֣אמֶר שָׁא֔וּל הַגִּ֣שׁוּ אֵלַ֔י הָעֹלָ֖ה וְהַשְּׁלָמִ֑ים וַיַּ֖עַל הָעֹלָֽה׃ (יד) וַיְהִ֗י כְּכַלֹּתוֹ֙ לְהַעֲל֣וֹת הָעֹלָ֔ה וְהִנֵּ֥ה שְׁמוּאֵ֖ל בָּ֑א וַיֵּצֵ֥א שָׁא֛וּל לִקְרָאת֖וֹ לְבָרֲכֽוֹ׃ (טו) וַיֹּ֥אמֶר שְׁמוּאֵ֖ל מֶ֣ה עָשִׂ֑יתָ וַיֹּ֣אמֶר שָׁא֡וּל כִּֽי־ רָאִיתִי֩ כִֽי־ נָפַ֨ץ הָעָ֜ם מֵעָלַ֗י וְאַתָּה֙ לֹא־ בָ֙אתָ֙ לְמוֹעֵ֣ד הַיָּמִ֔ים וּפְלִשְׁתִּ֖ים נֶאֱסָפִ֥ים מִכְמָֽשׂ׃ (טז) וָאֹמַ֗ר עַ֠תָּה יֵרְד֨וּ פְלִשְׁתִּ֤ים אֵלַי֙ הַגִּלְגָּ֔ל וּפְנֵ֥י יְהוָ֖ה לֹ֣א חִלִּ֑יתִי וָֽאֶתְאַפַּ֔ק וָאַעֲלֶ֖ה הָעֹלָֽה׃ (יז) וַיֹּ֧אמֶר שְׁמוּאֵ֛ל אֶל־ שָׁא֖וּל נִסְכָּ֑לְתָּ לֹ֣א שָׁמַ֗רְתָּ אֶת־ מִצְוַ֞ת יְהוָ֤ה אֱלֹהֶ֙יךָ֙ אֲשֶׁ֣ר צִוָּ֔ךְ כִּ֣י עַתָּ֗ה הֵכִ֨ין יְהוָ֧ה אֶת־ מַֽמְלַכְתְּךָ֛ אֶל־ יִשְׂרָאֵ֖ל עַד־ עוֹלָֽם׃ (יח) וְעַתָּ֖ה מַמְלַכְתְּךָ֣ לֹא־ תָק֑וּם בִּקֵּשׁ֩ יְהוָ֨ה ל֜וֹ אִ֣ישׁ כִּלְבָב֗וֹ וַיְצַוֵּ֨הוּ יְהוָ֤ה לְנָגִיד֙ עַל־ עַמּ֔וֹ כִּ֚י לֹ֣א שָׁמַ֔רְתָּ אֵ֥ת אֲשֶֽׁר־ צִוְּךָ֖ יְהוָֽה׃ (יט) וַיָּ֣קָם שְׁמוּאֵ֗ל וַיַּ֛עַל מִן־ הַגִּלְגָּ֖ל גִּבְעַ֣ת בִּנְיָמִ֑ן וַיִּפְקֹ֣ד שָׁא֗וּל אֶת־ הָעָם֙ הַנִּמְצְאִ֣ים עִמּ֔וֹ כְּשֵׁ֥שׁ מֵא֖וֹת אִֽישׁ׃ (כ) וְשָׁא֞וּל וְיוֹנָתָ֣ן בְּנ֗וֹ וְהָעָם֙ הַנִּמְצָ֣א עִמָּ֔ם יֹשְׁבִ֖ים בְּגֶ֣בַע בִּנְיָמִ֑ן וּפְלִשְׁתִּ֖ים חָנ֥וּ בְמִכְמָֽשׂ׃ (כא) וַיֵּצֵ֧א הַמַּשְׁחִ֛ית מִמַּחֲנֵ֥ה פְלִשְׁתִּ֖ים שְׁלֹשָׁ֣ה רָאשִׁ֑ים הָרֹ֨אשׁ אֶחָ֥ד יִפְנֶ֛ה אֶל־ דֶּ֥רֶךְ עָפְרָ֖ה אֶל־ אֶ֥רֶץ שׁוּעָֽל׃ (כב) וְהָרֹ֤אשׁ אֶחָד֙ יִפְנֶ֔ה דֶּ֖רֶךְ בֵּ֣ית חֹר֑וֹן וְהָרֹ֨אשׁ אֶחָ֤ד יִפְנֶה֙ דֶּ֣רֶךְ הַגְּב֔וּל הַנִּשְׁקָ֛ף עַל־ גֵּ֥י הַצְּבֹעִ֖ים הַמִּדְבָּֽרָה׃ (כג) וְחָרָשׁ֙ לֹ֣א יִמָּצֵ֔א בְּכֹ֖ל אֶ֣רֶץ יִשְׂרָאֵ֑ל כִּֽי־ (אמר) אָמְר֣וּ פְלִשְׁתִּ֔ים פֶּ֚ן יַעֲשׂ֣וּ הָעִבְרִ֔ים חֶ֖רֶב א֥וֹ חֲנִֽית׃ (כד) וַיֵּרְד֥וּ כָל־ יִשְׂרָאֵ֖ל הַפְּלִשְׁתִּ֑ים לִ֠לְטוֹשׁ אִ֣ישׁ אֶת־ מַחֲרַשְׁתּ֤וֹ וְאֶת־ אֵתוֹ֙ וְאֶת־ קַרְדֻּמּ֔וֹ וְאֵ֖ת מַחֲרֵשָׁתֽוֹ׃ (כה) וְֽהָיְתָ֞ה הַפְּצִ֣ירָה פִ֗ים לַמַּֽחֲרֵשֹׁת֙ וְלָ֣אֵתִ֔ים וְלִשְׁלֹ֥שׁ קִלְּשׁ֖וֹן וּלְהַקַּרְדֻּמִּ֑ים וּלְהַצִּ֖יב הַדָּרְבָֽן׃ (כו) וְהָיָה֙ בְּי֣וֹם מִלְחֶ֔מֶת וְלֹ֨א נִמְצָ֜א חֶ֤רֶב וַחֲנִית֙ בְּיַ֣ד כָּל־ הָעָ֔ם אֲשֶׁ֥ר אֶת־ שָׁא֖וּל וְאֶת־ יוֹנָתָ֑ן וַתִּמָּצֵ֣א לְשָׁא֔וּל וּלְיוֹנָתָ֖ן בְּנֽוֹ׃ (כז) וַיֵּצֵא֙ מַצַּ֣ב פְּלִשְׁתִּ֔ים אֶֽל־ מַעֲבַ֖ר מִכְמָֽשׂ׃ (כח) וַיְהִ֣י הַיּ֗וֹם וַיֹּ֨אמֶר יוֹנָתָ֤ן בֶּן־ שָׁאוּל֙ אֶל־ הַנַּ֙עַר֙ נֹשֵׂ֣א כֵלָ֔יו לְכָ֗ה וְנַעְבְּרָה֙ אֶל־ מַצַּ֣ב פְּלִשְׁתִּ֔ים אֲשֶׁ֖ר מֵעֵ֣בֶר הַלָּ֑ז וּלְאָבִ֖יו לֹ֥א הִגִּֽיד׃ (כט) וְשָׁא֗וּל יוֹשֵׁב֙ בִּקְצֵ֣ה הַגִּבְעָ֔ה תַּ֥חַת הָרִמּ֖וֹן אֲשֶׁ֣ר בְּמִגְר֑וֹן וְהָעָם֙ אֲשֶׁ֣ר עִמּ֔וֹ כְּשֵׁ֥שׁ מֵא֖וֹת אִֽישׁ׃ (ל) וַאֲחִיָּ֣ה בֶן־ אֲחִט֡וּב אֲחִ֡י אִיכָב֣וֹד ׀ בֶּן־ פִּינְחָ֨ס בֶּן־ עֵלִ֜י כֹּהֵ֧ן ׀ יְהוָ֛ה בְּשִׁל֖וֹ נֹשֵׂ֣א אֵפ֑וֹד וְהָעָם֙ לֹ֣א יָדַ֔ע כִּ֥י הָלַ֖ךְ יוֹנָתָֽן׃ (לא) וּבֵ֣ין הַֽמַּעְבְּר֗וֹת אֲשֶׁ֨ר בִּקֵּ֤שׁ יֽוֹנָתָן֙ לַֽעֲבֹר֙ עַל־ מַצַּ֣ב פְּלִשְׁתִּ֔ים שֵׁן־ הַסֶּ֤לַע מֵהָעֵ֙בֶר֙ מִזֶּ֔ה וְשֵׁן־ הַסֶּ֥לַע מֵהָעֵ֖בֶר מִזֶּ֑ה וְשֵׁ֤ם הָֽאֶחָד֙ בּוֹצֵ֔ץ וְשֵׁ֥ם הָאֶחָ֖ד סֶֽנֶּה׃ (לב) הַשֵּׁ֧ן הָאֶחָ֛ד מָצ֥וּק מִצָּפ֖וֹן מ֣וּל מִכְמָ֑שׂ וְהָאֶחָ֥ד מִנֶּ֖גֶב מ֥וּל גָּֽבַע׃ (לג) וַיֹּ֨אמֶר יְהוֹנָתָ֜ן אֶל־ הַנַּ֣עַר ׀ נֹשֵׂ֣א כֵלָ֗יו לְכָה֙ וְנַעְבְּרָ֗ה אֶל־ מַצַּב֙ הָעֲרֵלִ֣ים הָאֵ֔לֶּה אוּלַ֛י יַעֲשֶׂ֥ה יְהוָ֖ה לָ֑נוּ כִּ֣י אֵ֤ין לַֽיהוָה֙ מַעְצ֔וֹר לְהוֹשִׁ֥יעַ בְּרַ֖ב א֥וֹ בִמְעָֽט׃ (לד) וַיֹּ֤אמֶר לוֹ֙ נֹשֵׂ֣א כֵלָ֔יו עֲשֵׂ֖ה כָּל־ אֲשֶׁ֣ר בִּלְבָבֶ֑ךָ נְטֵ֣ה לָ֔ךְ הִנְנִ֥י עִמְּךָ֖ כִּלְבָבֶֽךָ׃ (לה) וַיֹּ֙אמֶר֙ יְה֣וֹנָתָ֔ן הִנֵּ֛ה אֲנַ֥חְנוּ עֹבְרִ֖ים אֶל־ הָאֲנָשִׁ֑ים וְנִגְלִ֖ינוּ אֲלֵיהֶֽם׃ (לו) אִם־ כֹּ֤ה יֹֽאמְרוּ֙ אֵלֵ֔ינוּ דֹּ֕מּוּ עַד־ הַגִּיעֵ֖נוּ אֲלֵיכֶ֑ם וְעָמַ֣דְנוּ תַחְתֵּ֔ינוּ וְלֹ֥א נַעֲלֶ֖ה אֲלֵיהֶֽם׃ (לז) וְאִם־ כֹּ֨ה יֹאמְר֜וּ עֲל֤וּ עָלֵ֙ינוּ֙ וְעָלִ֔ינוּ כִּֽי־ נְתָנָ֥ם יְהוָ֖ה בְּיָדֵ֑נוּ וְזֶה־ לָּ֖נוּ הָאֽוֹת׃ (לח) וַיִּגָּל֣וּ שְׁנֵיהֶ֔ם אֶל־ מַצַּ֖ב פְּלִשְׁתִּ֑ים וַיֹּאמְר֣וּ פְלִשְׁתִּ֔ים הִנֵּ֤ה עִבְרִים֙ יֹֽצְאִ֔ים מִן־ הַחֹרִ֖ים אֲשֶׁ֥ר הִתְחַבְּאוּ־ שָֽׁם׃ (לט) וַיַּעֲנוּ֩ אַנְשֵׁ֨י הַמַּצָּבָ֜ה אֶת־ יוֹנָתָ֣ן ׀ וְאֶת־ נֹשֵׂ֣א כֵלָ֗יו וַיֹּֽאמְרוּ֙ עֲל֣וּ אֵלֵ֔ינוּ וְנוֹדִ֥יעָה אֶתְכֶ֖ם דָּבָ֑ר וַיֹּ֨אמֶר יוֹנָתָ֜ן אֶל־ נֹשֵׂ֤א כֵלָיו֙ עֲלֵ֣ה אַחֲרַ֔י כִּֽי־ נְתָנָ֥ם יְהוָ֖ה בְּיַ֥ד יִשְׂרָאֵֽל׃ (מ) וַיַּ֣עַל יוֹנָתָ֗ן עַל־ יָדָיו֙ וְעַל־ רַגְלָ֔יו וְנֹשֵׂ֥א כֵלָ֖יו אַחֲרָ֑יו וַֽיִּפְּלוּ֙ לִפְנֵ֣י יוֹנָתָ֔ן וְנֹשֵׂ֥א כֵלָ֖יו מְמוֹתֵ֥ת אַחֲרָֽיו׃ (מא) וַתְּהִ֞י הַמַּכָּ֣ה הָרִאשֹׁנָ֗ה אֲשֶׁ֨ר הִכָּ֧ה יוֹנָתָ֛ן וְנֹשֵׂ֥א כֵלָ֖יו כְּעֶשְׂרִ֣ים אִ֑ישׁ כְּבַחֲצִ֥י מַעֲנָ֖ה צֶ֥מֶד שָׂדֶֽה׃ (מב) וַתְּהִי֩ חֲרָדָ֨ה בַמַּחֲנֶ֤ה בַשָּׂדֶה֙ וּבְכָל־ הָעָ֔ם הַמַּצָּב֙ וְהַמַּשְׁחִ֔ית חָרְד֖וּ גַּם־ הֵ֑מָּה וַתִּרְגַּ֣ז הָאָ֔רֶץ וַתְּהִ֖י לְחֶרְדַּ֥ת אֱלֹהִֽים׃ (מג) וַיִּרְא֤וּ הַצֹּפִים֙ לְשָׁא֔וּל בְּגִבְעַ֖ת בִּנְיָמִ֑ן וְהִנֵּ֧ה הֶהָמ֛וֹן נָמ֖וֹג וַיֵּ֥לֶךְ וַהֲלֹֽם׃ (מד) וַיֹּ֣אמֶר שָׁא֗וּל לָעָם֙ אֲשֶׁ֣ר אִתּ֔וֹ פִּקְדוּ־ נָ֣א וּרְא֔וּ מִ֖י הָלַ֣ךְ מֵעִמָּ֑נוּ וַֽיִּפְקְד֔וּ וְהִנֵּ֛ה אֵ֥ין יוֹנָתָ֖ן וְנֹשֵׂ֥א כֵלָֽיו׃ (מה) וַיֹּ֤אמֶר שָׁאוּל֙ לַֽאֲחִיָּ֔ה הַגִּ֖ישָׁה אֲר֣וֹן הָאֱלֹהִ֑ים כִּֽי־ הָיָ֞ה אֲר֧וֹן הָאֱלֹהִ֛ים בַּיּ֥וֹם הַה֖וּא וּבְנֵ֥י יִשְׂרָאֵֽל׃ (מו) וַיְהִ֗י עַ֣ד דִּבֶּ֤ר שָׁאוּל֙ אֶל־ הַכֹּהֵ֔ן וְהֶהָמ֗וֹן אֲשֶׁר֙ בְּמַחֲנֵ֣ה פְלִשְׁתִּ֔ים וַיֵּ֥לֶךְ הָל֖וֹךְ וָרָ֑ב וַיֹּ֧אמֶר שָׁא֛וּל אֶל־ הַכֹּהֵ֖ן אֱסֹ֥ף יָדֶֽךָ׃ (מז) וַיִּזָּעֵ֣ק שָׁא֗וּל וְכָל־ הָעָם֙ אֲשֶׁ֣ר אִתּ֔וֹ וַיָּבֹ֖אוּ עַד־ הַמִּלְחָמָ֑ה וְהִנֵּ֨ה הָיְתָ֜ה חֶ֤רֶב אִישׁ֙ בְּרֵעֵ֔הוּ מְהוּמָ֖ה גְּדוֹלָ֥ה מְאֹֽד׃ (מח) וְהָעִבְרִ֗ים הָי֤וּ לַפְּלִשְׁתִּים֙ כְּאֶתְמ֣וֹל שִׁלְשׁ֔וֹם אֲשֶׁ֨ר עָל֥וּ עִמָּ֛ם בַּֽמַּחֲנֶ֖ה סָבִ֑יב וְגַם־ הֵ֗מָּה לִֽהְיוֹת֙ עִם־ יִשְׂרָאֵ֔ל אֲשֶׁ֥ר עִם־ שָׁא֖וּל וְיוֹנָתָֽן׃ (מט) וְכֹל֩ אִ֨ישׁ יִשְׂרָאֵ֜ל הַמִּֽתְחַבְּאִ֤ים בְּהַר־ אֶפְרַ֙יִם֙ שָֽׁמְע֔וּ כִּֽי־ נָ֖סוּ פְּלִשְׁתִּ֑ים וַֽיַּדְבְּק֥וּ גַם־ הֵ֛מָּה אַחֲרֵיהֶ֖ם בַּמִּלְחָמָֽה׃
