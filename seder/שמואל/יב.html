(א) וַיְהִ֥י דָוִ֛ד לְכָל־ דָּרְכָ֖ו מַשְׂכִּ֑יל וַֽיהוָ֖ה עִמּֽוֹ׃ (ב) וַיַּ֣רְא שָׁא֔וּל אֲשֶׁר־ ה֖וּא מַשְׂכִּ֣יל מְאֹ֑ד וַיָּ֖גָר מִפָּנָֽיו׃ (ג) וְכָל־ יִשְׂרָאֵל֙ וִיהוּדָ֔ה אֹהֵ֖ב אֶת־ דָּוִ֑ד כִּֽי־ ה֛וּא יוֹצֵ֥א וָבָ֖א לִפְנֵיהֶֽם׃ (ד) וַיֹּ֨אמֶר שָׁא֜וּל אֶל־ דָּוִ֗ד הִנֵּה֩ בִתִּ֨י הַגְּדוֹלָ֤ה מֵרַב֙ אֹתָהּ֙ אֶתֶּן־ לְךָ֣ לְאִשָּׁ֔ה אַ֚ךְ הֱיֵה־ לִּ֣י לְבֶן־ חַ֔יִל וְהִלָּחֵ֖ם מִלְחֲמ֣וֹת יְהוָ֑ה וְשָׁא֣וּל אָמַ֗ר אַל־ תְּהִ֤י יָדִי֙ בּ֔וֹ וּתְהִי־ ב֖וֹ יַד־ פְּלִשְׁתִּֽים׃ (ה) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ שָׁא֗וּל מִ֤י אָֽנֹכִי֙ וּמִ֣י חַיַּ֔י מִשְׁפַּ֥חַת אָבִ֖י בְּיִשְׂרָאֵ֑ל כִּֽי־ אֶהְיֶ֥ה חָתָ֖ן לַמֶּֽלֶךְ׃ (ו) וַיְהִ֗י בְּעֵ֥ת תֵּ֛ת אֶת־ מֵרַ֥ב בַּת־ שָׁא֖וּל לְדָוִ֑ד וְהִ֧יא נִתְּנָ֛ה לְעַדְרִיאֵ֥ל הַמְּחֹלָתִ֖י לְאִשָּֽׁה׃ (ז) וַתֶּאֱהַ֛ב מִיכַ֥ל בַּת־ שָׁא֖וּל אֶת־ דָּוִ֑ד וַיַּגִּ֣דוּ לְשָׁא֔וּל וַיִּשַׁ֥ר הַדָּבָ֖ר בְּעֵינָֽיו׃ (ח) וַיֹּ֨אמֶר שָׁא֜וּל אֶתְּנֶ֤נָּה לּוֹ֙ וּתְהִי־ ל֣וֹ לְמוֹקֵ֔שׁ וּתְהִי־ ב֖וֹ יַד־ פְּלִשְׁתִּ֑ים וַיֹּ֤אמֶר שָׁאוּל֙ אֶל־ דָּוִ֔ד בִּשְׁתַּ֛יִם תִּתְחַתֵּ֥ן בִּ֖י הַיּֽוֹם׃ (ט) וַיְצַ֨ו שָׁא֜וּל אֶת־ עֲבָדָ֗ו דַּבְּר֨וּ אֶל־ דָּוִ֤ד בַּלָּט֙ לֵאמֹ֔ר הִנֵּ֨ה חָפֵ֤ץ בְּךָ֙ הַמֶּ֔לֶךְ וְכָל־ עֲבָדָ֖יו אֲהֵב֑וּךָ וְעַתָּ֖ה הִתְחַתֵּ֥ן בַּמֶּֽלֶךְ׃ (י) וַֽיְדַבְּר֞וּ עַבְדֵ֤י שָׁאוּל֙ בְּאָזְנֵ֣י דָוִ֔ד אֶת־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיֹּ֣אמֶר דָּוִ֗ד הַֽנְקַלָּ֤ה בְעֵֽינֵיכֶם֙ הִתְחַתֵּ֣ן בַּמֶּ֔לֶךְ וְאָנֹכִ֖י אִֽישׁ־ רָ֥שׁ וְנִקְלֶֽה׃ (יא) וַיַּגִּ֜דוּ עַבְדֵ֥י שָׁא֛וּל ל֖וֹ לֵאמֹ֑ר כַּדְּבָרִ֥ים הָאֵ֖לֶּה דִּבֶּ֥ר דָּוִֽד׃ (יב) וַיֹּ֨אמֶר שָׁא֜וּל כֹּֽה־ תֹאמְר֣וּ לְדָוִ֗ד אֵֽין־ חֵ֤פֶץ לַמֶּ֙לֶךְ֙ בְּמֹ֔הַר כִּ֗י בְּמֵאָה֙ עָרְל֣וֹת פְּלִשְׁתִּ֔ים לְהִנָּקֵ֖ם בְּאֹיְבֵ֣י הַמֶּ֑לֶךְ וְשָׁא֣וּל חָשַׁ֔ב לְהַפִּ֥יל אֶת־ דָּוִ֖ד בְּיַד־ פְּלִשְׁתִּֽים׃ (יג) וַיַּגִּ֨דוּ עֲבָדָ֤יו לְדָוִד֙ אֶת־ הַדְּבָרִ֣ים הָאֵ֔לֶּה וַיִּשַׁ֤ר הַדָּבָר֙ בְּעֵינֵ֣י דָוִ֔ד לְהִתְחַתֵּ֖ן בַּמֶּ֑לֶךְ וְלֹ֥א מָלְא֖וּ הַיָּמִֽים׃ (יד) וַיָּ֨קָם דָּוִ֜ד וַיֵּ֣לֶךְ ׀ ה֣וּא וַאֲנָשָׁ֗יו וַיַּ֣ךְ בַּפְּלִשְׁתִּים֮ מָאתַ֣יִם אִישׁ֒ וַיָּבֵ֤א דָוִד֙ אֶת־ עָרְלֹ֣תֵיהֶ֔ם וַיְמַלְא֣וּם לַמֶּ֔לֶךְ לְהִתְחַתֵּ֖ן בַּמֶּ֑לֶךְ וַיִּתֶּן־ ל֥וֹ שָׁא֛וּל אֶת־ מִיכַ֥ל בִּתּ֖וֹ לְאִשָּֽׁה׃ (טו) וַיַּ֤רְא שָׁאוּל֙ וַיֵּ֔דַע כִּ֥י יְהוָ֖ה עִם־ דָּוִ֑ד וּמִיכַ֥ל בַּת־ שָׁא֖וּל אֲהֵבַֽתְהוּ׃ (טז) וַיֹּ֣אסֶף שָׁא֗וּל לֵרֹ֛א מִפְּנֵ֥י דָוִ֖ד ע֑וֹד וַיְהִ֥י שָׁא֛וּל אֹיֵ֥ב אֶת־ דָּוִ֖ד כָּל־ הַיָּמִֽים׃ (יז) וַיֵּצְא֖וּ שָׂרֵ֣י פְלִשְׁתִּ֑ים וַיְהִ֣י ׀ מִדֵּ֣י צֵאתָ֗ם שָׂכַ֤ל דָּוִד֙ מִכֹּל֙ עַבְדֵ֣י שָׁא֔וּל וַיִּיקַ֥ר שְׁמ֖וֹ מְאֹֽד׃ (יח) וַיְדַבֵּ֣ר שָׁא֗וּל אֶל־ יוֹנָתָ֤ן בְּנוֹ֙ וְאֶל־ כָּל־ עֲבָדָ֔יו לְהָמִ֖ית אֶת־ דָּוִ֑ד וִיהֽוֹנָתָן֙ בֶּן־ שָׁא֔וּל חָפֵ֥ץ בְּדָוִ֖ד מְאֹֽד׃ (יט) וַיַּגֵּ֤ד יְהוֹנָתָן֙ לְדָוִ֣ד לֵאמֹ֔ר מְבַקֵּ֛שׁ שָׁא֥וּל אָבִ֖י לַהֲמִיתֶ֑ךָ וְעַתָּה֙ הִשָּֽׁמֶר־ נָ֣א בַבֹּ֔קֶר וְיָשַׁבְתָּ֥ בַסֵּ֖תֶר וְנַחְבֵּֽאתָ׃ (כ) וַאֲנִ֨י אֵצֵ֜א וְעָמַדְתִּ֣י לְיַד־ אָבִ֗י בַּשָּׂדֶה֙ אֲשֶׁ֣ר אַתָּ֣ה שָׁ֔ם וַאֲנִ֕י אֲדַבֵּ֥ר בְּךָ֖ אֶל־ אָבִ֑י וְרָאִ֥יתִי מָ֖ה וְהִגַּ֥דְתִּי לָֽךְ׃ (כא) וַיְדַבֵּ֨ר יְהוֹנָתָ֤ן בְּדָוִד֙ ט֔וֹב אֶל־ שָׁא֖וּל אָבִ֑יו וַיֹּ֣אמֶר אֵ֠לָיו אַל־ יֶחֱטָ֨א הַמֶּ֜לֶךְ בְּעַבְדּ֣וֹ בְדָוִ֗ד כִּ֣י ל֤וֹא חָטָא֙ לָ֔ךְ וְכִ֥י מַעֲשָׂ֖יו טוֹב־ לְךָ֥ מְאֹֽד׃ (כב) וַיָּשֶׂם֩ אֶת־ נַפְשׁ֨וֹ בְכַפּ֜וֹ וַיַּ֣ךְ אֶת־ הַפְּלִשְׁתִּ֗י וַיַּ֨עַשׂ יְהוָ֜ה תְּשׁוּעָ֤ה גְדוֹלָה֙ לְכָל־ יִשְׂרָאֵ֔ל רָאִ֖יתָ וַתִּשְׂמָ֑ח וְלָ֤מָּה תֶֽחֱטָא֙ בְּדָ֣ם נָקִ֔י לְהָמִ֥ית אֶת־ דָּוִ֖ד חִנָּֽם׃ (כג) וַיִּשְׁמַ֥ע שָׁא֖וּל בְּק֣וֹל יְהוֹנָתָ֑ן וַיִּשָּׁבַ֣ע שָׁא֔וּל חַי־ יְהוָ֖ה אִם־ יוּמָֽת׃ (כד) וַיִּקְרָ֤א יְהוֹנָתָן֙ לְדָוִ֔ד וַיַּגֶּד־ לוֹ֙ יְה֣וֹנָתָ֔ן אֵ֥ת כָּל־ הַדְּבָרִ֖ים הָאֵ֑לֶּה וַיָּבֵ֨א יְהוֹנָתָ֤ן אֶת־ דָּוִד֙ אֶל־ שָׁא֔וּל וַיְהִ֥י לְפָנָ֖יו כְּאֶתְמ֥וֹל שִׁלְשֽׁוֹם׃ (כה) וַתּ֥וֹסֶף הַמִּלְחָמָ֖ה לִֽהְי֑וֹת וַיֵּצֵ֨א דָוִ֜ד וַיִּלָּ֣חֶם בַּפְּלִשְׁתִּ֗ים וַיַּ֤ךְ בָּהֶם֙ מַכָּ֣ה גְדוֹלָ֔ה וַיָּנֻ֖סוּ מִפָּנָֽיו׃ (כו) וַתְּהִי֩ ר֨וּחַ יְהוָ֤ה ׀ רָעָה֙ אֶל־ שָׁא֔וּל וְהוּא֙ בְּבֵית֣וֹ יוֹשֵׁ֔ב וַחֲנִית֖וֹ בְּיָד֑וֹ וְדָוִ֖ד מְנַגֵּ֥ן בְּיָֽד׃ (כז) וַיְבַקֵּ֨שׁ שָׁא֜וּל לְהַכּ֤וֹת בַּֽחֲנִית֙ בְּדָוִ֣ד וּבַקִּ֔יר וַיִּפְטַר֙ מִפְּנֵ֣י שָׁא֔וּל וַיַּ֥ךְ אֶֽת־ הַחֲנִ֖ית בַּקִּ֑יר וְדָוִ֛ד נָ֥ס וַיִּמָּלֵ֖ט בַּלַּ֥יְלָה הֽוּא׃ (כח) וַיִּשְׁלַח֩ שָׁא֨וּל מַלְאָכִ֜ים אֶל־ בֵּ֤ית דָּוִד֙ לְשָׁמְר֔וֹ וְלַהֲמִית֖וֹ בַּבֹּ֑קֶר וַתַּגֵּ֣ד לְדָוִ֗ד מִיכַ֤ל אִשְׁתּוֹ֙ לֵאמֹ֔ר אִם־ אֵ֨ינְךָ֜ מְמַלֵּ֤ט אֶֽת־ נַפְשְׁךָ֙ הַלַּ֔יְלָה מָחָ֖ר אַתָּ֥ה מוּמָֽת׃ (כט) וַתֹּ֧רֶד מִיכַ֛ל אֶת־ דָּוִ֖ד בְּעַ֣ד הַחַלּ֑וֹן וַיֵּ֥לֶךְ וַיִּבְרַ֖ח וַיִּמָּלֵֽט׃ (ל) וַתִּקַּ֨ח מִיכַ֜ל אֶת־ הַתְּרָפִ֗ים וַתָּ֙שֶׂם֙ אֶל־ הַמִּטָּ֔ה וְאֵת֙ כְּבִ֣יר הָֽעִזִּ֔ים שָׂ֖מָה מְרַֽאֲשֹׁתָ֑יו וַתְּכַ֖ס בַּבָּֽגֶד׃ (לא) וַיִּשְׁלַ֥ח שָׁא֛וּל מַלְאָכִ֖ים לָקַ֣חַת אֶת־ דָּוִ֑ד וַתֹּ֖אמֶר חֹלֶ֥ה הֽוּא׃ (לב) וַיִּשְׁלַ֤ח שָׁאוּל֙ אֶת־ הַמַּלְאָכִ֔ים לִרְא֥וֹת אֶת־ דָּוִ֖ד לֵאמֹ֑ר הַעֲל֨וּ אֹת֧וֹ בַמִּטָּ֛ה אֵלַ֖י לַהֲמִתֽוֹ׃ (לג) וַיָּבֹ֙אוּ֙ הַמַּלְאָכִ֔ים וְהִנֵּ֥ה הַתְּרָפִ֖ים אֶל־ הַמִּטָּ֑ה וּכְבִ֥יר הָעִזִּ֖ים מְרַאֲשֹׁתָֽיו׃ (לד) וַיֹּ֨אמֶר שָׁא֜וּל אֶל־ מִיכַ֗ל לָ֤מָּה כָּ֙כָה֙ רִמִּיתִ֔נִי וַתְּשַׁלְּחִ֥י אֶת־ אֹיְבִ֖י וַיִּמָּלֵ֑ט וַתֹּ֤אמֶר מִיכַל֙ אֶל־ שָׁא֔וּל הוּא־ אָמַ֥ר אֵלַ֛י שַׁלְּחִ֖נִי לָמָ֥ה אֲמִיתֵֽךְ׃ (לה) וְדָוִ֨ד בָּרַ֜ח וַיִּמָּלֵ֗ט וַיָּבֹ֤א אֶל־ שְׁמוּאֵל֙ הָרָמָ֔תָה וַיַּ֨גֶּד־ ל֔וֹ אֵ֛ת כָּל־ אֲשֶׁ֥ר עָֽשָׂה־ ל֖וֹ שָׁא֑וּל וַיֵּ֤לֶךְ הוּא֙ וּשְׁמוּאֵ֔ל וַיֵּשְׁב֖וּ (בנוית) בְּנָֽיוֹת׃ (לו) וַיֻּגַּ֥ד לְשָׁא֖וּל לֵאמֹ֑ר הִנֵּ֣ה דָוִ֔ד (בנוית) בְּנָי֖וֹת בָּרָמָֽה׃ (לז) וַיִּשְׁלַ֨ח שָׁא֣וּל מַלְאָכִים֮ לָקַ֣חַת אֶת־ דָּוִד֒ וַיַּ֗רְא אֶֽת־ לַהֲקַ֤ת הַנְּבִיאִים֙ נִבְּאִ֔ים וּשְׁמוּאֵ֕ל עֹמֵ֥ד נִצָּ֖ב עֲלֵיהֶ֑ם וַתְּהִ֞י עַֽל־ מַלְאֲכֵ֤י שָׁאוּל֙ ר֣וּחַ אֱלֹהִ֔ים וַיִּֽתְנַבְּא֖וּ גַּם־ הֵֽמָּה׃ (לח) וַיַּגִּ֣דוּ לְשָׁא֗וּל וַיִּשְׁלַח֙ מַלְאָכִ֣ים אֲחֵרִ֔ים וַיִּֽתְנַבְּא֖וּ גַּם־ הֵ֑מָּה וַיֹּ֣סֶף שָׁא֗וּל וַיִּשְׁלַח֙ מַלְאָכִ֣ים שְׁלִשִׁ֔ים וַיִּֽתְנַבְּא֖וּ גַּם־ הֵֽמָּה׃ (לט) וַיֵּ֨לֶךְ גַּם־ ה֜וּא הָרָמָ֗תָה וַיָּבֹא֙ עַד־ בּ֤וֹר הַגָּדוֹל֙ אֲשֶׁ֣ר בַּשֶּׂ֔כוּ וַיִּשְׁאַ֣ל וַיֹּ֔אמֶר אֵיפֹ֥ה שְׁמוּאֵ֖ל וְדָוִ֑ד וַיֹּ֕אמֶר הִנֵּ֖ה (בנוית) בְּנָי֥וֹת בָּרָמָֽה׃ (מ) וַיֵּ֣לֶךְ שָׁ֔ם אֶל־ (נוית) נָי֖וֹת בָּרָמָ֑ה וַתְּהִי֩ עָלָ֨יו גַּם־ ה֜וּא ר֣וּחַ אֱלֹהִ֗ים וַיֵּ֤לֶךְ הָלוֹךְ֙ וַיִּתְנַבֵּ֔א עַד־ בֹּא֖וֹ (בנוית) בְּנָי֥וֹת בָּרָמָֽה׃ (מא) וַיִּפְשַׁ֨ט גַּם־ ה֜וּא בְּגָדָ֗יו וַיִּתְנַבֵּ֤א גַם־ הוּא֙ לִפְנֵ֣י שְׁמוּאֵ֔ל וַיִּפֹּ֣ל עָרֹ֔ם כָּל־ הַיּ֥וֹם הַה֖וּא וְכָל־ הַלָּ֑יְלָה עַל־ כֵּן֙ יֹֽאמְר֔וּ הֲגַ֥ם שָׁא֖וּל בַּנְּבִיאִֽם׃ (מב) וַיִּבְרַ֣ח דָּוִ֔ד (מנוות) מִנָּי֖וֹת בָּרָמָ֑ה וַיָּבֹ֞א וַיֹּ֣אמֶר ׀ לִפְנֵ֣י יְהוֹנָתָ֗ן מֶ֤ה עָשִׂ֙יתִי֙ מֶֽה־ עֲוֺנִ֤י וּמֶֽה־ חַטָּאתִי֙ לִפְנֵ֣י אָבִ֔יךָ כִּ֥י מְבַקֵּ֖שׁ אֶת־ נַפְשִֽׁי׃ (מג) וַיֹּ֨אמֶר ל֣וֹ חָלִילָה֮ לֹ֣א תָמוּת֒ הִנֵּ֡ה (לו)־ (עשה) לֹֽא־ יַעֲשֶׂ֨ה אָבִ֜י דָּבָ֣ר גָּד֗וֹל א֚וֹ דָּבָ֣ר קָטֹ֔ן וְלֹ֥א יִגְלֶ֖ה אֶת־ אָזְנִ֑י וּמַדּוּעַ֩ יַסְתִּ֨יר אָבִ֥י מִמֶּ֛נִּי אֶת־ הַדָּבָ֥ר הַזֶּ֖ה אֵ֥ין זֹֽאת׃ (מד) וַיִּשָּׁבַ֨ע ע֜וֹד דָּוִ֗ד וַיֹּ֙אמֶר֙ יָדֹ֨עַ יָדַ֜ע אָבִ֗יךָ כִּֽי־ מָצָ֤אתִי חֵן֙ בְּעֵינֶ֔יךָ וַיֹּ֛אמֶר אַל־ יֵֽדַע־ זֹ֥את יְהוֹנָתָ֖ן פֶּן־ יֵֽעָצֵ֑ב וְאוּלָ֗ם חַי־ יְהוָה֙ וְחֵ֣י נַפְשֶׁ֔ךָ כִּ֣י כְפֶ֔שַׂע בֵּינִ֖י וּבֵ֥ין הַמָּֽוֶת׃
