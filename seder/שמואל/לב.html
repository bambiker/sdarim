שמואל סדר לב

לפי החלוקה לפרקים: שמואל ב יט, מ – כא, ו · 37 פסוקים

פתיחה באתר · ספר שמואל כולו ·

(א) וַיַּעֲבֹ֧ר כָּל־ הָעָ֛ם אֶת־ הַיַּרְדֵּ֖ן וְהַמֶּ֣לֶךְ עָבָ֑ר וַיִּשַּׁ֨ק הַמֶּ֤לֶךְ לְבַרְזִלַּי֙ וַיְבָ֣רֲכֵ֔הוּ וַיָּ֖שָׁב לִמְקֹמֽוֹ׃ (ב) וַיַּעֲבֹ֤ר הַמֶּ֙לֶךְ֙ הַגִּלְגָּ֔לָה וְכִמְהָ֖ן עָבַ֣ר עִמּ֑וֹ וְכָל־ עַ֤ם יְהוּדָה֙ (ויעברו) הֶעֱבִ֣ירוּ אֶת־ הַמֶּ֔לֶךְ וְגַ֕ם חֲצִ֖י עַ֥ם יִשְׂרָאֵֽל׃ (ג) וְהִנֵּ֛ה כָּל־ אִ֥ישׁ יִשְׂרָאֵ֖ל בָּאִ֣ים אֶל־ הַמֶּ֑לֶךְ וַיֹּאמְר֣וּ אֶל־ הַמֶּ֡לֶךְ מַדּוּעַ֩ גְּנָב֨וּךָ אַחֵ֜ינוּ אִ֣ישׁ יְהוּדָ֗ה וַיַּעֲבִ֨רוּ אֶת־ הַמֶּ֤לֶךְ וְאֶת־ בֵּיתוֹ֙ אֶת־ הַיַּרְדֵּ֔ן וְכָל־ אַנְשֵׁ֥י דָוִ֖ד עִמּֽוֹ׃ (ד) וַיַּעַן֩ כָּל־ אִ֨ישׁ יְהוּדָ֜ה עַל־ אִ֣ישׁ יִשְׂרָאֵ֗ל כִּֽי־ קָר֤וֹב הַמֶּ֙לֶךְ֙ אֵלַ֔י וְלָ֤מָּה זֶּה֙ חָרָ֣ה לְךָ֔ עַל־ הַדָּבָ֖ר הַזֶּ֑ה הֶאָכ֤וֹל אָכַ֙לְנוּ֙ מִן־ הַמֶּ֔לֶךְ אִם־ נִשֵּׂ֥את נִשָּׂ֖א לָֽנוּ׃ (ה) וַיַּ֣עַן אִֽישׁ־ יִשְׂרָאֵל֩ אֶת־ אִ֨ישׁ יְהוּדָ֜ה וַיֹּ֗אמֶר עֶֽשֶׂר־ יָד֨וֹת לִ֣י בַמֶּלֶךְ֮ וְגַם־ בְּדָוִד֮ אֲנִ֣י מִמְּךָ֒ וּמַדּ֙וּעַ֙ הֱקִלֹּתַ֔נִי וְלֹא־ הָיָ֨ה דְבָרִ֥י רִאשׁ֛וֹן לִ֖י לְהָשִׁ֣יב אֶת־ מַלְכִּ֑י וַיִּ֙קֶשׁ֙ דְּבַר־ אִ֣ישׁ יְהוּדָ֔ה מִדְּבַ֖ר אִ֥ישׁ יִשְׂרָאֵֽל׃ (ו) וְשָׁ֨ם נִקְרָ֜א אִ֣ישׁ בְּלִיַּ֗עַל וּשְׁמ֛וֹ שֶׁ֥בַע בֶּן־ בִּכְרִ֖י אִ֣ישׁ יְמִינִ֑י וַיִּתְקַ֣ע בַּשֹּׁפָ֗ר וַ֠יֹּאמֶר אֵֽין־ לָ֨נוּ חֵ֜לֶק בְּדָוִ֗ד וְלֹ֤א נַֽחֲלָה־ לָ֙נוּ֙ בְּבֶן־ יִשַׁ֔י אִ֥ישׁ לְאֹהָלָ֖יו יִשְׂרָאֵֽל׃ (ז) וַיַּ֜עַל כָּל־ אִ֤ישׁ יִשְׂרָאֵל֙ מֵאַחֲרֵ֣י דָוִ֔ד אַחֲרֵ֖י שֶׁ֣בַע בֶּן־ בִּכְרִ֑י וְאִ֤ישׁ יְהוּדָה֙ דָּבְק֣וּ בְמַלְכָּ֔ם מִן־ הַיַּרְדֵּ֖ן וְעַד־ יְרוּשָׁלִָֽם׃ (ח) וַיָּבֹ֨א דָוִ֣ד אֶל־ בֵּיתוֹ֮ יְרֽוּשָׁלִַם֒ וַיִּקַּ֣ח הַמֶּ֡לֶךְ אֵ֣ת עֶֽשֶׂר־ נָשִׁ֣ים ׀ פִּלַגְשִׁ֡ים אֲשֶׁ֣ר הִנִּיחַ֩ לִשְׁמֹ֨ר הַבַּ֜יִת וַֽיִּתְּנֵ֤ם בֵּית־ מִשְׁמֶ֙רֶת֙ וַֽיְכַלְכְּלֵ֔ם וַאֲלֵיהֶ֖ם לֹא־ בָ֑א וַתִּהְיֶ֧ינָה צְרֻר֛וֹת עַד־ י֥וֹם מֻתָ֖ן אַלְמְנ֥וּת חַיּֽוּת׃ (ט) וַיֹּ֤אמֶר הַמֶּ֙לֶךְ֙ אֶל־ עֲמָשָׂ֔א הַזְעֶק־ לִ֥י אֶת־ אִישׁ־ יְהוּדָ֖ה שְׁלֹ֣שֶׁת יָמִ֑ים וְאַתָּ֖ה פֹּ֥ה עֲמֹֽד׃ (י) וַיֵּ֥לֶךְ עֲמָשָׂ֖א לְהַזְעִ֣יק אֶת־ יְהוּדָ֑ה (וייחר) וַיּ֕וֹחֶר מִן־ הַמּוֹעֵ֖ד אֲשֶׁ֥ר יְעָדֽוֹ׃ (יא) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ אֲבִישַׁ֔י עַתָּ֗ה יֵ֧רַֽע לָ֛נוּ שֶׁ֥בַע בֶּן־ בִּכְרִ֖י מִן־ אַבְשָׁל֑וֹם אַ֠תָּה קַ֞ח אֶת־ עַבְדֵ֤י אֲדֹנֶ֙יךָ֙ וּרְדֹ֣ף אַחֲרָ֔יו פֶּן־ מָ֥צָא ל֛וֹ עָרִ֥ים בְּצֻר֖וֹת וְהִצִּ֥יל עֵינֵֽנוּ׃ (יב) וַיֵּצְא֤וּ אַֽחֲרָיו֙ אַנְשֵׁ֣י יוֹאָ֔ב וְהַכְּרֵתִ֥י וְהַפְּלֵתִ֖י וְכָל־ הַגִּבֹּרִ֑ים וַיֵּֽצְאוּ֙ מִיר֣וּשָׁלִַ֔ם לִרְדֹּ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (יג) הֵ֗ם עִם־ הָאֶ֤בֶן הַגְּדוֹלָה֙ אֲשֶׁ֣ר בְּגִבְע֔וֹן וַעֲמָשָׂ֖א בָּ֣א לִפְנֵיהֶ֑ם וְיוֹאָ֞ב חָג֣וּר ׀ מִדּ֣וֹ לְבֻשׁ֗וּ (ועלו) וְעָלָ֞יו חֲג֥וֹר חֶ֙רֶב֙ מְצֻמֶּ֤דֶת עַל־ מָתְנָיו֙ בְּתַעְרָ֔הּ וְה֥וּא יָצָ֖א וַתִּפֹּֽל׃ (יד) וַיֹּ֤אמֶר יוֹאָב֙ לַעֲמָשָׂ֔א הֲשָׁל֥וֹם אַתָּ֖ה אָחִ֑י וַתֹּ֜חֶז יַד־ יְמִ֥ין יוֹאָ֛ב בִּזְקַ֥ן עֲמָשָׂ֖א לִנְשָׁק־ לֽוֹ׃ (טו) וַעֲמָשָׂ֨א לֹֽא־ נִשְׁמַ֜ר בַּחֶ֣רֶב ׀ אֲשֶׁ֣ר בְּיַד־ יוֹאָ֗ב וַיַּכֵּהוּ֩ בָ֨הּ אֶל־ הַחֹ֜מֶשׁ וַיִּשְׁפֹּ֨ךְ מֵעָ֥יו אַ֛רְצָה וְלֹא־ שָׁ֥נָה ל֖וֹ וַיָּמֹ֑ת וְיוֹאָב֙ וַאֲבִישַׁ֣י אָחִ֔יו רָדַ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (טז) וְאִישׁ֙ עָמַ֣ד עָלָ֔יו מִֽנַּעֲרֵ֖י יוֹאָ֑ב וַיֹּ֗אמֶר מִי֩ אֲשֶׁ֨ר חָפֵ֧ץ בְּיוֹאָ֛ב וּמִ֥י אֲשֶׁר־ לְדָוִ֖ד אַחֲרֵ֥י יוֹאָֽב׃ (יז) וַעֲמָשָׂ֛א מִתְגֹּלֵ֥ל בַּדָּ֖ם בְּת֣וֹךְ הַֽמְסִּלָּ֑ה וַיַּ֨רְא הָאִ֜ישׁ כִּֽי־ עָמַ֣ד כָּל־ הָעָ֗ם וַיַּסֵּב֩ אֶת־ עֲמָשָׂ֨א מִן־ הַֽמְסִלָּ֤ה הַשָּׂדֶה֙ וַיַּשְׁלֵ֤ךְ עָלָיו֙ בֶּ֔גֶד כַּאֲשֶׁ֣ר רָאָ֔ה כָּל־ הַבָּ֥א עָלָ֖יו וְעָמָֽד׃ (יח) כַּאֲשֶׁ֥ר הֹגָ֖ה מִן־ הַֽמְסִלָּ֑ה עָבַ֤ר כָּל־ אִישׁ֙ אַחֲרֵ֣י יוֹאָ֔ב לִרְדֹּ֕ף אַחֲרֵ֖י שֶׁ֥בַע בֶּן־ בִּכְרִֽי׃ (יט) וַֽיַּעֲבֹ֞ר בְּכָל־ שִׁבְטֵ֣י יִשְׂרָאֵ֗ל אָבֵ֛לָה וּבֵ֥ית מַעֲכָ֖ה וְכָל־ הַבֵּרִ֑ים (ויקלהו) וַיִּקָּ֣הֲל֔וּ וַיָּבֹ֖אוּ אַף־ אַחֲרָֽיו׃ (כ) וַיָּבֹ֜אוּ וַיָּצֻ֣רוּ עָלָ֗יו בְּאָבֵ֙לָה֙ בֵּ֣ית הַֽמַּעֲכָ֔ה וַיִּשְׁפְּכ֤וּ סֹֽלְלָה֙ אֶל־ הָעִ֔יר וַֽתַּעֲמֹ֖ד בַּחֵ֑ל וְכָל־ הָעָם֙ אֲשֶׁ֣ר אֶת־ יוֹאָ֔ב מַשְׁחִיתִ֖ם לְהַפִּ֥יל הַחוֹמָֽה׃ (כא) וַתִּקְרָ֛א אִשָּׁ֥ה חֲכָמָ֖ה מִן־ הָעִ֑יר שִׁמְע֤וּ שִׁמְעוּ֙ אִמְרוּ־ נָ֣א אֶל־ יוֹאָ֔ב קְרַ֣ב עַד־ הֵ֔נָּה וַאֲדַבְּרָ֖ה אֵלֶֽיךָ׃ (כב) וַיִּקְרַ֣ב אֵלֶ֔יהָ וַתֹּ֧אמֶר הָאִשָּׁ֛ה הַאַתָּ֥ה יוֹאָ֖ב וַיֹּ֣אמֶר אָ֑נִי וַתֹּ֣אמֶר ל֗וֹ שְׁמַע֙ דִּבְרֵ֣י אֲמָתֶ֔ךָ וַיֹּ֖אמֶר שֹׁמֵ֥עַ אָנֹֽכִי׃ (כג) וַתֹּ֖אמֶר לֵאמֹ֑ר דַּבֵּ֨ר יְדַבְּר֤וּ בָרִֽאשֹׁנָה֙ לֵאמֹ֔ר שָׁאֹ֧ל יְשָׁאֲל֛וּ בְּאָבֵ֖ל וְכֵ֥ן הֵתַֽמּוּ׃ (כד) אָנֹכִ֕י שְׁלֻמֵ֖י אֱמוּנֵ֣י יִשְׂרָאֵ֑ל אַתָּ֣ה מְבַקֵּ֗שׁ לְהָמִ֨ית עִ֤יר וְאֵם֙ בְּיִשְׂרָאֵ֔ל לָ֥מָּה תְבַלַּ֖ע נַחֲלַ֥ת יְהוָֽה׃ (כה) וַיַּ֥עַן יוֹאָ֖ב וַיֹּאמַ֑ר חָלִ֤ילָה חָלִ֙ילָה֙ לִ֔י אִם־ אֲבַלַּ֖ע וְאִם־ אַשְׁחִֽית׃ (כו) לֹא־ כֵ֣ן הַדָּבָ֗ר כִּ֡י אִישׁ֩ מֵהַ֨ר אֶפְרַ֜יִם שֶׁ֧בַע בֶּן־ בִּכְרִ֣י שְׁמ֗וֹ נָשָׂ֤א יָדוֹ֙ בַּמֶּ֣לֶךְ בְּדָוִ֔ד תְּנֽוּ־ אֹת֣וֹ לְבַדּ֔וֹ וְאֵלְכָ֖ה מֵעַ֣ל הָעִ֑יר וַתֹּ֤אמֶר הָֽאִשָּׁה֙ אֶל־ יוֹאָ֔ב הִנֵּ֥ה רֹאשׁ֛וֹ מֻשְׁלָ֥ךְ אֵלֶ֖יךָ בְּעַ֥ד הַחוֹמָֽה׃ (כז) וַתָּבוֹא֩ הָאִשָּׁ֨ה אֶל־ כָּל־ הָעָ֜ם בְּחָכְמָתָ֗הּ וַֽיִּכְרְת֞וּ אֶת־ רֹ֨אשׁ שֶׁ֤בַע בֶּן־ בִּכְרִי֙ וַיַּשְׁלִ֣כוּ אֶל־ יוֹאָ֔ב וַיִּתְקַע֙ בַּשּׁוֹפָ֔ר וַיָּפֻ֥צוּ מֵֽעַל־ הָעִ֖יר אִ֣ישׁ לְאֹהָלָ֑יו וְיוֹאָ֛ב שָׁ֥ב יְרוּשָׁלִַ֖ם אֶל־ הַמֶּֽלֶךְ׃ (כח) וְיוֹאָ֕ב אֶ֥ל כָּל־ הַצָּבָ֖א יִשְׂרָאֵ֑ל וּבְנָיָה֙ בֶּן־ יְה֣וֹיָדָ֔ע עַל־ (הכרי) הַכְּרֵתִ֖י וְעַל־ הַפְּלֵתִֽי׃ (כט) וַאֲדֹרָ֖ם עַל־ הַמַּ֑ס וִיהוֹשָׁפָ֥ט בֶּן־ אֲחִיל֖וּד הַמַּזְכִּֽיר׃ (ל) (ושיא) וּשְׁוָ֖א סֹפֵ֑ר וְצָד֥וֹק וְאֶבְיָתָ֖ר כֹּהֲנִֽים׃ (לא) וְגַ֗ם עִירָא֙ הַיָּ֣אִרִ֔י הָיָ֥ה כֹהֵ֖ן לְדָוִֽד׃ (לב) וַיְהִ֣י רָעָב֩ בִּימֵ֨י דָוִ֜ד שָׁלֹ֣שׁ שָׁנִ֗ים שָׁנָה֙ אַחֲרֵ֣י שָׁנָ֔ה וַיְבַקֵּ֥שׁ דָּוִ֖ד אֶת־ פְּנֵ֣י יְהוָ֑ה וַיֹּ֣אמֶר יְהוָ֗ה אֶל־ שָׁאוּל֙ וְאֶל־ בֵּ֣ית הַדָּמִ֔ים עַל־ אֲשֶׁר־ הֵמִ֖ית אֶת־ הַגִּבְעֹנִֽים׃ (לג) וַיִּקְרָ֥א הַמֶּ֛לֶךְ לַגִּבְעֹנִ֖ים וַיֹּ֣אמֶר אֲלֵיהֶ֑ם וְהַגִּבְעֹנִ֞ים לֹ֣א מִבְּנֵ֧י יִשְׂרָאֵ֣ל הֵ֗מָּה כִּ֚י אִם־ מִיֶּ֣תֶר הָאֱמֹרִ֔י וּבְנֵ֤י יִשְׂרָאֵל֙ נִשְׁבְּע֣וּ לָהֶ֔ם וַיְבַקֵּ֤שׁ שָׁאוּל֙ לְהַכֹּתָ֔ם בְּקַנֹּאת֥וֹ לִבְנֵֽי־ יִשְׂרָאֵ֖ל וִיהוּדָֽה׃ (לד) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ הַגִּבְעֹנִ֔ים מָ֥ה אֶעֱשֶׂ֖ה לָכֶ֑ם וּבַמָּ֣ה אֲכַפֵּ֔ר וּבָרְכ֖וּ אֶת־ נַחֲלַ֥ת יְהוָֽה׃ (לה) וַיֹּ֧אמְרוּ ל֣וֹ הַגִּבְעֹנִ֗ים אֵֽין־ (לי) לָ֜נוּ כֶּ֤סֶף וְזָהָב֙ עִם־ שָׁא֣וּל וְעִם־ בֵּית֔וֹ וְאֵֽין־ לָ֥נוּ אִ֖ישׁ לְהָמִ֣ית בְּיִשְׂרָאֵ֑ל וַיֹּ֛אמֶר מָֽה־ אַתֶּ֥ם אֹמְרִ֖ים אֶעֱשֶׂ֥ה לָכֶֽם׃ (לו) וַיֹּֽאמְרוּ֙ אֶל־ הַמֶּ֔לֶךְ הָאִישׁ֙ אֲשֶׁ֣ר כִּלָּ֔נוּ וַאֲשֶׁ֖ר דִּמָּה־ לָ֑נוּ נִשְׁמַ֕דְנוּ מֵֽהִתְיַצֵּ֖ב בְּכָל־ גְּבֻ֥ל יִשְׂרָאֵֽל׃ (לז) (ינתן)־ יֻתַּן־ לָ֜נוּ שִׁבְעָ֤ה אֲנָשִׁים֙ מִבָּנָ֔יו וְהוֹקַֽעֲנוּם֙ לַֽיהוָ֔ה בְּגִבְעַ֥ת שָׁא֖וּל בְּחִ֣יר יְהוָ֑ה וַיֹּ֥אמֶר הַמֶּ֖לֶךְ אֲנִ֥י אֶתֵּֽן׃
