שמואל סדר לג

פתיחה באתר · ספר שמואל כולו ·

(א) וַיַּחְמֹ֣ל הַמֶּ֔לֶךְ עַל־ מְפִי־ בֹ֖שֶׁת בֶּן־ יְהוֹנָתָ֣ן בֶּן־ שָׁא֑וּל עַל־ שְׁבֻעַ֤ת יְהוָה֙ אֲשֶׁ֣ר בֵּֽינֹתָ֔ם בֵּ֣ין דָּוִ֔ד וּבֵ֖ין יְהוֹנָתָ֥ן בֶּן־ שָׁאֽוּל׃ (ב) וַיִּקַּ֣ח הַמֶּ֡לֶךְ אֶת־ שְׁ֠נֵי בְּנֵ֨י רִצְפָּ֤ה בַת־ אַיָּה֙ אֲשֶׁ֣ר יָלְדָ֣ה לְשָׁא֔וּל אֶת־ אַרְמֹנִ֖י וְאֶת־ מְפִבֹ֑שֶׁת וְאֶת־ חֲמֵ֗שֶׁת בְּנֵי֙ מִיכַ֣ל בַּת־ שָׁא֔וּל אֲשֶׁ֥ר יָלְדָ֛ה לְעַדְרִיאֵ֥ל בֶּן־ בַּרְזִלַּ֖י הַמְּחֹלָתִֽי׃ (ג) וַֽיִּתְּנֵ֞ם בְּיַ֣ד הַגִּבְעֹנִ֗ים וַיֹּקִיעֻ֤ם בָּהָר֙ לִפְנֵ֣י יְהוָ֔ה וַיִּפְּל֥וּ (שבעתים) שְׁבַעְתָּ֖ם יָ֑חַד (והם) וְהֵ֨מָּה הֻמְת֜וּ בִּימֵ֤י קָצִיר֙ בָּרִ֣אשֹׁנִ֔ים (תחלת) בִּתְחִלַּ֖ת קְצִ֥יר שְׂעֹרִֽים׃ (ד) וַתִּקַּ֣ח רִצְפָּה֩ בַת־ אַיָּ֨ה אֶת־ הַשַּׂ֜ק וַתַּטֵּ֨הוּ לָ֤הּ אֶל־ הַצּוּר֙ מִתְּחִלַּ֣ת קָצִ֔יר עַ֛ד נִתַּךְ־ מַ֥יִם עֲלֵיהֶ֖ם מִן־ הַשָּׁמָ֑יִם וְלֹֽא־ נָתְנָה֩ ע֨וֹף הַשָּׁמַ֜יִם לָנ֤וּחַ עֲלֵיהֶם֙ יוֹמָ֔ם וְאֶת־ חַיַּ֥ת הַשָּׂדֶ֖ה לָֽיְלָה׃ (ה) וַיֻּגַּ֖ד לְדָוִ֑ד אֵ֧ת אֲשֶׁר־ עָשְׂתָ֛ה רִצְפָּ֥ה בַת־ אַיָּ֖ה פִּלֶ֥גֶשׁ שָׁאֽוּל׃ (ו) וַיֵּ֣לֶךְ דָּוִ֗ד וַיִּקַּ֞ח אֶת־ עַצְמ֤וֹת שָׁאוּל֙ וְאֶת־ עַצְמוֹת֙ יְהוֹנָתָ֣ן בְּנ֔וֹ מֵאֵ֕ת בַּעֲלֵ֖י יָבֵ֣ישׁ גִּלְעָ֑ד אֲשֶׁר֩ גָּנְב֨וּ אֹתָ֜ם מֵרְחֹ֣ב בֵּֽית־ שַׁ֗ן אֲשֶׁ֨ר (תלום) תְּלָא֥וּם (שם הפלשתים) שָׁ֙מָּה֙ פְּלִשְׁתִּ֔ים בְּי֨וֹם הַכּ֧וֹת פְּלִשְׁתִּ֛ים אֶת־ שָׁא֖וּל בַּגִּלְבֹּֽעַ׃ (ז) וַיַּ֤עַל מִשָּׁם֙ אֶת־ עַצְמ֣וֹת שָׁא֔וּל וְאֶת־ עַצְמ֖וֹת יְהוֹנָתָ֣ן בְּנ֑וֹ וַיַּ֣אַסְפ֔וּ אֶת־ עַצְמ֖וֹת הַמּוּקָעִֽים׃ (ח) וַיִּקְבְּר֣וּ אֶת־ עַצְמוֹת־ שָׁא֣וּל וִיהוֹנָֽתָן־ בְּ֠נוֹ בְּאֶ֨רֶץ בִּנְיָמִ֜ן בְּצֵלָ֗ע בְּקֶ֙בֶר֙ קִ֣ישׁ אָבִ֔יו וַֽיַּעֲשׂ֔וּ כֹּ֥ל אֲשֶׁר־ צִוָּ֖ה הַמֶּ֑לֶךְ וַיֵּעָתֵ֧ר אֱלֹהִ֛ים לָאָ֖רֶץ אַֽחֲרֵי־ כֵֽן׃ (ט) וַתְּהִי־ ע֧וֹד מִלְחָמָ֛ה לַפְּלִשְׁתִּ֖ים אֶת־ יִשְׂרָאֵ֑ל וַיֵּ֨רֶד דָּוִ֜ד וַעֲבָדָ֥יו עִמּ֛וֹ וַיִּלָּחֲמ֥וּ אֶת־ פְּלִשְׁתִּ֖ים וַיָּ֥עַף דָּוִֽד׃ (י) (וישבו) וְיִשְׁבִּ֨י בְּנֹ֜ב אֲשֶׁ֣ר ׀ בִּילִידֵ֣י הָרָפָ֗ה וּמִשְׁקַ֤ל קֵינוֹ֙ שְׁלֹ֤שׁ מֵאוֹת֙ מִשְׁקַ֣ל נְחֹ֔שֶׁת וְה֖וּא חָג֣וּר חֲדָשָׁ֑ה וַיֹּ֖אמֶר לְהַכּ֥וֹת אֶת־ דָּוִֽד׃ (יא) וַיַּֽעֲזָר־ לוֹ֙ אֲבִישַׁ֣י בֶּן־ צְרוּיָ֔ה וַיַּ֥ךְ אֶת־ הַפְּלִשְׁתִּ֖י וַיְמִיתֵ֑הוּ אָ֣ז נִשְׁבְּעוּ֩ אַנְשֵׁי־ דָוִ֨ד ל֜וֹ לֵאמֹ֗ר לֹא־ תֵצֵ֨א ע֤וֹד אִתָּ֙נוּ֙ לַמִּלְחָמָ֔ה וְלֹ֥א תְכַבֶּ֖ה אֶת־ נֵ֥ר יִשְׂרָאֵֽל׃ (יב) וַֽיְהִי֙ אַֽחֲרֵי־ כֵ֔ן וַתְּהִי־ ע֧וֹד הַמִּלְחָמָ֛ה בְּג֖וֹב עִם־ פְּלִשְׁתִּ֑ים אָ֣ז הִכָּ֗ה סִבְּכַי֙ הַחֻ֣שָׁתִ֔י אֶת־ סַ֕ף אֲשֶׁ֖ר בִּילִדֵ֥י הָרָפָֽה׃ (יג) וַתְּהִי־ ע֧וֹד הַמִּלְחָמָ֛ה בְּג֖וֹב עִם־ פְּלִשְׁתִּ֑ים וַיַּ֡ךְ אֶלְחָנָן֩ בֶּן־ יַעְרֵ֨י אֹרְגִ֜ים בֵּ֣ית הַלַּחְמִ֗י אֵ֚ת גָּלְיָ֣ת הַגִּתִּ֔י וְעֵ֣ץ חֲנִית֔וֹ כִּמְנ֖וֹר אֹרְגִֽים׃ (יד) וַתְּהִי־ ע֥וֹד מִלְחָמָ֖ה בְּגַ֑ת וַיְהִ֣י ׀ אִ֣ישׁ (מדין) מָד֗וֹן וְאֶצְבְּעֹ֣ת יָדָיו֩ וְאֶצְבְּעֹ֨ת רַגְלָ֜יו שֵׁ֣שׁ וָשֵׁ֗שׁ עֶשְׂרִ֤ים וְאַרְבַּע֙ מִסְפָּ֔ר וְגַם־ ה֖וּא יֻלַּ֥ד לְהָרָפָֽה׃ (טו) וַיְחָרֵ֖ף אֶת־ יִשְׂרָאֵ֑ל וַיַּכֵּ֙הוּ֙ יְה֣וֹנָתָ֔ן בֶּן־ (שמעי) שִׁמְעָ֖ה אֲחִ֥י דָוִֽד׃ (טז) אֶת־ אַרְבַּ֥עַת אֵ֛לֶּה יֻלְּד֥וּ לְהָרָפָ֖ה בְּגַ֑ת וַיִּפְּל֥וּ בְיַד־ דָּוִ֖ד וּבְיַ֥ד עֲבָדָֽיו׃ (יז) וַיְדַבֵּ֤ר דָּוִד֙ לַֽיהוָ֔ה אֶת־ דִּבְרֵ֖י הַשִּׁירָ֣ה הַזֹּ֑את בְּיוֹם֩ הִצִּ֨יל יְהוָ֥ה אֹת֛וֹ מִכַּ֥ף כָּל־ אֹיְבָ֖יו וּמִכַּ֥ף שָׁאֽוּל׃ (יח) וַיֹּאמַ֑ר יְהוָ֛ה סַֽלְעִ֥י וּמְצֻדָתִ֖י וּמְפַלְטִי־ לִֽי׃ (יט) אֱלֹהֵ֥י צוּרִ֖י אֶחֱסֶה־ בּ֑וֹ מָגִנִּ֞י וְקֶ֣רֶן יִשְׁעִ֗י מִשְׂגַּבִּי֙ וּמְנוּסִ֔י מֹשִׁעִ֕י מֵחָמָ֖ס תֹּשִׁעֵֽנִי׃ (כ) מְהֻלָּ֖ל אֶקְרָ֣א יְהוָ֑ה וּמֵאֹיְבַ֖י אִוָּשֵֽׁעַ׃ (כא) כִּ֥י אֲפָפֻ֖נִי מִשְׁבְּרֵי־ מָ֑וֶת נַחֲלֵ֥י בְלִיַּ֖עַל יְבַעֲתֻֽנִי׃ (כב) חֶבְלֵ֥י שְׁא֖וֹל סַבֻּ֑נִי קִדְּמֻ֖נִי מֹֽקְשֵׁי־ מָֽוֶת׃ (כג) בַּצַּר־ לִי֙ אֶקְרָ֣א יְהוָ֔ה וְאֶל־ אֱלֹהַ֖י אֶקְרָ֑א וַיִּשְׁמַ֤ע מֵהֵֽיכָלוֹ֙ קוֹלִ֔י וְשַׁוְעָתִ֖י בְּאָזְנָֽיו׃ (כד) (ותגעש) וַיִּתְגָּעַ֤שׁ וַתִּרְעַשׁ֙ הָאָ֔רֶץ מוֹסְד֥וֹת הַשָּׁמַ֖יִם יִרְגָּ֑זוּ וַיִּֽתְגָּעֲשׁ֖וּ כִּֽי־ חָ֥רָה לֽוֹ׃ (כה) עָלָ֤ה עָשָׁן֙ בְּאַפּ֔וֹ וְאֵ֥שׁ מִפִּ֖יו תֹּאכֵ֑ל גֶּחָלִ֖ים בָּעֲר֥וּ מִמֶּֽנּוּ׃ (כו) וַיֵּ֥ט שָׁמַ֖יִם וַיֵּרַ֑ד וַעֲרָפֶ֖ל תַּ֥חַת רַגְלָֽיו׃ (כז) וַיִּרְכַּ֥ב עַל־ כְּר֖וּב וַיָּעֹ֑ף וַיֵּרָ֖א עַל־ כַּנְפֵי־ רֽוּחַ׃ (כח) וַיָּ֥שֶׁת חֹ֛שֶׁךְ סְבִיבֹתָ֖יו סֻכּ֑וֹת חַֽשְׁרַת־ מַ֖יִם עָבֵ֥י שְׁחָקִֽים׃ (כט) מִנֹּ֖גַהּ נֶגְדּ֑וֹ בָּעֲר֖וּ גַּחֲלֵי־ אֵֽשׁ׃ (ל) יַרְעֵ֥ם מִן־ שָׁמַ֖יִם יְהוָ֑ה וְעֶלְי֖וֹן יִתֵּ֥ן קוֹלֽוֹ׃ (לא) וַיִּשְׁלַ֥ח חִצִּ֖ים וַיְפִיצֵ֑ם בָּרָ֖ק (ויהמם) וַיָּהֹֽם׃ (לב) וַיֵּֽרָאוּ֙ אֲפִ֣קֵי יָ֔ם יִגָּל֖וּ מֹסְד֣וֹת תֵּבֵ֑ל בְּגַעֲרַ֣ת יְהוָ֔ה מִנִּשְׁמַ֖ת ר֥וּחַ אַפּֽוֹ׃ (לג) יִשְׁלַ֥ח מִמָּר֖וֹם יִקָּחֵ֑נִי יַֽמְשֵׁ֖נִי מִמַּ֥יִם רַבִּֽים׃ (לד) יַצִּילֵ֕נִי מֵאֹיְבִ֖י עָ֑ז מִשֹּׂ֣נְאַ֔י כִּ֥י אָמְצ֖וּ מִמֶּֽנִּי׃ (לה) יְקַדְּמֻ֖נִי בְּי֣וֹם אֵידִ֑י וַיְהִ֧י יְהוָ֛ה מִשְׁעָ֖ן לִֽי׃ (לו) וַיֹּצֵ֥א לַמֶּרְחָ֖ב אֹתִ֑י יְחַלְּצֵ֖נִי כִּי־ חָ֥פֵֽץ בִּֽי׃ (לז) יִגְמְלֵ֥נִי יְהוָ֖ה כְּצִדְקָתִ֑י כְּבֹ֥ר יָדַ֖י יָשִׁ֥יב לִֽי׃ (לח) כִּ֥י שָׁמַ֖רְתִּי דַּרְכֵ֣י יְהוָ֑ה וְלֹ֥א רָשַׁ֖עְתִּי מֵאֱלֹהָֽי׃ (לט) כִּ֥י כָל־ (משפטו) מִשְׁפָּטָ֖יו לְנֶגְדִּ֑י וְחֻקֹּתָ֖יו לֹא־ אָס֥וּר מִמֶּֽנָּה׃ (מ) וָאֶהְיֶ֥ה תָמִ֖ים ל֑וֹ וָאֶשְׁתַּמְּרָ֖ה מֵעֲוֺנִֽי׃ (מא) וַיָּ֧שֶׁב יְהוָ֛ה לִ֖י כְּצִדְקָתִ֑י כְּבֹרִ֖י לְנֶ֥גֶד עֵינָֽיו׃ (מב) עִם־ חָסִ֖יד תִּתְחַסָּ֑ד עִם־ גִּבּ֥וֹר תָּמִ֖ים תִּתַּמָּֽם׃ (מג) עִם־ נָבָ֖ר תִּתָּבָ֑ר וְעִם־ עִקֵּ֖שׁ תִּתַּפָּֽל׃ (מד) וְאֶת־ עַ֥ם עָנִ֖י תּוֹשִׁ֑יעַ וְעֵינֶ֖יךָ עַל־ רָמִ֥ים תַּשְׁפִּֽיל׃ (מה) כִּֽי־ אַתָּ֥ה נֵירִ֖י יְהוָ֑ה וַיהוָ֖ה יַגִּ֥יהַּ חָשְׁכִּֽי׃ (מו) כִּ֥י בְכָ֖ה אָר֣וּץ גְּד֑וּד בֵּאלֹהַ֖י אֲדַלֶּג־ שֽׁוּר׃ (מז) הָאֵ֖ל תָּמִ֣ים דַּרְכּ֑וֹ אִמְרַ֤ת יְהוָה֙ צְרוּפָ֔ה מָגֵ֣ן ה֔וּא לְכֹ֖ל הַחֹסִ֥ים בּֽוֹ׃ (מח) כִּ֥י מִי־ אֵ֖ל מִבַּלְעֲדֵ֣י יְהוָ֑ה וּמִ֥י צ֖וּר מִֽבַּלְעֲדֵ֥י אֱלֹהֵֽינוּ׃ (מט) הָאֵ֥ל מָעוּזִּ֖י חָ֑יִל וַיַּתֵּ֥ר תָּמִ֖ים (דרכו) דַּרְכִּֽי׃ (נ) מְשַׁוֶּ֥ה (רגליו) רַגְלַ֖י כָּאַיָּל֑וֹת וְעַ֥ל בָּמוֹתַ֖י יַעֲמִדֵֽנִי׃ (נא) מְלַמֵּ֥ד יָדַ֖י לַמִּלְחָמָ֑ה וְנִחַ֥ת קֶֽשֶׁת־ נְחוּשָׁ֖ה זְרֹעֹתָֽי׃ (נב) וַתִּתֶּן־ לִ֖י מָגֵ֣ן יִשְׁעֶ֑ךָ וַעֲנֹתְךָ֖ תַּרְבֵּֽנִי׃ (נג) תַּרְחִ֥יב צַעֲדִ֖י תַּחְתֵּ֑נִי וְלֹ֥א מָעֲד֖וּ קַרְסֻלָּֽי׃ (נד) אֶרְדְּפָ֥ה אֹיְבַ֖י וָאַשְׁמִידֵ֑ם וְלֹ֥א אָשׁ֖וּב עַד־ כַּלּוֹתָֽם׃ (נה) וָאֲכַלֵּ֥ם וָאֶמְחָצֵ֖ם וְלֹ֣א יְקוּמ֑וּן וַֽיִּפְּל֖וּ תַּ֥חַת רַגְלָֽי׃ (נו) וַתַּזְרֵ֥נִי חַ֖יִל לַמִּלְחָמָ֑ה תַּכְרִ֥יעַ קָמַ֖י תַּחְתֵּֽנִי׃ (נז) וְאֹ֣יְבַ֔י תַּ֥תָּה לִּ֖י עֹ֑רֶף מְשַׂנְאַ֖י וָאַצְמִיתֵֽם׃ (נח) יִשְׁע֖וּ וְאֵ֣ין מֹשִׁ֑יעַ אֶל־ יְהוָ֖ה וְלֹ֥א עָנָֽם׃ (נט) וְאֶשְׁחָקֵ֖ם כַּעֲפַר־ אָ֑רֶץ כְּטִיט־ חוּצ֥וֹת אֲדִקֵּ֖ם אֶרְקָעֵֽם׃ (ס) וַֽתְּפַלְּטֵ֔נִי מֵרִיבֵ֖י עַמִּ֑י תִּשְׁמְרֵ֙נִי֙ לְרֹ֣אשׁ גּוֹיִ֔ם עַ֥ם לֹא־ יָדַ֖עְתִּי יַעַבְדֻֽנִי׃ (סא) בְּנֵ֥י נֵכָ֖ר יִתְכַּֽחֲשׁוּ־ לִ֑י לִשְׁמ֥וֹעַ אֹ֖זֶן יִשָּׁ֥מְעוּ לִֽי׃ (סב) בְּנֵ֥י נֵכָ֖ר יִבֹּ֑לוּ וְיַחְגְּר֖וּ מִמִּסְגְּרוֹתָֽם׃ (סג) חַי־ יְהוָ֖ה וּבָר֣וּךְ צוּרִ֑י וְיָרֻ֕ם אֱלֹהֵ֖י צ֥וּר יִשְׁעִֽי׃ (סד) הָאֵ֕ל הַנֹּתֵ֥ן נְקָמֹ֖ת לִ֑י וּמוֹרִ֥יד עַמִּ֖ים תַּחְתֵּֽנִי׃ (סה) וּמוֹצִיאִ֖י מֵאֹֽיְבָ֑י וּמִקָּמַי֙ תְּר֣וֹמְמֵ֔נִי מֵאִ֥ישׁ חֲמָסִ֖ים תַּצִּילֵֽנִי׃ (סו) עַל־ כֵּ֛ן אוֹדְךָ֥ יְהוָ֖ה בַּגּוֹיִ֑ם וּלְשִׁמְךָ֖ אֲזַמֵּֽר׃
