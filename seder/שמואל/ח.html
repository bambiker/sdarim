שמואל סדר ח

פתיחה באתר · ספר שמואל כולו ·

(א) וַיּ֧וֹשַׁע יְהוָ֛ה בַּיּ֥וֹם הַה֖וּא אֶת־ יִשְׂרָאֵ֑ל וְהַ֨מִּלְחָמָ֔ה עָבְרָ֖ה אֶת־ בֵּ֥ית אָֽוֶן׃ (ב) וְאִֽישׁ־ יִשְׂרָאֵ֥ל נִגַּ֖שׂ בַּיּ֣וֹם הַה֑וּא וַיֹּאֶל֩ שָׁא֨וּל אֶת־ הָעָ֜ם לֵאמֹ֗ר אָר֣וּר הָ֠אִישׁ אֲשֶׁר־ יֹ֨אכַל לֶ֜חֶם עַד־ הָעֶ֗רֶב וְנִקַּמְתִּי֙ מֵאֹ֣יְבַ֔י וְלֹֽא טָעַ֥ם כָּל־ הָעָ֖ם לָֽחֶם׃ (ג) וְכָל־ הָאָ֖רֶץ בָּ֣אוּ בַיָּ֑עַר וַיְהִ֥י דְבַ֖שׁ עַל־ פְּנֵ֥י הַשָּׂדֶֽה׃ (ד) וַיָּבֹ֤א הָעָם֙ אֶל־ הַיַּ֔עַר וְהִנֵּ֖ה הֵ֣לֶךְ דְּבָ֑שׁ וְאֵין־ מַשִּׂ֤יג יָדוֹ֙ אֶל־ פִּ֔יו כִּֽי־ יָרֵ֥א הָעָ֖ם אֶת־ הַשְּׁבֻעָֽה׃ (ה) וְיוֹנָתָ֣ן לֹֽא־ שָׁמַ֗ע בְּהַשְׁבִּ֣יעַ אָבִיו֮ אֶת־ הָעָם֒ וַיִּשְׁלַ֗ח אֶת־ קְצֵ֤ה הַמַּטֶּה֙ אֲשֶׁ֣ר בְּיָד֔וֹ וַיִּטְבֹּ֥ל אוֹתָ֖הּ בְּיַעְרַ֣ת הַדְּבָ֑שׁ וַיָּ֤שֶׁב יָדוֹ֙ אֶל־ פִּ֔יו (ותראנה) וַתָּאֹ֖רְנָה עֵינָֽיו׃ (ו) וַיַּעַן֩ אִ֨ישׁ מֵֽהָעָ֜ם וַיֹּ֗אמֶר הַשְׁבֵּעַ֩ הִשְׁבִּ֨יעַ אָבִ֤יךָ אֶת־ הָעָם֙ לֵאמֹ֔ר אָר֥וּר הָאִ֛ישׁ אֲשֶׁר־ יֹ֥אכַל לֶ֖חֶם הַיּ֑וֹם וַיָּ֖עַף הָעָֽם׃ (ז) וַיֹּ֙אמֶר֙ יֽוֹנָתָ֔ן עָכַ֥ר אָבִ֖י אֶת־ הָאָ֑רֶץ רְאוּ־ נָא֙ כִּֽי־ אֹ֣רוּ עֵינַ֔י כִּ֣י טָעַ֔מְתִּי מְעַ֖ט דְּבַ֥שׁ הַזֶּֽה׃ (ח) אַ֗ף כִּ֡י לוּא֩ אָכֹ֨ל אָכַ֤ל הַיּוֹם֙ הָעָ֔ם מִשְּׁלַ֥ל אֹיְבָ֖יו אֲשֶׁ֣ר מָצָ֑א כִּ֥י עַתָּ֛ה לֹֽא־ רָבְתָ֥ה מַכָּ֖ה בַּפְּלִשְׁתִּֽים׃ (ט) וַיַּכּ֞וּ בַּיּ֤וֹם הַהוּא֙ בַּפְּלִשְׁתִּ֔ים מִמִּכְמָ֖שׂ אַיָּלֹ֑נָה וַיָּ֥עַף הָעָ֖ם מְאֹֽד׃ (י) (ויעש) וַיַּ֤עַט הָעָם֙ אֶל־ (שלל) הַשָּׁלָ֔ל וַיִּקְח֨וּ צֹ֧אן וּבָקָ֛ר וּבְנֵ֥י בָקָ֖ר וַיִּשְׁחֲטוּ־ אָ֑רְצָה וַיֹּ֥אכַל הָעָ֖ם עַל־ הַדָּֽם׃ (יא) וַיַּגִּ֤ידוּ לְשָׁאוּל֙ לֵאמֹ֔ר הִנֵּ֥ה הָעָ֛ם חֹטִ֥אים לַֽיהוָ֖ה לֶאֱכֹ֣ל עַל־ הַדָּ֑ם וַיֹּ֣אמֶר בְּגַדְתֶּ֔ם גֹּֽלּוּ־ אֵלַ֥י הַיּ֖וֹם אֶ֥בֶן גְּדוֹלָֽה׃ (יב) וַיֹּ֣אמֶר שָׁא֣וּל פֻּ֣צוּ בָעָ֡ם וַאֲמַרְתֶּ֣ם לָהֶ֡ם הַגִּ֣ישׁוּ אֵלַי֩ אִ֨ישׁ שׁוֹר֜וֹ וְאִ֣ישׁ שְׂיֵ֗הוּ וּשְׁחַטְתֶּ֤ם בָּזֶה֙ וַאֲכַלְתֶּ֔ם וְלֹֽא־ תֶחֶטְא֥וּ לַֽיהוָ֖ה לֶאֱכֹ֣ל אֶל־ הַדָּ֑ם וַיַּגִּ֨שׁוּ כָל־ הָעָ֜ם אִ֣ישׁ שׁוֹר֧וֹ בְיָד֛וֹ הַלַּ֖יְלָה וַיִּשְׁחֲטוּ־ שָֽׁם׃ (יג) וַיִּ֧בֶן שָׁא֛וּל מִזְבֵּ֖חַ לַֽיהוָ֑ה אֹת֣וֹ הֵחֵ֔ל לִבְנ֥וֹת מִזְבֵּ֖חַ לַֽיהוָֽה׃ (יד) וַיֹּ֣אמֶר שָׁא֡וּל נֵרְדָ֣ה אַחֲרֵי֩ פְלִשְׁתִּ֨ים ׀ לַ֜יְלָה וְֽנָבֹ֥זָה בָהֶ֣ם ׀ עַד־ א֣וֹר הַבֹּ֗קֶר וְלֹֽא־ נַשְׁאֵ֤ר בָּהֶם֙ אִ֔ישׁ וַיֹּ֣אמְר֔וּ כָּל־ הַטּ֥וֹב בְּעֵינֶ֖יךָ עֲשֵׂ֑ה וַיֹּ֙אמֶר֙ הַכֹּהֵ֔ן נִקְרְבָ֥ה הֲלֹ֖ם אֶל־ הָאֱלֹהִֽים׃ (טו) וַיִּשְׁאַ֤ל שָׁאוּל֙ בֵּֽאלֹהִ֔ים הַֽאֵרֵד֙ אַחֲרֵ֣י פְלִשְׁתִּ֔ים הֲתִתְּנֵ֖ם בְּיַ֣ד יִשְׂרָאֵ֑ל וְלֹ֥א עָנָ֖הוּ בַּיּ֥וֹם הַהֽוּא׃ (טז) וַיֹּ֣אמֶר שָׁא֔וּל גֹּ֣שֽׁוּ הֲלֹ֔ם כֹּ֖ל פִּנּ֣וֹת הָעָ֑ם וּדְע֣וּ וּרְא֔וּ בַּמָּ֗ה הָֽיְתָ֛ה הַחַטָּ֥את הַזֹּ֖את הַיּֽוֹם׃ (יז) כִּ֣י חַי־ יְהוָ֗ה הַמּוֹשִׁ֙יעַ֙ אֶת־ יִשְׂרָאֵ֔ל כִּ֧י אִם־ יֶשְׁנ֛וֹ בְּיוֹנָתָ֥ן בְּנִ֖י כִּ֣י מ֣וֹת יָמ֑וּת וְאֵ֥ין עֹנֵ֖הוּ מִכָּל־ הָעָֽם׃ (יח) וַיֹּ֣אמֶר אֶל־ כָּל־ יִשְׂרָאֵ֗ל אַתֶּם֙ תִּֽהְיוּ֙ לְעֵ֣בֶר אֶחָ֔ד וַֽאֲנִי֙ וְיוֹנָתָ֣ן בְּנִ֔י נִהְיֶ֖ה לְעֵ֣בֶר אֶחָ֑ד וַיֹּאמְר֤וּ הָעָם֙ אֶל־ שָׁא֔וּל הַטּ֥וֹב בְּעֵינֶ֖יךָ עֲשֵֽׂה׃ (יט) וַיֹּ֣אמֶר שָׁא֗וּל אֶל־ יְהוָ֛ה אֱלֹהֵ֥י יִשְׂרָאֵ֖ל הָ֣בָה תָמִ֑ים וַיִּלָּכֵ֧ד יוֹנָתָ֛ן וְשָׁא֖וּל וְהָעָ֥ם יָצָֽאוּ׃ (כ) וַיֹּ֣אמֶר שָׁא֔וּל הַפִּ֕ילוּ בֵּינִ֕י וּבֵ֖ין יוֹנָתָ֣ן בְּנִ֑י וַיִּלָּכֵ֖ד יוֹנָתָֽן׃ (כא) וַיֹּ֤אמֶר שָׁאוּל֙ אֶל־ י֣וֹנָתָ֔ן הַגִּ֥ידָה לִּ֖י מֶ֣ה עָשִׂ֑יתָה וַיַּגֶּד־ ל֣וֹ יוֹנָתָ֗ן וַיֹּאמֶר֩ טָעֹ֨ם טָעַ֜מְתִּי בִּקְצֵ֨ה הַמַּטֶּ֧ה אֲשֶׁר־ בְּיָדִ֛י מְעַ֥ט דְּבַ֖שׁ הִנְנִ֥י אָמֽוּת׃ (כב) וַיֹּ֣אמֶר שָׁא֔וּל כֹּֽה־ יַעֲשֶׂ֥ה אֱלֹהִ֖ים וְכֹ֣ה יוֹסִ֑ף כִּֽי־ מ֥וֹת תָּמ֖וּת יוֹנָתָֽן׃ (כג) וַיֹּ֨אמֶר הָעָ֜ם אֶל־ שָׁא֗וּל הֲ‍ֽיוֹנָתָ֤ן ׀ יָמוּת֙ אֲשֶׁ֣ר עָ֠שָׂה הַיְשׁוּעָ֨ה הַגְּדוֹלָ֣ה הַזֹּאת֮ בְּיִשְׂרָאֵל֒ חָלִ֗ילָה חַי־ יְהוָה֙ אִם־ יִפֹּ֞ל מִשַּׂעֲרַ֤ת רֹאשׁוֹ֙ אַ֔רְצָה כִּֽי־ עִם־ אֱלֹהִ֥ים עָשָׂ֖ה הַיּ֣וֹם הַזֶּ֑ה וַיִּפְדּ֥וּ הָעָ֛ם אֶת־ יוֹנָתָ֖ן וְלֹא־ מֵֽת׃ (כד) וַיַּ֣עַל שָׁא֔וּל מֵאַחֲרֵ֖י פְּלִשְׁתִּ֑ים וּפְלִשְׁתִּ֖ים הָלְכ֥וּ לִמְקוֹמָֽם׃ (כה) וְשָׁא֛וּל לָכַ֥ד הַמְּלוּכָ֖ה עַל־ יִשְׂרָאֵ֑ל וַיִּלָּ֣חֶם סָבִ֣יב ׀ בְּֽכָל־ אֹיְבָ֡יו בְּמוֹאָ֣ב ׀ וּבִבְנֵי־ עַמּ֨וֹן וּבֶאֱד֜וֹם וּבְמַלְכֵ֤י צוֹבָה֙ וּבַפְּלִשְׁתִּ֔ים וּבְכֹ֥ל אֲשֶׁר־ יִפְנֶ֖ה יַרְשִֽׁיעַ׃ (כו) וַיַּ֣עַשׂ חַ֔יִל וַיַּ֖ךְ אֶת־ עֲמָלֵ֑ק וַיַּצֵּ֥ל אֶת־ יִשְׂרָאֵ֖ל מִיַּ֥ד שֹׁסֵֽהוּ׃ (כז) וַיִּֽהְיוּ֙ בְּנֵ֣י שָׁא֔וּל יוֹנָתָ֥ן וְיִשְׁוִ֖י וּמַלְכִּי־ שׁ֑וּעַ וְשֵׁם֙ שְׁתֵּ֣י בְנֹתָ֔יו שֵׁ֤ם הַבְּכִירָה֙ מֵרַ֔ב וְשֵׁ֥ם הַקְּטַנָּ֖ה מִיכַֽל׃ (כח) וְשֵׁם֙ אֵ֣שֶׁת שָׁא֔וּל אֲחִינֹ֖עַם בַּת־ אֲחִימָ֑עַץ וְשֵׁ֤ם שַׂר־ צְבָאוֹ֙ אֲבִינֵ֔ר בֶּן־ נֵ֖ר דּ֥וֹד שָׁאֽוּל׃ (כט) וְקִ֧ישׁ אֲבִֽי־ שָׁא֛וּל וְנֵ֥ר אֲבִֽי־ אַבְנֵ֖ר בֶּן־ אֲבִיאֵֽל׃ (ל) וַתְּהִ֤י הַמִּלְחָמָה֙ חֲזָקָ֣ה עַל־ פְּלִשְׁתִּ֔ים כֹּ֖ל יְמֵ֣י שָׁא֑וּל וְרָאָ֨ה שָׁא֜וּל כָּל־ אִ֤ישׁ גִּבּוֹר֙ וְכָל־ בֶּן־ חַ֔יִל וַיַּאַסְפֵ֖הוּ אֵלָֽיו׃ (לא) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ שָׁא֔וּל אֹתִ֨י שָׁלַ֤ח יְהוָה֙ לִמְשָׁחֳךָ֣ לְמֶ֔לֶךְ עַל־ עַמּ֖וֹ עַל־ יִשְׂרָאֵ֑ל וְעַתָּ֣ה שְׁמַ֔ע לְק֖וֹל דִּבְרֵ֥י יְהוָֽה׃ (לב) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת פָּקַ֕דְתִּי אֵ֛ת אֲשֶׁר־ עָשָׂ֥ה עֲמָלֵ֖ק לְיִשְׂרָאֵ֑ל אֲשֶׁר־ שָׂ֥ם לוֹ֙ בַּדֶּ֔רֶךְ בַּעֲלֹת֖וֹ מִמִּצְרָֽיִם׃ (לג) עַתָּה֩ לֵ֨ךְ וְהִכִּֽיתָ֜ה אֶת־ עֲמָלֵ֗ק וְהַֽחֲרַמְתֶּם֙ אֶת־ כָּל־ אֲשֶׁר־ ל֔וֹ וְלֹ֥א תַחְמֹ֖ל עָלָ֑יו וְהֵמַתָּ֞ה מֵאִ֣ישׁ עַד־ אִשָּׁ֗ה מֵֽעֹלֵל֙ וְעַד־ יוֹנֵ֔ק מִשּׁ֣וֹר וְעַד־ שֶׂ֔ה מִגָּמָ֖ל וְעַד־ חֲמֽוֹר׃ (לד) וַיְשַׁמַּ֤ע שָׁאוּל֙ אֶת־ הָעָ֔ם וַֽיִּפְקְדֵם֙ בַּטְּלָאִ֔ים מָאתַ֥יִם אֶ֖לֶף רַגְלִ֑י וַעֲשֶׂ֥רֶת אֲלָפִ֖ים אֶת־ אִ֥ישׁ יְהוּדָֽה׃ (לה) וַיָּבֹ֥א שָׁא֖וּל עַד־ עִ֣יר עֲמָלֵ֑ק וַיָּ֖רֶב בַּנָּֽחַל׃ (לו) וַיֹּ֣אמֶר שָׁא֣וּל אֶֽל־ הַקֵּינִ֡י לְכוּ֩ סֻּ֨רוּ רְד֜וּ מִתּ֣וֹךְ עֲמָלֵקִ֗י פֶּן־ אֹֽסִפְךָ֙ עִמּ֔וֹ וְאַתָּ֞ה עָשִׂ֤יתָה חֶ֙סֶד֙ עִם־ כָּל־ בְּנֵ֣י יִשְׂרָאֵ֔ל בַּעֲלוֹתָ֖ם מִמִּצְרָ֑יִם וַיָּ֥סַר קֵינִ֖י מִתּ֥וֹךְ עֲמָלֵֽק׃ (לז) וַיַּ֥ךְ שָׁא֖וּל אֶת־ עֲמָלֵ֑ק מֵֽחֲוִילָה֙ בּוֹאֲךָ֣ שׁ֔וּר אֲשֶׁ֖ר עַל־ פְּנֵ֥י מִצְרָֽיִם׃ (לח) וַיִּתְפֹּ֛שׂ אֶת־ אֲגַ֥ג מֶֽלֶךְ־ עֲמָלֵ֖ק חָ֑י וְאֶת־ כָּל־ הָעָ֖ם הֶחֱרִ֥ים לְפִי־ חָֽרֶב׃ (לט) וַיַּחְמֹל֩ שָׁא֨וּל וְהָעָ֜ם עַל־ אֲגָ֗ג וְעַל־ מֵיטַ֣ב הַצֹּאן֩ וְהַבָּקָ֨ר וְהַמִּשְׁנִ֤ים וְעַל־ הַכָּרִים֙ וְעַל־ כָּל־ הַטּ֔וֹב וְלֹ֥א אָב֖וּ הַחֲרִימָ֑ם וְכָל־ הַמְּלָאכָ֛ה נְמִבְזָ֥ה וְנָמֵ֖ס אֹתָ֥הּ הֶחֱרִֽימוּ׃ (מ) וַֽיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ שְׁמוּאֵ֖ל לֵאמֹֽר׃ (מא) נִחַ֗מְתִּי כִּֽי־ הִמְלַ֤כְתִּי אֶת־ שָׁאוּל֙ לְמֶ֔לֶךְ כִּֽי־ שָׁב֙ מֵאַֽחֲרַ֔י וְאֶת־ דְּבָרַ֖י לֹ֣א הֵקִ֑ים וַיִּ֙חַר֙ לִשְׁמוּאֵ֔ל וַיִּזְעַ֥ק אֶל־ יְהוָ֖ה כָּל־ הַלָּֽיְלָה׃ (מב) וַיַּשְׁכֵּ֧ם שְׁמוּאֵ֛ל לִקְרַ֥את שָׁא֖וּל בַּבֹּ֑קֶר וַיֻּגַּ֨ד לִשְׁמוּאֵ֜ל לֵאמֹ֗ר בָּֽא־ שָׁא֤וּל הַכַּרְמֶ֙לָה֙ וְהִנֵּ֨ה מַצִּ֥יב לוֹ֙ יָ֔ד וַיִּסֹּב֙ וַֽיַּעֲבֹ֔ר וַיֵּ֖רֶד הַגִּלְגָּֽל׃ (מג) וַיָּבֹ֥א שְׁמוּאֵ֖ל אֶל־ שָׁא֑וּל וַיֹּ֧אמֶר ל֣וֹ שָׁא֗וּל בָּר֤וּךְ אַתָּה֙ לַֽיהוָ֔ה הֲקִימֹ֖תִי אֶת־ דְּבַ֥ר יְהוָֽה׃ (מד) וַיֹּ֣אמֶר שְׁמוּאֵ֔ל וּמֶ֛ה קֽוֹל־ הַצֹּ֥אן הַזֶּ֖ה בְּאָזְנָ֑י וְק֣וֹל הַבָּקָ֔ר אֲשֶׁ֥ר אָנֹכִ֖י שֹׁמֵֽעַ׃ (מה) וַיֹּ֨אמֶר שָׁא֜וּל מֵעֲמָלֵקִ֣י הֱבִיא֗וּם אֲשֶׁ֨ר חָמַ֤ל הָעָם֙ עַל־ מֵיטַ֤ב הַצֹּאן֙ וְהַבָּקָ֔ר לְמַ֥עַן זְבֹ֖חַ לַיהוָ֣ה אֱלֹהֶ֑יךָ וְאֶת־ הַיּוֹתֵ֖ר הֶחֱרַֽמְנוּ׃ (מו) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ שָׁא֔וּל הֶ֚רֶף וְאַגִּ֣ידָה לְּךָ֔ אֵת֩ אֲשֶׁ֨ר דִּבֶּ֧ר יְהוָ֛ה אֵלַ֖י הַלָּ֑יְלָה (ויאמרו) וַיֹּ֥אמֶר ל֖וֹ דַּבֵּֽר׃
