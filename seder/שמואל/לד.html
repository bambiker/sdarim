שמואל סדר לד

פתיחה באתר · ספר שמואל כולו ·

(א) (מגדיל) מִגְדּ֖וֹל יְשׁוּע֣וֹת מַלְכּ֑וֹ וְעֹֽשֶׂה־ חֶ֧סֶד לִמְשִׁיח֛וֹ לְדָוִ֥ד וּלְזַרְע֖וֹ עַד־ עוֹלָֽם׃ (ב) וְאֵ֛לֶּה דִּבְרֵ֥י דָוִ֖ד הָאַֽחֲרֹנִ֑ים נְאֻ֧ם דָּוִ֣ד בֶּן־ יִשַׁ֗י וּנְאֻ֤ם הַגֶּ֙בֶר֙ הֻ֣קַם עָ֔ל מְשִׁ֙יחַ֙ אֱלֹהֵ֣י יַֽעֲקֹ֔ב וּנְעִ֖ים זְמִר֥וֹת יִשְׂרָאֵֽל׃ (ג) ר֥וּחַ יְהוָ֖ה דִּבֶּר־ בִּ֑י וּמִלָּת֖וֹ עַל־ לְשׁוֹנִֽי׃ (ד) אָמַר֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל לִ֥י דִבֶּ֖ר צ֣וּר יִשְׂרָאֵ֑ל מוֹשֵׁל֙ בָּאָדָ֔ם צַדִּ֕יק מוֹשֵׁ֖ל יִרְאַ֥ת אֱלֹהִֽים׃ (ה) וּכְא֥וֹר בֹּ֖קֶר יִזְרַח־ שָׁ֑מֶשׁ בֹּ֚קֶר לֹ֣א עָב֔וֹת מִנֹּ֥גַהּ מִמָּטָ֖ר דֶּ֥שֶׁא מֵאָֽרֶץ׃ (ו) כִּֽי־ לֹא־ כֵ֥ן בֵּיתִ֖י עִם־ אֵ֑ל כִּי֩ בְרִ֨ית עוֹלָ֜ם שָׂ֣ם לִ֗י עֲרוּכָ֤ה בַכֹּל֙ וּשְׁמֻרָ֔ה כִּֽי־ כָל־ יִשְׁעִ֥י וְכָל־ חֵ֖פֶץ כִּֽי־ לֹ֥א יַצְמִֽיחַ׃ (ז) וּבְלִיַּ֕עַל כְּק֥וֹץ מֻנָ֖ד כֻּלָּ֑הַם כִּֽי־ לֹ֥א בְיָ֖ד יִקָּֽחוּ׃ (ח) וְאִישׁ֙ יִגַּ֣ע בָּהֶ֔ם יִמָּלֵ֥א בַרְזֶ֖ל וְעֵ֣ץ חֲנִ֑ית וּבָאֵ֕שׁ שָׂר֥וֹף יִשָּׂרְפ֖וּ בַּשָּֽׁבֶת׃ (ט) אֵ֛לֶּה שְׁמ֥וֹת הַגִּבֹּרִ֖ים אֲשֶׁ֣ר לְדָוִ֑ד יֹשֵׁ֨ב בַּשֶּׁ֜בֶת תַּחְכְּמֹנִ֣י ׀ רֹ֣אשׁ הַשָּׁלִשִׁ֗י ה֚וּא עֲדִינ֣וֹ (העצנו) הָֽעֶצְנִ֔י עַל־ שְׁמֹנֶ֥ה מֵא֛וֹת חָלָ֖ל בְּפַ֥עַם (אחד) אֶחָֽת׃ (י) (ואחרו) וְאַחֲרָ֛יו אֶלְעָזָ֥ר בֶּן־ (דדי) דֹּד֖וֹ בֶּן־ אֲחֹחִ֑י בִּשְׁלֹשָׁ֨ה (גברים) הַגִּבֹּרִ֜ים עִם־ דָּוִ֗ד בְּחָֽרְפָ֤ם בַּפְּלִשְׁתִּים נֶאֶסְפוּ־ שָׁ֣ם לַמִּלְחָמָ֔ה וַֽיַּעֲל֖וּ אִ֥ישׁ יִשְׂרָאֵֽל׃ (יא) ה֣וּא קָם֩ וַיַּ֨ךְ בַּפְּלִשְׁתִּ֜ים עַ֣ד ׀ כִּֽי־ יָגְעָ֣ה יָד֗וֹ וַתִּדְבַּ֤ק יָדוֹ֙ אֶל־ הַחֶ֔רֶב וַיַּ֧עַשׂ יְהוָ֛ה תְּשׁוּעָ֥ה גְדוֹלָ֖ה בַּיּ֣וֹם הַה֑וּא וְהָעָ֛ם יָשֻׁ֥בוּ אַחֲרָ֖יו אַךְ־ לְפַשֵּֽׁט׃ (יב) וְאַחֲרָ֛יו שַׁמָּ֥א בֶן־ אָגֵ֖א הָרָרִ֑י וַיֵּאָסְפ֨וּ פְלִשְׁתִּ֜ים לַחַיָּ֗ה וַתְּהִי־ שָׁ֞ם חֶלְקַ֤ת הַשָּׂדֶה֙ מְלֵאָ֣ה עֲדָשִׁ֔ים וְהָעָ֥ם נָ֖ס מִפְּנֵ֥י פְלִשְׁתִּֽים׃ (יג) וַיִּתְיַצֵּ֤ב בְּתוֹךְ־ הַֽחֶלְקָה֙ וַיַּצִּילֶ֔הָ וַיַּ֖ךְ אֶת־ פְּלִשְׁתִּ֑ים וַיַּ֥עַשׂ יְהוָ֖ה תְּשׁוּעָ֥ה גְדוֹלָֽה׃ (יד) וַיֵּרְד֨וּ (שלשים) שְׁלֹשָׁ֜ה מֵהַשְּׁלֹשִׁ֣ים רֹ֗אשׁ וַיָּבֹ֤אוּ אֶל־ קָצִיר֙ אֶל־ דָּוִ֔ד אֶל־ מְעָרַ֖ת עֲדֻלָּ֑ם וְחַיַּ֣ת פְּלִשְׁתִּ֔ים חֹנָ֖ה בְּעֵ֥מֶק רְפָאִֽים׃ (טו) וְדָוִ֖ד אָ֣ז בַּמְּצוּדָ֑ה וּמַצַּ֣ב פְּלִשְׁתִּ֔ים אָ֖ז בֵּ֥ית לָֽחֶם׃ (טז) וַיִּתְאַוֶּ֥ה דָוִ֖ד וַיֹּאמַ֑ר מִ֚י יַשְׁקֵ֣נִי מַ֔יִם מִבֹּ֥אר בֵּֽית־ לֶ֖חֶם אֲשֶׁ֥ר בַּשָּֽׁעַר׃ (יז) וַיִּבְקְעוּ֩ שְׁלֹ֨שֶׁת הַגִּבֹּרִ֜ים בְּמַחֲנֵ֣ה פְלִשְׁתִּ֗ים וַיִּֽשְׁאֲבוּ־ מַ֙יִם֙ מִבֹּ֤אר בֵּֽית־ לֶ֙חֶם֙ אֲשֶׁ֣ר בַּשַּׁ֔עַר וַיִּשְׂא֖וּ וַיָּבִ֣אוּ אֶל־ דָּוִ֑ד וְלֹ֤א אָבָה֙ לִשְׁתּוֹתָ֔ם וַיַּסֵּ֥ךְ אֹתָ֖ם לַֽיהוָֽה׃ (יח) וַיֹּ֡אמֶר חָלִילָה֩ לִּ֨י יְהוָ֜ה מֵעֲשֹׂ֣תִי זֹ֗את הֲדַ֤ם הָֽאֲנָשִׁים֙ הַהֹלְכִ֣ים בְּנַפְשׁוֹתָ֔ם וְלֹ֥א אָבָ֖ה לִשְׁתּוֹתָ֑ם אֵ֣לֶּה עָשׂ֔וּ שְׁלֹ֖שֶׁת הַגִּבֹּרִֽים׃ (יט) וַאֲבִישַׁ֞י אֲחִ֣י ׀ יוֹאָ֣ב בֶּן־ צְרוּיָ֗ה ה֚וּא רֹ֣אשׁ (השלשי) הַשְּׁלֹשָׁ֔ה וְהוּא֙ עוֹרֵ֣ר אֶת־ חֲנִית֔וֹ עַל־ שְׁלֹ֥שׁ מֵא֖וֹת חָלָ֑ל וְלוֹ־ שֵׁ֖ם בַּשְּׁלֹשָֽׁה׃ (כ) מִן־ הַשְּׁלֹשָׁה֙ הֲכִ֣י נִכְבָּ֔ד וַיְהִ֥י לָהֶ֖ם לְשָׂ֑ר וְעַד־ הַשְּׁלֹשָׁ֖ה לֹא־ בָֽא׃ (כא) וּבְנָיָ֨הוּ בֶן־ יְהוֹיָדָ֧ע בֶּן־ אִֽישׁ־ (חי) חַ֛יִל רַב־ פְּעָלִ֖ים מִֽקַּבְצְאֵ֑ל ה֣וּא הִכָּ֗ה אֵ֣ת שְׁנֵ֤י אֲרִאֵל֙ מוֹאָ֔ב וְ֠הוּא יָרַ֞ד וְהִכָּ֧ה אֶֽת־ (האריה) הָאֲרִ֛י בְּת֥וֹךְ הַבֹּ֖אר בְּי֥וֹם הַשָּֽׁלֶג׃ (כב) וְהוּא־ הִכָּה֩ אֶת־ אִ֨ישׁ מִצְרִ֜י (אשר) אִ֣ישׁ מַרְאֶ֗ה וּבְיַ֤ד הַמִּצְרִי֙ חֲנִ֔ית וַיֵּ֥רֶד אֵלָ֖יו בַּשָּׁ֑בֶט וַיִּגְזֹ֤ל אֶֽת־ הַחֲנִית֙ מִיַּ֣ד הַמִּצְרִ֔י וַיַּהַרְגֵ֖הוּ בַּחֲנִיתֽוֹ׃ (כג) אֵ֣לֶּה עָשָׂ֔ה בְּנָיָ֖הוּ בֶּן־ יְהוֹיָדָ֑ע וְלוֹ־ שֵׁ֖ם בִּשְׁלֹשָׁ֥ה הַגִּבֹּרִֽים׃ (כד) מִן־ הַשְּׁלֹשִׁ֣ים נִכְבָּ֔ד וְאֶל־ הַשְּׁלֹשָׁ֖ה לֹא־ בָ֑א וַיְשִׂמֵ֥הוּ דָוִ֖ד אֶל־ מִשְׁמַעְתּֽוֹ׃ (כה) עֲשָׂה־ אֵ֥ל אֲחִֽי־ יוֹאָ֖ב בַּשְּׁלֹשִׁ֑ים אֶלְחָנָ֥ן בֶּן־ דֹּד֖וֹ בֵּ֥ית לָֽחֶם׃ (כו) שַׁמָּה֙ הַֽחֲרֹדִ֔י אֱלִיקָ֖א הַחֲרֹדִֽי׃ (כז) חֶ֚לֶץ הַפַּלְטִ֔י עִירָ֥א בֶן־ עִקֵּ֖שׁ הַתְּקוֹעִֽי׃ (כח) אֲבִיעֶ֙זֶר֙ הָֽעַנְּתֹתִ֔י מְבֻנַּ֖י הַחֻשָׁתִֽי׃ (כט) צַלְמוֹן֙ הָֽאֲחֹחִ֔י מַהְרַ֖י הַנְּטֹפָתִֽי׃ (ל) חֵ֥לֶב בֶּֽן־ בַּעֲנָ֖ה הַנְּטֹפָתִ֑י אִתַּי֙ בֶּן־ רִיבַ֔י מִגִּבְעַ֖ת בְּנֵ֥י בִנְיָמִֽן׃ (לא) בְּנָיָ֙הוּ֙ פִּרְעָ֣תֹנִ֔י הִדַּ֖י מִנַּ֥חֲלֵי גָֽעַשׁ׃ (לב) אֲבִֽי־ עַלְבוֹן֙ הָֽעַרְבָתִ֔י עַזְמָ֖וֶת הַבַּרְחֻמִֽי׃ (לג) אֶלְיַחְבָּא֙ הַשַּׁ֣עַלְבֹנִ֔י בְּנֵ֥י יָשֵׁ֖ן יְהוֹנָתָֽן׃ (לד) שַׁמָּה֙ הַֽהֲרָרִ֔י אֲחִיאָ֥ם בֶּן־ שָׁרָ֖ר הָארָרִֽי׃ (לה) אֱלִיפֶ֥לֶט בֶּן־ אֲחַסְבַּ֖י בֶּן־ הַמַּֽעֲכָתִ֑י אֱלִיעָ֥ם בֶּן־ אֲחִיתֹ֖פֶל הַגִּלֹנִֽי׃ (לו) (חצרו) חֶצְרַי֙ הַֽכַּרְמְלִ֔י פַּעֲרַ֖י הָאַרְבִּֽי׃ (לז) יִגְאָ֤ל בֶּן־ נָתָן֙ מִצֹּבָ֔ה בָּנִ֖י הַגָּדִֽי׃ (לח) צֶ֖לֶק הָעַמֹּנִ֑י נַחְרַי֙ הַבְּאֵ֣רֹתִ֔י (נשאי) נֹשֵׂ֕א כְּלֵ֖י יוֹאָ֥ב בֶּן־ צְרֻיָֽה׃ (לט) עִירָא֙ הַיִּתְרִ֔י גָּרֵ֖ב הַיִּתְרִֽי׃ (מ) אֽוּרִיָּה֙ הַֽחִתִּ֔י כֹּ֖ל שְׁלֹשִׁ֥ים וְשִׁבְעָֽה׃ (מא) וַיֹּ֙סֶף֙ אַף־ יְהוָ֔ה לַחֲר֖וֹת בְּיִשְׂרָאֵ֑ל וַיָּ֨סֶת אֶת־ דָּוִ֤ד בָּהֶם֙ לֵאמֹ֔ר לֵ֛ךְ מְנֵ֥ה אֶת־ יִשְׂרָאֵ֖ל וְאֶת־ יְהוּדָֽה׃ (מב) וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֶל־ יוֹאָ֣ב ׀ שַׂר־ הַחַ֣יִל אֲשֶׁר־ אִתּ֗וֹ שֽׁוּט־ נָ֞א בְּכָל־ שִׁבְטֵ֤י יִשְׂרָאֵל֙ מִדָּן֙ וְעַד־ בְּאֵ֣ר שֶׁ֔בַע וּפִקְד֖וּ אֶת־ הָעָ֑ם וְיָ֣דַעְתִּ֔י אֵ֖ת מִסְפַּ֥ר הָעָֽם׃ (מג) וַיֹּ֨אמֶר יוֹאָ֜ב אֶל־ הַמֶּ֗לֶךְ וְיוֹסֵ֣ף יְהוָה֩ אֱלֹהֶ֨יךָ אֶל־ הָעָ֜ם כָּהֵ֤ם ׀ וְכָהֵם֙ מֵאָ֣ה פְעָמִ֔ים וְעֵינֵ֥י אֲדֹנִֽי־ הַמֶּ֖לֶךְ רֹא֑וֹת וַאדֹנִ֣י הַמֶּ֔לֶךְ לָ֥מָּה חָפֵ֖ץ בַּדָּבָ֥ר הַזֶּֽה׃ (מד) וַיֶּחֱזַ֤ק דְּבַר־ הַמֶּ֙לֶךְ֙ אֶל־ יוֹאָ֔ב וְעַ֖ל שָׂרֵ֣י הֶחָ֑יִל וַיֵּצֵ֨א יוֹאָ֜ב וְשָׂרֵ֤י הַחַ֙יִל֙ לִפְנֵ֣י הַמֶּ֔לֶךְ לִפְקֹ֥ד אֶת־ הָעָ֖ם אֶת־ יִשְׂרָאֵֽל׃ (מה) וַיַּעַבְר֖וּ אֶת־ הַיַּרְדֵּ֑ן וַיַּחֲנ֣וּ בַעֲרוֹעֵ֗ר יְמִ֥ין הָעִ֛יר אֲשֶׁ֛ר בְּתוֹךְ־ הַנַּ֥חַל הַגָּ֖ד וְאֶל־ יַעְזֵֽר׃ (מו) וַיָּבֹ֙אוּ֙ הַגִּלְעָ֔דָה וְאֶל־ אֶ֥רֶץ תַּחְתִּ֖ים חָדְשִׁ֑י וַיָּבֹ֙אוּ֙ דָּ֣נָה יַּ֔עַן וְסָבִ֖יב אֶל־ צִידֽוֹן׃ (מז) וַיָּבֹ֙אוּ֙ מִבְצַר־ צֹ֔ר וְכָל־ עָרֵ֥י הַחִוִּ֖י וְהַֽכְּנַעֲנִ֑י וַיֵּֽצְא֛וּ אֶל־ נֶ֥גֶב יְהוּדָ֖ה בְּאֵ֥ר שָֽׁבַע׃ (מח) וַיָּשֻׁ֖טוּ בְּכָל־ הָאָ֑רֶץ וַיָּבֹ֜אוּ מִקְצֵ֨ה תִשְׁעָ֧ה חֳדָשִׁ֛ים וְעֶשְׂרִ֥ים י֖וֹם יְרוּשָׁלִָֽם׃ (מט) וַיִּתֵּ֥ן יוֹאָ֛ב אֶת־ מִסְפַּ֥ר מִפְקַד־ הָעָ֖ם אֶל־ הַמֶּ֑לֶךְ וַתְּהִ֣י יִשְׂרָאֵ֡ל שְׁמֹנֶה֩ מֵא֨וֹת אֶ֤לֶף אִֽישׁ־ חַ֙יִל֙ שֹׁ֣לֵֽף חֶ֔רֶב וְאִ֣ישׁ יְהוּדָ֔ה חֲמֵשׁ־ מֵא֥וֹת אֶ֖לֶף אִֽישׁ׃ (נ) וַיַּ֤ךְ לֵב־ דָּוִד֙ אֹת֔וֹ אַחֲרֵי־ כֵ֖ן סָפַ֣ר אֶת־ הָעָ֑ם וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ יְהוָ֗ה חָטָ֤אתִי מְאֹד֙ אֲשֶׁ֣ר עָשִׂ֔יתִי וְעַתָּ֣ה יְהוָ֔ה הַֽעֲבֶר־ נָא֙ אֶת־ עֲוֺ֣ן עַבְדְּךָ֔ כִּ֥י נִסְכַּ֖לְתִּי מְאֹֽד׃ (נא) וַיָּ֥קָם דָּוִ֖ד בַּבֹּ֑קֶר וּדְבַר־ יְהוָ֗ה הָיָה֙ אֶל־ גָּ֣ד הַנָּבִ֔יא חֹזֵ֥ה דָוִ֖ד לֵאמֹֽר׃ (נב) הָל֞וֹךְ וְדִבַּרְתָּ֣ אֶל־ דָּוִ֗ד כֹּ֚ה אָמַ֣ר יְהוָ֔ה שָׁלֹ֕שׁ אָנֹכִ֖י נוֹטֵ֣ל עָלֶ֑יךָ בְּחַר־ לְךָ֥ אַֽחַת־ מֵהֶ֖ם וְאֶֽעֱשֶׂה־ לָּֽךְ׃ (נג) וַיָּבֹא־ גָ֥ד אֶל־ דָּוִ֖ד וַיַּגֶּד־ ל֑וֹ וַיֹּ֣אמֶר ל֡וֹ הֲתָב֣וֹא לְךָ֣ שֶֽׁבַע שָׁנִ֣ים ׀ רָעָ֣ב ׀ בְּאַרְצֶ֡ךָ אִם־ שְׁלֹשָׁ֣ה חֳ֠דָשִׁים נֻסְךָ֨ לִפְנֵֽי־ צָרֶ֜יךָ וְה֣וּא רֹדְפֶ֗ךָ וְאִם־ הֱ֠יוֹת שְׁלֹ֨שֶׁת יָמִ֥ים דֶּ֙בֶר֙ בְּאַרְצֶ֔ךָ עַתָּה֙ דַּ֣ע וּרְאֵ֔ה מָה־ אָשִׁ֥יב שֹׁלְחִ֖י דָּבָֽר׃ (נד) וַיֹּ֧אמֶר דָּוִ֛ד אֶל־ גָּ֖ד צַר־ לִ֣י מְאֹ֑ד נִפְּלָה־ נָּ֤א בְיַד־ יְהוָה֙ כִּֽי־ רַבִּ֣ים (רחמו) רַֽחֲמָ֔יו וּבְיַד־ אָדָ֖ם אַל־ אֶפֹּֽלָה׃ (נה) וַיִּתֵּ֨ן יְהוָ֥ה דֶּ֙בֶר֙ בְּיִשְׂרָאֵ֔ל מֵהַבֹּ֖קֶר וְעַד־ עֵ֣ת מוֹעֵ֑ד וַיָּ֣מָת מִן־ הָעָ֗ם מִדָּן֙ וְעַד־ בְּאֵ֣ר שֶׁ֔בַע שִׁבְעִ֥ים אֶ֖לֶף אִֽישׁ׃ (נו) וַיִּשְׁלַח֩ יָד֨וֹ הַמַּלְאָ֥ךְ ׀ יְרֽוּשָׁלִַם֮ לְשַׁחֲתָהּ֒ וַיִּנָּ֤חֶם יְהוָה֙ אֶל־ הָ֣רָעָ֔ה וַ֠יֹּאמֶר לַמַּלְאָ֞ךְ הַמַּשְׁחִ֤ית בָּעָם֙ רַ֔ב עַתָּ֖ה הֶ֣רֶף יָדֶ֑ךָ וּמַלְאַ֤ךְ יְהוָה֙ הָיָ֔ה עִם־ גֹּ֖רֶן (האורנה) הָאֲרַ֥וְנָה הַיְבֻסִֽי׃ (נז) וַיֹּאמֶר֩ דָּוִ֨ד אֶל־ יְהוָ֜ה בִּרְאֹת֣וֹ ׀ אֶֽת־ הַמַּלְאָ֣ךְ ׀ הַמַּכֶּ֣ה בָעָ֗ם וַיֹּ֙אמֶר֙ הִנֵּ֨ה אָנֹכִ֤י חָטָ֙אתִי֙ וְאָנֹכִ֣י הֶעֱוֵ֔יתִי וְאֵ֥לֶּה הַצֹּ֖אן מֶ֣ה עָשׂ֑וּ תְּהִ֨י נָ֥א יָדְךָ֛ בִּ֖י וּבְבֵ֥ית אָבִֽי׃ (נח) וַיָּבֹא־ גָ֥ד אֶל־ דָּוִ֖ד בַּיּ֣וֹם הַה֑וּא וַיֹּ֣אמֶר ל֗וֹ עֲלֵה֙ הָקֵ֤ם לַֽיהוָה֙ מִזְבֵּ֔חַ בְּגֹ֖רֶן (ארניה) אֲרַ֥וְנָה הַיְבֻסִֽי׃ (נט) וַיַּ֤עַל דָּוִד֙ כִּדְבַר־ גָּ֔ד כַּאֲשֶׁ֖ר צִוָּ֥ה יְהוָֽה׃ (ס) וַיַּשְׁקֵ֣ף אֲרַ֗וְנָה וַיַּ֤רְא אֶת־ הַמֶּ֙לֶךְ֙ וְאֶת־ עֲבָדָ֔יו עֹבְרִ֖ים עָלָ֑יו וַיֵּצֵ֣א אֲרַ֔וְנָה וַיִּשְׁתַּ֧חוּ לַמֶּ֛לֶךְ אַפָּ֖יו אָֽרְצָה׃ (סא) וַיֹּ֣אמֶר אֲרַ֔וְנָה מַדּ֛וּעַ בָּ֥א אֲדֹנִֽי־ הַמֶּ֖לֶךְ אֶל־ עַבְדּ֑וֹ וַיֹּ֨אמֶר דָּוִ֜ד לִקְנ֧וֹת מֵעִמְּךָ֣ אֶת־ הַגֹּ֗רֶן לִבְנ֤וֹת מִזְבֵּ֙חַ֙ לַֽיהוָ֔ה וְתֵעָצַ֥ר הַמַּגֵּפָ֖ה מֵעַ֥ל הָעָֽם׃ (סב) וַיֹּ֤אמֶר אֲרַ֙וְנָה֙ אֶל־ דָּוִ֔ד יִקַּ֥ח וְיַ֛עַל אֲדֹנִ֥י הַמֶּ֖לֶךְ הַטּ֣וֹב (בעינו) בְּעֵינָ֑יו רְאֵה֙ הַבָּקָ֣ר לָעֹלָ֔ה וְהַמֹּרִגִּ֛ים וּכְלֵ֥י הַבָּקָ֖ר לָעֵצִֽים׃ (סג) הַכֹּ֗ל נָתַ֛ן אֲרַ֥וְנָה הַמֶּ֖לֶךְ לַמֶּ֑לֶךְ וַיֹּ֤אמֶר אֲרַ֙וְנָה֙ אֶל־ הַמֶּ֔לֶךְ יְהוָ֥ה אֱלֹהֶ֖יךָ יִרְצֶֽךָ׃ (סד) וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֶל־ אֲרַ֗וְנָה לֹ֚א כִּֽי־ קָנ֨וֹ אֶקְנֶ֤ה מֵאֽוֹתְךָ֙ בִּמְחִ֔יר וְלֹ֧א אַעֲלֶ֛ה לַיהוָ֥ה אֱלֹהַ֖י עֹל֣וֹת חִנָּ֑ם וַיִּ֨קֶן דָּוִ֤ד אֶת־ הַגֹּ֙רֶן֙ וְאֶת־ הַבָּקָ֔ר בְּכֶ֖סֶף שְׁקָלִ֥ים חֲמִשִּֽׁים׃ (סה) וַיִּבֶן֩ שָׁ֨ם דָּוִ֤ד מִזְבֵּ֙חַ֙ לַֽיהוָ֔ה וַיַּ֥עַל עֹל֖וֹת וּשְׁלָמִ֑ים וַיֵּעָתֵ֤ר יְהוָה֙ לָאָ֔רֶץ וַתֵּעָצַ֥ר הַמַּגֵּפָ֖ה מֵעַ֥ל יִשְׂרָאֵֽל׃
