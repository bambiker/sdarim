שמואל סדר יח

פתיחה באתר · ספר שמואל כולו ·

(א) וַיֹּ֨אמֶר שָׁא֜וּל אֶל־ דָּוִ֗ד בָּר֤וּךְ אַתָּה֙ בְּנִ֣י דָוִ֔ד גַּ֚ם עָשֹׂ֣ה תַעֲשֶׂ֔ה וְגַ֖ם יָכֹ֣ל תּוּכָ֑ל וַיֵּ֤לֶךְ דָּוִד֙ לְדַרְכּ֔וֹ וְשָׁא֖וּל שָׁ֥ב לִמְקוֹמֽוֹ׃ (ב) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ לִבּ֔וֹ עַתָּ֛ה אֶסָּפֶ֥ה יוֹם־ אֶחָ֖ד בְּיַד־ שָׁא֑וּל אֵֽין־ לִ֨י ט֜וֹב כִּ֣י הִמָּלֵ֥ט אִמָּלֵ֣ט ׀ אֶל־ אֶ֣רֶץ פְּלִשְׁתִּ֗ים וְנוֹאַ֨שׁ מִמֶּ֤נִּי שָׁאוּל֙ לְבַקְשֵׁ֤נִי עוֹד֙ בְּכָל־ גְּב֣וּל יִשְׂרָאֵ֔ל וְנִמְלַטְתִּ֖י מִיָּדֽוֹ׃ (ג) וַיָּ֣קָם דָּוִ֔ד וַיַּעֲבֹ֣ר ה֔וּא וְשֵׁשׁ־ מֵא֥וֹת אִ֖ישׁ אֲשֶׁ֣ר עִמּ֑וֹ אֶל־ אָכִ֥ישׁ בֶּן־ מָע֖וֹךְ מֶ֥לֶךְ גַּֽת׃ (ד) וַיֵּשֶׁב֩ דָּוִ֨ד עִם־ אָכִ֥ישׁ בְּגַ֛ת ה֥וּא וַאֲנָשָׁ֖יו אִ֣ישׁ וּבֵית֑וֹ דָּוִד֙ וּשְׁתֵּ֣י נָשָׁ֔יו אֲחִינֹ֙עַם֙ הַיִּזְרְעֵאלִ֔ית וַאֲבִיגַ֥יִל אֵֽשֶׁת־ נָבָ֖ל הַֽכַּרְמְלִֽית׃ (ה) וַיֻּגַּ֣ד לְשָׁא֔וּל כִּֽי־ בָרַ֥ח דָּוִ֖ד גַּ֑ת וְלֹֽא־ (יוסף) יָסַ֥ף ע֖וֹד לְבַקְשֽׁוֹ׃ (ו) וַיֹּ֨אמֶר דָּוִ֜ד אֶל־ אָכִ֗ישׁ אִם־ נָא֩ מָצָ֨אתִי חֵ֤ן בְּעֵינֶ֙יךָ֙ יִתְּנוּ־ לִ֣י מָק֗וֹם בְּאַחַ֛ת עָרֵ֥י הַשָּׂדֶ֖ה וְאֵ֣שְׁבָה שָּׁ֑ם וְלָ֨מָּה יֵשֵׁ֧ב עַבְדְּךָ֛ בְּעִ֥יר הַמַּמְלָכָ֖ה עִמָּֽךְ׃ (ז) וַיִּתֶּן־ ל֥וֹ אָכִ֛ישׁ בַּיּ֥וֹם הַה֖וּא אֶת־ צִֽקְלָ֑ג לָכֵ֞ן הָיְתָ֤ה צִֽקְלַג֙ לְמַלְכֵ֣י יְהוּדָ֔ה עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (ח) וַֽיְהִי֙ מִסְפַּ֣ר הַיָּמִ֔ים אֲשֶׁר־ יָשַׁ֥ב דָּוִ֖ד בִּשְׂדֵ֣ה פְלִשְׁתִּ֑ים יָמִ֖ים וְאַרְבָּעָ֥ה חֳדָשִֽׁים׃ (ט) וַיַּ֤עַל דָּוִד֙ וַֽאֲנָשָׁ֔יו וַֽיִּפְשְׁט֛וּ אֶל־ הַגְּשׁוּרִ֥י (והגרזי) וְהַגִּזְרִ֖י וְהָעֲמָלֵקִ֑י כִּ֣י הֵ֜נָּה יֹשְׁב֤וֹת הָאָ֙רֶץ֙ אֲשֶׁ֣ר מֵֽעוֹלָ֔ם בּוֹאֲךָ֥ שׁ֖וּרָה וְעַד־ אֶ֥רֶץ מִצְרָֽיִם׃ (י) וְהִכָּ֤ה דָוִד֙ אֶת־ הָאָ֔רֶץ וְלֹ֥א יְחַיֶּ֖ה אִ֣ישׁ וְאִשָּׁ֑ה וְלָקַח֩ צֹ֨אן וּבָקָ֜ר וַחֲמֹרִ֤ים וּגְמַלִּים֙ וּבְגָדִ֔ים וַיָּ֖שָׁב וַיָּבֹ֥א אֶל־ אָכִֽישׁ׃ (יא) וַיֹּ֣אמֶר אָכִ֔ישׁ אַל־ פְּשַׁטְתֶּ֖ם הַיּ֑וֹם וַיֹּ֣אמֶר דָּוִ֗ד עַל־ נֶ֤גֶב יְהוּדָה֙ וְעַל־ נֶ֣גֶב הַיַּרְחְמְאֵלִ֔י וְאֶל־ נֶ֖גֶב הַקֵּינִֽי׃ (יב) וְאִ֨ישׁ וְאִשָּׁ֜ה לֹֽא־ יְחַיֶּ֣ה דָוִ֗ד לְהָבִ֥יא גַת֙ לֵאמֹ֔ר פֶּן־ יַגִּ֥דוּ עָלֵ֖ינוּ לֵאמֹ֑ר כֹּֽה־ עָשָׂ֤ה דָוִד֙ וְכֹ֣ה מִשְׁפָּט֔וֹ כָּל־ הַ֨יָּמִ֔ים אֲשֶׁ֥ר יָשַׁ֖ב בִּשְׂדֵ֥ה פְלִשְׁתִּֽים׃ (יג) וַיַּאֲמֵ֥ן אָכִ֖ישׁ בְּדָוִ֣ד לֵאמֹ֑ר הַבְאֵ֤שׁ הִבְאִישׁ֙ בְּעַמּ֣וֹ בְיִשְׂרָאֵ֔ל וְהָ֥יָה לִ֖י לְעֶ֥בֶד עוֹלָֽם׃ (יד) וַֽיְהִי֙ בַּיָּמִ֣ים הָהֵ֔ם וַיִּקְבְּצ֨וּ פְלִשְׁתִּ֤ים אֶת־ מַֽחֲנֵיהֶם֙ לַצָּבָ֔א לְהִלָּחֵ֖ם בְּיִשְׂרָאֵ֑ל וַיֹּ֤אמֶר אָכִישׁ֙ אֶל־ דָּוִ֔ד יָדֹ֣עַ תֵּדַ֗ע כִּ֤י אִתִּי֙ תֵּצֵ֣א בַֽמַּחֲנֶ֔ה אַתָּ֖ה וַאֲנָשֶֽׁיךָ׃ (טו) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ אָכִ֔ישׁ לָכֵן֙ אַתָּ֣ה תֵדַ֔ע אֵ֥ת אֲשֶׁר־ יַעֲשֶׂ֖ה עַבְדֶּ֑ךָ וַיֹּ֤אמֶר אָכִישׁ֙ אֶל־ דָּוִ֔ד לָכֵ֗ן שֹׁמֵ֧ר לְרֹאשִׁ֛י אֲשִֽׂימְךָ֖ כָּל־ הַיָּמִֽים׃ (טז) וּשְׁמוּאֵ֣ל מֵ֔ת וַיִּסְפְּדוּ־ לוֹ֙ כָּל־ יִשְׂרָאֵ֔ל וַיִּקְבְּרֻ֥הוּ בָרָמָ֖ה וּבְעִיר֑וֹ וְשָׁא֗וּל הֵסִ֛יר הָאֹב֥וֹת וְאֶת־ הַיִּדְּעֹנִ֖ים מֵהָאָֽרֶץ׃ (יז) וַיִּקָּבְצ֣וּ פְלִשְׁתִּ֔ים וַיָּבֹ֖אוּ וַיַּחֲנ֣וּ בְשׁוּנֵ֑ם וַיִּקְבֹּ֤ץ שָׁאוּל֙ אֶת־ כָּל־ יִשְׂרָאֵ֔ל וַֽיַּחֲנ֖וּ בַּגִּלְבֹּֽעַ׃ (יח) וַיַּ֥רְא שָׁא֖וּל אֶת־ מַחֲנֵ֣ה פְלִשְׁתִּ֑ים וַיִּרָ֕א וַיֶּחֱרַ֥ד לִבּ֖וֹ מְאֹֽד׃ (יט) וַיִּשְׁאַ֤ל שָׁאוּל֙ בַּֽיהוָ֔ה וְלֹ֥א עָנָ֖הוּ יְהוָ֑ה גַּ֧ם בַּחֲלֹמ֛וֹת גַּ֥ם בָּאוּרִ֖ים גַּ֥ם בַּנְּבִיאִֽם׃ (כ) וַיֹּ֨אמֶר שָׁא֜וּל לַעֲבָדָ֗יו בַּקְּשׁוּ־ לִי֙ אֵ֣שֶׁת בַּעֲלַת־ א֔וֹב וְאֵלְכָ֥ה אֵלֶ֖יהָ וְאֶדְרְשָׁה־ בָּ֑הּ וַיֹּאמְר֤וּ עֲבָדָיו֙ אֵלָ֔יו הִנֵּ֛ה אֵ֥שֶׁת בַּֽעֲלַת־ א֖וֹב בְּעֵ֥ין דּֽוֹר׃ (כא) וַיִּתְחַפֵּ֣שׂ שָׁא֗וּל וַיִּלְבַּשׁ֙ בְּגָדִ֣ים אֲחֵרִ֔ים וַיֵּ֣לֶךְ ה֗וּא וּשְׁנֵ֤י אֲנָשִׁים֙ עִמּ֔וֹ וַיָּבֹ֥אוּ אֶל־ הָאִשָּׁ֖ה לָ֑יְלָה וַיֹּ֗אמֶר (קסומי)־ קָֽסֳמִי־ נָ֥א לִי֙ בָּא֔וֹב וְהַ֣עֲלִי לִ֔י אֵ֥ת אֲשֶׁר־ אֹמַ֖ר אֵלָֽיִךְ׃ (כב) וַתֹּ֨אמֶר הָֽאִשָּׁ֜ה אֵלָ֗יו הִנֵּ֨ה אַתָּ֤ה יָדַ֙עְתָּ֙ אֵ֣ת אֲשֶׁר־ עָשָׂ֣ה שָׁא֔וּל אֲשֶׁ֥ר הִכְרִ֛ית אֶת־ הָאֹב֥וֹת וְאֶת־ הַיִּדְּעֹנִ֖י מִן־ הָאָ֑רֶץ וְלָמָ֥ה אַתָּ֛ה מִתְנַקֵּ֥שׁ בְּנַפְשִׁ֖י לַהֲמִיתֵֽנִי׃ (כג) וַיִּשָּׁ֤בַֽע לָהּ֙ שָׁא֔וּל בַּֽיהוָ֖ה לֵאמֹ֑ר חַי־ יְהוָ֕ה אִֽם־ יִקְּרֵ֥ךְ עָוֺ֖ן בַּדָּבָ֥ר הַזֶּֽה׃ (כד) וַתֹּ֙אמֶר֙ הָֽאִשָּׁ֔ה אֶת־ מִ֖י אַֽעֲלֶה־ לָּ֑ךְ וַיֹּ֕אמֶר אֶת־ שְׁמוּאֵ֖ל הַֽעֲלִי־ לִֽי׃ (כה) וַתֵּ֤רֶא הָֽאִשָּׁה֙ אֶת־ שְׁמוּאֵ֔ל וַתִּזְעַ֖ק בְּק֣וֹל גָּד֑וֹל וַתֹּאמֶר֩ הָאִשָּׁ֨ה אֶל־ שָׁא֧וּל לֵאמֹ֛ר לָ֥מָּה רִמִּיתָ֖נִי וְאַתָּ֥ה שָׁאֽוּל׃ (כו) וַיֹּ֨אמֶר לָ֥הּ הַמֶּ֛לֶךְ אַל־ תִּֽירְאִ֖י כִּ֣י מָ֣ה רָאִ֑ית וַתֹּ֤אמֶר הָֽאִשָּׁה֙ אֶל־ שָׁא֔וּל אֱלֹהִ֥ים רָאִ֖יתִי עֹלִ֥ים מִן־ הָאָֽרֶץ׃ (כז) וַיֹּ֤אמֶר לָהּ֙ מַֽה־ תָּאֳר֔וֹ וַתֹּ֗אמֶר אִ֤ישׁ זָקֵן֙ עֹלֶ֔ה וְה֥וּא עֹטֶ֖ה מְעִ֑יל וַיֵּ֤דַע שָׁאוּל֙ כִּֽי־ שְׁמוּאֵ֣ל ה֔וּא וַיִּקֹּ֥ד אַפַּ֛יִם אַ֖רְצָה וַיִּשְׁתָּֽחוּ׃ (כח) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ שָׁא֔וּל לָ֥מָּה הִרְגַּזְתַּ֖נִי לְהַעֲל֣וֹת אֹתִ֑י וַיֹּ֣אמֶר שָׁ֠אוּל צַר־ לִ֨י מְאֹ֜ד וּפְלִשְׁתִּ֣ים ׀ נִלְחָמִ֣ים בִּ֗י וֵֽאלֹהִ֞ים סָ֤ר מֵֽעָלַי֙ וְלֹֽא־ עָנָ֣נִי ע֗וֹד גַּ֤ם בְּיַֽד־ הַנְּבִיאִם֙ גַּם־ בַּ֣חֲלֹמ֔וֹת וָאֶקְרָאֶ֣ה לְךָ֔ לְהוֹדִיעֵ֖נִי מָ֥ה אֶעֱשֶֽׂה׃ (כט) וַיֹּ֣אמֶר שְׁמוּאֵ֔ל וְלָ֖מָּה תִּשְׁאָלֵ֑נִי וַיהוָ֛ה סָ֥ר מֵעָלֶ֖יךָ וַיְהִ֥י עָרֶֽךָ׃ (ל) וַיַּ֤עַשׂ יְהוָה֙ ל֔וֹ כַּאֲשֶׁ֖ר דִּבֶּ֣ר בְּיָדִ֑י וַיִּקְרַ֨ע יְהוָ֤ה אֶת־ הַמַּמְלָכָה֙ מִיָּדֶ֔ךָ וַֽיִּתְּנָ֖הּ לְרֵעֲךָ֥ לְדָוִֽד׃ (לא) כַּאֲשֶׁ֤ר לֹֽא־ שָׁמַ֙עְתָּ֙ בְּק֣וֹל יְהוָ֔ה וְלֹֽא־ עָשִׂ֥יתָ חֲרוֹן־ אַפּ֖וֹ בַּעֲמָלֵ֑ק עַל־ כֵּן֙ הַדָּבָ֣ר הַזֶּ֔ה עָשָֽׂה־ לְךָ֥ יְהוָ֖ה הַיּ֥וֹם הַזֶּֽה׃ (לב) וְיִתֵּ֣ן יְ֠הוָה גַּ֣ם אֶת־ יִשְׂרָאֵ֤ל עִמְּךָ֙ בְּיַד־ פְּלִשְׁתִּ֔ים וּמָחָ֕ר אַתָּ֥ה וּבָנֶ֖יךָ עִמִּ֑י גַּ֚ם אֶת־ מַחֲנֵ֣ה יִשְׂרָאֵ֔ל יִתֵּ֥ן יְהוָ֖ה בְּיַד־ פְּלִשְׁתִּֽים׃ (לג) וַיְמַהֵ֣ר שָׁא֗וּל וַיִּפֹּ֤ל מְלֹא־ קֽוֹמָתוֹ֙ אַ֔רְצָה וַיִּרָ֥א מְאֹ֖ד מִדִּבְרֵ֣י שְׁמוּאֵ֑ל גַּם־ כֹּ֙חַ֙ לֹא־ הָ֣יָה ב֔וֹ כִּ֣י לֹ֤א אָכַל֙ לֶ֔חֶם כָּל־ הַיּ֖וֹם וְכָל־ הַלָּֽיְלָה׃ (לד) וַתָּב֤וֹא הָֽאִשָּׁה֙ אֶל־ שָׁא֔וּל וַתֵּ֖רֶא כִּי־ נִבְהַ֣ל מְאֹ֑ד וַתֹּ֣אמֶר אֵלָ֗יו הִנֵּ֨ה שָׁמְעָ֤ה שִׁפְחָֽתְךָ֙ בְּקוֹלֶ֔ךָ וָאָשִׂ֤ים נַפְשִׁי֙ בְּכַפִּ֔י וָֽאֶשְׁמַע֙ אֶת־ דְּבָרֶ֔יךָ אֲשֶׁ֥ר דִּבַּ֖רְתָּ אֵלָֽי׃ (לה) וְעַתָּ֗ה שְׁמַֽע־ נָ֤א גַם־ אַתָּה֙ בְּק֣וֹל שִׁפְחָתֶ֔ךָ וְאָשִׂ֧מָה לְפָנֶ֛יךָ פַּת־ לֶ֖חֶם וֶאֱכ֑וֹל וִיהִ֤י בְךָ֙ כֹּ֔חַ כִּ֥י תֵלֵ֖ךְ בַּדָּֽרֶךְ׃ (לו) וַיְמָאֵ֗ן וַיֹּ֙אמֶר֙ לֹ֣א אֹכַ֔ל וַיִּפְרְצוּ־ ב֤וֹ עֲבָדָיו֙ וְגַם־ הָ֣אִשָּׁ֔ה וַיִּשְׁמַ֖ע לְקֹלָ֑ם וַיָּ֙קָם֙ מֵֽהָאָ֔רֶץ וַיֵּ֖שֶׁב אֶל־ הַמִּטָּֽה׃
