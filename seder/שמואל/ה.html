שמואל סדר ה

פתיחה באתר · ספר שמואל כולו ·

(א) וְלוֹ־ הָיָ֨ה בֵ֜ן וּשְׁמ֤וֹ שָׁאוּל֙ בָּח֣וּר וָט֔וֹב וְאֵ֥ין אִ֛ישׁ מִבְּנֵ֥י יִשְׂרָאֵ֖ל ט֣וֹב מִמֶּ֑נּוּ מִשִּׁכְמ֣וֹ וָמַ֔עְלָה גָּבֹ֖הַּ מִכָּל־ הָעָֽם׃ (ב) וַתֹּאבַ֙דְנָה֙ הָאֲתֹנ֔וֹת לְקִ֖ישׁ אֲבִ֣י שָׁא֑וּל וַיֹּ֨אמֶר קִ֜ישׁ אֶל־ שָׁא֣וּל בְּנ֗וֹ קַח־ נָ֤א אִתְּךָ֙ אֶת־ אַחַ֣ד מֵֽהַנְּעָרִ֔ים וְק֣וּם לֵ֔ךְ בַּקֵּ֖שׁ אֶת־ הָאֲתֹנֹֽת׃ (ג) וַיַּעֲבֹ֧ר בְּהַר־ אֶפְרַ֛יִם וַיַּעֲבֹ֥ר בְּאֶֽרֶץ־ שָׁלִ֖שָׁה וְלֹ֣א מָצָ֑אוּ וַיַּעַבְר֤וּ בְאֶֽרֶץ־ שַׁעֲלִים֙ וָאַ֔יִן וַיַּעֲבֹ֥ר בְּאֶֽרֶץ־ יְמִינִ֖י וְלֹ֥א מָצָֽאוּ׃ (ד) הֵ֗מָּה בָּ֚אוּ בְּאֶ֣רֶץ צ֔וּף וְשָׁא֥וּל אָמַ֛ר לְנַעֲר֥וֹ אֲשֶׁר־ עִמּ֖וֹ לְכָ֣ה וְנָשׁ֑וּבָה פֶּן־ יֶחְדַּ֥ל אָבִ֛י מִן־ הָאֲתֹנ֖וֹת וְדָ֥אַג לָֽנוּ׃ (ה) וַיֹּ֣אמֶר ל֗וֹ הִנֵּה־ נָ֤א אִישׁ־ אֱלֹהִים֙ בָּעִ֣יר הַזֹּ֔את וְהָאִ֣ישׁ נִכְבָּ֔ד כֹּ֥ל אֲשֶׁר־ יְדַבֵּ֖ר בּ֣וֹא יָב֑וֹא עַתָּה֙ נֵ֣לֲכָה שָּׁ֔ם אוּלַי֙ יַגִּ֣יד לָ֔נוּ אֶת־ דַּרְכֵּ֖נוּ אֲשֶׁר־ הָלַ֥כְנוּ עָלֶֽיהָ׃ (ו) וַיֹּ֨אמֶר שָׁא֜וּל לְנַעֲר֗וֹ וְהִנֵּ֣ה נֵלֵךְ֮ וּמַה־ נָּבִ֣יא לָאִישׁ֒ כִּ֤י הַלֶּ֙חֶם֙ אָזַ֣ל מִכֵּלֵ֔ינוּ וּתְשׁוּרָ֥ה אֵין־ לְהָבִ֖יא לְאִ֣ישׁ הָאֱלֹהִ֑ים מָ֖ה אִתָּֽנוּ׃ (ז) וַיֹּ֤סֶף הַנַּ֙עַר֙ לַעֲנ֣וֹת אֶת־ שָׁא֔וּל וַיֹּ֕אמֶר הִנֵּה֙ נִמְצָ֣א בְיָדִ֔י רֶ֖בַע שֶׁ֣קֶל כָּ֑סֶף וְנָֽתַתִּי֙ לְאִ֣ישׁ הָאֱלֹהִ֔ים וְהִגִּ֥יד לָ֖נוּ אֶת־ דַּרְכֵּֽנוּ׃ (ח) לְפָנִ֣ים ׀ בְּיִשְׂרָאֵ֗ל כֹּֽה־ אָמַ֤ר הָאִישׁ֙ בְּלֶכְתּוֹ֙ לִדְר֣וֹשׁ אֱלֹהִ֔ים לְכ֥וּ וְנֵלְכָ֖ה עַד־ הָרֹאֶ֑ה כִּ֤י לַנָּבִיא֙ הַיּ֔וֹם יִקָּרֵ֥א לְפָנִ֖ים הָרֹאֶֽה׃ (ט) וַיֹּ֨אמֶר שָׁא֧וּל לְנַעֲר֛וֹ ט֥וֹב דְּבָרְךָ֖ לְכָ֣ה ׀ נֵלֵ֑כָה וַיֵּֽלְכוּ֙ אֶל־ הָעִ֔יר אֲשֶׁר־ שָׁ֖ם אִ֥ישׁ הָאֱלֹהִֽים׃ (י) הֵ֗מָּה עֹלִים֙ בְּמַעֲלֵ֣ה הָעִ֔יר וְהֵ֙מָּה֙ מָצְא֣וּ נְעָר֔וֹת יֹצְא֖וֹת לִשְׁאֹ֣ב מָ֑יִם וַיֹּאמְר֣וּ לָהֶ֔ן הֲיֵ֥שׁ בָּזֶ֖ה הָרֹאֶֽה׃ (יא) וַתַּעֲנֶ֧ינָה אוֹתָ֛ם וַתֹּאמַ֥רְנָה יֵּ֖שׁ הִנֵּ֣ה לְפָנֶ֑יךָ מַהֵ֣ר ׀ עַתָּ֗ה כִּ֤י הַיּוֹם֙ בָּ֣א לָעִ֔יר כִּ֣י זֶ֧בַח הַיּ֛וֹם לָעָ֖ם בַּבָּמָֽה׃ (יב) כְּבֹאֲכֶ֣ם הָעִ֣יר כֵּ֣ן תִּמְצְא֣וּן אֹת֡וֹ בְּטֶרֶם֩ יַעֲלֶ֨ה הַבָּמָ֜תָה לֶאֱכֹ֗ל כִּ֠י לֹֽא־ יֹאכַ֤ל הָעָם֙ עַד־ בֹּא֔וֹ כִּֽי־ הוּא֙ יְבָרֵ֣ךְ הַזֶּ֔בַח אַחֲרֵי־ כֵ֖ן יֹאכְל֣וּ הַקְּרֻאִ֑ים וְעַתָּ֣ה עֲל֔וּ כִּֽי־ אֹת֥וֹ כְהַיּ֖וֹם תִּמְצְא֥וּן אֹתֽוֹ׃ (יג) וַֽיַּעֲל֖וּ הָעִ֑יר הֵ֗מָּה בָּאִים֙ בְּת֣וֹךְ הָעִ֔יר וְהִנֵּ֤ה שְׁמוּאֵל֙ יֹצֵ֣א לִקְרָאתָ֔ם לַעֲל֖וֹת הַבָּמָֽה׃ (יד) וַֽיהוָ֔ה גָּלָ֖ה אֶת־ אֹ֣זֶן שְׁמוּאֵ֑ל י֣וֹם אֶחָ֔ד לִפְנֵ֥י בֽוֹא־ שָׁא֖וּל לֵאמֹֽר׃ (טו) כָּעֵ֣ת ׀ מָחָ֡ר אֶשְׁלַח֩ אֵלֶ֨יךָ אִ֜ישׁ מֵאֶ֣רֶץ בִּנְיָמִ֗ן וּמְשַׁחְתּ֤וֹ לְנָגִיד֙ עַל־ עַמִּ֣י יִשְׂרָאֵ֔ל וְהוֹשִׁ֥יעַ אֶת־ עַמִּ֖י מִיַּ֣ד פְּלִשְׁתִּ֑ים כִּ֤י רָאִ֙יתִי֙ אֶת־ עַמִּ֔י כִּ֛י בָּ֥אָה צַעֲקָת֖וֹ אֵלָֽי׃ (טז) וּשְׁמוּאֵ֖ל רָאָ֣ה אֶת־ שָׁא֑וּל וַיהוָ֣ה עָנָ֔הוּ הִנֵּ֤ה הָאִישׁ֙ אֲשֶׁ֣ר אָמַ֣רְתִּי אֵלֶ֔יךָ זֶ֖ה יַעְצֹ֥ר בְּעַמִּֽי׃ (יז) וַיִּגַּ֥שׁ שָׁא֛וּל אֶת־ שְׁמוּאֵ֖ל בְּת֣וֹךְ הַשָּׁ֑עַר וַיֹּ֙אמֶר֙ הַגִּֽידָה־ נָּ֣א לִ֔י אֵי־ זֶ֖ה בֵּ֥ית הָרֹאֶֽה׃ (יח) וַיַּ֨עַן שְׁמוּאֵ֜ל אֶת־ שָׁא֗וּל וַיֹּ֙אמֶר֙ אָנֹכִ֣י הָרֹאֶ֔ה עֲלֵ֤ה לְפָנַי֙ הַבָּמָ֔ה וַאֲכַלְתֶּ֥ם עִמִּ֖י הַיּ֑וֹם וְשִׁלַּחְתִּ֣יךָ בַבֹּ֔קֶר וְכֹ֛ל אֲשֶׁ֥ר בִּֽלְבָבְךָ֖ אַגִּ֥יד לָֽךְ׃ (יט) וְלָאֲתֹנ֞וֹת הָאֹבְד֣וֹת לְךָ֗ הַיּוֹם֙ שְׁלֹ֣שֶׁת הַיָּמִ֔ים אַל־ תָּ֧שֶׂם אֶֽת־ לִבְּךָ֛ לָהֶ֖ם כִּ֣י נִמְצָ֑אוּ וּלְמִי֙ כָּל־ חֶמְדַּ֣ת יִשְׂרָאֵ֔ל הֲל֣וֹא לְךָ֔ וּלְכֹ֖ל בֵּ֥ית אָבִֽיךָ׃ (כ) וַיַּ֨עַן שָׁא֜וּל וַיֹּ֗אמֶר הֲל֨וֹא בֶן־ יְמִינִ֤י אָ֙נֹכִי֙ מִקַּטַנֵּי֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל וּמִשְׁפַּחְתִּי֙ הַצְּעִרָ֔ה מִכָּֽל־ מִשְׁפְּח֖וֹת שִׁבְטֵ֣י בִנְיָמִ֑ן וְלָ֙מָּה֙ דִּבַּ֣רְתָּ אֵלַ֔י כַּדָּבָ֖ר הַזֶּֽה׃ (כא) וַיִּקַּ֤ח שְׁמוּאֵל֙ אֶת־ שָׁא֣וּל וְאֶֽת־ נַעֲר֔וֹ וַיְבִיאֵ֖ם לִשְׁכָּ֑תָה וַיִּתֵּ֨ן לָהֶ֤ם מָקוֹם֙ בְּרֹ֣אשׁ הַקְּרוּאִ֔ים וְהֵ֖מָּה כִּשְׁלֹשִׁ֥ים אִֽישׁ׃ (כב) וַיֹּ֤אמֶר שְׁמוּאֵל֙ לַטַּבָּ֔ח תְּנָה֙ אֶת־ הַמָּנָ֔ה אֲשֶׁ֥ר נָתַ֖תִּי לָ֑ךְ אֲשֶׁר֙ אָמַ֣רְתִּי אֵלֶ֔יךָ שִׂ֥ים אֹתָ֖הּ עִמָּֽךְ׃ (כג) וַיָּ֣רֶם הַ֠טַּבָּח אֶת־ הַשּׁ֨וֹק וְהֶעָלֶ֜יהָ וַיָּ֣שֶׂם ׀ לִפְנֵ֣י שָׁא֗וּל וַיֹּ֙אמֶר֙ הִנֵּ֤ה הַנִּשְׁאָר֙ שִׂים־ לְפָנֶ֣יךָ אֱכֹ֔ל כִּ֧י לַמּוֹעֵ֛ד שָֽׁמוּר־ לְךָ֥ לֵאמֹ֖ר הָעָ֣ם ׀ קָרָ֑אתִי וַיֹּ֧אכַל שָׁא֛וּל עִם־ שְׁמוּאֵ֖ל בַּיּ֥וֹם הַהֽוּא׃ (כד) וַיֵּרְד֥וּ מֵהַבָּמָ֖ה הָעִ֑יר וַיְדַבֵּ֥ר עִם־ שָׁא֖וּל עַל־ הַגָּֽג׃ (כה) וַיַּשְׁכִּ֗מוּ וַיְהִ֞י כַּעֲל֤וֹת הַשַּׁ֙חַר֙ וַיִּקְרָ֨א שְׁמוּאֵ֤ל אֶל־ שָׁאוּל֙ (הגג) הַגָּ֣גָה לֵאמֹ֔ר ק֖וּמָה וַאֲשַׁלְּחֶ֑ךָּ וַיָּ֣קָם שָׁא֗וּל וַיֵּצְא֧וּ שְׁנֵיהֶ֛ם ה֥וּא וּשְׁמוּאֵ֖ל הַחֽוּצָה׃ (כו) הֵ֗מָּה יֽוֹרְדִים֙ בִּקְצֵ֣ה הָעִ֔יר וּשְׁמוּאֵ֞ל אָמַ֣ר אֶל־ שָׁא֗וּל אֱמֹ֥ר לַנַּ֛עַר וְיַעֲבֹ֥ר לְפָנֵ֖ינוּ וַֽיַּעֲבֹ֑ר וְאַתָּה֙ עֲמֹ֣ד כַּיּ֔וֹם וְאַשְׁמִיעֲךָ֖ אֶת־ דְּבַ֥ר אֱלֹהִֽים׃ (כז) וַיִּקַּ֨ח שְׁמוּאֵ֜ל אֶת־ פַּ֥ךְ הַשֶּׁ֛מֶן וַיִּצֹ֥ק עַל־ רֹאשׁ֖וֹ וַיִּשָּׁקֵ֑הוּ וַיֹּ֕אמֶר הֲל֗וֹא כִּֽי־ מְשָׁחֲךָ֧ יְהוָ֛ה עַל־ נַחֲלָת֖וֹ לְנָגִֽיד׃ (כח) בְּלֶכְתְּךָ֤ הַיּוֹם֙ מֵעִמָּדִ֔י וּמָצָאתָ֩ שְׁנֵ֨י אֲנָשִׁ֜ים עִם־ קְבֻרַ֥ת רָחֵ֛ל בִּגְב֥וּל בִּנְיָמִ֖ן בְּצֶלְצַ֑ח וְאָמְר֣וּ אֵלֶ֗יךָ נִמְצְא֤וּ הָאֲתֹנוֹת֙ אֲשֶׁ֣ר הָלַ֣כְתָּ לְבַקֵּ֔שׁ וְהִנֵּ֨ה נָטַ֤שׁ אָבִ֙יךָ֙ אֶת־ דִּבְרֵ֣י הָאֲתֹנ֔וֹת וְדָאַ֤ג לָכֶם֙ לֵאמֹ֔ר מָ֥ה אֶעֱשֶׂ֖ה לִבְנִֽי׃ (כט) וְחָלַפְתָּ֨ מִשָּׁ֜ם וָהָ֗לְאָה וּבָ֙אתָ֙ עַד־ אֵל֣וֹן תָּב֔וֹר וּמְצָא֤וּךָ שָּׁם֙ שְׁלֹשָׁ֣ה אֲנָשִׁ֔ים עֹלִ֥ים אֶל־ הָאֱלֹהִ֖ים בֵּֽית־ אֵ֑ל אֶחָ֞ד נֹשֵׂ֣א ׀ שְׁלֹשָׁ֣ה גְדָיִ֗ים וְאֶחָד֙ נֹשֵׂ֗א שְׁלֹ֙שֶׁת֙ כִּכְּר֣וֹת לֶ֔חֶם וְאֶחָ֥ד נֹשֵׂ֖א נֵֽבֶל־ יָֽיִן׃ (ל) וְשָׁאֲל֥וּ לְךָ֖ לְשָׁל֑וֹם וְנָתְנ֤וּ לְךָ֙ שְׁתֵּי־ לֶ֔חֶם וְלָקַחְתָּ֖ מִיָּדָֽם׃ (לא) אַ֣חַר כֵּ֗ן תָּבוֹא֙ גִּבְעַ֣ת הָאֱלֹהִ֔ים אֲשֶׁר־ שָׁ֖ם נְצִבֵ֣י פְלִשְׁתִּ֑ים וִיהִי֩ כְבֹאֲךָ֨ שָׁ֜ם הָעִ֗יר וּפָגַעְתָּ֞ חֶ֤בֶל נְבִיאִים֙ יֹרְדִ֣ים מֵֽהַבָּמָ֔ה וְלִפְנֵיהֶ֞ם נֵ֤בֶל וְתֹף֙ וְחָלִ֣יל וְכִנּ֔וֹר וְהֵ֖מָּה מִֽתְנַבְּאִֽים׃ (לב) וְצָלְחָ֤ה עָלֶ֙יךָ֙ ר֣וּחַ יְהוָ֔ה וְהִתְנַבִּ֖יתָ עִמָּ֑ם וְנֶהְפַּכְתָּ֖ לְאִ֥ישׁ אַחֵֽר׃ (לג) וְהָיָ֗ה כִּ֥י (תבאינה) תָבֹ֛אנָה הָאֹת֥וֹת הָאֵ֖לֶּה לָ֑ךְ עֲשֵׂ֤ה לְךָ֙ אֲשֶׁ֣ר תִּמְצָ֣א יָדֶ֔ךָ כִּ֥י הָאֱלֹהִ֖ים עִמָּֽךְ׃ (לד) וְיָרַדְתָּ֣ לְפָנַי֮ הַגִּלְגָּל֒ וְהִנֵּ֤ה אָֽנֹכִי֙ יֹרֵ֣ד אֵלֶ֔יךָ לְהַעֲל֣וֹת עֹל֔וֹת לִזְבֹּ֖חַ זִבְחֵ֣י שְׁלָמִ֑ים שִׁבְעַ֨ת יָמִ֤ים תּוֹחֵל֙ עַד־ בּוֹאִ֣י אֵלֶ֔יךָ וְהוֹדַעְתִּ֣י לְךָ֔ אֵ֖ת אֲשֶׁ֥ר תַּעֲשֶֽׂה׃ (לה) וְהָיָ֗ה כְּהַפְנֹת֤וֹ שִׁכְמוֹ֙ לָלֶ֙כֶת֙ מֵעִ֣ם שְׁמוּאֵ֔ל וַיַּהֲפָךְ־ ל֥וֹ אֱלֹהִ֖ים לֵ֣ב אַחֵ֑ר וַיָּבֹ֛אוּ כָּל־ הָאֹת֥וֹת הָאֵ֖לֶּה בַּיּ֥וֹם הַהֽוּא׃ (לו) וַיָּבֹ֤אוּ שָׁם֙ הַגִּבְעָ֔תָה וְהִנֵּ֥ה חֶֽבֶל־ נְבִאִ֖ים לִקְרָאת֑וֹ וַתִּצְלַ֤ח עָלָיו֙ ר֣וּחַ אֱלֹהִ֔ים וַיִּתְנַבֵּ֖א בְּתוֹכָֽם׃ (לז) וַיְהִ֗י כָּל־ יֽוֹדְעוֹ֙ מֵאִתְּמ֣וֹל שִׁלְשׁ֔וֹם וַיִּרְא֕וּ וְהִנֵּ֥ה עִם־ נְבִאִ֖ים נִבָּ֑א וַיֹּ֨אמֶר הָעָ֜ם אִ֣ישׁ אֶל־ רֵעֵ֗הוּ מַה־ זֶּה֙ הָיָ֣ה לְבֶן־ קִ֔ישׁ הֲגַ֥ם שָׁא֖וּל בַּנְּבִיאִֽים׃ (לח) וַיַּ֨עַן אִ֥ישׁ מִשָּׁ֛ם וַיֹּ֖אמֶר וּמִ֣י אֲבִיהֶ֑ם עַל־ כֵּן֙ הָיְתָ֣ה לְמָשָׁ֔ל הֲגַ֥ם שָׁא֖וּל בַּנְּבִאִֽים׃ (לט) וַיְכַל֙ מֵֽהִתְנַבּ֔וֹת וַיָּבֹ֖א הַבָּמָֽה׃ (מ) וַיֹּאמֶר֩ דּ֨וֹד שָׁא֥וּל אֵלָ֛יו וְאֶֽל־ נַעֲר֖וֹ אָ֣ן הֲלַכְתֶּ֑ם וַיֹּ֕אמֶר לְבַקֵּשׁ֙ אֶת־ הָ֣אֲתֹנ֔וֹת וַנִּרְאֶ֣ה כִי־ אַ֔יִן וַנָּב֖וֹא אֶל־ שְׁמוּאֵֽל׃ (מא) וַיֹּ֖אמֶר דּ֣וֹד שָׁא֑וּל הַגִּֽידָה־ נָּ֣א לִ֔י מָֽה־ אָמַ֥ר לָכֶ֖ם שְׁמוּאֵֽל׃ (מב) וַיֹּ֤אמֶר שָׁאוּל֙ אֶל־ דּוֹד֔וֹ הַגֵּ֤ד הִגִּיד֙ לָ֔נוּ כִּ֥י נִמְצְא֖וּ הָאֲתֹנ֑וֹת וְאֶת־ דְּבַ֤ר הַמְּלוּכָה֙ לֹֽא־ הִגִּ֣יד ל֔וֹ אֲשֶׁ֖ר אָמַ֥ר שְׁמוּאֵֽל׃ (מג) וַיַּצְעֵ֤ק שְׁמוּאֵל֙ אֶת־ הָעָ֔ם אֶל־ יְהוָ֖ה הַמִּצְפָּֽה׃ (מד) וַיֹּ֣אמֶר ׀ אֶל־ בְּנֵ֣י יִשְׂרָאֵ֗ל כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל אָנֹכִ֛י הֶעֱלֵ֥יתִי אֶת־ יִשְׂרָאֵ֖ל מִמִּצְרָ֑יִם וָאַצִּ֤יל אֶתְכֶם֙ מִיַּ֣ד מִצְרַ֔יִם וּמִיַּד֙ כָּל־ הַמַּמְלָכ֔וֹת הַלֹּחֲצִ֖ים אֶתְכֶֽם׃ (מה) וְאַתֶּ֨ם הַיּ֜וֹם מְאַסְתֶּ֣ם אֶת־ אֱלֹהֵיכֶ֗ם אֲשֶׁר־ ה֣וּא מוֹשִׁ֣יעַ לָכֶם֮ מִכָּל־ רָעוֹתֵיכֶ֣ם וְצָרֹֽתֵיכֶם֒ וַתֹּ֣אמְרוּ ל֔וֹ כִּי־ מֶ֖לֶךְ תָּשִׂ֣ים עָלֵ֑ינוּ וְעַתָּ֗ה הִֽתְיַצְּבוּ֙ לִפְנֵ֣י יְהוָ֔ה לְשִׁבְטֵיכֶ֖ם וּלְאַלְפֵיכֶֽם׃ (מו) וַיַּקְרֵ֣ב שְׁמוּאֵ֔ל אֵ֖ת כָּל־ שִׁבְטֵ֣י יִשְׂרָאֵ֑ל וַיִּלָּכֵ֖ד שֵׁ֥בֶט בִּנְיָמִֽן׃ (מז) וַיַּקְרֵ֞ב אֶת־ שֵׁ֤בֶט בִּנְיָמִן֙ (למשפחתו) לְמִשְׁפְּחֹתָ֔יו וַתִּלָּכֵ֖ד מִשְׁפַּ֣חַת הַמַּטְרִ֑י וַיִּלָּכֵד֙ שָׁא֣וּל בֶּן־ קִ֔ישׁ וַיְבַקְשֻׁ֖הוּ וְלֹ֥א נִמְצָֽא׃ (מח) וַיִּשְׁאֲלוּ־ עוֹד֙ בַּֽיהוָ֔ה הֲבָ֥א ע֖וֹד הֲלֹ֣ם אִ֑ישׁ וַיֹּ֣אמֶר יְהוָ֔ה הִנֵּה־ ה֥וּא נֶחְבָּ֖א אֶל־ הַכֵּלִֽים׃ (מט) וַיָּרֻ֙צוּ֙ וַיִּקָּחֻ֣הוּ מִשָּׁ֔ם וַיִּתְיַצֵּ֖ב בְּת֣וֹךְ הָעָ֑ם וַיִּגְבַּהּ֙ מִכָּל־ הָעָ֔ם מִשִּׁכְמ֖וֹ וָמָֽעְלָה׃
