שמואל סדר י

לפי החלוקה לפרקים: שמואל א טז, יח – יז, לו · 42 פסוקים

פתיחה באתר · ספר שמואל כולו ·

(א) וַיַּעַן֩ אֶחָ֨ד מֵהַנְּעָרִ֜ים וַיֹּ֗אמֶר הִנֵּ֨ה רָאִ֜יתִי בֵּ֣ן לְיִשַׁי֮ בֵּ֣ית הַלַּחְמִי֒ יֹדֵ֣עַ נַ֠גֵּן וְגִבּ֨וֹר חַ֜יִל וְאִ֧ישׁ מִלְחָמָ֛ה וּנְב֥וֹן דָּבָ֖ר וְאִ֣ישׁ תֹּ֑אַר וַיהוָ֖ה עִמּֽוֹ׃ (ב) וַיִּשְׁלַ֥ח שָׁא֛וּל מַלְאָכִ֖ים אֶל־ יִשָׁ֑י וַיֹּ֕אמֶר שִׁלְחָ֥ה אֵלַ֛י אֶת־ דָּוִ֥ד בִּנְךָ֖ אֲשֶׁ֥ר בַּצֹּֽאן׃ (ג) וַיִּקַּ֨ח יִשַׁ֜י חֲמ֥וֹר לֶ֙חֶם֙ וְנֹ֣אד יַ֔יִן וּגְדִ֥י עִזִּ֖ים אֶחָ֑ד וַיִּשְׁלַ֛ח בְּיַד־ דָּוִ֥ד בְּנ֖וֹ אֶל־ שָׁאֽוּל׃ (ד) וַיָּבֹ֤א דָוִד֙ אֶל־ שָׁא֔וּל וַֽיַּעֲמֹ֖ד לְפָנָ֑יו וַיֶּאֱהָבֵ֣הֽוּ מְאֹ֔ד וַֽיְהִי־ ל֖וֹ נֹשֵׂ֥א כֵלִֽים׃ (ה) וַיִּשְׁלַ֣ח שָׁא֔וּל אֶל־ יִשַׁ֖י לֵאמֹ֑ר יַעֲמָד־ נָ֤א דָוִד֙ לְפָנַ֔י כִּֽי־ מָ֥צָא חֵ֖ן בְּעֵינָֽי׃ (ו) וְהָיָ֗ה בִּֽהְי֤וֹת רֽוּחַ־ אֱלֹהִים֙ אֶל־ שָׁא֔וּל וְלָקַ֥ח דָּוִ֛ד אֶת־ הַכִּנּ֖וֹר וְנִגֵּ֣ן בְּיָד֑וֹ וְרָוַ֤ח לְשָׁאוּל֙ וְט֣וֹב ל֔וֹ וְסָ֥רָה מֵעָלָ֖יו ר֥וּחַ הָרָעָֽה׃ (ז) וַיַּאַסְפ֨וּ פְלִשְׁתִּ֤ים אֶת־ מַֽחֲנֵיהֶם֙ לַמִּלְחָמָ֔ה וַיֵּאָ֣סְפ֔וּ שֹׂכֹ֖ה אֲשֶׁ֣ר לִיהוּדָ֑ה וַֽיַּחֲנ֛וּ בֵּין־ שׂוֹכֹ֥ה וּבֵין־ עֲזֵקָ֖ה בְּאֶ֥פֶס דַּמִּֽים׃ (ח) וְשָׁא֤וּל וְאִֽישׁ־ יִשְׂרָאֵל֙ נֶאֶסְפ֔וּ וַֽיַּחֲנ֖וּ בְּעֵ֣מֶק הָאֵלָ֑ה וַיַּעַרְכ֥וּ מִלְחָמָ֖ה לִקְרַ֥את פְּלִשְׁתִּֽים׃ (ט) וּפְלִשְׁתִּ֞ים עֹמְדִ֤ים אֶל־ הָהָר֙ מִזֶּ֔ה וְיִשְׂרָאֵ֛ל עֹמְדִ֥ים אֶל־ הָהָ֖ר מִזֶּ֑ה וְהַגַּ֖יְא בֵּינֵיהֶֽם׃ (י) וַיֵּצֵ֤א אִֽישׁ־ הַבֵּנַ֙יִם֙ מִמַּחֲנ֣וֹת פְּלִשְׁתִּ֔ים גָּלְיָ֥ת שְׁמ֖וֹ מִגַּ֑ת גָּבְה֕וֹ שֵׁ֥שׁ אַמּ֖וֹת וָזָֽרֶת׃ (יא) וְכ֤וֹבַע נְחֹ֙שֶׁת֙ עַל־ רֹאשׁ֔וֹ וְשִׁרְי֥וֹן קַשְׂקַשִּׂ֖ים ה֣וּא לָב֑וּשׁ וּמִשְׁקַל֙ הַשִּׁרְי֔וֹן חֲמֵשֶׁת־ אֲלָפִ֥ים שְׁקָלִ֖ים נְחֹֽשֶֽׁת׃ (יב) וּמִצְחַ֥ת נְחֹ֖שֶׁת עַל־ רַגְלָ֑יו וְכִיד֥וֹן נְחֹ֖שֶׁת בֵּ֥ין כְּתֵפָֽיו׃ (יג) (וחץ) וְעֵ֣ץ חֲנִית֗וֹ כִּמְנוֹר֙ אֹֽרְגִ֔ים וְלַהֶ֣בֶת חֲנִית֔וֹ שֵׁשׁ־ מֵא֥וֹת שְׁקָלִ֖ים בַּרְזֶ֑ל וְנֹשֵׂ֥א הַצִּנָּ֖ה הֹלֵ֥ךְ לְפָנָֽיו׃ (יד) וַֽיַּעֲמֹ֗ד וַיִּקְרָא֙ אֶל־ מַעַרְכֹ֣ת יִשְׂרָאֵ֔ל וַיֹּ֣אמֶר לָהֶ֔ם לָ֥מָּה תֵצְא֖וּ לַעֲרֹ֣ךְ מִלְחָמָ֑ה הֲל֧וֹא אָנֹכִ֣י הַפְּלִשְׁתִּ֗י וְאַתֶּם֙ עֲבָדִ֣ים לְשָׁא֔וּל בְּרוּ־ לָכֶ֥ם אִ֖ישׁ וְיֵרֵ֥ד אֵלָֽי׃ (טו) אִם־ יוּכַ֞ל לְהִלָּחֵ֤ם אִתִּי֙ וְהִכָּ֔נִי וְהָיִ֥ינוּ לָכֶ֖ם לַעֲבָדִ֑ים וְאִם־ אֲנִ֤י אֽוּכַל־ לוֹ֙ וְהִכִּיתִ֔יו וִהְיִ֤יתֶם לָ֙נוּ֙ לַעֲבָדִ֔ים וַעֲבַדְתֶּ֖ם אֹתָֽנוּ׃ (טז) וַיֹּ֙אמֶר֙ הַפְּלִשְׁתִּ֔י אֲנִ֗י חֵרַ֛פְתִּי אֶת־ מַעַרְכ֥וֹת יִשְׂרָאֵ֖ל הַיּ֣וֹם הַזֶּ֑ה תְּנוּ־ לִ֣י אִ֔ישׁ וְנִֽלָּחֲמָ֖ה יָֽחַד׃ (יז) וַיִּשְׁמַ֤ע שָׁאוּל֙ וְכָל־ יִשְׂרָאֵ֔ל אֶת־ דִּבְרֵ֥י הַפְּלִשְׁתִּ֖י הָאֵ֑לֶּה וַיֵּחַ֥תּוּ וַיִּֽרְא֖וּ מְאֹֽד׃ (יח) וְדָוִד֩ בֶּן־ אִ֨ישׁ אֶפְרָתִ֜י הַזֶּ֗ה מִבֵּ֥ית לֶ֙חֶם֙ יְהוּדָ֔ה וּשְׁמ֣וֹ יִשַׁ֔י וְל֖וֹ שְׁמֹנָ֣ה בָנִ֑ים וְהָאִישׁ֙ בִּימֵ֣י שָׁא֔וּל זָקֵ֖ן בָּ֥א בַאֲנָשִֽׁים׃ (יט) וַיֵּ֨לְכ֜וּ שְׁלֹ֤שֶׁת בְּנֵֽי־ יִשַׁי֙ הַגְּדֹלִ֔ים הָלְכ֥וּ אַחֲרֵי־ שָׁא֖וּל לַמִּלְחָמָ֑ה וְשֵׁ֣ם ׀ שְׁלֹ֣שֶׁת בָּנָ֗יו אֲשֶׁ֤ר הָלְכוּ֙ בַּמִּלְחָמָ֔ה אֱלִיאָ֣ב הַבְּכ֗וֹר וּמִשְׁנֵ֙הוּ֙ אֲבִ֣ינָדָ֔ב וְהַשְּׁלִשִׁ֖י שַׁמָּֽה׃ (כ) וְדָוִ֖ד ה֣וּא הַקָּטָ֑ן וּשְׁלֹשָׁה֙ הַגְּדֹלִ֔ים הָלְכ֖וּ אַחֲרֵ֥י שָׁאֽוּל׃ (כא) וְדָוִ֛ד הֹלֵ֥ךְ וָשָׁ֖ב מֵעַ֣ל שָׁא֑וּל לִרְע֛וֹת אֶת־ צֹ֥אן אָבִ֖יו בֵּֽית־ לָֽחֶם׃ (כב) וַיִּגַּ֥שׁ הַפְּלִשְׁתִּ֖י הַשְׁכֵּ֣ם וְהַעֲרֵ֑ב וַיִּתְיַצֵּ֖ב אַרְבָּעִ֥ים יֽוֹם׃ (כג) וַיֹּ֨אמֶר יִשַׁ֜י לְדָוִ֣ד בְּנ֗וֹ קַח־ נָ֤א לְאַחֶ֙יךָ֙ אֵיפַ֤ת הַקָּלִיא֙ הַזֶּ֔ה וַעֲשָׂרָ֥ה לֶ֖חֶם הַזֶּ֑ה וְהָרֵ֥ץ הַֽמַּחֲנֶ֖ה לְאַחֶֽיךָ׃ (כד) וְ֠אֵת עֲשֶׂ֜רֶת חֲרִצֵ֤י הֶֽחָלָב֙ הָאֵ֔לֶּה תָּבִ֖יא לְשַׂר־ הָאָ֑לֶף וְאֶת־ אַחֶ֙יךָ֙ תִּפְקֹ֣ד לְשָׁל֔וֹם וְאֶת־ עֲרֻבָּתָ֖ם תִּקָּֽח׃ (כה) וְשָׁא֤וּל וְהֵ֙מָּה֙ וְכָל־ אִ֣ישׁ יִשְׂרָאֵ֔ל בְּעֵ֖מֶק הָֽאֵלָ֑ה נִלְחָמִ֖ים עִם־ פְּלִשְׁתִּֽים׃ (כו) וַיַּשְׁכֵּ֨ם דָּוִ֜ד בַּבֹּ֗קֶר וַיִּטֹּ֤שׁ אֶת־ הַצֹּאן֙ עַל־ שֹׁמֵ֔ר וַיִּשָּׂ֣א וַיֵּ֔לֶךְ כַּאֲשֶׁ֥ר צִוָּ֖הוּ יִשָׁ֑י וַיָּבֹא֙ הַמַּעְגָּ֔לָה וְהַחַ֗יִל הַיֹּצֵא֙ אֶל־ הַמַּ֣עֲרָכָ֔ה וְהֵרֵ֖עוּ בַּמִּלְחָמָֽה׃ (כז) וַתַּעֲרֹ֤ךְ יִשְׂרָאֵל֙ וּפְלִשְׁתִּ֔ים מַעֲרָכָ֖ה לִקְרַ֥את מַעֲרָכָֽה׃ (כח) וַיִּטֹּשׁ֩ דָּוִ֨ד אֶת־ הַכֵּלִ֜ים מֵעָלָ֗יו עַל־ יַד֙ שׁוֹמֵ֣ר הַכֵּלִ֔ים וַיָּ֖רָץ הַמַּעֲרָכָ֑ה וַיָּבֹ֕א וַיִּשְׁאַ֥ל לְאֶחָ֖יו לְשָׁלֽוֹם׃ (כט) וְה֣וּא ׀ מְדַבֵּ֣ר עִמָּ֗ם וְהִנֵּ֣ה אִ֣ישׁ הַבֵּנַ֡יִם עוֹלֶ֞ה גָּלְיָת֩ הַפְּלִשְׁתִּ֨י שְׁמ֤וֹ מִגַּת֙ (ממערות) מִמַּעַרְכ֣וֹת פְּלִשְׁתִּ֔ים וַיְדַבֵּ֖ר כַּדְּבָרִ֣ים הָאֵ֑לֶּה וַיִּשְׁמַ֖ע דָּוִֽד׃ (ל) וְכֹל֙ אִ֣ישׁ יִשְׂרָאֵ֔ל בִּרְאוֹתָ֖ם אֶת־ הָאִ֑ישׁ וַיָּנֻ֙סוּ֙ מִפָּנָ֔יו וַיִּֽירְא֖וּ מְאֹֽד׃ (לא) וַיֹּ֣אמֶר ׀ אִ֣ישׁ יִשְׂרָאֵ֗ל הַרְּאִיתֶם֙ הָאִ֤ישׁ הָֽעֹלֶה֙ הַזֶּ֔ה כִּ֛י לְחָרֵ֥ף אֶת־ יִשְׂרָאֵ֖ל עֹלֶ֑ה וְֽ֠הָיָה הָאִ֨ישׁ אֲשֶׁר־ יַכֶּ֜נּוּ יַעְשְׁרֶ֥נּוּ הַמֶּ֣לֶךְ ׀ עֹ֣שֶׁר גָּד֗וֹל וְאֶת־ בִּתּוֹ֙ יִתֶּן־ ל֔וֹ וְאֵת֙ בֵּ֣ית אָבִ֔יו יַעֲשֶׂ֥ה חָפְשִׁ֖י בְּיִשְׂרָאֵֽל׃ (לב) וַיֹּ֣אמֶר דָּוִ֗ד אֶֽל־ הָאֲנָשִׁ֞ים הָעֹמְדִ֣ים עִמּוֹ֮ לֵאמֹר֒ מַה־ יֵּעָשֶׂ֗ה לָאִישׁ֙ אֲשֶׁ֤ר יַכֶּה֙ אֶת־ הַפְּלִשְׁתִּ֣י הַלָּ֔ז וְהֵסִ֥יר חֶרְפָּ֖ה מֵעַ֣ל יִשְׂרָאֵ֑ל כִּ֣י מִ֗י הַפְּלִשְׁתִּ֤י הֶֽעָרֵל֙ הַזֶּ֔ה כִּ֣י חֵרֵ֔ף מַעַרְכ֖וֹת אֱלֹהִ֥ים חַיִּֽים׃ (לג) וַיֹּ֤אמֶר לוֹ֙ הָעָ֔ם כַּדָּבָ֥ר הַזֶּ֖ה לֵאמֹ֑ר כֹּ֣ה יֵעָשֶׂ֔ה לָאִ֖ישׁ אֲשֶׁ֥ר יַכֶּֽנּוּ׃ (לד) וַיִּשְׁמַ֤ע אֱלִיאָב֙ אָחִ֣יו הַגָּד֔וֹל בְּדַבְּר֖וֹ אֶל־ הָאֲנָשִׁ֑ים וַיִּֽחַר־ אַף֩ אֱלִיאָ֨ב בְּדָוִ֜ד וַיֹּ֣אמֶר ׀ לָמָּה־ זֶּ֣ה יָרַ֗דְתָּ וְעַל־ מִ֨י נָטַ֜שְׁתָּ מְעַ֨ט הַצֹּ֤אן הָהֵ֙נָּה֙ בַּמִּדְבָּ֔ר אֲנִ֧י יָדַ֣עְתִּי אֶת־ זְדֹנְךָ֗ וְאֵת֙ רֹ֣עַ לְבָבֶ֔ךָ כִּ֗י לְמַ֛עַן רְא֥וֹת הַמִּלְחָמָ֖ה יָרָֽדְתָּ׃ (לה) וַיֹּ֣אמֶר דָּוִ֔ד מֶ֥ה עָשִׂ֖יתִי עָ֑תָּה הֲל֖וֹא דָּבָ֥ר הֽוּא׃ (לו) וַיִּסֹּ֤ב מֵֽאֶצְלוֹ֙ אֶל־ מ֣וּל אַחֵ֔ר וַיֹּ֖אמֶר כַּדָּבָ֣ר הַזֶּ֑ה וַיְשִׁבֻ֤הוּ הָעָם֙ דָּבָ֔ר כַּדָּבָ֖ר הָרִאשֽׁוֹן׃ (לז) וַיְּשָּֽׁמְעוּ֙ הַדְּבָרִ֔ים אֲשֶׁ֖ר דִּבֶּ֣ר דָּוִ֑ד וַיַּגִּ֥דוּ לִפְנֵֽי־ שָׁא֖וּל וַיִּקָּחֵֽהוּ׃ (לח) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ שָׁא֔וּל אַל־ יִפֹּ֥ל לֵב־ אָדָ֖ם עָלָ֑יו עַבְדְּךָ֣ יֵלֵ֔ךְ וְנִלְחַ֖ם עִם־ הַפְּלִשְׁתִּ֥י הַזֶּֽה׃ (לט) וַיֹּ֨אמֶר שָׁא֜וּל אֶל־ דָּוִ֗ד לֹ֤א תוּכַל֙ לָלֶ֙כֶת֙ אֶל־ הַפְּלִשְׁתִּ֣י הַזֶּ֔ה לְהִלָּחֵ֖ם עִמּ֑וֹ כִּֽי־ נַ֣עַר אַ֔תָּה וְה֛וּא אִ֥ישׁ מִלְחָמָ֖ה מִנְּעֻרָֽיו׃ (מ) וַיֹּ֤אמֶר דָּוִד֙ אֶל־ שָׁא֔וּל רֹעֶ֨ה הָיָ֧ה עַבְדְּךָ֛ לְאָבִ֖יו בַּצֹּ֑אן וּבָ֤א הָֽאֲרִי֙ וְאֶת־ הַדּ֔וֹב וְנָשָׂ֥א שֶׂ֖ה מֵהָעֵֽדֶר׃ (מא) וְיָצָ֧אתִי אַחֲרָ֛יו וְהִכִּתִ֖יו וְהִצַּ֣לְתִּי מִפִּ֑יו וַיָּ֣קָם עָלַ֔י וְהֶחֱזַ֙קְתִּי֙ בִּזְקָנ֔וֹ וְהִכִּתִ֖יו וַהֲמִיתִּֽיו׃ (מב) גַּ֧ם אֶֽת־ הָאֲרִ֛י גַּם־ הַדּ֖וֹב הִכָּ֣ה עַבְדֶּ֑ךָ וְֽ֠הָיָה הַפְּלִשְׁתִּ֨י הֶעָרֵ֤ל הַזֶּה֙ כְּאַחַ֣ד מֵהֶ֔ם כִּ֣י חֵרֵ֔ף מַעַרְכֹ֖ת אֱלֹהִ֥ים חַיִּֽים׃
