שמואל סדר כ

פתיחה באתר · ספר שמואל כולו ·

(א) וַיְהִ֕י מֵֽהַיּ֥וֹם הַה֖וּא וָמָ֑עְלָה וַיְשִׂמֶ֜הָ לְחֹ֤ק וּלְמִשְׁפָּט֙ לְיִשְׂרָאֵ֔ל עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (ב) וַיָּבֹ֤א דָוִד֙ אֶל־ צִ֣קְלַ֔ג וַיְשַׁלַּ֧ח מֵֽהַשָּׁלָ֛ל לְזִקְנֵ֥י יְהוּדָ֖ה לְרֵעֵ֣הוּ לֵאמֹ֑ר הִנֵּ֤ה לָכֶם֙ בְּרָכָ֔ה מִשְּׁלַ֖ל אֹיְבֵ֥י יְהוָֽה׃ (ג) לַאֲשֶׁ֧ר בְּבֵֽית־ אֵ֛ל וְלַאֲשֶׁ֥ר בְּרָמֽוֹת־ נֶ֖גֶב וְלַאֲשֶׁ֥ר בְּיַתִּֽר׃ (ד) וְלַאֲשֶׁ֧ר בַּעֲרֹעֵ֛ר וְלַאֲשֶׁ֥ר בְּשִֽׂפְמ֖וֹת וְלַאֲשֶׁ֥ר בְּאֶשְׁתְּמֹֽעַ׃ (ה) וְלַאֲשֶׁ֣ר בְּרָכָ֗ל וְלַֽאֲשֶׁר֙ בְּעָרֵ֣י הַיְּרַחְמְאֵלִ֔י וְלַאֲשֶׁ֖ר בְּעָרֵ֥י הַקֵּינִֽי׃ (ו) וְלַאֲשֶׁ֧ר בְּחָרְמָ֛ה וְלַאֲשֶׁ֥ר בְּבוֹר־ עָשָׁ֖ן וְלַאֲשֶׁ֥ר בַּעֲתָֽךְ׃ (ז) וְלַאֲשֶׁ֖ר בְּחֶבְר֑וֹן וּֽלְכָל־ הַמְּקֹמ֛וֹת אֲשֶֽׁר־ הִתְהַלֶּךְ־ שָׁ֥ם דָּוִ֖ד ה֥וּא וַאֲנָשָֽׁיו׃ (ח) וּפְלִשְׁתִּ֖ים נִלְחָמִ֣ים בְּיִשְׂרָאֵ֑ל וַיָּנֻ֜סוּ אַנְשֵׁ֤י יִשְׂרָאֵל֙ מִפְּנֵ֣י פְלִשְׁתִּ֔ים וַיִּפְּל֥וּ חֲלָלִ֖ים בְּהַ֥ר הַגִּלְבֹּֽעַ׃ (ט) וַיַּדְבְּק֣וּ פְלִשְׁתִּ֔ים אֶת־ שָׁא֖וּל וְאֶת־ בָּנָ֑יו וַיַּכּ֣וּ פְלִשְׁתִּ֗ים אֶת־ יְהוֹנָתָ֧ן וְאֶת־ אֲבִינָדָ֛ב וְאֶת־ מַלְכִּי־ שׁ֖וּעַ בְּנֵ֥י שָׁאֽוּל׃ (י) וַתִּכְבַּ֤ד הַמִּלְחָמָה֙ אֶל־ שָׁא֔וּל וַיִּמְצָאֻ֥הוּ הַמּוֹרִ֖ים אֲנָשִׁ֣ים בַּקָּ֑שֶׁת וַיָּ֥חֶל מְאֹ֖ד מֵהַמּוֹרִֽים׃ (יא) וַיֹּ֣אמֶר שָׁאוּל֩ לְנֹשֵׂ֨א כֵלָ֜יו שְׁלֹ֥ף חַרְבְּךָ֣ ׀ וְדָקְרֵ֣נִי בָ֗הּ פֶּן־ יָ֠בוֹאוּ הָעֲרֵלִ֨ים הָאֵ֤לֶּה וּדְקָרֻ֙נִי֙ וְהִתְעַלְּלוּ־ בִ֔י וְלֹ֤א אָבָה֙ נֹשֵׂ֣א כֵלָ֔יו כִּ֥י יָרֵ֖א מְאֹ֑ד וַיִּקַּ֤ח שָׁאוּל֙ אֶת־ הַחֶ֔רֶב וַיִּפֹּ֖ל עָלֶֽיהָ׃ (יב) וַיַּ֥רְא נֹשֵֽׂא־ כֵלָ֖יו כִּ֣י מֵ֣ת שָׁא֑וּל וַיִּפֹּ֥ל גַּם־ ה֛וּא עַל־ חַרְבּ֖וֹ וַיָּ֥מָת עִמּֽוֹ׃ (יג) וַיָּ֣מָת שָׁא֡וּל וּשְׁלֹ֣שֶׁת בָּנָיו֩ וְנֹשֵׂ֨א כֵלָ֜יו גַּ֧ם כָּל־ אֲנָשָׁ֛יו בַּיּ֥וֹם הַה֖וּא יַחְדָּֽו׃ (יד) וַיִּרְא֣וּ אַנְשֵֽׁי־ יִ֠שְׂרָאֵל אֲשֶׁר־ בְּעֵ֨בֶר הָעֵ֜מֶק וַאֲשֶׁ֣ר ׀ בְּעֵ֣בֶר הַיַּרְדֵּ֗ן כִּֽי־ נָ֙סוּ֙ אַנְשֵׁ֣י יִשְׂרָאֵ֔ל וְכִי־ מֵ֖תוּ שָׁא֣וּל וּבָנָ֑יו וַיַּעַזְב֤וּ אֶת־ הֶֽעָרִים֙ וַיָּנֻ֔סוּ וַיָּבֹ֣אוּ פְלִשְׁתִּ֔ים וַיֵּֽשְׁב֖וּ בָּהֶֽן׃ (טו) וַֽיְהִי֙ מִֽמָּחֳרָ֔ת וַיָּבֹ֣אוּ פְלִשְׁתִּ֔ים לְפַשֵּׁ֖ט אֶת־ הַחֲלָלִ֑ים וַֽיִּמְצְא֤וּ אֶת־ שָׁאוּל֙ וְאֶת־ שְׁלֹ֣שֶׁת בָּנָ֔יו נֹפְלִ֖ים בְּהַ֥ר הַגִּלְבֹּֽעַ׃ (טז) וַֽיִּכְרְתוּ֙ אֶת־ רֹאשׁ֔וֹ וַיַּפְשִׁ֖יטוּ אֶת־ כֵּלָ֑יו וַיְשַׁלְּח֨וּ בְאֶֽרֶץ־ פְּלִשְׁתִּ֜ים סָבִ֗יב לְבַשֵּׂ֛ר בֵּ֥ית עֲצַבֵּיהֶ֖ם וְאֶת־ הָעָֽם׃ (יז) וַיָּשִׂ֙מוּ֙ אֶת־ כֵּלָ֔יו בֵּ֖ית עַשְׁתָּר֑וֹת וְאֶת־ גְּוִיָּתוֹ֙ תָּקְע֔וּ בְּחוֹמַ֖ת בֵּ֥ית שָֽׁן׃ (יח) וַיִּשְׁמְע֣וּ אֵלָ֔יו יֹשְׁבֵ֖י יָבֵ֣ישׁ גִּלְעָ֑ד אֵ֛ת אֲשֶׁר־ עָשׂ֥וּ פְלִשְׁתִּ֖ים לְשָׁאֽוּל׃ (יט) וַיָּק֜וּמוּ כָּל־ אִ֣ישׁ חַיִל֮ וַיֵּלְכ֣וּ כָל־ הַלַּיְלָה֒ וַיִּקְח֞וּ אֶת־ גְּוִיַּ֣ת שָׁא֗וּל וְאֵת֙ גְּוִיֹּ֣ת בָּנָ֔יו מֵחוֹמַ֖ת בֵּ֣ית שָׁ֑ן וַיָּבֹ֣אוּ יָבֵ֔שָׁה וַיִּשְׂרְפ֥וּ אֹתָ֖ם שָֽׁם׃ (כ) וַיִּקְחוּ֙ אֶת־ עַצְמֹ֣תֵיהֶ֔ם וַיִּקְבְּר֥וּ תַֽחַת־ הָאֶ֖שֶׁל בְּיָבֵ֑שָׁה וַיָּצֻ֖מוּ שִׁבְעַ֥ת יָמִֽים׃ (כא) וַיְהִ֗י אַֽחֲרֵי֙ מ֣וֹת שָׁא֔וּל וְדָוִ֣ד שָׁ֔ב מֵהַכּ֖וֹת אֶת־ הָעֲמָלֵ֑ק וַיֵּ֧שֶׁב דָּוִ֛ד בְּצִקְלָ֖ג יָמִ֥ים שְׁנָֽיִם׃ (כב) וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁלִישִׁ֗י וְהִנֵּה֩ אִ֨ישׁ בָּ֤א מִן־ הַֽמַּחֲנֶה֙ מֵעִ֣ם שָׁא֔וּל וּבְגָדָ֣יו קְרֻעִ֔ים וַאֲדָמָ֖ה עַל־ רֹאשׁ֑וֹ וַיְהִי֙ בְּבֹא֣וֹ אֶל־ דָּוִ֔ד וַיִּפֹּ֥ל אַ֖רְצָה וַיִּשְׁתָּֽחוּ׃ (כג) וַיֹּ֤אמֶר לוֹ֙ דָּוִ֔ד אֵ֥י מִזֶּ֖ה תָּב֑וֹא וַיֹּ֣אמֶר אֵלָ֔יו מִמַּחֲנֵ֥ה יִשְׂרָאֵ֖ל נִמְלָֽטְתִּי׃ (כד) וַיֹּ֨אמֶר אֵלָ֥יו דָּוִ֛ד מֶה־ הָיָ֥ה הַדָּבָ֖ר הַגֶּד־ נָ֣א לִ֑י וַ֠יֹּאמֶר אֲשֶׁר־ נָ֨ס הָעָ֜ם מִן־ הַמִּלְחָמָ֗ה וְגַם־ הַרְבֵּ֞ה נָפַ֤ל מִן־ הָעָם֙ וַיָּמֻ֔תוּ וְגַ֗ם שָׁא֛וּל וִיהוֹנָתָ֥ן בְּנ֖וֹ מֵֽתוּ׃ (כה) וַיֹּ֣אמֶר דָּוִ֔ד אֶל־ הַנַּ֖עַר הַמַּגִּ֣יד ל֑וֹ אֵ֣יךְ יָדַ֔עְתָּ כִּי־ מֵ֥ת שָׁא֖וּל וִיהֽוֹנָתָ֥ן בְּנֽוֹ׃ (כו) וַיֹּ֜אמֶר הַנַּ֣עַר ׀ הַמַּגִּ֣יד ל֗וֹ נִקְרֹ֤א נִקְרֵ֙יתִי֙ בְּהַ֣ר הַגִּלְבֹּ֔עַ וְהִנֵּ֥ה שָׁא֖וּל נִשְׁעָ֣ן עַל־ חֲנִית֑וֹ וְהִנֵּ֥ה הָרֶ֛כֶב וּבַעֲלֵ֥י הַפָּרָשִׁ֖ים הִדְבִּקֻֽהוּ׃ (כז) וַיִּ֥פֶן אַחֲרָ֖יו וַיִּרְאֵ֑נִי וַיִּקְרָ֣א אֵלָ֔י וָאֹמַ֖ר הִנֵּֽנִי׃ (כח) וַיֹּ֥אמֶר לִ֖י מִי־ אָ֑תָּה (ויאמר) וָאֹמַ֣ר אֵלָ֔יו עֲמָלֵקִ֖י אָנֹֽכִי׃ (כט) וַיֹּ֣אמֶר אֵלַ֗י עֲמָד־ נָ֤א עָלַי֙ וּמֹ֣תְתֵ֔נִי כִּ֥י אֲחָזַ֖נִי הַשָּׁבָ֑ץ כִּֽי־ כָל־ ע֥וֹד נַפְשִׁ֖י בִּֽי׃ (ל) וָאֶעֱמֹ֤ד עָלָיו֙ וַאֲמֹ֣תְתֵ֔הוּ כִּ֣י יָדַ֔עְתִּי כִּ֛י לֹ֥א יִֽחְיֶ֖ה אַחֲרֵ֣י נִפְל֑וֹ וָאֶקַּ֞ח הַנֵּ֣זֶר ׀ אֲשֶׁ֣ר עַל־ רֹאשׁ֗וֹ וְאֶצְעָדָה֙ אֲשֶׁ֣ר עַל־ זְרֹע֔וֹ וָאֲבִיאֵ֥ם אֶל־ אֲדֹנִ֖י הֵֽנָּה׃ (לא) וַיַּחֲזֵ֥ק דָּוִ֛ד (בבגדו) בִּבְגָדָ֖יו וַיִּקְרָעֵ֑ם וְגַ֥ם כָּל־ הָאֲנָשִׁ֖ים אֲשֶׁ֥ר אִתּֽוֹ׃ (לב) וַֽיִּסְפְּדוּ֙ וַיִּבְכּ֔וּ וַיָּצֻ֖מוּ עַד־ הָעָ֑רֶב עַל־ שָׁא֞וּל וְעַל־ יְהוֹנָתָ֣ן בְּנ֗וֹ וְעַל־ עַ֤ם יְהוָה֙ וְעַל־ בֵּ֣ית יִשְׂרָאֵ֔ל כִּ֥י נָפְל֖וּ בֶּחָֽרֶב׃ (לג) וַיֹּ֣אמֶר דָּוִ֗ד אֶל־ הַנַּ֙עַר֙ הַמַּגִּ֣יד ל֔וֹ אֵ֥י מִזֶּ֖ה אָ֑תָּה וַיֹּ֕אמֶר בֶּן־ אִ֛ישׁ גֵּ֥ר עֲמָלֵקִ֖י אָנֹֽכִי׃ (לד) וַיֹּ֥אמֶר אֵלָ֖יו דָּוִ֑ד אֵ֚יךְ לֹ֣א יָרֵ֔אתָ לִשְׁלֹ֙חַ֙ יָֽדְךָ֔ לְשַׁחֵ֖ת אֶת־ מְשִׁ֥יחַ יְהוָֽה׃ (לה) וַיִּקְרָ֣א דָוִ֗ד לְאַחַד֙ מֵֽהַנְּעָרִ֔ים וַיֹּ֖אמֶר גַּ֣שׁ פְּגַע־ בּ֑וֹ וַיַּכֵּ֖הוּ וַיָּמֹֽת׃ (לו) וַיֹּ֤אמֶר אֵלָיו֙ דָּוִ֔ד (דמיך) דָּמְךָ֖ עַל־ רֹאשֶׁ֑ךָ כִּ֣י פִ֗יךָ עָנָ֤ה בְךָ֙ לֵאמֹ֔ר אָנֹכִ֥י מֹתַ֖תִּי אֶת־ מְשִׁ֥יחַ יְהוָֽה׃ (לז) וַיְקֹנֵ֣ן דָּוִ֔ד אֶת־ הַקִּינָ֖ה הַזֹּ֑את עַל־ שָׁא֖וּל וְעַל־ יְהוֹנָתָ֥ן בְּנֽוֹ׃ (לח) וַיֹּ֕אמֶר לְלַמֵּ֥ד בְּנֵֽי־ יְהוּדָ֖ה קָ֑שֶׁת הִנֵּ֥ה כְתוּבָ֖ה עַל־ סֵ֥פֶר הַיָּשָֽׁר׃ (לט) הַצְּבִי֙ יִשְׂרָאֵ֔ל עַל־ בָּמוֹתֶ֖יךָ חָלָ֑ל אֵ֖יךְ נָפְל֥וּ גִבּוֹרִֽים׃ (מ) אַל־ תַּגִּ֣ידוּ בְגַ֔ת אַֽל־ תְּבַשְּׂר֖וּ בְּחוּצֹ֣ת אַשְׁקְל֑וֹן פֶּן־ תִּשְׂמַ֙חְנָה֙ בְּנ֣וֹת פְּלִשְׁתִּ֔ים פֶּֽן־ תַּעֲלֹ֖זְנָה בְּנ֥וֹת הָעֲרֵלִֽים׃ (מא) הָרֵ֣י בַגִּלְבֹּ֗עַ אַל־ טַ֧ל וְאַל־ מָטָ֛ר עֲלֵיכֶ֖ם וּשְׂדֵ֣י תְרוּמֹ֑ת כִּ֣י שָׁ֤ם נִגְעַל֙ מָגֵ֣ן גִּבּוֹרִ֔ים מָגֵ֣ן שָׁא֔וּל בְּלִ֖י מָשִׁ֥יחַ בַּשָּֽׁמֶן׃ (מב) מִדַּ֣ם חֲלָלִ֗ים מֵחֵ֙לֶב֙ גִּבּוֹרִ֔ים קֶ֚שֶׁת יְה֣וֹנָתָ֔ן לֹ֥א נָשׂ֖וֹג אָח֑וֹר וְחֶ֣רֶב שָׁא֔וּל לֹ֥א תָשׁ֖וּב רֵיקָֽם׃ (מג) שָׁא֣וּל וִיהוֹנָתָ֗ן הַנֶּאֱהָבִ֤ים וְהַנְּעִימִם֙ בְּחַיֵּיהֶ֔ם וּבְמוֹתָ֖ם לֹ֣א נִפְרָ֑דוּ מִנְּשָׁרִ֣ים קַ֔לּוּ מֵאֲרָי֖וֹת גָּבֵֽרוּ׃ (מד) בְּנוֹת֙ יִשְׂרָאֵ֔ל אֶל־ שָׁא֖וּל בְּכֶ֑ינָה הַמַּלְבִּֽשְׁכֶ֤ם שָׁנִי֙ עִם־ עֲדָנִ֔ים הַֽמַּעֲלֶה֙ עֲדִ֣י זָהָ֔ב עַ֖ל לְבוּשְׁכֶֽן׃ (מה) אֵ֚יךְ נָפְל֣וּ גִבֹּרִ֔ים בְּת֖וֹךְ הַמִּלְחָמָ֑ה יְה֣וֹנָתָ֔ן עַל־ בָּמוֹתֶ֖יךָ חָלָֽל׃ (מו) צַר־ לִ֣י עָלֶ֗יךָ אָחִי֙ יְה֣וֹנָתָ֔ן נָעַ֥מְתָּ לִּ֖י מְאֹ֑ד נִפְלְאַ֤תָה אַהֲבָֽתְךָ֙ לִ֔י מֵאַהֲבַ֖ת נָשִֽׁים׃ (מז) אֵ֚יךְ נָפְל֣וּ גִבּוֹרִ֔ים וַיֹּאבְד֖וּ כְּלֵ֥י מִלְחָמָֽה׃ (מח) וַיְהִ֣י אַֽחֲרֵי־ כֵ֗ן וַיִּשְׁאַל֩ דָּוִ֨ד בַּֽיהוָ֤ה ׀ לֵאמֹר֙ הַאֶעֱלֶ֗ה בְּאַחַת֙ עָרֵ֣י יְהוּדָ֔ה וַיֹּ֧אמֶר יְהוָ֛ה אֵלָ֖יו עֲלֵ֑ה וַיֹּ֧אמֶר דָּוִ֛ד אָ֥נָה אֶעֱלֶ֖ה וַיֹּ֥אמֶר חֶבְרֹֽנָה׃ (מט) וַיַּ֤עַל שָׁם֙ דָּוִ֔ד וְגַ֖ם שְׁתֵּ֣י נָשָׁ֑יו אֲחִינֹ֙עַם֙ הַיִּזְרְעֵלִ֔ית וַאֲבִיגַ֕יִל אֵ֖שֶׁת נָבָ֥ל הַֽכַּרְמְלִֽי׃ (נ) וַאֲנָשָׁ֧יו אֲשֶׁר־ עִמּ֛וֹ הֶעֱלָ֥ה דָוִ֖ד אִ֣ישׁ וּבֵית֑וֹ וַיֵּשְׁב֖וּ בְּעָרֵ֥י חֶבְרֽוֹן׃ (נא) וַיָּבֹ֙אוּ֙ אַנְשֵׁ֣י יְהוּדָ֔ה וַיִּמְשְׁחוּ־ שָׁ֧ם אֶת־ דָּוִ֛ד לְמֶ֖לֶךְ עַל־ בֵּ֣ית יְהוּדָ֑ה וַיַּגִּ֤דוּ לְדָוִד֙ לֵאמֹ֔ר אַנְשֵׁי֙ יָבֵ֣ישׁ גִּלְעָ֔ד אֲשֶׁ֥ר קָבְר֖וּ אֶת־ שָׁאֽוּל׃ (נב) וַיִּשְׁלַ֤ח דָּוִד֙ מַלְאָכִ֔ים אֶל־ אַנְשֵׁ֖י יָבֵ֣ישׁ גִּלְעָ֑ד וַיֹּ֣אמֶר אֲלֵיהֶ֗ם בְּרֻכִ֤ים אַתֶּם֙ לַֽיהוָ֔ה אֲשֶׁ֨ר עֲשִׂיתֶ֜ם הַחֶ֣סֶד הַזֶּ֗ה עִם־ אֲדֹֽנֵיכֶם֙ עִם־ שָׁא֔וּל וַֽתִּקְבְּר֖וּ אֹתֽוֹ׃ (נג) וְעַתָּ֕ה יַֽעַשׂ־ יְהוָ֥ה עִמָּכֶ֖ם חֶ֣סֶד וֶאֱמֶ֑ת וְגַ֣ם אָנֹכִ֗י אֶעֱשֶׂ֤ה אִתְּכֶם֙ הַטּוֹבָ֣ה הַזֹּ֔את אֲשֶׁ֥ר עֲשִׂיתֶ֖ם הַדָּבָ֥ר הַזֶּֽה׃
