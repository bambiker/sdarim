שמואל סדר ו

לפי החלוקה לפרקים: שמואל א י, כד – יב, כא · 40 פסוקים

פתיחה באתר · ספר שמואל כולו ·

(א) וַיֹּ֨אמֶר שְׁמוּאֵ֜ל אֶל־ כָּל־ הָעָ֗ם הַרְּאִיתֶם֙ אֲשֶׁ֣ר בָּֽחַר־ בּ֣וֹ יְהוָ֔ה כִּ֛י אֵ֥ין כָּמֹ֖הוּ בְּכָל־ הָעָ֑ם וַיָּרִ֧עוּ כָל־ הָעָ֛ם וַיֹּאמְר֖וּ יְחִ֥י הַמֶּֽלֶךְ׃ (ב) וַיְדַבֵּ֨ר שְׁמוּאֵ֜ל אֶל־ הָעָ֗ם אֵ֚ת מִשְׁפַּ֣ט הַמְּלֻכָ֔ה וַיִּכְתֹּ֣ב בַּסֵּ֔פֶר וַיַּנַּ֖ח לִפְנֵ֣י יְהוָ֑ה וַיְשַׁלַּ֧ח שְׁמוּאֵ֛ל אֶת־ כָּל־ הָעָ֖ם אִ֥ישׁ לְבֵיתֽוֹ׃ (ג) וְגַ֨ם־ שָׁא֔וּל הָלַ֥ךְ לְבֵית֖וֹ גִּבְעָ֑תָה וַיֵּלְכ֣וּ עִמּ֔וֹ הַחַ֕יִל אֲשֶׁר־ נָגַ֥ע אֱלֹהִ֖ים בְּלִבָּֽם׃ (ד) וּבְנֵ֧י בְלִיַּ֣עַל אָמְר֗וּ מַה־ יֹּשִׁעֵ֙נוּ֙ זֶ֔ה וַיִּבְזֻ֕הוּ וְלֹֽא־ הֵבִ֥יאוּ ל֖וֹ מִנְחָ֑ה וַיְהִ֖י כְּמַחֲרִֽישׁ׃ (ה) וַיַּ֗עַל נָחָשׁ֙ הָֽעַמּוֹנִ֔י וַיִּ֖חַן עַל־ יָבֵ֣שׁ גִּלְעָ֑ד וַיֹּ֨אמְר֜וּ כָּל־ אַנְשֵׁ֤י יָבֵישׁ֙ אֶל־ נָחָ֔שׁ כְּרָת־ לָ֥נוּ בְרִ֖ית וְנַעַבְדֶֽךָּ׃ (ו) וַיֹּ֣אמֶר אֲלֵיהֶ֗ם נָחָשׁ֙ הָעַמּוֹנִ֔י בְּזֹאת֙ אֶכְרֹ֣ת לָכֶ֔ם בִּנְק֥וֹר לָכֶ֖ם כָּל־ עֵ֣ין יָמִ֑ין וְשַׂמְתִּ֥יהָ חֶרְפָּ֖ה עַל־ כָּל־ יִשְׂרָאֵֽל׃ (ז) וַיֹּאמְר֨וּ אֵלָ֜יו זִקְנֵ֣י יָבֵ֗ישׁ הֶ֤רֶף לָ֙נוּ֙ שִׁבְעַ֣ת יָמִ֔ים וְנִשְׁלְחָה֙ מַלְאָכִ֔ים בְּכֹ֖ל גְּב֣וּל יִשְׂרָאֵ֑ל וְאִם־ אֵ֥ין מוֹשִׁ֛יעַ אֹתָ֖נוּ וְיָצָ֥אנוּ אֵלֶֽיךָ׃ (ח) וַיָּבֹ֤אוּ הַמַּלְאָכִים֙ גִּבְעַ֣ת שָׁא֔וּל וַיְדַבְּר֥וּ הַדְּבָרִ֖ים בְּאָזְנֵ֣י הָעָ֑ם וַיִּשְׂא֧וּ כָל־ הָעָ֛ם אֶת־ קוֹלָ֖ם וַיִּבְכּֽוּ׃ (ט) וְהִנֵּ֣ה שָׁא֗וּל בָּ֣א אַחֲרֵ֤י הַבָּקָר֙ מִן־ הַשָּׂדֶ֔ה וַיֹּ֣אמֶר שָׁא֔וּל מַה־ לָּעָ֖ם כִּ֣י יִבְכּ֑וּ וַיְסַ֨פְּרוּ־ ל֔וֹ אֶת־ דִּבְרֵ֖י אַנְשֵׁ֥י יָבֵֽישׁ׃ (י) וַתִּצְלַ֤ח רֽוּחַ־ אֱלֹהִים֙ עַל־ שָׁא֔וּל (בשמעו) כְּשָׁמְע֖וֹ אֶת־ הַדְּבָרִ֣ים הָאֵ֑לֶּה וַיִּ֥חַר אַפּ֖וֹ מְאֹֽד׃ (יא) וַיִּקַּח֩ צֶ֨מֶד בָּקָ֜ר וַֽיְנַתְּחֵ֗הוּ וַיְשַׁלַּ֞ח בְּכָל־ גְּב֣וּל יִשְׂרָאֵל֮ בְּיַ֣ד הַמַּלְאָכִ֣ים ׀ לֵאמֹר֒ אֲשֶׁר֩ אֵינֶ֨נּוּ יֹצֵ֜א אַחֲרֵ֤י שָׁאוּל֙ וְאַחַ֣ר שְׁמוּאֵ֔ל כֹּ֥ה יֵעָשֶׂ֖ה לִבְקָר֑וֹ וַיִּפֹּ֤ל פַּֽחַד־ יְהוָה֙ עַל־ הָעָ֔ם וַיֵּצְא֖וּ כְּאִ֥ישׁ אֶחָֽד׃ (יב) וַֽיִּפְקְדֵ֖ם בְּבָ֑זֶק וַיִּהְי֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ שְׁלֹ֣שׁ מֵא֣וֹת אֶ֔לֶף וְאִ֥ישׁ יְהוּדָ֖ה שְׁלֹשִׁ֥ים אָֽלֶף׃ (יג) וַיֹּאמְר֞וּ לַמַּלְאָכִ֣ים הַבָּאִ֗ים כֹּ֤ה תֹֽאמְרוּן֙ לְאִישׁ֙ יָבֵ֣ישׁ גִּלְעָ֔ד מָחָ֛ר תִּהְיֶֽה־ לָכֶ֥ם תְּשׁוּעָ֖ה (בחם) כְּחֹ֣ם הַשָּׁ֑מֶשׁ וַיָּבֹ֣אוּ הַמַּלְאָכִ֗ים וַיַּגִּ֛ידוּ לְאַנְשֵׁ֥י יָבֵ֖ישׁ וַיִּשְׂמָֽחוּ׃ (יד) וַֽיֹּאמְרוּ֙ אַנְשֵׁ֣י יָבֵ֔ישׁ מָחָ֖ר נֵצֵ֣א אֲלֵיכֶ֑ם וַעֲשִׂיתֶ֣ם לָּ֔נוּ כְּכָל־ הַטּ֖וֹב בְּעֵינֵיכֶֽם׃ (טו) וַיְהִ֣י מִֽמָּחֳרָ֗ת וַיָּ֨שֶׂם שָׁא֣וּל אֶת־ הָעָם֮ שְׁלֹשָׁ֣ה רָאשִׁים֒ וַיָּבֹ֤אוּ בְתוֹךְ־ הַֽמַּחֲנֶה֙ בְּאַשְׁמֹ֣רֶת הַבֹּ֔קֶר וַיַּכּ֥וּ אֶת־ עַמּ֖וֹן עַד־ חֹ֣ם הַיּ֑וֹם וַיְהִ֤י הַנִּשְׁאָרִים֙ וַיָּפֻ֔צוּ וְלֹ֥א נִשְׁאֲרוּ־ בָ֖ם שְׁנַ֥יִם יָֽחַד׃ (טז) וַיֹּ֤אמֶר הָעָם֙ אֶל־ שְׁמוּאֵ֔ל מִ֣י הָאֹמֵ֔ר שָׁא֖וּל יִמְלֹ֣ךְ עָלֵ֑ינוּ תְּנ֥וּ הָאֲנָשִׁ֖ים וּנְמִיתֵֽם׃ (יז) וַיֹּ֣אמֶר שָׁא֔וּל לֹֽא־ יוּמַ֥ת אִ֖ישׁ בַּיּ֣וֹם הַזֶּ֑ה כִּ֥י הַיּ֛וֹם עָשָֽׂה־ יְהוָ֥ה תְּשׁוּעָ֖ה בְּיִשְׂרָאֵֽל׃ (יח) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ הָעָ֔ם לְכ֖וּ וְנֵלְכָ֣ה הַגִּלְגָּ֑ל וּנְחַדֵּ֥שׁ שָׁ֖ם הַמְּלוּכָֽה׃ (יט) וַיֵּלְכ֨וּ כָל־ הָעָ֜ם הַגִּלְגָּ֗ל וַיַּמְלִכוּ֩ שָׁ֨ם אֶת־ שָׁא֜וּל לִפְנֵ֤י יְהוָה֙ בַּגִּלְגָּ֔ל וַיִּזְבְּחוּ־ שָׁ֛ם זְבָחִ֥ים שְׁלָמִ֖ים לִפְנֵ֣י יְהוָ֑ה וַיִּשְׂמַ֨ח שָׁ֥ם שָׁא֛וּל וְכָל־ אַנְשֵׁ֥י יִשְׂרָאֵ֖ל עַד־ מְאֹֽד׃ (כ) וַיֹּ֤אמֶר שְׁמוּאֵל֙ אֶל־ כָּל־ יִשְׂרָאֵ֔ל הִנֵּה֙ שָׁמַ֣עְתִּי בְקֹֽלְכֶ֔ם לְכֹ֥ל אֲשֶׁר־ אֲמַרְתֶּ֖ם לִ֑י וָאַמְלִ֥יךְ עֲלֵיכֶ֖ם מֶֽלֶךְ׃ (כא) וְעַתָּ֞ה הִנֵּ֥ה הַמֶּ֣לֶךְ ׀ מִתְהַלֵּ֣ךְ לִפְנֵיכֶ֗ם וַאֲנִי֙ זָקַ֣נְתִּי וָשַׂ֔בְתִּי וּבָנַ֖י הִנָּ֣ם אִתְּכֶ֑ם וַאֲנִי֙ הִתְהַלַּ֣כְתִּי לִפְנֵיכֶ֔ם מִנְּעֻרַ֖י עַד־ הַיּ֥וֹם הַזֶּֽה׃ (כב) הִנְנִ֣י עֲנ֣וּ בִי֩ נֶ֨גֶד יְהוָ֜ה וְנֶ֣גֶד מְשִׁיח֗וֹ אֶת־ שׁוֹר֩ ׀ מִ֨י לָקַ֜חְתִּי וַחֲמ֧וֹר מִ֣י לָקַ֗חְתִּי וְאֶת־ מִ֤י עָשַׁ֙קְתִּי֙ אֶת־ מִ֣י רַצּ֔וֹתִי וּמִיַּד־ מִי֙ לָקַ֣חְתִּי כֹ֔פֶר וְאַעְלִ֥ים עֵינַ֖י בּ֑וֹ וְאָשִׁ֖יב לָכֶֽם׃ (כג) וַיֹּ֣אמְר֔וּ לֹ֥א עֲשַׁקְתָּ֖נוּ וְלֹ֣א רַצּוֹתָ֑נוּ וְלֹֽא־ לָקַ֥חְתָּ מִיַּד־ אִ֖ישׁ מְאֽוּמָה׃ (כד) וַיֹּ֨אמֶר אֲלֵיהֶ֜ם עֵ֧ד יְהוָ֣ה בָּכֶ֗ם וְעֵ֤ד מְשִׁיחוֹ֙ הַיּ֣וֹם הַזֶּ֔ה כִּ֣י לֹ֧א מְצָאתֶ֛ם בְּיָדִ֖י מְא֑וּמָה וַיֹּ֖אמֶר עֵֽד׃ (כה) וַיֹּ֥אמֶר שְׁמוּאֵ֖ל אֶל־ הָעָ֑ם יְהוָ֗ה אֲשֶׁ֤ר עָשָׂה֙ אֶת־ מֹשֶׁ֣ה וְאֶֽת־ אַהֲרֹ֔ן וַאֲשֶׁ֧ר הֶעֱלָ֛ה אֶת־ אֲבֹתֵיכֶ֖ם מֵאֶ֥רֶץ מִצְרָֽיִם׃ (כו) וְעַתָּ֗ה הִֽתְיַצְּב֛וּ וְאִשָּׁפְטָ֥ה אִתְּכֶ֖ם לִפְנֵ֣י יְהוָ֑ה אֵ֚ת כָּל־ צִדְק֣וֹת יְהוָ֔ה אֲשֶׁר־ עָשָׂ֥ה אִתְּכֶ֖ם וְאֶת־ אֲבוֹתֵיכֶֽם׃ (כז) כַּֽאֲשֶׁר־ בָּ֥א יַעֲקֹ֖ב מִצְרָ֑יִם וַיִּזְעֲק֤וּ אֲבֽוֹתֵיכֶם֙ אֶל־ יְהוָ֔ה וַיִּשְׁלַ֨ח יְהוָ֜ה אֶת־ מֹשֶׁ֣ה וְאֶֽת־ אַהֲרֹ֗ן וַיּוֹצִ֤יאוּ אֶת־ אֲבֹֽתֵיכֶם֙ מִמִּצְרַ֔יִם וַיֹּשִׁב֖וּם בַּמָּק֥וֹם הַזֶּֽה׃ (כח) וַֽיִּשְׁכְּח֖וּ אֶת־ יְהוָ֣ה אֱלֹהֵיהֶ֑ם וַיִּמְכֹּ֣ר אֹתָ֡ם בְּיַ֣ד סִֽיסְרָא֩ שַׂר־ צְבָ֨א חָצ֜וֹר וּבְיַד־ פְּלִשְׁתִּ֗ים וּבְיַד֙ מֶ֣לֶךְ מוֹאָ֔ב וַיִּֽלָּחֲמ֖וּ בָּֽם׃ (כט) וַיִּזְעֲק֤וּ אֶל־ יְהוָה֙ (ויאמר) וַיֹּאמְר֣וּ חָטָ֔אנוּ כִּ֤י עָזַ֙בְנוּ֙ אֶת־ יְהוָ֔ה וַנַּעֲבֹ֥ד אֶת־ הַבְּעָלִ֖ים וְאֶת־ הָעַשְׁתָּר֑וֹת וְעַתָּ֗ה הַצִּילֵ֛נוּ מִיַּ֥ד אֹיְבֵ֖ינוּ וְנַעַבְדֶֽךָּ׃ (ל) וַיִּשְׁלַ֤ח יְהוָה֙ אֶת־ יְרֻבַּ֣עַל וְאֶת־ בְּדָ֔ן וְאֶת־ יִפְתָּ֖ח וְאֶת־ שְׁמוּאֵ֑ל וַיַּצֵּ֨ל אֶתְכֶ֜ם מִיַּ֤ד אֹֽיְבֵיכֶם֙ מִסָּבִ֔יב וַתֵּשְׁב֖וּ בֶּֽטַח׃ (לא) וַתִּרְא֗וּ כִּֽי־ נָחָ֞שׁ מֶ֣לֶךְ בְּנֵֽי־ עַמּוֹן֮ בָּ֣א עֲלֵיכֶם֒ וַתֹּ֣אמְרוּ לִ֔י לֹ֕א כִּי־ מֶ֖לֶךְ יִמְלֹ֣ךְ עָלֵ֑ינוּ וַיהוָ֥ה אֱלֹהֵיכֶ֖ם מַלְכְּכֶֽם׃ (לב) וְעַתָּ֗ה הִנֵּ֥ה הַמֶּ֛לֶךְ אֲשֶׁ֥ר בְּחַרְתֶּ֖ם אֲשֶׁ֣ר שְׁאֶלְתֶּ֑ם וְהִנֵּ֨ה נָתַ֧ן יְהוָ֛ה עֲלֵיכֶ֖ם מֶֽלֶךְ׃ (לג) אִם־ תִּֽירְא֣וּ אֶת־ יְהוָ֗ה וַעֲבַדְתֶּ֤ם אֹתוֹ֙ וּשְׁמַעְתֶּ֣ם בְּקֹל֔וֹ וְלֹ֥א תַמְר֖וּ אֶת־ פִּ֣י יְהוָ֑ה וִהְיִתֶ֣ם גַּם־ אַתֶּ֗ם וְגַם־ הַמֶּ֙לֶךְ֙ אֲשֶׁ֣ר מָלַ֣ךְ עֲלֵיכֶ֔ם אַחַ֖ר יְהוָ֥ה אֱלֹהֵיכֶֽם׃ (לד) וְאִם־ לֹ֤א תִשְׁמְעוּ֙ בְּק֣וֹל יְהוָ֔ה וּמְרִיתֶ֖ם אֶת־ פִּ֣י יְהוָ֑ה וְהָיְתָ֧ה יַד־ יְהוָ֛ה בָּכֶ֖ם וּבַאֲבֹתֵיכֶֽם׃ (לה) גַּם־ עַתָּה֙ הִתְיַצְּב֣וּ וּרְא֔וּ אֶת־ הַדָּבָ֥ר הַגָּד֖וֹל הַזֶּ֑ה אֲשֶׁ֣ר יְהוָ֔ה עֹשֶׂ֖ה לְעֵינֵיכֶֽם׃ (לו) הֲל֤וֹא קְצִיר־ חִטִּים֙ הַיּ֔וֹם אֶקְרָא֙ אֶל־ יְהוָ֔ה וְיִתֵּ֥ן קֹל֖וֹת וּמָטָ֑ר וּדְע֣וּ וּרְא֗וּ כִּֽי־ רָעַתְכֶ֤ם רַבָּה֙ אֲשֶׁ֤ר עֲשִׂיתֶם֙ בְּעֵינֵ֣י יְהוָ֔ה לִשְׁא֥וֹל לָכֶ֖ם מֶֽלֶךְ׃ (לז) וַיִּקְרָ֤א שְׁמוּאֵל֙ אֶל־ יְהוָ֔ה וַיִּתֵּ֧ן יְהוָ֛ה קֹלֹ֥ת וּמָטָ֖ר בַּיּ֣וֹם הַה֑וּא וַיִּירָ֨א כָל־ הָעָ֥ם מְאֹ֛ד אֶת־ יְהוָ֖ה וְאֶת־ שְׁמוּאֵֽל׃ (לח) וַיֹּאמְר֨וּ כָל־ הָעָ֜ם אֶל־ שְׁמוּאֵ֗ל הִתְפַּלֵּ֧ל בְּעַד־ עֲבָדֶ֛יךָ אֶל־ יְהוָ֥ה אֱלֹהֶ֖יךָ וְאַל־ נָמ֑וּת כִּֽי־ יָסַ֤פְנוּ עַל־ כָּל־ חַטֹּאתֵ֙ינוּ֙ רָעָ֔ה לִשְׁאֹ֥ל לָ֖נוּ מֶֽלֶךְ׃ (לט) וַיֹּ֨אמֶר שְׁמוּאֵ֤ל אֶל־ הָעָם֙ אַל־ תִּירָ֔אוּ אַתֶּ֣ם עֲשִׂיתֶ֔ם אֵ֥ת כָּל־ הָרָעָ֖ה הַזֹּ֑את אַ֗ךְ אַל־ תָּס֙וּרוּ֙ מֵאַחֲרֵ֣י יְהוָ֔ה וַעֲבַדְתֶּ֥ם אֶת־ יְהוָ֖ה בְּכָל־ לְבַבְכֶֽם׃ (מ) וְלֹ֖א תָּס֑וּרוּ כִּ֣י ׀ אַחֲרֵ֣י הַתֹּ֗הוּ אֲשֶׁ֧ר לֹֽא־ יוֹעִ֛ילוּ וְלֹ֥א יַצִּ֖ילוּ כִּי־ תֹ֥הוּ הֵֽמָּה׃
