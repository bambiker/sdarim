דברים סדר יז

פרשת כי תצא

פתיחה באתר · ספר דברים כולו ·

(א) כִּ֣י יִקָּרֵ֣א קַן־ צִפּ֣וֹר ׀ לְפָנֶ֡יךָ בַּדֶּ֜רֶךְ בְּכָל־ עֵ֣ץ ׀ א֣וֹ עַל־ הָאָ֗רֶץ אֶפְרֹחִים֙ א֣וֹ בֵיצִ֔ים וְהָאֵ֤ם רֹבֶ֙צֶת֙ עַל־ הָֽאֶפְרֹחִ֔ים א֖וֹ עַל־ הַבֵּיצִ֑ים לֹא־ תִקַּ֥ח הָאֵ֖ם עַל־ הַבָּנִֽים׃ (ב) שַׁלֵּ֤חַ תְּשַׁלַּח֙ אֶת־ הָאֵ֔ם וְאֶת־ הַבָּנִ֖ים תִּֽקַּֽח־ לָ֑ךְ לְמַ֙עַן֙ יִ֣יטַב לָ֔ךְ וְהַאֲרַכְתָּ֖ יָמִֽים׃ (ג) כִּ֤י תִבְנֶה֙ בַּ֣יִת חָדָ֔שׁ וְעָשִׂ֥יתָ מַעֲקֶ֖ה לְגַגֶּ֑ךָ וְלֹֽא־ תָשִׂ֤ים דָּמִים֙ בְּבֵיתֶ֔ךָ כִּֽי־ יִפֹּ֥ל הַנֹּפֵ֖ל מִמֶּֽנּוּ׃ (ד) לֹא־ תִזְרַ֥ע כַּרְמְךָ֖ כִּלְאָ֑יִם פֶּן־ תִּקְדַּ֗שׁ הַֽמְלֵאָ֤ה הַזֶּ֙רַע֙ אֲשֶׁ֣ר תִּזְרָ֔ע וּתְבוּאַ֖ת הַכָּֽרֶם׃ (ה) לֹֽא־ תַחֲרֹ֥שׁ בְּשׁוֹר־ וּבַחֲמֹ֖ר יַחְדָּֽו׃ (ו) לֹ֤א תִלְבַּשׁ֙ שַֽׁעַטְנֵ֔ז צֶ֥מֶר וּפִשְׁתִּ֖ים יַחְדָּֽו׃ (ז) גְּדִלִ֖ים תַּעֲשֶׂה־ לָּ֑ךְ עַל־ אַרְבַּ֛ע כַּנְפ֥וֹת כְּסוּתְךָ֖ אֲשֶׁ֥ר תְּכַסֶּה־ בָּֽהּ׃ (ח) כִּֽי־ יִקַּ֥ח אִ֖ישׁ אִשָּׁ֑ה וּבָ֥א אֵלֶ֖יהָ וּשְׂנֵאָֽהּ׃ (ט) וְשָׂ֥ם לָהּ֙ עֲלִילֹ֣ת דְּבָרִ֔ים וְהוֹצִ֥יא עָלֶ֖יהָ שֵׁ֣ם רָ֑ע וְאָמַ֗ר אֶת־ הָאִשָּׁ֤ה הַזֹּאת֙ לָקַ֔חְתִּי וָאֶקְרַ֣ב אֵלֶ֔יהָ וְלֹא־ מָצָ֥אתִי לָ֖הּ בְּתוּלִֽים׃ (י) וְלָקַ֛ח אֲבִ֥י (הנער) הַֽנַּעֲרָ֖ה וְאִמָּ֑הּ וְהוֹצִ֜יאוּ אֶת־ בְּתוּלֵ֧י (הנער) הַֽנַּעֲרָ֛ה אֶל־ זִקְנֵ֥י הָעִ֖יר הַשָּֽׁעְרָה׃ (יא) וְאָמַ֛ר אֲבִ֥י (הנער) הַֽנַּעַרָ֖ה אֶל־ הַזְּקֵנִ֑ים אֶת־ בִּתִּ֗י נָתַ֜תִּי לָאִ֥ישׁ הַזֶּ֛ה לְאִשָּׁ֖ה וַיִּשְׂנָאֶֽהָ׃ (יב) וְהִנֵּה־ ה֡וּא שָׂם֩ עֲלִילֹ֨ת דְּבָרִ֜ים לֵאמֹ֗ר לֹֽא־ מָצָ֤אתִי לְבִתְּךָ֙ בְּתוּלִ֔ים וְאֵ֖לֶּה בְּתוּלֵ֣י בִתִּ֑י וּפָֽרְשׂוּ֙ הַשִּׂמְלָ֔ה לִפְנֵ֖י זִקְנֵ֥י הָעִֽיר׃ (יג) וְלָֽקְח֛וּ זִקְנֵ֥י הָֽעִיר־ הַהִ֖וא אֶת־ הָאִ֑ישׁ וְיִסְּר֖וּ אֹתֽוֹ׃ (יד) וְעָנְשׁ֨וּ אֹת֜וֹ מֵ֣אָה כֶ֗סֶף וְנָתְנוּ֙ לַאֲבִ֣י הַֽנַּעֲרָ֔ה כִּ֤י הוֹצִיא֙ שֵׁ֣ם רָ֔ע עַ֖ל בְּתוּלַ֣ת יִשְׂרָאֵ֑ל וְלֽוֹ־ תִהְיֶ֣ה לְאִשָּׁ֔ה לֹא־ יוּכַ֥ל לְשַּׁלְּחָ֖הּ כָּל־ יָמָֽיו׃ (טו) וְאִם־ אֱמֶ֣ת הָיָ֔ה הַדָּבָ֖ר הַזֶּ֑ה לֹא־ נִמְצְא֥וּ בְתוּלִ֖ים (לנער) לַֽנַּעֲרָֽה׃ (טז) וְהוֹצִ֨יאוּ אֶת־ (הנער) הַֽנַּעֲרָ֜ה אֶל־ פֶּ֣תַח בֵּית־ אָבִ֗יהָ וּסְקָלוּהָ֩ אַנְשֵׁ֨י עִירָ֤הּ בָּאֲבָנִים֙ וָמֵ֔תָה כִּֽי־ עָשְׂתָ֤ה נְבָלָה֙ בְּיִשְׂרָאֵ֔ל לִזְנ֖וֹת בֵּ֣ית אָבִ֑יהָ וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּֽךָ׃ (יז) כִּֽי־ יִמָּצֵ֨א אִ֜ישׁ שֹׁכֵ֣ב ׀ עִם־ אִשָּׁ֣ה בְעֻֽלַת־ בַּ֗עַל וּמֵ֙תוּ֙ גַּם־ שְׁנֵיהֶ֔ם הָאִ֛ישׁ הַשֹּׁכֵ֥ב עִם־ הָאִשָּׁ֖ה וְהָאִשָּׁ֑ה וּבִֽעַרְתָּ֥ הָרָ֖ע מִיִּשְׂרָאֵֽל׃ (יח) כִּ֤י יִהְיֶה֙ (נער) נַעֲרָ֣ה בְתוּלָ֔ה מְאֹרָשָׂ֖ה לְאִ֑ישׁ וּמְצָאָ֥הּ אִ֛ישׁ בָּעִ֖יר וְשָׁכַ֥ב עִמָּֽהּ׃ (יט) וְהוֹצֵאתֶ֨ם אֶת־ שְׁנֵיהֶ֜ם אֶל־ שַׁ֣עַר ׀ הָעִ֣יר הַהִ֗וא וּסְקַלְתֶּ֨ם אֹתָ֥ם בָּאֲבָנִים֮ וָמֵתוּ֒ אֶת־ (הנער) הַֽנַּעֲרָ֗ה עַל־ דְּבַר֙ אֲשֶׁ֣ר לֹא־ צָעֲקָ֣ה בָעִ֔יר וְאֶ֨ת־ הָאִ֔ישׁ עַל־ דְּבַ֥ר אֲשֶׁר־ עִנָּ֖ה אֶת־ אֵ֣שֶׁת רֵעֵ֑הוּ וּבִֽעַרְתָּ֥ הָרָ֖ע מִקִּרְבֶּֽךָ׃ (כ) וְֽאִם־ בַּשָּׂדֶ֞ה יִמְצָ֣א הָאִ֗ישׁ אֶת־ (הנער) הַֽנַּעֲרָה֙ הַמְאֹ֣רָשָׂ֔ה וְהֶחֱזִֽיק־ בָּ֥הּ הָאִ֖ישׁ וְשָׁכַ֣ב עִמָּ֑הּ וּמֵ֗ת הָאִ֛ישׁ אֲשֶׁר־ שָׁכַ֥ב עִמָּ֖הּ לְבַדּֽוֹ׃ (כא) (ולנער) וְלַֽנַּעֲרָה֙ לֹא־ תַעֲשֶׂ֣ה דָבָ֔ר אֵ֥ין (לנער) לַֽנַּעֲרָ֖ה חֵ֣טְא מָ֑וֶת כִּ֡י כַּאֲשֶׁר֩ יָק֨וּם אִ֤ישׁ עַל־ רֵעֵ֙הוּ֙ וּרְצָח֣וֹ נֶ֔פֶשׁ כֵּ֖ן הַדָּבָ֥ר הַזֶּֽה׃ (כב) כִּ֥י בַשָּׂדֶ֖ה מְצָאָ֑הּ צָעֲקָ֗ה (הנער) הַֽנַּעֲרָה֙ הַמְאֹ֣רָשָׂ֔ה וְאֵ֥ין מוֹשִׁ֖יעַ לָֽהּ׃ (כג) כִּֽי־ יִמְצָ֣א אִ֗ישׁ (נער) נַעֲרָ֤ה בְתוּלָה֙ אֲשֶׁ֣ר לֹא־ אֹרָ֔שָׂה וּתְפָשָׂ֖הּ וְשָׁכַ֣ב עִמָּ֑הּ וְנִמְצָֽאוּ׃ (כד) וְ֠נָתַן הָאִ֨ישׁ הַשֹּׁכֵ֥ב עִמָּ֛הּ לַאֲבִ֥י (הנער) הַֽנַּעֲרָ֖ה חֲמִשִּׁ֣ים כָּ֑סֶף וְלֽוֹ־ תִהְיֶ֣ה לְאִשָּׁ֗ה תַּ֚חַת אֲשֶׁ֣ר עִנָּ֔הּ לֹא־ יוּכַ֥ל שַׁלְּחָ֖ה כָּל־ יָמָֽיו׃ (כה) לֹא־ יִקַּ֥ח אִ֖ישׁ אֶת־ אֵ֣שֶׁת אָבִ֑יו וְלֹ֥א יְגַלֶּ֖ה כְּנַ֥ף אָבִֽיו׃ (כו) לֹֽא־ יָבֹ֧א פְצֽוּעַ־ דַּכָּ֛א וּכְר֥וּת שָׁפְכָ֖ה בִּקְהַ֥ל יְהוָֽה׃ (כז) לֹא־ יָבֹ֥א מַמְזֵ֖ר בִּקְהַ֣ל יְהוָ֑ה גַּ֚ם דּ֣וֹר עֲשִׂירִ֔י לֹא־ יָ֥בֹא ל֖וֹ בִּקְהַ֥ל יְהוָֽה׃ (כח) לֹֽא־ יָבֹ֧א עַמּוֹנִ֛י וּמוֹאָבִ֖י בִּקְהַ֣ל יְהוָ֑ה גַּ֚ם דּ֣וֹר עֲשִׂירִ֔י לֹא־ יָבֹ֥א לָהֶ֛ם בִּקְהַ֥ל יְהוָ֖ה עַד־ עוֹלָֽם׃ (כט) עַל־ דְּבַ֞ר אֲשֶׁ֨ר לֹא־ קִדְּמ֤וּ אֶתְכֶם֙ בַּלֶּ֣חֶם וּבַמַּ֔יִם בַּדֶּ֖רֶךְ בְּצֵאתְכֶ֣ם מִמִּצְרָ֑יִם וַאֲשֶׁר֩ שָׂכַ֨ר עָלֶ֜יךָ אֶת־ בִּלְעָ֣ם בֶּן־ בְּע֗וֹר מִפְּת֛וֹר אֲרַ֥ם נַהֲרַ֖יִם לְקַֽלְלֶֽךָּ׃ (ל) וְלֹֽא־ אָבָ֞ה יְהוָ֤ה אֱלֹהֶ֙יךָ֙ לִשְׁמֹ֣עַ אֶל־ בִּלְעָ֔ם וַיַּהֲפֹךְ֩ יְהוָ֨ה אֱלֹהֶ֧יךָ לְּךָ֛ אֶת־ הַקְּלָלָ֖ה לִבְרָכָ֑ה כִּ֥י אֲהֵֽבְךָ֖ יְהוָ֥ה אֱלֹהֶֽיךָ׃ (לא) לֹא־ תִדְרֹ֥שׁ שְׁלֹמָ֖ם וְטֹבָתָ֑ם כָּל־ יָמֶ֖יךָ לְעוֹלָֽם׃ (לב) לֹֽא־ תְתַעֵ֣ב אֲדֹמִ֔י כִּ֥י אָחִ֖יךָ ה֑וּא לֹא־ תְתַעֵ֣ב מִצְרִ֔י כִּי־ גֵ֖ר הָיִ֥יתָ בְאַרְצֽוֹ׃ (לג) בָּנִ֛ים אֲשֶׁר־ יִוָּלְד֥וּ לָהֶ֖ם דּ֣וֹר שְׁלִישִׁ֑י יָבֹ֥א לָהֶ֖ם בִּקְהַ֥ל יְהוָֽה׃
