עזרא ונחמיה סדר ה

פתיחה באתר · ספר עזרא ונחמיה כולו ·

(א) אָנָּ֣א אֲדֹנָ֗י תְּהִ֣י נָ֣א אָזְנְךָֽ־ קַ֠שֶּׁבֶת אֶל־ תְּפִלַּ֨ת עַבְדְּךָ֜ וְאֶל־ תְּפִלַּ֣ת עֲבָדֶ֗יךָ הַֽחֲפֵצִים֙ לְיִרְאָ֣ה אֶת־ שְׁמֶ֔ךָ וְהַצְלִֽיחָה־ נָּ֤א לְעַבְדְּךָ֙ הַיּ֔וֹם וּתְנֵ֣הוּ לְרַחֲמִ֔ים לִפְנֵ֖י הָאִ֣ישׁ הַזֶּ֑ה וַאֲנִ֛י הָיִ֥יתִי מַשְׁקֶ֖ה לַמֶּֽלֶךְ׃ (ב) וַיְהִ֣י ׀ בְּחֹ֣דֶשׁ נִיסָ֗ן שְׁנַ֥ת עֶשְׂרִ֛ים לְאַרְתַּחְשַׁ֥סְתְּא הַמֶּ֖לֶךְ יַ֣יִן לְפָנָ֑יו וָאֶשָּׂ֤א אֶת־ הַיַּ֙יִן֙ וָאֶתְּנָ֣ה לַמֶּ֔לֶךְ וְלֹא־ הָיִ֥יתִי רַ֖ע לְפָנָֽיו׃ (ג) וַיֹּאמֶר֩ לִ֨י הַמֶּ֜לֶךְ מַדּ֣וּעַ ׀ פָּנֶ֣יךָ רָעִ֗ים וְאַתָּה֙ אֵֽינְךָ֣ חוֹלֶ֔ה אֵ֣ין זֶ֔ה כִּי־ אִ֖ם רֹ֣עַֽ לֵ֑ב וָאִירָ֖א הַרְבֵּ֥ה מְאֹֽד׃ (ד) וָאֹמַ֣ר לַמֶּ֔לֶךְ הַמֶּ֖לֶךְ לְעוֹלָ֣ם יִחְיֶ֑ה מַדּ֜וּעַ לֹא־ יֵרְע֣וּ פָנַ֗י אֲשֶׁ֨ר הָעִ֜יר בֵּית־ קִבְר֤וֹת אֲבֹתַי֙ חֲרֵבָ֔ה וּשְׁעָרֶ֖יהָ אֻכְּל֥וּ בָאֵֽשׁ׃ (ה) וַיֹּ֤אמֶר לִי֙ הַמֶּ֔לֶךְ עַל־ מַה־ זֶּ֖ה אַתָּ֣ה מְבַקֵּ֑שׁ וָֽאֶתְפַּלֵּ֔ל אֶל־ אֱלֹהֵ֖י הַשָּׁמָֽיִם׃ (ו) וָאֹמַ֣ר לַמֶּ֔לֶךְ אִם־ עַל־ הַמֶּ֣לֶךְ ט֔וֹב וְאִם־ יִיטַ֥ב עַבְדְּךָ֖ לְפָנֶ֑יךָ אֲשֶׁ֧ר תִּשְׁלָחֵ֣נִי אֶל־ יְהוּדָ֗ה אֶל־ עִ֛יר קִבְר֥וֹת אֲבֹתַ֖י וְאֶבְנֶֽנָּה׃ (ז) וַיֹּאמֶר֩ לִ֨י הַמֶּ֜לֶךְ וְהַשֵּׁגַ֣ל ׀ יוֹשֶׁ֣בֶת אֶצְל֗וֹ עַד־ מָתַ֛י יִהְיֶ֥ה מַֽהֲלָכֲךָ֖ וּמָתַ֣י תָּשׁ֑וּב וַיִּיטַ֤ב לִפְנֵֽי־ הַמֶּ֙לֶךְ֙ וַיִּשְׁלָחֵ֔נִי וָֽאֶתְּנָ֥ה ל֖וֹ זְמָֽן׃ (ח) וָאוֹמַר֮ לַמֶּלֶךְ֒ אִם־ עַל־ הַמֶּ֣לֶךְ ט֔וֹב אִגְּרוֹת֙ יִתְּנוּ־ לִ֔י עַֽל־ פַּחֲו֖וֹת עֵ֣בֶר הַנָּהָ֑ר אֲשֶׁר֙ יַעֲבִיר֔וּנִי עַ֥ד אֲשֶׁר־ אָב֖וֹא אֶל־ יְהוּדָֽה׃ (ט) וְאִגֶּ֡רֶת אֶל־ אָסָף֩ שֹׁמֵ֨ר הַפַּרְדֵּ֜ס אֲשֶׁ֣ר לַמֶּ֗לֶךְ אֲשֶׁ֣ר יִתֶּן־ לִ֣י עֵצִ֡ים לְ֠קָרוֹת אֶת־ שַׁעֲרֵ֨י הַבִּירָ֤ה אֲשֶׁר־ לַבַּ֙יִת֙ וּלְחוֹמַ֣ת הָעִ֔יר וְלַבַּ֖יִת אֲשֶׁר־ אָב֣וֹא אֵלָ֑יו וַיִּתֶּן־ לִ֣י הַמֶּ֔לֶךְ כְּיַד־ אֱלֹהַ֖י הַטּוֹבָ֥ה עָלָֽי׃ (י) וָֽאָב֗וֹא אֶֽל־ פַּֽחֲווֹת֙ עֵ֣בֶר הַנָּהָ֔ר וָאֶתְּנָ֣ה לָהֶ֔ם אֵ֖ת אִגְּר֣וֹת הַמֶּ֑לֶךְ וַיִּשְׁלַ֤ח עִמִּי֙ הַמֶּ֔לֶךְ שָׂ֥רֵי חַ֖יִל וּפָרָשִֽׁים׃ (יא) וַיִּשְׁמַ֞ע סַנְבַלַּ֣ט הַחֹרֹנִ֗י וְטֽוֹבִיָּה֙ הָעֶ֣בֶד הָֽעַמֹּנִ֔י וַיֵּ֥רַע לָהֶ֖ם רָעָ֣ה גְדֹלָ֑ה אֲשֶׁר־ בָּ֥א אָדָ֔ם לְבַקֵּ֥שׁ טוֹבָ֖ה לִבְנֵ֥י יִשְׂרָאֵֽל׃ (יב) וָאָב֖וֹא אֶל־ יְרוּשָׁלִָ֑ם וָאֱהִי־ שָׁ֖ם יָמִ֥ים שְׁלֹשָֽׁה׃ (יג) וָאָק֣וּם ׀ לַ֗יְלָה אֲנִי֮ וַאֲנָשִׁ֣ים ׀ מְעַט֮ עִמִּי֒ וְלֹא־ הִגַּ֣דְתִּי לְאָדָ֔ם מָ֗ה אֱלֹהַי֙ נֹתֵ֣ן אֶל־ לִבִּ֔י לַעֲשׂ֖וֹת לִירוּשָׁלִָ֑ם וּבְהֵמָה֙ אֵ֣ין עִמִּ֔י כִּ֚י אִם־ הַבְּהֵמָ֔ה אֲשֶׁ֥ר אֲנִ֖י רֹכֵ֥ב בָּֽהּ׃ (יד) וָאֵצְאָ֨ה בְשַֽׁעַר־ הַגַּ֜יא לַ֗יְלָה וְאֶל־ פְּנֵי֙ עֵ֣ין הַתַּנִּ֔ין וְאֶל־ שַׁ֖עַר הָאַשְׁפֹּ֑ת וָאֱהִ֨י שֹׂבֵ֜ר בְּחוֹמֹ֤ת יְרוּשָׁלִַ֙ם֙ אֲשֶׁר־ (המפרוצים) הֵ֣ם ׀ פְּרוּצִ֔ים וּשְׁעָרֶ֖יהָ אֻכְּל֥וּ בָאֵֽשׁ׃ (טו) וָאֶֽעֱבֹר֙ אֶל־ שַׁ֣עַר הָעַ֔יִן וְאֶל־ בְּרֵכַ֖ת הַמֶּ֑לֶךְ וְאֵין־ מָק֥וֹם לַבְּהֵמָ֖ה לַעֲבֹ֥ר תַּחְתָּֽי׃ (טז) וָאֱהִ֨י עֹלֶ֤ה בַנַּ֙חַל֙ לַ֔יְלָה וָאֱהִ֥י שֹׂבֵ֖ר בַּחוֹמָ֑ה וָאָשׁ֗וּב וָאָב֛וֹא בְּשַׁ֥עַר הַגַּ֖יְא וָאָשֽׁוּב׃ (יז) וְהַסְּגָנִ֗ים לֹ֤א יָדְעוּ֙ אָ֣נָה הָלַ֔כְתִּי וּמָ֖ה אֲנִ֣י עֹשֶׂ֑ה וְלַיְּהוּדִ֨ים וְלַכֹּהֲנִ֜ים וְלַחֹרִ֣ים וְלַסְּגָנִ֗ים וּלְיֶ֙תֶר֙ עֹשֵׂ֣ה הַמְּלָאכָ֔ה עַד־ כֵּ֖ן לֹ֥א הִגַּֽדְתִּי׃ (יח) וָאוֹמַ֣ר אֲלֵהֶ֗ם אַתֶּ֤ם רֹאִים֙ הָרָעָה֙ אֲשֶׁ֣ר אֲנַ֣חְנוּ בָ֔הּ אֲשֶׁ֤ר יְרוּשָׁלִַ֙ם֙ חֲרֵבָ֔ה וּשְׁעָרֶ֖יהָ נִצְּת֣וּ בָאֵ֑שׁ לְכ֗וּ וְנִבְנֶה֙ אֶת־ חוֹמַ֣ת יְרוּשָׁלִַ֔ם וְלֹא־ נִהְיֶ֥ה ע֖וֹד חֶרְפָּֽה׃ (יט) וָאַגִּ֨יד לָהֶ֜ם אֶת־ יַ֣ד אֱלֹהַ֗י אֲשֶׁר־ הִיא֙ טוֹבָ֣ה עָלַ֔י וְאַף־ דִּבְרֵ֥י הַמֶּ֖לֶךְ אֲשֶׁ֣ר אָֽמַר־ לִ֑י וַיֹּֽאמְרוּ֙ נָק֣וּם וּבָנִ֔ינוּ וַיְחַזְּק֥וּ יְדֵיהֶ֖ם לַטּוֹבָֽה׃ (כ) וַיִּשְׁמַע֩ סַנְבַלַּ֨ט הַחֹרֹנִ֜י וְטֹבִיָּ֣ה ׀ הָעֶ֣בֶד הָֽעַמּוֹנִ֗י וְגֶ֙שֶׁם֙ הָֽעַרְבִ֔י וַיַּלְעִ֣גוּ לָ֔נוּ וַיִּבְז֖וּ עָלֵ֑ינוּ וַיֹּאמְר֗וּ מָֽה־ הַדָּבָ֤ר הַזֶּה֙ אֲשֶׁ֣ר אַתֶּ֣ם עֹשִׂ֔ים הַעַ֥ל הַמֶּ֖לֶךְ אַתֶּ֥ם מֹרְדִֽים׃ (כא) וָאָשִׁ֨יב אוֹתָ֜ם דָּבָ֗ר וָאוֹמַ֤ר לָהֶם֙ אֱלֹהֵ֣י הַשָּׁמַ֔יִם ה֚וּא יַצְלִ֣יחַֽ לָ֔נוּ וַאֲנַ֥חְנוּ עֲבָדָ֖יו נָק֣וּם וּבָנִ֑ינוּ וְלָכֶ֗ם אֵֽין־ חֵ֧לֶק וּצְדָקָ֛ה וְזִכָּר֖וֹן בִּירוּשָׁלִָֽם׃ (כב) וַיָּ֡קָם אֶלְיָשִׁיב֩ הַכֹּהֵ֨ן הַגָּד֜וֹל וְאֶחָ֣יו הַכֹּהֲנִ֗ים וַיִּבְנוּ֙ אֶת־ שַׁ֣עַר הַצֹּ֔אן הֵ֣מָּה קִדְּשׁ֔וּהוּ וַֽיַּעֲמִ֖ידוּ דַּלְתֹתָ֑יו וְעַד־ מִגְדַּ֤ל הַמֵּאָה֙ קִדְּשׁ֔וּהוּ עַ֖ד מִגְדַּ֥ל חֲנַנְאֵֽל׃ (כג) וְעַל־ יָד֥וֹ בָנ֖וּ אַנְשֵׁ֣י יְרֵח֑וֹ וְעַל־ יָד֣וֹ בָנָ֔ה זַכּ֖וּר בֶּן־ אִמְרִֽי׃ (כד) וְאֵת֙ שַׁ֣עַר הַדָּגִ֔ים בָּנ֖וּ בְּנֵ֣י הַסְּנָאָ֑ה הֵ֣מָּה קֵר֔וּהוּ וַֽיַּעֲמִ֙ידוּ֙ דַּלְתֹתָ֔יו מַנְעוּלָ֖יו וּבְרִיחָֽיו׃ (כה) וְעַל־ יָדָ֣ם הֶחֱזִ֗יק מְרֵמ֤וֹת בֶּן־ אוּרִיָּה֙ בֶּן־ הַקּ֔וֹץ וְעַל־ יָדָ֣ם הֶחֱזִ֔יק מְשֻׁלָּ֥ם בֶּן־ בֶּרֶכְיָ֖ה בֶּן־ מְשֵׁיזַבְאֵ֑ל וְעַל־ יָדָ֣ם הֶֽחֱזִ֔יק צָד֖וֹק בֶּֽן־ בַּעֲנָֽא׃ (כו) וְעַל־ יָדָ֖ם הֶחֱזִ֣יקוּ הַתְּקוֹעִ֑ים וְאַדִּֽירֵיהֶם֙ לֹא־ הֵבִ֣יאוּ צַוָּרָ֔ם בַּעֲבֹדַ֖ת אֲדֹנֵיהֶֽם׃ (כז) וְאֵת֩ שַׁ֨עַר הַיְשָׁנָ֜ה הֶחֱזִ֗יקוּ יֽוֹיָדָע֙ בֶּן־ פָּסֵ֔חַ וּמְשֻׁלָּ֖ם בֶּן־ בְּסֽוֹדְיָ֑ה הֵ֣מָּה קֵר֔וּהוּ וַֽיַּעֲמִ֙ידוּ֙ דַּלְתֹתָ֔יו וּמַנְעֻלָ֖יו וּבְרִיחָֽיו׃ (כח) וְעַל־ יָדָ֨ם הֶחֱזִ֜יק מְלַטְיָ֣ה הַגִּבְעֹנִ֗י וְיָדוֹן֙ הַמֵּרֹ֣נֹתִ֔י אַנְשֵׁ֥י גִבְע֖וֹן וְהַמִּצְפָּ֑ה לְכִסֵּ֕א פַּחַ֖ת עֵ֥בֶר הַנָּהָֽר׃ (כט) עַל־ יָד֣וֹ הֶחֱזִ֗יק עֻזִּיאֵ֤ל בֶּֽן־ חַרְהֲיָה֙ צֽוֹרְפִ֔ים וְעַל־ יָד֣וֹ הֶחֱזִ֔יק חֲנַנְיָ֖ה בֶּן־ הָרַקָּחִ֑ים וַיַּֽעַזְבוּ֙ יְר֣וּשָׁלִַ֔ם עַ֖ד הַחוֹמָ֥ה הָרְחָבָֽה׃ (ל) וְעַל־ יָדָ֤ם הֶחֱזִיק֙ רְפָיָ֣ה בֶן־ ח֔וּר שַׂ֕ר חֲצִ֖י פֶּ֥לֶךְ יְרוּשָׁלִָֽם׃ (לא) וְעַל־ יָדָ֧ם הֶחֱזִ֛יק יְדָיָ֥ה בֶן־ חֲרוּמַ֖ף וְנֶ֣גֶד בֵּית֑וֹ וְעַל־ יָד֣וֹ הֶחֱזִ֔יק חַטּ֖וּשׁ בֶּן־ חֲשַׁבְנְיָֽה׃ (לב) מִדָּ֣ה שֵׁנִ֗ית הֶחֱזִיק֙ מַלְכִּיָּ֣ה בֶן־ חָרִ֔ם וְחַשּׁ֖וּב בֶּן־ פַּחַ֣ת מוֹאָ֑ב וְאֵ֖ת מִגְדַּ֥ל הַתַּנּוּרִֽים׃ (לג) וְעַל־ יָד֣וֹ הֶחֱזִ֗יק שַׁלּוּם֙ בֶּן־ הַלּוֹחֵ֔שׁ שַׂ֕ר חֲצִ֖י פֶּ֣לֶךְ יְרוּשָׁלִָ֑ם ה֖וּא וּבְנוֹתָֽיו׃ (לד) אֵת֩ שַׁ֨עַר הַגַּ֜יְא הֶחֱזִ֣יק חָנוּן֮ וְיֹשְׁבֵ֣י זָנוֹחַ֒ הֵ֣מָּה בָנ֔וּהוּ וַֽיַּעֲמִ֙ידוּ֙ דַּלְתֹתָ֔יו מַנְעֻלָ֖יו וּבְרִיחָ֑יו וְאֶ֤לֶף אַמָּה֙ בַּחוֹמָ֔ה עַ֖ד שַׁ֥עַר הָשֲׁפֽוֹת׃ (לה) וְאֵ֣ת ׀ שַׁ֣עַר הָאַשְׁפּ֗וֹת הֶחֱזִיק֙ מַלְכִּיָּ֣ה בֶן־ רֵכָ֔ב שַׂ֖ר פֶּ֣לֶךְ בֵּית־ הַכָּ֑רֶם ה֣וּא יִבְנֶ֔נּוּ וְיַעֲמִיד֙ דַּלְתֹתָ֔יו מַנְעֻלָ֖יו וּבְרִיחָֽיו׃ (לו) וְאֵת֩ שַׁ֨עַר הָעַ֜יִן הֶ֠חֱזִיק שַׁלּ֣וּן בֶּן־ כָּל־ חֹזֶה֮ שַׂ֣ר פֶּ֣לֶךְ הַמִּצְפָּה֒ ה֤וּא יִבְנֶ֙נּוּ֙ וִיטַֽלְלֶ֔נּוּ (ויעמידו) וְיַעֲמִיד֙ דַּלְתֹתָ֔יו מַנְעֻלָ֖יו וּבְרִיחָ֑יו וְ֠אֵת חוֹמַ֞ת בְּרֵכַ֤ת הַשֶּׁ֙לַח֙ לְגַן־ הַמֶּ֔לֶךְ וְעַד־ הַֽמַּעֲל֔וֹת הַיּוֹרְד֖וֹת מֵעִ֥יר דָּוִֽיד׃ (לז) אַחֲרָ֤יו הֶחֱזִיק֙ נְחֶמְיָ֣ה בֶן־ עַזְבּ֔וּק שַׂ֕ר חֲצִ֖י פֶּ֣לֶךְ בֵּֽית־ צ֑וּר עַד־ נֶ֙גֶד֙ קִבְרֵ֣י דָוִ֔יד וְעַד־ הַבְּרֵכָה֙ הָעֲשׂוּיָ֔ה וְעַ֖ד בֵּ֥ית הַגִּבֹּרִֽים׃ (לח) אַחֲרָ֛יו הֶחֱזִ֥יקוּ הַלְוִיִּ֖ם רְח֣וּם בֶּן־ בָּנִ֑י עַל־ יָד֣וֹ הֶחֱזִ֗יק חֲשַׁבְיָ֛ה שַׂר־ חֲצִי־ פֶ֥לֶךְ קְעִילָ֖ה לְפִלְכּֽוֹ׃ (לט) אַחֲרָיו֙ הֶחֱזִ֣יקוּ אֲחֵיהֶ֔ם בַּוַּ֖י בֶּן־ חֵנָדָ֑ד שַׂ֕ר חֲצִ֖י פֶּ֥לֶךְ קְעִילָֽה׃ (מ) וַיְחַזֵּ֨ק עַל־ יָד֜וֹ עֵ֧זֶר בֶּן־ יֵשׁ֛וּעַ שַׂ֥ר הַמִּצְפָּ֖ה מִדָּ֣ה שֵׁנִ֑ית מִנֶּ֕גֶד עֲלֹ֥ת הַנֶּ֖שֶׁק הַמִּקְצֹֽעַ׃ (מא) אַחֲרָ֨יו הֶחֱרָ֧ה הֶחֱזִ֛יק בָּר֥וּךְ בֶּן־ (זבי) זַכַּ֖י מִדָּ֣ה שֵׁנִ֑ית מִן־ הַ֨מִּקְצ֔וֹעַ עַד־ פֶּ֙תַח֙ בֵּ֣ית אֶלְיָשִׁ֔יב הַכֹּהֵ֖ן הַגָּדֽוֹל׃ (מב) אַחֲרָ֣יו הֶחֱזִ֗יק מְרֵמ֧וֹת בֶּן־ אוּרִיָּ֛ה בֶּן־ הַקּ֖וֹץ מִדָּ֣ה שֵׁנִ֑ית מִפֶּ֙תַח֙ בֵּ֣ית אֶלְיָשִׁ֔יב וְעַד־ תַּכְלִ֖ית בֵּ֥ית אֶלְיָשִֽׁיב׃ (מג) וְאַחֲרָ֛יו הֶחֱזִ֥יקוּ הַכֹּהֲנִ֖ים אַנְשֵׁ֥י הַכִּכָּֽר׃ (מד) אַחֲרָ֨יו הֶחֱזִ֧יק בִּנְיָמִ֛ן וְחַשּׁ֖וּב נֶ֣גֶד בֵּיתָ֑ם אַחֲרָ֣יו הֶחֱזִ֗יק עֲזַרְיָ֧ה בֶן־ מַעֲשֵׂיָ֛ה בֶּן־ עֲנָֽנְיָ֖ה אֵ֥צֶל בֵּיתֽוֹ׃ (מה) אַחֲרָ֣יו הֶחֱזִ֗יק בִּנּ֛וּי בֶּן־ חֵנָדָ֖ד מִדָּ֣ה שֵׁנִ֑ית מִבֵּ֣ית עֲזַרְיָ֔ה עַד־ הַמִּקְצ֖וֹעַ וְעַד־ הַפִּנָּֽה׃ (מו) פָּלָ֣ל בֶּן־ אוּזַי֮ מִנֶּ֣גֶד הַמִּקְצוֹעַ֒ וְהַמִּגְדָּ֗ל הַיּוֹצֵא֙ מִבֵּ֤ית הַמֶּ֙לֶךְ֙ הָֽעֶלְי֔וֹן אֲשֶׁ֖ר לַחֲצַ֣ר הַמַּטָּרָ֑ה אַחֲרָ֖יו פְּדָיָ֥ה בֶן־ פַּרְעֹֽשׁ׃ (מז) וְהַ֨נְּתִינִ֔ים הָי֥וּ יֹשְׁבִ֖ים בָּעֹ֑פֶל עַ֠ד נֶ֜גֶד שַׁ֤עַר הַמַּ֙יִם֙ לַמִּזְרָ֔ח וְהַמִּגְדָּ֖ל הַיּוֹצֵֽא׃ (מח) אַחֲרָ֛יו הֶחֱזִ֥יקוּ הַתְּקֹעִ֖ים מִדָּ֣ה שֵׁנִ֑ית מִנֶּ֜גֶד הַמִּגְדָּ֤ל הַגָּדוֹל֙ הַיּוֹצֵ֔א וְעַ֖ד חוֹמַ֥ת הָעֹֽפֶל׃ (מט) מֵעַ֣ל ׀ שַׁ֣עַר הַסּוּסִ֗ים הֶחֱזִ֙יקוּ֙ הַכֹּ֣הֲנִ֔ים אִ֖ישׁ לְנֶ֥גֶד בֵּיתֽוֹ׃ (נ) אַחֲרָ֧יו הֶחֱזִ֛יק צָד֥וֹק בֶּן־ אִמֵּ֖ר נֶ֣גֶד בֵּית֑וֹ וְאַחֲרָ֤יו הֶחֱזִיק֙ שְׁמַֽעְיָ֣ה בֶן־ שְׁכַנְיָ֔ה שֹׁמֵ֖ר שַׁ֥עַר הַמִּזְרָֽח׃ (נא) (אחרי) אַחֲרָ֨יו הֶחֱזִ֜יק חֲנַנְיָ֣ה בֶן־ שֶׁלֶמְיָ֗ה וְחָנ֧וּן בֶּן־ צָלָ֛ף הַשִּׁשִּׁ֖י מִדָּ֣ה שֵׁנִ֑י אַחֲרָ֣יו הֶחֱזִ֗יק מְשֻׁלָּם֙ בֶּן־ בֶּ֣רֶכְיָ֔ה נֶ֖גֶד נִשְׁכָּתֽוֹ׃ (נב) (אחרי) אַחֲרָ֣יו הֶחֱזִ֗יק מַלְכִּיָּה֙ בֶּן־ הַצֹּ֣רְפִ֔י עַד־ בֵּ֥ית הַנְּתִינִ֖ים וְהָרֹכְלִ֑ים נֶ֚גֶד שַׁ֣עַר הַמִּפְקָ֔ד וְעַ֖ד עֲלִיַּ֥ת הַפִּנָּֽה׃ (נג) וּבֵ֨ין עֲלִיַּ֤ת הַפִּנָּה֙ לְשַׁ֣עַר הַצֹּ֔אן הֶחֱזִ֥יקוּ הַצֹּרְפִ֖ים וְהָרֹכְלִֽים׃ (נד) וַיְהִ֞י כַּאֲשֶׁ֧ר שָׁמַ֣ע סַנְבַלַּ֗ט כִּֽי־ אֲנַ֤חְנוּ בוֹנִים֙ אֶת־ הַ֣חוֹמָ֔ה וַיִּ֣חַר ל֔וֹ וַיִּכְעַ֖ס הַרְבֵּ֑ה וַיַּלְעֵ֖ג עַל־ הַיְּהוּדִֽים׃ (נה) וַיֹּ֣אמֶר ׀ לִפְנֵ֣י אֶחָ֗יו וְחֵיל֙ שֹֽׁמְר֔וֹן וַיֹּ֕אמֶר מָ֛ה הַיְּהוּדִ֥ים הָאֲמֵלָלִ֖ים עֹשִׂ֑ים הֲיַעַזְב֨וּ לָהֶ֤ם הֲיִזְבָּ֙חוּ֙ הַיְכַלּ֣וּ בַיּ֔וֹם הַיְחַיּ֧וּ אֶת־ הָאֲבָנִ֛ים מֵעֲרֵמ֥וֹת הֶעָפָ֖ר וְהֵ֥מָּה שְׂרוּפֽוֹת׃ (נו) וְטוֹבִיָּ֥ה הָעַמֹּנִ֖י אֶצְל֑וֹ וַיֹּ֗אמֶר גַּ֚ם אֲשֶׁר־ הֵ֣ם בּוֹנִ֔ים אִם־ יַעֲלֶ֣ה שׁוּעָ֔ל וּפָרַ֖ץ חוֹמַ֥ת אַבְנֵיהֶֽם׃ (נז) שְׁמַ֤ע אֱלֹהֵ֙ינוּ֙ כִּֽי־ הָיִ֣ינוּ בוּזָ֔ה וְהָשֵׁ֥ב חֶרְפָּתָ֖ם אֶל־ רֹאשָׁ֑ם וּתְנֵ֥ם לְבִזָּ֖ה בְּאֶ֥רֶץ שִׁבְיָֽה׃ (נח) וְאַל־ תְּכַס֙ עַל־ עֲוֺנָ֔ם וְחַטָּאתָ֖ם מִלְּפָנֶ֣יךָ אַל־ תִּמָּחֶ֑ה כִּ֥י הִכְעִ֖יסוּ לְנֶ֥גֶד הַבּוֹנִֽים׃
