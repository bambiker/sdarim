(א) וּבְכָל־ זֹ֕את אֲנַ֛חְנוּ כֹּרְתִ֥ים אֲמָנָ֖ה וְכֹתְבִ֑ים וְעַל֙ הֶֽחָת֔וּם שָׂרֵ֥ינוּ לְוִיֵּ֖נוּ כֹּהֲנֵֽינוּ׃ (ב) וְעַ֖ל הַחֲתוּמִ֑ים נְחֶמְיָ֧ה הַתִּרְשָׁ֛תָא בֶּן־ חֲכַלְיָ֖ה וְצִדְקִיָּֽה׃ (ג) שְׂרָיָ֥ה עֲזַרְיָ֖ה יִרְמְיָֽה׃ (ד) פַּשְׁח֥וּר אֲמַרְיָ֖ה מַלְכִּיָּֽה׃ (ה) חַטּ֥וּשׁ שְׁבַנְיָ֖ה מַלּֽוּךְ׃ (ו) חָרִ֥ם מְרֵמ֖וֹת עֹֽבַדְיָֽה׃ (ז) דָּנִיֵּ֥אל גִּנְּת֖וֹן בָּרֽוּךְ׃ (ח) מְשֻׁלָּ֥ם אֲבִיָּ֖ה מִיָּמִֽן׃ (ט) מַֽעַזְיָ֥ה בִלְגַּ֖י שְׁמַֽעְיָ֑ה אֵ֖לֶּה הַכֹּהֲנִֽים׃ (י) וְֽהַלְוִיִּ֑ם וְיֵשׁ֙וּעַ֙ בֶּן־ אֲזַנְיָ֔ה בִּנּ֕וּי מִבְּנֵ֥י חֵנָדָ֖ד קַדְמִיאֵֽל׃ (יא) וַאֲחֵיהֶ֑ם שְׁבַנְיָ֧ה הֽוֹדִיָּ֛ה קְלִיטָ֖א פְּלָאיָ֥ה חָנָֽן׃ (יב) מִיכָ֥א רְח֖וֹב חֲשַׁבְיָֽה׃ (יג) זַכּ֥וּר שֵׁרֵֽבְיָ֖ה שְׁבַנְיָֽה׃ (יד) הוֹדִיָּ֥ה בָנִ֖י בְּנִֽינוּ׃ (טו) רָאשֵׁ֖י הָעָ֑ם פַּרְעֹשׁ֙ פַּחַ֣ת מוֹאָ֔ב עֵילָ֥ם זַתּ֖וּא בָּנִֽי׃ (טז) בֻּנִּ֥י עַזְגָּ֖ד בֵּבָֽי׃ (יז) אֲדֹנִיָּ֥ה בִגְוַ֖י עָדִֽין׃ (יח) אָטֵ֥ר חִזְקִיָּ֖ה עַזּֽוּר׃ (יט) הוֹדִיָּ֥ה חָשֻׁ֖ם בֵּצָֽי׃ (כ) חָרִ֥יף עֲנָת֖וֹת (נובי) נֵיבָֽי׃ (כא) מַגְפִּיעָ֥שׁ מְשֻׁלָּ֖ם חֵזִֽיר׃ (כב) מְשֵׁיזַבְאֵ֥ל צָד֖וֹק יַדּֽוּעַ׃ (כג) פְּלַטְיָ֥ה חָנָ֖ן עֲנָיָֽה׃ (כד) הוֹשֵׁ֥עַ חֲנַנְיָ֖ה חַשּֽׁוּב׃ (כה) הַלּוֹחֵ֥שׁ פִּלְחָ֖א שׁוֹבֵֽק׃ (כו) רְח֥וּם חֲשַׁבְנָ֖ה מַעֲשֵׂיָֽה׃ (כז) וַאֲחִיָּ֥ה חָנָ֖ן עָנָֽן׃ (כח) מַלּ֥וּךְ חָרִ֖ם בַּעֲנָֽה׃ (כט) וּשְׁאָ֣ר הָעָ֡ם הַכֹּהֲנִ֣ים הַ֠לְוִיִּם הַשּׁוֹעֲרִ֨ים הַמְשֹׁרְרִ֜ים הַנְּתִינִ֗ים וְֽכָל־ הַנִּבְדָּ֞ל מֵעַמֵּ֤י הָאֲרָצוֹת֙ אֶל־ תּוֹרַ֣ת הָאֱלֹהִ֔ים נְשֵׁיהֶ֖ם בְּנֵיהֶ֣ם וּבְנֹתֵיהֶ֑ם כֹּ֖ל יוֹדֵ֥עַ מֵבִֽין׃ (ל) מַחֲזִיקִ֣ים עַל־ אֲחֵיהֶם֮ אַדִּירֵיהֶם֒ וּבָאִ֞ים בְּאָלָ֣ה וּבִשְׁבוּעָ֗ה לָלֶ֙כֶת֙ בְּתוֹרַ֣ת הָאֱלֹהִ֔ים אֲשֶׁ֣ר נִתְּנָ֔ה בְּיַ֖ד מֹשֶׁ֣ה עֶֽבֶד־ הָֽאֱלֹהִ֑ים וְלִשְׁמ֣וֹר וְלַעֲשׂ֗וֹת אֶת־ כָּל־ מִצְוֺת֙ יְהוָ֣ה אֲדֹנֵ֔ינוּ וּמִשְׁפָּטָ֖יו וְחֻקָּֽיו׃ (לא) וַאֲשֶׁ֛ר לֹא־ נִתֵּ֥ן בְּנֹתֵ֖ינוּ לְעַמֵּ֣י הָאָ֑רֶץ וְאֶת־ בְּנֹ֣תֵיהֶ֔ם לֹ֥א נִקַּ֖ח לְבָנֵֽינוּ׃ (לב) וְעַמֵּ֣י הָאָ֡רֶץ הַֽמְבִיאִים֩ אֶת־ הַמַּקָּח֨וֹת וְכָל־ שֶׁ֜בֶר בְּי֤וֹם הַשַּׁבָּת֙ לִמְכּ֔וֹר לֹא־ נִקַּ֥ח מֵהֶ֛ם בַּשַּׁבָּ֖ת וּבְי֣וֹם קֹ֑דֶשׁ וְנִטֹּ֛שׁ אֶת־ הַשָּׁנָ֥ה הַשְּׁבִיעִ֖ית וּמַשָּׁ֥א כָל־ יָֽד׃ (לג) וְהֶעֱמַ֤דְנוּ עָלֵ֙ינוּ֙ מִצְוֺ֔ת לָתֵ֥ת עָלֵ֛ינוּ שְׁלִשִׁ֥ית הַשֶּׁ֖קֶל בַּשָּׁנָ֑ה לַעֲבֹדַ֖ת בֵּ֥ית אֱלֹהֵֽינוּ׃ (לד) לְלֶ֣חֶם הַֽמַּעֲרֶ֡כֶת וּמִנְחַ֣ת הַתָּמִ֣יד וּלְעוֹלַ֣ת הַ֠תָּמִיד הַשַּׁבָּת֨וֹת הֶחֳדָשִׁ֜ים לַמּוֹעֲדִ֗ים וְלַקֳּדָשִׁים֙ וְלַ֣חַטָּא֔וֹת לְכַפֵּ֖ר עַל־ יִשְׂרָאֵ֑ל וְכֹ֖ל מְלֶ֥אכֶת בֵּית־ אֱלֹהֵֽינוּ׃ (לה) וְהַגּוֹרָל֨וֹת הִפַּ֜לְנוּ עַל־ קֻרְבַּ֣ן הָעֵצִ֗ים הַכֹּהֲנִ֣ים הַלְוִיִּם֮ וְהָעָם֒ לְ֠הָבִיא לְבֵ֨ית אֱלֹהֵ֧ינוּ לְבֵית־ אֲבֹתֵ֛ינוּ לְעִתִּ֥ים מְזֻמָּנִ֖ים שָׁנָ֣ה בְשָׁנָ֑ה לְבַעֵ֗ר עַל־ מִזְבַּח֙ יְהוָ֣ה אֱלֹהֵ֔ינוּ כַּכָּת֖וּב בַּתּוֹרָֽה׃ (לו) וּלְהָבִ֞יא אֶת־ בִּכּוּרֵ֣י אַדְמָתֵ֗נוּ וּבִכּוּרֵ֛י כָּל־ פְּרִ֥י כָל־ עֵ֖ץ שָׁנָ֣ה בְשָׁנָ֑ה לְבֵ֖ית יְהוָֽה׃ (לז) וְאֶת־ בְּכֹר֤וֹת בָּנֵ֙ינוּ֙ וּבְהֶמְתֵּ֔ינוּ כַּכָּת֖וּב בַּתּוֹרָ֑ה וְאֶת־ בְּכוֹרֵ֨י בְקָרֵ֜ינוּ וְצֹאנֵ֗ינוּ לְהָבִיא֙ לְבֵ֣ית אֱלֹהֵ֔ינוּ לַכֹּ֣הֲנִ֔ים הַמְשָׁרְתִ֖ים בְּבֵ֥ית אֱלֹהֵֽינוּ׃ (לח) וְאֶת־ רֵאשִׁ֣ית עֲרִיסֹתֵ֣ינוּ וּ֠תְרוּמֹתֵינוּ וּפְרִ֨י כָל־ עֵ֜ץ תִּיר֣וֹשׁ וְיִצְהָ֗ר נָבִ֤יא לַכֹּהֲנִים֙ אֶל־ לִשְׁכ֣וֹת בֵּית־ אֱלֹהֵ֔ינוּ וּמַעְשַׂ֥ר אַדְמָתֵ֖נוּ לַלְוִיִּ֑ם וְהֵם֙ הַלְוִיִּ֔ם הַֽמְעַשְּׂרִ֔ים בְּכֹ֖ל עָרֵ֥י עֲבֹדָתֵֽנוּ׃ (לט) וְהָיָ֨ה הַכֹּהֵ֧ן בֶּֽן־ אַהֲרֹ֛ן עִם־ הַלְוִיִּ֖ם בַּעְשֵׂ֣ר הַלְוִיִּ֑ם וְהַלְוִיִּ֞ם יַעֲל֨וּ אֶת־ מַעֲשַׂ֤ר הַֽמַּעֲשֵׂר֙ לְבֵ֣ית אֱלֹהֵ֔ינוּ אֶל־ הַלְּשָׁכ֖וֹת לְבֵ֥ית הָאוֹצָֽר׃ (מ) כִּ֣י אֶל־ הַ֠לְּשָׁכוֹת יָבִ֨יאוּ בְנֵי־ יִשְׂרָאֵ֜ל וּבְנֵ֣י הַלֵּוִ֗י אֶת־ תְּרוּמַ֣ת הַדָּגָן֮ הַתִּיר֣וֹשׁ וְהַיִּצְהָר֒ וְשָׁם֙ כְּלֵ֣י הַמִּקְדָּ֔שׁ וְהַכֹּהֲנִים֙ הַמְשָׁ֣רְתִ֔ים וְהַשּׁוֹעֲרִ֖ים וְהַמְשֹׁרְרִ֑ים וְלֹ֥א נַעֲזֹ֖ב אֶת־ בֵּ֥ית אֱלֹהֵֽינוּ׃ (מא) וַיֵּשְׁב֥וּ שָׂרֵֽי־ הָעָ֖ם בִּירוּשָׁלִָ֑ם וּשְׁאָ֣ר הָ֠עָם הִפִּ֨ילוּ גוֹרָל֜וֹת לְהָבִ֣יא ׀ אֶחָ֣ד מִן־ הָעֲשָׂרָ֗ה לָשֶׁ֙בֶת֙ בִּֽירוּשָׁלִַ֙ם֙ עִ֣יר הַקֹּ֔דֶשׁ וְתֵ֥שַׁע הַיָּד֖וֹת בֶּעָרִֽים׃ (מב) וַֽיְבָרֲכ֖וּ הָעָ֑ם לְכֹל֙ הָֽאֲנָשִׁ֔ים הַמִּֽתְנַדְּבִ֔ים לָשֶׁ֖בֶת בִּירוּשָׁלִָֽם׃ (מג) וְאֵ֙לֶּה֙ רָאשֵׁ֣י הַמְּדִינָ֔ה אֲשֶׁ֥ר יָשְׁב֖וּ בִּירוּשָׁלִָ֑ם וּבְעָרֵ֣י יְהוּדָ֗ה יָֽשְׁב֞וּ אִ֤ישׁ בַּאֲחֻזָּתוֹ֙ בְּעָ֣רֵיהֶ֔ם יִשְׂרָאֵ֤ל הַכֹּהֲנִים֙ וְהַלְוִיִּ֣ם וְהַנְּתִינִ֔ים וּבְנֵ֖י עַבְדֵ֥י שְׁלֹמֹֽה׃ (מד) וּבִֽירוּשָׁלִַ֙ם֙ יָֽשְׁב֔וּ מִבְּנֵ֥י יְהוּדָ֖ה וּמִבְּנֵ֣י בִנְיָמִ֑ן מִבְּנֵ֣י יְ֠הוּדָה עֲתָיָ֨ה בֶן־ עֻזִּיָּ֜ה בֶּן־ זְכַרְיָ֧ה בֶן־ אֲמַרְיָ֛ה בֶּן־ שְׁפַטְיָ֥ה בֶן־ מַהֲלַלְאֵ֖ל מִבְּנֵי־ פָֽרֶץ׃ (מה) וּמַעֲשֵׂיָ֣ה בֶן־ בָּר֣וּךְ בֶּן־ כָּל־ חֹ֠זֶה בֶּן־ חֲזָיָ֨ה בֶן־ עֲדָיָ֧ה בֶן־ יוֹיָרִ֛יב בֶּן־ זְכַרְיָ֖ה בֶּן־ הַשִּׁלֹנִֽי׃ (מו) כָּל־ בְּנֵי־ פֶ֕רֶץ הַיֹּשְׁבִ֖ים בִּירוּשָׁלִָ֑ם אַרְבַּ֥ע מֵא֛וֹת שִׁשִּׁ֥ים וּשְׁמֹנָ֖ה אַנְשֵׁי־ חָֽיִל׃ (מז) וְאֵ֖לֶּה בְּנֵ֣י בִנְיָמִ֑ן סַלֻּ֡א בֶּן־ מְשֻׁלָּ֡ם בֶּן־ יוֹעֵ֡ד בֶּן־ פְּדָיָה֩ בֶן־ ק֨וֹלָיָ֧ה בֶן־ מַעֲשֵׂיָ֛ה בֶּן־ אִֽיתִיאֵ֖ל בֶּן־ יְשַֽׁעְיָֽה׃ (מח) וְאַחֲרָ֖יו גַּבַּ֣י סַלָּ֑י תְּשַׁ֥ע מֵא֖וֹת עֶשְׂרִ֥ים וּשְׁמֹנָֽה׃ (מט) וְיוֹאֵ֥ל בֶּן־ זִכְרִ֖י פָּקִ֣יד עֲלֵיהֶ֑ם וִיהוּדָ֧ה בֶן־ הַסְּנוּאָ֛ה עַל־ הָעִ֖יר מִשְׁנֶֽה׃ (נ) מִן־ הַֽכֹּהֲנִ֑ים יְדַֽעְיָ֥ה בֶן־ יוֹיָרִ֖יב יָכִֽין׃ (נא) שְׂרָיָ֨ה בֶן־ חִלְקִיָּ֜ה בֶּן־ מְשֻׁלָּ֣ם בֶּן־ צָד֗וֹק בֶּן־ מְרָיוֹת֙ בֶּן־ אֲחִיט֔וּב נְגִ֖ד בֵּ֥ית הָאֱלֹהִֽים׃ (נב) וַאֲחֵיהֶ֗ם עֹשֵׂ֤י הַמְּלָאכָה֙ לַבַּ֔יִת שְׁמֹנֶ֥ה מֵא֖וֹת עֶשְׂרִ֣ים וּשְׁנָ֑יִם וַ֠עֲדָיָה בֶּן־ יְרֹחָ֤ם בֶּן־ פְּלַלְיָה֙ בֶּן־ אַמְצִ֣י בֶן־ זְכַרְיָ֔ה בֶּן־ פַּשְׁח֖וּר בֶּן־ מַלְכִּיָּֽה׃ (נג) וְאֶחָיו֙ רָאשִׁ֣ים לְאָב֔וֹת מָאתַ֖יִם אַרְבָּעִ֣ים וּשְׁנָ֑יִם וַעֲמַשְׁסַ֧י בֶּן־ עֲזַרְאֵ֛ל בֶּן־ אַחְזַ֥י בֶּן־ מְשִׁלֵּמ֖וֹת בֶּן־ אִמֵּֽר׃ (נד) וַאֲחֵיהֶם֙ גִּבּ֣וֹרֵי חַ֔יִל מֵאָ֖ה עֶשְׂרִ֣ים וּשְׁמֹנָ֑ה וּפָקִ֣יד עֲלֵיהֶ֔ם זַבְדִּיאֵ֖ל בֶּן־ הַגְּדוֹלִֽים׃ (נה) וּמִֽן־ הַלְוִיִּ֑ם שְׁמַעְיָ֧ה בֶן־ חַשּׁ֛וּב בֶּן־ עַזְרִיקָ֥ם בֶּן־ חֲשַׁבְיָ֖ה בֶּן־ בּוּנִּֽי׃ (נו) וְשַׁבְּתַ֨י וְיוֹזָבָ֜ד עַל־ הַמְּלָאכָ֤ה הַחִֽיצֹנָה֙ לְבֵ֣ית הָאֱלֹהִ֔ים מֵרָאשֵׁ֖י הַלְוִיִּֽם׃ (נז) וּמַתַּנְיָ֣ה בֶן־ מִ֠יכָה בֶּן־ זַבְדִּ֨י בֶן־ אָסָ֜ף רֹ֗אשׁ הַתְּחִלָּה֙ יְהוֹדֶ֣ה לַתְּפִלָּ֔ה וּבַקְבֻּקְיָ֖ה מִשְׁנֶ֣ה מֵאֶחָ֑יו וְעַבְדָּא֙ בֶּן־ שַׁמּ֔וּעַ בֶּן־ גָּלָ֖ל בֶּן־ (ידיתון) יְדוּתֽוּן׃ (נח) כָּל־ הַלְוִיִּם֙ בְּעִ֣יר הַקֹּ֔דֶשׁ מָאתַ֖יִם שְׁמֹנִ֥ים וְאַרְבָּעָֽה׃ (נט) וְהַשּֽׁוֹעֲרִים֙ עַקּ֣וּב טַלְמ֔וֹן וַאֲחֵיהֶ֖ם הַשֹּׁמְרִ֣ים בַּשְּׁעָרִ֑ים מֵאָ֖ה שִׁבְעִ֥ים וּשְׁנָֽיִם׃ (ס) וּשְׁאָ֨ר יִשְׂרָאֵ֜ל הַכֹּהֲנִ֤ים הַלְוִיִּם֙ בְּכָל־ עָרֵ֣י יְהוּדָ֔ה אִ֖ישׁ בְּנַחֲלָתֽוֹ׃ (סא) וְהַנְּתִינִ֖ים יֹשְׁבִ֣ים בָּעֹ֑פֶל וְצִיחָ֥א וְגִשְׁפָּ֖א עַל־ הַנְּתִינִֽים׃ (סב) וּפְקִ֤יד הַלְוִיִּם֙ בִּיר֣וּשָׁלִַ֔ם עֻזִּ֤י בֶן־ בָּנִי֙ בֶּן־ חֲשַׁבְיָ֔ה בֶּן־ מַתַּנְיָ֖ה בֶּן־ מִיכָ֑א מִבְּנֵ֤י אָסָף֙ הַמְשֹׁ֣רְרִ֔ים לְנֶ֖גֶד מְלֶ֥אכֶת בֵּית־ הָאֱלֹהִֽים׃ (סג) כִּֽי־ מִצְוַ֥ת הַמֶּ֖לֶךְ עֲלֵיהֶ֑ם וַאֲמָנָ֥ה עַל־ הַמְשֹׁרְרִ֖ים דְּבַר־ י֥וֹם בְּיוֹמֽוֹ׃ (סד) וּפְתַֽחְיָ֨ה בֶּן־ מְשֵֽׁיזַבְאֵ֜ל מִבְּנֵי־ זֶ֤רַח בֶּן־ יְהוּדָה֙ לְיַ֣ד הַמֶּ֔לֶךְ לְכָל־ דָּבָ֖ר לָעָֽם׃ (סה) וְאֶל־ הַחֲצֵרִ֖ים בִּשְׂדֹתָ֑ם מִבְּנֵ֣י יְהוּדָ֗ה יָֽשְׁב֞וּ בְּקִרְיַ֤ת הָֽאַרְבַּע֙ וּבְנֹתֶ֔יהָ וּבְדִיבֹן֙ וּבְנֹתֶ֔יהָ וּבִֽיקַּבְצְאֵ֖ל וַחֲצֵרֶֽיהָ׃ (סו) וּבְיֵשׁ֥וּעַ וּבְמוֹלָדָ֖ה וּבְבֵ֥ית פָּֽלֶט׃ (סז) וּבַחֲצַ֥ר שׁוּעָ֛ל וּבִבְאֵ֥ר שֶׁ֖בַע וּבְנֹתֶֽיהָ׃ (סח) וּבְצִֽקְלַ֥ג וּבִמְכֹנָ֖ה וּבִבְנֹתֶֽיהָ׃ (סט) וּבְעֵ֥ין רִמּ֛וֹן וּבְצָרְעָ֖ה וּבְיַרְמֽוּת׃ (ע) זָנֹ֤חַ עֲדֻלָּם֙ וְחַצְרֵיהֶ֔ם לָכִישׁ֙ וּשְׂדֹתֶ֔יהָ עֲזֵקָ֖ה וּבְנֹתֶ֑יהָ וַיַּחֲנ֥וּ מִבְּאֵֽר־ שֶׁ֖בַע עַד־ גֵּֽיא־ הִנֹּֽם׃ (עא) וּבְנֵ֥י בִנְיָמִ֖ן מִגָּ֑בַע מִכְמָ֣שׂ וְעַיָּ֔ה וּבֵֽית־ אֵ֖ל וּבְנֹתֶֽיהָ׃ (עב) עֲנָת֥וֹת נֹ֖ב עֲנָֽנְיָֽה׃ (עג) חָצ֥וֹר ׀ רָמָ֖ה גִּתָּֽיִם׃ (עד) חָדִ֥יד צְבֹעִ֖ים נְבַלָּֽט׃ (עה) לֹ֥ד וְאוֹנ֖וֹ גֵּ֥י הַחֲרָשִֽׁים׃ (עו) וּמִן־ הַלְוִיִּ֔ם מַחְלְק֥וֹת יְהוּדָ֖ה לְבִנְיָמִֽין׃ (עז) וְאֵ֙לֶּה֙ הַכֹּהֲנִ֣ים וְהַלְוִיִּ֔ם אֲשֶׁ֥ר עָל֛וּ עִם־ זְרֻבָּבֶ֥ל בֶּן־ שְׁאַלְתִּיאֵ֖ל וְיֵשׁ֑וּעַ שְׂרָיָ֥ה יִרְמְיָ֖ה עֶזְרָֽא׃ (עח) אֲמַרְיָ֥ה מַלּ֖וּךְ חַטּֽוּשׁ׃ (עט) שְׁכַנְיָ֥ה רְחֻ֖ם מְרֵמֹֽת׃ (פ) עִדּ֥וֹא גִנְּת֖וֹי אֲבִיָּֽה׃ (פא) מִיָּמִ֥ין מַֽעַדְיָ֖ה בִּלְגָּֽה׃ (פב) שְׁמַֽעְיָ֥ה וְיוֹיָרִ֖יב יְדַֽעְיָֽה׃ (פג) סַלּ֣וּ עָמ֔וֹק חִלְקִיָּ֖ה יְדַֽעְיָ֑ה אֵ֣לֶּה רָאשֵׁ֧י הַכֹּהֲנִ֛ים וַאֲחֵיהֶ֖ם בִּימֵ֥י יֵשֽׁוּעַ׃ (פד) וְהַלְוִיִּ֗ם יֵשׁ֧וּעַ בִּנּ֛וּי קַדְמִיאֵ֥ל שֵׁרֵבְיָ֖ה יְהוּדָ֣ה מַתַּנְיָ֑ה עַֽל־ הֻיְּד֖וֹת ה֥וּא וְאֶחָֽיו׃ (פה) וּבַקְבֻּֽקְיָ֨ה (וענו) וְעֻנִּ֧י אֲחֵיהֶ֛ם לְנֶגְדָּ֖ם לְמִשְׁמָרֽוֹת׃ (פו) וְיֵשׁ֖וּעַ הוֹלִ֣יד אֶת־ יֽוֹיָקִ֑ים וְיֽוֹיָקִים֙ הוֹלִ֣יד אֶת־ אֶלְיָשִׁ֔יב וְאֶלְיָשִׁ֖יב אֶת־ יוֹיָדָֽע׃ (פז) וְיוֹיָדָע֙ הוֹלִ֣יד אֶת־ יוֹנָתָ֔ן וְיוֹנָתָ֖ן הוֹלִ֥יד אֶת־ יַדּֽוּעַ׃ (פח) וּבִימֵי֙ יֽוֹיָקִ֔ים הָי֥וּ כֹהֲנִ֖ים רָאשֵׁ֣י הָאָב֑וֹת לִשְׂרָיָ֣ה מְרָיָ֔ה לְיִרְמְיָ֖ה חֲנַנְיָֽה׃ (פט) לְעֶזְרָ֣א מְשֻׁלָּ֔ם לַאֲמַרְיָ֖ה יְהוֹחָנָֽן׃ (צ) (למלוכי) לִמְלִ֙יכוּ֙ יֽוֹנָתָ֔ן לִשְׁבַנְיָ֖ה יוֹסֵֽף׃ (צא) לְחָרִ֣ם עַדְנָ֔א לִמְרָי֖וֹת חֶלְקָֽי׃ (צב) (לעדיא) לְעִדּ֥וֹא זְכַרְיָ֖ה לְגִנְּת֥וֹן מְשֻׁלָּֽם׃ (צג) לַאֲבִיָּ֖ה זִכְרִ֑י לְמִ֨נְיָמִ֔ין לְמוֹעַדְיָ֖ה פִּלְטָֽי׃ (צד) לְבִלְגָּ֣ה שַׁמּ֔וּעַ לִֽשְׁמַעְיָ֖ה יְהוֹנָתָֽן׃ (צה) וּלְיוֹיָרִ֣יב מַתְּנַ֔י לִֽידַֽעְיָ֖ה עֻזִּֽי׃ (צו) לְסַלַּ֥י קַלָּ֖י לְעָמ֥וֹק עֵֽבֶר׃ (צז) לְחִלְקִיָּ֣ה חֲשַׁבְיָ֔ה לִֽידַֽעְיָ֖ה נְתַנְאֵֽל׃ (צח) הַלְוִיִּם֩ בִּימֵ֨י אֶלְיָשִׁ֜יב יוֹיָדָ֤ע וְיוֹחָנָן֙ וְיַדּ֔וּעַ כְּתוּבִ֖ים רָאשֵׁ֣י אָב֑וֹת וְהַכֹּ֣הֲנִ֔ים עַל־ מַלְכ֖וּת דָּרְיָ֥וֶשׁ הַפָּֽרְסִֽי׃ (צט) בְּנֵ֤י לֵוִי֙ רָאשֵׁ֣י הָֽאָב֔וֹת כְּתוּבִ֕ים עַל־ סֵ֖פֶר דִּבְרֵ֣י הַיָּמִ֑ים וְעַד־ יְמֵ֖י יֽוֹחָנָ֥ן בֶּן־ אֶלְיָשִֽׁיב׃ (ק) וְרָאשֵׁ֣י הַ֠לְוִיִּם חֲשַׁבְיָ֨ה שֵֽׁרֵבְיָ֜ה וְיֵשׁ֤וּעַ בֶּן־ קַדְמִיאֵל֙ וַאֲחֵיהֶ֣ם לְנֶגְדָּ֔ם לְהַלֵּ֣ל לְהוֹד֔וֹת בְּמִצְוַ֖ת דָּוִ֣יד אִישׁ־ הָאֱלֹהִ֑ים מִשְׁמָ֖ר לְעֻמַּ֥ת מִשְׁמָֽר׃ (קא) מַתַּנְיָ֧ה וּבַקְבֻּֽקְיָ֛ה עֹבַדְיָ֥ה מְשֻׁלָּ֖ם טַלְמ֣וֹן עַקּ֑וּב שֹׁמְרִ֤ים שֽׁוֹעֲרִים֙ מִשְׁמָ֔ר בַּאֲסֻפֵּ֖י הַשְּׁעָרִֽים׃ (קב) אֵ֕לֶּה בִּימֵ֛י יוֹיָקִ֥ים בֶּן־ יֵשׁ֖וּעַ בֶּן־ יוֹצָדָ֑ק וּבִימֵי֙ נְחֶמְיָ֣ה הַפֶּחָ֔ה וְעֶזְרָ֥א הַכֹּהֵ֖ן הַסּוֹפֵֽר׃
