עזרא ונחמיה סדר ז

פתיחה באתר · ספר עזרא ונחמיה כולו ·

(א) וַתִּשְׁלַם֙ הַֽחוֹמָ֔ה בְּעֶשְׂרִ֥ים וַחֲמִשָּׁ֖ה לֶאֱל֑וּל לַחֲמִשִּׁ֥ים וּשְׁנַ֖יִם יֽוֹם׃ (ב) וַיְהִ֗י כַּאֲשֶׁ֤ר שָֽׁמְעוּ֙ כָּל־ א֣וֹיְבֵ֔ינוּ וַיִּֽרְא֗וּ כָּל־ הַגּוֹיִם֙ אֲשֶׁ֣ר סְבִֽיבֹתֵ֔ינוּ וַיִּפְּל֥וּ מְאֹ֖ד בְּעֵינֵיהֶ֑ם וַיֵּ֣דְע֔וּ כִּ֚י מֵאֵ֣ת אֱלֹהֵ֔ינוּ נֶעֶשְׂתָ֖ה הַמְּלָאכָ֥ה הַזֹּֽאת׃ (ג) גַּ֣ם ׀ בַּיָּמִ֣ים הָהֵ֗ם מַרְבִּ֞ים חֹרֵ֤י יְהוּדָה֙ אִגְּרֹ֣תֵיהֶ֔ם הוֹלְכ֖וֹת עַל־ טוֹבִיָּ֑ה וַאֲשֶׁ֥ר לְטוֹבִיָּ֖ה בָּא֥וֹת אֲלֵיהֶֽם׃ (ד) כִּי־ רַבִּ֣ים בִּֽיהוּדָ֗ה בַּעֲלֵ֤י שְׁבוּעָה֙ ל֔וֹ כִּי־ חָתָ֥ן ה֖וּא לִשְׁכַנְיָ֣ה בֶן־ אָרַ֑ח וִֽיהוֹחָנָ֣ן בְּנ֔וֹ לָקַ֕ח אֶת־ בַּת־ מְשֻׁלָּ֖ם בֶּ֥ן בֶּֽרֶכְיָֽה׃ (ה) גַּ֣ם טוֹבֹתָ֗יו הָי֤וּ אֹמְרִים֙ לְפָנַ֔י וּדְבָרַ֕י הָי֥וּ מוֹצִיאִ֖ים ל֑וֹ אִגְּר֛וֹת שָׁלַ֥ח טוֹבִיָּ֖ה לְיָֽרְאֵֽנִי׃ (ו) וַיְהִ֗י כַּאֲשֶׁ֤ר נִבְנְתָה֙ הַחוֹמָ֔ה וָאַעֲמִ֖יד הַדְּלָת֑וֹת וַיִּפָּֽקְד֛וּ הַשּׁוֹעֲרִ֥ים וְהַמְשֹׁרְרִ֖ים וְהַלְוִיִּֽם׃ (ז) וָאֲצַוֶּ֞ה אֶת־ חֲנָ֣נִי אָחִ֗י וְאֶת־ חֲנַנְיָ֛ה שַׂ֥ר הַבִּירָ֖ה עַל־ יְרוּשָׁלִָ֑ם כִּי־ הוּא֙ כְּאִ֣ישׁ אֱמֶ֔ת וְיָרֵ֥א אֶת־ הָאֱלֹהִ֖ים מֵרַבִּֽים׃ (ח) (ויאמר) וָאֹמַ֣ר לָהֶ֗ם לֹ֣א יִפָּֽתְח֞וּ שַׁעֲרֵ֤י יְרוּשָׁלִַ֙ם֙ עַד־ חֹ֣ם הַשֶּׁ֔מֶשׁ וְעַ֨ד הֵ֥ם עֹמְדִ֛ים יָגִ֥יפוּ הַדְּלָת֖וֹת וֶאֱחֹ֑זוּ וְהַעֲמֵ֗יד מִשְׁמְרוֹת֙ יֹשְׁבֵ֣י יְרוּשָׁלִַ֔ם אִ֚ישׁ בְּמִשְׁמָר֔וֹ וְאִ֖ישׁ נֶ֥גֶד בֵּיתֽוֹ׃ (ט) וְהָעִ֞יר רַחֲבַ֤ת יָדַ֙יִם֙ וּגְדוֹלָ֔ה וְהָעָ֥ם מְעַ֖ט בְּתוֹכָ֑הּ וְאֵ֥ין בָּתִּ֖ים בְּנוּיִֽם׃ (י) וַיִּתֵּ֤ן אֱלֹהַי֙ אֶל־ לִבִּ֔י וָאֶקְבְּצָ֞ה אֶת־ הַחֹרִ֧ים וְאֶת־ הַסְּגָנִ֛ים וְאֶת־ הָעָ֖ם לְהִתְיַחֵ֑שׂ וָֽאֶמְצָ֗א סֵ֤פֶר הַיַּ֙חַשׂ֙ הָעוֹלִ֣ים בָּרִאשׁוֹנָ֔ה וָאֶמְצָ֖א כָּת֥וּב בּֽוֹ׃ (יא) אֵ֣לֶּה ׀ בְּנֵ֣י הַמְּדִינָ֗ה הָעֹלִים֙ מִשְּׁבִ֣י הַגּוֹלָ֔ה אֲשֶׁ֣ר הֶגְלָ֔ה נְבוּכַדְנֶצַּ֖ר מֶ֣לֶךְ בָּבֶ֑ל וַיָּשׁ֧וּבוּ לִֽירוּשָׁלִַ֛ם וְלִיהוּדָ֖ה אִ֥ישׁ לְעִירֽוֹ׃ (יב) הַבָּאִ֣ים עִם־ זְרֻבָּבֶ֗ל יֵשׁ֡וּעַ נְחֶמְיָ֡ה עֲ֠זַרְיָה רַֽעַמְיָ֨ה נַחֲמָ֜נִי מָרְדֳּכַ֥י בִּלְשָׁ֛ן מִסְפֶּ֥רֶת בִּגְוַ֖י נְח֣וּם בַּעֲנָ֑ה מִסְפַּ֕ר אַנְשֵׁ֖י עַ֥ם יִשְׂרָאֵֽל׃ (יג) בְּנֵ֣י פַרְעֹ֔שׁ אַלְפַּ֕יִם מֵאָ֖ה וְשִׁבְעִ֥ים וּשְׁנָֽיִם׃ (יד) בְּנֵ֣י שְׁפַטְיָ֔ה שְׁלֹ֥שׁ מֵא֖וֹת שִׁבְעִ֥ים וּשְׁנָֽיִם׃ (טו) בְּנֵ֣י אָרַ֔ח שֵׁ֥שׁ מֵא֖וֹת חֲמִשִּׁ֥ים וּשְׁנָֽיִם׃ (טז) בְּנֵֽי־ פַחַ֥ת מוֹאָ֛ב לִבְנֵ֥י יֵשׁ֖וּעַ וְיוֹאָ֑ב אַלְפַּ֕יִם וּשְׁמֹנֶ֥ה מֵא֖וֹת שְׁמֹנָ֥ה עָשָֽׂר׃ (יז) בְּנֵ֣י עֵילָ֔ם אֶ֕לֶף מָאתַ֖יִם חֲמִשִּׁ֥ים וְאַרְבָּעָֽה׃ (יח) בְּנֵ֣י זַתּ֔וּא שְׁמֹנֶ֥ה מֵא֖וֹת אַרְבָּעִ֥ים וַחֲמִשָּֽׁה׃ (יט) בְּנֵ֣י זַכָּ֔י שְׁבַ֥ע מֵא֖וֹת וְשִׁשִּֽׁים׃ (כ) בְּנֵ֣י בִנּ֔וּי שֵׁ֥שׁ מֵא֖וֹת אַרְבָּעִ֥ים וּשְׁמֹנָֽה׃ (כא) בְּנֵ֣י בֵבָ֔י שֵׁ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וּשְׁמֹנָֽה׃ (כב) בְּנֵ֣י עַזְגָּ֔ד אַלְפַּ֕יִם שְׁלֹ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וּשְׁנָֽיִם׃ (כג) בְּנֵי֙ אֲדֹ֣נִיקָ֔ם שֵׁ֥שׁ מֵא֖וֹת שִׁשִּׁ֥ים וְשִׁבְעָֽה׃ (כד) בְּנֵ֣י בִגְוָ֔י אַלְפַּ֖יִם שִׁשִּׁ֥ים וְשִׁבְעָֽה׃ (כה) בְּנֵ֣י עָדִ֔ין שֵׁ֥שׁ מֵא֖וֹת חֲמִשִּׁ֥ים וַחֲמִשָּֽׁה׃ (כו) בְּנֵֽי־ אָטֵ֥ר לְחִזְקִיָּ֖ה תִּשְׁעִ֥ים וּשְׁמֹנָֽה׃ (כז) בְּנֵ֣י חָשֻׁ֔ם שְׁלֹ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וּשְׁמֹנָֽה׃ (כח) בְּנֵ֣י בֵצָ֔י שְׁלֹ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וְאַרְבָּעָֽה׃ (כט) בְּנֵ֣י חָרִ֔יף מֵאָ֖ה שְׁנֵ֥ים עָשָֽׂר׃ (ל) בְּנֵ֥י גִבְע֖וֹן תִּשְׁעִ֥ים וַחֲמִשָּֽׁה׃ (לא) אַנְשֵׁ֤י בֵֽית־ לֶ֙חֶם֙ וּנְטֹפָ֔ה מֵאָ֖ה שְׁמֹנִ֥ים וּשְׁמֹנָֽה׃ (לב) אַנְשֵׁ֣י עֲנָת֔וֹת מֵאָ֖ה עֶשְׂרִ֥ים וּשְׁמֹנָֽה׃ (לג) אַנְשֵׁ֥י בֵית־ עַזְמָ֖וֶת אַרְבָּעִ֥ים וּשְׁנָֽיִם׃ (לד) אַנְשֵׁ֨י קִרְיַ֤ת יְעָרִים֙ כְּפִירָ֣ה וּבְאֵר֔וֹת שְׁבַ֥ע מֵא֖וֹת אַרְבָּעִ֥ים וּשְׁלֹשָֽׁה׃ (לה) אַנְשֵׁ֤י הָֽרָמָה֙ וָגָ֔בַע שֵׁ֥שׁ מֵא֖וֹת עֶשְׂרִ֥ים וְאֶחָֽד׃ (לו) אַנְשֵׁ֣י מִכְמָ֔ס מֵאָ֖ה וְעֶשְׂרִ֥ים וּשְׁנָֽיִם׃ (לז) אַנְשֵׁ֤י בֵֽית־ אֵל֙ וְהָעָ֔י מֵאָ֖ה עֶשְׂרִ֥ים וּשְׁלֹשָֽׁה׃ (לח) אַנְשֵׁ֥י נְב֛וֹ אַחֵ֖ר חֲמִשִּׁ֥ים וּשְׁנָֽיִם׃ (לט) בְּנֵי֙ עֵילָ֣ם אַחֵ֔ר אֶ֕לֶף מָאתַ֖יִם חֲמִשִּׁ֥ים וְאַרְבָּעָֽה׃ (מ) בְּנֵ֣י חָרִ֔ם שְׁלֹ֥שׁ מֵא֖וֹת וְעֶשְׂרִֽים׃ (מא) בְּנֵ֣י יְרֵח֔וֹ שְׁלֹ֥שׁ מֵא֖וֹת אַרְבָּעִ֥ים וַחֲמִשָּֽׁה׃ (מב) בְּנֵי־ לֹד֙ חָדִ֣יד וְאוֹנ֔וֹ שְׁבַ֥ע מֵא֖וֹת וְעֶשְׂרִ֥ים וְאֶחָֽד׃ (מג) בְּנֵ֣י סְנָאָ֔ה שְׁלֹ֣שֶׁת אֲלָפִ֔ים תְּשַׁ֥ע מֵא֖וֹת וּשְׁלֹשִֽׁים׃ (מד) הַֽכֹּהֲנִ֑ים בְּנֵ֤י יְדַֽעְיָה֙ לְבֵ֣ית יֵשׁ֔וּעַ תְּשַׁ֥ע מֵא֖וֹת שִׁבְעִ֥ים וּשְׁלֹשָֽׁה׃ (מה) בְּנֵ֣י אִמֵּ֔ר אֶ֖לֶף חֲמִשִּׁ֥ים וּשְׁנָֽיִם׃ (מו) בְּנֵ֣י פַשְׁח֔וּר אֶ֕לֶף מָאתַ֖יִם אַרְבָּעִ֥ים וְשִׁבְעָֽה׃ (מז) בְּנֵ֣י חָרִ֔ם אֶ֖לֶף שִׁבְעָ֥ה עָשָֽׂר׃ (מח) הַלְוִיִּ֑ם בְּנֵֽי־ יֵשׁ֧וּעַ לְקַדְמִיאֵ֛ל לִבְנֵ֥י לְהוֹדְוָ֖ה שִׁבְעִ֥ים וְאַרְבָּעָֽה׃ (מט) הַֽמְשֹׁרְרִ֑ים בְּנֵ֣י אָסָ֔ף מֵאָ֖ה אַרְבָּעִ֥ים וּשְׁמֹנָֽה׃ (נ) הַשֹּֽׁעֲרִ֗ים בְּנֵֽי־ שַׁלּ֤וּם בְּנֵֽי־ אָטֵר֙ בְּנֵֽי־ טַלְמֹ֣ן בְּנֵֽי־ עַקּ֔וּב בְּנֵ֥י חֲטִיטָ֖א בְּנֵ֣י שֹׁבָ֑י מֵאָ֖ה שְׁלֹשִׁ֥ים וּשְׁמֹנָֽה׃ (נא) הַנְּתִינִ֑ים בְּנֵי־ צִחָ֥א בְנֵי־ חֲשֻׂפָ֖א בְּנֵ֥י טַבָּעֽוֹת׃ (נב) בְּנֵי־ קֵירֹ֥ס בְּנֵי־ סִיעָ֖א בְּנֵ֥י פָדֽוֹן׃ (נג) בְּנֵי־ לְבָנָ֥ה בְנֵי־ חֲגָבָ֖ה בְּנֵ֥י שַׁלְמָֽי׃ (נד) בְּנֵי־ חָנָ֥ן בְּנֵי־ גִדֵּ֖ל בְּנֵי־ גָֽחַר׃ (נה) בְּנֵי־ רְאָיָ֥ה בְנֵי־ רְצִ֖ין בְּנֵ֥י נְקוֹדָֽא׃ (נו) בְּנֵי־ גַזָּ֥ם בְּנֵי־ עֻזָּ֖א בְּנֵ֥י פָסֵֽחַ׃ (נז) בְּנֵי־ בֵסַ֥י בְּנֵי־ מְעוּנִ֖ים בְּנֵ֥י (נפושסים) נְפִֽישְׁסִֽים׃ (נח) בְּנֵי־ בַקְבּ֥וּק בְּנֵֽי־ חֲקוּפָ֖א בְּנֵ֥י חַרְחֽוּר׃ (נט) בְּנֵי־ בַצְלִ֥ית בְּנֵֽי־ מְחִידָ֖א בְּנֵ֥י חַרְשָֽׁא׃ (ס) בְּנֵי־ בַרְק֥וֹס בְּֽנֵי־ סִֽיסְרָ֖א בְּנֵי־ תָֽמַח׃ (סא) בְּנֵ֥י נְצִ֖יחַ בְּנֵ֥י חֲטִיפָֽא׃ (סב) בְּנֵ֖י עַבְדֵ֣י שְׁלֹמֹ֑ה בְּנֵי־ סוֹטַ֥י בְּנֵי־ סוֹפֶ֖רֶת בְּנֵ֥י פְרִידָֽא׃ (סג) בְּנֵי־ יַעְלָ֥א בְנֵי־ דַרְק֖וֹן בְּנֵ֥י גִדֵּֽל׃ (סד) בְּנֵ֧י שְׁפַטְיָ֣ה בְנֵֽי־ חַטִּ֗יל בְּנֵ֛י פֹּכֶ֥רֶת הַצְּבָיִ֖ים בְּנֵ֥י אָמֽוֹן׃ (סה) כָּל־ הַ֨נְּתִינִ֔ים וּבְנֵ֖י עַבְדֵ֣י שְׁלֹמֹ֑ה שְׁלֹ֥שׁ מֵא֖וֹת תִּשְׁעִ֥ים וּשְׁנָֽיִם׃ (סו) וְאֵ֗לֶּה הָֽעוֹלִים֙ מִתֵּ֥ל מֶ֙לַח֙ תֵּ֣ל חַרְשָׁ֔א כְּר֥וּב אַדּ֖וֹן וְאִמֵּ֑ר וְלֹ֣א יָכְל֗וּ לְהַגִּ֤יד בֵּית־ אֲבוֹתָם֙ וְזַרְעָ֔ם אִ֥ם מִיִּשְׂרָאֵ֖ל הֵֽם׃ (סז) בְּנֵי־ דְלָיָ֥ה בְנֵֽי־ טוֹבִיָּ֖ה בְּנֵ֣י נְקוֹדָ֑א שֵׁ֥שׁ מֵא֖וֹת וְאַרְבָּעִ֥ים וּשְׁנָֽיִם׃ (סח) וּמִן־ הַכֹּ֣הֲנִ֔ים בְּנֵ֥י חֳבַיָּ֖ה בְּנֵ֣י הַקּ֑וֹץ בְּנֵ֣י בַרְזִלַּ֗י אֲשֶׁ֣ר לָ֠קַח מִבְּנ֞וֹת בַּרְזִלַּ֤י הַגִּלְעָדִי֙ אִשָּׁ֔ה וַיִּקָּרֵ֖א עַל־ שְׁמָֽם׃ (סט) אֵ֗לֶּה בִּקְשׁ֧וּ כְתָבָ֛ם הַמִּתְיַחְשִׂ֖ים וְלֹ֣א נִמְצָ֑א וַיְגֹֽאֲל֖וּ מִן־ הַכְּהֻנָּֽה׃ (ע) וַיֹּ֤אמֶר הַתִּרְשָׁ֙תָא֙ לָהֶ֔ם אֲשֶׁ֥ר לֹא־ יֹאכְל֖וּ מִקֹּ֣דֶשׁ הַקֳּדָשִׁ֑ים עַ֛ד עֲמֹ֥ד הַכֹּהֵ֖ן לְאוּרִ֥ים וְתוּמִּֽים׃ (עא) כָּל־ הַקָּהָ֖ל כְּאֶחָ֑ד אַרְבַּ֣ע רִבּ֔וֹא אַלְפַּ֖יִם שְׁלֹשׁ־ מֵא֥וֹת וְשִׁשִּֽׁים׃ (עב) מִ֠לְּבַד עַבְדֵיהֶ֤ם וְאַמְהֹֽתֵיהֶם֙ אֵ֔לֶּה שִׁבְעַ֣ת אֲלָפִ֔ים שְׁלֹ֥שׁ מֵא֖וֹת שְׁלֹשִׁ֣ים וְשִׁבְעָ֑ה וְלָהֶ֗ם מְשֹֽׁרֲרִים֙ וּמְשֹׁ֣רֲר֔וֹת מָאתַ֖יִם וְאַרְבָּעִ֥ים וַחֲמִשָּֽׁה׃ (עג) גְּמַלִּ֕ים אַרְבַּ֥ע מֵא֖וֹת שְׁלֹשִׁ֣ים וַחֲמִשָּׁ֑ה חֲמֹרִ֕ים שֵׁ֣שֶׁת אֲלָפִ֔ים שְׁבַ֥ע מֵא֖וֹת וְעֶשְׂרִֽים׃ (עד) וּמִקְצָת֙ רָאשֵׁ֣י הָֽאָב֔וֹת נָתְנ֖וּ לַמְּלָאכָ֑ה הַתִּרְשָׁ֜תָא נָתַ֣ן לָאוֹצָ֗ר זָהָ֞ב דַּרְכְּמֹנִ֥ים אֶ֙לֶף֙ מִזְרָק֣וֹת חֲמִשִּׁ֔ים כָּתְנוֹת֙ כֹּֽהֲנִ֔ים שְׁלֹשִׁ֖ים וַחֲמֵ֥שׁ מֵאֽוֹת׃ (עה) וּמֵֽרָאשֵׁ֣י הָֽאָב֗וֹת נָֽתְנוּ֙ לְאוֹצַ֣ר הַמְּלָאכָ֔ה זָהָ֕ב דַּרְכְּמוֹנִ֖ים שְׁתֵּ֣י רִבּ֑וֹת וְכֶ֕סֶף מָנִ֖ים אַלְפַּ֥יִם וּמָאתָֽיִם׃ (עו) וַאֲשֶׁ֣ר נָתְנוּ֮ שְׁאֵרִ֣ית הָעָם֒ זָהָ֗ב דַּרְכְּמוֹנִים֙ שְׁתֵּ֣י רִבּ֔וֹא וְכֶ֖סֶף מָנִ֣ים אַלְפָּ֑יִם וְכָתְנֹ֥ת כֹּֽהֲנִ֖ים שִׁשִּׁ֥ים וְשִׁבְעָֽה׃ (עז) וַיֵּשְׁב֣וּ הַכֹּהֲנִ֣ים וְהַלְוִיִּ֡ם וְהַשּׁוֹעֲרִים֩ וְהַמְשֹׁרְרִ֨ים וּמִן־ הָעָ֧ם וְהַנְּתִינִ֛ים וְכָל־ יִשְׂרָאֵ֖ל בְּעָרֵיהֶ֑ם וַיִּגַּע֙ הַחֹ֣דֶשׁ הַשְּׁבִיעִ֔י וּבְנֵ֥י יִשְׂרָאֵ֖ל בְּעָרֵיהֶֽם׃ (עח) וַיֵּאָסְפ֤וּ כָל־ הָעָם֙ כְּאִ֣ישׁ אֶחָ֔ד אֶל־ הָ֣רְח֔וֹב אֲשֶׁ֖ר לִפְנֵ֣י שַֽׁעַר־ הַמָּ֑יִם וַיֹּֽאמְרוּ֙ לְעֶזְרָ֣א הַסֹּפֵ֔ר לְהָבִ֗יא אֶת־ סֵ֙פֶר֙ תּוֹרַ֣ת מֹשֶׁ֔ה אֲשֶׁר־ צִוָּ֥ה יְהוָ֖ה אֶת־ יִשְׂרָאֵֽל׃ (עט) וַיָּבִ֣יא עֶזְרָ֣א הַ֠כֹּהֵן אֶֽת־ הַתּוֹרָ֞ה לִפְנֵ֤י הַקָּהָל֙ מֵאִ֣ישׁ וְעַד־ אִשָּׁ֔ה וְכֹ֖ל מֵבִ֣ין לִשְׁמֹ֑עַ בְּי֥וֹם אֶחָ֖ד לַחֹ֥דֶשׁ הַשְּׁבִיעִֽי׃ (פ) וַיִּקְרָא־ בוֹ֩ לִפְנֵ֨י הָרְח֜וֹב אֲשֶׁ֣ר ׀ לִפְנֵ֣י שַֽׁעַר־ הַמַּ֗יִם מִן־ הָאוֹר֙ עַד־ מַחֲצִ֣ית הַיּ֔וֹם נֶ֛גֶד הָאֲנָשִׁ֥ים וְהַנָּשִׁ֖ים וְהַמְּבִינִ֑ים וְאָזְנֵ֥י כָל־ הָעָ֖ם אֶל־ סֵ֥פֶר הַתּוֹרָֽה׃ (פא) וַֽיַּעֲמֹ֞ד עֶזְרָ֣א הַסֹּפֵ֗ר עַֽל־ מִגְדַּל־ עֵץ֮ אֲשֶׁ֣ר עָשׂ֣וּ לַדָּבָר֒ וַיַּֽעֲמֹ֣ד אֶצְל֡וֹ מַתִּתְיָ֡ה וְשֶׁ֡מַע וַ֠עֲנָיָה וְאוּרִיָּ֧ה וְחִלְקִיָּ֛ה וּמַעֲשֵׂיָ֖ה עַל־ יְמִינ֑וֹ וּמִשְּׂמֹאל֗וֹ פְּ֠דָיָה וּמִֽישָׁאֵ֧ל וּמַלְכִּיָּ֛ה וְחָשֻׁ֥ם וְחַשְׁבַּדָּ֖נָה זְכַרְיָ֥ה מְשֻׁלָּֽם׃ (פב) וַיִּפְתַּ֨ח עֶזְרָ֤א הַסֵּ֙פֶר֙ לְעֵינֵ֣י כָל־ הָעָ֔ם כִּֽי־ מֵעַ֥ל כָּל־ הָעָ֖ם הָיָ֑ה וּכְפִתְח֖וֹ עָֽמְד֥וּ כָל־ הָעָֽם׃ (פג) וַיְבָ֣רֶךְ עֶזְרָ֔א אֶת־ יְהוָ֥ה הָאֱלֹהִ֖ים הַגָּד֑וֹל וַיַּֽעֲנ֨וּ כָל־ הָעָ֜ם אָמֵ֤ן ׀ אָמֵן֙ בְּמֹ֣עַל יְדֵיהֶ֔ם וַיִּקְּד֧וּ וַיִּשְׁתַּחֲוֻּ֛ לַיהוָ֖ה אַפַּ֥יִם אָֽרְצָה׃ (פד) וְיֵשׁ֡וּעַ וּבָנִ֡י וְשֵׁרֵ֥בְיָ֣ה ׀ יָמִ֡ין עַקּ֡וּב שַׁבְּתַ֣י ׀ הֽוֹדִיָּ֡ה מַעֲשֵׂיָ֡ה קְלִיטָ֣א עֲזַרְיָה֩ יוֹזָבָ֨ד חָנָ֤ן פְּלָאיָה֙ וְהַלְוִיִּ֔ם מְבִינִ֥ים אֶת־ הָעָ֖ם לַתּוֹרָ֑ה וְהָעָ֖ם עַל־ עָמְדָֽם׃ (פה) וַֽיִּקְרְא֥וּ בַסֵּ֛פֶר בְּתוֹרַ֥ת הָאֱלֹהִ֖ים מְפֹרָ֑שׁ וְשׂ֣וֹם שֶׂ֔כֶל וַיָּבִ֖ינוּ בַּמִּקְרָֽא׃ (פו) וַיֹּ֣אמֶר נְחֶמְיָ֣ה ה֣וּא הַתִּרְשָׁ֡תָא וְעֶזְרָ֣א הַכֹּהֵ֣ן ׀ הַסֹּפֵ֡ר וְהַלְוִיִּם֩ הַמְּבִינִ֨ים אֶת־ הָעָ֜ם לְכָל־ הָעָ֗ם הַיּ֤וֹם קָדֹֽשׁ־ הוּא֙ לַיהוָ֣ה אֱלֹהֵיכֶ֔ם אַל־ תִּֽתְאַבְּל֖וּ וְאַל־ תִּבְכּ֑וּ כִּ֤י בוֹכִים֙ כָּל־ הָעָ֔ם כְּשָׁמְעָ֖ם אֶת־ דִּבְרֵ֥י הַתּוֹרָֽה׃
