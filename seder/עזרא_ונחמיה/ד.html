עזרא ונחמיה סדר ד

לפי החלוקה לפרקים: עזרא ח, לה – נחמיה א, י · 71 פסוקים

פתיחה באתר · ספר עזרא ונחמיה כולו ·

(א) הַ֠בָּאִים מֵֽהַשְּׁבִ֨י בְנֵֽי־ הַגּוֹלָ֜ה הִקְרִ֥יבוּ עֹל֣וֹת ׀ לֵאלֹהֵ֣י יִשְׂרָאֵ֗ל פָּרִ֨ים שְׁנֵים־ עָשָׂ֤ר עַל־ כָּל־ יִשְׂרָאֵל֙ אֵילִ֣ים ׀ תִּשְׁעִ֣ים וְשִׁשָּׁ֗ה כְּבָשִׂים֙ שִׁבְעִ֣ים וְשִׁבְעָ֔ה צְפִירֵ֥י חַטָּ֖את שְׁנֵ֣ים עָשָׂ֑ר הַכֹּ֖ל עוֹלָ֥ה לַיהוָֽה׃ (ב) וַֽיִּתְּנ֣וּ ׀ אֶת־ דָּתֵ֣י הַמֶּ֗לֶךְ לַאֲחַשְׁדַּרְפְּנֵי֙ הַמֶּ֔לֶךְ וּפַחֲו֖וֹת עֵ֣בֶר הַנָּהָ֑ר וְנִשְּׂא֥וּ אֶת־ הָעָ֖ם וְאֶת־ בֵּֽית־ הָאֱלֹהִֽים׃ (ג) וּכְכַלּ֣וֹת אֵ֗לֶּה נִגְּשׁ֨וּ אֵלַ֤י הַשָּׂרִים֙ לֵאמֹ֔ר לֹֽא־ נִבְדְּל֞וּ הָעָ֤ם יִשְׂרָאֵל֙ וְהַכֹּהֲנִ֣ים וְהַלְוִיִּ֔ם מֵעַמֵּ֖י הָאֲרָצ֑וֹת כְּ֠תוֹעֲבֹֽתֵיהֶם לַכְּנַעֲנִ֨י הַחִתִּ֜י הַפְּרִזִּ֣י הַיְבוּסִ֗י הָֽעַמֹּנִי֙ הַמֹּ֣אָבִ֔י הַמִּצְרִ֖י וְהָאֱמֹרִֽי׃ (ד) כִּֽי־ נָשְׂא֣וּ מִבְּנֹֽתֵיהֶ֗ם לָהֶם֙ וְלִבְנֵיהֶ֔ם וְהִתְעָֽרְבוּ֙ זֶ֣רַע הַקֹּ֔דֶשׁ בְּעַמֵּ֖י הָאֲרָצ֑וֹת וְיַ֧ד הַשָּׂרִ֣ים וְהַסְּגָנִ֗ים הָֽיְתָ֛ה בַּמַּ֥עַל הַזֶּ֖ה רִאשׁוֹנָֽה׃ (ה) וּכְשָׁמְעִי֙ אֶת־ הַדָּבָ֣ר הַזֶּ֔ה קָרַ֥עְתִּי אֶת־ בִּגְדִ֖י וּמְעִילִ֑י וָאֶמְרְטָ֞ה מִשְּׂעַ֤ר רֹאשִׁי֙ וּזְקָנִ֔י וָאֵשְׁבָ֖ה מְשׁוֹמֵֽם׃ (ו) וְאֵלַ֣י יֵאָסְפ֗וּ כֹּ֤ל חָרֵד֙ בְּדִבְרֵ֣י אֱלֹהֵֽי־ יִשְׂרָאֵ֔ל עַ֖ל מַ֣עַל הַגּוֹלָ֑ה וַאֲנִי֙ יֹשֵׁ֣ב מְשׁוֹמֵ֔ם עַ֖ד לְמִנְחַ֥ת הָעָֽרֶב׃ (ז) וּבְמִנְחַ֣ת הָעֶ֗רֶב קַ֚מְתִּי מִתַּֽעֲנִיתִ֔י וּבְקָרְעִ֥י בִגְדִ֖י וּמְעִילִ֑י וָֽאֶכְרְעָה֙ עַל־ בִּרְכַּ֔י וָאֶפְרְשָׂ֥ה כַפַּ֖י אֶל־ יְהוָ֥ה אֱלֹהָֽי׃ (ח) וָאֹמְרָ֗ה אֱלֹהַי֙ בֹּ֣שְׁתִּי וְנִכְלַ֔מְתִּי לְהָרִ֧ים אֱלֹהַ֛י פָּנַ֖י אֵלֶ֑יךָ כִּ֣י עֲוֺנֹתֵ֤ינוּ רָבוּ֙ לְמַ֣עְלָה רֹּ֔אשׁ וְאַשְׁמָתֵ֥נוּ גָדְלָ֖ה עַ֥ד לַשָּׁמָֽיִם׃ (ט) מִימֵ֣י אֲבֹתֵ֗ינוּ אֲנַ֙חְנוּ֙ בְּאַשְׁמָ֣ה גְדֹלָ֔ה עַ֖ד הַיּ֣וֹם הַזֶּ֑ה וּבַעֲוֺנֹתֵ֡ינוּ נִתַּ֡נּוּ אֲנַחְנוּ֩ מְלָכֵ֨ינוּ כֹהֲנֵ֜ינוּ בְּיַ֣ד ׀ מַלְכֵ֣י הָאֲרָצ֗וֹת בַּחֶ֜רֶב בַּשְּׁבִ֧י וּבַבִּזָּ֛ה וּבְבֹ֥שֶׁת פָּנִ֖ים כְּהַיּ֥וֹם הַזֶּֽה׃ (י) וְעַתָּ֡ה כִּמְעַט־ רֶגַע֩ הָיְתָ֨ה תְחִנָּ֜ה מֵאֵ֣ת ׀ יְהוָ֣ה אֱלֹהֵ֗ינוּ לְהַשְׁאִ֥יר לָ֙נוּ֙ פְּלֵיטָ֔ה וְלָתֶת־ לָ֥נוּ יָתֵ֖ד בִּמְק֣וֹם קָדְשׁ֑וֹ לְהָאִ֤יר עֵינֵ֙ינוּ֙ אֱלֹהֵ֔ינוּ וּלְתִתֵּ֛נוּ מִֽחְיָ֥ה מְעַ֖ט בְּעַבְדֻתֵֽנוּ׃ (יא) כִּֽי־ עֲבָדִ֣ים אֲנַ֔חְנוּ וּבְעַבְדֻ֔תֵנוּ לֹ֥א עֲזָבָ֖נוּ אֱלֹהֵ֑ינוּ וַֽיַּט־ עָלֵ֣ינוּ חֶ֡סֶד לִפְנֵי֩ מַלְכֵ֨י פָרַ֜ס לָֽתֶת־ לָ֣נוּ מִֽחְיָ֗ה לְרוֹמֵ֞ם אֶת־ בֵּ֤ית אֱלֹהֵ֙ינוּ֙ וּלְהַעֲמִ֣יד אֶת־ חָרְבֹתָ֔יו וְלָֽתֶת־ לָ֣נוּ גָדֵ֔ר בִּֽיהוּדָ֖ה וּבִירוּשָׁלִָֽם׃ (יב) וְעַתָּ֛ה מַה־ נֹּאמַ֥ר אֱלֹהֵ֖ינוּ אַֽחֲרֵי־ זֹ֑את כִּ֥י עָזַ֖בְנוּ מִצְוֺתֶֽיךָ׃ (יג) אֲשֶׁ֣ר צִוִּ֗יתָ בְּיַ֨ד עֲבָדֶ֣יךָ הַנְּבִיאִים֮ לֵאמֹר֒ הָאָ֗רֶץ אֲשֶׁ֨ר אַתֶּ֤ם בָּאִים֙ לְרִשְׁתָּ֔הּ אֶ֤רֶץ נִדָּה֙ הִ֔יא בְּנִדַּ֖ת עַמֵּ֣י הָאֲרָצ֑וֹת בְּתוֹעֲבֹֽתֵיהֶ֗ם אֲשֶׁ֥ר מִלְא֛וּהָ מִפֶּ֥ה אֶל־ פֶּ֖ה בְּטֻמְאָתָֽם׃ (יד) וְ֠עַתָּה בְּֽנוֹתֵיכֶ֞ם אַל־ תִּתְּנ֣וּ לִבְנֵיהֶ֗ם וּבְנֹֽתֵיהֶם֙ אַל־ תִּשְׂא֣וּ לִבְנֵיכֶ֔ם וְלֹֽא־ תִדְרְשׁ֧וּ שְׁלֹמָ֛ם וְטוֹבָתָ֖ם עַד־ עוֹלָ֑ם לְמַ֣עַן תֶּחֶזְק֗וּ וַאֲכַלְתֶּם֙ אֶת־ ט֣וּב הָאָ֔רֶץ וְהוֹרַשְׁתֶּ֥ם לִבְנֵיכֶ֖ם עַד־ עוֹלָֽם׃ (טו) וְאַֽחֲרֵי֙ כָּל־ הַבָּ֣א עָלֵ֔ינוּ בְּמַעֲשֵׂ֙ינוּ֙ הָרָעִ֔ים וּבְאַשְׁמָתֵ֖נוּ הַגְּדֹלָ֑ה כִּ֣י ׀ אַתָּ֣ה אֱלֹהֵ֗ינוּ חָשַׂ֤כְתָּֽ לְמַ֙טָּה֙ מֵֽעֲוֺנֵ֔נוּ וְנָתַ֥תָּה לָּ֛נוּ פְּלֵיטָ֖ה כָּזֹֽאת׃ (טז) הֲנָשׁוּב֙ לְהָפֵ֣ר מִצְוֺתֶ֔יךָ וּ֨לְהִתְחַתֵּ֔ן בְּעַמֵּ֥י הַתֹּעֵב֖וֹת הָאֵ֑לֶּה הֲל֤וֹא תֶֽאֱנַף־ בָּ֙נוּ֙ עַד־ כַּלֵּ֔ה לְאֵ֥ין שְׁאֵרִ֖ית וּפְלֵיטָֽה׃ (יז) יְהוָ֞ה אֱלֹהֵ֤י יִשְׂרָאֵל֙ צַדִּ֣יק אַ֔תָּה כִּֽי־ נִשְׁאַ֥רְנוּ פְלֵיטָ֖ה כְּהַיּ֣וֹם הַזֶּ֑ה הִנְנ֤וּ לְפָנֶ֙יךָ֙ בְּאַשְׁמָתֵ֔ינוּ כִּ֣י אֵ֥ין לַעֲמ֛וֹד לְפָנֶ֖יךָ עַל־ זֹֽאת׃ (יח) וּכְהִתְפַּלֵּ֤ל עֶזְרָא֙ וּכְ֨הִתְוַדֹּת֔וֹ בֹּכֶה֙ וּמִתְנַפֵּ֔ל לִפְנֵ֖י בֵּ֣ית הָאֱלֹהִ֑ים נִקְבְּצוּ֩ אֵלָ֨יו מִיִּשְׂרָאֵ֜ל קָהָ֣ל רַב־ מְאֹ֗ד אֲנָשִׁ֤ים וְנָשִׁים֙ וִֽילָדִ֔ים כִּֽי־ בָכ֥וּ הָעָ֖ם הַרְבֵּה־ בֶֽכֶה׃ (יט) וַיַּעַן֩ שְׁכַנְיָ֨ה בֶן־ יְחִיאֵ֜ל מִבְּנֵ֤י (עולם) עֵילָם֙ וַיֹּ֣אמֶר לְעֶזְרָ֔א אֲנַ֙חְנוּ֙ מָעַ֣לְנוּ בֵאלֹהֵ֔ינוּ וַנֹּ֛שֶׁב נָשִׁ֥ים נָכְרִיּ֖וֹת מֵעַמֵּ֣י הָאָ֑רֶץ וְעַתָּ֛ה יֵשׁ־ מִקְוֶ֥ה לְיִשְׂרָאֵ֖ל עַל־ זֹֽאת׃ (כ) וְעַתָּ֣ה נִֽכְרָת־ בְּרִ֣ית לֵ֠אלֹהֵינוּ לְהוֹצִ֨יא כָל־ נָשִׁ֜ים וְהַנּוֹלָ֤ד מֵהֶם֙ בַּעֲצַ֣ת אֲדֹנָ֔י וְהַחֲרֵדִ֖ים בְּמִצְוַ֣ת אֱלֹהֵ֑ינוּ וְכַתּוֹרָ֖ה יֵעָשֶֽׂה׃ (כא) ק֛וּם כִּֽי־ עָלֶ֥יךָ הַדָּבָ֖ר וַאֲנַ֣חְנוּ עִמָּ֑ךְ חֲזַ֖ק וַעֲשֵֽׂה׃ (כב) וַיָּ֣קָם עֶזְרָ֡א וַיַּשְׁבַּ֣ע אֶת־ שָׂרֵי֩ הַכֹּהֲנִ֨ים הַלְוִיִּ֜ם וְכָל־ יִשְׂרָאֵ֗ל לַעֲשׂ֛וֹת כַּדָּבָ֥ר הַזֶּ֖ה וַיִּשָּׁבֵֽעוּ׃ (כג) וַיָּ֣קָם עֶזְרָ֗א מִלִּפְנֵי֙ בֵּ֣ית הָֽאֱלֹהִ֔ים וַיֵּ֕לֶךְ אֶל־ לִשְׁכַּ֖ת יְהוֹחָנָ֣ן בֶּן־ אֶלְיָשִׁ֑יב וַיֵּ֣לֶךְ שָׁ֗ם לֶ֤חֶם לֹֽא־ אָכַל֙ וּמַ֣יִם לֹֽא־ שָׁתָ֔ה כִּ֥י מִתְאַבֵּ֖ל עַל־ מַ֥עַל הַגּוֹלָֽה׃ (כד) וַיַּעֲבִ֨ירוּ ק֜וֹל בִּיהוּדָ֣ה וִירֽוּשָׁלִַ֗ם לְכֹל֙ בְּנֵ֣י הַגּוֹלָ֔ה לְהִקָּבֵ֖ץ יְרוּשָׁלִָֽם׃ (כה) וְכֹל֩ אֲשֶׁ֨ר לֹֽא־ יָב֜וֹא לִשְׁלֹ֣שֶׁת הַיָּמִ֗ים כַּעֲצַ֤ת הַשָּׂרִים֙ וְהַזְּקֵנִ֔ים יָחֳרַ֖ם כָּל־ רְכוּשׁ֑וֹ וְה֥וּא יִבָּדֵ֖ל מִקְּהַ֥ל הַגּוֹלָֽה׃ (כו) וַיִּקָּבְצ֣וּ כָל־ אַנְשֵֽׁי־ יְהוּדָה֩ וּבִנְיָמִ֨ן ׀ יְרוּשָׁלִַ֜ם לִשְׁלֹ֣שֶׁת הַיָּמִ֗ים ה֛וּא חֹ֥דֶשׁ הַתְּשִׁיעִ֖י בְּעֶשְׂרִ֣ים בַּחֹ֑דֶשׁ וַיֵּשְׁב֣וּ כָל־ הָעָ֗ם בִּרְחוֹב֙ בֵּ֣ית הָאֱלֹהִ֔ים מַרְעִידִ֥ים עַל־ הַדָּבָ֖ר וּמֵהַגְּשָׁמִֽים׃ (כז) וַיָּ֨קָם עֶזְרָ֤א הַכֹּהֵן֙ וַיֹּ֣אמֶר אֲלֵהֶ֔ם אַתֶּ֣ם מְעַלְתֶּ֔ם וַתֹּשִׁ֖יבוּ נָשִׁ֣ים נָכְרִיּ֑וֹת לְהוֹסִ֖יף עַל־ אַשְׁמַ֥ת יִשְׂרָאֵֽל׃ (כח) וְעַתָּ֗ה תְּנ֥וּ תוֹדָ֛ה לַיהוָ֥ה אֱלֹהֵֽי־ אֲבֹתֵיכֶ֖ם וַעֲשׂ֣וּ רְצוֹנ֑וֹ וְהִבָּֽדְלוּ֙ מֵעַמֵּ֣י הָאָ֔רֶץ וּמִן־ הַנָּשִׁ֖ים הַנָּכְרִיּֽוֹת׃ (כט) וַיַּֽעְנ֧וּ כָֽל־ הַקָּהָ֛ל וַיֹּאמְר֖וּ ק֣וֹל גָּד֑וֹל כֵּ֛ן (כדבריך) כִּדְבָרְךָ֥ עָלֵ֖ינוּ לַעֲשֽׂוֹת׃ (ל) אֲבָ֞ל הָעָ֥ם רָב֙ וְהָעֵ֣ת גְּשָׁמִ֔ים וְאֵ֥ין כֹּ֖חַ לַעֲמ֣וֹד בַּח֑וּץ וְהַמְּלָאכָ֗ה לֹֽא־ לְי֤וֹם אֶחָד֙ וְלֹ֣א לִשְׁנַ֔יִם כִּֽי־ הִרְבִּ֥ינוּ לִפְשֹׁ֖עַ בַּדָּבָ֥ר הַזֶּֽה׃ (לא) יַֽעֲמְדוּ־ נָ֣א שָׂ֠רֵינוּ לְֽכָל־ הַקָּהָ֞ל וְכֹ֣ל ׀ אֲשֶׁ֣ר בֶּעָרֵ֗ינוּ הַהֹשִׁ֞יב נָשִׁ֤ים נָכְרִיּוֹת֙ יָבֹא֙ לְעִתִּ֣ים מְזֻמָּנִ֔ים וְעִמָּהֶ֛ם זִקְנֵי־ עִ֥יר וָעִ֖יר וְשֹׁפְטֶ֑יהָ עַ֠ד לְהָשִׁ֞יב חֲר֤וֹן אַף־ אֱלֹהֵ֙ינוּ֙ מִמֶּ֔נּוּ עַ֖ד לַדָּבָ֥ר הַזֶּֽה׃ (לב) אַ֣ךְ יוֹנָתָ֧ן בֶּן־ עֲשָׂהאֵ֛ל וְיַחְזְיָ֥ה בֶן־ תִּקְוָ֖ה עָמְד֣וּ עַל־ זֹ֑את וּמְשֻׁלָּ֛ם וְשַׁבְּתַ֥י הַלֵּוִ֖י עֲזָרֻֽם׃ (לג) וַיַּֽעֲשׂוּ־ כֵן֮ בְּנֵ֣י הַגּוֹלָה֒ וַיִּבָּדְלוּ֩ עֶזְרָ֨א הַכֹּהֵ֜ן אֲנָשִׁ֨ים רָאשֵׁ֧י הָאָב֛וֹת לְבֵ֥ית אֲבֹתָ֖ם וְכֻלָּ֣ם בְּשֵׁמ֑וֹת וַיֵּשְׁב֗וּ בְּי֤וֹם אֶחָד֙ לַחֹ֣דֶשׁ הָעֲשִׂירִ֔י לְדַרְי֖וֹשׁ הַדָּבָֽר׃ (לד) וַיְכַלּ֣וּ בַכֹּ֔ל אֲנָשִׁ֕ים הַהֹשִׁ֖יבוּ נָשִׁ֣ים נָכְרִיּ֑וֹת עַ֛ד י֥וֹם אֶחָ֖ד לַחֹ֥דֶשׁ הָרִאשֽׁוֹן׃ (לה) וַיִּמָּצֵא֙ מִבְּנֵ֣י הַכֹּהֲנִ֔ים אֲשֶׁ֥ר הֹשִׁ֖יבוּ נָשִׁ֣ים נָכְרִיּ֑וֹת מִבְּנֵ֨י יֵשׁ֤וּעַ בֶּן־ יֽוֹצָדָק֙ וְאֶחָ֔יו מַֽעֲשֵׂיָה֙ וֶֽאֱלִיעֶ֔זֶר וְיָרִ֖יב וּגְדַלְיָֽה׃ (לו) וַיִּתְּנ֥וּ יָדָ֖ם לְהוֹצִ֣יא נְשֵׁיהֶ֑ם וַאֲשֵׁמִ֥ים אֵֽיל־ צֹ֖אן עַל־ אַשְׁמָתָֽם׃ (לז) וּמִבְּנֵ֣י אִמֵּ֔ר חֲנָ֖נִי וּזְבַדְיָֽה׃ (לח) וּמִבְּנֵ֖י חָרִ֑ם מַעֲשֵׂיָ֤ה וְאֵֽלִיָּה֙ וּֽשְׁמַֽעְיָ֔ה וִיחִיאֵ֖ל וְעֻזִיָּֽה׃ (לט) וּמִבְּנֵ֖י פַּשְׁח֑וּר אֶלְיוֹעֵינַ֤י מַֽעֲשֵׂיָה֙ יִשְׁמָעֵ֣אל נְתַנְאֵ֔ל יוֹזָבָ֖ד וְאֶלְעָשָֽׂה׃ (מ) וּמִֽן־ הַלְוִיִּ֑ם יוֹזָבָ֣ד וְשִׁמְעִ֗י וְקֵֽלָיָה֙ ה֣וּא קְלִיטָ֔א פְּתַֽחְיָ֥ה יְהוּדָ֖ה וֶאֱלִיעֶֽזֶר׃ (מא) וּמִן־ הַמְשֹׁרְרִ֖ים אֶלְיָשִׁ֑יב וּמִן־ הַשֹּׁ֣עֲרִ֔ים שַׁלֻּ֥ם וָטֶ֖לֶם וְאוּרִֽי׃ (מב) וּמִֽיִּשְׂרָאֵ֑ל מִבְּנֵ֣י פַרְעֹ֡שׁ רַ֠מְיָה וְיִזִּיָּ֤ה וּמַלְכִּיָּה֙ וּמִיָּמִ֣ן וְאֶלְעָזָ֔ר וּמַלְכִּיָּ֖ה וּבְנָיָֽה׃ (מג) וּמִבְּנֵ֖י עֵילָ֑ם מַתַּנְיָ֤ה זְכַרְיָה֙ וִיחִיאֵ֣ל וְעַבְדִּ֔י וִירֵמ֖וֹת וְאֵלִיָּֽה׃ (מד) וּמִבְּנֵ֖י זַתּ֑וּא אֶלְיוֹעֵנַ֤י אֶלְיָשִׁיב֙ מַתַּנְיָ֣ה וִֽירֵמ֔וֹת וְזָבָ֖ד וַעֲזִיזָֽא׃ (מה) וּמִבְּנֵ֖י בֵּבָ֑י יְהוֹחָנָ֥ן חֲנַנְיָ֖ה זַבַּ֥י עַתְלָֽי׃ (מו) וּמִבְּנֵ֖י בָּנִ֑י מְשֻׁלָּ֤ם מַלּוּךְ֙ וַעֲדָיָ֔ה יָשׁ֖וּב וּשְׁאָ֥ל (ירמות) וְרָמֽוֹת׃ (מז) וּמִבְּנֵ֛י פַּחַ֥ת מוֹאָ֖ב עַדְנָ֣א וּכְלָ֑ל בְּנָיָ֤ה מַעֲשֵׂיָה֙ מַתַּנְיָ֣ה בְצַלְאֵ֔ל וּבִנּ֖וּי וּמְנַשֶּֽׁה׃ (מח) וּבְנֵ֖י חָרִ֑ם אֱלִיעֶ֧זֶר יִשִּׁיָּ֛ה מַלְכִּיָּ֖ה שְׁמַֽעְיָ֥ה שִׁמְעֽוֹן׃ (מט) בְּנְיָמִ֥ן מַלּ֖וּךְ שְׁמַרְיָֽה׃ (נ) מִבְּנֵ֖י חָשֻׁ֑ם מַתְּנַ֤י מַתַּתָּה֙ זָבָ֣ד אֱלִיפֶ֔לֶט יְרֵמַ֥י מְנַשֶּׁ֖ה שִׁמְעִֽי׃ (נא) מִבְּנֵ֣י בָנִ֔י מַעֲדַ֥י עַמְרָ֖ם וְאוּאֵֽל׃ (נב) בְּנָיָ֥ה בֵדְיָ֖ה (כלהי) כְּלֽוּהוּ׃ (נג) וַנְיָ֥ה מְרֵמ֖וֹת אֶלְיָשִֽׁיב׃ (נד) מַתַּנְיָ֥ה מַתְּנַ֖י (ויעשו) וְיַעֲשָֽׂי׃ (נה) וּבָנִ֥י וּבִנּ֖וּי שִׁמְעִֽי׃ (נו) וְשֶֽׁלֶמְיָ֥ה וְנָתָ֖ן וַעֲדָיָֽה׃ (נז) מַכְנַדְבַ֥י שָׁשַׁ֖י שָׁרָֽי׃ (נח) עֲזַרְאֵ֥ל וְשֶׁלֶמְיָ֖הוּ שְׁמַרְיָֽה׃ (נט) שַׁלּ֥וּם אֲמַרְיָ֖ה יוֹסֵֽף׃ (ס) מִבְּנֵ֖י נְב֑וֹ יְעִיאֵ֤ל מַתִּתְיָה֙ זָבָ֣ד זְבִינָ֔א (ידו) יַדַּ֥י וְיוֹאֵ֖ל בְּנָיָֽה׃ (סא) כָּל־ אֵ֕לֶּה (נשאי) נָשְׂא֖וּ נָשִׁ֣ים נָכְרִיּ֑וֹת וְיֵ֣שׁ מֵהֶ֣ם נָשִׁ֔ים וַיָּשִׂ֖ימוּ בָּנִֽים׃ (סב) דִּבְרֵ֥י נְחֶמְיָ֖ה בֶּן־ חֲכַלְיָ֑ה וַיְהִ֤י בְחֹֽדֶשׁ־ (כסלו) כִּסְלֵיו֙ שְׁנַ֣ת עֶשְׂרִ֔ים וַאֲנִ֥י הָיִ֖יתִי בְּשׁוּשַׁ֥ן הַבִּירָֽה׃ (סג) וַיָּבֹ֨א חֲנָ֜נִי אֶחָ֧ד מֵאַחַ֛י ה֥וּא וַאֲנָשִׁ֖ים מִֽיהוּדָ֑ה וָאֶשְׁאָלֵ֞ם עַל־ הַיְּהוּדִ֧ים הַפְּלֵיטָ֛ה אֲשֶֽׁר־ נִשְׁאֲר֥וּ מִן־ הַשֶּׁ֖בִי וְעַל־ יְרוּשָׁלִָֽם׃ (סד) וַיֹּאמְרוּ֮ לִי֒ הַֽנִּשְׁאָרִ֞ים אֲשֶֽׁר־ נִשְׁאֲר֤וּ מִן־ הַשְּׁבִי֙ שָׁ֣ם בַּמְּדִינָ֔ה בְּרָעָ֥ה גְדֹלָ֖ה וּבְחֶרְפָּ֑ה וְחוֹמַ֤ת יְרוּשָׁלִַ֙ם֙ מְפֹרָ֔צֶת וּשְׁעָרֶ֖יהָ נִצְּת֥וּ בָאֵֽשׁ׃ (סה) וַיְהִ֞י כְּשָׁמְעִ֣י ׀ אֶת־ הַדְּבָרִ֣ים הָאֵ֗לֶּה יָשַׁ֙בְתִּי֙ וָֽאֶבְכֶּ֔ה וָאֶתְאַבְּלָ֖ה יָמִ֑ים וָֽאֱהִ֥י צָם֙ וּמִתְפַּלֵּ֔ל לִפְנֵ֖י אֱלֹהֵ֥י הַשָּׁמָֽיִם׃ (סו) וָאֹמַ֗ר אָֽנָּ֤א יְהוָה֙ אֱלֹהֵ֣י הַשָּׁמַ֔יִם הָאֵ֥ל הַגָּד֖וֹל וְהַנּוֹרָ֑א שֹׁמֵ֤ר הַבְּרִית֙ וָחֶ֔סֶד לְאֹהֲבָ֖יו וּלְשֹׁמְרֵ֥י מִצְוֺתָֽיו׃ (סז) תְּהִ֣י נָ֣א אָזְנְךָֽ־ קַשֶּׁ֣בֶת וְֽעֵינֶ֪יךָ פְתֻוּח֟וֹת לִשְׁמֹ֣עַ אֶל־ תְּפִלַּ֣ת עַבְדְּךָ֡ אֲשֶׁ֣ר אָנֹכִי֩ מִתְפַּלֵּ֨ל לְפָנֶ֤יךָ הַיּוֹם֙ יוֹמָ֣ם וָלַ֔יְלָה עַל־ בְּנֵ֥י יִשְׂרָאֵ֖ל עֲבָדֶ֑יךָ וּמִתְוַדֶּ֗ה עַל־ חַטֹּ֤אות בְּנֵֽי־ יִשְׂרָאֵל֙ אֲשֶׁ֣ר חָטָ֣אנוּ לָ֔ךְ וַאֲנִ֥י וּבֵית־ אָבִ֖י חָטָֽאנוּ׃ (סח) חֲבֹ֖ל חָבַ֣לְנוּ לָ֑ךְ וְלֹא־ שָׁמַ֣רְנוּ אֶת־ הַמִּצְוֺ֗ת וְאֶת־ הַֽחֻקִּים֙ וְאֶת־ הַמִּשְׁפָּטִ֔ים אֲשֶׁ֥ר צִוִּ֖יתָ אֶת־ מֹשֶׁ֥ה עַבְדֶּֽךָ׃ (סט) זְכָר־ נָא֙ אֶת־ הַדָּבָ֔ר אֲשֶׁ֥ר צִוִּ֛יתָ אֶת־ מֹשֶׁ֥ה עַבְדְּךָ֖ לֵאמֹ֑ר אַתֶּ֣ם תִּמְעָ֔לוּ אֲנִ֕י אָפִ֥יץ אֶתְכֶ֖ם בָּעַמִּֽים׃ (ע) וְשַׁבְתֶּ֣ם אֵלַ֔י וּשְׁמַרְתֶּם֙ מִצְוֺתַ֔י וַעֲשִׂיתֶ֖ם אֹתָ֑ם אִם־ יִהְיֶ֨ה נִֽדַּחֲכֶ֜ם בִּקְצֵ֤ה הַשָּׁמַ֙יִם֙ מִשָּׁ֣ם אֲקַבְּצֵ֔ם (והבואתים) וַהֲבִֽיאוֹתִים֙ אֶל־ הַמָּק֔וֹם אֲשֶׁ֣ר בָּחַ֔רְתִּי לְשַׁכֵּ֥ן אֶת־ שְׁמִ֖י שָֽׁם׃ (עא) וְהֵ֥ם עֲבָדֶ֖יךָ וְעַמֶּ֑ךָ אֲשֶׁ֤ר פָּדִ֙יתָ֙ בְּכֹחֲךָ֣ הַגָּד֔וֹל וּבְיָדְךָ֖ הַחֲזָקָֽה׃
