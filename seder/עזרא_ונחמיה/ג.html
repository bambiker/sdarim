עזרא ונחמיה סדר ג

פתיחה באתר · ספר עזרא ונחמיה כולו ·

(א) וַהֲקִ֨ימוּ כָהֲנַיָּ֜א בִּפְלֻגָּתְה֗וֹן וְלֵוָיֵא֙ בְּמַחְלְקָ֣תְה֔וֹן עַל־ עֲבִידַ֥ת אֱלָהָ֖א דִּ֣י בִירוּשְׁלֶ֑ם כִּכְתָ֖ב סְפַ֥ר מֹשֶֽׁה׃ (ב) וַיַּעֲשׂ֥וּ בְנֵי־ הַגּוֹלָ֖ה אֶת־ הַפָּ֑סַח בְּאַרְבָּעָ֥ה עָשָׂ֖ר לַחֹ֥דֶשׁ הָרִאשֽׁוֹן׃ (ג) כִּ֣י הִֽטַּהֲר֞וּ הַכֹּהֲנִ֧ים וְהַלְוִיִּ֛ם כְּאֶחָ֖ד כֻּלָּ֣ם טְהוֹרִ֑ים וַיִּשְׁחֲט֤וּ הַפֶּ֙סַח֙ לְכָל־ בְּנֵ֣י הַגּוֹלָ֔ה וְלַאֲחֵיהֶ֥ם הַכֹּהֲנִ֖ים וְלָהֶֽם׃ (ד) וַיֹּאכְל֣וּ בְנֵֽי־ יִשְׂרָאֵ֗ל הַשָּׁבִים֙ מֵֽהַגּוֹלָ֔ה וְכֹ֗ל הַנִּבְדָּ֛ל מִטֻּמְאַ֥ת גּוֹיֵֽ־ הָאָ֖רֶץ אֲלֵהֶ֑ם לִדְרֹ֕שׁ לַֽיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (ה) וַיַּֽעֲשׂ֧וּ חַג־ מַצּ֛וֹת שִׁבְעַ֥ת יָמִ֖ים בְּשִׂמְחָ֑ה כִּ֣י ׀ שִׂמְּחָ֣ם יְהוָ֗ה וְֽהֵסֵ֞ב לֵ֤ב מֶֽלֶךְ־ אַשּׁוּר֙ עֲלֵיהֶ֔ם לְחַזֵּ֣ק יְדֵיהֶ֔ם בִּמְלֶ֥אכֶת בֵּית־ הָאֱלֹהִ֖ים אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (ו) וְאַחַר֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה בְּמַלְכ֖וּת אַרְתַּחְשַׁ֣סְתְּא מֶֽלֶךְ־ פָּרָ֑ס עֶזְרָא֙ בֶּן־ שְׂרָיָ֔ה בֶּן־ עֲזַרְיָ֖ה בֶּן־ חִלְקִיָּֽה׃ (ז) בֶּן־ שַׁלּ֥וּם בֶּן־ צָד֖וֹק בֶּן־ אֲחִיטֽוּב׃ (ח) בֶּן־ אֲמַרְיָ֥ה בֶן־ עֲזַרְיָ֖ה בֶּן־ מְרָיֽוֹת׃ (ט) בֶּן־ זְרַֽחְיָ֥ה בֶן־ עֻזִּ֖י בֶּן־ בֻּקִּֽי׃ (י) בֶּן־ אֲבִישׁ֗וּעַ בֶּן־ פִּֽינְחָס֙ בֶּן־ אֶלְעָזָ֔ר בֶּן־ אַהֲרֹ֥ן הַכֹּהֵ֖ן הָרֹֽאשׁ׃ (יא) ה֤וּא עֶזְרָא֙ עָלָ֣ה מִבָּבֶ֔ל וְהֽוּא־ סֹפֵ֤ר מָהִיר֙ בְּתוֹרַ֣ת מֹשֶׁ֔ה אֲשֶׁר־ נָתַ֥ן יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וַיִּתֶּן־ ל֣וֹ הַמֶּ֗לֶךְ כְּיַד־ יְהוָ֤ה אֱלֹהָיו֙ עָלָ֔יו כֹּ֖ל בַּקָּשָׁתֽוֹ׃ (יב) וַיַּֽעֲל֣וּ מִבְּנֵֽי־ יִ֠שְׂרָאֵל וּמִן־ הַכֹּהֲנִ֨ים וְהַלְוִיִּ֜ם וְהַמְשֹׁרְרִ֧ים וְהַשֹּׁעֲרִ֛ים וְהַנְּתִינִ֖ים אֶל־ יְרוּשָׁלִָ֑ם בִּשְׁנַת־ שֶׁ֖בַע לְאַרְתַּחְשַׁ֥סְתְּא הַמֶּֽלֶךְ׃ (יג) וַיָּבֹ֥א יְרוּשָׁלִַ֖ם בַּחֹ֣דֶשׁ הַחֲמִישִׁ֑י הִ֛יא שְׁנַ֥ת הַשְּׁבִיעִ֖ית לַמֶּֽלֶךְ׃ (יד) כִּ֗י בְּאֶחָד֙ לַחֹ֣דֶשׁ הָרִאשׁ֔וֹן ה֣וּא יְסֻ֔ד הַֽמַּעֲלָ֖ה מִבָּבֶ֑ל וּבְאֶחָ֞ד לַחֹ֣דֶשׁ הַחֲמִישִׁ֗י בָּ֚א אֶל־ יְר֣וּשָׁלִַ֔ם כְּיַד־ אֱלֹהָ֖יו הַטּוֹבָ֥ה עָלָֽיו׃ (טו) כִּ֤י עֶזְרָא֙ הֵכִ֣ין לְבָב֔וֹ לִדְר֛וֹשׁ אֶת־ תּוֹרַ֥ת יְהוָ֖ה וְלַעֲשֹׂ֑ת וּלְלַמֵּ֥ד בְּיִשְׂרָאֵ֖ל חֹ֥ק וּמִשְׁפָּֽט׃ (טז) וְזֶ֣ה ׀ פַּרְשֶׁ֣גֶן הַֽנִּשְׁתְּוָ֗ן אֲשֶׁ֤ר נָתַן֙ הַמֶּ֣לֶךְ אַרְתַּחְשַׁ֔סְתְּא לְעֶזְרָ֥א הַכֹּהֵ֖ן הַסֹּפֵ֑ר סֹפֵ֞ר דִּבְרֵ֧י מִצְוֺת־ יְהוָ֛ה וְחֻקָּ֖יו עַל־ יִשְׂרָאֵֽל׃ (יז) אַ֨רְתַּחְשַׁ֔סְתְּא מֶ֖לֶךְ מַלְכַיָּ֑א לְעֶזְרָ֣א כָ֠הֲנָא סָפַ֨ר דָּתָ֜א דִּֽי־ אֱלָ֧הּ שְׁמַיָּ֛א גְּמִ֖יר וּכְעֶֽנֶת׃ (יח) מִנִּי֮ שִׂ֣ים טְעֵם֒ דִּ֣י כָל־ מִתְנַדַּ֣ב בְּמַלְכוּתִי֩ מִן־ עַמָּ֨ה יִשְׂרָאֵ֜ל וְכָהֲנ֣וֹהִי וְלֵוָיֵ֗א לִמְהָ֧ךְ לִֽירוּשְׁלֶ֛ם עִמָּ֖ךְ יְהָֽךְ׃ (יט) כָּל־ קֳבֵ֗ל דִּי֩ מִן־ קֳדָ֨ם מַלְכָּ֜א וְשִׁבְעַ֤ת יָעֲטֹ֙הִי֙ שְׁלִ֔יחַ לְבַקָּרָ֥א עַל־ יְה֖וּד וְלִֽירוּשְׁלֶ֑ם בְּדָ֥ת אֱלָהָ֖ךְ דִּ֥י בִידָֽךְ׃ (כ) וּלְהֵיבָלָ֖ה כְּסַ֣ף וּדְהַ֑ב דִּֽי־ מַלְכָּ֣א וְיָעֲט֗וֹהִי הִתְנַדַּ֙בוּ֙ לֶאֱלָ֣הּ יִשְׂרָאֵ֔ל דִּ֥י בִֽירוּשְׁלֶ֖ם מִשְׁכְּנֵֽהּ׃ (כא) וְכֹל֙ כְּסַ֣ף וּדְהַ֔ב דִּ֣י תְהַשְׁכַּ֔ח בְּכֹ֖ל מְדִינַ֣ת בָּבֶ֑ל עִם֩ הִתְנַדָּב֨וּת עַמָּ֤א וְכָֽהֲנַיָּא֙ מִֽתְנַדְּבִ֔ין לְבֵ֥ית אֱלָהֲהֹ֖ם דִּ֥י בִירוּשְׁלֶֽם׃ (כב) כָּל־ קֳבֵ֣ל דְּנָה֩ אָסְפַּ֨רְנָא תִקְנֵ֜א בְּכַסְפָּ֣א דְנָ֗ה תּוֹרִ֤ין ׀ דִּכְרִין֙ אִמְּרִ֔ין וּמִנְחָתְה֖וֹן וְנִסְכֵּיה֑וֹן וּתְקָרֵ֣ב הִמּ֔וֹ עַֽל־ מַדְבְּחָ֔ה דִּ֛י בֵּ֥ית אֱלָהֲכֹ֖ם דִּ֥י בִירוּשְׁלֶֽם׃ (כג) וּמָ֣ה דִי֩ (עליך) עֲלָ֨ךְ וְעַל־ (אחיך) אֶחָ֜ךְ יֵיטַ֗ב בִּשְׁאָ֛ר כַּסְפָּ֥א וְדַהֲבָ֖ה לְמֶעְבַּ֑ד כִּרְע֥וּת אֱלָהֲכֹ֖ם תַּעַבְדֽוּן׃ (כד) וּמָֽאנַיָּא֙ דִּֽי־ מִתְיַהֲבִ֣ין לָ֔ךְ לְפָלְחָ֖ן בֵּ֣ית אֱלָהָ֑ךְ הַשְׁלֵ֕ם קֳדָ֖ם אֱלָ֥הּ יְרוּשְׁלֶֽם׃ (כה) וּשְׁאָ֗ר חַשְׁחוּת֙ בֵּ֣ית אֱלָהָ֔ךְ דִּ֥י יִפֶּל־ לָ֖ךְ לְמִנְתַּ֑ן תִּנְתֵּ֕ן מִן־ בֵּ֖ית גִּנְזֵ֥י מַלְכָּֽא׃ (כו) וּ֠מִנִּי אֲנָ֞ה אַרְתַּחְשַׁ֤סְתְּא מַלְכָּא֙ שִׂ֣ים טְעֵ֔ם לְכֹל֙ גִּזַּֽבְרַיָּ֔א דִּ֖י בַּעֲבַ֣ר נַהֲרָ֑ה דִּ֣י כָל־ דִּ֣י יִ֠שְׁאֲלֶנְכוֹן עֶזְרָ֨א כָהֲנָ֜ה סָפַ֤ר דָּתָא֙ דִּֽי־ אֱלָ֣הּ שְׁמַיָּ֔א אָסְפַּ֖רְנָא יִתְעֲבִֽד׃ (כז) עַד־ כְּסַף֮ כַּכְּרִ֣ין מְאָה֒ וְעַד־ חִנְטִין֙ כֹּרִ֣ין מְאָ֔ה וְעַד־ חֲמַר֙ בַּתִּ֣ין מְאָ֔ה וְעַד־ בַּתִּ֥ין מְשַׁ֖ח מְאָ֑ה וּמְלַ֖ח דִּי־ לָ֥א כְתָֽב׃ (כח) כָּל־ דִּ֗י מִן־ טַ֙עַם֙ אֱלָ֣הּ שְׁמַיָּ֔א יִתְעֲבֵד֙ אַדְרַזְדָּ֔א לְבֵ֖ית אֱלָ֣הּ שְׁמַיָּ֑א דִּֽי־ לְמָ֤ה לֶֽהֱוֵא֙ קְצַ֔ף עַל־ מַלְכ֥וּת מַלְכָּ֖א וּבְנֽוֹהִי׃ (כט) וּלְכֹ֣ם מְהוֹדְעִ֗ין דִּ֣י כָל־ כָּהֲנַיָּ֣א וְ֠לֵוָיֵא זַמָּ֨רַיָּ֤א תָרָֽעַיָּא֙ נְתִ֣ינַיָּ֔א וּפָ֣לְחֵ֔י בֵּ֖ית אֱלָהָ֣א דְנָ֑ה מִנְדָּ֤ה בְלוֹ֙ וַהֲלָ֔ךְ לָ֥א שַׁלִּ֖יט לְמִרְמֵ֥א עֲלֵיהֹֽם׃ (ל) וְאַ֣נְתְּ עֶזְרָ֗א כְּחָכְמַ֨ת אֱלָהָ֤ךְ דִּֽי־ בִידָךְ֙ מֶ֣נִּי שָׁפְטִ֞ין וְדַיָּנִ֗ין דִּי־ לֶהֱוֺ֤ן (דאנין) דָּאיְנִין֙ לְכָל־ עַמָּה֙ דִּ֚י בַּעֲבַ֣ר נַהֲרָ֔ה לְכָל־ יָדְעֵ֖י דָּתֵ֣י אֱלָהָ֑ךְ וְדִ֧י לָ֦א יָדַ֖ע תְּהוֹדְעֽוּן׃ (לא) וְכָל־ דִּי־ לָא֩ לֶהֱוֵ֨א עָבֵ֜ד דָּתָ֣א דִֽי־ אֱלָהָ֗ךְ וְדָתָא֙ דִּ֣י מַלְכָּ֔א אָסְפַּ֕רְנָא דִּינָ֕ה לֶהֱוֵ֥א מִתְעֲבֵ֖ד מִנֵּ֑הּ הֵ֤ן לְמוֹת֙ הֵ֣ן (לשרשו) לִשְׁרֹשִׁ֔י הֵן־ לַעֲנָ֥שׁ נִכְסִ֖ין וְלֶאֱסוּרִֽין׃ (לב) בָּר֥וּךְ יְהוָ֖ה אֱלֹהֵ֣י אֲבוֹתֵ֑ינוּ אֲשֶׁ֨ר נָתַ֤ן כָּזֹאת֙ בְּלֵ֣ב הַמֶּ֔לֶךְ לְפָאֵ֕ר אֶת־ בֵּ֥ית יְהוָ֖ה אֲשֶׁ֥ר בִּירוּשָׁלִָֽם׃ (לג) וְעָלַ֣י הִטָּה־ חֶ֗סֶד לִפְנֵ֤י הַמֶּ֙לֶךְ֙ וְיֽוֹעֲצָ֔יו וּלְכָל־ שָׂרֵ֥י הַמֶּ֖לֶךְ הַגִּבֹּרִ֑ים וַאֲנִ֣י הִתְחַזַּ֗קְתִּי כְּיַד־ יְהוָ֤ה אֱלֹהַי֙ עָלַ֔י וָאֶקְבְּצָ֧ה מִיִּשְׂרָאֵ֛ל רָאשִׁ֖ים לַעֲל֥וֹת עִמִּֽי׃ (לד) וְאֵ֛לֶּה רָאשֵׁ֥י אֲבֹתֵיהֶ֖ם וְהִתְיַחְשָׂ֑ם הָעֹלִ֣ים עִמִּ֗י בְּמַלְכ֛וּת אַרְתַּחְשַׁ֥סְתְּא הַמֶּ֖לֶךְ מִבָּבֶֽל׃ (לה) מִבְּנֵ֤י פִֽינְחָס֙ גֵּֽרְשֹׁ֔ם מִבְּנֵ֥י אִיתָמָ֖ר דָּנִיֵּ֑אל מִבְּנֵ֥י דָוִ֖יד חַטּֽוּשׁ׃ (לו) מִבְּנֵ֣י שְׁכַנְיָ֔ה מִבְּנֵ֥י פַרְעֹ֖שׁ זְכַרְיָ֑ה וְעִמּ֛וֹ הִתְיַחֵ֥שׂ לִזְכָרִ֖ים מֵאָ֥ה וַחֲמִשִּֽׁים׃ (לז) מִבְּנֵי֙ פַּחַ֣ת מוֹאָ֔ב אֶלְיְהֽוֹעֵינַ֖י בֶּן־ זְרַֽחְיָ֑ה וְעִמּ֖וֹ מָאתַ֥יִם הַזְּכָרִֽים׃ (לח) מִבְּנֵ֥י שְׁכַנְיָ֖ה בֶּן־ יַחֲזִיאֵ֑ל וְעִמּ֕וֹ שְׁלֹ֥שׁ מֵא֖וֹת הַזְּכָרִֽים׃ (לט) וּמִבְּנֵ֣י עָדִ֔ין עֶ֖בֶד בֶּן־ יוֹנָתָ֑ן וְעִמּ֖וֹ חֲמִשִּׁ֥ים הַזְּכָרִֽים׃ (מ) וּמִבְּנֵ֣י עֵילָ֔ם יְשַֽׁעְיָ֖ה בֶּן־ עֲתַלְיָ֑ה וְעִמּ֖וֹ שִׁבְעִ֥ים הַזְּכָרִֽים׃ (מא) וּמִבְּנֵ֣י שְׁפַטְיָ֔ה זְבַדְיָ֖ה בֶּן־ מִֽיכָאֵ֑ל וְעִמּ֖וֹ שְׁמֹנִ֥ים הַזְּכָרִֽים׃ (מב) מִבְּנֵ֣י יוֹאָ֔ב עֹבַדְיָ֖ה בֶּן־ יְחִיאֵ֑ל וְעִמּ֕וֹ מָאתַ֛יִם וּשְׁמֹנָ֥ה עָשָׂ֖ר הַזְּכָרִֽים׃ (מג) וּמִבְּנֵ֥י שְׁלוֹמִ֖ית בֶּן־ יוֹסִפְיָ֑ה וְעִמּ֕וֹ מֵאָ֥ה וְשִׁשִּׁ֖ים הַזְּכָרִֽים׃ (מד) וּמִבְּנֵ֣י בֵבַ֔י זְכַרְיָ֖ה בֶּן־ בֵּבָ֑י וְעִמּ֕וֹ עֶשְׂרִ֥ים וּשְׁמֹנָ֖ה הַזְּכָרִֽים׃ (מה) וּמִבְּנֵ֣י עַזְגָּ֔ד יוֹחָנָ֖ן בֶּן־ הַקָּטָ֑ן וְעִמּ֕וֹ מֵאָ֥ה וַעֲשָׂרָ֖ה הַזְּכָרִֽים׃ (מו) וּמִבְּנֵ֣י אֲדֹנִיקָם֮ אַחֲרֹנִים֒ וְאֵ֣לֶּה שְׁמוֹתָ֔ם אֱלִיפֶ֖לֶט יְעִיאֵ֣ל וּֽשְׁמַעְיָ֑ה וְעִמָּהֶ֖ם שִׁשִּׁ֥ים הַזְּכָרִֽים׃ (מז) וּמִבְּנֵ֥י בִגְוַ֖י עוּתַ֣י (וזבוד) וְזַכּ֑וּר וְעִמּ֖וֹ שִׁבְעִ֥ים הַזְּכָרִֽים׃ (מח) וָֽאֶקְבְּצֵ֗ם אֶל־ הַנָּהָר֙ הַבָּ֣א אֶֽל־ אַהֲוָ֔א וַנַּחֲנֶ֥ה שָׁ֖ם יָמִ֣ים שְׁלֹשָׁ֑ה וָאָבִ֤ינָה בָעָם֙ וּבַכֹּ֣הֲנִ֔ים וּמִבְּנֵ֥י לֵוִ֖י לֹא־ מָצָ֥אתִי שָֽׁם׃ (מט) וָאֶשְׁלְחָ֡ה לֶאֱלִיעֶ֡זֶר לַאֲרִיאֵ֡ל לִֽ֠שְׁמַעְיָה וּלְאֶלְנָתָ֨ן וּלְיָרִ֜יב וּלְאֶלְנָתָ֧ן וּלְנָתָ֛ן וְלִזְכַרְיָ֥ה וְלִמְשֻׁלָּ֖ם רָאשִׁ֑ים וּלְיוֹיָרִ֥יב וּלְאֶלְנָתָ֖ן מְבִינִֽים׃ (נ) (ואוצאה) וָאֲצַוֶּ֤ה אוֹתָם֙ עַל־ אִדּ֣וֹ הָרֹ֔אשׁ בְּכָסִפְיָ֖א הַמָּק֑וֹם וָאָשִׂימָה֩ בְּפִיהֶ֨ם דְּבָרִ֜ים לְ֠דַבֵּר אֶל־ אִדּ֨וֹ אָחִ֤יו (הנתונים) הַנְּתִינִים֙ בְּכָסִפְיָ֣א הַמָּק֔וֹם לְהָֽבִיא־ לָ֥נוּ מְשָׁרְתִ֖ים לְבֵ֥ית אֱלֹהֵֽינוּ׃ (נא) וַיָּבִ֨יאּוּ לָ֜נוּ כְּיַד־ אֱלֹהֵ֨ינוּ הַטּוֹבָ֤ה עָלֵ֙ינוּ֙ אִ֣ישׁ שֶׂ֔כֶל מִבְּנֵ֣י מַחְלִ֔י בֶּן־ לֵוִ֖י בֶּן־ יִשְׂרָאֵ֑ל וְשֵׁרֵֽבְיָ֛ה וּבָנָ֥יו וְאֶחָ֖יו שְׁמֹנָ֥ה עָשָֽׂר׃ (נב) וְאֶת־ חֲשַׁבְיָ֔ה וְאִתּ֥וֹ יְשַֽׁעְיָ֖ה מִבְּנֵ֣י מְרָרִ֑י אֶחָ֥יו וּבְנֵיהֶ֖ם עֶשְׂרִֽים׃ (נג) וּמִן־ הַנְּתִינִ֗ים שֶׁנָּתַ֨ן דָּוִ֤יד וְהַשָּׂרִים֙ לַעֲבֹדַ֣ת הַלְוִיִּ֔ם נְתִינִ֖ים מָאתַ֣יִם וְעֶשְׂרִ֑ים כֻּלָּ֖ם נִקְּב֥וּ בְשֵׁמֽוֹת׃ (נד) וָאֶקְרָ֨א שָׁ֥ם צוֹם֙ עַל־ הַנָּהָ֣ר אַהֲוָ֔א לְהִתְעַנּ֖וֹת לִפְנֵ֣י אֱלֹהֵ֑ינוּ לְבַקֵּ֤שׁ מִמֶּ֙נּוּ֙ דֶּ֣רֶךְ יְשָׁרָ֔ה לָ֥נוּ וּלְטַפֵּ֖נוּ וּלְכָל־ רְכוּשֵֽׁנוּ׃ (נה) כִּ֣י בֹ֗שְׁתִּי לִשְׁא֤וֹל מִן־ הַמֶּ֙לֶךְ֙ חַ֣יִל וּפָרָשִׁ֔ים לְעָזְרֵ֥נוּ מֵאוֹיֵ֖ב בַּדָּ֑רֶךְ כִּֽי־ אָמַ֨רְנוּ לַמֶּ֜לֶךְ לֵאמֹ֗ר יַד־ אֱלֹהֵ֤ינוּ עַל־ כָּל־ מְבַקְשָׁיו֙ לְטוֹבָ֔ה וְעֻזּ֣וֹ וְאַפּ֔וֹ עַ֖ל כָּל־ עֹזְבָֽיו׃ (נו) וַנָּצ֛וּמָה וַנְּבַקְשָׁ֥ה מֵאֱלֹהֵ֖ינוּ עַל־ זֹ֑את וַיֵּעָתֵ֖ר לָֽנוּ׃ (נז) וָאַבְדִּ֛ילָה מִשָּׂרֵ֥י הַכֹּהֲנִ֖ים שְׁנֵ֣ים עָשָׂ֑ר לְשֵׁרֵֽבְיָ֣ה חֲשַׁבְיָ֔ה וְעִמָּהֶ֥ם מֵאֲחֵיהֶ֖ם עֲשָׂרָֽה׃ (נח) (ואשקולה) וָאֶשְׁקֳלָ֣ה לָהֶ֔ם אֶת־ הַכֶּ֥סֶף וְאֶת־ הַזָּהָ֖ב וְאֶת־ הַכֵּלִ֑ים תְּרוּמַ֣ת בֵּית־ אֱלֹהֵ֗ינוּ הַהֵרִ֙ימוּ֙ הַמֶּ֙לֶךְ֙ וְיֹעֲצָ֣יו וְשָׂרָ֔יו וְכָל־ יִשְׂרָאֵ֖ל הַנִּמְצָאִֽים׃ (נט) וָאֶשְׁקֲלָ֨ה עַל־ יָדָ֜ם כֶּ֗סֶף כִּכָּרִים֙ שֵֽׁשׁ־ מֵא֣וֹת וַחֲמִשִּׁ֔ים וּכְלֵי־ כֶ֥סֶף מֵאָ֖ה לְכִכָּרִ֑ים זָהָ֖ב מֵאָ֥ה כִכָּֽר׃ (ס) וּכְפֹרֵ֤י זָהָב֙ עֶשְׂרִ֔ים לַאֲדַרְכֹנִ֖ים אָ֑לֶף וּכְלֵ֨י נְחֹ֜שֶׁת מֻצְהָ֤ב טוֹבָה֙ שְׁנַ֔יִם חֲמוּדֹ֖ת כַּזָּהָֽב׃ (סא) וָאֹמְרָ֣ה אֲלֵהֶ֗ם אַתֶּ֥ם קֹ֙דֶשׁ֙ לַיהוָ֔ה וְהַכֵּלִ֖ים קֹ֑דֶשׁ וְהַכֶּ֤סֶף וְהַזָּהָב֙ נְדָבָ֔ה לַיהוָ֖ה אֱלֹהֵ֥י אֲבֹתֵיכֶֽם׃ (סב) שִׁקְד֣וּ וְשִׁמְר֗וּ עַֽד־ תִּשְׁקְל֡וּ לִפְנֵי֩ שָׂרֵ֨י הַכֹּהֲנִ֧ים וְהַלְוִיִּ֛ם וְשָׂרֵֽי־ הָאָב֥וֹת לְיִשְׂרָאֵ֖ל בִּירוּשָׁלִָ֑ם הַלִּשְׁכ֖וֹת בֵּ֥ית יְהוָֽה׃ (סג) וְקִבְּלוּ֙ הַכֹּהֲנִ֣ים וְהַלְוִיִּ֔ם מִשְׁקַ֛ל הַכֶּ֥סֶף וְהַזָּהָ֖ב וְהַכֵּלִ֑ים לְהָבִ֥יא לִירוּשָׁלִַ֖ם לְבֵ֥ית אֱלֹהֵֽינוּ׃ (סד) וַֽנִּסְעָ֞ה מִנְּהַ֣ר אַֽהֲוָ֗א בִּשְׁנֵ֤ים עָשָׂר֙ לַחֹ֣דֶשׁ הָרִאשׁ֔וֹן לָלֶ֖כֶת יְרוּשָׁלִָ֑ם וְיַד־ אֱלֹהֵ֙ינוּ֙ הָיְתָ֣ה עָלֵ֔ינוּ וַיַּ֨צִּילֵ֔נוּ מִכַּ֥ף אוֹיֵ֛ב וְאוֹרֵ֖ב עַל־ הַדָּֽרֶךְ׃ (סה) וַנָּב֖וֹא יְרוּשָׁלִָ֑ם וַנֵּ֥שֶׁב שָׁ֖ם יָמִ֥ים שְׁלֹשָֽׁה׃ (סו) וּבַיּ֣וֹם הָרְבִיעִ֡י נִשְׁקַ֣ל הַכֶּסֶף֩ וְהַזָּהָ֨ב וְהַכֵּלִ֜ים בְּבֵ֣ית אֱלֹהֵ֗ינוּ עַ֠ל יַד־ מְרֵמ֤וֹת בֶּן־ אֽוּרִיָּה֙ הַכֹּהֵ֔ן וְעִמּ֖וֹ אֶלְעָזָ֣ר בֶּן־ פִּֽינְחָ֑ס וְעִמָּהֶ֞ם יוֹזָבָ֧ד בֶּן־ יֵשׁ֛וּעַ וְנֽוֹעַדְיָ֥ה בֶן־ בִּנּ֖וּי הַלְוִיִּֽם׃ (סז) בְּמִסְפָּ֥ר בְּמִשְׁקָ֖ל לַכֹּ֑ל
