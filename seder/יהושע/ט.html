יהושע סדר ט

פתיחה באתר · ספר יהושע כולו ·

(א) וְשֵׁ֨ם חֶבְר֤וֹן לְפָנִים֙ קִרְיַ֣ת אַרְבַּ֔ע הָאָדָ֧ם הַגָּד֛וֹל בָּעֲנָקִ֖ים ה֑וּא וְהָאָ֥רֶץ שָׁקְטָ֖ה מִמִּלְחָמָֽה׃ (ב) וַיְהִ֣י הַגּוֹרָ֗ל לְמַטֵּ֛ה בְּנֵ֥י יְהוּדָ֖ה לְמִשְׁפְּחֹתָ֑ם אֶל־ גְּב֨וּל אֱד֧וֹם מִדְבַּר־ צִ֛ן נֶ֖גְבָּה מִקְצֵ֥ה תֵימָֽן׃ (ג) וַיְהִ֤י לָהֶם֙ גְּב֣וּל נֶ֔גֶב מִקְצֵ֖ה יָ֣ם הַמֶּ֑לַח מִן־ הַלָּשֹׁ֖ן הַפֹּנֶ֥ה נֶֽגְבָּה׃ (ד) וְ֠יָצָא אֶל־ מִנֶּ֜גֶב לְמַעֲלֵ֤ה עַקְרַבִּים֙ וְעָ֣בַר צִ֔נָה וְעָלָ֥ה מִנֶּ֖גֶב לְקָדֵ֣שׁ בַּרְנֵ֑עַ וְעָבַ֤ר חֶצְרוֹן֙ וְעָלָ֣ה אַדָּ֔רָה וְנָסַ֖ב הַקַּרְקָֽעָה׃ (ה) וְעָבַ֣ר עַצְמ֗וֹנָה וְיָצָא֙ נַ֣חַל מִצְרַ֔יִם (והיה) וְהָי֛וּ תֹּצְא֥וֹת הַגְּב֖וּל יָ֑מָּה זֶה־ יִהְיֶ֥ה לָכֶ֖ם גְּב֥וּל נֶֽגֶב׃ (ו) וּגְב֥וּל קֵ֙דְמָה֙ יָ֣ם הַמֶּ֔לַח עַד־ קְצֵ֖ה הַיַּרְדֵּ֑ן וּגְב֞וּל לִפְאַ֤ת צָפ֙וֹנָה֙ מִלְּשׁ֣וֹן הַיָּ֔ם מִקְצֵ֖ה הַיַּרְדֵּֽן׃ (ז) וְעָלָ֤ה הַגְּבוּל֙ בֵּ֣ית חָגְלָ֔ה וְעָבַ֕ר מִצְּפ֖וֹן לְבֵ֣ית הָעֲרָבָ֑ה וְעָלָ֣ה הַגְּב֔וּל אֶ֥בֶן בֹּ֖הַן בֶּן־ רְאוּבֵֽן׃ (ח) וְעָלָ֨ה הַגְּב֥וּל ׀ דְּבִרָה֮ מֵעֵ֣מֶק עָכוֹר֒ וְצָפ֜וֹנָה פֹּנֶ֣ה אֶל־ הַגִּלְגָּ֗ל אֲשֶׁר־ נֹ֙כַח֙ לְמַעֲלֵ֣ה אֲדֻמִּ֔ים אֲשֶׁ֥ר מִנֶּ֖גֶב לַנָּ֑חַל וְעָבַ֤ר הַגְּבוּל֙ אֶל־ מֵי־ עֵ֣ין שֶׁ֔מֶשׁ וְהָי֥וּ תֹצְאֹתָ֖יו אֶל־ עֵ֥ין רֹגֵֽל׃ (ט) וְעָלָ֨ה הַגְּב֜וּל גֵּ֣י בֶן־ הִנֹּ֗ם אֶל־ כֶּ֤תֶף הַיְבוּסִי֙ מִנֶּ֔גֶב הִ֖יא יְרֽוּשָׁלִָ֑ם וְעָלָ֨ה הַגְּב֜וּל אֶל־ רֹ֣אשׁ הָהָ֗ר אֲ֠שֶׁר עַל־ פְּנֵ֤י גֵֽי־ הִנֹּם֙ יָ֔מָּה אֲשֶׁ֛ר בִּקְצֵ֥ה עֵֽמֶק־ רְפָאִ֖ים צָפֹֽנָה׃ (י) וְתָאַ֨ר הַגְּב֜וּל מֵרֹ֣אשׁ הָהָ֗ר אֶל־ מַעְיַן֙ מֵ֣י נֶפְתּ֔וֹחַ וְיָצָ֖א אֶל־ עָרֵ֣י הַר־ עֶפְר֑וֹן וְתָאַ֤ר הַגְּבוּל֙ בַּעֲלָ֔ה הִ֖יא קִרְיַ֥ת יְעָרִֽים׃ (יא) וְנָסַב֩ הַגְּב֨וּל מִבַּעֲלָ֥ה יָ֙מָּה֙ אֶל־ הַ֣ר שֵׂעִ֔יר וְעָבַ֕ר אֶל־ כֶּ֧תֶף הַר־ יְעָרִ֛ים מִצָּפ֖וֹנָה הִ֣יא כְסָל֑וֹן וְיָרַ֥ד בֵּֽית־ שֶׁ֖מֶשׁ וְעָבַ֥ר תִּמְנָֽה׃ (יב) וְיָצָ֨א הַגְּב֜וּל אֶל־ כֶּ֣תֶף עֶקְרוֹן֮ צָפוֹנָה֒ וְתָאַ֤ר הַגְּבוּל֙ שִׁכְּר֔וֹנָה וְעָבַ֥ר הַר־ הַֽבַּעֲלָ֖ה וְיָצָ֣א יַבְנְאֵ֑ל וְהָי֛וּ תֹּצְא֥וֹת הַגְּב֖וּל יָֽמָּה׃ (יג) וּגְב֣וּל יָ֔ם הַיָּ֥מָּה הַגָּד֖וֹל וּגְב֑וּל זֶ֠ה גְּב֧וּל בְּנֵֽי־ יְהוּדָ֛ה סָבִ֖יב לְמִשְׁפְּחֹתָֽם׃ (יד) וּלְכָלֵ֣ב בֶּן־ יְפֻנֶּ֗ה נָ֤תַן חֵ֙לֶק֙ בְּת֣וֹךְ בְּנֵֽי־ יְהוּדָ֔ה אֶל־ פִּ֥י יְהוָ֖ה לִֽיהוֹשֻׁ֑עַ אֶת־ קִרְיַ֥ת אַרְבַּ֛ע אֲבִ֥י הָעֲנָ֖ק הִ֥יא חֶבְרֽוֹן׃ (טו) וַיֹּ֤רֶשׁ מִשָּׁם֙ כָּלֵ֔ב אֶת־ שְׁלוֹשָׁ֖ה בְּנֵ֣י הָעֲנָ֑ק אֶת־ שֵׁשַׁ֤י וְאֶת־ אֲחִימַן֙ וְאֶת־ תַּלְמַ֔י יְלִידֵ֖י הָעֲנָֽק׃ (טז) וַיַּ֣עַל מִשָּׁ֔ם אֶל־ יֹשְׁבֵ֖י דְּבִ֑ר וְשֵׁם־ דְּבִ֥ר לְפָנִ֖ים קִרְיַת־ סֵֽפֶר׃ (יז) וַיֹּ֣אמֶר כָּלֵ֔ב אֲשֶׁר־ יַכֶּ֥ה אֶת־ קִרְיַת־ סֵ֖פֶר וּלְכָדָ֑הּ וְנָתַ֥תִּי ל֛וֹ אֶת־ עַכְסָ֥ה בִתִּ֖י לְאִשָּֽׁה׃ (יח) וַֽיִּלְכְּדָ֛הּ עָתְנִיאֵ֥ל בֶּן־ קְנַ֖ז אֲחִ֣י כָלֵ֑ב וַיִּתֶּן־ ל֛וֹ אֶת־ עַכְסָ֥ה בִתּ֖וֹ לְאִשָּֽׁה׃ (יט) וַיְהִ֣י בְּבוֹאָ֗הּ וַתְּסִיתֵ֙הוּ֙ לִשְׁא֤וֹל מֵֽאֵת־ אָבִ֙יהָ֙ שָׂדֶ֔ה וַתִּצְנַ֖ח מֵעַ֣ל הַחֲמ֑וֹר וַיֹּֽאמֶר־ לָ֥הּ כָּלֵ֖ב מַה־ לָּֽךְ׃ (כ) וַתֹּ֜אמֶר תְּנָה־ לִּ֣י בְרָכָ֗ה כִּ֣י אֶ֤רֶץ הַנֶּ֙גֶב֙ נְתַתָּ֔נִי וְנָתַתָּ֥ה לִ֖י גֻּלֹּ֣ת מָ֑יִם וַיִּתֶּן־ לָ֗הּ אֵ֚ת גֻּלֹּ֣ת עִלִּיּ֔וֹת וְאֵ֖ת גֻּלֹּ֥ת תַּחְתִּיּֽוֹת׃ (כא) זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵי־ יְהוּדָ֖ה לְמִשְׁפְּחֹתָֽם׃ (כב) וַיִּֽהְי֣וּ הֶעָרִ֗ים מִקְצֵה֙ לְמַטֵּ֣ה בְנֵֽי־ יְהוּדָ֔ה אֶל־ גְּב֥וּל אֱד֖וֹם בַּנֶּ֑גְבָּה קַבְצְאֵ֥ל וְעֵ֖דֶר וְיָגֽוּר׃ (כג) וְקִינָ֥ה וְדִֽימוֹנָ֖ה וְעַדְעָדָֽה׃ (כד) וְקֶ֥דֶשׁ וְחָצ֖וֹר וְיִתְנָֽן׃ (כה) זִ֥יף וָטֶ֖לֶם וּבְעָלֽוֹת׃ (כו) וְחָצ֤וֹר ׀ חֲדַתָּה֙ וּקְרִיּ֔וֹת חֶצְר֖וֹן הִ֥יא חָצֽוֹר׃ (כז) אֲמָ֥ם וּשְׁמַ֖ע וּמוֹלָדָֽה׃ (כח) וַחֲצַ֥ר גַּדָּ֛ה וְחֶשְׁמ֖וֹן וּבֵ֥ית פָּֽלֶט׃ (כט) וַחֲצַ֥ר שׁוּעָ֛ל וּבְאֵ֥ר שֶׁ֖בַע וּבִזְיוֹתְיָֽה׃ (ל) בַּעֲלָ֥ה וְעִיִּ֖ים וָעָֽצֶם׃ (לא) וְאֶלְתּוֹלַ֥ד וּכְסִ֖יל וְחָרְמָֽה׃ (לב) וְצִֽקְלַ֥ג וּמַדְמַנָּ֖ה וְסַנְסַנָּֽה׃ (לג) וּלְבָא֥וֹת וְשִׁלְחִ֖ים וְעַ֣יִן וְרִמּ֑וֹן כָּל־ עָרִ֛ים עֶשְׂרִ֥ים וָתֵ֖שַׁע וְחַצְרֵיהֶֽן׃ (לד) בַּשְּׁפֵלָ֑ה אֶשְׁתָּא֥וֹל וְצָרְעָ֖ה וְאַשְׁנָֽה׃ (לה) וְזָנ֙וֹחַ֙ וְעֵ֣ין גַּנִּ֔ים תַּפּ֖וּחַ וְהָעֵינָֽם׃ (לו) יַרְמוּת֙ וַעֲדֻלָּ֔ם שׂוֹכֹ֖ה וַעֲזֵקָֽה׃ (לז) וְשַׁעֲרַ֙יִם֙ וַעֲדִיתַ֔יִם וְהַגְּדֵרָ֖ה וּגְדֵרֹתָ֑יִם עָרִ֥ים אַרְבַּֽע־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (לח) צְנָ֥ן וַחֲדָשָׁ֖ה וּמִגְדַּל־ גָּֽד׃ (לט) וְדִלְעָ֥ן וְהַמִּצְפֶּ֖ה וְיָקְתְאֵֽל׃ (מ) לָכִ֥ישׁ וּבָצְקַ֖ת וְעֶגְלֽוֹן׃ (מא) וְכַבּ֥וֹן וְלַחְמָ֖ס וְכִתְלִֽישׁ׃ (מב) וּגְדֵר֕וֹת בֵּית־ דָּג֥וֹן וְנַעֲמָ֖ה וּמַקֵּדָ֑ה עָרִ֥ים שֵׁשׁ־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (מג) לִבְנָ֥ה וָעֶ֖תֶר וְעָשָֽׁן׃ (מד) וְיִפְתָּ֥ח וְאַשְׁנָ֖ה וּנְצִֽיב׃ (מה) וּקְעִילָ֥ה וְאַכְזִ֖יב וּמָֽרֵאשָׁ֑ה עָרִ֥ים תֵּ֖שַׁע וְחַצְרֵיהֶֽן׃ (מו) עֶקְר֥וֹן וּבְנֹתֶ֖יהָ וַחֲצֵרֶֽיהָ׃ (מז) מֵעֶקְר֖וֹן וָיָ֑מָּה כֹּ֛ל אֲשֶׁר־ עַל־ יַ֥ד אַשְׁדּ֖וֹד וְחַצְרֵיהֶֽן׃ (מח) אַשְׁדּ֞וֹד בְּנוֹתֶ֣יהָ וַחֲצֵרֶ֗יהָ עַזָּ֥ה בְּנוֹתֶ֥יהָ וַחֲצֵרֶ֖יהָ עַד־ נַ֣חַל מִצְרָ֑יִם וְהַיָּ֥ם (הגבול) הַגָּד֖וֹל וּגְבֽוּל׃ (מט) וּבָהָ֑ר שָׁמִ֥יר וְיַתִּ֖יר וְשׂוֹכֹֽה׃ (נ) וְדַנָּ֥ה וְקִרְיַת־ סַנָּ֖ה הִ֥יא דְבִֽר׃ (נא) וַעֲנָ֥ב וְאֶשְׁתְּמֹ֖ה וְעָנִֽים׃ (נב) וְגֹ֥שֶׁן וְחֹלֹ֖ן וְגִלֹ֑ה עָרִ֥ים אַֽחַת־ עֶשְׂרֵ֖ה וְחַצְרֵיהֶֽן׃ (נג) אֲרַ֥ב וְרוּמָ֖ה וְאֶשְׁעָֽן׃ (נד) (וינים) וְיָנ֥וּם וּבֵית־ תַּפּ֖וּחַ וַאֲפֵֽקָה׃ (נה) וְחֻמְטָ֗ה וְקִרְיַ֥ת אַרְבַּ֛ע הִ֥יא חֶבְר֖וֹן וְצִיעֹ֑ר עָרִ֥ים תֵּ֖שַׁע וְחַצְרֵיהֶֽן׃ (נו) מָע֥וֹן ׀ כַּרְמֶ֖ל וָזִ֥יף וְיוּטָּֽה׃ (נז) וְיִזְרְעֶ֥אל וְיָקְדְעָ֖ם וְזָנֽוֹחַ׃ (נח) הַקַּ֖יִן גִּבְעָ֣ה וְתִמְנָ֑ה עָרִ֥ים עֶ֖שֶׂר וְחַצְרֵיהֶֽן׃ (נט) חַלְח֥וּל בֵּֽית־ צ֖וּר וּגְדֽוֹר׃ (ס) וּמַעֲרָ֥ת וּבֵית־ עֲנ֖וֹת וְאֶלְתְּקֹ֑ן עָרִ֥ים שֵׁ֖שׁ וְחַצְרֵיהֶֽן׃ (סא) קִרְיַת־ בַּ֗עַל הִ֛יא קִרְיַ֥ת יְעָרִ֖ים וְהָֽרַבָּ֑ה עָרִ֥ים שְׁתַּ֖יִם וְחַצְרֵיהֶֽן׃ (סב) בַּמִּדְבָּ֑ר בֵּ֚ית הָעֲרָבָ֔ה מִדִּ֖ין וּסְכָכָֽה׃ (סג) וְהַנִּבְשָׁ֥ן וְעִיר־ הַמֶּ֖לַח וְעֵ֣ין גֶּ֑דִי עָרִ֥ים שֵׁ֖שׁ וְחַצְרֵיהֶֽן׃ (סד) וְאֶת־ הַיְבוּסִי֙ יוֹשְׁבֵ֣י יְרֽוּשָׁלִַ֔ם לֹֽא־ (יוכלו) יָכְל֥וּ בְנֵֽי־ יְהוּדָ֖ה לְהֽוֹרִישָׁ֑ם וַיֵּ֨שֶׁב הַיְבוּסִ֜י אֶת־ בְּנֵ֤י יְהוּדָה֙ בִּיר֣וּשָׁלִַ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (סה) וַיֵּצֵ֨א הַגּוֹרָ֜ל לִבְנֵ֤י יוֹסֵף֙ מִיַּרְדֵּ֣ן יְרִיח֔וֹ לְמֵ֥י יְרִיח֖וֹ מִזְרָ֑חָה הַמִּדְבָּ֗ר עֹלֶ֧ה מִירִיח֛וֹ בָּהָ֖ר בֵּֽית־ אֵֽל׃ (סו) וְיָצָ֥א מִבֵּֽית־ אֵ֖ל ל֑וּזָה וְעָבַ֛ר אֶל־ גְּב֥וּל הָאַרְכִּ֖י עֲטָרֽוֹת׃ (סז) וְיָֽרַד־ יָ֜מָּה אֶל־ גְּב֣וּל הַיַּפְלֵטִ֗י עַ֣ד גְּב֧וּל בֵּית־ חוֹרֹ֛ן תַּחְתּ֖וֹן וְעַד־ גָּ֑זֶר וְהָי֥וּ (תצאתו) תֹצְאֹתָ֖יו יָֽמָּה׃ (סח) וַיִּנְחֲל֥וּ בְנֵי־ יוֹסֵ֖ף מְנַשֶּׁ֥ה וְאֶפְרָֽיִם׃ (סט) וַיְהִ֛י גְּב֥וּל בְּנֵֽי־ אֶפְרַ֖יִם לְמִשְׁפְּחֹתָ֑ם וַיְהִ֞י גְּב֤וּל נַחֲלָתָם֙ מִזְרָ֔חָה עַטְר֣וֹת אַדָּ֔ר עַד־ בֵּ֥ית חוֹרֹ֖ן עֶלְיֽוֹן׃ (ע) וְיָצָ֨א הַגְּב֜וּל הַיָּ֗מָּה הַֽמִּכְמְתָת֙ מִצָּפ֔וֹן וְנָסַ֧ב הַגְּב֛וּל מִזְרָ֖חָה תַּאֲנַ֣ת שִׁלֹ֑ה וְעָבַ֣ר אוֹת֔וֹ מִמִּזְרַ֖ח יָנֽוֹחָה׃ (עא) וְיָרַ֥ד מִיָּנ֖וֹחָה עֲטָר֣וֹת וְנַעֲרָ֑תָה וּפָגַע֙ בִּֽירִיח֔וֹ וְיָצָ֖א הַיַּרְדֵּֽן׃ (עב) מִתַּפּ֜וּחַ יֵלֵ֨ךְ הַגְּב֥וּל יָ֙מָּה֙ נַ֣חַל קָנָ֔ה וְהָי֥וּ תֹצְאֹתָ֖יו הַיָּ֑מָּה זֹ֗את נַחֲלַ֛ת מַטֵּ֥ה בְנֵי־ אֶפְרַ֖יִם לְמִשְׁפְּחֹתָֽם׃ (עג) וְהֶעָרִ֗ים הַמִּבְדָּלוֹת֙ לִבְנֵ֣י אֶפְרַ֔יִם בְּת֖וֹךְ נַחֲלַ֣ת בְּנֵֽי־ מְנַשֶּׁ֑ה כָּֽל־ הֶעָרִ֖ים וְחַצְרֵיהֶֽן׃ (עד) וְלֹ֣א הוֹרִ֔ישׁוּ אֶת־ הַֽכְּנַעֲנִ֖י הַיּוֹשֵׁ֣ב בְּגָ֑זֶר וַיֵּ֨שֶׁב הַֽכְּנַעֲנִ֜י בְּקֶ֤רֶב אֶפְרַ֙יִם֙ עַד־ הַיּ֣וֹם הַזֶּ֔ה וַיְהִ֖י לְמַס־ עֹבֵֽד׃ (עה) וַיְהִ֤י הַגּוֹרָל֙ לְמַטֵּ֣ה מְנַשֶּׁ֔ה כִּי־ ה֖וּא בְּכ֣וֹר יוֹסֵ֑ף לְמָכִיר֩ בְּכ֨וֹר מְנַשֶּׁ֜ה אֲבִ֣י הַגִּלְעָ֗ד כִּ֣י ה֤וּא הָיָה֙ אִ֣ישׁ מִלְחָמָ֔ה וַֽיְהִי־ ל֖וֹ הַגִּלְעָ֥ד וְהַבָּשָֽׁן׃ (עו) וַ֠יְהִי לִבְנֵ֨י מְנַשֶּׁ֥ה הַנּוֹתָרִים֮ לְמִשְׁפְּחֹתָם֒ לִבְנֵ֨י אֲבִיעֶ֜זֶר וְלִבְנֵי־ חֵ֗לֶק וְלִבְנֵ֤י אַשְׂרִיאֵל֙ וְלִבְנֵי־ שֶׁ֔כֶם וְלִבְנֵי־ חֵ֖פֶר וְלִבְנֵ֣י שְׁמִידָ֑ע אֵ֠לֶּה בְּנֵ֨י מְנַשֶּׁ֧ה בֶּן־ יוֹסֵ֛ף הַזְּכָרִ֖ים לְמִשְׁפְּחֹתָֽם׃ (עז) וְלִצְלָפְחָד֩ בֶּן־ חֵ֨פֶר בֶּן־ גִּלְעָ֜ד בֶּן־ מָכִ֣יר בֶּן־ מְנַשֶּׁ֗ה לֹא־ הָ֥יוּ ל֛וֹ בָּנִ֖ים כִּ֣י אִם־ בָּנ֑וֹת וְאֵ֙לֶּה֙ שְׁמ֣וֹת בְּנֹתָ֔יו מַחְלָ֣ה וְנֹעָ֔ה חָגְלָ֥ה מִלְכָּ֖ה וְתִרְצָֽה׃
