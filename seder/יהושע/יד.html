יהושע סדר יד

לפי החלוקה לפרקים: יהושע כב, לד – כד, לג · 50 פסוקים

פתיחה באתר · ספר יהושע כולו ·

(א) וַֽיִּקְרְא֛וּ בְּנֵי־ רְאוּבֵ֥ן וּבְנֵי־ גָ֖ד לַמִּזְבֵּ֑חַ כִּ֣י עֵ֥ד הוּא֙ בֵּֽינֹתֵ֔ינוּ כִּ֥י יְהוָ֖ה הָאֱלֹהִֽים׃ (ב) וַֽיְהִי֙ מִיָּמִ֣ים רַבִּ֔ים אַ֠חֲרֵי אֲשֶׁר־ הֵנִ֨יחַ יְהוָ֧ה לְיִשְׂרָאֵ֛ל מִכָּל־ אֹיְבֵיהֶ֖ם מִסָּבִ֑יב וִיהוֹשֻׁ֣עַ זָקֵ֔ן בָּ֖א בַּיָּמִֽים׃ (ג) וַיִּקְרָ֤א יְהוֹשֻׁ֙עַ֙ לְכָל־ יִשְׂרָאֵ֔ל לִזְקֵנָיו֙ וּלְרָאשָׁ֔יו וּלְשֹׁפְטָ֖יו וּלְשֹֽׁטְרָ֑יו וַיֹּ֣אמֶר אֲלֵהֶ֔ם אֲנִ֣י זָקַ֔נְתִּי בָּ֖אתִי בַּיָּמִֽים׃ (ד) וְאַתֶּ֣ם רְאִיתֶ֗ם אֵת֩ כָּל־ אֲשֶׁ֨ר עָשָׂ֜ה יְהוָ֧ה אֱלֹהֵיכֶ֛ם לְכָל־ הַגּוֹיִ֥ם הָאֵ֖לֶּה מִפְּנֵיכֶ֑ם כִּ֚י יְהוָ֣ה אֱלֹהֵיכֶ֔ם ה֖וּא הַנִּלְחָ֥ם לָכֶֽם׃ (ה) רְאוּ֩ הִפַּ֨לְתִּי לָכֶ֜ם אֶֽת־ הַ֠גּוֹיִם הַנִּשְׁאָרִ֥ים הָאֵ֛לֶּה בְּנַחֲלָ֖ה לְשִׁבְטֵיכֶ֑ם מִן־ הַיַּרְדֵּ֗ן וְכָל־ הַגּוֹיִם֙ אֲשֶׁ֣ר הִכְרַ֔תִּי וְהַיָּ֥ם הַגָּד֖וֹל מְב֥וֹא הַשָּֽׁמֶשׁ׃ (ו) וַיהוָ֣ה אֱלֹֽהֵיכֶ֗ם ה֚וּא יֶהְדֳּפֵ֣ם מִפְּנֵיכֶ֔ם וְהוֹרִ֥ישׁ אֹתָ֖ם מִלִּפְנֵיכֶ֑ם וִֽירִשְׁתֶּם֙ אֶת־ אַרְצָ֔ם כַּאֲשֶׁ֥ר דִּבֶּ֛ר יְהוָ֥ה אֱלֹהֵיכֶ֖ם לָכֶֽם׃ (ז) וַחֲזַקְתֶּ֣ם מְאֹ֔ד לִשְׁמֹ֣ר וְלַעֲשׂ֔וֹת אֵ֚ת כָּל־ הַכָּת֔וּב בְּסֵ֖פֶר תּוֹרַ֣ת מֹשֶׁ֑ה לְבִלְתִּ֥י סוּר־ מִמֶּ֖נּוּ יָמִ֥ין וּשְׂמֹֽאול׃ (ח) לְבִלְתִּי־ בוֹא֙ בַּגּוֹיִ֣ם הָאֵ֔לֶּה הַנִּשְׁאָרִ֥ים הָאֵ֖לֶּה אִתְּכֶ֑ם וּבְשֵׁ֨ם אֱלֹהֵיהֶ֤ם לֹא־ תַזְכִּ֙ירוּ֙ וְלֹ֣א תַשְׁבִּ֔יעוּ וְלֹ֣א תַעַבְד֔וּם וְלֹ֥א תִֽשְׁתַּחֲו֖וּ לָהֶֽם׃ (ט) כִּ֛י אִם־ בַּיהוָ֥ה אֱלֹהֵיכֶ֖ם תִּדְבָּ֑קוּ כַּאֲשֶׁ֣ר עֲשִׂיתֶ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (י) וַיּ֤וֹרֶשׁ יְהוָה֙ מִפְּנֵיכֶ֔ם גּוֹיִ֖ם גְּדֹלִ֣ים וַעֲצוּמִ֑ים וְאַתֶּ֗ם לֹא־ עָ֤מַד אִישׁ֙ בִּפְנֵיכֶ֔ם עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (יא) אִישׁ־ אֶחָ֥ד מִכֶּ֖ם יִרְדָּף־ אָ֑לֶף כִּ֣י ׀ יְהוָ֣ה אֱלֹהֵיכֶ֗ם ה֚וּא הַנִּלְחָ֣ם לָכֶ֔ם כַּאֲשֶׁ֖ר דִּבֶּ֥ר לָכֶֽם׃ (יב) וְנִשְׁמַרְתֶּ֥ם מְאֹ֖ד לְנַפְשֹֽׁתֵיכֶ֑ם לְאַהֲבָ֖ה אֶת־ יְהוָ֥ה אֱלֹהֵיכֶֽם׃ (יג) כִּ֣י ׀ אִם־ שׁ֣וֹב תָּשׁ֗וּבוּ וּדְבַקְתֶּם֙ בְּיֶ֙תֶר֙ הַגּוֹיִ֣ם הָאֵ֔לֶּה הַנִּשְׁאָרִ֥ים הָאֵ֖לֶּה אִתְּכֶ֑ם וְהִֽתְחַתַּנְתֶּ֥ם בָּהֶ֛ם וּבָאתֶ֥ם בָּהֶ֖ם וְהֵ֥ם בָּכֶֽם׃ (יד) יָד֙וֹעַ֙ תֵּֽדְע֔וּ כִּי֩ לֹ֨א יוֹסִ֜יף יְהוָ֣ה אֱלֹהֵיכֶ֗ם לְהוֹרִ֛ישׁ אֶת־ הַגּוֹיִ֥ם הָאֵ֖לֶּה מִלִּפְנֵיכֶ֑ם וְהָי֨וּ לָכֶ֜ם לְפַ֣ח וּלְמוֹקֵ֗שׁ וּלְשֹׁטֵ֤ט בְּצִדֵּיכֶם֙ וְלִצְנִנִ֣ים בְּעֵינֵיכֶ֔ם עַד־ אֲבָדְכֶ֗ם מֵ֠עַל הָאֲדָמָ֤ה הַטּוֹבָה֙ הַזֹּ֔את אֲשֶׁר֙ נָתַ֣ן לָכֶ֔ם יְהוָ֖ה אֱלֹהֵיכֶֽם׃ (טו) וְהִנֵּ֨ה אָנֹכִ֤י הוֹלֵךְ֙ הַיּ֔וֹם בְּדֶ֖רֶךְ כָּל־ הָאָ֑רֶץ וִידַעְתֶּ֞ם בְּכָל־ לְבַבְכֶ֣ם וּבְכָל־ נַפְשְׁכֶ֗ם כִּ֣י לֹֽא־ נָפַל֩ דָּבָ֨ר אֶחָ֜ד מִכֹּ֣ל ׀ הַדְּבָרִ֣ים הַטּוֹבִ֗ים אֲשֶׁ֨ר דִּבֶּ֜ר יְהוָ֤ה אֱלֹהֵיכֶם֙ עֲלֵיכֶ֔ם הַכֹּל֙ בָּ֣אוּ לָכֶ֔ם לֹֽא־ נָפַ֥ל מִמֶּ֖נּוּ דָּבָ֥ר אֶחָֽד׃ (טז) וְהָיָ֗ה כַּאֲשֶׁר־ בָּ֤א עֲלֵיכֶם֙ כָּל־ הַדָּבָ֣ר הַטּ֔וֹב אֲשֶׁ֥ר דִּבֶּ֛ר יְהוָ֥ה אֱלֹהֵיכֶ֖ם אֲלֵיכֶ֑ם כֵּן֩ יָבִ֨יא יְהוָ֜ה עֲלֵיכֶ֗ם אֵ֚ת כָּל־ הַדָּבָ֣ר הָרָ֔ע עַד־ הַשְׁמִיד֣וֹ אוֹתְכֶ֗ם מֵ֠עַל הָאֲדָמָ֤ה הַטּוֹבָה֙ הַזֹּ֔את אֲשֶׁר֙ נָתַ֣ן לָכֶ֔ם יְהוָ֖ה אֱלֹהֵיכֶֽם׃ (יז) בְּ֠עָבְרְכֶם אֶת־ בְּרִ֨ית יְהוָ֥ה אֱלֹהֵיכֶם֮ אֲשֶׁ֣ר צִוָּ֣ה אֶתְכֶם֒ וַהֲלַכְתֶּ֗ם וַעֲבַדְתֶּם֙ אֱלֹהִ֣ים אֲחֵרִ֔ים וְהִשְׁתַּחֲוִיתֶ֖ם לָהֶ֑ם וְחָרָ֤ה אַף־ יְהוָה֙ בָּכֶ֔ם וַאֲבַדְתֶּ֣ם מְהֵרָ֔ה מֵעַל֙ הָאָ֣רֶץ הַטּוֹבָ֔ה אֲשֶׁ֖ר נָתַ֥ן לָכֶֽם׃ (יח) וַיֶּאֶסֹ֧ף יְהוֹשֻׁ֛עַ אֶת־ כָּל־ שִׁבְטֵ֥י יִשְׂרָאֵ֖ל שְׁכֶ֑מָה וַיִּקְרָא֩ לְזִקְנֵ֨י יִשְׂרָאֵ֜ל וּלְרָאשָׁ֗יו וּלְשֹֽׁפְטָיו֙ וּלְשֹׁ֣טְרָ֔יו וַיִּֽתְיַצְּב֖וּ לִפְנֵ֥י הָאֱלֹהִֽים׃ (יט) וַיֹּ֨אמֶר יְהוֹשֻׁ֜עַ אֶל־ כָּל־ הָעָ֗ם כֹּֽה־ אָמַ֣ר יְהוָה֮ אֱלֹהֵ֣י יִשְׂרָאֵל֒ בְּעֵ֣בֶר הַנָּהָ֗ר יָשְׁב֤וּ אֲבֽוֹתֵיכֶם֙ מֵֽעוֹלָ֔ם תֶּ֛רַח אֲבִ֥י אַבְרָהָ֖ם וַאֲבִ֣י נָח֑וֹר וַיַּעַבְד֖וּ אֱלֹהִ֥ים אֲחֵרִֽים׃ (כ) וָ֠אֶקַּח אֶת־ אֲבִיכֶ֤ם אֶת־ אַבְרָהָם֙ מֵעֵ֣בֶר הַנָּהָ֔ר וָאוֹלֵ֥ךְ אוֹת֖וֹ בְּכָל־ אֶ֣רֶץ כְּנָ֑עַן (וארב) וָאַרְבֶּה֙ אֶת־ זַרְע֔וֹ וָֽאֶתֶּן־ ל֖וֹ אֶת־ יִצְחָֽק׃ (כא) וָאֶתֵּ֣ן לְיִצְחָ֔ק אֶֽת־ יַעֲקֹ֖ב וְאֶת־ עֵשָׂ֑ו וָאֶתֵּ֨ן לְעֵשָׂ֜ו אֶת־ הַ֤ר שֵׂעִיר֙ לָרֶ֣שֶׁת אוֹת֔וֹ וְיַעֲקֹ֥ב וּבָנָ֖יו יָרְד֥וּ מִצְרָֽיִם׃ (כב) וָאֶשְׁלַ֞ח אֶת־ מֹשֶׁ֤ה וְאֶֽת־ אַהֲרֹן֙ וָאֶגֹּ֣ף אֶת־ מִצְרַ֔יִם כַּאֲשֶׁ֥ר עָשִׂ֖יתִי בְּקִרְבּ֑וֹ וְאַחַ֖ר הוֹצֵ֥אתִי אֶתְכֶֽם׃ (כג) וָֽאוֹצִ֤יא אֶת־ אֲבֽוֹתֵיכֶם֙ מִמִּצְרַ֔יִם וַתָּבֹ֖אוּ הַיָּ֑מָּה וַיִּרְדְּפ֨וּ מִצְרַ֜יִם אַחֲרֵ֧י אֲבוֹתֵיכֶ֛ם בְּרֶ֥כֶב וּבְפָרָשִׁ֖ים יַם־ סֽוּף׃ (כד) וַיִּצְעֲק֣וּ אֶל־ יְהוָ֗ה וַיָּ֨שֶׂם מַֽאֲפֵ֜ל בֵּינֵיכֶ֣ם ׀ וּבֵ֣ין הַמִּצְרִ֗ים וַיָּבֵ֨א עָלָ֤יו אֶת־ הַיָּם֙ וַיְכַסֵּ֔הוּ וַתִּרְאֶ֙ינָה֙ עֵינֵיכֶ֔ם אֵ֥ת אֲשֶׁר־ עָשִׂ֖יתִי בְּמִצְרָ֑יִם וַתֵּשְׁב֥וּ בַמִּדְבָּ֖ר יָמִ֥ים רַבִּֽים׃ (כה) (ואבאה) וָאָבִ֣יא אֶתְכֶ֗ם אֶל־ אֶ֤רֶץ הָאֱמֹרִי֙ הַיּוֹשֵׁב֙ בְּעֵ֣בֶר הַיַּרְדֵּ֔ן וַיִּֽלָּחֲמ֖וּ אִתְּכֶ֑ם וָאֶתֵּ֨ן אוֹתָ֤ם בְּיֶדְכֶם֙ וַתִּֽירְשׁ֣וּ אֶת־ אַרְצָ֔ם וָאַשְׁמִידֵ֖ם מִפְּנֵיכֶֽם׃ (כו) וַיָּ֨קָם בָּלָ֤ק בֶּן־ צִפּוֹר֙ מֶ֣לֶךְ מוֹאָ֔ב וַיִּלָּ֖חֶם בְּיִשְׂרָאֵ֑ל וַיִּשְׁלַ֗ח וַיִּקְרָ֛א לְבִלְעָ֥ם בֶּן־ בְּע֖וֹר לְקַלֵּ֥ל אֶתְכֶֽם׃ (כז) וְלֹ֥א אָבִ֖יתִי לִשְׁמֹ֣עַ לְבִלְעָ֑ם וַיְבָ֤רֶךְ בָּרוֹךְ֙ אֶתְכֶ֔ם וָאַצִּ֥ל אֶתְכֶ֖ם מִיָּדֽוֹ׃ (כח) וַתַּעַבְר֣וּ אֶת־ הַיַּרְדֵּן֮ וַתָּבֹ֣אוּ אֶל־ יְרִיחוֹ֒ וַיִּלָּחֲמ֣וּ בָכֶ֣ם בַּעֲלֵֽי־ יְ֠רִיחוֹ הָֽאֱמֹרִ֨י וְהַפְּרִזִּ֜י וְהַֽכְּנַעֲנִ֗י וְהַֽחִתִּי֙ וְהַגִּרְגָּשִׁ֔י הַֽחִוִּ֖י וְהַיְבוּסִ֑י וָאֶתֵּ֥ן אוֹתָ֖ם בְּיֶדְכֶֽם׃ (כט) וָאֶשְׁלַ֤ח לִפְנֵיכֶם֙ אֶת־ הַצִּרְעָ֔ה וַתְּגָ֤רֶשׁ אוֹתָם֙ מִפְּנֵיכֶ֔ם שְׁנֵ֖י מַלְכֵ֣י הָאֱמֹרִ֑י לֹ֥א בְחַרְבְּךָ֖ וְלֹ֥א בְקַשְׁתֶּֽךָ׃ (ל) וָאֶתֵּ֨ן לָכֶ֜ם אֶ֣רֶץ ׀ אֲשֶׁ֧ר לֹֽא־ יָגַ֣עְתָּ בָּ֗הּ וְעָרִים֙ אֲשֶׁ֣ר לֹא־ בְנִיתֶ֔ם וַתֵּשְׁב֖וּ בָּהֶ֑ם כְּרָמִ֤ים וְזֵיתִים֙ אֲשֶׁ֣ר לֹֽא־ נְטַעְתֶּ֔ם אַתֶּ֖ם אֹכְלִֽים׃ (לא) וְעַתָּ֞ה יְר֧אוּ אֶת־ יְהוָ֛ה וְעִבְד֥וּ אֹת֖וֹ בְּתָמִ֣ים וּבֶֽאֱמֶ֑ת וְהָסִ֣ירוּ אֶת־ אֱלֹהִ֗ים אֲשֶׁר֩ עָבְד֨וּ אֲבוֹתֵיכֶ֜ם בְּעֵ֤בֶר הַנָּהָר֙ וּבְמִצְרַ֔יִם וְעִבְד֖וּ אֶת־ יְהוָֽה׃ (לב) וְאִם֩ רַ֨ע בְּֽעֵינֵיכֶ֜ם לַעֲבֹ֣ד אֶת־ יְהוָ֗ה בַּחֲר֨וּ לָכֶ֣ם הַיּוֹם֮ אֶת־ מִ֣י תַעֲבֹדוּן֒ אִ֣ם אֶת־ אֱלֹהִ֞ים אֲשֶׁר־ עָבְד֣וּ אֲבוֹתֵיכֶ֗ם אֲשֶׁר֙ (בעבר) מֵעֵ֣בֶר הַנָּהָ֔ר וְאִם֙ אֶת־ אֱלֹהֵ֣י הָאֱמֹרִ֔י אֲשֶׁ֥ר אַתֶּ֖ם יֹשְׁבִ֣ים בְּאַרְצָ֑ם וְאָנֹכִ֣י וּבֵיתִ֔י נַעֲבֹ֖ד אֶת־ יְהוָֽה׃ (לג) וַיַּ֤עַן הָעָם֙ וַיֹּ֔אמֶר חָלִ֣ילָה לָּ֔נוּ מֵעֲזֹ֖ב אֶת־ יְהוָ֑ה לַעֲבֹ֖ד אֱלֹהִ֥ים אֲחֵרִֽים׃ (לד) כִּ֚י יְהוָ֣ה אֱלֹהֵ֔ינוּ הוּא֩ הַמַּעֲלֶ֨ה אֹתָ֧נוּ וְאֶת־ אֲבוֹתֵ֛ינוּ מֵאֶ֥רֶץ מִצְרַ֖יִם מִבֵּ֣ית עֲבָדִ֑ים וַאֲשֶׁ֧ר עָשָׂ֣ה לְעֵינֵ֗ינוּ אֶת־ הָאֹת֤וֹת הַגְּדֹלוֹת֙ הָאֵ֔לֶּה וַֽיִּשְׁמְרֵ֗נוּ בְּכָל־ הַדֶּ֙רֶךְ֙ אֲשֶׁ֣ר הָלַ֣כְנוּ בָ֔הּ וּבְכֹל֙ הָֽעַמִּ֔ים אֲשֶׁ֥ר עָבַ֖רְנוּ בְּקִרְבָּֽם׃ (לה) וַיְגָ֨רֶשׁ יְהוָ֜ה אֶת־ כָּל־ הָעַמִּ֗ים וְאֶת־ הָאֱמֹרִ֛י יֹשֵׁ֥ב הָאָ֖רֶץ מִפָּנֵ֑ינוּ גַּם־ אֲנַ֙חְנוּ֙ נַעֲבֹ֣ד אֶת־ יְהוָ֔ה כִּי־ ה֖וּא אֱלֹהֵֽינוּ׃ (לו) וַיֹּ֨אמֶר יְהוֹשֻׁ֜עַ אֶל־ הָעָ֗ם לֹ֤א תֽוּכְלוּ֙ לַעֲבֹ֣ד אֶת־ יְהוָ֔ה כִּֽי־ אֱלֹהִ֥ים קְדֹשִׁ֖ים ה֑וּא אֵֽל־ קַנּ֣וֹא ה֔וּא לֹֽא־ יִשָּׂ֥א לְפִשְׁעֲכֶ֖ם וּלְחַטֹּאותֵיכֶֽם׃ (לז) כִּ֤י תַֽעַזְבוּ֙ אֶת־ יְהוָ֔ה וַעֲבַדְתֶּ֖ם אֱלֹהֵ֣י נֵכָ֑ר וְשָׁ֨ב וְהֵרַ֤ע לָכֶם֙ וְכִלָּ֣ה אֶתְכֶ֔ם אַחֲרֵ֖י אֲשֶׁר־ הֵיטִ֥יב לָכֶֽם׃ (לח) וַיֹּ֥אמֶר הָעָ֖ם אֶל־ יְהוֹשֻׁ֑עַ לֹ֕א כִּ֥י אֶת־ יְהוָ֖ה נַעֲבֹֽד׃ (לט) וַיֹּ֨אמֶר יְהוֹשֻׁ֜עַ אֶל־ הָעָ֗ם עֵדִ֤ים אַתֶּם֙ בָּכֶ֔ם כִּֽי־ אַתֶּ֞ם בְּחַרְתֶּ֥ם לָכֶ֛ם אֶת־ יְהוָ֖ה לַעֲבֹ֣ד אוֹת֑וֹ וַיֹּאמְר֖וּ עֵדִֽים׃ (מ) וְעַתָּ֕ה הָסִ֛ירוּ אֶת־ אֱלֹהֵ֥י הַנֵּכָ֖ר אֲשֶׁ֣ר בְּקִרְבְּכֶ֑ם וְהַטּוּ֙ אֶת־ לְבַבְכֶ֔ם אֶל־ יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (מא) וַיֹּאמְר֥וּ הָעָ֖ם אֶל־ יְהוֹשֻׁ֑עַ אֶת־ יְהוָ֤ה אֱלֹהֵ֙ינוּ֙ נַעֲבֹ֔ד וּבְקוֹל֖וֹ נִשְׁמָֽע׃ (מב) וַיִּכְרֹ֨ת יְהוֹשֻׁ֧עַ בְּרִ֛ית לָעָ֖ם בַּיּ֣וֹם הַה֑וּא וַיָּ֥שֶׂם ל֛וֹ חֹ֥ק וּמִשְׁפָּ֖ט בִּשְׁכֶֽם׃ (מג) וַיִּכְתֹּ֤ב יְהוֹשֻׁ֙עַ֙ אֶת־ הַדְּבָרִ֣ים הָאֵ֔לֶּה בְּסֵ֖פֶר תּוֹרַ֣ת אֱלֹהִ֑ים וַיִּקַּח֙ אֶ֣בֶן גְּדוֹלָ֔ה וַיְקִימֶ֣הָ שָּׁ֔ם תַּ֚חַת הָֽאַלָּ֔ה אֲשֶׁ֖ר בְּמִקְדַּ֥שׁ יְהוָֽה׃ (מד) וַיֹּ֨אמֶר יְהוֹשֻׁ֜עַ אֶל־ כָּל־ הָעָ֗ם הִנֵּ֨ה הָאֶ֤בֶן הַזֹּאת֙ תִּֽהְיֶה־ בָּ֣נוּ לְעֵדָ֔ה כִּֽי־ הִ֣יא שָׁמְעָ֗ה אֵ֚ת כָּל־ אִמְרֵ֣י יְהוָ֔ה אֲשֶׁ֥ר דִּבֶּ֖ר עִמָּ֑נוּ וְהָיְתָ֤ה בָכֶם֙ לְעֵדָ֔ה פֶּֽן־ תְּכַחֲשׁ֖וּן בֵּאלֹהֵיכֶֽם׃ (מה) וַיְשַׁלַּ֤ח יְהוֹשֻׁ֙עַ֙ אֶת־ הָעָ֔ם אִ֖ישׁ לְנַחֲלָתֽוֹ׃ (מו) וַיְהִ֗י אַֽחֲרֵי֙ הַדְּבָרִ֣ים הָאֵ֔לֶּה וַיָּ֛מָת יְהוֹשֻׁ֥עַ בִּן־ נ֖וּן עֶ֣בֶד יְהוָ֑ה בֶּן־ מֵאָ֥ה וָעֶ֖שֶׂר שָׁנִֽים׃ (מז) וַיִּקְבְּר֤וּ אֹתוֹ֙ בִּגְב֣וּל נַחֲלָת֔וֹ בְּתִמְנַת־ סֶ֖רַח אֲשֶׁ֣ר בְּהַר־ אֶפְרָ֑יִם מִצְּפ֖וֹן לְהַר־ גָּֽעַשׁ׃ (מח) וַיַּעֲבֹ֤ד יִשְׂרָאֵל֙ אֶת־ יְהוָ֔ה כֹּ֖ל יְמֵ֣י יְהוֹשֻׁ֑עַ וְכֹ֣ל ׀ יְמֵ֣י הַזְּקֵנִ֗ים אֲשֶׁ֨ר הֶאֱרִ֤יכוּ יָמִים֙ אַחֲרֵ֣י יְהוֹשֻׁ֔עַ וַאֲשֶׁ֣ר יָדְע֗וּ אֵ֚ת כָּל־ מַעֲשֵׂ֣ה יְהוָ֔ה אֲשֶׁ֥ר עָשָׂ֖ה לְיִשְׂרָאֵֽל׃ (מט) וְאֶת־ עַצְמ֣וֹת י֠וֹסֵף אֲשֶׁר־ הֶעֱל֨וּ בְנֵי־ יִשְׂרָאֵ֥ל ׀ מִמִּצְרַיִם֮ קָבְר֣וּ בִשְׁכֶם֒ בְּחֶלְקַ֣ת הַשָּׂדֶ֗ה אֲשֶׁ֨ר קָנָ֧ה יַעֲקֹ֛ב מֵאֵ֛ת בְּנֵֽי־ חֲמ֥וֹר אֲבִֽי־ שְׁכֶ֖ם בְּמֵאָ֣ה קְשִׂיטָ֑ה וַיִּֽהְי֥וּ לִבְנֵֽי־ יוֹסֵ֖ף לְנַחֲלָֽה׃ (נ) וְאֶלְעָזָ֥ר בֶּֽן־ אַהֲרֹ֖ן מֵ֑ת וַיִּקְבְּר֣וּ אֹת֗וֹ בְּגִבְעַת֙ פִּֽינְחָ֣ס בְּנ֔וֹ אֲשֶׁ֥ר נִתַּן־ ל֖וֹ בְּהַ֥ר אֶפְרָֽיִם׃
