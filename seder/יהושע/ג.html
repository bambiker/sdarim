לפי החלוקה לפרקים: יהושע ד, כד – ו, כו · 42 פסוקים

פתיחה באתר · ספר יהושע כולו ·

(א) לְ֠מַעַן דַּ֜עַת כָּל־ עַמֵּ֤י הָאָ֙רֶץ֙ אֶת־ יַ֣ד יְהוָ֔ה כִּ֥י חֲזָקָ֖ה הִ֑יא לְמַ֧עַן יְרָאתֶ֛ם אֶת־ יְהוָ֥ה אֱלֹהֵיכֶ֖ם כָּל־ הַיָּמִֽים׃ (ב) וַיְהִ֣י כִשְׁמֹ֣עַ כָּל־ מַלְכֵ֣י הָאֱמֹרִ֡י אֲשֶׁר֩ בְּעֵ֨בֶר הַיַּרְדֵּ֜ן יָ֗מָּה וְכָל־ מַלְכֵ֤י הַֽכְּנַעֲנִי֙ אֲשֶׁ֣ר עַל־ הַיָּ֔ם אֵ֠ת אֲשֶׁר־ הוֹבִ֨ישׁ יְהוָ֜ה אֶת־ מֵ֧י הַיַּרְדֵּ֛ן מִפְּנֵ֥י בְנֵֽי־ יִשְׂרָאֵ֖ל עַד־ (עברנו) עָבְרָ֑ם וַיִּמַּ֣ס לְבָבָ֗ם וְלֹא־ הָ֨יָה בָ֥ם עוֹד֙ ר֔וּחַ מִפְּנֵ֖י בְּנֵֽי־ יִשְׂרָאֵֽל׃ (ג) בָּעֵ֣ת הַהִ֗יא אָמַ֤ר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ עֲשֵׂ֥ה לְךָ֖ חַֽרְב֣וֹת צֻרִ֑ים וְשׁ֛וּב מֹ֥ל אֶת־ בְּנֵֽי־ יִשְׂרָאֵ֖ל שֵׁנִֽית׃ (ד) וַיַּעַשׂ־ ל֥וֹ יְהוֹשֻׁ֖עַ חַֽרְב֣וֹת צֻרִ֑ים וַיָּ֙מָל֙ אֶת־ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶל־ גִּבְעַ֖ת הָעֲרָלֽוֹת׃ (ה) וְזֶ֥ה הַדָּבָ֖ר אֲשֶׁר־ מָ֣ל יְהוֹשֻׁ֑עַ כָּל־ הָעָ֣ם הַיֹּצֵא֩ מִמִּצְרַ֨יִם הַזְּכָרִ֜ים כֹּ֣ל ׀ אַנְשֵׁ֣י הַמִּלְחָמָ֗ה מֵ֤תוּ בַמִּדְבָּר֙ בַּדֶּ֔רֶךְ בְּצֵאתָ֖ם מִמִּצְרָֽיִם׃ (ו) כִּֽי־ מֻלִ֣ים הָי֔וּ כָּל־ הָעָ֖ם הַיֹּֽצְאִ֑ים וְכָל־ הָ֠עָם הַיִּלֹּדִ֨ים בַּמִּדְבָּ֥ר בַּדֶּ֛רֶךְ בְּצֵאתָ֥ם מִמִּצְרַ֖יִם לֹא־ מָֽלוּ׃ (ז) כִּ֣י ׀ אַרְבָּעִ֣ים שָׁנָ֗ה הָלְכ֣וּ בְנֵֽי־ יִשְׂרָאֵל֮ בַּמִּדְבָּר֒ עַד־ תֹּ֨ם כָּל־ הַגּ֜וֹי אַנְשֵׁ֤י הַמִּלְחָמָה֙ הַיֹּצְאִ֣ים מִמִּצְרַ֔יִם אֲשֶׁ֥ר לֹֽא־ שָׁמְע֖וּ בְּק֣וֹל יְהוָ֑ה אֲשֶׁ֨ר נִשְׁבַּ֤ע יְהוָה֙ לָהֶ֔ם לְבִלְתִּ֞י הַרְאוֹתָ֣ם אֶת־ הָאָ֗רֶץ אֲשֶׁר֩ נִשְׁבַּ֨ע יְהוָ֤ה לַֽאֲבוֹתָם֙ לָ֣תֶת לָ֔נוּ אֶ֛רֶץ זָבַ֥ת חָלָ֖ב וּדְבָֽשׁ׃ (ח) וְאֶת־ בְּנֵיהֶם֙ הֵקִ֣ים תַּחְתָּ֔ם אֹתָ֖ם מָ֣ל יְהוֹשֻׁ֑עַ כִּי־ עֲרֵלִ֣ים הָי֔וּ כִּ֛י לֹא־ מָ֥לוּ אוֹתָ֖ם בַּדָּֽרֶךְ׃ (ט) וַיְהִ֛י כַּאֲשֶׁר־ תַּ֥מּוּ כָל־ הַגּ֖וֹי לְהִמּ֑וֹל וַיֵּשְׁב֥וּ תַחְתָּ֛ם בַּֽמַּחֲנֶ֖ה עַ֥ד חֲיוֹתָֽם׃ (י) וַיֹּ֤אמֶר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ הַיּ֗וֹם גַּלּ֛וֹתִי אֶת־ חֶרְפַּ֥ת מִצְרַ֖יִם מֵעֲלֵיכֶ֑ם וַיִּקְרָ֞א שֵׁ֣ם הַמָּק֤וֹם הַהוּא֙ גִּלְגָּ֔ל עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (יא) וַיַּחֲנ֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל בַּגִּלְגָּ֑ל וַיַּעֲשׂ֣וּ אֶת־ הַפֶּ֡סַח בְּאַרְבָּעָה֩ עָשָׂ֨ר י֥וֹם לַחֹ֛דֶשׁ בָּעֶ֖רֶב בְּעַֽרְב֥וֹת יְרִיחֽוֹ׃ (יב) וַיֹּ֨אכְל֜וּ מֵעֲב֥וּר הָאָ֛רֶץ מִמָּֽחֳרַ֥ת הַפֶּ֖סַח מַצּ֣וֹת וְקָל֑וּי בְּעֶ֖צֶם הַיּ֥וֹם הַזֶּֽה׃ (יג) וַיִּשְׁבֹּ֨ת הַמָּ֜ן מִֽמָּחֳרָ֗ת בְּאָכְלָם֙ מֵעֲב֣וּר הָאָ֔רֶץ וְלֹא־ הָ֥יָה ע֛וֹד לִבְנֵ֥י יִשְׂרָאֵ֖ל מָ֑ן וַיֹּאכְל֗וּ מִתְּבוּאַת֙ אֶ֣רֶץ כְּנַ֔עַן בַּשָּׁנָ֖ה הַהִֽיא׃ (יד) וַיְהִ֗י בִּֽהְי֣וֹת יְהוֹשֻׁעַ֮ בִּירִיחוֹ֒ וַיִּשָּׂ֤א עֵינָיו֙ וַיַּ֔רְא וְהִנֵּה־ אִישׁ֙ עֹמֵ֣ד לְנֶגְדּ֔וֹ וְחַרְבּ֥וֹ שְׁלוּפָ֖ה בְּיָד֑וֹ וַיֵּ֨לֶךְ יְהוֹשֻׁ֤עַ אֵלָיו֙ וַיֹּ֣אמֶר ל֔וֹ הֲלָ֥נוּ אַתָּ֖ה אִם־ לְצָרֵֽינוּ׃ (טו) וַיֹּ֣אמֶר ׀ לֹ֗א כִּ֛י אֲנִ֥י שַׂר־ צְבָֽא־ יְהוָ֖ה עַתָּ֣ה בָ֑אתִי וַיִּפֹּל֩ יְהוֹשֻׁ֨עַ אֶל־ פָּנָ֥יו אַ֙רְצָה֙ וַיִּשְׁתָּ֔חוּ וַיֹּ֣אמֶר ל֔וֹ מָ֥ה אֲדֹנִ֖י מְדַבֵּ֥ר אֶל־ עַבְדּֽוֹ׃ (טז) וַיֹּאמֶר֩ שַׂר־ צְבָ֨א יְהוָ֜ה אֶל־ יְהוֹשֻׁ֗עַ שַׁל־ נַֽעַלְךָ֙ מֵעַ֣ל רַגְלֶ֔ךָ כִּ֣י הַמָּק֗וֹם אֲשֶׁ֥ר אַתָּ֛ה עֹמֵ֥ד עָלָ֖יו קֹ֣דֶשׁ ה֑וּא וַיַּ֥עַשׂ יְהוֹשֻׁ֖עַ כֵּֽן׃ (יז) וִֽירִיחוֹ֙ סֹגֶ֣רֶת וּמְסֻגֶּ֔רֶת מִפְּנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל אֵ֥ין יוֹצֵ֖א וְאֵ֥ין בָּֽא׃ (יח) וַיֹּ֤אמֶר יְהוָה֙ אֶל־ יְהוֹשֻׁ֔עַ רְאֵה֙ נָתַ֣תִּי בְיָֽדְךָ֔ אֶת־ יְרִיח֖וֹ וְאֶת־ מַלְכָּ֑הּ גִּבּוֹרֵ֖י הֶחָֽיִל׃ (יט) וְסַבֹּתֶ֣ם אֶת־ הָעִ֗יר כֹּ֚ל אַנְשֵׁ֣י הַמִּלְחָמָ֔ה הַקֵּ֥יף אֶת־ הָעִ֖יר פַּ֣עַם אֶחָ֑ת כֹּ֥ה תַעֲשֶׂ֖ה שֵׁ֥שֶׁת יָמִֽים׃ (כ) וְשִׁבְעָ֣ה כֹהֲנִ֡ים יִשְׂאוּ֩ שִׁבְעָ֨ה שׁוֹפְר֤וֹת הַיּֽוֹבְלִים֙ לִפְנֵ֣י הָאָר֔וֹן וּבַיּוֹם֙ הַשְּׁבִיעִ֔י תָּסֹ֥בּוּ אֶת־ הָעִ֖יר שֶׁ֣בַע פְּעָמִ֑ים וְהַכֹּ֣הֲנִ֔ים יִתְקְע֖וּ בַּשּׁוֹפָרֽוֹת׃ (כא) וְהָיָ֞ה בִּמְשֹׁ֣ךְ ׀ בְּקֶ֣רֶן הַיּוֹבֵ֗ל (בשמעכם) כְּשָׁמְעֲכֶם֙ אֶת־ ק֣וֹל הַשּׁוֹפָ֔ר יָרִ֥יעוּ כָל־ הָעָ֖ם תְּרוּעָ֣ה גְדוֹלָ֑ה וְנָ֨פְלָ֜ה חוֹמַ֤ת הָעִיר֙ תַּחְתֶּ֔יהָ וְעָל֥וּ הָעָ֖ם אִ֥ישׁ נֶגְדּֽוֹ׃ (כב) וַיִּקְרָ֞א יְהוֹשֻׁ֤עַ בִּן־ נוּן֙ אֶל־ הַכֹּ֣הֲנִ֔ים וַיֹּ֣אמֶר אֲלֵהֶ֔ם שְׂא֖וּ אֶת־ אֲר֣וֹן הַבְּרִ֑ית וְשִׁבְעָ֣ה כֹֽהֲנִ֗ים יִשְׂאוּ֙ שִׁבְעָ֤ה שֽׁוֹפְרוֹת֙ יוֹבְלִ֔ים לִפְנֵ֖י אֲר֥וֹן יְהוָֽה׃ (כג) (ויאמרו) וַיֹּ֙אמֶר֙ אֶל־ הָעָ֔ם עִבְר֖וּ וְסֹ֣בּוּ אֶת־ הָעִ֑יר וְהֶ֣חָל֔וּץ יַעֲבֹ֕ר לִפְנֵ֖י אֲר֥וֹן יְהוָֽה׃ (כד) וַיְהִ֗י כֶּאֱמֹ֣ר יְהוֹשֻׁעַ֮ אֶל־ הָעָם֒ וְשִׁבְעָ֣ה הַכֹּהֲנִ֡ים נֹשְׂאִים֩ שִׁבְעָ֨ה שׁוֹפְר֤וֹת הַיּֽוֹבְלִים֙ לִפְנֵ֣י יְהוָ֔ה עָבְר֕וּ וְתָקְע֖וּ בַּשּֽׁוֹפָר֑וֹת וַֽאֲרוֹן֙ בְּרִ֣ית יְהוָ֔ה הֹלֵ֖ךְ אַחֲרֵיהֶֽם׃ (כה) וְהֶחָל֣וּץ הֹלֵ֔ךְ לִפְנֵי֙ הַכֹּ֣הֲנִ֔ים (תקעו) תֹּקְעֵ֖י הַשּֽׁוֹפָר֑וֹת וְהַֽמְאַסֵּ֗ף הֹלֵךְ֙ אַחֲרֵ֣י הָאָר֔וֹן הָל֖וֹךְ וְתָק֥וֹעַ בַּשּׁוֹפָרֽוֹת׃ (כו) וְאֶת־ הָעָם֩ צִוָּ֨ה יְהוֹשֻׁ֜עַ לֵאמֹ֗ר לֹ֤א תָרִ֙יעוּ֙ וְלֹֽא־ תַשְׁמִ֣יעוּ אֶת־ קוֹלְכֶ֔ם וְלֹא־ יֵצֵ֥א מִפִּיכֶ֖ם דָּבָ֑ר עַ֠ד י֣וֹם אָמְרִ֧י אֲלֵיכֶ֛ם הָרִ֖יעוּ וַהֲרִיעֹתֶֽם׃ (כז) וַיַּסֵּ֤ב אֲרוֹן־ יְהוָה֙ אֶת־ הָעִ֔יר הַקֵּ֖ף פַּ֣עַם אֶחָ֑ת וַיָּבֹ֙אוּ֙ הַֽמַּחֲנֶ֔ה וַיָּלִ֖ינוּ בַּֽמַּחֲנֶֽה׃ (כח) וַיַּשְׁכֵּ֥ם יְהוֹשֻׁ֖עַ בַּבֹּ֑קֶר וַיִּשְׂא֥וּ הַכֹּהֲנִ֖ים אֶת־ אֲר֥וֹן יְהוָֽה׃ (כט) וְשִׁבְעָ֣ה הַכֹּהֲנִ֡ים נֹשְׂאִים֩ שִׁבְעָ֨ה שׁוֹפְר֜וֹת הַיֹּבְלִ֗ים לִפְנֵי֙ אֲר֣וֹן יְהוָ֔ה הֹלְכִ֣ים הָל֔וֹךְ וְתָקְע֖וּ בַּשּׁוֹפָר֑וֹת וְהֶחָלוּץ֙ הֹלֵ֣ךְ לִפְנֵיהֶ֔ם וְהַֽמְאַסֵּ֗ף הֹלֵךְ֙ אַֽחֲרֵי֙ אֲר֣וֹן יְהוָ֔ה (הולך) הָל֖וֹךְ וְתָק֥וֹעַ בַּשּׁוֹפָרֽוֹת׃ (ל) וַיָּסֹ֨בּוּ אֶת־ הָעִ֜יר בַּיּ֤וֹם הַשֵּׁנִי֙ פַּ֣עַם אַחַ֔ת וַיָּשֻׁ֖בוּ הַֽמַּחֲנֶ֑ה כֹּ֥ה עָשׂ֖וּ שֵׁ֥שֶׁת יָמִֽים׃ (לא) וַיְהִ֣י ׀ בַּיּ֣וֹם הַשְּׁבִיעִ֗י וַיַּשְׁכִּ֙מוּ֙ כַּעֲל֣וֹת הַשַּׁ֔חַר וַיָּסֹ֧בּוּ אֶת־ הָעִ֛יר כַּמִּשְׁפָּ֥ט הַזֶּ֖ה שֶׁ֣בַע פְּעָמִ֑ים רַ֚ק בַּיּ֣וֹם הַה֔וּא סָבְב֥וּ אֶת־ הָעִ֖יר שֶׁ֥בַע פְּעָמִֽים׃ (לב) וַיְהִי֙ בַּפַּ֣עַם הַשְּׁבִיעִ֔ית תָּקְע֥וּ הַכֹּהֲנִ֖ים בַּשּׁוֹפָר֑וֹת וַיֹּ֨אמֶר יְהוֹשֻׁ֤עַ אֶל־ הָעָם֙ הָרִ֔יעוּ כִּֽי־ נָתַ֧ן יְהוָ֛ה לָכֶ֖ם אֶת־ הָעִֽיר׃ (לג) וְהָיְתָ֨ה הָעִ֥יר חֵ֛רֶם הִ֥יא וְכָל־ אֲשֶׁר־ בָּ֖הּ לַֽיהוָ֑ה רַק֩ רָחָ֨ב הַזּוֹנָ֜ה תִּֽחְיֶ֗ה הִ֚יא וְכָל־ אֲשֶׁ֣ר אִתָּ֣הּ בַּבַּ֔יִת כִּ֣י הֶחְבְּאַ֔תָה אֶת־ הַמַּלְאָכִ֖ים אֲשֶׁ֥ר שָׁלָֽחְנוּ׃ (לד) וְרַק־ אַתֶּם֙ שִׁמְר֣וּ מִן־ הַחֵ֔רֶם פֶּֽן־ תַּחֲרִ֖ימוּ וּלְקַחְתֶּ֣ם מִן־ הַחֵ֑רֶם וְשַׂמְתֶּ֞ם אֶת־ מַחֲנֵ֤ה יִשְׂרָאֵל֙ לְחֵ֔רֶם וַעֲכַרְתֶּ֖ם אוֹתֽוֹ׃ (לה) וְכֹ֣ל ׀ כֶּ֣סֶף וְזָהָ֗ב וּכְלֵ֤י נְחֹ֙שֶׁת֙ וּבַרְזֶ֔ל קֹ֥דֶשׁ ה֖וּא לַֽיהוָ֑ה אוֹצַ֥ר יְהוָ֖ה יָבֽוֹא׃ (לו) וַיָּ֣רַע הָעָ֔ם וַֽיִּתְקְע֖וּ בַּשֹּֽׁפָר֑וֹת וַיְהִי֩ כִשְׁמֹ֨עַ הָעָ֜ם אֶת־ ק֣וֹל הַשּׁוֹפָ֗ר וַיָּרִ֤יעוּ הָעָם֙ תְּרוּעָ֣ה גְדוֹלָ֔ה וַתִּפֹּ֨ל הַֽחוֹמָ֜ה תַּחְתֶּ֗יהָ וַיַּ֨עַל הָעָ֤ם הָעִ֙ירָה֙ אִ֣ישׁ נֶגְדּ֔וֹ וַֽיִּלְכְּד֖וּ אֶת־ הָעִֽיר׃ (לז) וַֽיַּחֲרִ֙ימוּ֙ אֶת־ כָּל־ אֲשֶׁ֣ר בָּעִ֔יר מֵאִישׁ֙ וְעַד־ אִשָּׁ֔ה מִנַּ֖עַר וְעַד־ זָקֵ֑ן וְעַ֨ד שׁ֥וֹר וָשֶׂ֛ה וַחֲמ֖וֹר לְפִי־ חָֽרֶב׃ (לח) וְלִשְׁנַ֨יִם הָאֲנָשִׁ֜ים הַֽמְרַגְּלִ֤ים אֶת־ הָאָ֙רֶץ֙ אָמַ֣ר יְהוֹשֻׁ֔עַ בֹּ֖אוּ בֵּית־ הָאִשָּׁ֣ה הַזּוֹנָ֑ה וְהוֹצִ֨יאוּ מִשָּׁ֤ם אֶת־ הָֽאִשָּׁה֙ וְאֶת־ כָּל־ אֲשֶׁר־ לָ֔הּ כַּאֲשֶׁ֥ר נִשְׁבַּעְתֶּ֖ם לָֽהּ׃ (לט) וַיָּבֹ֜אוּ הַנְּעָרִ֣ים הַֽמְרַגְּלִ֗ים וַיֹּצִ֡יאוּ אֶת־ רָ֠חָב וְאֶת־ אָבִ֨יהָ וְאֶת־ אִמָּ֤הּ וְאֶת־ אַחֶ֙יהָ֙ וְאֶת־ כָּל־ אֲשֶׁר־ לָ֔הּ וְאֵ֥ת כָּל־ מִשְׁפְּחוֹתֶ֖יהָ הוֹצִ֑יאוּ וַיַּ֨נִּיח֔וּם מִח֖וּץ לְמַחֲנֵ֥ה יִשְׂרָאֵֽל׃ (מ) וְהָעִ֛יר שָׂרְפ֥וּ בָאֵ֖שׁ וְכָל־ אֲשֶׁר־ בָּ֑הּ רַ֣ק ׀ הַכֶּ֣סֶף וְהַזָּהָ֗ב וּכְלֵ֤י הַנְּחֹ֙שֶׁת֙ וְהַבַּרְזֶ֔ל נָתְנ֖וּ אוֹצַ֥ר בֵּית־ יְהוָֽה׃ (מא) וְֽאֶת־ רָחָ֣ב הַ֠זּוֹנָה וְאֶת־ בֵּ֨ית אָבִ֤יהָ וְאֶת־ כָּל־ אֲשֶׁר־ לָהּ֙ הֶחֱיָ֣ה יְהוֹשֻׁ֔עַ וַתֵּ֙שֶׁב֙ בְּקֶ֣רֶב יִשְׂרָאֵ֔ל עַ֖ד הַיּ֣וֹם הַזֶּ֑ה כִּ֤י הֶחְבִּ֙יאָה֙ אֶת־ הַמַּלְאָכִ֔ים אֲשֶׁר־ שָׁלַ֥ח יְהוֹשֻׁ֖עַ לְרַגֵּ֥ל אֶת־ יְרִיחֽוֹ׃ (מב) וַיַּשְׁבַּ֣ע יְהוֹשֻׁ֔עַ בָּעֵ֥ת הַהִ֖יא לֵאמֹ֑ר אָר֨וּר הָאִ֜ישׁ לִפְנֵ֣י יְהוָ֗ה אֲשֶׁ֤ר יָקוּם֙ וּבָנָ֞ה אֶת־ הָעִ֤יר הַזֹּאת֙ אֶת־ יְרִיח֔וֹ בִּבְכֹר֣וֹ יְיַסְּדֶ֔נָּה וּבִצְעִיר֖וֹ יַצִּ֥יב דְּלָתֶֽיהָ׃
