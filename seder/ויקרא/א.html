ויקרא סדר א

פרשת ויקרא

פתיחה באתר · ספר ויקרא כולו ·

(א) וַיִּקְרָ֖א אֶל־ מֹשֶׁ֑ה וַיְדַבֵּ֤ר יְהוָה֙ אֵלָ֔יו מֵאֹ֥הֶל מוֹעֵ֖ד לֵאמֹֽר׃ (ב) דַּבֵּ֞ר אֶל־ בְּנֵ֤י יִשְׂרָאֵל֙ וְאָמַרְתָּ֣ אֲלֵהֶ֔ם אָדָ֗ם כִּֽי־ יַקְרִ֥יב מִכֶּ֛ם קָרְבָּ֖ן לַֽיהוָ֑ה מִן־ הַבְּהֵמָ֗ה מִן־ הַבָּקָר֙ וּמִן־ הַצֹּ֔אן תַּקְרִ֖יבוּ אֶת־ קָרְבַּנְכֶֽם׃ (ג) אִם־ עֹלָ֤ה קָרְבָּנוֹ֙ מִן־ הַבָּקָ֔ר זָכָ֥ר תָּמִ֖ים יַקְרִיבֶ֑נּוּ אֶל־ פֶּ֝תַח אֹ֤הֶל מוֹעֵד֙ יַקְרִ֣יב אֹת֔וֹ לִרְצֹנ֖וֹ לִפְנֵ֥י יְהוָֽה׃ (ד) וְסָמַ֣ךְ יָד֔וֹ עַ֖ל רֹ֣אשׁ הָעֹלָ֑ה וְנִרְצָ֥ה ל֖וֹ לְכַפֵּ֥ר עָלָֽיו׃ (ה) וְשָׁחַ֛ט אֶת־ בֶּ֥ן הַבָּקָ֖ר לִפְנֵ֣י יְהוָ֑ה וְ֠הִקְרִיבוּ בְּנֵ֨י אַהֲרֹ֤ן הַֽכֹּֽהֲנִים֙ אֶת־ הַדָּ֔ם וְזָרְק֨וּ אֶת־ הַדָּ֤ם עַל־ הַמִּזְבֵּ֙חַ֙ סָבִ֔יב אֲשֶׁר־ פֶּ֖תַח אֹ֥הֶל מוֹעֵֽד׃ (ו) וְהִפְשִׁ֖יט אֶת־ הָעֹלָ֑ה וְנִתַּ֥ח אֹתָ֖הּ לִנְתָחֶֽיהָ׃ (ז) וְ֠נָתְנוּ בְּנֵ֨י אַהֲרֹ֧ן הַכֹּהֵ֛ן אֵ֖שׁ עַל־ הַמִּזְבֵּ֑חַ וְעָרְכ֥וּ עֵצִ֖ים עַל־ הָאֵֽשׁ׃ (ח) וְעָרְכ֗וּ בְּנֵ֤י אַהֲרֹן֙ הַכֹּ֣הֲנִ֔ים אֵ֚ת הַנְּתָחִ֔ים אֶת־ הָרֹ֖אשׁ וְאֶת־ הַפָּ֑דֶר עַל־ הָעֵצִים֙ אֲשֶׁ֣ר עַל־ הָאֵ֔שׁ אֲשֶׁ֖ר עַל־ הַמִּזְבֵּֽחַ׃ (ט) וְקִרְבּ֥וֹ וּכְרָעָ֖יו יִרְחַ֣ץ בַּמָּ֑יִם וְהִקְטִ֨יר הַכֹּהֵ֤ן אֶת־ הַכֹּל֙ הַמִּזְבֵּ֔חָה עֹלָ֛ה אִשֵּׁ֥ה רֵֽיחַ־ נִיח֖וֹחַ לַֽיהוָֽה׃ (י) וְאִם־ מִן־ הַצֹּ֨אן קָרְבָּנ֧וֹ מִן־ הַכְּשָׂבִ֛ים א֥וֹ מִן־ הָעִזִּ֖ים לְעֹלָ֑ה זָכָ֥ר תָּמִ֖ים יַקְרִיבֶֽנּוּ׃ (יא) וְשָׁחַ֨ט אֹת֜וֹ עַ֣ל יֶ֧רֶךְ הַמִּזְבֵּ֛חַ צָפֹ֖נָה לִפְנֵ֣י יְהוָ֑ה וְזָרְק֡וּ בְּנֵי֩ אַהֲרֹ֨ן הַכֹּהֲנִ֧ים אֶת־ דָּמ֛וֹ עַל־ הַמִּזְבֵּ֖חַ סָבִֽיב׃ (יב) וְנִתַּ֤ח אֹתוֹ֙ לִנְתָחָ֔יו וְאֶת־ רֹאשׁ֖וֹ וְאֶת־ פִּדְר֑וֹ וְעָרַ֤ךְ הַכֹּהֵן֙ אֹתָ֔ם עַל־ הָֽעֵצִים֙ אֲשֶׁ֣ר עַל־ הָאֵ֔שׁ אֲשֶׁ֖ר עַל־ הַמִּזְבֵּֽחַ׃ (יג) וְהַקֶּ֥רֶב וְהַכְּרָעַ֖יִם יִרְחַ֣ץ בַּמָּ֑יִם וְהִקְרִ֨יב הַכֹּהֵ֤ן אֶת־ הַכֹּל֙ וְהִקְטִ֣יר הַמִּזְבֵּ֔חָה עֹלָ֣ה ה֗וּא אִשֵּׁ֛ה רֵ֥יחַ נִיחֹ֖חַ לַיהוָֽה׃ (יד) וְאִ֧ם מִן־ הָע֛וֹף עֹלָ֥ה קָרְבָּנ֖וֹ לַֽיהוָ֑ה וְהִקְרִ֣יב מִן־ הַתֹּרִ֗ים א֛וֹ מִן־ בְּנֵ֥י הַיּוֹנָ֖ה אֶת־ קָרְבָּנֽוֹ׃ (טו) וְהִקְרִיב֤וֹ הַכֹּהֵן֙ אֶל־ הַמִּזְבֵּ֔חַ וּמָלַק֙ אֶת־ רֹאשׁ֔וֹ וְהִקְטִ֖יר הַמִּזְבֵּ֑חָה וְנִמְצָ֣ה דָמ֔וֹ עַ֖ל קִ֥יר הַמִּזְבֵּֽחַ׃ (טז) וְהֵסִ֥יר אֶת־ מֻרְאָת֖וֹ בְּנֹצָתָ֑הּ וְהִשְׁלִ֨יךְ אֹתָ֜הּ אֵ֤צֶל הַמִּזְבֵּ֙חַ֙ קֵ֔דְמָה אֶל־ מְק֖וֹם הַדָּֽשֶׁן׃ (יז) וְשִׁסַּ֨ע אֹת֣וֹ בִכְנָפָיו֮ לֹ֣א יַבְדִּיל֒ וְהִקְטִ֨יר אֹת֤וֹ הַכֹּהֵן֙ הַמִּזְבֵּ֔חָה עַל־ הָעֵצִ֖ים אֲשֶׁ֣ר עַל־ הָאֵ֑שׁ עֹלָ֣ה ה֗וּא אִשֵּׁ֛ה רֵ֥יחַ נִיחֹ֖חַ לַיהוָֽה׃ (יח) וְנֶ֗פֶשׁ כִּֽי־ תַקְרִ֞יב קָרְבַּ֤ן מִנְחָה֙ לַֽיהוָ֔ה סֹ֖לֶת יִהְיֶ֣ה קָרְבָּנ֑וֹ וְיָצַ֤ק עָלֶ֙יהָ֙ שֶׁ֔מֶן וְנָתַ֥ן עָלֶ֖יהָ לְבֹנָֽה׃ (יט) וֶֽהֱבִיאָ֗הּ אֶל־ בְּנֵ֣י אַהֲרֹן֮ הַכֹּהֲנִים֒ וְקָמַ֨ץ מִשָּׁ֜ם מְלֹ֣א קֻמְצ֗וֹ מִסָּלְתָּהּ֙ וּמִשַּׁמְנָ֔הּ עַ֖ל כָּל־ לְבֹנָתָ֑הּ וְהִקְטִ֨יר הַכֹּהֵ֜ן אֶת־ אַזְכָּרָתָהּ֙ הַמִּזְבֵּ֔חָה אִשֵּׁ֛ה רֵ֥יחַ נִיחֹ֖חַ לַיהוָֽה׃ (כ) וְהַנּוֹתֶ֙רֶת֙ מִן־ הַמִּנְחָ֔ה לְאַהֲרֹ֖ן וּלְבָנָ֑יו קֹ֥דֶשׁ קָֽדָשִׁ֖ים מֵאִשֵּׁ֥י יְהוָֽה׃ (כא) וְכִ֥י תַקְרִ֛ב קָרְבַּ֥ן מִנְחָ֖ה מַאֲפֵ֣ה תַנּ֑וּר סֹ֣לֶת חַלּ֤וֹת מַצֹּת֙ בְּלוּלֹ֣ת בַּשֶּׁ֔מֶן וּרְקִיקֵ֥י מַצּ֖וֹת מְשֻׁחִ֥ים בַּשָּֽׁמֶן׃ (כב) וְאִם־ מִנְחָ֥ה עַל־ הַֽמַּחֲבַ֖ת קָרְבָּנֶ֑ךָ סֹ֛לֶת בְּלוּלָ֥ה בַשֶּׁ֖מֶן מַצָּ֥ה תִהְיֶֽה׃ (כג) פָּת֤וֹת אֹתָהּ֙ פִּתִּ֔ים וְיָצַקְתָּ֥ עָלֶ֖יהָ שָׁ֑מֶן מִנְחָ֖ה הִֽוא׃ (כד) וְאִם־ מִנְחַ֥ת מַרְחֶ֖שֶׁת קָרְבָּנֶ֑ךָ סֹ֥לֶת בַּשֶּׁ֖מֶן תֵּעָשֶֽׂה׃ (כה) וְהֵבֵאתָ֣ אֶת־ הַמִּנְחָ֗ה אֲשֶׁ֧ר יֵעָשֶׂ֛ה מֵאֵ֖לֶּה לַיהוָ֑ה וְהִקְרִיבָהּ֙ אֶל־ הַכֹּהֵ֔ן וְהִגִּישָׁ֖הּ אֶל־ הַמִּזְבֵּֽחַ׃ (כו) וְהֵרִ֨ים הַכֹּהֵ֤ן מִן־ הַמִּנְחָה֙ אֶת־ אַזְכָּ֣רָתָ֔הּ וְהִקְטִ֖יר הַמִּזְבֵּ֑חָה אִשֵּׁ֛ה רֵ֥יחַ נִיחֹ֖חַ לַיהוָֽה׃ (כז) וְהַנּוֹתֶ֙רֶת֙ מִן־ הַמִּנְחָ֔ה לְאַהֲרֹ֖ן וּלְבָנָ֑יו קֹ֥דֶשׁ קָֽדָשִׁ֖ים מֵאִשֵּׁ֥י יְהוָֽה׃ (כח) כָּל־ הַמִּנְחָ֗ה אֲשֶׁ֤ר תַּקְרִ֙יבוּ֙ לַיהוָ֔ה לֹ֥א תֵעָשֶׂ֖ה חָמֵ֑ץ כִּ֤י כָל־ שְׂאֹר֙ וְכָל־ דְּבַ֔שׁ לֹֽא־ תַקְטִ֧ירוּ מִמֶּ֛נּוּ אִשֶּׁ֖ה לַֽיהוָֽה׃ (כט) קָרְבַּ֥ן רֵאשִׁ֛ית תַּקְרִ֥יבוּ אֹתָ֖ם לַיהוָ֑ה וְאֶל־ הַמִּזְבֵּ֥חַ לֹא־ יַעֲל֖וּ לְרֵ֥יחַ נִיחֹֽחַ׃ (ל) וְכָל־ קָרְבַּ֣ן מִנְחָתְךָ֮ בַּמֶּ֣לַח תִּמְלָח֒ וְלֹ֣א תַשְׁבִּ֗ית מֶ֚לַח בְּרִ֣ית אֱלֹהֶ֔יךָ מֵעַ֖ל מִנְחָתֶ֑ךָ עַ֥ל כָּל־ קָרְבָּנְךָ֖ תַּקְרִ֥יב מֶֽלַח׃ (לא) וְאִם־ תַּקְרִ֛יב מִנְחַ֥ת בִּכּוּרִ֖ים לַיהוָ֑ה אָבִ֞יב קָל֤וּי בָּאֵשׁ֙ גֶּ֣רֶשׂ כַּרְמֶ֔ל תַּקְרִ֕יב אֵ֖ת מִנְחַ֥ת בִּכּוּרֶֽיךָ׃ (לב) וְנָתַתָּ֤ עָלֶ֙יהָ֙ שֶׁ֔מֶן וְשַׂמְתָּ֥ עָלֶ֖יהָ לְבֹנָ֑ה מִנְחָ֖ה הִֽוא׃ (לג) וְהִקְטִ֨יר הַכֹּהֵ֜ן אֶת־ אַזְכָּרָתָ֗הּ מִגִּרְשָׂהּ֙ וּמִשַּׁמְנָ֔הּ עַ֖ל כָּל־ לְבֹנָתָ֑הּ אִשֶּׁ֖ה לַיהוָֽה׃ (לד) וְאִם־ זֶ֥בַח שְׁלָמִ֖ים קָרְבָּנ֑וֹ אִ֤ם מִן־ הַבָּקָר֙ ה֣וּא מַקְרִ֔יב אִם־ זָכָר֙ אִם־ נְקֵבָ֔ה תָּמִ֥ים יַקְרִיבֶ֖נּוּ לִפְנֵ֥י יְהוָֽה׃ (לה) וְסָמַ֤ךְ יָדוֹ֙ עַל־ רֹ֣אשׁ קָרְבָּנ֔וֹ וּשְׁחָט֕וֹ פֶּ֖תַח אֹ֣הֶל מוֹעֵ֑ד וְזָרְק֡וּ בְּנֵי֩ אַהֲרֹ֨ן הַכֹּהֲנִ֧ים אֶת־ הַדָּ֛ם עַל־ הַמִּזְבֵּ֖חַ סָבִֽיב׃ (לו) וְהִקְרִיב֙ מִזֶּ֣בַח הַשְּׁלָמִ֔ים אִשֶּׁ֖ה לַיהוָ֑ה אֶת־ הַחֵ֙לֶב֙ הַֽמְכַסֶּ֣ה אֶת־ הַקֶּ֔רֶב וְאֵת֙ כָּל־ הַחֵ֔לֶב אֲשֶׁ֖ר עַל־ הַקֶּֽרֶב׃ (לז) וְאֵת֙ שְׁתֵּ֣י הַכְּלָיֹ֔ת וְאֶת־ הַחֵ֙לֶב֙ אֲשֶׁ֣ר עֲלֵהֶ֔ן אֲשֶׁ֖ר עַל־ הַכְּסָלִ֑ים וְאֶת־ הַיֹּתֶ֙רֶת֙ עַל־ הַכָּבֵ֔ד עַל־ הַכְּלָי֖וֹת יְסִירֶֽנָּה׃ (לח) וְהִקְטִ֨ירוּ אֹת֤וֹ בְנֵֽי־ אַהֲרֹן֙ הַמִּזְבֵּ֔חָה עַל־ הָ֣עֹלָ֔ה אֲשֶׁ֥ר עַל־ הָעֵצִ֖ים אֲשֶׁ֣ר עַל־ הָאֵ֑שׁ אִשֵּׁ֛ה רֵ֥יחַ נִיחֹ֖חַ לַֽיהוָֽה׃ (לט) וְאִם־ מִן־ הַצֹּ֧אן קָרְבָּנ֛וֹ לְזֶ֥בַח שְׁלָמִ֖ים לַיהוָ֑ה זָכָר֙ א֣וֹ נְקֵבָ֔ה תָּמִ֖ים יַקְרִיבֶֽנּוּ׃ (מ) אִם־ כֶּ֥שֶׂב הֽוּא־ מַקְרִ֖יב אֶת־ קָרְבָּנ֑וֹ וְהִקְרִ֥יב אֹת֖וֹ לִפְנֵ֥י יְהוָֽה׃ (מא) וְסָמַ֤ךְ אֶת־ יָדוֹ֙ עַל־ רֹ֣אשׁ קָרְבָּנ֔וֹ וְשָׁחַ֣ט אֹת֔וֹ לִפְנֵ֖י אֹ֣הֶל מוֹעֵ֑ד וְ֠זָרְקוּ בְּנֵ֨י אַהֲרֹ֧ן אֶת־ דָּמ֛וֹ עַל־ הַמִּזְבֵּ֖חַ סָבִֽיב׃ (מב) וְהִקְרִ֨יב מִזֶּ֣בַח הַשְּׁלָמִים֮ אִשֶּׁ֣ה לַיהוָה֒ חֶלְבּוֹ֙ הָאַלְיָ֣ה תְמִימָ֔ה לְעֻמַּ֥ת הֶעָצֶ֖ה יְסִירֶ֑נָּה וְאֶת־ הַחֵ֙לֶב֙ הַֽמְכַסֶּ֣ה אֶת־ הַקֶּ֔רֶב וְאֵת֙ כָּל־ הַחֵ֔לֶב אֲשֶׁ֖ר עַל־ הַקֶּֽרֶב׃ (מג) וְאֵת֙ שְׁתֵּ֣י הַכְּלָיֹ֔ת וְאֶת־ הַחֵ֙לֶב֙ אֲשֶׁ֣ר עֲלֵהֶ֔ן אֲשֶׁ֖ר עַל־ הַכְּסָלִ֑ים וְאֶת־ הַיֹּתֶ֙רֶת֙ עַל־ הַכָּבֵ֔ד עַל־ הַכְּלָיֹ֖ת יְסִירֶֽנָּה׃ (מד) וְהִקְטִיר֥וֹ הַכֹּהֵ֖ן הַמִּזְבֵּ֑חָה לֶ֥חֶם אִשֶּׁ֖ה לַיהוָֽה׃ (מה) וְאִ֥ם עֵ֖ז קָרְבָּנ֑וֹ וְהִקְרִיב֖וֹ לִפְנֵ֥י יְהוָֽה׃ (מו) וְסָמַ֤ךְ אֶת־ יָדוֹ֙ עַל־ רֹאשׁ֔וֹ וְשָׁחַ֣ט אֹת֔וֹ לִפְנֵ֖י אֹ֣הֶל מוֹעֵ֑ד וְ֠זָרְקוּ בְּנֵ֨י אַהֲרֹ֧ן אֶת־ דָּמ֛וֹ עַל־ הַמִּזְבֵּ֖חַ סָבִֽיב׃ (מז) וְהִקְרִ֤יב מִמֶּ֙נּוּ֙ קָרְבָּנ֔וֹ אִשֶּׁ֖ה לַֽיהוָ֑ה אֶת־ הַחֵ֙לֶב֙ הַֽמְכַסֶּ֣ה אֶת־ הַקֶּ֔רֶב וְאֵת֙ כָּל־ הַחֵ֔לֶב אֲשֶׁ֖ר עַל־ הַקֶּֽרֶב׃ (מח) וְאֵת֙ שְׁתֵּ֣י הַכְּלָיֹ֔ת וְאֶת־ הַחֵ֙לֶב֙ אֲשֶׁ֣ר עֲלֵהֶ֔ן אֲשֶׁ֖ר עַל־ הַכְּסָלִ֑ים וְאֶת־ הַיֹּתֶ֙רֶת֙ עַל־ הַכָּבֵ֔ד עַל־ הַכְּלָיֹ֖ת יְסִירֶֽנָּה׃ (מט) וְהִקְטִירָ֥ם הַכֹּהֵ֖ן הַמִּזְבֵּ֑חָה לֶ֤חֶם אִשֶּׁה֙ לְרֵ֣יחַ נִיחֹ֔חַ כָּל־ חֵ֖לֶב לַיהוָֽה׃ (נ) חֻקַּ֤ת עוֹלָם֙ לְדֹרֹ֣תֵיכֶ֔ם בְּכֹ֖ל מֽוֹשְׁבֹתֵיכֶ֑ם כָּל־ חֵ֥לֶב וְכָל־ דָּ֖ם לֹ֥א תֹאכֵֽלוּ׃
