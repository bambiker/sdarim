ויקרא סדר ב

פרשת ויקרא · פרשת צו

פתיחה באתר · ספר ויקרא כולו ·

(א) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (ב) דַּבֵּ֞ר אֶל־ בְּנֵ֣י יִשְׂרָאֵ֘ל לֵאמֹר֒ נֶ֗פֶשׁ כִּֽי־ תֶחֱטָ֤א בִשְׁגָגָה֙ מִכֹּל֙ מִצְוֺ֣ת יְהוָ֔ה אֲשֶׁ֖ר לֹ֣א תֵעָשֶׂ֑ינָה וְעָשָׂ֕ה מֵאַחַ֖ת מֵהֵֽנָּה׃ (ג) אִ֣ם הַכֹּהֵ֧ן הַמָּשִׁ֛יחַ יֶחֱטָ֖א לְאַשְׁמַ֣ת הָעָ֑ם וְהִקְרִ֡יב עַ֣ל חַטָּאתוֹ֩ אֲשֶׁ֨ר חָטָ֜א פַּ֣ר בֶּן־ בָּקָ֥ר תָּמִ֛ים לַיהוָ֖ה לְחַטָּֽאת׃ (ד) וְהֵבִ֣יא אֶת־ הַפָּ֗ר אֶל־ פֶּ֛תַח אֹ֥הֶל מוֹעֵ֖ד לִפְנֵ֣י יְהוָ֑ה וְסָמַ֤ךְ אֶת־ יָדוֹ֙ עַל־ רֹ֣אשׁ הַפָּ֔ר וְשָׁחַ֥ט אֶת־ הַפָּ֖ר לִפְנֵ֥י יְהוָֽה׃ (ה) וְלָקַ֛ח הַכֹּהֵ֥ן הַמָּשִׁ֖יחַ מִדַּ֣ם הַפָּ֑ר וְהֵבִ֥יא אֹת֖וֹ אֶל־ אֹ֥הֶל מוֹעֵֽד׃ (ו) וְטָבַ֧ל הַכֹּהֵ֛ן אֶת־ אֶצְבָּע֖וֹ בַּדָּ֑ם וְהִזָּ֨ה מִן־ הַדָּ֜ם שֶׁ֤בַע פְּעָמִים֙ לִפְנֵ֣י יְהוָ֔ה אֶת־ פְּנֵ֖י פָּרֹ֥כֶת הַקֹּֽדֶשׁ׃ (ז) וְנָתַן֩ הַכֹּהֵ֨ן מִן־ הַדָּ֜ם עַל־ קַ֠רְנוֹת מִזְבַּ֨ח קְטֹ֤רֶת הַסַּמִּים֙ לִפְנֵ֣י יְהוָ֔ה אֲשֶׁ֖ר בְּאֹ֣הֶל מוֹעֵ֑ד וְאֵ֣ת ׀ כָּל־ דַּ֣ם הַפָּ֗ר יִשְׁפֹּךְ֙ אֶל־ יְסוֹד֙ מִזְבַּ֣ח הָעֹלָ֔ה אֲשֶׁר־ פֶּ֖תַח אֹ֥הֶל מוֹעֵֽד׃ (ח) וְאֶת־ כָּל־ חֵ֛לֶב פַּ֥ר הַֽחַטָּ֖את יָרִ֣ים מִמֶּ֑נּוּ אֶת־ הַחֵ֙לֶב֙ הַֽמְכַסֶּ֣ה עַל־ הַקֶּ֔רֶב וְאֵת֙ כָּל־ הַחֵ֔לֶב אֲשֶׁ֖ר עַל־ הַקֶּֽרֶב׃ (ט) וְאֵת֙ שְׁתֵּ֣י הַכְּלָיֹ֔ת וְאֶת־ הַחֵ֙לֶב֙ אֲשֶׁ֣ר עֲלֵיהֶ֔ן אֲשֶׁ֖ר עַל־ הַכְּסָלִ֑ים וְאֶת־ הַיֹּתֶ֙רֶת֙ עַל־ הַכָּבֵ֔ד עַל־ הַכְּלָי֖וֹת יְסִירֶֽנָּה׃ (י) כַּאֲשֶׁ֣ר יוּרַ֔ם מִשּׁ֖וֹר זֶ֣בַח הַשְּׁלָמִ֑ים וְהִקְטִירָם֙ הַכֹּהֵ֔ן עַ֖ל מִזְבַּ֥ח הָעֹלָֽה׃ (יא) וְאֶת־ ע֤וֹר הַפָּר֙ וְאֶת־ כָּל־ בְּשָׂר֔וֹ עַל־ רֹאשׁ֖וֹ וְעַל־ כְּרָעָ֑יו וְקִרְבּ֖וֹ וּפִרְשֽׁוֹ׃ (יב) וְהוֹצִ֣יא אֶת־ כָּל־ הַ֠פָּר אֶל־ מִח֨וּץ לַֽמַּחֲנֶ֜ה אֶל־ מָק֤וֹם טָהוֹר֙ אֶל־ שֶׁ֣פֶךְ הַדֶּ֔שֶׁן וְשָׂרַ֥ף אֹת֛וֹ עַל־ עֵצִ֖ים בָּאֵ֑שׁ עַל־ שֶׁ֥פֶךְ הַדֶּ֖שֶׁן יִשָּׂרֵֽף׃ (יג) וְאִ֨ם כָּל־ עֲדַ֤ת יִשְׂרָאֵל֙ יִשְׁגּ֔וּ וְנֶעְלַ֣ם דָּבָ֔ר מֵעֵינֵ֖י הַקָּהָ֑ל וְ֠עָשׂוּ אַחַ֨ת מִכָּל־ מִצְוֺ֧ת יְהוָ֛ה אֲשֶׁ֥ר לֹא־ תֵעָשֶׂ֖ינָה וְאָשֵֽׁמוּ׃ (יד) וְנֽוֹדְעָה֙ הַֽחַטָּ֔את אֲשֶׁ֥ר חָטְא֖וּ עָלֶ֑יהָ וְהִקְרִ֨יבוּ הַקָּהָ֜ל פַּ֤ר בֶּן־ בָּקָר֙ לְחַטָּ֔את וְהֵבִ֣יאוּ אֹת֔וֹ לִפְנֵ֖י אֹ֥הֶל מוֹעֵֽד׃ (טו) וְ֠סָמְכוּ זִקְנֵ֨י הָעֵדָ֧ה אֶת־ יְדֵיהֶ֛ם עַל־ רֹ֥אשׁ הַפָּ֖ר לִפְנֵ֣י יְהוָ֑ה וְשָׁחַ֥ט אֶת־ הַפָּ֖ר לִפְנֵ֥י יְהוָֽה׃ (טז) וְהֵבִ֛יא הַכֹּהֵ֥ן הַמָּשִׁ֖יחַ מִדַּ֣ם הַפָּ֑ר אֶל־ אֹ֖הֶל מוֹעֵֽד׃ (יז) וְטָבַ֧ל הַכֹּהֵ֛ן אֶצְבָּע֖וֹ מִן־ הַדָּ֑ם וְהִזָּ֞ה שֶׁ֤בַע פְּעָמִים֙ לִפְנֵ֣י יְהוָ֔ה אֵ֖ת פְּנֵ֥י הַפָּרֹֽכֶת׃ (יח) וּמִן־ הַדָּ֞ם יִתֵּ֣ן ׀ עַל־ קַרְנֹ֣ת הַמִּזְבֵּ֗חַ אֲשֶׁר֙ לִפְנֵ֣י יְהוָ֔ה אֲשֶׁ֖ר בְּאֹ֣הֶל מוֹעֵ֑ד וְאֵ֣ת כָּל־ הַדָּ֗ם יִשְׁפֹּךְ֙ אֶל־ יְסוֹד֙ מִזְבַּ֣ח הָעֹלָ֔ה אֲשֶׁר־ פֶּ֖תַח אֹ֥הֶל מוֹעֵֽד׃ (יט) וְאֵ֥ת כָּל־ חֶלְבּ֖וֹ יָרִ֣ים מִמֶּ֑נּוּ וְהִקְטִ֖יר הַמִּזְבֵּֽחָה׃ (כ) וְעָשָׂ֣ה לַפָּ֔ר כַּאֲשֶׁ֤ר עָשָׂה֙ לְפַ֣ר הַֽחַטָּ֔את כֵּ֖ן יַעֲשֶׂה־ לּ֑וֹ וְכִפֶּ֧ר עֲלֵהֶ֛ם הַכֹּהֵ֖ן וְנִסְלַ֥ח לָהֶֽם׃ (כא) וְהוֹצִ֣יא אֶת־ הַפָּ֗ר אֶל־ מִחוּץ֙ לַֽמַּחֲנֶ֔ה וְשָׂרַ֣ף אֹת֔וֹ כַּאֲשֶׁ֣ר שָׂרַ֔ף אֵ֖ת הַפָּ֣ר הָרִאשׁ֑וֹן חַטַּ֥את הַקָּהָ֖ל הֽוּא׃ (כב) אֲשֶׁ֥ר נָשִׂ֖יא יֶֽחֱטָ֑א וְעָשָׂ֡ה אַחַ֣ת מִכָּל־ מִצְוֺת֩ יְהוָ֨ה אֱלֹהָ֜יו אֲשֶׁ֧ר לֹא־ תֵעָשֶׂ֛ינָה בִּשְׁגָגָ֖ה וְאָשֵֽׁם׃ (כג) אֽוֹ־ הוֹדַ֤ע אֵלָיו֙ חַטָּאת֔וֹ אֲשֶׁ֥ר חָטָ֖א בָּ֑הּ וְהֵבִ֧יא אֶת־ קָרְבָּנ֛וֹ שְׂעִ֥יר עִזִּ֖ים זָכָ֥ר תָּמִֽים׃ (כד) וְסָמַ֤ךְ יָדוֹ֙ עַל־ רֹ֣אשׁ הַשָּׂעִ֔יר וְשָׁחַ֣ט אֹת֔וֹ בִּמְק֛וֹם אֲשֶׁר־ יִשְׁחַ֥ט אֶת־ הָעֹלָ֖ה לִפְנֵ֣י יְהוָ֑ה חַטָּ֖את הֽוּא׃ (כה) וְלָקַ֨ח הַכֹּהֵ֜ן מִדַּ֤ם הַֽחַטָּאת֙ בְּאֶצְבָּע֔וֹ וְנָתַ֕ן עַל־ קַרְנֹ֖ת מִזְבַּ֣ח הָעֹלָ֑ה וְאֶת־ דָּמ֣וֹ יִשְׁפֹּ֔ךְ אֶל־ יְס֖וֹד מִזְבַּ֥ח הָעֹלָֽה׃ (כו) וְאֶת־ כָּל־ חֶלְבּוֹ֙ יַקְטִ֣יר הַמִּזְבֵּ֔חָה כְּחֵ֖לֶב זֶ֣בַח הַשְּׁלָמִ֑ים וְכִפֶּ֨ר עָלָ֧יו הַכֹּהֵ֛ן מֵחַטָּאת֖וֹ וְנִסְלַ֥ח לֽוֹ׃ (כז) וְאִם־ נֶ֧פֶשׁ אַחַ֛ת תֶּחֱטָ֥א בִשְׁגָגָ֖ה מֵעַ֣ם הָאָ֑רֶץ בַּ֠עֲשֹׂתָהּ אַחַ֨ת מִמִּצְוֺ֧ת יְהוָ֛ה אֲשֶׁ֥ר לֹא־ תֵעָשֶׂ֖ינָה וְאָשֵֽׁם׃ (כח) א֚וֹ הוֹדַ֣ע אֵלָ֔יו חַטָּאת֖וֹ אֲשֶׁ֣ר חָטָ֑א וְהֵבִ֨יא קָרְבָּנ֜וֹ שְׂעִירַ֤ת עִזִּים֙ תְּמִימָ֣ה נְקֵבָ֔ה עַל־ חַטָּאת֖וֹ אֲשֶׁ֥ר חָטָֽא׃ (כט) וְסָמַךְ֙ אֶת־ יָד֔וֹ עַ֖ל רֹ֣אשׁ הַֽחַטָּ֑את וְשָׁחַט֙ אֶת־ הַ֣חַטָּ֔את בִּמְק֖וֹם הָעֹלָֽה׃ (ל) וְלָקַ֨ח הַכֹּהֵ֤ן מִדָּמָהּ֙ בְּאֶצְבָּע֔וֹ וְנָתַ֕ן עַל־ קַרְנֹ֖ת מִזְבַּ֣ח הָעֹלָ֑ה וְאֶת־ כָּל־ דָּמָ֣הּ יִשְׁפֹּ֔ךְ אֶל־ יְס֖וֹד הַמִּזְבֵּֽחַ׃ (לא) וְאֶת־ כָּל־ חֶלְבָּ֣הּ יָסִ֗יר כַּאֲשֶׁ֨ר הוּסַ֣ר חֵלֶב֮ מֵעַ֣ל זֶ֣בַח הַשְּׁלָמִים֒ וְהִקְטִ֤יר הַכֹּהֵן֙ הַמִּזְבֵּ֔חָה לְרֵ֥יחַ נִיחֹ֖חַ לַיהוָ֑ה וְכִפֶּ֥ר עָלָ֛יו הַכֹּהֵ֖ן וְנִסְלַ֥ח לֽוֹ׃ (לב) וְאִם־ כֶּ֛בֶשׂ יָבִ֥יא קָרְבָּנ֖וֹ לְחַטָּ֑את נְקֵבָ֥ה תְמִימָ֖ה יְבִיאֶֽנָּה׃ (לג) וְסָמַךְ֙ אֶת־ יָד֔וֹ עַ֖ל רֹ֣אשׁ הַֽחַטָּ֑את וְשָׁחַ֤ט אֹתָהּ֙ לְחַטָּ֔את בִּמְק֕וֹם אֲשֶׁ֥ר יִשְׁחַ֖ט אֶת־ הָעֹלָֽה׃ (לד) וְלָקַ֨ח הַכֹּהֵ֜ן מִדַּ֤ם הַֽחַטָּאת֙ בְּאֶצְבָּע֔וֹ וְנָתַ֕ן עַל־ קַרְנֹ֖ת מִזְבַּ֣ח הָעֹלָ֑ה וְאֶת־ כָּל־ דָּמָ֣הּ יִשְׁפֹּ֔ךְ אֶל־ יְס֖וֹד הַמִּזְבֵּֽחַ׃ (לה) וְאֶת־ כָּל־ חֶלְבָּ֣ה יָסִ֗יר כַּאֲשֶׁ֨ר יוּסַ֥ר חֵֽלֶב־ הַכֶּשֶׂב֮ מִזֶּ֣בַח הַשְּׁלָמִים֒ וְהִקְטִ֨יר הַכֹּהֵ֤ן אֹתָם֙ הַמִּזְבֵּ֔חָה עַ֖ל אִשֵּׁ֣י יְהוָ֑ה וְכִפֶּ֨ר עָלָ֧יו הַכֹּהֵ֛ן עַל־ חַטָּאת֥וֹ אֲשֶׁר־ חָטָ֖א וְנִסְלַ֥ח לֽוֹ׃ (לו) וְנֶ֣פֶשׁ כִּֽי־ תֶחֱטָ֗א וְשָֽׁמְעָה֙ ק֣וֹל אָלָ֔ה וְה֣וּא עֵ֔ד א֥וֹ רָאָ֖ה א֣וֹ יָדָ֑ע אִם־ ל֥וֹא יַגִּ֖יד וְנָשָׂ֥א עֲוֺנֽוֹ׃ (לז) א֣וֹ נֶ֗פֶשׁ אֲשֶׁ֣ר תִּגַּע֮ בְּכָל־ דָּבָ֣ר טָמֵא֒ אוֹ֩ בְנִבְלַ֨ת חַיָּ֜ה טְמֵאָ֗ה א֤וֹ בְּנִבְלַת֙ בְּהֵמָ֣ה טְמֵאָ֔ה א֕וֹ בְּנִבְלַ֖ת שֶׁ֣רֶץ טָמֵ֑א וְנֶעְלַ֣ם מִמֶּ֔נּוּ וְה֥וּא טָמֵ֖א וְאָשֵֽׁם׃ (לח) א֣וֹ כִ֤י יִגַּע֙ בְּטֻמְאַ֣ת אָדָ֔ם לְכֹל֙ טֻמְאָת֔וֹ אֲשֶׁ֥ר יִטְמָ֖א בָּ֑הּ וְנֶעְלַ֣ם מִמֶּ֔נּוּ וְה֥וּא יָדַ֖ע וְאָשֵֽׁם׃ (לט) א֣וֹ נֶ֡פֶשׁ כִּ֣י תִשָּׁבַע֩ לְבַטֵּ֨א בִשְׂפָתַ֜יִם לְהָרַ֣ע ׀ א֣וֹ לְהֵיטִ֗יב לְ֠כֹל אֲשֶׁ֨ר יְבַטֵּ֧א הָאָדָ֛ם בִּשְׁבֻעָ֖ה וְנֶעְלַ֣ם מִמֶּ֑נּוּ וְהוּא־ יָדַ֥ע וְאָשֵׁ֖ם לְאַחַ֥ת מֵאֵֽלֶּה׃ (מ) וְהָיָ֥ה כִֽי־ יֶאְשַׁ֖ם לְאַחַ֣ת מֵאֵ֑לֶּה וְהִ֨תְוַדָּ֔ה אֲשֶׁ֥ר חָטָ֖א עָלֶֽיהָ׃ (מא) וְהֵבִ֣יא אֶת־ אֲשָׁמ֣וֹ לַיהוָ֡ה עַ֣ל חַטָּאתוֹ֩ אֲשֶׁ֨ר חָטָ֜א נְקֵבָ֨ה מִן־ הַצֹּ֥אן כִּשְׂבָּ֛ה אֽוֹ־ שְׂעִירַ֥ת עִזִּ֖ים לְחַטָּ֑את וְכִפֶּ֥ר עָלָ֛יו הַכֹּהֵ֖ן מֵחַטָּאתֽוֹ׃ (מב) וְאִם־ לֹ֨א תַגִּ֣יע יָדוֹ֮ דֵּ֣י שֶׂה֒ וְהֵבִ֨יא אֶת־ אֲשָׁמ֜וֹ אֲשֶׁ֣ר חָטָ֗א שְׁתֵּ֥י תֹרִ֛ים אֽוֹ־ שְׁנֵ֥י בְנֵֽי־ יוֹנָ֖ה לַֽיהוָ֑ה אֶחָ֥ד לְחַטָּ֖את וְאֶחָ֥ד לְעֹלָֽה׃ (מג) וְהֵבִ֤יא אֹתָם֙ אֶל־ הַכֹּהֵ֔ן וְהִקְרִ֛יב אֶת־ אֲשֶׁ֥ר לַחַטָּ֖את רִאשׁוֹנָ֑ה וּמָלַ֧ק אֶת־ רֹאשׁ֛וֹ מִמּ֥וּל עָרְפּ֖וֹ וְלֹ֥א יַבְדִּֽיל׃ (מד) וְהִזָּ֞ה מִדַּ֤ם הַחַטָּאת֙ עַל־ קִ֣יר הַמִּזְבֵּ֔חַ וְהַנִּשְׁאָ֣ר בַּדָּ֔ם יִמָּצֵ֖ה אֶל־ יְס֣וֹד הַמִּזְבֵּ֑חַ חַטָּ֖את הֽוּא׃ (מה) וְאֶת־ הַשֵּׁנִ֛י יַעֲשֶׂ֥ה עֹלָ֖ה כַּמִּשְׁפָּ֑ט וְכִפֶּ֨ר עָלָ֧יו הַכֹּהֵ֛ן מֵחַטָּאת֥וֹ אֲשֶׁר־ חָטָ֖א וְנִסְלַ֥ח לֽוֹ׃ (מו) וְאִם־ לֹא֩ תַשִּׂ֨יג יָד֜וֹ לִשְׁתֵּ֣י תֹרִ֗ים אוֹ֮ לִשְׁנֵ֣י בְנֵי־ יוֹנָה֒ וְהֵבִ֨יא אֶת־ קָרְבָּנ֜וֹ אֲשֶׁ֣ר חָטָ֗א עֲשִׂירִ֧ת הָאֵפָ֛ה סֹ֖לֶת לְחַטָּ֑את לֹא־ יָשִׂ֨ים עָלֶ֜יהָ שֶׁ֗מֶן וְלֹא־ יִתֵּ֤ן עָלֶ֙יהָ֙ לְבֹנָ֔ה כִּ֥י חַטָּ֖את הִֽיא׃ (מז) וֶהֱבִיאָהּ֮ אֶל־ הַכֹּהֵן֒ וְקָמַ֣ץ הַכֹּהֵ֣ן ׀ מִ֠מֶּנָּה מְל֨וֹא קֻמְצ֜וֹ אֶת־ אַזְכָּרָתָה֙ וְהִקְטִ֣יר הַמִּזְבֵּ֔חָה עַ֖ל אִשֵּׁ֣י יְהוָ֑ה חַטָּ֖את הִֽוא׃ (מח) וְכִפֶּר֩ עָלָ֨יו הַכֹּהֵ֜ן עַל־ חַטָּאת֧וֹ אֲשֶׁר־ חָטָ֛א מֵֽאַחַ֥ת מֵאֵ֖לֶּה וְנִסְלַ֣ח ל֑וֹ וְהָיְתָ֥ה לַכֹּהֵ֖ן כַּמִּנְחָֽה׃ (מט) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (נ) נֶ֚פֶשׁ כִּֽי־ תִמְעֹ֣ל מַ֔עַל וְחָֽטְאָה֙ בִּשְׁגָגָ֔ה מִקָּדְשֵׁ֖י יְהוָ֑ה וְהֵבִיא֩ אֶת־ אֲשָׁמ֨וֹ לַֽיהוָ֜ה אַ֧יִל תָּמִ֣ים מִן־ הַצֹּ֗אן בְּעֶרְכְּךָ֛ כֶּֽסֶף־ שְׁקָלִ֥ים בְּשֶֽׁקֶל־ הַקֹּ֖דֶשׁ לְאָשָֽׁם׃ (נא) וְאֵ֣ת אֲשֶׁר֩ חָטָ֨א מִן־ הַקֹּ֜דֶשׁ יְשַׁלֵּ֗ם וְאֶת־ חֲמִֽישִׁתוֹ֙ יוֹסֵ֣ף עָלָ֔יו וְנָתַ֥ן אֹת֖וֹ לַכֹּהֵ֑ן וְהַכֹּהֵ֗ן יְכַפֵּ֥ר עָלָ֛יו בְּאֵ֥יל הָאָשָׁ֖ם וְנִסְלַ֥ח לֽוֹ׃ (נב) וְאִם־ נֶ֙פֶשׁ֙ כִּ֣י תֶֽחֱטָ֔א וְעָֽשְׂתָ֗ה אַחַת֙ מִכָּל־ מִצְוֺ֣ת יְהוָ֔ה אֲשֶׁ֖ר לֹ֣א תֵעָשֶׂ֑ינָה וְלֹֽא־ יָדַ֥ע וְאָשֵׁ֖ם וְנָשָׂ֥א עֲוֺנֽוֹ׃ (נג) וְ֠הֵבִיא אַ֣יִל תָּמִ֧ים מִן־ הַצֹּ֛אן בְּעֶרְכְּךָ֥ לְאָשָׁ֖ם אֶל־ הַכֹּהֵ֑ן וְכִפֶּר֩ עָלָ֨יו הַכֹּהֵ֜ן עַ֣ל שִׁגְגָת֧וֹ אֲשֶׁר־ שָׁגָ֛ג וְה֥וּא לֹֽא־ יָדַ֖ע וְנִסְלַ֥ח לֽוֹ׃ (נד) אָשָׁ֖ם ה֑וּא אָשֹׁ֥ם אָשַׁ֖ם לַיהוָֽה׃ (נה) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (נו) נֶ֚פֶשׁ כִּ֣י תֶחֱטָ֔א וּמָעֲלָ֥ה מַ֖עַל בַּיהוָ֑ה וְכִחֵ֨שׁ בַּעֲמִית֜וֹ בְּפִקָּד֗וֹן אֽוֹ־ בִתְשׂ֤וּמֶת יָד֙ א֣וֹ בְגָזֵ֔ל א֖וֹ עָשַׁ֥ק אֶת־ עֲמִיתֽוֹ׃ (נז) אֽוֹ־ מָצָ֧א אֲבֵדָ֛ה וְכִ֥חֶשׁ בָּ֖הּ וְנִשְׁבַּ֣ע עַל־ שָׁ֑קֶר עַל־ אַחַ֗ת מִכֹּ֛ל אֲשֶׁר־ יַעֲשֶׂ֥ה הָאָדָ֖ם לַחֲטֹ֥א בָהֵֽנָּה׃ (נח) וְהָיָה֮ כִּֽי־ יֶחֱטָ֣א וְאָשֵׁם֒ וְהֵשִׁ֨יב אֶת־ הַגְּזֵלָ֜ה אֲשֶׁ֣ר גָּזָ֗ל א֤וֹ אֶת־ הָעֹ֙שֶׁק֙ אֲשֶׁ֣ר עָשָׁ֔ק א֚וֹ אֶת־ הַפִּקָּד֔וֹן אֲשֶׁ֥ר הָפְקַ֖ד אִתּ֑וֹ א֥וֹ אֶת־ הָאֲבֵדָ֖ה אֲשֶׁ֥ר מָצָֽא׃ (נט) א֠וֹ מִכֹּ֞ל אֲשֶׁר־ יִשָּׁבַ֣ע עָלָיו֮ לַשֶּׁקֶר֒ וְשִׁלַּ֤ם אֹתוֹ֙ בְּרֹאשׁ֔וֹ וַחֲמִשִׁתָ֖יו יֹסֵ֣ף עָלָ֑יו לַאֲשֶׁ֨ר ה֥וּא ל֛וֹ יִתְּנֶ֖נּוּ בְּי֥וֹם אַשְׁמָתֽוֹ׃ (ס) וְאֶת־ אֲשָׁמ֥וֹ יָבִ֖יא לַיהוָ֑ה אַ֣יִל תָּמִ֧ים מִן־ הַצֹּ֛אן בְּעֶרְכְּךָ֥ לְאָשָׁ֖ם אֶל־ הַכֹּהֵֽן׃ (סא) וְכִפֶּ֨ר עָלָ֧יו הַכֹּהֵ֛ן לִפְנֵ֥י יְהוָ֖ה וְנִסְלַ֣ח ל֑וֹ עַל־ אַחַ֛ת מִכֹּ֥ל אֲשֶֽׁר־ יַעֲשֶׂ֖ה לְאַשְׁמָ֥ה בָֽהּ׃ (סב) וַיְדַבֵּ֥ר יְהוָ֖ה אֶל־ מֹשֶׁ֥ה לֵּאמֹֽר׃ (סג) צַ֤ו אֶֽת־ אַהֲרֹן֙ וְאֶת־ בָּנָ֣יו לֵאמֹ֔ר זֹ֥את תּוֹרַ֖ת הָעֹלָ֑ה הִ֣וא הָעֹלָ֡ה עַל֩ מוֹקְדָ֨ה עַל־ הַמִּזְבֵּ֤חַ כָּל־ הַלַּ֙יְלָה֙ עַד־ הַבֹּ֔קֶר וְאֵ֥שׁ הַמִּזְבֵּ֖חַ תּ֥וּקַד בּֽוֹ׃ (סד) וְלָבַ֨שׁ הַכֹּהֵ֜ן מִדּ֣וֹ בַ֗ד וּמִֽכְנְסֵי־ בַד֮ יִלְבַּ֣שׁ עַל־ בְּשָׂרוֹ֒ וְהֵרִ֣ים אֶת־ הַדֶּ֗שֶׁן אֲשֶׁ֨ר תֹּאכַ֥ל הָאֵ֛שׁ אֶת־ הָעֹלָ֖ה עַל־ הַמִּזְבֵּ֑חַ וְשָׂמ֕וֹ אֵ֖צֶל הַמִּזְבֵּֽחַ׃ (סה) וּפָשַׁט֙ אֶת־ בְּגָדָ֔יו וְלָבַ֖שׁ בְּגָדִ֣ים אֲחֵרִ֑ים וְהוֹצִ֤יא אֶת־ הַדֶּ֙שֶׁן֙ אֶל־ מִח֣וּץ לַֽמַּחֲנֶ֔ה אֶל־ מָק֖וֹם טָהֽוֹר׃ (סו) וְהָאֵ֨שׁ עַל־ הַמִּזְבֵּ֤חַ תּֽוּקַד־ בּוֹ֙ לֹ֣א תִכְבֶּ֔ה וּבִעֵ֨ר עָלֶ֧יהָ הַכֹּהֵ֛ן עֵצִ֖ים בַּבֹּ֣קֶר בַּבֹּ֑קֶר וְעָרַ֤ךְ עָלֶ֙יהָ֙ הָֽעֹלָ֔ה וְהִקְטִ֥יר עָלֶ֖יהָ חֶלְבֵ֥י הַשְּׁלָמִֽים׃ (סז) אֵ֗שׁ תָּמִ֛יד תּוּקַ֥ד עַל־ הַמִּזְבֵּ֖חַ לֹ֥א תִכְבֶֽה׃ (סח) וְזֹ֥את תּוֹרַ֖ת הַמִּנְחָ֑ה הַקְרֵ֨ב אֹתָ֤הּ בְּנֵֽי־ אַהֲרֹן֙ לִפְנֵ֣י יְהוָ֔ה אֶל־ פְּנֵ֖י הַמִּזְבֵּֽחַ׃ (סט) וְהֵרִ֨ים מִמֶּ֜נּוּ בְּקֻמְצ֗וֹ מִסֹּ֤לֶת הַמִּנְחָה֙ וּמִשַּׁמְנָ֔הּ וְאֵת֙ כָּל־ הַלְּבֹנָ֔ה אֲשֶׁ֖ר עַל־ הַמִּנְחָ֑ה וְהִקְטִ֣יר הַמִּזְבֵּ֗חַ רֵ֧יחַ נִיחֹ֛חַ אַזְכָּרָתָ֖הּ לַיהוָֽה׃ (ע) וְהַנּוֹתֶ֣רֶת מִמֶּ֔נָּה יֹאכְל֖וּ אַהֲרֹ֣ן וּבָנָ֑יו מַצּ֤וֹת תֵּֽאָכֵל֙ בְּמָק֣וֹם קָדֹ֔שׁ בַּחֲצַ֥ר אֹֽהֶל־ מוֹעֵ֖ד יֹאכְלֽוּהָ׃ (עא) לֹ֤א תֵאָפֶה֙ חָמֵ֔ץ חֶלְקָ֛ם נָתַ֥תִּי אֹתָ֖הּ מֵאִשָּׁ֑י קֹ֤דֶשׁ קָֽדָשִׁים֙ הִ֔וא כַּחַטָּ֖את וְכָאָשָֽׁם׃ (עב) כָּל־ זָכָ֞ר בִּבְנֵ֤י אַהֲרֹן֙ יֹֽאכֲלֶ֔נָּה חָק־ עוֹלָם֙ לְדֹרֹ֣תֵיכֶ֔ם מֵאִשֵּׁ֖י יְהוָ֑ה כֹּ֛ל אֲשֶׁר־ יִגַּ֥ע בָּהֶ֖ם יִקְדָּֽשׁ׃
