משלי סדר ב

פתיחה באתר · ספר משלי כולו ·

(א) יְהִֽי־ מְקוֹרְךָ֥ בָר֑וּךְ וּ֝שְׂמַ֗ח מֵאֵ֥שֶׁת נְעוּרֶֽךָ׃ (ב) אַיֶּ֥לֶת אֲהָבִ֗ים וְֽיַעֲלַ֫ת־ חֵ֥ן דַּ֭דֶּיהָ יְרַוֻּ֣ךָ בְכָל־ עֵ֑ת בְּ֝אַהֲבָתָ֗הּ תִּשְׁגֶּ֥ה תָמִֽיד׃ (ג) וְלָ֤מָּה תִשְׁגֶּ֣ה בְנִ֣י בְזָרָ֑ה וּ֝תְחַבֵּ֗ק חֵ֣ק נָכְרִיָּֽה׃ (ד) כִּ֤י נֹ֨כַח ׀ עֵינֵ֣י יְ֭הוָה דַּרְכֵי־ אִ֑ישׁ וְֽכָל־ מַעְגְּלֹתָ֥יו מְפַלֵּֽס׃ (ה) עַֽווֹנוֹתָ֗יו יִלְכְּדֻנ֥וֹ אֶת־ הָרָשָׁ֑ע וּבְחַבְלֵ֥י חַ֝טָּאת֗וֹ יִתָּמֵֽךְ׃ (ו) ה֗וּא יָ֭מוּת בְּאֵ֣ין מוּסָ֑ר וּבְרֹ֖ב אִוַּלְתּ֣וֹ יִשְׁגֶּֽה׃ (ז) בְּ֭נִי אִם־ עָרַ֣בְתָּ לְרֵעֶ֑ךָ תָּקַ֖עְתָּ לַזָּ֣ר כַּפֶּֽיךָ׃ (ח) נוֹקַ֥שְׁתָּ בְאִמְרֵי־ פִ֑יךָ נִ֝לְכַּ֗דְתָּ בְּאִמְרֵי־ פִֽיךָ׃ (ט) עֲשֵׂ֨ה זֹ֥את אֵפ֪וֹא ׀ בְּנִ֡י וְֽהִנָּצֵ֗ל כִּ֘י בָ֤אתָ בְכַף־ רֵעֶ֑ךָ לֵ֥ךְ הִ֝תְרַפֵּ֗ס וּרְהַ֥ב רֵעֶֽיךָ׃ (י) אַל־ תִּתֵּ֣ן שֵׁנָ֣ה לְעֵינֶ֑יךָ וּ֝תְנוּמָ֗ה לְעַפְעַפֶּֽיךָ׃ (יא) הִ֭נָּצֵל כִּצְבִ֣י מִיָּ֑ד וּ֝כְצִפּ֗וֹר מִיַּ֥ד יָקֽוּשׁ׃ (יב) לֵֽךְ־ אֶל־ נְמָלָ֥ה עָצֵ֑ל רְאֵ֖ה דְרָכֶ֣יהָ וַחֲכָֽם׃ (יג) אֲשֶׁ֖ר אֵֽין־ לָ֥הּ קָצִ֗ין שֹׁטֵ֥ר וּמֹשֵֽׁל׃ (יד) תָּכִ֣ין בַּקַּ֣יִץ לַחְמָ֑הּ אָגְרָ֥ה בַ֝קָּצִ֗יר מַאֲכָלָֽהּ׃ (טו) עַד־ מָתַ֖י עָצֵ֥ל ׀ תִּשְׁכָּ֑ב מָ֝תַ֗י תָּק֥וּם מִשְּׁנָתֶֽךָ׃ (טז) מְעַ֣ט שֵׁ֭נוֹת מְעַ֣ט תְּנוּמ֑וֹת מְעַ֓ט ׀ חִבֻּ֖ק יָדַ֣יִם לִשְׁכָּֽב׃ (יז) וּבָֽא־ כִמְהַלֵּ֥ךְ רֵאשֶׁ֑ךָ וּ֝מַחְסֹֽרְךָ֗ כְּאִ֣ישׁ מָגֵֽן׃ (יח) אָדָ֣ם בְּ֭לִיַּעַל אִ֣ישׁ אָ֑וֶן ה֝וֹלֵ֗ךְ עִקְּשׁ֥וּת פֶּֽה׃ (יט) קֹרֵ֣ץ בְּ֭עֵינָו מֹלֵ֣ל בְּרַגְלָ֑ו מֹ֝רֶ֗ה בְּאֶצְבְּעֹתָֽיו׃ (כ) תַּֽהְפֻּכ֨וֹת ׀ בְּלִבּ֗וֹ חֹרֵ֣שׁ רָ֣ע בְּכָל־ עֵ֑ת (מדנים) מִדְיָנִ֥ים יְשַׁלֵּֽחַ׃ (כא) עַל־ כֵּ֗ן פִּ֭תְאֹם יָב֣וֹא אֵיד֑וֹ פֶּ֥תַע יִ֝שָּׁבֵ֗ר וְאֵ֣ין מַרְפֵּֽא׃ (כב) שֶׁשׁ־ הֵ֭נָּה שָׂנֵ֣א יְהוָ֑ה וְ֝שֶׁ֗בַע (תועבות) תּוֹעֲבַ֥ת נַפְשֽׁוֹ׃ (כג) עֵינַ֣יִם רָ֭מוֹת לְשׁ֣וֹן שָׁ֑קֶר וְ֝יָדַ֗יִם שֹׁפְכ֥וֹת דָּם־ נָקִֽי׃ (כד) לֵ֗ב חֹ֭רֵשׁ מַחְשְׁב֣וֹת אָ֑וֶן רַגְלַ֥יִם מְ֝מַהֲר֗וֹת לָר֥וּץ לָֽרָעָה׃ (כה) יָפִ֣יחַ כְּ֭זָבִים עֵ֣ד שָׁ֑קֶר וּמְשַׁלֵּ֥חַ מְ֝דָנִ֗ים בֵּ֣ין אַחִֽים׃ (כו) נְצֹ֣ר בְּ֭נִי מִצְוַ֣ת אָבִ֑יךָ וְאַל־ תִּ֝טֹּ֗שׁ תּוֹרַ֥ת אִמֶּֽךָ׃ (כז) קָשְׁרֵ֣ם עַל־ לִבְּךָ֣ תָמִ֑יד עָ֝נְדֵ֗ם עַל־ גַּרְגְּרֹתֶֽךָ׃ (כח) בְּהִתְהַלֶּכְךָ֨ ׀ תַּנְחֶ֬ה אֹתָ֗ךְ בְּֽ֭שָׁכְבְּךָ תִּשְׁמֹ֣ר עָלֶ֑יךָ וַ֝הֲקִיצ֗וֹתָ הִ֣יא תְשִׂיחֶֽךָ׃ (כט) כִּ֤י נֵ֣ר מִ֭צְוָה וְת֣וֹרָה א֑וֹר וְדֶ֥רֶךְ חַ֝יִּ֗ים תּוֹכְח֥וֹת מוּסָֽר׃ (ל) לִ֭שְׁמָרְךָ מֵאֵ֣שֶׁת רָ֑ע מֵֽ֝חֶלְקַ֗ת לָשׁ֥וֹן נָכְרִיָּֽה׃ (לא) אַל־ תַּחְמֹ֣ד יָ֭פְיָהּ בִּלְבָבֶ֑ךָ וְאַל־ תִּ֝קָּֽחֲךָ֗ בְּעַפְעַפֶּֽיהָ׃ (לב) כִּ֤י בְעַד־ אִשָּׁ֥ה זוֹנָ֗ה עַֽד־ כִּכַּ֫ר לָ֥חֶם וְאֵ֥שֶׁת אִ֑ישׁ נֶ֖פֶשׁ יְקָרָ֣ה תָצֽוּד׃ (לג) הֲיַחְתֶּ֤ה אִ֓ישׁ אֵ֬שׁ בְּחֵיק֑וֹ וּ֝בְגָדָ֗יו לֹ֣א תִשָּׂרַֽפְנָה׃ (לד) אִם־ יְהַלֵּ֣ךְ אִ֭ישׁ עַל־ הַגֶּחָלִ֑ים וְ֝רַגְלָ֗יו לֹ֣א תִכָּוֶֽינָה׃ (לה) כֵּ֗ן הַ֭בָּא אֶל־ אֵ֣שֶׁת רֵעֵ֑הוּ לֹ֥א יִ֝נָּקֶ֗ה כָּֽל־ הַנֹּגֵ֥עַ בָּֽהּ׃ (לו) לֹא־ יָב֣וּזוּ לַ֭גַּנָּב כִּ֣י יִגְנ֑וֹב לְמַלֵּ֥א נַ֝פְשׁ֗וֹ כִּ֣י יִרְעָֽב׃ (לז) וְ֭נִמְצָא יְשַׁלֵּ֣ם שִׁבְעָתָ֑יִם אֶת־ כָּל־ ה֖וֹן בֵּית֣וֹ יִתֵּֽן׃ (לח) נֹאֵ֣ף אִשָּׁ֣ה חֲסַר־ לֵ֑ב מַֽשְׁחִ֥ית נַ֝פְשׁ֗וֹ ה֣וּא יַעֲשֶֽׂנָּה׃ (לט) נֶֽגַע־ וְקָל֥וֹן יִמְצָ֑א וְ֝חֶרְפָּת֗וֹ לֹ֣א תִמָּחֶֽה׃ (מ) כִּֽי־ קִנְאָ֥ה חֲמַת־ גָּ֑בֶר וְלֹֽא־ יַ֝חְמ֗וֹל בְּי֣וֹם נָקָֽם׃ (מא) לֹא־ יִ֭שָּׂא פְּנֵ֣י כָל־ כֹּ֑פֶר וְלֹֽא־ יֹ֝אבֶ֗ה כִּ֣י תַרְבֶּה־ שֹֽׁחַד׃ (מב) בְּ֭נִי שְׁמֹ֣ר אֲמָרָ֑י וּ֝מִצְוֺתַ֗י תִּצְפֹּ֥ן אִתָּֽךְ׃ (מג) שְׁמֹ֣ר מִצְוֺתַ֣י וֶחְיֵ֑ה וְ֝תוֹרָתִ֗י כְּאִישׁ֥וֹן עֵינֶֽיךָ׃ (מד) קָשְׁרֵ֥ם עַל־ אֶצְבְּעֹתֶ֑יךָ כָּ֝תְבֵ֗ם עַל־ ל֥וּחַ לִבֶּֽךָ׃ (מה) אֱמֹ֣ר לַֽ֭חָכְמָה אֲחֹ֣תִי אָ֑תְּ וּ֝מֹדָ֗ע לַבִּינָ֥ה תִקְרָֽא׃ (מו) לִ֭שְׁמָרְךָ מֵאִשָּׁ֣ה זָרָ֑ה מִ֝נָּכְרִיָּ֗ה אֲמָרֶ֥יהָ הֶחֱלִֽיקָה׃ (מז) כִּ֭י בְּחַלּ֣וֹן בֵּיתִ֑י בְּעַ֖ד אֶשְׁנַבִּ֣י נִשְׁקָֽפְתִּי׃ (מח) וָאֵ֤רֶא בַפְּתָאיִ֗ם אָ֘בִ֤ינָה בַבָּנִ֗ים נַ֣עַר חֲסַר־ לֵֽב׃ (מט) עֹבֵ֣ר בַּ֭שּׁוּק אֵ֣צֶל פִּנָּ֑הּ וְדֶ֖רֶךְ בֵּיתָ֣הּ יִצְעָֽד׃ (נ) בְּנֶֽשֶׁף־ בְּעֶ֥רֶב י֑וֹם בְּאִישׁ֥וֹן לַ֝֗יְלָה וַאֲפֵלָֽה׃ (נא) וְהִנֵּ֣ה אִ֭שָּׁה לִקְרָאת֑וֹ שִׁ֥ית ז֝וֹנָ֗ה וּנְצֻ֥רַת לֵֽב׃ (נב) הֹמִיָּ֣ה הִ֣יא וְסֹרָ֑רֶת בְּ֝בֵיתָ֗הּ לֹא־ יִשְׁכְּנ֥וּ רַגְלֶֽיהָ׃ (נג) פַּ֤עַם ׀ בַּח֗וּץ פַּ֥עַם בָּרְחֹב֑וֹת וְאֵ֖צֶל כָּל־ פִּנָּ֣ה תֶאֱרֹֽב׃ (נד) וְהֶחֱזִ֣יקָה בּ֭וֹ וְנָ֣שְׁקָה־ לּ֑וֹ הֵעֵ֥זָה פָ֝נֶ֗יהָ וַתֹּ֣אמַר לֽוֹ׃ (נה) זִבְחֵ֣י שְׁלָמִ֣ים עָלָ֑י הַ֝יּ֗וֹם שִׁלַּ֥מְתִּי נְדָרָֽי׃ (נו) עַל־ כֵּ֭ן יָצָ֣אתִי לִקְרָאתֶ֑ךָ לְשַׁחֵ֥ר פָּ֝נֶ֗יךָ וָאֶמְצָאֶֽךָּ׃ (נז) מַ֭רְבַדִּים רָבַ֣דְתִּי עַרְשִׂ֑י חֲ֝טֻב֗וֹת אֵט֥וּן מִצְרָֽיִם׃ (נח) נַ֥פְתִּי מִשְׁכָּבִ֑י מֹ֥ר אֲ֝הָלִ֗ים וְקִנָּמֽוֹן׃ (נט) לְכָ֤ה נִרְוֶ֣ה דֹ֭דִים עַד־ הַבֹּ֑קֶר נִ֝תְעַלְּסָ֗ה בָּאֳהָבִֽים׃ (ס) כִּ֤י אֵ֣ין הָאִ֣ישׁ בְּבֵית֑וֹ הָ֝לַ֗ךְ בְּדֶ֣רֶךְ מֵרָחֽוֹק׃ (סא) צְֽרוֹר־ הַ֭כֶּסֶף לָקַ֣ח בְּיָד֑וֹ לְי֥וֹם הַ֝כֵּ֗סֶא יָבֹ֥א בֵיתֽוֹ׃ (סב) הִ֭טַּתּוּ בְּרֹ֣ב לִקְחָ֑הּ בְּחֵ֥לֶק שְׂ֝פָתֶ֗יהָ תַּדִּיחֶֽנּוּ׃ (סג) ה֤וֹלֵ֥ךְ אַחֲרֶ֗יהָ פִּ֫תְאֹ֥ם כְּ֭שׁוֹר אֶל־ טָ֣בַח יָב֑וֹא וּ֝כְעֶ֗כֶס אֶל־ מוּסַ֥ר אֱוִֽיל׃ (סד) עַ֤ד יְפַלַּ֪ח חֵ֡ץ כְּֽבֵד֗וֹ כְּמַהֵ֣ר צִפּ֣וֹר אֶל־ פָּ֑ח וְלֹֽא־ יָ֝דַ֗ע כִּֽי־ בְנַפְשׁ֥וֹ הֽוּא׃ (סה) וְעַתָּ֣ה בָ֭נִים שִׁמְעוּ־ לִ֑י וְ֝הַקְשִׁ֗יבוּ לְאִמְרֵי־ פִֽי׃ (סו) אַל־ יֵ֣שְׂטְ אֶל־ דְּרָכֶ֣יהָ לִבֶּ֑ךָ אַל־ תֵּ֝תַע בִּנְתִיבוֹתֶֽיהָ׃ (סז) כִּֽי־ רַבִּ֣ים חֲלָלִ֣ים הִפִּ֑ילָה וַ֝עֲצֻמִ֗ים כָּל־ הֲרֻגֶֽיהָ׃ (סח) דַּרְכֵ֣י שְׁא֣וֹל בֵּיתָ֑הּ יֹ֝רְד֗וֹת אֶל־ חַדְרֵי־ מָֽוֶת׃ (סט) הֲלֹֽא־ חָכְמָ֥ה תִקְרָ֑א וּ֝תְבוּנָ֗ה תִּתֵּ֥ן קוֹלָֽהּ׃ (ע) בְּרֹאשׁ־ מְרוֹמִ֥ים עֲלֵי־ דָ֑רֶךְ בֵּ֖ית נְתִיב֣וֹת נִצָּֽבָה׃ (עא) לְיַד־ שְׁעָרִ֥ים לְפִי־ קָ֑רֶת מְב֖וֹא פְתָחִ֣ים תָּרֹֽנָּה׃ (עב) אֲלֵיכֶ֣ם אִישִׁ֣ים אֶקְרָ֑א וְ֝קוֹלִ֗י אֶל־ בְּנֵ֥י אָדָֽם׃ (עג) הָבִ֣ינוּ פְתָאיִ֣ם עָרְמָ֑ה וּ֝כְסִילִ֗ים הָבִ֥ינוּ לֵֽב׃ (עד) שִׁ֭מְעוּ כִּֽי־ נְגִידִ֣ים אֲדַבֵּ֑ר וּמִפְתַּ֥ח שְׂ֝פָתַ֗י מֵישָׁרִֽים׃ (עה) כִּֽי־ אֱ֭מֶת יֶהְגֶּ֣ה חִכִּ֑י וְתוֹעֲבַ֖ת שְׂפָתַ֣י רֶֽשַׁע׃ (עו) בְּצֶ֥דֶק כָּל־ אִמְרֵי־ פִ֑י אֵ֥ין בָּ֝הֶ֗ם נִפְתָּ֥ל וְעִקֵּֽשׁ׃ (עז) כֻּלָּ֣ם נְ֭כֹחִים לַמֵּבִ֑ין וִֽ֝ישָׁרִ֗ים לְמֹ֣צְאֵי דָֽעַת׃ (עח) קְחֽוּ־ מוּסָרִ֥י וְאַל־ כָּ֑סֶף וְ֝דַ֗עַת מֵחָר֥וּץ נִבְחָֽר׃ (עט) כִּֽי־ טוֹבָ֣ה חָ֭כְמָה מִפְּנִינִ֑ים וְכָל־ חֲ֝פָצִ֗ים לֹ֣א יִֽשְׁווּ־ בָֽהּ׃ (פ) אֲ‍ֽנִי־ חָ֭כְמָה שָׁכַ֣נְתִּי עָרְמָ֑ה וְדַ֖עַת מְזִמּ֣וֹת אֶמְצָֽא׃ (פא) יִֽרְאַ֣ת יְהוָה֮ שְֽׂנֹ֫את רָ֥ע גֵּ֘אָ֤ה וְגָא֨וֹן ׀ וְדֶ֣רֶךְ רָ֭ע וּפִ֨י תַהְפֻּכ֬וֹת שָׂנֵֽאתִי׃ (פב) לִֽי־ עֵ֭צָה וְתוּשִׁיָּ֑ה אֲנִ֥י בִ֝ינָ֗ה לִ֣י גְבוּרָֽה׃ (פג) בִּ֭י מְלָכִ֣ים יִמְלֹ֑כוּ וְ֝רוֹזְנִ֗ים יְחֹ֣קְקוּ צֶֽדֶק׃ (פד) בִּ֭י שָׂרִ֣ים יָשֹׂ֑רוּ וּ֝נְדִיבִ֗ים כָּל־ שֹׁ֥פְטֵי צֶֽדֶק׃ (פה) אֲ֭נִי (אהביה) אֹהֲבַ֣י אֵהָ֑ב וּ֝מְשַׁחֲרַ֗י יִמְצָאֻֽנְנִי׃ (פו) עֹֽשֶׁר־ וְכָב֥וֹד אִתִּ֑י ה֥וֹן עָ֝תֵ֗ק וּצְדָקָֽה׃ (פז) ט֣וֹב פִּ֭רְיִי מֵחָר֣וּץ וּמִפָּ֑ז וּ֝תְבוּאָתִ֗י מִכֶּ֥סֶף נִבְחָֽר׃ (פח) בְּאֹֽרַח־ צְדָקָ֥ה אֲהַלֵּ֑ך בְּ֝ת֗וֹךְ נְתִיב֥וֹת מִשְׁפָּֽט׃ (פט) לְהַנְחִ֖יל אֹהֲבַ֥י ׀ יֵ֑שׁ וְאֹצְרֹ֖תֵיהֶ֣ם אֲמַלֵּֽא׃ (צ) יְֽהוָ֗ה קָ֭נָנִי רֵאשִׁ֣ית דַּרְכּ֑וֹ קֶ֖דֶם מִפְעָלָ֣יו מֵאָֽז׃ (צא) מֵ֭עוֹלָם נִסַּ֥כְתִּי מֵרֹ֗אשׁ מִקַּדְמֵי־ אָֽרֶץ׃ (צב) בְּאֵין־ תְּהֹמ֥וֹת חוֹלָ֑לְתִּי בְּאֵ֥ין מַ֝עְיָנ֗וֹת נִכְבַּדֵּי־ מָֽיִם׃ (צג) בְּטֶ֣רֶם הָרִ֣ים הָטְבָּ֑עוּ לִפְנֵ֖י גְבָע֣וֹת חוֹלָֽלְתִּי׃ (צד) עַד־ לֹ֣א עָ֭שָׂה אֶ֣רֶץ וְחוּצ֑וֹת וְ֝רֹ֗אשׁ עָפְר֥וֹת תֵּבֵֽל׃ (צה) בַּהֲכִינ֣וֹ שָׁ֭מַיִם שָׁ֣ם אָ֑נִי בְּח֥וּקוֹ ח֝֗וּג עַל־ פְּנֵ֥י תְהֽוֹם׃ (צו) בְּאַמְּצ֣וֹ שְׁחָקִ֣ים מִמָּ֑עַל בַּ֝עֲז֗וֹז עִינ֥וֹת תְּהוֹם׃ (צז) בְּשׂ֘וּמ֤וֹ לַיָּ֨ם ׀ חֻקּ֗וֹ וּ֭מַיִם לֹ֣א יַֽעַבְרוּ־ פִ֑יו בְּ֝חוּק֗וֹ מ֣וֹסְדֵי אָֽרֶץ׃ (צח) וָֽאֶהְיֶ֥ה אֶצְל֗וֹ אָ֫מ֥וֹן וָֽאֶהְיֶ֣ה שַׁ֭עֲשֻׁעִים י֤וֹם ׀ י֑וֹם מְשַׂחֶ֖קֶת לְפָנָ֣יו בְּכָל־ עֵֽת׃ (צט) מְ֭שַׂחֶקֶת בְּתֵבֵ֣ל אַרְצ֑וֹ וְ֝שַׁעֲשֻׁעַ֗י אֶת־ בְּנֵ֥י אָדָֽם׃ (ק) וְעַתָּ֣ה בָ֭נִים שִׁמְעוּ־ לִ֑י וְ֝אַשְׁרֵ֗י דְּרָכַ֥י יִשְׁמֹֽרוּ׃ (קא) שִׁמְע֖וּ מוּסָ֥ר וַחֲכָ֗מוּ וְאַל־ תִּפְרָֽעוּ׃ (קב) אַ֥שְֽׁרֵי אָדָם֮ שֹׁמֵ֪עַֽ֫ לִ֥י לִשְׁקֹ֣ד עַל־ דַּ֭לְתֹתַי י֤וֹם ׀ י֑וֹם לִ֝שְׁמֹ֗ר מְזוּזֹ֥ת פְּתָחָֽי׃ (קג) כִּ֣י מֹ֭צְאִי (מצאי) מָצָ֣א חַיִּ֑ים וַיָּ֥פֶק רָ֝צ֗וֹן מֵיְהוָֽה׃ (קד) וְֽ֭חֹטְאִי חֹמֵ֣ס נַפְשׁ֑וֹ כָּל־ מְ֝שַׂנְאַ֗י אָ֣הֲבוּ מָֽוֶת׃ (קה) חָ֭כְמוֹת בָּנְתָ֣ה בֵיתָ֑הּ חָצְבָ֖ה עַמּוּדֶ֣יהָ שִׁבְעָֽה׃ (קו) טָבְחָ֣ה טִ֭בְחָהּ מָסְכָ֣ה יֵינָ֑הּ אַ֝֗ף עָֽרְכָ֥ה שֻׁלְחָנָֽהּ׃ (קז) שָֽׁלְחָ֣ה נַעֲרֹתֶ֣יהָ תִקְרָ֑א עַל־ גַּ֝פֵּ֗י מְרֹ֣מֵי קָֽרֶת׃ (קח) מִי־ פֶ֭תִי יָסֻ֣ר הֵ֑נָּה חֲסַר־ לֵ֝֗ב אָ֣מְרָה לּֽוֹ׃ (קט) לְ֭כוּ לַחֲמ֣וּ בְֽלַחֲמִ֑י וּ֝שְׁת֗וּ בְּיַ֣יִן מָסָֽכְתִּי׃ (קי) עִזְב֣וּ פְתָאיִ֣ם וִֽחְי֑וּ וְ֝אִשְׁר֗וּ בְּדֶ֣רֶךְ בִּינָֽה׃ (קיא) יֹ֤סֵ֨ר ׀ לֵ֗ץ לֹקֵ֣חַֽ ל֣וֹ קָל֑וֹן וּמוֹכִ֖יחַ לְרָשָׁ֣ע מוּמֽוֹ׃ (קיב) אַל־ תּ֣וֹכַח לֵ֭ץ פֶּן־ יִשְׂנָאֶ֑ךָּ הוֹכַ֥ח לְ֝חָכָ֗ם וְיֶאֱהָבֶֽךָּ׃ (קיג) תֵּ֣ן לְ֭חָכָם וְיֶחְכַּם־ ע֑וֹד הוֹדַ֥ע לְ֝צַדִּ֗יק וְי֣וֹסֶף לֶֽקַח׃ (קיד) תְּחִלַּ֣ת חָ֭כְמָה יִרְאַ֣ת יְהוָ֑ה וְדַ֖עַת קְדֹשִׁ֣ים בִּינָֽה׃
