משלי סדר ח

פתיחה באתר · ספר משלי כולו ·

(א) נָגִ֗יד חֲסַ֣ר תְּ֭בוּנוֹת וְרַ֥ב מַעֲשַׁקּ֑וֹת (שנאי) שֹׂ֥נֵא בֶ֝֗צַע יַאֲרִ֥יךְ יָמִֽים׃ (ב) אָ֭דָם עָשֻׁ֣ק בְּדַם־ נָפֶשׁ עַד־ בּ֥וֹר יָ֝נ֗וּס אַל־ יִתְמְכוּ־ בֽוֹ׃ (ג) הוֹלֵ֣ךְ תָּ֭מִים יִוָּשֵׁ֑עַ וְנֶעְקַ֥שׁ דְּ֝רָכַ֗יִם יִפּ֥וֹל בְּאֶחָֽת׃ (ד) עֹבֵ֣ד אַ֭דְמָתוֹ יִֽשְׂבַּֽע־ לָ֑חֶם וּמְרַדֵּ֥ף רֵ֝קִ֗ים יִֽשְׂבַּֽע־ רִֽישׁ׃ (ה) אִ֣ישׁ אֱ֭מוּנוֹת רַב־ בְּרָכ֑וֹת וְאָ֥ץ לְ֝הַעֲשִׁ֗יר לֹ֣א יִנָּקֶֽה׃ (ו) הַֽכֵּר־ פָּנִ֥ים לֹא־ ט֑וֹב וְעַל־ פַּת־ לֶ֝֗חֶם יִפְשַׁע־ גָּֽבֶר׃ (ז) נִֽבֳהָ֥ל לַה֗וֹן אִ֭ישׁ רַ֣ע עָ֑יִן וְלֹֽא־ יֵ֝דַע כִּי־ חֶ֥סֶר יְבֹאֶֽנּוּ׃ (ח) מ֘וֹכִ֤יחַ אָדָ֣ם אַ֭חֲרַי חֵ֣ן יִמְצָ֑א מִֽמַּחֲלִ֥יק לָשֽׁוֹן׃ (ט) גּוֹזֵ֤ל ׀ אָ֘בִ֤יו וְאִמּ֗וֹ וְאֹמֵ֥ר אֵֽין־ פָּ֑שַׁע חָבֵ֥ר ה֝֗וּא לְאִ֣ישׁ מַשְׁחִֽית׃ (י) רְחַב־ נֶ֭פֶשׁ יְגָרֶ֣ה מָד֑וֹן וּבוֹטֵ֖חַ עַל־ יְהוָ֣ה יְדֻשָּֽׁן׃ (יא) בּוֹטֵ֣חַ בְּ֭לִבּוֹ ה֣וּא כְסִ֑יל וְהוֹלֵ֥ךְ בְּ֝חָכְמָ֗ה ה֣וּא יִמָּלֵֽט׃ (יב) נוֹתֵ֣ן לָ֭רָשׁ אֵ֣ין מַחְס֑וֹר וּמַעְלִ֥ים עֵ֝ינָ֗יו רַב־ מְאֵרֽוֹת׃ (יג) בְּק֣וּם רְ֭שָׁעִים יִסָּתֵ֣ר אָדָ֑ם וּ֝בְאָבְדָ֗ם יִרְבּ֥וּ צַדִּיקִֽים׃ (יד) אִ֣ישׁ תּ֭וֹכָחוֹת מַקְשֶׁה־ עֹ֑רֶף פֶּ֥תַע יִ֝שָּׁבֵ֗ר וְאֵ֣ין מַרְפֵּֽא׃ (טו) בִּרְב֣וֹת צַ֭דִּיקִים יִשְׂמַ֣ח הָעָ֑ם וּבִמְשֹׁ֥ל רָ֝שָׁ֗ע יֵאָ֥נַֽח עָֽם׃ (טז) אִֽישׁ־ אֹהֵ֣ב חָ֭כְמָה יְשַׂמַּ֣ח אָבִ֑יו וְרֹעֶ֥ה ז֝וֹנ֗וֹת יְאַבֶּד־ הֽוֹן׃ (יז) מֶ֗לֶךְ בְּ֭מִשְׁפָּט יַעֲמִ֣יד אָ֑רֶץ וְאִ֖ישׁ תְּרוּמ֣וֹת יֶֽהֶרְסֶֽנָּה׃ (יח) גֶּ֭בֶר מַחֲלִ֣יק עַל־ רֵעֵ֑הוּ רֶ֝֗שֶׁת פּוֹרֵ֥שׂ עַל־ פְּעָמָֽיו׃ (יט) בְּפֶ֤שַֽׁע אִ֣ישׁ רָ֣ע מוֹקֵ֑שׁ וְ֝צַדִּ֗יק יָר֥וּן וְשָׂמֵֽחַ׃ (כ) יֹדֵ֣עַ צַ֭דִּיק דִּ֣ין דַּלִּ֑ים רָ֝שָׁ֗ע לֹא־ יָבִ֥ין דָּֽעַת׃ (כא) אַנְשֵׁ֣י לָ֭צוֹן יָפִ֣יחוּ קִרְיָ֑ה וַ֝חֲכָמִ֗ים יָשִׁ֥יבוּ אָֽף׃ (כב) אִֽישׁ־ חָכָ֗ם נִ֭שְׁפָּט אֶת־ אִ֣ישׁ אֱוִ֑יל וְרָגַ֥ז וְ֝שָׂחַ֗ק וְאֵ֣ין נָֽחַת׃ (כג) אַנְשֵׁ֣י דָ֭מִים יִשְׂנְאוּ־ תָ֑ם וִֽ֝ישָׁרִ֗ים יְבַקְשׁ֥וּ נַפְשֽׁוֹ׃ (כד) כָּל־ ר֭וּחוֹ יוֹצִ֣יא כְסִ֑יל וְ֝חָכָ֗ם בְּאָח֥וֹר יְשַׁבְּחֶֽנָּה׃ (כה) מֹ֭שֵׁל מַקְשִׁ֣יב עַל־ דְּבַר־ שָׁ֑קֶר כָּֽל־ מְשָׁרְתָ֥יו רְשָׁעִֽים׃ (כו) רָ֤שׁ וְאִ֣ישׁ תְּכָכִ֣ים נִפְגָּ֑שׁוּ מֵ֤אִיר־ עֵינֵ֖י שְׁנֵיהֶ֣ם יְהוָֽה׃ (כז) מֶ֤לֶךְ שׁוֹפֵ֣ט בֶּֽאֱמֶ֣ת דַּלִּ֑ים כִּ֝סְא֗וֹ לָעַ֥ד יִכּֽוֹן׃ (כח) שֵׁ֣בֶט וְ֭תוֹכַחַת יִתֵּ֣ן חָכְמָ֑ה וְנַ֥עַר מְ֝שֻׁלָּ֗ח מֵבִ֥ישׁ אִמּֽוֹ׃ (כט) בִּרְב֣וֹת רְ֭שָׁעִים יִרְבֶּה־ פָּ֑שַׁע וְ֝צַדִּיקִ֗ים בְּֽמַפַּלְתָּ֥ם יִרְאֽוּ׃ (ל) יַסֵּ֣ר בִּ֭נְךָ וִֽינִיחֶ֑ךָ וְיִתֵּ֖ן מַעֲדַנִּ֣ים לְנַפְשֶֽׁךָ׃ (לא) בְּאֵ֣ין חָ֭זוֹן יִפָּ֣רַֽע עָ֑ם וְשֹׁמֵ֖ר תּוֹרָ֣ה אַשְׁרֵֽהוּ׃ (לב) בִּ֭דְבָרִים לֹא־ יִוָּ֣סֶר עָ֑בֶד כִּֽי־ יָ֝בִ֗ין וְאֵ֣ין מַעֲנֶֽה׃ (לג) חָזִ֗יתָ אִ֭ישׁ אָ֣ץ בִּדְבָרָ֑יו תִּקְוָ֖ה לִכְסִ֣יל מִמֶּֽנּוּ׃ (לד) מְפַנֵּ֣ק מִנֹּ֣עַר עַבְדּ֑וֹ וְ֝אַחֲרִית֗וֹ יִהְיֶ֥ה מָנֽוֹן׃ (לה) אִֽישׁ־ אַ֭ף יְגָרֶ֣ה מָד֑וֹן וּבַ֖עַל חֵמָ֣ה רַב־ פָּֽשַׁע׃ (לו) גַּאֲוַ֣ת אָ֭דָם תַּשְׁפִּילֶ֑נּוּ וּשְׁפַל־ ר֝֗וּחַ יִתְמֹ֥ךְ כָּבֽוֹד׃ (לז) חוֹלֵ֣ק עִם־ גַּ֭נָּב שׂוֹנֵ֣א נַפְשׁ֑וֹ אָלָ֥ה יִ֝שְׁמַ֗ע וְלֹ֣א יַגִּֽיד׃ (לח) חֶרְדַּ֣ת אָ֭דָם יִתֵּ֣ן מוֹקֵ֑שׁ וּבוֹטֵ֖חַ בַּיהוָ֣ה יְשֻׂגָּֽב׃ (לט) רַ֭בִּים מְבַקְשִׁ֣ים פְּנֵי־ מוֹשֵׁ֑ל וּ֝מֵיְהוָ֗ה מִשְׁפַּט־ אִֽישׁ׃ (מ) תּוֹעֲבַ֣ת צַ֭דִּיקִים אִ֣ישׁ עָ֑וֶל וְתוֹעֲבַ֖ת רָשָׁ֣ע יְשַׁר־ דָּֽרֶךְ׃ (מא) דִּבְרֵ֤י ׀ אָג֥וּר בִּן־ יָקֶ֗ה הַמַּ֫שָּׂ֥א נְאֻ֣ם הַ֭גֶּבֶר לְאִֽיתִיאֵ֑ל לְאִ֖יתִיאֵ֣ל וְאֻכָֽל׃ (מב) כִּ֤י בַ֣עַר אָנֹכִ֣י מֵאִ֑ישׁ וְלֹֽא־ בִינַ֖ת אָדָ֣ם לִֽי׃ (מג) וְלֹֽא־ לָמַ֥דְתִּי חָכְמָ֑ה וְדַ֖עַת קְדֹשִׁ֣ים אֵדָֽע׃ (מד) מִ֤י עָלָֽה־ שָׁמַ֨יִם ׀ וַיֵּרַ֡ד מִ֤י אָֽסַף־ ר֨וּחַ ׀ בְּחָפְנָ֡יו מִ֤י צָֽרַר־ מַ֨יִם ׀ בַּשִּׂמְלָ֗ה מִ֭י הֵקִ֣ים כָּל־ אַפְסֵי־ אָ֑רֶץ מַה־ שְּׁמ֥וֹ וּמַֽה־ שֶּׁם־ בְּ֝נ֗וֹ כִּ֣י תֵדָֽע׃ (מה) כָּל־ אִמְרַ֣ת אֱל֣וֹהַּ צְרוּפָ֑ה מָגֵ֥ן ה֝֗וּא לַֽחֹסִ֥ים בּֽוֹ׃ (מו) אַל־ תּ֥וֹסְףְּ עַל־ דְּבָרָ֑יו פֶּן־ יוֹכִ֖יחַ בְּךָ֣ וְנִכְזָֽבְתָּ׃ (מז) שְׁ֭תַּיִם שָׁאַ֣לְתִּי מֵאִתָּ֑ךְ אַל־ תִּמְנַ֥ע מִ֝מֶּ֗נִּי בְּטֶ֣רֶם אָמֽוּת׃ (מח) שָׁ֤וְא ׀ וּֽדְבַר־ כָּזָ֡ב הַרְחֵ֬ק מִמֶּ֗נִּי רֵ֣אשׁ וָ֭עֹשֶׁר אַל־ תִּֽתֶּן־ לִ֑י הַ֝טְרִיפֵ֗נִי לֶ֣חֶם חֻקִּֽי׃ (מט) פֶּ֥ן אֶשְׂבַּ֨ע ׀ וְכִחַשְׁתִּי֮ וְאָמַ֗רְתִּי מִ֥י יְה֫וָ֥ה וּפֶֽן־ אִוָּרֵ֥שׁ וְגָנַ֑בְתִּי וְ֝תָפַ֗שְׂתִּי שֵׁ֣ם אֱלֹהָֽי׃ (נ) אַל־ תַּלְשֵׁ֣ן עֶ֭בֶד אֶל־ (אדנו) אֲדֹנָ֑יו פֶּֽן־ יְקַלֶּלְךָ֥ וְאָשָֽׁמְתָּ׃ (נא) דּ֭וֹר אָבִ֣יו יְקַלֵּ֑ל וְאֶת־ אִ֝מּ֗וֹ לֹ֣א יְבָרֵֽךְ׃ (נב) דּ֭וֹר טָה֣וֹר בְּעֵינָ֑יו וּ֝מִצֹּאָת֗וֹ לֹ֣א רֻחָֽץ׃ (נג) דּ֭וֹר מָה־ רָמ֣וּ עֵינָ֑יו וְ֝עַפְעַפָּ֗יו יִנָּשֵֽׂאוּ׃ (נד) דּ֤וֹר ׀ חֲרָב֣וֹת שִׁנָּיו֮ וּֽמַאֲכָל֪וֹת מְֽתַלְּעֹ֫תָ֥יו לֶאֱכֹ֣ל עֲנִיִּ֣ים מֵאֶ֑רֶץ וְ֝אֶבְיוֹנִ֗ים מֵאָדָֽם׃ (נה) לַֽעֲלוּקָ֨ה ׀ שְׁתֵּ֥י בָנוֹת֮ הַ֤ב ׀ הַ֥ב שָׁל֣וֹשׁ הֵ֭נָּה לֹ֣א תִשְׂבַּ֑עְנָה אַ֝רְבַּ֗ע לֹא־ אָ֥מְרוּ הֽוֹן׃ (נו) שְׁאוֹל֮ וְעֹ֪צֶ֫ר רָ֥חַם אֶ֭רֶץ לֹא־ שָׂ֣בְעָה מַּ֑יִם וְ֝אֵ֗שׁ לֹא־ אָ֥מְרָה הֽוֹן׃ (נז) עַ֤יִן ׀ תִּֽלְעַ֣ג לְאָב֮ וְתָב֪וּז לִֽיקֲּהַ֫ת־ אֵ֥ם יִקְּר֥וּהָ עֹרְבֵי־ נַ֑חַל וְֽיֹאכְל֥וּהָ בְנֵי־ נָֽשֶׁר׃ (נח) שְׁלֹשָׁ֣ה הֵ֭מָּה נִפְלְא֣וּ מִמֶּ֑נִּי (וארבע) וְ֝אַרְבָּעָ֗ה לֹ֣א יְדַעְתִּֽים׃ (נט) דֶּ֤רֶךְ הַנֶּ֨שֶׁר ׀ בַּשָּׁמַיִם֮ דֶּ֥רֶךְ נָחָ֗שׁ עֲלֵ֫י צ֥וּר דֶּֽרֶךְ־ אֳנִיָּ֥ה בְלֶב־ יָ֑ם וְדֶ֖רֶךְ גֶּ֣בֶר בְּעַלְמָֽה׃ (ס) כֵּ֤ן ׀ דֶּ֥רֶךְ אִשָּׁ֗ה מְנָ֫אָ֥פֶת אָ֭כְלָה וּמָ֣חֲתָה פִ֑יהָ וְ֝אָמְרָ֗ה לֹֽא־ פָעַ֥לְתִּי אָֽוֶן׃ (סא) תַּ֣חַת שָׁ֭לוֹשׁ רָ֣גְזָה אֶ֑רֶץ וְתַ֥חַת אַ֝רְבַּ֗ע לֹא־ תוּכַ֥ל שְׂאֵֽת׃ (סב) תַּֽחַת־ עֶ֭בֶד כִּ֣י יִמְל֑וֹךְ וְ֝נָבָ֗ל כִּ֣י יִֽשְׂבַּֽע־ לָֽחֶם׃ (סג) תַּ֣חַת שְׂ֭נוּאָה כִּ֣י תִבָּעֵ֑ל וְ֝שִׁפְחָ֗ה כִּֽי־ תִירַ֥שׁ גְּבִרְתָּֽהּ׃ (סד) אַרְבָּ֣עָה הֵ֭ם קְטַנֵּי־ אָ֑רֶץ וְ֝הֵ֗מָּה חֲכָמִ֥ים מְחֻכָּמִֽים׃ (סה) הַ֭נְּמָלִים עַ֣ם לֹא־ עָ֑ז וַיָּכִ֖ינוּ בַקַּ֣יִץ לַחְמָֽם׃ (סו) שְׁ֭פַנִּים עַ֣ם לֹא־ עָצ֑וּם וַיָּשִׂ֖ימוּ בַסֶּ֣לַע בֵּיתָֽם׃ (סז) מֶ֭לֶךְ אֵ֣ין לָאַרְבֶּ֑ה וַיֵּצֵ֖א חֹצֵ֣ץ כֻּלּֽוֹ׃ (סח) שְׂ֭מָמִית בְּיָדַ֣יִם תְּתַפֵּ֑שׂ וְ֝הִ֗יא בְּהֵ֣יכְלֵי מֶֽלֶךְ׃ (סט) שְׁלֹשָׁ֣ה הֵ֭מָּה מֵיטִ֣יבֵי צָ֑עַד וְ֝אַרְבָּעָ֗ה מֵיטִ֥בֵי לָֽכֶת׃ (ע) לַ֭יִשׁ גִּבּ֣וֹר בַּבְּהֵמָ֑ה וְלֹא־ יָ֝שׁ֗וּב מִפְּנֵי־ כֹֽל׃ (עא) זַרְזִ֣יר מָתְנַ֣יִם אוֹ־ תָ֑יִשׁ וּ֝מֶ֗לֶךְ אַלְק֥וּם עִמּֽוֹ׃ (עב) אִם־ נָבַ֥לְתָּ בְהִתְנַשֵּׂ֑א וְאִם־ זַ֝מּ֗וֹתָ יָ֣ד לְפֶֽה׃ (עג) כִּ֤י מִ֪יץ חָלָ֡ב י֘וֹצִ֤יא חֶמְאָ֗ה וּֽמִיץ־ אַ֭ף י֣וֹצִיא דָ֑ם וּמִ֥יץ אַ֝פַּ֗יִם י֣וֹצִיא רִֽיב׃ (עד) דִּ֭בְרֵי לְמוּאֵ֣ל מֶ֑לֶךְ מַ֝שָּׂ֗א אֲ‍ֽשֶׁר־ יִסְּרַ֥תּוּ אִמּֽוֹ׃ (עה) מַה־ בְּ֭רִי וּמַֽה־ בַּר־ בִּטְנִ֑י וּ֝מֶה בַּר־ נְדָרָֽי׃ (עו) אַל־ תִּתֵּ֣ן לַנָּשִׁ֣ים חֵילֶ֑ךָ וּ֝דְרָכֶ֗יךָ לַֽמְח֥וֹת מְלָכִֽין׃ (עז) אַ֤ל לַֽמְלָכִ֨ים ׀ לְֽמוֹאֵ֗ל אַ֣ל לַֽמְלָכִ֣ים שְׁתוֹ־ יָ֑יִן וּ֝לְרוֹזְנִ֗ים (או) אֵ֣י שֵׁכָֽר׃ (עח) פֶּן־ יִ֭שְׁתֶּה וְיִשְׁכַּ֣ח מְחֻקָּ֑ק וִֽ֝ישַׁנֶּה דִּ֣ין כָּל־ בְּנֵי־ עֹֽנִי׃ (עט) תְּנוּ־ שֵׁכָ֣ר לְאוֹבֵ֑ד וְ֝יַיִן לְמָ֣רֵי נָֽפֶשׁ׃ (פ) יִ֭שְׁתֶּה וְיִשְׁכַּ֣ח רִישׁ֑וֹ וַ֝עֲמָל֗וֹ לֹ֣א יִזְכָּר־ עֽוֹד׃ (פא) פְּתַח־ פִּ֥יךָ לְאִלֵּ֑ם אֶל־ דִּ֝֗ין כָּל־ בְּנֵ֥י חֲלֽוֹף׃ (פב) פְּתַח־ פִּ֥יךָ שְׁפָט־ צֶ֑דֶק וְ֝דִ֗ין עָנִ֥י וְאֶבְיֽוֹן׃ (פג) אֵֽשֶׁת־ חַ֭יִל מִ֣י יִמְצָ֑א וְרָחֹ֖ק מִפְּנִינִ֣ים מִכְרָֽהּ׃ (פד) בָּ֣טַח בָּ֭הּ לֵ֣ב בַּעְלָ֑הּ וְ֝שָׁלָ֗ל לֹ֣א יֶחְסָֽר׃ (פה) גְּמָלַ֣תְהוּ ט֣וֹב וְלֹא־ רָ֑ע כֹּ֝֗ל יְמֵ֣י חַיֶּֽיה׃ (פו) דָּ֭רְשָׁה צֶ֣מֶר וּפִשְׁתִּ֑ים וַ֝תַּ֗עַשׂ בְּחֵ֣פֶץ כַּפֶּֽיהָ׃ (פז) הָ֭יְתָה כָּאֳנִיּ֣וֹת סוֹחֵ֑ר מִ֝מֶּרְחָ֗ק תָּבִ֥יא לַחְמָֽהּ׃ (פח) וַתָּ֤קָם ׀ בְּע֬וֹד לַ֗יְלָה וַתִּתֵּ֣ן טֶ֣רֶף לְבֵיתָ֑הּ וְ֝חֹ֗ק לְנַעֲרֹתֶֽיהָ׃ (פט) זָמְמָ֣ה שָׂ֭דֶה וַתִּקָּחֵ֑הוּ מִפְּרִ֥י כַ֝פֶּ֗יהָ (נטע) נָ֣טְעָה כָּֽרֶם׃ (צ) חָֽגְרָ֣ה בְע֣וֹז מָתְנֶ֑יהָ וַ֝תְּאַמֵּ֗ץ זְרֹעוֹתֶֽיהָ׃ (צא) טָ֭עֲמָה כִּי־ ט֣וֹב סַחְרָ֑הּ לֹֽא־ יִכְבֶּ֖ה (בליל) בַלַּ֣יְלָה נֵרָֽהּ׃ (צב) יָ֭דֶיהָ שִׁלְּחָ֣ה בַכִּישׁ֑וֹר וְ֝כַפֶּ֗יהָ תָּ֣מְכוּ פָֽלֶךְ׃ (צג) כַּ֭פָּהּ פָּרְשָׂ֣ה לֶעָנִ֑י וְ֝יָדֶ֗יהָ שִׁלְּחָ֥ה לָֽאֶבְיֽוֹן׃ (צד) לֹא־ תִירָ֣א לְבֵיתָ֣הּ מִשָּׁ֑לֶג כִּ֥י כָל־ בֵּ֝יתָ֗הּ לָבֻ֥שׁ שָׁנִֽים׃ (צה) מַרְבַדִּ֥ים עָֽשְׂתָה־ לָּ֑הּ שֵׁ֖שׁ וְאַרְגָּמָ֣ן לְבוּשָֽׁהּ׃ (צו) נוֹדָ֣ע בַּשְּׁעָרִ֣ים בַּעְלָ֑הּ בְּ֝שִׁבְתּ֗וֹ עִם־ זִקְנֵי־ אָֽרֶץ׃ (צז) סָדִ֣ין עָ֭שְׂתָה וַתִּמְכֹּ֑ר וַ֝חֲג֗וֹר נָתְנָ֥ה לַֽכְּנַעֲנִֽי׃ (צח) עֹז־ וְהָדָ֥ר לְבוּשָׁ֑הּ וַ֝תִּשְׂחַ֗ק לְי֣וֹם אַחֲרֽוֹן׃ (צט) פִּ֭יהָ פָּתְחָ֣ה בְחָכְמָ֑ה וְתֽוֹרַת־ חֶ֝֗סֶד עַל־ לְשׁוֹנָֽהּ׃ (ק) צ֭וֹפִיָּה הֲלִיכ֣וֹת בֵּיתָ֑הּ וְלֶ֥חֶם עַ֝צְל֗וּת לֹ֣א תֹאכֵֽל׃ (קא) קָ֣מוּ בָ֭נֶיהָ וַֽיְאַשְּׁר֑וּהָ בַּ֝עְלָ֗הּ וַֽיְהַֽלְלָהּ׃ (קב) רַבּ֣וֹת בָּ֭נוֹת עָ֣שׂוּ חָ֑יִל וְ֝אַ֗תְּ עָלִ֥ית עַל־ כֻּלָּֽנָה׃ (קג) שֶׁ֣קֶר הַ֭חֵן וְהֶ֣בֶל הַיֹּ֑פִי אִשָּׁ֥ה יִרְאַת־ יְ֝הוָ֗ה הִ֣יא תִתְהַלָּֽל׃ (קד) תְּנוּ־ לָ֭הּ מִפְּרִ֣י יָדֶ֑יהָ וִֽיהַלְל֖וּהָ בַשְּׁעָרִ֣ים מַעֲשֶֽׂיהָ׃
