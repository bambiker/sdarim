משלי סדר ד

פתיחה באתר · ספר משלי כולו ·

(א) בְּאֵ֣ין אֲ֭לָפִים אֵב֣וּס בָּ֑ר וְרָב־ תְּ֝בוּא֗וֹת בְּכֹ֣חַ שֽׁוֹר׃ (ב) עֵ֣ד אֱ֭מוּנִים לֹ֣א יְכַזֵּ֑ב וְיָפִ֥יחַ כְּ֝זָבִ֗ים עֵ֣ד שָֽׁקֶר׃ (ג) בִּקֶּשׁ־ לֵ֣ץ חָכְמָ֣ה וָאָ֑יִן וְדַ֖עַת לְנָב֣וֹן נָקָֽל׃ (ד) לֵ֣ךְ מִ֭נֶּגֶד לְאִ֣ישׁ כְּסִ֑יל וּבַל־ יָ֝דַ֗עְתָּ שִׂפְתֵי־ דָֽעַת׃ (ה) חָכְמַ֣ת עָ֭רוּם הָבִ֣ין דַּרְכּ֑וֹ וְאִוֶּ֖לֶת כְּסִילִ֣ים מִרְמָֽה׃ (ו) אֱ֭וִלִים יָלִ֣יץ אָשָׁ֑ם וּבֵ֖ין יְשָׁרִ֣ים רָצֽוֹן׃ (ז) לֵ֗ב י֭וֹדֵעַ מָרַּ֣ת נַפְשׁ֑וֹ וּ֝בְשִׂמְחָת֗וֹ לֹא־ יִתְעָ֥רַב זָֽר׃ (ח) בֵּ֣ית רְ֭שָׁעִים יִשָּׁמֵ֑ד וְאֹ֖הֶל יְשָׁרִ֣ים יַפְרִֽיחַ׃ (ט) יֵ֤שׁ דֶּ֣רֶךְ יָ֭שָׁר לִפְנֵי־ אִ֑ישׁ וְ֝אַחֲרִיתָ֗הּ דַּרְכֵי־ מָֽוֶת׃ (י) גַּם־ בִּשְׂח֥וֹק יִכְאַב־ לֵ֑ב וְאַחֲרִיתָ֖הּ שִׂמְחָ֣ה תוּגָֽה׃ (יא) מִדְּרָכָ֣יו יִ֭שְׂבַּע ס֣וּג לֵ֑ב וּ֝מֵעָלָ֗יו אִ֣ישׁ טֽוֹב׃ (יב) פֶּ֭תִי יַאֲמִ֣ין לְכָל־ דָּבָ֑ר וְ֝עָר֗וּם יָבִ֥ין לַאֲשֻׁרֽוֹ׃ (יג) חָכָ֣ם יָ֭רֵא וְסָ֣ר מֵרָ֑ע וּ֝כְסִ֗יל מִתְעַבֵּ֥ר וּבוֹטֵֽחַ׃ (יד) קְֽצַר־ אַ֭פַּיִם יַעֲשֶׂ֣ה אִוֶּ֑לֶת וְאִ֥ישׁ מְ֝זִמּ֗וֹת יִשָּׂנֵֽא׃ (טו) נָחֲל֣וּ פְתָאיִ֣ם אִוֶּ֑לֶת וַֽ֝עֲרוּמִ֗ים יַכְתִּ֥רוּ דָֽעַת׃ (טז) שַׁח֣וּ רָ֭עִים לִפְנֵ֣י טוֹבִ֑ים וּ֝רְשָׁעִ֗ים עַֽל־ שַׁעֲרֵ֥י צַדִּֽיק׃ (יז) גַּם־ לְ֭רֵעֵהוּ יִשָּׂ֣נֵא רָ֑שׁ וְאֹהֲבֵ֖י עָשִׁ֣יר רַבִּֽים׃ (יח) בָּז־ לְרֵעֵ֥הוּ חוֹטֵ֑א וּמְחוֹנֵ֖ן (עניים) עֲנָוִ֣ים אַשְׁרָֽיו׃ (יט) הֲ‍ֽלוֹא־ יִ֭תְעוּ חֹ֣רְשֵׁי רָ֑ע וְחֶ֥סֶד וֶ֝אֱמֶ֗ת חֹ֣רְשֵׁי טֽוֹב׃ (כ) בְּכָל־ עֶ֭צֶב יִהְיֶ֣ה מוֹתָ֑ר וּדְבַר־ שְׂ֝פָתַ֗יִם אַךְ־ לְמַחְסֽוֹר׃ (כא) עֲטֶ֣רֶת חֲכָמִ֣ים עָשְׁרָ֑ם אִוֶּ֖לֶת כְּסִילִ֣ים אִוֶּֽלֶת׃ (כב) מַצִּ֣יל נְ֭פָשׁוֹת עֵ֣ד אֱמֶ֑ת וְיָפִ֖חַ כְּזָבִ֣ים מִרְמָֽה׃ (כג) בְּיִרְאַ֣ת יְ֭הוָה מִבְטַח־ עֹ֑ז וּ֝לְבָנָ֗יו יִהְיֶ֥ה מַחְסֶֽה׃ (כד) יִרְאַ֣ת יְ֭הוָה מְק֣וֹר חַיִּ֑ים לָ֝ס֗וּר מִמֹּ֥קְשֵׁי מָֽוֶת׃ (כה) בְּרָב־ עָ֥ם הַדְרַת־ מֶ֑לֶךְ וּבְאֶ֥פֶס לְ֝אֹ֗ם מְחִתַּ֥ת רָזֽוֹן׃ (כו) אֶ֣רֶךְ אַ֭פַּיִם רַב־ תְּבוּנָ֑ה וּקְצַר־ ר֝֗וּחַ מֵרִ֥ים אִוֶּֽלֶת׃ (כז) חַיֵּ֣י בְ֭שָׂרִים לֵ֣ב מַרְפֵּ֑א וּרְקַ֖ב עֲצָמ֣וֹת קִנְאָֽה׃ (כח) עֹ֣שֵֽׁק דָּ֭ל חֵרֵ֣ף עֹשֵׂ֑הוּ וּ֝מְכַבְּד֗וֹ חֹנֵ֥ן אֶבְיֽוֹן׃ (כט) בְּֽ֭רָעָתוֹ יִדָּחֶ֣ה רָשָׁ֑ע וְחֹסֶ֖ה בְמוֹת֣וֹ צַדִּֽיק׃ (ל) בְּלֵ֣ב נָ֭בוֹן תָּנ֣וּחַ חָכְמָ֑ה וּבְקֶ֥רֶב כְּ֝סִילִ֗ים תִּוָּדֵֽעַ׃ (לא) צְדָקָ֥ה תְרֽוֹמֵֽם־ גּ֑וֹי וְחֶ֖סֶד לְאֻמִּ֣ים חַטָּֽאת׃ (לב) רְֽצוֹן־ מֶ֭לֶךְ לְעֶ֣בֶד מַשְׂכִּ֑יל וְ֝עֶבְרָת֗וֹ תִּהְיֶ֥ה מֵבִֽישׁ׃ (לג) מַֽעֲנֶה־ רַּ֭ךְ יָשִׁ֣יב חֵמָ֑ה וּדְבַר־ עֶ֝֗צֶב יַעֲלֶה־ אָֽף׃ (לד) לְשׁ֣וֹן חֲ֭כָמִים תֵּיטִ֣יב דָּ֑עַת וּפִ֥י כְ֝סִילִ֗ים יַבִּ֥יעַ אִוֶּֽלֶת׃ (לה) בְּֽכָל־ מָ֭קוֹם עֵינֵ֣י יְהוָ֑ה צֹ֝פ֗וֹת רָעִ֥ים וטוֹבִֽים׃ (לו) מַרְפֵּ֣א לָ֭שׁוֹן עֵ֣ץ חַיִּ֑ים וְסֶ֥לֶף בָּ֝֗הּ שֶׁ֣בֶר בְּרֽוּחַ׃ (לז) אֱוִ֗יל יִ֭נְאַץ מוּסַ֣ר אָבִ֑יו וְשֹׁמֵ֖ר תּוֹכַ֣חַת יַעְרִֽם׃ (לח) בֵּ֣ית צַ֭דִּיק חֹ֣סֶן רָ֑ב וּבִתְבוּאַ֖ת רָשָׁ֣ע נֶעְכָּֽרֶת׃ (לט) שִׂפְתֵ֣י חֲ֭כָמִים יְזָ֣רוּ דָ֑עַת וְלֵ֖ב כְּסִילִ֣ים לֹא־ כֵֽן׃ (מ) זֶ֣בַח רְ֭שָׁעִים תּוֹעֲבַ֣ת יְהוָ֑ה וּתְפִלַּ֖ת יְשָׁרִ֣ים רְצוֹנֽוֹ׃ (מא) תּוֹעֲבַ֣ת יְ֭הוָה דֶּ֣רֶךְ רָשָׁ֑ע וּמְרַדֵּ֖ף צְדָקָ֣ה יֶאֱהָֽב׃ (מב) מוּסָ֣ר רָ֭ע לְעֹזֵ֣ב אֹ֑רַח שׂוֹנֵ֖א תוֹכַ֣חַת יָמֽוּת׃ (מג) שְׁא֣וֹל וַ֭אֲבַדּוֹן נֶ֣גֶד יְהוָ֑ה אַ֝֗ף כִּֽי־ לִבּ֥וֹת בְּֽנֵי־ אָדָֽם׃ (מד) לֹ֣א יֶאֱהַב־ לֵ֭ץ הוֹכֵ֣חַֽ ל֑וֹ אֶל־ חֲ֝כָמִ֗ים לֹ֣א יֵלֵֽךְ׃ (מה) לֵ֣ב שָׂ֭מֵחַ יֵיטִ֣ב פָּנִ֑ים וּבְעַצְּבַת־ לֵ֝ב ר֣וּחַ נְכֵאָֽה׃ (מו) לֵ֣ב נָ֭בוֹן יְבַקֶּשׁ־ דָּ֑עַת (ופני) וּפִ֥י כְ֝סִילִ֗ים יִרְעֶ֥ה אִוֶּֽלֶת׃ (מז) כָּל־ יְמֵ֣י עָנִ֣י רָעִ֑ים וְטֽוֹב־ לֵ֝֗ב מִשְׁתֶּ֥ה תָמִֽיד׃ (מח) טוֹב־ מְ֭עַט בְּיִרְאַ֣ת יְהוָ֑ה מֵאוֹצָ֥ר רָ֝֗ב וּמְה֥וּמָה בֽוֹ׃ (מט) ט֤וֹב אֲרֻחַ֣ת יָ֭רָק וְאַהֲבָה־ שָׁ֑ם מִשּׁ֥וֹר אָ֝ב֗וּס וְשִׂנְאָה־ בֽוֹ׃ (נ) אִ֣ישׁ חֵ֭מָה יְגָרֶ֣ה מָד֑וֹן וְאֶ֥רֶך אַ֝פַּ֗יִם יַשְׁקִ֥יט רִֽיב׃ (נא) דֶּ֣רֶךְ עָ֭צֵל כִּמְשֻׂ֣כַת חָ֑דֶק וְאֹ֖רַח יְשָׁרִ֣ים סְלֻלָֽה׃ (נב) בֵּ֣ן חָ֭כָם יְשַׂמַּח־ אָ֑ב וּכְסִ֥יל אָ֝דָ֗ם בּוֹזֶ֥ה אִמּֽוֹ׃ (נג) אִ֭וֶּלֶת שִׂמְחָ֣ה לַחֲסַר־ לֵ֑ב וְאִ֥ישׁ תְּ֝בוּנָ֗ה יְיַשֶׁר־ לָֽכֶת׃ (נד) הָפֵ֣ר מַ֭חֲשָׁבוֹת בְּאֵ֣ין ס֑וֹד וּבְרֹ֖ב יוֹעֲצִ֣ים תָּקֽוּם׃ (נה) שִׂמְחָ֣ה לָ֭אִישׁ בְּמַעֲנֵה־ פִ֑יו וְדָבָ֖ר בְּעִתּ֣וֹ מַה־ טּֽוֹב׃ (נו) אֹ֣רַח חַ֭יִּים לְמַ֣עְלָה לְמַשְׂכִּ֑יל לְמַ֥עַן ס֝֗וּר מִשְּׁא֥וֹל מָֽטָּה׃ (נז) בֵּ֣ית גֵּ֭אִים יִסַּ֥ח ׀ יְהוָ֑ה וְ֝יַצֵּ֗ב גְּב֣וּל אַלְמָנָֽה׃ (נח) תּוֹעֲבַ֣ת יְ֭הוָה מַחְשְׁב֣וֹת רָ֑ע וּ֝טְהֹרִ֗ים אִמְרֵי־ נֹֽעַם׃ (נט) עֹכֵ֣ר בֵּ֭יתוֹ בּוֹצֵ֣עַ בָּ֑צַע וְשׂוֹנֵ֖א מַתָּנֹ֣ת יִחְיֶֽה׃ (ס) לֵ֣ב צַ֭דִּיק יֶהְגֶּ֣ה לַעֲנ֑וֹת וּפִ֥י רְ֝שָׁעִ֗ים יַבִּ֥יעַ רָעֽוֹת׃ (סא) רָח֣וֹק יְ֭הוָה מֵרְשָׁעִ֑ים וּתְפִלַּ֖ת צַדִּיקִ֣ים יִשְׁמָֽע׃ (סב) מְֽאוֹר־ עֵ֭ינַיִם יְשַׂמַּֽח־ לֵ֑ב שְׁמוּעָ֥ה ט֝וֹבָ֗ה תְּדַשֶּׁן־ עָֽצֶם׃ (סג) אֹ֗זֶן שֹׁ֭מַעַת תּוֹכַ֣חַת חַיִּ֑ים בְּקֶ֖רֶב חֲכָמִ֣ים תָּלִֽין׃ (סד) פּוֹרֵ֣עַ מ֭וּסָר מוֹאֵ֣ס נַפְשׁ֑וֹ וְשׁוֹמֵ֥עַ תּ֝וֹכַ֗חַת ק֣וֹנֶה לֵּֽב׃ (סה) יִרְאַ֣ת יְ֭הוָה מוּסַ֣ר חָכְמָ֑ה וְלִפְנֵ֖י כָב֣וֹד עֲנָוָֽה׃ (סו) לְאָדָ֥ם מַֽעַרְכֵי־ לֵ֑ב וּ֝מֵיְהוָ֗ה מַעֲנֵ֥ה לָשֽׁוֹן׃ (סז) כָּֽל־ דַּרְכֵי־ אִ֭ישׁ זַ֣ךְ בְּעֵינָ֑יו וְתֹכֵ֖ן רוּח֣וֹת יְהוָֽה׃ (סח) גֹּ֣ל אֶל־ יְהוָ֣ה מַעֲשֶׂ֑יךָ וְ֝יִכֹּ֗נוּ מַחְשְׁבֹתֶֽיךָ׃ (סט) כֹּ֤ל פָּעַ֣ל יְ֭הוָה לַֽמַּעֲנֵ֑הוּ וְגַם־ רָ֝שָׁ֗ע לְי֣וֹם רָעָֽה׃ (ע) תּוֹעֲבַ֣ת יְ֭הוָה כָּל־ גְּבַהּ־ לֵ֑ב יָ֥ד לְ֝יָ֗ד לֹ֣א יִנָּקֶֽה׃ (עא) בְּחֶ֣סֶד וֶ֭אֱמֶת יְכֻפַּ֣ר עָוֺ֑ן וּבְיִרְאַ֥ת יְ֝הוָ֗ה ס֣וּר מֵרָֽע׃ (עב) בִּרְצ֣וֹת יְ֭הוָה דַּרְכֵי־ אִ֑ישׁ גַּם־ א֝וֹיְבָ֗יו יַשְׁלִ֥ם אִתּֽוֹ׃ (עג) טוֹב־ מְ֭עַט בִּצְדָקָ֑ה מֵרֹ֥ב תְּ֝בוּא֗וֹת בְּלֹ֣א מִשְׁפָּֽט׃ (עד) לֵ֣ב אָ֭דָם יְחַשֵּׁ֣ב דַּרְכּ֑וֹ וַֽ֝יהוָ֗ה יָכִ֥ין צַעֲדֽוֹ׃ (עה) קֶ֤סֶם ׀ עַֽל־ שִׂפְתֵי־ מֶ֑לֶךְ בְּ֝מִשְׁפָּ֗ט לֹ֣א יִמְעַל־ פִּֽיו׃ (עו) פֶּ֤לֶס ׀ וּמֹאזְנֵ֣י מִ֭שְׁפָּט לַֽיהוָ֑ה מַ֝עֲשֵׂ֗הוּ כָּל־ אַבְנֵי־ כִֽיס׃ (עז) תּוֹעֲבַ֣ת מְ֭לָכִים עֲשׂ֣וֹת רֶ֑שַׁע כִּ֥י בִ֝צְדָקָ֗ה יִכּ֥וֹן כִּסֵּֽא׃ (עח) רְצ֣וֹן מְ֭לָכִים שִׂפְתֵי־ צֶ֑דֶק וְדֹבֵ֖ר יְשָׁרִ֣ים יֶאֱהָֽב׃ (עט) חֲמַת־ מֶ֥לֶךְ מַלְאֲכֵי־ מָ֑וֶת וְאִ֖ישׁ חָכָ֣ם יְכַפְּרֶֽנָּה׃ (פ) בְּאוֹר־ פְּנֵי־ מֶ֥לֶךְ חַיִּ֑ים וּ֝רְצוֹנ֗וֹ כְּעָ֣ב מַלְקֽוֹשׁ׃ (פא) קְֽנֹה־ חָכְמָ֗ה מַה־ טּ֥וֹב מֵחָר֑וּץ וּקְנ֥וֹת בִּ֝ינָ֗ה נִבְחָ֥ר מִכָּֽסֶף׃ (פב) מְסִלַּ֣ת יְ֭שָׁרִים ס֣וּר מֵרָ֑ע שֹׁמֵ֥ר נַ֝פְשׁ֗וֹ נֹצֵ֥ר דַּרְכּֽוֹ׃ (פג) לִפְנֵי־ שֶׁ֥בֶר גָּא֑וֹן וְלִפְנֵ֥י כִ֝שָּׁל֗וֹן גֹּ֣בַהּ רֽוּחַ׃ (פד) ט֣וֹב שְׁפַל־ ר֭וּחַ אֶת־ (עניים) עֲנָוִ֑ים מֵֽחַלֵּ֥ק שָׁ֝לָ֗ל אֶת־ גֵּאִֽים׃ (פה) מַשְׂכִּ֣יל עַל־ דָּ֭בָר יִמְצָא־ ט֑וֹב וּבוֹטֵ֖חַ בַּיהוָ֣ה אַשְׁרָֽיו׃ (פו) לַחֲכַם־ לֵ֭ב יִקָּרֵ֣א נָב֑וֹן וּמֶ֥תֶק שְׂ֝פָתַ֗יִם יֹסִ֥יף לֶֽקַח׃ (פז) מְק֣וֹר חַ֭יִּים שֵׂ֣כֶל בְּעָלָ֑יו וּמוּסַ֖ר אֱוִלִ֣ים אִוֶּֽלֶת׃ (פח) לֵ֣ב חָ֭כָם יַשְׂכִּ֣יל פִּ֑יהוּ וְעַל־ שְׂ֝פָתָ֗יו יֹסִ֥יף לֶֽקַח׃ (פט) צוּף־ דְּ֭בַשׁ אִמְרֵי־ נֹ֑עַם מָת֥וֹק לַ֝נֶּפֶשׁ וּמַרְפֵּ֥א לָעָֽצֶם׃ (צ) יֵ֤שׁ דֶּ֣רֶךְ יָ֭שָׁר לִפְנֵי־ אִ֑ישׁ וְ֝אַחֲרִיתָ֗הּ דַּרְכֵי־ מָֽוֶת׃ (צא) נֶ֣פֶשׁ עָ֭מֵל עָ֣מְלָה לּ֑וֹ כִּֽי־ אָכַ֖ף עָלָ֣יו פִּֽיהוּ׃ (צב) אִ֣ישׁ בְּ֭לִיַּעַל כֹּרֶ֣ה רָעָ֑ה וְעַל־ (שפתיו) שְׂ֝פָת֗וֹ כְּאֵ֣שׁ צָרָֽבֶת׃ (צג) אִ֣ישׁ תַּ֭הְפֻּכוֹת יְשַׁלַּ֣ח מָד֑וֹן וְ֝נִרְגָּ֗ מַפְרִ֥יד אַלּֽוּף׃ (צד) אִ֣ישׁ חָ֭מָס יְפַתֶּ֣ה רֵעֵ֑הוּ וְ֝הוֹלִיכ֗וֹ בְּדֶ֣רֶךְ לֹא־ טֽוֹב׃ (צה) עֹצֶ֣ה עֵ֭ינָיו לַחְשֹׁ֣ב תַּהְפֻּכ֑וֹת קֹרֵ֥ץ שְׂ֝פָתָ֗יו כִּלָּ֥ה רָעָֽה׃ (צו) עֲטֶ֣רֶת תִּפְאֶ֣רֶת שֵׂיבָ֑ה בְּדֶ֥רֶךְ צְ֝דָקָ֗ה תִּמָּצֵֽא׃ (צז) ט֤וֹב אֶ֣רֶךְ אַ֭פַּיִם מִגִּבּ֑וֹר וּמֹשֵׁ֥ל בְּ֝רוּח֗וֹ מִלֹּכֵ֥ד עִֽיר׃ (צח) בַּ֭חֵיק יוּטַ֣ל אֶת־ הַגּוֹרָ֑ל וּ֝מֵיְהוָ֗ה כָּל־ מִשְׁפָּטֽוֹ׃ (צט) ט֤וֹב פַּ֣ת חֲ֭רֵבָה וְשַׁלְוָה־ בָ֑הּ מִ֝בַּ֗יִת מָלֵ֥א זִבְחֵי־ רִֽיב׃ (ק) עֶֽבֶד־ מַשְׂכִּ֗יל יִ֭מְשֹׁל בְּבֵ֣ן מֵבִ֑ישׁ וּבְת֥וֹךְ אַ֝חִ֗ים יַחֲלֹ֥ק נַחֲלָֽה׃ (קא) מַצְרֵ֣ף לַ֭כֶּסֶף וְכ֣וּר לַזָּהָ֑ב וּבֹחֵ֖ן לִבּ֣וֹת יְהוָֽה׃ (קב) מֵ֭רַע מַקְשִׁ֣יב עַל־ שְׂפַת־ אָ֑וֶן שֶׁ֥קֶר מֵ֝זִין עַל־ לְשׁ֥וֹן הַוֺּֽת׃ (קג) לֹעֵ֣ג לָ֭רָשׁ חֵרֵ֣ף עֹשֵׂ֑הוּ שָׂמֵ֥חַ לְ֝אֵ֗יד לֹ֣א יִנָּקֶֽה׃ (קד) עֲטֶ֣רֶת זְ֭קֵנִים בְּנֵ֣י בָנִ֑ים וְתִפְאֶ֖רֶת בָּנִ֣ים אֲבוֹתָֽם׃ (קה) לֹא־ נָאוָ֣ה לְנָבָ֣ל שְׂפַת־ יֶ֑תֶר אַ֝֗ף כִּֽי־ לְנָדִ֥יב שְׂפַת־ שָֽׁקֶר׃ (קו) אֶֽבֶן־ חֵ֣ן הַ֭שֹּׁחַד בְּעֵינֵ֣י בְעָלָ֑יו אֶֽל־ כָּל־ אֲשֶׁ֖ר יִפְנֶ֣ה יַשְׂכִּֽיל׃ (קז) מְֽכַסֶּה־ פֶּ֭שַׁע מְבַקֵּ֣שׁ אַהֲבָ֑ה וְשֹׁנֶ֥ה בְ֝דָבָ֗ר מַפְרִ֥יד אַלּֽוּף׃ (קח) תֵּ֣חַת גְּעָרָ֣ה בְמֵבִ֑ין מֵהַכּ֖וֹת כְּסִ֣יל מֵאָֽה׃ (קט) אַךְ־ מְרִ֥י יְבַקֶּשׁ־ רָ֑ע וּמַלְאָ֥ךְ אַ֝כְזָרִ֗י יְשֻׁלַּח־ בּֽוֹ׃ (קי) פָּג֬וֹשׁ דֹּ֣ב שַׁכּ֣וּל בְּאִ֑ישׁ וְאַל־ כְּ֝סִ֗יל בְּאִוַּלְתּֽוֹ׃ (קיא) מֵשִׁ֣יב רָ֭עָה תַּ֣חַת טוֹבָ֑ה לֹא־ (תמיש) תָמ֥וּשׁ רָ֝עָ֗ה מִבֵּיתֽוֹ׃ (קיב) פּ֣וֹטֵֽר מַ֭יִם רֵאשִׁ֣ית מָד֑וֹן וְלִפְנֵ֥י הִ֝תְגַּלַּ֗ע הָרִ֥יב נְטֽוֹשׁ׃ (קיג) מַצְדִּ֣יק רָ֭שָׁע וּמַרְשִׁ֣יעַ צַדִּ֑יק תּוֹעֲבַ֥ת יְ֝הוָ֗ה גַּם־ שְׁנֵיהֶֽם׃ (קיד) לָמָּה־ זֶּ֣ה מְחִ֣יר בְּיַד־ כְּסִ֑יל לִקְנ֖וֹת חָכְמָ֣ה וְלֶב־ אָֽיִן׃ (קטו) בְּכָל־ עֵ֭ת אֹהֵ֣ב הָרֵ֑עַ וְאָ֥ח לְ֝צָרָ֗ה יִוָּלֵֽד׃ (קטז) אָדָ֣ם חֲסַר־ לֵ֭ב תּוֹקֵ֣עַ כָּ֑ף עֹרֵ֥ב עֲ֝רֻבָּ֗ה לִפְנֵ֥י רֵעֵֽהוּ׃ (קיז) אֹ֣הֵֽב פֶּ֭שַׁע אֹהֵ֣ב מַצָּ֑ה מַגְבִּ֥יהַּ פִּ֝תְח֗וֹ מְבַקֶּשׁ־ שָֽׁבֶר׃ (קיח) עִקֶּשׁ־ לֵ֭ב לֹ֣א יִמְצָא־ ט֑וֹב וְנֶהְפָּ֥ךְ בִּ֝לְשׁוֹנ֗וֹ יִפּ֥וֹל בְּרָעָֽה׃ (קיט) יֹלֵ֣ד כְּ֭סִיל לְת֣וּגָה ל֑וֹ וְלֹֽא־ יִ֝שְׂמַ֗ח אֲבִ֣י נָבָֽל׃ (קכ) לֵ֣ב שָׂ֭מֵחַ יֵיטִ֣ב גֵּהָ֑ה וְר֥וּחַ נְ֝כֵאָ֗ה תְּיַבֶּשׁ־ גָּֽרֶם׃ (קכא) שֹׁ֣חַד מֵ֭חֵיק רָשָׁ֣ע יִקָּ֑ח לְ֝הַטּ֗וֹת אָרְח֥וֹת מִשְׁפָּֽט׃ (קכב) אֶת־ פְּנֵ֣י מֵבִ֣ין חָכְמָ֑ה וְעֵינֵ֥י כְ֝סִ֗יל בִּקְצֵה־ אָֽרֶץ׃ (קכג) כַּ֣עַס לְ֭אָבִיו בֵּ֣ן כְּסִ֑יל וּ֝מֶ֗מֶר לְיוֹלַדְתּֽוֹ׃ (קכד) גַּ֤ם עֲנ֣וֹשׁ לַצַּדִּ֣יק לֹא־ ט֑וֹב לְהַכּ֖וֹת נְדִיבִ֣ים עַל־ יֹֽשֶׁר׃ (קכה) חוֹשֵׂ֣ךְ אֲ֭מָרָיו יוֹדֵ֣עַ דָּ֑עַת (וקר)־ יְקַר־ ר֝֗וּחַ אִ֣ישׁ תְּבוּנָֽה׃ (קכו) גַּ֤ם אֱוִ֣יל מַ֭חֲרִישׁ חָכָ֣ם יֵחָשֵׁ֑ב אֹטֵ֖ם שְׂפָתָ֣יו נָבֽוֹן׃ (קכז) לְֽ֭תַאֲוָה יְבַקֵּ֣שׁ נִפְרָ֑ד בְּכָל־ תּ֝וּשִׁיָּ֗ה יִתְגַּלָּֽע׃ (קכח) לֹֽא־ יַחְפֹּ֣ץ כְּ֭סִיל בִּתְבוּנָ֑ה כִּ֝֗י אִֽם־ בְּהִתְגַּלּ֥וֹת לִבּֽוֹ׃ (קכט) בְּֽבוֹא־ רָ֭שָׁע בָּ֣א גַם־ בּ֑וּז וְֽעִם־ קָל֥וֹן חֶרְפָּֽה׃ (קל) מַ֣יִם עֲ֭מֻקִּים דִּבְרֵ֣י פִי־ אִ֑ישׁ נַ֥חַל נֹ֝בֵ֗עַ מְק֣וֹר חָכְמָֽה׃ (קלא) שְׂאֵ֣ת פְּנֵי־ רָשָׁ֣ע לֹא־ ט֑וֹב לְהַטּ֥וֹת צַ֝דִּ֗יק בַּמִּשְׁפָּֽט׃ (קלב) שִׂפְתֵ֣י כְ֭סִיל יָבֹ֣אֽוּ בְרִ֑יב וּ֝פִ֗יו לְֽמַהֲלֻמ֥וֹת יִקְרָֽא׃ (קלג) פִּֽי־ כְ֭סִיל מְחִתָּה־ ל֑וֹ וּ֝שְׂפָתָ֗יו מוֹקֵ֥שׁ נַפְשֽׁוֹ׃ (קלד) דִּבְרֵ֣י נִ֭רְגָּן כְּמִֽתְלַהֲמִ֑ים וְ֝הֵ֗ם יָרְד֥וּ חַדְרֵי־ בָֽטֶן׃ (קלה) גַּ֭ם מִתְרַפֶּ֣ה בִמְלַאכְתּ֑וֹ אָ֥ח ה֝֗וּא לְבַ֣עַל מַשְׁחִֽית׃
