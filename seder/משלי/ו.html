משלי סדר ו

פתיחה באתר · ספר משלי כולו ·

(א) לְהוֹדִֽיעֲךָ֗ קֹ֭שְׁטְ אִמְרֵ֣י אֱמֶ֑ת לְהָשִׁ֥יב אֲמָרִ֥ים אֱ֝מֶ֗ת לְשֹׁלְחֶֽיךָ׃ (ב) אַֽל־ תִּגְזָל־ דָּ֭ל כִּ֣י דַל־ ה֑וּא וְאַל־ תְּדַכֵּ֖א עָנִ֣י בַשָּֽׁעַר׃ (ג) כִּֽי־ יְ֭הוָה יָרִ֣יב רִיבָ֑ם וְקָבַ֖ע אֶת־ קֹבְעֵיהֶ֣ם נָֽפֶשׁ׃ (ד) אַל־ תִּ֭תְרַע אֶת־ בַּ֣עַל אָ֑ף וְאֶת־ אִ֥ישׁ חֵ֝מוֹת לֹ֣א תָבֽוֹא׃ (ה) פֶּן־ תֶּאֱלַ֥ף (ארחתו) אֹֽרְחֹתָ֑יו וְלָקַחְתָּ֖ מוֹקֵ֣שׁ לְנַפְשֶֽׁךָ׃ (ו) אַל־ תְּהִ֥י בְתֹֽקְעֵי־ כָ֑ף בַּ֝עֹרְבִ֗ים מַשָּׁאֽוֹת׃ (ז) אִם־ אֵֽין־ לְךָ֥ לְשַׁלֵּ֑ם לָ֥מָּה יִקַּ֥ח מִ֝שְׁכָּבְךָ֗ מִתַּחְתֶּֽיךָ׃ (ח) אַל־ תַּ֭סֵּג גְּב֣וּל עוֹלָ֑ם אֲשֶׁ֖ר עָשׂ֣וּ אֲבוֹתֶֽיךָ׃ (ט) חָזִ֡יתָ אִ֤ישׁ ׀ מָ֘הִ֤יר בִּמְלַאכְתּ֗וֹ לִֽפְנֵֽי־ מְלָכִ֥ים יִתְיַצָּ֑ב בַּל־ יִ֝תְיַצֵּב לִפְנֵ֥י חֲשֻׁכִּֽים׃ (י) כִּֽי־ תֵ֭שֵׁב לִלְח֣וֹם אֶת־ מוֹשֵׁ֑ל בִּ֥ין תָּ֝בִ֗ין אֶת־ אֲשֶׁ֥ר לְפָנֶֽיךָ׃ (יא) וְשַׂמְתָּ֣ שַׂכִּ֣ין בְּלֹעֶ֑ךָ אִם־ בַּ֖עַל נֶ֣פֶשׁ אָֽתָּה׃ (יב) אַל־ תִּ֭תְאָו לְמַטְעַמּוֹתָ֑יו וְ֝ה֗וּא לֶ֣חֶם כְּזָבִֽים׃ (יג) אַל־ תִּיגַ֥ע לְֽהַעֲשִׁ֑יר מִֽבִּינָתְךָ֥ חֲדָֽל׃ (יד) (התעוף) הֲתָ֤עִיף עֵינֶ֥יךָ בּ֗וֹ וְֽאֵ֫ינֶ֥נּוּ כִּ֤י עָשֹׂ֣ה יַעֲשֶׂה־ לּ֣וֹ כְנָפַ֑יִם כְּ֝נֶ֗שֶׁר (ועיף) יָע֥וּף הַשָּׁמָֽיִם׃ (טו) אַל־ תִּלְחַ֗ם אֶת־ לֶ֭חֶם רַ֣ע עָ֑יִן וְאַל־ (תתאו) תִּ֝תְאָ֗יו לְמַטְעַמֹּתָֽיו׃ (טז) כִּ֤י ׀ כְּמוֹ־ שָׁעַ֥ר בְּנַפְשׁ֗וֹ כֶּ֫ן־ ה֥וּא אֱכֹ֣ל וּ֭שְׁתֵה יֹ֣אמַר לָ֑ךְ וְ֝לִבּ֗וֹ בַּל־ עִמָּֽךְ׃ (יז) פִּֽתְּךָ־ אָכַ֥לְתָּ תְקִיאֶ֑נָּה וְ֝שִׁחַ֗תָּ דְּבָרֶ֥יךָ הַנְּעִימִֽים׃ (יח) בְּאָזְנֵ֣י כְ֭סִיל אַל־ תְּדַבֵּ֑ר כִּֽי־ יָ֝ב֗וּז לְשֵׂ֣כֶל מִלֶּֽיךָ׃ (יט) אַל־ תַּ֭סֵּג גְּב֣וּל עוֹלָ֑ם וּבִשְׂדֵ֥י יְ֝תוֹמִ֗ים אַל־ תָּבֹֽא׃ (כ) כִּֽי־ גֹאֲלָ֥ם חָזָ֑ק הֽוּא־ יָרִ֖יב אֶת־ רִיבָ֣ם אִתָּֽךְ׃ (כא) הָבִ֣יאָה לַמּוּסָ֣ר לִבֶּ֑ךָ וְ֝אָזְנֶ֗ךָ לְאִמְרֵי־ דָֽעַת׃ (כב) אַל־ תִּמְנַ֣ע מִנַּ֣עַר מוּסָ֑ר כִּֽי־ תַכֶּ֥נּוּ בַ֝שֵּׁ֗בֶט לֹ֣א יָמֽוּת׃ (כג) אַ֭תָּה בַּשֵּׁ֣בֶט תַּכֶּ֑נּוּ וְ֝נַפְשׁ֗וֹ מִשְּׁא֥וֹל תַּצִּֽיל׃ (כד) בְּ֭נִי אִם־ חָכַ֣ם לִבֶּ֑ךָ יִשְׂמַ֖ח לִבִּ֣י גַם־ אָֽנִי׃ (כה) וְתַעְלֹ֥זְנָה כִלְיוֹתָ֑י בְּדַבֵּ֥ר שְׂ֝פָתֶ֗יךָ מֵישָׁרִֽים׃ (כו) אַל־ יְקַנֵּ֣א לִ֭בְּךָ בַּֽחַטָּאִ֑ים כִּ֥י אִם־ בְּיִרְאַת־ יְ֝הוָ֗ה כָּל־ הַיּֽוֹם׃ (כז) כִּ֭י אִם־ יֵ֣שׁ אַחֲרִ֑ית וְ֝תִקְוָתְךָ֗ לֹ֣א תִכָּרֵֽת׃ (כח) שְׁמַע־ אַתָּ֣ה בְנִ֣י וַחֲכָ֑ם וְאַשֵּׁ֖ר בַּדֶּ֣רֶךְ לִבֶּֽךָ׃ (כט) אַל־ תְּהִ֥י בְסֹֽבְאֵי־ יָ֑יִן בְּזֹלֲלֵ֖י בָשָׂ֣ר לָֽמוֹ׃ (ל) כִּי־ סֹבֵ֣א וְ֭זוֹלֵל יִוָּרֵ֑שׁ וּ֝קְרָעִ֗ים תַּלְבִּ֥ישׁ נוּמָֽה׃ (לא) שְׁמַ֣ע לְ֭אָבִיךָ זֶ֣ה יְלָדֶ֑ךָ וְאַל־ תָּ֝ב֗וּז כִּֽי־ זָקְנָ֥ה אִמֶּֽךָ׃ (לב) אֱמֶ֣ת קְ֭נֵה וְאַל־ תִּמְכֹּ֑ר חָכְמָ֖ה וּמוּסָ֣ר וּבִינָֽה׃ (לג) (גול) גִּ֣יל (יגול) יָ֭גִיל אֲבִ֣י צַדִּ֑יק (יולד) וְיוֹלֵ֥ד חָ֝כָ֗ם (וישמח)־ יִשְׂמַח־ בּֽוֹ׃ (לד) יִֽשְׂמַח־ אָבִ֥יךָ וְאִמֶּ֑ךָ וְ֝תָגֵ֗ל יֽוֹלַדְתֶּֽךָ׃ (לה) תְּנָֽה־ בְנִ֣י לִבְּךָ֣ לִ֑י וְ֝עֵינֶ֗יךָ דְּרָכַ֥י (תרצנה) תִּצֹּֽרְנָה׃ (לו) כִּֽי־ שׁוּחָ֣ה עֲמֻקָּ֣ה זוֹנָ֑ה וּבְאֵ֥ר צָ֝רָ֗ה נָכְרִיָּֽה׃ (לז) אַף־ הִ֭יא כְּחֶ֣תֶף תֶּֽאֱרֹ֑ב וּ֝בוֹגְדִ֗ים בְּאָדָ֥ם תּוֹסִֽף׃ (לח) לְמִ֨י א֥וֹי לְמִ֪י אֲב֡וֹי לְמִ֤י (מדונים) מִדְיָנִ֨ים ׀ לְמִ֥י שִׂ֗יחַ לְ֭מִי פְּצָעִ֣ים חִנָּ֑ם לְ֝מִ֗י חַכְלִל֥וּת עֵינָֽיִם׃ (לט) לַֽמְאַחֲרִ֥ים עַל־ הַיָּ֑יִן לַ֝בָּאִ֗ים לַחְקֹ֥ר מִמְסָֽךְ׃ (מ) אַל־ תֵּ֥רֶא יַיִן֮ כִּ֪י יִתְאַ֫דָּ֥ם כִּֽי־ יִתֵּ֣ן (בכיס) בַּכּ֣וֹס עֵינ֑וֹ יִ֝תְהַלֵּ֗ךְ בְּמֵישָׁרִֽים׃ (מא) אַ֭חֲרִיתוֹ כְּנָחָ֣שׁ יִשָּׁ֑ךְ וּֽכְצִפְעֹנִ֥י יַפְרִֽשׁ׃ (מב) עֵ֭ינֶיךָ יִרְא֣וּ זָר֑וֹת וְ֝לִבְּךָ֗ יְדַבֵּ֥ר תַּהְפֻּכֽוֹת׃ (מג) וְ֭הָיִיתָ כְּשֹׁכֵ֣ב בְּלֶב־ יָ֑ם וּ֝כְשֹׁכֵ֗ב בְּרֹ֣אשׁ חִבֵּֽל׃ (מד) הִכּ֥וּנִי בַל־ חָלִיתִי֮ הֲלָמ֗וּנִי בַּל־ יָ֫דָ֥עְתִּי מָתַ֥י אָקִ֑יץ א֝וֹסִ֗יף אֲבַקְשֶׁ֥נּוּ עֽוֹד׃ (מה) אַל־ תְּ֭קַנֵּא בְּאַנְשֵׁ֣י רָעָ֑ה וְאַל־ (תתאו) תִּ֝תְאָ֗יו לִהְי֥וֹת אִתָּֽם׃ (מו) כִּי־ שֹׁ֭ד יֶהְגֶּ֣ה לִבָּ֑ם וְ֝עָמָ֗ל שִׂפְתֵיהֶ֥ם תְּדַבֵּֽרְנָה׃ (מז) בְּ֭חָכְמָה יִבָּ֣נֶה בָּ֑יִת וּ֝בִתְבוּנָ֗ה יִתְכּוֹנָֽן׃ (מח) וּ֭בְדַעַת חֲדָרִ֣ים יִמָּלְא֑וּ כָּל־ ה֖וֹן יָקָ֣ר וְנָעִֽים׃ (מט) גֶּֽבֶר־ חָכָ֥ם בַּע֑וֹז וְאִֽישׁ־ דַּ֝֗עַת מְאַמֶּץ־ כֹּֽחַ׃ (נ) כִּ֣י בְ֭תַחְבֻּלוֹת תַּעֲשֶׂה־ לְּךָ֣ מִלְחָמָ֑ה וּ֝תְשׁוּעָ֗ה בְּרֹ֣ב יוֹעֵֽץ׃ (נא) רָאמ֣וֹת לֶֽאֱוִ֣יל חָכְמ֑וֹת בַּ֝שַּׁ֗עַר לֹ֣א יִפְתַּח־ פִּֽיהוּ׃ (נב) מְחַשֵּׁ֥ב לְהָרֵ֑עַ ל֝֗וֹ בַּֽעַל־ מְזִמּ֥וֹת יִקְרָֽאוּ׃ (נג) זִמַּ֣ת אִוֶּ֣לֶת חַטָּ֑את וְתוֹעֲבַ֖ת לְאָדָ֣ם לֵֽץ׃ (נד) הִ֭תְרַפִּיתָ בְּי֥וֹם צָרָ֗ה צַ֣ר כֹּחֶֽכָה׃ (נה) הַ֭צֵּל לְקֻחִ֣ים לַמָּ֑וֶת וּמָטִ֥ים לַ֝הֶ֗רֶג אִם־ תַּחְשֽׂוֹךְ׃ (נו) כִּֽי־ תֹאמַ֗ר הֵן֮ לֹא־ יָדַ֪עְנ֫וּ זֶ֥ה הֲ‍ֽלֹא־ תֹ֘כֵ֤ן לִבּ֨וֹת ׀ הֽוּא־ יָבִ֗ין וְנֹצֵ֣ר נַ֭פְשְׁךָ ה֣וּא יֵדָ֑ע וְהֵשִׁ֖יב לְאָדָ֣ם כְּפָעֳלֽוֹ׃ (נז) אֱכָל־ בְּנִ֣י דְבַ֣שׁ כִּי־ ט֑וֹב וְנֹ֥פֶת מָ֝ת֗וֹק עַל־ חִכֶּֽךָ׃ (נח) כֵּ֤ן ׀ דְּעֶ֥ה חָכְמָ֗ה לְנַ֫פְשֶׁ֥ךָ אִם־ מָ֭צָאתָ וְיֵ֣שׁ אַחֲרִ֑ית וְ֝תִקְוָתְךָ֗ לֹ֣א תִכָּרֵֽת׃ (נט) אַל־ תֶּאֱרֹ֣ב רָ֭שָׁע לִנְוֵ֣ה צַדִּ֑יק אַֽל־ תְּשַׁדֵּ֥ד רִבְצוֹ׃ (ס) כִּ֤י שֶׁ֨בַע ׀ יִפּ֣וֹל צַדִּ֣יק וָקָ֑ם וּ֝רְשָׁעִ֗ים יִכָּשְׁל֥וּ בְרָעָֽה׃ (סא) בִּנְפֹ֣ל (אויביך) אֽ֭וֹיִבְךָ אַל־ תִּשְׂמָ֑ח וּ֝בִכָּשְׁל֗וֹ אַל־ יָגֵ֥ל לִבֶּֽךָ׃ (סב) פֶּן־ יִרְאֶ֣ה יְ֭הוָה וְרַ֣ע בְּעֵינָ֑יו וְהֵשִׁ֖יב מֵעָלָ֣יו אַפּֽוֹ׃ (סג) אַל־ תִּתְחַ֥ר בַּמְּרֵעִ֑ים אַל־ תְּ֝קַנֵּ֗א בָּרְשָׁעִֽים׃ (סד) כִּ֤י ׀ לֹֽא־ תִהְיֶ֣ה אַחֲרִ֣ית לָרָ֑ע נֵ֖ר רְשָׁעִ֣ים יִדְעָֽךְ׃ (סה) יְרָֽא־ אֶת־ יְהוָ֣ה בְּנִ֣י וָמֶ֑לֶךְ עִם־ שׁ֝וֹנִ֗ים אַל־ תִּתְעָרָֽב׃ (סו) כִּֽי־ פִ֭תְאֹם יָק֣וּם אֵידָ֑ם וּפִ֥יד שְׁ֝נֵיהֶ֗ם מִ֣י יוֹדֵֽעַ׃ (סז) גַּם־ אֵ֥לֶּה לַֽחֲכָמִ֑ים הַֽכֵּר־ פָּנִ֖ים בְּמִשְׁפָּ֣ט בַּל־ טֽוֹב׃ (סח) אֹ֤מֵ֨ר ׀ לְרָשָׁע֮ צַדִּ֪יק אָ֥תָּה יִקְּבֻ֥הוּ עַמִּ֑ים יִזְעָמ֥וּהוּ לְאֻמִּֽים׃ (סט) וְלַמּוֹכִיחִ֥ים יִנְעָ֑ם וַֽ֝עֲלֵיהֶ֗ם תָּב֥וֹא בִרְכַּת־ טֽוֹב׃ (ע) שְׂפָתַ֥יִם יִשָּׁ֑ק מֵ֝שִׁ֗יב דְּבָרִ֥ים נְכֹחִֽים׃ (עא) הָ֘כֵ֤ן בַּח֨וּץ ׀ מְלַאכְתֶּ֗ךָ וְעַתְּדָ֣הּ בַּשָּׂדֶ֣ה לָ֑ךְ אַ֝חַ֗ר וּבָנִ֥יתָ בֵיתֶֽךָ׃ (עב) אַל־ תְּהִ֣י עֵד־ חִנָּ֣ם בְּרֵעֶ֑ךָ וַ֝הֲפִתִּ֗יתָ בִּשְׂפָתֶֽיךָ׃ (עג) אַל־ תֹּאמַ֗ר כַּאֲשֶׁ֣ר עָֽשָׂה־ לִ֭י כֵּ֤ן אֶֽעֱשֶׂה־ לּ֑וֹ אָשִׁ֖יב לָאִ֣ישׁ כְּפָעֳלֽוֹ׃ (עד) עַל־ שְׂדֵ֣ה אִישׁ־ עָצֵ֣ל עָבַ֑רְתִּי וְעַל־ כֶּ֝֗רֶם אָדָ֥ם חֲסַר־ לֵֽב׃ (עה) וְהִנֵּ֨ה עָ֘לָ֤ה כֻלּ֨וֹ ׀ קִמְּשֹׂנִ֗ים כָּסּ֣וּ פָנָ֣יו חֲרֻלִּ֑ים וְגֶ֖דֶר אֲבָנָ֣יו נֶהֱרָֽסָה׃ (עו) וָֽאֶחֱזֶ֣ה אָ֭נֹכִֽי אָשִׁ֣ית לִבִּ֑י רָ֝אִ֗יתִי לָקַ֥חְתִּי מוּסָֽר׃ (עז) מְעַ֣ט שֵׁ֭נוֹת מְעַ֣ט תְּנוּמ֑וֹת מְעַ֓ט ׀ חִבֻּ֖ק יָדַ֣יִם לִשְׁכָּֽב׃ (עח) וּבָֽא־ מִתְהַלֵּ֥ךְ רֵישֶׁ֑ךָ וּ֝מַחְסֹרֶ֗יךָ כְּאִ֣ישׁ מָגֵֽן׃ (עט) גַּם־ אֵ֭לֶּה מִשְׁלֵ֣י שְׁלֹמֹ֑ה אֲשֶׁ֥ר הֶ֝עְתִּ֗יקוּ אַנְשֵׁ֤י ׀ חִזְקִיָּ֬ה מֶֽלֶךְ־ יְהוּדָֽה׃ (פ) כְּבֹ֣ד אֱ֭לֹהִים הַסְתֵּ֣ר דָּבָ֑ר וּכְבֹ֥ד מְ֝לָכִ֗ים חֲקֹ֣ר דָּבָֽר׃ (פא) שָׁמַ֣יִם לָ֭רוּם וָאָ֣רֶץ לָעֹ֑מֶק וְלֵ֥ב מְ֝לָכִ֗ים אֵ֣ין חֵֽקֶר׃ (פב) הָג֣וֹ סִיגִ֣ים מִכָּ֑סֶף וַיֵּצֵ֖א לַצֹּרֵ֣ף כֶּֽלִי׃ (פג) הָג֣וֹ רָ֭שָׁע לִפְנֵי־ מֶ֑לֶךְ וְיִכּ֖וֹן בַּצֶּ֣דֶק כִּסְאֽוֹ׃ (פד) אַל־ תִּתְהַדַּ֥ר לִפְנֵי־ מֶ֑לֶךְ וּבִמְק֥וֹם גְּ֝דֹלִ֗ים אַֽל־ תַּעֲמֹֽד׃ (פה) כִּ֤י ט֥וֹב אֲמָר־ לְךָ֗ עֲ‍ֽלֵ֫ה הֵ֥נָּה מֵֽ֭הַשְׁפִּ֣ילְךָ לִפְנֵ֣י נָדִ֑יב אֲשֶׁ֖ר רָא֣וּ עֵינֶֽיךָ׃ (פו) אַל־ תֵּצֵ֥א לָרִ֗ב מַ֫הֵ֥ר פֶּ֣ן מַה־ תַּ֭עֲשֶׂה בְּאַחֲרִיתָ֑הּ בְּהַכְלִ֖ים אֹתְךָ֣ רֵעֶֽךָ׃ (פז) רִֽ֭יבְךָ רִ֣יב אֶת־ רֵעֶ֑ךָ וְס֖וֹד אַחֵ֣ר אַל־ תְּגָֽל׃ (פח) פֶּֽן־ יְחַסֶּדְךָ֥ שֹׁמֵ֑עַ וְ֝דִבָּתְךָ֗ לֹ֣א תָשֽׁוּב׃ (פט) תַּפּוּחֵ֣י זָ֭הָב בְּמַשְׂכִּיּ֥וֹת כָּ֑סֶף דָּ֝בָ֗ר דָּבֻ֥ר עַל־ אָפְנָֽיו׃ (צ) נֶ֣זֶם
