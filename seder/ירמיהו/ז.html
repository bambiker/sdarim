ירמיהו סדר ז

פתיחה באתר · ספר ירמיהו כולו ·

(א) וְהָיָ֗ה אַֽחֲרֵי֙ נָתְשִׁ֣י אוֹתָ֔ם אָשׁ֖וּב וְרִֽחַמְתִּ֑ים וַהֲשִׁבֹתִ֛ים אִ֥ישׁ לְנַחֲלָת֖וֹ וְאִ֥ישׁ לְאַרְצֽוֹ׃ (ב) וְהָיָ֡ה אִם־ לָמֹ֣ד יִלְמְדוּ֩ אֶת־ דַּֽרְכֵ֨י עַמִּ֜י לְהִשָּׁבֵ֤עַ בִּשְׁמִי֙ חַי־ יְהוָ֔ה כַּאֲשֶׁ֤ר לִמְּדוּ֙ אֶת־ עַמִּ֔י לְהִשָּׁבֵ֖עַ בַּבָּ֑עַל וְנִבְנ֖וּ בְּת֥וֹךְ עַמִּֽי׃ (ג) וְאִ֖ם לֹ֣א יִשְׁמָ֑עוּ וְנָ֨תַשְׁתִּ֜י אֶת־ הַגּ֥וֹי הַה֛וּא נָת֥וֹשׁ וְאַבֵּ֖ד נְאֻם־ יְהוָֽה׃ (ד) כֹּֽה־ אָמַ֨ר יְהוָ֜ה אֵלַ֗י הָל֞וֹךְ וְקָנִ֤יתָ לְּךָ֙ אֵז֣וֹר פִּשְׁתִּ֔ים וְשַׂמְתּ֖וֹ עַל־ מָתְנֶ֑יךָ וּבַמַּ֖יִם לֹ֥א תְבִאֵֽהוּ׃ (ה) וָאֶקְנֶ֥ה אֶת־ הָאֵז֖וֹר כִּדְבַ֣ר יְהוָ֑ה וָאָשִׂ֖ם עַל־ מָתְנָֽי׃ (ו) וַיְהִ֧י דְבַר־ יְהוָ֛ה אֵלַ֖י שֵׁנִ֥ית לֵאמֹֽר׃ (ז) קַ֧ח אֶת־ הָאֵז֛וֹר אֲשֶׁ֥ר קָנִ֖יתָ אֲשֶׁ֣ר עַל־ מָתְנֶ֑יךָ וְקוּם֙ לֵ֣ךְ פְּרָ֔תָה וְטָמְנֵ֥הוּ שָׁ֖ם בִּנְקִ֥יק הַסָּֽלַע׃ (ח) וָאֵלֵ֕ךְ וָאֶטְמְנֵ֖הוּ בִּפְרָ֑ת כַּאֲשֶׁ֛ר צִוָּ֥ה יְהוָ֖ה אוֹתִֽי׃ (ט) וַיְהִ֕י מִקֵּ֖ץ יָמִ֣ים רַבִּ֑ים וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י ק֚וּם לֵ֣ךְ פְּרָ֔תָה וְקַ֤ח מִשָּׁם֙ אֶת־ הָ֣אֵז֔וֹר אֲשֶׁ֥ר צִוִּיתִ֖יךָ לְטָמְנוֹ־ שָֽׁם׃ (י) וָאֵלֵ֣ךְ פְּרָ֔תָה וָאֶחְפֹּ֗ר וָֽאֶקַּח֙ אֶת־ הָ֣אֵז֔וֹר מִן־ הַמָּק֖וֹם אֲשֶׁר־ טְמַנְתִּ֣יו שָׁ֑מָּה וְהִנֵּה֙ נִשְׁחַ֣ת הָאֵז֔וֹר לֹ֥א יִצְלַ֖ח לַכֹּֽל׃ (יא) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יב) כֹּ֖ה אָמַ֣ר יְהוָ֑ה כָּ֠כָה אַשְׁחִ֞ית אֶת־ גְּא֧וֹן יְהוּדָ֛ה וְאֶת־ גְּא֥וֹן יְרוּשָׁלִַ֖ם הָרָֽב׃ (יג) הָעָם֩ הַזֶּ֨ה הָרָ֜ע הַֽמֵּאֲנִ֣ים ׀ לִשְׁמ֣וֹעַ אֶת־ דְּבָרַ֗י הַהֹֽלְכִים֙ בִּשְׁרִר֣וּת לִבָּ֔ם וַיֵּלְכ֗וּ אַֽחֲרֵי֙ אֱלֹהִ֣ים אֲחֵרִ֔ים לְעָבְדָ֖ם וּלְהִשְׁתַּחֲוֺ֣ת לָהֶ֑ם וִיהִי֙ כָּאֵז֣וֹר הַזֶּ֔ה אֲשֶׁ֥ר לֹא־ יִצְלַ֖ח לַכֹּֽל׃ (יד) כִּ֡י כַּאֲשֶׁר֩ יִדְבַּ֨ק הָאֵז֜וֹר אֶל־ מָתְנֵי־ אִ֗ישׁ כֵּ֣ן הִדְבַּ֣קְתִּי אֵ֠לַי אֶת־ כָּל־ בֵּ֨ית יִשְׂרָאֵ֜ל וְאֶת־ כָּל־ בֵּ֤ית יְהוּדָה֙ נְאֻם־ יְהוָ֔ה לִֽהְי֥וֹת לִי֙ לְעָ֔ם וּלְשֵׁ֥ם וְלִתְהִלָּ֖ה וּלְתִפְאָ֑רֶת וְלֹ֖א שָׁמֵֽעוּ׃ (טו) וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם אֶת־ הַדָּבָ֣ר הַזֶּ֗ה כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל כָּל־ נֵ֖בֶל יִמָּ֣לֵא יָ֑יִן וְאָמְר֣וּ אֵלֶ֔יךָ הֲיָדֹ֙עַ֙ לֹ֣א נֵדַ֔ע כִּ֥י כָל־ נֵ֖בֶל יִמָּ֥לֵא יָֽיִן׃ (טז) וְאָמַרְתָּ֨ אֲלֵיהֶ֜ם כֹּֽה־ אָמַ֣ר יְהוָ֗ה הִנְנִ֣י מְמַלֵּ֣א אֶת־ כָּל־ יֹשְׁבֵ֣י הָאָ֪רֶץ הַזֹּ֟את וְאֶת־ הַמְּלָכִ֣ים הַיֹּשְׁבִים֩ לְדָוִ֨ד עַל־ כִּסְא֜וֹ וְאֶת־ הַכֹּהֲנִ֣ים וְאֶת־ הַנְּבִיאִ֗ים וְאֵ֛ת כָּל־ יֹשְׁבֵ֥י יְרוּשָׁלִָ֖ם שִׁכָּרֽוֹן׃ (יז) וְנִפַּצְתִּים֩ אִ֨ישׁ אֶל־ אָחִ֜יו וְהָאָב֧וֹת וְהַבָּנִ֛ים יַחְדָּ֖ו נְאֻם־ יְהוָ֑ה לֹֽא־ אֶחְמ֧וֹל וְלֹֽא־ אָח֛וּס וְלֹ֥א אֲרַחֵ֖ם מֵהַשְׁחִיתָֽם׃ (יח) שִׁמְע֥וּ וְהַאֲזִ֖ינוּ אַל־ תִּגְבָּ֑הוּ כִּ֥י יְהוָ֖ה דִּבֵּֽר׃ (יט) תְּנוּ֩ לַיהוָ֨ה אֱלֹהֵיכֶ֤ם כָּבוֹד֙ בְּטֶ֣רֶם יַחְשִׁ֔ךְ וּבְטֶ֛רֶם יִֽתְנַגְּפ֥וּ רַגְלֵיכֶ֖ם עַל־ הָ֣רֵי נָ֑שֶׁף וְקִוִּיתֶ֤ם לְאוֹר֙ וְשָׂמָ֣הּ לְצַלְמָ֔וֶת (ישית) וְשִׁ֖ית לַעֲרָפֶֽל׃ (כ) וְאִם֙ לֹ֣א תִשְׁמָע֔וּהָ בְּמִסְתָּרִ֥ים תִּבְכֶּֽה־ נַפְשִׁ֖י מִפְּנֵ֣י גֵוָ֑ה וְדָמֹ֨עַ תִּדְמַ֜ע וְתֵרַ֤ד עֵינִי֙ דִּמְעָ֔ה כִּ֥י נִשְׁבָּ֖ה עֵ֥דֶר יְהוָֽה׃ (כא) אֱמֹ֥ר לַמֶּ֛לֶךְ וְלַגְּבִירָ֖ה הַשְׁפִּ֣ילוּ שֵׁ֑בוּ כִּ֤י יָרַד֙ מַרְאֲשׁ֣וֹתֵיכֶ֔ם עֲטֶ֖רֶת תִּֽפְאַרְתְּכֶֽם׃ (כב) עָרֵ֥י הַנֶּ֛גֶב סֻגְּר֖וּ וְאֵ֣ין פֹּתֵ֑חַ הָגְלָ֧ת יְהוּדָ֛ה כֻּלָּ֖הּ הָגְלָ֥ת שְׁלוֹמִֽים׃ (כג) (שאי) שְׂא֤וּ עֵֽינֵיכֶם֙ (וראי) וּרְא֔וּ הַבָּאִ֖ים מִצָּפ֑וֹן אַיֵּ֗ה הָעֵ֙דֶר֙ נִתַּן־ לָ֔ךְ צֹ֖אן תִּפְאַרְתֵּֽךְ׃ (כד) מַה־ תֹּֽאמְרִי֙ כִּֽי־ יִפְקֹ֣ד עָלַ֔יִךְ וְ֠אַתְּ לִמַּ֨דְתְּ אֹתָ֥ם עָלַ֛יִךְ אַלֻּפִ֖ים לְרֹ֑אשׁ הֲל֤וֹא חֲבָלִים֙ יֹאחֱז֔וּךְ כְּמ֖וֹ אֵ֥שֶׁת לֵדָֽה׃ (כה) וְכִ֤י תֹאמְרִי֙ בִּלְבָבֵ֔ךְ מַדּ֖וּעַ קְרָאֻ֣נִי אֵ֑לֶּה בְּרֹ֧ב עֲוֺנֵ֛ךְ נִגְל֥וּ שׁוּלַ֖יִךְ נֶחְמְס֥וּ עֲקֵבָֽיִךְ׃ (כו) הֲיַהֲפֹ֤ךְ כּוּשִׁי֙ עוֹר֔וֹ וְנָמֵ֖ר חֲבַרְבֻּרֹתָ֑יו גַּם־ אַתֶּם֙ תּוּכְל֣וּ לְהֵיטִ֔יב לִמֻּדֵ֖י הָרֵֽעַ׃ (כז) וַאֲפִיצֵ֖ם כְּקַשׁ־ עוֹבֵ֑ר לְר֖וּחַ מִדְבָּֽר׃ (כח) זֶ֣ה גוֹרָלֵ֧ךְ מְנָת־ מִדַּ֛יִךְ מֵֽאִתִּ֖י נְאֻם־ יְהוָ֑ה אֲשֶׁר֙ שָׁכַ֣חַתְּ אוֹתִ֔י וַֽתִּבְטְחִ֖י בַּשָּֽׁקֶר׃ (כט) וְגַם־ אֲנִ֛י חָשַׂ֥פְתִּי שׁוּלַ֖יִךְ עַל־ פָּנָ֑יִךְ וְנִרְאָ֖ה קְלוֹנֵֽךְ׃ (ל) נִֽאֻפַ֤יִךְ וּמִצְהֲלוֹתַ֙יִךְ֙ זִמַּ֣ת זְנוּתֵ֔ךְ עַל־ גְּבָעוֹת֙ בַּשָּׂדֶ֔ה רָאִ֖יתִי שִׁקּוּצָ֑יִךְ א֥וֹי לָךְ֙ יְר֣וּשָׁלִַ֔ם לֹ֣א תִטְהֲרִ֔י אַחֲרֵ֥י מָתַ֖י עֹֽד׃ (לא) אֲשֶׁ֨ר הָיָ֤ה דְבַר־ יְהוָה֙ אֶֽל־ יִרְמְיָ֔הוּ עַל־ דִּבְרֵ֖י הַבַּצָּרֽוֹת׃ (לב) אָבְלָ֣ה יְהוּדָ֔ה וּשְׁעָרֶ֥יהָ אֻמְלְל֖וּ קָדְר֣וּ לָאָ֑רֶץ וְצִוְחַ֥ת יְרוּשָׁלִַ֖ם עָלָֽתָה׃ (לג) וְאַדִּ֣רֵיהֶ֔ם שָׁלְח֥וּ (צעוריהם) צְעִירֵיהֶ֖ם לַמָּ֑יִם בָּ֣אוּ עַל־ גֵּבִ֞ים לֹא־ מָ֣צְאוּ מַ֗יִם שָׁ֤בוּ כְלֵיהֶם֙ רֵיקָ֔ם בֹּ֥שׁוּ וְהָכְלְמ֖וּ וְחָפ֥וּ רֹאשָֽׁם׃ (לד) בַּעֲב֤וּר הָאֲדָמָה֙ חַ֔תָּה כִּ֛י לֹא־ הָיָ֥ה גֶ֖שֶׁם בָּאָ֑רֶץ בֹּ֥שׁוּ אִכָּרִ֖ים חָפ֥וּ רֹאשָֽׁם׃ (לה) כִּ֤י גַם־ אַיֶּ֙לֶת֙ בַּשָּׂדֶ֔ה יָלְדָ֖ה וְעָז֑וֹב כִּ֥י לֹֽא־ הָיָ֖ה דֶּֽשֶׁא׃ (לו) וּפְרָאִים֙ עָמְד֣וּ עַל־ שְׁפָיִ֔ם שָׁאֲפ֥וּ ר֖וּחַ כַּתַּנִּ֑ים כָּל֥וּ עֵינֵיהֶ֖ם כִּי־ אֵ֥ין עֵֽשֶׂב׃ (לז) אִם־ עֲוֺנֵ֙ינוּ֙ עָ֣נוּ בָ֔נוּ יְהוָ֕ה עֲשֵׂ֖ה לְמַ֣עַן שְׁמֶ֑ךָ כִּֽי־ רַבּ֥וּ מְשׁוּבֹתֵ֖ינוּ לְךָ֥ חָטָֽאנוּ׃ (לח) מִקְוֵה֙ יִשְׂרָאֵ֔ל מֽוֹשִׁיע֖וֹ בְּעֵ֣ת צָרָ֑ה לָ֤מָּה תִֽהְיֶה֙ כְּגֵ֣ר בָּאָ֔רֶץ וּכְאֹרֵ֖חַ נָטָ֥ה לָלֽוּן׃ (לט) לָ֤מָּה תִֽהְיֶה֙ כְּאִ֣ישׁ נִדְהָ֔ם כְּגִבּ֖וֹר לֹא־ יוּכַ֣ל לְהוֹשִׁ֑יעַ וְאַתָּ֧ה בְקִרְבֵּ֣נוּ יְהוָ֗ה וְשִׁמְךָ֛ עָלֵ֥ינוּ נִקְרָ֖א אַל־ תַּנִּחֵֽנוּ׃ (מ) כֹּֽה־ אָמַ֨ר יְהוָ֜ה לָעָ֣ם הַזֶּ֗ה כֵּ֤ן אָֽהֲבוּ֙ לָנ֔וּעַ רַגְלֵיהֶ֖ם לֹ֣א חָשָׂ֑כוּ וַיהוָה֙ לֹ֣א רָצָ֔ם עַתָּה֙ יִזְכֹּ֣ר עֲוֺנָ֔ם וְיִפְקֹ֖ד חַטֹּאתָֽם׃ (מא) וַיֹּ֥אמֶר יְהוָ֖ה אֵלָ֑י אַל־ תִּתְפַּלֵּ֛ל בְּעַד־ הָעָ֥ם הַזֶּ֖ה לְטוֹבָֽה׃ (מב) כִּ֣י יָצֻ֗מוּ אֵינֶ֤נִּי שֹׁמֵ֙עַ֙ אֶל־ רִנָּתָ֔ם וְכִ֧י יַעֲל֛וּ עֹלָ֥ה וּמִנְחָ֖ה אֵינֶ֣נִּי רֹצָ֑ם כִּ֗י בַּחֶ֙רֶב֙ וּבָרָעָ֣ב וּבַדֶּ֔בֶר אָנֹכִ֖י מְכַלֶּ֥ה אוֹתָֽם׃ (מג) וָאֹמַ֞ר אֲהָ֣הּ ׀ אֲדֹנָ֣י יְהוִ֗ה הִנֵּ֨ה הַנְּבִאִ֜ים אֹמְרִ֤ים לָהֶם֙ לֹֽא־ תִרְא֣וּ חֶ֔רֶב וְרָעָ֖ב לֹֽא־ יִהְיֶ֣ה לָכֶ֑ם כִּֽי־ שְׁל֤וֹם אֱמֶת֙ אֶתֵּ֣ן לָכֶ֔ם בַּמָּק֖וֹם הַזֶּֽה׃ (מד) וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י שֶׁ֚קֶר הַנְּבִאִים֙ נִבְּאִ֣ים בִּשְׁמִ֔י לֹ֤א שְׁלַחְתִּים֙ וְלֹ֣א צִוִּיתִ֔ים וְלֹ֥א דִבַּ֖רְתִּי אֲלֵיהֶ֑ם חֲז֨וֹן שֶׁ֜קֶר וְקֶ֤סֶם (ואלול) וֶֽאֱלִיל֙ (ותרמות) וְתַרְמִ֣ית לִבָּ֔ם הֵ֖מָּה מִֽתְנַבְּאִ֥ים לָכֶֽם׃ (מה) לָכֵ֞ן כֹּֽה־ אָמַ֣ר יְהוָ֗ה עַֽל־ הַנְּבִאִ֞ים הַנִּבְּאִ֣ים בִּשְׁמִי֮ וַאֲנִ֣י לֹֽא־ שְׁלַחְתִּים֒ וְהֵ֙מָּה֙ אֹֽמְרִ֔ים חֶ֣רֶב וְרָעָ֔ב לֹ֥א יִהְיֶ֖ה בָּאָ֣רֶץ הַזֹּ֑את בַּחֶ֤רֶב וּבָֽרָעָב֙ יִתַּ֔מּוּ הַנְּבִאִ֖ים הָהֵֽמָּה׃ (מו) וְהָעָ֣ם אֲשֶׁר־ הֵ֣מָּה נִבְּאִ֣ים לָהֶ֡ם יִֽהְי֣וּ מֻשְׁלָכִים֩ בְּחֻצ֨וֹת יְרוּשָׁלִַ֜ם מִפְּנֵ֣י ׀ הָרָעָ֣ב וְהַחֶ֗רֶב וְאֵ֤ין מְקַבֵּר֙ לָהֵ֔מָּה הֵ֣מָּה נְשֵׁיהֶ֔ם וּבְנֵיהֶ֖ם וּבְנֹֽתֵיהֶ֑ם וְשָׁפַכְתִּ֥י עֲלֵיהֶ֖ם אֶת־ רָעָתָֽם׃ (מז) וְאָמַרְתָּ֤ אֲלֵיהֶם֙ אֶת־ הַדָּבָ֣ר הַזֶּ֔ה תֵּרַ֨דְנָה עֵינַ֥י דִּמְעָ֛ה לַ֥יְלָה וְיוֹמָ֖ם וְאַל־ תִּדְמֶ֑ינָה כִּי֩ שֶׁ֨בֶר גָּד֜וֹל נִשְׁבְּרָ֗ה בְּתוּלַת֙ בַּת־ עַמִּ֔י מַכָּ֖ה נַחְלָ֥ה מְאֹֽד׃ (מח) אִם־ יָצָ֣אתִי הַשָּׂדֶ֗ה וְהִנֵּה֙ חַלְלֵי־ חֶ֔רֶב וְאִם֙ בָּ֣אתִי הָעִ֔יר וְהִנֵּ֖ה תַּחֲלוּאֵ֣י רָעָ֑ב כִּֽי־ גַם־ נָבִ֧יא גַם־ כֹּהֵ֛ן סָחֲר֥וּ אֶל־ אֶ֖רֶץ וְלֹ֥א יָדָֽעוּ׃ (מט) הֲמָאֹ֨ס מָאַ֜סְתָּ אֶת־ יְהוּדָ֗ה אִם־ בְּצִיּוֹן֙ גָּעֲלָ֣ה נַפְשֶׁ֔ךָ מַדּ֙וּעַ֙ הִכִּיתָ֔נוּ וְאֵ֥ין לָ֖נוּ מַרְפֵּ֑א קַוֵּ֤ה לְשָׁלוֹם֙ וְאֵ֣ין ט֔וֹב וּלְעֵ֥ת מַרְפֵּ֖א וְהִנֵּ֥ה בְעָתָֽה׃ (נ) יָדַ֧עְנוּ יְהוָ֛ה רִשְׁעֵ֖נוּ עֲוֺ֣ן אֲבוֹתֵ֑ינוּ כִּ֥י חָטָ֖אנוּ לָֽךְ׃ (נא) אַל־ תִּנְאַץ֙ לְמַ֣עַן שִׁמְךָ֔ אַל־ תְּנַבֵּ֖ל כִּסֵּ֣א כְבוֹדֶ֑ךָ זְכֹ֕ר אַל־ תָּפֵ֥ר בְּרִֽיתְךָ֖ אִתָּֽנוּ׃
