ירמיהו סדר ט

פתיחה באתר · ספר ירמיהו כולו ·

(א) בָּר֣וּךְ הַגֶּ֔בֶר אֲשֶׁ֥ר יִבְטַ֖ח בַּֽיהוָ֑ה וְהָיָ֥ה יְהוָ֖ה מִבְטַחֽוֹ׃ (ב) וְהָיָ֞ה כְּעֵ֣ץ ׀ שָׁת֣וּל עַל־ מַ֗יִם וְעַל־ יוּבַל֙ יְשַׁלַּ֣ח שָֽׁרָשָׁ֔יו וְלֹ֤א (ירא) יִרְאֶה֙ כִּֽי־ יָבֹ֣א חֹ֔ם וְהָיָ֥ה עָלֵ֖הוּ רַֽעֲנָ֑ן וּבִשְׁנַ֤ת בַּצֹּ֙רֶת֙ לֹ֣א יִדְאָ֔ג וְלֹ֥א יָמִ֖ישׁ מֵעֲשׂ֥וֹת פֶּֽרִי׃ (ג) עָקֹ֥ב הַלֵּ֛ב מִכֹּ֖ל וְאָנֻ֣שׁ ה֑וּא מִ֖י יֵדָעֶֽנּוּ׃ (ד) אֲנִ֧י יְהוָ֛ה חֹקֵ֥ר לֵ֖ב בֹּחֵ֣ן כְּלָי֑וֹת וְלָתֵ֤ת לְאִישׁ֙ (כדרכו) כִּדְרָכָ֔יו כִּפְרִ֖י מַעֲלָלָֽיו׃ (ה) קֹרֵ֤א דָגַר֙ וְלֹ֣א יָלָ֔ד עֹ֥שֶׂה עֹ֖שֶׁר וְלֹ֣א בְמִשְׁפָּ֑ט בַּחֲצִ֤י (ימו) יָמָיו֙ יַעַזְבֶ֔נּוּ וּבְאַחֲרִית֖וֹ יִהְיֶ֥ה נָבָֽל׃ (ו) כִּסֵּ֣א כָב֔וֹד מָר֖וֹם מֵֽרִאשׁ֑וֹן מְק֖וֹם מִקְדָּשֵֽׁנוּ׃ (ז) מִקְוֵ֤ה יִשְׂרָאֵל֙ יְהוָ֔ה כָּל־ עֹזְבֶ֖יךָ יֵבֹ֑שׁוּ (יסורי) וְסוּרַי֙ בָּאָ֣רֶץ יִכָּתֵ֔בוּ כִּ֥י עָזְב֛וּ מְק֥וֹר מַֽיִם־ חַיִּ֖ים אֶת־ יְהוָֽה׃ (ח) רְפָאֵ֤נִי יְהוָה֙ וְאֵ֣רָפֵ֔א הוֹשִׁיעֵ֖נִי וְאִוָּשֵׁ֑עָה כִּ֥י תְהִלָּתִ֖י אָֽתָּה׃ (ט) הִנֵּה־ הֵ֕מָּה אֹמְרִ֖ים אֵלָ֑י אַיֵּ֥ה דְבַר־ יְהוָ֖ה יָ֥בוֹא נָֽא׃ (י) וַאֲנִ֞י לֹא־ אַ֣צְתִּי ׀ מֵרֹעֶ֣ה אַחֲרֶ֗יךָ וְי֥וֹם אָנ֛וּשׁ לֹ֥א הִתְאַוֵּ֖יתִי אַתָּ֣ה יָדָ֑עְתָּ מוֹצָ֣א שְׂפָתַ֔י נֹ֥כַח פָּנֶ֖יךָ הָיָֽה׃ (יא) אַל־ תִּֽהְיֵה־ לִ֖י לִמְחִתָּ֑ה מַֽחֲסִי־ אַ֖תָּה בְּי֥וֹם רָעָֽה׃ (יב) יֵבֹ֤שׁוּ רֹדְפַי֙ וְאַל־ אֵבֹ֣שָׁה אָ֔נִי יֵחַ֣תּוּ הֵ֔מָּה וְאַל־ אֵחַ֖תָּה אָ֑נִי הָבִ֤יא עֲלֵיהֶם֙ י֣וֹם רָעָ֔ה וּמִשְׁנֶ֥ה שִׁבָּר֖וֹן שָׁבְרֵֽם׃ (יג) כֹּה־ אָמַ֨ר יְהוָ֜ה אֵלַ֗י הָלֹ֤ךְ וְעָֽמַדְתָּ֙ בְּשַׁ֣עַר בְּנֵֽי־ (עם) הָעָ֔ם אֲשֶׁ֨ר יָבֹ֤אוּ בוֹ֙ מַלְכֵ֣י יְהוּדָ֔ה וַאֲשֶׁ֖ר יֵ֣צְאוּ ב֑וֹ וּבְכֹ֖ל שַׁעֲרֵ֥י יְרוּשָׁלִָֽם׃ (יד) וְאָמַרְתָּ֣ אֲ֠לֵיהֶם שִׁמְע֨וּ דְבַר־ יְהוָ֜ה מַלְכֵ֤י יְהוּדָה֙ וְכָל־ יְהוּדָ֔ה וְכֹ֖ל יֹשְׁבֵ֣י יְרוּשָׁלִָ֑ם הַבָּאִ֖ים בַּשְּׁעָרִ֥ים הָאֵֽלֶּה׃ (טו) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִשָּׁמְר֖וּ בְּנַפְשֽׁוֹתֵיכֶ֑ם וְאַל־ תִּשְׂא֤וּ מַשָּׂא֙ בְּי֣וֹם הַשַּׁבָּ֔ת וַהֲבֵאתֶ֖ם בְּשַׁעֲרֵ֥י יְרוּשָׁלִָֽם׃ (טז) וְלֹא־ תוֹצִ֨יאוּ מַשָּׂ֤א מִבָּֽתֵּיכֶם֙ בְּי֣וֹם הַשַּׁבָּ֔ת וְכָל־ מְלָאכָ֖ה לֹ֣א תַֽעֲשׂ֑וּ וְקִדַּשְׁתֶּם֙ אֶת־ י֣וֹם הַשַּׁבָּ֔ת כַּאֲשֶׁ֥ר צִוִּ֖יתִי אֶת־ אֲבוֹתֵיכֶֽם׃ (יז) וְלֹ֣א שָֽׁמְע֔וּ וְלֹ֥א הִטּ֖וּ אֶת־ אָזְנָ֑ם וַיַּקְשׁוּ֙ אֶת־ עָרְפָּ֔ם לְבִלְתִּ֣י (שומע) שְׁמ֔וֹעַ וּלְבִלְתִּ֖י קַ֥חַת מוּסָֽר׃ (יח) וְ֠הָיָה אִם־ שָׁמֹ֨עַ תִּשְׁמְע֤וּן אֵלַי֙ נְאֻם־ יְהוָ֔ה לְבִלְתִּ֣י ׀ הָבִ֣יא מַשָּׂ֗א בְּשַׁעֲרֵ֛י הָעִ֥יר הַזֹּ֖את בְּי֣וֹם הַשַּׁבָּ֑ת וּלְקַדֵּשׁ֙ אֶת־ י֣וֹם הַשַּׁבָּ֔ת לְבִלְתִּ֥י עֲשֽׂוֹת־ (בה) בּ֖וֹ כָּל־ מְלָאכָֽה׃ (יט) וּבָ֣אוּ בְשַׁעֲרֵ֣י הָעִ֣יר הַזֹּ֡את מְלָכִ֣ים ׀ וְשָׂרִ֡ים יֹשְׁבִים֩ עַל־ כִּסֵּ֨א דָוִ֜ד רֹכְבִ֣ים ׀ בָּרֶ֣כֶב וּבַסּוּסִ֗ים הֵ֚מָּה וְשָׂ֣רֵיהֶ֔ם אִ֥ישׁ יְהוּדָ֖ה וְיֹשְׁבֵ֣י יְרוּשָׁלִָ֑ם וְיָשְׁבָ֥ה הָֽעִיר־ הַזֹּ֖את לְעוֹלָֽם׃ (כ) וּבָ֣אוּ מֵעָרֵֽי־ יְ֠הוּדָה וּמִסְּבִיב֨וֹת יְרוּשָׁלִַ֜ם וּמֵאֶ֣רֶץ בִּנְיָמִ֗ן וּמִן־ הַשְּׁפֵלָ֤ה וּמִן־ הָהָר֙ וּמִן־ הַנֶּ֔גֶב מְבִאִ֛ים עוֹלָ֥ה וְזֶ֖בַח וּמִנְחָ֣ה וּלְבוֹנָ֑ה וּמְבִאֵ֥י תוֹדָ֖ה בֵּ֥ית יְהוָֽה׃ (כא) וְאִם־ לֹ֨א תִשְׁמְע֜וּ אֵלַ֗י לְקַדֵּשׁ֙ אֶת־ י֣וֹם הַשַּׁבָּ֔ת וּלְבִלְתִּ֣י ׀ שְׂאֵ֣ת מַשָּׂ֗א וּבֹ֛א בְּשַׁעֲרֵ֥י יְרוּשָׁלִַ֖ם בְּי֣וֹם הַשַּׁבָּ֑ת וְהִצַּ֧תִּי אֵ֣שׁ בִּשְׁעָרֶ֗יהָ וְאָֽכְלָ֛ה אַרְמְנ֥וֹת יְרוּשָׁלִַ֖ם וְלֹ֥א תִכְבֶּֽה׃ (כב) הַדָּבָר֙ אֲשֶׁ֣ר הָיָ֣ה אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (כג) ק֥וּם וְיָרַדְתָּ֖ בֵּ֣ית הַיּוֹצֵ֑ר וְשָׁ֖מָּה אַשְׁמִֽיעֲךָ֥ אֶת־ דְּבָרָֽי׃ (כד) וָאֵרֵ֖ד בֵּ֣ית הַיּוֹצֵ֑ר (והנהו) וְהִנֵּה־ ה֛וּא עֹשֶׂ֥ה מְלָאכָ֖ה עַל־ הָאָבְנָֽיִם׃ (כה) וְנִשְׁחַ֣ת הַכְּלִ֗י אֲשֶׁ֨ר ה֥וּא עֹשֶׂ֛ה בַּחֹ֖מֶר בְּיַ֣ד הַיּוֹצֵ֑ר וְשָׁ֗ב וַֽיַּעֲשֵׂ֙הוּ֙ כְּלִ֣י אַחֵ֔ר כַּאֲשֶׁ֥ר יָשַׁ֛ר בְּעֵינֵ֥י הַיּוֹצֵ֖ר לַעֲשֽׂוֹת׃ (כו) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֽוֹר׃ (כז) הֲכַיּוֹצֵ֨ר הַזֶּ֜ה לֹא־ אוּכַ֨ל לַעֲשׂ֥וֹת לָכֶ֛ם בֵּ֥ית יִשְׂרָאֵ֖ל נְאֻם־ יְהוָ֑ה הִנֵּ֤ה כַחֹ֙מֶר֙ בְּיַ֣ד הַיּוֹצֵ֔ר כֵּן־ אַתֶּ֥ם בְּיָדִ֖י בֵּ֥ית יִשְׂרָאֵֽל׃ (כח) רֶ֣גַע אֲדַבֵּ֔ר עַל־ גּ֖וֹי וְעַל־ מַמְלָכָ֑ה לִנְת֥וֹשׁ וְלִנְת֖וֹץ וּֽלְהַאֲבִֽיד׃ (כט) וְשָׁב֙ הַגּ֣וֹי הַה֔וּא מֵרָ֣עָת֔וֹ אֲשֶׁ֥ר דִּבַּ֖רְתִּי עָלָ֑יו וְנִֽחַמְתִּי֙ עַל־ הָ֣רָעָ֔ה אֲשֶׁ֥ר חָשַׁ֖בְתִּי לַעֲשׂ֥וֹת לֽוֹ׃ (ל) וְרֶ֣גַע אֲדַבֵּ֔ר עַל־ גּ֖וֹי וְעַל־ מַמְלָכָ֑ה לִבְנֹ֖ת וְלִנְטֹֽעַ׃ (לא) וְעָשָׂ֤ה (הרעה) הָרַע֙ בְּעֵינַ֔י לְבִלְתִּ֖י שְׁמֹ֣עַ בְּקוֹלִ֑י וְנִֽחַמְתִּי֙ עַל־ הַטּוֹבָ֔ה אֲשֶׁ֥ר אָמַ֖רְתִּי לְהֵיטִ֥יב אוֹתֽוֹ׃ (לב) וְעַתָּ֡ה אֱמָר־ נָ֣א אֶל־ אִישׁ־ יְהוּדָה֩ וְעַל־ יוֹשְׁבֵ֨י יְרוּשָׁלִַ֜ם לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנֵּ֨ה אָנֹכִ֜י יוֹצֵ֤ר עֲלֵיכֶם֙ רָעָ֔ה וְחֹשֵׁ֥ב עֲלֵיכֶ֖ם מַֽחֲשָׁבָ֑ה שׁ֣וּבוּ נָ֗א אִ֚ישׁ מִדַּרְכּ֣וֹ הָֽרָעָ֔ה וְהֵיטִ֥יבוּ דַרְכֵיכֶ֖ם וּמַעַלְלֵיכֶֽם׃ (לג) וְאָמְר֖וּ נוֹאָ֑שׁ כִּֽי־ אַחֲרֵ֤י מַחְשְׁבוֹתֵ֙ינוּ֙ נֵלֵ֔ךְ וְאִ֛ישׁ שְׁרִר֥וּת לִבּֽוֹ־ הָרָ֖ע נַעֲשֶֽׂה׃ (לד) לָכֵ֗ן כֹּ֚ה אָמַ֣ר יְהוָ֔ה שַֽׁאֲלוּ־ נָא֙ בַּגּוֹיִ֔ם מִ֥י שָׁמַ֖ע כָּאֵ֑לֶּה שַֽׁעֲרֻרִת֙ עָשְׂתָ֣ה מְאֹ֔ד בְּתוּלַ֖ת יִשְׂרָאֵֽל׃ (לה) הֲיַעֲזֹ֥ב מִצּ֛וּר שָׂדַ֖י שֶׁ֣לֶג לְבָנ֑וֹן אִם־ יִנָּתְשׁ֗וּ מַ֛יִם זָרִ֥ים קָרִ֖ים נוֹזְלִֽים׃ (לו) כִּֽי־ שְׁכֵחֻ֥נִי עַמִּ֖י לַשָּׁ֣וְא יְקַטֵּ֑רוּ וַיַּכְשִׁל֤וּם בְּדַרְכֵיהֶם֙ שְׁבִילֵ֣י עוֹלָ֔ם לָלֶ֣כֶת נְתִיב֔וֹת דֶּ֖רֶךְ לֹ֥א סְלוּלָֽה׃ (לז) לָשׂ֥וּם אַרְצָ֛ם לְשַׁמָּ֖ה (שרוקת) שְׁרִיק֣וֹת עוֹלָ֑ם כֹּ֚ל עוֹבֵ֣ר עָלֶ֔יהָ יִשֹּׁ֖ם וְיָנִ֥יד בְּרֹאשֽׁוֹ׃ (לח) כְּרֽוּחַ־ קָדִ֥ים אֲפִיצֵ֖ם לִפְנֵ֣י אוֹיֵ֑ב עֹ֧רֶף וְלֹֽא־ פָנִ֛ים אֶרְאֵ֖ם בְּי֥וֹם אֵידָֽם׃ (לט) וַיֹּאמְר֗וּ לְכ֨וּ וְנַחְשְׁבָ֣ה עַֽל־ יִרְמְיָהוּ֮ מַחֲשָׁבוֹת֒ כִּי֩ לֹא־ תֹאבַ֨ד תּוֹרָ֜ה מִכֹּהֵ֗ן וְעֵצָה֙ מֵֽחָכָ֔ם וְדָבָ֖ר מִנָּבִ֑יא לְכוּ֙ וְנַכֵּ֣הוּ בַלָּשׁ֔וֹן וְאַל־ נַקְשִׁ֖יבָה אֶל־ כָּל־ דְּבָרָֽיו׃
