ירמיהו סדר ו

פתיחה באתר · ספר ירמיהו כולו ·

(א) כִּ֣י אִם־ בְּזֹ֞את יִתְהַלֵּ֣ל הַמִּתְהַלֵּ֗ל הַשְׂכֵּל֮ וְיָדֹ֣עַ אוֹתִי֒ כִּ֚י אֲנִ֣י יְהוָ֔ה עֹ֥שֶׂה חֶ֛סֶד מִשְׁפָּ֥ט וּצְדָקָ֖ה בָּאָ֑רֶץ כִּֽי־ בְאֵ֥לֶּה חָפַ֖צְתִּי נְאֻם־ יְהוָֽה׃ (ב) הִנֵּ֛ה יָמִ֥ים בָּאִ֖ים נְאֻם־ יְהוָ֑ה וּפָ֣קַדְתִּ֔י עַל־ כָּל־ מ֖וּל בְּעָרְלָֽה׃ (ג) עַל־ מִצְרַ֣יִם וְעַל־ יְהוּדָ֗ה וְעַל־ אֱד֞וֹם וְעַל־ בְּנֵ֤י עַמּוֹן֙ וְעַל־ מוֹאָ֔ב וְעַל֙ כָּל־ קְצוּצֵ֣י פֵאָ֔ה הַיֹּשְׁבִ֖ים בַּמִּדְבָּ֑ר כִּ֤י כָל־ הַגּוֹיִם֙ עֲרֵלִ֔ים וְכָל־ בֵּ֥ית יִשְׂרָאֵ֖ל עַרְלֵי־ לֵֽב׃ (ד) שִׁמְע֣וּ אֶת־ הַדָּבָ֗ר אֲשֶׁ֨ר דִּבֶּ֧ר יְהוָ֛ה עֲלֵיכֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל׃ (ה) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה אֶל־ דֶּ֤רֶךְ הַגּוֹיִם֙ אַל־ תִּלְמָ֔דוּ וּמֵאֹת֥וֹת הַשָּׁמַ֖יִם אַל־ תֵּחָ֑תּוּ כִּֽי־ יֵחַ֥תּוּ הַגּוֹיִ֖ם מֵהֵֽמָּה׃ (ו) כִּֽי־ חֻקּ֥וֹת הָֽעַמִּ֖ים הֶ֣בֶל ה֑וּא כִּֽי־ עֵץ֙ מִיַּ֣עַר כְּרָת֔וֹ מַעֲשֵׂ֥ה יְדֵ֥י־ חָרָ֖שׁ בַּֽמַּעֲצָֽד׃ (ז) בְּכֶ֥סֶף וּבְזָהָ֖ב יְיַפֵּ֑הוּ בְּמַסְמְר֧וֹת וּבְמַקָּב֛וֹת יְחַזְּק֖וּם וְל֥וֹא יָפִֽיק׃ (ח) כְּתֹ֨מֶר מִקְשָׁ֥ה הֵ֙מָּה֙ וְלֹ֣א יְדַבֵּ֔רוּ נָשׂ֥וֹא יִנָּשׂ֖וּא כִּ֣י לֹ֣א יִצְעָ֑דוּ אַל־ תִּֽירְא֤וּ מֵהֶם֙ כִּי־ לֹ֣א יָרֵ֔עוּ וְגַם־ הֵיטֵ֖יב אֵ֥ין אוֹתָֽם׃ (ט) מֵאֵ֥ין כָּמ֖וֹךָ יְהוָ֑ה גָּד֥וֹל אַתָּ֛ה וְגָד֥וֹל שִׁמְךָ֖ בִּגְבוּרָֽה׃ (י) מִ֣י לֹ֤א יִֽרָאֲךָ֙ מֶ֣לֶךְ הַגּוֹיִ֔ם כִּ֥י לְךָ֖ יָאָ֑תָה כִּ֣י בְכָל־ חַכְמֵ֧י הַגּוֹיִ֛ם וּבְכָל־ מַלְכוּתָ֖ם מֵאֵ֥ין כָּמֽוֹךָ׃ (יא) וּבְאַחַ֖ת יִבְעֲר֣וּ וְיִכְסָ֑לוּ מוּסַ֥ר הֲבָלִ֖ים עֵ֥ץ הֽוּא׃ (יב) כֶּ֣סֶף מְרֻקָּ֞ע מִתַּרְשִׁ֣ישׁ יוּבָ֗א וְזָהָב֙ מֵֽאוּפָ֔ז מַעֲשֵׂ֥ה חָרָ֖שׁ וִידֵ֣י צוֹרֵ֑ף תְּכֵ֤לֶת וְאַרְגָּמָן֙ לְבוּשָׁ֔ם מַעֲשֵׂ֥ה חֲכָמִ֖ים כֻּלָּֽם׃ (יג) וַֽיהוָ֤ה אֱלֹהִים֙ אֱמֶ֔ת הֽוּא־ אֱלֹהִ֥ים חַיִּ֖ים וּמֶ֣לֶךְ עוֹלָ֑ם מִקִּצְפּוֹ֙ תִּרְעַ֣שׁ הָאָ֔רֶץ וְלֹֽא־ יָכִ֥לוּ גוֹיִ֖ם זַעְמֽוֹ׃ (יד) כִּדְנָה֙ תֵּאמְר֣וּן לְה֔וֹם אֱלָ֣הַיָּ֔א דִּֽי־ שְׁמַיָּ֥א וְאַרְקָ֖א לָ֣א עֲבַ֑דוּ יֵאבַ֧דוּ מֵֽאַרְעָ֛א וּמִן־ תְּח֥וֹת שְׁמַיָּ֖א אֵֽלֶּה׃ (טו) עֹשֵׂ֥ה אֶ֙רֶץ֙ בְּכֹח֔וֹ מֵכִ֥ין תֵּבֵ֖ל בְּחָכְמָת֑וֹ וּבִתְבוּנָת֖וֹ נָטָ֥ה שָׁמָֽיִם׃ (טז) לְק֨וֹל תִּתּ֜וֹ הֲמ֥וֹן מַ֙יִם֙ בַּשָּׁמַ֔יִם וַיַּעֲלֶ֥ה נְשִׂאִ֖ים מִקְצֵ֣ה (ארץ) הָאָ֑רֶץ בְּרָקִ֤ים לַמָּטָר֙ עָשָׂ֔ה וַיּ֥וֹצֵא ר֖וּחַ מֵאֹצְרֹתָֽיו׃ (יז) נִבְעַ֤ר כָּל־ אָדָם֙ מִדַּ֔עַת הֹבִ֥ישׁ כָּל־ צוֹרֵ֖ף מִפָּ֑סֶל כִּ֛י שֶׁ֥קֶר נִסְכּ֖וֹ וְלֹא־ ר֥וּחַ בָּֽם׃ (יח) הֶ֣בֶל הֵ֔מָּה מַעֲשֵׂ֖ה תַּעְתֻּעִ֑ים בְּעֵ֥ת פְּקֻדָּתָ֖ם יֹאבֵֽדוּ׃ (יט) לֹֽא־ כְאֵ֜לֶּה חֵ֣לֶק יַעֲקֹ֗ב כִּֽי־ יוֹצֵ֤ר הַכֹּל֙ ה֔וּא וְיִ֨שְׂרָאֵ֔ל שֵׁ֖בֶט נַֽחֲלָת֑וֹ יְהוָ֥ה צְבָא֖וֹת שְׁמֽוֹ׃ (כ) אִסְפִּ֥י מֵאֶ֖רֶץ כִּנְעָתֵ֑ךְ (ישבתי) יֹשֶׁ֖בֶת בַּמָּצֽוֹר׃ (כא) כִּֽי־ כֹה֙ אָמַ֣ר יְהוָ֔ה הִנְנִ֥י קוֹלֵ֛עַ אֶת־ יוֹשְׁבֵ֥י הָאָ֖רֶץ בַּפַּ֣עַם הַזֹּ֑את וַהֲצֵר֥וֹתִי לָהֶ֖ם לְמַ֥עַן יִמְצָֽאוּ׃ (כב) א֥וֹי לִי֙ עַל־ שִׁבְרִ֔י נַחְלָ֖ה מַכָּתִ֑י וַאֲנִ֣י אָמַ֔רְתִּי אַ֛ךְ זֶ֥ה חֳלִ֖י וְאֶשָּׂאֶֽנּוּ׃ (כג) אָהֳלִ֣י שֻׁדָּ֔ד וְכָל־ מֵיתָרַ֖י נִתָּ֑קוּ בָּנַ֤י יְצָאֻ֙נִי֙ וְאֵינָ֔ם אֵין־ נֹטֶ֥ה עוֹד֙ אָהֳלִ֔י וּמֵקִ֖ים יְרִיעוֹתָֽי׃ (כד) כִּ֤י נִבְעֲרוּ֙ הָֽרֹעִ֔ים וְאֶת־ יְהוָ֖ה לֹ֣א דָרָ֑שׁוּ עַל־ כֵּן֙ לֹ֣א הִשְׂכִּ֔ילוּ וְכָל־ מַרְעִיתָ֖ם נָפֽוֹצָה׃ (כה) ק֤וֹל שְׁמוּעָה֙ הִנֵּ֣ה בָאָ֔ה וְרַ֥עַשׁ גָּד֖וֹל מֵאֶ֣רֶץ צָפ֑וֹן לָשׂ֞וּם אֶת־ עָרֵ֧י יְהוּדָ֛ה שְׁמָמָ֖ה מְע֥וֹן תַּנִּֽים׃ (כו) יָדַ֣עְתִּי יְהוָ֔ה כִּ֛י לֹ֥א לָאָדָ֖ם דַּרְכּ֑וֹ לֹֽא־ לְאִ֣ישׁ הֹלֵ֔ךְ וְהָכִ֖ין אֶֽת־ צַעֲדֽוֹ׃ (כז) יַסְּרֵ֥נִי יְהוָ֖ה אַךְ־ בְּמִשְׁפָּ֑ט אַל־ בְּאַפְּךָ֖ פֶּן־ תַּמְעִטֵֽנִי׃ (כח) שְׁפֹ֣ךְ חֲמָתְךָ֗ עַל־ הַגּוֹיִם֙ אֲשֶׁ֣ר לֹֽא־ יְדָע֔וּךָ וְעַל֙ מִשְׁפָּח֔וֹת אֲשֶׁ֥ר בְּשִׁמְךָ֖ לֹ֣א קָרָ֑אוּ כִּֽי־ אָכְל֣וּ אֶֽת־ יַעֲקֹ֗ב וַאֲכָלֻ֙הוּ֙ וַיְכַלֻּ֔הוּ וְאֶת־ נָוֵ֖הוּ הֵשַֽׁמּוּ׃ (כט) הַדָּבָר֙ אֲשֶׁ֣ר הָיָ֣ה אֶֽל־ יִרְמְיָ֔הוּ מֵאֵ֥ת יְהוָ֖ה לֵאמֹֽר׃ (ל) שִׁמְע֕וּ אֶת־ דִּבְרֵ֖י הַבְּרִ֣ית הַזֹּ֑את וְדִבַּרְתָּם֙ אֶל־ אִ֣ישׁ יְהוּדָ֔ה וְעַל־ יֹשְׁבֵ֖י יְרוּשָׁלִָֽם׃ (לא) וְאָמַרְתָּ֣ אֲלֵיהֶ֔ם כֹּֽה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל אָר֣וּר הָאִ֔ישׁ אֲשֶׁר֙ לֹ֣א יִשְׁמַ֔ע אֶת־ דִּבְרֵ֖י הַבְּרִ֥ית הַזֹּֽאת׃ (לב) אֲשֶׁ֣ר צִוִּ֣יתִי אֶת־ אֲבֽוֹתֵיכֶ֡ם בְּי֣וֹם הוֹצִיאִֽי־ אוֹתָ֣ם מֵאֶֽרֶץ־ מִצְרַיִם֩ מִכּ֨וּר הַבַּרְזֶ֜ל לֵאמֹ֗ר שִׁמְע֤וּ בְקוֹלִי֙ וַעֲשִׂיתֶ֣ם אוֹתָ֔ם כְּכֹ֥ל אֲשֶׁר־ אֲצַוֶּ֖ה אֶתְכֶ֑ם וִהְיִ֤יתֶם לִי֙ לְעָ֔ם וְאָ֣נֹכִ֔י אֶהְיֶ֥ה לָכֶ֖ם לֵאלֹהִֽים׃ (לג) לְמַעַן֩ הָקִ֨ים אֶת־ הַשְּׁבוּעָ֜ה אֲשֶׁר־ נִשְׁבַּ֣עְתִּי לַאֲבֽוֹתֵיכֶ֗ם לָתֵ֤ת לָהֶם֙ אֶ֣רֶץ זָבַ֥ת חָלָ֛ב וּדְבַ֖שׁ כַּיּ֣וֹם הַזֶּ֑ה וָאַ֥עַן וָאֹמַ֖ר אָמֵ֥ן ׀ יְהוָֽה׃ (לד) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י קְרָ֨א אֶת־ כָּל־ הַדְּבָרִ֤ים הָאֵ֙לֶּה֙ בְּעָרֵ֣י יְהוּדָ֔ה וּבְחֻצ֥וֹת יְרוּשָׁלִַ֖ם לֵאמֹ֑ר שִׁמְע֗וּ אֶת־ דִּבְרֵי֙ הַבְּרִ֣ית הַזֹּ֔את וַעֲשִׂיתֶ֖ם אוֹתָֽם׃ (לה) כִּי֩ הָעֵ֨ד הַעִדֹ֜תִי בַּאֲבֽוֹתֵיכֶ֗ם בְּיוֹם֩ הַעֲלוֹתִ֨י אוֹתָ֜ם מֵאֶ֤רֶץ מִצְרַ֙יִם֙ וְעַד־ הַיּ֣וֹם הַזֶּ֔ה הַשְׁכֵּ֥ם וְהָעֵ֖ד לֵאמֹ֑ר שִׁמְע֖וּ בְּקוֹלִֽי׃ (לו) וְלֹ֤א שָֽׁמְעוּ֙ וְלֹֽא־ הִטּ֣וּ אֶת־ אָזְנָ֔ם וַיֵּ֣לְכ֔וּ אִ֕ישׁ בִּשְׁרִיר֖וּת לִבָּ֣ם הָרָ֑ע וָאָבִ֨יא עֲלֵיהֶ֜ם אֶֽת־ כָּל־ דִּבְרֵ֧י הַבְּרִית־ הַזֹּ֛את אֲשֶׁר־ צִוִּ֥יתִי לַעֲשׂ֖וֹת וְלֹ֥א עָשֽׂוּ׃ (לז) וַיֹּ֥אמֶר יְהוָ֖ה אֵלָ֑י נִֽמְצָא־ קֶ֙שֶׁר֙ בְּאִ֣ישׁ יְהוּדָ֔ה וּבְיֹשְׁבֵ֖י יְרוּשָׁלִָֽם׃ (לח) שָׁ֩בוּ֩ עַל־ עֲוֺנֹ֨ת אֲבוֹתָ֜ם הָרִֽאשֹׁנִ֗ים אֲשֶׁ֤ר מֵֽאֲנוּ֙ לִשְׁמ֣וֹעַ אֶת־ דְּבָרַ֔י וְהֵ֣מָּה הָלְכ֗וּ אַחֲרֵ֛י אֱלֹהִ֥ים אֲחֵרִ֖ים לְעָבְדָ֑ם הֵפֵ֤רוּ בֵֽית־ יִשְׂרָאֵל֙ וּבֵ֣ית יְהוּדָ֔ה אֶת־ בְּרִיתִ֕י אֲשֶׁ֥ר כָּרַ֖תִּי אֶת־ אֲבוֹתָֽם׃ (לט) לָכֵ֗ן כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֨י מֵבִ֤יא אֲלֵיהֶם֙ רָעָ֔ה אֲשֶׁ֥ר לֹֽא־ יוּכְל֖וּ לָצֵ֣את מִמֶּ֑נָּה וְזָעֲק֣וּ אֵלַ֔י וְלֹ֥א אֶשְׁמַ֖ע אֲלֵיהֶֽם׃ (מ) וְהָֽלְכ֞וּ עָרֵ֣י יְהוּדָ֗ה וְיֹשְׁבֵי֙ יְר֣וּשָׁלִַ֔ם וְזָֽעֲקוּ֙ אֶל־ הָ֣אֱלֹהִ֔ים אֲשֶׁ֛ר הֵ֥ם מְקַטְּרִ֖ים לָהֶ֑ם וְהוֹשֵׁ֛עַ לֹֽא־ יוֹשִׁ֥יעוּ לָהֶ֖ם בְּעֵ֥ת רָעָתָֽם׃ (מא) כִּ֚י מִסְפַּ֣ר עָרֶ֔יךָ הָי֥וּ אֱלֹהֶ֖יךָ יְהוּדָ֑ה וּמִסְפַּ֞ר חֻצ֣וֹת יְרוּשָׁלִַ֗ם שַׂמְתֶּ֤ם מִזְבְּחוֹת֙ לַבֹּ֔שֶׁת מִזְבְּח֖וֹת לְקַטֵּ֥ר לַבָּֽעַל׃ (מב) וְאַתָּ֗ה אַל־ תִּתְפַּלֵּל֙ בְּעַד־ הָעָ֣ם הַזֶּ֔ה וְאַל־ תִּשָּׂ֥א בַעֲדָ֖ם רִנָּ֣ה וּתְפִלָּ֑ה כִּ֣י אֵינֶ֣נִּי שֹׁמֵ֗עַ בְּעֵ֛ת קָרְאָ֥ם אֵלַ֖י בְּעַ֥ד רָעָתָֽם׃ (מג) מֶ֣ה לִֽידִידִ֞י בְּבֵיתִ֗י עֲשׂוֹתָ֤הּ הַֽמְזִמָּ֙תָה֙ הָֽרַבִּ֔ים וּבְשַׂר־ קֹ֖דֶשׁ יַעַבְר֣וּ מֵֽעָלָ֑יִךְ כִּ֥י רָעָתֵ֖כִי אָ֥ז תַּעֲלֹֽזִי׃ (מד) זַ֤יִת רַֽעֲנָן֙ יְפֵ֣ה פְרִי־ תֹ֔אַר קָרָ֥א יְהוָ֖ה שְׁמֵ֑ךְ לְק֣וֹל ׀ הֲמוּלָּ֣ה גְדֹלָ֗ה הִצִּ֥ית אֵשׁ֙ עָלֶ֔יהָ וְרָע֖וּ דָּלִיּוֹתָֽיו׃ (מה) וַיהוָ֤ה צְבָאוֹת֙ הַנּוֹטֵ֣עַ אוֹתָ֔ךְ דִּבֶּ֥ר עָלַ֖יִךְ רָעָ֑ה בִּ֠גְלַל רָעַ֨ת בֵּֽית־ יִשְׂרָאֵ֜ל וּבֵ֣ית יְהוּדָ֗ה אֲשֶׁ֨ר עָשׂ֥וּ לָהֶ֛ם לְהַכְעִסֵ֖נִי לְקַטֵּ֥ר לַבָּֽעַל׃ (מו) וַֽיהוָ֥ה הֽוֹדִיעַ֖נִי וָֽאֵדָ֑עָה אָ֖ז הִרְאִיתַ֥נִי מַעַלְלֵיהֶֽם׃ (מז) וַאֲנִ֕י כְּכֶ֥בֶשׂ אַלּ֖וּף יוּבַ֣ל לִטְב֑וֹחַ וְלֹֽא־ יָדַ֜עְתִּי כִּֽי־ עָלַ֣י ׀ חָשְׁב֣וּ מַחֲשָׁב֗וֹת נַשְׁחִ֨יתָה עֵ֤ץ בְּלַחְמוֹ֙ וְנִכְרְתֶ֙נּוּ֙ מֵאֶ֣רֶץ חַיִּ֔ים וּשְׁמ֖וֹ לֹֽא־ יִזָּכֵ֥ר עֽוֹד׃ (מח) וַֽיהוָ֤ה צְבָאוֹת֙ שֹׁפֵ֣ט צֶ֔דֶק בֹּחֵ֥ן כְּלָי֖וֹת וָלֵ֑ב אֶרְאֶ֤ה נִקְמָֽתְךָ֙ מֵהֶ֔ם כִּ֥י אֵלֶ֖יךָ גִּלִּ֥יתִי אֶת־ רִיבִֽי׃ (מט) לָכֵ֗ן כֹּֽה־ אָמַ֤ר יְהוָה֙ עַל־ אַנְשֵׁ֣י עֲנָת֔וֹת הַֽמְבַקְשִׁ֥ים אֶֽת־ נַפְשְׁךָ֖ לֵאמֹ֑ר לֹ֤א תִנָּבֵא֙ בְּשֵׁ֣ם יְהוָ֔ה וְלֹ֥א תָמ֖וּת בְּיָדֵֽנוּ׃ (נ) לָכֵ֗ן כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת הִנְנִ֥י פֹקֵ֖ד עֲלֵיהֶ֑ם הַבַּֽחוּרִים֙ יָמֻ֣תוּ בַחֶ֔רֶב בְּנֵיהֶם֙ וּבְנ֣וֹתֵיהֶ֔ם יָמֻ֖תוּ בָּרָעָֽב׃ (נא) וּשְׁאֵרִ֕ית לֹ֥א תִֽהְיֶ֖ה לָהֶ֑ם כִּֽי־ אָבִ֥יא רָעָ֛ה אֶל־ אַנְשֵׁ֥י עֲנָת֖וֹת שְׁנַ֥ת פְּקֻדָּתָֽם׃ (נב) צַדִּ֤יק אַתָּה֙ יְהוָ֔ה כִּ֥י אָרִ֖יב אֵלֶ֑יךָ אַ֤ךְ מִשְׁפָּטִים֙ אֲדַבֵּ֣ר אוֹתָ֔ךְ מַדּ֗וּעַ דֶּ֤רֶךְ רְשָׁעִים֙ צָלֵ֔חָה שָׁל֖וּ כָּל־ בֹּ֥גְדֵי בָֽגֶד׃ (נג) נְטַעְתָּם֙ גַּם־ שֹׁרָ֔שׁוּ יֵלְכ֖וּ גַּם־ עָ֣שׂוּ פֶ֑רִי קָר֤וֹב אַתָּה֙ בְּפִיהֶ֔ם וְרָח֖וֹק מִכִּלְיוֹתֵיהֶֽם׃ (נד) וְאַתָּ֤ה יְהוָה֙ יְדַעְתָּ֔נִי תִּרְאֵ֕נִי וּבָחַנְתָּ֥ לִבִּ֖י אִתָּ֑ךְ הַתִּקֵם֙ כְּצֹ֣אן לְטִבְחָ֔ה וְהַקְדִּשֵׁ֖ם לְי֥וֹם הֲרֵגָֽה׃ (נה) עַד־ מָתַי֙ תֶּאֱבַ֣ל הָאָ֔רֶץ וְעֵ֥שֶׂב כָּל־ הַשָּׂדֶ֖ה יִיבָ֑שׁ מֵרָעַ֣ת יֹֽשְׁבֵי־ בָ֗הּ סָפְתָ֤ה בְהֵמוֹת֙ וָע֔וֹף כִּ֣י אָמְר֔וּ לֹ֥א יִרְאֶ֖ה אֶת־ אַחֲרִיתֵֽנוּ׃ (נו) כִּ֣י אֶת־ רַגְלִ֥ים ׀ רַ֙צְתָּה֙ וַיַּלְא֔וּךָ וְאֵ֥יךְ תְּתַֽחֲרֶ֖ה אֶת־ הַסּוּסִ֑ים וּבְאֶ֤רֶץ שָׁלוֹם֙ אַתָּ֣ה בוֹטֵ֔חַ וְאֵ֥יךְ תַּעֲשֶׂ֖ה בִּגְא֥וֹן הַיַּרְדֵּֽן׃ (נז) כִּ֧י גַם־ אַחֶ֣יךָ וּבֵית־ אָבִ֗יךָ גַּם־ הֵ֙מָּה֙ בָּ֣גְדוּ בָ֔ךְ גַּם־ הֵ֛מָּה קָרְא֥וּ אַחֲרֶ֖יךָ מָלֵ֑א אַל־ תַּאֲמֵ֣ן בָּ֔ם כִּֽי־ יְדַבְּר֥וּ אֵלֶ֖יךָ טוֹבֽוֹת׃ (נח) עָזַ֙בְתִּי֙ אֶת־ בֵּיתִ֔י נָטַ֖שְׁתִּי אֶת־ נַחֲלָתִ֑י נָתַ֛תִּי אֶת־ יְדִד֥וּת נַפְשִׁ֖י בְּכַ֥ף אֹיְבֶֽיהָ׃ (נט) הָיְתָה־ לִּ֥י נַחֲלָתִ֖י כְּאַרְיֵ֣ה בַיָּ֑עַר נָתְנָ֥ה עָלַ֛י בְּקוֹלָ֖הּ עַל־ כֵּ֥ן שְׂנֵאתִֽיהָ׃ (ס) הַעַ֨יִט צָב֤וּעַ נַחֲלָתִי֙ לִ֔י הַעַ֖יִט סָבִ֣יב עָלֶ֑יהָ לְכ֗וּ אִסְפ֛וּ כָּל־ חַיַּ֥ת הַשָּׂדֶ֖ה הֵתָ֥יוּ לְאָכְלָֽה׃ (סא) רֹעִ֤ים רַבִּים֙ שִֽׁחֲת֣וּ כַרְמִ֔י בֹּסְס֖וּ אֶת־ חֶלְקָתִ֑י נָֽתְנ֛וּ אֶת־ חֶלְקַ֥ת חֶמְדָּתִ֖י לְמִדְבַּ֥ר שְׁמָמָֽה׃ (סב) שָׂמָהּ֙ לִשְׁמָמָ֔ה אָבְלָ֥ה עָלַ֖י שְׁמֵמָ֑ה נָשַׁ֙מָּה֙ כָּל־ הָאָ֔רֶץ כִּ֛י אֵ֥ין אִ֖ישׁ שָׂ֥ם עַל־ לֵֽב׃ (סג) עַֽל־ כָּל־ שְׁפָיִ֣ם בַּמִּדְבָּ֗ר בָּ֚אוּ שֹֽׁדְדִ֔ים כִּ֣י חֶ֤רֶב לַֽיהוָה֙ אֹֽכְלָ֔ה מִקְצֵה־ אֶ֖רֶץ וְעַד־ קְצֵ֣ה הָאָ֑רֶץ אֵ֥ין שָׁל֖וֹם לְכָל־ בָּשָֽׂר׃ (סד) זָרְע֤וּ חִטִּים֙ וְקֹצִ֣ים קָצָ֔רוּ נֶחְל֖וּ לֹ֣א יוֹעִ֑לוּ וּבֹ֙שׁוּ֙ מִתְּבוּאֹ֣תֵיכֶ֔ם מֵחֲר֖וֹן אַף־ יְהוָֽה׃ (סה) כֹּ֣ה ׀ אָמַ֣ר יְהוָ֗ה עַל־ כָּל־ שְׁכֵנַי֙ הָֽרָעִ֔ים הַנֹּֽגְעִים֙ בַּֽנַּחֲלָ֔ה אֲשֶׁר־ הִנְחַ֥לְתִּי אֶת־ עַמִּ֖י אֶת־ יִשְׂרָאֵ֑ל הִנְנִ֤י נֹֽתְשָׁם֙ מֵעַ֣ל אַדְמָתָ֔ם וְאֶת־ בֵּ֥ית יְהוּדָ֖ה אֶתּ֥וֹשׁ מִתּוֹכָֽם׃
