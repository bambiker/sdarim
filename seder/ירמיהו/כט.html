ירמיהו סדר כט

פתיחה באתר · ספר ירמיהו כולו ·

(א) צִיּ֣וֹן יִשְׁאָ֔לוּ דֶּ֖רֶךְ הֵ֣נָּה פְנֵיהֶ֑ם בֹּ֚אוּ וְנִלְו֣וּ אֶל־ יְהוָ֔ה בְּרִ֥ית עוֹלָ֖ם לֹ֥א תִשָּׁכֵֽחַ׃ (ב) צֹ֤אן אֹֽבְדוֹת֙ (היה) הָי֣וּ עַמִּ֔י רֹעֵיהֶ֣ם הִתְע֔וּם הָרִ֖ים (שובבים) שֽׁוֹבְב֑וּם מֵהַ֤ר אֶל־ גִּבְעָה֙ הָלָ֔כוּ שָׁכְח֖וּ רִבְצָֽם׃ (ג) כָּל־ מוֹצְאֵיהֶ֣ם אֲכָל֔וּם וְצָרֵיהֶ֥ם אָמְר֖וּ לֹ֣א נֶאְשָׁ֑ם תַּ֗חַת אֲשֶׁ֨ר חָטְא֤וּ לַֽיהוָה֙ נְוֵה־ צֶ֔דֶק וּמִקְוֵ֥ה אֲבֽוֹתֵיהֶ֖ם יְהוָֽה׃ (ד) נֻ֚דוּ מִתּ֣וֹךְ בָּבֶ֔ל וּמֵאֶ֥רֶץ כַּשְׂדִּ֖ים (יצאו) צֵ֑אוּ וִהְי֕וּ כְּעַתּוּדִ֖ים לִפְנֵי־ צֹֽאן׃ (ה) כִּ֣י הִנֵּ֣ה אָנֹכִ֡י מֵעִיר֩ וּמַעֲלֶ֨ה עַל־ בָּבֶ֜ל קְהַל־ גּוֹיִ֤ם גְּדֹלִים֙ מֵאֶ֣רֶץ צָפ֔וֹן וְעָ֣רְכוּ לָ֔הּ מִשָּׁ֖ם תִּלָּכֵ֑ד חִצָּיו֙ כְּגִבּ֣וֹר מַשְׁכִּ֔יל לֹ֥א יָשׁ֖וּב רֵיקָֽם׃ (ו) וְהָיְתָ֥ה כַשְׂדִּ֖ים לְשָׁלָ֑ל כָּל־ שֹׁלְלֶ֥יהָ יִשְׂבָּ֖עוּ נְאֻם־ יְהוָֽה׃ (ז) כִּ֤י (תשמחי) תִשְׂמְחוּ֙ כִּ֣י (תעלזי) תַֽעֲלְז֔וּ שֹׁסֵ֖י נַחֲלָתִ֑י כִּ֤י (תפושי) תָפ֙וּשׁוּ֙ כְּעֶגְלָ֣ה דָשָׁ֔ה (ותצהלי) וְתִצְהֲל֖וּ כָּאֲבִּרִֽים׃ (ח) בּ֤וֹשָׁה אִמְּכֶם֙ מְאֹ֔ד חָפְרָ֖ה יֽוֹלַדְתְּכֶ֑ם הִנֵּה֙ אַחֲרִ֣ית גּוֹיִ֔ם מִדְבָּ֖ר צִיָּ֥ה וַעֲרָבָֽה׃ (ט) מִקֶּ֤צֶף יְהוָה֙ לֹ֣א תֵשֵׁ֔ב וְהָיְתָ֥ה שְׁמָמָ֖ה כֻּלָּ֑הּ כֹּ֚ל עֹבֵ֣ר עַל־ בָּבֶ֔ל יִשֹּׁ֥ם וְיִשְׁרֹ֖ק עַל־ כָּל־ מַכּוֹתֶֽיהָ׃ (י) עִרְכ֨וּ עַל־ בָּבֶ֤ל ׀ סָבִיב֙ כָּל־ דֹּ֣רְכֵי קֶ֔שֶׁת יְד֣וּ אֵלֶ֔יהָ אַֽל־ תַּחְמְל֖וּ אֶל־ חֵ֑ץ כִּ֥י לַֽיהוָ֖ה חָטָֽאָה׃ (יא) הָרִ֨יעוּ עָלֶ֤יהָ סָבִיב֙ נָתְנָ֣ה יָדָ֔הּ נָֽפְלוּ֙ (אשויתיה) אָשְׁיוֹתֶ֔יהָ נֶהֶרְס֖וּ חֽוֹמוֹתֶ֑יהָ כִּי֩ נִקְמַ֨ת יְהוָ֥ה הִיא֙ הִנָּ֣קְמוּ בָ֔הּ כַּאֲשֶׁ֥ר עָשְׂתָ֖ה עֲשׂוּ־ לָֽהּ׃ (יב) כִּרְת֤וּ זוֹרֵ֙עַ֙ מִבָּבֶ֔ל וְתֹפֵ֥שׂ מַגָּ֖ל בְּעֵ֣ת קָצִ֑יר מִפְּנֵי֙ חֶ֣רֶב הַיּוֹנָ֔ה אִ֤ישׁ אֶל־ עַמּוֹ֙ יִפְנ֔וּ וְאִ֥ישׁ לְאַרְצ֖וֹ יָנֻֽסוּ׃ (יג) שֶׂ֧ה פְזוּרָ֛ה יִשְׂרָאֵ֖ל אֲרָי֣וֹת הִדִּ֑יחוּ הָרִאשׁ֤וֹן אֲכָלוֹ֙ מֶ֣לֶךְ אַשּׁ֔וּר וְזֶ֤ה הָאַחֲרוֹן֙ עִצְּמ֔וֹ נְבוּכַדְרֶאצַּ֖ר מֶ֥לֶךְ בָּבֶֽל׃ (יד) לָכֵ֗ן כֹּֽה־ אָמַ֞ר יְהוָ֤ה צְבָאוֹת֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הִנְנִ֥י פֹקֵ֛ד אֶל־ מֶ֥לֶךְ בָּבֶ֖ל וְאֶל־ אַרְצ֑וֹ כַּאֲשֶׁ֥ר פָּקַ֖דְתִּי אֶל־ מֶ֥לֶךְ אַשּֽׁוּר׃ (טו) וְשֹׁבַבְתִּ֤י אֶת־ יִשְׂרָאֵל֙ אֶל־ נָוֵ֔הוּ וְרָעָ֥ה הַכַּרְמֶ֖ל וְהַבָּשָׁ֑ן וּבְהַ֥ר אֶפְרַ֛יִם וְהַגִּלְעָ֖ד תִּשְׂבַּ֥ע נַפְשֽׁוֹ׃ (טז) בַּיָּמִ֣ים הָהֵם֩ וּבָעֵ֨ת הַהִ֜יא נְאֻם־ יְהוָ֗ה יְבֻקַּ֞שׁ אֶת־ עֲוֺ֤ן יִשְׂרָאֵל֙ וְאֵינֶ֔נּוּ וְאֶת־ חַטֹּ֥את יְהוּדָ֖ה וְלֹ֣א תִמָּצֶ֑אינָה כִּ֥י אֶסְלַ֖ח לַאֲשֶׁ֥ר אַשְׁאִֽיר׃ (יז) עַל־ הָאָ֤רֶץ מְרָתַ֙יִם֙ עֲלֵ֣ה עָלֶ֔יהָ וְאֶל־ יוֹשְׁבֵ֖י פְּק֑וֹד חֲרֹ֨ב וְהַחֲרֵ֤ם אַֽחֲרֵיהֶם֙ נְאֻם־ יְהוָ֔ה וַעֲשֵׂ֕ה כְּכֹ֖ל אֲשֶׁ֥ר צִוִּיתִֽיךָ׃ (יח) ק֥וֹל מִלְחָמָ֖ה בָּאָ֑רֶץ וְשֶׁ֖בֶר גָּדֽוֹל׃ (יט) אֵ֤יךְ נִגְדַּע֙ וַיִּשָּׁבֵ֔ר פַּטִּ֖ישׁ כָּל־ הָאָ֑רֶץ אֵ֣יךְ הָיְתָ֧ה לְשַׁמָּ֛ה בָּבֶ֖ל בַּגּוֹיִֽם׃ (כ) יָקֹ֨שְׁתִּי לָ֤ךְ וְגַם־ נִלְכַּדְתְּ֙ בָּבֶ֔ל וְאַ֖תְּ לֹ֣א יָדָ֑עַתְּ נִמְצֵאת֙ וְגַם־ נִתְפַּ֔שְׂתְּ כִּ֥י בַֽיהוָ֖ה הִתְגָּרִֽית׃ (כא) פָּתַ֤ח יְהוָה֙ אֶת־ א֣וֹצָר֔וֹ וַיּוֹצֵ֖א אֶת־ כְּלֵ֣י זַעְמ֑וֹ כִּי־ מְלָאכָ֣ה הִ֗יא לַֽאדֹנָ֧י יְהוִ֛ה צְבָא֖וֹת בְּאֶ֥רֶץ כַּשְׂדִּֽים׃ (כב) בֹּֽאוּ־ לָ֤הּ מִקֵּץ֙ פִּתְח֣וּ מַאֲבֻסֶ֔יהָ סָלּ֥וּהָ כְמוֹ־ עֲרֵמִ֖ים וְהַחֲרִימ֑וּהָ אַל־ תְּהִי־ לָ֖הּ שְׁאֵרִֽית׃ (כג) חִרְבוּ֙ כָּל־ פָּרֶ֔יהָ יֵרְד֖וּ לַטָּ֑בַח ה֣וֹי עֲלֵיהֶ֔ם כִּֽי־ בָ֥א יוֹמָ֖ם עֵ֥ת פְּקֻדָּתָֽם׃ (כד) ק֥וֹל נָסִ֛ים וּפְלֵטִ֖ים מֵאֶ֣רֶץ בָּבֶ֑ל לְהַגִּ֣יד בְּצִיּ֗וֹן אֶת־ נִקְמַת֙ יְהוָ֣ה אֱלֹהֵ֔ינוּ נִקְמַ֖ת הֵיכָלֽוֹ׃ (כה) הַשְׁמִ֣יעוּ אֶל־ בָּבֶ֣ל ׀ רַ֠בִּים כָּל־ דֹּ֨רְכֵי קֶ֜שֶׁת חֲנ֧וּ עָלֶ֣יהָ סָבִ֗יב אַל־ יְהִי־ לָהּ֙ פְּלֵטָ֔ה שַׁלְּמוּ־ לָ֣הּ כְּפָעֳלָ֔הּ כְּכֹ֛ל אֲשֶׁ֥ר עָשְׂתָ֖ה עֲשׂוּ־ לָ֑הּ כִּ֧י אֶל־ יְהוָ֛ה זָ֖דָה אֶל־ קְד֥וֹשׁ יִשְׂרָאֵֽל׃ (כו) לָכֵ֛ן יִפְּל֥וּ בַחוּרֶ֖יהָ בִּרְחֹבֹתֶ֑יהָ וְכָל־ אַנְשֵׁ֨י מִלְחַמְתָּ֥הּ יִדַּ֛מּוּ בַּיּ֥וֹם הַה֖וּא נְאֻם־ יְהוָֽה׃ (כז) הִנְנִ֤י אֵלֶ֙יךָ֙ זָד֔וֹן נְאֻם־ אֲדֹנָ֥י יְהוִ֖ה צְבָא֑וֹת כִּ֛י בָּ֥א יוֹמְךָ֖ עֵ֥ת פְּקַדְתִּֽיךָ׃ (כח) וְכָשַׁ֤ל זָדוֹן֙ וְנָפַ֔ל וְאֵ֥ין ל֖וֹ מֵקִ֑ים וְהִצַּ֤תִּי אֵשׁ֙ בְּעָרָ֔יו וְאָכְלָ֖ה כָּל־ סְבִיבֹתָֽיו׃ (כט) כֹּ֤ה אָמַר֙ יְהוָ֣ה צְבָא֔וֹת עֲשׁוּקִ֛ים בְּנֵי־ יִשְׂרָאֵ֥ל וּבְנֵי־ יְהוּדָ֖ה יַחְדָּ֑ו וְכָל־ שֹֽׁבֵיהֶם֙ הֶחֱזִ֣יקוּ בָ֔ם מֵאֲנ֖וּ שַׁלְּחָֽם׃ (ל) גֹּאֲלָ֣ם ׀ חָזָ֗ק יְהוָ֤ה צְבָאוֹת֙ שְׁמ֔וֹ רִ֥יב יָרִ֖יב אֶת־ רִיבָ֑ם לְמַ֙עַן֙ הִרְגִּ֣יעַ אֶת־ הָאָ֔רֶץ וְהִרְגִּ֖יז לְיֹשְׁבֵ֥י בָבֶֽל׃ (לא) חֶ֥רֶב עַל־ כַּשְׂדִּ֖ים נְאֻם־ יְהוָ֑ה וְאֶל־ יֹשְׁבֵ֣י בָבֶ֔ל וְאֶל־ שָׂרֶ֖יהָ וְאֶל־ חֲכָמֶֽיהָ׃ (לב) חֶ֥רֶב אֶל־ הַבַּדִּ֖ים וְנֹאָ֑לוּ חֶ֥רֶב אֶל־ גִּבּוֹרֶ֖יהָ וָחָֽתּוּ׃ (לג) חֶ֜רֶב אֶל־ סוּסָ֣יו וְאֶל־ רִכְבּ֗וֹ וְאֶל־ כָּל־ הָעֶ֛רֶב אֲשֶׁ֥ר בְּתוֹכָ֖הּ וְהָי֣וּ לְנָשִׁ֑ים חֶ֥רֶב אֶל־ אוֹצְרֹתֶ֖יהָ וּבֻזָּֽזוּ׃ (לד) חֹ֥רֶב אֶל־ מֵימֶ֖יהָ וְיָבֵ֑שׁוּ כִּ֣י אֶ֤רֶץ פְּסִלִים֙ הִ֔יא וּבָאֵימִ֖ים יִתְהֹלָֽלוּ׃ (לה) לָכֵ֗ן יֵשְׁב֤וּ צִיִּים֙ אֶת־ אִיִּ֔ים וְיָ֥שְׁבוּ בָ֖הּ בְּנ֣וֹת יַֽעֲנָ֑ה וְלֹֽא־ תֵשֵׁ֥ב עוֹד֙ לָנֶ֔צַח וְלֹ֥א תִשְׁכּ֖וֹן עַד־ דּ֥וֹר וָדֽוֹר׃ (לו) כְּמַהְפֵּכַ֨ת אֱלֹהִ֜ים אֶת־ סְדֹ֧ם וְאֶת־ עֲמֹרָ֛ה וְאֶת־ שְׁכֵנֶ֖יהָ נְאֻם־ יְהוָ֑ה לֹֽא־ יֵשֵׁ֥ב שָׁם֙ אִ֔ישׁ וְלֹֽא־ יָג֥וּר בָּ֖הּ בֶּן־ אָדָֽם׃ (לז) הִנֵּ֛ה עַ֥ם בָּ֖א מִצָּפ֑וֹן וְג֤וֹי גָּדוֹל֙ וּמְלָכִ֣ים רַבִּ֔ים יֵעֹ֖רוּ מִיַּרְכְּתֵי־ אָֽרֶץ׃ (לח) קֶ֣שֶׁת וְכִידֹ֞ן יַחֲזִ֗יקוּ אַכְזָרִ֥י הֵ֙מָּה֙ וְלֹ֣א יְרַחֵ֔מוּ קוֹלָם֙ כַּיָּ֣ם יֶהֱמֶ֔ה וְעַל־ סוּסִ֖ים יִרְכָּ֑בוּ עָר֗וּךְ כְּאִישׁ֙ לַמִּלְחָמָ֔ה עָלַ֖יִךְ בַּת־ בָּבֶֽל׃ (לט) שָׁמַ֧ע מֶֽלֶךְ־ בָּבֶ֛ל אֶת־ שִׁמְעָ֖ם וְרָפ֣וּ יָדָ֑יו צָרָה֙ הֶחֱזִיקַ֔תְהוּ חִ֖יל כַּיּוֹלֵדָֽה׃ (מ) הִ֠נֵּה כְּאַרְיֵ֞ה יַעֲלֶ֨ה מִגְּא֣וֹן הַיַּרְדֵּן֮ אֶל־ נְוֵ֣ה אֵיתָן֒ כִּֽי־ אַרְגִּ֤עָה (ארוצם) אֲרִיצֵם֙ מֵֽעָלֶ֔יהָ וּמִ֥י בָח֖וּר אֵלֶ֣יהָ אֶפְקֹ֑ד כִּ֣י מִ֤י כָמ֙וֹנִי֙ וּמִ֣י יוֹעִדֶ֔נִּי וּמִֽי־ זֶ֣ה רֹעֶ֔ה אֲשֶׁ֥ר יַעֲמֹ֖ד לְפָנָֽי׃ (מא) לָכֵ֞ן שִׁמְע֣וּ עֲצַת־ יְהוָ֗ה אֲשֶׁ֤ר יָעַץ֙ אֶל־ בָּבֶ֔ל וּמַ֨חְשְׁבוֹתָ֔יו אֲשֶׁ֥ר חָשַׁ֖ב אֶל־ אֶ֣רֶץ כַּשְׂדִּ֑ים אִם־ לֹ֤א יִסְחָבוּם֙ צְעִירֵ֣י הַצֹּ֔אן אִם־ לֹ֥א יַשִּׁ֛ים עֲלֵיהֶ֖ם נָוֶֽה׃ (מב) מִקּוֹל֙ נִתְפְּשָׂ֣ה בָבֶ֔ל נִרְעֲשָׁ֖ה הָאָ֑רֶץ וּזְעָקָ֖ה בַּגּוֹיִ֥ם נִשְׁמָֽע׃ (מג) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִי֙ מֵעִ֣יר עַל־ בָּבֶ֔ל וְאֶל־ יֹשְׁבֵ֖י לֵ֣ב קָמָ֑י ר֖וּחַ מַשְׁחִֽית׃ (מד) וְשִׁלַּחְתִּ֨י לְבָבֶ֤ל ׀ זָרִים֙ וְזֵר֔וּהָ וִיבֹקְק֖וּ אֶת־ אַרְצָ֑הּ כִּֽי־ הָי֥וּ עָלֶ֛יהָ מִסָּבִ֖יב בְּי֥וֹם רָעָֽה׃ (מה) אֶֽל־ יִדְרֹ֤ךְ (ידרך) הַדֹּרֵךְ֙ קַשְׁתּ֔וֹ וְאֶל־ יִתְעַ֖ל בְּסִרְיֹנ֑וֹ וְאַֽל־ תַּחְמְלוּ֙ אֶל־ בַּ֣חֻרֶ֔יהָ הַחֲרִ֖ימוּ כָּל־ צְבָאָֽהּ׃ (מו) וְנָפְל֥וּ חֲלָלִ֖ים בְּאֶ֣רֶץ כַּשְׂדִּ֑ים וּמְדֻקָּרִ֖ים בְּחוּצוֹתֶֽיהָ׃ (מז) כִּ֠י לֹֽא־ אַלְמָ֨ן יִשְׂרָאֵ֤ל וִֽיהוּדָה֙ מֵֽאֱלֹהָ֔יו מֵֽיְהוָ֖ה צְבָא֑וֹת כִּ֤י אַרְצָם֙ מָלְאָ֣ה אָשָׁ֔ם מִקְּד֖וֹשׁ יִשְׂרָאֵֽל׃ (מח) נֻ֣סוּ ׀ מִתּ֣וֹךְ בָּבֶ֗ל וּמַלְּטוּ֙ אִ֣ישׁ נַפְשׁ֔וֹ אַל־ תִּדַּ֖מּוּ בַּעֲוֺנָ֑הּ כִּי֩ עֵ֨ת נְקָמָ֥ה הִיא֙ לַֽיהוָ֔ה גְּמ֕וּל ה֥וּא מְשַׁלֵּ֖ם לָֽהּ׃ (מט) כּוֹס־ זָהָ֤ב בָּבֶל֙ בְּיַד־ יְהוָ֔ה מְשַׁכֶּ֖רֶת כָּל־ הָאָ֑רֶץ מִיֵּינָהּ֙ שָׁת֣וּ גוֹיִ֔ם עַל־ כֵּ֖ן יִתְהֹלְל֥וּ גוֹיִֽם׃ (נ) פִּתְאֹ֛ם נָפְלָ֥ה בָבֶ֖ל וַתִּשָּׁבֵ֑ר הֵילִ֣ילוּ עָלֶ֗יהָ קְח֤וּ צֳרִי֙ לְמַכְאוֹבָ֔הּ אוּלַ֖י תֵּרָפֵֽא׃ (נא) (רפאנו) רִפִּ֣ינוּ אֶת־ בָּבֶל֙ וְלֹ֣א נִרְפָּ֔תָה עִזְב֕וּהָ וְנֵלֵ֖ךְ אִ֣ישׁ לְאַרְצ֑וֹ כִּֽי־ נָגַ֤ע אֶל־ הַשָּׁמַ֙יִם֙ מִשְׁפָּטָ֔הּ וְנִשָּׂ֖א עַד־ שְׁחָקִֽים׃
