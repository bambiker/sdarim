ירמיהו סדר ב

פתיחה באתר · ספר ירמיהו כולו ·

(א) הֲל֣וֹא מֵעַ֔תָּה (קראתי) קָרָ֥את לִ֖י אָבִ֑י אַלּ֥וּף נְעֻרַ֖י אָֽתָּה׃ (ב) הֲיִנְטֹ֣ר לְעוֹלָ֔ם אִם־ יִשְׁמֹ֖ר לָנֶ֑צַח הִנֵּ֥ה (דברתי) דִבַּ֛רְתְּ וַתַּעֲשִׂ֥י הָרָע֖וֹת וַתּוּכָֽל׃ (ג) וַיֹּ֨אמֶר יְהוָ֜ה אֵלַ֗י בִּימֵי֙ יֹאשִׁיָּ֣הוּ הַמֶּ֔לֶךְ הֲ‍ֽרָאִ֔יתָ אֲשֶׁ֥ר עָשְׂתָ֖ה מְשֻׁבָ֣ה יִשְׂרָאֵ֑ל הֹלְכָ֨ה הִ֜יא עַל־ כָּל־ הַ֣ר גָּבֹ֗הַּ וְאֶל־ תַּ֛חַת כָּל־ עֵ֥ץ רַעֲנָ֖ן וַתִּזְנִי־ שָֽׁם׃ (ד) וָאֹמַ֗ר אַחֲרֵ֨י עֲשׂוֹתָ֧הּ אֶת־ כָּל־ אֵ֛לֶּה אֵלַ֥י תָּשׁ֖וּב וְלֹא־ שָׁ֑בָה (ותראה) וַתֵּ֛רֶא בָּגוֹדָ֥ה אֲחוֹתָ֖הּ יְהוּדָֽה׃ (ה) וָאֵ֗רֶא כִּ֤י עַל־ כָּל־ אֹדוֹת֙ אֲשֶׁ֤ר נִֽאֲפָה֙ מְשֻׁבָ֣ה יִשְׂרָאֵ֔ל שִׁלַּחְתִּ֕יהָ וָאֶתֵּ֛ן אֶת־ סֵ֥פֶר כְּרִיתֻתֶ֖יהָ אֵלֶ֑יהָ וְלֹ֨א יָֽרְאָ֜ה בֹּֽגֵדָ֤ה יְהוּדָה֙ אֲחוֹתָ֔הּ וַתֵּ֖לֶךְ וַתִּ֥זֶן גַּם־ הִֽיא׃ (ו) וְהָיָה֙ מִקֹּ֣ל זְנוּתָ֔הּ וַתֶּחֱנַ֖ף אֶת־ הָאָ֑רֶץ וַתִּנְאַ֥ף אֶת־ הָאֶ֖בֶן וְאֶת־ הָעֵֽץ׃ (ז) וְגַם־ בְּכָל־ זֹ֗את לֹא־ שָׁ֨בָה אֵלַ֜י בָּגוֹדָ֧ה אֲחוֹתָ֛הּ יְהוּדָ֖ה בְּכָל־ לִבָּ֑הּ כִּ֥י אִם־ בְּשֶׁ֖קֶר נְאֻם־ יְהוָֽה׃ (ח) וַיֹּ֤אמֶר יְהוָה֙ אֵלַ֔י צִדְּקָ֥ה נַפְשָׁ֖הּ מְשֻׁבָ֣ה יִשְׂרָאֵ֑ל מִבֹּגֵדָ֖ה יְהוּדָֽה׃ (ט) הָלֹ֡ךְ וְקָֽרָאתָ֩ אֶת־ הַדְּבָרִ֨ים הָאֵ֜לֶּה צָפ֗וֹנָה וְ֠אָמַרְתָּ שׁ֣וּבָה מְשֻׁבָ֤ה יִשְׂרָאֵל֙ נְאֻם־ יְהוָ֔ה לֽוֹא־ אַפִּ֥יל פָּנַ֖י בָּכֶ֑ם כִּֽי־ חָסִ֤יד אֲנִי֙ נְאֻם־ יְהוָ֔ה לֹ֥א אֶטּ֖וֹר לְעוֹלָֽם׃ (י) אַ֚ךְ דְּעִ֣י עֲוֺנֵ֔ךְ כִּ֛י בַּיהוָ֥ה אֱלֹהַ֖יִךְ פָּשָׁ֑עַתְּ וַתְּפַזְּרִ֨י אֶת־ דְּרָכַ֜יִךְ לַזָּרִ֗ים תַּ֚חַת כָּל־ עֵ֣ץ רַעֲנָ֔ן וּבְקוֹלִ֥י לֹא־ שְׁמַעְתֶּ֖ם נְאֻם־ יְהֹוָֽה׃ (יא) שׁ֣וּבוּ בָנִ֤ים שׁוֹבָבִים֙ נְאֻם־ יְהוָ֔ה כִּ֥י אָנֹכִ֖י בָּעַ֣לְתִּי בָכֶ֑ם וְלָקַחְתִּ֨י אֶתְכֶ֜ם אֶחָ֣ד מֵעִ֗יר וּשְׁנַ֙יִם֙ מִמִּשְׁפָּחָ֔ה וְהֵבֵאתִ֥י אֶתְכֶ֖ם צִיּֽוֹן׃ (יב) וְנָתַתִּ֥י לָכֶ֛ם רֹעִ֖ים כְּלִבִּ֑י וְרָע֥וּ אֶתְכֶ֖ם דֵּעָ֥ה וְהַשְׂכֵּֽיל׃ (יג) וְהָיָ֡ה כִּ֣י תִרְבּוּ֩ וּפְרִיתֶ֨ם בָּאָ֜רֶץ בַּיָּמִ֤ים הָהֵ֙מָּה֙ נְאֻם־ יְהוָ֔ה לֹא־ יֹ֣אמְרוּ ע֗וֹד אֲרוֹן֙ בְּרִית־ יְהוָ֔ה וְלֹ֥א יַעֲלֶ֖ה עַל־ לֵ֑ב וְלֹ֤א יִזְכְּרוּ־ בוֹ֙ וְלֹ֣א יִפְקֹ֔דוּ וְלֹ֥א יֵעָשֶׂ֖ה עֽוֹד׃ (יד) בָּעֵ֣ת הַהִ֗יא יִקְרְא֤וּ לִירוּשָׁלִַ֙ם֙ כִּסֵּ֣א יְהוָ֔ה וְנִקְוּ֨וּ אֵלֶ֧יהָ כָֽל־ הַגּוֹיִ֛ם לְשֵׁ֥ם יְהוָ֖ה לִירוּשָׁלִָ֑ם וְלֹא־ יֵלְכ֣וּ ע֔וֹד אַחֲרֵ֕י שְׁרִר֖וּת לִבָּ֥ם הָרָֽע׃ (טו) בַּיָּמִ֣ים הָהֵ֔מָּה יֵלְכ֥וּ בֵית־ יְהוּדָ֖ה עַל־ בֵּ֣ית יִשְׂרָאֵ֑ל וְיָבֹ֤אוּ יַחְדָּו֙ מֵאֶ֣רֶץ צָפ֔וֹן עַל־ הָאָ֕רֶץ אֲשֶׁ֥ר הִנְחַ֖לְתִּי אֶת־ אֲבוֹתֵיכֶֽם׃ (טז) וְאָנֹכִ֣י אָמַ֗רְתִּי אֵ֚יךְ אֲשִׁיתֵ֣ךְ בַּבָּנִ֔ים וְאֶתֶּן־ לָךְ֙ אֶ֣רֶץ חֶמְדָּ֔ה נַחֲלַ֥ת צְבִ֖י צִבְא֣וֹת גּוֹיִ֑ם וָאֹמַ֗ר אָבִי֙ (תקראו)־ תִּקְרְאִי־ לִ֔י וּמֵאַחֲרַ֖י לֹ֥א (תשובו) תָשֽׁוּבִי׃ (יז) אָכֵ֛ן בָּגְדָ֥ה אִשָּׁ֖ה מֵרֵעָ֑הּ כֵּ֣ן בְּגַדְתֶּ֥ם בִּ֛י בֵּ֥ית יִשְׂרָאֵ֖ל נְאֻם־ יְהוָֽה׃ (יח) ק֚וֹל עַל־ שְׁפָיִ֣ים נִשְׁמָ֔ע בְּכִ֥י תַחֲנוּנֵ֖י בְּנֵ֣י יִשְׂרָאֵ֑ל כִּ֤י הֶעֱוּוּ֙ אֶת־ דַּרְכָּ֔ם שָׁכְח֖וּ אֶת־ יְהֹוָ֥ה אֱלֹהֵיהֶֽם׃ (יט) שׁ֚וּבוּ בָּנִ֣ים שׁוֹבָבִ֔ים אֶרְפָּ֖ה מְשׁוּבֹֽתֵיכֶ֑ם הִנְנוּ֙ אָתָ֣נוּ לָ֔ךְ כִּ֥י אַתָּ֖ה יְהֹוָ֥ה אֱלֹהֵֽינוּ׃ (כ) אָכֵ֥ן לַשֶּׁ֛קֶר מִגְּבָע֖וֹת הָמ֣וֹן הָרִ֑ים אָכֵן֙ בַּיהֹוָ֣ה אֱלֹהֵ֔ינוּ תְּשׁוּעַ֖ת יִשְׂרָאֵֽל׃ (כא) וְהַבֹּ֗שֶׁת אָֽכְלָ֛ה אֶת־ יְגִ֥יעַ אֲבוֹתֵ֖ינוּ מִנְּעוּרֵ֑ינוּ אֶת־ צֹאנָם֙ וְאֶת־ בְּקָרָ֔ם אֶת־ בְּנֵיהֶ֖ם וְאֶת־ בְּנוֹתֵיהֶֽם׃ (כב) נִשְׁכְּבָ֣ה בְּבָשְׁתֵּ֗נוּ וּֽתְכַסֵּנוּ֮ כְּלִמָּתֵנוּ֒ כִּי֩ לַיהוָ֨ה אֱלֹהֵ֜ינוּ חָטָ֗אנוּ אֲנַ֙חְנוּ֙ וַאֲבוֹתֵ֔ינוּ מִנְּעוּרֵ֖ינוּ וְעַד־ הַיּ֣וֹם הַזֶּ֑ה וְלֹ֣א שָׁמַ֔עְנוּ בְּק֖וֹל יְהֹוָ֥ה אֱלֹהֵֽינוּ׃ (כג) אִם־ תָּשׁ֨וּב יִשְׂרָאֵ֧ל ׀ נְאֻם־ יְהוָ֛ה אֵלַ֖י תָּשׁ֑וּב וְאִם־ תָּסִ֧יר שִׁקּוּצֶ֛יךָ מִפָּנַ֖י וְלֹ֥א תָנֽוּד׃ (כד) וְנִשְׁבַּ֙עְתָּ֙ חַי־ יְהוָ֔ה בֶּאֱמֶ֖ת בְּמִשְׁפָּ֣ט וּבִצְדָקָ֑ה וְהִתְבָּ֥רְכוּ ב֛וֹ גּוֹיִ֖ם וּב֥וֹ יִתְהַלָּֽלוּ׃ (כה) כִּי־ כֹ֣ה ׀ אָמַ֣ר יְהֹוָ֗ה לְאִ֤ישׁ יְהוּדָה֙ וְלִיר֣וּשָׁלִַ֔ם נִ֥ירוּ לָכֶ֖ם נִ֑יר וְאַֽל־ תִּזְרְע֖וּ אֶל־ קוֹצִֽים׃ (כו) הִמֹּ֣לוּ לַיהֹוָ֗ה וְהָסִ֙רוּ֙ עָרְל֣וֹת לְבַבְכֶ֔ם אִ֥ישׁ יְהוּדָ֖ה וְיֹשְׁבֵ֣י יְרוּשָׁלִָ֑ם פֶּן־ תֵּצֵ֨א כָאֵ֜שׁ חֲמָתִ֗י וּבָעֲרָה֙ וְאֵ֣ין מְכַבֶּ֔ה מִפְּנֵ֖י רֹ֥עַ מַעַלְלֵיכֶֽם׃ (כז) הַגִּ֣ידוּ בִֽיהוּדָ֗ה וּבִירוּשָׁלִַ֙ם֙ הַשְׁמִ֔יעוּ וְאִמְר֕וּ (ותקעו) תִּקְע֥וּ שׁוֹפָ֖ר בָּאָ֑רֶץ קִרְא֤וּ מַלְאוּ֙ וְאִמְר֔וּ הֵאָסְפ֥וּ וְנָב֖וֹאָה אֶל־ עָרֵ֥י הַמִּבְצָֽר׃ (כח) שְׂאוּ־ נֵ֣ס צִיּ֔וֹנָה הָעִ֖יזוּ אַֽל־ תַּעֲמֹ֑דוּ כִּ֣י רָעָ֗ה אָנֹכִ֛י מֵבִ֥יא מִצָּפ֖וֹן וְשֶׁ֥בֶר גָּדֽוֹל׃ (כט) עָלָ֤ה אַרְיֵה֙ מִֽסֻּבְּכ֔וֹ וּמַשְׁחִ֣ית גּוֹיִ֔ם נָסַ֖ע יָצָ֣א מִמְּקֹמ֑וֹ לָשׂ֤וּם אַרְצֵךְ֙ לְשַׁמָּ֔ה עָרַ֥יִךְ תִּצֶּ֖ינָה מֵאֵ֥ין יוֹשֵֽׁב׃ (ל) עַל־ זֹ֛את חִגְר֥וּ שַׂקִּ֖ים סִפְד֣וּ וְהֵילִ֑ילוּ כִּ֥י לֹא־ שָׁ֛ב חֲר֥וֹן אַף־ יְהֹוָ֖ה מִמֶּֽנּוּ׃ (לא) וְהָיָ֤ה בַיּוֹם־ הַהוּא֙ נְאֻם־ יְהוָ֔ה יֹאבַ֥ד לֵב־ הַמֶּ֖לֶךְ וְלֵ֣ב הַשָּׂרִ֑ים וְנָשַׁ֙מּוּ֙ הַכֹּ֣הֲנִ֔ים וְהַנְּבִיאִ֖ים יִתְמָֽהוּ׃ (לב) וָאֹמַ֞ר אֲהָ֣הּ ׀ אֲדֹנָ֣י יְהוִ֗ה אָכֵן֩ הַשֵּׁ֨א הִשֵּׁ֜אתָ לָעָ֤ם הַזֶּה֙ וְלִירוּשָׁלִַ֣ם לֵאמֹ֔ר שָׁל֖וֹם יִהְיֶ֣ה לָכֶ֑ם וְנָגְעָ֥ה חֶ֖רֶב עַד־ הַנָּֽפֶשׁ׃ (לג) בָּעֵ֣ת הַהִ֗יא יֵאָמֵ֤ר לָֽעָם־ הַזֶּה֙ וְלִיר֣וּשָׁלִַ֔ם ר֣וּחַ צַ֤ח שְׁפָיִים֙ בַּמִּדְבָּ֔ר דֶּ֖רֶךְ בַּת־ עַמִּ֑י ל֥וֹא לִזְר֖וֹת וְל֥וֹא לְהָבַֽר׃ (לד) ר֧וּחַ מָלֵ֛א מֵאֵ֖לֶּה יָ֣בוֹא לִ֑י עַתָּ֕ה גַּם־ אֲנִ֛י אֲדַבֵּ֥ר מִשְׁפָּטִ֖ים אוֹתָֽם׃ (לה) הִנֵּ֣ה ׀ כַּעֲנָנִ֣ים יַעֲלֶ֗ה וְכַסּוּפָה֙ מַרְכְּבוֹתָ֔יו קַלּ֥וּ מִנְּשָׁרִ֖ים סוּסָ֑יו א֥וֹי לָ֖נוּ כִּ֥י שֻׁדָּֽדְנוּ׃ (לו) כַּבְּסִ֨י מֵרָעָ֤ה לִבֵּךְ֙ יְר֣וּשָׁלִַ֔ם לְמַ֖עַן תִּוָּשֵׁ֑עִי עַד־ מָתַ֛י תָּלִ֥ין בְּקִרְבֵּ֖ךְ מַחְשְׁב֥וֹת אוֹנֵֽךְ׃ (לז) כִּ֛י ק֥וֹל מַגִּ֖יד מִדָּ֑ן וּמַשְׁמִ֥יעַ אָ֖וֶן מֵהַ֥ר אֶפְרָֽיִם׃ (לח) הַזְכִּ֣ירוּ לַגּוֹיִ֗ם הִנֵּה֙ הַשְׁמִ֣יעוּ עַל־ יְרוּשָׁלִַ֔ם נֹצְרִ֥ים בָּאִ֖ים מֵאֶ֣רֶץ הַמֶּרְחָ֑ק וַֽיִּתְּנ֛וּ עַל־ עָרֵ֥י יְהוּדָ֖ה קוֹלָֽם׃ (לט) כְּשֹׁמְרֵ֣י שָׂדַ֔י הָי֥וּ עָלֶ֖יהָ מִסָּבִ֑יב כִּי־ אֹתִ֥י מָרָ֖תָה נְאֻם־ יְהוָֽה׃ (מ) דַּרְכֵּךְ֙ וּמַ֣עֲלָלַ֔יִךְ עָשׂ֥וֹ אֵ֖לֶּה לָ֑ךְ זֹ֤את רָעָתֵךְ֙ כִּ֣י מָ֔ר כִּ֥י נָגַ֖ע עַד־ לִבֵּֽךְ׃ (מא) מֵעַ֣י ׀ מֵעַ֨י ׀ (אחולה) אוֹחִ֜ילָה קִיר֥וֹת לִבִּ֛י הֹֽמֶה־ לִּ֥י לִבִּ֖י לֹ֣א אַחֲרִ֑ישׁ כִּ֣י ק֤וֹל שׁוֹפָר֙ (שמעתי) שָׁמַ֣עַתְּ נַפְשִׁ֔י תְּרוּעַ֖ת מִלְחָמָֽה׃ (מב) שֶׁ֤בֶר עַל־ שֶׁ֙בֶר֙ נִקְרָ֔א כִּ֥י שֻׁדְּדָ֖ה כָּל־ הָאָ֑רֶץ פִּתְאֹם֙ שֻׁדְּד֣וּ אֹהָלַ֔י רֶ֖גַע יְרִיעֹתָֽי׃ (מג) עַד־ מָתַ֖י אֶרְאֶה־ נֵּ֑ס אֶשְׁמְעָ֖ה ק֥וֹל שׁוֹפָֽר׃ (מד) כִּ֣י ׀ אֱוִ֣יל עַמִּ֗י אוֹתִי֙ לֹ֣א יָדָ֔עוּ בָּנִ֤ים סְכָלִים֙ הֵ֔מָּה וְלֹ֥א נְבוֹנִ֖ים הֵ֑מָּה חֲכָמִ֥ים הֵ֙מָּה֙ לְהָרַ֔ע וּלְהֵיטִ֖יב לֹ֥א יָדָֽעוּ׃ (מה) רָאִ֙יתִי֙ אֶת־ הָאָ֔רֶץ וְהִנֵּה־ תֹ֖הוּ וָבֹ֑הוּ וְאֶל־ הַשָּׁמַ֖יִם וְאֵ֥ין אוֹרָֽם׃ (מו) רָאִ֙יתִי֙ הֶֽהָרִ֔ים וְהִנֵּ֖ה רֹעֲשִׁ֑ים וְכָל־ הַגְּבָע֖וֹת הִתְקַלְקָֽלוּ׃ (מז) רָאִ֕יתִי וְהִנֵּ֖ה אֵ֣ין הָאָדָ֑ם וְכָל־ ע֥וֹף הַשָּׁמַ֖יִם נָדָֽדוּ׃ (מח) רָאִ֕יתִי וְהִנֵּ֥ה הַכַּרְמֶ֖ל הַמִּדְבָּ֑ר וְכָל־ עָרָ֗יו נִתְּצוּ֙ מִפְּנֵ֣י יְהוָ֔ה מִפְּנֵ֖י חֲר֥וֹן אַפּֽוֹ׃ (מט) כִּי־ כֹה֙ אָמַ֣ר יְהוָ֔ה שְׁמָמָ֥ה תִהְיֶ֖ה כָּל־ הָאָ֑רֶץ וְכָלָ֖ה לֹ֥א אֶעֱשֶֽׂה׃ (נ) עַל־ זֹאת֙ תֶּאֱבַ֣ל הָאָ֔רֶץ וְקָדְר֥וּ הַשָּׁמַ֖יִם מִמָּ֑עַל עַ֤ל כִּי־ דִבַּ֙רְתִּי֙ זַמֹּ֔תִי וְלֹ֥א נִחַ֖מְתִּי וְלֹא־ אָשׁ֥וּב מִמֶּֽנָּה׃ (נא) מִקּ֨וֹל פָּרָ֜שׁ וְרֹ֣מֵה קֶ֗שֶׁת בֹּרַ֙חַת֙ כָּל־ הָעִ֔יר בָּ֚אוּ בֶּעָבִ֔ים וּבַכֵּפִ֖ים עָל֑וּ כָּל־ הָעִ֣יר עֲזוּבָ֔ה וְאֵין־ יוֹשֵׁ֥ב בָּהֵ֖ן אִֽישׁ׃ (נב) (ואתי) וְאַ֨תְּ שָׁד֜וּד מַֽה־ תַּעֲשִׂ֗י כִּֽי־ תִלְבְּשִׁ֨י שָׁנִ֜י כִּי־ תַעְדִּ֣י עֲדִי־ זָהָ֗ב כִּֽי־ תִקְרְעִ֤י בַפּוּךְ֙ עֵינַ֔יִךְ לַשָּׁ֖וְא תִּתְיַפִּ֑י מָאֲסוּ־ בָ֥ךְ עֹגְבִ֖ים נַפְשֵׁ֥ךְ יְבַקֵּֽשׁוּ׃ (נג) כִּי֩ ק֨וֹל כְּחוֹלָ֜ה שָׁמַ֗עְתִּי צָרָה֙ כְּמַבְכִּירָ֔ה ק֧וֹל בַּת־ צִיּ֛וֹן תִּתְיַפֵּ֖חַ תְּפָרֵ֣שׂ כַּפֶּ֑יהָ אֽוֹי־ נָ֣א לִ֔י כִּֽי־ עָיְפָ֥ה נַפְשִׁ֖י לְהֹרְגִֽים׃
