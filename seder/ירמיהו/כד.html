ירמיהו סדר כד

לפי החלוקה לפרקים: ירמיהו לט, יח – מב, יא · 46 פסוקים

פתיחה באתר · ספר ירמיהו כולו ·

(א) כִּ֤י מַלֵּט֙ אֲמַלֶּטְךָ֔ וּבַחֶ֖רֶב לֹ֣א תִפֹּ֑ל וְהָיְתָ֨ה לְךָ֤ נַפְשְׁךָ֙ לְשָׁלָ֔ל כִּֽי־ בָטַ֥חְתָּ בִּ֖י נְאֻם־ יְהוָֽה׃ (ב) הַדָּבָ֞ר אֲשֶׁר־ הָיָ֤ה אֶֽל־ יִרְמְיָ֙הוּ֙ מֵאֵ֣ת יְהוָ֔ה אַחַ֣ר ׀ שַׁלַּ֣ח אֹת֗וֹ נְבוּזַרְאֲדָ֛ן רַב־ טַבָּחִ֖ים מִן־ הָֽרָמָ֑ה בְּקַחְתּ֣וֹ אֹת֗וֹ וְהֽוּא־ אָס֤וּר בָּֽאזִקִּים֙ בְּת֨וֹךְ כָּל־ גָּל֤וּת יְרוּשָׁלִַ֙ם֙ וִֽיהוּדָ֔ה הַמֻּגְלִ֖ים בָּבֶֽלָה׃ (ג) וַיִּקַּ֥ח רַב־ טַבָּחִ֖ים לְיִרְמְיָ֑הוּ וַיֹּ֣אמֶר אֵלָ֔יו יְהוָ֣ה אֱלֹהֶ֗יךָ דִּבֶּר֙ אֶת־ הָרָעָ֣ה הַזֹּ֔את אֶל־ הַמָּק֖וֹם הַזֶּֽה׃ (ד) וַיָּבֵ֥א וַיַּ֛עַשׂ יְהוָ֖ה כַּאֲשֶׁ֣ר דִּבֵּ֑ר כִּֽי־ חֲטָאתֶ֤ם לַֽיהוָה֙ וְלֹֽא־ שְׁמַעְתֶּ֣ם בְּקוֹל֔וֹ וְהָיָ֥ה לָכֶ֖ם (דבר) הַדָּבָ֥ר הַזֶּֽה׃ (ה) וְעַתָּ֞ה הִנֵּ֧ה פִתַּחְתִּ֣יךָ הַיּ֗וֹם מִֽן־ הָאזִקִּים֮ אֲשֶׁ֣ר עַל־ יָדֶךָ֒ אִם־ ט֨וֹב בְּעֵינֶ֜יךָ לָב֧וֹא אִתִּ֣י בָבֶ֗ל בֹּ֚א וְאָשִׂ֤ים אֶת־ עֵינִי֙ עָלֶ֔יךָ וְאִם־ רַ֧ע בְּעֵינֶ֛יךָ לָבֽוֹא־ אִתִּ֥י בָבֶ֖ל חֲדָ֑ל רְאֵה֙ כָּל־ הָאָ֣רֶץ לְפָנֶ֔יךָ אֶל־ ט֨וֹב וְאֶל־ הַיָּשָׁ֧ר בְּעֵינֶ֛יךָ לָלֶ֥כֶת שָׁ֖מָּה לֵֽךְ׃ (ו) וְעוֹדֶ֣נּוּ לֹֽא־ יָשׁ֗וּב וְשֻׁ֡בָה אֶל־ גְּדַלְיָ֣ה בֶן־ אֲחִיקָ֣ם בֶּן־ שָׁפָ֡ן אֲשֶׁר֩ הִפְקִ֨יד מֶֽלֶךְ־ בָּבֶ֜ל בְּעָרֵ֣י יְהוּדָ֗ה וְשֵׁ֤ב אִתּוֹ֙ בְּת֣וֹךְ הָעָ֔ם א֠וֹ אֶל־ כָּל־ הַיָּשָׁ֧ר בְּעֵינֶ֛יךָ לָלֶ֖כֶת לֵ֑ךְ וַיִּתֶּן־ ל֧וֹ רַב־ טַבָּחִ֛ים אֲרֻחָ֥ה וּמַשְׂאֵ֖ת וַֽיְשַׁלְּחֵֽהוּ׃ (ז) וַיָּבֹ֧א יִרְמְיָ֛הוּ אֶל־ גְּדַלְיָ֥ה בֶן־ אֲחִיקָ֖ם הַמִּצְפָּ֑תָה וַיֵּ֤שֶׁב אִתּוֹ֙ בְּת֣וֹךְ הָעָ֔ם הַנִּשְׁאָרִ֖ים בָּאָֽרֶץ׃ (ח) וַיִּשְׁמְעוּ֩ כָל־ שָׂרֵ֨י הַחֲיָלִ֜ים אֲשֶׁ֣ר בַּשָּׂדֶ֗ה הֵ֚מָּה וְאַנְשֵׁיהֶ֔ם כִּֽי־ הִפְקִ֧יד מֶֽלֶךְ־ בָּבֶ֛ל אֶת־ גְּדַלְיָ֥הוּ בֶן־ אֲחִיקָ֖ם בָּאָ֑רֶץ וְכִ֣י ׀ הִפְקִ֣יד אִתּ֗וֹ אֲנָשִׁ֤ים וְנָשִׁים֙ וָטָ֔ף וּמִדַּלַּ֣ת הָאָ֔רֶץ מֵאֲשֶׁ֥ר לֹֽא־ הָגְל֖וּ בָּבֶֽלָה׃ (ט) וַיָּבֹ֥אוּ אֶל־ גְּדַלְיָ֖ה הַמִּצְפָּ֑תָה וְיִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֡הוּ וְיוֹחָנָ֣ן וְיוֹנָתָ֣ן בְּנֵֽי־ קָ֠רֵחַ וּשְׂרָיָ֨ה בֶן־ תַּנְחֻ֜מֶת וּבְנֵ֣י ׀ (עופי) עֵיפַ֣י הַנְּטֹפָתִ֗י וִֽיזַנְיָ֙הוּ֙ בֶּן־ הַמַּ֣עֲכָתִ֔י הֵ֖מָּה וְאַנְשֵׁיהֶֽם׃ (י) וַיִּשָּׁבַ֨ע לָהֶ֜ם גְּדַלְיָ֨הוּ בֶן־ אֲחִיקָ֤ם בֶּן־ שָׁפָן֙ וּלְאַנְשֵׁיהֶ֣ם לֵאמֹ֔ר אַל־ תִּֽירְא֖וּ מֵעֲב֣וֹד הַכַּשְׂדִּ֑ים שְׁב֣וּ בָאָ֗רֶץ וְעִבְד֛וּ אֶת־ מֶ֥לֶךְ בָּבֶ֖ל וְיִיטַ֥ב לָכֶֽם׃ (יא) וַאֲנִ֗י הִנְנִ֤י יֹשֵׁב֙ בַּמִּצְפָּ֔ה לַֽעֲמֹד֙ לִפְנֵ֣י הַכַּשְׂדִּ֔ים אֲשֶׁ֥ר יָבֹ֖אוּ אֵלֵ֑ינוּ וְאַתֶּ֡ם אִסְפוּ֩ יַ֨יִן וְקַ֜יִץ וְשֶׁ֗מֶן וְשִׂ֙מוּ֙ בִּכְלֵיכֶ֔ם וּשְׁב֖וּ בְּעָרֵיכֶ֥ם אֲשֶׁר־ תְּפַשְׂתֶּֽם׃ (יב) וְגַ֣ם כָּֽל־ הַיְּהוּדִ֡ים אֲשֶׁר־ בְּמוֹאָ֣ב ׀ וּבִבְנֵֽי־ עַמּ֨וֹן וּבֶאֱד֜וֹם וַאֲשֶׁ֤ר בְּכָל־ הָֽאֲרָצוֹת֙ שָֽׁמְע֔וּ כִּֽי־ נָתַ֧ן מֶֽלֶךְ־ בָּבֶ֛ל שְׁאֵרִ֖ית לִֽיהוּדָ֑ה וְכִי֙ הִפְקִ֣יד עֲלֵיהֶ֔ם אֶת־ גְּדַלְיָ֖הוּ בֶּן־ אֲחִיקָ֥ם בֶּן־ שָׁפָֽן׃ (יג) וַיָּשֻׁ֣בוּ כָל־ הַיְּהוּדִ֗ים מִכָּל־ הַמְּקֹמוֹת֙ אֲשֶׁ֣ר נִדְּחוּ־ שָׁ֔ם וַיָּבֹ֧אוּ אֶֽרֶץ־ יְהוּדָ֛ה אֶל־ גְּדַלְיָ֖הוּ הַמִּצְפָּ֑תָה וַיַּאַסְפ֛וּ יַ֥יִן וָקַ֖יִץ הַרְבֵּ֥ה מְאֹֽד׃ (יד) וְיֽוֹחָנָן֙ בֶּן־ קָרֵ֔חַ וְכָל־ שָׂרֵ֥י הַחֲיָלִ֖ים אֲשֶׁ֣ר בַּשָּׂדֶ֑ה בָּ֥אוּ אֶל־ גְּדַלְיָ֖הוּ הַמִּצְפָּֽתָה׃ (טו) וַיֹּאמְר֣וּ אֵלָ֗יו הֲיָדֹ֤עַ תֵּדַע֙ כִּ֞י בַּעֲלִ֣יס ׀ מֶ֣לֶךְ בְּנֵֽי־ עַמּ֗וֹן שָׁלַח֙ אֶת־ יִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֔ה לְהַכֹּתְךָ֖ נָ֑פֶשׁ וְלֹא־ הֶאֱמִ֣ין לָהֶ֔ם גְּדַלְיָ֖הוּ בֶּן־ אֲחִיקָֽם׃ (טז) וְיוֹחָנָ֣ן בֶּן־ קָרֵ֡חַ אָמַ֣ר אֶל־ גְּדַלְיָהוּ֩ בַסֵּ֨תֶר בַּמִּצְפָּ֜ה לֵאמֹ֗ר אֵ֤לְכָה נָּא֙ וְאַכֶּה֙ אֶת־ יִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֔ה וְאִ֖ישׁ לֹ֣א יֵדָ֑ע לָ֧מָּה יַכֶּ֣כָּה נֶּ֗פֶשׁ וְנָפֹ֙צוּ֙ כָּל־ יְהוּדָ֔ה הַנִּקְבָּצִ֣ים אֵלֶ֔יךָ וְאָבְדָ֖ה שְׁאֵרִ֥ית יְהוּדָֽה׃ (יז) וַיֹּ֨אמֶר גְּדַלְיָ֤הוּ בֶן־ אֲחִיקָם֙ אֶל־ יוֹחָנָ֣ן בֶּן־ קָרֵ֔חַ אַֽל־ (תעש) תַּעֲשֵׂ֖ה אֶת־ הַדָּבָ֣ר הַזֶּ֑ה כִּֽי־ שֶׁ֛קֶר אַתָּ֥ה דֹבֵ֖ר אֶל־ יִשְׁמָעֵֽאל׃ (יח) וַיְהִ֣י ׀ בַּחֹ֣דֶשׁ הַשְּׁבִיעִ֗י בָּ֣א יִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֣ה בֶן־ אֱלִישָׁמָ֣ע מִזֶּ֣רַע הַ֠מְּלוּכָה וְרַבֵּ֨י הַמֶּ֜לֶךְ וַעֲשָׂרָ֨ה אֲנָשִׁ֥ים אִתּ֛וֹ אֶל־ גְּדַלְיָ֥הוּ בֶן־ אֲחִיקָ֖ם הַמִּצְפָּ֑תָה וַיֹּ֨אכְלוּ שָׁ֥ם לֶ֛חֶם יַחְדָּ֖ו בַּמִּצְפָּֽה׃ (יט) וַיָּקָם֩ יִשְׁמָעֵ֨אל בֶּן־ נְתַנְיָ֜ה וַעֲשֶׂ֥רֶת הָאֲנָשִׁ֣ים ׀ אֲשֶׁר־ הָי֣וּ אִתּ֗וֹ וַ֠יַּכּוּ אֶת־ גְּדַלְיָ֨הוּ בֶן־ אֲחִיקָ֧ם בֶּן־ שָׁפָ֛ן בַּחֶ֖רֶב וַיָּ֣מֶת אֹת֑וֹ אֲשֶׁר־ הִפְקִ֥יד מֶֽלֶךְ־ בָּבֶ֖ל בָּאָֽרֶץ׃ (כ) וְאֵ֣ת כָּל־ הַיְּהוּדִ֗ים אֲשֶׁר־ הָי֨וּ אִתּ֤וֹ אֶת־ גְּדַלְיָ֙הוּ֙ בַּמִּצְפָּ֔ה וְאֶת־ הַכַּשְׂדִּ֖ים אֲשֶׁ֣ר נִמְצְאוּ־ שָׁ֑ם אֵ֚ת אַנְשֵׁ֣י הַמִּלְחָמָ֔ה הִכָּ֖ה יִשְׁמָעֵֽאל׃ (כא) וַיְהִ֛י בַּיּ֥וֹם הַשֵּׁנִ֖י לְהָמִ֣ית אֶת־ גְּדַלְיָ֑הוּ וְאִ֖ישׁ לֹ֥א יָדָֽע׃ (כב) וַיָּבֹ֣אוּ אֲ֠נָשִׁים מִשְּׁכֶ֞ם מִשִּׁל֤וֹ וּמִשֹּֽׁמְרוֹן֙ שְׁמֹנִ֣ים אִ֔ישׁ מְגֻלְּחֵ֥י זָקָ֛ן וּקְרֻעֵ֥י בְגָדִ֖ים וּמִתְגֹּֽדְדִ֑ים וּמִנְחָ֤ה וּלְבוֹנָה֙ בְּיָדָ֔ם לְהָבִ֖יא בֵּ֥ית יְהוָֽה׃ (כג) וַ֠יֵּצֵא יִשְׁמָעֵ֨אל בֶּן־ נְתַנְיָ֤ה לִקְרָאתָם֙ מִן־ הַמִּצְפָּ֔ה הֹלֵ֥ךְ הָלֹ֖ךְ וּבֹכֶ֑ה וַֽיְהִי֙ כִּפְגֹ֣שׁ אֹתָ֔ם וַיֹּ֣אמֶר אֲלֵיהֶ֔ם בֹּ֖אוּ אֶל־ גְּדַלְיָ֥הוּ בֶן־ אֲחִיקָֽם׃ (כד) וַיְהִ֕י כְּבוֹאָ֖ם אֶל־ תּ֣וֹךְ הָעִ֑יר וַיִּשְׁחָטֵ֞ם יִשְׁמָעֵ֤אל בֶּן־ נְתַנְיָה֙ אֶל־ תּ֣וֹךְ הַבּ֔וֹר ה֖וּא וְהָאֲנָשִׁ֥ים אֲשֶׁר־ אִתּֽוֹ׃ (כה) וַעֲשָׂרָ֨ה אֲנָשִׁ֜ים נִמְצְאוּ־ בָ֗ם וַיֹּאמְר֤וּ אֶל־ יִשְׁמָעֵאל֙ אַל־ תְּמִתֵ֔נוּ כִּֽי־ יֶשׁ־ לָ֤נוּ מַטְמֹנִים֙ בַּשָּׂדֶ֔ה חִטִּ֥ים וּשְׂעֹרִ֖ים וְשֶׁ֣מֶן וּדְבָ֑שׁ וַיֶּחְדַּ֕ל וְלֹ֥א הֱמִיתָ֖ם בְּת֥וֹךְ אֲחֵיהֶֽם׃ (כו) וְהַבּ֗וֹר אֲשֶׁר֩ הִשְׁלִ֨יךְ שָׁ֤ם יִשְׁמָעֵאל֙ אֵ֣ת ׀ כָּל־ פִּגְרֵ֣י הָאֲנָשִׁ֗ים אֲשֶׁ֤ר הִכָּה֙ בְּיַד־ גְּדַלְיָ֔הוּ ה֗וּא אֲשֶׁ֤ר עָשָׂה֙ הַמֶּ֣לֶךְ אָסָ֔א מִפְּנֵ֖י בַּעְשָׁ֣א מֶֽלֶךְ־ יִשְׂרָאֵ֑ל אֹת֗וֹ מִלֵּ֛א יִשְׁמָעֵ֥אל בֶּן־ נְתַנְיָ֖הוּ חֲלָלִֽים׃ (כז) וַיִּ֣שְׁבְּ ׀ יִ֠שְׁמָעֵאל אֶת־ כָּל־ שְׁאֵרִ֨ית הָעָ֜ם אֲשֶׁ֣ר בַּמִּצְפָּ֗ה אֶת־ בְּנ֤וֹת הַמֶּ֙לֶךְ֙ וְאֶת־ כָּל־ הָעָם֙ הַנִּשְׁאָרִ֣ים בַּמִּצְפָּ֔ה אֲשֶׁ֣ר הִפְקִ֗יד נְבֽוּזַרְאֲדָן֙ רַב־ טַבָּחִ֔ים אֶת־ גְּדַלְיָ֖הוּ בֶּן־ אֲחִיקָ֑ם וַיִּשְׁבֵּם֙ יִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֔ה וַיֵּ֕לֶךְ לַעֲבֹ֖ר אֶל־ בְּנֵ֥י עַמּֽוֹן׃ (כח) וַיִּשְׁמַע֙ יוֹחָנָ֣ן בֶּן־ קָרֵ֔חַ וְכָל־ שָׂרֵ֥י הַחֲיָלִ֖ים אֲשֶׁ֣ר אִתּ֑וֹ אֵ֤ת כָּל־ הָֽרָעָה֙ אֲשֶׁ֣ר עָשָׂ֔ה יִשְׁמָעֵ֖אל בֶּן־ נְתַנְיָֽה׃ (כט) וַיִּקְחוּ֙ אֶת־ כָּל־ הָ֣אֲנָשִׁ֔ים וַיֵּ֣לְכ֔וּ לְהִלָּחֵ֖ם עִם־ יִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֑ה וַיִּמְצְא֣וּ אֹת֔וֹ אֶל־ מַ֥יִם רַבִּ֖ים אֲשֶׁ֥ר בְּגִבְעֽוֹן׃ (ל) וַיְהִ֗י כִּרְא֤וֹת כָּל־ הָעָם֙ אֲשֶׁ֣ר אֶת־ יִשְׁמָעֵ֔אל אֶת־ יֽוֹחָנָן֙ בֶּן־ קָרֵ֔חַ וְאֵ֛ת כָּל־ שָׂרֵ֥י הַחֲיָלִ֖ים אֲשֶׁ֣ר אִתּ֑וֹ וַיִּשְׂמָֽחוּ׃ (לא) וַיָּסֹ֙בּוּ֙ כָּל־ הָעָ֔ם אֲשֶׁר־ שָׁבָ֥ה יִשְׁמָעֵ֖אל מִן־ הַמִּצְפָּ֑ה וַיָּשֻׁ֙בוּ֙ וַיֵּ֣לְכ֔וּ אֶל־ יֽוֹחָנָ֖ן בֶּן־ קָרֵֽחַ׃ (לב) וְיִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֗ה נִמְלַט֙ בִּשְׁמֹנָ֣ה אֲנָשִׁ֔ים מִפְּנֵ֖י יֽוֹחָנָ֑ן וַיֵּ֖לֶךְ אֶל־ בְּנֵ֥י עַמּֽוֹן׃ (לג) וַיִּקַּח֩ יוֹחָנָ֨ן בֶּן־ קָרֵ֜חַ וְכָל־ שָׂרֵ֧י הַחֲיָלִ֣ים אֲשֶׁר־ אִתּ֗וֹ אֵ֣ת כָּל־ שְׁאֵרִ֤ית הָעָם֙ אֲשֶׁ֣ר הֵ֠שִׁיב מֵאֵ֨ת יִשְׁמָעֵ֤אל בֶּן־ נְתַנְיָה֙ מִן־ הַמִּצְפָּ֔ה אַחַ֣ר הִכָּ֔ה אֶת־ גְּדַלְיָ֖ה בֶּן־ אֲחִיקָ֑ם גְּבָרִ֞ים אַנְשֵׁ֣י הַמִּלְחָמָ֗ה וְנָשִׁ֤ים וְטַף֙ וְסָ֣רִסִ֔ים אֲשֶׁ֥ר הֵשִׁ֖יב מִגִּבְעֽוֹן׃ (לד) וַיֵּלְכ֗וּ וַיֵּֽשְׁבוּ֙ בְּגֵר֣וּת (כמוהם) כִּמְהָ֔ם אֲשֶׁר־ אֵ֖צֶל בֵּ֣ית לָ֑חֶם לָלֶ֖כֶת לָב֥וֹא מִצְרָֽיִם׃ (לה) מִפְּנֵי֙ הַכַּשְׂדִּ֔ים כִּ֥י יָרְא֖וּ מִפְּנֵיהֶ֑ם כִּֽי־ הִכָּ֞ה יִשְׁמָעֵ֣אל בֶּן־ נְתַנְיָ֗ה אֶת־ גְּדַלְיָ֙הוּ֙ בֶּן־ אֲחִיקָ֔ם אֲשֶׁר־ הִפְקִ֥יד מֶֽלֶךְ־ בָּבֶ֖ל בָּאָֽרֶץ׃ (לו) וַֽיִּגְּשׁוּ֙ כָּל־ שָׂרֵ֣י הַחֲיָלִ֔ים וְיֽוֹחָנָן֙ בֶּן־ קָרֵ֔חַ וִֽיזַנְיָ֖ה בֶּן־ הוֹשַֽׁעְיָ֑ה וְכָל־ הָעָ֖ם מִקָּטֹ֥ן וְעַד־ גָּדֽוֹל׃ (לז) וַיֹּאמְר֞וּ אֶֽל־ יִרְמְיָ֣הוּ הַנָּבִ֗יא תִּפָּל־ נָ֤א תְחִנָּתֵ֙נוּ֙ לְפָנֶ֔יךָ וְהִתְפַּלֵּ֤ל בַּעֲדֵ֙נוּ֙ אֶל־ יְהוָ֣ה אֱלֹהֶ֔יךָ בְּעַ֖ד כָּל־ הַשְּׁאֵרִ֣ית הַזֹּ֑את כִּֽי־ נִשְׁאַ֤רְנוּ מְעַט֙ מֵֽהַרְבֵּ֔ה כַּאֲשֶׁ֥ר עֵינֶ֖יךָ רֹא֥וֹת אֹתָֽנוּ׃ (לח) וְיַגֶּד־ לָ֙נוּ֙ יְהוָ֣ה אֱלֹהֶ֔יךָ אֶת־ הַדֶּ֖רֶךְ אֲשֶׁ֣ר נֵֽלֶךְ־ בָּ֑הּ וְאֶת־ הַדָּבָ֖ר אֲשֶׁ֥ר נַעֲשֶֽׂה׃ (לט) וַיֹּ֨אמֶר אֲלֵיהֶ֜ם יִרְמְיָ֤הוּ הַנָּבִיא֙ שָׁמַ֔עְתִּי הִנְנִ֧י מִתְפַּלֵּ֛ל אֶל־ יְהוָ֥ה אֱלֹהֵיכֶ֖ם כְּדִבְרֵיכֶ֑ם וְֽהָיָ֡ה כָּֽל־ הַדָּבָר֩ אֲשֶׁר־ יַעֲנֶ֨ה יְהוָ֤ה אֶתְכֶם֙ אַגִּ֣יד לָכֶ֔ם לֹֽא־ אֶמְנַ֥ע מִכֶּ֖ם דָּבָֽר׃ (מ) וְהֵ֙מָּה֙ אָמְר֣וּ אֶֽל־ יִרְמְיָ֔הוּ יְהִ֤י יְהוָה֙ בָּ֔נוּ לְעֵ֖ד אֱמֶ֣ת וְנֶאֱמָ֑ן אִם־ לֹ֡א כְּֽכָל־ הַ֠דָּבָר אֲשֶׁ֨ר יִֽשְׁלָחֲךָ֜ יְהוָ֧ה אֱלֹהֶ֛יךָ אֵלֵ֖ינוּ כֵּ֥ן נַעֲשֶֽׂה׃ (מא) אִם־ ט֣וֹב וְאִם־ רָ֔ע בְּק֣וֹל ׀ יְהוָ֣ה אֱלֹהֵ֗ינוּ אֲשֶׁ֨ר (אנו) אֲנַ֜חְנוּ שֹׁלְחִ֥ים אֹתְךָ֛ אֵלָ֖יו נִשְׁמָ֑ע לְמַ֙עַן֙ אֲשֶׁ֣ר יִֽיטַב־ לָ֔נוּ כִּ֣י נִשְׁמַ֔ע בְּק֖וֹל יְהוָ֥ה אֱלֹהֵֽינוּ׃ (מב) וַיְהִ֕י מִקֵּ֖ץ עֲשֶׂ֣רֶת יָמִ֑ים וַיְהִ֥י דְבַר־ יְהוָ֖ה אֶֽל־ יִרְמְיָֽהוּ׃ (מג) וַיִּקְרָ֗א אֶל־ יֽוֹחָנָן֙ בֶּן־ קָרֵ֔חַ וְאֶ֛ל כָּל־ שָׂרֵ֥י הַחֲיָלִ֖ים אֲשֶׁ֣ר אִתּ֑וֹ וּלְכָ֨ל־ הָעָ֔ם לְמִקָּטֹ֥ן וְעַד־ גָּדֽוֹל׃ (מד) וַיֹּ֣אמֶר אֲלֵיהֶ֔ם כֹּֽה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל אֲשֶׁ֨ר שְׁלַחְתֶּ֤ם אֹתִי֙ אֵלָ֔יו לְהַפִּ֥יל תְּחִנַּתְכֶ֖ם לְפָנָֽיו׃ (מה) אִם־ שׁ֤וֹב תֵּֽשְׁבוּ֙ בָּאָ֣רֶץ הַזֹּ֔את וּבָנִ֤יתִי אֶתְכֶם֙ וְלֹ֣א אֶהֱרֹ֔ס וְנָטַעְתִּ֥י אֶתְכֶ֖ם וְלֹ֣א אֶתּ֑וֹשׁ כִּ֤י נִחַ֙מְתִּי֙ אֶל־ הָ֣רָעָ֔ה אֲשֶׁ֥ר עָשִׂ֖יתִי לָכֶֽם׃ (מו) אַל־ תִּֽירְא֗וּ מִפְּנֵי֙ מֶ֣לֶךְ בָּבֶ֔ל אֲשֶׁר־ אַתֶּ֥ם יְרֵאִ֖ים מִפָּנָ֑יו אַל־ תִּֽירְא֤וּ מִמֶּ֙נּוּ֙ נְאֻם־ יְהוָ֔ה כִּֽי־ אִתְּכֶ֣ם אָ֔נִי לְהוֹשִׁ֧יעַ אֶתְכֶ֛ם וּלְהַצִּ֥יל אֶתְכֶ֖ם
