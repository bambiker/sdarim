שופטים סדר יד

פתיחה באתר · ספר שופטים כולו ·

(א) וַיִּשְׁאֲל֥וּ בְנֵֽי־ יִשְׂרָאֵ֖ל בַּֽיהוָ֑ה וְשָׁ֗ם אֲרוֹן֙ בְּרִ֣ית הָאֱלֹהִ֔ים בַּיָּמִ֖ים הָהֵֽם׃ (ב) וּ֠פִינְחָס בֶּן־ אֶלְעָזָ֨ר בֶּֽן־ אַהֲרֹ֜ן עֹמֵ֣ד ׀ לְפָנָ֗יו בַּיָּמִ֣ים הָהֵם֮ לֵאמֹר֒ הַאוֹסִ֨ף ע֜וֹד לָצֵ֧את לַמִּלְחָמָ֛ה עִם־ בְּנֵֽי־ בִנְיָמִ֥ן אָחִ֖י אִם־ אֶחְדָּ֑ל וַיֹּ֤אמֶר יְהוָה֙ עֲל֔וּ כִּ֥י מָחָ֖ר אֶתְּנֶ֥נּוּ בְיָדֶֽךָ׃ (ג) וַיָּ֤שֶׂם יִשְׂרָאֵל֙ אֹֽרְבִ֔ים אֶל־ הַגִּבְעָ֖ה סָבִֽיב׃ (ד) וַיַּעֲל֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶל־ בְּנֵ֥י בִנְיָמִ֖ן בַּיּ֣וֹם הַשְּׁלִישִׁ֑י וַיַּעַרְכ֥וּ אֶל־ הַגִּבְעָ֖ה כְּפַ֥עַם בְּפָֽעַם׃ (ה) וַיֵּצְא֤וּ בְנֵֽי־ בִנְיָמִן֙ לִקְרַ֣את הָעָ֔ם הָנְתְּק֖וּ מִן־ הָעִ֑יר וַיָּחֵ֡לּוּ לְהַכּוֹת֩ מֵהָעָ֨ם חֲלָלִ֜ים כְּפַ֣עַם ׀ בְּפַ֗עַם בַּֽמְסִלּוֹת֙ אֲשֶׁ֨ר אַחַ֜ת עֹלָ֣ה בֵֽית־ אֵ֗ל וְאַחַ֤ת גִּבְעָ֙תָה֙ בַּשָּׂדֶ֔ה כִּשְׁלֹשִׁ֥ים אִ֖ישׁ בְּיִשְׂרָאֵֽל׃ (ו) וַיֹּֽאמְרוּ֙ בְּנֵ֣י בִנְיָמִ֔ן נִגָּפִ֥ים הֵ֛ם לְפָנֵ֖ינוּ כְּבָרִאשֹׁנָ֑ה וּבְנֵ֧י יִשְׂרָאֵ֣ל אָמְר֗וּ נָנ֙וּסָה֙ וּֽנְתַקְּנֻ֔הוּ מִן־ הָעִ֖יר אֶל־ הַֽמְסִלּֽוֹת׃ (ז) וְכֹ֣ל ׀ אִ֣ישׁ יִשְׂרָאֵ֗ל קָ֚מוּ מִמְּקוֹמ֔וֹ וַיַּעַרְכ֖וּ בְּבַ֣עַל תָּמָ֑ר וְאֹרֵ֧ב יִשְׂרָאֵ֛ל מֵגִ֥יחַ מִמְּקֹמ֖וֹ מִמַּֽעֲרֵה־ גָֽבַע׃ (ח) וַיָּבֹאוּ֩ מִנֶּ֨גֶד לַגִּבְעָ֜ה עֲשֶׂרֶת֩ אֲלָפִ֨ים אִ֤ישׁ בָּחוּר֙ מִכָּל־ יִשְׂרָאֵ֔ל וְהַמִּלְחָמָ֖ה כָּבֵ֑דָה וְהֵם֙ לֹ֣א יָדְע֔וּ כִּֽי־ נֹגַ֥עַת עֲלֵיהֶ֖ם הָרָעָֽה׃ (ט) וַיִּגֹּ֨ף יְהוָ֥ה ׀ אֶֽת־ בִּנְיָמִן֮ לִפְנֵ֣י יִשְׂרָאֵל֒ וַיַּשְׁחִיתוּ֩ בְנֵ֨י יִשְׂרָאֵ֤ל בְּבִנְיָמִן֙ בַּיּ֣וֹם הַה֔וּא עֶשְׂרִ֨ים וַחֲמִשָּׁ֥ה אֶ֛לֶף וּמֵאָ֖ה אִ֑ישׁ כָּל־ אֵ֖לֶּה שֹׁ֥לֵף חָֽרֶב׃ (י) וַיִּרְא֥וּ בְנֵֽי־ בִנְיָמִ֖ן כִּ֣י נִגָּ֑פוּ וַיִּתְּנ֨וּ אִֽישׁ־ יִשְׂרָאֵ֤ל מָקוֹם֙ לְבִנְיָמִ֔ן כִּ֤י בָֽטְחוּ֙ אֶל־ הָ֣אֹרֵ֔ב אֲשֶׁר שָׂ֖מוּ אֶל־ הַגִּבְעָֽה׃ (יא) וְהָאֹרֵ֣ב הֵחִ֔ישׁוּ וַֽיִּפְשְׁט֖וּ אֶל־ הַגִּבְעָ֑ה וַיִּמְשֹׁךְ֙ הָאֹרֵ֔ב וַיַּ֥ךְ אֶת־ כָּל־ הָעִ֖יר לְפִי־ חָֽרֶב׃ (יב) וְהַמּוֹעֵ֗ד הָיָ֛ה לְאִ֥ישׁ יִשְׂרָאֵ֖ל עִם־ הָאֹרֵ֑ב הֶ֕רֶב לְהַעֲלוֹתָ֛ם מַשְׂאַ֥ת הֶעָשָׁ֖ן מִן־ הָעִֽיר׃ (יג) וַיַּהֲפֹ֥ךְ אִֽישׁ־ יִשְׂרָאֵ֖ל בַּמִּלְחָמָ֑ה וּבִנְיָמִ֡ן הֵחֵל֩ לְהַכּ֨וֹת חֲלָלִ֤ים בְּאִֽישׁ־ יִשְׂרָאֵל֙ כִּשְׁלֹשִׁ֣ים אִ֔ישׁ כִּ֣י אָמְר֔וּ אַךְ֩ נִגּ֨וֹף נִגָּ֥ף הוּא֙ לְפָנֵ֔ינוּ כַּמִּלְחָמָ֖ה הָרִאשֹׁנָֽה׃ (יד) וְהַמַּשְׂאֵ֗ת הֵחֵ֛לָּה לַעֲל֥וֹת מִן־ הָעִ֖יר עַמּ֣וּד עָשָׁ֑ן וַיִּ֤פֶן בִּנְיָמִן֙ אַחֲרָ֔יו וְהִנֵּ֛ה עָלָ֥ה כְלִיל־ הָעִ֖יר הַשָּׁמָֽיְמָה׃ (טו) וְאִ֤ישׁ יִשְׂרָאֵל֙ הָפַ֔ךְ וַיִּבָּהֵ֖ל אִ֣ישׁ בִּנְיָמִ֑ן כִּ֣י רָאָ֔ה כִּֽי־ נָגְעָ֥ה עָלָ֖יו הָרָעָֽה׃ (טז) וַיִּפְנ֞וּ לִפְנֵ֨י אִ֤ישׁ יִשְׂרָאֵל֙ אֶל־ דֶּ֣רֶךְ הַמִּדְבָּ֔ר וְהַמִּלְחָמָ֖ה הִדְבִּיקָ֑תְהוּ וַאֲשֶׁר֙ מֵהֶ֣עָרִ֔ים מַשְׁחִיתִ֥ים אוֹת֖וֹ בְּתוֹכֽוֹ׃ (יז) כִּתְּר֤וּ אֶת־ בִּנְיָמִן֙ הִרְדִיפֻ֔הוּ מְנוּחָ֖ה הִדְרִיכֻ֑הוּ עַ֛ד נֹ֥כַח הַגִּבְעָ֖ה מִמִּזְרַח־ שָֽׁמֶשׁ׃ (יח) וַֽיִּפְּלוּ֙ מִבִּנְיָמִ֔ן שְׁמֹנָֽה־ עָשָׂ֥ר אֶ֖לֶף אִ֑ישׁ אֶת־ כָּל־ אֵ֖לֶּה אַנְשֵׁי־ חָֽיִל׃ (יט) וַיִּפְנ֞וּ וַיָּנֻ֤סוּ הַמִּדְבָּ֙רָה֙ אֶל־ סֶ֣לַע הָֽרִמּ֔וֹן וַיְעֹֽלְלֻ֙הוּ֙ בַּֽמְסִלּ֔וֹת חֲמֵ֥שֶׁת אֲלָפִ֖ים אִ֑ישׁ וַיַּדְבִּ֤יקוּ אַחֲרָיו֙ עַד־ גִּדְעֹ֔ם וַיַּכּ֥וּ מִמֶּ֖נּוּ אַלְפַּ֥יִם אִֽישׁ׃ (כ) וַיְהִי֩ כָל־ הַנֹּ֨פְלִ֜ים מִבִּנְיָמִ֗ן עֶשְׂרִים֩ וַחֲמִשָּׁ֨ה אֶ֥לֶף אִ֛ישׁ שֹׁ֥לֵֽף חֶ֖רֶב בַּיּ֣וֹם הַה֑וּא אֶֽת־ כָּל־ אֵ֖לֶּה אַנְשֵׁי־ חָֽיִל׃ (כא) וַיִּפְנ֞וּ וַיָּנֻ֤סוּ הַמִּדְבָּ֙רָה֙ אֶל־ סֶ֣לַע הָֽרִמּ֔וֹן שֵׁ֥שׁ מֵא֖וֹת אִ֑ישׁ וַיֵּֽשְׁבוּ֙ בְּסֶ֣לַע רִמּ֔וֹן אַרְבָּעָ֖ה חֳדָשִֽׁים׃ (כב) וְאִ֨ישׁ יִשְׂרָאֵ֜ל שָׁ֣בוּ אֶל־ בְּנֵ֤י בִנְיָמִן֙ וַיַּכּ֣וּם לְפִי־ חֶ֔רֶב מֵעִ֤יר מְתֹם֙ עַד־ בְּהֵמָ֔ה עַ֖ד כָּל־ הַנִּמְצָ֑א גַּ֛ם כָּל־ הֶעָרִ֥ים הַנִּמְצָא֖וֹת שִׁלְּח֥וּ בָאֵֽשׁ׃ (כג) וְאִ֣ישׁ יִשְׂרָאֵ֔ל נִשְׁבַּ֥ע בַּמִּצְפָּ֖ה לֵאמֹ֑ר אִ֣ישׁ מִמֶּ֔נּוּ לֹא־ יִתֵּ֥ן בִּתּ֛וֹ לְבִנְיָמִ֖ן לְאִשָּֽׁה׃ (כד) וַיָּבֹ֤א הָעָם֙ בֵּֽית־ אֵ֔ל וַיֵּ֤שְׁבוּ שָׁם֙ עַד־ הָעֶ֔רֶב לִפְנֵ֖י הָאֱלֹהִ֑ים וַיִּשְׂא֣וּ קוֹלָ֔ם וַיִּבְכּ֖וּ בְּכִ֥י גָדֽוֹל׃ (כה) וַיֹּ֣אמְר֔וּ לָמָ֗ה יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הָ֥יְתָה זֹּ֖את בְּיִשְׂרָאֵ֑ל לְהִפָּקֵ֥ד הַיּ֛וֹם מִיִּשְׂרָאֵ֖ל שֵׁ֥בֶט אֶחָֽד׃ (כו) וַֽיְהִי֙ מִֽמָּחֳרָ֔ת וַיַּשְׁכִּ֣ימוּ הָעָ֔ם וַיִּבְנוּ־ שָׁ֖ם מִזְבֵּ֑חַ וַיַּעֲל֥וּ עֹל֖וֹת וּשְׁלָמִֽים׃ (כז) וַיֹּֽאמְרוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל מִ֠י אֲשֶׁ֨ר לֹא־ עָלָ֧ה בַקָּהָ֛ל מִכָּל־ שִׁבְטֵ֥י יִשְׂרָאֵ֖ל אֶל־ יְהוָ֑ה כִּי֩ הַשְּׁבוּעָ֨ה הַגְּדוֹלָ֜ה הָיְתָ֗ה לַ֠אֲשֶׁר לֹא־ עָלָ֨ה אֶל־ יְהוָ֧ה הַמִּצְפָּ֛ה לֵאמֹ֖ר מ֥וֹת יוּמָֽת׃ (כח) וַיִּנָּֽחֲמוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל אֶל־ בִּנְיָמִ֖ן אָחִ֑יו וַיֹּ֣אמְר֔וּ נִגְדַּ֥ע הַיּ֛וֹם שֵׁ֥בֶט אֶחָ֖ד מִיִּשְׂרָאֵֽל׃ (כט) מַה־ נַּעֲשֶׂ֥ה לָהֶ֛ם לַנּוֹתָרִ֖ים לְנָשִׁ֑ים וַאֲנַ֙חְנוּ֙ נִשְׁבַּ֣עְנוּ בַֽיהוָ֔ה לְבִלְתִּ֛י תֵּת־ לָהֶ֥ם מִבְּנוֹתֵ֖ינוּ לְנָשִֽׁים׃ (ל) וַיֹּ֣אמְר֔וּ מִ֗י אֶחָד֙ מִשִּׁבְטֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁ֛ר לֹֽא־ עָלָ֥ה אֶל־ יְהוָ֖ה הַמִּצְפָּ֑ה וְ֠הִנֵּה לֹ֣א בָא־ אִ֧ישׁ אֶל־ הַֽמַּחֲנֶ֛ה מִיָּבֵ֥ישׁ גִּלְעָ֖ד אֶל־ הַקָּהָֽל׃ (לא) וַיִּתְפָּקֵ֖ד הָעָ֑ם וְהִנֵּ֤ה אֵֽין־ שָׁם֙ אִ֔ישׁ מִיּוֹשְׁבֵ֖י יָבֵ֥שׁ גִּלְעָֽד׃ (לב) וַיִּשְׁלְחוּ־ שָׁ֣ם הָעֵדָ֗ה שְׁנֵים־ עָשָׂ֥ר אֶ֛לֶף אִ֖ישׁ מִבְּנֵ֣י הֶחָ֑יִל וַיְצַוּ֨וּ אוֹתָ֜ם לֵאמֹ֗ר לְ֠כוּ וְהִכִּיתֶ֞ם אֶת־ יוֹשְׁבֵ֨י יָבֵ֤שׁ גִּלְעָד֙ לְפִי־ חֶ֔רֶב וְהַנָּשִׁ֖ים וְהַטָּֽף׃ (לג) וְזֶ֥ה הַדָּבָ֖ר אֲשֶׁ֣ר תַּעֲשׂ֑וּ כָּל־ זָכָ֗ר וְכָל־ אִשָּׁ֛ה יֹדַ֥עַת מִשְׁכַּב־ זָכָ֖ר תַּחֲרִֽימוּ׃ (לד) וַֽיִּמְצְא֞וּ מִיּוֹשְׁבֵ֣י ׀ יָבֵ֣ישׁ גִּלְעָ֗ד אַרְבַּ֤ע מֵאוֹת֙ נַעֲרָ֣ה בְתוּלָ֔ה אֲשֶׁ֧ר לֹֽא־ יָדְעָ֛ה אִ֖ישׁ לְמִשְׁכַּ֣ב זָכָ֑ר וַיָּבִ֨יאוּ אוֹתָ֤ם אֶל־ הַֽמַּחֲנֶה֙ שִׁלֹ֔ה אֲשֶׁ֖ר בְּאֶ֥רֶץ כְּנָֽעַן׃ (לה) וַֽיִּשְׁלְחוּ֙ כָּל־ הָ֣עֵדָ֔ה וַֽיְדַבְּרוּ֙ אֶל־ בְּנֵ֣י בִנְיָמִ֔ן אֲשֶׁ֖ר בְּסֶ֣לַע רִמּ֑וֹן וַיִּקְרְא֥וּ לָהֶ֖ם שָׁלֽוֹם׃ (לו) וַיָּ֤שָׁב בִּנְיָמִן֙ בָּעֵ֣ת הַהִ֔יא וַיִּתְּנ֤וּ לָהֶם֙ הַנָּשִׁ֔ים אֲשֶׁ֣ר חִיּ֔וּ מִנְּשֵׁ֖י יָבֵ֣שׁ גִּלְעָ֑ד וְלֹֽא־ מָצְא֥וּ לָהֶ֖ם כֵּֽן׃ (לז) וְהָעָ֥ם נִחָ֖ם לְבִנְיָמִ֑ן כִּֽי־ עָשָׂ֧ה יְהוָ֛ה פֶּ֖רֶץ בְּשִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (לח) וַיֹּֽאמְר֨וּ זִקְנֵ֣י הָעֵדָ֔ה מַה־ נַּעֲשֶׂ֥ה לַנּוֹתָרִ֖ים לְנָשִׁ֑ים כִּֽי־ נִשְׁמְדָ֥ה מִבִּנְיָמִ֖ן אִשָּֽׁה׃ (לט) וַיֹּ֣אמְר֔וּ יְרֻשַּׁ֥ת פְּלֵיטָ֖ה לְבִנְיָמִ֑ן וְלֹֽא־ יִמָּחֶ֥ה שֵׁ֖בֶט מִיִּשְׂרָאֵֽל׃ (מ) וַאֲנַ֗חְנוּ לֹ֥א נוּכַ֛ל לָתֵת־ לָהֶ֥ם נָשִׁ֖ים מִבְּנוֹתֵ֑ינוּ כִּֽי־ נִשְׁבְּע֤וּ בְנֵֽי־ יִשְׂרָאֵל֙ לֵאמֹ֔ר אָר֕וּר נֹתֵ֥ן אִשָּׁ֖ה לְבִנְיָמִֽן׃ (מא) וַיֹּאמְר֡וּ הִנֵּה֩ חַג־ יְהוָ֨ה בְּשִׁל֜וֹ מִיָּמִ֣ים ׀ יָמִ֗ימָה אֲשֶׁ֞ר מִצְּפ֤וֹנָה לְבֵֽית־ אֵל֙ מִזְרְחָ֣ה הַשֶּׁ֔מֶשׁ לִמְסִלָּ֔ה הָעֹלָ֥ה מִבֵּֽית־ אֵ֖ל שְׁכֶ֑מָה וּמִנֶּ֖גֶב לִלְבוֹנָֽה׃ (מב) (ויצו) וַיְצַוּ֕וּ אֶת־ בְּנֵ֥י בִנְיָמִ֖ן לֵאמֹ֑ר לְכ֖וּ וַאֲרַבְתֶּ֥ם בַּכְּרָמִֽים׃ (מג) וּרְאִיתֶ֗ם וְ֠הִנֵּה אִם־ יֵ֨צְא֥וּ בְנוֹת־ שִׁילוֹ֮ לָח֣וּל בַּמְּחֹלוֹת֒ וִֽיצָאתֶם֙ מִן־ הַכְּרָמִ֔ים וַחֲטַפְתֶּ֥ם לָכֶ֛ם אִ֥ישׁ אִשְׁתּ֖וֹ מִבְּנ֣וֹת שִׁיל֑וֹ וַהֲלַכְתֶּ֖ם אֶ֥רֶץ בִּנְיָמִֽן׃ (מד) וְהָיָ֡ה כִּֽי־ יָבֹ֣אוּ אֲבוֹתָם֩ א֨וֹ אֲחֵיהֶ֜ם (לרוב) לָרִ֣יב ׀ אֵלֵ֗ינוּ וְאָמַ֤רְנוּ אֲלֵיהֶם֙ חָנּ֣וּנוּ אוֹתָ֔ם כִּ֣י לֹ֥א לָקַ֛חְנוּ אִ֥ישׁ אִשְׁתּ֖וֹ בַּמִּלְחָמָ֑ה כִּ֣י לֹ֥א אַתֶּ֛ם נְתַתֶּ֥ם לָהֶ֖ם כָּעֵ֥ת תֶּאְשָֽׁמוּ׃ (מה) וַיַּֽעֲשׂוּ־ כֵן֙ בְּנֵ֣י בִנְיָמִ֔ן וַיִּשְׂא֤וּ נָשִׁים֙ לְמִסְפָּרָ֔ם מִן־ הַמְּחֹלְל֖וֹת אֲשֶׁ֣ר גָּזָ֑לוּ וַיֵּלְכ֗וּ וַיָּשׁ֙וּבוּ֙ אֶל־ נַ֣חֲלָתָ֔ם וַיִּבְנוּ֙ אֶת־ הֶ֣עָרִ֔ים וַיֵּשְׁב֖וּ בָּהֶֽם׃ (מו) וַיִּתְהַלְּכ֨וּ מִשָּׁ֤ם בְּנֵֽי־ יִשְׂרָאֵל֙ בָּעֵ֣ת הַהִ֔יא אִ֥ישׁ לְשִׁבְט֖וֹ וּלְמִשְׁפַּחְתּ֑וֹ וַיֵּצְא֣וּ מִשָּׁ֔ם אִ֖ישׁ לְנַחֲלָתֽוֹ׃ (מז) בַּיָּמִ֣ים הָהֵ֔ם אֵ֥ין מֶ֖לֶךְ בְּיִשְׂרָאֵ֑ל אִ֛ישׁ הַיָּשָׁ֥ר בְּעֵינָ֖יו יַעֲשֶֽׂה׃
