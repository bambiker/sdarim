שופטים סדר ט

פתיחה באתר · ספר שופטים כולו ·

(א) וַיַּעֲבֹ֥ר יִפְתָּ֛ח אֶל־ בְּנֵ֥י עַמּ֖וֹן לְהִלָּ֣חֶם בָּ֑ם וַיִתְּנֵ֥ם יְהוָ֖ה בְּיָדֽוֹ׃ (ב) וַיַּכֵּ֡ם מֵעֲרוֹעֵר֩ וְעַד־ בּוֹאֲךָ֨ מִנִּ֜ית עֶשְׂרִ֣ים עִ֗יר וְעַד֙ אָבֵ֣ל כְּרָמִ֔ים מַכָּ֖ה גְּדוֹלָ֣ה מְאֹ֑ד וַיִּכָּֽנְעוּ֙ בְּנֵ֣י עַמּ֔וֹן מִפְּנֵ֖י בְּנֵ֥י יִשְׂרָאֵֽל׃ (ג) וַיָּבֹ֨א יִפְתָּ֣ח הַמִּצְפָּה֮ אֶל־ בֵּיתוֹ֒ וְהִנֵּ֤ה בִתּוֹ֙ יֹצֵ֣את לִקְרָאת֔וֹ בְתֻפִּ֖ים וּבִמְחֹל֑וֹת וְרַק֙ הִ֣יא יְחִידָ֔ה אֵֽין־ ל֥וֹ מִמֶּ֛נּוּ בֵּ֖ן אוֹ־ בַֽת׃ (ד) וַיְהִי֩ כִרְאוֹת֨וֹ אוֹתָ֜הּ וַיִּקְרַ֣ע אֶת־ בְּגָדָ֗יו וַיֹּ֙אמֶר֙ אֲהָ֤הּ בִּתִּי֙ הַכְרֵ֣עַ הִכְרַעְתִּ֔נִי וְאַ֖תְּ הָיִ֣יתְ בְּעֹֽכְרָ֑י וְאָנֹכִ֗י פָּצִ֤יתִי־ פִי֙ אֶל־ יְהוָ֔ה וְלֹ֥א אוּכַ֖ל לָשֽׁוּב׃ (ה) וַתֹּ֣אמֶר אֵלָ֗יו אָבִי֙ פָּצִ֤יתָה אֶת־ פִּ֙יךָ֙ אֶל־ יְהוָ֔ה עֲשֵׂ֣ה לִ֔י כַּאֲשֶׁ֖ר יָצָ֣א מִפִּ֑יךָ אַחֲרֵ֡י אֲשֶׁ֣ר עָשָׂה֩ לְךָ֙ יְהוָ֧ה נְקָמ֛וֹת מֵאֹיְבֶ֖יךָ מִבְּנֵ֥י עַמּֽוֹן׃ (ו) וַתֹּ֙אמֶר֙ אֶל־ אָבִ֔יהָ יֵעָ֥שֶׂה לִּ֖י הַדָּבָ֣ר הַזֶּ֑ה הַרְפֵּ֨ה מִמֶּ֜נִּי שְׁנַ֣יִם חֳדָשִׁ֗ים וְאֵֽלְכָה֙ וְיָרַדְתִּ֣י עַל־ הֶֽהָרִ֔ים וְאֶבְכֶּה֙ עַל־ בְּתוּלַ֔י אָנֹכִ֖י (ורעיתי) וְרֵעוֹתָֽי׃ (ז) וַיֹּ֣אמֶר לֵ֔כִי וַיִּשְׁלַ֥ח אוֹתָ֖הּ שְׁנֵ֣י חֳדָשִׁ֑ים וַתֵּ֤לֶךְ הִיא֙ וְרֵ֣עוֹתֶ֔יהָ וַתֵּ֥בְךְּ עַל־ בְּתוּלֶ֖יהָ עַל־ הֶהָרִֽים׃ (ח) וַיְהִ֞י מִקֵּ֣ץ ׀ שְׁנַ֣יִם חֳדָשִׁ֗ים וַתָּ֙שָׁב֙ אֶל־ אָבִ֔יהָ וַיַּ֣עַשׂ לָ֔הּ אֶת־ נִדְר֖וֹ אֲשֶׁ֣ר נָדָ֑ר וְהִיא֙ לֹא־ יָדְעָ֣ה אִ֔ישׁ וַתְּהִי־ חֹ֖ק בְּיִשְׂרָאֵֽל׃ (ט) מִיָּמִ֣ים ׀ יָמִ֗ימָה תֵּלַ֙כְנָה֙ בְּנ֣וֹת יִשְׂרָאֵ֔ל לְתַנּ֕וֹת לְבַת־ יִפְתָּ֖ח הַגִּלְעָדִ֑י אַרְבַּ֥עַת יָמִ֖ים בַּשָּׁנָֽה׃ (י) וַיִּצָּעֵק֙ אִ֣ישׁ אֶפְרַ֔יִם וַֽיַּעֲבֹ֖ר צָפ֑וֹנָה וַיֹּאמְר֨וּ לְיִפְתָּ֜ח מַדּ֣וּעַ ׀ עָבַ֣רְתָּ ׀ לְהִלָּחֵ֣ם בִּבְנֵי־ עַמּ֗וֹן וְלָ֙נוּ֙ לֹ֤א קָרָ֙אתָ֙ לָלֶ֣כֶת עִמָּ֔ךְ בֵּיתְךָ֕ נִשְׂרֹ֥ף עָלֶ֖יךָ בָּאֵֽשׁ׃ (יא) וַיֹּ֤אמֶר יִפְתָּח֙ אֲלֵיהֶ֔ם אִ֣ישׁ רִ֗יב הָיִ֛יתִי אֲנִ֛י וְעַמִּ֥י וּבְנֵֽי־ עַמּ֖וֹן מְאֹ֑ד וָאֶזְעַ֣ק אֶתְכֶ֔ם וְלֹֽא־ הוֹשַׁעְתֶּ֥ם אוֹתִ֖י מִיָּדָֽם׃ (יב) וָֽאֶרְאֶ֞ה כִּֽי־ אֵינְךָ֣ מוֹשִׁ֗יע וָאָשִׂ֨ימָה נַפְשִׁ֤י בְכַפִּי֙ וָֽאֶעְבְּרָה֙ אֶל־ בְּנֵ֣י עַמּ֔וֹן וַיִּתְּנֵ֥ם יְהוָ֖ה בְּיָדִ֑י וְלָמָ֞ה עֲלִיתֶ֥ם אֵלַ֛י הַיּ֥וֹם הַזֶּ֖ה לְהִלָּ֥חֶם בִּֽי׃ (יג) וַיִּקְבֹּ֤ץ יִפְתָּח֙ אֶת־ כָּל־ אַנְשֵׁ֣י גִלְעָ֔ד וַיִּלָּ֖חֶם אֶת־ אֶפְרָ֑יִם וַיַּכּוּ֩ אַנְשֵׁ֨י גִלְעָ֜ד אֶת־ אֶפְרַ֗יִם כִּ֤י אָמְרוּ֙ פְּלִיטֵ֤י אֶפְרַ֙יִם֙ אַתֶּ֔ם גִּלְעָ֕ד בְּת֥וֹךְ אֶפְרַ֖יִם בְּת֥וֹךְ מְנַשֶּֽׁה׃ (יד) וַיִּלְכֹּ֥ד גִּלְעָ֛ד אֶֽת־ מַעְבְּר֥וֹת הַיַּרְדֵּ֖ן לְאֶפְרָ֑יִם וְֽ֠הָיָה כִּ֣י יֹאמְר֞וּ פְּלִיטֵ֤י אֶפְרַ֙יִם֙ אֶעֱבֹ֔רָה וַיֹּ֨אמְרוּ ל֧וֹ אַנְשֵֽׁי־ גִלְעָ֛ד הַֽאֶפְרָתִ֥י אַ֖תָּה וַיֹּ֥אמֶֽר ׀ לֹֽא׃ (טו) וַיֹּ֣אמְרוּ לוֹ֩ אֱמָר־ נָ֨א שִׁבֹּ֜לֶת וַיֹּ֣אמֶר סִבֹּ֗לֶת וְלֹ֤א יָכִין֙ לְדַבֵּ֣ר כֵּ֔ן וַיֹּאחֲז֣וּ אוֹת֔וֹ וַיִּשְׁחָט֖וּהוּ אֶל־ מַעְבְּר֣וֹת הַיַּרְדֵּ֑ן וַיִּפֹּ֞ל בָּעֵ֤ת הַהִיא֙ מֵֽאֶפְרַ֔יִם אַרְבָּעִ֥ים וּשְׁנַ֖יִם אָֽלֶף׃ (טז) וַיִּשְׁפֹּ֥ט יִפְתָּ֛ח אֶת־ יִשְׂרָאֵ֖ל שֵׁ֣שׁ שָׁנִ֑ים וַיָּ֗מָת יִפְתָּח֙ הַגִּלְעָדִ֔י וַיִּקָּבֵ֖ר בְּעָרֵ֥י גִלְעָֽד׃ (יז) וַיִּשְׁפֹּ֤ט אַֽחֲרָיו֙ אֶת־ יִשְׂרָאֵ֔ל אִבְצָ֖ן מִבֵּ֥ית לָֽחֶם׃ (יח) וַיְהִי־ ל֞וֹ שְׁלֹשִׁ֣ים בָּנִ֗ים וּשְׁלֹשִׁ֤ים בָּנוֹת֙ שִׁלַּ֣ח הַח֔וּצָה וּשְׁלֹשִׁ֣ים בָּנ֔וֹת הֵבִ֥יא לְבָנָ֖יו מִן־ הַח֑וּץ וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל שֶׁ֥בַע שָׁנִֽים׃ (יט) וַיָּ֣מָת אִבְצָ֔ן וַיִּקָּבֵ֖ר בְּבֵ֥ית לָֽחֶם׃ (כ) וַיִּשְׁפֹּ֤ט אַֽחֲרָיו֙ אֶת־ יִשְׂרָאֵ֔ל אֵיל֖וֹן הַזְּבֽוּלֹנִ֑י וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל עֶ֥שֶׂר שָׁנִֽים׃ (כא) וַיָּ֖מָת אֵל֣וֹן הַזְּבֽוּלֹנִ֑י וַיִּקָּבֵ֥ר בְּאַיָּל֖וֹן בְּאֶ֥רֶץ זְבוּלֻֽן׃ (כב) וַיִּשְׁפֹּ֥ט אַחֲרָ֖יו אֶת־ יִשְׂרָאֵ֑ל עַבְדּ֥וֹן בֶּן־ הִלֵּ֖ל הַפִּרְעָתוֹנִֽי׃ (כג) וַיְהִי־ ל֞וֹ אַרְבָּעִ֣ים בָּנִ֗ים וּשְׁלֹשִׁים֙ בְּנֵ֣י בָנִ֔ים רֹכְבִ֖ים עַל־ שִׁבְעִ֣ים עֲיָרִ֑ם וַיִּשְׁפֹּ֥ט אֶת־ יִשְׂרָאֵ֖ל שְׁמֹנֶ֥ה שָׁנִֽים׃ (כד) וַיָּ֛מָת עַבְדּ֥וֹן בֶּן־ הִלֵּ֖ל הַפִּרְעָתוֹנִ֑י וַיִּקָּבֵ֤ר בְּפִרְעָתוֹן֙ בְּאֶ֣רֶץ אֶפְרַ֔יִם בְּהַ֖ר הָעֲמָלֵקִֽי׃ (כה) וַיֹּסִ֙פוּ֙ בְּנֵ֣י יִשְׂרָאֵ֔ל לַעֲשׂ֥וֹת הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַיִּתְּנֵ֧ם יְהוָ֛ה בְּיַד־ פְּלִשְׁתִּ֖ים אַרְבָּעִ֥ים שָׁנָֽה׃ (כו) וַיְהִי֩ אִ֨ישׁ אֶחָ֧ד מִצָּרְעָ֛ה מִמִּשְׁפַּ֥חַת הַדָּנִ֖י וּשְׁמ֣וֹ מָנ֑וֹחַ וְאִשְׁתּ֥וֹ עֲקָרָ֖ה וְלֹ֥א יָלָֽדָה׃ (כז) וַיֵּרָ֥א מַלְאַךְ־ יְהוָ֖ה אֶל־ הָאִשָּׁ֑ה וַיֹּ֣אמֶר אֵלֶ֗יהָ הִנֵּה־ נָ֤א אַתְּ־ עֲקָרָה֙ וְלֹ֣א יָלַ֔דְתְּ וְהָרִ֖ית וְיָלַ֥דְתְּ בֵּֽן׃ (כח) וְעַתָּה֙ הִשָּׁ֣מְרִי נָ֔א וְאַל־ תִּשְׁתִּ֖י יַ֣יִן וְשֵׁכָ֑ר וְאַל־ תֹּאכְלִ֖י כָּל־ טָמֵֽא׃ (כט) כִּי֩ הִנָּ֨ךְ הָרָ֜ה וְיֹלַ֣דְתְּ בֵּ֗ן וּמוֹרָה֙ לֹא־ יַעֲלֶ֣ה עַל־ רֹאשׁ֔וֹ כִּֽי־ נְזִ֧יר אֱלֹהִ֛ים יִהְיֶ֥ה הַנַּ֖עַר מִן־ הַבָּ֑טֶן וְה֗וּא יָחֵ֛ל לְהוֹשִׁ֥יעַ אֶת־ יִשְׂרָאֵ֖ל מִיַּ֥ד פְּלִשְׁתִּֽים׃ (ל) וַתָּבֹ֣א הָאִשָּׁ֗ה וַתֹּ֣אמֶר לְאִישָׁהּ֮ לֵאמֹר֒ אִ֤ישׁ הָאֱלֹהִים֙ בָּ֣א אֵלַ֔י וּמַרְאֵ֕הוּ כְּמַרְאֵ֛ה מַלְאַ֥ךְ הָאֱלֹהִ֖ים נוֹרָ֣א מְאֹ֑ד וְלֹ֤א שְׁאִלְתִּ֙יהוּ֙ אֵֽי־ מִזֶּ֣ה ה֔וּא וְאֶת־ שְׁמ֖וֹ לֹֽא־ הִגִּ֥יד לִֽי׃ (לא) וַיֹּ֣אמֶר לִ֔י הִנָּ֥ךְ הָרָ֖ה וְיֹלַ֣דְתְּ בֵּ֑ן וְעַתָּ֞ה אַל־ תִּשְׁתִּ֣י ׀ יַ֣יִן וְשֵׁכָ֗ר וְאַל־ תֹּֽאכְלִי֙ כָּל־ טֻמְאָ֔ה כִּֽי־ נְזִ֤יר אֱלֹהִים֙ יִהְיֶ֣ה הַנַּ֔עַר מִן־ הַבֶּ֖טֶן עַד־ י֥וֹם מוֹתֽוֹ׃ (לב) וַיֶּעְתַּ֥ר מָנ֛וֹחַ אֶל־ יְהוָ֖ה וַיֹּאמַ֑ר בִּ֣י אֲדוֹנָ֔י אִ֣ישׁ הָאֱלֹהִ֞ים אֲשֶׁ֣ר שָׁלַ֗חְתָּ יָבוֹא־ נָ֥א עוֹד֙ אֵלֵ֔ינוּ וְיוֹרֵ֕נוּ מַֽה־ נַּעֲשֶׂ֖ה לַנַּ֥עַר הַיּוּלָּֽד׃ (לג) וַיִּשְׁמַ֥ע הָאֱלֹהִ֖ים בְּק֣וֹל מָנ֑וֹחַ וַיָּבֹ֣א מַלְאַךְ֩ הָאֱלֹהִ֨ים ע֜וֹד אֶל־ הָאִשָּׁ֗ה וְהִיא֙ יוֹשֶׁ֣בֶת בַּשָּׂדֶ֔ה וּמָנ֥וֹחַ אִישָׁ֖הּ אֵ֥ין עִמָּֽהּ׃ (לד) וַתְּמַהֵר֙ הָֽאִשָּׁ֔ה וַתָּ֖רָץ וַתַּגֵּ֣ד לְאִישָׁ֑הּ וַתֹּ֣אמֶר אֵלָ֔יו הִנֵּ֨ה נִרְאָ֤ה אֵלַי֙ הָאִ֔ישׁ אֲשֶׁר־ בָּ֥א בַיּ֖וֹם אֵלָֽי׃ (לה) וַיָּ֛קָם וַיֵּ֥לֶךְ מָנ֖וֹחַ אַחֲרֵ֣י אִשְׁתּ֑וֹ וַיָּבֹא֙ אֶל־ הָאִ֔ישׁ וַיֹּ֣אמֶר ל֗וֹ הַאַתָּ֥ה הָאִ֛ישׁ אֲשֶׁר־ דִּבַּ֥רְתָּ אֶל־ הָאִשָּׁ֖ה וַיֹּ֥אמֶר אָֽנִי׃ (לו) וַיֹּ֣אמֶר מָנ֔וֹחַ עַתָּ֖ה יָבֹ֣א דְבָרֶ֑יךָ מַה־ יִּֽהְיֶ֥ה מִשְׁפַּט־ הַנַּ֖עַר וּמַעֲשֵֽׂהוּ׃ (לז) וַיֹּ֛אמֶר מַלְאַ֥ךְ יְהוָ֖ה אֶל־ מָנ֑וֹחַ מִכֹּ֛ל אֲשֶׁר־ אָמַ֥רְתִּי אֶל־ הָאִשָּׁ֖ה תִּשָּׁמֵֽר׃ (לח) מִכֹּ֣ל אֲשֶׁר־ יֵצֵא֩ מִגֶּ֨פֶן הַיַּ֜יִן לֹ֣א תֹאכַ֗ל וְיַ֤יִן וְשֵׁכָר֙ אַל־ תֵּ֔שְׁתְּ וְכָל־ טֻמְאָ֖ה אַל־ תֹּאכַ֑ל כֹּ֥ל אֲשֶׁר־ צִוִּיתִ֖יהָ תִּשְׁמֹֽר׃ (לט) וַיֹּ֥אמֶר מָנ֖וֹחַ אֶל־ מַלְאַ֣ךְ יְהוָ֑ה נַעְצְרָה־ נָּ֣א אוֹתָ֔ךְ וְנַעֲשֶׂ֥ה לְפָנֶ֖יךָ גְּדִ֥י עִזִּֽים׃ (מ) וַיֹּאמֶר֩ מַלְאַ֨ךְ יְהוָ֜ה אֶל־ מָנ֗וֹחַ אִם־ תַּעְצְרֵ֙נִי֙ לֹא־ אֹכַ֣ל בְּלַחְמֶ֔ךָ וְאִם־ תַּעֲשֶׂ֣ה עֹלָ֔ה לַיהוָ֖ה תַּעֲלֶ֑נָּה כִּ֚י לֹא־ יָדַ֣ע מָנ֔וֹחַ כִּֽי־ מַלְאַ֥ךְ יְהוָ֖ה הֽוּא׃ (מא) וַיֹּ֧אמֶר מָנ֛וֹחַ אֶל־ מַלְאַ֥ךְ יְהוָ֖ה מִ֣י שְׁמֶ֑ךָ כִּֽי־ יָבֹ֥א (דבריך) דְבָרְךָ֖ וְכִבַּדְנֽוּךָ׃ (מב) וַיֹּ֤אמֶר לּוֹ֙ מַלְאַ֣ךְ יְהוָ֔ה לָ֥מָּה זֶּ֖ה תִּשְׁאַ֣ל לִשְׁמִ֑י וְהוּא־ פֶ֛לִאי׃ (מג) וַיִּקַּ֨ח מָנ֜וֹחַ אֶת־ גְּדִ֤י הָעִזִּים֙ וְאֶת־ הַמִּנְחָ֔ה וַיַּ֥עַל עַל־ הַצּ֖וּר לַֽיהוָ֑ה וּמַפְלִ֣א לַעֲשׂ֔וֹת וּמָנ֥וֹחַ וְאִשְׁתּ֖וֹ רֹאִֽים׃ (מד) וַיְהִי֩ בַעֲל֨וֹת הַלַּ֜הַב מֵעַ֤ל הַמִּזְבֵּ֙חַ֙ הַשָּׁמַ֔יְמָה וַיַּ֥עַל מַלְאַךְ־ יְהוָ֖ה בְּלַ֣הַב הַמִּזְבֵּ֑חַ וּמָנ֤וֹחַ וְאִשְׁתּוֹ֙ רֹאִ֔ים וַיִּפְּל֥וּ עַל־ פְּנֵיהֶ֖ם אָֽרְצָה׃ (מה) וְלֹא־ יָ֤סַף עוֹד֙ מַלְאַ֣ךְ יְהוָ֔ה לְהֵרָאֹ֖ה אֶל־ מָנ֣וֹחַ וְאֶל־ אִשְׁתּ֑וֹ אָ֚ז יָדַ֣ע מָנ֔וֹחַ כִּֽי־ מַלְאַ֥ךְ יְהוָ֖ה הֽוּא׃ (מו) וַיֹּ֧אמֶר מָנ֛וֹחַ אֶל־ אִשְׁתּ֖וֹ מ֣וֹת נָמ֑וּת כִּ֥י אֱלֹהִ֖ים רָאִֽינוּ׃ (מז) וַתֹּ֧אמֶר ל֣וֹ אִשְׁתּ֗וֹ לוּ֩ חָפֵ֨ץ יְהוָ֤ה לַהֲמִיתֵ֙נוּ֙ לֹֽא־ לָקַ֤ח מִיָּדֵ֙נוּ֙ עֹלָ֣ה וּמִנְחָ֔ה וְלֹ֥א הֶרְאָ֖נוּ אֶת־ כָּל־ אֵ֑לֶּה וְכָעֵ֕ת לֹ֥א הִשְׁמִיעָ֖נוּ כָּזֹֽאת׃
