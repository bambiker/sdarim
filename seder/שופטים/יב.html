שופטים סדר יב

פתיחה באתר · ספר שופטים כולו ·

(א) וַיֹּ֧אמֶר לָהֶ֛ם הַכֹּהֵ֖ן לְכ֣וּ לְשָׁל֑וֹם נֹ֣כַח יְהוָ֔ה דַּרְכְּכֶ֖ם אֲשֶׁ֥ר תֵּֽלְכוּ־ בָֽהּ׃ (ב) וַיֵּלְכוּ֙ חֲמֵ֣שֶׁת הָאֲנָשִׁ֔ים וַיָּבֹ֖אוּ לָ֑יְשָׁה וַיִּרְא֣וּ אֶת־ הָעָ֣ם אֲשֶׁר־ בְּקִרְבָּ֣הּ יוֹשֶֽׁבֶת־ לָ֠בֶטַח כְּמִשְׁפַּ֨ט צִדֹנִ֜ים שֹׁקֵ֣ט ׀ וּבֹטֵ֗חַ וְאֵין־ מַכְלִ֨ים דָּבָ֤ר בָּאָ֙רֶץ֙ יוֹרֵ֣שׁ עֶ֔צֶר וּרְחֹקִ֥ים הֵ֙מָּה֙ מִצִּ֣דֹנִ֔ים וְדָבָ֥ר אֵין־ לָהֶ֖ם עִם־ אָדָֽם׃ (ג) וַיָּבֹ֙אוּ֙ אֶל־ אֲחֵיהֶ֔ם צָרְעָ֖ה וְאֶשְׁתָּאֹ֑ל וַיֹּאמְר֥וּ לָהֶ֛ם אֲחֵיהֶ֖ם מָ֥ה אַתֶּֽם׃ (ד) וַיֹּאמְר֗וּ ק֚וּמָה וְנַעֲלֶ֣ה עֲלֵיהֶ֔ם כִּ֤י רָאִ֙ינוּ֙ אֶת־ הָאָ֔רֶץ וְהִנֵּ֥ה טוֹבָ֖ה מְאֹ֑ד וְאַתֶּ֣ם מַחְשִׁ֔ים אַל־ תֵּעָ֣צְל֔וּ לָלֶ֥כֶת לָבֹ֖א לָרֶ֥שֶׁת אֶת־ הָאָֽרֶץ׃ (ה) כְּבֹאֲכֶ֞ם תָּבֹ֣אוּ ׀ אֶל־ עַ֣ם בֹּטֵ֗חַ וְהָאָ֙רֶץ֙ רַחֲבַ֣ת יָדַ֔יִם כִּֽי־ נְתָנָ֥הּ אֱלֹהִ֖ים בְּיֶדְכֶ֑ם מָקוֹם֙ אֲשֶׁ֣ר אֵֽין־ שָׁ֣ם מַחְס֔וֹר כָּל־ דָּבָ֖ר אֲשֶׁ֥ר בָּאָֽרֶץ׃ (ו) וַיִּסְע֤וּ מִשָּׁם֙ מִמִּשְׁפַּ֣חַת הַדָּנִ֔י מִצָּרְעָ֖ה וּמֵאֶשְׁתָּאֹ֑ל שֵֽׁשׁ־ מֵא֣וֹת אִ֔ישׁ חָג֖וּר כְּלֵ֥י מִלְחָמָֽה׃ (ז) וַֽיַּעֲל֗וּ וַֽיַּחֲנ֛וּ בְּקִרְיַ֥ת יְעָרִ֖ים בִּֽיהוּדָ֑ה עַל־ כֵּ֡ן קָרְאוּ֩ לַמָּק֨וֹם הַה֜וּא מַחֲנֵה־ דָ֗ן עַ֚ד הַיּ֣וֹם הַזֶּ֔ה הִנֵּ֕ה אַחֲרֵ֖י קִרְיַ֥ת יְעָרִֽים׃ (ח) וַיַּעַבְר֥וּ מִשָּׁ֖ם הַר־ אֶפְרָ֑יִם וַיָּבֹ֖אוּ עַד־ בֵּ֥ית מִיכָֽה׃ (ט) וַֽיַּעֲנ֞וּ חֲמֵ֣שֶׁת הָאֲנָשִׁ֗ים הַהֹלְכִים֮ לְרַגֵּל֮ אֶת־ הָאָ֣רֶץ לַיִשׁ֒ וַיֹּֽאמְרוּ֙ אֶל־ אֲחֵיהֶ֔ם הַיְדַעְתֶּ֗ם כִּ֠י יֵ֣שׁ בַּבָּתִּ֤ים הָאֵ֙לֶּה֙ אֵפ֣וֹד וּתְרָפִ֔ים וּפֶ֖סֶל וּמַסֵּכָ֑ה וְעַתָּ֖ה דְּע֥וּ מַֽה־ תַּעֲשֽׂוּ׃ (י) וַיָּס֣וּרוּ שָׁ֔מָּה וַיָּבֹ֛אוּ אֶל־ בֵּֽית־ הַנַּ֥עַר הַלֵּוִ֖י בֵּ֣ית מִיכָ֑ה וַיִּשְׁאֲלוּ־ ל֖וֹ לְשָׁלֽוֹם׃ (יא) וְשֵׁשׁ־ מֵא֣וֹת אִ֗ישׁ חֲגוּרִים֙ כְּלֵ֣י מִלְחַמְתָּ֔ם נִצָּבִ֖ים פֶּ֣תַח הַשָּׁ֑עַר אֲשֶׁ֖ר מִבְּנֵי־ דָֽן׃ (יב) וַֽיַּעֲל֞וּ חֲמֵ֣שֶׁת הָאֲנָשִׁ֗ים הַהֹלְכִים֮ לְרַגֵּ֣ל אֶת־ הָאָרֶץ֒ בָּ֣אוּ שָׁ֔מָּה לָקְח֗וּ אֶת־ הַפֶּ֙סֶל֙ וְאֶת־ הָ֣אֵפ֔וֹד וְאֶת־ הַתְּרָפִ֖ים וְאֶת־ הַמַּסֵּכָ֑ה וְהַכֹּהֵ֗ן נִצָּב֙ פֶּ֣תַח הַשַּׁ֔עַר וְשֵׁשׁ־ מֵא֣וֹת הָאִ֔ישׁ הֶחָג֖וּר כְּלֵ֥י הַמִּלְחָמָֽה׃ (יג) וְאֵ֗לֶּה בָּ֚אוּ בֵּ֣ית מִיכָ֔ה וַיִּקְחוּ֙ אֶת־ פֶּ֣סֶל הָאֵפ֔וֹד וְאֶת־ הַתְּרָפִ֖ים וְאֶת־ הַמַּסֵּכָ֑ה וַיֹּ֤אמֶר אֲלֵיהֶם֙ הַכֹּהֵ֔ן מָ֥ה אַתֶּ֖ם עֹשִֽׂים׃ (יד) וַיֹּאמְרוּ֩ ל֨וֹ הַחֲרֵ֜שׁ שִֽׂים־ יָדְךָ֤ עַל־ פִּ֙יךָ֙ וְלֵ֣ךְ עִמָּ֔נוּ וֶֽהְיֵה־ לָ֖נוּ לְאָ֣ב וּלְכֹהֵ֑ן הֲט֣וֹב ׀ הֱיוֹתְךָ֣ כֹהֵ֗ן לְבֵית֙ אִ֣ישׁ אֶחָ֔ד א֚וֹ הֱיוֹתְךָ֣ כֹהֵ֔ן לְשֵׁ֥בֶט וּלְמִשְׁפָּחָ֖ה בְּיִשְׂרָאֵֽל׃ (טו) וַיִּיטַב֙ לֵ֣ב הַכֹּהֵ֔ן וַיִּקַּח֙ אֶת־ הָ֣אֵפ֔וֹד וְאֶת־ הַתְּרָפִ֖ים וְאֶת־ הַפָּ֑סֶל וַיָּבֹ֖א בְּקֶ֥רֶב הָעָֽם׃ (טז) וַיִּפְנ֖וּ וַיֵּלֵ֑כוּ וַיָּשִׂ֨ימוּ אֶת־ הַטַּ֧ף וְאֶת־ הַמִּקְנֶ֛ה וְאֶת־ הַכְּבוּדָּ֖ה לִפְנֵיהֶֽם׃ (יז) הֵ֥מָּה הִרְחִ֖יקוּ מִבֵּ֣ית מִיכָ֑ה וְהָאֲנָשִׁ֗ים אֲשֶׁ֤ר בַּבָּתִּים֙ אֲשֶׁר֙ עִם־ בֵּ֣ית מִיכָ֔ה נִֽזְעֲק֔וּ וַיַּדְבִּ֖יקוּ אֶת־ בְּנֵי־ דָֽן׃ (יח) וַֽיִּקְרְאוּ֙ אֶל־ בְּנֵי־ דָ֔ן וַיַּסֵּ֖בּוּ פְּנֵיהֶ֑ם וַיֹּאמְר֣וּ לְמִיכָ֔ה מַה־ לְּךָ֖ כִּ֥י נִזְעָֽקְתָּ׃ (יט) וַיֹּ֡אמֶר אֶת־ אֱלֹהַי֩ אֲשֶׁר־ עָשִׂ֨יתִי לְקַחְתֶּ֧ם וְֽאֶת־ הַכֹּהֵ֛ן וַתֵּלְכ֖וּ וּמַה־ לִּ֣י ע֑וֹד וּמַה־ זֶּ֛ה תֹּאמְר֥וּ אֵלַ֖י מַה־ לָּֽךְ׃ (כ) וַיֹּאמְר֤וּ אֵלָיו֙ בְּנֵי־ דָ֔ן אַל־ תַּשְׁמַ֥ע קוֹלְךָ֖ עִמָּ֑נוּ פֶּֽן־ יִפְגְּע֣וּ בָכֶ֗ם אֲנָשִׁים֙ מָ֣רֵי נֶ֔פֶשׁ וְאָסַפְתָּ֥ה נַפְשְׁךָ֖ וְנֶ֥פֶשׁ בֵּיתֶֽךָ׃ (כא) וַיֵּלְכ֥וּ בְנֵי־ דָ֖ן לְדַרְכָּ֑ם וַיַּ֣רְא מִיכָ֗ה כִּי־ חֲזָקִ֥ים הֵ֙מָּה֙ מִמֶּ֔נּוּ וַיִּ֖פֶן וַיָּ֥שָׁב אֶל־ בֵּיתֽוֹ׃ (כב) וְהֵ֨מָּה לָקְח֜וּ אֵ֧ת אֲשֶׁר־ עָשָׂ֣ה מִיכָ֗ה וְֽאֶת־ הַכֹּהֵן֮ אֲשֶׁ֣ר הָיָה־ לוֹ֒ וַיָּבֹ֣אוּ עַל־ לַ֗יִשׁ עַל־ עַם֙ שֹׁקֵ֣ט וּבֹטֵ֔חַ וַיַּכּ֥וּ אוֹתָ֖ם לְפִי־ חָ֑רֶב וְאֶת־ הָעִ֖יר שָׂרְפ֥וּ בָאֵֽשׁ׃ (כג) וְאֵ֨ין מַצִּ֜יל כִּ֧י רְֽחוֹקָה־ הִ֣יא מִצִּיד֗וֹן וְדָבָ֤ר אֵין־ לָהֶם֙ עִם־ אָדָ֔ם וְהִ֕יא בָּעֵ֖מֶק אֲשֶׁ֣ר לְבֵית־ רְח֑וֹב וַיִּבְנ֥וּ אֶת־ הָעִ֖יר וַיֵּ֥שְׁבוּ בָֽהּ׃ (כד) וַיִּקְרְא֤וּ שֵׁם־ הָעִיר֙ דָּ֔ן בְּשֵׁם֙ דָּ֣ן אֲבִיהֶ֔ם אֲשֶׁ֥ר יוּלַּ֖ד לְיִשְׂרָאֵ֑ל וְאוּלָ֛ם לַ֥יִשׁ שֵׁם־ הָעִ֖יר לָרִאשֹׁנָֽה׃ (כה) וַיָּקִ֧ימוּ לָהֶ֛ם בְּנֵי־ דָ֖ן אֶת־ הַפָּ֑סֶל וִ֠יהוֹנָתָן בֶּן־ גֵּרְשֹׁ֨ם בֶּן־ מְ ה֣וּא וּבָנָ֗יו הָי֤וּ כֹהֲנִים֙ לְשֵׁ֣בֶט הַדָּנִ֔י עַד־ י֖וֹם גְּל֥וֹת הָאָֽרֶץ׃ (כו) וַיָּשִׂ֣ימוּ לָהֶ֔ם אֶת־ פֶּ֥סֶל מִיכָ֖ה אֲשֶׁ֣ר עָשָׂ֑ה כָּל־ יְמֵ֛י הֱי֥וֹת בֵּית־ הָאֱלֹהִ֖ים בְּשִׁלֹֽה׃ (כז) וַיְהִי֙ בַּיָּמִ֣ים הָהֵ֔ם וּמֶ֖לֶךְ אֵ֣ין בְּיִשְׂרָאֵ֑ל וַיְהִ֣י ׀ אִ֣ישׁ לֵוִ֗י גָּ֚ר בְּיַרְכְּתֵ֣י הַר־ אֶפְרַ֔יִם וַיִּֽקַּֽח־ לוֹ֙ אִשָּׁ֣ה פִילֶ֔גֶשׁ מִבֵּ֥ית לֶ֖חֶם יְהוּדָֽה׃ (כח) וַתִּזְנֶ֤ה עָלָיו֙ פִּֽילַגְשׁ֔וֹ וַתֵּ֤לֶךְ מֵֽאִתּוֹ֙ אֶל־ בֵּ֣ית אָבִ֔יהָ אֶל־ בֵּ֥ית לֶ֖חֶם יְהוּדָ֑ה וַתְּהִי־ שָׁ֕ם יָמִ֖ים אַרְבָּעָ֥ה חֳדָשִֽׁים׃ (כט) וַיָּ֨קָם אִישָׁ֜הּ וַיֵּ֣לֶךְ אַחֲרֶ֗יהָ לְדַבֵּ֤ר עַל־ לִבָּהּ֙ (להשיבו) לַהֲשִׁיבָ֔הּ וְנַעֲר֥וֹ עִמּ֖וֹ וְצֶ֣מֶד חֲמֹרִ֑ים וַתְּבִיאֵ֙הוּ֙ בֵּ֣ית אָבִ֔יהָ וַיִּרְאֵ֙הוּ֙ אֲבִ֣י הַֽנַּעֲרָ֔ה וַיִּשְׂמַ֖ח לִקְרָאתֽוֹ׃ (ל) וַיֶּחֱזַק־ בּ֤וֹ חֹֽתְנוֹ֙ אֲבִ֣י הַֽנַּעֲרָ֔ה וַיֵּ֥שֶׁב אִתּ֖וֹ שְׁלֹ֣שֶׁת יָמִ֑ים וַיֹּאכְלוּ֙ וַיִּשְׁתּ֔וּ וַיָּלִ֖ינוּ שָֽׁם׃ (לא) וַֽיְהִי֙ בַּיּ֣וֹם הָרְבִיעִ֔י וַיַּשְׁכִּ֥ימוּ בַבֹּ֖קֶר וַיָּ֣קָם לָלֶ֑כֶת וַיֹּאמֶר֩ אֲבִ֨י הַֽנַּעֲרָ֜ה אֶל־ חֲתָנ֗וֹ סְעָ֧ד לִבְּךָ֛ פַּת־ לֶ֖חֶם וְאַחַ֥ר תֵּלֵֽכוּ׃ (לב) וַיֵּשְׁב֗וּ וַיֹּאכְל֧וּ שְׁנֵיהֶ֛ם יַחְדָּ֖ו וַיִּשְׁתּ֑וּ וַיֹּ֜אמֶר אֲבִ֤י הַֽנַּעֲרָה֙ אֶל־ הָאִ֔ישׁ הֽוֹאֶל־ נָ֥א וְלִ֖ין וְיִטַ֥ב לִבֶּֽךָ׃ (לג) וַיָּ֥קָם הָאִ֖ישׁ לָלֶ֑כֶת וַיִּפְצַר־ בּוֹ֙ חֹתְנ֔וֹ וַיָּ֖שָׁב וַיָּ֥לֶן שָֽׁם׃ (לד) וַיַּשְׁכֵּ֨ם בַּבֹּ֜קֶר בַּיּ֣וֹם הַחֲמִישִׁי֮ לָלֶכֶת֒ וַיֹּ֣אמֶר ׀ אֲבִ֣י הַֽנַּעֲרָ֗ה סְעָד־ נָא֙ לְבָ֣בְךָ֔ וְהִֽתְמַהְמְה֖וּ עַד־ נְט֣וֹת הַיּ֑וֹם וַיֹּאכְל֖וּ שְׁנֵיהֶֽם׃ (לה) וַיָּ֤קָם הָאִישׁ֙ לָלֶ֔כֶת ה֥וּא וּפִילַגְשׁ֖וֹ וְנַעֲר֑וֹ וַיֹּ֣אמֶר ל֣וֹ חֹתְנ֣וֹ אֲבִ֣י הַֽנַּעֲרָ֡ה הִנֵּ֣ה נָא֩ רָפָ֨ה הַיּ֜וֹם לַעֲרֹ֗ב לִֽינוּ־ נָ֞א הִנֵּ֨ה חֲנ֤וֹת הַיּוֹם֙ לִ֥ין פֹּה֙ וְיִיטַ֣ב לְבָבֶ֔ךָ וְהִשְׁכַּמְתֶּ֤ם מָחָר֙ לְדַרְכְּכֶ֔ם וְהָלַכְתָּ֖ לְאֹהָלֶֽךָ׃ (לו) וְלֹֽא־ אָבָ֤ה הָאִישׁ֙ לָל֔וּן וַיָּ֣קָם וַיֵּ֗לֶךְ וַיָּבֹא֙ עַד־ נֹ֣כַח יְב֔וּס הִ֖יא יְרוּשָׁלִָ֑ם וְעִמּ֗וֹ צֶ֤מֶד חֲמוֹרִים֙ חֲבוּשִׁ֔ים וּפִילַגְשׁ֖וֹ עִמּֽוֹ׃ (לז) הֵ֣ם עִם־ יְב֔וּס וְהַיּ֖וֹם רַ֣ד מְאֹ֑ד וַיֹּ֨אמֶר הַנַּ֜עַר אֶל־ אֲדֹנָ֗יו לְכָה־ נָּ֛א וְנָס֛וּרָה אֶל־ עִֽיר־ הַיְבוּסִ֥י הַזֹּ֖את וְנָלִ֥ין בָּֽהּ׃ (לח) וַיֹּ֤אמֶר אֵלָיו֙ אֲדֹנָ֔יו לֹ֤א נָסוּר֙ אֶל־ עִ֣יר נָכְרִ֔י אֲשֶׁ֛ר לֹֽא־ מִבְּנֵ֥י יִשְׂרָאֵ֖ל הֵ֑נָּה וְעָבַ֖רְנוּ עַד־ גִּבְעָֽה׃ (לט) וַיֹּ֣אמֶר לְנַעֲר֔וֹ לְךָ֥ וְנִקְרְבָ֖ה בְּאַחַ֣ד הַמְּקֹמ֑וֹת וְלַ֥נּוּ בַגִּבְעָ֖ה א֥וֹ בָרָמָֽה׃ (מ) וַיַּעַבְר֖וּ וַיֵּלֵ֑כוּ וַתָּבֹ֤א לָהֶם֙ הַשֶּׁ֔מֶשׁ אֵ֥צֶל הַגִּבְעָ֖ה אֲשֶׁ֥ר לְבִנְיָמִֽן׃ (מא) וַיָּסֻ֣רוּ שָׁ֔ם לָב֖וֹא לָל֣וּן בַּגִּבְעָ֑ה וַיָּבֹ֗א וַיֵּ֙שֶׁב֙ בִּרְח֣וֹב הָעִ֔יר וְאֵ֥ין אִ֛ישׁ מְאַסֵּֽף־ אוֹתָ֥ם הַבַּ֖יְתָה לָלֽוּן׃ (מב) וְהִנֵּ֣ה ׀ אִ֣ישׁ זָקֵ֗ן בָּ֣א מִֽן־ מַעֲשֵׂ֤הוּ מִן־ הַשָּׂדֶה֙ בָּעֶ֔רֶב וְהָאִישׁ֙ מֵהַ֣ר אֶפְרַ֔יִם וְהוּא־ גָ֖ר בַּגִּבְעָ֑ה וְאַנְשֵׁ֥י הַמָּק֖וֹם בְּנֵ֥י יְמִינִֽי׃ (מג) וַיִּשָּׂ֣א עֵינָ֗יו וַיַּ֛רְא אֶת־ הָאִ֥ישׁ הָאֹרֵ֖חַ בִּרְחֹ֣ב הָעִ֑יר וַיֹּ֨אמֶר הָאִ֧ישׁ הַזָּקֵ֛ן אָ֥נָה תֵלֵ֖ךְ וּמֵאַ֥יִן תָּבֽוֹא׃ (מד) וַיֹּ֣אמֶר אֵלָ֗יו עֹבְרִ֨ים אֲנַ֜חְנוּ מִבֵּֽית־ לֶ֣חֶם יְהוּדָה֮ עַד־ יַרְכְּתֵ֣י הַר־ אֶפְרַיִם֒ מִשָּׁ֣ם אָנֹ֔כִי וָאֵלֵ֕ךְ עַד־ בֵּ֥ית לֶ֖חֶם יְהוּדָ֑ה וְאֶת־ בֵּ֤ית יְהוָה֙ אֲנִ֣י הֹלֵ֔ךְ וְאֵ֣ין אִ֔ישׁ מְאַסֵּ֥ף אוֹתִ֖י הַבָּֽיְתָה׃ (מה) וְגַם־ תֶּ֤בֶן גַּם־ מִסְפּוֹא֙ יֵ֣שׁ לַחֲמוֹרֵ֔ינוּ וְ֠גַם לֶ֣חֶם וָיַ֤יִן יֶשׁ־ לִי֙ וְלַֽאֲמָתֶ֔ךָ וְלַנַּ֖עַר עִם־ עֲבָדֶ֑יךָ אֵ֥ין מַחְס֖וֹר כָּל־ דָּבָֽר׃
