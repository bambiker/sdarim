איוב סדר ז

פתיחה באתר · ספר איוב כולו ·

(א) אִם־ אַ֭יִן אַתָּ֥ה שְֽׁמַֽע־ לִ֑י הַ֝חֲרֵ֗שׁ וַאֲאַלֶּפְךָ֥ חָכְמָֽה׃ (ב) וַיַּ֥עַן אֱלִיה֗וּא וַיֹּאמַֽר׃ (ג) שִׁמְע֣וּ חֲכָמִ֣ים מִלָּ֑י וְ֝יֹדְעִ֗ים הַאֲזִ֥ינוּ לִֽי׃ (ד) כִּי־ אֹ֭זֶן מִלִּ֣ין תִּבְחָ֑ן וְ֝חֵ֗ךְ יִטְעַ֥ם לֶאֱכֹֽל׃ (ה) מִשְׁפָּ֥ט נִבְחֲרָה־ לָּ֑נוּ נֵדְעָ֖ה בֵינֵ֣ינוּ מַה־ טּֽוֹב׃ (ו) כִּֽי־ אָ֭מַר אִיּ֣וֹב צָדַ֑קְתִּי וְ֝אֵ֗ל הֵסִ֥יר מִשְׁפָּטִֽי׃ (ז) עַל־ מִשְׁפָּטִ֥י אֲכַזֵּ֑ב אָנ֖וּשׁ חִצִּ֣י בְלִי־ פָֽשַׁע׃ (ח) מִי־ גֶ֥בֶר כְּאִיּ֑וֹב יִֽשְׁתֶּה־ לַּ֥עַג כַּמָּֽיִם׃ (ט) וְאָרַ֣ח לְ֭חֶבְרָה עִם־ פֹּ֣עֲלֵי אָ֑וֶן וְ֝לָלֶ֗כֶת עִם־ אַנְשֵׁי־ רֶֽשַׁע׃ (י) כִּֽי־ אָ֭מַר לֹ֣א יִסְכָּן־ גָּ֑בֶר בִּ֝רְצֹת֗וֹ עִם־ אֱלֹהִֽים׃ (יא) לָכֵ֤ן ׀ אַ֥נֲשֵׁ֥י לֵבָ֗ב שִׁמְע֫וּ לִ֥י חָלִ֖לָה לָאֵ֥ל מֵרֶ֗שַׁע וְשַׁדַּ֥י מֵעָֽוֶל׃ (יב) כִּ֤י פֹ֣עַל אָ֭דָם יְשַׁלֶּם־ ל֑וֹ וּֽכְאֹ֥רַח אִ֝֗ישׁ יַמְצִאֶֽנּוּ׃ (יג) אַף־ אָמְנָ֗ם אֵ֥ל לֹֽא־ יַרְשִׁ֑יעַ וְ֝שַׁדַּ֗י לֹֽא־ יְעַוֵּ֥ת מִשְׁפָּֽט׃ (יד) מִֽי־ פָקַ֣ד עָלָ֣יו אָ֑רְצָה וּמִ֥י שָׂ֝֗ם תֵּבֵ֥ל כֻּלָּֽהּ׃ (טו) אִם־ יָשִׂ֣ים אֵלָ֣יו לִבּ֑וֹ רוּח֥וֹ וְ֝נִשְׁמָת֗וֹ אֵלָ֥יו יֶאֱסֹֽף׃ (טז) יִגְוַ֣ע כָּל־ בָּשָׂ֣ר יָ֑חַד וְ֝אָדָ֗ם עַל־ עָפָ֥ר יָשֽׁוּב׃ (יז) וְאִם־ בִּ֥ינָה שִׁמְעָה־ זֹּ֑את הַ֝אֲזִ֗ינָה לְק֣וֹל מִלָּֽי׃ (יח) הַאַ֬ף שׂוֹנֵ֣א מִשְׁפָּ֣ט יַחֲב֑וֹשׁ וְאִם־ צַדִּ֖יק כַּבִּ֣יר תַּרְשִֽׁיעַ׃ (יט) הַאֲמֹ֣ר לְמֶ֣לֶךְ בְּלִיָּ֑עַל רָ֝שָׁ֗ע אֶל־ נְדִיבִֽים׃ (כ) אֲשֶׁ֤ר לֹֽא־ נָשָׂ֨א ׀ פְּנֵ֥י שָׂרִ֗ים וְלֹ֣א נִכַּר־ שׁ֭וֹעַ לִפְנֵי־ דָ֑ל כִּֽי־ מַעֲשֵׂ֖ה יָדָ֣יו כֻּלָּֽם׃ (כא) רֶ֤גַע ׀ יָמֻתוּ֮ וַחֲצ֪וֹת לָ֥יְלָה יְגֹעֲשׁ֣וּ עָ֣ם וְיַעֲבֹ֑רוּ וְיָסִ֥ירוּ אַ֝בִּ֗יר לֹ֣א בְיָֽד׃ (כב) כִּי־ עֵ֭ינָיו עַל־ דַּרְכֵי־ אִ֑ישׁ וְֽכָל־ צְעָדָ֥יו יִרְאֶֽה׃ (כג) אֵֽין־ חֹ֭שֶׁךְ וְאֵ֣ין צַלְמָ֑וֶת לְהִסָּ֥תֶר שָׁ֝֗ם פֹּ֣עֲלֵי אָֽוֶן׃ (כד) כִּ֤י לֹ֣א עַל־ אִ֭ישׁ יָשִׂ֣ים ע֑וֹד לַהֲלֹ֥ךְ אֶל־ אֵ֝֗ל בַּמִּשְׁפָּֽט׃ (כה) יָרֹ֣עַ כַּבִּירִ֣ים לֹא־ חֵ֑קֶר וַיַּעֲמֵ֖ד אֲחֵרִ֣ים תַּחְתָּֽם׃ (כו) לָכֵ֗ן יַ֭כִּיר מַעְבָּֽדֵיהֶ֑ם וְהָ֥פַךְ לַ֝֗יְלָה וְיִדַּכָּֽאוּ׃ (כז) תַּֽחַת־ רְשָׁעִ֥ים סְפָקָ֗ם בִּמְק֥וֹם רֹאִֽים׃ (כח) אֲשֶׁ֣ר עַל־ כֵּ֭ן סָ֣רוּ מֵֽאַחֲרָ֑יו וְכָל־ דְּ֝רָכָ֗יו לֹ֣א הִשְׂכִּֽילוּ׃ (כט) לְהָבִ֣יא עָ֭לָיו צַֽעֲקַת־ דָּ֑ל וְצַעֲקַ֖ת עֲנִיִּ֣ים יִשְׁמָֽע׃ (ל) וְה֤וּא יַשְׁקִ֨ט ׀ וּמִ֥י יַרְשִׁ֗עַ וְיַסְתֵּ֣ר פָּ֭נִים וּמִ֣י יְשׁוּרֶ֑נּוּ וְעַל־ גּ֖וֹי וְעַל־ אָדָ֣ם יָֽחַד׃ (לא) מִ֭מְּלֹךְ אָדָ֥ם חָנֵ֗ף מִמֹּ֥קְשֵׁי עָֽם׃ (לב) כִּֽי־ אֶל־ אֵ֭ל הֶאָמַ֥ר נָשָׂ֗אתִי לֹ֣א אֶחְבֹּֽל׃ (לג) בִּלְעֲדֵ֣י אֶ֭חֱזֶה אַתָּ֣ה הֹרֵ֑נִי אִֽם־ עָ֥וֶל פָּ֝עַ֗לְתִּי לֹ֣א אֹסִֽיף׃ (לד) הַֽמֵעִמְּךָ֬ יְשַׁלְמֶ֨נָּה כִּֽי־ מָאַ֗סְתָּ כִּי־ אַתָּ֣ה תִבְחַ֣ר וְלֹא־ אָ֑נִי וּֽמַה־ יָדַ֥עְתָּ דַבֵּֽר׃ (לה) אַנְשֵׁ֣י לֵ֭בָב יֹ֣אמְרוּ לִ֑י וְגֶ֥בֶר חָ֝כָ֗ם שֹׁמֵ֥עַֽ לִֽי׃ (לו) אִ֭יּוֹב לֹא־ בְדַ֣עַת יְדַבֵּ֑ר וּ֝דְבָרָ֗יו לֹ֣א בְהַשְׂכֵּֽיל׃ (לז) אָבִ֗י יִבָּחֵ֣ן אִיּ֣וֹב עַד־ נֶ֑צַח עַל־ תְּ֝שֻׁבֹ֗ת בְּאַנְשֵׁי־ אָֽוֶן׃ (לח) כִּ֥י יֹ֘סִ֤יף עַֽל־ חַטָּאת֣וֹ פֶ֭שַׁע בֵּינֵ֣ינוּ יִסְפּ֑וֹק וְיֶ֖רֶב אֲמָרָ֣יו לָאֵֽל׃ (לט) וַיַּ֥עַן אֱלִיה֗וּ וַיֹּאמַֽר׃ (מ) הֲ֭זֹאת חָשַׁ֣בְתָּ לְמִשְׁפָּ֑ט אָ֝מַ֗רְתָּ צִדְקִ֥י מֵאֵֽל׃ (מא) כִּֽי־ תֹ֭אמַר מַה־ יִּסְכָּן־ לָ֑ךְ מָֽה־ אֹ֝עִ֗יל מֵֽחַטָּאתִֽי׃ (מב) אֲ֭נִי אֲשִֽׁיבְךָ֣ מִלִּ֑ין וְֽאֶת־ רֵעֶ֥יךָ עִמָּֽךְ׃ (מג) הַבֵּ֣ט שָׁמַ֣יִם וּרְאֵ֑ה וְשׁ֥וּר שְׁ֝חָקִ֗ים גָּבְה֥וּ מִמֶּֽךָּ׃ (מד) אִם־ חָ֭טָאתָ מַה־ תִּפְעָל־ בּ֑וֹ וְרַבּ֥וּ פְ֝שָׁעֶ֗יךָ מַה־ תַּעֲשֶׂה־ לּֽוֹ׃ (מה) אִם־ צָ֭דַקְתָּ מַה־ תִּתֶּן־ ל֑וֹ א֥וֹ מַה־ מִיָּדְךָ֥ יִקָּֽח׃ (מו) לְאִישׁ־ כָּמ֥וֹךָ רִשְׁעֶ֑ךָ וּלְבֶן־ אָ֝דָ֗ם צִדְקָתֶֽךָ׃ (מז) מֵ֭רֹב עֲשׁוּקִ֣ים יַזְעִ֑יקוּ יְשַׁוְּע֖וּ מִזְּר֣וֹעַ רַבִּֽים׃ (מח) וְֽלֹא־ אָמַ֗ר אַ֭יֵּה אֱל֣וֹהַּ עֹשָׂ֑י נֹתֵ֖ן זְמִר֣וֹת בַּלָּֽיְלָה׃ (מט) מַ֭לְּפֵנוּ מִבַּהֲמ֣וֹת אָ֑רֶץ וּמֵע֖וֹף הַשָּׁמַ֣יִם יְחַכְּמֵֽנוּ׃ (נ) שָׁ֣ם יִ֭צְעֲקוּ וְלֹ֣א יַעֲנֶ֑ה מִ֝פְּנֵ֗י גְּא֣וֹן רָעִֽים׃ (נא) אַךְ־ שָׁ֭וְא לֹא־ יִשְׁמַ֥ע ׀ אֵ֑ל וְ֝שַׁדַּ֗י לֹ֣א יְשׁוּרֶֽנָּה׃ (נב) אַ֣ף כִּֽי־ תֹ֭אמַר לֹ֣א תְשׁוּרֶ֑נּוּ דִּ֥ין לְ֝פָנָ֗יו וּתְח֥וֹלֵֽל לֽוֹ׃ (נג) וְעַתָּ֗ה כִּי־ אַ֭יִן פָּקַ֣ד אַפּ֑וֹ וְלֹֽא־ יָדַ֖ע בַּפַּ֣שׁ מְאֹֽד׃ (נד) וְ֭אִיּוֹב הֶ֣בֶל יִפְצֶה־ פִּ֑יהוּ בִּבְלִי־ דַ֝֗עַת מִלִּ֥ין יַכְבִּֽר׃ (נה) וַיֹּ֥סֶף אֱלִיה֗וּא וַיֹּאמַֽר׃ (נו) כַּתַּר־ לִ֣י זְ֭עֵיר וַאֲחַוֶּ֑ךָּ כִּ֤י ע֖וֹד לֶאֱל֣וֹהַּ מִלִּֽים׃ (נז) אֶשָּׂ֣א דֵ֭עִי לְמֵרָח֑וֹק וּ֝לְפֹעֲלִ֗י אֶֽתֵּֽן־ צֶֽדֶק׃ (נח) כִּֽי־ אָ֭מְנָם לֹא־ שֶׁ֣קֶר מִלָּ֑י תְּמִ֖ים דֵּע֣וֹת עִמָּֽךְ׃ (נט) הֶן־ אֵ֣ל כַּ֭בִּיר וְלֹ֣א יִמְאָ֑ס כַּ֝בִּ֗יר כֹּ֣חַֽ לֵֽב׃ (ס) לֹא־ יְחַיֶּ֥ה רָשָׁ֑ע וּמִשְׁפַּ֖ט עֲנִיִּ֣ים יִתֵּֽן׃ (סא) לֹֽא־ יִגְרַ֥ע מִצַּדִּ֗יק עֵ֫ינָ֥יו וְאֶת־ מְלָכִ֥ים לַכִּסֵּ֑א וַיֹּשִׁיבֵ֥ם לָ֝נֶ֗צַח וַיִּגְבָּֽהוּ׃ (סב) וְאִם־ אֲסוּרִ֥ים בַּזִּקִּ֑ים יִ֝לָּכְד֗וּן בְּחַבְלֵי־ עֹֽנִי׃ (סג) וַיַּגֵּ֣ד לָהֶ֣ם פָּעֳלָ֑ם וּ֝פִשְׁעֵיהֶ֗ם כִּ֣י יִתְגַּבָּֽרוּ׃ (סד) וַיִּ֣גֶל אָ֭זְנָם לַמּוּסָ֑ר וַ֝יֹּ֗אמֶר כִּֽי־ יְשֻׁב֥וּן מֵאָֽוֶן׃ (סה) אִֽם־ יִשְׁמְע֗וּ וְֽיַ֫עֲבֹ֥דוּ יְכַלּ֣וּ יְמֵיהֶ֣ם בַּטּ֑וֹב וּ֝שְׁנֵיהֶ֗ם בַּנְּעִימִֽים׃ (סו) וְאִם־ לֹ֣א יִ֭שְׁמְעוּ בְּשֶׁ֣לַח יַעֲבֹ֑רוּ וְ֝יִגְוְע֗וּ כִּבְלִי־ דָֽעַת׃ (סז) וְֽחַנְפֵי־ לֵ֭ב יָשִׂ֣ימוּ אָ֑ף לֹ֥א יְ֝שַׁוְּע֗וּ כִּ֣י אֲסָרָֽם׃ (סח) תָּמֹ֣ת בַּנֹּ֣עַר נַפְשָׁ֑ם וְ֝חַיָּתָ֗ם בַּקְּדֵשִֽׁים׃ (סט) יְחַלֵּ֣ץ עָנִ֣י בְעָנְי֑וֹ וְיִ֖גֶל בַּלַּ֣חַץ אָזְנָֽם׃ (ע) וְאַ֤ף הֲסִיתְךָ֨ ׀ מִפִּי־ צָ֗ר רַ֭חַב לֹא־ מוּצָ֣ק תַּחְתֶּ֑יהָ וְנַ֥חַת שֻׁ֝לְחָנְךָ֗ מָ֣לֵא דָֽשֶׁן׃ (עא) וְדִין־ רָשָׁ֥ע מָלֵ֑אתָ דִּ֖ין וּמִשְׁפָּ֣ט יִתְמֹֽכוּ׃ (עב) כִּֽי־ חֵ֭מָה פֶּן־ יְסִֽיתְךָ֣ בְסָ֑פֶק וְרָב־ כֹּ֝֗פֶר אַל־ יַטֶּֽךָּ׃ (עג) הֲיַעֲרֹ֣ךְ שׁ֭וּעֲךָ לֹ֣א בְצָ֑ר וְ֝כֹ֗ל מַאֲמַצֵּי־ כֹֽחַ׃ (עד) אַל־ תִּשְׁאַ֥ף הַלָּ֑יְלָה לַעֲל֖וֹת עַמִּ֣ים תַּחְתָּֽם׃ (עה) הִ֭שָּׁמֶר אַל־ תֵּ֣פֶן אֶל־ אָ֑וֶן כִּֽי־ עַל־ זֶ֝֗ה בָּחַ֥רְתָּ מֵעֹֽנִי׃ (עו) הֶן־ אֵ֭ל יַשְׂגִּ֣יב בְּכֹח֑וֹ מִ֖י כָמֹ֣הוּ מוֹרֶֽה׃ (עז) מִֽי־ פָקַ֣ד עָלָ֣יו דַּרְכּ֑וֹ וּמִֽי־ אָ֝מַ֗ר פָּעַ֥לְתָּ עַוְלָֽה׃ (עח) זְ֭כֹר כִּֽי־ תַשְׂגִּ֣יא פָעֳל֑וֹ אֲשֶׁ֖ר שֹׁרְר֣וּ אֲנָשִֽׁים׃ (עט) כָּל־ אָדָ֥ם חָֽזוּ־ ב֑וֹ אֱ֝נ֗וֹשׁ יַבִּ֥יט מֵרָחֽוֹק׃ (פ) הֶן־ אֵ֣ל שַׂ֭גִּיא וְלֹ֣א נֵדָ֑ע מִסְפַּ֖ר שָׁנָ֣יו וְלֹא־ חֵֽקֶר׃ (פא) כִּ֭י יְגָרַ֣ע נִטְפֵי־ מָ֑יִם יָזֹ֖קּוּ מָטָ֣ר לְאֵדֽוֹ׃ (פב) אֲשֶֽׁר־ יִזְּל֥וּ שְׁחָקִ֑ים יִ֝רְעֲפ֗וּ עֲלֵ֤י ׀ אָדָ֬ם רָֽב׃ (פג) אַ֣ף אִם־ יָ֭בִין מִפְרְשֵׂי־ עָ֑ב תְּ֝שֻׁא֗וֹת סֻכָּתֽוֹ׃ (פד) הֵן־ פָּרַ֣שׂ עָלָ֣יו אוֹר֑וֹ וְשָׁרְשֵׁ֖י הַיָּ֣ם כִּסָּֽה׃ (פה) כִּי־ בָ֭ם יָדִ֣ין עַמִּ֑ים יִֽתֶּן־ אֹ֥כֶל לְמַכְבִּֽיר׃ (פו) עַל־ כַּפַּ֥יִם כִּסָּה־ א֑וֹר וַיְצַ֖ו עָלֶ֣יהָ בְמַפְגִּֽיעַ׃ (פז) יַגִּ֣יד עָלָ֣יו רֵע֑וֹ מִ֝קְנֶ֗ה אַ֣ף עַל־ עוֹלֶֽה׃ (פח) אַף־ לְ֭זֹאת יֶחֱרַ֣ד לִבִּ֑י וְ֝יִתַּ֗ר מִמְּקוֹמֽוֹ׃ (פט) שִׁמְע֤וּ שָׁמ֣וֹעַ בְּרֹ֣גֶז קֹל֑וֹ וְ֝הֶ֗גֶה מִפִּ֥יו יֵצֵֽא׃ (צ) תַּֽחַת־ כָּל־ הַשָּׁמַ֥יִם יִשְׁרֵ֑הוּ וְ֝אוֹר֗וֹ עַל־ כַּנְפ֥וֹת הָאָֽרֶץ׃ (צא) אַחֲרָ֤יו ׀ יִשְׁאַג־ ק֗וֹל יַ֭רְעֵם בְּק֣וֹל גְּאוֹנ֑וֹ וְלֹ֥א יְ֝עַקְּבֵ֗ם כִּֽי־ יִשָּׁמַ֥ע קוֹלֽוֹ׃ (צב) יַרְעֵ֤ם אֵ֣ל בְּ֭קוֹלוֹ נִפְלָא֑וֹת עֹשֶׂ֥ה גְ֝דֹל֗וֹת וְלֹ֣א נֵדָֽע׃ (צג) כִּ֤י לַשֶּׁ֨לַג ׀ יֹאמַ֗ר הֱוֵ֫א אָ֥רֶץ וְגֶ֥שֶׁם מָטָ֑ר וְ֝גֶ֗שֶׁם מִטְר֥וֹת עֻזּֽוֹ׃ (צד) בְּיַד־ כָּל־ אָדָ֥ם יַחְתּ֑וֹם לָ֝דַ֗עַת כָּל־ אַנְשֵׁ֥י מַעֲשֵֽׂהוּ׃ (צה) וַתָּבֹ֣א חַיָּ֣ה בְמוֹ־ אָ֑רֶב וּבִמְע֖וֹנֹתֶ֣יהָ תִשְׁכֹּֽן׃ (צו) מִן־ הַ֭חֶדֶר תָּב֣וֹא סוּפָ֑ה וּֽמִמְּזָרִ֥ים קָרָֽה׃ (צז) מִנִּשְׁמַת־ אֵ֥ל יִתֶּן־ קָ֑רַח וְרֹ֖חַב מַ֣יִם בְּמוּצָֽק׃ (צח) אַף־ בְּ֭רִי יַטְרִ֣יחַ עָ֑ב יָ֝פִ֗יץ עֲנַ֣ן אוֹרֽוֹ׃ (צט) וְה֤וּא מְסִבּ֨וֹת ׀ מִתְהַפֵּ֣ךְ (בתחבולתו) בְּתַחְבּוּלֹתָ֣יו לְפָעֳלָ֑ם כֹּ֖ל אֲשֶׁ֥ר יְצַוֵּ֓ם ׀ עַל־ פְּנֵ֖י תֵבֵ֣ל אָֽרְצָה׃ (ק) אִם־ לְשֵׁ֥בֶט אִם־ לְאַרְצ֑וֹ אִם־ לְ֝חֶ֗סֶד יַמְצִאֵֽהוּ׃ (קא) הַאֲזִ֣ינָה זֹּ֣את אִיּ֑וֹב עֲ֝מֹ֗ד וְהִתְבּוֹנֵ֤ן ׀ נִפְלְא֬וֹת אֵֽל׃ (קב) הֲ֭תֵדַע בְּשׂוּם־ אֱל֣וֹהַּ עֲלֵיהֶ֑ם וְ֝הוֹפִ֗יעַ א֣וֹר עֲנָנֽוֹ׃ (קג) הֲ֭תֵדַע עַל־ מִפְלְשֵׂי־ עָ֑ב מִ֝פְלְא֗וֹת תְּמִ֣ים דֵּעִֽים׃ (קד) אֲשֶׁר־ בְּגָדֶ֥יךָ חַמִּ֑ים בְּהַשְׁקִ֥ט אֶ֝֗רֶץ מִדָּרֽוֹם׃ (קה) תַּרְקִ֣יעַ עִ֭מּוֹ לִשְׁחָקִ֑ים חֲ֝זָקִ֗ים כִּרְאִ֥י מוּצָֽק׃ (קו) ה֭וֹדִיעֵנוּ מַה־ נֹּ֣אמַר ל֑וֹ לֹ֥א־ נַ֝עֲרֹ֗ךְ מִפְּנֵי־ חֹֽשֶׁךְ׃ (קז) הַֽיְסֻפַּר־ ל֭וֹ כִּ֣י אֲדַבֵּ֑ר אִֽם־ אָ֥מַר אִ֝֗ישׁ כִּ֣י יְבֻלָּֽע׃ (קח) וְעַתָּ֤ה ׀ לֹ֘א רָ֤אוּ א֗וֹר בָּהִ֣יר ה֭וּא בַּשְּׁחָקִ֑ים וְר֥וּחַ עָ֝בְרָ֗ה וַֽתְּטַהֲרֵֽם׃ (קט) מִ֭צָּפוֹן זָהָ֣ב יֶֽאֱתֶ֑ה עַל־ אֱ֝ל֗וֹהַּ נ֣וֹרָא הֽוֹד׃ (קי) שַׁדַּ֣י לֹֽא־ מְ֭צָאנֻהוּ שַׂגִּיא־ כֹ֑חַ וּמִשְׁפָּ֥ט וְרֹב־ צְ֝דָקָ֗ה לֹ֣א יְעַנֶּֽה׃ (קיא) לָ֭כֵן יְרֵא֣וּהוּ אֲנָשִׁ֑ים לֹֽא־ יִ֝רְאֶ֗ה כָּל־ חַכְמֵי־ לֵֽב׃ (קיב) וַיַּֽעַן־ יְהוָ֣ה אֶת־ אִ֭יּוֹב (מנ הסערה) מִ֥ן ׀ הַסְּעָרָ֗ה וַיֹּאמַֽר׃ (קיג) מִ֤י זֶ֨ה ׀ מַחְשִׁ֖יךְ עֵצָ֥ה בְמִלִּ֗ין בְּֽלִי־ דָֽעַת׃ (קיד) אֱזָר־ נָ֣א כְגֶ֣בֶר חֲלָצֶ֑יךָ וְ֝אֶשְׁאָלְךָ֗ וְהוֹדִיעֵֽנִי׃ (קטו) אֵיפֹ֣ה הָ֭יִיתָ בְּיָסְדִי־ אָ֑רֶץ הַ֝גֵּ֗ד אִם־ יָדַ֥עְתָּ בִינָֽה׃ (קטז) מִי־ שָׂ֣ם מְ֭מַדֶּיהָ כִּ֣י תֵדָ֑ע א֤וֹ מִֽי־ נָטָ֖ה עָלֶ֣יהָ קָּֽו׃ (קיז) עַל־ מָ֭ה אֲדָנֶ֣יהָ הָטְבָּ֑עוּ א֥וֹ מִֽי־ יָ֝רָ֗ה אֶ֣בֶן פִּנָּתָֽהּ׃ (קיח) בְּרָן־ יַ֭חַד כּ֣וֹכְבֵי בֹ֑קֶר וַ֝יָּרִ֗יעוּ כָּל־ בְּנֵ֥י אֱלֹהִֽים׃ (קיט) וַיָּ֣סֶךְ בִּדְלָתַ֣יִם יָ֑ם בְּ֝גִיח֗וֹ מֵרֶ֥חֶם יֵצֵֽא׃ (קכ) בְּשׂוּמִ֣י עָנָ֣ן לְבֻשׁ֑וֹ וַ֝עֲרָפֶ֗ל חֲתֻלָּתֽוֹ׃ (קכא) וָאֶשְׁבֹּ֣ר עָלָ֣יו חֻקִּ֑י וָֽ֝אָשִׂ֗ים בְּרִ֣יחַ וּדְלָתָֽיִם׃ (קכב) וָאֹמַ֗ר עַד־ פֹּ֣ה תָ֭בוֹא וְלֹ֣א תֹסִ֑יף וּפֹ֥א־ יָ֝שִׁ֗ית בִּגְא֥וֹן גַּלֶּֽיךָ׃ (קכג) הְֽ֭מִיָּמֶיךָ צִוִּ֣יתָ בֹּ֑קֶר (ידעתה שחר) יִדַּ֖עְתָּה הַשַּׁ֣חַר מְקֹמֽוֹ׃ (קכד) לֶ֭אֱחֹז בְּכַנְפ֣וֹת הָאָ֑רֶץ וְיִנָּעֲר֖וּ רְשָׁ מִמֶּֽנָּה׃ (קכה) תִּ֭תְהַפֵּךְ כְּחֹ֣מֶר חוֹתָ֑ם וְ֝יִֽתְיַצְּב֗וּ כְּמ֣וֹ לְבֽוּשׁ׃ (קכו) וְיִמָּנַ֣ע מֵרְשָׁ אוֹרָ֑ם וּזְר֥וֹעַ רָ֝מָ֗ה תִּשָּׁבֵֽר׃ (קכז) הֲ֭בָאתָ עַד־ נִבְכֵי־ יָ֑ם וּבְחֵ֥קֶר תְּ֝ה֗וֹם הִתְהַלָּֽכְתָּ׃ (קכח) הֲנִגְל֣וּ לְ֭ךָ שַׁעֲרֵי־ מָ֑וֶת וְשַׁעֲרֵ֖י צַלְמָ֣וֶת תִּרְאֶֽה׃ (קכט) הִ֭תְבֹּנַנְתָּ עַד־ רַחֲבֵי־ אָ֑רֶץ הַ֝גֵּ֗ד אִם־ יָדַ֥עְתָּ כֻלָּֽהּ׃ (קל) אֵי־ זֶ֣ה הַ֭דֶּרֶךְ יִשְׁכָּן־ א֑וֹר וְ֝חֹ֗שֶׁךְ אֵי־ זֶ֥ה מְקֹמֽוֹ׃ (קלא) כִּ֣י תִ֭קָּחֶנּוּ אֶל־ גְּבוּל֑וֹ וְכִֽי־ תָ֝בִ֗ין נְתִיב֥וֹת בֵּיתֽוֹ׃ (קלב) יָ֭דַעְתָּ כִּי־ אָ֣ז תִּוָּלֵ֑ד וּמִסְפַּ֖ר יָמֶ֣יךָ רַבִּֽים׃ (קלג) הֲ֭בָאתָ אֶל־ אֹצְר֣וֹת שָׁ֑לֶג וְאֹצְר֖וֹת בָּרָ֣ד תִּרְאֶֽה׃ (קלד) אֲשֶׁר־ חָשַׂ֥כְתִּי לְעֶת־ צָ֑ר לְי֥וֹם קְ֝רָ֗ב וּמִלְחָמָֽה׃ (קלה) אֵי־ זֶ֣ה הַ֭דֶּרֶךְ יֵחָ֣לֶק א֑וֹר יָפֵ֖ץ קָדִ֣ים עֲלֵי־ אָֽרֶץ׃ (קלו) מִֽי־ פִלַּ֣ג לַשֶּׁ֣טֶף תְּעָלָ֑ה וְ֝דֶ֗רֶךְ לַחֲזִ֥יז קֹלֽוֹת׃ (קלז) לְ֭הַמְטִיר עַל־ אֶ֣רֶץ לֹא־ אִ֑ישׁ מִ֝דְבָּ֗ר לֹא־ אָדָ֥ם בּֽוֹ׃ (קלח) לְהַשְׂבִּ֣יעַ שֹׁ֭אָה וּמְשֹׁאָ֑ה וּ֝לְהַצְמִ֗יחַ מֹ֣צָא דֶֽשֶׁא׃ (קלט) הֲיֵשׁ־ לַמָּטָ֥ר אָ֑ב א֥וֹ מִי־ ה֝וֹלִ֗יד אֶגְלֵי־ טָֽל׃ (קמ) מִבֶּ֣טֶן מִ֭י יָצָ֣א הַקָּ֑רַח וּכְפֹ֥ר שָׁ֝מַיִם מִ֣י יְלָדֽוֹ׃ (קמא) כָּ֭אֶבֶן מַ֣יִם יִתְחַבָּ֑אוּ וּפְנֵ֥י תְ֝ה֗וֹם יִתְלַכָּֽדוּ׃ (קמב) הַֽ֭תְקַשֵּׁר מַעֲדַנּ֣וֹת כִּימָ֑ה אֽוֹ־ מֹשְׁכ֖וֹת כְּסִ֣יל תְּפַתֵּֽחַ׃ (קמג) הֲתֹצִ֣יא מַזָּר֣וֹת בְּעִתּ֑וֹ וְ֝עַ֗יִשׁ עַל־ בָּנֶ֥יהָ תַנְחֵֽם׃ (קמד) הֲ֭יָדַעְתָּ חֻקּ֣וֹת שָׁמָ֑יִם אִם־ תָּשִׂ֖ים מִשְׁטָר֣וֹ בָאָֽרֶץ׃ (קמה) הֲתָרִ֣ים
