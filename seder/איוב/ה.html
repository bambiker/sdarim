איוב סדר ה

פתיחה באתר · ספר איוב כולו ·

(א) יְֽמַלֵּ֥ט אִֽי־ נָקִ֑י וְ֝נִמְלַ֗ט בְּבֹ֣ר כַּפֶּֽיךָ׃ (ב) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (ג) גַּם־ הַ֭יּוֹם מְרִ֣י שִׂחִ֑י יָ֝דִ֗י כָּבְדָ֥ה עַל־ אַנְחָתִֽי׃ (ד) מִֽי־ יִתֵּ֣ן יָ֭דַעְתִּי וְאֶמְצָאֵ֑הוּ אָ֝ב֗וֹא עַד־ תְּכוּנָתֽוֹ׃ (ה) אֶעֶרְכָ֣ה לְפָנָ֣יו מִשְׁפָּ֑ט וּ֝פִ֗י אֲמַלֵּ֥א תוֹכָחֽוֹת׃ (ו) אֵ֭דְעָה מִלִּ֣ים יַעֲנֵ֑נִי וְ֝אָבִ֗ינָה מַה־ יֹּ֥אמַר לִֽי׃ (ז) הַבְּרָב־ כֹּ֭חַ יָרִ֣יב עִמָּדִ֑י לֹ֥א אַךְ־ ה֝֗וּא יָשִׂ֥ם בִּֽי׃ (ח) שָׁ֗ם יָ֭שָׁר נוֹכָ֣ח עִמּ֑וֹ וַאֲפַלְּטָ֥ה לָ֝נֶ֗צַח מִשֹּׁפְטִֽי׃ (ט) הֵ֤ן קֶ֣דֶם אֶהֱלֹ֣ךְ וְאֵינֶ֑נּוּ וְ֝אָח֗וֹר וְֽלֹא־ אָבִ֥ין לֽוֹ׃ (י) שְׂמֹ֣אול בַּעֲשֹׂת֣וֹ וְלֹא־ אָ֑חַז יַעְטֹ֥ף יָ֝מִ֗ין וְלֹ֣א אֶרְאֶֽה׃ (יא) כִּֽי־ יָ֭דַע דֶּ֣רֶךְ עִמָּדִ֑י בְּ֝חָנַ֗נִי כַּזָּהָ֥ב אֵצֵֽא׃ (יב) בַּ֭אֲשֻׁרוֹ אָחֲזָ֣ה רַגְלִ֑י דַּרְכּ֖וֹ שָׁמַ֣רְתִּי וְלֹא־ אָֽט׃ (יג) מִצְוַ֣ת שְׂ֭פָתָיו וְלֹ֣א אָמִ֑ישׁ מֵ֝חֻקִּ֗י צָפַ֥נְתִּי אִמְרֵי־ פִֽיו׃ (יד) וְה֣וּא בְ֭אֶחָד וּמִ֣י יְשִׁיבֶ֑נּוּ וְנַפְשׁ֖וֹ אִוְּתָ֣ה וַיָּֽעַשׂ׃ (טו) כִּ֭י יַשְׁלִ֣ים חֻקִּ֑י וְכָהֵ֖נָּה רַבּ֣וֹת עִמּֽוֹ׃ (טז) עַל־ כֵּ֭ן מִפָּנָ֣יו אֶבָּהֵ֑ל אֶ֝תְבּוֹנֵ֗ן וְאֶפְחַ֥ד מִמֶּֽנּוּ׃ (יז) וְ֭אֵל הֵרַ֣ךְ לִבִּ֑י וְ֝שַׁדַּ֗י הִבְהִילָֽנִי׃ (יח) כִּֽי־ לֹ֣א נִ֭צְמַתִּי מִפְּנֵי־ חֹ֑שֶׁךְ וּ֝מִפָּנַ֗י כִּסָּה־ אֹֽפֶל׃ (יט) מַדּ֗וּעַ מִ֭שַּׁדַּי לֹא־ נִצְפְּנ֣וּ עִתִּ֑ים (וידעו) וְ֝יֹדְעָ֗יו לֹא־ חָ֥זוּ יָמָֽיו׃ (כ) גְּבֻל֥וֹת יַשִּׂ֑יגוּ עֵ֥דֶר גָּ֝זְל֗וּ וַיִּרְעֽוּ׃ (כא) חֲמ֣וֹר יְתוֹמִ֣ים יִנְהָ֑גוּ יַ֝חְבְּל֗וּ שׁ֣וֹר אַלְמָנָֽה׃ (כב) יַטּ֣וּ אֶבְיוֹנִ֣ים מִדָּ֑רֶךְ יַ֥חַד חֻ֝בְּא֗וּ עֲנִיֵּי־ אָֽרֶץ׃ (כג) הֵ֤ן פְּרָאִ֨ים ׀ בַּֽמִּדְבָּ֗ר יָצְא֣וּ בְּ֭פָעֳלָם מְשַׁחֲרֵ֣י לַטָּ֑רֶף עֲרָבָ֥ה ל֥וֹ לֶ֝֗חֶם לַנְּעָרִֽים׃ (כד) בַּ֭שָּׂדֶה בְּלִיל֣וֹ (יקצירו) יִקְצ֑וֹרוּ וְכֶ֖רֶם רָשָׁ֣ע יְלַקֵּֽשׁוּ׃ (כה) עָר֣וֹם יָ֭לִינוּ מִבְּלִ֣י לְב֑וּשׁ וְאֵ֥ין כְּ֝ס֗וּת בַּקָּרָֽה׃ (כו) מִזֶּ֣רֶם הָרִ֣ים יִרְטָ֑בוּ וּֽמִבְּלִ֥י מַ֝חְסֶ֗ה חִבְּקוּ־ צֽוּר׃ (כז) יִ֭גְזְלוּ מִשֹּׁ֣ד יָת֑וֹם וְֽעַל־ עָנִ֥י יַחְבֹּֽלוּ׃ (כח) עָר֣וֹם הִ֭לְּכוּ בְּלִ֣י לְב֑וּשׁ וּ֝רְעֵבִ֗ים נָ֣שְׂאוּ עֹֽמֶר׃ (כט) בֵּין־ שׁוּרֹתָ֥ם יַצְהִ֑ירוּ יְקָבִ֥ים דָּ֝רְכ֗וּ וַיִּצְמָֽאוּ׃ (ל) מֵ֘עִ֤יר מְתִ֨ים ׀ יִנְאָ֗קוּ וְנֶֽפֶשׁ־ חֲלָלִ֥ים תְּשַׁוֵּ֑עַ וֶ֝אֱל֗וֹהַּ לֹא־ יָשִׂ֥ים תִּפְלָֽה׃ (לא) הֵ֤מָּה ׀ הָיוּ֮ בְּֽמֹרְדֵ֫י־ א֥וֹר לֹֽא־ הִכִּ֥ירוּ דְרָכָ֑יו וְלֹ֥א יָ֝שְׁב֗וּ בִּנְתִיבֹתָֽיו׃ (לב) לָא֡וֹר יָ֘ק֤וּם רוֹצֵ֗חַ יִֽקְטָל־ עָנִ֥י וְאֶבְי֑וֹן וּ֝בַלַּ֗יְלָה יְהִ֣י כַגַּנָּֽב׃ (לג) וְעֵ֤ין נֹאֵ֨ף ׀ שָׁ֤מְרָֽה נֶ֣שֶׁף לֵ֭אמֹר לֹא־ תְשׁוּרֵ֣נִי עָ֑יִן וְסֵ֖תֶר פָּנִ֣ים יָשִֽׂים׃ (לד) חָתַ֥ר בַּחֹ֗שֶׁךְ בָּ֫תִּ֥ים יוֹמָ֥ם חִתְּמוּ־ לָ֗מוֹ לֹא־ יָ֥דְעוּ אֽוֹר׃ (לה) כִּ֤י יַחְדָּ֨ו ׀ בֹּ֣קֶר לָ֣מוֹ צַלְמָ֑וֶת כִּֽי־ יַ֝כִּ֗יר בַּלְה֥וֹת צַלְמָֽוֶת׃ (לו) קַֽל־ ה֤וּא ׀ עַל־ פְּנֵי־ מַ֗יִם תְּקֻלַּ֣ל חֶלְקָתָ֣ם בָּאָ֑רֶץ לֹֽא־ יִ֝פְנֶה דֶּ֣רֶךְ כְּרָמִֽים׃ (לז) צִיָּ֤ה גַם־ חֹ֗ם יִגְזְל֥וּ מֵֽימֵי־ שֶׁ֗לֶג שְׁא֣וֹל חָטָֽאוּ׃ (לח) יִשְׁכָּ֘חֵ֤הוּ רֶ֨חֶם ׀ מְתָ֘ק֤וֹ רִמָּ֗ה ע֥וֹד לֹֽא־ יִזָּכֵ֑ר וַתִּשָּׁבֵ֖ר כָּעֵ֣ץ עַוְלָֽה׃ (לט) רֹעֶ֣ה עֲ֭קָרָה לֹ֣א תֵלֵ֑ד וְ֝אַלְמָנָ֗ה לֹ֣א יְיֵטִֽיב׃ (מ) וּמָשַׁ֣ךְ אַבִּירִ֣ים בְּכֹח֑וֹ יָ֝ק֗וּם וְֽלֹא־ יַאֲמִ֥ין בַּֽחַיִּֽין׃ (מא) יִתֶּן־ ל֣וֹ לָ֭בֶטַח וְיִשָּׁעֵ֑ן וְ֝עֵינֵ֗יהוּ עַל־ דַּרְכֵיהֶֽם׃ (מב) ר֤וֹמּוּ מְּעַ֨ט ׀ וְֽאֵינֶ֗נּוּ וְֽהֻמְּכ֗וּ כַּכֹּ֥ל יִקָּפְצ֑וּן וּכְרֹ֖אשׁ שִׁבֹּ֣לֶת יִמָּֽלוּ׃ (מג) וְאִם־ לֹ֣א אֵ֭פוֹ מִ֣י יַכְזִיבֵ֑נִי וְיָשֵׂ֥ם לְ֝אַ֗ל מִלָּתִֽי׃ (מד) וַ֭יַּעַן בִּלְדַּ֥ד הַשֻּׁחִ֗י וַיֹּאמַֽר׃ (מה) הַמְשֵׁ֣ל וָפַ֣חַד עִמּ֑וֹ עֹשֶׂ֥ה שָׁ֝ל֗וֹם בִּמְרוֹמָֽיו׃ (מו) הֲיֵ֣שׁ מִ֭סְפָּר לִגְדוּדָ֑יו וְעַל־ מִ֝֗י לֹא־ יָק֥וּם אוֹרֵֽהוּ׃ (מז) וּמַה־ יִּצְדַּ֣ק אֱנ֣וֹשׁ עִם־ אֵ֑ל וּמַה־ יִּ֝זְכֶּ֗ה יְל֣וּד אִשָּֽׁה׃ (מח) הֵ֣ן עַד־ יָ֭רֵחַ וְלֹ֣א יַאֲהִ֑יל וְ֝כוֹכָבִ֗ים לֹא־ זַכּ֥וּ בְעֵינָֽיו׃ (מט) אַ֭ף כִּֽי־ אֱנ֣וֹשׁ רִמָּ֑ה וּבֶן־ אָ֝דָ֗ם תּוֹלֵעָֽה׃ (נ) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (נא) מֶה־ עָזַ֥רְתָּ לְלֹא־ כֹ֑חַ ה֝וֹשַׁ֗עְתָּ זְר֣וֹעַ לֹא־ עֹֽז׃ (נב) מַה־ יָּ֭עַצְתָּ לְלֹ֣א חָכְמָ֑ה וְ֝תוּשִׁיָּ֗ה לָרֹ֥ב הוֹדָֽעְתָּ׃ (נג) אֶת־ מִ֭י הִגַּ֣דְתָּ מִלִּ֑ין וְנִשְׁמַת־ מִ֝י יָצְאָ֥ה מִמֶּֽךָּ׃ (נד) הָרְפָאִ֥ים יְחוֹלָ֑לוּ מִתַּ֥חַת מַ֝֗יִם וְשֹׁכְנֵיהֶֽם׃ (נה) עָר֣וֹם שְׁא֣וֹל נֶגְדּ֑וֹ וְאֵ֥ין כְּ֝ס֗וּת לָֽאֲבַדּֽוֹן׃ (נו) נֹטֶ֣ה צָפ֣וֹן עַל־ תֹּ֑הוּ תֹּ֥לֶה אֶ֝֗רֶץ עַל־ בְּלִי־ מָֽה׃ (נז) צֹרֵֽר־ מַ֥יִם בְּעָבָ֑יו וְלֹא־ נִבְקַ֖ע עָנָ֣ן תַּחְתָּֽם׃ (נח) מְאַחֵ֥ז פְּנֵי־ כִסֵּ֑ה פַּרְשֵׁ֖ז עָלָ֣יו עֲנָנֽוֹ׃ (נט) חֹֽק־ חָ֭ג עַל־ פְּנֵי־ מָ֑יִם עַד־ תַּכְלִ֖ית א֣וֹר עִם־ חֹֽשֶׁךְ׃ (ס) עַמּוּדֵ֣י שָׁמַ֣יִם יְרוֹפָ֑פוּ וְ֝יִתְמְה֗וּ מִגַּעֲרָתֽוֹ׃ (סא) בְּ֭כֹחוֹ רָגַ֣ע הַיָּ֑ם (ובתובנתו) וּ֝בִתְבוּנָת֗וֹ מָ֣חַץ רָֽהַב׃ (סב) בְּ֭רוּחוֹ שָׁמַ֣יִם שִׁפְרָ֑ה חֹֽלֲלָ֥ה יָ֝ד֗וֹ נָחָ֥שׁ בָּרִֽיחַ׃ (סג) הֶן־ אֵ֤לֶּה ׀ קְצ֬וֹת (דרכו) דְּרָכָ֗יו וּמַה־ שֵּׁ֣מֶץ דָּ֭בָר נִשְׁמַע־ בּ֑וֹ וְרַ֥עַם (גבורתו) גְּ֝בוּרוֹתָ֗יו מִ֣י יִתְבּוֹנָֽן׃ (סד) וַיֹּ֣סֶף אִ֭יּוֹב שְׂאֵ֥ת מְשָׁל֗וֹ וַיֹּאמַֽר׃ (סה) חַי־ אֵ֭ל הֵסִ֣יר מִשְׁפָּטִ֑י וְ֝שַׁדַּ֗י הֵמַ֥ר נַפְשִֽׁי׃ (סו) כִּֽי־ כָל־ ע֣וֹד נִשְׁמָתִ֣י בִ֑י וְר֖וּחַ אֱל֣וֹהַּ בְּאַפִּֽי׃ (סז) אִם־ תְּדַבֵּ֣רְנָה שְׂפָתַ֣י עַוְלָ֑ה וּ֝לְשׁוֹנִ֗י אִם־ יֶהְגֶּ֥ה רְמִיָּֽה׃ (סח) חָלִ֣ילָה לִּי֮ אִם־ אַצְדִּ֪יק אֶ֫תְכֶ֥ם עַד־ אֶגְוָ֑ע לֹא־ אָסִ֖יר תֻּמָּתִ֣י מִמֶּֽנִּי׃ (סט) בְּצִדְקָתִ֣י הֶ֭חֱזַקְתִּי וְלֹ֣א אַרְפֶּ֑הָ לֹֽא־ יֶחֱרַ֥ף לְ֝בָבִ֗י מִיָּמָֽי׃ (ע) יְהִ֣י כְ֭רָשָׁע אֹ֣יְבִ֑י וּמִתְקוֹמְמִ֥י כְעַוָּֽל׃ (עא) כִּ֤י מַה־ תִּקְוַ֣ת חָ֭נֵף כִּ֣י יִבְצָ֑ע כִּ֤י יֵ֖שֶׁל אֱל֣וֹהַּ נַפְשֽׁוֹ׃ (עב) הַֽ֭צַעֲקָתוֹ יִשְׁמַ֥ע ׀ אֵ֑ל כִּֽי־ תָב֖וֹא עָלָ֣יו צָרָֽה׃ (עג) אִם־ עַל־ שַׁדַּ֥י יִתְעַנָּ֑ג יִקְרָ֖א אֱל֣וֹהַּ בְּכָל־ עֵֽת׃ (עד) אוֹרֶ֣ה אֶתְכֶ֣ם בְּיַד־ אֵ֑ל אֲשֶׁ֥ר עִם־ שַׁ֝דַּ֗י לֹ֣א אֲכַחֵֽד׃ (עה) הֵן־ אַתֶּ֣ם כֻּלְּכֶ֣ם חֲזִיתֶ֑ם וְלָמָּה־ זֶּ֝֗ה הֶ֣בֶל תֶּהְבָּֽלוּ׃ (עו) זֶ֤ה ׀ חֵֽלֶק־ אָדָ֖ם רָשָׁ֥ע ׀ עִם־ אֵ֑ל וְֽנַחֲלַ֥ת עָ֝רִיצִ֗ים מִשַּׁדַּ֥י יִקָּֽחוּ׃ (עז) אִם־ יִרְבּ֣וּ בָנָ֣יו לְמוֹ־ חָ֑רֶב וְ֝צֶאֱצָאָ֗יו לֹ֣א יִשְׂבְּעוּ־ לָֽחֶם׃ (עח) (שרידו) שְׂ֭רִידָיו בַּמָּ֣וֶת יִקָּבֵ֑רוּ וְ֝אַלְמְנֹתָ֗יו לֹ֣א תִבְכֶּֽינָה׃ (עט) אִם־ יִצְבֹּ֣ר כֶּעָפָ֣ר כָּ֑סֶף וְ֝כַחֹ֗מֶר יָכִ֥ין מַלְבּֽוּשׁ׃ (פ) יָ֭כִין וְצַדִּ֣יק יִלְבָּ֑שׁ וְ֝כֶ֗סֶף נָקִ֥י יַחֲלֹֽק׃ (פא) בָּנָ֣ה כָעָ֣שׁ בֵּית֑וֹ וּ֝כְסֻכָּ֗ה עָשָׂ֥ה נֹצֵֽר׃ (פב) עָשִׁ֣יר יִ֭שְׁכַּב וְלֹ֣א יֵאָסֵ֑ף עֵינָ֖יו פָּקַ֣ח וְאֵינֶֽנּוּ׃ (פג) תַּשִּׂיגֵ֣הוּ כַ֭מַּיִם בַּלָּה֑וֹת לַ֝֗יְלָה גְּנָבַ֥תּוּ סוּפָֽה׃ (פד) יִשָּׂאֵ֣הוּ קָדִ֣ים וְיֵלַ֑ךְ וִֽ֝ישָׂעֲרֵ֗הוּ מִמְּקֹמֽוֹ׃ (פה) וְיַשְׁלֵ֣ךְ עָ֭לָיו וְלֹ֣א יַחְמֹ֑ל מִ֝יָּד֗וֹ בָּר֥וֹחַ יִבְרָֽח׃ (פו) יִשְׂפֹּ֣ק עָלֵ֣ימוֹ כַפֵּ֑ימוֹ וְיִשְׁרֹ֥ק עָ֝לָ֗יו מִמְּקֹמֽוֹ׃ (פז) כִּ֤י יֵ֣שׁ לַכֶּ֣סֶף מוֹצָ֑א וּ֝מָק֗וֹם לַזָּהָ֥ב יָזֹֽקּוּ׃ (פח) בַּ֭רְזֶל מֵעָפָ֣ר יֻקָּ֑ח וְ֝אֶ֗בֶן יָצ֥וּק נְחוּשָֽׁה׃ (פט) קֵ֤ץ ׀ שָׂ֤ם לַחֹ֗שֶׁךְ וּֽלְכָל־ תַּ֭כְלִית ה֣וּא חוֹקֵ֑ר אֶ֖בֶן אֹ֣פֶל וְצַלְמָֽוֶת׃ (צ) פָּ֤רַץ נַ֨חַל ׀ מֵֽעִם־ גָּ֗ר הַֽנִּשְׁכָּחִ֥ים מִנִּי־ רָ֑גֶל דַּ֖לּוּ מֵאֱנ֣וֹשׁ נָֽעוּ׃ (צא) אֶ֗רֶץ מִמֶּ֥נָּה יֵֽצֵא־ לָ֑חֶם וְ֝תַחְתֶּ֗יהָ נֶהְפַּ֥ךְ כְּמוֹ־ אֵֽשׁ׃ (צב) מְקוֹם־ סַפִּ֥יר אֲבָנֶ֑יהָ וְעַפְרֹ֖ת זָהָ֣ב לֽוֹ׃ (צג) נָ֭תִיב לֹֽא־ יְדָ֣עוֹ עָ֑יִט וְלֹ֥א שְׁ֝זָפַ֗תּוּ עֵ֣ין אַיָּֽה׃ (צד) לֹֽא־ הִדְרִיכֻ֥הוּ בְנֵי־ שָׁ֑חַץ לֹֽא־ עָדָ֖ה עָלָ֣יו שָֽׁחַל׃ (צה) בַּֽ֭חַלָּמִישׁ שָׁלַ֣ח יָד֑וֹ הָפַ֖ךְ מִשֹּׁ֣רֶשׁ הָרִֽים׃ (צו) בַּ֭צּוּרוֹת יְאֹרִ֣ים בִּקֵּ֑עַ וְכָל־ יְ֝קָ֗ר רָאֲתָ֥ה עֵינֽוֹ׃ (צז) מִ֭בְּכִי נְהָר֣וֹת חִבֵּ֑שׁ וְ֝תַעֲלֻמָ֗הּ יֹ֣צִא אֽוֹר׃ (צח) וְֽ֭הַחָכְמָה מֵאַ֣יִן תִּמָּצֵ֑א וְאֵ֥י זֶ֝ה מְק֣וֹם בִּינָֽה׃ (צט) לֹא־ יָדַ֣ע אֱנ֣וֹשׁ עֶרְכָּ֑הּ וְלֹ֥א תִ֝מָּצֵ֗א בְּאֶ֣רֶץ הַֽחַיִּֽים׃ (ק) תְּה֣וֹם אָ֭מַר לֹ֣א בִי־ הִ֑יא וְיָ֥ם אָ֝מַ֗ר אֵ֣ין עִמָּדִֽי׃ (קא) לֹא־ יֻתַּ֣ן סְג֣וֹר תַּחְתֶּ֑יהָ וְלֹ֥א יִ֝שָּׁקֵ֗ל כֶּ֣סֶף מְחִירָֽהּ׃ (קב) לֹֽא־ תְ֭סֻלֶּה בְּכֶ֣תֶם אוֹפִ֑יר בְּשֹׁ֖הַם יָקָ֣ר וְסַפִּֽיר׃ (קג) לֹא־ יַעַרְכֶ֣נָּה זָ֭הָב וּזְכוֹכִ֑ית וּתְמ֖וּרָתָ֣הּ כְּלִי־ פָֽז׃ (קד) רָאמ֣וֹת וְ֭גָבִישׁ לֹ֣א יִזָּכֵ֑ר וּמֶ֥שֶׁךְ חָ֝כְמָ֗ה מִפְּנִינִֽים׃ (קה) לֹֽא־ יַ֭עַרְכֶנָּה פִּטְדַת־ כּ֑וּשׁ בְּכֶ֥תֶם טָ֝ה֗וֹר לֹ֣א תְסֻלֶּֽה׃ (קו) וְֽ֭הַחָכְמָה מֵאַ֣יִן תָּב֑וֹא וְאֵ֥י זֶ֝֗ה מְק֣וֹם בִּינָֽה׃ (קז) וְֽ֭נֶעֶלְמָה מֵעֵינֵ֣י כָל־ חָ֑י וּמֵע֖וֹף הַשָּׁמַ֣יִם נִסְתָּֽרָה׃ (קח) אֲבַדּ֣וֹן וָ֭מָוֶת אָ֣מְר֑וּ בְּ֝אָזְנֵ֗ינוּ שָׁמַ֥עְנוּ שִׁמְעָֽהּ׃ (קט) אֱ֭לֹהִים הֵבִ֣ין דַּרְכָּ֑הּ וְ֝ה֗וּא יָדַ֥ע אֶת־ מְקוֹמָֽהּ׃ (קי) כִּי־ ה֭וּא לִקְצוֹת־ הָאָ֣רֶץ יַבִּ֑יט תַּ֖חַת כָּל־ הַשָּׁמַ֣יִם יִרְאֶֽה׃ (קיא) לַעֲשׂ֣וֹת לָר֣וּחַ מִשְׁקָ֑ל וּ֝מַ֗יִם תִּכֵּ֥ן בְּמִדָּֽה׃ (קיב) בַּעֲשֹׂת֣וֹ לַמָּטָ֣ר חֹ֑ק וְ֝דֶ֗רֶךְ לַחֲזִ֥יז קֹלֽוֹת׃ (קיג) אָ֣ז רָ֭אָהּ וַֽיְסַפְּרָ֑הּ הֱ֝כִינָ֗הּ וְגַם־ חֲקָרָֽהּ׃ (קיד) וַיֹּ֤אמֶר ׀ לָֽאָדָ֗ם הֵ֤ן יִרְאַ֣ת אֲ֭דֹנָי הִ֣יא חָכְמָ֑ה וְס֖וּר מֵרָ֣ע בִּינָֽה׃ (קטו) וַיֹּ֣סֶף אִ֭יּוֹב שְׂאֵ֥ת מְשָׁל֗וֹ וַיֹּאמַֽר׃ (קטז) מִֽי־ יִתְּנֵ֥נִי כְיַרְחֵי־ קֶ֑דֶם כִּ֝ימֵ֗י אֱל֣וֹהַּ יִשְׁמְרֵֽנִי׃ (קיז) בְּהִלּ֣וֹ נֵ֭רוֹ עֲלֵ֣י רֹאשִׁ֑י לְ֝אוֹרוֹ אֵ֣לֶךְ חֹֽשֶׁךְ׃ (קיח) כַּאֲשֶׁ֣ר הָ֭יִיתִי בִּימֵ֣י חָרְפִּ֑י בְּס֥וֹד אֱ֝ל֗וֹהַּ עֲלֵ֣י אָהֳלִֽי׃ (קיט) בְּע֣וֹד שַׁ֭דַּי עִמָּדִ֑י סְבִ֖יבוֹתַ֣י נְעָרָֽי׃ (קכ) בִּרְחֹ֣ץ הֲלִיכַ֣י בְּחֵמָ֑ה וְצ֥וּר יָצ֥וּק עִ֝מָּדִ֗י פַּלְגֵי־ שָֽׁמֶן׃ (קכא) בְּצֵ֣אתִי שַׁ֣עַר עֲלֵי־ קָ֑רֶת בָּ֝רְח֗וֹב אָכִ֥ין מוֹשָׁבִֽי׃ (קכב) רָא֣וּנִי נְעָרִ֣ים וְנֶחְבָּ֑אוּ וִֽ֝ישִׁישִׁים קָ֣מוּ עָמָֽדוּ׃ (קכג) שָׂ֭רִים עָצְר֣וּ בְמִלִּ֑ים וְ֝כַ֗ף יָשִׂ֥ימוּ לְפִיהֶֽם׃ (קכד) קוֹל־ נְגִידִ֥ים נֶחְבָּ֑אוּ וּ֝לְשׁוֹנָ֗ם לְחִכָּ֥ם דָּבֵֽקָה׃ (קכה) כִּ֤י אֹ֣זֶן שָׁ֭מְעָה וַֽתְּאַשְּׁרֵ֑נִי וְעַ֥יִן רָ֝אֲתָ֗ה וַתְּעִידֵֽנִי׃ (קכו) כִּֽי־ אֲ֭מַלֵּט עָנִ֣י מְשַׁוֵּ֑עַ וְ֝יָת֗וֹם וְֽלֹא־ עֹזֵ֥ר לֽוֹ׃ (קכז) בִּרְכַּ֣ת אֹ֭בֵד עָלַ֣י תָּבֹ֑א וְלֵ֖ב אַלְמָנָ֣ה אַרְנִֽן׃
