איוב סדר ו

פתיחה באתר · ספר איוב כולו ·

(א) צֶ֣דֶק לָ֭בַשְׁתִּי וַיִּלְבָּשֵׁ֑נִי כִּֽמְעִ֥יל וְ֝צָנִ֗יף מִשְׁפָּטִֽי׃ (ב) עֵינַ֣יִם הָ֭יִיתִי לַֽעִוֵּ֑ר וְרַגְלַ֖יִם לַפִּסֵּ֣חַ אָֽנִי׃ (ג) אָ֣ב אָ֭נֹכִֽי לָֽאֶבְיוֹנִ֑ים וְרִ֖ב לֹא־ יָדַ֣עְתִּי אֶחְקְרֵֽהוּ׃ (ד) וָֽ֭אֲשַׁבְּרָה מְתַלְּע֣וֹת עַוָּ֑ל וּ֝מִשִּׁנָּ֗יו אַשְׁלִ֥יךְ טָֽרֶף׃ (ה) וָ֭אֹמַר עִם־ קִנִּ֣י אֶגְוָ֑ע וְ֝כַח֗וֹל אַרְבֶּ֥ה יָמִֽים׃ (ו) שָׁרְשִׁ֣י פָת֣וּחַ אֱלֵי־ מָ֑יִם וְ֝טַ֗ל יָלִ֥ין בִּקְצִירִֽי׃ (ז) כְּ֭בוֹדִי חָדָ֣שׁ עִמָּדִ֑י וְ֝קַשְׁתִּ֗י בְּיָדִ֥י תַחֲלִֽיף׃ (ח) לִֽי־ שָׁמְע֥וּ וְיִחֵ֑לּוּ וְ֝יִדְּמ֗וּ לְמ֣וֹ עֲצָתִֽי׃ (ט) אַחֲרֵ֣י דְ֭בָרִי לֹ֣א יִשְׁנ֑וּ וְ֝עָלֵ֗ימוֹ תִּטֹּ֥ף מִלָּתִֽי׃ (י) וְיִֽחֲל֣וּ כַמָּטָ֣ר לִ֑י וּ֝פִיהֶ֗ם פָּעֲר֥וּ לְמַלְקֽוֹשׁ׃ (יא) אֶשְׂחַ֣ק אֲ֭לֵהֶם לֹ֣א יַאֲמִ֑ינוּ וְא֥וֹר פָּ֝נַ֗י לֹ֣א יַפִּילֽוּן׃ (יב) אֶֽבֲחַ֣ר דַּרְכָּם֮ וְאֵשֵׁ֪ב רֹ֥אשׁ וְ֭אֶשְׁכּוֹן כְּמֶ֣לֶךְ בַּגְּד֑וּד כַּאֲשֶׁ֖ר אֲבֵלִ֣ים יְנַחֵֽם׃ (יג) וְעַתָּ֤ה ׀ שָֽׂחֲק֣וּ עָלַי֮ צְעִירִ֥ים מִמֶּ֗נִּי לְיָ֫מִ֥ים אֲשֶׁר־ מָאַ֥סְתִּי אֲבוֹתָ֑ם לָ֝שִׁ֗ית עִם־ כַּלְבֵ֥י צֹאנִֽי׃ (יד) גַּם־ כֹּ֣חַ יְ֭דֵיהֶם לָ֣מָּה לִּ֑י עָ֝לֵ֗ימוֹ אָ֣בַד כָּֽלַח׃ (טו) בְּחֶ֥סֶר וּבְכָפָ֗ן גַּ֫לְמ֥וּד הַֽעֹרְקִ֥ים צִיָּ֑ה אֶ֝֗מֶשׁ שׁוֹאָ֥ה וּמְשֹׁאָֽה׃ (טז) הַקֹּטְפִ֣ים מַלּ֣וּחַ עֲלֵי־ שִׂ֑יחַ וְשֹׁ֖רֶשׁ רְתָמִ֣ים לַחְמָֽם׃ (יז) מִן־ גֵּ֥ו יְגֹרָ֑שׁוּ יָרִ֥יעוּ עָ֝לֵ֗ימוֹ כַּגַּנָּֽב׃ (יח) בַּעֲר֣וּץ נְחָלִ֣ים לִשְׁכֹּ֑ן חֹרֵ֖י עָפָ֣ר וְכֵפִֽים׃ (יט) בֵּין־ שִׂיחִ֥ים יִנְהָ֑קוּ תַּ֖חַת חָר֣וּל יְסֻפָּֽחוּ׃ (כ) בְּֽנֵי־ נָ֭בָל גַּם־ בְּנֵ֣י בְלִי־ שֵׁ֑ם נִ֝כְּא֗וּ מִן־ הָאָֽרֶץ׃ (כא) וְ֭עַתָּה נְגִינָתָ֣ם הָיִ֑יתִי וָאֱהִ֖י לָהֶ֣ם לְמִלָּֽה׃ (כב) תִּֽ֭עֲבוּנִי רָ֣חֲקוּ מֶ֑נִּי וּ֝מִפָּנַ֗י לֹא־ חָ֥שְׂכוּ רֹֽק׃ (כג) כִּֽי־ (יתרו) יִתְרִ֣י פִ֭תַּח וַיְעַנֵּ֑נִי וְ֝רֶ֗סֶן מִפָּנַ֥י שִׁלֵּֽחוּ׃ (כד) עַל־ יָמִין֮ פִּרְחַ֪ח יָ֫ק֥וּמוּ רַגְלַ֥י שִׁלֵּ֑חוּ וַיָּסֹ֥לּוּ עָ֝לַ֗י אָרְח֥וֹת אֵידָֽם׃ (כה) נָתְס֗וּ נְֽתִיבָ֫תִ֥י לְהַוָּתִ֥י יֹעִ֑ילוּ לֹ֖א עֹזֵ֣ר לָֽמוֹ׃ (כו) כְּפֶ֣רֶץ רָחָ֣ב יֶאֱתָ֑יוּ תַּ֥חַת שֹׁ֝אָ֗ה הִתְגַּלְגָּֽלוּ׃ (כז) הָהְפַּ֥ךְ עָלַ֗י בַּלָּ֫ה֥וֹת תִּרְדֹּ֣ף כָּ֭רוּחַ נְדִבָתִ֑י וּ֝כְעָ֗ב עָבְרָ֥ה יְשֻׁעָתִֽי׃ (כח) וְעַתָּ֗ה עָ֭לַי תִּשְׁתַּפֵּ֣ךְ נַפְשִׁ֑י יֹ֭אחֲז֣וּנִי יְמֵי־ עֹֽנִי׃ (כט) לַ֗יְלָה עֲ֭צָמַי נִקַּ֣ר מֵעָלָ֑י וְ֝עֹרְקַ֗י לֹ֣א יִשְׁכָּבֽוּן׃ (ל) בְּרָב־ כֹּ֭חַ יִתְחַפֵּ֣שׂ לְבוּשִׁ֑י כְּפִ֖י כֻתָּנְתִּ֣י יַֽאַזְרֵֽנִי׃ (לא) הֹרָ֥נִי לַחֹ֑מֶר וָ֝אֶתְמַשֵּׁ֗ל כֶּעָפָ֥ר וָאֵֽפֶר׃ (לב) אֲשַׁוַּ֣ע אֵ֭לֶיךָ וְלֹ֣א תַעֲנֵ֑נִי עָ֝מַ֗דְתִּי וַתִּתְבֹּ֥נֶן בִּֽי׃ (לג) תֵּהָפֵ֣ךְ לְאַכְזָ֣ר לִ֑י בְּעֹ֖צֶם יָדְךָ֣ תִשְׂטְמֵֽנִי׃ (לד) תִּשָּׂאֵ֣נִי אֶל־ ר֭וּחַ תַּרְכִּיבֵ֑נִי וּ֝תְמֹגְגֵ֗נִי (תשוה) תּוּשִׁיָּֽה׃ (לה) כִּֽי־ יָ֭דַעְתִּי מָ֣וֶת תְּשִׁיבֵ֑נִי וּבֵ֖ית מוֹעֵ֣ד לְכָל־ חָֽי׃ (לו) אַ֣ךְ לֹא־ בְ֭עִי יִשְׁלַח־ יָ֑ד אִם־ בְּ֝פִיד֗וֹ לָהֶ֥ן שֽׁוּעַ׃ (לז) אִם־ לֹ֣א בָ֭כִיתִי לִקְשֵׁה־ י֑וֹם עָֽגְמָ֥ה נַ֝פְשִׁ֗י לָאֶבְיֽוֹן׃ (לח) כִּ֤י ט֣וֹב קִ֭וִּיתִי וַיָּ֣בֹא רָ֑ע וַֽאֲיַחֲלָ֥ה לְ֝א֗וֹר וַיָּ֥בֹא אֹֽפֶל׃ (לט) מֵעַ֖י רֻתְּח֥וּ וְלֹא־ דָ֗מּוּ קִדְּמֻ֥נִי יְמֵי־ עֹֽנִי׃ (מ) קֹדֵ֣ר הִ֭לַּכְתִּי בְּלֹ֣א חַמָּ֑ה קַ֖מְתִּי בַקָּהָ֣ל אֲשַׁוֵּֽעַ׃ (מא) אָ֭ח הָיִ֣יתִי לְתַנִּ֑ים וְ֝רֵ֗עַ לִבְנ֥וֹת יַעֲנָֽה׃ (מב) ע֭וֹרִי שָׁחַ֣ר מֵעָלָ֑י וְעַצְמִי־ חָ֝֗רָה מִנִּי־ חֹֽרֶב׃ (מג) וַיְהִ֣י לְ֭אֵבֶל כִּנֹּרִ֑י וְ֝עֻגָבִ֗י לְק֣וֹל בֹּכִֽים׃ (מד) בְּ֭רִית כָּרַ֣תִּי לְעֵינָ֑י וּמָ֥ה אֶ֝תְבּוֹנֵ֗ן עַל־ בְּתוּלָֽה׃ (מה) וּמֶ֤ה ׀ חֵ֣לֶק אֱל֣וֹהַּ מִמָּ֑עַל וְֽנַחֲלַ֥ת שַׁ֝דַּ֗י מִמְּרֹמִֽים׃ (מו) הֲלֹא־ אֵ֥יד לְעַוָּ֑ל וְ֝נֵ֗כֶר לְפֹ֣עֲלֵי אָֽוֶן׃ (מז) הֲלֹא־ ה֭וּא יִרְאֶ֣ה דְרָכָ֑י וְֽכָל־ צְעָדַ֥י יִסְפּֽוֹר׃ (מח) אִם־ הָלַ֥כְתִּי עִם־ שָׁ֑וְא וַתַּ֖חַשׁ עַל־ מִרְמָ֣ה רַגְלִֽי׃ (מט) יִשְׁקְלֵ֥נִי בְמֹאזְנֵי־ צֶ֑דֶק וְיֵדַ֥ע אֱ֝ל֗וֹהַּ תֻּמָּתִֽי׃ (נ) אִ֥ם תִּטֶּ֣ה אַשֻּׁרִי֮ מִנִּ֪י הַ֫דָּ֥רֶךְ וְאַחַ֣ר עֵ֭ינַי הָלַ֣ךְ לִבִּ֑י וּ֝בְכַפַּ֗י דָּ֣בַק מֻאֽוּם׃ (נא) אֶ֭זְרְעָה וְאַחֵ֣ר יֹאכֵ֑ל וְֽצֶאֱצָאַ֥י יְשֹׁרָֽשׁוּ׃ (נב) אִם־ נִפְתָּ֣ה לִ֭בִּי עַל־ אִשָּׁ֑ה וְעַל־ פֶּ֖תַח רֵעִ֣י אָרָֽבְתִּי׃ (נג) תִּטְחַ֣ן לְאַחֵ֣ר אִשְׁתִּ֑י וְ֝עָלֶ֗יהָ יִכְרְע֥וּן אֲחֵרִֽין׃ (נד) כִּי־ (הוא) הִ֥יא זִמָּ֑ה (והיא) וְ֝ה֗וּא עָוֺ֥ן פְּלִילִֽים׃ (נה) כִּ֤י אֵ֣שׁ הִ֭יא עַד־ אֲבַדּ֣וֹן תֹּאכֵ֑ל וּֽבְכָל־ תְּב֖וּאָתִ֣י תְשָׁרֵֽשׁ׃ (נו) אִם־ אֶמְאַ֗ס מִשְׁפַּ֣ט עַ֭בְדִּי וַאֲמָתִ֑י בְּ֝רִבָ֗ם עִמָּדִֽי׃ (נז) וּמָ֣ה אֶֽ֭עֱשֶׂה כִּֽי־ יָק֣וּם אֵ֑ל וְכִֽי־ יִ֝פְקֹ֗ד מָ֣ה אֲשִׁיבֶֽנּוּ׃ (נח) הֲ‍ֽ֝לֹא־ בַ֭בֶּטֶן עֹשֵׂ֣נִי עָשָׂ֑הוּ וַ֝יְכֻנֶ֗נּוּ בָּרֶ֥חֶם אֶחָֽד׃ (נט) אִם־ אֶ֭מְנַע מֵחֵ֣פֶץ דַּלִּ֑ים וְעֵינֵ֖י אַלְמָנָ֣ה אֲכַלֶּֽה׃ (ס) וְאֹכַ֣ל פִּתִּ֣י לְבַדִּ֑י וְלֹא־ אָכַ֖ל יָת֣וֹם מִמֶּֽנָּה׃ (סא) כִּ֣י מִ֭נְּעוּרַי גְּדֵלַ֣נִי כְאָ֑ב וּמִבֶּ֖טֶן אִמִּ֣י אַנְחֶֽנָּה׃ (סב) אִם־ אֶרְאֶ֣ה א֭וֹבֵד מִבְּלִ֣י לְב֑וּשׁ וְאֵ֥ין כְּ֝ס֗וּת לָאֶבְיֽוֹן׃ (סג) אִם־ לֹ֣א בֵרֲכ֣וּנִי (חלצו) חֲלָצָ֑יו וּמִגֵּ֥ז כְּ֝בָשַׂי יִתְחַמָּֽם׃ (סד) אִם־ הֲנִיפ֣וֹתִי עַל־ יָת֣וֹם יָדִ֑י כִּֽי־ אֶרְאֶ֥ה בַ֝שַּׁ֗עַר עֶזְרָתִֽי׃ (סה) כְּ֭תֵפִי מִשִּׁכְמָ֣ה תִפּ֑וֹל וְ֝אֶזְרֹעִ֗י מִקָּנָ֥ה תִשָּׁבֵֽר׃ (סו) כִּ֤י פַ֣חַד אֵ֭לַי אֵ֣יד אֵ֑ל וּ֝מִשְּׂאֵת֗וֹ לֹ֣א אוּכָֽל׃ (סז) אִם־ שַׂ֣מְתִּי זָהָ֣ב כִּסְלִ֑י וְ֝לַכֶּ֗תֶם אָמַ֥רְתִּי מִבְטַחִֽי׃ (סח) אִם־ אֶ֭שְׂמַח כִּי־ רַ֣ב חֵילִ֑י וְכִֽי־ כַ֝בִּ֗יר מָצְאָ֥ה יָדִֽי׃ (סט) אִם־ אֶרְאֶ֣ה א֖וֹר כִּ֣י יָהֵ֑ל וְ֝יָרֵ֗חַ יָקָ֥ר הֹלֵֽךְ׃ (ע) וַיִּ֣פְתְּ בַּסֵּ֣תֶר לִבִּ֑י וַתִּשַּׁ֖ק יָדִ֣י לְפִֽי׃ (עא) גַּם־ ה֭וּא עָוֺ֣ן פְּלִילִ֑י כִּֽי־ כִחַ֖שְׁתִּי לָאֵ֣ל מִמָּֽעַל׃ (עב) אִם־ אֶ֭שְׂמַח בְּפִ֣יד מְשַׂנְאִ֑י וְ֝הִתְעֹרַ֗רְתִּי כִּֽי־ מְצָ֥אוֹ רָֽע׃ (עג) וְלֹא־ נָתַ֣תִּי לַחֲטֹ֣א חִכִּ֑י לִשְׁאֹ֖ל בְּאָלָ֣ה נַפְשֽׁוֹ׃ (עד) אִם־ לֹ֣א אָ֭מְרוּ מְתֵ֣י אָהֳלִ֑י מִֽי־ יִתֵּ֥ן מִ֝בְּשָׂר֗וֹ לֹ֣א נִשְׂבָּֽע׃ (עה) בַּ֭חוּץ לֹא־ יָלִ֣ין גֵּ֑ר דְּ֝לָתַ֗י לָאֹ֥רַח אֶפְתָּֽח׃ (עו) אִם־ כִּסִּ֣יתִי כְאָדָ֣ם פְּשָׁעָ֑י לִטְמ֖וֹן בְּחֻבִּ֣י עֲוֺֽנִי׃ (עז) כִּ֤י אֶֽעֱר֨וֹץ ׀ הָ֘מ֤וֹן רַבָּ֗ה וּבוּז־ מִשְׁפָּח֥וֹת יְחִתֵּ֑נִי וָ֝אֶדֹּ֗ם לֹא־ אֵ֥צֵא פָֽתַח׃ (עח) מִ֤י יִתֶּן־ לִ֨י ׀ שֹׁ֘מֵ֤עַֽ לִ֗י הֶן־ תָּ֭וִי שַׁדַּ֣י יַעֲנֵ֑נִי וְסֵ֥פֶר כָּ֝תַ֗ב אִ֣ישׁ רִיבִֽי׃ (עט) אִם־ לֹ֣א עַל־ שִׁ֭כְמִי אֶשָּׂאֶ֑נּוּ אֶֽעֶנְדֶ֖נּוּ עֲטָר֣וֹת לִֽי׃ (פ) מִסְפַּ֣ר צְ֭עָדַי אַגִּידֶ֑נּוּ כְּמוֹ־ נָ֝גִ֗יד אֲקָרֲבֶֽנּוּ׃ (פא) אִם־ עָ֭לַי אַדְמָתִ֣י תִזְעָ֑ק וְ֝יַ֗חַד תְּלָמֶ֥יהָ יִבְכָּיֽוּן׃ (פב) אִם־ כֹּ֭חָהּ אָכַ֣לְתִּי בְלִי־ כָ֑סֶף וְנֶ֖פֶשׁ בְּעָלֶ֣יהָ הִפָּֽחְתִּי׃ (פג) תַּ֤חַת חִטָּ֨ה ׀ יֵ֥צֵא ח֗וֹחַ וְתַֽחַת־ שְׂעֹרָ֥ה בָאְשָׁ֑ה תַּ֝֗מּוּ דִּבְרֵ֥י אִיּֽוֹב׃ (פד) וַֽיִּשְׁבְּת֡וּ שְׁלֹ֤שֶׁת הָאֲנָשִׁ֣ים הָ֭אֵלֶּה מֵעֲנ֣וֹת אֶת־ אִיּ֑וֹב כִּ֤י ה֖וּא צַדִּ֣יק בְּעֵינָֽיו׃ (פה) וַיִּ֤חַר אַ֨ף ׀ אֱלִיה֣וּא בֶן־ בַּרַכְאֵ֣ל הַבּוּזִי֮ מִמִּשְׁפַּ֪חַ֫ת רָ֥ם בְּ֭אִיּוֹב חָרָ֣ה אַפּ֑וֹ עַֽל־ צַדְּק֥וֹ נַ֝פְשׁ֗וֹ מֵאֱלֹהִֽים׃ (פו) וּבִשְׁלֹ֣שֶׁת רֵעָיו֮ חָרָ֪ה אַ֫פּ֥וֹ עַ֤ל אֲשֶׁ֣ר לֹא־ מָצְא֣וּ מַעֲנֶ֑ה וַ֝יַּרְשִׁ֗יעוּ אֶת־ אִיּֽוֹב׃ (פז) וֶֽאֱלִיה֗וּ חִכָּ֣ה אֶת־ אִ֭יּוֹב בִּדְבָרִ֑ים כִּ֤י זְֽקֵנִים־ הֵ֖מָּה מִמֶּ֣נּוּ לְיָמִֽים׃ (פח) וַיַּ֤רְא אֱלִיה֗וּא כִּ֘י אֵ֤ין מַעֲנֶ֗ה בְּ֭פִי שְׁלֹ֥שֶׁת הָאֲנָשִׁ֗ים וַיִּ֥חַר אַפּֽוֹ׃ (פט) וַיַּ֤עַן ׀ אֱלִיה֖וּא בֶן־ בַּֽרַכְאֵ֥ל הַבּוּזִ֗י וַיֹּ֫אמַ֥ר צָ֘עִ֤יר אֲנִ֣י לְ֭יָמִים וְאַתֶּ֣ם יְשִׁישִׁ֑ים עַל־ כֵּ֖ן זָחַ֥לְתִּי וָֽאִירָ֓א ׀ מֵחַוֺּ֖ת דֵּעִ֣י אֶתְכֶֽם׃ (צ) אָ֭מַרְתִּי יָמִ֣ים יְדַבֵּ֑רוּ וְרֹ֥ב שָׁ֝נִ֗ים יֹדִ֥יעוּ חָכְמָֽה׃ (צא) אָ֭כֵן רֽוּחַ־ הִ֣יא בֶאֱנ֑וֹשׁ וְנִשְׁמַ֖ת שַׁדַּ֣י תְּבִינֵֽם׃ (צב) לֹֽא־ רַבִּ֥ים יֶחְכָּ֑מוּ וּ֝זְקֵנִ֗ים יָבִ֥ינוּ מִשְׁפָּֽט׃ (צג) לָכֵ֣ן אָ֭מַרְתִּי שִׁמְעָה־ לִּ֑י אֲחַוֶּ֖ה דֵּעִ֣י אַף־ אָֽנִי׃ (צד) הֵ֤ן הוֹחַ֨לְתִּי לְֽדִבְרֵיכֶ֗ם אָ֭זִין עַד־ תְּב֥וּנֹֽתֵיכֶ֑ם עַֽד־ תַּחְקְר֥וּן מִלִּֽין׃ (צה) וְעָֽדֵיכֶ֗ם אֶתְבּ֫וֹנָ֥ן וְהִנֵּ֤ה אֵ֣ין לְאִיּ֣וֹב מוֹכִ֑יחַ עוֹנֶ֖ה אֲמָרָ֣יו מִכֶּֽם׃ (צו) פֶּן־ תֹּ֣֭אמְרוּ מָצָ֣אנוּ חָכְמָ֑ה אֵ֖ל יִדְּפֶ֣נּוּ לֹא־ אִֽישׁ׃ (צז) וְלֹא־ עָרַ֣ךְ אֵלַ֣י מִלִּ֑ין וּ֝בְאִמְרֵיכֶ֗ם לֹ֣א אֲשִׁיבֶֽנּוּ׃ (צח) חַ֭תּוּ לֹא־ עָ֣נוּ ע֑וֹד הֶעְתִּ֖יקוּ מֵהֶ֣ם מִלִּֽים׃ (צט) וְ֭הוֹחַלְתִּי כִּי־ לֹ֣א יְדַבֵּ֑רוּ כִּ֥י עָ֝מְד֗וּ לֹא־ עָ֥נוּ עֽוֹד׃ (ק) אַעֲנֶ֣ה אַף־ אֲנִ֣י חֶלְקִ֑י אֲחַוֶּ֖ה דֵעִ֣י אַף־ אָֽנִי׃ (קא) כִּ֭י מָלֵ֣תִי מִלִּ֑ים הֱ֝צִיקַ֗תְנִי ר֣וּחַ בִּטְנִֽי׃ (קב) הִנֵּֽה־ בִטְנִ֗י כְּיַ֥יִן לֹא־ יִפָּתֵ֑חַ כְּאֹב֥וֹת חֲ֝דָשִׁ֗ים יִבָּקֵֽעַ׃ (קג) אֲדַבְּרָ֥ה וְיִֽרְוַֽח־ לִ֑י אֶפְתַּ֖ח שְׂפָתַ֣י וְאֶֽעֱנֶֽה׃ (קד) אַל־ נָ֭א אֶשָּׂ֣א פְנֵי־ אִ֑ישׁ וְאֶל־ אָ֝דָ֗ם לֹ֣א אֲכַנֶּֽה׃ (קה) כִּ֤י לֹ֣א יָדַ֣עְתִּי אֲכַנֶּ֑ה כִּ֝מְעַ֗ט יִשָּׂאֵ֥נִי עֹשֵֽׂנִי׃ (קו) וְֽאוּלָ֗ם שְׁמַֽע־ נָ֣א אִיּ֣וֹב מִלָּ֑י וְֽכָל־ דְּבָרַ֥י הַאֲזִֽינָה׃ (קז) הִנֵּה־ נָ֭א פָּתַ֣חְתִּי פִ֑י דִּבְּרָ֖ה לְשׁוֹנִ֣י בְחִכִּֽי׃ (קח) יֹֽשֶׁר־ לִבִּ֥י אֲמָרָ֑י וְדַ֥עַת שְׂ֝פָתַ֗י בָּר֥וּר מִלֵּֽלוּ׃ (קט) רֽוּחַ־ אֵ֥ל עָשָׂ֑תְנִי וְנִשְׁמַ֖ת שַׁדַּ֣י תְּחַיֵּֽנִי׃ (קי) אִם־ תּוּכַ֥ל הֲשִׁיבֵ֑נִי עֶרְכָ֥ה לְ֝פָנַ֗י הִתְיַצָּֽבָה׃ (קיא) הֵן־ אֲנִ֣י כְפִ֣יךָ לָאֵ֑ל מֵ֝חֹ֗מֶר קֹרַ֥צְתִּי גַם־ אָֽנִי׃ (קיב) הִנֵּ֣ה אֵ֭מָתִי לֹ֣א תְבַעֲתֶ֑ךָּ וְ֝אַכְפִּ֗י עָלֶ֥יךָ לֹא־ יִכְבָּֽד׃ (קיג) אַ֭ךְ אָמַ֣רְתָּ בְאָזְנָ֑י וְק֖וֹל מִלִּ֣ין אֶשְׁמָֽע׃ (קיד) זַ֥ךְ אֲנִ֗י בְּֽלִ֫י פָ֥שַׁע חַ֥ף אָנֹכִ֑י וְלֹ֖א עָוֺ֣ן לִֽי׃ (קטו) הֵ֣ן תְּ֭נוּאוֹת עָלַ֣י יִמְצָ֑א יַחְשְׁבֵ֖נִי לְאוֹיֵ֣ב לֽוֹ׃ (קטז) יָשֵׂ֣ם בַּסַּ֣ד רַגְלָ֑י יִ֝שְׁמֹ֗ר כָּל־ אָרְחֹתָֽי׃ (קיז) הֶן־ זֹ֣את לֹא־ צָדַ֣קְתָּ אֶעֱנֶ֑ךָּ כִּֽי־ יִרְבֶּ֥ה אֱ֝ל֗וֹהַ מֵאֱנֽוֹשׁ׃ (קיח) מַ֭דּוּעַ אֵלָ֣יו רִיב֑וֹתָ כִּ֥י כָל־ דְּ֝בָרָ֗יו לֹ֣א־ יַעֲנֶֽה׃ (קיט) כִּֽי־ בְאַחַ֥ת יְדַבֶּר־ אֵ֑ל וּ֝בִשְׁתַּ֗יִם לֹ֣א יְשׁוּרֶֽנָּה׃ (קכ) בַּחֲל֤וֹם ׀ חֶזְי֬וֹן לַ֗יְלָה בִּנְפֹ֣ל תַּ֭רְדֵּמָה עַל־ אֲנָשִׁ֑ים בִּ֝תְנוּמ֗וֹת עֲלֵ֣י מִשְׁכָּֽב׃ (קכא) אָ֣ז יִ֭גְלֶה אֹ֣זֶן אֲנָשִׁ֑ים וּבְמֹ֖סָרָ֣ם יַחְתֹּֽם׃ (קכב) לְ֭הָסִיר אָדָ֣ם מַעֲשֶׂ֑ה וְגֵוָ֖ה מִגֶּ֣בֶר יְכַסֶּֽה׃ (קכג) יַחְשֹׂ֣ךְ נַ֭פְשׁוֹ מִנִּי־ שָׁ֑חַת וְ֝חַיָּת֗וֹ מֵעֲבֹ֥ר בַּשָּֽׁלַח׃ (קכד) וְהוּכַ֣ח בְּ֭מַכְאוֹב עַל־ מִשְׁכָּב֑וֹ (וריב) וְר֖וֹב עֲצָמָ֣יו אֵתָֽן׃ (קכה) וְזִֽהֲמַ֣תּוּ חַיָּת֣וֹ לָ֑חֶם וְ֝נַפְשׁ֗וֹ מַאֲכַ֥ל תַּאֲוָֽה׃ (קכו) יִ֣כֶל בְּשָׂר֣וֹ מֵרֹ֑אִי (ושפי) וְשֻׁפּ֥וּ עַ֝צְמוֹתָ֗יו לֹ֣א רֻאּֽוּ׃ (קכז) וַתִּקְרַ֣ב לַשַּׁ֣חַת נַפְשׁ֑וֹ וְ֝חַיָּת֗וֹ לַֽמְמִתִֽים׃ (קכח) אִם־ יֵ֤שׁ עָלָ֨יו ׀ מַלְאָ֗ךְ מֵלִ֗יץ אֶחָ֥ד מִנִּי־ אָ֑לֶף לְהַגִּ֖יד לְאָדָ֣ם יָשְׁרֽוֹ׃ (קכט) וַיְחֻנֶּ֗נּוּ וַיֹּ֗אמֶר פְּ֭דָעֵהוּ מֵרֶ֥דֶת שָׁ֗חַת מָצָ֥אתִי כֹֽפֶר׃ (קל) רֻֽטֲפַ֣שׁ בְּשָׂר֣וֹ מִנֹּ֑עַר יָ֝שׁ֗וּב לִימֵ֥י עֲלוּמָֽיו׃ (קלא) יֶעְתַּ֤ר אֶל־ אֱל֨וֹהַּ ׀ וַיִּרְצֵ֗הוּ וַיַּ֣רְא פָּ֭נָיו בִּתְרוּעָ֑ה וַיָּ֥שֶׁב לֶ֝אֱנ֗וֹשׁ צִדְקָתֽוֹ׃ (קלב) יָשֹׁ֤ר ׀ עַל־ אֲנָשִׁ֗ים וַיֹּ֗אמֶר חָ֭טָאתִי וְיָשָׁ֥ר הֶעֱוֵ֗יתִי וְלֹא־ שָׁ֥וָה לִֽי׃ (קלג) פָּדָ֣ה (נפשי) נַ֭פְשׁוֹ מֵעֲבֹ֣ר בַּשָּׁ֑חַת (וחיתי) וְ֝חַיָּתוֹ בָּא֥וֹר תִּרְאֶֽה׃ (קלד) הֶן־ כָּל־ אֵ֭לֶּה יִפְעַל־ אֵ֑ל פַּעֲמַ֖יִם שָׁל֣וֹשׁ עִם־ גָּֽבֶר׃ (קלה) לְהָשִׁ֣יב נַ֭פְשׁוֹ מִנִּי־ שָׁ֑חַת לֵ֝א֗וֹר בְּא֣וֹר הַֽחַיִּים׃ (קלו) הַקְשֵׁ֖ב אִיּ֥וֹב שְֽׁמַֽע־ לִ֑י הַ֝חֲרֵ֗שׁ וְאָנֹכִ֥י אֲדַבֵּֽר׃ (קלז) אִם־ יֵשׁ־ מִלִּ֥ין הֲשִׁיבֵ֑נִי דַּ֝בֵּ֗ר כִּֽי־ חָפַ֥צְתִּי צַדְּקֶֽךָּ׃
