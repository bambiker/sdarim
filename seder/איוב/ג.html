איוב סדר ג

לפי החלוקה לפרקים: איוב יא, יט – יז, ח · 142 פסוקים

פתיחה באתר · ספר איוב כולו ·

(א) וְֽ֭רָבַצְתָּ וְאֵ֣ין מַחֲרִ֑יד וְחִלּ֖וּ פָנֶ֣יךָ רַבִּֽים׃ (ב) וְעֵינֵ֥י רְשָׁעִ֗ים תִּ֫כְלֶ֥ינָה וּ֭מָנוֹס אָבַ֣ד מִנְהֶ֑ם וְ֝תִקְוָתָ֗ם מַֽפַּח־ נָֽפֶשׁ׃ (ג) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (ד) אָ֭מְנָם כִּ֣י אַתֶּם־ עָ֑ם וְ֝עִמָּכֶ֗ם תָּמ֥וּת חָכְמָֽה׃ (ה) גַּם־ לִ֤י לֵבָ֨ב ׀ כְּֽמוֹכֶ֗ם לֹא־ נֹפֵ֣ל אָנֹכִ֣י מִכֶּ֑ם וְאֶת־ מִי־ אֵ֥ין כְּמוֹ־ אֵֽלֶּה׃ (ו) שְׂחֹ֤ק לְרֵעֵ֨הוּ ׀ אֶֽהְיֶ֗ה קֹרֵ֣א לֶ֭אֱלוֹהַּ וַֽיַּעֲנֵ֑הוּ שְׂ֝ח֗וֹק צַדִּ֥יק תָּמִֽים׃ (ז) לַפִּ֣יד בּ֭וּז לְעַשְׁתּ֣וּת שַׁאֲנָ֑ן נָ֝כ֗וֹן לְמ֣וֹעֲדֵי רָֽגֶל׃ (ח) יִשְׁלָ֤יוּ אֹֽהָלִ֨ים ׀ לְשֹׁ֥דְדִ֗ים וּֽ֭בַטֻּחוֹת לְמַרְגִּ֣יזֵי אֵ֑ל לַאֲשֶׁ֤ר הֵבִ֖יא אֱל֣וֹהַּ בְּיָדֽוֹ׃ (ט) וְֽאוּלָ֗ם שְׁאַל־ נָ֣א בְהֵמ֣וֹת וְתֹרֶ֑ךָּ וְע֥וֹף הַ֝שָּׁמַ֗יִם וְיַגֶּד־ לָֽךְ׃ (י) א֤וֹ שִׂ֣יחַ לָאָ֣רֶץ וְתֹרֶ֑ךָּ וִֽיסַפְּר֥וּ לְ֝ךָ֗ דְּגֵ֣י הַיָּֽם׃ (יא) מִ֭י לֹא־ יָדַ֣ע בְּכָל־ אֵ֑לֶּה כִּ֥י יַד־ יְ֝הוָה עָ֣שְׂתָה זֹּֽאת׃ (יב) אֲשֶׁ֣ר בְּ֭יָדוֹ נֶ֣פֶשׁ כָּל־ חָ֑י וְ֝ר֗וּחַ כָּל־ בְּשַׂר־ אִֽישׁ׃ (יג) הֲלֹא־ אֹ֭זֶן מִלִּ֣ין תִּבְחָ֑ן וְ֝חֵ֗ךְ אֹ֣כֶל יִטְעַם־ לֽוֹ׃ (יד) בִּֽישִׁישִׁ֥ים חָכְמָ֑ה וְאֹ֖רֶךְ יָמִ֣ים תְּבוּנָֽה׃ (טו) עִ֭מּוֹ חָכְמָ֣ה וּגְבוּרָ֑ה ל֝֗וֹ עֵצָ֥ה וּתְבוּנָֽה׃ (טז) הֵ֣ן יַ֭הֲרוֹס וְלֹ֣א יִבָּנֶ֑ה יִסְגֹּ֥ר עַל־ אִ֝֗ישׁ וְלֹ֣א יִפָּתֵֽחַ׃ (יז) הֵ֤ן יַעְצֹ֣ר בַּמַּ֣יִם וְיִבָ֑שׁוּ וִֽ֝ישַׁלְּחֵ֗ם וְיַ֖הַפְכוּ אָֽרֶץ׃ (יח) עִ֭מּוֹ עֹ֣ז וְתֽוּשִׁיָּ֑ה ל֝֗וֹ שֹׁגֵ֥ג וּמַשְׁגֶּֽה׃ (יט) מוֹלִ֣יךְ יוֹעֲצִ֣ים שׁוֹלָ֑ל וְֽשֹׁפְטִ֥ים יְהוֹלֵֽל׃ (כ) מוּסַ֣ר מְלָכִ֣ים פִּתֵּ֑חַ וַיֶּאְסֹ֥ר אֵ֝ז֗וֹר בְּמָתְנֵיהֶֽם׃ (כא) מוֹלִ֣יךְ כֹּהֲנִ֣ים שׁוֹלָ֑ל וְאֵֽתָנִ֣ים יְסַלֵּֽף׃ (כב) מֵסִ֣יר שָׂ֭פָה לְנֶאֱמָנִ֑ים וְטַ֖עַם זְקֵנִ֣ים יִקָּֽח׃ (כג) שׁוֹפֵ֣ךְ בּ֭וּז עַל־ נְדִיבִ֑ים וּמְזִ֖יחַ אֲפִיקִ֣ים רִפָּֽה׃ (כד) מְגַלֶּ֣ה עֲ֭מֻקוֹת מִנִּי־ חֹ֑שֶׁךְ וַיֹּצֵ֖א לָא֣וֹר צַלְמָֽוֶת׃ (כה) מַשְׂגִּ֣יא לַ֭גּוֹיִם וַֽיְאַבְּדֵ֑ם שֹׁטֵ֥חַ לַ֝גּוֹיִ֗ם וַיַּנְחֵֽם׃ (כו) מֵסִ֗יר לֵ֭ב רָאשֵׁ֣י עַם־ הָאָ֑רֶץ וַ֝יַּתְעֵ֗ם בְּתֹ֣הוּ לֹא־ דָֽרֶךְ׃ (כז) יְמַֽשְׁשׁוּ־ חֹ֥שֶׁךְ וְלֹא־ א֑וֹר וַ֝יַּתְעֵ֗ם כַּשִּׁכּֽוֹר׃ (כח) הֶן־ כֹּ֭ל רָאֲתָ֣ה עֵינִ֑י שָֽׁמְעָ֥ה אָ֝זְנִ֗י וַתָּ֥בֶן לָֽהּ׃ (כט) כְּֽ֭דַעְתְּכֶם יָדַ֣עְתִּי גַם־ אָ֑נִי לֹא־ נֹפֵ֖ל אָנֹכִ֣י מִכֶּֽם׃ (ל) אוּלָ֗ם אֲ֭נִי אֶל־ שַׁדַּ֣י אֲדַבֵּ֑ר וְהוֹכֵ֖חַ אֶל־ אֵ֣ל אֶחְפָּֽץ׃ (לא) וְֽאוּלָ֗ם אַתֶּ֥ם טֹֽפְלֵי־ שָׁ֑קֶר רֹפְאֵ֖י אֱלִ֣ל כֻּלְּכֶֽם׃ (לב) מִֽי־ יִ֭תֵּן הַחֲרֵ֣שׁ תַּחֲרִישׁ֑וּן וּתְהִ֖י לָכֶ֣ם לְחָכְמָֽה׃ (לג) שִׁמְעוּ־ נָ֥א תוֹכַחְתִּ֑י וְרִב֖וֹת שְׂפָתַ֣י הַקְשִֽׁיבוּ׃ (לד) הַ֭לְאֵל תְּדַבְּר֣וּ עַוְלָ֑ה וְ֝ל֗וֹ תְּֽדַבְּר֥וּ רְמִיָּֽה׃ (לה) הֲפָנָ֥יו תִּשָּׂא֑וּן אִם־ לָאֵ֥ל תְּרִיבֽוּן׃ (לו) הֲ֭טוֹב כִּֽי־ יַחְקֹ֣ר אֶתְכֶ֑ם אִם־ כְּהָתֵ֥ל בֶּ֝אֱנ֗וֹשׁ תְּהָתֵ֥לּוּ בֽוֹ׃ (לז) הוֹכֵ֣חַ יוֹכִ֣יחַ אֶתְכֶ֑ם אִם־ בַּ֝סֵּ֗תֶר פָּנִ֥ים תִּשָּׂאֽוּן׃ (לח) הֲלֹ֣א שְׂ֭אֵתוֹ תְּבַעֵ֣ת אֶתְכֶ֑ם וּ֝פַחְדּ֗וֹ יִפֹּ֥ל עֲלֵיכֶֽם׃ (לט) זִֽ֭כְרֹנֵיכֶם מִשְׁלֵי־ אֵ֑פֶר לְגַבֵּי־ חֹ֝֗מֶר גַּבֵּיכֶֽם׃ (מ) הַחֲרִ֣ישׁוּ מִ֭מֶּנִּי וַאֲדַבְּרָה־ אָ֑נִי וְיַעֲבֹ֖ר עָלַ֣י מָֽה׃ (מא) עַל־ מָ֤ה ׀ אֶשָּׂ֣א בְשָׂרִ֣י בְשִׁנָּ֑י וְ֝נַפְשִׁ֗י אָשִׂ֥ים בְּכַפִּֽי׃ (מב) הֵ֣ן יִ֭קְטְלֵנִי (לא) ל֣וֹ אֲיַחֵ֑ל אַךְ־ דְּ֝רָכַ֗י אֶל־ פָּנָ֥יו אוֹכִֽיחַ׃ (מג) גַּם־ הוּא־ לִ֥י לִֽישׁוּעָ֑ה כִּי־ לֹ֥א לְ֝פָנָ֗יו חָנֵ֥ף יָבֽוֹא׃ (מד) שִׁמְע֣וּ שָׁ֭מוֹעַ מִלָּתִ֑י וְ֝אַֽחֲוָתִ֗י בְּאָזְנֵיכֶֽם׃ (מה) הִנֵּה־ נָ֭א עָרַ֣כְתִּי מִשְׁפָּ֑ט יָ֝דַ֗עְתִּי כִּֽי־ אֲנִ֥י אֶצְדָּֽק׃ (מו) מִי־ ה֭וּא יָרִ֣יב עִמָּדִ֑י כִּֽי־ עַתָּ֖ה אַחֲרִ֣ישׁ וְאֶגְוָֽע׃ (מז) אַךְ־ שְׁ֭תַּיִם אַל־ תַּ֣עַשׂ עִמָּדִ֑י אָ֥ז מִ֝פָּנֶ֗יךָ לֹ֣א אֶסָּתֵֽר׃ (מח) כַּ֭פְּךָ מֵעָלַ֣י הַרְחַ֑ק וְ֝אֵ֥מָתְךָ֗ אַֽל־ תְּבַעֲתַֽנִּי׃ (מט) וּ֭קְרָא וְאָנֹכִ֣י אֶֽעֱנֶ֑ה אֽוֹ־ אֲ֝דַבֵּ֗ר וַהֲשִׁיבֵֽנִי׃ (נ) כַּמָּ֣ה לִ֭י עֲוֺנ֣וֹת וְחַטָּא֑וֹת פִּֽשְׁעִ֥י וְ֝חַטָּאתִ֗י הֹדִיעֵֽנִי׃ (נא) לָֽמָּה־ פָנֶ֥יךָ תַסְתִּ֑יר וְתַחְשְׁבֵ֖נִי לְאוֹיֵ֣ב לָֽךְ׃ (נב) הֶעָלֶ֣ה נִדָּ֣ף תַּעֲר֑וֹץ וְאֶת־ קַ֖שׁ יָבֵ֣שׁ תִּרְדֹּֽף׃ (נג) כִּֽי־ תִכְתֹּ֣ב עָלַ֣י מְרֹר֑וֹת וְ֝תוֹרִישֵׁ֗נִי עֲוֺנ֥וֹת נְעוּרָֽי׃ (נד) וְתָ֘שֵׂ֤ם בַּסַּ֨ד ׀ רַגְלַ֗י וְתִשְׁמ֥וֹר כָּל־ אָרְחוֹתָ֑י עַל־ שָׁרְשֵׁ֥י רַ֝גְלַ֗י תִּתְחַקֶּֽה׃ (נה) וְ֭הוּא כְּרָקָ֣ב יִבְלֶ֑ה כְּ֝בֶ֗גֶד אֲכָ֣לוֹ עָֽשׁ׃ (נו) אָ֭דָם יְל֣וּד אִשָּׁ֑ה קְצַ֥ר יָ֝מִ֗ים וּֽשְׂבַֽע־ רֹֽגֶז׃ (נז) כְּצִ֣יץ יָ֭צָא וַיִּמָּ֑ל וַיִּבְרַ֥ח כַּ֝צֵּ֗ל וְלֹ֣א יַעֲמֽוֹד׃ (נח) אַף־ עַל־ זֶ֭ה פָּקַ֣חְתָּ עֵינֶ֑ךָ וְאֹ֘תִ֤י תָבִ֖יא בְמִשְׁפָּ֣ט עִמָּֽךְ׃ (נט) מִֽי־ יִתֵּ֣ן טָ֭הוֹר מִטָּמֵ֗א לֹ֣א אֶחָֽד׃ (ס) אִ֥ם חֲרוּצִ֨ים ׀ יָמָ֗יו מִֽסְפַּר־ חֳדָשָׁ֥יו אִתָּ֑ךְ (חקו) חֻקָּ֥יו עָ֝שִׂ֗יתָ וְלֹ֣א יַעֲבֽוֹר׃ (סא) שְׁעֵ֣ה מֵעָלָ֣יו וְיֶחְדָּ֑ל עַד־ יִ֝רְצֶ֗ה כְּשָׂכִ֥יר יוֹמֽוֹ׃ (סב) כִּ֤י יֵ֥שׁ לָעֵ֗ץ תִּ֫קְוָ֥ה אִֽם־ יִ֭כָּרֵת וְע֣וֹד יַחֲלִ֑יף וְ֝יֹֽנַקְתּ֗וֹ לֹ֣א תֶחְדָּֽל׃ (סג) אִם־ יַזְקִ֣ין בָּאָ֣רֶץ שָׁרְשׁ֑וֹ וּ֝בֶעָפָ֗ר יָמ֥וּת גִּזְעֽוֹ׃ (סד) מֵרֵ֣יחַ מַ֣יִם יַפְרִ֑חַ וְעָשָׂ֖ה קָצִ֣יר כְּמוֹ־ נָֽטַע׃ (סה) וְגֶ֣בֶר יָ֭מוּת וַֽיֶּחֱלָ֑שׁ וַיִּגְוַ֖ע אָדָ֣ם וְאַיּֽוֹ׃ (סו) אָֽזְלוּ־ מַ֭יִם מִנִּי־ יָ֑ם וְ֝נָהָ֗ר יֶחֱרַ֥ב וְיָבֵֽשׁ׃ (סז) וְאִ֥ישׁ שָׁכַ֗ב וְֽלֹא־ יָ֫ק֥וּם עַד־ בִּלְתִּ֣י שָׁ֭מַיִם לֹ֣א יָקִ֑יצוּ וְלֹֽא־ יֵ֝עֹ֗רוּ מִשְּׁנָתָֽם׃ (סח) מִ֤י יִתֵּ֨ן ׀ בִּשְׁא֬וֹל תַּצְפִּנֵ֗נִי תַּ֭סְתִּירֵנִי עַד־ שׁ֣וּב אַפֶּ֑ךָ תָּ֤שִׁ֥ית לִ֖י חֹ֣ק וְתִזְכְּרֵֽנִי׃ (סט) אִם־ יָמ֥וּת גֶּ֗בֶר הֲיִ֫חְיֶ֥ה כָּל־ יְמֵ֣י צְבָאִ֣י אֲיַחֵ֑ל עַד־ בּ֝֗וֹא חֲלִיפָתִֽי׃ (ע) תִּ֭קְרָא וְאָנֹכִ֣י אֶֽעֱנֶ֑ךָּ לְֽמַעֲשֵׂ֖ה יָדֶ֣יךָ תִכְסֹֽף׃ (עא) כִּֽי־ עַ֭תָּה צְעָדַ֣י תִּסְפּ֑וֹר לֹֽא־ תִ֝שְׁמ֗וֹר עַל־ חַטָּאתִֽי׃ (עב) חָתֻ֣ם בִּצְר֣וֹר פִּשְׁעִ֑י וַ֝תִּטְפֹּ֗ל עַל־ עֲוֺנִֽי׃ (עג) וְ֭אוּלָם הַר־ נוֹפֵ֣ל יִבּ֑וֹל וְ֝צ֗וּר יֶעְתַּ֥ק מִמְּקֹמֽוֹ׃ (עד) אֲבָנִ֤ים ׀ שָׁ֥חֲקוּ מַ֗יִם תִּשְׁטֹֽף־ סְפִיחֶ֥יהָ עֲפַר־ אָ֑רֶץ וְתִקְוַ֖ת אֱנ֣וֹשׁ הֶאֱבַֽדְתָּ׃ (עה) תִּתְקְפֵ֣הוּ לָ֭נֶצַח וַֽיַּהֲלֹ֑ךְ מְשַׁנֶּ֥ה פָ֝נָ֗יו וַֽתְּשַׁלְּחֵֽהוּ׃ (עו) יִכְבְּד֣וּ בָ֭נָיו וְלֹ֣א יֵדָ֑ע וְ֝יִצְעֲר֗וּ וְֽלֹא־ יָבִ֥ין לָֽמוֹ׃ (עז) אַךְ־ בְּ֭שָׂרוֹ עָלָ֣יו יִכְאָ֑ב וְ֝נַפְשׁ֗וֹ עָלָ֥יו תֶּאֱבָֽל׃ (עח) וַ֭יַּעַן אֱלִיפַ֥ז הַֽתֵּימָנִ֗י וַיֹּאמַֽר׃ (עט) הֶֽחָכָ֗ם יַעֲנֶ֥ה דַֽעַת־ ר֑וּחַ וִֽימַלֵּ֖א קָדִ֣ים בִּטְנֽוֹ׃ (פ) הוֹכֵ֣חַ בְּ֭דָבָר לֹ֣א יִסְכּ֑וֹן וּ֝מִלִּ֗ים לֹא־ יוֹעִ֥יל בָּֽם׃ (פא) אַף־ אַ֭תָּה תָּפֵ֣ר יִרְאָ֑ה וְתִגְרַ֥ע שִׂ֝יחָ֗ה לִפְנֵי־ אֵֽל׃ (פב) כִּ֤י יְאַלֵּ֣ף עֲוֺנְךָ֣ פִ֑יךָ וְ֝תִבְחַ֗ר לְשׁ֣וֹן עֲרוּמִֽים׃ (פג) יַרְשִֽׁיעֲךָ֣ פִ֣יךָ וְלֹא־ אָ֑נִי וּ֝שְׂפָתֶ֗יךָ יַעֲנוּ־ בָֽךְ׃ (פד) הֲרִאישׁ֣וֹן אָ֭דָם תִּוָּלֵ֑ד וְלִפְנֵ֖י גְבָע֣וֹת חוֹלָֽלְתָּ׃ (פה) הַבְס֣וֹד אֱל֣וֹהַ תִּשְׁמָ֑ע וְתִגְרַ֖ע אֵלֶ֣יךָ חָכְמָֽה׃ (פו) מַה־ יָּ֭דַעְתָּ וְלֹ֣א נֵדָ֑ע תָּ֝בִ֗ין וְֽלֹא־ עִמָּ֥נוּ הֽוּא׃ (פז) גַּם־ שָׂ֣ב גַּם־ יָשִׁ֣ישׁ בָּ֑נוּ כַּבִּ֖יר מֵאָבִ֣יךָ יָמִֽים׃ (פח) הַמְעַ֣ט מִ֭מְּךָ תַּנְחֻמ֣וֹת אֵ֑ל וְ֝דָבָ֗ר לָאַ֥ט עִמָּֽךְ׃ (פט) מַה־ יִּקָּחֲךָ֥ לִבֶּ֑ךָ וּֽמַה־ יִּרְזְמ֥וּן עֵינֶֽיךָ׃ (צ) כִּֽי־ תָשִׁ֣יב אֶל־ אֵ֣ל רוּחֶ֑ךָ וְהֹצֵ֖אתָ מִפִּ֣יךָ מִלִּֽין׃ (צא) מָֽה־ אֱנ֥וֹשׁ כִּֽי־ יִזְכֶּ֑ה וְכִֽי־ יִ֝צְדַּ֗ק יְל֣וּד אִשָּֽׁה׃ (צב) הֵ֣ן (בקדשו) בִּ֭קְדֹשָׁיו לֹ֣א יַאֲמִ֑ין וְ֝שָׁמַ֗יִם לֹא־ זַכּ֥וּ בְעֵינָֽיו׃ (צג) אַ֭ף כִּֽי־ נִתְעָ֥ב וְֽנֶאֱלָ֑ח אִישׁ־ שֹׁתֶ֖ה כַמַּ֣יִם עַוְלָֽה׃ (צד) אֲחַוְךָ֥ שְֽׁמַֽע־ לִ֑י וְזֶֽה־ חָ֝זִ֗יתִי וַאֲסַפֵּֽרָה׃ (צה) אֲשֶׁר־ חֲכָמִ֥ים יַגִּ֑ידוּ וְלֹ֥א כִֽ֝חֲד֗וּ מֵאֲבוֹתָֽם׃ (צו) לָהֶ֣ם לְ֭בַדָּם נִתְּנָ֣ה הָאָ֑רֶץ וְלֹא־ עָ֖בַר זָ֣ר בְּתוֹכָֽם׃ (צז) כָּל־ יְמֵ֣י רָ֭שָׁע ה֣וּא מִתְחוֹלֵ֑ל וּמִסְפַּ֥ר שָׁ֝נִ֗ים נִצְפְּנ֥וּ לֶעָרִֽיץ׃ (צח) קוֹל־ פְּחָדִ֥ים בְּאָזְנָ֑יו בַּ֝שָּׁל֗וֹם שׁוֹדֵ֥ד יְבוֹאֶֽנּוּ׃ (צט) לֹא־ יַאֲמִ֣ין שׁ֭וּב מִנִּי־ חֹ֑שֶׁךְ (וצפו) וְצָפ֖וּי ה֣וּא אֱלֵי־ חָֽרֶב׃ (ק) נֹ֘דֵ֤ד ה֣וּא לַלֶּ֣חֶם אַיֵּ֑ה יָדַ֓ע ׀ כִּֽי־ נָכ֖וֹן בְּיָד֣וֹ יֽוֹם־ חֹֽשֶׁךְ׃ (קא) יְֽ֭בַעֲתֻהוּ צַ֣ר וּמְצוּקָ֑ה תִּ֝תְקְפֵ֗הוּ כְּמֶ֤לֶךְ ׀ עָתִ֬יד לַכִּידֽוֹר׃ (קב) כִּֽי־ נָטָ֣ה אֶל־ אֵ֣ל יָד֑וֹ וְאֶל־ שַׁ֝דַּ֗י יִתְגַּבָּֽר׃ (קג) יָר֣וּץ אֵלָ֣יו בְּצַוָּ֑אר בַּ֝עֲבִ֗י גַּבֵּ֥י מָֽגִנָּֽיו׃ (קד) כִּֽי־ כִסָּ֣ה פָנָ֣יו בְּחֶלְבּ֑וֹ וַיַּ֖עַשׂ פִּימָ֣ה עֲלֵי־ כָֽסֶל׃ (קה) וַיִּשְׁכּ֤וֹן ׀ עָ֘רִ֤ים נִכְחָד֗וֹת בָּ֭תִּים לֹא־ יֵ֣שְׁבוּ לָ֑מוֹ אֲשֶׁ֖ר הִתְעַתְּד֣וּ לְגַלִּֽים׃ (קו) לֹֽא־ יֶ֭עְשַׁר וְלֹא־ יָק֣וּם חֵיל֑וֹ וְלֹֽא־ יִטֶּ֖ה לָאָ֣רֶץ מִנְלָֽם׃ (קז) לֹֽא־ יָס֨וּר ׀ מִנִּי־ חֹ֗שֶׁךְ יֹֽ֭נַקְתּוֹ תְּיַבֵּ֣שׁ שַׁלְהָ֑בֶת וְ֝יָס֗וּר בְּר֣וּחַ פִּֽיו׃ (קח) אַל־ יַאֲמֵ֣ן (בשו) בַּשָּׁ֣יו נִתְעָ֑ה כִּי־ שָׁ֝֗וְא תִּהְיֶ֥ה תְמוּרָתֽוֹ׃ (קט) בְּֽלֹא־ י֭וֹמוֹ תִּמָּלֵ֑א וְ֝כִפָּת֗וֹ לֹ֣א רַעֲנָֽנָה׃ (קי) יַחְמֹ֣ס כַּגֶּ֣פֶן בִּסְר֑וֹ וְיַשְׁלֵ֥ךְ כַּ֝זַּ֗יִת נִצָּתֽוֹ׃ (קיא) כִּֽי־ עֲדַ֣ת חָנֵ֣ף גַּלְמ֑וּד וְ֝אֵ֗שׁ אָכְלָ֥ה אָֽהֳלֵי־ שֹֽׁחַד׃ (קיב) הָרֹ֣ה עָ֭מָל וְיָ֣לֹד אָ֑וֶן וּ֝בִטְנָ֗ם תָּכִ֥ין מִרְמָֽה׃ (קיג) וַיַּ֥עַן אִיּ֗וֹב וַיֹּאמַֽר׃ (קיד) שָׁמַ֣עְתִּי כְאֵ֣לֶּה רַבּ֑וֹת מְנַחֲמֵ֖י עָמָ֣ל כֻּלְּכֶֽם׃ (קטו) הֲקֵ֥ץ לְדִבְרֵי־ ר֑וּחַ א֥וֹ מַה־ יַּ֝מְרִֽיצְךָ֗ כִּ֣י תַעֲנֶֽה׃ (קטז) גַּ֤ם ׀ אָנֹכִי֮ כָּכֶ֪ם אֲדַ֫בֵּ֥רָה ל֤וּ־ יֵ֪שׁ נַפְשְׁכֶ֡ם תַּ֤חַת נַפְשִׁ֗י אַחְבִּ֣ירָה עֲלֵיכֶ֣ם בְּמִלִּ֑ים וְאָנִ֥יעָה עֲ֝לֵיכֶ֗ם בְּמ֣וֹ רֹאשִֽׁי׃ (קיז) אֲאַמִּצְכֶ֥ם בְּמוֹ־ פִ֑י וְנִ֖יד שְׂפָתַ֣י יַחְשֹֽׂךְ׃ (קיח) אִֽם־ אֲ֭דַבְּרָה לֹא־ יֵחָשֵׂ֣ךְ כְּאֵבִ֑י וְ֝אַחְדְּלָ֗ה מַה־ מִנִּ֥י יַהֲלֹֽךְ׃ (קיט) אַךְ־ עַתָּ֥ה הֶלְאָ֑נִי הֲ֝שִׁמּ֗וֹתָ כָּל־ עֲדָתִֽי׃ (קכ) וַֽ֭תִּקְמְטֵנִי לְעֵ֣ד הָיָ֑ה וַיָּ֥קָם בִּ֥י כַ֝חֲשִׁ֗י בְּפָנַ֥י יַעֲנֶֽה׃ (קכא) אַפּ֤וֹ טָרַ֨ף ׀ וַֽיִּשְׂטְמֵ֗נִי חָרַ֣ק עָלַ֣י בְּשִׁנָּ֑יו צָרִ֓י ׀ יִלְט֖וֹשׁ עֵינָ֣יו לִֽי׃ (קכב) פָּעֲר֬וּ עָלַ֨י ׀ בְּפִיהֶ֗ם בְּ֭חֶרְפָּה הִכּ֣וּ לְחָיָ֑י יַ֝֗חַד עָלַ֥י יִתְמַלָּאֽוּן׃ (קכג) יַסְגִּירֵ֣נִי אֵ֭ל אֶ֣ל עֲוִ֑יל וְעַל־ יְדֵ֖י רְשָׁעִ֣ים יִרְטֵֽנִי׃ (קכד) שָׁ֘לֵ֤ו הָיִ֨יתִי ׀ וַֽיְפַרְפְּרֵ֗נִי וְאָחַ֣ז בְּ֭עָרְפִּי וַֽיְפַצְפְּצֵ֑נִי וַיְקִימֵ֥נִי ל֝֗וֹ לְמַטָּרָֽה׃ (קכה) יָ֘סֹ֤בּוּ עָלַ֨י ׀ רַבָּ֗יו יְפַלַּ֣ח כִּ֭לְיוֹתַי וְלֹ֣א יַחְמ֑וֹל יִשְׁפֹּ֥ךְ לָ֝אָ֗רֶץ מְרֵרָֽתִי׃ (קכו) יִפְרְצֵ֣נִי פֶ֭רֶץ עַל־ פְּנֵי־ פָ֑רֶץ יָרֻ֖ץ עָלַ֣י כְּגִבּֽוֹר׃ (קכז) שַׂ֣ק תָּ֭פַרְתִּי עֲלֵ֣י גִלְדִּ֑י וְעֹלַ֖לְתִּי בֶעָפָ֣ר קַרְנִֽי׃ (קכח) פָּנַ֣י (חמרמרה) חֳ֭מַרְמְרוּ מִנִּי־ בֶ֑כִי וְעַ֖ל עַפְעַפַּ֣י צַלְמָֽוֶת׃ (קכט) עַ֭ל לֹא־ חָמָ֣ס בְּכַפָּ֑י וּֽתְפִלָּתִ֥י זַכָּֽה׃ (קל) אֶ֭רֶץ אַל־ תְּכַסִּ֣י דָמִ֑י וְֽאַל־ יְהִ֥י מָ֝ק֗וֹם לְזַעֲקָתִֽי׃ (קלא) גַּם־ עַ֭תָּה הִנֵּה־ בַשָּׁמַ֣יִם עֵדִ֑י וְ֝שָׂהֲדִ֗י בַּמְּרוֹמִֽים׃ (קלב) מְלִיצַ֥י רֵעָ֑י אֶל־ אֱ֝ל֗וֹהַ דָּלְפָ֥ה עֵינִֽי׃ (קלג) וְיוֹכַ֣ח לְגֶ֣בֶר עִם־ אֱל֑וֹהַּ וּֽבֶן־ אָדָ֥ם לְרֵעֵֽהוּ׃ (קלד) כִּֽי־ שְׁנ֣וֹת מִסְפָּ֣ר יֶאֱתָ֑יוּ וְאֹ֖רַח לֹא־ אָשׁ֣וּב אֶהֱלֹֽךְ׃ (קלה) רוּחִ֣י חֻ֭בָּלָה יָמַ֥י נִזְעָ֗כוּ קְבָרִ֥ים לִֽי׃ (קלו) אִם־ לֹ֣א הֲ֭תֻלִים עִמָּדִ֑י וּ֝בְהַמְּרוֹתָ֗ם תָּלַ֥ן עֵינִֽי׃ (קלז) שִֽׂימָה־ נָּ֭א עָרְבֵ֣נִי עִמָּ֑ךְ מִֽי ה֝֗וּא לְיָדִ֥י יִתָּקֵֽעַ׃ (קלח) כִּֽי־ לִ֭בָּם צָפַ֣נְתָּ מִּשָּׂ֑כֶל עַל־ כֵּ֝֗ן לֹ֣א תְרֹמֵֽם׃ (קלט) לְ֭חֵלֶק יַגִּ֣יד רֵעִ֑ים וְעֵינֵ֖י בָנָ֣יו תִּכְלֶֽנָה׃ (קמ) וְֽ֭הִצִּגַנִי לִמְשֹׁ֣ל עַמִּ֑ים וְתֹ֖פֶת לְפָנִ֣ים אֶֽהְיֶֽה׃ (קמא) וַתֵּ֣כַהּ מִכַּ֣עַשׂ עֵינִ֑י וִֽיצֻרַ֖י כַּצֵּ֣ל כֻּלָּֽם׃ (קמב) יָשֹׁ֣מּוּ יְשָׁרִ֣ים עַל־ זֹ֑את וְ֝נָקִ֗י עַל־ חָנֵ֥ף יִתְעֹרָֽר׃
