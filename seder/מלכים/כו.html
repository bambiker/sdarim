מלכים סדר כו

פתיחה באתר · ספר מלכים כולו ·

(א) וַיַּ֨עַשׂ יְהוֹאָ֧שׁ הַיָּשָׁ֛ר בְּעֵינֵ֥י יְהוָ֖ה כָּל־ יָמָ֑יו אֲשֶׁ֣ר הוֹרָ֔הוּ יְהוֹיָדָ֖ע הַכֹּהֵֽן׃ (ב) רַ֥ק הַבָּמ֖וֹת לֹא־ סָ֑רוּ ע֥וֹד הָעָ֛ם מְזַבְּחִ֥ים וּֽמְקַטְּרִ֖ים בַּבָּמֽוֹת׃ (ג) וַיֹּ֨אמֶר יְהוֹאָ֜שׁ אֶל־ הַכֹּהֲנִ֗ים כֹּל֩ כֶּ֨סֶף הַקֳּדָשִׁ֜ים אֲשֶׁר־ יוּבָ֤א בֵית־ יְהוָה֙ כֶּ֣סֶף עוֹבֵ֔ר אִ֕ישׁ כֶּ֥סֶף נַפְשׁ֖וֹת עֶרְכּ֑וֹ כָּל־ כֶּ֗סֶף אֲשֶׁ֤ר יַֽעֲלֶה֙ עַ֣ל לֶב־ אִ֔ישׁ לְהָבִ֖יא בֵּ֥ית יְהוָֽה׃ (ד) יִקְח֤וּ לָהֶם֙ הַכֹּ֣הֲנִ֔ים אִ֖ישׁ מֵאֵ֣ת מַכָּר֑וֹ וְהֵ֗ם יְחַזְּקוּ֙ אֶת־ בֶּ֣דֶק הַבַּ֔יִת לְכֹ֛ל אֲשֶׁר־ יִמָּצֵ֥א שָׁ֖ם בָּֽדֶק׃ (ה) וַיְהִ֗י בִּשְׁנַ֨ת עֶשְׂרִ֧ים וְשָׁלֹ֛שׁ שָׁנָ֖ה לַמֶּ֣לֶךְ יְהוֹאָ֑שׁ לֹֽא־ חִזְּק֥וּ הַכֹּהֲנִ֖ים אֶת־ בֶּ֥דֶק הַבָּֽיִת׃ (ו) וַיִּקְרָא֩ הַמֶּ֨לֶךְ יְהוֹאָ֜שׁ לִיהוֹיָדָ֤ע הַכֹּהֵן֙ וְלַכֹּ֣הֲנִ֔ים וַיֹּ֣אמֶר אֲלֵהֶ֔ם מַדּ֛וּעַ אֵינְכֶ֥ם מְחַזְּקִ֖ים אֶת־ בֶּ֣דֶק הַבָּ֑יִת וְעַתָּ֗ה אַל־ תִּקְחוּ־ כֶ֙סֶף֙ מֵאֵ֣ת מַכָּֽרֵיכֶ֔ם כִּֽי־ לְבֶ֥דֶק הַבַּ֖יִת תִּתְּנֻֽהוּ׃ (ז) וַיֵּאֹ֖תוּ הַכֹּֽהֲנִ֑ים לְבִלְתִּ֤י קְחַת־ כֶּ֙סֶף֙ מֵאֵ֣ת הָעָ֔ם וּלְבִלְתִּ֥י חַזֵּ֖ק אֶת־ בֶּ֥דֶק הַבָּֽיִת׃ (ח) וַיִּקַּ֞ח יְהוֹיָדָ֤ע הַכֹּהֵן֙ אֲר֣וֹן אֶחָ֔ד וַיִּקֹּ֥ב חֹ֖ר בְּדַלְתּ֑וֹ וַיִּתֵּ֣ן אֹתוֹ֩ אֵ֨צֶל הַמִּזְבֵּ֜חַ (בימין) מִיָּמִ֗ין בְּבֽוֹא־ אִישׁ֙ בֵּ֣ית יְהוָ֔ה וְנָֽתְנוּ־ שָׁ֤מָּה הַכֹּֽהֲנִים֙ שֹׁמְרֵ֣י הַסַּ֔ף אֶת־ כָּל־ הַכֶּ֖סֶף הַמּוּבָ֥א בֵית־ יְהוָֽה׃ (ט) וַֽיְהִי֙ כִּרְאוֹתָ֔ם כִּֽי־ רַ֥ב הַכֶּ֖סֶף בָּֽאָר֑וֹן וַיַּ֨עַל סֹפֵ֤ר הַמֶּ֙לֶךְ֙ וְהַכֹּהֵ֣ן הַגָּד֔וֹל וַיָּצֻ֙רוּ֙ וַיִּמְנ֔וּ אֶת־ הַכֶּ֖סֶף הַנִּמְצָ֥א בֵית־ יְהוָֽה׃ (י) וְנָתְנוּ֙ אֶת־ הַכֶּ֣סֶף הַֽמְתֻכָּ֔ן עַל־ (יד) יְדֵי֙ עֹשֵׂ֣י הַמְּלָאכָ֔ה (הפקדים) הַמֻּפְקָדִ֖ים בֵּ֣ית יְהוָ֑ה וַיּוֹצִיאֻ֜הוּ לְחָרָשֵׁ֤י הָעֵץ֙ וְלַבֹּנִ֔ים הָעֹשִׂ֖ים בֵּ֥ית יְהוָֽה׃ (יא) וְלַגֹּֽדְרִים֙ וּלְחֹצְבֵ֣י הָאֶ֔בֶן וְלִקְנ֤וֹת עֵצִים֙ וְאַבְנֵ֣י מַחְצֵ֔ב לְחַזֵּ֖ק אֶת־ בֶּ֣דֶק בֵּית־ יְהוָ֑ה וּלְכֹ֛ל אֲשֶׁר־ יֵצֵ֥א עַל־ הַבַּ֖יִת לְחָזְקָֽה׃ (יב) אַךְ֩ לֹ֨א יֵעָשֶׂ֜ה בֵּ֣ית יְהוָ֗ה סִפּ֥וֹת כֶּ֙סֶף֙ מְזַמְּר֤וֹת מִזְרָקוֹת֙ חֲצֹ֣צְר֔וֹת כָּל־ כְּלִ֥י זָהָ֖ב וּכְלִי־ כָ֑סֶף מִן־ הַכֶּ֖סֶף הַמּוּבָ֥א בֵית־ יְהוָֽה׃ (יג) כִּֽי־ לְעֹשֵׂ֥י הַמְּלָאכָ֖ה יִתְּנֻ֑הוּ וְחִזְּקוּ־ ב֖וֹ אֶת־ בֵּ֥ית יְהוָֽה׃ (יד) וְלֹ֧א יְחַשְּׁב֣וּ אֶת־ הָאֲנָשִׁ֗ים אֲשֶׁ֨ר יִתְּנ֤וּ אֶת־ הַכֶּ֙סֶף֙ עַל־ יָדָ֔ם לָתֵ֖ת לְעֹשֵׂ֣י הַמְּלָאכָ֑ה כִּ֥י בֶאֱמֻנָ֖ה הֵ֥ם עֹשִֽׂים׃ (טו) כֶּ֤סֶף אָשָׁם֙ וְכֶ֣סֶף חַטָּא֔וֹת לֹ֥א יוּבָ֖א בֵּ֣ית יְהוָ֑ה לַכֹּהֲנִ֖ים יִהְיֽוּ׃ (טז) אָ֣ז יַעֲלֶ֗ה חֲזָאֵל֙ מֶ֣לֶךְ אֲרָ֔ם וַיִּלָּ֥חֶם עַל־ גַּ֖ת וַֽיִּלְכְּדָ֑הּ וַיָּ֤שֶׂם חֲזָאֵל֙ פָּנָ֔יו לַעֲל֖וֹת עַל־ יְרוּשָׁלִָֽם׃ (יז) וַיִּקַּ֞ח יְהוֹאָ֣שׁ מֶֽלֶךְ־ יְהוּדָ֗ה אֵ֣ת כָּל־ הַקֳּדָשִׁ֡ים אֲשֶׁר־ הִקְדִּ֣ישׁוּ יְהוֹשָׁפָ֣ט וִיהוֹרָם֩ וַאֲחַזְיָ֨הוּ אֲבֹתָ֜יו מַלְכֵ֤י יְהוּדָה֙ וְאֶת־ קֳדָשָׁ֔יו וְאֵ֣ת כָּל־ הַזָּהָ֗ב הַנִּמְצָ֛א בְּאֹצְר֥וֹת בֵּית־ יְהוָ֖ה וּבֵ֣ית הַמֶּ֑לֶךְ וַיִּשְׁלַ֗ח לַֽחֲזָאֵל֙ מֶ֣לֶךְ אֲרָ֔ם וַיַּ֖עַל מֵעַ֥ל יְרוּשָׁלִָֽם׃ (יח) וְיֶ֛תֶר דִּבְרֵ֥י יוֹאָ֖שׁ וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (יט) וַיָּקֻ֥מוּ עֲבָדָ֖יו וַיִּקְשְׁרֽוּ־ קָ֑שֶׁר וַיַּכּוּ֙ אֶת־ יוֹאָ֔שׁ בֵּ֥ית מִלֹּ֖א הַיּוֹרֵ֥ד סִלָּֽא׃ (כ) וְיוֹזָבָ֣ד בֶּן־ שִׁ֠מְעָת וִיהוֹזָבָ֨ד בֶּן־ שֹׁמֵ֤ר ׀ עֲבָדָיו֙ הִכֻּ֣הוּ וַיָּמֹ֔ת וַיִּקְבְּר֥וּ אֹת֛וֹ עִם־ אֲבֹתָ֖יו בְּעִ֣יר דָּוִ֑ד וַיִּמְלֹ֛ךְ אֲמַצְיָ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ (כא) בִּשְׁנַ֨ת עֶשְׂרִ֤ים וְשָׁלֹשׁ֙ שָׁנָ֔ה לְיוֹאָ֥שׁ בֶּן־ אֲחַזְיָ֖הוּ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ יְהוֹאָחָ֨ז בֶּן־ יֵה֤וּא עַל־ יִשְׂרָאֵל֙ בְּשֹׁ֣מְר֔וֹן שְׁבַ֥ע עֶשְׂרֵ֖ה שָׁנָֽה׃ (כב) וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה וַ֠יֵּלֶךְ אַחַ֨ר חַטֹּ֜את יָרָבְעָ֧ם בֶּן־ נְבָ֛ט אֲשֶׁר־ הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל לֹא־ סָ֥ר מִמֶּֽנָּה׃ (כג) וַיִּֽחַר־ אַ֥ף יְהוָ֖ה בְּיִשְׂרָאֵ֑ל וַֽיִּתְּנֵ֞ם בְּיַ֣ד ׀ חֲזָאֵ֣ל מֶֽלֶךְ־ אֲרָ֗ם וּבְיַ֛ד בֶּן־ הֲדַ֥ד בֶּן־ חֲזָאֵ֖ל כָּל־ הַיָּמִֽים׃ (כד) וַיְחַ֥ל יְהוֹאָחָ֖ז אֶת־ פְּנֵ֣י יְהוָ֑ה וַיִּשְׁמַ֤ע אֵלָיו֙ יְהוָ֔ה כִּ֤י רָאָה֙ אֶת־ לַ֣חַץ יִשְׂרָאֵ֔ל כִּֽי־ לָחַ֥ץ אֹתָ֖ם מֶ֥לֶךְ אֲרָֽם׃ (כה) וַיִּתֵּ֨ן יְהוָ֤ה לְיִשְׂרָאֵל֙ מוֹשִׁ֔יעַ וַיֵּ֣צְא֔וּ מִתַּ֖חַת יַד־ אֲרָ֑ם וַיֵּשְׁב֧וּ בְנֵֽי־ יִשְׂרָאֵ֛ל בְּאָהֳלֵיהֶ֖ם כִּתְמ֥וֹל שִׁלְשֽׁוֹם׃ (כו) אַ֠ךְ לֹֽא־ סָ֜רוּ מֵחַטֹּ֧אות בֵּית־ יָרָבְעָ֛ם אֲשֶׁר־ (החטי) הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל בָּ֣הּ הָלָ֑ךְ וְגַם֙ הָאֲשֵׁרָ֔ה עָמְדָ֖ה בְּשֹׁמְרֽוֹן׃ (כז) כִּ֣י לֹא֩ הִשְׁאִ֨יר לִיהוֹאָחָ֜ז עָ֗ם כִּ֣י אִם־ חֲמִשִּׁ֤ים פָּֽרָשִׁים֙ וַעֲשָׂ֣רָה רֶ֔כֶב וַעֲשֶׂ֥רֶת אֲלָפִ֖ים רַגְלִ֑י כִּ֤י אִבְּדָם֙ מֶ֣לֶךְ אֲרָ֔ם וַיְשִׂמֵ֥ם כֶּֽעָפָ֖ר לָדֻֽשׁ׃ (כח) וְיֶ֨תֶר דִּבְרֵ֧י יְהוֹאָחָ֛ז וְכָל־ אֲשֶׁ֥ר עָשָׂ֖ה וּגְבוּרָת֑וֹ הֲלוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כט) וַיִּשְׁכַּ֤ב יְהֽוֹאָחָז֙ עִם־ אֲבֹתָ֔יו וַֽיִּקְבְּרֻ֖הוּ בְּשֹׁמְר֑וֹן וַיִּמְלֹ֛ךְ יוֹאָ֥שׁ בְּנ֖וֹ תַּחְתָּֽיו׃ (ל) בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָשֶׁ֙בַע֙ שָׁנָ֔ה לְיוֹאָ֖שׁ מֶ֣לֶךְ יְהוּדָ֑ה מָ֠לַךְ יְהוֹאָ֨שׁ בֶּן־ יְהוֹאָחָ֤ז עַל־ יִשְׂרָאֵל֙ בְּשֹׁ֣מְר֔וֹן שֵׁ֥שׁ עֶשְׂרֵ֖ה שָׁנָֽה׃ (לא) וַיַּֽעֲשֶׂ֥ה הָרַ֖ע בְּעֵינֵ֣י יְהוָ֑ה לֹ֣א סָ֗ר מִכָּל־ חַטֹּ֞אות יָרָבְעָ֧ם בֶּן־ נְבָ֛ט אֲשֶׁר־ הֶחֱטִ֥יא אֶת־ יִשְׂרָאֵ֖ל בָּ֥הּ הָלָֽךְ׃ (לב) וְיֶ֨תֶר דִּבְרֵ֤י יוֹאָשׁ֙ וְכָל־ אֲשֶׁ֣ר עָשָׂ֔ה וּגְב֣וּרָת֔וֹ אֲשֶׁ֣ר נִלְחַ֔ם עִ֖ם אֲמַצְיָ֣ה מֶֽלֶךְ־ יְהוּדָ֑ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (לג) וַיִּשְׁכַּ֤ב יוֹאָשׁ֙ עִם־ אֲבֹתָ֔יו וְיָרָבְעָ֖ם יָשַׁ֣ב עַל־ כִּסְא֑וֹ וַיִּקָּבֵ֤ר יוֹאָשׁ֙ בְּשֹׁ֣מְר֔וֹן עִ֖ם מַלְכֵ֥י יִשְׂרָאֵֽל׃ (לד) וֶֽאֱלִישָׁע֙ חָלָ֣ה אֶת־ חָלְי֔וֹ אֲשֶׁ֥ר יָמ֖וּת בּ֑וֹ וַיֵּ֨רֶד אֵלָ֜יו יוֹאָ֣שׁ מֶֽלֶךְ־ יִשְׂרָאֵ֗ל וַיֵּ֤בְךְּ עַל־ פָּנָיו֙ וַיֹּאמַ֔ר אָבִ֣י ׀ אָבִ֔י רֶ֥כֶב יִשְׂרָאֵ֖ל וּפָרָשָֽׁיו׃ (לה) וַיֹּ֤אמֶר לוֹ֙ אֱלִישָׁ֔ע קַ֖ח קֶ֣שֶׁת וְחִצִּ֑ים וַיִּקַּ֥ח אֵלָ֖יו קֶ֥שֶׁת וְחִצִּֽים׃ (לו) וַיֹּ֣אמֶר ׀ לְמֶ֣לֶךְ יִשְׂרָאֵ֗ל הַרְכֵּ֤ב יָֽדְךָ֙ עַל־ הַקֶּ֔שֶׁת וַיַּרְכֵּ֖ב יָד֑וֹ וַיָּ֧שֶׂם אֱלִישָׁ֛ע יָדָ֖יו עַל־ יְדֵ֥י הַמֶּֽלֶךְ׃ (לז) וַיֹּ֗אמֶר פְּתַ֧ח הַחַלּ֛וֹן קֵ֖דְמָה וַיִּפְתָּ֑ח וַיֹּ֤אמֶר אֱלִישָׁ֤ע יְרֵה֙ וַיּ֔וֹר וַיֹּ֗אמֶר חֵץ־ תְּשׁוּעָ֤ה לַֽיהוָה֙ וְחֵ֣ץ תְּשׁוּעָ֣ה בַֽאֲרָ֔ם וְהִכִּיתָ֧ אֶת־ אֲרָ֛ם בַּאֲפֵ֖ק עַד־ כַּלֵּֽה׃ (לח) וַיֹּ֛אמֶר קַ֥ח הַחִצִּ֖ים וַיִּקָּ֑ח וַיֹּ֤אמֶר לְמֶֽלֶךְ־ יִשְׂרָאֵל֙ הַךְ־ אַ֔רְצָה וַיַּ֥ךְ שָֽׁלֹשׁ־ פְּעָמִ֖ים וַֽיַּעֲמֹֽד׃ (לט) וַיִּקְצֹ֨ף עָלָ֜יו אִ֣ישׁ הָאֱלֹהִ֗ים וַיֹּ֙אמֶר֙ לְהַכּ֨וֹת חָמֵ֤שׁ אוֹ־ שֵׁשׁ֙ פְּעָמִ֔ים אָ֛ז הִכִּ֥יתָ אֶת־ אֲרָ֖ם עַד־ כַּלֵּ֑ה וְעַתָּ֕ה שָׁלֹ֥שׁ פְּעָמִ֖ים תַּכֶּ֥ה אֶת־ אֲרָֽם׃ (מ) וַיָּ֥מָת אֱלִישָׁ֖ע וַֽיִּקְבְּרֻ֑הוּ וּגְדוּדֵ֥י מוֹאָ֛ב יָבֹ֥אוּ בָאָ֖רֶץ בָּ֥א שָׁנָֽה׃ (מא) וַיְהִ֞י הֵ֣ם ׀ קֹבְרִ֣ים אִ֗ישׁ וְהִנֵּה֙ רָא֣וּ אֶֽת־ הַגְּד֔וּד וַיַּשְׁלִ֥יכוּ אֶת־ הָאִ֖ישׁ בְּקֶ֣בֶר אֱלִישָׁ֑ע וַיֵּ֜לֶךְ וַיִּגַּ֤ע הָאִישׁ֙ בְּעַצְמ֣וֹת אֱלִישָׁ֔ע וַיְחִ֖י וַיָּ֥קָם עַל־ רַגְלָֽיו׃ (מב) וַֽחֲזָאֵל֙ מֶ֣לֶךְ אֲרָ֔ם לָחַ֖ץ אֶת־ יִשְׂרָאֵ֑ל כֹּ֖ל יְמֵ֥י יְהוֹאָחָֽז׃
