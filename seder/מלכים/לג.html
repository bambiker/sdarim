מלכים סדר לג

לפי החלוקה לפרקים: מלכים ב כב, ב – כג, כד · 43 פסוקים

פתיחה באתר · ספר מלכים כולו ·

(א) וַיַּ֥עַשׂ הַיָּשָׁ֖ר בְּעֵינֵ֣י יְהוָ֑ה וַיֵּ֗לֶךְ בְּכָל־ דֶּ֙רֶךְ֙ דָּוִ֣ד אָבִ֔יו וְלֹא־ סָ֖ר יָמִ֥ין וּשְׂמֹֽאול׃ (ב) וַיְהִ֗י בִּשְׁמֹנֶ֤ה עֶשְׂרֵה֙ שָׁנָ֔ה לַמֶּ֖לֶךְ יֹאשִׁיָּ֑הוּ שָׁלַ֣ח הַ֠מֶּלֶךְ אֶת־ שָׁפָ֨ן בֶּן־ אֲצַלְיָ֤הוּ בֶן־ מְשֻׁלָּם֙ הַסֹּפֵ֔ר בֵּ֥ית יְהוָ֖ה לֵאמֹֽר׃ (ג) עֲלֵ֗ה אֶל־ חִלְקִיָּ֙הוּ֙ הַכֹּהֵ֣ן הַגָּד֔וֹל וְיַתֵּ֣ם אֶת־ הַכֶּ֔סֶף הַמּוּבָ֖א בֵּ֣ית יְהוָ֑ה אֲשֶׁ֥ר אָסְפ֛וּ שֹׁמְרֵ֥י הַסַּ֖ף מֵאֵ֥ת הָעָֽם׃ (ד) (ויתנה) וְיִתְּנֻ֗הוּ עַל־ יַד֙ עֹשֵׂ֣י הַמְּלָאכָ֔ה הַמֻּפְקָדִ֖ים (בבית) בֵּ֣ית יְהוָ֑ה וְיִתְּנ֣וּ אֹת֗וֹ לְעֹשֵׂ֤י הַמְּלָאכָה֙ אֲשֶׁר֙ בְּבֵ֣ית יְהוָ֔ה לְחַזֵּ֖ק בֶּ֥דֶק הַבָּֽיִת׃ (ה) לֶחָ֣רָשִׁ֔ים וְלַבֹּנִ֖ים וְלַגֹּֽדְרִ֑ים וְלִקְנ֤וֹת עֵצִים֙ וְאַבְנֵ֣י מַחְצֵ֔ב לְחַזֵּ֖ק אֶת־ הַבָּֽיִת׃ (ו) אַ֚ךְ לֹא־ יֵחָשֵׁ֣ב אִתָּ֔ם הַכֶּ֖סֶף הַנִּתָּ֣ן עַל־ יָדָ֑ם כִּ֥י בֶאֱמוּנָ֖ה הֵ֥ם עֹשִֽׂים׃ (ז) וַ֠יֹּאמֶר חִלְקִיָּ֜הוּ הַכֹּהֵ֤ן הַגָּדוֹל֙ עַל־ שָׁפָ֣ן הַסֹּפֵ֔ר סֵ֧פֶר הַתּוֹרָ֛ה מָצָ֖אתִי בְּבֵ֣ית יְהוָ֑ה וַיִּתֵּ֨ן חִלְקִיָּ֧ה אֶת־ הַסֵּ֛פֶר אֶל־ שָׁפָ֖ן וַיִּקְרָאֵֽהוּ׃ (ח) וַיָּבֹ֞א שָׁפָ֤ן הַסֹּפֵר֙ אֶל־ הַמֶּ֔לֶךְ וַיָּ֥שֶׁב אֶת־ הַמֶּ֖לֶךְ דָּבָ֑ר וַיֹּ֗אמֶר הִתִּ֤יכוּ עֲבָדֶ֙יךָ֙ אֶת־ הַכֶּ֙סֶף֙ הַנִּמְצָ֣א בַבַּ֔יִת וַֽיִּתְּנֻ֗הוּ עַל־ יַד֙ עֹשֵׂ֣י הַמְּלָאכָ֔ה הַמֻּפְקָדִ֖ים בֵּ֥ית יְהוָֽה׃ (ט) וַיַּגֵּ֞ד שָׁפָ֤ן הַסֹּפֵר֙ לַמֶּ֣לֶךְ לֵאמֹ֔ר סֵ֚פֶר נָ֣תַן לִ֔י חִלְקִיָּ֖ה הַכֹּהֵ֑ן וַיִּקְרָאֵ֥הוּ שָׁפָ֖ן לִפְנֵ֥י הַמֶּֽלֶךְ׃ (י) וַֽיְהִי֙ כִּשְׁמֹ֣עַ הַמֶּ֔לֶךְ אֶת־ דִּבְרֵ֖י סֵ֣פֶר הַתּוֹרָ֑ה וַיִּקְרַ֖ע אֶת־ בְּגָדָֽיו׃ (יא) וַיְצַ֣ו הַמֶּ֡לֶךְ אֶת־ חִלְקִיָּ֣ה הַכֹּהֵ֡ן וְאֶת־ אֲחִיקָ֣ם בֶּן־ שָׁ֠פָן וְאֶת־ עַכְבּ֨וֹר בֶּן־ מִיכָיָ֜ה וְאֵ֣ת ׀ שָׁפָ֣ן הַסֹּפֵ֗ר וְאֵ֛ת עֲשָׂיָ֥ה עֶֽבֶד־ הַמֶּ֖לֶךְ לֵאמֹֽר׃ (יב) לְכוּ֩ דִרְשׁ֨וּ אֶת־ יְהוָ֜ה בַּעֲדִ֣י וּבְעַד־ הָעָ֗ם וּבְעַד֙ כָּל־ יְהוּדָ֔ה עַל־ דִּבְרֵ֛י הַסֵּ֥פֶר הַנִּמְצָ֖א הַזֶּ֑ה כִּֽי־ גְדוֹלָ֞ה חֲמַ֣ת יְהוָ֗ה אֲשֶׁר־ הִיא֙ נִצְּתָ֣ה בָ֔נוּ עַל֩ אֲשֶׁ֨ר לֹֽא־ שָׁמְע֜וּ אֲבֹתֵ֗ינוּ עַל־ דִּבְרֵי֙ הַסֵּ֣פֶר הַזֶּ֔ה לַעֲשׂ֖וֹת כְּכָל־ הַכָּת֥וּב עָלֵֽינוּ׃ (יג) וַיֵּ֣לֶךְ חִלְקִיָּ֣הוּ הַ֠כֹּהֵן וַאֲחִיקָ֨ם וְעַכְבּ֜וֹר וְשָׁפָ֣ן וַעֲשָׂיָ֗ה אֶל־ חֻלְדָּ֨ה הַנְּבִיאָ֜ה אֵ֣שֶׁת ׀ שַׁלֻּ֣ם בֶּן־ תִּקְוָ֗ה בֶּן־ חַרְחַס֙ שֹׁמֵ֣ר הַבְּגָדִ֔ים וְהִ֛יא יֹשֶׁ֥בֶת בִּירוּשָׁלִַ֖ם בַּמִּשְׁנֶ֑ה וַֽיְדַבְּר֖וּ אֵלֶֽיהָ׃ (יד) וַתֹּ֣אמֶר אֲלֵיהֶ֔ם כֹּֽה־ אָמַ֥ר יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל אִמְר֣וּ לָאִ֔ישׁ אֲשֶׁר־ שָׁלַ֥ח אֶתְכֶ֖ם אֵלָֽי׃ (טו) כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֨י מֵבִ֥יא רָעָ֛ה אֶל־ הַמָּק֥וֹם הַזֶּ֖ה וְעַל־ יֹֽשְׁבָ֑יו אֵ֚ת כָּל־ דִּבְרֵ֣י הַסֵּ֔פֶר אֲשֶׁ֥ר קָרָ֖א מֶ֥לֶךְ יְהוּדָֽה׃ (טז) תַּ֣חַת ׀ אֲשֶׁ֣ר עֲזָב֗וּנִי וַֽיְקַטְּרוּ֙ לֵאלֹהִ֣ים אֲחֵרִ֔ים לְמַ֙עַן֙ הַכְעִיסֵ֔נִי בְּכֹ֖ל מַעֲשֵׂ֣ה יְדֵיהֶ֑ם וְנִצְּתָ֧ה חֲמָתִ֛י בַּמָּק֥וֹם הַזֶּ֖ה וְלֹ֥א תִכְבֶּֽה׃ (יז) וְאֶל־ מֶ֣לֶךְ יְהוּדָ֗ה הַשֹּׁלֵ֤חַ אֶתְכֶם֙ לִדְרֹ֣שׁ אֶת־ יְהוָ֔ה כֹּ֥ה תֹאמְר֖וּ אֵלָ֑יו כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל הַדְּבָרִ֖ים אֲשֶׁ֥ר שָׁמָֽעְתָּ׃ (יח) יַ֠עַן רַךְ־ לְבָ֨בְךָ֜ וַתִּכָּנַ֣ע ׀ מִפְּנֵ֣י יְהוָ֗ה בְּֽשָׁמְעֲךָ֡ אֲשֶׁ֣ר דִּבַּרְתִּי֩ עַל־ הַמָּק֨וֹם הַזֶּ֜ה וְעַל־ יֹשְׁבָ֗יו לִהְי֤וֹת לְשַׁמָּה֙ וְלִקְלָלָ֔ה וַתִּקְרַע֙ אֶת־ בְּגָדֶ֔יךָ וַתִּבְכֶּ֖ה לְפָנָ֑י וְגַ֧ם אָנֹכִ֛י שָׁמַ֖עְתִּי נְאֻם־ יְהוָֽה׃ (יט) לָכֵן֩ הִנְנִ֨י אֹֽסִפְךָ֜ עַל־ אֲבֹתֶ֗יךָ וְנֶאֱסַפְתָּ֣ אֶל־ קִבְרֹתֶיךָ֮ בְּשָׁלוֹם֒ וְלֹא־ תִרְאֶ֣ינָה עֵינֶ֔יךָ בְּכֹל֙ הָֽרָעָ֔ה אֲשֶׁר־ אֲנִ֥י מֵבִ֖יא עַל־ הַמָּק֣וֹם הַזֶּ֑ה וַיָּשִׁ֥יבוּ אֶת־ הַמֶּ֖לֶךְ דָּבָֽר׃ (כ) וַיִּשְׁלַ֖ח הַמֶּ֑לֶךְ וַיַּאַסְפ֣וּ אֵלָ֔יו כָּל־ זִקְנֵ֥י יְהוּדָ֖ה וִירוּשָׁלִָֽם׃ (כא) וַיַּ֣עַל הַמֶּ֣לֶךְ בֵּית־ יְהוָ֡ה וְכָל־ אִ֣ישׁ יְהוּדָה֩ וְכָל־ יֹשְׁבֵ֨י יְרוּשָׁלִַ֜ם אִתּ֗וֹ וְהַכֹּֽהֲנִים֙ וְהַנְּבִיאִ֔ים וְכָל־ הָעָ֖ם לְמִקָּטֹ֣ן וְעַד־ גָּד֑וֹל וַיִּקְרָ֣א בְאָזְנֵיהֶ֗ם אֶת־ כָּל־ דִּבְרֵי֙ סֵ֣פֶר הַבְּרִ֔ית הַנִּמְצָ֖א בְּבֵ֥ית יְהוָֽה׃ (כב) וַיַּעֲמֹ֣ד הַ֠מֶּלֶךְ עַֽל־ הָ֨עַמּ֜וּד וַיִּכְרֹ֥ת אֶֽת־ הַבְּרִ֣ית ׀ לִפְנֵ֣י יְהוָ֗ה לָלֶ֜כֶת אַחַ֤ר יְהוָה֙ וְלִשְׁמֹ֨ר מִצְוֺתָ֜יו וְאֶת־ עֵדְוֺתָ֤יו וְאֶת־ חֻקֹּתָיו֙ בְּכָל־ לֵ֣ב וּבְכָל־ נֶ֔פֶשׁ לְהָקִ֗ים אֶת־ דִּבְרֵי֙ הַבְּרִ֣ית הַזֹּ֔את הַכְּתֻבִ֖ים עַל־ הַסֵּ֣פֶר הַזֶּ֑ה וַיַּעֲמֹ֥ד כָּל־ הָעָ֖ם בַּבְּרִֽית׃ (כג) וַיְצַ֣ו הַמֶּ֡לֶךְ אֶת־ חִלְקִיָּהוּ֩ הַכֹּהֵ֨ן הַגָּד֜וֹל וְאֶת־ כֹּהֲנֵ֣י הַמִּשְׁנֶה֮ וְאֶת־ שֹׁמְרֵ֣י הַסַּף֒ לְהוֹצִיא֙ מֵהֵיכַ֣ל יְהוָ֔ה אֵ֣ת כָּל־ הַכֵּלִ֗ים הָֽעֲשׂוּיִם֙ לַבַּ֣עַל וְלָֽאֲשֵׁרָ֔ה וּלְכֹ֖ל צְבָ֣א הַשָּׁמָ֑יִם וַֽיִּשְׂרְפֵ֞ם מִח֤וּץ לִירוּשָׁלִַ֙ם֙ בְּשַׁדְמ֣וֹת קִדְר֔וֹן וְנָשָׂ֥א אֶת־ עֲפָרָ֖ם בֵּֽית־ אֵֽל׃ (כד) וְהִשְׁבִּ֣ית אֶת־ הַכְּמָרִ֗ים אֲשֶׁ֤ר נָֽתְנוּ֙ מַלְכֵ֣י יְהוּדָ֔ה וַיְקַטֵּ֤ר בַּבָּמוֹת֙ בְּעָרֵ֣י יְהוּדָ֔ה וּמְסִבֵּ֖י יְרוּשָׁלִָ֑ם וְאֶת־ הַֽמְקַטְּרִ֣ים לַבַּ֗עַל לַשֶּׁ֤מֶשׁ וְלַיָּרֵ֙חַ֙ וְלַמַּזָּל֔וֹת וּלְכֹ֖ל צְבָ֥א הַשָּׁמָֽיִם׃ (כה) וַיֹּצֵ֣א אֶת־ הָאֲשֵׁרָה֩ מִבֵּ֨ית יְהוָ֜ה מִח֤וּץ לִירוּשָׁלִַ֙ם֙ אֶל־ נַ֣חַל קִדְר֔וֹן וַיִּשְׂרֹ֥ף אֹתָ֛הּ בְּנַ֥חַל קִדְר֖וֹן וַיָּ֣דֶק לְעָפָ֑ר וַיַּשְׁלֵךְ֙ אֶת־ עֲפָרָ֔הּ עַל־ קֶ֖בֶר בְּנֵ֥י הָעָֽם׃ (כו) וַיִּתֹּץ֙ אֶת־ בָּתֵּ֣י הַקְּדֵשִׁ֔ים אֲשֶׁ֖ר בְּבֵ֣ית יְהוָ֑ה אֲשֶׁ֣ר הַנָּשִׁ֗ים אֹרְג֥וֹת שָׁ֛ם בָּתִּ֖ים לָאֲשֵׁרָֽה׃ (כז) וַיָּבֵ֤א אֶת־ כָּל־ הַכֹּֽהֲנִים֙ מֵעָרֵ֣י יְהוּדָ֔ה וַיְטַמֵּ֣א אֶת־ הַבָּמ֗וֹת אֲשֶׁ֤ר קִטְּרוּ־ שָׁ֙מָּה֙ הַכֹּ֣הֲנִ֔ים מִגֶּ֖בַע עַד־ בְּאֵ֣ר שָׁ֑בַע וְנָתַ֞ץ אֶת־ בָּמ֣וֹת הַשְּׁעָרִ֗ים אֲשֶׁר־ פֶּ֜תַח שַׁ֤עַר יְהוֹשֻׁ֙עַ֙ שַׂר־ הָעִ֔יר אֲשֶֽׁר־ עַל־ שְׂמֹ֥אול אִ֖ישׁ בְּשַׁ֥עַר הָעִֽיר׃ (כח) אַ֗ךְ לֹ֤א יַֽעֲלוּ֙ כֹּהֲנֵ֣י הַבָּמ֔וֹת אֶל־ מִזְבַּ֥ח יְהוָ֖ה בִּירוּשָׁלִָ֑ם כִּ֛י אִם־ אָכְל֥וּ מַצּ֖וֹת בְּת֥וֹךְ אֲחֵיהֶֽם׃ (כט) וְטִמֵּ֣א אֶת־ הַתֹּ֔פֶת אֲשֶׁ֖ר בְּגֵ֣י (בני)־ בֶן־ הִנֹּ֑ם לְבִלְתִּ֗י לְהַעֲבִ֨יר אִ֜ישׁ אֶת־ בְּנ֧וֹ וְאֶת־ בִּתּ֛וֹ בָּאֵ֖שׁ לַמֹּֽלֶךְ׃ (ל) וַיַּשְׁבֵּ֣ת אֶת־ הַסּוּסִ֗ים אֲשֶׁ֣ר נָתְנוּ֩ מַלְכֵ֨י יְהוּדָ֤ה לַשֶּׁ֙מֶשׁ֙ מִבֹּ֣א בֵית־ יְהוָ֔ה אֶל־ לִשְׁכַּת֙ נְתַן־ מֶ֣לֶךְ הַסָּרִ֔יס אֲשֶׁ֖ר בַּפַּרְוָרִ֑ים וְאֶת־ מַרְכְּב֥וֹת הַשֶּׁ֖מֶשׁ שָׂרַ֥ף בָּאֵֽשׁ׃ (לא) וְאֶֽת־ הַֽמִּזְבְּח֡וֹת אֲשֶׁ֣ר עַל־ הַגָּג֩ עֲלִיַּ֨ת אָחָ֜ז אֲשֶׁר־ עָשׂ֣וּ ׀ מַלְכֵ֣י יְהוּדָ֗ה וְאֶת־ הַֽמִּזְבְּחוֹת֙ אֲשֶׁר־ עָשָׂ֣ה מְנַשֶּׁ֔ה בִּשְׁתֵּ֛י חַצְר֥וֹת בֵּית־ יְהוָ֖ה נָתַ֣ץ הַמֶּ֑לֶךְ וַיָּ֣רָץ מִשָּׁ֔ם וְהִשְׁלִ֥יךְ אֶת־ עֲפָרָ֖ם אֶל־ נַ֥חַל קִדְרֽוֹן׃ (לב) וְֽאֶת־ הַבָּמ֞וֹת אֲשֶׁ֣ר ׀ עַל־ פְּנֵ֣י יְרוּשָׁלִַ֗ם אֲשֶׁר֮ מִימִ֣ין לְהַר־ הַמַּשְׁחִית֒ אֲשֶׁ֣ר בָּ֠נָה שְׁלֹמֹ֨ה מֶֽלֶךְ־ יִשְׂרָאֵ֜ל לְעַשְׁתֹּ֣רֶת ׀ שִׁקֻּ֣ץ צִידֹנִ֗ים וְלִכְמוֹשׁ֙ שִׁקֻּ֣ץ מוֹאָ֔ב וּלְמִלְכֹּ֖ם תּוֹעֲבַ֣ת בְּנֵֽי־ עַמּ֑וֹן טִמֵּ֖א הַמֶּֽלֶךְ׃ (לג) וְשִׁבַּר֙ אֶת־ הַמַּצֵּב֔וֹת וַיִּכְרֹ֖ת אֶת־ הָאֲשֵׁרִ֑ים וַיְמַלֵּ֥א אֶת־ מְקוֹמָ֖ם עַצְמ֥וֹת אָדָֽם׃ (לד) וְגַ֨ם אֶת־ הַמִּזְבֵּ֜חַ אֲשֶׁ֣ר בְּבֵֽית־ אֵ֗ל הַבָּמָה֙ אֲשֶׁ֨ר עָשָׂ֜ה יָרָבְעָ֤ם בֶּן־ נְבָט֙ אֲשֶׁ֣ר הֶחֱטִ֣יא אֶת־ יִשְׂרָאֵ֔ל גַּ֣ם אֶת־ הַמִּזְבֵּ֧חַ הַה֛וּא וְאֶת־ הַבָּמָ֖ה נָתָ֑ץ וַיִּשְׂרֹ֧ף אֶת־ הַבָּמָ֛ה הֵדַ֥ק לְעָפָ֖ר וְשָׂרַ֥ף אֲשֵׁרָֽה׃ (לה) וַיִּ֣פֶן יֹאשִׁיָּ֗הוּ וַיַּ֨רְא אֶת־ הַקְּבָרִ֤ים אֲשֶׁר־ שָׁם֙ בָּהָ֔ר וַיִּשְׁלַ֗ח וַיִּקַּ֤ח אֶת־ הָֽעֲצָמוֹת֙ מִן־ הַקְּבָרִ֔ים וַיִּשְׂרֹ֥ף עַל־ הַמִּזְבֵּ֖חַ וַֽיְטַמְּאֵ֑הוּ כִּדְבַ֣ר יְהוָ֗ה אֲשֶׁ֤ר קָרָא֙ אִ֣ישׁ הָאֱלֹהִ֔ים אֲשֶׁ֣ר קָרָ֔א אֶת־ הַדְּבָרִ֖ים הָאֵֽלֶּה׃ (לו) וַיֹּ֕אמֶר מָ֚ה הַצִּיּ֣וּן הַלָּ֔ז אֲשֶׁ֖ר אֲנִ֣י רֹאֶ֑ה וַיֹּאמְר֨וּ אֵלָ֜יו אַנְשֵׁ֣י הָעִ֗יר הַקֶּ֤בֶר אִישׁ־ הָֽאֱלֹהִים֙ אֲשֶׁר־ בָּ֣א מִֽיהוּדָ֔ה וַיִּקְרָ֗א אֶת־ הַדְּבָרִ֤ים הָאֵ֙לֶּה֙ אֲשֶׁ֣ר עָשִׂ֔יתָ עַ֖ל הַמִּזְבַּ֥ח בֵּֽית־ אֵֽל׃ (לז) וַיֹּ֙אמֶר֙ הַנִּ֣יחוּ ל֔וֹ אִ֖ישׁ אַל־ יָנַ֣ע עַצְמֹתָ֑יו וַֽיְמַלְּטוּ֙ עַצְמֹתָ֔יו אֵ֚ת עַצְמ֣וֹת הַנָּבִ֔יא אֲשֶׁר־ בָּ֖א מִשֹּׁמְרֽוֹן׃ (לח) וְגַם֩ אֶת־ כָּל־ בָּתֵּ֨י הַבָּמ֜וֹת אֲשֶׁ֣ר ׀ בְּעָרֵ֣י שֹׁמְר֗וֹן אֲשֶׁ֨ר עָשׂ֜וּ מַלְכֵ֤י יִשְׂרָאֵל֙ לְהַכְעִ֔יס הֵסִ֖יר יֹֽאשִׁיָּ֑הוּ וַיַּ֣עַשׂ לָהֶ֔ם כְּכָל־ הַֽמַּעֲשִׂ֔ים אֲשֶׁ֥ר עָשָׂ֖ה בְּבֵֽית־ אֵֽל׃ (לט) וַ֠יִּזְבַּח אֶת־ כָּל־ כֹּהֲנֵ֨י הַבָּמ֤וֹת אֲשֶׁר־ שָׁם֙ עַל־ הַֽמִּזְבְּח֔וֹת וַיִּשְׂרֹ֛ף אֶת־ עַצְמ֥וֹת אָדָ֖ם עֲלֵיהֶ֑ם וַיָּ֖שָׁב יְרוּשָׁלִָֽם׃ (מ) וַיְצַ֤ו הַמֶּ֙לֶךְ֙ אֶת־ כָּל־ הָעָ֣ם לֵאמֹ֔ר עֲשׂ֣וּ פֶ֔סַח לַֽיהוָ֖ה אֱלֹֽהֵיכֶ֑ם כַּכָּת֕וּב עַ֛ל סֵ֥פֶר הַבְּרִ֖ית הַזֶּֽה׃ (מא) כִּ֣י לֹ֤א נַֽעֲשָׂה֙ כַּפֶּ֣סַח הַזֶּ֔ה מִימֵי֙ הַשֹּׁ֣פְטִ֔ים אֲשֶׁ֥ר שָׁפְט֖וּ אֶת־ יִשְׂרָאֵ֑ל וְכֹ֗ל יְמֵ֛י מַלְכֵ֥י יִשְׂרָאֵ֖ל וּמַלְכֵ֥י יְהוּדָֽה׃ (מב) כִּ֗י אִם־ בִּשְׁמֹנֶ֤ה עֶשְׂרֵה֙ שָׁנָ֔ה לַמֶּ֖לֶךְ יֹֽאשִׁיָּ֑הוּ נַעֲשָׂ֞ה הַפֶּ֧סַח הַזֶּ֛ה לַיהוָ֖ה בִּירוּשָׁלִָֽם׃ (מג) וְגַ֣ם אֶת־ הָאֹב֣וֹת וְאֶת־ הַ֠יִּדְּעֹנִים וְאֶת־ הַתְּרָפִ֨ים וְאֶת־ הַגִּלֻּלִ֜ים וְאֵ֣ת כָּל־ הַשִּׁקֻּצִ֗ים אֲשֶׁ֤ר נִרְאוּ֙ בְּאֶ֤רֶץ יְהוּדָה֙ וּבִיר֣וּשָׁלִַ֔ם בִּעֵ֖ר יֹֽאשִׁיָּ֑הוּ לְ֠מַעַן הָקִ֞ים אֶת־ דִּבְרֵ֤י הַתּוֹרָה֙ הַכְּתֻבִ֣ים עַל־ הַסֵּ֔פֶר אֲשֶׁ֥ר מָצָ֛א חִלְקִיָּ֥הוּ הַכֹּהֵ֖ן בֵּ֥ית יְהוָֽה׃
