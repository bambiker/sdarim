מלכים סדר כה

פתיחה באתר · ספר מלכים כולו ·

(א) וַיֵּ֣לֶךְ מִשָּׁ֡ם וַיִּמְצָ֣א אֶת־ יְהוֹנָדָב֩ בֶּן־ רֵכָ֨ב לִקְרָאת֜וֹ וַֽיְבָרְכֵ֗הוּ וַיֹּ֨אמֶר אֵלָ֜יו הֲיֵ֧שׁ אֶת־ לְבָבְךָ֣ יָשָׁ֗ר כַּאֲשֶׁ֤ר לְבָבִי֙ עִם־ לְבָבֶ֔ךָ וַיֹּ֨אמֶר יְהוֹנָדָ֥ב יֵ֛שׁ וָיֵ֖שׁ תְּנָ֣ה אֶת־ יָדֶ֑ךָ וַיִּתֵּ֣ן יָד֔וֹ וַיַּעֲלֵ֥הוּ אֵלָ֖יו אֶל־ הַמֶּרְכָּבָֽה׃ (ב) וַיֹּ֙אמֶר֙ לְכָ֣ה אִתִּ֔י וּרְאֵ֖ה בְּקִנְאָתִ֣י לַיהוָ֑ה וַיַּרְכִּ֥בוּ אֹת֖וֹ בְּרִכְבּֽוֹ׃ (ג) וַיָּבֹא֙ שֹֽׁמְר֔וֹן וַ֠יַּךְ אֶת־ כָּל־ הַנִּשְׁאָרִ֧ים לְאַחְאָ֛ב בְּשֹׁמְר֖וֹן עַד־ הִשְׁמִיד֑וֹ כִּדְבַ֣ר יְהוָ֔ה אֲשֶׁ֥ר דִּבֶּ֖ר אֶל־ אֵלִיָּֽהוּ׃ (ד) וַיִּקְבֹּ֤ץ יֵהוּא֙ אֶת־ כָּל־ הָעָ֔ם וַיֹּ֣אמֶר אֲלֵהֶ֔ם אַחְאָ֕ב עָבַ֥ד אֶת־ הַבַּ֖עַל מְעָ֑ט יֵה֖וּא יַעַבְדֶ֥נּוּ הַרְבֵּֽה׃ (ה) וְעַתָּ֣ה כָל־ נְבִיאֵ֣י הַבַּ֡עַל כָּל־ עֹבְדָ֣יו וְכָל־ כֹּהֲנָיו֩ קִרְא֨וּ אֵלַ֜י אִ֣ישׁ אַל־ יִפָּקֵ֗ד כִּי֩ זֶ֨בַח גָּד֥וֹל לִי֙ לַבַּ֔עַל כֹּ֥ל אֲשֶׁר־ יִפָּקֵ֖ד לֹ֣א יִֽחְיֶ֑ה וְיֵהוּא֙ עָשָׂ֣ה בְעָקְבָּ֔ה לְמַ֥עַן הַאֲבִ֖יד אֶת־ עֹבְדֵ֥י הַבָּֽעַל׃ (ו) וַיֹּ֣אמֶר יֵה֗וּא קַדְּשׁ֧וּ עֲצָרָ֛ה לַבַּ֖עַל וַיִּקְרָֽאוּ׃ (ז) וַיִּשְׁלַ֤ח יֵהוּא֙ בְּכָל־ יִשְׂרָאֵ֔ל וַיָּבֹ֙אוּ֙ כָּל־ עֹבְדֵ֣י הַבַּ֔עַל וְלֹֽא־ נִשְׁאַ֥ר אִ֖ישׁ אֲשֶׁ֣ר לֹֽא־ בָ֑א וַיָּבֹ֙אוּ֙ בֵּ֣ית הַבַּ֔עַל וַיִּמָּלֵ֥א בֵית־ הַבַּ֖עַל פֶּ֥ה לָפֶֽה׃ (ח) וַיֹּ֗אמֶר לַֽאֲשֶׁר֙ עַל־ הַמֶּלְתָּחָ֔ה הוֹצֵ֣א לְב֔וּשׁ לְכֹ֖ל עֹבְדֵ֣י הַבָּ֑עַל וַיֹּצֵ֥א לָהֶ֖ם הַמַּלְבּֽוּשׁ׃ (ט) וַיָּבֹ֥א יֵה֛וּא וִיהוֹנָדָ֥ב בֶּן־ רֵכָ֖ב בֵּ֣ית הַבָּ֑עַל וַיֹּ֜אמֶר לְעֹבְדֵ֣י הַבַּ֗עַל חַפְּשׂ֤וּ וּרְאוּ֙ פֶּן־ יֶשׁ־ פֹּ֤ה עִמָּכֶם֙ מֵעַבְדֵ֣י יְהוָ֔ה כִּ֛י אִם־ עֹבְדֵ֥י הַבַּ֖עַל לְבַדָּֽם׃ (י) וַיָּבֹ֕אוּ לַעֲשׂ֖וֹת זְבָחִ֣ים וְעֹל֑וֹת וְיֵה֞וּא שָׂם־ ל֤וֹ בַחוּץ֙ שְׁמֹנִ֣ים אִ֔ישׁ וַיֹּ֗אמֶר הָאִ֤ישׁ אֲשֶׁר־ יִמָּלֵט֙ מִן־ הָאֲנָשִׁ֗ים אֲשֶׁ֤ר אֲנִי֙ מֵבִ֣יא עַל־ יְדֵיכֶ֔ם נַפְשׁ֖וֹ תַּ֥חַת נַפְשֽׁוֹ׃ (יא) וַיְהִ֞י כְּכַלֹּת֣וֹ ׀ לַעֲשׂ֣וֹת הָעֹלָ֗ה וַיֹּ֣אמֶר יֵ֠הוּא לָרָצִ֨ים וְלַשָּׁלִשִׁ֜ים בֹּ֤אוּ הַכּוּם֙ אִ֣ישׁ אַל־ יֵצֵ֔א וַיַּכּ֖וּם לְפִי־ חָ֑רֶב וַיַּשְׁלִ֗כוּ הָֽרָצִים֙ וְהַשָּׁ֣לִשִׁ֔ים וַיֵּלְכ֖וּ עַד־ עִ֥יר בֵּית־ הַבָּֽעַל׃ (יב) וַיֹּצִ֛אוּ אֶת־ מַצְּב֥וֹת בֵּית־ הַבַּ֖עַל וַֽיִּשְׂרְפֽוּהָ׃ (יג) וַֽיִּתְּצ֔וּ אֵ֖ת מַצְּבַ֣ת הַבָּ֑עַל וַֽיִּתְּצוּ֙ אֶת־ בֵּ֣ית הַבַּ֔עַל וַיְשִׂמֻ֥הוּ (למחראות) לְמֽוֹצָא֖וֹת עַד־ הַיּֽוֹם׃ (יד) וַיַּשְׁמֵ֥ד יֵה֛וּא אֶת־ הַבַּ֖עַל מִיִּשְׂרָאֵֽל׃ (טו) רַ֠ק חֲטָאֵ֞י יָרָבְעָ֤ם בֶּן־ נְבָט֙ אֲשֶׁ֣ר הֶחֱטִ֣יא אֶת־ יִשְׂרָאֵ֔ל לֹֽא־ סָ֥ר יֵה֖וּא מֵאַֽחֲרֵיהֶ֑ם עֶגְלֵי֙ הַזָּהָ֔ב אֲשֶׁ֥ר בֵּֽית־ אֵ֖ל וַאֲשֶׁ֥ר בְּדָֽן׃ (טז) וַיֹּ֨אמֶר יְהוָ֜ה אֶל־ יֵה֗וּא יַ֤עַן אֲשֶׁר־ הֱטִיבֹ֙תָ֙ לַעֲשׂ֤וֹת הַיָּשָׁר֙ בְּעֵינַ֔י כְּכֹל֙ אֲשֶׁ֣ר בִּלְבָבִ֔י עָשִׂ֖יתָ לְבֵ֣ית אַחְאָ֑ב בְּנֵ֣י רְבִעִ֔ים יֵשְׁב֥וּ לְךָ֖ עַל־ כִּסֵּ֥א יִשְׂרָאֵֽל׃ (יז) וְיֵה֗וּא לֹ֥א שָׁמַ֛ר לָלֶ֛כֶת בְּתֽוֹרַת־ יְהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בְּכָל־ לְבָב֑וֹ לֹ֣א סָ֗ר מֵעַל֙ חַטֹּ֣אות יָֽרָבְעָ֔ם אֲשֶׁ֥ר הֶחֱטִ֖יא אֶת־ יִשְׂרָאֵֽל׃ (יח) בַּיָּמִ֣ים הָהֵ֔ם הֵחֵ֣ל יְהוָ֔ה לְקַצּ֖וֹת בְּיִשְׂרָאֵ֑ל וַיַּכֵּ֥ם חֲזָאֵ֖ל בְּכָל־ גְּב֥וּל יִשְׂרָאֵֽל׃ (יט) מִן־ הַיַּרְדֵּן֙ מִזְרַ֣ח הַשֶּׁ֔מֶשׁ אֵ֚ת כָּל־ אֶ֣רֶץ הַגִּלְעָ֔ד הַגָּדִ֥י וְהָרֻאובֵנִ֖י וְהַֽמְנַשִּׁ֑י מֵעֲרֹעֵר֙ אֲשֶׁ֣ר עַל־ נַ֣חַל אַרְנֹ֔ן וְהַגִּלְעָ֖ד וְהַבָּשָֽׁן׃ (כ) וְיֶ֨תֶר דִּבְרֵ֥י יֵה֛וּא וְכָל־ אֲשֶׁ֥ר עָשָׂ֖ה וְכָל־ גְּבוּרָת֑וֹ הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יִשְׂרָאֵֽל׃ (כא) וַיִּשְׁכַּ֤ב יֵהוּא֙ עִם־ אֲבֹתָ֔יו וַיִּקְבְּר֥וּ אֹת֖וֹ בְּשֹׁמְר֑וֹן וַיִּמְלֹ֛ךְ יְהוֹאָחָ֥ז בְּנ֖וֹ תַּחְתָּֽיו׃ (כב) וְהַיָּמִ֗ים אֲשֶׁ֨ר מָלַ֤ךְ יֵהוּא֙ עַל־ יִשְׂרָאֵ֔ל עֶשְׂרִ֥ים וּשְׁמֹנֶֽה־ שָׁנָ֖ה בְּשֹׁמְרֽוֹן׃ (כג) וַֽעֲתַלְיָה֙ אֵ֣ם אֲחַזְיָ֔הוּ (וראתה) רָאֲתָ֖ה כִּ֣י מֵ֣ת בְּנָ֑הּ וַתָּ֙קָם֙ וַתְּאַבֵּ֔ד אֵ֖ת כָּל־ זֶ֥רַע הַמַּמְלָכָֽה׃ (כד) וַתִּקַּ֣ח יְהוֹשֶׁ֣בַע בַּת־ הַמֶּֽלֶךְ־ י֠וֹרָם אֲח֨וֹת אֲחַזְיָ֜הוּ אֶת־ יוֹאָ֣שׁ בֶּן־ אֲחַזְיָ֗ה וַתִּגְנֹ֤ב אֹתוֹ֙ מִתּ֤וֹךְ בְּנֵֽי־ הַמֶּ֙לֶךְ֙ (הממותתים) הַמּ֣וּמָתִ֔ים אֹת֥וֹ וְאֶת־ מֵינִקְתּ֖וֹ בַּחֲדַ֣ר הַמִּטּ֑וֹת וַיַּסְתִּ֧רוּ אֹת֛וֹ מִפְּנֵ֥י עֲתַלְיָ֖הוּ וְלֹ֥א הוּמָֽת׃ (כה) וַיְהִ֤י אִתָּהּ֙ בֵּ֣ית יְהוָ֔ה מִתְחַבֵּ֖א שֵׁ֣שׁ שָׁנִ֑ים וַעֲתַלְיָ֖ה מֹלֶ֥כֶת עַל־ הָאָֽרֶץ׃ (כו) וּבַשָּׁנָ֣ה הַ֠שְּׁבִיעִית שָׁלַ֨ח יְהוֹיָדָ֜ע וַיִּקַּ֣ח ׀ אֶת־ שָׂרֵ֣י (המאיות) הַמֵּא֗וֹת לַכָּרִי֙ וְלָ֣רָצִ֔ים וַיָּבֵ֥א אֹתָ֛ם אֵלָ֖יו בֵּ֣ית יְהוָ֑ה וַיִּכְרֹת֩ לָהֶ֨ם בְּרִ֜ית וַיַּשְׁבַּ֤ע אֹתָם֙ בְּבֵ֣ית יְהוָ֔ה וַיַּ֥רְא אֹתָ֖ם אֶת־ בֶּן־ הַמֶּֽלֶךְ׃ (כז) וַיְצַוֵּ֣ם לֵאמֹ֔ר זֶ֥ה הַדָּבָ֖ר אֲשֶׁ֣ר תַּעֲשׂ֑וּן הַשְּׁלִשִׁ֤ית מִכֶּם֙ בָּאֵ֣י הַשַּׁבָּ֔ת וְשֹׁ֣מְרֵ֔י מִשְׁמֶ֖רֶת בֵּ֥ית הַמֶּֽלֶךְ׃ (כח) וְהַשְּׁלִשִׁית֙ בְּשַׁ֣עַר ס֔וּר וְהַשְּׁלִשִׁ֥ית בַּשַּׁ֖עַר אַחַ֣ר הָרָצִ֑ים וּשְׁמַרְתֶּ֛ם אֶת־ מִשְׁמֶ֥רֶת הַבַּ֖יִת מַסָּֽח׃ (כט) וּשְׁתֵּ֤י הַיָּדוֹת֙ בָּכֶ֔ם כֹּ֖ל יֹצְאֵ֣י הַשַּׁבָּ֑ת וְשָֽׁמְר֛וּ אֶת־ מִשְׁמֶ֥רֶת בֵּית־ יְהוָ֖ה אֶל־ הַמֶּֽלֶךְ׃ (ל) וְהִקַּפְתֶּ֨ם עַל־ הַמֶּ֜לֶךְ סָבִ֗יב אִ֚ישׁ וְכֵלָ֣יו בְּיָד֔וֹ וְהַבָּ֥א אֶל־ הַשְּׂדֵר֖וֹת יוּמָ֑ת וִהְי֥וּ אֶת־ הַמֶּ֖לֶךְ בְּצֵאת֥וֹ וּבְבֹאֽוֹ׃ (לא) וַֽיַּעֲשׂ֞וּ שָׂרֵ֣י (המאיות) הַמֵּא֗וֹת כְּכֹ֣ל אֲשֶׁר־ צִוָּה֮ יְהוֹיָדָ֣ע הַכֹּהֵן֒ וַיִּקְחוּ֙ אִ֣ישׁ אֶת־ אֲנָשָׁ֔יו בָּאֵ֣י הַשַּׁבָּ֔ת עִ֖ם יֹצְאֵ֣י הַשַּׁבָּ֑ת וַיָּבֹ֖אוּ אֶל־ יְהוֹיָדָ֥ע הַכֹּהֵֽן׃ (לב) וַיִּתֵּ֨ן הַכֹּהֵ֜ן לְשָׂרֵ֣י (המאיות) הַמֵּא֗וֹת אֶֽת־ הַחֲנִית֙ וְאֶת־ הַשְּׁלָטִ֔ים אֲשֶׁ֖ר לַמֶּ֣לֶךְ דָּוִ֑ד אֲשֶׁ֖ר בְּבֵ֥ית יְהוָֽה׃ (לג) וַיַּעַמְד֨וּ הָרָצִ֜ים אִ֣ישׁ ׀ וְכֵלָ֣יו בְּיָד֗וֹ מִכֶּ֨תֶף הַבַּ֤יִת הַיְמָנִית֙ עַד־ כֶּ֤תֶף הַבַּ֙יִת֙ הַשְּׂמָאלִ֔ית לַמִּזְבֵּ֖חַ וְלַבָּ֑יִת עַל־ הַמֶּ֖לֶךְ סָבִֽיב׃ (לד) וַיּוֹצִ֣א אֶת־ בֶּן־ הַמֶּ֗לֶךְ וַיִּתֵּ֤ן עָלָיו֙ אֶת־ הַנֵּ֙זֶר֙ וְאֶת־ הָ֣עֵד֔וּת וַיַּמְלִ֥כוּ אֹת֖וֹ וַיִּמְשָׁחֻ֑הוּ וַיַּכּוּ־ כָ֔ף וַיֹּאמְר֖וּ יְחִ֥י הַמֶּֽלֶךְ׃ (לה) וַתִּשְׁמַ֣ע עֲתַלְיָ֔ה אֶת־ ק֥וֹל הָֽרָצִ֖ין הָעָ֑ם וַתָּבֹ֥א אֶל־ הָעָ֖ם בֵּ֥ית יְהוָֽה׃ (לו) וַתֵּ֡רֶא וְהִנֵּ֣ה הַמֶּלֶךְ֩ עֹמֵ֨ד עַֽל־ הָעַמּ֜וּד כַּמִּשְׁפָּ֗ט וְהַשָּׂרִ֤ים וְהַחֲצֹֽצְרוֹת֙ אֶל־ הַמֶּ֔לֶךְ וְכָל־ עַ֤ם הָאָ֙רֶץ֙ שָׂמֵ֔חַ וְתֹקֵ֖עַ בַּחֲצֹֽצְר֑וֹת וַתִּקְרַ֤ע עֲתַלְיָה֙ אֶת־ בְּגָדֶ֔יהָ וַתִּקְרָ֖א קֶ֥שֶׁר קָֽשֶׁר׃ (לז) וַיְצַו֩ יְהוֹיָדָ֨ע הַכֹּהֵ֜ן אֶת־ שָׂרֵ֥י (המיאות) הַמֵּא֣וֹת ׀ פְּקֻדֵ֣י הַחַ֗יִל וַיֹּ֤אמֶר אֲלֵיהֶם֙ הוֹצִ֤יאוּ אֹתָהּ֙ אֶל־ מִבֵּ֣ית לַשְּׂדֵרֹ֔ת וְהַבָּ֥א אַחֲרֶ֖יהָ הָמֵ֣ת בֶּחָ֑רֶב כִּ֚י אָמַ֣ר הַכֹּהֵ֔ן אַל־ תּוּמַ֖ת בֵּ֥ית יְהוָֽה׃ (לח) וַיָּשִׂ֤מוּ לָהּ֙ יָדַ֔יִם וַתָּב֛וֹא דֶּֽרֶךְ־ מְב֥וֹא הַסּוּסִ֖ים בֵּ֣ית הַמֶּ֑לֶךְ וַתּוּמַ֖ת שָֽׁם׃ (לט) וַיִּכְרֹ֨ת יְהוֹיָדָ֜ע אֶֽת־ הַבְּרִ֗ית בֵּ֤ין יְהוָה֙ וּבֵ֤ין הַמֶּ֙לֶךְ֙ וּבֵ֣ין הָעָ֔ם לִהְי֥וֹת לְעָ֖ם לַֽיהוָ֑ה וּבֵ֥ין הַמֶּ֖לֶךְ וּבֵ֥ין הָעָֽם׃ (מ) וַיָּבֹ֣אוּ כָל־ עַם֩ הָאָ֨רֶץ בֵּית־ הַבַּ֜עַל וַֽיִּתְּצֻ֗הוּ אֶת־ (מזבחתו) מִזְבְּחֹתָ֤יו וְאֶת־ צְלָמָיו֙ שִׁבְּר֣וּ הֵיטֵ֔ב וְאֵ֗ת מַתָּן֙ כֹּהֵ֣ן הַבַּ֔עַל הָרְג֖וּ לִפְנֵ֣י הַֽמִּזְבְּח֑וֹת וַיָּ֧שֶׂם הַכֹּהֵ֛ן פְּקֻדּ֖וֹת עַל־ בֵּ֥ית יְהוָֽה׃ (מא) וַיִּקַּ֣ח אֶת־ שָׂרֵ֣י הַ֠מֵּאוֹת וְאֶת־ הַכָּרִ֨י וְאֶת־ הָרָצִ֜ים וְאֵ֣ת ׀ כָּל־ עַ֣ם הָאָ֗רֶץ וַיֹּרִ֤ידוּ אֶת־ הַמֶּ֙לֶךְ֙ מִבֵּ֣ית יְהוָ֔ה וַיָּב֛וֹאוּ דֶּֽרֶך־ שַׁ֥עַר הָרָצִ֖ים בֵּ֣ית הַמֶּ֑לֶךְ וַיֵּ֖שֶׁב עַל־ כִּסֵּ֥א הַמְּלָכִֽים׃ (מב) וַיִּשְׂמַ֥ח כָּל־ עַם־ הָאָ֖רֶץ וְהָעִ֣יר שָׁקָ֑טָה וְאֶת־ עֲתַלְיָ֛הוּ הֵמִ֥יתוּ בַחֶ֖רֶב בֵּ֥ית (מלך) הַמֶּֽלֶךְ׃ (מג) בֶּן־ שֶׁ֥בַע שָׁנִ֖ים יְהוֹאָ֥שׁ בְּמָלְכֽוֹ׃ (מד) בִּשְׁנַת־ שֶׁ֤בַע לְיֵהוּא֙ מָלַ֣ךְ יְהוֹאָ֔שׁ וְאַרְבָּעִ֣ים שָׁנָ֔ה מָלַ֖ךְ בִּירֽוּשָׁלִָ֑ם וְשֵׁ֣ם אִמּ֔וֹ צִבְיָ֖ה מִבְּאֵ֥ר שָֽׁבַע׃
