מלכים סדר כג

פתיחה באתר · ספר מלכים כולו ·

(א) וַיֵּצֵ֣א הָעָ֔ם וַיָּבֹ֕זּוּ אֵ֖ת מַחֲנֵ֣ה אֲרָ֑ם וַיְהִ֨י סְאָה־ סֹ֜לֶת בְּשֶׁ֗קֶל וְסָאתַ֧יִם שְׂעֹרִ֛ים בְּשֶׁ֖קֶל כִּדְבַ֥ר יְהוָֽה׃ (ב) וְהַמֶּלֶךְ֩ הִפְקִ֨יד אֶת־ הַשָּׁלִ֜ישׁ אֲשֶׁר־ נִשְׁעָ֤ן עַל־ יָדוֹ֙ עַל־ הַשַּׁ֔עַר וַיִּרְמְסֻ֧הוּ הָעָ֛ם בַּשַּׁ֖עַר וַיָּמֹ֑ת כַּאֲשֶׁ֤ר דִּבֶּר֙ אִ֣ישׁ הָאֱלֹהִ֔ים אֲשֶׁ֣ר דִּבֶּ֔ר בְּרֶ֥דֶת הַמֶּ֖לֶךְ אֵלָֽיו׃ (ג) וַיְהִ֗י כְּדַבֵּר֙ אִ֣ישׁ הָאֱלֹהִ֔ים אֶל־ הַמֶּ֖לֶךְ לֵאמֹ֑ר סָאתַ֨יִם שְׂעֹרִ֜ים בְּשֶׁ֗קֶל וּֽסְאָה־ סֹ֙לֶת֙ בְּשֶׁ֔קֶל יִהְיֶה֙ כָּעֵ֣ת מָחָ֔ר בְּשַׁ֖עַר שֹׁמְרֽוֹן׃ (ד) וַיַּ֨עַן הַשָּׁלִ֜ישׁ אֶת־ אִ֣ישׁ הָאֱלֹהִים֮ וַיֹּאמַר֒ וְהִנֵּ֣ה יְהוָ֗ה עֹשֶׂ֤ה אֲרֻבּוֹת֙ בַּשָּׁמַ֔יִם הֲיִהְיֶ֖ה כַּדָּבָ֣ר הַזֶּ֑ה וַיֹּ֗אמֶר הִנְּךָ֤ רֹאֶה֙ בְּעֵינֶ֔יךָ וּמִשָּׁ֖ם לֹ֥א תֹאכֵֽל׃ (ה) וַיְהִי־ ל֖וֹ כֵּ֑ן וַיִּרְמְס֨וּ אֹת֥וֹ הָעָ֛ם בַּשַּׁ֖עַר וַיָּמֹֽת׃ (ו) וֶאֱלִישָׁ֡ע דִּבֶּ֣ר אֶל־ הָאִשָּׁה֩ אֲשֶׁר־ הֶחֱיָ֨ה אֶת־ בְּנָ֜הּ לֵאמֹ֗ר ק֤וּמִי וּלְכִי֙ (אתי) אַ֣תְּ וּבֵיתֵ֔ךְ וְג֖וּרִי בַּאֲשֶׁ֣ר תָּג֑וּרִי כִּֽי־ קָרָ֤א יְהוָה֙ לָֽרָעָ֔ב וְגַם־ בָּ֥א אֶל־ הָאָ֖רֶץ שֶׁ֥בַע שָׁנִֽים׃ (ז) וַתָּ֙קָם֙ הָֽאִשָּׁ֔ה וַתַּ֕עַשׂ כִּדְבַ֖ר אִ֣ישׁ הָאֱלֹהִ֑ים וַתֵּ֤לֶךְ הִיא֙ וּבֵיתָ֔הּ וַתָּ֥גָר בְּאֶֽרֶץ־ פְּלִשְׁתִּ֖ים שֶׁ֥בַע שָׁנִֽים׃ (ח) וַיְהִ֗י מִקְצֵה֙ שֶׁ֣בַע שָׁנִ֔ים וַתָּ֥שָׁב הָאִשָּׁ֖ה מֵאֶ֣רֶץ פְּלִשְׁתִּ֑ים וַתֵּצֵא֙ לִצְעֹ֣ק אֶל־ הַמֶּ֔לֶךְ אֶל־ בֵּיתָ֖הּ וְאֶל־ שָׂדָֽהּ׃ (ט) וְהַמֶּ֗לֶךְ מְדַבֵּר֙ אֶל־ גֵּ֣חֲזִ֔י נַ֥עַר אִישׁ־ הָאֱלֹהִ֖ים לֵאמֹ֑ר סַפְּרָה־ נָּ֣א לִ֔י אֵ֥ת כָּל־ הַגְּדֹל֖וֹת אֲשֶׁר־ עָשָׂ֥ה אֱלִישָֽׁע׃ (י) וַ֠יְהִי ה֥וּא מְסַפֵּ֣ר לַמֶּלֶךְ֮ אֵ֣ת אֲשֶׁר־ הֶחֱיָ֣ה אֶת־ הַמֵּת֒ וְהִנֵּ֨ה הָאִשָּׁ֜ה אֲשֶׁר־ הֶחֱיָ֤ה אֶת־ בְּנָהּ֙ צֹעֶ֣קֶת אֶל־ הַמֶּ֔לֶךְ עַל־ בֵּיתָ֖הּ וְעַל־ שָׂדָ֑הּ וַיֹּ֤אמֶר גֵּֽחֲזִי֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ זֹ֚את הָֽאִשָּׁ֔ה וְזֶה־ בְּנָ֖הּ אֲשֶׁר־ הֶחֱיָ֥ה אֱלִישָֽׁע׃ (יא) וַיִּשְׁאַ֥ל הַמֶּ֛לֶךְ לָאִשָּׁ֖ה וַתְּסַפֶּר־ ל֑וֹ וַיִּתֶּן־ לָ֣הּ הַמֶּלֶךְ֩ סָרִ֨יס אֶחָ֜ד לֵאמֹ֗ר הָשֵׁ֤יב אֶת־ כָּל־ אֲשֶׁר־ לָהּ֙ וְאֵת֙ כָּל־ תְּבוּאֹ֣ת הַשָּׂדֶ֔ה מִיּ֛וֹם עָזְבָ֥ה אֶת־ הָאָ֖רֶץ וְעַד־ עָֽתָּה׃ (יב) וַיָּבֹ֤א אֱלִישָׁע֙ דַּמֶּ֔שֶׂק וּבֶן־ הֲדַ֥ד מֶֽלֶךְ־ אֲרָ֖ם חֹלֶ֑ה וַיֻּגַּד־ ל֣וֹ לֵאמֹ֔ר בָּ֛א אִ֥ישׁ הָאֱלֹהִ֖ים עַד־ הֵֽנָּה׃ (יג) וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֶל־ חֲזָהאֵ֗ל קַ֤ח בְּיָֽדְךָ֙ מִנְחָ֔ה וְלֵ֕ךְ לִקְרַ֖את אִ֣ישׁ הָאֱלֹהִ֑ים וְדָרַשְׁתָּ֨ אֶת־ יְהוָ֤ה מֵֽאוֹתוֹ֙ לֵאמֹ֔ר הַאֶחְיֶ֖ה מֵחֳלִ֥י זֶֽה׃ (יד) וַיֵּ֣לֶךְ חֲזָאֵל֮ לִקְרָאתוֹ֒ וַיִּקַּ֨ח מִנְחָ֤ה בְיָדוֹ֙ וְכָל־ ט֣וּב דַּמֶּ֔שֶׂק מַשָּׂ֖א אַרְבָּעִ֣ים גָּמָ֑ל וַיָּבֹא֙ וַיַּעֲמֹ֣ד לְפָנָ֔יו וַיֹּ֗אמֶר בִּנְךָ֨ בֶן־ הֲדַ֤ד מֶֽלֶךְ־ אֲרָם֙ שְׁלָחַ֤נִי אֵלֶ֙יךָ֙ לֵאמֹ֔ר הַאֶחְיֶ֖ה מֵחֳלִ֥י זֶֽה׃ (טו) וַיֹּ֤אמֶר אֵלָיו֙ אֱלִישָׁ֔ע לֵ֥ךְ אֱמָר־ (לא) ל֖וֹ חָיֹ֣ה תִחְיֶ֑ה וְהִרְאַ֥נִי יְהוָ֖ה כִּֽי־ מ֥וֹת יָמֽוּת׃ (טז) וַיַּעֲמֵ֥ד אֶת־ פָּנָ֖יו וַיָּ֣שֶׂם עַד־ בֹּ֑שׁ וַיֵּ֖בְךְּ אִ֥ישׁ הָאֱלֹהִֽים׃ (יז) וַיֹּ֣אמֶר חֲזָאֵ֔ל מַדּ֖וּעַ אֲדֹנִ֣י בֹכֶ֑ה וַיֹּ֡אמֶר כִּֽי־ יָדַ֡עְתִּי אֵ֣ת אֲשֶׁר־ תַּעֲשֶׂה֩ לִבְנֵ֨י יִשְׂרָאֵ֜ל רָעָ֗ה מִבְצְרֵיהֶ֞ם תְּשַׁלַּ֤ח בָּאֵשׁ֙ וּבַחֻֽרֵיהֶם֙ בַּחֶ֣רֶב תַּהֲרֹ֔ג וְעֹלְלֵיהֶ֣ם תְּרַטֵּ֔שׁ וְהָרֹתֵיהֶ֖ם תְּבַקֵּֽעַ׃ (יח) וַיֹּ֣אמֶר חֲזָהאֵ֔ל כִּ֣י מָ֤ה עַבְדְּךָ֙ הַכֶּ֔לֶב כִּ֣י יַעֲשֶׂ֔ה הַדָּבָ֥ר הַגָּד֖וֹל הַזֶּ֑ה וַיֹּ֣אמֶר אֱלִישָׁ֔ע הִרְאַ֧נִי יְהוָ֛ה אֹתְךָ֖ מֶ֥לֶךְ עַל־ אֲרָֽם׃ (יט) וַיֵּ֣לֶךְ ׀ מֵאֵ֣ת אֱלִישָׁ֗ע וַיָּבֹא֙ אֶל־ אֲדֹנָ֔יו וַיֹּ֣אמֶר ל֔וֹ מָֽה־ אָמַ֥ר לְךָ֖ אֱלִישָׁ֑ע וַיֹּ֕אמֶר אָ֥מַר לִ֖י חָיֹ֥ה תִחְיֶֽה׃ (כ) וַיְהִ֣י מִֽמָּחֳרָ֗ת וַיִּקַּ֤ח הַמַּכְבֵּר֙ וַיִּטְבֹּ֣ל בַּמַּ֔יִם וַיִּפְרֹ֥שׂ עַל־ פָּנָ֖יו וַיָּמֹ֑ת וַיִּמְלֹ֥ךְ חֲזָהאֵ֖ל תַּחְתָּֽיו׃ (כא) וּבִשְׁנַ֣ת חָמֵ֗שׁ לְיוֹרָ֤ם בֶּן־ אַחְאָב֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וִיהוֹשָׁפָ֖ט מֶ֣לֶךְ יְהוּדָ֑ה מָלַ֛ךְ יְהוֹרָ֥ם בֶּן־ יְהוֹשָׁפָ֖ט מֶ֥לֶךְ יְהוּדָֽה׃ (כב) בֶּן־ שְׁלֹשִׁ֥ים וּשְׁתַּ֛יִם שָׁנָ֖ה הָיָ֣ה בְמָלְכ֑וֹ וּשְׁמֹנֶ֣ה (שנה) שָׁנִ֔ים מָלַ֖ךְ בִּירוּשָׁלִָֽם׃ (כג) וַיֵּ֜לֶךְ בְּדֶ֣רֶךְ ׀ מַלְכֵ֣י יִשְׂרָאֵ֗ל כַּאֲשֶׁ֤ר עָשׂוּ֙ בֵּ֣ית אַחְאָ֔ב כִּ֚י בַּת־ אַחְאָ֔ב הָֽיְתָה־ לּ֖וֹ לְאִשָּׁ֑ה וַיַּ֥עַשׂ הָרַ֖ע בְּעֵינֵ֥י יְהוָֽה׃ (כד) וְלֹֽא־ אָבָ֤ה יְהוָה֙ לְהַשְׁחִ֣ית אֶת־ יְהוּדָ֔ה לְמַ֖עַן דָּוִ֣ד עַבְדּ֑וֹ כַּאֲשֶׁ֣ר אָֽמַר־ ל֗וֹ לָתֵ֨ת ל֥וֹ נִ֛יר לְבָנָ֖יו כָּל־ הַיָּמִֽים׃ (כה) בְּיָמָיו֙ פָּשַׁ֣ע אֱד֔וֹם מִתַּ֖חַת יַד־ יְהוּדָ֑ה וַיַּמְלִ֥כוּ עֲלֵיהֶ֖ם מֶֽלֶךְ׃ (כו) וַיַּעֲבֹ֤ר יוֹרָם֙ צָעִ֔ירָה וְכָל־ הָרֶ֖כֶב עִמּ֑וֹ וַֽיְהִי־ ה֞וּא קָ֣ם לַ֗יְלָה וַיַּכֶּ֨ה אֶת־ אֱד֜וֹם הַסֹּבֵ֤יב אֵלָיו֙ וְאֵת֙ שָׂרֵ֣י הָרֶ֔כֶב וַיָּ֥נָס הָעָ֖ם לְאֹהָלָֽיו׃ (כז) וַיִּפְשַׁ֣ע אֱד֗וֹם מִתַּ֙חַת֙ יַד־ יְהוּדָ֔ה עַ֖ד הַיּ֣וֹם הַזֶּ֑ה אָ֛ז תִּפְשַׁ֥ע לִבְנָ֖ה בָּעֵ֥ת הַהִֽיא׃ (כח) וְיֶ֛תֶר דִּבְרֵ֥י יוֹרָ֖ם וְכָל־ אֲשֶׁ֣ר עָשָׂ֑ה הֲלֽוֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ סֵ֛פֶר דִּבְרֵ֥י הַיָּמִ֖ים לְמַלְכֵ֥י יְהוּדָֽה׃ (כט) וַיִּשְׁכַּ֤ב יוֹרָם֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֥ר עִם־ אֲבֹתָ֖יו בְּעִ֣יר דָּוִ֑ד וַיִּמְלֹ֛ךְ אֲחַזְיָ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ (ל) בִּשְׁנַת֙ שְׁתֵּים־ עֶשְׂרֵ֣ה שָׁנָ֔ה לְיוֹרָ֥ם בֶּן־ אַחְאָ֖ב מֶ֣לֶךְ יִשְׂרָאֵ֑ל מָלַ֛ךְ אֲחַזְיָ֥הוּ בֶן־ יְהוֹרָ֖ם מֶ֥לֶךְ יְהוּדָֽה׃ (לא) בֶּן־ עֶשְׂרִ֨ים וּשְׁתַּ֤יִם שָׁנָה֙ אֲחַזְיָ֣הוּ בְמָלְכ֔וֹ וְשָׁנָ֣ה אַחַ֔ת מָלַ֖ךְ בִּירוּשָׁלִָ֑ם וְשֵׁ֤ם אִמּוֹ֙ עֲתַלְיָ֔הוּ בַּת־ עָמְרִ֖י מֶ֥לֶךְ יִשְׂרָאֵֽל׃ (לב) וַיֵּ֗לֶךְ בְּדֶ֙רֶךְ֙ בֵּ֣ית אַחְאָ֔ב וַיַּ֧עַשׂ הָרַ֛ע בְּעֵינֵ֥י יְהוָ֖ה כְּבֵ֣ית אַחְאָ֑ב כִּ֛י חֲתַ֥ן בֵּית־ אַחְאָ֖ב הֽוּא׃ (לג) וַיֵּ֜לֶךְ אֶת־ יוֹרָ֣ם בֶּן־ אַחְאָ֗ב לַמִּלְחָמָ֛ה עִם־ חֲזָהאֵ֥ל מֶֽלֶךְ־ אֲרָ֖ם בְּרָמֹ֣ת גִּלְעָ֑ד וַיַּכּ֥וּ אֲרַמִּ֖ים אֶת־ יוֹרָֽם׃ (לד) וַיָּשָׁב֩ יוֹרָ֨ם הַמֶּ֜לֶךְ לְהִתְרַפֵּ֣א בְיִזְרְעֶ֗אל מִן־ הַמַּכִּים֙ אֲשֶׁ֨ר יַכֻּ֤הוּ אֲרַמִּים֙ בָּֽרָמָ֔ה בְּהִלָּ֣חֲמ֔וֹ אֶת־ חֲזָהאֵ֖ל מֶ֣לֶךְ אֲרָ֑ם וַאֲחַזְיָ֨הוּ בֶן־ יְהוֹרָ֜ם מֶ֣לֶךְ יְהוּדָ֗ה יָרַ֡ד לִרְא֞וֹת אֶת־ יוֹרָ֧ם בֶּן־ אַחְאָ֛ב בְּיִזְרְעֶ֖אל כִּֽי־ חֹלֶ֥ה הֽוּא׃ (לה) וֶאֱלִישָׁע֙ הַנָּבִ֔יא קָרָ֕א לְאַחַ֖ד מִבְּנֵ֣י הַנְּבִיאִ֑ים וַיֹּ֨אמֶר ל֜וֹ חֲגֹ֣ר מָתְנֶ֗יךָ וְ֠קַח פַּ֣ךְ הַשֶּׁ֤מֶן הַזֶּה֙ בְּיָדֶ֔ךָ וְלֵ֖ךְ רָמֹ֥ת גִּלְעָֽד׃ (לו) וּבָ֖אתָ שָׁ֑מָּה וּרְאֵֽה־ שָׁ֠ם יֵה֨וּא בֶן־ יְהוֹשָׁפָ֜ט בֶּן־ נִמְשִׁ֗י וּבָ֙אתָ֙ וַהֲקֵֽמֹתוֹ֙ מִתּ֣וֹך אֶחָ֔יו וְהֵבֵיאתָ֥ אֹת֖וֹ חֶ֥דֶר בְּחָֽדֶר׃ (לז) וְלָקַחְתָּ֤ פַךְ־ הַשֶּׁ֙מֶן֙ וְיָצַקְתָּ֣ עַל־ רֹאשׁ֔וֹ וְאָֽמַרְתָּ֙ כֹּֽה־ אָמַ֣ר יְהוָ֔ה מְשַׁחְתִּ֥יךָֽ לְמֶ֖לֶךְ אֶל־ יִשְׂרָאֵ֑ל וּפָתַחְתָּ֥ הַדֶּ֛לֶת וְנַ֖סְתָּה וְלֹ֥א תְחַכֶּֽה׃ (לח) וַיֵּ֧לֶךְ הַנַּ֛עַר הַנַּ֥עַר הַנָּבִ֖יא רָמֹ֥ת גִּלְעָֽד׃ (לט) וַיָּבֹ֗א וְהִנֵּ֨ה שָׂרֵ֤י הַחַ֙יִל֙ יֹֽשְׁבִ֔ים וַיֹּ֕אמֶר דָּבָ֥ר לִ֛י אֵלֶ֖יךָ הַשָּׂ֑ר וַיֹּ֤אמֶר יֵהוּא֙ אֶל־ מִ֣י מִכֻּלָּ֔נוּ וַיֹּ֖אמֶר אֵלֶ֥יךָ הַשָּֽׂר׃ (מ) וַיָּ֙קָם֙ וַיָּבֹ֣א הַבַּ֔יְתָה וַיִּצֹ֥ק הַשֶּׁ֖מֶן אֶל־ רֹאשׁ֑וֹ וַיֹּ֣אמֶר ל֗וֹ כֹּֽה־ אָמַ֤ר יְהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל מְשַׁחְתִּ֧יךָֽ לְמֶ֛לֶךְ אֶל־ עַ֥ם יְהוָ֖ה אֶל־ יִשְׂרָאֵֽל׃ (מא) וְהִ֨כִּיתָ֔ה אֶת־ בֵּ֥ית אַחְאָ֖ב אֲדֹנֶ֑יךָ וְנִקַּמְתִּ֞י דְּמֵ֣י ׀ עֲבָדַ֣י הַנְּבִיאִ֗ים וּדְמֵ֛י כָּל־ עַבְדֵ֥י יְהוָ֖ה מִיַּ֥ד אִיזָֽבֶל׃ (מב) וְאָבַ֖ד כָּל־ בֵּ֣ית אַחְאָ֑ב וְהִכְרַתִּ֤י לְאַחְאָב֙ מַשְׁתִּ֣ין בְּקִ֔יר וְעָצ֥וּר וְעָז֖וּב בְּיִשְׂרָאֵֽל׃ (מג) וְנָֽתַתִּי֙ אֶת־ בֵּ֣ית אַחְאָ֔ב כְּבֵ֖ית יָרָבְעָ֣ם בֶּן־ נְבָ֑ט וּכְבֵ֖ית בַּעְשָׁ֥א בֶן־ אֲחִיָּֽה׃ (מד) וְאֶת־ אִיזֶ֜בֶל יֹאכְל֧וּ הַכְּלָבִ֛ים בְּחֵ֥לֶק יִזְרְעֶ֖אל וְאֵ֣ין קֹבֵ֑ר וַיִּפְתַּ֥ח הַדֶּ֖לֶת וַיָּנֹֽס׃ (מה) וְיֵה֗וּא יָצָא֙ אֶל־ עַבְדֵ֣י אֲדֹנָ֔יו וַיֹּ֤אמֶר לוֹ֙ הֲשָׁל֔וֹם מַדּ֛וּעַ בָּֽא־ הַמְשֻׁגָּ֥ע הַזֶּ֖ה אֵלֶ֑יךָ וַיֹּ֣אמֶר אֲלֵיהֶ֔ם אַתֶּ֛ם יְדַעְתֶּ֥ם אֶת־ הָאִ֖ישׁ וְאֶת־ שִׂיחֽוֹ׃ (מו) וַיֹּאמְר֣וּ שֶׁ֔קֶר הַגֶּד־ נָ֖א לָ֑נוּ וַיֹּ֗אמֶר כָּזֹ֤את וְכָזֹאת֙ אָמַ֤ר אֵלַי֙ לֵאמֹ֔ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה מְשַׁחְתִּ֥יךָֽ לְמֶ֖לֶךְ אֶל־ יִשְׂרָאֵֽל׃
