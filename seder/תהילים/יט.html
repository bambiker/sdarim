(א) אַ֣ךְ צַ֭דִּיקִים יוֹד֣וּ לִשְׁמֶ֑ךָ יֵשְׁב֥וּ יְ֝שָׁרִ֗ים אֶת־ פָּנֶֽיךָ׃ (ב) מִזְמ֗וֹר לְדָ֫וִ֥ד יְהוָ֣ה קְ֭רָאתִיךָ ח֣וּשָׁה לִּ֑י הַאֲזִ֥ינָה ק֝וֹלִ֗י בְּקָרְאִי־ לָֽךְ׃ (ג) תִּכּ֤וֹן תְּפִלָּתִ֣י קְטֹ֣רֶת לְפָנֶ֑יךָ מַֽשְׂאַ֥ת כַּ֝פַּ֗י מִנְחַת־ עָֽרֶב׃ (ד) שִׁיתָ֣ה יְ֭הוָה שָׁמְרָ֣ה לְפִ֑י נִ֝צְּרָ֗ה עַל־ דַּ֥ל שְׂפָתָֽי׃ (ה) אַל־ תַּט־ לִבִּ֨י לְדָבָ֪ר ׀ רָ֡ע לְהִתְע֘וֹלֵ֤ל עֲלִל֨וֹת ׀ בְּרֶ֗שַׁע אֶת־ אִישִׁ֥ים פֹּֽעֲלֵי־ אָ֑וֶן וּבַל־ אֶ֝לְחַ֗ם בְּמַנְעַמֵּיהֶֽם׃ (ו) יֶֽהֶלְמֵֽנִי־ צַדִּ֨יק ׀ חֶ֡סֶד וְֽיוֹכִיחֵ֗נִי שֶׁ֣מֶן רֹ֭אשׁ אַל־ יָנִ֣י רֹאשִׁ֑י כִּי־ ע֥וֹד וּ֝תְפִלָּתִ֗י בְּרָעוֹתֵיהֶֽם׃ (ז) נִשְׁמְט֣וּ בִֽידֵי־ סֶ֭לַע שֹׁפְטֵיהֶ֑ם וְשָׁמְע֥וּ אֲ֝מָרַ֗י כִּ֣י נָעֵֽמוּ׃ (ח) כְּמ֤וֹ פֹלֵ֣חַ וּבֹקֵ֣עַ בָּאָ֑רֶץ נִפְזְר֥וּ עֲ֝צָמֵ֗ינוּ לְפִ֣י שְׁאֽוֹל׃ (ט) כִּ֤י אֵלֶ֨יךָ ׀ יְהֹוִ֣ה אֲדֹנָ֣י עֵינָ֑י בְּכָ֥ה חָ֝סִ֗יתִי אַל־ תְּעַ֥ר נַפְשִֽׁי׃ (י) שָׁמְרֵ֗נִי מִ֣ידֵי פַ֭ח יָ֣קְשׁוּ לִ֑י וּ֝מֹקְשׁ֗וֹת פֹּ֣עֲלֵי אָֽוֶן׃ (יא) יִפְּל֣וּ בְמַכְמֹרָ֣יו רְשָׁעִ֑ים יַ֥חַד אָ֝נֹכִ֗י עַֽד־ אֶעֱבֽוֹר׃ (יב) מַשְׂכִּ֥יל לְדָוִ֑ד בִּהְיוֹת֖וֹ בַמְּעָרָ֣ה תְפִלָּֽה׃ (יג) ק֭וֹלִי אֶל־ יְהוָ֣ה אֶזְעָ֑ק ק֝וֹלִ֗י אֶל־ יְהוָ֥ה אֶתְחַנָּֽן׃ (יד) אֶשְׁפֹּ֣ךְ לְפָנָ֣יו שִׂיחִ֑י צָ֝רָתִ֗י לְפָנָ֥יו אַגִּֽיד׃ (טו) בְּהִתְעַטֵּ֬ף עָלַ֨י ׀ רוּחִ֗י וְאַתָּה֮ יָדַ֪עְתָּ נְֽתִיבָ֫תִ֥י בְּאֹֽרַח־ ז֥וּ אֲהַלֵּ֑ךְ טָמְנ֖וּ פַ֣ח לִֽי׃ (טז) הַבֵּ֤יט יָמִ֨ין ׀ וּרְאֵה֮ וְאֵֽין־ לִ֪י מַ֫כִּ֥יר אָבַ֣ד מָנ֣וֹס מִמֶּ֑נִּי אֵ֖ין דּוֹרֵ֣שׁ לְנַפְשִֽׁי׃ (יז) זָעַ֥קְתִּי אֵלֶ֗יךָ יְה֫וָ֥ה אָ֭מַרְתִּי אַתָּ֣ה מַחְסִ֑י חֶ֝לְקִ֗י בְּאֶ֣רֶץ הַֽחַיִּים׃ (יח) הַקְשִׁ֤יבָה ׀ אֶֽל־ רִנָּתִי֮ כִּֽי־ דַלּ֪וֹתִ֫י מְאֹ֥ד הַצִּילֵ֥נִי מֵרֹדְפַ֑י כִּ֖י אָמְצ֣וּ מִמֶּֽנִּי׃ (יט) ה֘וֹצִ֤יאָה מִמַּסְגֵּ֨ר ׀ נַפְשִׁי֮ לְהוֹד֪וֹת אֶת־ שְׁ֫מֶ֥ךָ בִּ֭י יַכְתִּ֣רוּ צַדִּיקִ֑ים כִּ֖י תִגְמֹ֣ל עָלָֽי׃ (כ) מִזְמ֗וֹר לְדָ֫וִ֥ד יְהוָ֤ה ׀ שְׁמַ֬ע תְּפִלָּתִ֗י הַאֲזִ֥ינָה אֶל־ תַּחֲנוּנַ֑י בֶּאֱמֻנָתְךָ֥ עֲ֝נֵ֗נִי בְּצִדְקָתֶֽךָ׃ (כא) וְאַל־ תָּב֣וֹא בְ֭מִשְׁפָּט אֶת־ עַבְדֶּ֑ךָ כִּ֤י לֹֽא־ יִצְדַּ֖ק לְפָנֶ֣יךָ כָל־ חָֽי׃ (כב) כִּ֥י רָ֘דַ֤ף אוֹיֵ֨ב ׀ נַפְשִׁ֗י דִּכָּ֣א לָ֭אָרֶץ חַיָּתִ֑י הוֹשִׁיבַ֥נִי בְ֝מַחֲשַׁכִּ֗ים כְּמֵתֵ֥י עוֹלָֽם׃ (כג) וַתִּתְעַטֵּ֣ף עָלַ֣י רוּחִ֑י בְּ֝תוֹכִ֗י יִשְׁתּוֹמֵ֥ם לִבִּֽי׃ (כד) זָ֘כַ֤רְתִּי יָמִ֨ים ׀ מִקֶּ֗דֶם הָגִ֥יתִי בְכָל־ פָּעֳלֶ֑ךָ בְּֽמַעֲשֵׂ֖ה יָדֶ֣יךָ אֲשׂוֹחֵֽחַ׃ (כה) פֵּרַ֣שְׂתִּי יָדַ֣י אֵלֶ֑יךָ נַפְשִׁ֓י ׀ כְּאֶֽרֶץ־ עֲיֵפָ֖ה לְךָ֣ סֶֽלָה׃ (כו) מַ֘הֵ֤ר עֲנֵ֨נִי ׀ יְהוָה֮ כָּלְתָ֪ה ר֫וּחִ֥י אַל־ תַּסְתֵּ֣ר פָּנֶ֣יךָ מִמֶּ֑נִּי וְ֝נִמְשַׁ֗לְתִּי עִם־ יֹ֥רְדֵי בֽוֹר׃ (כז) הַשְׁמִ֘יעֵ֤נִי בַבֹּ֨קֶר ׀ חַסְדֶּךָ֮ כִּֽי־ בְךָ֪ בָ֫טָ֥חְתִּי הוֹדִיעֵ֗נִי דֶּֽרֶךְ־ ז֥וּ אֵלֵ֑ךְ כִּֽי־ אֵ֝לֶיךָ נָשָׂ֥אתִי נַפְשִֽׁי׃ (כח) הַצִּילֵ֖נִי מֵאֹיְבַ֥י ׀ יְהוָ֗ה אֵלֶ֥יךָ כִסִּֽתִי׃ (כט) לַמְּדֵ֤נִי ׀ לַֽעֲשׂ֣וֹת רְצוֹנֶךָ֮ כִּֽי־ אַתָּ֪ה אֱל֫וֹהָ֥י רוּחֲךָ֥ טוֹבָ֑ה תַּ֝נְחֵ֗נִי בְּאֶ֣רֶץ מִישֽׁוֹר׃ (ל) לְמַֽעַן־ שִׁמְךָ֣ יְהוָ֣ה תְּחַיֵּ֑נִי בְּצִדְקָתְךָ֓ ׀ תוֹצִ֖יא מִצָּרָ֣ה נַפְשִֽׁי׃ (לא) וּֽבְחַסְדְּךָ֮ תַּצְמִ֪ית אֹ֫יְבָ֥י וְֽ֭הַאֲבַדְתָּ כָּל־ צֹרֲרֵ֣י נַפְשִׁ֑י כִּ֝֗י אֲנִ֣י עַבְדֶּֽךָ׃ (לב) לְדָוִ֨ד ׀ בָּ֘ר֤וּךְ יְהוָ֨ה ׀ צוּרִ֗י הַֽמְלַמֵּ֣ד יָדַ֣י לַקְרָ֑ב אֶ֝צְבְּעוֹתַ֗י לַמִּלְחָמָֽה׃ (לג) חַסְדִּ֥י וּמְצוּדָתִי֮ מִשְׂגַּבִּ֪י וּֽמְפַלְטִ֫י לִ֥י מָ֭גִנִּי וּב֣וֹ חָסִ֑יתִי הָרוֹדֵ֖ד עַמִּ֣י תַחְתָּֽי׃ (לד) יְֽהוָ֗ה מָה־ אָ֭דָם וַתֵּדָעֵ֑הוּ בֶּן־ אֱ֝נ֗וֹשׁ וַֽתְּחַשְּׁבֵֽהוּ׃ (לה) אָ֭דָם לַהֶ֣בֶל דָּמָ֑ה יָ֝מָ֗יו כְּצֵ֣ל עוֹבֵֽר׃ (לו) יְ֭הוָה הַט־ שָׁמֶ֣יךָ וְתֵרֵ֑ד גַּ֖ע בֶּהָרִ֣ים וְֽיֶעֱשָֽׁנוּ׃ (לז) בְּר֣וֹק בָּ֭רָק וּתְפִיצֵ֑ם שְׁלַ֥ח חִ֝צֶּ֗יךָ וּתְהֻמֵּֽם׃ (לח) שְׁלַ֥ח יָדֶ֗יךָ מִמָּ֫ר֥וֹם פְּצֵ֣נִי וְ֭הַצִּילֵנִי מִמַּ֣יִם רַבִּ֑ים מִ֝יַּ֗ד בְּנֵ֣י נֵכָֽר׃ (לט) אֲשֶׁ֣ר פִּ֭יהֶם דִּבֶּר־ שָׁ֑וְא וִֽ֝ימִינָ֗ם יְמִ֣ין שָֽׁקֶר׃ (מ) אֱ‍ֽלֹהִ֗ים שִׁ֣יר חָ֭דָשׁ אָשִׁ֣ירָה לָּ֑ךְ בְּנֵ֥בֶל עָ֝שׂ֗וֹר אֲזַמְּרָה־ לָּֽךְ׃ (מא) הַנּוֹתֵ֥ן תְּשׁוּעָ֗ה לַמְּלָ֫כִ֥ים הַ֭פּוֹצֶה אֶת־ דָּוִ֥ד עַבְדּ֗וֹ מֵחֶ֥רֶב רָעָֽה׃ (מב) פְּצֵ֥נִי וְהַצִּילֵנִי֮ מִיַּ֪ד בְּֽנֵי־ נֵ֫כָ֥ר אֲשֶׁ֣ר פִּ֭יהֶם דִּבֶּר־ שָׁ֑וְא וִֽ֝ימִינָ֗ם יְמִ֣ין שָֽׁקֶר׃ (מג) אֲשֶׁ֤ר בָּנֵ֨ינוּ ׀ כִּנְטִעִים֮ מְגֻדָּלִ֪ים בִּֽנְעוּרֵ֫יהֶ֥ם בְּנוֹתֵ֥ינוּ כְזָוִיֹּ֑ת מְ֝חֻטָּב֗וֹת תַּבְנִ֥ית הֵיכָֽל׃ (מד) מְזָוֵ֣ינוּ מְלֵאִים֮ מְפִיקִ֥ים מִזַּ֗ן אֶ֫ל־ זַ֥ן צֹאונֵ֣נוּ מַ֭אֲלִיפוֹת מְרֻבָּב֗וֹת בְּחוּצוֹתֵֽינוּ׃ (מה) אַלּוּפֵ֗ינוּ מְֽסֻבָּ֫לִ֥ים אֵֽין־ פֶּ֭רֶץ וְאֵ֣ין יוֹצֵ֑את וְאֵ֥ין צְ֝וָחָ֗ה בִּרְחֹבֹתֵֽינוּ׃ (מו) אַשְׁרֵ֣י הָ֭עָם שֶׁכָּ֣כָה לּ֑וֹ אַֽשְׁרֵ֥י הָ֝עָ֗ם שֶׁיֲהוָ֥ה אֱלֹהָֽיו׃ (מז) תְּהִלָּ֗ה לְדָ֫וִ֥ד אֲרוֹמִמְךָ֣ אֱלוֹהַ֣י הַמֶּ֑לֶךְ וַאֲבָרֲכָ֥ה שִׁ֝מְךָ֗ לְעוֹלָ֥ם וָעֶֽד׃ (מח) בְּכָל־ י֥וֹם אֲבָרֲכֶ֑ךָּ וַאֲהַלְלָ֥ה שִׁ֝מְךָ֗ לְעוֹלָ֥ם וָעֶֽד׃ (מט) גָּ֘ד֤וֹל יְהוָ֣ה וּמְהֻלָּ֣ל מְאֹ֑ד וְ֝לִגְדֻלָּת֗וֹ אֵ֣ין חֵֽקֶר׃ (נ) דּ֣וֹר לְ֭דוֹר יְשַׁבַּ֣ח מַעֲשֶׂ֑יךָ וּגְב֖וּרֹתֶ֣יךָ יַגִּֽידוּ׃ (נא) הֲ֭דַר כְּב֣וֹד הוֹדֶ֑ךָ וְדִבְרֵ֖י נִפְלְאוֹתֶ֣יךָ אָשִֽׂיחָה׃ (נב) וֶעֱז֣וּז נוֹרְאֹתֶ֣יךָ יֹאמֵ֑רוּ (וגדולתיך) וּגְדוּלָּתְךָ֥ אֲסַפְּרֶֽנָּה׃ (נג) זֵ֣כֶר רַב־ טוּבְךָ֣ יַבִּ֑יעוּ וְצִדְקָתְךָ֥ יְרַנֵּֽנוּ׃ (נד) חַנּ֣וּן וְרַח֣וּם יְהוָ֑ה אֶ֥רֶךְ אַ֝פַּ֗יִם וּגְדָל־ חָֽסֶד׃ (נה) טוֹב־ יְהוָ֥ה לַכֹּ֑ל וְ֝רַחֲמָ֗יו עַל־ כָּל־ מַעֲשָֽׂיו׃ (נו) יוֹד֣וּךָ יְ֭הוָה כָּל־ מַעֲשֶׂ֑יךָ וַ֝חֲסִידֶ֗יךָ יְבָרֲכֽוּכָה׃ (נז) כְּב֣וֹד מַלְכוּתְךָ֣ יֹאמֵ֑רוּ וּגְבוּרָתְךָ֥ יְדַבֵּֽרוּ׃ (נח) לְהוֹדִ֤יעַ ׀ לִבְנֵ֣י הָ֭אָדָם גְּבוּרֹתָ֑יו וּ֝כְב֗וֹד הֲדַ֣ר מַלְכוּתֽוֹ׃ (נט) מַֽלְכוּתְךָ֗ מַלְכ֥וּת כָּל־ עֹֽלָמִ֑ים וּ֝מֶֽמְשֶׁלְתְּךָ֗ בְּכָל־ דּ֥וֹר וָדֽוֹר׃ (ס) סוֹמֵ֣ךְ יְ֭הוָה לְכָל־ הַנֹּפְלִ֑ים וְ֝זוֹקֵ֗ף לְכָל־ הַכְּפוּפִֽים׃ (סא) עֵֽינֵי־ כֹ֭ל אֵלֶ֣יךָ יְשַׂבֵּ֑רוּ וְאַתָּ֤ה נֽוֹתֵן־ לָהֶ֖ם אֶת־ אָכְלָ֣ם בְּעִתּֽוֹ׃ (סב) פּוֹתֵ֥חַ אֶת־ יָדֶ֑ךָ וּמַשְׂבִּ֖יעַ לְכָל־ חַ֣י רָצֽוֹן׃ (סג) צַדִּ֣יק יְ֭הוָה בְּכָל־ דְּרָכָ֑יו וְ֝חָסִ֗יד בְּכָל־ מַעֲשָֽׂיו׃ (סד) קָר֣וֹב הָרְשָׁעִ֣ים יַשְׁמִֽיד׃ (סז) תְּהִלַּ֥ת יְהוָ֗ה יְֽדַבֶּ֫ר־ פִּ֥י וִיבָרֵ֣ךְ כָּל־ בָּ֭שָׂר שֵׁ֥ם קָדְשׁ֗וֹ לְעוֹלָ֥ם וָעֶֽד׃ (סח) הַֽלְלוּ־ יָ֡הּ הַלְלִ֥י נַ֝פְשִׁ֗י אֶת־ יְהוָֽה׃ (סט) אֲהַלְלָ֣ה יְהוָ֣ה בְּחַיָּ֑י אֲזַמְּרָ֖ה לֵֽאלֹהַ֣י בְּעוֹדִֽי׃ (ע) אַל־ תִּבְטְח֥וּ בִנְדִיבִ֑ים בְּבֶן־ אָדָ֓ם ׀ שֶׁ֤אֵֽין ל֥וֹ תְשׁוּעָֽה׃ (עא) תֵּצֵ֣א ר֭וּחוֹ יָשֻׁ֣ב לְאַדְמָת֑וֹ בַּיּ֥וֹם הַ֝ה֗וּא אָבְד֥וּ עֶשְׁתֹּנֹתָֽיו׃ (עב) אַשְׁרֵ֗י שֶׁ֤אֵ֣ל יַעֲקֹ֣ב בְּעֶזְר֑וֹ שִׂ֝בְר֗וֹ עַל־ יְהוָ֥ה אֱלֹהָֽיו׃ (עג) עֹשֶׂ֤ה ׀ שָׁ֘מַ֤יִם וָאָ֗רֶץ אֶת־ הַיָּ֥ם וְאֶת־ כָּל־ אֲשֶׁר־ בָּ֑ם הַשֹּׁמֵ֖ר אֱמֶ֣ת לְעוֹלָֽם׃ (עד) עֹשֶׂ֤ה מִשְׁפָּ֨ט ׀ לָעֲשׁוּקִ֗ים נֹתֵ֣ן לֶ֭חֶם לָרְעֵבִ֑ים יְ֝הוָ֗ה מַתִּ֥יר אֲסוּרִֽים׃ (עה) יְהוָ֤ה ׀ פֹּ֘קֵ֤חַ עִוְרִ֗ים יְ֭הוָה זֹקֵ֣ף כְּפוּפִ֑ים יְ֝הוָ֗ה אֹהֵ֥ב צַדִּיקִֽים׃ (עו) יְהוָ֤ה ׀ שֹׁ֘מֵ֤ר אֶת־ גֵּרִ֗ים יָת֣וֹם וְאַלְמָנָ֣ה יְעוֹדֵ֑ד וְדֶ֖רֶךְ רְשָׁעִ֣ים יְעַוֵּֽת׃ (עז) יִמְלֹ֤ךְ יְהוָ֨ה ׀ לְעוֹלָ֗ם אֱלֹהַ֣יִךְ צִ֭יּוֹן לְדֹ֥ר וָדֹ֗ר הַֽלְלוּ־ יָֽהּ׃ (עח) הַ֥לְלוּ יָ֨הּ ׀ כִּי־ ט֭וֹב זַמְּרָ֣ה אֱלֹהֵ֑ינוּ כִּֽי־ נָ֝עִים נָאוָ֥ה תְהִלָּֽה׃ (עט) בּוֹנֵ֣ה יְרוּשָׁלִַ֣ם יְהוָ֑ה נִדְחֵ֖י יִשְׂרָאֵ֣ל יְכַנֵּֽס׃ (פ) הָ֭רֹפֵא לִשְׁב֣וּרֵי לֵ֑ב וּ֝מְחַבֵּ֗שׁ לְעַצְּבוֹתָֽם׃ (פא) מוֹנֶ֣ה מִ֭סְפָּר לַכּוֹכָבִ֑ים לְ֝כֻלָּ֗ם שֵׁמ֥וֹת יִקְרָֽא׃ (פב) גָּד֣וֹל אֲדוֹנֵ֣ינוּ וְרַב־ כֹּ֑חַ לִ֝תְבוּנָת֗וֹ אֵ֣ין מִסְפָּֽר׃ (פג) מְעוֹדֵ֣ד עֲנָוִ֣ים יְהוָ֑ה מַשְׁפִּ֖יל רְשָׁעִ֣ים עֲדֵי־ אָֽרֶץ׃ (פד) עֱנ֣וּ לַיהוָ֣ה בְּתוֹדָ֑ה זַמְּר֖וּ לֵאלֹהֵ֣ינוּ בְכִנּֽוֹר׃ (פה) הַֽמְכַסֶּ֬ה שָׁמַ֨יִם ׀ בְּעָבִ֗ים הַמֵּכִ֣ין לָאָ֣רֶץ מָטָ֑ר הַמַּצְמִ֖יחַ הָרִ֣ים חָצִֽיר׃ (פו) נוֹתֵ֣ן לִבְהֵמָ֣ה לַחְמָ֑הּ לִבְנֵ֥י עֹ֝רֵ֗ב אֲשֶׁ֣ר יִקְרָֽאוּ׃ (פז) לֹ֤א בִגְבוּרַ֣ת הַסּ֣וּס יֶחְפָּ֑ץ לֹֽא־ בְשׁוֹקֵ֖י הָאִ֣ישׁ יִרְצֶֽה׃ (פח) רוֹצֶ֣ה יְ֭הוָה אֶת־ יְרֵאָ֑יו אֶת־ הַֽמְיַחֲלִ֥ים לְחַסְדּֽוֹ׃ (פט) שַׁבְּחִ֣י יְ֭רוּשָׁלִַם אֶת־ יְהוָ֑ה הַֽלְלִ֖י אֱלֹהַ֣יִךְ צִיּֽוֹן׃ (צ) כִּֽי־ חִ֭זַּק בְּרִיחֵ֣י שְׁעָרָ֑יִךְ בֵּרַ֖ךְ בָּנַ֣יִךְ בְּקִרְבֵּֽךְ׃ (צא) הַשָּׂם־ גְּבוּלֵ֥ךְ שָׁל֑וֹם חֵ֥לֶב חִ֝טִּ֗ים יַשְׂבִּיעֵֽךְ׃ (צב) הַשֹּׁלֵ֣חַ אִמְרָת֣וֹ אָ֑רֶץ עַד־ מְ֝הֵרָ֗ה יָר֥וּץ דְּבָרֽוֹ׃ (צג) הַנֹּתֵ֣ן שֶׁ֣לֶג כַּצָּ֑מֶר כְּ֝פ֗וֹר כָּאֵ֥פֶר יְפַזֵּֽר׃ (צד) מַשְׁלִ֣יךְ קַֽרְח֣וֹ כְפִתִּ֑ים לִפְנֵ֥י קָ֝רָת֗וֹ מִ֣י יַעֲמֹֽד׃ (צה) יִשְׁלַ֣ח דְּבָר֣וֹ וְיַמְסֵ֑ם יַשֵּׁ֥ב ר֝וּח֗וֹ יִזְּלוּ־ מָֽיִם׃ (צו) מַגִּ֣יד (דברו) דְּבָרָ֣יו לְיַעֲקֹ֑ב חֻקָּ֥יו וּ֝מִשְׁפָּטָ֗יו לְיִשְׂרָאֵֽל׃ (צז) לֹ֘א עָ֤שָׂה כֵ֨ן ׀ לְכָל־ גּ֗וֹי וּמִשְׁפָּטִ֥ים בַּל־ יְדָע֗וּם הַֽלְלוּ־ יָֽהּ׃ (צח) הַ֥לְלוּ יָ֨הּ ׀ הַֽלְל֣וּ אֶת־ יְ֭הוָה מִן־ הַשָּׁמַ֑יִם הַֽ֝לְל֗וּהוּ בַּמְּרוֹמִֽים׃ (צט) הַֽלְל֥וּהוּ כָל־ מַלְאָכָ֑יו הַֽ֝לְל֗וּהוּ כָּל־ (צבאו) צְבָאָֽיו׃ (ק) הַֽ֭לְלוּהוּ שֶׁ֣מֶשׁ וְיָרֵ֑חַ הַ֝לְל֗וּהוּ כָּל־ כּ֥וֹכְבֵי אֽוֹר׃ (קא) הַֽ֭לְלוּהוּ שְׁמֵ֣י הַשָּׁמָ֑יִם וְ֝הַמַּ֗יִם אֲשֶׁ֤ר ׀ מֵעַ֬ל הַשָּׁמָֽיִם׃ (קב) יְֽ֭הַֽלְלוּ אֶת־ שֵׁ֣ם יְהוָ֑ה כִּ֤י ה֭וּא צִוָּ֣ה וְנִבְרָֽאוּ׃ (קג) וַיַּעֲמִידֵ֣ם לָעַ֣ד לְעוֹלָ֑ם חָק־ נָ֝תַ֗ן וְלֹ֣א יַעֲבֽוֹר׃ (קד) הַֽלְל֣וּ אֶת־ יְ֭הוָה מִן־ הָאָ֑רֶץ תַּ֝נִּינִ֗ים וְכָל־ תְּהֹמֽוֹת׃ (קה) אֵ֣שׁ וּ֭בָרָד שֶׁ֣לֶג וְקִיט֑וֹר ר֥וּחַ סְ֝עָרָ֗ה עֹשָׂ֥ה דְבָרֽוֹ׃ (קו) הֶהָרִ֥ים וְכָל־ גְּבָע֑וֹת עֵ֥ץ פְּ֝רִ֗י וְכָל־ אֲרָזִֽים׃ (קז) הַֽחַיָּ֥ה וְכָל־ בְּהֵמָ֑ה רֶ֝֗מֶשׂ וְצִפּ֥וֹר כָּנָֽף׃ (קח) מַלְכֵי־ אֶ֭רֶץ וְכָל־ לְאֻמִּ֑ים שָׂ֝רִ֗ים וְכָל־ שֹׁ֥פְטֵי אָֽרֶץ׃ (קט) בַּחוּרִ֥ים וְגַם־ בְּתוּל֑וֹת זְ֝קֵנִ֗ים עִם־ נְעָרִֽים׃ (קי) יְהַלְל֤וּ ׀ אֶת־ שֵׁ֬ם יְהוָ֗ה כִּֽי־ נִשְׂגָּ֣ב שְׁמ֣וֹ לְבַדּ֑וֹ ה֝וֹד֗וֹ עַל־ אֶ֥רֶץ וְשָׁמָֽיִם׃ (קיא) וַיָּ֤רֶם קֶ֨רֶן ׀ לְעַמּ֡וֹ תְּהִלָּ֤ה לְֽכָל־ חֲסִידָ֗יו לִבְנֵ֣י יִ֭שְׂרָאֵל עַֽם־ קְרֹב֗וֹ הַֽלְלוּ־ יָֽהּ׃ (קיב) הַ֥לְלוּ יָ֨הּ ׀ שִׁ֣ירוּ לַֽ֭יהוָה שִׁ֣יר חָדָ֑שׁ תְּ֝הִלָּת֗וֹ בִּקְהַ֥ל חֲסִידִֽים׃ (קיג) יִשְׂמַ֣ח יִשְׂרָאֵ֣ל בְּעֹשָׂ֑יו בְּנֵֽי־ צִ֝יּ֗וֹן יָגִ֥ילוּ בְמַלְכָּֽם׃ (קיד) יְהַֽלְל֣וּ בִּרְקִ֥יעַ עֻזּֽוֹ׃ (קכב) הַֽלְל֥וּהוּ בִגְבוּרֹתָ֑יו הַֽ֝לְל֗וּהוּ כְּרֹ֣ב גֻּדְלֽוֹ׃ (קכג) הַֽ֭לְלוּהוּ בְּתֵ֣קַע שׁוֹפָ֑ר הַֽ֝לְל֗וּהוּ בְּנֵ֣בֶל וְכִנּֽוֹר׃ (קכד) הַֽ֭לְלוּהוּ בְתֹ֣ף וּמָח֑וֹל
