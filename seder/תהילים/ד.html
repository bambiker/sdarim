(א) יְֽהוָ֗ה עֹ֭ז לְעַמּ֣וֹ יִתֵּ֑ן יְהוָ֓ה ׀ יְבָרֵ֖ךְ אֶת־ עַמּ֣וֹ בַשָּׁלֽוֹם׃ (ב) מִזְמ֡וֹר שִׁיר־ חֲנֻכַּ֖ת הַבַּ֣יִת לְדָוִֽד׃ (ג) אֲרוֹמִמְךָ֣ יְ֭הוָה כִּ֣י דִלִּיתָ֑נִי וְלֹא־ שִׂמַּ֖חְתָּ אֹיְבַ֣י לִֽי׃ (ד) יְהוָ֥ה אֱלֹהָ֑י שִׁוַּ֥עְתִּי אֵ֝לֶ֗יךָ וַתִּרְפָּאֵֽנִי׃ (ה) יְֽהוָ֗ה הֶֽעֱלִ֣יתָ מִן־ שְׁא֣וֹל נַפְשִׁ֑י חִ֝יִּיתַ֗נִי (מיורדי)־ מִיָּֽרְדִי־ בֽוֹר׃ (ו) זַמְּר֣וּ לַיהוָ֣ה חֲסִידָ֑יו וְ֝הוֹד֗וּ לְזֵ֣כֶר קָדְשֽׁוֹ׃ (ז) כִּ֤י רֶ֨גַע ׀ בְּאַפּוֹ֮ חַיִּ֪ים בִּרְצ֫וֹנ֥וֹ בָּ֭עֶרֶב יָלִ֥ין בֶּ֗כִי וְלַבֹּ֥קֶר רִנָּֽה׃ (ח) וַ֭אֲנִי אָמַ֣רְתִּי בְשַׁלְוִ֑י בַּל־ אֶמּ֥וֹט לְעוֹלָֽם׃ (ט) יְֽהוָ֗ה בִּרְצוֹנְךָ֮ הֶעֱמַ֪דְתָּה לְֽהַרְרִ֫י עֹ֥ז הִסְתַּ֥רְתָּ פָנֶ֗יךָ הָיִ֥יתִי נִבְהָֽל׃ (י) אֵלֶ֣יךָ יְהוָ֣ה אֶקְרָ֑א וְאֶל־ אֲ֝דֹנָ֗י אֶתְחַנָּֽן׃ (יא) מַה־ בֶּ֥צַע בְּדָמִי֮ בְּרִדְתִּ֪י אֶ֫ל־ שָׁ֥חַת הֲיוֹדְךָ֥ עָפָ֑ר הֲיַגִּ֥יד אֲמִתֶּֽךָ׃ (יב) שְׁמַע־ יְהוָ֥ה וְחָנֵּ֑נִי יְ֝הוָה הֱ‍ֽיֵה־ עֹזֵ֥ר לִֽי׃ (יג) הָפַ֣כְתָּ מִסְפְּדִי֮ לְמָח֪וֹל לִ֥י פִּתַּ֥חְתָּ שַׂקִּ֑י וַֽתְּאַזְּרֵ֥נִי שִׂמְחָֽה׃ (יד) לְמַ֤עַן ׀ יְזַמֶּרְךָ֣ כָ֭בוֹד וְלֹ֣א יִדֹּ֑ם יְהוָ֥ה אֱ֝לֹהַ֗י לְעוֹלָ֥ם אוֹדֶֽךָּ׃ (טו) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (טז) בְּךָ֖ יְהוָ֣ה חָ֭סִיתִי אַל־ אֵב֣וֹשָׁה לְעוֹלָ֑ם בְּצִדְקָתְךָ֥ פַלְּטֵֽנִי׃ (יז) הַטֵּ֤ה אֵלַ֨י ׀ אָזְנְךָ֮ מְהֵרָ֪ה הַצִּ֫ילֵ֥נִי הֱיֵ֤ה לִ֨י ׀ לְֽצוּר־ מָ֭עוֹז לְבֵ֥ית מְצוּד֗וֹת לְהוֹשִׁיעֵֽנִי׃ (יח) כִּֽי־ סַלְעִ֣י וּמְצוּדָתִ֣י אָ֑תָּה וּלְמַ֥עַן שִׁ֝מְךָ֗ תַּֽנְחֵ֥נִי וּֽתְנַהֲלֵֽנִי׃ (יט) תּוֹצִיאֵ֗נִי מֵרֶ֣שֶׁת ז֭וּ טָ֣מְנוּ לִ֑י כִּֽי־ אַ֝תָּה מָֽעוּזִּֽי׃ (כ) בְּיָדְךָ֮ אַפְקִ֪יד ר֫וּחִ֥י פָּדִ֖יתָה אוֹתִ֥י יְהוָ֗ה אֵ֣ל אֱמֶֽת׃ (כא) שָׂנֵ֗אתִי הַשֹּׁמְרִ֥ים הַבְלֵי־ שָׁ֑וְא וַ֝אֲנִ֗י אֶל־ יְהוָ֥ה בָּטָֽחְתִּי׃ (כב) אָגִ֥ילָה וְאֶשְׂמְחָ֗ה בְּחַ֫סְדֶּ֥ךָ אֲשֶׁ֣ר רָ֭אִיתָ אֶת־ עָנְיִ֑י יָ֝דַ֗עְתָּ בְּצָר֥וֹת נַפְשִֽׁי׃ (כג) וְלֹ֣א הִ֭סְגַּרְתַּנִי בְּיַד־ אוֹיֵ֑ב הֶֽעֱמַ֖דְתָּ בַמֶּרְחָ֣ב רַגְלָֽי׃ (כד) חָנֵּ֥נִי יְהוָה֮ כִּ֤י צַ֫ר־ לִ֥י עָשְׁשָׁ֖ה בְכַ֥עַס עֵינִ֗י נַפְשִׁ֥י וּבִטְנִֽי׃ (כה) כִּ֤י כָל֪וּ בְיָג֡וֹן חַיַּי֮ וּשְׁנוֹתַ֪י בַּאֲנָ֫חָ֥ה כָּשַׁ֣ל בַּעֲוֺנִ֣י כֹחִ֑י וַעֲצָמַ֥י עָשֵֽׁשׁוּ׃ (כו) מִכָּל־ צֹרְרַ֨י הָיִ֪יתִי חֶרְפָּ֡ה וְלִשֲׁכֵנַ֨י ׀ מְאֹד֮ וּפַ֪חַד לִֽמְיֻדָּ֫עָ֥י רֹאַ֥י בַּח֑וּץ נָדְד֥וּ מִמֶּֽנִּי׃ (כז) נִ֭שְׁכַּחְתִּי כְּמֵ֣ת מִלֵּ֑ב הָ֝יִ֗יתִי כִּכְלִ֥י אֹבֵֽד׃ (כח) כִּ֤י שָׁמַ֨עְתִּי ׀ דִּבַּ֥ת רַבִּים֮ מָג֪וֹר מִסָּ֫בִ֥יב בְּהִוָּסְדָ֣ם יַ֣חַד עָלַ֑י לָקַ֖חַת נַפְשִׁ֣י זָמָֽמוּ׃ (כט) וַאֲנִ֤י ׀ עָלֶ֣יךָ בָטַ֣חְתִּי יְהוָ֑ה אָ֝מַ֗רְתִּי אֱלֹהַ֥י אָֽתָּה׃ (ל) בְּיָדְךָ֥ עִתֹּתָ֑י הַצִּ֘ילֵ֤נִי מִיַּד־ א֝וֹיְבַ֗י וּמֵרֹדְפָֽי׃ (לא) הָאִ֣ירָה פָ֭נֶיךָ עַל־ עַבְדֶּ֑ךָ ה֖וֹשִׁיעֵ֣נִי בְחַסְדֶּֽךָ׃ (לב) יְֽהוָ֗ה אַל־ אֵ֭בוֹשָׁה כִּ֣י קְרָאתִ֑יךָ יֵבֹ֥שׁוּ רְ֝שָׁעִ֗ים יִדְּמ֥וּ לִשְׁאֽוֹל׃ (לג) תֵּ֥אָלַ֗מְנָה שִׂפְתֵ֫י שָׁ֥קֶר הַדֹּבְר֖וֹת עַל־ צַדִּ֥יק עָתָ֗ק בְּגַאֲוָ֥ה וָבֽוּז׃ (לד) מָ֤ה רַֽב־ טוּבְךָ֮ אֲשֶׁר־ צָפַ֪נְתָּ לִּֽירֵ֫אֶ֥יךָ פָּ֭עַלְתָּ לַחֹסִ֣ים בָּ֑ךְ נֶ֝֗גֶד בְּנֵ֣י אָדָם׃ (לה) תַּסְתִּירֵ֤ם ׀ בְּסֵ֥תֶר פָּנֶיךָ֮ מֵֽרֻכְסֵ֫י אִ֥ישׁ תִּצְפְּנֵ֥ם בְּסֻכָּ֗ה מֵרִ֥יב לְשֹׁנֽוֹת׃ (לו) בָּר֥וּךְ יְהוָ֑ה כִּ֥י הִפְלִ֘יא חַסְדּ֥וֹ לִ֝֗י בְּעִ֣יר מָצֽוֹר׃ (לז) וַאֲנִ֤י ׀ אָ֘מַ֤רְתִּי בְחָפְזִ֗י נִגְרַזְתִּי֮ מִנֶּ֪גֶד עֵ֫ינֶ֥יךָ אָכֵ֗ן שָׁ֭מַעְתָּ ק֥וֹל תַּחֲנוּנַ֗י בְּשַׁוְּעִ֥י אֵלֶֽיךָ׃ (לח) אֶֽהֱב֥וּ אֶת־ יְהוָ֗ה כָּֽל־ חֲסִ֫ידָ֥יו אֱ֭מוּנִים נֹצֵ֣ר יְהוָ֑ה וּמְשַׁלֵּ֥ם עַל־ יֶ֝֗תֶר עֹשֵׂ֥ה גַאֲוָֽה׃ (לט) חִ֭זְקוּ וְיַאֲמֵ֣ץ לְבַבְכֶ֑ם כָּל־ הַ֝מְיַחֲלִ֗ים לַיהוָֽה׃ (מ) לְדָוִ֗ד מַ֫שְׂכִּ֥יל אַשְׁרֵ֥י נְֽשׂוּי־ פֶּ֗שַׁע כְּס֣וּי חֲטָאָֽה׃ (מא) אַ֥שְֽׁרֵי אָדָ֗ם לֹ֤א יַחְשֹׁ֬ב יְהוָ֣ה ל֣וֹ עָוֺ֑ן וְאֵ֖ין בְּרוּח֣וֹ רְמִיָּה׃ (מב) כִּֽי־ הֶ֭חֱרַשְׁתִּי בָּל֣וּ עֲצָמָ֑י בְּ֝שַׁאֲגָתִ֗י כָּל־ הַיּֽוֹם׃ (מג) כִּ֤י ׀ יוֹמָ֣ם וָלַיְלָה֮ תִּכְבַּ֥ד עָלַ֗י יָ֫דֶ֥ךָ נֶהְפַּ֥ךְ לְשַׁדִּ֑י בְּחַרְבֹ֖נֵי קַ֣יִץ סֶֽלָה׃ (מד) חַטָּאתִ֨י אוֹדִ֪יעֲךָ֡ וַעֲוֺ֘נִ֤י לֹֽא־ כִסִּ֗יתִי אָמַ֗רְתִּי אוֹדֶ֤ה עֲלֵ֣י פְ֭שָׁעַי לַיהוָ֑ה וְאַתָּ֨ה נָ֘שָׂ֤אתָ עֲוֺ֖ן חַטָּאתִ֣י סֶֽלָה׃ (מה) עַל־ זֹ֡את יִתְפַּלֵּ֬ל כָּל־ חָסִ֨יד ׀ אֵלֶיךָ֮ לְעֵ֪ת מְ֫צֹ֥א רַ֗ק לְ֭שֵׁטֶף מַ֣יִם רַבִּ֑ים אֵ֝לָ֗יו לֹ֣א יַגִּֽיעוּ׃ (מו) אַתָּ֤ה ׀ סֵ֥תֶר לִי֮ מִצַּ֪ר תִּ֫צְּרֵ֥נִי רָנֵּ֥י פַלֵּ֑ט תְּס֖וֹבְבֵ֣נִי סֶֽלָה׃ (מז) אַשְׂכִּֽילְךָ֨ ׀ וְֽאוֹרְךָ֗ בְּדֶֽרֶךְ־ ז֥וּ תֵלֵ֑ךְ אִֽיעֲצָ֖ה עָלֶ֣יךָ עֵינִֽי׃ (מח) אַל־ תִּֽהְי֤וּ ׀ כְּס֥וּס כְּפֶרֶד֮ אֵ֤ין הָ֫בִ֥ין בְּמֶֽתֶג־ וָרֶ֣סֶן עֶדְי֣וֹ לִבְל֑וֹם בַּ֝֗ל קְרֹ֣ב אֵלֶֽיךָ׃ (מט) רַבִּ֥ים מַכְאוֹבִ֗ים לָרָ֫שָׁ֥ע וְהַבּוֹטֵ֥חַ בַּיהוָ֑ה חֶ֝֗סֶד יְסוֹבְבֶֽנּוּ׃ (נ) שִׂמְח֬וּ בַֽיהוָ֣ה וְ֭גִילוּ צַדִּיקִ֑ים וְ֝הַרְנִ֗ינוּ כָּל־ יִשְׁרֵי־ לֵֽב׃ (נא) רַנְּנ֣וּ צַ֭דִּיקִים בַּֽיהוָ֑ה לַ֝יְשָׁרִ֗ים נָאוָ֥ה תְהִלָּֽה׃ (נב) הוֹד֣וּ לַיהוָ֣ה בְּכִנּ֑וֹר בְּנֵ֥בֶל עָ֝שׂ֗וֹר זַמְּרוּ־ לֽוֹ׃ (נג) שִֽׁירוּ־ ל֭וֹ שִׁ֣יר חָדָ֑שׁ הֵיטִ֥יבוּ נַ֝גֵּ֗ן בִּתְרוּעָֽה׃ (נד) כִּֽי־ יָשָׁ֥ר דְּבַר־ יְהוָ֑ה וְכָל־ מַ֝עֲשֵׂ֗הוּ בֶּאֱמוּנָֽה׃ (נה) אֹ֭הֵב צְדָקָ֣ה וּמִשְׁפָּ֑ט חֶ֥סֶד יְ֝הוָ֗ה מָלְאָ֥ה הָאָֽרֶץ׃ (נו) בִּדְבַ֣ר יְ֭הוָה שָׁמַ֣יִם נַעֲשׂ֑וּ וּבְר֥וּחַ פִּ֝֗יו כָּל־ צְבָאָֽם׃ (נז) כֹּנֵ֣ס כַּ֭נֵּד מֵ֣י הַיָּ֑ם נֹתֵ֖ן בְּאֹצָר֣וֹת תְּהוֹמֽוֹת׃ (נח) יִֽירְא֣וּ מֵ֭יְהוָה כָּל־ הָאָ֑רֶץ מִמֶּ֥נּוּ יָ֝ג֗וּרוּ כָּל־ יֹשְׁבֵ֥י תֵבֵֽל׃ (נט) כִּ֤י ה֣וּא אָמַ֣ר וַיֶּ֑הִי הֽוּא־ צִ֝וָּ֗ה וַֽיַּעֲמֹֽד׃ (ס) יְֽהוָ֗ה הֵפִ֥יר עֲצַת־ גּוֹיִ֑ם הֵ֝נִ֗יא מַחְשְׁב֥וֹת עַמִּֽים׃ (סא) עֲצַ֣ת יְ֭הוָה לְעוֹלָ֣ם תַּעֲמֹ֑ד מַחְשְׁב֥וֹת לִ֝בּ֗וֹ לְדֹ֣ר וָדֹֽר׃ (סב) אַשְׁרֵ֣י הַ֭גּוֹי אֲשֶׁר־ יְהוָ֣ה אֱלֹהָ֑יו הָעָ֓ם ׀ בָּחַ֖ר לְנַחֲלָ֣ה לֽוֹ׃ (סג) מִ֭שָּׁמַיִם הִבִּ֣יט יְהוָ֑ה רָ֝אָ֗ה אֶֽת־ כָּל־ בְּנֵ֥י הָאָדָֽם׃ (סד) מִֽמְּכוֹן־ שִׁבְתּ֥וֹ הִשְׁגִּ֑יחַ אֶ֖ל כָּל־ יֹשְׁבֵ֣י הָאָֽרֶץ׃ (סה) הַיֹּצֵ֣ר יַ֣חַד לִבָּ֑ם הַ֝מֵּבִ֗ין אֶל־ כָּל־ מַעֲשֵׂיהֶֽם׃ (סו) אֵֽין־ הַ֭מֶּלֶךְ נוֹשָׁ֣ע בְּרָב־ חָ֑יִל גִּ֝בּ֗וֹר לֹֽא־ יִנָּצֵ֥ל בְּרָב־ כֹּֽחַ׃ (סז) שֶׁ֣קֶר הַ֭סּוּס לִתְשׁוּעָ֑ה וּבְרֹ֥ב חֵ֝יל֗וֹ לֹ֣א יְמַלֵּֽט׃ (סח) הִנֵּ֤ה עֵ֣ין יְ֭הוָה אֶל־ יְרֵאָ֑יו לַֽמְיַחֲלִ֥ים לְחַסְדּֽוֹ׃ (סט) לְהַצִּ֣יל מִמָּ֣וֶת נַפְשָׁ֑ם וּ֝לְחַיּוֹתָ֗ם בָּרָעָֽב׃ (ע) נַ֭פְשֵׁנוּ חִכְּתָ֣ה לַֽיהוָ֑ה עֶזְרֵ֖נוּ וּמָגִנֵּ֣נוּ הֽוּא׃ (עא) כִּי־ ב֭וֹ יִשְׂמַ֣ח לִבֵּ֑נוּ כִּ֤י בְשֵׁ֖ם קָדְשׁ֣וֹ בָטָֽחְנוּ׃ (עב) יְהִֽי־ חַסְדְּךָ֣ יְהוָ֣ה עָלֵ֑ינוּ כַּ֝אֲשֶׁ֗ר יִחַ֥לְנוּ לָֽךְ׃ (עג) לְדָוִ֗ד בְּשַׁנּוֹת֣וֹ אֶת־ טַ֭עְמוֹ לִפְנֵ֣י אֲבִימֶ֑לֶךְ וַֽ֝יְגָרֲשֵׁ֗הוּ וַיֵּלַֽךְ׃ (עד) אֲבָרֲכָ֣ה אֶת־ יְהוָ֣ה בְּכָל־ עֵ֑ת תָּ֝מִ֗יד תְּֽהִלָּת֥וֹ בְּפִֽי׃ (עה) בַּ֭יהוָה תִּתְהַלֵּ֣ל נַפְשִׁ֑י יִשְׁמְע֖וּ עֲנָוִ֣ים וְיִשְׂמָֽחוּ׃ (עו) גַּדְּל֣וּ לַיהוָ֣ה אִתִּ֑י וּנְרוֹמְמָ֖ה שְׁמ֣וֹ יַחְדָּֽו׃ (עז) דָּרַ֣שְׁתִּי אֶת־ יְהוָ֣ה וְעָנָ֑נִי וּמִכָּל־ מְ֝גוּרוֹתַ֗י הִצִּילָֽנִי׃ (עח) הִבִּ֣יטוּ אֵלָ֣יו וְנָהָ֑רוּ וּ֝פְנֵיהֶ֗ם אַל־ יֶחְפָּֽרוּ׃ (עט) זֶ֤ה עָנִ֣י קָ֭רָא וַיהוָ֣ה שָׁמֵ֑עַ וּמִכָּל־ צָ֝רוֹתָ֗יו הוֹשִׁיעֽוֹ׃ (פ) חֹנֶ֤ה מַלְאַךְ־ יְהוָ֓ה סָ֘בִ֤יב לִֽירֵאָ֗יו וַֽיְחַלְּצֵֽם׃ (פא) טַעֲמ֣וּ וּ֭רְאוּ כִּי־ ט֣וֹב יְהוָ֑ה אַֽשְׁרֵ֥י הַ֝גֶּ֗בֶר יֶחֱסֶה־ בּֽוֹ׃ (פב) יְר֣אוּ אֶת־ יְהוָ֣ה קְדֹשָׁ֑יו כִּי־ אֵ֥ין מַ֝חְס֗וֹר לִירֵאָֽיו׃ (פג) כְּ֭פִירִים רָשׁ֣וּ וְרָעֵ֑בוּ וְדֹרְשֵׁ֥י יְ֝הוָ֗ה לֹא־ יַחְסְר֥וּ כָל־ טֽוֹב׃ (פד) לְֽכוּ־ בָ֭נִים שִׁמְעוּ־ לִ֑י יִֽרְאַ֥ת יְ֝הוָ֗ה אֲלַמֶּדְכֶֽם׃ (פה) מִֽי־ הָ֭אִישׁ הֶחָפֵ֣ץ חַיִּ֑ים אֹהֵ֥ב יָ֝מִ֗ים לִרְא֥וֹת טֽוֹב׃ (פו) נְצֹ֣ר לְשׁוֹנְךָ֣ מֵרָ֑ע וּ֝שְׂפָתֶ֗יךָ מִדַּבֵּ֥ר מִרְמָֽה׃ (פז) ס֣וּר מֵ֭רָע וַעֲשֵׂה־ ט֑וֹב בַּקֵּ֖שׁ שָׁל֣וֹם וְרָדְפֵֽהוּ׃ (פח) עֵינֵ֣י יְ֭הוָה אֶל־ צַדִּיקִ֑ים וְ֝אָזְנָ֗יו אֶל־ שַׁוְעָתָֽם׃ (פט) פְּנֵ֣י יְ֭הוָה בְּעֹ֣שֵׂי רָ֑ע לְהַכְרִ֖ית מֵאֶ֣רֶץ זִכְרָֽם׃ (צ) צָעֲק֣וּ וַיהוָ֣ה שָׁמֵ֑עַ וּמִכָּל־ צָ֝רוֹתָ֗ם הִצִּילָֽם׃ (צא) קָר֣וֹב יְ֭הוָה לְנִשְׁבְּרֵי־ לֵ֑ב וְֽאֶת־ דַּכְּאֵי־ ר֥וּחַ יוֹשִֽׁיעַ׃ (צב) רַ֭בּוֹת רָע֣וֹת צַדִּ֑יק וּ֝מִכֻּלָּ֗ם יַצִּילֶ֥נּוּ יְהוָֽה׃ (צג) שֹׁמֵ֥ר כָּל־ עַצְמוֹתָ֑יו אַחַ֥ת מֵ֝הֵ֗נָּה לֹ֣א נִשְׁבָּֽרָה׃ (צד) תְּמוֹתֵ֣ת רָשָׁ֣ע רָעָ֑ה וְשֹׂנְאֵ֖י צַדִּ֣יק יֶאְשָֽׁמוּ׃ (צה) פּוֹדֶ֣ה יְ֭הוָה נֶ֣פֶשׁ עֲבָדָ֑יו וְלֹ֥א יֶ֝אְשְׁמ֗וּ כָּֽל־ הַחֹסִ֥ים בּֽוֹ׃ (צו) לְדָוִ֨ד ׀ רִיבָ֣ה יְ֭הוָה אֶת־ יְרִיבַ֑י לְ֝חַ֗ם אֶת־ לֹֽחֲמָֽי׃ (צז) הַחֲזֵ֣ק מָגֵ֣ן וְצִנָּ֑ה וְ֝ק֗וּמָה בְּעֶזְרָתִֽי׃ (צח) וְהָ֘רֵ֤ק חֲנִ֣ית וּ֭סְגֹר לִקְרַ֣את רֹדְפָ֑י אֱמֹ֥ר לְ֝נַפְשִׁ֗י יְֽשֻׁעָתֵ֥ךְ אָֽנִי׃ (צט) יֵבֹ֣שׁוּ וְיִכָּלְמוּ֮ מְבַקְשֵׁ֪י נַ֫פְשִׁ֥י יִסֹּ֣גוּ אָח֣וֹר וְיַחְפְּר֑וּ חֹ֝שְׁבֵ֗י רָעָתִֽי׃ (ק) יִֽהְי֗וּ כְּמֹ֥ץ לִפְנֵי־ ר֑וּחַ וּמַלְאַ֖ךְ יְהוָ֣ה דּוֹחֶֽה׃ (קא) יְֽהִי־ דַרְכָּ֗ם חֹ֥שֶׁךְ וַחֲלַקְלַקּ֑וֹת וּמַלְאַ֥ךְ יְ֝הוָ֗ה רֹדְפָֽם׃ (קב) כִּֽי־ חִנָּ֣ם טָֽמְנוּ־ לִ֭י שַׁ֣חַת רִשְׁתָּ֑ם חִ֝נָּ֗ם חָפְר֥וּ לְנַפְשִֽׁי׃ (קג) תְּבוֹאֵ֣הוּ שׁוֹאָה֮ לֹֽא־ יֵ֫דָ֥ע וְרִשְׁתּ֣וֹ אֲשֶׁר־ טָמַ֣ן תִּלְכְּד֑וֹ בְּ֝שׁוֹאָ֗ה יִפָּל־ בָּֽהּ׃ (קד) וְ֭נַפְשִׁי תָּגִ֣יל בַּיהוָ֑ה תָּ֝שִׂישׂ בִּישׁוּעָתֽוֹ׃ (קה) כָּ֥ל עַצְמוֹתַ֨י ׀ תֹּאמַרְנָה֮ יְהוָ֗ה מִ֥י כָ֫מ֥וֹךָ מַצִּ֣יל עָ֭נִי מֵחָזָ֣ק מִמֶּ֑נּוּ וְעָנִ֥י וְ֝אֶבְי֗וֹן מִגֹּזְלֽוֹ׃ (קו) יְ֭קוּמוּן עֵדֵ֣י חָמָ֑ס אֲשֶׁ֥ר לֹא־ יָ֝דַ֗עְתִּי יִשְׁאָלֽוּנִי׃ (קז) יְשַׁלְּמ֣וּנִי רָ֭עָה תַּ֥חַת טוֹבָ֗ה שְׁכ֣וֹל לְנַפְשִֽׁי׃ (קח) וַאֲנִ֤י ׀ בַּחֲלוֹתָ֡ם לְב֬וּשִׁי שָׂ֗ק עִנֵּ֣יתִי בַצּ֣וֹם נַפְשִׁ֑י וּ֝תְפִלָּתִ֗י עַל־ חֵיקִ֥י תָשֽׁוּב׃ (קט) כְּרֵֽעַ־ כְּאָ֣ח לִ֭י הִתְהַלָּ֑כְתִּי כַּאֲבֶל־ אֵ֝֗ם קֹדֵ֥ר שַׁחֽוֹתִי׃ (קי) וּבְצַלְעִי֮ שָׂמְח֪וּ וְֽנֶאֱ֫סָ֥פוּ נֶאֶסְפ֬וּ עָלַ֣י נֵ֭כִים וְלֹ֣א יָדַ֑עְתִּי קָֽרְע֥וּ וְלֹא־ דָֽמּוּ׃ (קיא) בְּ֭חַנְפֵי לַעֲגֵ֣י מָע֑וֹג חָרֹ֖ק עָלַ֣י שִׁנֵּֽימוֹ׃ (קיב) אֲדֹנָי֮ כַּמָּ֪ה תִּ֫רְאֶ֥ה הָשִׁ֣יבָה נַ֭פְשִׁי מִשֹּׁאֵיהֶ֑ם מִ֝כְּפִירִ֗ים יְחִידָתִֽי׃ (קיג) א֭וֹדְךָ בְּקָהָ֣ל רָ֑ב בְּעַ֖ם עָצ֣וּם אֲהַֽלְלֶֽךָּ׃ (קיד) אַֽל־ יִשְׂמְחוּ־ לִ֣י אֹיְבַ֣י שֶׁ֑קֶר שֹׂנְאַ֥י חִ֝נָּ֗ם יִקְרְצוּ־ עָֽיִן׃ (קטו) כִּ֤י לֹ֥א שָׁל֗וֹם יְדַ֫בֵּ֥רוּ וְעַ֥ל רִגְעֵי־ אֶ֑רֶץ דִּבְרֵ֥י מִ֝רְמוֹת יַחֲשֹׁבֽוּן׃ (קטז) וַיַּרְחִ֥יבוּ עָלַ֗י פִּ֫יהֶ֥ם אָ֭מְרוּ הֶאָ֣ח ׀ הֶאָ֑ח רָאֲתָ֥ה עֵינֵֽינוּ׃ (קיז) רָאִ֣יתָה יְ֭הוָה אַֽל־ תֶּחֱרַ֑שׁ אֲ֝דֹנָ֗י אֲל־ תִּרְחַ֥ק מִמֶּֽנִּי׃ (קיח) הָעִ֣ירָה וְ֭הָקִיצָה לְמִשְׁפָּטִ֑י אֱלֹהַ֖י וַֽאדֹנָ֣י לְרִיבִֽי׃ (קיט) שָׁפְטֵ֣נִי כְ֭צִדְקְךָ יְהוָ֥ה אֱלֹהָ֗י וְאַל־ יִשְׂמְחוּ־ לִֽי׃ (קכ) אַל־ יֹאמְר֣וּ בְ֭לִבָּם הֶאָ֣ח נַפְשֵׁ֑נוּ אַל־ יֹ֝אמְר֗וּ בִּֽלַּעֲנֽוּהוּ׃ (קכא) יֵ֘בֹ֤שׁוּ וְיַחְפְּר֨וּ ׀ יַחְדָּו֮ שְׂמֵחֵ֪י רָעָ֫תִ֥י יִֽלְבְּשׁוּ־ בֹ֥שֶׁת וּכְלִמָּ֑ה הַֽמַּגְדִּילִ֥ים עָלָֽי׃ (קכב) יָרֹ֣נּוּ וְיִשְׂמְחוּ֮ חֲפֵצֵ֪י צִ֫דְקִ֥י וְיֹאמְר֣וּ תָ֭מִיד יִגְדַּ֣ל יְהוָ֑ה הֶ֝חָפֵ֗ץ שְׁל֣וֹם עַבְדּֽוֹ׃
