תהילים סדר יב

פתיחה באתר · ספר תהילים כולו ·

(א) יְהוָ֥ה צְבָא֑וֹת אַֽשְׁרֵ֥י אָ֝דָ֗ם בֹּטֵ֥חַ בָּֽךְ׃ (ב) לַמְנַצֵּ֬חַ ׀ לִבְנֵי־ קֹ֬רַח מִזְמֽוֹר׃ (ג) רָצִ֣יתָ יְהוָ֣ה אַרְצֶ֑ךָ שַׁ֝֗בְתָּ (שבות) שְׁבִ֣ית יַעֲקֹֽב׃ (ד) נָ֭שָׂאתָ עֲוֺ֣ן עַמֶּ֑ךָ כִּסִּ֖יתָ כָל־ חַטָּאתָ֣ם סֶֽלָה׃ (ה) אָסַ֥פְתָּ כָל־ עֶבְרָתֶ֑ךָ הֱ֝שִׁיב֗וֹתָ מֵחֲר֥וֹן אַפֶּֽךָ׃ (ו) שׁ֭וּבֵנוּ אֱלֹהֵ֣י יִשְׁעֵ֑נוּ וְהָפֵ֖ר כַּֽעַסְךָ֣ עִמָּֽנוּ׃ (ז) הַלְעוֹלָ֥ם תֶּֽאֱנַף־ בָּ֑נוּ תִּמְשֹׁ֥ךְ אַ֝פְּךָ֗ לְדֹ֣ר וָדֹֽר׃ (ח) הֲֽלֹא־ אַ֭תָּה תָּשׁ֣וּב תְּחַיֵּ֑נוּ וְ֝עַמְּךָ֗ יִשְׂמְחוּ־ בָֽךְ׃ (ט) הַרְאֵ֣נוּ יְהוָ֣ה חַסְדֶּ֑ךָ וְ֝יֶשְׁעֲךָ֗ תִּתֶּן־ לָֽנוּ׃ (י) אֶשְׁמְעָ֗ה מַה־ יְדַבֵּר֮ הָאֵ֪ל ׀ יְה֫וָ֥ה כִּ֤י ׀ יְדַבֵּ֬ר שָׁל֗וֹם אֶל־ עַמּ֥וֹ וְאֶל־ חֲסִידָ֑יו וְֽאַל־ יָשׁ֥וּבוּ לְכִסְלָֽה׃ (יא) אַ֤ךְ ׀ קָר֣וֹב לִירֵאָ֣יו יִשְׁע֑וֹ לִשְׁכֹּ֖ן כָּב֣וֹד בְּאַרְצֵֽנוּ׃ (יב) חֶֽסֶד־ וֶאֱמֶ֥ת נִפְגָּ֑שׁוּ צֶ֖דֶק וְשָׁל֣וֹם נָשָֽׁקוּ׃ (יג) אֱ֭מֶת מֵאֶ֣רֶץ תִּצְמָ֑ח וְ֝צֶ֗דֶק מִשָּׁמַ֥יִם נִשְׁקָֽף׃ (יד) גַּם־ יְ֭הוָה יִתֵּ֣ן הַטּ֑וֹב וְ֝אַרְצֵ֗נוּ תִּתֵּ֥ן יְבוּלָֽהּ׃ (טו) צֶ֭דֶק לְפָנָ֣יו יְהַלֵּ֑ךְ וְיָשֵׂ֖ם לְדֶ֣רֶךְ פְּעָמָֽיו׃ (טז) תְּפִלָּ֗ה לְדָ֫וִ֥ד הַטֵּֽה־ יְהוָ֣ה אָזְנְךָ֣ עֲנֵ֑נִי כִּֽי־ עָנִ֖י וְאֶבְי֣וֹן אָֽנִי׃ (יז) שָֽׁמְרָ֣ה נַפְשִׁי֮ כִּֽי־ חָסִ֪יד אָ֥נִי הוֹשַׁ֣ע עַ֭בְדְּךָ אַתָּ֣ה אֱלֹהַ֑י הַבּוֹטֵ֥חַ אֵלֶֽיךָ׃ (יח) חָנֵּ֥נִי אֲדֹנָ֑י כִּ֥י אֵלֶ֥יךָ אֶ֝קְרָ֗א כָּל־ הַיּֽוֹם׃ (יט) שַׂ֭מֵּחַ נֶ֣פֶשׁ עַבְדֶּ֑ךָ כִּ֥י אֵלֶ֥יךָ אֲ֝דֹנָ֗י נַפְשִׁ֥י אֶשָּֽׂא׃ (כ) כִּֽי־ אַתָּ֣ה אֲ֭דֹנָי ט֣וֹב וְסַלָּ֑ח וְרַב־ חֶ֝֗סֶד לְכָל־ קֹרְאֶֽיךָ׃ (כא) הַאֲזִ֣ינָה יְ֭הוָה תְּפִלָּתִ֑י וְ֝הַקְשִׁ֗יבָה בְּק֣וֹל תַּחֲנוּנוֹתָֽי׃ (כב) בְּי֣וֹם צָ֭רָתִ֥י אֶקְרָאֶ֗ךָּ כִּ֣י תַעֲנֵֽנִי׃ (כג) אֵין־ כָּמ֖וֹךָ בָאֱלֹהִ֥ים ׀ אֲדֹנָ֗י וְאֵ֣ין כְּֽמַעֲשֶֽׂיךָ׃ (כד) כָּל־ גּוֹיִ֤ם ׀ אֲשֶׁ֥ר עָשִׂ֗יתָ יָב֤וֹאוּ ׀ וְיִשְׁתַּחֲו֣וּ לְפָנֶ֣יךָ אֲדֹנָ֑י וִֽיכַבְּד֣וּ לִשְׁמֶֽךָ׃ (כה) כִּֽי־ גָד֣וֹל אַ֭תָּה וְעֹשֵׂ֣ה נִפְלָא֑וֹת אַתָּ֖ה אֱלֹהִ֣ים לְבַדֶּֽךָ׃ (כו) ה֘וֹרֵ֤נִי יְהוָ֨ה ׀ דַּרְכֶּ֗ךָ אֲהַלֵּ֥ךְ בַּאֲמִתֶּ֑ךָ יַחֵ֥ד לְ֝בָבִ֗י לְיִרְאָ֥ה שְׁמֶֽךָ׃ (כז) אוֹדְךָ֤ ׀ אֲדֹנָ֣י אֱ֭לֹהַי בְּכָל־ לְבָבִ֑י וַאֲכַבְּדָ֖ה שִׁמְךָ֣ לְעוֹלָֽם׃ (כח) כִּֽי־ חַ֭סְדְּךָ גָּד֣וֹל עָלָ֑י וְהִצַּ֥לְתָּ נַ֝פְשִׁ֗י מִשְּׁא֥וֹל תַּחְתִּיָּֽה׃ (כט) אֱלֹהִ֤ים ׀ זֵ֘דִ֤ים קָֽמוּ־ עָלַ֗י וַעֲדַ֣ת עָ֭רִיצִים בִּקְשׁ֣וּ נַפְשִׁ֑י וְלֹ֖א שָׂמ֣וּךָ לְנֶגְדָּֽם׃ (ל) וְאַתָּ֣ה אֲ֭דֹנָי אֵל־ רַח֣וּם וְחַנּ֑וּן אֶ֥רֶךְ אַ֝פַּ֗יִם וְרַב־ חֶ֥סֶד וֶאֱמֶֽת׃ (לא) פְּנֵ֥ה אֵלַ֗י וְחָ֫נֵּ֥נִי תְּנָֽה־ עֻזְּךָ֥ לְעַבְדֶּ֑ךָ וְ֝הוֹשִׁ֗יעָה לְבֶן־ אֲמָתֶֽךָ׃ (לב) עֲשֵֽׂה־ עִמִּ֥י א֗וֹת לְט֫וֹבָ֥ה וְיִרְא֣וּ שֹׂנְאַ֣י וְיֵבֹ֑שׁוּ כִּֽי־ אַתָּ֥ה יְ֝הוָ֗ה עֲזַרְתַּ֥נִי וְנִחַמְתָּֽנִי׃ (לג) לִבְנֵי־ קֹ֭רַח מִזְמ֣וֹר שִׁ֑יר יְ֝סוּדָת֗וֹ בְּהַרְרֵי־ קֹֽדֶשׁ׃ (לד) אֹהֵ֣ב יְ֭הוָה שַׁעֲרֵ֣י צִיּ֑וֹן מִ֝כֹּ֗ל מִשְׁכְּנ֥וֹת יַעֲקֹֽב׃ (לה) נִ֭כְבָּדוֹת מְדֻבָּ֣ר בָּ֑ךְ עִ֖יר הָאֱלֹהִ֣ים סֶֽלָה׃ (לו) אַזְכִּ֤יר ׀ רַ֥הַב וּבָבֶ֗ל לְֽיֹ֫דְעָ֥י הִנֵּ֤ה פְלֶ֣שֶׁת וְצ֣וֹר עִם־ כּ֑וּשׁ זֶ֝֗ה יֻלַּד־ שָֽׁם׃ (לז) וּֽלֲצִיּ֨וֹן ׀ יֵאָמַ֗ר אִ֣ישׁ וְ֭אִישׁ יֻלַּד־ בָּ֑הּ וְה֖וּא יְכוֹנְנֶ֣הָ עֶלְיֽוֹן׃ (לח) יְֽהוָ֗ה יִ֭סְפֹּר בִּכְת֣וֹב עַמִּ֑ים זֶ֖ה יֻלַּד־ שָׁ֣ם סֶֽלָה׃ (לט) וְשָׁרִ֥ים כְּחֹלְלִ֑ים כָּֽל־ מַעְיָנַ֥י בָּֽךְ׃ (מ) שִׁ֥יר מִזְמ֗וֹר לִבְנֵ֫י קֹ֥רַח לַמְנַצֵּ֣חַ עַל־ מָחֲלַ֣ת לְעַנּ֑וֹת מַ֝שְׂכִּ֗יל לְהֵימָ֥ן הָאֶזְרָחִֽי׃ (מא) יְ֭הוָה אֱלֹהֵ֣י יְשׁוּעָתִ֑י יוֹם־ צָעַ֖קְתִּי בַלַּ֣יְלָה נֶגְדֶּֽךָ׃ (מב) תָּב֣וֹא לְ֭פָנֶיךָ תְּפִלָּתִ֑י הַטֵּֽה־ אָ֝זְנְךָ֗ לְרִנָּתִֽי׃ (מג) כִּֽי־ שָֽׂבְעָ֣ה בְרָע֣וֹת נַפְשִׁ֑י וְחַיַּ֗י לִשְׁא֥וֹל הִגִּֽיעוּ׃ (מד) נֶ֭חְשַׁבְתִּי עִם־ י֣וֹרְדֵי ב֑וֹר הָ֝יִ֗יתִי כְּגֶ֣בֶר אֵֽין־ אֱיָֽל׃ (מה) בַּמֵּתִ֗ים חָ֫פְשִׁ֥י כְּמ֤וֹ חֲלָלִ֨ים ׀ שֹׁ֥כְבֵי קֶ֗בֶר אֲשֶׁ֤ר לֹ֣א זְכַרְתָּ֣ם ע֑וֹד וְ֝הֵ֗מָּה מִיָּדְךָ֥ נִגְזָֽרוּ׃ (מו) שַׁ֭תַּנִי בְּב֣וֹר תַּחְתִּיּ֑וֹת בְּ֝מַחֲשַׁכִּ֗ים בִּמְצֹלֽוֹת׃ (מז) עָ֭לַי סָמְכָ֣ה חֲמָתֶ֑ךָ וְכָל־ מִ֝שְׁבָּרֶ֗יךָ עִנִּ֥יתָ סֶּֽלָה׃ (מח) הִרְחַ֥קְתָּ מְיֻדָּעַ֗י מִ֫מֶּ֥נִּי שַׁתַּ֣נִי תוֹעֵב֣וֹת לָ֑מוֹ כָּ֝לֻ֗א וְלֹ֣א אֵצֵֽא׃ (מט) עֵינִ֥י דָאֲבָ֗ה מִנִּ֫י עֹ֥נִי קְרָאתִ֣יךָ יְהוָ֣ה בְּכָל־ י֑וֹם שִׁטַּ֖חְתִּי אֵלֶ֣יךָ כַפָּֽי׃ (נ) הֲלַמֵּתִ֥ים תַּעֲשֶׂה־ פֶּ֑לֶא אִם־ רְ֝פָאִ֗ים יָק֤וּמוּ ׀ יוֹד֬וּךָ סֶּֽלָה׃ (נא) הַיְסֻפַּ֣ר בַּקֶּ֣בֶר חַסְדֶּ֑ךָ אֱ֝מֽוּנָתְךָ֗ בָּאֲבַדּֽוֹן׃ (נב) הֲיִוָּדַ֣ע בַּחֹ֣שֶׁךְ פִּלְאֶ֑ךָ וְ֝צִדְקָתְךָ֗ בְּאֶ֣רֶץ נְשִׁיָּֽה׃ (נג) וַאֲנִ֤י ׀ אֵלֶ֣יךָ יְהוָ֣ה שִׁוַּ֑עְתִּי וּ֝בַבֹּ֗קֶר תְּֽפִלָּתִ֥י תְקַדְּמֶֽךָּ׃ (נד) לָמָ֣ה יְ֭הוָה תִּזְנַ֣ח נַפְשִׁ֑י תַּסְתִּ֖יר פָּנֶ֣יךָ מִמֶּֽנִּי׃ (נה) עָ֘נִ֤י אֲנִ֣י וְגֹוֵ֣עַ מִנֹּ֑עַר נָשָׂ֖אתִי אֵמֶ֣יךָ אָפֽוּנָה׃ (נו) עָ֭לַי עָבְר֣וּ חֲרוֹנֶ֑יךָ בִּ֝עוּתֶ֗יךָ צִמְּתוּתֻֽנִי׃ (נז) סַבּ֣וּנִי כַ֭מַּיִם כָּל־ הַיּ֑וֹם הִקִּ֖יפוּ עָלַ֣י יָֽחַד׃ (נח) הִרְחַ֣קְתָּ מִ֭מֶּנִּי אֹהֵ֣ב וָרֵ֑עַ מְֽיֻדָּעַ֥י מַחְשָֽׁךְ׃ (נט) מַ֝שְׂכִּ֗יל לְאֵיתָ֥ן הָֽאֶזְרָחִֽי׃ (ס) חַֽסְדֵ֣י יְ֭הוָה עוֹלָ֣ם אָשִׁ֑ירָה לְדֹ֥ר וָדֹ֓ר ׀ אוֹדִ֖יעַ אֱמוּנָתְךָ֣ בְּפִֽי׃ (סא) כִּֽי־ אָמַ֗רְתִּי ע֭וֹלָם חֶ֣סֶד יִבָּנֶ֑ה שָׁמַ֓יִם ׀ תָּכִ֖ן אֱמוּנָתְךָ֣ בָהֶֽם׃ (סב) כָּרַ֣תִּֽי בְ֭רִית לִבְחִירִ֑י נִ֝שְׁבַּ֗עְתִּי לְדָוִ֥ד עַבְדִּֽי׃ (סג) עַד־ ע֭וֹלָם אָכִ֣ין זַרְעֶ֑ךָ וּבָנִ֨יתִי לְדֹר־ וָד֖וֹר כִּסְאֲךָ֣ סֶֽלָה׃ (סד) וְי֘וֹד֤וּ שָׁמַ֣יִם פִּלְאֲךָ֣ יְהוָ֑ה אַף־ אֱ֝מֽוּנָתְךָ֗ בִּקְהַ֥ל קְדֹשִֽׁים׃ (סה) כִּ֤י מִ֣י בַ֭שַּׁחַק יַעֲרֹ֣ךְ לַיהוָ֑ה יִדְמֶ֥ה לַ֝יהוָ֗ה בִּבְנֵ֥י אֵלִים׃ (סו) אֵ֣ל נַ֭עֲרָץ בְּסוֹד־ קְדֹשִׁ֣ים רַבָּ֑ה וְ֝נוֹרָ֗א עַל־ כָּל־ סְבִיבָֽיו׃ (סז) יְהוָ֤ה ׀ אֱלֹ֘הֵ֤י צְבָא֗וֹת מִֽי־ כָֽמ֖וֹךָ חֲסִ֥ין ׀ יָ֑הּ וֶ֝אֱמֽוּנָתְךָ֗ סְבִיבוֹתֶֽיךָ׃ (סח) אַתָּ֣ה מ֭וֹשֵׁל בְּגֵא֣וּת הַיָּ֑ם בְּשׂ֥וֹא גַ֝לָּ֗יו אַתָּ֥ה תְשַׁבְּחֵֽם׃ (סט) אַתָּ֤ה דִכִּ֣אתָ כֶחָלָ֣ל רָ֑הַב בִּזְר֥וֹעַ עֻ֝זְּךָ֗ פִּזַּ֥רְתָּ אוֹיְבֶֽיךָ׃ (ע) לְךָ֣ שָׁ֭מַיִם אַף־ לְךָ֥ אָ֑רֶץ תֵּבֵ֥ל וּ֝מְלֹאָ֗הּ אַתָּ֥ה יְסַדְתָּֽם׃ (עא) צָפ֣וֹן וְ֭יָמִין אַתָּ֣ה בְרָאתָ֑ם תָּב֥וֹר וְ֝חֶרְמ֗וֹן בְּשִׁמְךָ֥ יְרַנֵּֽנוּ׃ (עב) לְךָ֣ זְ֭רוֹעַ עִם־ גְּבוּרָ֑ה תָּעֹ֥ז יָ֝דְךָ֗ תָּר֥וּם יְמִינֶֽךָ׃ (עג) צֶ֣דֶק וּ֭מִשְׁפָּט מְכ֣וֹן כִּסְאֶ֑ךָ חֶ֥סֶד וֶ֝אֱמֶ֗ת יְֽקַדְּמ֥וּ פָנֶֽיךָ׃ (עד) אַשְׁרֵ֣י הָ֭עָם יוֹדְעֵ֣י תְרוּעָ֑ה יְ֝הוָ֗ה בְּֽאוֹר־ פָּנֶ֥יךָ יְהַלֵּכֽוּן׃ (עה) בְּ֭שִׁמְךָ יְגִיל֣וּן כָּל־ הַיּ֑וֹם וּבְצִדְקָתְךָ֥ יָרֽוּמוּ׃ (עו) כִּֽי־ תִפְאֶ֣רֶת עֻזָּ֣מוֹ אָ֑תָּה וּ֝בִרְצֹנְךָ֗ (תרים) תָּר֥וּם קַרְנֵֽנוּ׃ (עז) כִּ֣י לַֽ֭יהוָה מָֽגִנֵּ֑נוּ וְלִקְד֖וֹשׁ יִשְׂרָאֵ֣ל מַלְכֵּֽנוּ׃ (עח) אָ֤ז דִּבַּ֥רְתָּֽ־ בְחָ֡זוֹן לַֽחֲסִידֶ֗יךָ וַתֹּ֗אמֶר שִׁוִּ֣יתִי עֵ֭זֶר עַל־ גִּבּ֑וֹר הֲרִימ֖וֹתִי בָח֣וּר מֵעָֽם׃ (עט) מָ֭צָאתִי דָּוִ֣ד עַבְדִּ֑י בְּשֶׁ֖מֶן קָדְשִׁ֣י מְשַׁחְתִּֽיו׃ (פ) אֲשֶׁ֣ר יָ֭דִי תִּכּ֣וֹן עִמּ֑וֹ אַף־ זְרוֹעִ֥י תְאַמְּצֶֽנּוּ׃ (פא) לֹֽא־ יַשִּׁ֣א אוֹיֵ֣ב בּ֑וֹ וּבֶן־ עַ֝וְלָ֗ה לֹ֣א יְעַנֶּֽנּוּ׃ (פב) וְכַתּוֹתִ֣י מִפָּנָ֣יו צָרָ֑יו וּמְשַׂנְאָ֥יו אֶגּֽוֹף׃ (פג) וֶֽאֶֽמוּנָתִ֣י וְחַסְדִּ֣י עִמּ֑וֹ וּ֝בִשְׁמִ֗י תָּר֥וּם קַרְנֽוֹ׃ (פד) וְשַׂמְתִּ֣י בַיָּ֣ם יָד֑וֹ וּֽבַנְּהָר֥וֹת יְמִינֽוֹ׃ (פה) ה֣וּא יִ֭קְרָאֵנִי אָ֣בִי אָ֑תָּה אֵ֝לִ֗י וְצ֣וּר יְשׁוּעָתִֽי׃ (פו) אַף־ אָ֭נִי בְּכ֣וֹר אֶתְּנֵ֑הוּ עֶ֝לְי֗וֹן לְמַלְכֵי־ אָֽרֶץ׃ (פז) לְ֭עוֹלָ֗ם (אשמור)־ אֶשְׁמָר־ ל֣וֹ חַסְדִּ֑י וּ֝בְרִיתִ֗י נֶאֱמֶ֥נֶת לֽוֹ׃ (פח) וְשַׂמְתִּ֣י לָעַ֣ד זַרְע֑וֹ וְ֝כִסְא֗וֹ כִּימֵ֥י שָׁמָֽיִם׃ (פט) אִם־ יַֽעַזְב֣וּ בָ֭נָיו תּוֹרָתִ֑י וּ֝בְמִשְׁפָּטַ֗י לֹ֣א יֵלֵכֽוּן׃ (צ) אִם־ חֻקֹּתַ֥י יְחַלֵּ֑לוּ וּ֝מִצְוֺתַ֗י לֹ֣א יִשְׁמֹֽרוּ׃ (צא) וּפָקַדְתִּ֣י בְשֵׁ֣בֶט פִּשְׁעָ֑ם וּבִנְגָעִ֥ים עֲוֺנָֽם׃ (צב) וְ֭חַסְדִּי לֹֽא־ אָפִ֣יר מֵֽעִמּ֑וֹ וְלֹֽא־ אֲ֝שַׁקֵּ֗ר בֶּאֱמוּנָתִֽי׃ (צג) לֹא־ אֲחַלֵּ֥ל בְּרִיתִ֑י וּמוֹצָ֥א שְׂ֝פָתַ֗י לֹ֣א אֲשַׁנֶּֽה׃ (צד) אַ֭חַת נִשְׁבַּ֣עְתִּי בְקָדְשִׁ֑י אִֽם־ לְדָוִ֥ד אֲכַזֵּֽב׃ (צה) זַ֭רְעוֹ לְעוֹלָ֣ם יִהְיֶ֑ה וְכִסְא֖וֹ כַשֶּׁ֣מֶשׁ נֶגְדִּֽי׃ (צו) כְּ֭יָרֵחַ יִכּ֣וֹן עוֹלָ֑ם וְעֵ֥ד בַּ֝שַּׁ֗חַק נֶאֱמָ֥ן סֶֽלָה׃ (צז) וְאַתָּ֣ה זָ֭נַחְתָּ וַתִּמְאָ֑ס הִ֝תְעַבַּ֗רְתָּ עִם־ מְשִׁיחֶֽךָ׃ (צח) נֵ֭אַרְתָּה בְּרִ֣ית עַבְדֶּ֑ךָ חִלַּ֖לְתָּ לָאָ֣רֶץ נִזְרֽוֹ׃ (צט) פָּרַ֥צְתָּ כָל־ גְּדֵרֹתָ֑יו שַׂ֖מְתָּ מִבְצָרָ֣יו מְחִתָּה׃ (ק) שַׁ֭סֻּהוּ כָּל־ עֹ֣בְרֵי דָ֑רֶךְ הָיָ֥ה חֶ֝רְפָּ֗ה לִשְׁכֵנָֽיו׃ (קא) הֲ֭רִימוֹתָ יְמִ֣ין צָרָ֑יו הִ֝שְׂמַ֗חְתָּ כָּל־ אוֹיְבָֽיו׃ (קב) אַף־ תָּ֭שִׁיב צ֣וּר חַרְבּ֑וֹ וְלֹ֥א הֲ֝קֵימֹת֗וֹ בַּמִּלְחָמָֽה׃ (קג) הִשְׁבַּ֥תָּ מִטְּהָר֑וֹ וְ֝כִסְא֗וֹ לָאָ֥רֶץ מִגַּֽרְתָּה׃ (קד) הִ֭קְצַרְתָּ יְמֵ֣י עֲלוּמָ֑יו הֶֽעֱטִ֨יתָ עָלָ֖יו בּוּשָׁ֣ה סֶֽלָה׃ (קה) עַד־ מָ֣ה יְ֭הוָה תִּסָּתֵ֣ר לָנֶ֑צַח תִּבְעַ֖ר כְּמוֹ־ אֵ֣שׁ חֲמָתֶֽךָ׃ (קו) זְכָר־ אֲנִ֥י מֶה־ חָ֑לֶד עַל־ מַה־ שָּׁ֝֗וְא בָּרָ֥אתָ כָל־ בְּנֵי־ אָדָֽם׃ (קז) מִ֤י גֶ֣בֶר יִֽ֭חְיֶה וְלֹ֣א יִרְאֶה־ מָּ֑וֶת יְמַלֵּ֨ט נַפְשׁ֖וֹ מִיַּד־ שְׁא֣וֹל סֶֽלָה׃ (קח) אַיֵּ֤ה ׀ חֲסָדֶ֖יךָ הָרִאשֹׁנִ֥ים ׀ אֲדֹנָ֑י נִשְׁבַּ֥עְתָּ לְ֝דָוִ֗ד בֶּאֱמוּנָתֶֽךָ׃ (קט) זְכֹ֣ר אֲ֭דֹנָי חֶרְפַּ֣ת עֲבָדֶ֑יךָ שְׂאֵתִ֥י בְ֝חֵיקִ֗י כָּל־ רַבִּ֥ים עַמִּֽים׃ (קי) אֲשֶׁ֤ר חֵרְפ֖וּ אוֹיְבֶ֥יךָ ׀ יְהוָ֑ה אֲשֶׁ֥ר חֵ֝רְפ֗וּ עִקְּב֥וֹת מְשִׁיחֶֽךָ׃ (קיא) בָּר֖וּךְ יְהוָ֥ה לְ֝עוֹלָ֗ם אָ֘מֵ֥ן ׀ וְאָמֵֽן׃ (קיב) תְּפִלָּה֮ לְמֹשֶׁ֪ה אִֽישׁ־ הָאֱלֹ֫הִ֥ים אֲ‍ֽדֹנָ֗י מָע֣וֹן אַ֭תָּה הָיִ֥יתָ לָּ֗נוּ בְּדֹ֣ר וָדֹֽר׃ (קיג) בְּטֶ֤רֶם ׀ הָ֘רִ֤ים יֻלָּ֗דוּ וַתְּח֣וֹלֵֽל אֶ֣רֶץ וְתֵבֵ֑ל וּֽמֵעוֹלָ֥ם עַד־ ע֝וֹלָ֗ם אַתָּ֥ה אֵֽל׃ (קיד) תָּשֵׁ֣ב אֱ֭נוֹשׁ עַד־ דַּכָּ֑א וַ֝תֹּ֗אמֶר שׁ֣וּבוּ בְנֵי־ אָדָֽם׃ (קטו) כִּ֤י אֶ֪לֶף שָׁנִ֡ים בְּֽעֵינֶ֗יךָ כְּי֣וֹם אֶ֭תְמוֹל כִּ֣י יַעֲבֹ֑ר וְאַשְׁמוּרָ֥ה בַלָּֽיְלָה׃ (קטז) זְ֭רַמְתָּם שֵׁנָ֣ה יִהְי֑וּ בַּ֝בֹּ֗קֶר כֶּחָצִ֥יר יַחֲלֹֽף׃ (קיז) בַּ֭בֹּקֶר יָצִ֣יץ וְחָלָ֑ף לָ֝עֶ֗רֶב יְמוֹלֵ֥ל וְיָבֵֽשׁ׃ (קיח) כִּֽי־ כָלִ֥ינוּ בְאַפֶּ֑ךָ וּֽבַחֲמָתְךָ֥ נִבְהָֽלְנוּ׃ (קיט) (שת) שַׁתָּ֣ה עֲוֺנֹתֵ֣ינוּ לְנֶגְדֶּ֑ךָ עֲ֝לֻמֵ֗נוּ לִמְא֥וֹר פָּנֶֽיךָ׃ (קכ) כִּ֣י כָל־ יָ֭מֵינוּ פָּנ֣וּ בְעֶבְרָתֶ֑ךָ כִּלִּ֖ינוּ שָׁנֵ֣ינוּ כְמוֹ־ הֶֽגֶה׃ (קכא) יְמֵֽי־ שְׁנוֹתֵ֨ינוּ בָהֶ֥ם שִׁבְעִ֪ים שָׁנָ֡ה וְאִ֤ם בִּגְבוּרֹ֨ת ׀ שְׁמ֘וֹנִ֤ים שָׁנָ֗ה וְ֭רָהְבָּם עָמָ֣ל וָאָ֑וֶן כִּי־ גָ֥ז חִ֝֗ישׁ וַנָּעֻֽפָה׃ (קכב) מִֽי־ י֭וֹדֵעַ עֹ֣ז אַפֶּ֑ךָ וּ֝כְיִרְאָתְךָ֗ עֶבְרָתֶֽךָ׃ (קכג) לִמְנ֣וֹת יָ֭מֵינוּ כֵּ֣ן הוֹדַ֑ע וְ֝נָבִ֗א לְבַ֣ב חָכְמָֽה׃ (קכד) שׁוּבָ֣ה יְ֭הוָה עַד־ מָתָ֑י וְ֝הִנָּחֵ֗ם עַל־ עֲבָדֶֽיךָ׃ (קכה) שַׂבְּעֵ֣נוּ בַבֹּ֣קֶר חַסְדֶּ֑ךָ וּֽנְרַנְּנָ֥ה וְ֝נִשְׂמְחָ֗ה בְּכָל־ יָמֵֽינוּ׃ (קכו) שַׂ֭מְּחֵנוּ כִּימ֣וֹת עִנִּיתָ֑נוּ שְׁ֝נ֗וֹת רָאִ֥ינוּ רָעָֽה׃ (קכז) יֵרָאֶ֣ה אֶל־ עֲבָדֶ֣יךָ פָעֳלֶ֑ךָ וַ֝הֲדָרְךָ֗ עַל־ בְּנֵיהֶֽם׃
