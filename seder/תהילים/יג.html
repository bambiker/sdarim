תהילים סדר יג

פתיחה באתר · ספר תהילים כולו ·

(א) וִיהִ֤י ׀ נֹ֤עַם אֲדֹנָ֥י אֱלֹהֵ֗ינוּ עָ֫לֵ֥ינוּ וּמַעֲשֵׂ֣ה יָ֭דֵינוּ כּוֹנְנָ֥ה עָלֵ֑ינוּ וּֽמַעֲשֵׂ֥ה יָ֝דֵ֗ינוּ כּוֹנְנֵֽהוּ׃ (ב) יֹ֭שֵׁב בְּסֵ֣תֶר עֶלְי֑וֹן בְּצֵ֥ל שַׁ֝דַּ֗י יִתְלוֹנָֽן׃ (ג) אֹמַ֗ר לַֽ֭יהוָה מַחְסִ֣י וּמְצוּדָתִ֑י אֱ֝לֹהַ֗י אֶבְטַח־ בּֽוֹ׃ (ד) כִּ֤י ה֣וּא יַ֭צִּֽילְךָ מִפַּ֥ח יָק֗וּשׁ מִדֶּ֥בֶר הַוּֽוֹת׃ (ה) בְּאֶבְרָת֨וֹ ׀ יָ֣סֶךְ לָ֭ךְ וְתַֽחַת־ כְּנָפָ֣יו תֶּחְסֶ֑ה צִנָּ֖ה וְֽסֹחֵרָ֣ה אֲמִתּֽוֹ׃ (ו) לֹא־ תִ֭ירָא מִפַּ֣חַד לָ֑יְלָה מֵ֝חֵ֗ץ יָע֥וּף יוֹמָֽם׃ (ז) מִ֭דֶּבֶר בָּאֹ֣פֶל יַהֲלֹ֑ךְ מִ֝קֶּ֗טֶב יָשׁ֥וּד צָהֳרָֽיִם׃ (ח) יִפֹּ֤ל מִצִּדְּךָ֨ ׀ אֶ֗לֶף וּרְבָבָ֥ה מִימִינֶ֑ךָ אֵ֝לֶ֗יךָ לֹ֣א יִגָּֽשׁ׃ (ט) רַ֭ק בְּעֵינֶ֣יךָ תַבִּ֑יט וְשִׁלֻּמַ֖ת רְשָׁעִ֣ים תִּרְאֶֽה׃ (י) כִּֽי־ אַתָּ֣ה יְהוָ֣ה מַחְסִ֑י עֶ֝לְי֗וֹן שַׂ֣מְתָּ מְעוֹנֶֽךָ׃ (יא) לֹֽא־ תְאֻנֶּ֣ה אֵלֶ֣יךָ רָעָ֑ה וְ֝נֶ֗גַע לֹא־ יִקְרַ֥ב בְּאָהֳלֶֽךָ׃ (יב) כִּ֣י מַ֭לְאָכָיו יְצַוֶּה־ לָּ֑ךְ לִ֝שְׁמָרְךָ֗ בְּכָל־ דְּרָכֶֽיךָ׃ (יג) עַל־ כַּפַּ֥יִם יִשָּׂא֑וּנְךָ פֶּן־ תִּגֹּ֖ף בָּאֶ֣בֶן רַגְלֶֽךָ׃ (יד) עַל־ שַׁ֣חַל וָפֶ֣תֶן תִּדְרֹ֑ךְ תִּרְמֹ֖ס כְּפִ֣יר וְתַנִּֽין׃ (טו) כִּ֤י בִ֣י חָ֭שַׁק וַאֲפַלְּטֵ֑הוּ אֲ֝שַׂגְּבֵ֗הוּ כִּֽי־ יָדַ֥ע שְׁמִֽי׃ (טז) יִקְרָאֵ֨נִי ׀ וְֽאֶעֱנֵ֗הוּ עִמּֽוֹ־ אָנֹכִ֥י בְצָרָ֑ה אֲ֝חַלְּצֵ֗הוּ וַֽאֲכַבְּדֵֽהוּ׃ (יז) אֹ֣רֶךְ יָ֭מִים אַשְׂבִּיעֵ֑הוּ וְ֝אַרְאֵ֗הוּ בִּֽישׁוּעָתִֽי׃ (יח) מִזְמ֥וֹר שִׁ֗יר לְי֣וֹם הַשַּׁבָּֽת׃ (יט) ט֗וֹב לְהֹד֥וֹת לַיהוָ֑ה וּלְזַמֵּ֖ר לְשִׁמְךָ֣ עֶלְיֽוֹן׃ (כ) לְהַגִּ֣יד בַּבֹּ֣קֶר חַסְֽדֶּ֑ךָ וֶ֝אֱמֽוּנָתְךָ֗ בַּלֵּילֽוֹת׃ (כא) עֲ‍ֽלֵי־ עָ֭שׂוֹר וַעֲלֵי־ נָ֑בֶל עֲלֵ֖י הִגָּי֣וֹן בְּכִנּֽוֹר׃ (כב) כִּ֤י שִׂמַּחְתַּ֣נִי יְהוָ֣ה בְּפָעֳלֶ֑ךָ בְּֽמַעֲשֵׂ֖י יָדֶ֣יךָ אֲרַנֵּֽן׃ (כג) מַה־ גָּדְל֣וּ מַעֲשֶׂ֣יךָ יְהוָ֑ה מְ֝אֹ֗ד עָמְק֥וּ מַחְשְׁבֹתֶֽיךָ׃ (כד) אִֽישׁ־ בַּ֭עַר לֹ֣א יֵדָ֑ע וּ֝כְסִ֗יל לֹא־ יָבִ֥ין אֶת־ זֹֽאת׃ (כה) בִּפְרֹ֤חַ רְשָׁעִ֨ים ׀ כְּמ֥וֹ עֵ֗שֶׂב וַ֭יָּצִיצוּ כָּל־ פֹּ֣עֲלֵי אָ֑וֶן לְהִשָּֽׁמְדָ֥ם עֲדֵי־ עַֽד׃ (כו) וְאַתָּ֥ה מָר֗וֹם לְעֹלָ֥ם יְהוָֽה׃ (כז) כִּ֤י הִנֵּ֪ה אֹיְבֶ֡יךָ ׀ יְֽהוָ֗ה כִּֽי־ הִנֵּ֣ה אֹיְבֶ֣יךָ יֹאבֵ֑דוּ יִ֝תְפָּרְד֗וּ כָּל־ פֹּ֥עֲלֵי אָֽוֶן׃ (כח) וַתָּ֣רֶם כִּרְאֵ֣ים קַרְנִ֑י בַּ֝לֹּתִ֗י בְּשֶׁ֣מֶן רַעֲנָֽן׃ (כט) וַתַּבֵּ֥ט עֵינִ֗י בְּשׁ֫וּרָ֥י בַּקָּמִ֖ים עָלַ֥י מְרֵעִ֗ים תִּשְׁמַ֥עְנָה אָזְנָֽי׃ (ל) צַ֭דִּיק כַּתָּמָ֣ר יִפְרָ֑ח כְּאֶ֖רֶז בַּלְּבָנ֣וֹן יִשְׂגֶּֽה׃ (לא) שְׁ֭תוּלִים בְּבֵ֣ית יְהוָ֑ה בְּחַצְר֖וֹת אֱלֹהֵ֣ינוּ יַפְרִֽיחוּ׃ (לב) ע֭וֹד יְנוּב֣וּן בְּשֵׂיבָ֑ה דְּשֵׁנִ֖ים וְרַֽעֲנַנִּ֣ים יִהְיֽוּ׃ (לג) לְ֭הַגִּיד כִּֽי־ יָשָׁ֣ר יְהוָ֑ה צ֝וּרִ֗י וְֽלֹא־ (עלתה) עַוְלָ֥תָה בּֽוֹ׃ (לד) יְהוָ֣ה מָלָךְ֮ גֵּא֪וּת לָ֫בֵ֥שׁ לָבֵ֣שׁ יְ֭הוָה עֹ֣ז הִתְאַזָּ֑ר אַף־ תִּכּ֥וֹן תֵּ֝בֵ֗ל בַּל־ תִּמּֽוֹט׃ (לה) נָכ֣וֹן כִּסְאֲךָ֣ מֵאָ֑ז מֵֽעוֹלָ֣ם אָֽתָּה׃ (לו) נָשְׂא֤וּ נְהָר֨וֹת ׀ יְֽהוָ֗ה נָשְׂא֣וּ נְהָר֣וֹת קוֹלָ֑ם יִשְׂא֖וּ נְהָר֣וֹת דָּכְיָֽם׃ (לז) מִקֹּל֨וֹת ׀ מַ֤יִם רַבִּ֗ים אַדִּירִ֣ים מִשְׁבְּרֵי־ יָ֑ם אַדִּ֖יר בַּמָּר֣וֹם יְהוָֽה׃ (לח) עֵֽדֹתֶ֨יךָ ׀ נֶאֶמְנ֬וּ מְאֹ֗ד לְבֵיתְךָ֥ נַאֲוָה־ קֹ֑דֶשׁ יְ֝הוָ֗ה לְאֹ֣רֶךְ יָמִֽים׃ (לט) אֵל־ נְקָמ֥וֹת יְהוָ֑ה אֵ֖ל נְקָמ֣וֹת הוֹפִֽיַע׃ (מ) הִ֭נָּשֵׂא שֹׁפֵ֣ט הָאָ֑רֶץ הָשֵׁ֥ב גְּ֝מ֗וּל עַל־ גֵּאִֽים׃ (מא) עַד־ מָתַ֖י רְשָׁעִ֥ים ׀ יְהוָ֑ה עַד־ מָ֝תַ֗י רְשָׁעִ֥ים יַעֲלֹֽזוּ׃ (מב) יַבִּ֣יעוּ יְדַבְּר֣וּ עָתָ֑ק יִֽ֝תְאַמְּר֗וּ כָּל־ פֹּ֥עֲלֵי אָֽוֶן׃ (מג) עַמְּךָ֣ יְהוָ֣ה יְדַכְּא֑וּ וְֽנַחֲלָתְךָ֥ יְעַנּֽוּ׃ (מד) אַ֭לְמָנָה וְגֵ֣ר יַהֲרֹ֑גוּ וִֽיתוֹמִ֣ים יְרַצֵּֽחוּ׃ (מה) וַ֭יֹּ֣אמְרוּ לֹ֣א יִרְאֶה־ יָּ֑הּ וְלֹא־ יָ֝בִ֗ין אֱלֹהֵ֥י יַעֲקֹֽב׃ (מו) בִּ֭ינוּ בֹּעֲרִ֣ים בָּעָ֑ם וּ֝כְסִילִ֗ים מָתַ֥י תַּשְׂכִּֽילוּ׃ (מז) הֲנֹ֣טַֽע אֹ֭זֶן הֲלֹ֣א יִשְׁמָ֑ע אִֽם־ יֹ֥צֵֽר עַ֝֗יִן הֲלֹ֣א יַבִּֽיט׃ (מח) הֲיֹסֵ֣ר גּ֭וֹיִם הֲלֹ֣א יוֹכִ֑יחַ הַֽמְלַמֵּ֖ד אָדָ֣ם דָּֽעַת׃ (מט) יְֽהוָ֗ה יֹ֭דֵעַ מַחְשְׁב֣וֹת אָדָ֑ם כִּי־ הֵ֥מָּה הָֽבֶל׃ (נ) אַשְׁרֵ֤י ׀ הַגֶּ֣בֶר אֲשֶׁר־ תְּיַסְּרֶ֣נּוּ יָּ֑הּ וּֽמִתּוֹרָתְךָ֥ תְלַמְּדֶֽנּוּ׃ (נא) לְהַשְׁקִ֣יט ל֭וֹ מִ֣ימֵי רָ֑ע עַ֤ד יִכָּרֶ֖ה לָרָשָׁ֣ע שָֽׁחַת׃ (נב) כִּ֤י ׀ לֹא־ יִטֹּ֣שׁ יְהוָ֣ה עַמּ֑וֹ וְ֝נַחֲלָת֗וֹ לֹ֣א יַעֲזֹֽב׃ (נג) כִּֽי־ עַד־ צֶ֭דֶק יָשׁ֣וּב מִשְׁפָּ֑ט וְ֝אַחֲרָ֗יו כָּל־ יִשְׁרֵי־ לֵֽב׃ (נד) מִֽי־ יָק֣וּם לִ֭י עִם־ מְרֵעִ֑ים מִֽי־ יִתְיַצֵּ֥ב לִ֝י עִם־ פֹּ֥עֲלֵי אָֽוֶן׃ (נה) לוּלֵ֣י יְ֭הוָה עֶזְרָ֣תָה לִּ֑י כִּמְעַ֓ט ׀ שָֽׁכְנָ֖ה דוּמָ֣ה נַפְשִֽׁי׃ (נו) אִם־ אָ֭מַרְתִּי מָ֣טָה רַגְלִ֑י חַסְדְּךָ֥ יְ֝הוָ֗ה יִסְעָדֵֽנִי׃ (נז) בְּרֹ֣ב שַׂרְעַפַּ֣י בְּקִרְבִּ֑י תַּ֝נְחוּמֶ֗יךָ יְֽשַׁעַשְׁע֥וּ נַפְשִֽׁי׃ (נח) הַֽ֭יְחָבְרְךָ כִּסֵּ֣א הַוּ֑וֹת יֹצֵ֖ר עָמָ֣ל עֲלֵי־ חֹֽק׃ (נט) יָ֭גוֹדּוּ עַל־ נֶ֣פֶשׁ צַדִּ֑יק וְדָ֖ם נָקִ֣י יַרְשִֽׁיעוּ׃ (ס) וַיְהִ֬י יְהוָ֣ה לִ֣י לְמִשְׂגָּ֑ב וֵ֝אלֹהַ֗י לְצ֣וּר מַחְסִֽי׃ (סא) וַיָּ֤שֶׁב עֲלֵיהֶ֨ם ׀ אֶת־ אוֹנָ֗ם וּבְרָעָתָ֥ם יַצְמִיתֵ֑ם יַ֝צְמִיתֵ֗ם יְהוָ֥ה אֱלֹהֵֽינוּ׃ (סב) לְ֭כוּ נְרַנְּנָ֣ה לַיהוָ֑ה נָ֝רִ֗יעָה לְצ֣וּר יִשְׁעֵֽנוּ׃ (סג) נְקַדְּמָ֣ה פָנָ֣יו בְּתוֹדָ֑ה בִּ֝זְמִר֗וֹת נָרִ֥יעַֽ לֽוֹ׃ (סד) כִּ֤י אֵ֣ל גָּד֣וֹל יְהוָ֑ה וּמֶ֥לֶךְ גָּ֝ד֗וֹל עַל־ כָּל־ אֱלֹהִֽים׃ (סה) אֲשֶׁ֣ר בְּ֭יָדוֹ מֶחְקְרֵי־ אָ֑רֶץ וְתוֹעֲפ֖וֹת הָרִ֣ים לֽוֹ׃ (סו) אֲשֶׁר־ ל֣וֹ הַ֭יָּם וְה֣וּא עָשָׂ֑הוּ וְ֝יַבֶּ֗שֶׁת יָדָ֥יו יָצָֽרוּ׃ (סז) בֹּ֭אוּ נִשְׁתַּחֲוֶ֣ה וְנִכְרָ֑עָה נִ֝בְרְכָ֗ה לִֽפְנֵי־ יְהוָ֥ה עֹשֵֽׂנוּ׃ (סח) כִּ֘י ה֤וּא אֱלֹהֵ֗ינוּ וַאֲנַ֤חְנוּ עַ֣ם מַ֭רְעִיתוֹ וְצֹ֣אן יָד֑וֹ הַ֝יּ֗וֹם אִֽם־ בְּקֹל֥וֹ תִשְׁמָֽעוּ׃ (סט) אַל־ תַּקְשׁ֣וּ לְ֭בַבְכֶם כִּמְרִיבָ֑ה כְּי֥וֹם מַ֝סָּ֗ה בַּמִּדְבָּֽר׃ (ע) אֲשֶׁ֣ר נִ֭סּוּנִי אֲבוֹתֵיכֶ֑ם בְּ֝חָנ֗וּנִי גַּם־ רָא֥וּ פָעֳלִֽי׃ (עא) אַרְבָּ֘עִ֤ים שָׁנָ֨ה ׀ אָ֘ק֤וּט בְּד֗וֹר וָאֹמַ֗ר עַ֤ם תֹּעֵ֣י לֵבָ֣ב הֵ֑ם וְ֝הֵ֗ם לֹא־ יָדְע֥וּ דְרָכָֽי׃ (עב) אֲשֶׁר־ נִשְׁבַּ֥עְתִּי בְאַפִּ֑י אִם־ יְ֝בֹא֗וּן אֶל־ מְנוּחָתִֽי׃ (עג) שִׁ֣ירוּ לַ֭יהוָה שִׁ֣יר חָדָ֑שׁ שִׁ֥ירוּ לַ֝יהוָ֗ה כָּל־ הָאָֽרֶץ׃ (עד) שִׁ֣ירוּ לַ֭יהוָה בָּרֲכ֣וּ שְׁמ֑וֹ בַּשְּׂר֥וּ מִיּֽוֹם־ לְ֝י֗וֹם יְשׁוּעָתֽוֹ׃ (עה) סַפְּר֣וּ בַגּוֹיִ֣ם כְּבוֹד֑וֹ בְּכָל־ הָֽ֝עַמִּ֗ים נִפְלְאוֹתָֽיו׃ (עו) כִּ֥י גָ֘ד֤וֹל יְהוָ֣ה וּמְהֻלָּ֣ל מְאֹ֑ד נוֹרָ֥א ה֝֗וּא עַל־ כָּל־ אֱלֹהִֽים׃ (עז) כִּ֤י ׀ כָּל־ אֱלֹהֵ֣י הָעַמִּ֣ים אֱלִילִ֑ים וַֽ֝יהוָ֗ה שָׁמַ֥יִם עָשָֽׂה׃ (עח) הוֹד־ וְהָדָ֥ר לְפָנָ֑יו עֹ֥ז וְ֝תִפְאֶ֗רֶת בְּמִקְדָּשֽׁוֹ׃ (עט) הָב֣וּ לַ֭יהוָה מִשְׁפְּח֣וֹת עַמִּ֑ים הָב֥וּ לַ֝יהוָ֗ה כָּב֥וֹד וָעֹֽז׃ (פ) הָב֣וּ לַ֭יהוָה כְּב֣וֹד שְׁמ֑וֹ שְׂאֽוּ־ מִ֝נְחָ֗ה וּבֹ֥אוּ לְחַצְרוֹתָֽיו׃ (פא) הִשְׁתַּחֲו֣וּ לַ֭יהוָה בְּהַדְרַת־ קֹ֑דֶשׁ חִ֥ילוּ מִ֝פָּנָ֗יו כָּל־ הָאָֽרֶץ׃ (פב) אִמְר֤וּ בַגּוֹיִ֨ם ׀ יְה֘וָ֤ה מָלָ֗ךְ אַף־ תִּכּ֣וֹן תֵּ֭בֵל בַּל־ תִּמּ֑וֹט יָדִ֥ין עַ֝מִּ֗ים בְּמֵישָׁרִֽים׃ (פג) יִשְׂמְח֣וּ הַ֭שָּׁמַיִם וְתָגֵ֣ל הָאָ֑רֶץ יִֽרְעַ֥ם הַ֝יָּ֗ם וּמְלֹאֽוֹ׃ (פד) יַעֲלֹ֣ז שָׂ֭דַי וְכָל־ אֲשֶׁר־ בּ֑וֹ אָ֥ז יְ֝רַנְּנ֗וּ כָּל־ עֲצֵי־ יָֽעַר׃ (פה) לִפְנֵ֤י יְהוָ֨ה ׀ כִּ֬י בָ֗א כִּ֥י בָא֮ לִשְׁפֹּ֪ט הָ֫אָ֥רֶץ יִשְׁפֹּֽט־ תֵּבֵ֥ל בְּצֶ֑דֶק וְ֝עַמִּ֗ים בֶּאֱמוּנָתֽוֹ׃ (פו) יְהוָ֣ה מָ֭לָךְ תָּגֵ֣ל הָאָ֑רֶץ יִ֝שְׂמְח֗וּ אִיִּ֥ים רַבִּֽים׃ (פז) עָנָ֣ן וַעֲרָפֶ֣ל סְבִיבָ֑יו צֶ֥דֶק וּ֝מִשְׁפָּ֗ט מְכ֣וֹן כִּסְאֽוֹ׃ (פח) אֵ֭שׁ לְפָנָ֣יו תֵּלֵ֑ךְ וּתְלַהֵ֖ט סָבִ֣יב צָרָֽיו׃ (פט) הֵאִ֣ירוּ בְרָקָ֣יו תֵּבֵ֑ל רָאֲתָ֖ה וַתָּחֵ֣ל הָאָֽרֶץ׃ (צ) הָרִ֗ים כַּדּוֹנַ֗ג נָ֭מַסּוּ מִלִּפְנֵ֣י יְהוָ֑ה מִ֝לִּפְנֵ֗י אֲד֣וֹן כָּל־ הָאָֽרֶץ׃ (צא) הִגִּ֣ידוּ הַשָּׁמַ֣יִם צִדְק֑וֹ וְרָא֖וּ כָל־ הָעַמִּ֣ים כְּבוֹדֽוֹ׃ (צב) יֵבֹ֤שׁוּ ׀ כָּל־ עֹ֬בְדֵי פֶ֗סֶל הַמִּֽתְהַלְלִ֥ים בָּאֱלִילִ֑ים הִשְׁתַּחֲווּ־ ל֝וֹ כָּל־ אֱלֹהִֽים׃ (צג) שָׁמְעָ֬ה וַתִּשְׂמַ֨ח ׀ צִיּ֗וֹן וַ֭תָּגֵלְנָה בְּנ֣וֹת יְהוּדָ֑ה לְמַ֖עַן מִשְׁפָּטֶ֣יךָ יְהוָֽה׃ (צד) כִּֽי־ אַתָּ֤ה יְהוָ֗ה עֶלְי֥וֹן עַל־ כָּל־ הָאָ֑רֶץ מְאֹ֥ד נַ֝עֲלֵ֗יתָ עַל־ כָּל־ אֱלֹהִֽים׃ (צה) אֹהֲבֵ֥י יְהוָ֗ה שִׂנְא֫וּ רָ֥ע שֹׁ֭מֵר נַפְשׁ֣וֹת חֲסִידָ֑יו מִיַּ֥ד רְ֝שָׁעִ֗ים יַצִּילֵֽם׃ (צו) א֭וֹר זָרֻ֣עַ לַצַּדִּ֑יק וּֽלְיִשְׁרֵי־ לֵ֥ב שִׂמְחָֽה׃ (צז) שִׂמְח֣וּ צַ֭דִּיקִים בַּֽיהוָ֑ה וְ֝הוֹד֗וּ לְזֵ֣כֶר קָדְשֽׁוֹ׃ (צח) מִזְמ֡וֹר שִׁ֤ירוּ לַֽיהוָ֨ה ׀ שִׁ֣יר חָ֭דָשׁ כִּֽי־ נִפְלָא֣וֹת עָשָׂ֑ה הוֹשִֽׁיעָה־ לּ֥וֹ יְ֝מִינ֗וֹ וּזְר֥וֹעַ קָדְשֽׁוֹ׃ (צט) הוֹדִ֣יעַ יְ֭הוָה יְשׁוּעָת֑וֹ לְעֵינֵ֥י הַ֝גּוֹיִ֗ם גִּלָּ֥ה צִדְקָתֽוֹ׃ (ק) זָ֘כַ֤ר חַסְדּ֨וֹ ׀ וֶֽאֱֽמוּנָתוֹ֮ לְבֵ֪ית יִשְׂרָ֫אֵ֥ל רָא֥וּ כָל־ אַפְסֵי־ אָ֑רֶץ אֵ֝֗ת יְשׁוּעַ֥ת אֱלֹהֵֽינוּ׃ (קא) הָרִ֣יעוּ לַֽ֭יהוָה כָּל־ הָאָ֑רֶץ פִּצְח֖וּ וְרַנְּנ֣וּ וְזַמֵּֽרוּ׃ (קב) זַמְּר֣וּ לַיהוָ֣ה בְּכִנּ֑וֹר בְּ֝כִנּ֗וֹר וְק֣וֹל זִמְרָֽה׃ (קג) בַּ֭חֲצֹ֣צְרוֹת וְק֣וֹל שׁוֹפָ֑ר הָ֝רִ֗יעוּ לִפְנֵ֤י ׀ הַמֶּ֬לֶךְ יְהוָֽה׃ (קד) יִרְעַ֣ם הַ֭יָּם וּמְלֹא֑וֹ תֵּ֝בֵ֗ל וְיֹ֣שְׁבֵי בָֽהּ׃ (קה) נְהָר֥וֹת יִמְחֲאוּ־ כָ֑ף יַ֝֗חַד הָרִ֥ים יְרַנֵּֽנוּ׃ (קו) לִֽפְֽנֵי־ יְהוָ֗ה כִּ֥י בָא֮ לִשְׁפֹּ֪ט הָ֫אָ֥רֶץ יִשְׁפֹּֽט־ תֵּבֵ֥ל בְּצֶ֑דֶק וְ֝עַמִּ֗ים בְּמֵישָׁרִֽים׃ (קז) יְהוָ֣ה מָ֭לָךְ יִרְגְּז֣וּ עַמִּ֑ים יֹשֵׁ֥ב כְּ֝רוּבִ֗ים תָּנ֥וּט הָאָֽרֶץ׃ (קח) יְ֭הוָה בְּצִיּ֣וֹן גָּד֑וֹל וְרָ֥ם ה֝֗וּא עַל־ כָּל־ הָֽעַמִּֽים׃ (קט) יוֹד֣וּ שִׁ֭מְךָ גָּד֥וֹל וְנוֹרָ֗א קָד֥וֹשׁ הֽוּא׃ (קי) וְעֹ֥ז מֶלֶךְ֮ מִשְׁפָּ֪ט אָ֫הֵ֥ב אַ֭תָּה כּוֹנַ֣נְתָּ מֵישָׁרִ֑ים מִשְׁפָּ֥ט וּ֝צְדָקָ֗ה בְּיַעֲקֹ֤ב ׀ אַתָּ֬ה עָשִֽׂיתָ׃ (קיא) רֽוֹמְמ֡וּ יְה֘וָ֤ה אֱלֹהֵ֗ינוּ וְֽ֭הִשְׁתַּחֲווּ לַהֲדֹ֥ם רַגְלָ֗יו קָד֥וֹשׁ הֽוּא׃ (קיב) מֹ֘שֶׁ֤ה וְאַהֲרֹ֨ן ׀ בְּֽכֹהֲנָ֗יו וּ֭שְׁמוּאֵל בְּקֹרְאֵ֣י שְׁמ֑וֹ קֹרִ֥אים אֶל־ יְ֝הוָ֗ה וְה֣וּא יַעֲנֵֽם׃ (קיג) בְּעַמּ֣וּד עָ֭נָן יְדַבֵּ֣ר אֲלֵיהֶ֑ם שָׁמְר֥וּ עֵ֝דֹתָ֗יו וְחֹ֣ק נָֽתַן־ לָֽמוֹ׃ (קיד) יְהוָ֣ה אֱלֹהֵינוּ֮ אַתָּ֪ה עֲנִ֫יתָ֥ם אֵ֣ל נֹ֭שֵׂא הָיִ֣יתָ לָהֶ֑ם וְ֝נֹקֵ֗ם עַל־ עֲלִילוֹתָֽם׃ (קטו) רֽוֹמְמ֡וּ יְה֘וָ֤ה אֱלֹהֵ֗ינוּ וְ֭הִֽשְׁתַּחֲווּ לְהַ֣ר קָדְשׁ֑וֹ כִּֽי־ קָ֝ד֗וֹשׁ יְהוָ֥ה אֱלֹהֵֽינוּ׃ (קטז) מִזְמ֥וֹר לְתוֹדָ֑ה הָרִ֥יעוּ לַ֝יהוָ֗ה כָּל־ הָאָֽרֶץ׃ (קיז) עִבְד֣וּ אֶת־ יְהוָ֣ה בְּשִׂמְחָ֑ה בֹּ֥אוּ לְ֝פָנָ֗יו בִּרְנָנָֽה׃ (קיח) דְּע֗וּ כִּֽי־ יְהוָה֮ ה֤וּא אֱלֹ֫הִ֥ים הֽוּא־ עָ֭שָׂנוּ (ולא) וְל֣וֹ אֲנַ֑חְנוּ עַ֝מּ֗וֹ וְצֹ֣אן מַרְעִיתֽוֹ׃ (קיט) בֹּ֤אוּ שְׁעָרָ֨יו ׀ בְּתוֹדָ֗ה חֲצֵרֹתָ֥יו בִּתְהִלָּ֑ה הֽוֹדוּ־ ל֝֗וֹ בָּרֲכ֥וּ שְׁמֽוֹ׃ (קכ) כִּי־ ט֣וֹב יְ֭הֹוָה לְעוֹלָ֣ם חַסְדּ֑וֹ וְעַד־ דֹּ֥ר וָ֝דֹ֗ר אֱמוּנָתֽוֹ׃
