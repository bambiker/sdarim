תהילים סדר ז

פתיחה באתר · ספר תהילים כולו ·

(א) כִּֽי־ נַ֭פְשׁוֹ בְּחַיָּ֣יו יְבָרֵ֑ךְ וְ֝יוֹדֻ֗ךָ כִּי־ תֵיטִ֥יב לָֽךְ׃ (ב) תָּ֭בוֹא עַד־ דּ֣וֹר אֲבוֹתָ֑יו עַד־ נֵ֝֗צַח לֹ֣א יִרְאוּ־ אֽוֹר׃ (ג) אָדָ֣ם בִּ֭יקָר וְלֹ֣א יָבִ֑ין נִמְשַׁ֖ל כַּבְּהֵמ֣וֹת נִדְמֽוּ׃ (ד) מִזְמ֗וֹר לְאָ֫סָ֥ף אֵ֤ל ׀ אֱ‍ֽלֹהִ֡ים יְֽהוָ֗ה דִּבֶּ֥ר וַיִּקְרָא־ אָ֑רֶץ מִמִּזְרַח־ שֶׁ֝֗מֶשׁ עַד־ מְבֹאֽוֹ׃ (ה) מִצִּיּ֥וֹן מִכְלַל־ יֹ֗פִי אֱלֹהִ֥ים הוֹפִֽיעַ׃ (ו) יָ֤בֹ֥א אֱלֹהֵ֗ינוּ וְֽאַל־ יֶ֫חֱרַ֥שׁ אֵשׁ־ לְפָנָ֥יו תֹּאכֵ֑ל וּ֝סְבִיבָ֗יו נִשְׂעֲרָ֥ה מְאֹֽד׃ (ז) יִקְרָ֣א אֶל־ הַשָּׁמַ֣יִם מֵעָ֑ל וְאֶל־ הָ֝אָ֗רֶץ לָדִ֥ין עַמּֽוֹ׃ (ח) אִסְפוּ־ לִ֥י חֲסִידָ֑י כֹּרְתֵ֖י בְרִיתִ֣י עֲלֵי־ זָֽבַח׃ (ט) וַיַּגִּ֣ידוּ שָׁמַ֣יִם צִדְק֑וֹ כִּֽי־ אֱלֹהִ֓ים ׀ שֹׁפֵ֖ט ה֣וּא סֶֽלָה׃ (י) שִׁמְעָ֤ה עַמִּ֨י ׀ וַאֲדַבֵּ֗רָה יִ֭שְׂרָאֵל וְאָעִ֣ידָה בָּ֑ךְ אֱלֹהִ֖ים אֱלֹהֶ֣יךָ אָנֹֽכִי׃ (יא) לֹ֣א עַל־ זְ֭בָחֶיךָ אוֹכִיחֶ֑ךָ וְעוֹלֹתֶ֖יךָ לְנֶגְדִּ֣י תָמִֽיד׃ (יב) לֹא־ אֶקַּ֣ח מִבֵּיתְךָ֣ פָ֑ר מִ֝מִּכְלְאֹתֶ֗יךָ עַתּוּדִֽים׃ (יג) כִּי־ לִ֥י כָל־ חַיְתוֹ־ יָ֑עַר בְּ֝הֵמ֗וֹת בְּהַרְרֵי־ אָֽלֶף׃ (יד) יָ֭דַעְתִּי כָּל־ ע֣וֹף הָרִ֑ים וְזִ֥יז שָׂ֝דַ֗י עִמָּדִֽי׃ (טו) אִם־ אֶ֭רְעַב לֹא־ אֹ֣מַר לָ֑ךְ כִּי־ לִ֥י תֵ֝בֵ֗ל וּמְלֹאָֽהּ׃ (טז) הַֽ֭אוֹכַל בְּשַׂ֣ר אַבִּירִ֑ים וְדַ֖ם עַתּוּדִ֣ים אֶשְׁתֶּֽה׃ (יז) זְבַ֣ח לֵאלֹהִ֣ים תּוֹדָ֑ה וְשַׁלֵּ֖ם לְעֶלְי֣וֹן נְדָרֶֽיךָ׃ (יח) וּ֭קְרָאֵנִי בְּי֣וֹם צָרָ֑ה אֲ֝חַלֶּצְךָ֗ וּֽתְכַבְּדֵֽנִי׃ (יט) וְלָ֤רָשָׁ֨ע ׀ אָ֘מַ֤ר אֱלֹהִ֗ים מַה־ לְּ֭ךָ לְסַפֵּ֣ר חֻקָּ֑י וַתִּשָּׂ֖א בְרִיתִ֣י עֲלֵי־ פִֽיךָ׃ (כ) וְ֭אַתָּה שָׂנֵ֣אתָ מוּסָ֑ר וַתַּשְׁלֵ֖ךְ דְּבָרַ֣י אַחֲרֶֽיךָ׃ (כא) אִם־ רָאִ֣יתָ גַ֭נָּב וַתִּ֣רֶץ עִמּ֑וֹ וְעִ֖ם מְנָאֲפִ֣ים חֶלְקֶֽךָ׃ (כב) פִּ֭יךָ שָׁלַ֣חְתָּ בְרָעָ֑ה וּ֝לְשׁוֹנְךָ֗ תַּצְמִ֥יד מִרְמָֽה׃ (כג) תֵּ֭שֵׁב בְּאָחִ֣יךָ תְדַבֵּ֑ר בְּבֶֽן־ אִ֝מְּךָ֗ תִּתֶּן־ דֹּֽפִי׃ (כד) אֵ֤לֶּה עָשִׂ֨יתָ ׀ וְֽהֶחֱרַ֗שְׁתִּי דִּמִּ֗יתָ הֱֽיוֹת־ אֶֽהְיֶ֥ה כָמ֑וֹךָ אוֹכִיחֲךָ֖ וְאֶֽעֶרְכָ֣ה לְעֵינֶֽיךָ׃ (כה) בִּֽינוּ־ נָ֣א זֹ֭את שֹׁכְחֵ֣י אֱל֑וֹהַּ פֶּן־ אֶ֝טְרֹ֗ף וְאֵ֣ין מַצִּֽיל׃ (כו) זֹבֵ֥חַ תּוֹדָ֗ה יְֽכַ֫בְּדָ֥נְנִי וְשָׂ֥ם דֶּ֑רֶךְ אַ֝רְאֶ֗נּוּ בְּיֵ֣שַׁע אֱלֹהִֽים׃ (כז) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (כח) בְּֽבוֹא־ אֵ֭לָיו נָתָ֣ן הַנָּבִ֑יא כַּֽאֲשֶׁר־ בָּ֝֗א אֶל־ בַּת־ שָֽׁבַע׃ (כט) חָנֵּ֣נִי אֱלֹהִ֣ים כְּחַסְדֶּ֑ךָ כְּרֹ֥ב רַ֝חֲמֶ֗יךָ מְחֵ֣ה פְשָׁעָֽי׃ (ל) (הרבה) הֶ֭רֶב כַּבְּסֵ֣נִי מֵעֲוֺנִ֑י וּֽמֵחַטָּאתִ֥י טַהֲרֵֽנִי׃ (לא) כִּֽי־ פְ֭שָׁעַי אֲנִ֣י אֵדָ֑ע וְחַטָּאתִ֖י נֶגְדִּ֣י תָמִֽיד׃ (לב) לְךָ֤ לְבַדְּךָ֨ ׀ חָטָאתִי֮ וְהָרַ֥ע בְּעֵינֶ֗יךָ עָ֫שִׂ֥יתִי לְ֭מַעַן תִּצְדַּ֥ק בְּדָבְרֶ֗ךָ תִּזְכֶּ֥ה בְשָׁפְטֶֽךָ׃ (לג) הֵן־ בְּעָו֥וֹן חוֹלָ֑לְתִּי וּ֝בְחֵ֗טְא יֶֽחֱמַ֥תְנִי אִמִּֽי׃ (לד) הֵן־ אֱ֭מֶת חָפַ֣צְתָּ בַטֻּח֑וֹת וּ֝בְסָתֻ֗ם חָכְמָ֥ה תוֹדִיעֵֽנִי׃ (לה) תְּחַטְּאֵ֣נִי בְאֵז֣וֹב וְאֶטְהָ֑ר תְּ֝כַבְּסֵ֗נִי וּמִשֶּׁ֥לֶג אַלְבִּֽין׃ (לו) תַּ֭שְׁמִיעֵנִי שָׂשׂ֣וֹן וְשִׂמְחָ֑ה תָּ֝גֵ֗לְנָה עֲצָמ֥וֹת דִּכִּֽיתָ׃ (לז) הַסְתֵּ֣ר פָּ֭נֶיךָ מֵחֲטָאָ֑י וְֽכָל־ עֲוֺ֖נֹתַ֣י מְחֵֽה׃ (לח) לֵ֣ב טָ֭הוֹר בְּרָא־ לִ֣י אֱלֹהִ֑ים וְר֥וּחַ נָ֝כ֗וֹן חַדֵּ֥שׁ בְּקִרְבִּֽי׃ (לט) אַל־ תַּשְׁלִיכֵ֥נִי מִלְּפָנֶ֑יךָ וְר֥וּחַ קָ֝דְשְׁךָ֗ אַל־ תִּקַּ֥ח מִמֶּֽנִּי׃ (מ) הָשִׁ֣יבָה לִּ֭י שְׂשׂ֣וֹן יִשְׁעֶ֑ךָ וְר֖וּחַ נְדִיבָ֣ה תִסְמְכֵֽנִי׃ (מא) אֲלַמְּדָ֣ה פֹשְׁעִ֣ים דְּרָכֶ֑יךָ וְ֝חַטָּאִ֗ים אֵלֶ֥יךָ יָשֽׁוּבוּ׃ (מב) הַצִּ֘ילֵ֤נִי מִדָּמִ֨ים ׀ אֱ‍ֽלֹהִ֗ים אֱלֹהֵ֥י תְּשׁוּעָתִ֑י תְּרַנֵּ֥ן לְ֝שׁוֹנִ֗י צִדְקָתֶֽךָ׃ (מג) אֲ֭דֹנָי שְׂפָתַ֣י תִּפְתָּ֑ח וּ֝פִ֗י יַגִּ֥יד תְּהִלָּתֶֽךָ׃ (מד) כִּ֤י ׀ לֹא־ תַחְפֹּ֣ץ זֶ֣בַח וְאֶתֵּ֑נָה ע֝וֹלָ֗ה לֹ֣א תִרְצֶֽה׃ (מה) זִֽבְחֵ֣י אֱלֹהִים֮ ר֪וּחַ נִשְׁבָּ֫רָ֥ה לֵב־ נִשְׁבָּ֥ר וְנִדְכֶּ֑ה אֱ֝לֹהִ֗ים לֹ֣א תִבְזֶֽה׃ (מו) הֵיטִ֣יבָה בִ֭רְצוֹנְךָ אֶת־ צִיּ֑וֹן תִּ֝בְנֶ֗ה חוֹמ֥וֹת יְרוּשָׁלִָֽם׃ (מז) אָ֤ז תַּחְפֹּ֣ץ זִבְחֵי־ צֶ֭דֶק עוֹלָ֣ה וְכָלִ֑יל אָ֤ז יַעֲל֖וּ עַל־ מִזְבַּחֲךָ֣ פָרִֽים׃ (מח) לַמְנַצֵּ֗חַ מַשְׂכִּ֥יל לְדָוִֽד׃ (מט) בְּב֤וֹא ׀ דּוֹאֵ֣ג הָאֲדֹמִי֮ וַיַּגֵּ֪ד לְשָׁ֫א֥וּל וַיֹּ֥אמֶר ל֑וֹ בָּ֥א דָ֝וִ֗ד אֶל־ בֵּ֥ית אֲחִימֶֽלֶךְ׃ (נ) מַה־ תִּתְהַלֵּ֣ל בְּ֭רָעָה הַגִּבּ֑וֹר חֶ֥סֶד אֵ֝֗ל כָּל־ הַיּֽוֹם׃ (נא) הַ֭וּוֹת תַּחְשֹׁ֣ב לְשׁוֹנֶ֑ךָ כְּתַ֥עַר מְ֝לֻטָּ֗שׁ עֹשֵׂ֥ה רְמִיָּֽה׃ (נב) אָהַ֣בְתָּ רָּ֣ע מִטּ֑וֹב שֶׁ֓קֶר ׀ מִדַּבֵּ֖ר צֶ֣דֶק סֶֽלָה׃ (נג) אָהַ֥בְתָּ כָֽל־ דִּבְרֵי־ בָ֗לַע לְשׁ֣וֹן מִרְמָֽה׃ (נד) גַּם־ אֵל֮ יִתָּצְךָ֪ לָ֫נֶ֥צַח יַחְתְּךָ֣ וְיִסָּחֲךָ֣ מֵאֹ֑הֶל וְשֵֽׁרֶשְׁךָ֨ מֵאֶ֖רֶץ חַיִּ֣ים סֶֽלָה׃ (נה) וְיִרְא֖וּ צַדִּיקִ֥ים וְיִירָ֗אוּ וְעָלָ֥יו יִשְׂחָֽקוּ׃ (נו) הִנֵּ֤ה הַגֶּ֗בֶר לֹ֤א יָשִׂ֥ים אֱלֹהִ֗ים מָֽע֫וּזּ֥וֹ וַ֭יִּבְטַח בְּרֹ֣ב עָשְׁר֑וֹ יָ֝עֹ֗ז בְּהַוָּתֽוֹ׃ (נז) וַאֲנִ֤י ׀ כְּזַ֣יִת רַ֭עֲנָן בְּבֵ֣ית אֱלֹהִ֑ים בָּטַ֥חְתִּי בְחֶֽסֶד־ אֱ֝לֹהִ֗ים עוֹלָ֥ם וָעֶֽד׃ (נח) אוֹדְךָ֣ לְ֭עוֹלָם כִּ֣י עָשִׂ֑יתָ וַאֲקַוֶּ֖ה שִׁמְךָ֥ כִֽי־ ט֝֗וֹב נֶ֣גֶד חֲסִידֶֽיךָ׃ (נט) לַמְנַצֵּ֥חַ עַֽל־ מָחֲלַ֗ת מַשְׂכִּ֥יל לְדָוִֽד׃ (ס) אָ֘מַ֤ר נָבָ֣ל בְּ֭לִבּוֹ אֵ֣ין אֱלֹהִ֑ים הִֽ֝שְׁחִ֗יתוּ וְהִֽתְעִ֥יבוּ עָ֝֗וֶל אֵ֣ין עֹֽשֵׂה־ טֽוֹב׃ (סא) אֱ‍ֽלֹהִ֗ים מִשָּׁמַיִם֮ הִשְׁקִ֪יף עַֽל־ בְּנֵ֫י אָדָ֥ם לִ֭רְאוֹת הֲיֵ֣שׁ מַשְׂכִּ֑יל דֹּ֝רֵ֗שׁ אֶת־ אֱלֹהִֽים׃ (סב) כֻּלּ֥וֹ סָג֮ יַחְדָּ֪ו נֶ֫אֱלָ֥חוּ אֵ֤ין עֹֽשֵׂה־ ט֑וֹב אֵ֝֗ין גַּם־ אֶחָֽד׃ (סג) הֲלֹ֥א יָדְעוּ֮ פֹּ֤עֲלֵ֫י אָ֥וֶן אֹכְלֵ֣י עַ֭מִּי אָ֣כְלוּ לֶ֑חֶם אֱ֝לֹהִ֗ים לֹ֣א קָרָֽאוּ׃ (סד) שָׁ֤ם ׀ פָּֽחֲדוּ־ פַחַד֮ לֹא־ הָ֪יָה֫ פָ֥חַד כִּֽי־ אֱלֹהִ֗ים פִּ֭זַּר עַצְמ֣וֹת חֹנָ֑ךְ הֱ֝בִשֹׁ֗תָה כִּֽי־ אֱלֹהִ֥ים מְאָסָֽם׃ (סה) מִ֥י יִתֵּ֣ן מִצִּיּוֹן֮ יְשֻׁע֪וֹת יִשְׂרָ֫אֵ֥ל בְּשׁ֣וּב אֱ֭לֹהִים שְׁב֣וּת עַמּ֑וֹ יָגֵ֥ל יַ֝עֲקֹ֗ב יִשְׂמַ֥ח יִשְׂרָאֵֽל׃ (סו) לַמְנַצֵּ֥חַ בִּנְגִינֹ֗ת מַשְׂכִּ֥יל לְדָוִֽד׃ (סז) בְּב֣וֹא הַ֭זִּיפִים וַיֹּאמְר֣וּ לְשָׁא֑וּל הֲלֹ֥א דָ֝וִ֗ד מִסְתַּתֵּ֥ר עִמָּֽנוּ׃ (סח) אֱ֭לֹהִים בְּשִׁמְךָ֣ הוֹשִׁיעֵ֑נִי וּבִגְבוּרָתְךָ֥ תְדִינֵֽנִי׃ (סט) אֱ֭לֹהִים שְׁמַ֣ע תְּפִלָּתִ֑י הַ֝אֲזִ֗ינָה לְאִמְרֵי־ פִֽי׃ (ע) כִּ֤י זָרִ֨ים ׀ קָ֤מוּ עָלַ֗י וְֽ֭עָרִיצִים בִּקְשׁ֣וּ נַפְשִׁ֑י לֹ֤א שָׂ֨מוּ אֱלֹהִ֖ים לְנֶגְדָּ֣ם סֶֽלָה׃ (עא) הִנֵּ֣ה אֱ֭לֹהִים עֹזֵ֣ר לִ֑י אֲ֝דֹנָ֗י בְּֽסֹמְכֵ֥י נַפְשִֽׁי׃ (עב) (ישוב) יָשִׁ֣יב הָ֭רַע לְשֹׁרְרָ֑י בַּ֝אֲמִתְּךָ֗ הַצְמִיתֵֽם׃ (עג) בִּנְדָבָ֥ה אֶזְבְּחָה־ לָּ֑ךְ א֤וֹדֶה שִּׁמְךָ֖ יְהוָ֣ה כִּי־ טֽוֹב׃ (עד) כִּ֣י מִכָּל־ צָ֭רָה הִצִּילָ֑נִי וּ֝בְאֹיְבַ֗י רָאֲתָ֥ה עֵינִֽי׃ (עה) לַמְנַצֵּ֥חַ בִּנְגִינֹ֗ת מַשְׂכִּ֥יל לְדָוִֽד׃ (עו) הַאֲזִ֣ינָה אֱ֭לֹהִים תְּפִלָּתִ֑י וְאַל־ תִּ֝תְעַלַּ֗ם מִתְּחִנָּתִֽי׃ (עז) הַקְשִׁ֣יבָה לִּ֣י וַעֲנֵ֑נִי אָרִ֖יד בְּשִׂיחִ֣י וְאָהִֽימָה׃ (עח) מִקּ֤וֹל אוֹיֵ֗ב מִפְּנֵ֣י עָקַ֣ת רָשָׁ֑ע כִּי־ יָמִ֥יטוּ עָלַ֥י אָ֝֗וֶן וּבְאַ֥ף יִשְׂטְמֽוּנִי׃ (עט) לִ֭בִּי יָחִ֣יל בְּקִרְבִּ֑י וְאֵימ֥וֹת מָ֝֗וֶת נָפְל֥וּ עָלָֽי׃ (פ) יִרְאָ֣ה וָ֭רַעַד יָ֣בֹא בִ֑י וַ֝תְּכַסֵּ֗נִי פַּלָּצֽוּת׃ (פא) וָאֹמַ֗ר מִֽי־ יִתֶּן־ לִּ֣י אֵ֭בֶר כַּיּוֹנָ֗ה אָע֥וּפָה וְאֶשְׁכֹּֽנָה׃ (פב) הִ֭נֵּה אַרְחִ֣יק נְדֹ֑ד אָלִ֖ין בַּמִּדְבָּ֣ר סֶֽלָה׃ (פג) אָחִ֣ישָׁה מִפְלָ֣ט לִ֑י מֵר֖וּחַ סֹעָ֣ה מִסָּֽעַר׃ (פד) בַּלַּ֣ע אֲ֭דֹנָי פַּלַּ֣ג לְשׁוֹנָ֑ם כִּֽי־ רָאִ֨יתִי חָמָ֖ס וְרִ֣יב בָּעִֽיר׃ (פה) יוֹמָ֤ם וָלַ֗יְלָה יְסוֹבְבֻ֥הָ עַל־ חוֹמֹתֶ֑יהָ וְאָ֖וֶן וְעָמָ֣ל בְּקִרְבָּֽהּ׃ (פו) הַוּ֥וֹת בְּקִרְבָּ֑הּ וְֽלֹא־ יָמִ֥ישׁ מֵ֝רְחֹבָ֗הּ תֹּ֣ךְ וּמִרְמָֽה׃ (פז) כִּ֤י לֹֽא־ אוֹיֵ֥ב יְחָֽרְפֵ֗נִי וְאֶ֫שָּׂ֥א לֹֽא־ מְ֭שַׂנְאִי עָלַ֣י הִגְדִּ֑יל וְאֶסָּתֵ֥ר מִמֶּֽנּוּ׃ (פח) וְאַתָּ֣ה אֱנ֣וֹשׁ כְּעֶרְכִּ֑י אַ֝לּוּפִ֗י וּמְיֻדָּֽעִי׃ (פט) אֲשֶׁ֣ר יַ֭חְדָּו נַמְתִּ֣יק ס֑וֹד בְּבֵ֥ית אֱ֝לֹהִ֗ים נְהַלֵּ֥ךְ בְּרָֽגֶשׁ׃ (צ) (ישימות) יַשִּׁ֤י מָ֨וֶת ׀ עָלֵ֗ימוֹ יֵרְד֣וּ שְׁא֣וֹל חַיִּ֑ים כִּֽי־ רָע֖וֹת בִּמְגוּרָ֣ם בְּקִרְבָּֽם׃ (צא) אֲ֭נִי אֶל־ אֱלֹהִ֣ים אֶקְרָ֑א וַ֝יהוָ֗ה יוֹשִׁיעֵֽנִי׃ (צב) עֶ֤רֶב וָבֹ֣קֶר וְ֭צָהֳרַיִם אָשִׂ֣יחָה וְאֶהֱמֶ֑ה וַיִּשְׁמַ֥ע קוֹלִֽי׃ (צג) פָּ֘דָ֤ה בְשָׁל֣וֹם נַ֭פְשִׁי מִקֲּרָב־ לִ֑י כִּֽי־ בְ֝רַבִּ֗ים הָי֥וּ עִמָּדִֽי׃ (צד) יִשְׁמַ֤ע ׀ אֵ֨ל ׀ וְֽיַעֲנֵם֮ וְיֹ֤שֵׁ֥ב קֶ֗דֶם סֶ֥לָה אֲשֶׁ֤ר אֵ֣ין חֲלִיפ֣וֹת לָ֑מוֹ וְלֹ֖א יָרְא֣וּ אֱלֹהִֽים׃ (צה) שָׁלַ֣ח יָ֭דָיו בִּשְׁלֹמָ֗יו חִלֵּ֥ל בְּרִיתֽוֹ׃ (צו) חָלְק֤וּ ׀ מַחְמָאֹ֣ת פִּיו֮ וּֽקֲרָב־ לִ֫בּ֥וֹ רַכּ֖וּ דְבָרָ֥יו מִשֶּׁ֗מֶן וְהֵ֣מָּה פְתִחֽוֹת׃ (צז) הַשְׁלֵ֤ךְ עַל־ יְהוָ֨ה ׀ יְהָבְךָ֮ וְה֪וּא יְכַ֫לְכְּלֶ֥ךָ לֹא־ יִתֵּ֖ן לְעוֹלָ֥ם מ֗וֹט לַצַּדִּֽיק׃ (צח) וְאַתָּ֤ה אֱלֹהִ֨ים ׀ תּוֹרִדֵ֬ם ׀ לִבְאֵ֬ר שַׁ֗חַת אַנְשֵׁ֤י דָמִ֣ים וּ֭מִרְמָה לֹא־ יֶחֱצ֣וּ יְמֵיהֶ֑ם וַ֝אֲנִ֗י אֶבְטַח־ בָּֽךְ׃ (צט) לַמְנַצֵּ֤חַ ׀ עַל־ י֬וֹנַת אֵ֣לֶם רְ֭חֹקִים לְדָוִ֣ד מִכְתָּ֑ם בֶּֽאֱחֹ֨ז אֹת֖וֹ פְלִשְׁתִּ֣ים בְּגַֽת׃ (ק) חָנֵּ֣נִי אֱ֭לֹהִים כִּֽי־ שְׁאָפַ֣נִי אֱנ֑וֹשׁ כָּל־ הַ֝יּ֗וֹם לֹחֵ֥ם יִלְחָצֵֽנִי׃ (קא) שָׁאֲפ֣וּ שׁ֭וֹרְרַי כָּל־ הַיּ֑וֹם כִּֽי־ רַבִּ֨ים לֹחֲמִ֖ים לִ֣י מָרֽוֹם׃ (קב) י֥וֹם אִירָ֑א אֲ֝נִ֗י אֵלֶ֥יךָ אֶבְטָֽח׃ (קג) בֵּאלֹהִים֮ אֲהַלֵּ֪ל דְּבָ֫ר֥וֹ בֵּאלֹהִ֣ים בָּ֭טַחְתִּי לֹ֣א אִירָ֑א מַה־ יַּעֲשֶׂ֖ה בָשָׂ֣ר לִֽי׃ (קד) כָּל־ הַ֭יּוֹם דְּבָרַ֣י יְעַצֵּ֑בוּ עָלַ֖י כָּל־ מַחְשְׁבֹתָ֣ם לָרָֽע׃ (קה) יָג֤וּרוּ ׀ (יצפינו) יִצְפּ֗וֹנוּ הֵ֭מָּה עֲקֵבַ֣י יִשְׁמֹ֑רוּ כַּ֝אֲשֶׁ֗ר קִוּ֥וּ נַפְשִֽׁי׃ (קו) עַל־ אָ֥וֶן פַּלֶּט־ לָ֑מוֹ בְּ֝אַ֗ף עַמִּ֤ים ׀ הוֹרֵ֬ד אֱלֹהִֽים׃ (קז) נֹדִי֮ סָפַ֪רְתָּ֫ה אָ֥תָּה שִׂ֣ימָה דִמְעָתִ֣י בְנֹאדֶ֑ךָ הֲ֝לֹ֗א בְּסִפְרָתֶֽךָ׃ (קח) אָ֥֨ז יָ֘שׁ֤וּבוּ אוֹיְבַ֣י אָ֭חוֹר בְּי֣וֹם אֶקְרָ֑א זֶה־ יָ֝דַ֗עְתִּי כִּֽי־ אֱלֹהִ֥ים לִֽי׃ (קט) בֵּֽ֭אלֹהִים אֲהַלֵּ֣ל דָּבָ֑ר בַּ֝יהוָ֗ה אֲהַלֵּ֥ל דָּבָֽר׃ (קי) בֵּֽאלֹהִ֣ים בָּ֭טַחְתִּי לֹ֣א אִירָ֑א מַה־ יַּעֲשֶׂ֖ה אָדָ֣ם לִֽי׃ (קיא) עָלַ֣י אֱלֹהִ֣ים נְדָרֶ֑יךָ אֲשַׁלֵּ֖ם תּוֹדֹ֣ת לָֽךְ׃ (קיב) כִּ֤י הִצַּ֪לְתָּ נַפְשִׁ֡י מִמָּוֶת֮ הֲלֹ֥א רַגְלַ֗י מִ֫דֶּ֥חִי לְ֭הִֽתְהַלֵּךְ לִפְנֵ֣י אֱלֹהִ֑ים בְּ֝א֗וֹר הַֽחַיִּֽים׃ (קיג) לַמְנַצֵּ֣חַ אַל־ תַּ֭שְׁחֵת לְדָוִ֣ד מִכְתָּ֑ם בְּבָרְח֥וֹ מִפְּנֵי־ שָׁ֝א֗וּל בַּמְּעָרָֽה׃ (קיד) חָנֵּ֤נִי אֱלֹהִ֨ים ׀ חָנֵּ֗נִי כִּ֥י בְךָ֮ חָסָ֪יָה נַ֫פְשִׁ֥י וּבְצֵֽל־ כְּנָפֶ֥יךָ אֶחְסֶ֑ה עַ֝֗ד יַעֲבֹ֥ר הַוּֽוֹת׃ (קטו) אֶ֭קְרָא לֵֽאלֹהִ֣ים עֶלְי֑וֹן לָ֝אֵ֗ל גֹּמֵ֥ר עָלָֽי׃ (קטז) יִשְׁלַ֤ח מִשָּׁמַ֨יִם ׀ וְֽיוֹשִׁיעֵ֗נִי חֵרֵ֣ף שֹׁאֲפִ֣י סֶ֑לָה יִשְׁלַ֥ח אֱ֝לֹהִ֗ים חַסְדּ֥וֹ וַאֲמִתּֽוֹ׃ (קיז) נַפְשִׁ֤י ׀ בְּת֥וֹךְ לְבָאִם֮ אֶשְׁכְּבָ֪ה לֹ֫הֲטִ֥ים בְּֽנֵי־ אָדָ֗ם שִׁ֭נֵּיהֶם חֲנִ֣ית וְחִצִּ֑ים וּ֝לְשׁוֹנָ֗ם חֶ֣רֶב חַדָּֽה׃ (קיח) ר֣וּמָה עַל־ הַשָּׁמַ֣יִם אֱלֹהִ֑ים עַ֖ל כָּל־ הָאָ֣רֶץ כְּבוֹדֶֽךָ׃ (קיט) רֶ֤שֶׁת ׀ הֵכִ֣ינוּ לִפְעָמַי֮ כָּפַ֪ף נַ֫פְשִׁ֥י כָּר֣וּ לְפָנַ֣י שִׁיחָ֑ה נָפְל֖וּ בְתוֹכָ֣הּ סֶֽלָה׃ (קכ) נָ֘כ֤וֹן לִבִּ֣י אֱ֭לֹהִים נָכ֣וֹן לִבִּ֑י אָ֝שִׁ֗ירָה וַאֲזַמֵּֽרָה׃
