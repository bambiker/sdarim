(א) וְה֤וּא רַח֨וּם ׀ יְכַפֵּ֥ר עָוֺן֮ וְֽלֹא־ יַ֫שְׁחִ֥ית וְ֭הִרְבָּה לְהָשִׁ֣יב אַפּ֑וֹ וְלֹֽא־ יָ֝עִיר כָּל־ חֲמָתֽוֹ׃ (ב) וַ֭יִּזְכֹּר כִּי־ בָשָׂ֣ר הֵ֑מָּה ר֥וּחַ ה֝וֹלֵ֗ךְ וְלֹ֣א יָשֽׁוּב׃ (ג) כַּ֭מָּה יַמְר֣וּהוּ בַמִּדְבָּ֑ר יַ֝עֲצִיב֗וּהוּ בִּֽישִׁימֽוֹן׃ (ד) וַיָּשׁ֣וּבוּ וַיְנַסּ֣וּ אֵ֑ל וּקְד֖וֹשׁ יִשְׂרָאֵ֣ל הִתְווּ׃ (ה) לֹא־ זָכְר֥וּ אֶת־ יָד֑וֹ י֝֗וֹם אֲ‍ֽשֶׁר־ פָּדָ֥ם מִנִּי־ צָֽר׃ (ו) אֲשֶׁר־ שָׂ֣ם בְּ֭מִצְרַיִם אֹֽתוֹתָ֑יו וּ֝מוֹפְתָ֗יו בִּשְׂדֵה־ צֹֽעַן׃ (ז) וַיַּהֲפֹ֣ךְ לְ֭דָם יְאֹרֵיהֶ֑ם וְ֝נֹזְלֵיהֶ֗ם בַּל־ יִשְׁתָּיֽוּן׃ (ח) יְשַׁלַּ֬ח בָּהֶ֣ם עָ֭רֹב וַיֹּאכְלֵ֑ם וּ֝צְפַרְדֵּ֗עַ וַתַּשְׁחִיתֵֽם׃ (ט) וַיִּתֵּ֣ן לֶחָסִ֣יל יְבוּלָ֑ם וִֽ֝יגִיעָ֗ם לָאַרְבֶּֽה׃ (י) יַהֲרֹ֣ג בַּבָּרָ֣ד גַּפְנָ֑ם וְ֝שִׁקְמוֹתָ֗ם בַּֽחֲנָמַֽל׃ (יא) וַיַּסְגֵּ֣ר לַבָּרָ֣ד בְּעִירָ֑ם וּ֝מִקְנֵיהֶ֗ם לָרְשָׁפִֽים׃ (יב) יְשַׁלַּח־ בָּ֨ם ׀ חֲר֬וֹן אַפּ֗וֹ עֶבְרָ֣ה וָזַ֣עַם וְצָרָ֑ה מִ֝שְׁלַ֗חַת מַלְאֲכֵ֥י רָעִֽים׃ (יג) יְפַלֵּ֥ס נָתִ֗יב לְאַ֫פּ֥וֹ לֹא־ חָשַׂ֣ךְ מִמָּ֣וֶת נַפְשָׁ֑ם וְ֝חַיָּתָ֗ם לַדֶּ֥בֶר הִסְגִּֽיר׃ (יד) וַיַּ֣ךְ כָּל־ בְּכ֣וֹר בְּמִצְרָ֑יִם רֵאשִׁ֥ית א֝וֹנִ֗ים בְּאָהֳלֵי־ חָֽם׃ (טו) וַיַּסַּ֣ע כַּצֹּ֣אן עַמּ֑וֹ וַֽיְנַהֲגֵ֥ם כַּ֝עֵ֗דֶר בַּמִּדְבָּֽר׃ (טז) וַיַּנְחֵ֣ם לָ֭בֶטַח וְלֹ֣א פָחָ֑דוּ וְאֶת־ א֝וֹיְבֵיהֶ֗ם כִּסָּ֥ה הַיָּֽם׃ (יז) וַ֭יְבִיאֵם אֶל־ גְּב֣וּל קָדְשׁ֑וֹ הַר־ זֶ֝֗ה קָנְתָ֥ה יְמִינֽוֹ׃ (יח) וַיְגָ֤רֶשׁ מִפְּנֵיהֶ֨ם ׀ גּוֹיִ֗ם וַֽ֭יַּפִּילֵם בְּחֶ֣בֶל נַחֲלָ֑ה וַיַּשְׁכֵּ֥ן בְּ֝אָהֳלֵיהֶ֗ם שִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (יט) וַיְנַסּ֣וּ וַ֭יַּמְרוּ אֶת־ אֱלֹהִ֣ים עֶלְי֑וֹן וְ֝עֵדוֹתָ֗יו לֹ֣א שָׁמָֽרוּ׃ (כ) וַיִּסֹּ֣גוּ וַֽ֭יִּבְגְּדוּ כַּאֲבוֹתָ֑ם נֶ֝הְפְּכ֗וּ כְּקֶ֣שֶׁת רְמִיָּֽה׃ (כא) וַיַּכְעִיס֥וּהוּ בְּבָמוֹתָ֑ם וּ֝בִפְסִילֵיהֶ֗ם יַקְנִיאֽוּהוּ׃ (כב) שָׁמַ֣ע אֱ֭לֹהִים וַֽיִּתְעַבָּ֑ר וַיִּמְאַ֥ס מְ֝אֹ֗ד בְּיִשְׂרָאֵֽל׃ (כג) וַ֭יִּטֹּשׁ מִשְׁכַּ֣ן שִׁל֑וֹ אֹ֝֗הֶל שִׁכֵּ֥ן בָּאָדָֽם׃ (כד) וַיִּתֵּ֣ן לַשְּׁבִ֣י עֻזּ֑וֹ וְֽתִפְאַרְתּ֥וֹ בְיַד־ צָֽר׃ (כה) וַיַּסְגֵּ֣ר לַחֶ֣רֶב עַמּ֑וֹ וּ֝בְנַחֲלָת֗וֹ הִתְעַבָּֽר׃ (כו) בַּחוּרָ֥יו אָֽכְלָה־ אֵ֑שׁ וּ֝בְתוּלֹתָ֗יו לֹ֣א הוּלָּֽלוּ׃ (כז) כֹּ֭הֲנָיו בַּחֶ֣רֶב נָפָ֑לוּ וְ֝אַלְמְנֹתָ֗יו לֹ֣א תִבְכֶּֽינָה׃ (כח) וַיִּקַ֖ץ כְּיָשֵׁ֥ן ׀ אֲדֹנָ֑י כְּ֝גִבּ֗וֹר מִתְרוֹנֵ֥ן מִיָּֽיִן׃ (כט) וַיַּךְ־ צָרָ֥יו אָח֑וֹר חֶרְפַּ֥ת ע֝וֹלָ֗ם נָ֣תַן לָֽמוֹ׃ (ל) וַ֭יִּמְאַס בְּאֹ֣הֶל יוֹסֵ֑ף וּֽבְשֵׁ֥בֶט אֶ֝פְרַ֗יִם לֹ֣א בָחָֽר׃ (לא) וַ֭יִּבְחַר אֶת־ שֵׁ֣בֶט יְהוּדָ֑ה אֶֽת־ הַ֥ר צִ֝יּ֗וֹן אֲשֶׁ֣ר אָהֵֽב׃ (לב) וַיִּ֣בֶן כְּמוֹ־ רָ֭מִים מִקְדָּשׁ֑וֹ כְּ֝אֶ֗רֶץ יְסָדָ֥הּ לְעוֹלָֽם׃ (לג) וַ֭יִּבְחַר בְּדָוִ֣ד עַבְדּ֑וֹ וַ֝יִּקָּחֵ֗הוּ מִֽמִּכְלְאֹ֥ת צֹֽאן׃ (לד) מֵאַחַ֥ר עָל֗וֹת הֱ֫בִיא֥וֹ לִ֭רְעוֹת בְּיַעֲקֹ֣ב עַמּ֑וֹ וּ֝בְיִשְׂרָאֵ֗ל נַחֲלָתֽוֹ׃ (לה) וַ֭יִּרְעֵם כְּתֹ֣ם לְבָב֑וֹ וּבִתְבוּנ֖וֹת כַּפָּ֣יו יַנְחֵֽם׃ (לו) מִזְמ֗וֹר לְאָ֫סָ֥ף אֱ‍ֽלֹהִ֡ים בָּ֤אוּ גוֹיִ֨ם ׀ בְּֽנַחֲלָתֶ֗ךָ טִ֭מְּאוּ אֶת־ הֵיכַ֣ל קָדְשֶׁ֑ךָ שָׂ֖מוּ אֶת־ יְרוּשָׁלִַ֣ם לְעִיִּֽים׃ (לז) נָֽתְנ֡וּ אֶת־ נִבְלַ֬ת עֲבָדֶ֗יךָ מַ֭אֲכָל לְע֣וֹף הַשָּׁמָ֑יִם בְּשַׂ֥ר חֲ֝סִידֶ֗יךָ לְחַיְתוֹ־ אָֽרֶץ׃ (לח) שָׁפְכ֬וּ דָמָ֨ם ׀ כַּמַּ֗יִם סְֽבִ֘יב֤וֹת יְֽרוּשָׁלִָ֗ם וְאֵ֣ין קוֹבֵֽר׃ (לט) הָיִ֣ינוּ חֶ֭רְפָּה לִשְׁכֵנֵ֑ינוּ לַ֥עַג וָ֝קֶ֗לֶס לִסְבִיבוֹתֵֽינוּ׃ (מ) עַד־ מָ֣ה יְ֭הוָה תֶּאֱנַ֣ף לָנֶ֑צַח תִּבְעַ֥ר כְּמוֹ־ אֵ֝֗שׁ קִנְאָתֶֽךָ׃ (מא) שְׁפֹ֤ךְ חֲמָתְךָ֨ אֶֽל־ הַגּוֹיִם֮ אֲשֶׁ֪ר לֹא־ יְדָ֫ע֥וּךָ וְעַ֥ל מַמְלָכ֑וֹת אֲשֶׁ֥ר בְּ֝שִׁמְךָ֗ לֹ֣א קָרָֽאוּ׃ (מב) כִּ֭י אָכַ֣ל אֶֽת־ יַעֲקֹ֑ב וְֽאֶת־ נָוֵ֥הוּ הֵשַֽׁמּוּ׃ (מג) אַֽל־ תִּזְכָּר־ לָנוּ֮ עֲוֺנֹ֪ת רִאשֹׁ֫נִ֥ים מַ֭הֵר יְקַדְּמ֣וּנוּ רַחֲמֶ֑יךָ כִּ֖י דַלּ֣וֹנוּ מְאֹֽד׃ (מד) עָזְרֵ֤נוּ ׀ אֱלֹ֘הֵ֤י יִשְׁעֵ֗נוּ עַל־ דְּבַ֥ר כְּבֽוֹד־ שְׁמֶ֑ךָ וְהַצִּילֵ֥נוּ וְכַפֵּ֥ר עַל־ חַ֝טֹּאתֵ֗ינוּ לְמַ֣עַן שְׁמֶֽךָ׃ (מה) לָ֤מָּה ׀ יֹאמְר֣וּ הַגּוֹיִם֮ אַיֵּ֪ה אֱ‍ֽלֹהֵ֫יהֶ֥ם יִוָּדַ֣ע (בגיים) בַּגּוֹיִ֣ם לְעֵינֵ֑ינוּ נִ֝קְמַ֗ת דַּֽם־ עֲבָדֶ֥יךָ הַשָּׁפֽוּךְ׃ (מו) תָּ֤ב֣וֹא לְפָנֶיךָ֮ אֶנְקַ֪ת אָ֫סִ֥יר כְּגֹ֥דֶל זְרוֹעֲךָ֑ ה֝וֹתֵ֗ר בְּנֵ֣י תְמוּתָֽה׃ (מז) וְהָ֘שֵׁ֤ב לִשְׁכֵנֵ֣ינוּ שִׁ֭בְעָתַיִם אֶל־ חֵיקָ֑ם חֶרְפָּ֘תָ֤ם אֲשֶׁ֖ר חֵרְפ֣וּךָ אֲדֹנָֽי׃ (מח) וַאֲנַ֤חְנוּ עַמְּךָ֨ ׀ וְצֹ֥אן מַרְעִיתֶךָ֮ נ֤וֹדֶ֥ה לְּךָ֗ לְע֫וֹלָ֥ם לְדֹ֥ר וָדֹ֑ר נְ֝סַפֵּ֗ר תְּהִלָּתֶֽךָ׃ (מט) לַמְנַצֵּ֥חַ אֶל־ שֹׁשַׁנִּ֑ים עֵד֖וּת לְאָסָ֣ף מִזְמֽוֹר׃ (נ) רֹ֘עֵ֤ה יִשְׂרָאֵ֨ל ׀ הַאֲזִ֗ינָה נֹהֵ֣ג כַּצֹּ֣אן יוֹסֵ֑ף יֹשֵׁ֖ב הַכְּרוּבִ֣ים הוֹפִֽיעָה׃ (נא) לִפְנֵ֤י אֶפְרַ֨יִם ׀ וּבִנְיָ֘מִ֤ן וּמְנַשֶּׁ֗ה עוֹרְרָ֥ה אֶת־ גְּבֽוּרָתֶ֑ךָ וּלְכָ֖ה לִישֻׁעָ֣תָה לָּֽנוּ׃ (נב) אֱלֹהִ֥ים הֲשִׁיבֵ֑נוּ וְהָאֵ֥ר פָּ֝נֶ֗יךָ וְנִוָּשֵֽׁעָה׃ (נג) יְהוָ֣ה אֱלֹהִ֣ים צְבָא֑וֹת עַד־ מָתַ֥י עָ֝שַׁ֗נְתָּ בִּתְפִלַּ֥ת עַמֶּֽךָ׃ (נד) הֶ֭אֱכַלְתָּם לֶ֣חֶם דִּמְעָ֑ה וַ֝תַּשְׁקֵ֗מוֹ בִּדְמָע֥וֹת שָׁלִֽישׁ׃ (נה) תְּשִׂימֵ֣נוּ מָ֭דוֹן לִשְׁכֵנֵ֑ינוּ וְ֝אֹיְבֵ֗ינוּ יִלְעֲגוּ־ לָֽמוֹ׃ (נו) אֱלֹהִ֣ים צְבָא֣וֹת הֲשִׁיבֵ֑נוּ וְהָאֵ֥ר פָּ֝נֶ֗יךָ וְנִוָּשֵֽׁעָה׃ (נז) גֶּ֭פֶן מִמִּצְרַ֣יִם תַּסִּ֑יעַ תְּגָרֵ֥שׁ גּ֝וֹיִ֗ם וַתִּטָּעֶֽהָ׃ (נח) פִּנִּ֥יתָ לְפָנֶ֑יהָ וַתַּשְׁרֵ֥שׁ שָׁ֝רָשֶׁ֗יהָ וַתְּמַלֵּא־ אָֽרֶץ׃ (נט) כָּסּ֣וּ הָרִ֣ים צִלָּ֑הּ וַ֝עֲנָפֶ֗יהָ אַֽרְזֵי־ אֵֽל׃ (ס) תְּשַׁלַּ֣ח קְצִירֶ֣הָ עַד־ יָ֑ם וְאֶל־ נָ֝הָ֗ר יֽוֹנְקוֹתֶֽיהָ׃ (סא) לָ֭מָּה פָּרַ֣צְתָּ גְדֵרֶ֑יהָ וְ֝אָר֗וּהָ כָּל־ עֹ֥בְרֵי דָֽרֶךְ׃ (סב) יְכַרְסְמֶ֣נָּֽה חֲזִ֣יר מִיָּ֑ וְזִ֖יז שָׂדַ֣י יִרְעֶֽנָּה׃ (סג) אֱלֹהִ֣ים צְבָאוֹת֮ שֽׁ֫וּב־ נָ֥א הַבֵּ֣ט מִשָּׁמַ֣יִם וּרְאֵ֑ה וּ֝פְקֹ֗ד גֶּ֣פֶן זֹֽאת׃ (סד) וְ֭כַנָּה אֲשֶׁר־ נָטְעָ֣ה יְמִינֶ֑ךָ וְעַל־ בֵּ֝֗ן אִמַּ֥צְתָּה לָּֽךְ׃ (סה) שְׂרֻפָ֣ה בָאֵ֣שׁ כְּסוּחָ֑ה מִגַּעֲרַ֖ת פָּנֶ֣יךָ יֹאבֵֽדוּ׃ (סו) תְּֽהִי־ יָ֭דְךָ עַל־ אִ֣ישׁ יְמִינֶ֑ךָ עַל־ בֶּן־ אָ֝דָ֗ם אִמַּ֥צְתָּ לָּֽךְ׃ (סז) וְלֹא־ נָס֥וֹג מִמֶּ֑ךָּ תְּ֝חַיֵּ֗נוּ וּבְשִׁמְךָ֥ נִקְרָֽא׃ (סח) יְה֘וָ֤ה אֱלֹהִ֣ים צְבָא֣וֹת הֲשִׁיבֵ֑נוּ הָאֵ֥ר פָּ֝נֶ֗יךָ וְנִוָּשֵֽׁעָה׃ (סט) לַמְנַצֵּ֬חַ ׀ עַֽל־ הַגִּתִּ֬ית לְאָסָֽף׃ (ע) הַ֭רְנִינוּ לֵאלֹהִ֣ים עוּזֵּ֑נוּ הָ֝רִ֗יעוּ לֵאלֹהֵ֥י יַעֲקֹֽב׃ (עא) שְֽׂאוּ־ זִ֭מְרָה וּתְנוּ־ תֹ֑ף כִּנּ֖וֹר נָעִ֣ים עִם־ נָֽבֶל׃ (עב) תִּקְע֣וּ בַחֹ֣דֶשׁ שׁוֹפָ֑ר בַּ֝כֵּ֗סֶה לְי֣וֹם חַגֵּֽנוּ׃ (עג) כִּ֤י חֹ֣ק לְיִשְׂרָאֵ֣ל ה֑וּא מִ֝שְׁפָּ֗ט לֵאלֹהֵ֥י יַעֲקֹֽב׃ (עד) עֵ֤דוּת ׀ בִּֽיה֘וֹסֵ֤ף שָׂמ֗וֹ בְּ֭צֵאתוֹ עַל־ אֶ֣רֶץ מִצְרָ֑יִם שְׂפַ֖ת לֹא־ יָדַ֣עְתִּי אֶשְׁמָֽע׃ (עה) הֲסִיר֣וֹתִי מִסֵּ֣בֶל שִׁכְמ֑וֹ כַּ֝פָּ֗יו מִדּ֥וּד תַּעֲבֹֽרְנָה׃ (עו) בַּצָּרָ֥ה קָרָ֗אתָ וָאֲחַ֫לְּצֶ֥ךָּ אֶ֭עֶנְךָ בְּסֵ֣תֶר רַ֑עַם אֶבְחָֽנְךָ֨ עַל־ מֵ֖י מְרִיבָ֣ה סֶֽלָה׃ (עז) שְׁמַ֣ע עַ֭מִּי וְאָעִ֣ידָה בָּ֑ךְ יִ֝שְׂרָאֵ֗ל אִם־ תִּֽשְׁמַֽע־ לִֽי׃ (עח) לֹֽא־ יִהְיֶ֣ה בְ֭ךָ אֵ֣ל זָ֑ר וְלֹ֥א תִ֝שְׁתַּחֲוֶ֗ה לְאֵ֣ל נֵכָֽר׃ (עט) אָנֹכִ֨י ׀ יְה֘וָ֤ה אֱלֹהֶ֗יךָ הַֽ֭מַּעַלְךָ מֵאֶ֣רֶץ מִצְרָ֑יִם הַרְחֶב־ פִּ֝֗יךָ וַאֲמַלְאֵֽהוּ׃ (פ) וְלֹא־ שָׁמַ֣ע עַמִּ֣י לְקוֹלִ֑י וְ֝יִשְׂרָאֵ֗ל לֹא־ אָ֥בָה לִֽי׃ (פא) וָֽ֭אֲשַׁלְּחֵהוּ בִּשְׁרִיר֣וּת לִבָּ֑ם יֵ֝לְכ֗וּ בְּֽמוֹעֲצוֹתֵיהֶֽם׃ (פב) ל֗וּ עַ֭מִּי שֹׁמֵ֣עַֽ לִ֑י יִ֝שְׂרָאֵ֗ל בִּדְרָכַ֥י יְהַלֵּֽכוּ׃ (פג) כִּ֭מְעַט אוֹיְבֵיהֶ֣ם אַכְנִ֑יעַ וְעַ֥ל צָ֝רֵיהֶ֗ם אָשִׁ֥יב יָדִֽי׃ (פד) מְשַׂנְאֵ֣י יְ֭הוָה יְכַֽחֲשׁוּ־ ל֑וֹ וִיהִ֖י עִתָּ֣ם לְעוֹלָֽם׃ (פה) וַֽ֭יַּאֲכִילֵהוּ מֵחֵ֣לֶב חִטָּ֑ה וּ֝מִצּ֗וּר דְּבַ֣שׁ אַשְׂבִּיעֶֽךָ׃ (פו) מִזְמ֗וֹר לְאָ֫סָ֥ף אֱ‍ֽלֹהִ֗ים נִצָּ֥ב בַּעֲדַת־ אֵ֑ל בְּקֶ֖רֶב אֱלֹהִ֣ים יִשְׁפֹּֽט׃ (פז) עַד־ מָתַ֥י תִּשְׁפְּטוּ־ עָ֑וֶל וּפְנֵ֥י רְ֝שָׁעִ֗ים תִּשְׂאוּ־ סֶֽלָה׃ (פח) שִׁפְטוּ־ דַ֥ל וְיָת֑וֹם עָנִ֖י וָרָ֣שׁ הַצְדִּֽיקוּ׃ (פט) פַּלְּטוּ־ דַ֥ל וְאֶבְי֑וֹן מִיַּ֖ד רְשָׁעִ֣ים הַצִּֽילוּ׃ (צ) לֹ֤א יָֽדְע֨וּ ׀ וְלֹ֥א יָבִ֗ינוּ בַּחֲשֵׁכָ֥ה יִתְהַלָּ֑כוּ יִ֝מּ֗וֹטוּ כָּל־ מ֥וֹסְדֵי אָֽרֶץ׃ (צא) אֲ‍ֽנִי־ אָ֭מַרְתִּי אֱלֹהִ֣ים אַתֶּ֑ם וּבְנֵ֖י עֶלְי֣וֹן כֻּלְּכֶֽם׃ (צב) אָ֭כֵן כְּאָדָ֣ם תְּמוּת֑וּן וּכְאַחַ֖ד הַשָּׂרִ֣ים תִּפֹּֽלוּ׃ (צג) קוּמָ֣ה אֱ֭לֹהִים שָׁפְטָ֣ה הָאָ֑רֶץ כִּֽי־ אַתָּ֥ה תִ֝נְחַ֗ל בְּכָל־ הַגּוֹיִֽם׃ (צד) שִׁ֖יר מִזְמ֣וֹר לְאָסָֽף׃ (צה) אֱלֹהִ֥ים אַל־ דֳּמִי־ לָ֑ךְ אַל־ תֶּחֱרַ֖שׁ וְאַל־ תִּשְׁקֹ֣ט אֵֽל׃ (צו) כִּֽי־ הִנֵּ֣ה א֭וֹיְבֶיךָ יֶהֱמָי֑וּן וּ֝מְשַׂנְאֶ֗יךָ נָ֣שְׂאוּ רֹֽאשׁ׃ (צז) עַֽל־ עַ֭מְּךָ יַעֲרִ֣ימוּ ס֑וֹד וְ֝יִתְיָעֲצ֗וּ עַל־ צְפוּנֶֽיךָ׃ (צח) אָמְר֗וּ לְ֭כוּ וְנַכְחִידֵ֣ם מִגּ֑וֹי וְלֹֽא־ יִזָּכֵ֖ר שֵֽׁם־ יִשְׂרָאֵ֣ל עֽוֹד׃ (צט) כִּ֤י נוֹעֲצ֣וּ לֵ֣ב יַחְדָּ֑ו עָ֝לֶ֗יךָ בְּרִ֣ית יִכְרֹֽתוּ׃ (ק) אָהֳלֵ֣י אֱ֭דוֹם וְיִשְׁמְעֵאלִ֗ים מוֹאָ֥ב וְהַגְרִֽים׃ (קא) גְּבָ֣ל וְ֭עַמּוֹן וַעֲמָלֵ֑ק פְּ֝לֶ֗שֶׁת עִם־ יֹ֥שְׁבֵי צֽוֹר׃ (קב) גַּם־ אַ֭שּׁוּר נִלְוָ֣ה עִמָּ֑ם הָ֤י֥וּ זְר֖וֹעַ לִבְנֵי־ ל֣וֹט סֶֽלָה׃ (קג) עֲשֵֽׂה־ לָהֶ֥ם כְּמִדְיָ֑ן כְּֽסִֽיסְרָ֥א כְ֝יָבִ֗ין בְּנַ֣חַל קִישֽׁוֹן׃ (קד) נִשְׁמְד֥וּ בְֽעֵין־ דֹּ֑אר הָ֥יוּ דֹּ֝֗מֶן לָאֲדָמָֽה׃ (קה) שִׁיתֵ֣מוֹ נְ֭דִיבֵמוֹ כְּעֹרֵ֣ב וְכִזְאֵ֑ב וּֽכְזֶ֥בַח וּ֝כְצַלְמֻנָּ֗ע כָּל־ נְסִיכֵֽמוֹ׃ (קו) אֲשֶׁ֣ר אָ֭מְרוּ נִ֣ירֲשָׁה לָּ֑נוּ אֵ֝֗ת נְא֣וֹת אֱלֹהִֽים׃ (קז) אֱ‍ֽלֹהַ֗י שִׁיתֵ֥מוֹ כַגַּלְגַּ֑ל כְּ֝קַ֗שׁ לִפְנֵי־ רֽוּחַ׃ (קח) כְּאֵ֥שׁ תִּבְעַר־ יָ֑עַר וּ֝כְלֶהָבָ֗ה תְּלַהֵ֥ט הָרִֽים׃ (קט) כֵּ֭ן תִּרְדְּפֵ֣ם בְּסַעֲרֶ֑ךָ וּבְסוּפָתְךָ֥ תְבַהֲלֵֽם׃ (קי) מַלֵּ֣א פְנֵיהֶ֣ם קָל֑וֹן וִֽיבַקְשׁ֖וּ שִׁמְךָ֣ יְהוָֽה׃ (קיא) יֵבֹ֖שׁוּ וְיִבָּהֲל֥וּ עֲדֵי־ עַ֗ד וְֽיַחְפְּר֥וּ וְיֹאבֵֽדוּ׃ (קיב) וְֽיֵדְע֗וּ כִּֽי־ אַתָּ֬ה שִׁמְךָ֣ יְהוָ֣ה לְבַדֶּ֑ךָ עֶ֝לְי֗וֹן עַל־ כָּל־ הָאָֽרֶץ׃ (קיג) לַמְנַצֵּ֥חַ עַֽל־ הַגִּתִּ֑ית לִבְנֵי־ קֹ֥רַח מִזְמֽוֹר׃ (קיד) מַה־ יְּדִיד֥וֹת מִשְׁכְּנוֹתֶ֗יךָ יְהוָ֥ה צְבָאֽוֹת׃ (קטו) נִכְסְפָ֬ה וְגַם־ כָּלְתָ֨ה ׀ נַפְשִׁי֮ לְחַצְר֪וֹת יְה֫וָ֥ה לִבִּ֥י וּבְשָׂרִ֑י יְ֝רַנְּנ֗וּ אֶ֣ל אֵֽל־ חָֽי׃ (קטז) גַּם־ צִפּ֨וֹר ׀ מָ֪צְאָה בַ֡יִת וּדְר֤וֹר ׀ קֵ֥ן לָהּ֮ אֲשֶׁר־ שָׁ֪תָה אֶפְרֹ֫חֶ֥יהָ אֶֽת־ מִ֭זְבְּחוֹתֶיךָ יְהוָ֣ה צְבָא֑וֹת מַ֝לְכִּ֗י וֵאלֹהָֽי׃ (קיז) אַ֭שְׁרֵי יוֹשְׁבֵ֣י בֵיתֶ֑ךָ ע֝֗וֹד יְֽהַלְל֥וּךָ סֶּֽלָה׃ (קיח) אַשְׁרֵ֣י אָ֭דָם עֽוֹז־ ל֥וֹ בָ֑ךְ מְ֝סִלּ֗וֹת בִּלְבָבָֽם׃ (קיט) עֹבְרֵ֤י ׀ בְּעֵ֣מֶק הַ֭בָּכָא מַעְיָ֣ן יְשִׁית֑וּהוּ גַּם־ בְּ֝רָכ֗וֹת יַעְטֶ֥ה מוֹרֶֽה׃ (קכ) יֵ֭לְכוּ מֵחַ֣יִל אֶל־ חָ֑יִל יֵרָאֶ֖ה אֶל־ אֱלֹהִ֣ים בְּצִיּֽוֹן׃ (קכא) יְה֘וָ֤ה אֱלֹהִ֣ים צְ֭בָאוֹת שִׁמְעָ֣ה תְפִלָּתִ֑י הַאֲזִ֨ינָה אֱלֹהֵ֖י יַעֲקֹ֣ב סֶֽלָה׃ (קכב) מָ֭גִנֵּנוּ רְאֵ֣ה אֱלֹהִ֑ים וְ֝הַבֵּ֗ט פְּנֵ֣י מְשִׁיחֶֽךָ׃ (קכג) כִּ֤י טֽוֹב־ י֥וֹם בַּחֲצֵרֶ֗יךָ מֵ֫אָ֥לֶף בָּחַ֗רְתִּי הִ֭סְתּוֹפֵף בְּבֵ֣ית אֱלֹהַ֑י מִ֝דּ֗וּר בְּאָהֳלֵי־ רֶֽשַׁע׃ (קכד) כִּ֤י שֶׁ֨מֶשׁ ׀ וּמָגֵן֮ יְהוָ֪ה אֱלֹ֫הִ֥ים חֵ֣ן וְ֭כָבוֹד יִתֵּ֣ן יְהוָ֑ה לֹ֥א יִמְנַע־ ט֝֗וֹב לַֽהֹלְכִ֥ים בְּתָמִֽים׃
