תהילים סדר טז

פתיחה באתר · ספר תהילים כולו ·

(א) רֵ֘אשִׁ֤ית חָכְמָ֨ה ׀ יִרְאַ֬ת יְהוָ֗ה שֵׂ֣כֶל ט֭וֹב לְכָל־ עֹשֵׂיהֶ֑ם תְּ֝הִלָּת֗וֹ עֹמֶ֥דֶת לָעַֽד׃ (ב) הַ֥לְלוּ יָ֨הּ ׀ אַשְׁרֵי־ אִ֭ישׁ יָרֵ֣א אֶת־ יְהוָ֑ה בְּ֝מִצְוֺתָ֗יו חָפֵ֥ץ מְאֹֽד׃ (ג) גִּבּ֣וֹר בָּ֭אָרֶץ יִהְיֶ֣ה זַרְע֑וֹ דּ֭וֹר יְשָׁרִ֣ים יְבֹרָֽךְ׃ (ד) הוֹן־ וָעֹ֥שֶׁר בְּבֵית֑וֹ וְ֝צִדְקָת֗וֹ עֹמֶ֥דֶת לָעַֽד׃ (ה) זָ֘רַ֤ח בַּחֹ֣שֶׁךְ א֭וֹר לַיְשָׁרִ֑ים חַנּ֖וּן וְרַח֣וּם וְצַדִּֽיק׃ (ו) טֽוֹב־ אִ֭ישׁ חוֹנֵ֣ן וּמַלְוֶ֑ה יְכַלְכֵּ֖ל דְּבָרָ֣יו בְּמִשְׁפָּֽט׃ (ז) כִּֽי־ לְעוֹלָ֥ם לֹא־ יִמּ֑וֹט לְזֵ֥כֶר ע֝וֹלָ֗ם יִהְיֶ֥ה צַדִּֽיק׃ (ח) מִשְּׁמוּעָ֣ה רָ֭עָה לֹ֣א יִירָ֑א נָכ֥וֹן לִ֝בּ֗וֹ בָּטֻ֥חַ בַּיהוָֽה׃ (ט) סָמ֣וּךְ לִ֭בּוֹ לֹ֣א יִירָ֑א עַ֖ד אֲשֶׁר־ יִרְאֶ֣ה בְצָרָֽיו׃ (י) פִּזַּ֤ר ׀ נָ֘תַ֤ן לָאֶבְיוֹנִ֗ים צִ֭דְקָתוֹ עֹמֶ֣דֶת לָעַ֑ד קַ֝רְנ֗וֹ תָּר֥וּם בְּכָבֽוֹד׃ (יא) רָ֘שָׁ֤ע יִרְאֶ֨ה ׀ וְכָעָ֗ס שִׁנָּ֣יו יַחֲרֹ֣ק וְנָמָ֑ס תַּאֲוַ֖ת רְשָׁעִ֣ים תֹּאבֵֽד׃ (יב) הַ֥לְלוּ יָ֨הּ ׀ הַ֭לְלוּ עַבְדֵ֣י יְהוָ֑ה הַֽ֝לְלוּ אֶת־ שֵׁ֥ם יְהוָֽה׃ (יג) יְהִ֤י שֵׁ֣ם יְהוָ֣ה מְבֹרָ֑ךְ מֵֽ֝עַתָּ֗ה וְעַד־ עוֹלָֽם׃ (יד) מִמִּזְרַח־ שֶׁ֥מֶשׁ עַד־ מְבוֹא֑וֹ מְ֝הֻלָּ֗ל שֵׁ֣ם יְהוָֽה׃ (טו) רָ֖ם עַל־ כָּל־ גּוֹיִ֥ם ׀ יְהוָ֑ה עַ֖ל הַשָּׁמַ֣יִם כְּבוֹדֽוֹ׃ (טז) מִ֭י כַּיהוָ֣ה אֱלֹהֵ֑ינוּ הַֽמַּגְבִּיהִ֥י לָשָֽׁבֶת׃ (יז) הַֽמַּשְׁפִּילִ֥י לִרְא֑וֹת בַּשָּׁמַ֥יִם וּבָאָֽרֶץ׃ (יח) מְקִֽימִ֣י מֵעָפָ֣ר דָּ֑ל מֵֽ֝אַשְׁפֹּ֗ת יָרִ֥ים אֶבְיֽוֹן׃ (יט) לְהוֹשִׁיבִ֥י עִם־ נְדִיבִ֑ים עִ֝֗ם נְדִיבֵ֥י עַמּֽוֹ׃ (כ) מֽוֹשִׁיבִ֨י ׀ עֲקֶ֬רֶת הַבַּ֗יִת אֵֽם־ הַבָּנִ֥ים שְׂמֵחָ֗ה הַֽלְלוּ־ יָֽהּ׃ (כא) בְּצֵ֣את יִ֭שְׂרָאֵל מִמִּצְרָ֑יִם בֵּ֥ית יַ֝עֲקֹ֗ב מֵעַ֥ם לֹעֵֽז׃ (כב) הָיְתָ֣ה יְהוּדָ֣ה לְקָדְשׁ֑וֹ יִ֝שְׂרָאֵ֗ל מַמְשְׁלוֹתָֽיו׃ (כג) הַיָּ֣ם רָ֭אָה וַיָּנֹ֑ס הַ֝יַּרְדֵּ֗ן יִסֹּ֥ב לְאָחֽוֹר׃ (כד) הֶֽ֭הָרִים רָקְד֣וּ כְאֵילִ֑ים גְּ֝בָע֗וֹת כִּבְנֵי־ צֹֽאן׃ (כה) מַה־ לְּךָ֣ הַ֭יָּם כִּ֣י תָנ֑וּס הַ֝יַּרְדֵּ֗ן תִּסֹּ֥ב לְאָחֽוֹר׃ (כו) הֶֽ֭הָרִים תִּרְקְד֣וּ כְאֵילִ֑ים גְּ֝בָע֗וֹת כִּבְנֵי־ צֹֽאן׃ (כז) מִלִּפְנֵ֣י אָ֭דוֹן ח֣וּלִי אָ֑רֶץ מִ֝לִּפְנֵ֗י אֱל֣וֹהַּ יַעֲקֹֽב׃ (כח) הַהֹפְכִ֣י הַצּ֣וּר אֲגַם־ מָ֑יִם חַ֝לָּמִ֗ישׁ לְמַעְיְנוֹ־ מָֽיִם׃ (כט) לֹ֤א לָ֥נוּ יְהוָ֗ה לֹ֫א לָ֥נוּ כִּֽי־ לְ֭שִׁמְךָ תֵּ֣ן כָּב֑וֹד עַל־ חַ֝סְדְּךָ֗ עַל־ אֲמִתֶּֽךָ׃ (ל) לָ֭מָּה יֹאמְר֣וּ הַגּוֹיִ֑ם אַיֵּה־ נָ֝֗א אֱלֹהֵיהֶֽם׃ (לא) וֵֽאלֹהֵ֥ינוּ בַשָּׁמָ֑יִם כֹּ֭ל אֲשֶׁר־ חָפֵ֣ץ עָשָֽׂה׃ (לב) עֲ‍ֽ֭צַבֵּיהֶם כֶּ֣סֶף וְזָהָ֑ב מַ֝עֲשֵׂ֗ה יְדֵ֣י אָדָֽם׃ (לג) פֶּֽה־ לָ֭הֶם וְלֹ֣א יְדַבֵּ֑רוּ עֵינַ֥יִם לָ֝הֶ֗ם וְלֹ֣א יִרְאֽוּ׃ (לד) אָזְנַ֣יִם לָ֭הֶם וְלֹ֣א יִשְׁמָ֑עוּ אַ֥ף לָ֝הֶ֗ם וְלֹ֣א יְרִיחֽוּן׃ (לה) יְדֵיהֶ֤ם ׀ וְלֹ֬א יְמִישׁ֗וּן רַ֭גְלֵיהֶם וְלֹ֣א יְהַלֵּ֑כוּ לֹֽא־ יֶ֝הְגּ֗וּ בִּגְרוֹנָֽם׃ (לו) כְּ֭מוֹהֶם יִהְי֣וּ עֹשֵׂיהֶ֑ם כֹּ֭ל אֲשֶׁר־ בֹּטֵ֣חַ בָּהֶֽם׃ (לז) יִ֭שְׂרָאֵל בְּטַ֣ח בַּיהוָ֑ה עֶזְרָ֖ם וּמָגִנָּ֣ם הֽוּא׃ (לח) בֵּ֣ית אַ֭הֲרֹן בִּטְח֣וּ בַיהוָ֑ה עֶזְרָ֖ם וּמָגִנָּ֣ם הֽוּא׃ (לט) יִרְאֵ֣י יְ֭הוָה בִּטְח֣וּ בַיהוָ֑ה עֶזְרָ֖ם וּמָגִנָּ֣ם הֽוּא׃ (מ) יְהוָה֮ זְכָרָ֪נוּ יְבָ֫רֵ֥ךְ יְ֭בָרֵךְ אֶת־ בֵּ֣ית יִשְׂרָאֵ֑ל יְ֝בָרֵ֗ךְ אֶת־ בֵּ֥ית אַהֲרֹֽן׃ (מא) יְ֭בָרֵךְ יִרְאֵ֣י יְהוָ֑ה הַ֝קְּטַנִּ֗ים עִם־ הַגְּדֹלִֽים׃ (מב) יֹסֵ֣ף יְהוָ֣ה עֲלֵיכֶ֑ם עֲ֝לֵיכֶ֗ם וְעַל־ בְּנֵיכֶֽם׃ (מג) בְּרוּכִ֣ים אַ֭תֶּם לַיהוָ֑ה עֹ֝שֵׂ֗ה שָׁמַ֥יִם וָאָֽרֶץ׃ (מד) הַשָּׁמַ֣יִם שָׁ֭מַיִם לַיהוָ֑ה וְ֝הָאָ֗רֶץ נָתַ֥ן לִבְנֵי־ אָדָֽם׃ (מה) לֹ֣א הַ֭מֵּתִים יְהַֽלְלוּ־ יָ֑הּ וְ֝לֹ֗א כָּל־ יֹרְדֵ֥י דוּמָֽה׃ (מו) וַאֲנַ֤חְנוּ ׀ נְבָ֘רֵ֤ךְ יָ֗הּ מֵֽעַתָּ֥ה וְעַד־ עוֹלָ֗ם הַֽלְלוּ־ יָֽהּ׃ (מז) אָ֭הַבְתִּי כִּֽי־ יִשְׁמַ֥ע ׀ יְהוָ֑ה אֶת־ ק֝וֹלִ֗י תַּחֲנוּנָֽי׃ (מח) כִּֽי־ הִטָּ֣ה אָזְנ֣וֹ לִ֑י וּבְיָמַ֥י אֶקְרָֽא׃ (מט) אֲפָפ֤וּנִי ׀ חֶבְלֵי־ מָ֗וֶת וּמְצָרֵ֣י שְׁא֣וֹל מְצָא֑וּנִי צָרָ֖ה וְיָג֣וֹן אֶמְצָֽא׃ (נ) וּבְשֵֽׁם־ יְהוָ֥ה אֶקְרָ֑א אָנָּ֥ה יְ֝הוָ֗ה מַלְּטָ֥ה נַפְשִֽׁי׃ (נא) חַנּ֣וּן יְהֹוָ֣ה וְצַדִּ֑יק וֵ֖אלֹהֵ֣ינוּ מְרַחֵֽם׃ (נב) שֹׁמֵ֣ר פְּתָאיִ֣ם יְהֹוָ֑ה דַּ֝לּוֹתִ֗י וְלִ֣י יְהוֹשִֽׁיעַ׃ (נג) שׁוּבִ֣י נַ֭פְשִׁי לִמְנוּחָ֑יְכִי כִּֽי־ יְ֝הוָ֗ה גָּמַ֥ל עָלָֽיְכִי׃ (נד) כִּ֤י חִלַּ֥צְתָּ נַפְשִׁ֗י מִ֫מָּ֥וֶת אֶת־ עֵינִ֥י מִן־ דִּמְעָ֑ה אֶת־ רַגְלִ֥י מִדֶּֽחִי׃ (נה) אֶ֭תְהַלֵּךְ לִפְנֵ֣י יְהוָ֑ה בְּ֝אַרְצ֗וֹת הַֽחַיִּֽים׃ (נו) הֶ֭אֱמַנְתִּי כִּ֣י אֲדַבֵּ֑ר אֲ֝נִ֗י עָנִ֥יתִי מְאֹֽד׃ (נז) אֲ֭נִי אָמַ֣רְתִּי בְחָפְזִ֑י כָּֽל־ הָאָדָ֥ם כֹּזֵֽב׃ (נח) מָֽה־ אָשִׁ֥יב לַיהוָ֑ה כָּֽל־ תַּגְמוּל֥וֹהִי עָלָֽי׃ (נט) כּוֹס־ יְשׁוּע֥וֹת אֶשָּׂ֑א וּבְשֵׁ֖ם יְהוָ֣ה אֶקְרָֽא׃ (ס) נְ֭דָרַי לַיהוָ֣ה אֲשַׁלֵּ֑ם נֶגְדָה־ נָּ֝֗א לְכָל־ עַמּֽוֹ׃ (סא) יָ֭קָר בְּעֵינֵ֣י יְהוָ֑ה הַ֝מָּ֗וְתָה לַחֲסִידָֽיו׃ (סב) אָֽנָּ֣ה יְהוָה֮ כִּֽי־ אֲנִ֪י עַ֫בְדֶּ֥ךָ אֲ‍ֽנִי־ עַ֭בְדְּךָ בֶּן־ אֲמָתֶ֑ךָ פִּ֝תַּ֗חְתָּ לְמוֹסֵרָֽי׃ (סג) לְֽךָ־ אֶ֭זְבַּח זֶ֣בַח תּוֹדָ֑ה וּבְשֵׁ֖ם יְהוָ֣ה אֶקְרָֽא׃ (סד) נְ֭דָרַי לַיהוָ֣ה אֲשַׁלֵּ֑ם נֶגְדָה־ נָּ֝֗א לְכָל־ עַמּֽוֹ׃ (סה) בְּחַצְר֤וֹת ׀ בֵּ֤ית יְהוָ֗ה בְּֽת֘וֹכֵ֤כִי יְֽרוּשָׁלִָ֗ם הַֽלְלוּ־ יָֽהּ׃ (סו) הַֽלְל֣וּ אֶת־ יְ֭הוָה כָּל־ גּוֹיִ֑ם שַׁ֝בְּח֗וּהוּ כָּל־ הָאֻמִּֽים׃ (סז) כִּ֥י גָ֘בַ֤ר עָלֵ֨ינוּ ׀ חַסְדּ֗וֹ וֶֽאֱמֶת־ יְהוָ֥ה לְעוֹלָ֗ם הַֽלְלוּ־ יָֽהּ׃ (סח) הוֹד֣וּ לַיהוָ֣ה כִּי־ ט֑וֹב כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (סט) יֹֽאמַר־ נָ֥א יִשְׂרָאֵ֑ל כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (ע) יֹֽאמְרוּ־ נָ֥א בֵֽית־ אַהֲרֹ֑ן כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (עא) יֹֽאמְרוּ־ נָ֭א יִרְאֵ֣י יְהוָ֑ה כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (עב) מִֽן־ הַ֭מֵּצַ֥ר קָרָ֣אתִי יָּ֑הּ עָנָ֖נִי בַמֶּרְחָ֣ב יָֽהּ׃ (עג) יְהוָ֣ה לִ֭י לֹ֣א אִירָ֑א מַה־ יַּעֲשֶׂ֖ה לִ֣י אָדָֽם׃ (עד) יְהוָ֣ה לִ֭י בְּעֹזְרָ֑י וַ֝אֲנִ֗י אֶרְאֶ֥ה בְשֹׂנְאָֽי׃ (עה) ט֗וֹב לַחֲס֥וֹת בַּיהוָ֑ה מִ֝בְּטֹ֗חַ בָּאָדָֽם׃ (עו) ט֗וֹב לַחֲס֥וֹת בַּיהוָ֑ה מִ֝בְּטֹ֗חַ בִּנְדִיבִֽים׃ (עז) כָּל־ גּוֹיִ֥ם סְבָב֑וּנִי בְּשֵׁ֥ם יְ֝הוָ֗ה כִּ֣י אֲמִילַֽם׃ (עח) סַבּ֥וּנִי גַם־ סְבָב֑וּנִי בְּשֵׁ֥ם יְ֝הוָ֗ה כִּ֣י אֲמִילַֽם׃ (עט) סַבּ֤וּנִי כִדְבוֹרִ֗ים דֹּ֭עֲכוּ כְּאֵ֣שׁ קוֹצִ֑ים בְּשֵׁ֥ם יְ֝הוָ֗ה כִּ֣י אֲמִילַֽם׃ (פ) דַּחֹ֣ה דְחִיתַ֣נִי לִנְפֹּ֑ל וַ֖יהוָ֣ה עֲזָרָֽנִי׃ (פא) עָזִּ֣י כְּאִמְרָתֶֽךָ׃ (קנה) חִשַּׁ֥בְתִּי דְרָכָ֑י וָאָשִׁ֥יבָה רַ֝גְלַ֗י אֶל־ עֵדֹתֶֽיךָ׃ (קנו) חַ֭שְׁתִּי וְלֹ֣א הִתְמַהְמָ֑הְתִּי לִ֝שְׁמֹ֗ר מִצְוֺתֶֽיךָ׃ (קנז) חֶבְלֵ֣י רְשָׁעִ֣ים עִוְּדֻ֑נִי תּֽ֝וֹרָתְךָ֗ לֹ֣א שָׁכָֽחְתִּי׃ (קנח) חֲצֽוֹת־ לַ֗יְלָה אָ֭קוּם לְהוֹד֣וֹת לָ֑ךְ עַ֝֗ל מִשְׁפְּטֵ֥י צִדְקֶֽךָ׃ (קנט) חָבֵ֣ר אָ֭נִי לְכָל־ אֲשֶׁ֣ר יְרֵא֑וּךָ וּ֝לְשֹׁמְרֵ֗י פִּקּוּדֶֽיךָ׃ (קס) חַסְדְּךָ֣ יְ֭הוָה מָלְאָ֥ה הָאָ֗רֶץ חֻקֶּ֥יךָ לַמְּדֵֽנִי׃ (קסא) ט֭וֹב עָשִׂ֣יתָ עִֽם־ עַבְדְּךָ֑ יְ֝הוָ֗ה כִּדְבָרֶֽךָ׃ (קסב) ט֤וּב טַ֣עַם וָדַ֣עַת לַמְּדֵ֑נִי כִּ֖י בְמִצְוֺתֶ֣יךָ הֶאֱמָֽנְתִּי׃ (קסג) טֶ֣רֶם אֶ֭עֱנֶה אֲנִ֣י שֹׁגֵ֑ג וְ֝עַתָּ֗ה אִמְרָתְךָ֥ שָׁמָֽרְתִּי׃ (קסד) טוֹב־ אַתָּ֥ה וּמֵטִ֗יב לַמְּדֵ֥נִי חֻקֶּֽיךָ׃ (קסה) טָפְל֬וּ עָלַ֣י שֶׁ֣קֶר זֵדִ֑ים אֲ֝נִ֗י בְּכָל־ לֵ֤ב ׀ אֱצֹּ֬ר פִּקּוּדֶֽיךָ׃ (קסו) טָפַ֣שׁ כַּחֵ֣לֶב לִבָּ֑ם אֲ֝נִ֗י תּוֹרָתְךָ֥ שִֽׁעֲשָֽׁעְתִּי׃ (קסז) טֽוֹב־ לִ֥י כִֽי־ עֻנֵּ֑יתִי לְ֝מַ֗עַן אֶלְמַ֥ד חֻקֶּֽיךָ׃
