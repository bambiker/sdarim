פתיחה באתר · ספר תהילים כולו ·

(א) בַּעֲב֤וּר ׀ יִשְׁמְר֣וּ חֻ֭קָּיו וְתוֹרֹתָ֥יו יִנְצֹ֗רוּ הַֽלְלוּ־ יָֽהּ׃ (ב) הַֽלְלוּיָ֨הּ ׀ הוֹד֣וּ לַיהוָ֣ה כִּי־ ט֑וֹב כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (ג) מִ֗י יְ֭מַלֵּל גְּבוּר֣וֹת יְהוָ֑ה יַ֝שְׁמִ֗יעַ כָּל־ תְּהִלָּתֽוֹ׃ (ד) אַ֭שְׁרֵי שֹׁמְרֵ֣י מִשְׁפָּ֑ט עֹשֵׂ֖ה צְדָקָ֣ה בְכָל־ עֵֽת׃ (ה) זָכְרֵ֣נִי יְ֭הוָה בִּרְצ֣וֹן עַמֶּ֑ךָ פָּ֝קְדֵ֗נִי בִּישׁוּעָתֶֽךָ׃ (ו) לִרְא֤וֹת ׀ בְּט֘וֹבַ֤ת בְּחִירֶ֗יךָ לִ֭שְׂמֹחַ בְּשִׂמְחַ֣ת גּוֹיֶ֑ךָ לְ֝הִתְהַלֵּ֗ל עִם־ נַחֲלָתֶֽךָ׃ (ז) חָטָ֥אנוּ עִם־ אֲבוֹתֵ֗ינוּ הֶעֱוִ֥ינוּ הִרְשָֽׁעְנוּ׃ (ח) אֲב֘וֹתֵ֤ינוּ בְמִצְרַ֨יִם ׀ לֹא־ הִשְׂכִּ֬ילוּ נִפְלְאוֹתֶ֗יךָ לֹ֣א זָ֭כְרוּ אֶת־ רֹ֣ב חֲסָדֶ֑יךָ וַיַּמְר֖וּ עַל־ יָ֣ם בְּיַם־ סֽוּף׃ (ט) וַֽ֭יּוֹשִׁיעֵם לְמַ֣עַן שְׁמ֑וֹ לְ֝הוֹדִ֗יעַ אֶת־ גְּבוּרָתֽוֹ׃ (י) וַיִּגְעַ֣ר בְּיַם־ ס֭וּף וַֽיֶּחֱרָ֑ב וַיּוֹלִיכֵ֥ם בַּ֝תְּהֹמ֗וֹת כַּמִּדְבָּֽר׃ (יא) וַֽ֭יּוֹשִׁיעֵם מִיַּ֣ד שׂוֹנֵ֑א וַ֝יִּגְאָלֵ֗ם מִיַּ֥ד אוֹיֵֽב׃ (יב) וַיְכַסּוּ־ מַ֥יִם צָרֵיהֶ֑ם אֶחָ֥ד מֵ֝הֶ֗ם לֹ֣א נוֹתָֽר׃ (יג) וַיַּאֲמִ֥ינוּ בִדְבָרָ֑יו יָ֝שִׁ֗ירוּ תְּהִלָּתֽוֹ׃ (יד) מִֽ֭הֲרוּ שָׁכְח֣וּ מַעֲשָׂ֑יו לֹֽא־ חִ֝כּ֗וּ לַעֲצָתֽוֹ׃ (טו) וַיִּתְאַוּ֣וּ תַ֭אֲוָה בַּמִּדְבָּ֑ר וַיְנַסּוּ־ אֵ֝֗ל בִּֽישִׁימֽוֹן׃ (טז) וַיִּתֵּ֣ן לָ֭הֶם שֶׁאֱלָתָ֑ם וַיְשַׁלַּ֖ח רָז֣וֹן בְּנַפְשָֽׁם׃ (יז) וַיְקַנְא֣וּ לְ֭מֹשֶׁה בַּֽמַּחֲנֶ֑ה לְ֝אַהֲרֹ֗ן קְד֣וֹשׁ יְהוָֽה׃ (יח) תִּפְתַּח־ אֶ֭רֶץ וַתִּבְלַ֣ע דָּתָ֑ן וַ֝תְּכַ֗ס עַל־ עֲדַ֥ת אֲבִירָֽם׃ (יט) וַתִּבְעַר־ אֵ֥שׁ בַּעֲדָתָ֑ם לֶ֝הָבָ֗ה תְּלַהֵ֥ט רְשָׁעִֽים׃ (כ) יַעֲשׂוּ־ עֵ֥גֶל בְּחֹרֵ֑ב וַ֝יִּשְׁתַּחֲו֗וּ לְמַסֵּכָֽה׃ (כא) וַיָּמִ֥ירוּ אֶת־ כְּבוֹדָ֑ם בְּתַבְנִ֥ית שׁ֝֗וֹר אֹכֵ֥ל עֵֽשֶׂב׃ (כב) שָׁ֭כְחוּ אֵ֣ל מוֹשִׁיעָ֑ם עֹשֶׂ֖ה גְדֹל֣וֹת בְּמִצְרָֽיִם׃ (כג) נִ֭פְלָאוֹת בְּאֶ֣רֶץ חָ֑ם נ֝וֹרָא֗וֹת עַל־ יַם־ סֽוּף׃ (כד) וַיֹּ֗אמֶר לְֽהַשְׁמִ֫ידָ֥ם לוּלֵ֡י מֹ֘שֶׁ֤ה בְחִיר֗וֹ עָמַ֣ד בַּפֶּ֣רֶץ לְפָנָ֑יו לְהָשִׁ֥יב חֲ֝מָת֗וֹ מֵֽהַשְׁחִֽית׃ (כה) וַֽ֭יִּמְאֲסוּ בְּאֶ֣רֶץ חֶמְדָּ֑ה לֹֽא־ הֶ֝אֱמִ֗ינוּ לִדְבָרֽוֹ׃ (כו) וַיֵּרָגְנ֥וּ בְאָהֳלֵיהֶ֑ם לֹ֥א שָׁ֝מְע֗וּ בְּק֣וֹל יְהוָֽה׃ (כז) וַיִּשָּׂ֣א יָד֣וֹ לָהֶ֑ם לְהַפִּ֥יל א֝וֹתָ֗ם בַּמִּדְבָּֽר׃ (כח) וּלְהַפִּ֣יל זַ֭רְעָם בַּגּוֹיִ֑ם וּ֝לְזָרוֹתָ֗ם בָּאֲרָצֽוֹת׃ (כט) וַ֭יִּצָּ֣מְדוּ לְבַ֣עַל פְּע֑וֹר וַ֝יֹּאכְל֗וּ זִבְחֵ֥י מֵתִֽים׃ (ל) וַ֭יַּכְעִיסוּ בְּמַֽעַלְלֵיהֶ֑ם וַתִּפְרָץ־ בָּ֝֗ם מַגֵּפָֽה׃ (לא) וַיַּעֲמֹ֣ד פִּֽ֭ינְחָס וַיְפַלֵּ֑ל וַ֝תֵּעָצַ֗ר הַמַּגֵּפָֽה׃ (לב) וַתֵּחָ֣שֶׁב ל֭וֹ לִצְדָקָ֑ה לְדֹ֥ר וָ֝דֹ֗ר עַד־ עוֹלָֽם׃ (לג) וַ֭יַּקְצִיפוּ עַל־ מֵ֥י מְרִיבָ֑ה וַיֵּ֥רַע לְ֝מֹשֶׁ֗ה בַּעֲבוּרָֽם׃ (לד) כִּֽי־ הִמְר֥וּ אֶת־ רוּח֑וֹ וַ֝יְבַטֵּ֗א בִּשְׂפָתָֽיו׃ (לה) לֹֽא־ הִ֭שְׁמִידוּ אֶת־ הָֽעַמִּ֑ים אֲשֶׁ֤ר אָמַ֖ר יְהוָ֣ה לָהֶֽם׃ (לו) וַיִּתְעָרְב֥וּ בַגּוֹיִ֑ם וַֽ֝יִּלְמְד֗וּ מַֽעֲשֵׂיהֶֽם׃ (לז) וַיַּעַבְד֥וּ אֶת־ עֲצַבֵּיהֶ֑ם וַיִּהְי֖וּ לָהֶ֣ם לְמוֹקֵֽשׁ׃ (לח) וַיִּזְבְּח֣וּ אֶת־ בְּ֭נֵיהֶם וְאֶת־ בְּנֽוֹתֵיהֶ֗ם לַשֵּֽׁדִים׃ (לט) וַיִּֽשְׁפְּכ֨וּ דָ֪ם נָקִ֡י דַּם־ בְּנֵ֘יהֶ֤ם וּֽבְנוֹתֵיהֶ֗ם אֲשֶׁ֣ר זִ֭בְּחוּ לַעֲצַבֵּ֣י כְנָ֑עַן וַתֶּחֱנַ֥ף הָ֝אָ֗רֶץ בַּדָּמִֽים׃ (מ) וַיִּטְמְא֥וּ בְמַעֲשֵׂיהֶ֑ם וַ֝יִּזְנוּ֗ בְּמַֽעַלְלֵיהֶֽם׃ (מא) וַיִּֽחַר־ אַ֣ף יְהוָ֣ה בְּעַמּ֑וֹ וַ֝יְתָעֵ֗ב אֶת־ נַחֲלָתֽוֹ׃ (מב) וַיִּתְּנֵ֥ם בְּיַד־ גּוֹיִ֑ם וַֽיִּמְשְׁל֥וּ בָ֝הֶ֗ם שֹׂנְאֵיהֶֽם׃ (מג) וַיִּלְחָצ֥וּם אוֹיְבֵיהֶ֑ם וַ֝יִּכָּנְע֗וּ תַּ֣חַת יָדָֽם׃ (מד) פְּעָמִ֥ים רַבּ֗וֹת יַצִּ֫ילֵ֥ם וְ֭הֵמָּה יַמְר֣וּ בַעֲצָתָ֑ם וַ֝יָּמֹ֗כּוּ בַּעֲוֺנָֽם׃ (מה) וַ֭יַּרְא בַּצַּ֣ר לָהֶ֑ם בְּ֝שָׁמְע֗וֹ אֶת־ רִנָּתָֽם׃ (מו) וַיִּזְכֹּ֣ר לָהֶ֣ם בְּרִית֑וֹ וַ֝יִּנָּחֵ֗ם כְּרֹ֣ב (חסדו) חֲסָדָֽיו׃ (מז) וַיִּתֵּ֣ן אוֹתָ֣ם לְרַחֲמִ֑ים לִ֝פְנֵ֗י כָּל־ שׁוֹבֵיהֶֽם׃ (מח) הוֹשִׁיעֵ֨נוּ ׀ יְה֘וָ֤ה אֱלֹהֵ֗ינוּ וְקַבְּצֵנוּ֮ מִֽן־ הַגּ֫וֹיִ֥ם לְ֭הֹדוֹת לְשֵׁ֣ם קָדְשֶׁ֑ךָ לְ֝הִשְׁתַּבֵּ֗חַ בִּתְהִלָּתֶֽךָ׃ (מט) בָּר֤וּךְ־ יְהוָ֨ה אֱלֹהֵ֪י יִשְׂרָאֵ֡ל מִן־ הָ֤עוֹלָ֨ם ׀ וְעַ֬ד הָעוֹלָ֗ם וְאָמַ֖ר כָּל־ הָעָ֥ם אָמֵ֗ן הַֽלְלוּ־ יָֽהּ׃ (נ) הֹד֣וּ לַיהוָ֣ה כִּי־ ט֑וֹב כִּ֖י לְעוֹלָ֣ם חַסְדּֽוֹ׃ (נא) יֹ֭אמְרוּ גְּאוּלֵ֣י יְהוָ֑ה אֲשֶׁ֥ר גְּ֝אָלָ֗ם מִיַּד־ צָֽר׃ (נב) וּֽמֵאֲרָצ֗וֹת קִ֫בְּצָ֥ם מִמִּזְרָ֥ח וּמִֽמַּעֲרָ֑ב מִצָּפ֥וֹן וּמִיָּֽם׃ (נג) תָּע֣וּ בַ֭מִּדְבָּר בִּישִׁימ֣וֹן דָּ֑רֶךְ עִ֥יר מ֝וֹשָׁ֗ב לֹ֣א מָצָֽאוּ׃ (נד) רְעֵבִ֥ים גַּם־ צְמֵאִ֑ים נַ֝פְשָׁ֗ם בָּהֶ֥ם תִּתְעַטָּֽף׃ (נה) וַיִּצְעֲק֣וּ אֶל־ יְ֭הוָה בַּצַּ֣ר לָהֶ֑ם מִ֝מְּצֽוּקוֹתֵיהֶ֗ם יַצִּילֵֽם׃ (נו) וַ֭יַּֽדְרִיכֵם בְּדֶ֣רֶךְ יְשָׁרָ֑ה לָ֝לֶ֗כֶת אֶל־ עִ֥יר מוֹשָֽׁב׃ (נז) יוֹד֣וּ לַיהוָ֣ה חַסְדּ֑וֹ וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃ (נח) כִּי־ הִ֭שְׂבִּיעַ נֶ֣פֶשׁ שֹׁקֵקָ֑ה וְנֶ֥פֶשׁ רְ֝עֵבָה מִלֵּא־ טֽוֹב׃ (נט) יֹ֭שְׁבֵי חֹ֣שֶׁךְ וְצַלְמָ֑וֶת אֲסִירֵ֖י עֳנִ֣י וּבַרְזֶֽל׃ (ס) כִּֽי־ הִמְר֥וּ אִמְרֵי־ אֵ֑ל וַעֲצַ֖ת עֶלְי֣וֹן נָאָֽצוּ׃ (סא) וַיַּכְנַ֣ע בֶּעָמָ֣ל לִבָּ֑ם כָּ֝שְׁל֗וּ וְאֵ֣ין עֹזֵֽר׃ (סב) וַיִּזְעֲק֣וּ אֶל־ יְ֭הוָה בַּצַּ֣ר לָהֶ֑ם מִ֝מְּצֻֽקוֹתֵיהֶ֗ם יוֹשִׁיעֵֽם׃ (סג) יֽ֭וֹצִיאֵם מֵחֹ֣שֶׁךְ וְצַלְמָ֑וֶת וּמוֹסְר֖וֹתֵיהֶ֣ם יְנַתֵּֽק׃ (סד) יוֹד֣וּ לַיהוָ֣ה חַסְדּ֑וֹ וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃ (סה) כִּֽי־ שִׁ֭בַּר דַּלְת֣וֹת נְחֹ֑שֶׁת וּבְרִיחֵ֖י בַרְזֶ֣ל גִּדֵּֽעַ׃ (סו) אֱ֭וִלִים מִדֶּ֣רֶךְ פִּשְׁעָ֑ם וּֽ֝מֵעֲוֺֽנֹתֵיהֶ֗ם יִתְעַנּֽוּ׃ (סז) כָּל־ אֹ֭כֶל תְּתַעֵ֣ב נַפְשָׁ֑ם וַ֝יַּגִּ֗יעוּ עַד־ שַׁ֥עֲרֵי מָֽוֶת׃ (סח) וַיִּזְעֲק֣וּ אֶל־ יְ֭הוָה בַּצַּ֣ר לָהֶ֑ם מִ֝מְּצֻֽקוֹתֵיהֶ֗ם יוֹשִׁיעֵֽם׃ (סט) יִשְׁלַ֣ח דְּ֭בָרוֹ וְיִרְפָּאֵ֑ם וִֽ֝ימַלֵּ֗ט מִשְּׁחִיתוֹתָֽם׃ (ע) יוֹד֣וּ לַיהוָ֣ה חַסְדּ֑וֹ וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃ (עא) וְ֭יִזְבְּחוּ זִבְחֵ֣י תוֹדָ֑ה וִֽיסַפְּר֖וּ מַעֲשָׂ֣יו בְּרִנָּֽה׃ (עב) יוֹרְדֵ֣י הַ֭יָּם בָּאֳנִיּ֑וֹת עֹשֵׂ֥י מְ֝לָאכָ֗ה בְּמַ֣יִם רַבִּֽים׃ (עג) הֵ֣מָּה רָ֭אוּ מַעֲשֵׂ֣י יְהוָ֑ה וְ֝נִפְלְאוֹתָ֗יו בִּמְצוּלָֽה׃ (עד) וַיֹּ֗אמֶר וַֽ֭יַּעֲמֵד ר֣וּחַ סְעָרָ֑ה וַתְּרוֹמֵ֥ם גַּלָּֽיו׃ (עה) יַעֲל֣וּ שָׁ֭מַיִם יֵרְד֣וּ תְהוֹמ֑וֹת נַ֝פְשָׁ֗ם בְּרָעָ֥ה תִתְמוֹגָֽג׃ (עו) יָח֣וֹגּוּ וְ֭יָנוּעוּ כַּשִּׁכּ֑וֹר וְכָל־ חָ֝כְמָתָ֗ם תִּתְבַּלָּֽע׃ (עז) וַיִּצְעֲק֣וּ אֶל־ יְ֭הוָה בַּצַּ֣ר לָהֶ֑ם וּֽ֝מִמְּצֽוּקֹתֵיהֶ֗ם יוֹצִיאֵֽם׃ (עח) יָקֵ֣ם סְ֭עָרָה לִדְמָמָ֑ה וַ֝יֶּחֱשׁ֗וּ גַּלֵּיהֶֽם׃ (עט) וַיִּשְׂמְח֥וּ כִֽי־ יִשְׁתֹּ֑קוּ וַ֝יַּנְחֵ֗ם אֶל־ מְח֥וֹז חֶפְצָֽם׃ (פ) יוֹד֣וּ לַיהוָ֣ה חַסְדּ֑וֹ וְ֝נִפְלְאוֹתָ֗יו לִבְנֵ֥י אָדָֽם׃ (פא) וִֽ֭ירֹמְמוּהוּ בִּקְהַל־ עָ֑ם וּבְמוֹשַׁ֖ב זְקֵנִ֣ים יְהַלְלֽוּהוּ׃ (פב) יָשֵׂ֣ם נְהָר֣וֹת לְמִדְבָּ֑ר נָכ֣וֹן לִבִּ֣י אֱלֹהִ֑ים אָשִׁ֥ירָה וַ֝אֲזַמְּרָ֗ה אַף־ כְּבוֹדִֽי׃ (צה) ע֭וּרָֽה הַנֵּ֥בֶל וְכִנּ֗וֹר אָעִ֥ירָה שָּֽׁחַר׃ (צו) אוֹדְךָ֖ בָעַמִּ֥ים ׀ יְהוָ֑ה וַ֝אֲזַמֶּרְךָ֗ בַּל־ אֻמִּֽים׃ (צז) יְמִֽינְךָ֣ וַעֲנֵֽנִי׃ (ק) אֱלֹהִ֤ים ׀ דִּבֶּ֥ר בְּקָדְשׁ֗וֹ אֶעְלֹ֥זָה אֲחַלְּקָ֥ה שְׁכֶ֑ם וְעֵ֖מֶק סֻכּ֣וֹת אֲמַדֵּֽד׃ (קא) לִ֤י גִלְעָ֨ד ׀ לִ֤י מְנַשֶּׁ֗ה וְ֭אֶפְרַיִם מָע֣וֹז רֹאשִׁ֑י יְ֝הוּדָ֗ה מְחֹקְקִֽי׃ (קב) מוֹאָ֤ב ׀ סִ֬יר רַחְצִ֗י עַל־ אֱ֭דוֹם אַשְׁלִ֣יךְ נַעֲלִ֑י עֲלֵֽי־ פְ֝לֶ֗שֶׁת אֶתְרוֹעָֽע׃ (קג) מִ֣י יֹ֭בִלֵנִי עִ֣יר מִבְצָ֑ר מִ֖י נָחַ֣נִי עַד־ אֱדֽוֹם׃ (קד) הֲלֹֽא־ אֱלֹהִ֥ים נִשְׁבַּ֤ע יְהוָ֨ה ׀ וְלֹ֥א יִנָּחֵ֗ם אַתָּֽה־ כֹהֵ֥ן לְעוֹלָ֑ם עַל־ דִּ֝בְרָתִ֗י מַלְכִּי־ צֶֽדֶק׃ (קמב) אֲדֹנָ֥י עַל־ יְמִֽינְךָ֑ מָחַ֖ץ בְּיוֹם־ אַפּ֣וֹ מְלָכִֽים׃ (קמג) יָדִ֣ין בַּ֭גּוֹיִם מָלֵ֣א גְוִיּ֑וֹת מָ֥חַץ רֹ֝֗אשׁ עַל־ אֶ֥רֶץ רַבָּֽה׃ (קמד) מִ֭נַּחַל בַּדֶּ֣רֶךְ יִשְׁתֶּ֑ה עַל־ כֵּ֝֗ן יָרִ֥ים רֹֽאשׁ׃ (קמה) הַ֥לְלוּ יָ֨הּ ׀ אוֹדֶ֣ה יְ֭הוָה בְּכָל־ לֵבָ֑ב בְּס֖וֹד יְשָׁרִ֣ים וְעֵדָֽה׃ (קמו) גְּ֭דֹלִים מַעֲשֵׂ֣י יְהוָ֑ה דְּ֝רוּשִׁ֗ים לְכָל־ חֶפְצֵיהֶֽם׃ (קמז) הוֹד־ וְהָדָ֥ר פָּֽעֳל֑וֹ וְ֝צִדְקָת֗וֹ עֹמֶ֥דֶת לָעַֽד׃ (קמח) זֵ֣כֶר עָ֭שָׂה לְנִפְלְאֹתָ֑יו חַנּ֖וּן וְרַח֣וּם יְהוָֽה׃ (קמט) טֶ֭רֶף נָתַ֣ן לִֽירֵאָ֑יו יִזְכֹּ֖ר לְעוֹלָ֣ם בְּרִיתֽוֹ׃ (קנ) כֹּ֣חַ מַ֭עֲשָׂיו הִגִּ֣יד לְעַמּ֑וֹ לָתֵ֥ת לָ֝הֶ֗ם נַחֲלַ֥ת גּוֹיִֽם׃ (קנא) מַעֲשֵׂ֣י יָ֭דָיו אֱמֶ֣ת וּמִשְׁפָּ֑ט נֶ֝אֱמָנִ֗ים כָּל־ פִּקּוּדָֽיו׃ (קנב) סְמוּכִ֣ים לָעַ֣ד לְעוֹלָ֑ם עֲ֝שׂוּיִ֗ם בֶּאֱמֶ֥ת וְיָשָֽׁר׃ (קנג) פְּד֤וּת ׀ שָׁ֘לַ֤ח לְעַמּ֗וֹ צִוָּֽה־ לְעוֹלָ֥ם בְּרִית֑וֹ קָד֖וֹשׁ וְנוֹרָ֣א שְׁמֽוֹ׃
