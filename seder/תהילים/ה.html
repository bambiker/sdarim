(א) וּ֭לְשׁוֹנִי תֶּהְגֶּ֣ה צִדְקֶ֑ךָ כָּל־ הַ֝יּוֹם תְּהִלָּתֶֽךָ׃ (ב) לַמְנַצֵּ֬חַ ׀ לְעֶֽבֶד־ יְהוָ֬ה לְדָוִֽד׃ (ג) נְאֻֽם־ פֶּ֣שַׁע לָ֭רָשָׁע בְּקֶ֣רֶב לִבִּ֑י אֵֽין־ פַּ֥חַד אֱ֝לֹהִ֗ים לְנֶ֣גֶד עֵינָֽיו׃ (ד) כִּֽי־ הֶחֱלִ֣יק אֵלָ֣יו בְּעֵינָ֑יו לִמְצֹ֖א עֲוֺנ֣וֹ לִשְׂנֹֽא׃ (ה) דִּבְרֵי־ פִ֭יו אָ֣וֶן וּמִרְמָ֑ה חָדַ֖ל לְהַשְׂכִּ֣יל לְהֵיטִֽיב׃ (ו) אָ֤וֶן ׀ יַחְשֹׁ֗ב עַֽל־ מִשְׁכָּ֫ב֥וֹ יִ֭תְיַצֵּב עַל־ דֶּ֣רֶךְ לֹא־ ט֑וֹב רָ֝֗ע לֹ֣א יִמְאָֽס׃ (ז) יְ֭הוָה בְּהַשָּׁמַ֣יִם חַסְדֶּ֑ךָ אֱ֝מֽוּנָתְךָ֗ עַד־ שְׁחָקִֽים׃ (ח) צִדְקָֽתְךָ֨ ׀ כְּֽהַרְרֵי־ אֵ֗ל מִ֭שְׁפָּטֶךָ תְּה֣וֹם רַבָּ֑ה אָ֤דָֽם־ וּבְהֵמָ֖ה תוֹשִׁ֣יעַ יְהוָֽה׃ (ט) מַה־ יָּקָ֥ר חַסְדְּךָ֗ אֱלֹ֫הִ֥ים וּבְנֵ֥י אָדָ֑ם בְּצֵ֥ל כְּ֝נָפֶ֗יךָ יֶחֱסָיֽוּן׃ (י) יִ֭רְוְיֻן מִדֶּ֣שֶׁן בֵּיתֶ֑ךָ וְנַ֖חַל עֲדָנֶ֣יךָ תַשְׁקֵֽם׃ (יא) כִּֽי־ עִ֭מְּךָ מְק֣וֹר חַיִּ֑ים בְּ֝אוֹרְךָ֗ נִרְאֶה־ אֽוֹר׃ (יב) מְשֹׁ֣ךְ חַ֭סְדְּךָ לְיֹדְעֶ֑יךָ וְ֝צִדְקָֽתְךָ֗ לְיִשְׁרֵי־ לֵֽב׃ (יג) אַל־ תְּ֭בוֹאֵנִי רֶ֣גֶל גַּאֲוָ֑ה וְיַד־ רְ֝שָׁעִ֗ים אַל־ תְּנִדֵֽנִי׃ (יד) שָׁ֣ם נָ֭פְלוּ פֹּ֣עֲלֵי אָ֑וֶן דֹּ֝ח֗וּ וְלֹא־ יָ֥כְלוּ קֽוּם׃ (טו) לְדָוִ֨ד ׀ אַל־ תִּתְחַ֥ר בַּמְּרֵעִ֑ים אַל־ תְּ֝קַנֵּ֗א בְּעֹשֵׂ֥י עַוְלָֽה׃ (טז) כִּ֣י כֶ֭חָצִיר מְהֵרָ֣ה יִמָּ֑לוּ וּכְיֶ֥רֶק דֶּ֝֗שֶׁא יִבּוֹלֽוּן׃ (יז) בְּטַ֣ח בַּֽ֭יהוָה וַעֲשֵׂה־ ט֑וֹב שְׁכָן־ אֶ֝֗רֶץ וּרְעֵ֥ה אֱמוּנָֽה׃ (יח) וְהִתְעַנַּ֥ג עַל־ יְהוָ֑ה וְיִֽתֶּן־ לְ֝ךָ֗ מִשְׁאֲלֹ֥ת לִבֶּֽךָ׃ (יט) גּ֣וֹל עַל־ יְהוָ֣ה דַּרְכֶּ֑ךָ וּבְטַ֥ח עָ֝לָ֗יו וְה֣וּא יַעֲשֶֽׂה׃ (כ) וְהוֹצִ֣יא כָא֣וֹר צִדְקֶ֑ךָ וּ֝מִשְׁפָּטֶ֗ךָ כַּֽצָּהֳרָֽיִם׃ (כא) דּ֤וֹם ׀ לַיהוָה֮ וְהִתְח֪וֹלֵ֫ל ל֥וֹ אַל־ תִּ֭תְחַר בְּמַצְלִ֣יחַ דַּרְכּ֑וֹ בְּ֝אִ֗ישׁ עֹשֶׂ֥ה מְזִמּֽוֹת׃ (כב) הֶ֣רֶף מֵ֭אַף וַעֲזֹ֣ב חֵמָ֑ה אַל־ תִּ֝תְחַ֗ר אַךְ־ לְהָרֵֽעַ׃ (כג) כִּֽי־ מְ֭רֵעִים יִכָּרֵת֑וּן וְקֹוֵ֥י יְ֝הוָ֗ה הֵ֣מָּה יִֽירְשׁוּ־ אָֽרֶץ׃ (כד) וְע֣וֹד מְ֭עַט וְאֵ֣ין רָשָׁ֑ע וְהִתְבּוֹנַ֖נְתָּ עַל־ מְקוֹמ֣וֹ וְאֵינֶֽנּוּ׃ (כה) וַעֲנָוִ֥ים יִֽירְשׁוּ־ אָ֑רֶץ וְ֝הִתְעַנְּג֗וּ עַל־ רֹ֥ב שָׁלֽוֹם׃ (כו) זֹמֵ֣ם רָ֭שָׁע לַצַּדִּ֑יק וְחֹרֵ֖ק עָלָ֣יו שִׁנָּֽיו׃ (כז) אֲדֹנָ֥י יִשְׂחַק־ ל֑וֹ כִּֽי־ רָ֝אָ֗ה כִּֽי־ יָבֹ֥א יוֹמֽוֹ׃ (כח) חֶ֤רֶב ׀ פָּֽתְח֣וּ רְשָׁעִים֮ וְדָרְכ֪וּ קַ֫שְׁתָּ֥ם לְ֭הַפִּיל עָנִ֣י וְאֶבְי֑וֹן לִ֝טְב֗וֹחַ יִשְׁרֵי־ דָֽרֶךְ׃ (כט) חַ֭רְבָּם תָּב֣וֹא בְלִבָּ֑ם וְ֝קַשְּׁתוֹתָ֗ם תִּשָּׁבַֽרְנָה׃ (ל) טוֹב־ מְ֭עַט לַצַּדִּ֑יק מֵ֝הֲמ֗וֹן רְשָׁעִ֥ים רַבִּֽים׃ (לא) כִּ֤י זְרוֹע֣וֹת רְ֭שָׁעִים תִּשָּׁבַ֑רְנָה וְסוֹמֵ֖ךְ צַדִּיקִ֣ים יְהוָֽה׃ (לב) יוֹדֵ֣עַ יְ֭הוָה יְמֵ֣י תְמִימִ֑ם וְ֝נַחֲלָתָ֗ם לְעוֹלָ֥ם תִּהְיֶֽה׃ (לג) לֹֽא־ יֵ֭בֹשׁוּ בְּעֵ֣ת רָעָ֑ה וּבִימֵ֖י רְעָב֣וֹן יִשְׂבָּֽעוּ׃ (לד) כִּ֤י רְשָׁעִ֨ים ׀ יֹאבֵ֗דוּ וְאֹיְבֵ֣י יְ֭הוָה כִּיקַ֣ר כָּרִ֑ים כָּל֖וּ בֶעָשָׁ֣ן כָּֽלוּ׃ (לה) לֹוֶ֣ה רָ֭שָׁע וְלֹ֣א יְשַׁלֵּ֑ם וְ֝צַדִּ֗יק חוֹנֵ֥ן וְנוֹתֵֽן׃ (לו) כִּ֣י מְ֭בֹרָכָיו יִ֣ירְשׁוּ אָ֑רֶץ וּ֝מְקֻלָּלָ֗יו יִכָּרֵֽתוּ׃ (לז) מֵ֭יְהוָה מִֽצְעֲדֵי־ גֶ֥בֶר כּוֹנָ֗נוּ וְדַרְכּ֥וֹ יֶחְפָּֽץ׃ (לח) כִּֽי־ יִפֹּ֥ל לֹֽא־ יוּטָ֑ל כִּֽי־ יְ֝הוָ֗ה סוֹמֵ֥ךְ יָדֽוֹ׃ (לט) נַ֤עַר ׀ הָיִ֗יתִי גַּם־ זָ֫קַ֥נְתִּי וְֽלֹא־ רָ֭אִיתִי צַדִּ֣יק נֶעֱזָ֑ב וְ֝זַרְע֗וֹ מְבַקֶּשׁ־ לָֽחֶם׃ (מ) כָּל־ הַ֭יּוֹם חוֹנֵ֣ן וּמַלְוֶ֑ה וְ֝זַרְע֗וֹ לִבְרָכָֽה׃ (מא) ס֣וּר מֵ֭רָע וַעֲשֵׂה־ ט֗וֹב וּשְׁכֹ֥ן לְעוֹלָֽם׃ (מב) כִּ֤י יְהוָ֨ה ׀ אֹ֘הֵ֤ב מִשְׁפָּ֗ט וְלֹא־ יַעֲזֹ֣ב אֶת־ חֲ֭סִידָיו לְעוֹלָ֣ם נִשְׁמָ֑רוּ וְזֶ֖רַע רְשָׁעִ֣ים נִכְרָֽת׃ (מג) צַדִּיקִ֥ים יִֽירְשׁוּ־ אָ֑רֶץ וְיִשְׁכְּנ֖וּ לָעַ֣ד עָלֶֽיהָ׃ (מד) פִּֽי־ צַ֭דִּיק יֶהְגֶּ֣ה חָכְמָ֑ה וּ֝לְשׁוֹנ֗וֹ תְּדַבֵּ֥ר מִשְׁפָּֽט׃ (מה) תּוֹרַ֣ת אֱלֹהָ֣יו בְּלִבּ֑וֹ לֹ֖א תִמְעַ֣ד אֲשֻׁרָיו׃ (מו) צוֹפֶ֣ה רָ֭שָׁע לַצַּדִּ֑יק וּ֝מְבַקֵּ֗שׁ לַהֲמִיתוֹ׃ (מז) יְ֭הוָה לֹא־ יַעַזְבֶ֣נּוּ בְיָד֑וֹ וְלֹ֥א יַ֝רְשִׁיעֶ֗נּוּ בְּהִשָּׁפְטֽוֹ׃ (מח) קַוֵּ֤ה אֶל־ יְהוָ֨ה ׀ וּשְׁמֹ֬ר דַּרְכּ֗וֹ וִֽ֭ירוֹמִמְךָ לָרֶ֣שֶׁת אָ֑רֶץ בְּהִכָּרֵ֖ת רְשָׁעִ֣ים תִּרְאֶֽה׃ (מט) רָ֭אִיתִי רָשָׁ֣ע עָרִ֑יץ וּ֝מִתְעָרֶ֗ה כְּאֶזְרָ֥ח רַעֲנָֽן׃ (נ) וַ֭יַּֽעֲבֹר וְהִנֵּ֣ה אֵינֶ֑נּוּ וָֽ֝אֲבַקְשֵׁ֗הוּ וְלֹ֣א נִמְצָֽא׃ (נא) שְׁמָר־ תָּ֭ם וּרְאֵ֣ה יָשָׁ֑ר כִּֽי־ אַחֲרִ֖ית לְאִ֣ישׁ שָׁלֽוֹם׃ (נב) וּֽ֭פֹשְׁעִים נִשְׁמְד֣וּ יַחְדָּ֑ו אַחֲרִ֖ית רְשָׁעִ֣ים נִכְרָֽתָה׃ (נג) וּתְשׁוּעַ֣ת צַ֭דִּיקִים מֵיְהוָ֑ה מָֽ֝עוּזָּ֗ם בְּעֵ֣ת צָרָֽה׃ (נד) וַֽיַּעְזְרֵ֥ם יְהוָ֗ה וַֽיְפַ֫לְּטֵ֥ם יְפַלְּטֵ֣ם מֵ֭רְשָׁעִים וְיוֹשִׁיעֵ֑ם כִּי־ חָ֥סוּ בֽוֹ׃ (נה) מִזְמ֖וֹר לְדָוִ֣ד לְהַזְכִּֽיר׃ (נו) יְֽהוָ֗ה אַל־ בְּקֶצְפְּךָ֥ תוֹכִיחֵ֑נִי וּֽבַחֲמָתְךָ֥ תְיַסְּרֵֽנִי׃ (נז) כִּֽי־ חִ֭צֶּיךָ נִ֣חֲתוּ בִ֑י וַתִּנְחַ֖ת עָלַ֣י יָדֶֽךָ׃ (נח) אֵין־ מְתֹ֣ם בִּ֭בְשָׂרִי מִפְּנֵ֣י זַעְמֶ֑ךָ אֵין־ שָׁל֥וֹם בַּ֝עֲצָמַ֗י מִפְּנֵ֥י חַטָּאתִֽי׃ (נט) כִּ֣י עֲ֭וֺנֹתַי עָבְר֣וּ רֹאשִׁ֑י כְּמַשָּׂ֥א כָ֝בֵ֗ד יִכְבְּד֥וּ מִמֶּֽנִּי׃ (ס) הִבְאִ֣ישׁוּ נָ֭מַקּוּ חַבּוּרֹתָ֑י מִ֝פְּנֵ֗י אִוַּלְתִּֽי׃ (סא) נַעֲוֵ֣יתִי שַׁחֹ֣תִי עַד־ מְאֹ֑ד כָּל־ הַ֝יּ֗וֹם קֹדֵ֥ר הִלָּֽכְתִּי׃ (סב) כִּֽי־ כְ֭סָלַי מָלְא֣וּ נִקְלֶ֑ה וְאֵ֥ין מְ֝תֹ֗ם בִּבְשָׂרִֽי׃ (סג) נְפוּג֣וֹתִי וְנִדְכֵּ֣יתִי עַד־ מְאֹ֑ד שָׁ֝אַ֗גְתִּי מִֽנַּהֲמַ֥ת לִבִּֽי׃ (סד) אֲ‍ֽדֹנָי נֶגְדְּךָ֥ כָל־ תַּאֲוָתִ֑י וְ֝אַנְחָתִ֗י מִמְּךָ֥ לֹא־ נִסְתָּֽרָה׃ (סה) לִבִּ֣י סְ֭חַרְחַר עֲזָבַ֣נִי כֹחִ֑י וְֽאוֹר־ עֵינַ֥י גַּם־ הֵ֝֗ם אֵ֣ין אִתִּֽי׃ (סו) אֹֽהֲבַ֨י ׀ וְרֵעַ֗י מִנֶּ֣גֶד נִגְעִ֣י יַעֲמֹ֑דוּ וּ֝קְרוֹבַ֗י מֵרָחֹ֥ק עָמָֽדוּ׃ (סז) וַיְנַקְשׁ֤וּ ׀ מְבַקְשֵׁ֬י נַפְשִׁ֗י וְדֹרְשֵׁ֣י רָ֭עָתִי דִּבְּר֣וּ הַוּ֑וֹת וּ֝מִרְמ֗וֹת כָּל־ הַיּ֥וֹם יֶהְגּֽוּ׃ (סח) וַאֲנִ֣י כְ֭חֵרֵשׁ לֹ֣א אֶשְׁמָ֑ע וּ֝כְאִלֵּ֗ם לֹ֣א יִפְתַּח־ פִּֽיו׃ (סט) וָאֱהִ֗י כְּ֭אִישׁ אֲשֶׁ֣ר לֹא־ שֹׁמֵ֑עַ וְאֵ֥ין בְּ֝פִ֗יו תּוֹכָחֽוֹת׃ (ע) כִּֽי־ לְךָ֣ יְהוָ֣ה הוֹחָ֑לְתִּי אַתָּ֥ה תַ֝עֲנֶ֗ה אֲדֹנָ֥י אֱלֹהָֽי׃ (עא) כִּֽי־ אָ֭מַרְתִּי פֶּן־ יִשְׂמְחוּ־ לִ֑י בְּמ֥וֹט רַ֝גְלִ֗י עָלַ֥י הִגְדִּֽילוּ׃ (עב) כִּֽי־ אֲ֭נִי לְצֶ֣לַע נָכ֑וֹן וּמַכְאוֹבִ֖י נֶגְדִּ֣י תָמִֽיד׃ (עג) כִּֽי־ עֲוֺנִ֥י אַגִּ֑יד אֶ֝דְאַ֗ג מֵֽחַטָּאתִֽי׃ (עד) וְֽ֭אֹיְבַי חַיִּ֣ים עָצֵ֑מוּ וְרַבּ֖וּ שֹׂנְאַ֣י שָֽׁקֶר׃ (עה) וּמְשַׁלְּמֵ֣י רָ֭עָה תַּ֣חַת טוֹבָ֑ה יִ֝שְׂטְנ֗וּנִי תַּ֣חַת (רדופי)־ רָֽדְפִי־ טֽוֹב׃ (עו) אַל־ תַּֽעַזְבֵ֥נִי יְהוָ֑ה אֱ֝לֹהַ֗י אַל־ תִּרְחַ֥ק מִמֶּֽנִּי׃ (עז) ח֥וּשָׁה לְעֶזְרָתִ֑י אֲ֝דֹנָ֗י תְּשׁוּעָתִֽי׃ (עח) לַמְנַצֵּ֥חַ (לידיתון) לִֽידוּת֗וּן מִזְמ֥וֹר לְדָוִֽד׃ (עט) אָמַ֗רְתִּי אֶֽשְׁמְרָ֣ה דְרָכַי֮ מֵחֲט֪וֹא בִלְשׁ֫וֹנִ֥י אֶשְׁמְרָ֥ה לְפִ֥י מַחְס֑וֹם בְּעֹ֖ד רָשָׁ֣ע לְנֶגְדִּֽי׃ (פ) נֶאֱלַ֣מְתִּי ד֭וּמִיָּה הֶחֱשֵׁ֣יתִי מִטּ֑וֹב וּכְאֵבִ֥י נֶעְכָּֽר׃ (פא) חַם־ לִבִּ֨י ׀ בְּקִרְבִּ֗י בַּהֲגִיגִ֥י תִבְעַר־ אֵ֑שׁ דִּ֝בַּ֗רְתִּי בִּלְשֽׁוֹנִי׃ (פב) הוֹדִ֘יעֵ֤נִי יְהוָ֨ה ׀ קִצִּ֗י וּמִדַּ֣ת יָמַ֣י מַה־ הִ֑יא אֵ֝דְעָ֗ה מֶה־ חָדֵ֥ל אָֽנִי׃ (פג) הִנֵּ֤ה טְפָח֨וֹת ׀ נָ֘תַ֤תָּה יָמַ֗י וְחֶלְדִּ֣י כְאַ֣יִן נֶגְדֶּ֑ךָ אַ֥ךְ כָּֽל־ הֶ֥בֶל כָּל־ אָ֝דָ֗ם נִצָּ֥ב סֶֽלָה׃ (פד) אַךְ־ בְּצֶ֤לֶם ׀ יִֽתְהַלֶּךְ־ אִ֗ישׁ אַךְ־ הֶ֥בֶל יֶהֱמָי֑וּן יִ֝צְבֹּ֗ר וְֽלֹא־ יֵדַ֥ע מִי־ אֹסְפָֽם׃ (פה) וְעַתָּ֣ה מַה־ קִּוִּ֣יתִי אֲדֹנָ֑י תּ֝וֹחַלְתִּ֗י לְךָ֣ הִֽיא׃ (פו) מִכָּל־ פְּשָׁעַ֥י הַצִּילֵ֑נִי חֶרְפַּ֥ת נָ֝בָ֗ל אַל־ תְּשִׂימֵֽנִי׃ (פז) נֶ֭אֱלַמְתִּי לֹ֣א אֶפְתַּח־ פִּ֑י כִּ֖י אַתָּ֣ה עָשִֽׂיתָ׃ (פח) הָסֵ֣ר מֵעָלַ֣י נִגְעֶ֑ךָ מִתִּגְרַ֥ת יָ֝דְךָ֗ אֲנִ֣י כָלִֽיתִי׃ (פט) בְּֽתוֹכָ֘ח֤וֹת עַל־ עָוֺ֨ן ׀ יִסַּ֬רְתָּ אִ֗ישׁ וַתֶּ֣מֶס כָּעָ֣שׁ חֲמוּד֑וֹ אַ֤ךְ הֶ֖בֶל כָּל־ אָדָ֣ם סֶֽלָה׃ (צ) שִֽׁמְעָ֥ה־ תְפִלָּתִ֨י ׀ יְהוָ֡ה וְשַׁוְעָתִ֨י ׀ הַאֲזִינָה֮ אֶֽל־ דִּמְעָתִ֗י אַֽל־ תֶּ֫חֱרַ֥שׁ כִּ֤י גֵ֣ר אָנֹכִ֣י עִמָּ֑ךְ תּ֝וֹשָׁ֗ב כְּכָל־ אֲבוֹתָֽי׃ (צא) הָשַׁ֣ע מִמֶּ֣נִּי וְאַבְלִ֑יגָה בְּטֶ֖רֶם אֵלֵ֣ךְ וְאֵינֶֽנִּי׃ (צב) לַ֝מְנַצֵּ֗חַ לְדָוִ֥ד מִזְמֽוֹר׃ (צג) קַוֺּ֣ה קִוִּ֣יתִי יְהוָ֑ה וַיֵּ֥ט אֵ֝לַ֗י וַיִּשְׁמַ֥ע שַׁוְעָתִֽי׃ (צד) וַיַּעֲלֵ֤נִי ׀ מִבּ֥וֹר שָׁאוֹן֮ מִטִּ֪יט הַיָּ֫וֵ֥ן וַיָּ֖קֶם עַל־ סֶ֥לַע רַגְלַ֗י כּוֹנֵ֥ן אֲשֻׁרָֽי׃ (צה) וַיִּתֵּ֬ן בְּפִ֨י ׀ שִׁ֥יר חָדָשׁ֮ תְּהִלָּ֪ה לֵֽאלֹ֫הֵ֥ינוּ יִרְא֣וּ רַבִּ֣ים וְיִירָ֑אוּ וְ֝יִבְטְח֗וּ בַּיהוָֽה׃ (צו) אַ֥שְֽׁרֵי הַגֶּ֗בֶר אֲשֶׁר־ שָׂ֣ם יְ֭הֹוָה מִבְטַח֑וֹ וְֽלֹא־ פָנָ֥ה אֶל־ רְ֝הָבִ֗ים וְשָׂטֵ֥י כָזָֽב׃ (צז) רַבּ֤וֹת עָשִׂ֨יתָ ׀ אַתָּ֤ה ׀ יְהוָ֣ה אֱלֹהַי֮ נִֽפְלְאֹתֶ֥יךָ וּמַחְשְׁבֹתֶ֗יךָ אֵ֫לֵ֥ינוּ אֵ֤ין ׀ עֲרֹ֬ךְ אֵלֶ֗יךָ אַגִּ֥ידָה וַאֲדַבֵּ֑רָה עָ֝צְמ֗וּ מִסַּפֵּֽר׃ (צח) זֶ֤בַח וּמִנְחָ֨ה ׀ לֹֽא־ חָפַ֗צְתָּ אָ֭זְנַיִם כָּרִ֣יתָ לִּ֑י עוֹלָ֥ה וַ֝חֲטָאָ֗ה לֹ֣א שָׁאָֽלְתָּ׃ (צט) אָ֣ז אָ֭מַרְתִּי הִנֵּה־ בָ֑אתִי בִּמְגִלַּת־ סֵ֝֗פֶר כָּת֥וּב עָלָֽי׃ (ק) לַֽעֲשֽׂוֹת־ רְצוֹנְךָ֣ אֱלֹהַ֣י חָפָ֑צְתִּי וְ֝ת֥וֹרָתְךָ֗ בְּת֣וֹךְ מֵעָֽי׃ (קא) בִּשַּׂ֤רְתִּי צֶ֨דֶק ׀ בְּקָ֘הָ֤ל רָ֗ב הִנֵּ֣ה שְׂ֭פָתַי לֹ֣א אֶכְלָ֑א יְ֝הוָ֗ה אַתָּ֥ה יָדָֽעְתָּ׃ (קב) צִדְקָתְךָ֬ לֹא־ כִסִּ֨יתִי ׀ בְּת֬וֹךְ לִבִּ֗י אֱמוּנָתְךָ֣ וּתְשׁוּעָתְךָ֣ אָמָ֑רְתִּי לֹא־ כִחַ֥דְתִּי חַסְדְּךָ֥ וַ֝אֲמִתְּךָ֗ לְקָהָ֥ל רָֽב׃ (קג) אַתָּ֤ה יְהוָ֗ה לֹא־ תִכְלָ֣א רַחֲמֶ֣יךָ מִמֶּ֑נִּי חַסְדְּךָ֥ וַ֝אֲמִתְּךָ֗ תָּמִ֥יד יִצְּרֽוּנִי׃ (קד) כִּ֤י אָפְפ֥וּ־ עָלַ֨י ׀ רָע֡וֹת עַד־ אֵ֬ין מִסְפָּ֗ר הִשִּׂיג֣וּנִי עֲ֭וֺנֹתַי וְלֹא־ יָכֹ֣לְתִּי לִרְא֑וֹת עָצְמ֥וּ מִשַּֽׂעֲר֥וֹת רֹ֝אשִׁ֗י וְלִבִּ֥י עֲזָבָֽנִי׃ (קה) רְצֵ֣ה יְ֭הוָה לְהַצִּילֵ֑נִי יְ֝הוָ֗ה לְעֶזְרָ֥תִי חֽוּשָׁה׃ (קו) יֵ֘בֹ֤שׁוּ וְיַחְפְּר֨וּ ׀ יַחַד֮ מְבַקְשֵׁ֥י נַפְשִׁ֗י לִסְפּ֫וֹתָ֥הּ יִסֹּ֣גוּ אָ֭חוֹר וְיִכָּלְמ֑וּ חֲ֝פֵצֵ֗י רָעָתִֽי׃ (קז) יָ֭שֹׁמּוּ עַל־ עֵ֣קֶב בָּשְׁתָּ֑ם הָאֹמְרִ֥ים לִ֝֗י הֶ֘אָ֥ח ׀ הֶאָֽח׃ (קח) יָ֘שִׂ֤ישׂוּ וְיִשְׂמְח֨וּ ׀ בְּךָ֗ כָּֽל־ מְבַ֫קְשֶׁ֥יךָ יֹאמְר֣וּ תָ֭מִיד יִגְדַּ֣ל יְהוָ֑ה אֹֽ֝הֲבֵ֗י תְּשׁוּעָתֶֽךָ׃ (קט) וַאֲנִ֤י ׀ עָנִ֣י וְאֶבְיוֹן֮ אֲדֹנָ֪י יַחֲשָׁ֫ב לִ֥י עֶזְרָתִ֣י וּמְפַלְטִ֣י אַ֑תָּה אֱ֝לֹהַ֗י אַל־ תְּאַחַֽר׃ (קי) לַמְנַצֵּ֗חַ מִזְמ֥וֹר לְדָוִֽד׃ (קיא) אַ֭שְׁרֵי מַשְׂכִּ֣יל אֶל־ דָּ֑ל בְּי֥וֹם רָ֝עָ֗ה יְֽמַלְּטֵ֥הוּ יְהוָֽה׃ (קיב) יְהוָ֤ה ׀ יִשְׁמְרֵ֣הוּ וִֽ֭יחַיֵּהוּ (יאשר) וְאֻשַּׁ֣ר בָּאָ֑רֶץ וְאַֽל־ תִּ֝תְּנֵ֗הוּ בְּנֶ֣פֶשׁ אֹיְבָֽיו׃ (קיג) יְֽהוָ֗ה יִ֭סְעָדֶנּוּ עַל־ עֶ֣רֶשׂ דְּוָ֑י כָּל־ מִ֝שְׁכָּב֗וֹ הָפַ֥כְתָּ בְחָלְיֽוֹ׃ (קיד) אֲ‍ֽנִי־ אָ֭מַרְתִּי יְהוָ֣ה חָנֵּ֑נִי רְפָאָ֥ה נַ֝פְשִׁ֗י כִּי־ חָטָ֥אתִי לָֽךְ׃ (קטו) אוֹיְבַ֗י יֹאמְר֣וּ רַ֣ע לִ֑י מָתַ֥י יָ֝מ֗וּת וְאָבַ֥ד שְׁמֽוֹ׃ (קטז) וְאִם־ בָּ֤א לִרְא֨וֹת ׀ שָׁ֤וְא יְדַבֵּ֗ר לִבּ֗וֹ יִקְבָּץ־ אָ֥וֶן ל֑וֹ יֵצֵ֖א לַח֣וּץ יְדַבֵּֽר׃ (קיז) יַ֗חַד עָלַ֣י יִ֭תְלַחֲשׁוּ כָּל־ שֹׂנְאָ֑י עָלַ֓י ׀ יַחְשְׁב֖וּ רָעָ֣ה לִֽי׃ (קיח) דְּֽבַר־ בְּ֭לִיַּעַל יָצ֣וּק בּ֑וֹ וַאֲשֶׁ֥ר שָׁ֝כַ֗ב לֹא־ יוֹסִ֥יף לָקֽוּם׃ (קיט) גַּם־ אִ֤ישׁ שְׁלוֹמִ֨י ׀ אֲשֶׁר־ בָּטַ֣חְתִּי ב֭וֹ אוֹכֵ֣ל לַחְמִ֑י הִגְדִּ֖יל עָלַ֣י עָקֵֽב׃ (קכ) וְאַתָּ֤ה יְהוָ֗ה חָנֵּ֥נִי וַהֲקִימֵ֑נִי וַֽאֲשַׁלְּמָ֥ה לָהֶֽם׃ (קכא) בְּזֹ֣את יָ֭דַעְתִּי כִּֽי־ חָפַ֣צְתָּ בִּ֑י כִּ֤י לֹֽא־ יָרִ֖יעַ אֹיְבִ֣י עָלָֽי׃ (קכב) וַאֲנִ֗י בְּ֭תֻמִּי תָּמַ֣כְתָּ בִּ֑י וַתַּצִּיבֵ֖נִי
