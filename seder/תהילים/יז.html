תהילים סדר יז

פתיחה באתר · ספר תהילים כולו ·

(א) טֽוֹב־ לִ֥י תֽוֹרַת־ פִּ֑יךָ מֵ֝אַלְפֵ֗י זָהָ֥ב וָכָֽסֶף׃ (ב) יָדֶ֣יךָ עָ֭שׂוּנִי וַֽיְכוֹנְנ֑וּנִי הֲ֝בִינֵ֗נִי וְאֶלְמְדָ֥ה מִצְוֺתֶֽיךָ׃ (ג) יְ֭רֵאֶיךָ יִרְא֣וּנִי וְיִשְׂמָ֑חוּ כִּ֖י לִדְבָרְךָ֣ יִחָֽלְתִּי׃ (ד) יָדַ֣עְתִּי יְ֭הוָה כִּי־ צֶ֣דֶק מִשְׁפָּטֶ֑יךָ וֶ֝אֱמוּנָ֗ה עִנִּיתָֽנִי׃ (ה) יְהִי־ נָ֣א חַסְדְּךָ֣ לְנַחֲמֵ֑נִי כְּאִמְרָתְךָ֥ לְעַבְדֶּֽךָ׃ (ו) יְבֹא֣וּנִי רַחֲמֶ֣יךָ וְאֶֽחְיֶ֑ה כִּי־ תֽ֝וֹרָתְךָ֗ שַֽׁעֲשֻׁעָֽי׃ (ז) יֵבֹ֣שׁוּ זֵ֭דִים כִּי־ שֶׁ֣קֶר עִוְּת֑וּנִי אֲ֝נִ֗י אָשִׂ֥יחַ בְּפִקּוּדֶֽיךָ׃ (ח) יָשׁ֣וּבוּ לִ֣י יְרֵאֶ֑יךָ (וידעו) וְ֝יֹדְעֵ֗י עֵדֹתֶֽיךָ׃ (ט) יְהִֽי־ לִבִּ֣י תָמִ֣ים בְּחֻקֶּ֑יךָ לְ֝מַ֗עַן לֹ֣א אֵבֽוֹשׁ׃ (י) כָּלְתָ֣ה לִתְשׁוּעָתְךָ֣ נַפְשִׁ֑י לִדְבָרְךָ֥ יִחָֽלְתִּי׃ (יא) כָּל֣וּ עֵ֭ינַי לְאִמְרָתֶ֑ךָ לֵ֝אמֹ֗ר מָתַ֥י תְּֽנַחֲמֵֽנִי׃ (יב) כִּֽי־ הָ֭יִיתִי כְּנֹ֣אד בְּקִיט֑וֹר חֻ֝קֶּ֗יךָ לֹ֣א שָׁכָֽחְתִּי׃ (יג) כַּמָּ֥ה יְמֵֽי־ עַבְדֶּ֑ךָ מָתַ֬י תַּעֲשֶׂ֖ה בְרֹדְפַ֣י מִשְׁפָּֽט׃ (יד) כָּֽרוּ־ לִ֣י זֵדִ֣ים שִׁיח֑וֹת אֲ֝שֶׁ֗ר לֹ֣א כְתוֹרָתֶֽךָ׃ (טו) כָּל־ מִצְוֺתֶ֥יךָ אֱמוּנָ֑ה שֶׁ֖קֶר רְדָפ֣וּנִי עָזְרֵֽנִי׃ (טז) כִּ֭מְעַט כִּלּ֣וּנִי בָאָ֑רֶץ וַ֝אֲנִ֗י לֹא־ עָזַ֥בְתִּי פִקֻּודֶֽיךָ׃ (יז) כְּחַסְדְּךָ֥ חַיֵּ֑נִי וְ֝אֶשְׁמְרָ֗ה עֵד֥וּת פִּֽיךָ׃ (יח) לְעוֹלָ֥ם יְהוָ֑ה דְּ֝בָרְךָ֗ נִצָּ֥ב בַּשָּׁמָֽיִם׃ (יט) לְדֹ֣ר וָ֭דֹר אֱמֽוּנָתֶ֑ךָ כּוֹנַ֥נְתָּ אֶ֝֗רֶץ וַֽתַּעֲמֹֽד׃ (כ) לְֽ֭מִשְׁפָּטֶיךָ עָמְד֣וּ הַיּ֑וֹם כִּ֖י הַכֹּ֣ל עֲבָדֶֽיךָ׃ (כא) לוּלֵ֣י ת֭וֹרָתְךָ שַׁעֲשֻׁעָ֑י אָ֝֗ז אָבַ֥דְתִּי בְעָנְיִֽי׃ (כב) לְ֭עוֹלָם לֹא־ אֶשְׁכַּ֣ח פִּקּוּדֶ֑יךָ כִּ֥י בָ֝֗ם חִיִּיתָֽנִי׃ (כג) לְֽךָ־ אֲ֭נִי הוֹשִׁיעֵ֑נִי כִּ֖י פִקּוּדֶ֣יךָ דָרָֽשְׁתִּי׃ (כד) לִ֤י קִוּ֣וּ רְשָׁעִ֣ים לְאַבְּדֵ֑נִי עֵ֝דֹתֶ֗יךָ אֶתְבּוֹנָֽן׃ (כה) לְֽכָל תִּ֭כְלָה רָאִ֣יתִי קֵ֑ץ רְחָבָ֖ה מִצְוָתְךָ֣ מְאֹֽד׃ (כו) מָֽה־ אָהַ֥בְתִּי תוֹרָתֶ֑ךָ כָּל־ הַ֝יּ֗וֹם הִ֣יא שִׂיחָתִֽי׃ (כז) מֵ֭אֹ֣יְבַי תְּחַכְּמֵ֣נִי מִצְוֺתֶ֑ךָ כִּ֖י לְעוֹלָ֣ם הִיא־ לִֽי׃ (כח) מִכָּל־ מְלַמְּדַ֥י הִשְׂכַּ֑לְתִּי כִּ֥י עֵ֝דְוֺתֶ֗יךָ שִׂ֣יחָה לִֽֿי׃ (כט) מִזְּקֵנִ֥ים אֶתְבּוֹנָ֑ן כִּ֖י פִקּוּדֶ֣יךָ נָצָֽרְתִּי׃ (ל) מִכָּל־ אֹ֣רַח רָ֭ע כָּלִ֣אתִי רַגְלָ֑י לְ֝מַ֗עַן אֶשְׁמֹ֥ר דְּבָרֶֽךָ׃ (לא) מִמִּשְׁפָּטֶ֥יךָ לֹא־ סָ֑רְתִּי כִּֽי־ אַ֝תָּ֗ה הוֹרֵתָֽנִי׃ (לב) מַה־ נִּמְלְצ֣וּ לְ֭חִכִּי אִמְרָתֶ֗ךָ מִדְּבַ֥שׁ לְפִֽי׃ (לג) מִפִּקּוּדֶ֥יךָ אֶתְבּוֹנָ֑ן עַל־ כֵּ֝֗ן שָׂנֵ֤אתִי ׀ כָּל־ אֹ֬רַח שָֽׁקֶר׃ (לד) נֵר־ לְרַגְלִ֥י דְבָרֶ֑ךָ וְ֝א֗וֹר לִנְתִיבָתִֽי׃ (לה) נִשְׁבַּ֥עְתִּי וָאֲקַיֵּ֑מָה לִ֝שְׁמֹ֗ר מִשְׁפְּטֵ֥י צִדְקֶֽךָ׃ (לו) נַעֲנֵ֥יתִי עַד־ מְאֹ֑ד יְ֝הוָ֗ה חַיֵּ֥נִי כִדְבָרֶֽךָ׃ (לז) נִדְב֣וֹת פִּ֭י רְצֵה־ נָ֣א יְהוָ֑ה וּֽמִשְׁפָּטֶ֥יךָ לַמְּדֵֽנִי׃ (לח) נַפְשִׁ֣י בְכַפִּ֣י תָמִ֑יד וְ֝תֽוֹרָתְךָ֗ לֹ֣א שָׁכָֽחְתִּי׃ (לט) נָתְנ֬וּ רְשָׁעִ֣ים פַּ֣ח לִ֑י וּ֝מִפִּקּוּדֶ֗יךָ לֹ֣א תָעִֽיתִי׃ (מ) נָחַ֣לְתִּי עֵדְוֺתֶ֣יךָ לְעוֹלָ֑ם כִּֽי־ שְׂשׂ֖וֹן לִבִּ֣י הֵֽמָּה׃ (מא) נָטִ֣יתִי לִ֭בִּי לַעֲשׂ֥וֹת חֻקֶּ֗יךָ לְעוֹלָ֥ם עֵֽקֶב׃ (מב) סֵעֲפִ֥ים שָׂנֵ֑אתִי וְֽתוֹרָתְךָ֥ אָהָֽבְתִּי׃ (מג) סִתְרִ֣י וּמָגִנִּ֣י אָ֑תָּה לִדְבָרְךָ֥ יִחָֽלְתִּי׃ (מד) סֽוּרוּ־ מִמֶּ֥נִּי מְרֵעִ֑ים וְ֝אֶצְּרָ֗ה מִצְוֺ֥ת אֱלֹהָֽי׃ (מה) סָמְכֵ֣נִי כְאִמְרָתְךָ֣ וְאֶֽחְיֶ֑ה וְאַל־ תְּ֝בִישֵׁ֗נִי מִשִּׂבְרִֽי׃ (מו) סְעָדֵ֥נִי וְאִוָּשֵׁ֑עָה וְאֶשְׁעָ֖ה בְחֻקֶּ֣יךָ תָמִֽיד׃ (מז) סָ֭לִיתָ כָּל־ שׁוֹגִ֣ים מֵחֻקֶּ֑יךָ כִּי־ שֶׁ֝֗קֶר תַּרְמִיתָֽם׃ (מח) סִגִ֗ים הִשְׁבַּ֥תָּ כָל־ רִשְׁעֵי־ אָ֑רֶץ לָ֝כֵ֗ן אָהַ֥בְתִּי עֵדֹתֶֽיךָ׃ (מט) סָמַ֣ר מִפַּחְדְּךָ֣ בְשָׂרִ֑י וּֽמִמִּשְׁפָּטֶ֥יךָ יָרֵֽאתִי׃ (נ) עָ֭שִׂיתִי מִשְׁפָּ֣ט וָצֶ֑דֶק בַּל־ תַּ֝נִּיחֵ֗נִי לְעֹֽשְׁקָֽי׃ (נא) עֲרֹ֣ב עַבְדְּךָ֣ לְט֑וֹב אַֽל־ יַעַשְׁקֻ֥נִי זֵדִֽים׃ (נב) עֵ֭ינַי כָּל֣וּ לִֽישׁוּעָתֶ֑ךָ וּלְאִמְרַ֥ת צִדְקֶֽךָ׃ (נג) עֲשֵׂ֖ה עִם־ עַבְדְּךָ֥ כְחַסְדֶּ֗ךָ וְחֻקֶּ֥יךָ לַמְּדֵֽנִי׃ (נד) עַבְדְּךָ־ אָ֥נִי הֲבִינֵ֑נִי וְ֝אֵדְעָ֗ה עֵדֹתֶֽיךָ׃ (נה) עֵ֭ת לַעֲשׂ֣וֹת לַיהוָ֑ה הֵ֝פֵ֗רוּ תּוֹרָתֶֽךָ׃ (נו) עַל־ כֵּ֭ן אָהַ֣בְתִּי מִצְוֺתֶ֑יךָ מִזָּהָ֥ב וּמִפָּֽז׃ (נז) עַל־ כֵּ֤ן ׀ כָּל־ פִּקּ֣וּדֵי כֹ֣ל יִשָּׁ֑רְתִּי כָּל־ אֹ֖רַח שֶׁ֣קֶר שָׂנֵֽאתִי׃ (נח) פְּלָא֥וֹת עֵדְוֺתֶ֑יךָ עַל־ כֵּ֝֗ן נְצָרָ֥תַם נַפְשִֽׁי׃ (נט) פֵּ֖תַח דְּבָרֶ֥יךָ יָאִ֗יר מֵבִ֥ין פְּתָיִֽים׃ (ס) פִּֽי־ פָ֭עַרְתִּי וָאֶשְׁאָ֑פָה כִּ֖י לְמִצְוֺתֶ֣יךָ יָאָֽבְתִּי׃ (סא) פְּנֵה־ אֵלַ֥י וְחָנֵּ֑נִי כְּ֝מִשְׁפָּ֗ט לְאֹהֲבֵ֥י שְׁמֶֽךָ׃ (סב) פְּ֭עָמַי הָכֵ֣ן בְּאִמְרָתֶ֑ךָ וְֽאַל־ תַּשְׁלֶט־ בִּ֥י כָל־ אָֽוֶן׃ (סג) פְּ֭דֵנִי מֵעֹ֣שֶׁק אָדָ֑ם וְ֝אֶשְׁמְרָ֗ה פִּקּוּדֶֽיךָ׃ (סד) פָּ֭נֶיךָ הָאֵ֣ר בְּעַבְדֶּ֑ךָ וְ֝לַמְּדֵ֗נִי אֶת־ חֻקֶּֽיךָ׃ (סה) פַּלְגֵי־ מַ֭יִם יָרְד֣וּ עֵינָ֑י עַ֝֗ל לֹא־ שָׁמְר֥וּ תוֹרָתֶֽךָ׃ (סו) צַדִּ֣יק אַתָּ֣ה יְהוָ֑ה וְ֝יָשָׁ֗ר מִשְׁפָּטֶֽיךָ׃ (סז) צִ֭וִּיתָ צֶ֣דֶק עֵדֹתֶ֑יךָ וֶֽאֱמוּנָ֥ה מְאֹֽד׃ (סח) צִמְּתַ֥תְנִי קִנְאָתִ֑י כִּֽי־ שָׁכְח֖וּ דְבָרֶ֣יךָ צָרָֽי׃ (סט) צְרוּפָ֖ה אִמְרָתְךָ֥ מְאֹ֗ד וְֽעַבְדְּךָ֥ אֲהֵבָֽהּ׃ (ע) צָעִ֣יר אָנֹכִ֣י וְנִבְזֶ֑ה פִּ֝קֻּדֶ֗יךָ לֹ֣א שָׁכָֽחְתִּי׃ (עא) צִדְקָתְךָ֣ צֶ֣דֶק לְעוֹלָ֑ם וְֽתוֹרָתְךָ֥ אֱמֶֽת׃ (עב) צַר־ וּמָצ֥וֹק מְצָא֑וּנִי מִ֝צְוֺתֶ֗יךָ שַׁעֲשֻׁעָֽי׃ (עג) צֶ֖דֶק עֵדְוֺתֶ֥יךָ לְעוֹלָ֗ם הֲבִינֵ֥נִי וְאֶחְיֶֽה׃ (עד) קָרָ֣אתִי בְכָל־ לֵ֭ב עֲנֵ֥נִי יְהוָ֗ה חֻקֶּ֥יךָ אֶצֹּֽרָה׃ (עה) קְרָאתִ֥יךָ הוֹשִׁיעֵ֑נִי וְ֝אֶשְׁמְרָ֗ה עֵדֹתֶֽיךָ׃ (עו) קִדַּ֣מְתִּי בַ֭נֶּשֶׁף וָאֲשַׁוֵּ֑עָה (לדבריך) לִדְבָרְךָ֥ יִחָֽלְתִּי׃ (עז) קִדְּמ֣וּ עֵ֭ינַי אַשְׁמֻר֑וֹת לָ֝שִׂ֗יחַ בְּאִמְרָתֶֽךָ׃ (עח) ק֭וֹלִי שִׁמְעָ֣ה כְחַסְדֶּ֑ךָ יְ֝הוָ֗ה כְּֽמִשְׁפָּטֶ֥ךָ חַיֵּֽנִי׃ (עט) קָ֭רְבוּ רֹדְפֵ֣י זִמָּ֑ה מִתּוֹרָתְךָ֥ רָחָֽקוּ׃ (פ) קָר֣וֹב אַתָּ֣ה יְהוָ֑ה וְֽכָל־ מִצְוֺתֶ֥יךָ אֱמֶֽת׃ (פא) קֶ֣דֶם יָ֭דַעְתִּי מֵעֵדֹתֶ֑יךָ כִּ֖י לְעוֹלָ֣ם יְסַדְתָּֽם׃ (פב) רְאֵֽה־ עָנְיִ֥י וְחַלְּצֵ֑נִי כִּי־ תֽ֝וֹרָתְךָ֗ לֹ֣א שָׁכָֽחְתִּי׃ (פג) רִיבָ֣ה יְ֭הוָה אֶת־ שִׁיבַ֣ת צִיּ֑וֹן הָ֝יִ֗ינוּ כְּחֹלְמִֽים׃ (קמח) אָ֤ז יִמָּלֵ֪א שְׂח֡וֹק פִּינוּ֮ וּלְשׁוֹנֵ֪נוּ רִ֫נָּ֥ה אָ֭ז יֹאמְר֣וּ בַגּוֹיִ֑ם הִגְדִּ֥יל יְ֝הוָ֗ה לַעֲשׂ֥וֹת עִם־ אֵֽלֶּה׃ (קמט) הִגְדִּ֣יל יְ֭הוָה לַעֲשׂ֥וֹת עִמָּ֗נוּ הָיִ֥ינוּ שְׂמֵחִֽים׃ (קנ) שׁוּבָ֣ה יְ֭הוָה אֶת־ (שבותנו) שְׁבִיתֵ֑נוּ כַּאֲפִיקִ֥ים בַּנֶּֽגֶב׃ (קנא) הַזֹּרְעִ֥ים בְּדִמְעָ֗ה בְּרִנָּ֥ה יִקְצֹֽרוּ׃ (קנב) הָ֘ל֤וֹךְ יֵלֵ֨ךְ ׀ וּבָכֹה֮ נֹשֵׂ֪א מֶֽשֶׁךְ־ הַ֫זָּ֥רַע בֹּֽ֬א־ יָב֥וֹא בְרִנָּ֑ה נֹ֝שֵׂ֗א אֲלֻמֹּתָֽיו׃ (קנג) שִׁ֥יר הַֽמַּֽעֲל֗וֹת לִשְׁלֹ֫מֹ֥ה אִם־ יְהוָ֤ה ׀ לֹא־ יִבְנֶ֬ה בַ֗יִת שָׁ֤וְא ׀ עָמְל֣וּ בוֹנָ֣יו בּ֑וֹ אִם־ יְהוָ֥ה לֹֽא־ יִשְׁמָר־ עִ֝֗יר שָׁ֤וְא ׀ שָׁקַ֬ד שׁוֹמֵֽר׃ (קנד) שָׁ֤וְא לָכֶ֨ם ׀ מַשְׁכִּ֪ימֵי ק֡וּם מְאַֽחֲרֵי־ שֶׁ֗בֶת אֹ֭כְלֵי לֶ֣חֶם הָעֲצָבִ֑ים כֵּ֤ן יִתֵּ֖ן לִֽידִיד֣וֹ שֵׁנָֽא׃ (קנה) הִנֵּ֤ה נַחֲלַ֣ת יְהוָ֣ה בָּנִ֑ים שָׂ֝כָ֗ר פְּרִ֣י הַבָּֽטֶן׃ (קנו) כְּחִצִּ֥ים בְּיַד־ גִּבּ֑וֹר כֵּ֝֗ן בְּנֵ֣י הַנְּעוּרִֽים׃ (קנז) אַשְׁרֵ֤י הַגֶּ֗בֶר אֲשֶׁ֤ר מִלֵּ֥א אֶת־ אַשְׁפָּת֗וֹ מֵ֫הֶ֥ם לֹֽא־ יֵבֹ֑שׁוּ כִּֽי־ יְדַבְּר֖וּ אֶת־ אוֹיְבִ֣ים בַּשָּֽׁעַר׃ (קנח) שִׁ֗יר הַֽמַּ֫עֲל֥וֹת אַ֭שְׁרֵי כָּל־ יְרֵ֣א יְהוָ֑ה הַ֝הֹלֵ֗ךְ בִּדְרָכָֽיו׃ (קנט) יְגִ֣יעַ כַּ֭פֶּיךָ כִּ֣י תֹאכֵ֑ל אַ֝שְׁרֶ֗יךָ וְט֣וֹב לָֽךְ׃ (קס) אֶשְׁתְּךָ֤ ׀ כְּגֶ֥פֶן פֹּרִיָּה֮ בְּיַרְכְּתֵ֪י בֵ֫יתֶ֥ךָ בָּ֭נֶיךָ כִּשְׁתִלֵ֣י זֵיתִ֑ים סָ֝בִ֗יב לְשֻׁלְחָנֶֽךָ׃ (קסא) הִנֵּ֣ה כִי־ כֵ֭ן יְבֹ֥רַךְ גָּ֗בֶר יְרֵ֣א יְהוָֽה׃ (קסב) יְבָרֶכְךָ֥ יְהוָ֗ה מִצִּ֫יּ֥וֹן וּ֭רְאֵה בְּט֣וּב יְרוּשָׁלִָ֑ם כֹּ֝֗ל יְמֵ֣י חַיֶּֽיךָ׃
