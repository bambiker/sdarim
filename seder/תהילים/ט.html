תהילים סדר ט

פתיחה באתר · ספר תהילים כולו ·

(א) יְבָרְכֵ֥נוּ אֱלֹהִ֑ים וְיִֽירְא֥וּ אֹ֝ת֗וֹ כָּל־ אַפְסֵי־ אָֽרֶץ׃ (ב) לַמְנַצֵּ֥חַ לְדָוִ֗ד מִזְמ֥וֹר שִֽׁיר׃ (ג) יָק֣וּם אֱ֭לֹהִים יָפ֣וּצוּ אוֹיְבָ֑יו וְיָנ֥וּסוּ מְ֝שַׂנְאָ֗יו מִפָּנָֽיו׃ (ד) כְּהִנְדֹּ֥ף עָשָׁ֗ן תִּ֫נְדֹּ֥ף כְּהִמֵּ֣ס דּ֭וֹנַג מִפְּנֵי־ אֵ֑שׁ יֹאבְד֥וּ רְ֝שָׁעִ֗ים מִפְּנֵ֥י אֱלֹהִֽים׃ (ה) וְֽצַדִּיקִ֗ים יִשְׂמְח֣וּ יַֽ֭עַלְצוּ לִפְנֵ֥י אֱלֹהִ֗ים וְיָשִׂ֥ישׂוּ בְשִׂמְחָֽה׃ (ו) שִׁ֤ירוּ ׀ לֵֽאלֹהִים֮ זַמְּר֪וּ שְׁ֫מ֥וֹ סֹ֡לּוּ לָרֹכֵ֣ב בָּ֭עֲרָבוֹת בְּיָ֥הּ שְׁמ֗וֹ וְעִלְז֥וּ לְפָנָֽיו׃ (ז) אֲבִ֣י יְ֭תוֹמִים וְדַיַּ֣ן אַלְמָנ֑וֹת אֱ֝לֹהִ֗ים בִּמְע֥וֹן קָדְשֽׁוֹ׃ (ח) אֱלֹהִ֤ים ׀ מ֘וֹשִׁ֤יב יְחִידִ֨ים ׀ בַּ֗יְתָה מוֹצִ֣יא אֲ֭סִירִים בַּכּוֹשָׁר֑וֹת אַ֥ךְ ס֝וֹרֲרִ֗ים שָׁכְנ֥וּ צְחִיחָֽה׃ (ט) אֱ‍ֽלֹהִ֗ים בְּ֭צֵאתְךָ לִפְנֵ֣י עַמֶּ֑ךָ בְּצַעְדְּךָ֖ בִֽישִׁימ֣וֹן סֶֽלָה׃ (י) אֶ֤רֶץ רָעָ֨שָׁה ׀ אַף־ שָׁמַ֣יִם נָטְפוּ֮ מִפְּנֵ֪י אֱלֹ֫הִ֥ים זֶ֥ה סִינַ֑י מִפְּנֵ֥י אֱ֝לֹהִ֗ים אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (יא) גֶּ֣שֶׁם נְ֭דָבוֹת תָּנִ֣יף אֱלֹהִ֑ים נַחֲלָתְךָ֥ וְ֝נִלְאָ֗ה אַתָּ֥ה כֽוֹנַנְתָּֽהּ׃ (יב) חַיָּתְךָ֥ יָֽשְׁבוּ־ בָ֑הּ תָּ֤כִ֥ין בְּטוֹבָתְךָ֖ לֶעָנִ֣י אֱלֹהִֽים׃ (יג) אֲדֹנָ֥י יִתֶּן־ אֹ֑מֶר הַֽ֝מְבַשְּׂר֗וֹת צָבָ֥א רָֽב׃ (יד) מַלְכֵ֣י צְ֭בָאוֹת יִדֹּד֣וּן יִדֹּד֑וּן וּנְוַת בַּ֝֗יִת תְּחַלֵּ֥ק שָׁלָֽל׃ (טו) אִֽם־ תִּשְׁכְּבוּן֮ בֵּ֪ין שְׁפַ֫תָּ֥יִם כַּנְפֵ֣י י֭וֹנָה נֶחְפָּ֣ה בַכֶּ֑סֶף וְ֝אֶבְרוֹתֶ֗יהָ בִּֽירַקְרַ֥ק חָרֽוּץ׃ (טז) בְּפָ֘רֵ֤שׂ שַׁדַּ֓י מְלָ֘כִ֤ים בָּ֗הּ תַּשְׁלֵ֥ג בְּצַלְמֽוֹן׃ (יז) הַר־ אֱ֭לֹהִים הַר־ בָּשָׁ֑ן הַ֥ר גַּ֝בְנֻנִּ֗ים הַר־ בָּשָֽׁן׃ (יח) לָ֤מָּה ׀ תְּֽרַצְּדוּן֮ הָרִ֪ים גַּבְנֻ֫נִּ֥ים הָהָ֗ר חָמַ֣ד אֱלֹהִ֣ים לְשִׁבְתּ֑וֹ אַף־ יְ֝הוָ֗ה יִשְׁכֹּ֥ן לָנֶֽצַח׃ (יט) רֶ֤כֶב אֱלֹהִ֗ים רִבֹּתַ֣יִם אַלְפֵ֣י שִׁנְאָ֑ן אֲדֹנָ֥י בָ֝֗ם סִינַ֥י בַּקֹּֽדֶשׁ׃ (כ) עָ֘לִ֤יתָ לַמָּר֨וֹם ׀ שָׁ֘בִ֤יתָ שֶּׁ֗בִי לָקַ֣חְתָּ מַ֭תָּנוֹת בָּאָדָ֑ם וְאַ֥ף ס֝וֹרְרִ֗ים לִשְׁכֹּ֤ן ׀ יָ֬הּ אֱלֹהִֽים׃ (כא) בָּ֤ר֣וּךְ אֲדֹנָי֮ י֤וֹם ׀ י֥וֹם יַֽעֲמָס־ לָ֗נוּ הָ֘אֵ֤ל יְֽשׁוּעָתֵ֬נוּ סֶֽלָה׃ (כב) הָ֤אֵ֣ל ׀ לָנוּ֮ אֵ֤ל לְֽמוֹשָׁ֫ע֥וֹת וְלֵיהוִ֥ה אֲדֹנָ֑י לַ֝מָּ֗וֶת תּוֹצָאֽוֹת׃ (כג) אַךְ־ אֱלֹהִ֗ים יִמְחַץ֮ רֹ֤אשׁ אֹ֫יְבָ֥יו קָדְקֹ֥ד שֵׂעָ֑ר מִ֝תְהַלֵּ֗ךְ בַּאֲשָׁמָֽיו׃ (כד) אָמַ֣ר אֲ֭דֹנָי מִבָּשָׁ֣ן אָשִׁ֑יב אָ֝שִׁ֗יב מִֽמְּצֻל֥וֹת יָֽם׃ (כה) לְמַ֤עַן ׀ תִּֽמְחַ֥ץ רַגְלְךָ֗ בְּ֫דָ֥ם לְשׁ֥וֹן כְּלָבֶ֑יךָ מֵאֹיְבִ֥ים מִנֵּֽהוּ׃ (כו) רָא֣וּ הֲלִיכוֹתֶ֣יךָ אֱלֹהִ֑ים הֲלִ֘יכ֤וֹת אֵלִ֖י מַלְכִּ֣י בַקֹּֽדֶשׁ׃ (כז) קִדְּמ֣וּ שָׁ֭רִים אַחַ֣ר נֹגְנִ֑ים בְּת֥וֹךְ עֲ֝לָמ֗וֹת תּוֹפֵפֽוֹת׃ (כח) בְּֽ֭מַקְהֵלוֹת בָּרְכ֣וּ אֱלֹהִ֑ים יְ֝הוָ֗ה מִמְּק֥וֹר יִשְׂרָאֵֽל׃ (כט) שָׁ֤ם בִּנְיָמִ֨ן ׀ צָעִ֡יר רֹדֵ֗ם שָׂרֵ֣י יְ֭הוּדָה רִגְמָתָ֑ם שָׂרֵ֥י זְ֝בֻל֗וּן שָׂרֵ֥י נַפְתָּלִֽי׃ (ל) צִוָּ֥ה אֱלֹהֶ֗יךָ עֻ֫זֶּ֥ךָ עוּזָּ֥ה אֱלֹהִ֑ים ז֝֗וּ פָּעַ֥לְתָּ לָּֽנוּ׃ (לא) מֵֽ֭הֵיכָלֶךָ עַל־ יְרוּשָׁלִָ֑ם לְךָ֤ יוֹבִ֖ילוּ מְלָכִ֣ים שָֽׁי׃ (לב) גְּעַ֨ר חַיַּ֪ת קָנֶ֡ה עֲדַ֤ת אַבִּירִ֨ים ׀ בְּעֶגְלֵ֬י עַמִּ֗ים מִתְרַפֵּ֥ס בְּרַצֵּי־ כָ֑סֶף בִּזַּ֥ר עַ֝מִּ֗ים קְרָב֥וֹת יֶחְפָּֽצוּ׃ (לג) יֶאֱתָ֣יוּ חַ֭שְׁמַנִּים מִנִּ֣י מִצְרָ֑יִם כּ֥וּשׁ תָּרִ֥יץ יָ֝דָ֗יו לֵאלֹהִֽים׃ (לד) מַמְלְכ֣וֹת הָ֭אָרֶץ שִׁ֣ירוּ לֵאלֹהִ֑ים זַמְּר֖וּ אֲדֹנָ֣י סֶֽלָה׃ (לה) לָ֭רֹכֵב בִּשְׁמֵ֣י שְׁמֵי־ קֶ֑דֶם הֵ֥ן יִתֵּ֥ן בְּ֝קוֹלוֹ ק֣וֹל עֹֽז׃ (לו) תְּנ֥וּ עֹ֗ז לֵֽאלֹ֫הִ֥ים עַֽל־ יִשְׂרָאֵ֥ל גַּאֲוָת֑וֹ וְ֝עֻזּ֗וֹ בַּשְּׁחָקִֽים׃ (לז) נ֤וֹרָ֥א אֱלֹהִ֗ים מִֽמִּקְדָּ֫שֶׁ֥יךָ אֵ֤ל יִשְׂרָאֵ֗ל ה֤וּא נֹתֵ֨ן ׀ עֹ֖ז וְתַעֲצֻמ֥וֹת לָעָ֗ם בָּר֥וּךְ אֱלֹהִֽים׃ (לח) לַמְנַצֵּ֬חַ עַֽל־ שׁוֹשַׁנִּ֬ים לְדָוִֽד׃ (לט) הוֹשִׁיעֵ֥נִי אֱלֹהִ֑ים כִּ֤י בָ֖אוּ מַ֣יִם עַד־ נָֽפֶשׁ׃ (מ) טָבַ֤עְתִּי ׀ בִּיוֵ֣ן מְ֭צוּלָה וְאֵ֣ין מָעֳמָ֑ד בָּ֥אתִי בְמַעֲמַקֵּי־ מַ֝֗יִם וְשִׁבֹּ֥לֶת שְׁטָפָֽתְנִי׃ (מא) יָגַ֣עְתִּי בְקָרְאִי֮ נִחַ֪ר גְּר֫וֹנִ֥י כָּל֥וּ עֵינַ֑י מְ֝יַחֵ֗ל לֵאלֹהָֽי׃ (מב) רַבּ֤וּ ׀ מִשַּׂעֲר֣וֹת רֹאשִׁי֮ שֹׂנְאַ֪י חִ֫נָּ֥ם עָצְמ֣וּ מַ֭צְמִיתַי אֹיְבַ֣י שֶׁ֑קֶר אֲשֶׁ֥ר לֹא־ גָ֝זַ֗לְתִּי אָ֣ז אָשִֽׁיב׃ (מג) אֱ‍ֽלֹהִ֗ים אַתָּ֣ה יָ֭דַעְתָּ לְאִוַּלְתִּ֑י וְ֝אַשְׁמוֹתַ֗י מִמְּךָ֥ לֹא־ נִכְחָֽדוּ׃ (מד) אַל־ יֵ֘בֹ֤שׁוּ בִ֨י ׀ קֹוֶיךָ֮ אֲדֹנָ֥י יְהוִ֗ה צְבָ֫א֥וֹת אַל־ יִכָּ֣לְמוּ בִ֣י מְבַקְשֶׁ֑יךָ אֱ֝לֹהֵ֗י יִשְׂרָאֵֽל׃ (מה) כִּֽי־ עָ֭לֶיךָ נָשָׂ֣אתִי חֶרְפָּ֑ה כִּסְּתָ֖ה כְלִמָּ֣ה פָנָֽי׃ (מו) מ֭וּזָר הָיִ֣יתִי לְאֶחָ֑י וְ֝נָכְרִ֗י לִבְנֵ֥י אִמִּֽי׃ (מז) כִּֽי־ קִנְאַ֣ת בֵּיתְךָ֣ אֲכָלָ֑תְנִי וְחֶרְפּ֥וֹת ח֝וֹרְפֶ֗יךָ נָפְל֥וּ עָלָֽי׃ (מח) וָאֶבְכֶּ֣ה בַצּ֣וֹם נַפְשִׁ֑י וַתְּהִ֖י לַחֲרָפ֣וֹת לִֽי׃ (מט) וָאֶתְּנָ֣ה לְבוּשִׁ֣י שָׂ֑ק וָאֱהִ֖י לָהֶ֣ם לְמָשָֽׁל׃ (נ) יָשִׂ֣יחוּ בִ֭י יֹ֣שְׁבֵי שָׁ֑עַר וּ֝נְגִינ֗וֹת שׁוֹתֵ֥י שֵׁכָֽר׃ (נא) וַאֲנִ֤י תְפִלָּתִֽי־ לְךָ֨ ׀ יְהוָ֡ה עֵ֤ת רָצ֗וֹן אֱלֹהִ֥ים בְּרָב־ חַסְדֶּ֑ךָ עֲ֝נֵ֗נִי בֶּאֱמֶ֥ת יִשְׁעֶֽךָ׃ (נב) הַצִּילֵ֣נִי מִ֭טִּיט וְאַל־ אֶטְבָּ֑עָה אִנָּצְלָ֥ה מִ֝שֹּֽׂנְאַ֗י וּמִמַּֽעֲמַקֵּי־ מָֽיִם׃ (נג) אַל־ תִּשְׁטְפֵ֤נִי ׀ שִׁבֹּ֣לֶת מַ֭יִם וְאַל־ תִּבְלָעֵ֣נִי מְצוּלָ֑ה וְאַל־ תֶּאְטַר־ עָלַ֖י בְּאֵ֣ר פִּֽיהָ׃ (נד) עֲנֵ֣נִי יְ֭הוָה כִּי־ ט֣וֹב חַסְדֶּ֑ךָ כְּרֹ֥ב רַ֝חֲמֶ֗יךָ פְּנֵ֣ה אֵלָֽי׃ (נה) וְאַל־ תַּסְתֵּ֣ר פָּ֭נֶיךָ מֵֽעַבְדֶּ֑ךָ כִּֽי־ צַר־ לִ֝֗י מַהֵ֥ר עֲנֵֽנִי׃ (נו) קָרְבָ֣ה אֶל־ נַפְשִׁ֣י גְאָלָ֑הּ לְמַ֖עַן אֹיְבַ֣י פְּדֵֽנִי׃ (נז) אַתָּ֤ה יָדַ֗עְתָּ חֶרְפָּתִ֣י וּ֭בָשְׁתִּי וּכְלִמָּתִ֑י נֶ֝גְדְּךָ֗ כָּל־ צוֹרְרָֽי׃ (נח) חֶרְפָּ֤ה ׀ שָֽׁבְרָ֥ה לִבִּ֗י וָֽאָ֫נ֥וּשָׁה וָאֲקַוֶּ֣ה לָנ֣וּד וָאַ֑יִן וְ֝לַמְנַחֲמִ֗ים וְלֹ֣א מָצָֽאתִי׃ (נט) וַיִּתְּנ֣וּ בְּבָרוּתִ֣י רֹ֑אשׁ וְ֝לִצְמָאִ֗י יַשְׁק֥וּנִי חֹֽמֶץ׃ (ס) יְהִֽי־ שֻׁלְחָנָ֣ם לִפְנֵיהֶ֣ם לְפָ֑ח וְלִשְׁלוֹמִ֥ים לְמוֹקֵֽשׁ׃ (סא) תֶּחְשַׁ֣כְנָה עֵ֭ינֵיהֶם מֵרְא֑וֹת וּ֝מָתְנֵ֗יהֶם תָּמִ֥יד הַמְעַֽד׃ (סב) שְׁפָךְ־ עֲלֵיהֶ֥ם זַעְמֶ֑ךָ וַחֲר֥וֹן אַ֝פְּךָ֗ יַשִּׂיגֵֽם׃ (סג) תְּהִי־ טִֽירָתָ֥ם נְשַׁמָּ֑ה בְּ֝אָהֳלֵיהֶ֗ם אַל־ יְהִ֥י יֹשֵֽׁב׃ (סד) כִּֽי־ אַתָּ֣ה אֲשֶׁר־ הִכִּ֣יתָ רָדָ֑פוּ וְאֶל־ מַכְא֖וֹב חֲלָלֶ֣יךָ יְסַפֵּֽרוּ׃ (סה) תְּֽנָה־ עָ֭וֺן עַל־ עֲוֺנָ֑ם וְאַל־ יָ֝בֹ֗אוּ בְּצִדְקָתֶֽךָ׃ (סו) יִ֭מָּחֽוּ מִסֵּ֣פֶר חַיִּ֑ים וְעִ֥ם צַ֝דִּיקִ֗ים אַל־ יִכָּתֵֽבוּ׃ (סז) וַ֭אֲנִי עָנִ֣י וְכוֹאֵ֑ב יְשׁוּעָתְךָ֖ אֱלֹהִ֣ים תְּשַׂגְּבֵֽנִי׃ (סח) אֲהַֽלְלָ֣ה שֵׁם־ אֱלֹהִ֣ים בְּשִׁ֑יר וַאֲגַדְּלֶ֥נּוּ בְתוֹדָֽה׃ (סט) וְתִיטַ֣ב לַֽ֭יהוָה מִשּׁ֥וֹר פָּ֗ר מַקְרִ֥ן מַפְרִֽיס׃ (ע) רָא֣וּ עֲנָוִ֣ים יִשְׂמָ֑חוּ דֹּרְשֵׁ֥י אֱ֝לֹהִ֗ים וִיחִ֥י לְבַבְכֶֽם׃ (עא) כִּֽי־ שֹׁמֵ֣עַ אֶל־ אֶבְיוֹנִ֣ים יְהוָ֑ה וְאֶת־ אֲ֝סִירָ֗יו לֹ֣א בָזָֽה׃ (עב) יְֽ֭הַלְלוּהוּ שָׁמַ֣יִם וָאָ֑רֶץ יַ֝מִּ֗ים וְֽכָל־ רֹמֵ֥שׂ בָּֽם׃ (עג) כִּ֤י אֱלֹהִ֨ים ׀ י֘וֹשִׁ֤יעַ צִיּ֗וֹן וְ֭יִבְנֶה עָרֵ֣י יְהוּדָ֑ה וְיָ֥שְׁבוּ שָׁ֝֗ם וִירֵשֽׁוּהָ׃ (עד) וְזֶ֣רַע עֲ֭בָדָיו יִנְחָל֑וּהָ וְאֹהֲבֵ֥י שְׁ֝מ֗וֹ יִשְׁכְּנוּ־ בָֽהּ׃ (עה) לַ֝מְנַצֵּ֗חַ לְדָוִ֥ד לְהַזְכִּֽיר׃ (עו) אֱלֹהִ֥ים לְהַצִּילֵ֑נִי יְ֝הוָ֗ה לְעֶזְרָ֥תִי חֽוּשָֽׁה׃ (עז) יֵבֹ֣שׁוּ וְיַחְפְּרוּ֮ מְבַקְשֵׁ֪י נַ֫פְשִׁ֥י יִסֹּ֣גוּ אָ֭חוֹר וְיִכָּלְמ֑וּ חֲ֝פֵצֵ֗י רָעָתִֽי׃ (עח) יָ֭שׁוּבוּ עַל־ עֵ֣קֶב בָּשְׁתָּ֑ם הָ֝אֹמְרִ֗ים הֶ֘אָ֥ח ׀ הֶאָֽח׃ (עט) יָ֘שִׂ֤ישׂוּ וְיִשְׂמְח֨וּ ׀ בְּךָ֗ כָּֽל־ מְבַ֫קְשֶׁ֥יךָ וְיֹאמְר֣וּ תָ֭מִיד יִגְדַּ֣ל אֱלֹהִ֑ים אֹ֝הֲבֵ֗י יְשׁוּעָתֶֽךָ׃ (פ) וַאֲנִ֤י ׀ עָנִ֣י וְאֶבְיוֹן֮ אֱלֹהִ֪ים חֽוּשָׁ֫ה־ לִּ֥י עֶזְרִ֣י וּמְפַלְטִ֣י אַ֑תָּה יְ֝הוָ֗ה אַל־ תְּאַחַֽר׃ (פא) בְּךָֽ־ יְהוָ֥ה חָסִ֑יתִי אַל־ אֵב֥וֹשָׁה לְעוֹלָֽם׃ (פב) בְּצִדְקָתְךָ֗ תַּצִּילֵ֥נִי וּֽתְפַלְּטֵ֑נִי הַטֵּֽה־ אֵלַ֥י אָ֝זְנְךָ֗ וְהוֹשִׁיעֵֽנִי׃ (פג) הֱיֵ֤ה לִ֨י ׀ לְצ֥וּר מָע֡וֹן לָב֗וֹא תָּמִ֗יד צִוִּ֥יתָ לְהוֹשִׁיעֵ֑נִי כִּֽי־ סַלְעִ֖י וּמְצוּדָתִ֣י אָֽתָּה׃ (פד) אֱ‍ֽלֹהַ֗י פַּ֭לְּטֵנִי מִיַּ֣ד רָשָׁ֑ע מִכַּ֖ף מְעַוֵּ֣ל וְחוֹמֵץ׃ (פה) כִּֽי־ אַתָּ֥ה תִקְוָתִ֑י אֲדֹנָ֥י יְ֝הוִ֗ה מִבְטַחִ֥י מִנְּעוּרָֽי׃ (פו) עָלֶ֤יךָ ׀ נִסְמַ֬כְתִּי מִבֶּ֗טֶן מִמְּעֵ֣י אִ֭מִּי אַתָּ֣ה גוֹזִ֑י בְּךָ֖ תְהִלָּתִ֣י תָמִֽיד׃ (פז) כְּ֭מוֹפֵת הָיִ֣יתִי לְרַבִּ֑ים וְ֝אַתָּ֗ה מַֽחֲסִי־ עֹֽז׃ (פח) יִמָּ֣לֵא פִ֭י תְּהִלָּתֶ֑ךָ כָּל־ הַ֝יּ֗וֹם תִּפְאַרְתֶּֽךָ׃ (פט) אַֽל־ תַּ֭שְׁלִיכֵנִי לְעֵ֣ת זִקְנָ֑ה כִּכְל֥וֹת כֹּ֝חִ֗י אַֽל־ תַּעַזְבֵֽנִי׃ (צ) כִּֽי־ אָמְר֣וּ אוֹיְבַ֣י לִ֑י וְשֹׁמְרֵ֥י נַ֝פְשִׁ֗י נוֹעֲצ֥וּ יַחְדָּֽו׃ (צא) לֵ֭אמֹר אֱלֹהִ֣ים עֲזָב֑וֹ רִֽדְפ֥וּ וְ֝תִפְשׂ֗וּהוּ כִּי־ אֵ֥ין מַצִּֽיל׃ (צב) אֱ֭לֹהִים אַל־ תִּרְחַ֣ק מִמֶּ֑נִּי אֱ֝לֹהַ֗י לְעֶזְרָ֥תִי (חישה) חֽוּשָׁה׃ (צג) יֵבֹ֣שׁוּ יִכְלוּ֮ שֹׂטְנֵ֪י נַ֫פְשִׁ֥י יַֽעֲט֣וּ חֶ֭רְפָּה וּכְלִמָּ֑ה מְ֝בַקְשֵׁ֗י רָעָתִֽי׃ (צד) וַ֭אֲנִי תָּמִ֣יד אֲיַחֵ֑ל וְ֝הוֹסַפְתִּ֗י עַל־ כָּל־ תְּהִלָּתֶֽךָ׃ (צה) פִּ֤י ׀ יְסַפֵּ֬ר צִדְקָתֶ֗ךָ כָּל־ הַיּ֥וֹם תְּשׁוּעָתֶ֑ךָ כִּ֤י לֹ֖א יָדַ֣עְתִּי סְפֹרֽוֹת׃ (צו) אָב֗וֹא בִּ֭גְבֻרוֹת אֲדֹנָ֣י יְהוִ֑ה אַזְכִּ֖יר צִדְקָתְךָ֣ לְבַדֶּֽךָ׃ (צז) אֱ‍ֽלֹהִ֗ים לִמַּדְתַּ֥נִי מִנְּעוּרָ֑י וְעַד־ הֵ֝֗נָּה אַגִּ֥יד נִפְלְאוֹתֶֽיךָ׃ (צח) וְגַ֤ם עַד־ זִקְנָ֨ה ׀ וְשֵׂיבָה֮ אֱלֹהִ֪ים אַֽל־ תַּעַ֫זְבֵ֥נִי עַד־ אַגִּ֣יד זְרוֹעֲךָ֣ לְד֑וֹר לְכָל־ יָ֝ב֗וֹא גְּבוּרָתֶֽךָ׃ (צט) וְצִדְקָתְךָ֥ אֱלֹהִ֗ים עַד־ מָ֫ר֥וֹם אֲשֶׁר־ עָשִׂ֥יתָ גְדֹל֑וֹת אֱ֝לֹהִ֗ים מִ֣י כָמֽוֹךָ׃ (ק) אֲשֶׁ֤ר (הראיתנו) הִרְאִיתַ֨נִי ׀ צָר֥וֹת רַבּ֗וֹת וְרָ֫ע֥וֹת תָּשׁ֥וּב (תחיינו) תְּחַיֵּ֑ינִי וּֽמִתְּהֹמ֥וֹת הָ֝אָ֗רֶץ תָּשׁ֥וּב תַּעֲלֵֽנִי׃ (קא) תֶּ֤רֶב ׀ גְּֽדֻלָּתִ֗י וְתִסֹּ֥ב תְּֽנַחֲמֵֽנִי׃ (קב) גַּם־ אֲנִ֤י ׀ אוֹדְךָ֣ בִכְלִי־ נֶבֶל֮ אֲמִתְּךָ֪ אֱלֹ֫הָ֥י אֲזַמְּרָ֣ה לְךָ֣ בְכִנּ֑וֹר קְ֝ד֗וֹשׁ יִשְׂרָאֵֽל׃ (קג) תְּרַנֵּ֣נָּ֣ה שְׂ֭פָתַי כִּ֣י אֲזַמְּרָה־ לָּ֑ךְ וְ֝נַפְשִׁ֗י אֲשֶׁ֣ר פָּדִֽיתָ׃ (קד) גַּם־ לְשׁוֹנִ֗י כָּל־ הַ֭יּוֹם תֶּהְגֶּ֣ה צִדְקָתֶ֑ךָ כִּי־ בֹ֥שׁוּ כִֽי־ חָ֝פְר֗וּ מְבַקְשֵׁ֥י רָעָתִֽי׃ (קה) לִשְׁלֹמֹ֨ה ׀ אֱ‍ֽלֹהִ֗ים מִ֭שְׁפָּטֶיךָ לְמֶ֣לֶךְ תֵּ֑ן וְצִדְקָתְךָ֥ לְבֶן־ מֶֽלֶךְ׃ (קו) יָדִ֣ין עַמְּךָ֣ בְצֶ֑דֶק וַעֲנִיֶּ֥יךָ בְמִשְׁפָּֽט׃ (קז) יִשְׂא֤וּ הָרִ֓ים שָׁ֘ל֥וֹם לָעָ֑ם וּ֝גְבָע֗וֹת בִּצְדָקָֽה׃ (קח) יִשְׁפֹּ֤ט ׀ עֲ‍ֽנִיֵּי־ עָ֗ם י֭וֹשִׁיעַ לִבְנֵ֣י אֶבְי֑וֹן וִֽידַכֵּ֣א עוֹשֵֽׁק׃ (קט) יִֽירָא֥וּךָ עִם־ שָׁ֑מֶשׁ וְלִפְנֵ֥י יָ֝רֵ֗חַ דּ֣וֹר דּוֹרִֽים׃ (קי) יֵ֭רֵד כְּמָטָ֣ר עַל־ גֵּ֑ז כִּ֝רְבִיבִ֗ים זַרְזִ֥יף אָֽרֶץ׃ (קיא) יִֽפְרַח־ בְּיָמָ֥יו צַדִּ֑יק וְרֹ֥ב שָׁ֝ל֗וֹם עַד־ בְּלִ֥י יָרֵֽחַ׃ (קיב) וְ֭יֵרְדְּ מִיָּ֣ם עַד־ יָ֑ם וּ֝מִנָּהָ֗ר עַד־ אַפְסֵי־ אָֽרֶץ׃ (קיג) לְ֭פָנָיו יִכְרְע֣וּ צִיִּ֑ים וְ֝אֹיְבָ֗יו עָפָ֥ר יְלַחֵֽכוּ׃ (קיד) מַלְכֵ֬י תַרְשִׁ֣ישׁ וְ֭אִיִּים מִנְחָ֣ה יָשִׁ֑יבוּ מַלְכֵ֥י שְׁבָ֥א וּ֝סְבָ֗א אֶשְׁכָּ֥ר יַקְרִֽיבוּ׃ (קטו) וְיִשְׁתַּחֲווּ־ ל֥וֹ כָל־ מְלָכִ֑ים כָּל־ גּוֹיִ֥ם יַֽעַבְדֽוּהוּ׃ (קטז) כִּֽי־ יַ֭צִּיל אֶבְי֣וֹן מְשַׁוֵּ֑עַ וְ֝עָנִ֗י וְֽאֵין־ עֹזֵ֥ר לֽוֹ׃ (קיז) יָ֭חֹס עַל־ דַּ֣ל וְאֶבְי֑וֹן וְנַפְשׁ֖וֹת אֶבְיוֹנִ֣ים יוֹשִֽׁיעַ׃ (קיח) מִתּ֣וֹךְ וּ֭מֵחָמָס יִגְאַ֣ל נַפְשָׁ֑ם וְיֵיקַ֖ר דָּמָ֣ם בְּעֵינָֽיו׃ (קיט) וִיחִ֗י וְיִתֶּן־ לוֹ֮ מִזְּהַ֪ב שְׁ֫בָ֥א וְיִתְפַּלֵּ֣ל בַּעֲד֣וֹ תָמִ֑יד כָּל־ הַ֝יּ֗וֹם יְבָרֲכֶֽנְהֽוּ׃ (קכ) יְהִ֤י פִסַּת־ בַּ֨ר ׀ בָּאָרֶץ֮ בְּרֹ֪אשׁ הָ֫רִ֥ים יִרְעַ֣שׁ כַּלְּבָנ֣וֹן פִּרְי֑וֹ וְיָצִ֥יצוּ מֵ֝עִ֗יר כְּעֵ֣שֶׂב הָאָֽרֶץ׃ (קכא) יְהִ֤י שְׁמ֨וֹ לְֽעוֹלָ֗ם לִפְנֵי־ שֶׁמֶשׁ֮ (ינין) יִנּ֪וֹן שְׁ֫מ֥וֹ וְיִתְבָּ֥רְכוּ ב֑וֹ כָּל־ גּוֹיִ֥ם יְאַשְּׁרֽוּהוּ׃ (קכב) בָּר֤וּךְ ׀ יְהוָ֣ה אֱ֭לֹהִים אֱלֹהֵ֣י יִשְׂרָאֵ֑ל עֹשֵׂ֖ה נִפְלָא֣וֹת לְבַדּֽוֹ׃ (קכג) וּבָר֤וּךְ ׀ שֵׁ֥ם כְּבוֹד֗וֹ לְע֫וֹלָ֥ם וְיִמָּלֵ֣א כְ֭בוֹדוֹ אֶת־ כֹּ֥ל הָאָ֗רֶץ אָ֘מֵ֥ן ׀ וְאָמֵֽן׃
