תהילים סדר יד

פתיחה באתר · ספר תהילים כולו ·

(א) לְדָוִ֗ד מִ֫זְמ֥וֹר חֶֽסֶד־ וּמִשְׁפָּ֥ט אָשִׁ֑ירָה לְךָ֖ יְהוָ֣ה אֲזַמֵּֽרָה׃ (ב) אַשְׂכִּ֤ילָה ׀ בְּדֶ֬רֶךְ תָּמִ֗ים מָ֭תַי תָּב֣וֹא אֵלָ֑י אֶתְהַלֵּ֥ךְ בְּתָם־ לְ֝בָבִ֗י בְּקֶ֣רֶב בֵּיתִֽי׃ (ג) לֹֽא־ אָשִׁ֨ית ׀ לְנֶ֥גֶד עֵינַ֗י דְּֽבַר־ בְּלִ֫יָּ֥עַל עֲשֹֽׂה־ סֵטִ֥ים שָׂנֵ֑אתִי לֹ֖א יִדְבַּ֣ק בִּֽי׃ (ד) לֵבָ֣ב עִ֭קֵּשׁ יָס֣וּר מִמֶּ֑נִּי רָ֝֗ע לֹ֣א אֵדָֽע׃ (ה) (מלושני) מְלָשְׁנִ֬י בַסֵּ֨תֶר ׀ רֵעֵהוּ֮ אוֹת֪וֹ אַ֫צְמִ֥ית גְּֽבַהּ־ עֵ֭ינַיִם וּרְחַ֣ב לֵבָ֑ב אֹ֝ת֗וֹ לֹ֣א אוּכָֽל׃ (ו) עֵינַ֤י ׀ בְּנֶֽאֶמְנֵי־ אֶרֶץ֮ לָשֶׁ֪בֶת עִמָּ֫דִ֥י הֹ֭לֵךְ בְּדֶ֣רֶךְ תָּמִ֑ים ה֝֗וּא יְשָׁרְתֵֽנִי׃ (ז) לֹֽא־ יֵשֵׁ֨ב ׀ בְּקֶ֥רֶב בֵּיתִי֮ עֹשֵׂ֪ה רְמִ֫יָּ֥ה דֹּבֵ֥ר שְׁקָרִ֑ים לֹֽא־ יִ֝כּ֗וֹן לְנֶ֣גֶד עֵינָֽי׃ (ח) לַבְּקָרִ֗ים אַצְמִ֥ית כָּל־ רִשְׁעֵי־ אָ֑רֶץ לְהַכְרִ֥ית מֵֽעִיר־ יְ֝הוָ֗ה כָּל־ פֹּ֥עֲלֵי אָֽוֶן׃ (ט) תְּ֭פִלָּה לְעָנִ֣י כִֽי־ יַעֲטֹ֑ף וְלִפְנֵ֥י יְ֝הוָ֗ה יִשְׁפֹּ֥ךְ שִׂיחֽוֹ׃ (י) יְ֭הוָה שִׁמְעָ֣ה תְפִלָּתִ֑י וְ֝שַׁוְעָתִ֗י אֵלֶ֥יךָ תָבֽוֹא׃ (יא) אַל־ תַּסְתֵּ֬ר פָּנֶ֨יךָ ׀ מִמֶּנִּי֮ בְּי֪וֹם צַ֫ר לִ֥י הַטֵּֽה־ אֵלַ֥י אָזְנֶ֑ךָ בְּי֥וֹם אֶ֝קְרָ֗א מַהֵ֥ר עֲנֵֽנִי׃ (יב) כִּֽי־ כָל֣וּ בְעָשָׁ֣ן יָמָ֑י וְ֝עַצְמוֹתַ֗י כְּמוֹ־ קֵ֥ד נִחָֽרוּ׃ (יג) הוּכָּֽה־ כָ֭עֵשֶׂב וַיִּבַ֣שׁ לִבִּ֑י כִּֽי־ שָׁ֝כַ֗חְתִּי מֵאֲכֹ֥ל לַחְמִֽי׃ (יד) מִקּ֥וֹל אַנְחָתִ֑י דָּבְקָ֥ה עַ֝צְמִ֗י לִבְשָׂרִֽי׃ (טו) דָּ֭מִיתִי לִקְאַ֣ת מִדְבָּ֑ר הָ֝יִ֗יתִי כְּכ֣וֹס חֳרָבֽוֹת׃ (טז) שָׁקַ֥דְתִּי וָאֶֽהְיֶ֑ה כְּ֝צִפּ֗וֹר בּוֹדֵ֥ד עַל־ גָּֽג׃ (יז) כָּל־ הַ֭יּוֹם חֵרְפ֣וּנִי אוֹיְבָ֑י מְ֝הוֹלָלַ֗י בִּ֣י נִשְׁבָּֽעוּ׃ (יח) כִּי־ אֵ֭פֶר כַּלֶּ֣חֶם אָכָ֑לְתִּי וְ֝שִׁקֻּוַ֗י בִּבְכִ֥י מָסָֽכְתִּי׃ (יט) מִפְּנֵֽי־ זַֽעַמְךָ֥ וְקִצְפֶּ֑ךָ כִּ֥י נְ֝שָׂאתַ֗נִי וַתַּשְׁלִיכֵֽנִי׃ (כ) יָ֭מַי כְּצֵ֣ל נָט֑וּי וַ֝אֲנִ֗י כָּעֵ֥שֶׂב אִיבָֽשׁ׃ (כא) וְאַתָּ֣ה יְ֭הוָה לְעוֹלָ֣ם תֵּשֵׁ֑ב וְ֝זִכְרְךָ֗ לְדֹ֣ר וָדֹֽר׃ (כב) אַתָּ֣ה תָ֭קוּם תְּרַחֵ֣ם צִיּ֑וֹן כִּי־ עֵ֥ת לְ֝חֶֽנְנָ֗הּ כִּי־ בָ֥א מוֹעֵֽד׃ (כג) כִּֽי־ רָצ֣וּ עֲ֭בָדֶיךָ אֶת־ אֲבָנֶ֑יהָ וְֽאֶת־ עֲפָרָ֥הּ יְחֹנֵֽנוּ׃ (כד) וְיִֽירְא֣וּ ג֭וֹיִם אֶת־ שֵׁ֣ם יְהוָ֑ה וְֽכָל־ מַלְכֵ֥י הָ֝אָ֗רֶץ אֶת־ כְּבוֹדֶֽךָ׃ (כה) כִּֽי־ בָנָ֣ה יְהוָ֣ה צִיּ֑וֹן נִ֝רְאָ֗ה בִּכְבוֹדֽוֹ׃ (כו) פָּ֭נָה אֶל־ תְּפִלַּ֣ת הָעַרְעָ֑ר וְלֹֽא־ בָ֝זָ֗ה אֶת־ תְּפִלָּתָֽם׃ (כז) תִּכָּ֣תֶב זֹ֭את לְד֣וֹר אַחֲר֑וֹן וְעַ֥ם נִ֝בְרָ֗א יְהַלֶּל־ יָֽהּ׃ (כח) כִּֽי־ הִ֭שְׁקִיף מִמְּר֣וֹם קָדְשׁ֑וֹ יְ֝הוָ֗ה מִשָּׁמַ֤יִם ׀ אֶל־ אֶ֬רֶץ הִבִּֽיט׃ (כט) לִ֭שְׁמֹעַ אֶנְקַ֣ת אָסִ֑יר לְ֝פַתֵּ֗חַ בְּנֵ֣י תְמוּתָֽה׃ (ל) לְסַפֵּ֣ר בְּ֭צִיּוֹן שֵׁ֣ם יְהוָ֑ה וּ֝תְהִלָּת֗וֹ בִּירוּשָׁלִָֽם׃ (לא) בְּהִקָּבֵ֣ץ עַמִּ֣ים יַחְדָּ֑ו וּ֝מַמְלָכ֗וֹת לַעֲבֹ֥ד אֶת־ יְהוָֽה׃ (לב) עִנָּ֖ה בַדֶּ֥רֶךְ (כחו) כֹּחִ֗י קִצַּ֥ר יָמָֽי׃ (לג) אֹמַ֗ר אֵלִ֗י אַֽל־ תַּ֭עֲלֵנִי בַּחֲצִ֣י יָמָ֑י בְּד֖וֹר דּוֹרִ֣ים שְׁנוֹתֶֽיךָ׃ (לד) לְ֭פָנִים הָאָ֣רֶץ יָסַ֑דְתָּ וּֽמַעֲשֵׂ֖ה יָדֶ֣יךָ שָׁמָֽיִם׃ (לה) הֵ֤מָּה ׀ יֹאבֵדוּ֮ וְאַתָּ֪ה תַ֫עֲמֹ֥ד וְ֭כֻלָּם כַּבֶּ֣גֶד יִבְל֑וּ כַּלְּב֖וּשׁ תַּחֲלִיפֵ֣ם וְֽיַחֲלֹֽפוּ׃ (לו) וְאַתָּה־ ה֑וּא וּ֝שְׁנוֹתֶ֗יךָ לֹ֣א יִתָּֽמּוּ׃ (לז) בְּנֵֽי־ עֲבָדֶ֥יךָ יִשְׁכּ֑וֹנוּ וְ֝זַרְעָ֗ם לְפָנֶ֥יךָ יִכּֽוֹן׃ (לח) לְדָוִ֨ד ׀ בָּרֲכִ֣י נַ֭פְשִׁי אֶת־ יְהוָ֑ה וְכָל־ קְ֝רָבַ֗י אֶת־ שֵׁ֥ם קָדְשֽׁוֹ׃ (לט) בָּרֲכִ֣י נַ֭פְשִׁי אֶת־ יְהוָ֑ה וְאַל־ תִּ֝שְׁכְּחִ֗י כָּל־ גְּמוּלָֽיו׃ (מ) הַסֹּלֵ֥חַ לְכָל־ עֲוֺנֵ֑כִי הָ֝רֹפֵ֗א לְכָל־ תַּחֲלֻאָֽיְכִי׃ (מא) הַגּוֹאֵ֣ל מִשַּׁ֣חַת חַיָּ֑יְכִי הַֽ֝מְעַטְּרֵ֗כִי חֶ֣סֶד וְרַחֲמִֽים׃ (מב) הַמַּשְׂבִּ֣יַע בַּטּ֣וֹב עֶדְיֵ֑ךְ תִּתְחַדֵּ֖שׁ כַּנֶּ֣שֶׁר נְעוּרָֽיְכִי׃ (מג) עֹשֵׂ֣ה צְדָק֣וֹת יְהוָ֑ה וּ֝מִשְׁפָּטִ֗ים לְכָל־ עֲשׁוּקִֽים׃ (מד) יוֹדִ֣יעַ דְּרָכָ֣יו לְמֹשֶׁ֑ה לִבְנֵ֥י יִ֝שְׂרָאֵ֗ל עֲלִילֽוֹתָיו׃ (מה) רַח֣וּם וְחַנּ֣וּן יְהוָ֑ה אֶ֖רֶךְ אַפַּ֣יִם וְרַב־ חָֽסֶד׃ (מו) לֹֽא־ לָנֶ֥צַח יָרִ֑יב וְלֹ֖א לְעוֹלָ֣ם יִטּֽוֹר׃ (מז) לֹ֣א כַ֭חֲטָאֵינוּ עָ֣שָׂה לָ֑נוּ וְלֹ֥א כַ֝עֲוֺנֹתֵ֗ינוּ גָּמַ֥ל עָלֵֽינוּ׃ (מח) כִּ֤י כִגְבֹ֣הַּ שָׁ֭מַיִם עַל־ הָאָ֑רֶץ גָּבַ֥ר חַ֝סְדּ֗וֹ עַל־ יְרֵאָֽיו׃ (מט) כִּרְחֹ֣ק מִ֭זְרָח מִֽמַּֽעֲרָ֑ב הִֽרְחִ֥יק מִ֝מֶּ֗נּוּ אֶת־ פְּשָׁעֵֽינוּ׃ (נ) כְּרַחֵ֣ם אָ֭ב עַל־ בָּנִ֑ים רִחַ֥ם יְ֝הוָ֗ה עַל־ יְרֵאָֽיו׃ (נא) כִּי־ ה֖וּא יָדַ֣ע יִצְרֵ֑נוּ זָ֝כ֗וּר כִּי־ עָפָ֥ר אֲנָֽחְנוּ׃ (נב) אֱ֭נוֹשׁ כֶּחָצִ֣יר יָמָ֑יו כְּצִ֥יץ הַ֝שָּׂדֶ֗ה כֵּ֣ן יָצִֽיץ׃ (נג) כִּ֤י ר֣וּחַ עָֽבְרָה־ בּ֣וֹ וְאֵינֶ֑נּוּ וְלֹא־ יַכִּירֶ֖נּוּ ע֣וֹד מְקוֹמֽוֹ׃ (נד) וְחֶ֤סֶד יְהוָ֨ה ׀ מֵעוֹלָ֣ם וְעַד־ ע֭וֹלָם עַל־ יְרֵאָ֑יו וְ֝צִדְקָת֗וֹ לִבְנֵ֥י בָנִֽים׃ (נה) לְשֹׁמְרֵ֥י בְרִית֑וֹ וּלְזֹכְרֵ֥י פִ֝קֻּדָ֗יו לַעֲשׂוֹתָֽם׃ (נו) יְֽהוָ֗ה בַּ֭שָּׁמַיִם הֵכִ֣ין כִּסְא֑וֹ וּ֝מַלְכוּת֗וֹ בַּכֹּ֥ל מָשָֽׁלָה׃ (נז) בָּרֲכ֥וּ יְהוָ֗ה מַלְאָ֫כָ֥יו גִּבֹּ֣רֵי כֹ֭חַ עֹשֵׂ֣י דְבָר֑וֹ לִ֝שְׁמֹ֗עַ בְּק֣וֹל דְּבָרֽוֹ׃ (נח) בָּרֲכ֣וּ יְ֭הוָה כָּל־ צְבָאָ֑יו מְ֝שָׁרְתָ֗יו עֹשֵׂ֥י רְצוֹנֽוֹ׃ (נט) בָּרֲכ֤וּ יְהוָ֨ה ׀ כָּֽל־ מַעֲשָׂ֗יו בְּכָל־ מְקֹמ֥וֹת מֶמְשַׁלְתּ֑וֹ בָּרֲכִ֥י נַ֝פְשִׁ֗י אֶת־ יְהוָֽה׃ (ס) בָּרֲכִ֥י נַפְשִׁ֗י אֶת־ יְה֫וָ֥ה יְהוָ֣ה אֱ֭לֹהַי גָּדַ֣לְתָּ מְּאֹ֑ד ה֭וֹד וְהָדָ֣ר לָבָֽשְׁתָּ׃ (סא) עֹֽטֶה־ א֭וֹר כַּשַּׂלְמָ֑ה נוֹטֶ֥ה שָׁ֝מַ֗יִם כַּיְרִיעָֽה׃ (סב) הַ֥מְקָרֶֽה בַמַּ֗יִם עֲ‍ֽלִיּ֫וֹתָ֥יו הַשָּׂם־ עָבִ֥ים רְכוּב֑וֹ הַֽ֝מְהַלֵּ֗ךְ עַל־ כַּנְפֵי־ רֽוּחַ׃ (סג) עֹשֶׂ֣ה מַלְאָכָ֣יו רוּח֑וֹת מְ֝שָׁרְתָ֗יו אֵ֣שׁ לֹהֵֽט׃ (סד) יָֽסַד־ אֶ֭רֶץ עַל־ מְכוֹנֶ֑יהָ בַּל־ תִּ֝מּ֗וֹט עוֹלָ֥ם וָעֶֽד׃ (סה) תְּ֭הוֹם כַּלְּב֣וּשׁ כִּסִּית֑וֹ עַל־ הָ֝רִ֗ים יַֽעַמְדוּ־ מָֽיִם׃ (סו) מִן־ גַּעֲרָ֣תְךָ֣ יְנוּס֑וּן מִן־ ק֥וֹל רַֽ֝עַמְךָ֗ יֵחָפֵזֽוּן׃ (סז) יַעֲל֣וּ הָ֭רִים יֵרְד֣וּ בְקָע֑וֹת אֶל־ מְ֝ק֗וֹם זֶ֤ה ׀ יָסַ֬דְתָּ לָהֶֽם׃ (סח) גְּֽבוּל־ שַׂ֭מְתָּ בַּל־ יַֽעֲבֹר֑וּן בַּל־ יְ֝שׁוּב֗וּן לְכַסּ֥וֹת הָאָֽרֶץ׃ (סט) הַֽמְשַׁלֵּ֣חַ מַ֭עְיָנִים בַּנְּחָלִ֑ים בֵּ֥ין הָ֝רִ֗ים יְהַלֵּכֽוּן׃ (ע) יַ֭שְׁקוּ כָּל־ חַיְת֣וֹ שָׂדָ֑י יִשְׁבְּר֖וּ פְרָאִ֣ים צְמָאָֽם׃ (עא) עֲ֭לֵיהֶם עוֹף־ הַשָּׁמַ֣יִם יִשְׁכּ֑וֹן מִבֵּ֥ין עֳ֝פָאיִ֗ם יִתְּנוּ־ קֽוֹל׃ (עב) מַשְׁקֶ֣ה הָ֭רִים מֵעֲלִיּוֹתָ֑יו מִפְּרִ֥י מַ֝עֲשֶׂ֗יךָ תִּשְׂבַּ֥ע הָאָֽרֶץ׃ (עג) מַצְמִ֤יחַ חָצִ֨יר ׀ לַבְּהֵמָ֗ה וְ֭עֵשֶׂב לַעֲבֹדַ֣ת הָאָדָ֑ם לְה֥וֹצִיא לֶ֝֗חֶם מִן־ הָאָֽרֶץ׃ (עד) וְיַ֤יִן ׀ יְשַׂמַּ֬ח לְֽבַב־ אֱנ֗וֹשׁ לְהַצְהִ֣יל פָּנִ֣ים מִשָּׁ֑מֶן וְ֝לֶ֗חֶם לְֽבַב־ אֱנ֥וֹשׁ יִסְעָֽד׃ (עה) יִ֭שְׂבְּעוּ עֲצֵ֣י יְהוָ֑ה אַֽרְזֵ֥י לְ֝בָנ֗וֹן אֲשֶׁ֣ר נָטָֽע׃ (עו) אֲשֶׁר־ שָׁ֭ם צִפֳּרִ֣ים יְקַנֵּ֑נוּ חֲ֝סִידָ֗ה בְּרוֹשִׁ֥ים בֵּיתָֽהּ׃ (עז) הָרִ֣ים הַ֭גְּבֹהִים לַיְּעֵלִ֑ים סְ֝לָעִ֗ים מַחְסֶ֥ה לַֽשְׁפַנִּֽים׃ (עח) עָשָׂ֣ה יָ֭רֵחַ לְמוֹעֲדִ֑ים שֶׁ֝֗מֶשׁ יָדַ֥ע מְבוֹאֽוֹ׃ (עט) תָּֽשֶׁת־ חֹ֭שֶׁךְ וִ֣יהִי לָ֑יְלָה בּֽוֹ־ תִ֝רְמֹ֗שׂ כָּל־ חַיְתוֹ־ יָֽעַר׃ (פ) הַ֭כְּפִירִים שֹׁאֲגִ֣ים לַטָּ֑רֶף וּלְבַקֵּ֖שׁ מֵאֵ֣ל אָכְלָֽם׃ (פא) תִּזְרַ֣ח הַ֭שֶּׁמֶשׁ יֵאָסֵפ֑וּן וְאֶל־ מְ֝עוֹנֹתָ֗ם יִרְבָּצֽוּן׃ (פב) יֵצֵ֣א אָדָ֣ם לְפָעֳל֑וֹ וְֽלַעֲבֹ֖דָת֣וֹ עֲדֵי־ עָֽרֶב׃ (פג) מָֽה־ רַבּ֬וּ מַעֲשֶׂ֨יךָ ׀ יְֽהוָ֗ה כֻּ֭לָּם בְּחָכְמָ֣ה עָשִׂ֑יתָ מָלְאָ֥ה הָ֝אָ֗רֶץ קִנְיָנֶֽךָ׃ (פד) זֶ֤ה ׀ הַיָּ֥ם גָּדוֹל֮ וּרְחַ֪ב יָ֫דָ֥יִם שָֽׁם־ רֶ֭מֶשׂ וְאֵ֣ין מִסְפָּ֑ר חַיּ֥וֹת קְ֝טַנּ֗וֹת עִם־ גְּדֹלֽוֹת׃ (פה) שָׁ֭ם אֳנִיּ֣וֹת יְהַלֵּכ֑וּן לִ֝וְיָתָ֗ן זֶֽה־ יָצַ֥רְתָּ לְשַֽׂחֶק־ בּֽוֹ׃ (פו) כֻּ֭לָּם אֵלֶ֣יךָ יְשַׂבֵּר֑וּן לָתֵ֖ת אָכְלָ֣ם בְּעִתּֽוֹ׃ (פז) תִּתֵּ֣ן לָ֭הֶם יִלְקֹט֑וּן תִּפְתַּ֥ח יָֽ֝דְךָ֗ יִשְׂבְּע֥וּן טֽוֹב׃ (פח) תַּסְתִּ֥יר פָּנֶיךָ֮ יִֽבָּהֵ֫ל֥וּן תֹּסֵ֣ף ר֭וּחָם יִגְוָע֑וּן וְֽאֶל־ עֲפָרָ֥ם יְשׁוּבֽוּן׃ (פט) תְּשַׁלַּ֣ח ר֭וּחֲךָ יִבָּרֵא֑וּן וּ֝תְחַדֵּ֗שׁ פְּנֵ֣י אֲדָמָֽה׃ (צ) יְהִ֤י כְב֣וֹד יְהוָ֣ה לְעוֹלָ֑ם יִשְׂמַ֖ח יְהוָ֣ה בְּמַעֲשָֽׂיו׃ (צא) הַמַּבִּ֣יט לָ֭אָרֶץ וַתִּרְעָ֑ד יִגַּ֖ע בֶּהָרִ֣ים וְֽיֶעֱשָֽׁנוּ׃ (צב) אָשִׁ֣ירָה לַיהוָ֣ה בְּחַיָּ֑י אֲזַמְּרָ֖ה לֵאלֹהַ֣י בְּעוֹדִֽי׃ (צג) יֶעֱרַ֣ב עָלָ֣יו שִׂיחִ֑י אָ֝נֹכִ֗י אֶשְׂמַ֥ח בַּיהוָֽה׃ (צד) יִתַּ֤מּוּ חַטָּאִ֨ים ׀ מִן־ הָאָ֡רֶץ וּרְשָׁעִ֤ים ׀ ע֤וֹד אֵינָ֗ם בָּרֲכִ֣י נַ֭פְשִׁי אֶת־ יְהוָ֗ה הַֽלְלוּ־ יָֽהּ׃ (צה) הוֹד֣וּ לַ֭יהוָה קִרְא֣וּ בִּשְׁמ֑וֹ הוֹדִ֥יעוּ בָ֝עַמִּ֗ים עֲלִילוֹתָֽיו׃ (צו) שִֽׁירוּ־ ל֭וֹ זַמְּרוּ־ ל֑וֹ שִׂ֝֗יחוּ בְּכָל־ נִפְלְאוֹתָֽיו׃ (צז) הִֽ֭תְהַלְלוּ בְּשֵׁ֣ם קָדְשׁ֑וֹ יִ֝שְׂמַ֗ח לֵ֤ב ׀ מְבַקְשֵׁ֬י יְהוָֽה׃ (צח) דִּרְשׁ֣וּ יְהוָ֣ה וְעֻזּ֑וֹ בַּקְּשׁ֖וּ פָנָ֣יו תָּמִֽיד׃ (צט) זִכְר֗וּ נִפְלְאוֹתָ֥יו אֲשֶׁר־ עָשָׂ֑ה מֹ֝פְתָ֗יו וּמִשְׁפְּטֵי־ פִֽיו׃ (ק) זֶ֭רַע אַבְרָהָ֣ם עַבְדּ֑וֹ בְּנֵ֖י יַעֲקֹ֣ב בְּחִירָֽיו׃ (קא) ה֭וּא יְהוָ֣ה אֱלֹהֵ֑ינוּ בְּכָל־ הָ֝אָ֗רֶץ מִשְׁפָּטָֽיו׃ (קב) זָכַ֣ר לְעוֹלָ֣ם בְּרִית֑וֹ דָּבָ֥ר צִ֝וָּ֗ה לְאֶ֣לֶף דּֽוֹר׃ (קג) אֲשֶׁ֣ר כָּ֭רַת אֶת־ אַבְרָהָ֑ם וּשְׁב֖וּעָת֣וֹ לְיִשְׂחָֽק׃ (קד) וַיַּֽעֲמִידֶ֣הָ לְיַעֲקֹ֣ב לְחֹ֑ק לְ֝יִשְׂרָאֵ֗ל בְּרִ֣ית עוֹלָֽם׃ (קה) לֵאמֹ֗ר לְךָ֗ אֶתֵּ֥ן אֶת־ אֶֽרֶץ־ כְּנָ֑עַן חֶ֝֗בֶל נַחֲלַתְכֶֽם׃ (קו) בִּֽ֭הְיוֹתָם מְתֵ֣י מִסְפָּ֑ר כִּ֝מְעַ֗ט וְגָרִ֥ים בָּֽהּ׃ (קז) וַֽ֭יִּתְהַלְּכוּ מִגּ֣וֹי אֶל־ גּ֑וֹי מִ֝מַּמְלָכָ֗ה אֶל־ עַ֥ם אַחֵֽר׃ (קח) לֹֽא־ הִנִּ֣יחַ אָדָ֣ם לְעָשְׁקָ֑ם וַיּ֖וֹכַח עֲלֵיהֶ֣ם מְלָכִֽים׃ (קט) אַֽל־ תִּגְּע֥וּ בִמְשִׁיחָ֑י וְ֝לִנְבִיאַי אַל־ תָּרֵֽעוּ׃ (קי) וַיִּקְרָ֣א רָ֭עָב עַל־ הָאָ֑רֶץ כָּֽל־ מַטֵּה־ לֶ֥חֶם שָׁבָֽר׃ (קיא) שָׁלַ֣ח לִפְנֵיהֶ֣ם אִ֑ישׁ לְ֝עֶ֗בֶד נִמְכַּ֥ר יוֹסֵֽף׃ (קיב) עִנּ֣וּ בַכֶּ֣בֶל (רגליו) רַגְל֑וֹ בַּ֝רְזֶ֗ל בָּ֣אָה נַפְשֽׁוֹ׃ (קיג) עַד־ עֵ֥ת בֹּֽא־ דְבָר֑וֹ אִמְרַ֖ת יְהוָ֣ה צְרָפָֽתְהוּ׃ (קיד) שָׁ֣לַח מֶ֭לֶךְ וַיַּתִּירֵ֑הוּ מֹשֵׁ֥ל עַ֝מִּ֗ים וַֽיְפַתְּחֵֽהוּ׃ (קטו) שָׂמ֣וֹ אָד֣וֹן לְבֵית֑וֹ וּ֝מֹשֵׁ֗ל בְּכָל־ קִנְיָנֽוֹ׃ (קטז) לֶאְסֹ֣ר שָׂרָ֣יו בְּנַפְשׁ֑וֹ וּזְקֵנָ֥יו יְחַכֵּֽם׃ (קיז) וַיָּבֹ֣א יִשְׂרָאֵ֣ל מִצְרָ֑יִם וְ֝יַעֲקֹ֗ב גָּ֣ר בְּאֶֽרֶץ־ חָֽם׃ (קיח) וַיֶּ֣פֶר אֶת־ עַמּ֣וֹ מְאֹ֑ד וַ֝יַּֽעֲצִמֵהוּ מִצָּרָֽיו׃ (קיט) הָפַ֣ךְ לִ֭בָּם לִשְׂנֹ֣א עַמּ֑וֹ לְ֝הִתְנַכֵּ֗ל בַּעֲבָדָֽיו׃ (קכ) שָׁ֭לַח מֹשֶׁ֣ה עַבְדּ֑וֹ אַ֝הֲרֹ֗ן אֲשֶׁ֣ר בָּֽחַר־ בּֽוֹ׃ (קכא) שָֽׂמוּ־ בָ֭ם דִּבְרֵ֣י אֹתוֹתָ֑יו וּ֝מֹפְתִ֗ים בְּאֶ֣רֶץ חָֽם׃ (קכב) שָׁ֣לַֽח חֹ֭שֶׁךְ וַיַּחְשִׁ֑ךְ וְלֹֽא־ מָ֝ר֗וּ אֶת־ (דברוו) דְּבָרֽוֹ׃ (קכג) הָפַ֣ךְ אֶת־ מֵימֵיהֶ֣ם לְדָ֑ם וַ֝יָּ֗מֶת אֶת־ דְּגָתָֽם׃ (קכד) שָׁרַ֣ץ אַרְצָ֣ם צְפַרְדְּעִ֑ים בְּ֝חַדְרֵ֗י מַלְכֵיהֶֽם׃ (קכה) אָ֭מַר וַיָּבֹ֣א עָרֹ֑ב כִּ֝נִּ֗ים בְּכָל־ גְּבוּלָֽם׃ (קכו) נָתַ֣ן גִּשְׁמֵיהֶ֣ם בָּרָ֑ד אֵ֖שׁ לֶהָב֣וֹת בְּאַרְצָֽם׃ (קכז) וַיַּ֣ךְ גַּ֭פְנָם וּתְאֵנָתָ֑ם וַ֝יְשַׁבֵּ֗ר עֵ֣ץ גְּבוּלָֽם׃ (קכח) אָ֭מַר וַיָּבֹ֣א אַרְבֶּ֑ה וְ֝יֶ֗לֶק וְאֵ֣ין מִסְפָּֽר׃ (קכט) וַיֹּ֣אכַל כָּל־ עֵ֣שֶׂב בְּאַרְצָ֑ם וַ֝יֹּ֗אכַל פְּרִ֣י אַדְמָתָֽם׃ (קל) וַיַּ֣ךְ כָּל־ בְּכ֣וֹר בְּאַרְצָ֑ם רֵ֝אשִׁ֗ית לְכָל־ אוֹנָֽם׃ (קלא) וַֽ֭יּוֹצִיאֵם בְּכֶ֣סֶף וְזָהָ֑ב וְאֵ֖ין בִּשְׁבָטָ֣יו כּוֹשֵֽׁל׃ (קלב) שָׂמַ֣ח מִצְרַ֣יִם בְּצֵאתָ֑ם כִּֽי־ נָפַ֖ל פַּחְדָּ֣ם עֲלֵיהֶֽם׃ (קלג) פָּרַ֣שׂ עָנָ֣ן לְמָסָ֑ךְ וְ֝אֵ֗שׁ לְהָאִ֥יר לָֽיְלָה׃ (קלד) שָׁאַ֣ל וַיָּבֵ֣א שְׂלָ֑ו וְלֶ֥חֶם שָׁ֝מַ֗יִם יַשְׂבִּיעֵֽם׃ (קלה) פָּ֣תַח צ֭וּר וַיָּז֣וּבוּ מָ֑יִם הָ֝לְכ֗וּ בַּצִּיּ֥וֹת נָהָֽר׃ (קלו) כִּֽי־ זָ֭כַר אֶת־ דְּבַ֣ר קָדְשׁ֑וֹ אֶֽת־ אַבְרָהָ֥ם עַבְדּֽוֹ׃ (קלז) וַיּוֹצִ֣א עַמּ֣וֹ בְשָׂשׂ֑וֹן בְּ֝רִנָּ֗ה אֶת־ בְּחִירָֽיו׃ (קלח) וַיִּתֵּ֣ן לָ֭הֶם אַרְצ֣וֹת גּוֹיִ֑ם וַעֲמַ֖ל לְאֻמִּ֣ים יִירָֽשׁוּ׃
