דברי הימים סדר יח

פתיחה באתר · ספר דברי הימים כולו ·

(א) וַיֹּ֕אמֶר לָכֵ֖ן שִׁמְע֣וּ דְבַר־ יְהוָ֑ה רָאִ֤יתִי אֶת־ יְהוָה֙ יוֹשֵׁ֣ב עַל־ כִּסְא֔וֹ וְכָל־ צְבָ֤א הַשָּׁמַ֙יִם֙ עֹֽמְדִ֔ים עַל־ יְמִינ֖וֹ וּשְׂמֹאלֽוֹ׃ (ב) וַיֹּ֣אמֶר יְהוָ֗ה מִ֤י יְפַתֶּה֙ אֶת־ אַחְאָ֣ב מֶֽלֶךְ־ יִשְׂרָאֵ֔ל וְיַ֕עַל וְיִפֹּ֖ל בְּרָמ֣וֹת גִּלְעָ֑ד וַיֹּ֕אמֶר זֶ֚ה אֹמֵ֣ר כָּ֔כָה וְזֶ֖ה אֹמֵ֥ר כָּֽכָה׃ (ג) וַיֵּצֵ֣א הָר֗וּחַ וַֽיַּעֲמֹד֙ לִפְנֵ֣י יְהוָ֔ה וַיֹּ֖אמֶר אֲנִ֣י אֲפַתֶּ֑נּוּ וַיֹּ֧אמֶר יְהוָ֛ה אֵלָ֖יו בַּמָּֽה׃ (ד) וַיֹּ֗אמֶר אֵצֵא֙ וְהָיִ֙יתִי֙ לְר֣וּחַ שֶׁ֔קֶר בְּפִ֖י כָּל־ נְבִיאָ֑יו וַיֹּ֗אמֶר תְּפַתֶּה֙ וְגַם־ תּוּכָ֔ל צֵ֖א וַעֲשֵׂה־ כֵֽן׃ (ה) וְעַתָּ֗ה הִנֵּ֨ה נָתַ֤ן יְהוָה֙ ר֣וּחַ שֶׁ֔קֶר בְּפִ֖י נְבִיאֶ֣יךָ אֵ֑לֶּה וַֽיהוָ֔ה דִּבֶּ֥ר עָלֶ֖יךָ רָעָֽה׃ (ו) וַיִּגַּשׁ֙ צִדְקִיָּ֣הוּ בֶֽן־ כְּנַעֲנָ֔ה וַיַּ֥ךְ אֶת־ מִיכָ֖יְהוּ עַל־ הַלֶּ֑חִי וַיֹּ֗אמֶר אֵ֣י זֶ֤ה הַדֶּ֙רֶךְ֙ עָבַ֧ר רֽוּחַ־ יְהוָ֛ה מֵאִתִּ֖י לְדַבֵּ֥ר אֹתָֽךְ׃ (ז) וַיֹּ֣אמֶר מִיכָ֔יְהוּ הִנְּךָ֥ רֹאֶ֖ה בַּיּ֣וֹם הַה֑וּא אֲשֶׁ֥ר תָּב֛וֹא חֶ֥דֶר בְּחֶ֖דֶר לְהֵחָבֵֽא׃ (ח) וַיֹּ֙אמֶר֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל קְחוּ֙ אֶת־ מִיכָ֔יְהוּ וַהֲשִׁיבֻ֖הוּ אֶל־ אָמ֣וֹן שַׂר־ הָעִ֑יר וְאֶל־ יוֹאָ֖שׁ בֶּן־ הַמֶּֽלֶךְ׃ (ט) וַאֲמַרְתֶּ֗ם כֹּ֚ה אָמַ֣ר הַמֶּ֔לֶךְ שִׂ֥ימוּ זֶ֖ה בֵּ֣ית הַכֶּ֑לֶא וְהַאֲכִלֻ֜הוּ לֶ֤חֶם לַ֙חַץ֙ וּמַ֣יִם לַ֔חַץ עַ֖ד שׁוּבִ֥י בְשָׁלֽוֹם׃ (י) וַיֹּ֣אמֶר מִיכָ֔יְהוּ אִם־ שׁ֤וֹב תָּשׁוּב֙ בְּשָׁל֔וֹם לֹא־ דִבֶּ֥ר יְהוָ֖ה בִּ֑י וַיֹּ֕אמֶר שִׁמְע֖וּ עַמִּ֥ים כֻּלָּֽם׃ (יא) וַיַּ֧עַל מֶֽלֶךְ־ יִשְׂרָאֵ֛ל וִֽיהוֹשָׁפָ֥ט מֶֽלֶךְ־ יְהוּדָ֖ה אֶל־ רָמֹ֥ת גִּלְעָֽד׃ (יב) וַיֹּאמֶר֩ מֶ֨לֶךְ יִשְׂרָאֵ֜ל אֶל־ יְהוֹשָׁפָ֗ט הִתְחַפֵּשׂ֙ וָב֣וֹא בַמִּלְחָמָ֔ה וְאַתָּ֖ה לְבַ֣שׁ בְּגָדֶ֑יךָ וַיִּתְחַפֵּשׂ֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וַיָּבֹ֖אוּ בַּמִּלְחָמָֽה׃ (יג) וּמֶ֣לֶךְ אֲרָ֡ם צִוָּה֩ אֶת־ שָׂרֵ֨י הָרֶ֤כֶב אֲשֶׁר־ לוֹ֙ לֵאמֹ֔ר לֹ֚א תִּלָּ֣חֲמ֔וּ אֶת־ הַקָּטֹ֖ן אֶת־ הַגָּד֑וֹל כִּ֛י אִֽם־ אֶת־ מֶ֥לֶךְ יִשְׂרָאֵ֖ל לְבַדּֽוֹ׃ (יד) וַיְהִ֡י כִּרְאוֹת֩ שָׂרֵ֨י הָרֶ֜כֶב אֶת־ יְהוֹשָׁפָ֗ט וְהֵ֤מָּה אָֽמְרוּ֙ מֶ֣לֶךְ יִשְׂרָאֵ֣ל ה֔וּא וַיָּסֹ֥בּוּ עָלָ֖יו לְהִלָּחֵ֑ם וַיִּזְעַ֤ק יְהֽוֹשָׁפָט֙ וַֽיהוָ֣ה עֲזָר֔וֹ וַיְסִיתֵ֥ם אֱלֹהִ֖ים מִמֶּֽנּוּ׃ (טו) וַיְהִ֗י כִּרְאוֹת֙ שָׂרֵ֣י הָרֶ֔כֶב כִּ֥י לֹא־ הָיָ֖ה מֶ֣לֶךְ יִשְׂרָאֵ֑ל וַיָּשֻׁ֖בוּ מֵאַחֲרָֽיו׃ (טז) וְאִ֗ישׁ מָשַׁ֤ךְ בַּקֶּ֙שֶׁת֙ לְתֻמּ֔וֹ וַיַּךְ֙ אֶת־ מֶ֣לֶךְ יִשְׂרָאֵ֔ל בֵּ֥ין הַדְּבָקִ֖ים וּבֵ֣ין הַשִּׁרְיָ֑ן וַיֹּ֣אמֶר לָֽרַכָּ֗ב הֲפֹ֧ךְ (ידיך) יָדְךָ֛ וְהוֹצֵאתַ֥נִי מִן־ הַֽמַּחֲנֶ֖ה כִּ֥י הָחֳלֵֽיתִי׃ (יז) וַתַּ֤עַל הַמִּלְחָמָה֙ בַּיּ֣וֹם הַה֔וּא וּמֶ֣לֶךְ יִשְׂרָאֵ֗ל הָיָ֨ה מַעֲמִ֧יד בַּמֶּרְכָּבָ֛ה נֹ֥כַח אֲרָ֖ם עַד־ הָעָ֑רֶב וַיָּ֕מָת לְעֵ֖ת בּ֥וֹא הַשָּֽׁמֶשׁ׃ (יח) וַ֠יָּשָׁב יְהוֹשָׁפָ֨ט מֶֽלֶךְ־ יְהוּדָ֧ה אֶל־ בֵּית֛וֹ בְּשָׁל֖וֹם לִֽירוּשָׁלִָֽם׃ (יט) וַיֵּצֵ֣א אֶל־ פָּנָ֗יו יֵה֣וּא בֶן־ חֲנָ֘נִי֮ הַחֹזֶה֒ וַיֹּ֙אמֶר֙ אֶל־ הַמֶּ֣לֶךְ יְהוֹשָׁפָ֔ט הֲלָרָשָׁ֣ע לַעְזֹ֔ר וּלְשֹׂנְאֵ֥י יְהוָ֖ה תֶּאֱהָ֑ב וּבָזֹאת֙ עָלֶ֣יךָ קֶּ֔צֶף מִלִּפְנֵ֖י יְהוָֽה׃ (כ) אֲבָ֕ל דְּבָרִ֥ים טוֹבִ֖ים נִמְצְא֣וּ עִמָּ֑ךְ כִּֽי־ בִעַ֤רְתָּ הָאֲשֵׁרוֹת֙ מִן־ הָאָ֔רֶץ וַהֲכִינ֥וֹתָ לְבָבְךָ֖ לִדְרֹ֥שׁ הָֽאֱלֹהִֽים׃ (כא) וַיֵּ֥שֶׁב יְהוֹשָׁפָ֖ט בִּירוּשָׁלִָ֑ם וַיָּ֜שָׁב וַיֵּצֵ֣א בָעָ֗ם מִבְּאֵ֥ר שֶׁ֙בַע֙ עַד־ הַ֣ר אֶפְרַ֔יִם וַיְשִׁיבֵ֕ם אֶל־ יְהוָ֖ה אֱלֹהֵ֥י אֲבוֹתֵיהֶֽם׃ (כב) וַיַּעֲמֵ֨ד שֹֽׁפְטִ֜ים בָּאָ֗רֶץ בְּכָל־ עָרֵ֧י יְהוּדָ֛ה הַבְּצֻר֖וֹת לְעִ֥יר וָעִֽיר׃ (כג) וַיֹּ֣אמֶר אֶל־ הַשֹּֽׁפְטִ֗ים רְאוּ֙ מָֽה־ אַתֶּ֣ם עֹשִׂ֔ים כִּ֣י לֹ֧א לְאָדָ֛ם תִּשְׁפְּט֖וּ כִּ֣י לַיהוָ֑ה וְעִמָּכֶ֖ם בִּדְבַ֥ר מִשְׁפָּֽט׃ (כד) וְעַתָּ֕ה יְהִ֥י פַֽחַד־ יְהוָ֖ה עֲלֵיכֶ֑ם שִׁמְר֣וּ וַעֲשׂ֔וּ כִּֽי־ אֵ֞ין עִם־ יְהוָ֣ה אֱלֹהֵ֗ינוּ עַוְלָ֛ה וּמַשֹּׂ֥א פָנִ֖ים וּמִקַּח־ שֹֽׁחַד׃ (כה) וְגַ֣ם בִּ֠ירוּשָׁלִַם הֶעֱמִ֨יד יְהוֹשָׁפָ֜ט מִן־ הַלְוִיִּ֣ם וְהַכֹּהֲנִ֗ים וּמֵרָאשֵׁ֤י הָאָבוֹת֙ לְיִשְׂרָאֵ֔ל לְמִשְׁפַּ֥ט יְהוָ֖ה וְלָרִ֑יב וַיָּשֻׁ֖בוּ יְרוּשָׁלִָֽם׃ (כו) וַיְצַ֥ו עֲלֵיהֶ֖ם לֵאמֹ֑ר כֹּ֤ה תַעֲשׂוּן֙ בְּיִרְאַ֣ת יְהוָ֔ה בֶּאֱמוּנָ֖ה וּבְלֵבָ֥ב שָׁלֵֽם׃ (כז) וְכָל־ רִיב֩ אֲשֶׁר־ יָב֨וֹא עֲלֵיכֶ֜ם מֵאֲחֵיכֶ֣ם ׀ הַיֹּשְׁבִ֣ים בְּעָרֵיהֶ֗ם בֵּֽין־ דָּ֣ם ׀ לְדָם֮ בֵּין־ תּוֹרָ֣ה לְמִצְוָה֮ לְחֻקִּ֣ים וּלְמִשְׁפָּטִים֒ וְהִזְהַרְתֶּ֣ם אֹתָ֔ם וְלֹ֤א יֶאְשְׁמוּ֙ לַיהוָ֔ה וְהָֽיָה־ קֶ֥צֶף עֲלֵיכֶ֖ם וְעַל־ אֲחֵיכֶ֑ם כֹּ֥ה תַעֲשׂ֖וּן וְלֹ֥א תֶאְשָֽׁמוּ׃ (כח) וְהִנֵּ֡ה אֲמַרְיָ֣הוּ כֹהֵן֩ הָרֹ֨אשׁ עֲלֵיכֶ֜ם לְכֹ֣ל דְּבַר־ יְהוָ֗ה וּזְבַדְיָ֨הוּ בֶן־ יִשְׁמָעֵ֜אל הַנָּגִ֤יד לְבֵית־ יְהוּדָה֙ לְכֹ֣ל דְּבַר־ הַמֶּ֔לֶךְ וְשֹׁטְרִ֥ים הַלְוִיִּ֖ם לִפְנֵיכֶ֑ם חִזְק֣וּ וַעֲשׂ֔וּ וִיהִ֥י יְהוָ֖ה עִם־ הַטּֽוֹב׃ (כט) וַיְהִ֣י אַֽחֲרֵיכֵ֡ן בָּ֣אוּ בְנֵי־ מוֹאָב֩ וּבְנֵ֨י עַמּ֜וֹן וְעִמָּהֶ֧ם ׀ מֵֽהָעַמּוֹנִ֛ים עַל־ יְהוֹשָׁפָ֖ט לַמִּלְחָמָֽה׃ (ל) וַיָּבֹ֗אוּ וַיַּגִּ֤ידוּ לִֽיהוֹשָׁפָט֙ לֵאמֹ֔ר בָּ֣א עָלֶ֜יךָ הָמ֥וֹן רָ֛ב מֵעֵ֥בֶר לַיָּ֖ם מֵאֲרָ֑ם וְהִנָּם֙ בְּחַֽצְצ֣וֹן תָּמָ֔ר הִ֖יא עֵ֥ין גֶּֽדִי׃ (לא) וַיִּרָ֕א וַיִּתֵּ֧ן יְהוֹשָׁפָ֛ט אֶת־ פָּנָ֖יו לִדְר֣וֹשׁ לַיהוָ֑ה וַיִּקְרָא־ צ֖וֹם עַל־ כָּל־ יְהוּדָֽה׃ (לב) וַיִּקָּבְצ֣וּ יְהוּדָ֔ה לְבַקֵּ֖שׁ מֵֽיְהוָ֑ה גַּ֚ם מִכָּל־ עָרֵ֣י יְהוּדָ֔ה בָּ֖אוּ לְבַקֵּ֥שׁ אֶת־ יְהוָֽה׃ (לג) וַיַּעֲמֹ֣ד יְהוֹשָׁפָ֗ט בִּקְהַ֧ל יְהוּדָ֛ה וִירוּשָׁלִַ֖ם בְּבֵ֣ית יְהוָ֑ה לִפְנֵ֖י הֶחָצֵ֥ר הַחֲדָשָֽׁה׃ (לד) וַיֹּאמַ֗ר יְהוָ֞ה אֱלֹהֵ֤י אֲבֹתֵ֙ינוּ֙ הֲלֹ֨א אַתָּֽה־ ה֤וּא אֱלֹהִים֙ בַּשָּׁמַ֔יִם וְאַתָּ֣ה מוֹשֵׁ֔ל בְּכֹ֖ל מַמְלְכ֣וֹת הַגּוֹיִ֑ם וּבְיָדְךָ֙ כֹּ֣חַ וּגְבוּרָ֔ה וְאֵ֥ין עִמְּךָ֖ לְהִתְיַצֵּֽב׃ (לה) הֲלֹ֣א ׀ אַתָּ֣ה אֱלֹהֵ֗ינוּ הוֹרַ֙שְׁתָּ֙ אֶת־ יֹשְׁבֵי֙ הָאָ֣רֶץ הַזֹּ֔את מִלִּפְנֵ֖י עַמְּךָ֣ יִשְׂרָאֵ֑ל וַֽתִּתְּנָ֗הּ לְזֶ֛רַע אַבְרָהָ֥ם אֹֽהַבְךָ֖ לְעוֹלָֽם׃ (לו) וַיֵּשְׁב֖וּ־ בָ֑הּ וַיִּבְנ֨וּ לְךָ֧ ׀ בָּ֛הּ מִקְדָּ֖שׁ לְשִׁמְךָ֥ לֵאמֹֽר׃ (לז) אִם־ תָּב֨וֹא עָלֵ֜ינוּ רָעָ֗ה חֶרֶב֮ שְׁפוֹט֮ וְדֶ֣בֶר וְרָעָב֒ נַֽעַמְדָ֞ה לִפְנֵ֨י הַבַּ֤יִת הַזֶּה֙ וּלְפָנֶ֔יךָ כִּ֥י שִׁמְךָ֖ בַּבַּ֣יִת הַזֶּ֑ה וְנִזְעַ֥ק אֵלֶ֛יךָ מִצָּרָתֵ֖נוּ וְתִשְׁמַ֥ע וְתוֹשִֽׁיעַ׃ (לח) וְעַתָּ֡ה הִנֵּה֩ בְנֵֽי־ עַמּ֨וֹן וּמוֹאָ֜ב וְהַר־ שֵׂעִ֗יר אֲ֠שֶׁר לֹֽא־ נָתַ֤תָּה לְיִשְׂרָאֵל֙ לָב֣וֹא בָהֶ֔ם בְּבֹאָ֖ם מֵאֶ֣רֶץ מִצְרָ֑יִם כִּ֛י סָ֥רוּ מֵעֲלֵיהֶ֖ם וְלֹ֥א הִשְׁמִידֽוּם׃ (לט) וְהִ֨נֵּה־ הֵ֔ם גֹּמְלִ֖ים עָלֵ֑ינוּ לָבוֹא֙ לְגָ֣רְשֵׁ֔נוּ מִיְּרֻשָּׁתְךָ֖ אֲשֶׁ֥ר הֽוֹרַשְׁתָּֽנוּ׃ (מ) אֱלֹהֵ֙ינוּ֙ הֲלֹ֣א תִשְׁפָּט־ בָּ֔ם כִּ֣י אֵ֥ין בָּ֙נוּ֙ כֹּ֔חַ לִ֠פְנֵי הֶהָמ֥וֹן הָרָ֛ב הַזֶּ֖ה הַבָּ֣א עָלֵ֑ינוּ וַאֲנַ֗חְנוּ לֹ֤א נֵדַע֙ מַֽה־ נַּעֲשֶׂ֔ה כִּ֥י עָלֶ֖יךָ עֵינֵֽינוּ׃ (מא) וְכָ֨ל־ יְהוּדָ֔ה עֹמְדִ֖ים לִפְנֵ֣י יְהוָ֑ה גַּם־ טַפָּ֖ם נְשֵׁיהֶ֥ם וּבְנֵיהֶֽם׃ (מב) וְיַחֲזִיאֵ֡ל בֶּן־ זְכַרְיָ֡הוּ בֶּן־ בְּ֠נָיָה בֶּן־ יְעִיאֵ֧ל בֶּן־ מַתַּנְיָ֛ה הַלֵּוִ֖י מִן־ בְּנֵ֣י אָסָ֑ף הָיְתָ֤ה עָלָיו֙ ר֣וּחַ יְהוָ֔ה בְּת֖וֹךְ הַקָּהָֽל׃ (מג) וַיֹּ֗אמֶר הַקְשִׁ֤יבוּ כָל־ יְהוּדָה֙ וְיֹשְׁבֵ֣י יְרוּשָׁלִַ֔ם וְהַמֶּ֖לֶךְ יְהוֹשָׁפָ֑ט כֹּֽה־ אָמַ֨ר יְהוָ֜ה לָכֶ֗ם אַ֠תֶּם אַל־ תִּֽירְא֤וּ וְאַל־ תֵּחַ֙תּוּ֙ מִפְּנֵ֨י הֶהָמ֤וֹן הָרָב֙ הַזֶּ֔ה כִּ֣י לֹ֥א לָכֶ֛ם הַמִּלְחָמָ֖ה כִּ֥י לֵאלֹהִֽים׃ (מד) מָחָר֙ רְד֣וּ עֲלֵיהֶ֔ם הִנָּ֥ם עֹלִ֖ים בְּמַעֲלֵ֣ה הַצִּ֑יץ וּמְצָאתֶ֤ם אֹתָם֙ בְּס֣וֹף הַנַּ֔חַל פְּנֵ֖י מִדְבַּ֥ר יְרוּאֵֽל׃ (מה) לֹ֥א לָכֶ֖ם לְהִלָּחֵ֣ם בָּזֹ֑את הִתְיַצְּב֣וּ עִמְד֡וּ וּרְא֣וּ אֶת־ יְשׁוּעַת֩ יְהוָ֨ה עִמָּכֶ֜ם יְהוּדָ֣ה וִֽירוּשָׁלִַ֗ם אַל־ תִּֽירְאוּ֙ וְאַל־ תֵּחַ֔תּוּ מָחָר֙ צְא֣וּ לִפְנֵיהֶ֔ם וַיהוָ֖ה עִמָּכֶֽם׃ (מו) וַיִּקֹּ֧ד יְהוֹשָׁפָ֛ט אַפַּ֖יִם אָ֑רְצָה וְכָל־ יְהוּדָ֞ה וְיֹשְׁבֵ֣י יְרוּשָׁלִַ֗ם נָֽפְלוּ֙ לִפְנֵ֣י יְהוָ֔ה לְהִֽשְׁתַּחֲוֺ֖ת לַיהוָֽה׃ (מז) וַיָּקֻ֧מוּ הַלְוִיִּ֛ם מִן־ בְּנֵ֥י הַקְּהָתִ֖ים וּמִן־ בְּנֵ֣י הַקָּרְחִ֑ים לְהַלֵּ֗ל לַיהוָה֙ אֱלֹהֵ֣י יִשְׂרָאֵ֔ל בְּק֥וֹל גָּד֖וֹל לְמָֽעְלָה׃ (מח) וַיַּשְׁכִּ֣ימוּ בַבֹּ֔קֶר וַיֵּצְא֖וּ לְמִדְבַּ֣ר תְּק֑וֹעַ וּבְצֵאתָ֞ם עָמַ֣ד יְהוֹשָׁפָ֗ט וַיֹּ֙אמֶר֙ שְׁמָע֗וּנִי יְהוּדָה֙ וְיֹשְׁבֵ֣י יְרוּשָׁלִַ֔ם הַאֲמִ֜ינוּ בַּיהוָ֤ה אֱלֹהֵיכֶם֙ וְתֵ֣אָמֵ֔נוּ הַאֲמִ֥ינוּ בִנְבִיאָ֖יו וְהַצְלִֽיחוּ׃ (מט) וַיִּוָּעַץ֙ אֶל־ הָעָ֔ם וַיַּעֲמֵ֤ד מְשֹֽׁרֲרִים֙ לַיהוָ֔ה וּֽמְהַֽלְלִ֖ים לְהַדְרַת־ קֹ֑דֶשׁ בְּצֵאת֙ לִפְנֵ֣י הֶֽחָל֔וּץ וְאֹֽמְרִים֙ הוֹד֣וּ לַיהוָ֔ה כִּ֥י לְעוֹלָ֖ם חַסְדּֽוֹ׃ (נ) וּבְעֵת֩ הֵחֵ֨לּוּ בְרִנָּ֜ה וּתְהִלָּ֗ה נָתַ֣ן יְהוָ֣ה ׀ מְ֠אָֽרְבִים עַל־ בְּנֵ֨י עַמּ֜וֹן מוֹאָ֧ב וְהַר־ שֵׂעִ֛יר הַבָּאִ֥ים לִֽיהוּדָ֖ה וַיִּנָּגֵֽפוּ׃ (נא) וַ֠יַּֽעַמְדוּ בְּנֵ֨י עַמּ֧וֹן וּמוֹאָ֛ב עַל־ יֹשְׁבֵ֥י הַר־ שֵׂעִ֖יר לְהַחֲרִ֣ים וּלְהַשְׁמִ֑יד וּכְכַלּוֹתָם֙ בְּיוֹשְׁבֵ֣י שֵׂעִ֔יר עָזְר֥וּ אִישׁ־ בְּרֵעֵ֖הוּ לְמַשְׁחִֽית׃ (נב) וִֽיהוּדָ֛ה בָּ֥א עַל־ הַמִּצְפֶּ֖ה לַמִּדְבָּ֑ר וַיִּפְנוּ֙ אֶל־ הֶ֣הָמ֔וֹן וְהִנָּ֧ם פְּגָרִ֛ים נֹפְלִ֥ים אַ֖רְצָה וְאֵ֥ין פְּלֵיטָֽה׃ (נג) וַיָּבֹ֨א יְהוֹשָׁפָ֣ט וְעַמּוֹ֮ לָבֹ֣ז אֶת־ שְׁלָלָם֒ וַיִּמְצְאוּ֩ בָהֶ֨ם לָרֹ֜ב וּרְכ֤וּשׁ וּפְגָרִים֙ וּכְלֵ֣י חֲמֻד֔וֹת וַיְנַצְּל֥וּ לָהֶ֖ם לְאֵ֣ין מַשָּׂ֑א וַיִּֽהְי֞וּ יָמִ֧ים שְׁלוֹשָׁ֛ה בֹּזְזִ֥ים אֶת־ הַשָּׁלָ֖ל כִּ֥י רַב־ הֽוּא׃ (נד) וּבַיּ֣וֹם הָרְבִעִ֗י נִקְהֲלוּ֙ לְעֵ֣מֶק בְּרָכָ֔ה כִּי־ שָׁ֖ם בֵּרֲכ֣וּ אֶת־ יְהוָ֑ה עַל־ כֵּ֡ן קָֽרְא֞וּ אֶת־ שֵׁ֨ם הַמָּק֥וֹם הַה֛וּא עֵ֥מֶק בְּרָכָ֖ה עַד־ הַיּֽוֹם׃ (נה) וַ֠יָּשֻׁבוּ כָּל־ אִ֨ישׁ יְהוּדָ֤ה וִֽירוּשָׁלִַ֙ם֙ וִֽיהוֹשָׁפָ֣ט בְּרֹאשָׁ֔ם לָשׁ֥וּב אֶל־ יְרוּשָׁלִַ֖ם בְּשִׂמְחָ֑ה כִּֽי־ שִׂמְּחָ֥ם יְהוָ֖ה מֵֽאוֹיְבֵיהֶֽם׃ (נו) וַיָּבֹ֙אוּ֙ יְר֣וּשָׁלִַ֔ם בִּנְבָלִ֥ים וּבְכִנֹּר֖וֹת וּבַחֲצֹצְר֑וֹת אֶל־ בֵּ֖ית יְהוָֽה׃ (נז) וַיְהִי֙ פַּ֣חַד אֱלֹהִ֔ים עַ֖ל כָּל־ מַמְלְכ֣וֹת הָאֲרָצ֑וֹת בְּשָׁמְעָ֕ם כִּ֚י נִלְחַ֣ם יְהוָ֔ה עִ֖ם אוֹיְבֵ֥י יִשְׂרָאֵֽל׃ (נח) וַתִּשְׁקֹ֖ט מַלְכ֣וּת יְהוֹשָׁפָ֑ט וַיָּ֧נַֽח ל֦וֹ אֱלֹהָ֖יו מִסָּבִֽיב׃
