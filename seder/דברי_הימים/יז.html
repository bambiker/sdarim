דברי הימים סדר יז

לפי החלוקה לפרקים: דברי הימים ב טו, טו – יח, יז · 55 פסוקים

פתיחה באתר · ספר דברי הימים כולו ·

(א) וַיִּשְׂמְח֨וּ כָל־ יְהוּדָ֜ה עַל־ הַשְּׁבוּעָ֗ה כִּ֤י בְכָל־ לְבָבָם֙ נִשְׁבָּ֔עוּ וּבְכָל־ רְצוֹנָ֣ם בִּקְשֻׁ֔הוּ וַיִּמָּצֵ֖א לָהֶ֑ם וַיָּ֧נַח יְהוָ֛ה לָהֶ֖ם מִסָּבִֽיב׃ (ב) וְגַֽם־ מַעֲכָ֞ה אֵ֣ם ׀ אָסָ֣א הַמֶּ֗לֶךְ הֱסִירָהּ֙ מִגְּבִירָ֔ה אֲשֶׁר־ עָשְׂתָ֥ה לַאֲשֵׁרָ֖ה מִפְלָ֑צֶת וַיִּכְרֹ֤ת אָסָא֙ אֶת־ מִפְלַצְתָּ֔הּ וַיָּ֕דֶק וַיִּשְׂרֹ֖ף בְּנַ֥חַל קִדְרֽוֹן׃ (ג) וְהַ֨בָּמ֔וֹת לֹא־ סָ֖רוּ מִיִּשְׂרָאֵ֑ל רַ֧ק לְבַב־ אָסָ֛א הָיָ֥ה שָׁלֵ֖ם כָּל־ יָמָֽיו׃ (ד) וַיָּבֵ֞א אֶת־ קָדְשֵׁ֥י אָבִ֛יו וְקָֽדָשָׁ֖יו בֵּ֣ית הָאֱלֹהִ֑ים כֶּ֥סֶף וְזָהָ֖ב וְכֵלִֽים׃ (ה) וּמִלְחָמָ֖ה לֹ֣א הָיָ֑תָה עַ֛ד שְׁנַת־ שְׁלֹשִׁ֥ים וְחָמֵ֖שׁ לְמַלְכ֥וּת אָסָֽא׃ (ו) בִּשְׁנַ֨ת שְׁלֹשִׁ֤ים וָשֵׁשׁ֙ לְמַלְכ֣וּת אָסָ֔א עָלָ֞ה בַּעְשָׁ֤א מֶֽלֶךְ־ יִשְׂרָאֵל֙ עַל־ יְהוּדָ֔ה וַיִּ֖בֶן אֶת־ הָרָמָ֑ה לְבִלְתִּ֗י תֵּ֚ת יוֹצֵ֣א וָבָ֔א לְאָסָ֖א מֶ֥לֶךְ יְהוּדָֽה׃ (ז) וַיֹּצֵ֨א אָסָ֜א כֶּ֣סֶף וְזָהָ֗ב מֵאֹֽצְר֛וֹת בֵּ֥ית יְהוָ֖ה וּבֵ֣ית הַמֶּ֑לֶךְ וַיִּשְׁלַ֗ח אֶל־ בֶּן־ הֲדַד֙ מֶ֣לֶךְ אֲרָ֔ם הַיּוֹשֵׁ֥ב בְּדַרְמֶ֖שֶׂק לֵאמֹֽר׃ (ח) בְּרִית֙ בֵּינִ֣י וּבֵינֶ֔ךָ וּבֵ֥ין אָבִ֖י וּבֵ֣ין אָבִ֑יךָ הִנֵּ֨ה שָׁלַ֤חְתִּֽי לְךָ֙ כֶּ֣סֶף וְזָהָ֔ב לֵ֞ךְ הָפֵ֣ר בְּרִֽיתְךָ֗ אֶת־ בַּעְשָׁא֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל וְיַעֲלֶ֖ה מֵעָלָֽי׃ (ט) וַיִּשְׁמַ֨ע בֶּן הֲדַ֜ד אֶל־ הַמֶּ֣לֶךְ אָסָ֗א וַ֠יִּשְׁלַח אֶת־ שָׂרֵ֨י הַחֲיָלִ֤ים אֲשֶׁר־ לוֹ֙ אֶל־ עָרֵ֣י יִשְׂרָאֵ֔ל וַיַּכּוּ֙ אֶת־ עִיּ֣וֹן וְאֶת־ דָּ֔ן וְאֵ֖ת אָבֵ֣ל מָ֑יִם וְאֵ֥ת כָּֽל־ מִסְכְּנ֖וֹת עָרֵ֥י נַפְתָּלִֽי׃ (י) וַיְהִי֙ כִּשְׁמֹ֣עַ בַּעְשָׁ֔א וַיֶּחְדַּ֕ל מִבְּנ֖וֹת אֶת־ הָרָמָ֑ה וַיַּשְׁבֵּ֖ת אֶת־ מְלַאכְתּֽוֹ׃ (יא) וְאָסָ֣א הַמֶּ֗לֶךְ לָקַח֙ אֶת־ כָּל־ יְהוּדָ֔ה וַיִּשְׂא֞וּ אֶת־ אַבְנֵ֤י הָֽרָמָה֙ וְאֶת־ עֵצֶ֔יהָ אֲשֶׁ֥ר בָּנָ֖ה בַּעְשָׁ֑א וַיִּ֣בֶן בָּהֶ֔ם אֶת־ גֶּ֖בַע וְאֶת־ הַמִּצְפָּֽה׃ (יב) וּבָעֵ֣ת הַהִ֗יא בָּ֚א חֲנָ֣נִי הָרֹאֶ֔ה אֶל־ אָסָ֖א מֶ֣לֶךְ יְהוּדָ֑ה וַיֹּ֣אמֶר אֵלָ֗יו בְּהִשָּׁ֨עֶנְךָ֜ עַל־ מֶ֤לֶךְ אֲרָם֙ וְלֹ֤א נִשְׁעַ֙נְתָּ֙ עַל־ יְהוָ֣ה אֱלֹהֶ֔יךָ עַל־ כֵּ֗ן נִמְלַ֛ט חֵ֥יל מֶֽלֶךְ־ אֲרָ֖ם מִיָּדֶֽךָ׃ (יג) הֲלֹ֧א הַכּוּשִׁ֣ים וְהַלּוּבִ֗ים הָי֨וּ לְחַ֧יִל ׀ לָרֹ֛ב לְרֶ֥כֶב וּלְפָרָשִׁ֖ים לְהַרְבֵּ֣ה מְאֹ֑ד וּבְהִשָּֽׁעֶנְךָ֥ עַל־ יְהוָ֖ה נְתָנָ֥ם בְּיָדֶֽךָ׃ (יד) כִּ֣י יְהוָ֗ה עֵינָ֞יו מְשֹׁטְט֤וֹת בְּכָל־ הָאָ֙רֶץ֙ לְ֠הִתְחַזֵּק עִם־ לְבָבָ֥ם שָׁלֵ֛ם אֵלָ֖יו נִסְכַּ֣לְתָּ עַל־ זֹ֑את כִּ֣י מֵעַ֔תָּה יֵ֥שׁ עִמְּךָ֖ מִלְחָמֽוֹת׃ (טו) וַיִּכְעַ֨ס אָסָ֜א אֶל־ הָרֹאֶ֗ה וַֽיִּתְּנֵ֙הוּ֙ בֵּ֣ית הַמַּהְפֶּ֔כֶת כִּֽי־ בְזַ֥עַף עִמּ֖וֹ עַל־ זֹ֑את וַיְרַצֵּ֥ץ אָסָ֛א מִן־ הָעָ֖ם בָּעֵ֥ת הַהִֽיא׃ (טז) וְהִנֵּה֙ דִּבְרֵ֣י אָסָ֔א הָרִאשׁוֹנִ֖ים וְהָאַחֲרוֹנִ֑ים הִנָּ֤ם כְּתוּבִים֙ עַל־ סֵ֣פֶר הַמְּלָכִ֔ים לִיהוּדָ֖ה וְיִשְׂרָאֵֽל׃ (יז) וַיֶּחֱלֶ֣א אָסָ֡א בִּשְׁנַת֩ שְׁלוֹשִׁ֨ים וָתֵ֤שַׁע לְמַלְכוּתוֹ֙ בְּרַגְלָ֔יו עַד־ לְמַ֖עְלָה חָלְי֑וֹ וְגַם־ בְּחָלְיוֹ֙ לֹא־ דָרַ֣שׁ אֶת־ יְהוָ֔ה כִּ֖י בָּרֹפְאִֽים׃ (יח) וַיִּשְׁכַּ֥ב אָסָ֖א עִם־ אֲבֹתָ֑יו וַיָּ֕מָת בִּשְׁנַ֛ת אַרְבָּעִ֥ים וְאַחַ֖ת לְמָלְכֽוֹ׃ (יט) וַיִּקְבְּרֻ֣הוּ בְקִבְרֹתָ֗יו אֲשֶׁ֣ר כָּֽרָה־ לוֹ֮ בְּעִ֣יר דָּוִיד֒ וַיַּשְׁכִּיבֻ֗הוּ בַּמִּשְׁכָּב֙ אֲשֶׁ֤ר מִלֵּא֙ בְּשָׂמִ֣ים וּזְנִ֔ים מְרֻקָּחִ֖ים בְּמִרְקַ֣חַת מַעֲשֶׂ֑ה וַיִּשְׂרְפוּ־ ל֥וֹ שְׂרֵפָ֖ה גְּדוֹלָ֥ה עַד־ לִמְאֹֽד׃ (כ) וַיִּמְלֹ֛ךְ יְהוֹשָׁפָ֥ט בְּנ֖וֹ תַּחְתָּ֑יו וַיִּתְחַזֵּ֖ק עַל־ יִשְׂרָאֵֽל׃ (כא) וַיִּ֨תֶּן־ חַ֔יִל בְּכָל־ עָרֵ֥י יְהוּדָ֖ה הַבְּצֻר֑וֹת וַיִּתֵּ֤ן נְצִיבִים֙ בְּאֶ֣רֶץ יְהוּדָ֔ה וּבְעָרֵ֣י אֶפְרַ֔יִם אֲשֶׁ֥ר לָכַ֖ד אָסָ֥א אָבִֽיו׃ (כב) וַיְהִ֥י יְהוָ֖ה עִם־ יְהוֹשָׁפָ֑ט כִּ֣י הָלַ֗ךְ בְּדַרְכֵ֞י דָּוִ֤יד אָבִיו֙ הָרִ֣אשֹׁנִ֔ים וְלֹ֥א דָרַ֖שׁ לַבְּעָלִֽים׃ (כג) כִּ֠י לֵֽאלֹהֵ֤י אָבִיו֙ דָּרָ֔שׁ וּבְמִצְוֺתָ֖יו הָלָ֑ךְ וְלֹ֖א כְּמַעֲשֵׂ֥ה יִשְׂרָאֵֽל׃ (כד) וַיָּ֨כֶן יְהוָ֤ה אֶת־ הַמַּמְלָכָה֙ בְּיָד֔וֹ וַיִּתְּנ֧וּ כָל־ יְהוּדָ֛ה מִנְחָ֖ה לִיהוֹשָׁפָ֑ט וַֽיְהִי־ ל֥וֹ עֹֽשֶׁר־ וְכָב֖וֹד לָרֹֽב׃ (כה) וַיִּגְבַּ֥הּ לִבּ֖וֹ בְּדַרְכֵ֣י יְהוָ֑ה וְע֗וֹד הֵסִ֛יר אֶת־ הַבָּמ֥וֹת וְאֶת־ הָאֲשֵׁרִ֖ים מִיהוּדָֽה׃ (כו) וּבִשְׁנַ֨ת שָׁל֜וֹשׁ לְמָלְכ֗וֹ שָׁלַ֤ח לְשָׂרָיו֙ לְבֶן־ חַ֙יִל֙ וּלְעֹבַדְיָ֣ה וְלִזְכַרְיָ֔ה וְלִנְתַנְאֵ֖ל וּלְמִיכָיָ֑הוּ לְלַמֵּ֖ד בְּעָרֵ֥י יְהוּדָֽה׃ (כז) וְעִמָּהֶ֣ם הַלְוִיִּ֗ם שְֽׁמַֽעְיָ֡הוּ וּנְתַנְיָ֡הוּ וּזְבַדְיָ֡הוּ וַעֲשָׂהאֵ֡ל (ושמרימות) וּשְׁמִֽירָמ֡וֹת וִֽיהוֹנָתָן֩ וַאֲדֹ֨נִיָּ֧הוּ וְטֽוֹבִיָּ֛הוּ וְט֥וֹב אֲדוֹנִיָּ֖ה הַלְוִיִּ֑ם וְעִמָּהֶ֛ם אֱלִישָׁמָ֥ע וִֽיהוֹרָ֖ם הַכֹּהֲנִֽים׃ (כח) וַֽיְלַמְּדוּ֙ בִּֽיהוּדָ֔ה וְעִ֨מָּהֶ֔ם סֵ֖פֶר תּוֹרַ֣ת יְהוָ֑ה וַיָּסֹ֙בּוּ֙ בְּכָל־ עָרֵ֣י יְהוּדָ֔ה וַֽיְלַמְּד֖וּ בָּעָֽם׃ (כט) וַיְהִ֣י ׀ פַּ֣חַד יְהוָ֗ה עַ֚ל כָּל־ מַמְלְכ֣וֹת הָֽאֲרָצ֔וֹת אֲשֶׁ֖ר סְבִיב֣וֹת יְהוּדָ֑ה וְלֹ֥א נִלְחֲמ֖וּ עִם־ יְהוֹשָׁפָֽט׃ (ל) וּמִן־ פְּלִשְׁתִּ֗ים מְבִיאִ֧ים לִֽיהוֹשָׁפָ֛ט מִנְחָ֖ה וְכֶ֣סֶף מַשָּׂ֑א גַּ֣ם הָֽעַרְבִיאִ֗ים מְבִיאִ֥ים לוֹ֙ צֹ֕אן אֵילִ֔ים שִׁבְעַ֤ת אֲלָפִים֙ וּשְׁבַ֣ע מֵא֔וֹת וּתְיָשִׁ֕ים שִׁבְעַ֥ת אֲלָפִ֖ים וּשְׁבַ֥ע מֵאֽוֹת׃ (לא) וַיְהִ֧י יְהוֹשָׁפָ֛ט הֹלֵ֥ךְ וְגָדֵ֖ל עַד־ לְמָ֑עְלָה וַיִּ֧בֶן בִּֽיהוּדָ֛ה בִּירָנִיּ֖וֹת וְעָרֵ֥י מִסְכְּנֽוֹת׃ (לב) וּמְלָאכָ֥ה רַבָּ֛ה הָ֥יָה ל֖וֹ בְּעָרֵ֣י יְהוּדָ֑ה וְאַנְשֵׁ֧י מִלְחָמָ֛ה גִּבּ֥וֹרֵי חַ֖יִל בִּירוּשָׁלִָֽם׃ (לג) וְאֵ֥לֶּה פְקֻדָּתָ֖ם לְבֵ֣ית אֲבוֹתֵיהֶ֑ם לִֽיהוּדָה֙ שָׂרֵ֣י אֲלָפִ֔ים עַדְנָ֣ה הַשָּׂ֔ר וְעִמּוֹ֙ גִּבּ֣וֹרֵי חַ֔יִל שְׁלֹ֥שׁ מֵא֖וֹת אָֽלֶף׃ (לד) וְעַל־ יָד֖וֹ יְהוֹחָנָ֣ן הַשָּׂ֑ר וְעִמּ֕וֹ מָאתַ֥יִם וּשְׁמוֹנִ֖ים אָֽלֶף׃ (לה) וְעַל־ יָדוֹ֙ עֲמַסְיָ֣ה בֶן־ זִכְרִ֔י הַמִּתְנַדֵּ֖ב לַיהוָ֑ה וְעִמּ֛וֹ מָאתַ֥יִם אֶ֖לֶף גִּבּ֥וֹר חָֽיִל׃ (לו) וּמִ֨ן־ בִּנְיָמִ֔ן גִּבּ֥וֹר חַ֖יִל אֶלְיָדָ֑ע וְעִמּ֛וֹ נֹֽשְׁקֵי־ קֶ֥שֶׁת וּמָגֵ֖ן מָאתַ֥יִם אָֽלֶף׃ (לז) וְעַל־ יָד֖וֹ יְהוֹזָבָ֑ד וְעִמּ֛וֹ מֵאָֽה־ וּשְׁמוֹנִ֥ים אֶ֖לֶף חֲלוּצֵ֥י צָבָֽא׃ (לח) אֵ֖לֶּה הַמְשָׁרְתִ֣ים אֶת־ הַמֶּ֑לֶךְ מִלְּבַ֞ד אֲשֶׁר־ נָתַ֥ן הַמֶּ֛לֶךְ בְּעָרֵ֥י הַמִּבְצָ֖ר בְּכָל־ יְהוּדָֽה׃ (לט) וַיְהִ֧י לִֽיהוֹשָׁפָ֛ט עֹ֥שֶׁר וְכָב֖וֹד לָרֹ֑ב וַיִּתְחַתֵּ֖ן לְאַחְאָֽב׃ (מ) וַיֵּרֶד֩ לְקֵ֨ץ שָׁנִ֤ים אֶל־ אַחְאָב֙ לְשֹׁ֣מְר֔וֹן וַיִּֽזְבַּֽח־ ל֨וֹ אַחְאָ֜ב צֹ֤אן וּבָקָר֙ לָרֹ֔ב וְלָעָ֖ם אֲשֶׁ֣ר עִמּ֑וֹ וַיְסִיתֵ֕הוּ לַעֲל֖וֹת אֶל־ רָמ֥וֹת גִּלְעָֽד׃ (מא) וַיֹּ֜אמֶר אַחְאָ֣ב מֶֽלֶךְ־ יִשְׂרָאֵ֗ל אֶל־ יְהֽוֹשָׁפָט֙ מֶ֣לֶךְ יְהוּדָ֔ה הֲתֵלֵ֥ךְ עִמִּ֖י רָמֹ֣ת גִּלְעָ֑ד וַיֹּ֣אמֶר ל֗וֹ כָּמ֤וֹנִי כָמ֙וֹךָ֙ וּכְעַמְּךָ֣ עַמִּ֔י וְעִמְּךָ֖ בַּמִּלְחָמָֽה׃ (מב) וַיֹּ֥אמֶר יְהוֹשָׁפָ֖ט אֶל־ מֶ֣לֶךְ יִשְׂרָאֵ֑ל דְּרָשׁ־ נָ֥א כַיּ֖וֹם אֶת־ דְּבַ֥ר יְהוָֽה׃ (מג) וַיִּקְבֹּ֨ץ מֶֽלֶךְ־ יִשְׂרָאֵ֥ל אֶֽת־ הַנְּבִאִים֮ אַרְבַּ֣ע מֵא֣וֹת אִישׁ֒ וַיֹּ֣אמֶר אֲלֵהֶ֗ם הֲנֵלֵ֞ךְ אֶל־ רָמֹ֥ת גִּלְעָ֛ד לַמִּלְחָמָ֖ה אִם־ אֶחְדָּ֑ל וַיֹּאמְר֣וּ עֲלֵ֔ה וְיִתֵּ֥ן הָאֱלֹהִ֖ים בְּיַ֥ד הַמֶּֽלֶךְ׃ (מד) וַיֹּ֙אמֶר֙ יְה֣וֹשָׁפָ֔ט הַאֵ֨ין פֹּ֥ה נָבִ֛יא לַיהוָ֖ה ע֑וֹד וְנִדְרְשָׁ֖ה מֵאֹתֽוֹ׃ (מה) וַיֹּ֣אמֶר מֶֽלֶךְ־ יִשְׂרָאֵ֣ל ׀ אֶֽל־ יְהוֹשָׁפָ֡ט ע֣וֹד אִישׁ־ אֶחָ֡ד לִדְרוֹשׁ֩ אֶת־ יְהוָ֨ה מֵֽאֹת֜וֹ וַאֲנִ֣י שְׂנֵאתִ֗יהוּ כִּֽי־ אֵ֠ינֶנּוּ מִתְנַבֵּ֨א עָלַ֤י לְטוֹבָה֙ כִּ֣י כָל־ יָמָ֣יו לְרָעָ֔ה ה֖וּא מִיכָ֣יְהוּ בֶן־ יִמְלָ֑א וַיֹּ֙אמֶר֙ יְה֣וֹשָׁפָ֔ט אַל־ יֹאמַ֥ר הַמֶּ֖לֶךְ כֵּֽן׃ (מו) וַיִּקְרָא֙ מֶ֣לֶךְ יִשְׂרָאֵ֔ל אֶל־ סָרִ֖יס אֶחָ֑ד וַיֹּ֕אמֶר מַהֵ֖ר (מיכהו) מִיכָ֥יְהוּ בֶן־ יִמְלָֽא׃ (מז) וּמֶ֣לֶךְ יִשְׂרָאֵ֡ל וִֽיהוֹשָׁפָ֣ט מֶֽלֶךְ־ יְהוּדָ֡ה יוֹשְׁבִים֩ אִ֨ישׁ עַל־ כִּסְא֜וֹ מְלֻבָּשִׁ֤ים בְּגָדִים֙ וְיֹשְׁבִ֣ים בְּגֹ֔רֶן פֶּ֖תַח שַׁ֣עַר שֹׁמְר֑וֹן וְכָל־ הַ֨נְּבִיאִ֔ים מִֽתְנַבְּאִ֖ים לִפְנֵיהֶֽם׃ (מח) וַיַּ֥עַשׂ ל֛וֹ צִדְקִיָּ֥הוּ בֶֽן־ כְּנַעֲנָ֖ה קַרְנֵ֣י בַרְזֶ֑ל וַיֹּ֙אמֶר֙ כֹּֽה־ אָמַ֣ר יְהוָ֔ה בְּאֵ֛לֶּה תְּנַגַּ֥ח אֶת־ אֲרָ֖ם עַד־ כַּלּוֹתָֽם׃ (מט) וְכָל־ הַ֨נְּבִאִ֔ים נִבְּאִ֥ים כֵּ֖ן לֵאמֹ֑ר עֲלֵ֞ה רָמֹ֤ת גִּלְעָד֙ וְהַצְלַ֔ח וְנָתַ֥ן יְהוָ֖ה בְּיַ֥ד הַמֶּֽלֶךְ׃ (נ) וְהַמַּלְאָ֞ךְ אֲשֶׁר־ הָלַ֣ךְ ׀ לִקְרֹ֣א לְמִיכָ֗יְהוּ דִּבֶּ֤ר אֵלָיו֙ לֵאמֹ֔ר הִנֵּ֞ה דִּבְרֵ֧י הַנְּבִאִ֛ים פֶּֽה־ אֶחָ֥ד ט֖וֹב אֶל־ הַמֶּ֑לֶךְ וִֽיהִי־ נָ֧א דְבָרְךָ֛ כְּאַחַ֥ד מֵהֶ֖ם וְדִבַּ֥רְתָּ טּֽוֹב׃ (נא) וַיֹּ֖אמֶר מִיכָ֑יְהוּ חַי־ יְהוָ֕ה כִּ֛י אֶת־ אֲשֶׁר־ יֹאמַ֥ר אֱלֹהַ֖י אֹת֥וֹ אֲדַבֵּֽר׃ (נב) וַיָּבֹא֮ אֶל־ הַמֶּלֶךְ֒ וַיֹּ֨אמֶר הַמֶּ֜לֶךְ אֵלָ֗יו מִיכָה֙ הֲנֵלֵ֞ךְ אֶל־ רָמֹ֥ת גִּלְעָ֛ד לַמִּלְחָמָ֖ה אִם־ אֶחְדָּ֑ל וַיֹּ֙אמֶר֙ עֲל֣וּ וְהַצְלִ֔יחוּ וְיִנָּתְנ֖וּ בְּיֶדְכֶֽם׃ (נג) וַיֹּ֤אמֶר אֵלָיו֙ הַמֶּ֔לֶךְ עַד־ כַּמֶּ֥ה פְעָמִ֖ים אֲנִ֣י מַשְׁבִּיעֶ֑ךָ אֲ֠שֶׁר לֹֽא־ תְדַבֵּ֥ר אֵלַ֛י רַק־ אֱמֶ֖ת בְּשֵׁ֥ם יְהוָֽה׃ (נד) וַיֹּ֗אמֶר רָאִ֤יתִי אֶת־ כָּל־ יִשְׂרָאֵל֙ נְפוֹצִ֣ים עַל־ הֶֽהָרִ֔ים כַּצֹּ֕אן אֲשֶׁ֥ר אֵין־ לָהֶ֖ן רֹעֶ֑ה וַיֹּ֤אמֶר יְהוָה֙ לֹֽא־ אֲדֹנִ֣ים לָאֵ֔לֶּה יָשׁ֥וּבוּ אִישׁ־ לְבֵית֖וֹ בְּשָׁלֽוֹם׃ (נה) וַיֹּ֥אמֶר מֶֽלֶךְ־ יִשְׂרָאֵ֖ל אֶל־ יְהוֹשָׁפָ֑ט הֲלֹא֙ אָמַ֣רְתִּי אֵלֶ֔יךָ לֹא־ יִתְנַבֵּ֥א עָלַ֛י ט֖וֹב כִּ֥י אִם־ לְרָֽע׃
