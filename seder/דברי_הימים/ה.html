דברי הימים סדר ה

פתיחה באתר · ספר דברי הימים כולו ·

(א) וַיֵּ֥לֶךְ דָּוִ֖יד הָל֣וֹךְ וְגָד֑וֹל וַיהוָ֥ה צְבָא֖וֹת עִמּֽוֹ׃ (ב) וְאֵ֨לֶּה רָאשֵׁ֤י הַגִּבּוֹרִים֙ אֲשֶׁ֣ר לְדָוִ֔יד הַמִּתְחַזְּקִ֨ים עִמּ֧וֹ בְמַלְכוּת֛וֹ עִם־ כָּל־ יִשְׂרָאֵ֖ל לְהַמְלִיכ֑וֹ כִּדְבַ֥ר יְהוָ֖ה עַל־ יִשְׂרָאֵֽל׃ (ג) וְאֵ֛לֶּה מִסְפַּ֥ר הַגִּבֹּרִ֖ים אֲשֶׁ֣ר לְדָוִ֑יד יָשָׁבְעָ֣ם בֶּן־ חַכְמוֹנִ֗י רֹ֚אשׁ (השלושים) הַשָּׁ֣לִישִׁ֔ים הֽוּא־ עוֹרֵ֧ר אֶת־ חֲנִית֛וֹ עַל־ שְׁלֹשׁ־ מֵא֥וֹת חָלָ֖ל בְּפַ֥עַם אֶחָֽת׃ (ד) וְאַחֲרָ֛יו אֶלְעָזָ֥ר בֶּן־ דּוֹד֖וֹ הָאֲחוֹחִ֑י ה֖וּא בִּשְׁלוֹשָׁ֥ה הַגִּבֹּרִֽים׃ (ה) הֽוּא־ הָיָ֨ה עִם־ דָּוִ֜יד בַּפַּ֣ס דַּמִּ֗ים וְהַפְּלִשְׁתִּים֙ נֶאֱסְפוּ־ שָׁ֣ם לַמִּלְחָמָ֔ה וַתְּהִ֛י חֶלְקַ֥ת הַשָּׂדֶ֖ה מְלֵאָ֣ה שְׂעוֹרִ֑ים וְהָעָ֥ם נָ֖סוּ מִפְּנֵ֥י פְלִשְׁתִּֽים׃ (ו) וַיִּֽתְיַצְּב֤וּ בְתוֹךְ־ הַחֶלְקָה֙ וַיַּצִּיל֔וּהָ וַיַּכּ֖וּ אֶת־ פְּלִשְׁתִּ֑ים וַיּ֥וֹשַׁע יְהוָ֖ה תְּשׁוּעָ֥ה גְדוֹלָֽה׃ (ז) וַיֵּרְד֡וּ שְֽׁלוֹשָׁה֩ מִן־ הַשְּׁלוֹשִׁ֨ים רֹ֤אשׁ עַל־ הַצֻּר֙ אֶל־ דָּוִ֔יד אֶל־ מְעָרַ֖ת עֲדֻלָּ֑ם וּמַחֲנֵ֣ה פְלִשְׁתִּ֔ים חֹנָ֖ה בְּעֵ֥מֶק רְפָאִֽים׃ (ח) וְדָוִ֖יד אָ֣ז בַּמְּצוּדָ֑ה וּנְצִ֣יב פְּלִשְׁתִּ֔ים אָ֖ז בְּבֵ֥ית לָֽחֶם׃ (ט) (ויתאו) וַיִּתְאָ֥יו דָּוִ֖יד וַיֹּאמַ֑ר מִ֚י יַשְׁקֵ֣נִי מַ֔יִם מִבּ֥וֹר בֵּֽית־ לֶ֖חֶם אֲשֶׁ֥ר בַּשָּֽׁעַר׃ (י) וַיִּבְקְע֨וּ הַשְּׁלֹשָׁ֜ה בְּמַחֲנֵ֣ה פְלִשְׁתִּ֗ים וַיִּֽשְׁאֲבוּ־ מַ֙יִם֙ מִבּ֤וֹר בֵּֽית־ לֶ֙חֶם֙ אֲשֶׁ֣ר בַּשַּׁ֔עַר וַיִּשְׂא֖וּ וַיָּבִ֣אוּ אֶל־ דָּוִ֑יד וְלֹֽא־ אָבָ֤ה דָוִיד֙ לִשְׁתּוֹתָ֔ם וַיְנַסֵּ֥ךְ אֹתָ֖ם לַיהוָֽה׃ (יא) וַיֹּ֡אמֶר חָלִילָה֩ לִּ֨י מֵאֱלֹהַ֜י מֵעֲשׂ֣וֹת זֹ֗את הֲדַ֣ם הָאֲנָשִׁים֩ הָאֵ֨לֶּה אֶשְׁתֶּ֤ה בְנַפְשׁוֹתָם֙ כִּ֣י בְנַפְשׁוֹתָ֣ם הֱבִיא֔וּם וְלֹ֥א אָבָ֖ה לִשְׁתּוֹתָ֑ם אֵ֣לֶּה עָשׂ֔וּ שְׁלֹ֖שֶׁת הַגִּבּוֹרִֽים׃ (יב) וְאַבְשַׁ֣י אֲחִֽי־ יוֹאָ֗ב ה֤וּא הָיָה֙ רֹ֣אשׁ הַשְּׁלוֹשָׁ֔ה וְהוּא֙ עוֹרֵ֣ר אֶת־ חֲנִית֔וֹ עַל־ שְׁלֹ֥שׁ מֵא֖וֹת חָלָ֑ל (ולא)־ וְלוֹ־ שֵׁ֖ם בַּשְּׁלוֹשָֽׁה׃ (יג) מִן־ הַשְּׁלוֹשָׁ֤ה בַשְּׁנַ֙יִם֙ נִכְבָּ֔ד וַיְהִ֥י לָהֶ֖ם לְשָׂ֑ר וְעַד־ הַשְּׁלוֹשָׁ֖ה לֹֽא־ בָֽא׃ (יד) בְּנָיָ֨ה בֶן־ יְהוֹיָדָ֧ע בֶּן־ אִֽישׁ־ חַ֛יִל רַב־ פְּעָלִ֖ים מִֽן־ קַבְצְאֵ֑ל ה֣וּא הִכָּ֗ה אֵ֣ת שְׁנֵ֤י אֲרִיאֵל֙ מוֹאָ֔ב וְ֠הוּא יָרַ֞ד וְהִכָּ֧ה אֶֽת־ הָאֲרִ֛י בְּת֥וֹךְ הַבּ֖וֹר בְּי֥וֹם הַשָּֽׁלֶג׃ (טו) וְהֽוּא־ הִכָּה֩ אֶת־ הָאִ֨ישׁ הַמִּצְרִ֜י אִ֥ישׁ מִדָּ֣ה ׀ חָמֵ֣שׁ בָּאַמָּ֗ה וּבְיַ֨ד הַמִּצְרִ֤י חֲנִית֙ כִּמְנ֣וֹר אֹרְגִ֔ים וַיֵּ֥רֶד אֵלָ֖יו בַּשָּׁ֑בֶט וַיִּגְזֹ֤ל אֶֽת־ הַחֲנִית֙ מִיַּ֣ד הַמִּצְרִ֔י וַיַּהַרְגֵ֖הוּ בַּחֲנִיתֽוֹ׃ (טז) אֵ֣לֶּה עָשָׂ֔ה בְּנָיָ֖הוּ בֶּן־ יְהוֹיָדָ֑ע וְלוֹ־ שֵׁ֖ם בִּשְׁלוֹשָׁ֥ה הַגִּבֹּרִֽים׃ (יז) מִן־ הַשְּׁלוֹשִׁ֗ים הִנּ֤וֹ נִכְבָּד֙ ה֔וּא וְאֶל־ הַשְּׁלוֹשָׁ֖ה לֹא־ בָ֑א וַיְשִׂימֵ֥הוּ דָוִ֖יד עַל־ מִשְׁמַעְתּֽוֹ׃ (יח) וְגִבּוֹרֵ֖י הַחֲיָלִ֑ים עֲשָׂה־ אֵל֙ אֲחִ֣י יוֹאָ֔ב אֶלְחָנָ֥ן בֶּן־ דּוֹד֖וֹ מִבֵּ֥ית לָֽחֶם׃ (יט) שַׁמּוֹת֙ הַהֲרוֹרִ֔י חֶ֖לֶץ הַפְּלוֹנִֽי׃ (כ) עִירָ֤א בֶן־ עִקֵּשׁ֙ הַתְּקוֹעִ֔י אֲבִיעֶ֖זֶר הָעֲנְּתוֹתִֽי׃ (כא) סִבְּכַי֙ הַחֻ֣שָׁתִ֔י עִילַ֖י הָאֲחוֹחִֽי׃ (כב) מַהְרַי֙ הַנְּטֹ֣פָתִ֔י חֵ֥לֶד בֶּֽן־ בַּֽעֲנָ֖ה הַנְּטוֹפָתִֽי׃ (כג) אִיתַ֣י בֶּן־ רִיבַ֗י מִגִּבְעַת֙ בְּנֵ֣י בִנְיָמִ֔ן בְּנָיָ֖ה הַפִּרְעָתֹנִֽי׃ (כד) חוּרַי֙ מִנַּ֣חֲלֵי גָ֔עַשׁ אֲבִיאֵ֖ל הָעַרְבָתִֽי׃ (כה) עַזְמָ֙וֶת֙ הַבַּ֣חֲרוּמִ֔י אֶלְיַחְבָּ֖א הַשַּׁעַלְבֹנִֽי׃ (כו) בְּנֵ֗י הָשֵׁם֙ הַגִּ֣זוֹנִ֔י יוֹנָתָ֥ן בֶּן־ שָׁגֵ֖ה הַהֲרָרִֽי׃ (כז) אֲחִיאָ֧ם בֶּן־ שָׂכָ֛ר הַהֲרָרִ֖י אֱלִיפַ֥ל בֶּן־ אֽוּר׃ (כח) חֵ֚פֶר הַמְּכֵ֣רָתִ֔י אֲחִיָּ֖ה הַפְּלֹנִֽי׃ (כט) חֶצְרוֹ֙ הַֽכַּרְמְלִ֔י נַעֲרַ֖י בֶּן־ אֶזְבָּֽי׃ (ל) יוֹאֵל֙ אֲחִ֣י נָתָ֔ן מִבְחָ֖ר בֶּן־ הַגְרִֽי׃ (לא) צֶ֖לֶק הָעַמּוֹנִ֑י נַחְרַי֙ הַבֵּ֣רֹתִ֔י נֹשֵׂ֕א כְּלֵ֖י יוֹאָ֥ב בֶּן־ צְרוּיָֽה׃ (לב) עִירָא֙ הַיִּתְרִ֔י גָּרֵ֖ב הַיִּתְרִֽי׃ (לג) אֽוּרִיָּה֙ הַחִתִּ֔י זָבָ֖ד בֶּן־ אַחְלָֽי׃ (לד) עֲדִינָ֨א בֶן־ שִׁיזָ֜א הָרֽאוּבֵנִ֗י רֹ֛אשׁ לָרֽאוּבֵנִ֖י וְעָלָ֥יו שְׁלוֹשִֽׁים׃ (לה) חָנָן֙ בֶּֽן־ מַעֲכָ֔ה וְיוֹשָׁפָ֖ט הַמִּתְנִֽי׃ (לו) עֻזִיָּ֖א הָעֲשְׁתְּרָתִ֑י שָׁמָע֙ (ויעואל) וִֽיעִיאֵ֔ל בְּנֵ֖י חוֹתָ֥ם הָעֲרֹעֵרִֽי׃ (לז) יְדִֽיעֲאֵל֙ בֶּן־ שִׁמְרִ֔י וְיֹחָ֥א אָחִ֖יו הַתִּיצִֽי׃ (לח) אֱלִיאֵל֙ הַֽמַּחֲוִ֔ים וִירִיבַ֥י וְיוֹשַׁוְיָ֖ה בְּנֵ֣י אֶלְנָ֑עַם וְיִתְמָ֖ה הַמּוֹאָבִֽי׃ (לט) אֱלִיאֵ֣ל וְעוֹבֵ֔ד וְיַעֲשִׂיאֵ֖ל הַמְּצֹבָיָֽה׃ (מ) וְאֵ֗לֶּה הַבָּאִ֤ים אֶל־ דָּוִיד֙ לְצִ֣יקְלַ֔ג ע֣וֹד עָצ֔וּר מִפְּנֵ֖י שָׁא֣וּל בֶּן־ קִ֑ישׁ וְהֵ֙מָּה֙ בַּגִּבּוֹרִ֔ים עֹזְרֵ֖י הַמִּלְחָמָֽה׃ (מא) נֹ֣שְׁקֵי קֶ֗שֶׁת מַיְמִינִ֤ים וּמַשְׂמִאלִים֙ בָּֽאֲבָנִ֔ים וּבַחִצִּ֖ים בַּקָּ֑שֶׁת מֵאֲחֵ֥י שָׁא֖וּל מִבִּנְיָמִֽן׃ (מב) הָרֹ֨אשׁ אֲחִיעֶ֜זֶר וְיוֹאָ֗שׁ בְּנֵי֙ הַשְּׁמָעָ֣ה הַגִּבְעָתִ֔י (ויזואל) וִיזִיאֵ֥ל וָפֶ֖לֶט בְּנֵ֣י עַזְמָ֑וֶת וּבְרָכָ֕ה וְיֵה֖וּא הָעֲנְּתֹתִֽי׃ (מג) וְיִֽשְׁמַֽעְיָ֧ה הַגִּבְעוֹנִ֛י גִּבּ֥וֹר בַּשְּׁלֹשִׁ֖ים וְעַל־ הַשְּׁלֹשִֽׁים׃ (מד) וְיִרְמְיָ֤ה וְיַחֲזִיאֵל֙ וְי֣וֹחָנָ֔ן וְיוֹזָבָ֖ד הַגְּדֵרָתִֽי׃ (מה) אֶלְעוּזַ֤י וִירִימוֹת֙ וּבְעַלְיָ֣ה וּשְׁמַרְיָ֔הוּ וּשְׁפַטְיָ֖הוּ (החריפי) הַחֲרוּפִֽי׃ (מו) אֶלְקָנָ֡ה וְ֠יִשִּׁיָּהוּ וַעֲזַרְאֵ֧ל וְיוֹעֶ֛זֶר וְיָשָׁבְעָ֖ם הַקָּרְחִֽים׃ (מז) וְיוֹעֵאלָ֧ה וּזְבַדְיָ֛ה בְּנֵ֥י יְרֹחָ֖ם מִן־ הַגְּדֽוֹר׃ (מח) וּמִן־ הַגָּדִ֡י נִבְדְּל֣וּ אֶל־ דָּוִיד֩ לַמְצַ֨ד מִדְבָּ֜רָה גִּבֹּרֵ֣י הַחַ֗יִל אַנְשֵׁ֤י צָבָא֙ לַמִּלְחָמָ֔ה עֹרְכֵ֥י צִנָּ֖ה וָרֹ֑מַח וּפְנֵ֤י אַרְיֵה֙ פְּנֵיהֶ֔ם וְכִצְבָאיִ֥ם עַל־ הֶהָרִ֖ים לְמַהֵֽר׃ (מט) עֵ֖זֶר הָרֹ֑אשׁ עֹבַדְיָה֙ הַשֵּׁנִ֔י אֱלִיאָ֖ב הַשְּׁלִשִֽׁי׃ (נ) מִשְׁמַנָּה֙ הָרְבִיעִ֔י יִרְמְיָ֖ה הַחֲמִשִֽׁי׃ (נא) עַתַּי֙ הַשִּׁשִּׁ֔י אֱלִיאֵ֖ל הַשְּׁבִעִֽי׃ (נב) יֽוֹחָנָן֙ הַשְּׁמִינִ֔י אֶלְזָבָ֖ד הַתְּשִׁיעִֽי׃ (נג) יִרְמְיָ֙הוּ֙ הָעֲשִׂירִ֔י מַכְבַּנַּ֖י עַשְׁתֵּ֥י עָשָֽׂר׃ (נד) אֵ֥לֶּה מִבְּנֵי־ גָ֖ד רָאשֵׁ֣י הַצָּבָ֑א אֶחָ֤ד לְמֵאָה֙ הַקָּטָ֔ן וְהַגָּד֖וֹל לְאָֽלֶף׃ (נה) אֵ֣לֶּה הֵ֗ם אֲשֶׁ֨ר עָבְר֤וּ אֶת־ הַיַּרְדֵּן֙ בַּחֹ֣דֶשׁ הָרִאשׁ֔וֹן וְה֥וּא מְמַלֵּ֖א עַל־ כָּל־ (גדיתיו) גְּדוֹתָ֑יו וַיַּבְרִ֙יחוּ֙ אֶת־ כָּל־ הָ֣עֲמָקִ֔ים לַמִּזְרָ֖ח וְלַֽמַּעֲרָֽב׃ (נו) וַיָּבֹ֗אוּ מִן־ בְּנֵ֤י בִנְיָמִן֙ וִֽיהוּדָ֔ה עַד־ לַמְצָ֖ד לְדָוִֽיד׃ (נז) וַיֵּצֵ֣א דָוִיד֮ לִפְנֵיהֶם֒ וַיַּ֙עַן֙ וַיֹּ֣אמֶר לָהֶ֔ם אִם־ לְשָׁל֞וֹם בָּאתֶ֤ם אֵלַי֙ לְעָזְרֵ֔נִי יִֽהְיֶה־ לִּ֧י עֲלֵיכֶ֛ם לֵבָ֖ב לְיָ֑חַד וְאִֽם־ לְרַמּוֹתַ֣נִי לְצָרַ֗י בְּלֹ֤א חָמָס֙ בְּכַפַּ֔י יֵ֛רֶא אֱלֹהֵ֥י אֲבוֹתֵ֖ינוּ וְיוֹכַֽח׃ (נח) וְר֣וּחַ לָבְשָׁ֗ה אֶת־ עֲמָשַׂי֮ רֹ֣אשׁ (השלושים) הַשָּׁלִישִׁים֒ לְךָ֤ דָוִיד֙ וְעִמְּךָ֣ בֶן־ יִשַׁ֔י שָׁל֨וֹם ׀ שָׁל֜וֹם לְךָ֗ וְשָׁלוֹם֙ לְעֹ֣זְרֶ֔ךָ כִּ֥י עֲזָרְךָ֖ אֱלֹהֶ֑יךָ וַיְקַבְּלֵ֣ם דָּוִ֔יד וַֽיִּתְּנֵ֖ם בְּרָאשֵׁ֥י הַגְּדֽוּד׃ (נט) וּמִֽמְּנַשֶּׁ֞ה נָפְל֣וּ עַל־ דָּוִ֗יד בְּבֹא֨וֹ עִם־ פְּלִשְׁתִּ֧ים עַל־ שָׁא֛וּל לַמִּלְחָמָ֖ה וְלֹ֣א עֲזָרֻ֑ם כִּ֣י בְעֵצָ֗ה שִׁלְּחֻ֜הוּ סַרְנֵ֤י פְלִשְׁתִּים֙ לֵאמֹ֔ר בְּרָאשֵׁ֕ינוּ יִפּ֖וֹל אֶל־ אֲדֹנָ֥יו שָׁאֽוּל׃ (ס) בְּלֶכְתּ֣וֹ אֶל־ צִֽיקְלַ֗ג נָפְל֣וּ עָלָ֣יו ׀ מִֽמְּנַשֶּׁ֡ה עַ֠דְנַח וְיוֹזָבָ֤ד וִידִֽיעֲאֵל֙ וּמִיכָאֵ֣ל וְיוֹזָבָ֔ד וֶאֱלִיה֖וּא וְצִלְּתָ֑י רָאשֵׁ֥י הָאֲלָפִ֖ים אֲשֶׁ֥ר לִמְנַשֶּֽׁה׃ (סא) וְהֵ֗מָּה עָזְר֤וּ עִם־ דָּוִיד֙ עַֽל־ הַגְּד֔וּד כִּֽי־ גִבּ֥וֹרֵי חַ֖יִל כֻּלָּ֑ם וַיִּהְי֥וּ שָׂרִ֖ים בַּצָּבָֽא׃ (סב) כִּ֚י לְעֶת־ י֣וֹם בְּי֔וֹם יָבֹ֥אוּ עַל־ דָּוִ֖יד לְעָזְר֑וֹ עַד־ לְמַחֲנֶ֥ה גָד֖וֹל כְּמַחֲנֵ֥ה אֱלֹהִֽים׃ (סג) וְ֠אֵלֶּה מִסְפְּרֵ֞י רָאשֵׁ֤י הֶֽחָלוּץ֙ לַצָּבָ֔א בָּ֥אוּ עַל־ דָּוִ֖יד חֶבְר֑וֹנָה לְהָסֵ֞ב מַלְכ֥וּת שָׁא֛וּל אֵלָ֖יו כְּפִ֥י יְהוָֽה׃ (סד) בְּנֵ֣י יְהוּדָ֔ה נֹשְׂאֵ֥י צִנָּ֖ה וָרֹ֑מַח שֵׁ֧שֶׁת אֲלָפִ֛ים וּשְׁמוֹנֶ֥ה מֵא֖וֹת חֲלוּצֵ֥י צָבָֽא׃ (סה) מִן־ בְּנֵ֣י שִׁמְע֗וֹן גִּבּ֤וֹרֵי חַ֙יִל֙ לַצָּבָ֔א שִׁבְעַ֥ת אֲלָפִ֖ים וּמֵאָֽה׃ (סו) מִן־ בְּנֵי֙ הַלֵּוִ֔י אַרְבַּ֥עַת אֲלָפִ֖ים וְשֵׁ֥שׁ מֵאֽוֹת׃ (סז) וִיהוֹיָדָ֖ע הַנָּגִ֣יד לְאַהֲרֹ֑ן וְעִמּ֕וֹ שְׁלֹ֥שֶׁת אֲלָפִ֖ים וּשְׁבַ֥ע מֵאֽוֹת׃ (סח) וְצָד֥וֹק נַ֖עַר גִּבּ֣וֹר חָ֑יִל וּבֵית־ אָבִ֥יו שָׂרִ֖ים עֶשְׂרִ֥ים וּשְׁנָֽיִם׃ (סט) וּמִן־ בְּנֵ֧י בִנְיָמִ֛ן אֲחֵ֥י שָׁא֖וּל שְׁלֹ֣שֶׁת אֲלָפִ֑ים וְעַד־ הֵ֙נָּה֙ מַרְבִּיתָ֔ם שֹׁמְרִ֕ים מִשְׁמֶ֖רֶת בֵּ֥ית שָׁאֽוּל׃ (ע) וּמִן־ בְּנֵ֣י אֶפְרַ֔יִם עֶשְׂרִ֥ים אֶ֖לֶף וּשְׁמוֹנֶ֣ה מֵא֑וֹת גִּבּ֣וֹרֵי חַ֔יִל אַנְשֵׁ֥י שֵׁמ֖וֹת לְבֵ֥ית אֲבוֹתָֽם׃ (עא) וּמֵחֲצִי֙ מַטֵּ֣ה מְנַשֶּׁ֔ה שְׁמוֹנָ֥ה עָשָׂ֖ר אָ֑לֶף אֲשֶׁ֤ר נִקְּבוּ֙ בְּשֵׁמ֔וֹת לָב֖וֹא לְהַמְלִ֥יךְ אֶת־ דָּוִֽיד׃ (עב) וּמִבְּנֵ֣י יִשָּׂשכָ֗ר יוֹדְעֵ֤י בִינָה֙ לַֽעִתִּ֔ים לָדַ֖עַת מַה־ יַּעֲשֶׂ֣ה יִשְׂרָאֵ֑ל רָאשֵׁיהֶ֣ם מָאתַ֔יִם וְכָל־ אֲחֵיהֶ֖ם עַל־ פִּיהֶֽם׃ (עג) מִזְּבֻל֞וּן יוֹצְאֵ֣י צָבָ֗א עֹרְכֵ֧י מִלְחָמָ֛ה בְּכָל־ כְּלֵ֥י מִלְחָמָ֖ה חֲמִשִּׁ֣ים אָ֑לֶף וְלַעֲדֹ֖ר בְּלֹא־ לֵ֥ב וָלֵֽב׃ (עד) וּמִנַּפְתָּלִ֖י שָׂרִ֣ים אָ֑לֶף וְעִמָּהֶם֙ בְּצִנָּ֣ה וַחֲנִ֔ית שְׁלֹשִׁ֥ים וְשִׁבְעָ֖ה אָֽלֶף׃ (עה) וּמִן־ הַדָּנִי֙ עֹרְכֵ֣י מִלְחָמָ֔ה עֶשְׂרִֽים־ וּשְׁמוֹנָ֥ה אֶ֖לֶף וְשֵׁ֥שׁ מֵאֽוֹת׃ (עו) וּמֵאָשֵׁ֗ר יוֹצְאֵ֥י צָבָ֛א לַעֲרֹ֥ךְ מִלְחָמָ֖ה אַרְבָּעִ֥ים אָֽלֶף׃ (עז) וּמֵעֵ֣בֶר לַ֠יַּרְדֵּן מִן־ הָראוּבֵנִ֨י וְהַגָּדִ֜י וַחֲצִ֣י ׀ שֵׁ֣בֶט מְנַשֶּׁ֗ה בְּכֹל֙ כְּלֵי֙ צְבָ֣א מִלְחָמָ֔ה מֵאָ֥ה וְעֶשְׂרִ֖ים אָֽלֶף׃ (עח) כָּל־ אֵ֜לֶּה אַנְשֵׁ֣י מִלְחָמָה֮ עֹדְרֵ֣י מַעֲרָכָה֒ בְּלֵבָ֤ב שָׁלֵם֙ בָּ֣אוּ חֶבְר֔וֹנָה לְהַמְלִ֥יךְ אֶת־ דָּוִ֖יד עַל־ כָּל־ יִשְׂרָאֵ֑ל וְ֠גַם כָּל־ שֵׁרִ֧ית יִשְׂרָאֵ֛ל לֵ֥ב אֶחָ֖ד לְהַמְלִ֥יךְ אֶת־ דָּוִֽיד׃ (עט) וַיִּהְיוּ־ שָׁ֤ם עִם־ דָּוִיד֙ יָמִ֣ים שְׁלוֹשָׁ֔ה אֹכְלִ֖ים וְשׁוֹתִ֑ים כִּֽי־ הֵכִ֥ינוּ לָהֶ֖ם אֲחֵיהֶֽם׃
