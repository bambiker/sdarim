(א) עַתָּ֗ה תְּנ֤וּ לְבַבְכֶם֙ וְנַפְשְׁכֶ֔ם לִדְר֖וֹשׁ לַיהוָ֣ה אֱלֹהֵיכֶ֑ם וְק֗וּמוּ וּבְנוּ֙ אֶת־ מִקְדַּשׁ֙ יְהוָ֣ה הָֽאֱלֹהִ֔ים לְהָבִ֞יא אֶת־ אֲר֣וֹן בְּרִית־ יְהוָ֗ה וּכְלֵי֙ קֹ֣דֶשׁ הָֽאֱלֹהִ֔ים לַבַּ֖יִת הַנִּבְנֶ֥ה לְשֵׁם־ יְהוָֽה׃ (ב) וְדָוִ֥יד זָקֵ֖ן וְשָׂבַ֣ע יָמִ֑ים וַיַּמְלֵ֛ךְ אֶת־ שְׁלֹמֹ֥ה בְנ֖וֹ עַל־ יִשְׂרָאֵֽל׃ (ג) וַיֶּאֱסֹף֙ אֶת־ כָּל־ שָׂרֵ֣י יִשְׂרָאֵ֔ל וְהַכֹּהֲנִ֖ים וְהַלְוִיִּֽם׃ (ד) וַיִּסָּֽפְרוּ֙ הַלְוִיִּ֔ם מִבֶּ֛ן שְׁלֹשִׁ֥ים שָׁנָ֖ה וָמָ֑עְלָה וַיְהִ֨י מִסְפָּרָ֤ם לְגֻלְגְּלֹתָם֙ לִגְבָרִ֔ים שְׁלֹשִׁ֥ים וּשְׁמוֹנָ֖ה אָֽלֶף׃ (ה) מֵאֵ֗לֶּה לְנַצֵּ֙חַ֙ עַל־ מְלֶ֣אכֶת בֵּית־ יְהוָ֔ה עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָ֑לֶף וְשֹׁטְרִ֥ים וְשֹׁפְטִ֖ים שֵׁ֥שֶׁת אֲלָפִֽים׃ (ו) וְאַרְבַּ֥עַת אֲלָפִ֖ים שֹׁעֲרִ֑ים וְאַרְבַּ֤עַת אֲלָפִים֙ מְהַֽלְלִ֣ים לַיהוָ֔ה בַּכֵּלִ֕ים אֲשֶׁ֥ר עָשִׂ֖יתִי לְהַלֵּֽל׃ (ז) וַיֶּֽחָלְקֵ֥ם דָּוִ֖יד מַחְלְק֑וֹת לִבְנֵ֣י לֵוִ֔י לְגֵרְשׁ֖וֹן קְהָ֥ת וּמְרָרִֽי׃ (ח) לַגֵּרְשֻׁנִּ֖י לַעְדָּ֥ן וְשִׁמְעִֽי׃ (ט) בְּנֵ֣י לַעְדָּ֗ן הָרֹ֧אשׁ יְחִיאֵ֛ל וְזֵתָ֥ם וְיוֹאֵ֖ל שְׁלֹשָֽׁה׃ (י) בְּנֵ֣י שִׁמְעִ֗י (שלמות) שְׁלוֹמִ֧ית וַחֲזִיאֵ֛ל וְהָרָ֖ן שְׁלֹשָׁ֑ה אֵ֛לֶּה רָאשֵׁ֥י הָאָב֖וֹת לְלַעְדָּֽן׃ (יא) וּבְנֵ֣י שִׁמְעִ֔י יַ֣חַת זִינָ֔א וִיע֖וּשׁ וּבְרִיעָ֑ה אֵ֥לֶּה בְנֵי־ שִׁמְעִ֖י אַרְבָּעָֽה׃ (יב) וַֽיְהִי־ יַ֣חַת הָרֹ֔אשׁ וְזִיזָ֖ה הַשֵּׁנִ֑י וִיע֤וּשׁ וּבְרִיעָה֙ לֹֽא־ הִרְבּ֣וּ בָנִ֔ים וַיִּֽהְיוּ֙ לְבֵ֣ית אָ֔ב לִפְקֻדָּ֖ה אֶחָֽת׃ (יג) בְּנֵ֣י קְהָ֗ת עַמְרָ֥ם יִצְהָ֛ר חֶבְר֥וֹן וְעֻזִּיאֵ֖ל אַרְבָּעָֽה׃ (יד) בְּנֵ֥י עַמְרָ֖ם אַהֲרֹ֣ן וּמֹשֶׁ֑ה וַיִּבָּדֵ֣ל אַהֲרֹ֡ן לְֽהַקְדִּישׁוֹ֩ קֹ֨דֶשׁ קָֽדָשִׁ֤ים הֽוּא־ וּבָנָיו֙ עַד־ עוֹלָ֔ם לְהַקְטִיר֩ לִפְנֵ֨י יְהוָ֧ה לְשָׁרְת֛וֹ וּלְבָרֵ֥ךְ בִּשְׁמ֖וֹ עַד־ עוֹלָֽם׃ (טו) וּמֹשֶׁ֖ה אִ֣ישׁ הָאֱלֹהִ֑ים בָּנָ֕יו יִקָּרְא֖וּ עַל־ שֵׁ֥בֶט הַלֵּוִֽי׃ (טז) בְּנֵ֣י מֹשֶׁ֔ה גֵּרְשֹׁ֖ם וֶאֱלִיעֶֽזֶר׃ (יז) בְּנֵ֥י גֵרְשׁ֖וֹם שְׁבוּאֵ֥ל הָרֹֽאשׁ׃ (יח) וַיִּֽהְי֥וּ בְנֵי־ אֱלִיעֶ֖זֶר רְחַבְיָ֣ה הָרֹ֑אשׁ וְלֹא־ הָיָ֤ה לֶאֱלִיעֶ֙זֶר֙ בָּנִ֣ים אֲחֵרִ֔ים וּבְנֵ֥י רְחַבְיָ֖ה רָב֥וּ לְמָֽעְלָה׃ (יט) בְּנֵ֥י יִצְהָ֖ר שְׁלֹמִ֥ית הָרֹֽאשׁ׃ (כ) בְּנֵ֖י חֶבְר֑וֹן יְרִיָּ֤הוּ הָרֹאשׁ֙ אֲמַרְיָ֣ה הַשֵּׁנִ֔י יַחֲזִיאֵל֙ הַשְּׁלִישִׁ֔י וִֽיקַמְעָ֖ם הָרְבִיעִֽי׃ (כא) בְּנֵ֖י עֻזִּיאֵ֑ל מִיכָ֣ה הָרֹ֔אשׁ וְיִשִּׁיָּ֖ה הַשֵּׁנִֽי׃ (כב) בְּנֵ֤י מְרָרִי֙ מַחְלִ֣י וּמוּשִׁ֔י בְּנֵ֥י מַחְלִ֖י אֶלְעָזָ֥ר וְקִֽישׁ׃ (כג) וַיָּ֙מָת֙ אֶלְעָזָ֔ר וְלֹא־ הָ֥יוּ ל֛וֹ בָּנִ֖ים כִּ֣י אִם־ בָּנ֑וֹת וַיִּשָּׂא֥וּם בְּנֵי־ קִ֖ישׁ אֲחֵיהֶֽם׃ (כד) בְּנֵ֣י מוּשִׁ֗י מַחְלִ֥י וְעֵ֛דֶר וִירֵמ֖וֹת שְׁלֹשָֽׁה׃ (כה) אֵ֣לֶּה בְנֵֽי־ לֵוִי֩ לְבֵ֨ית אֲבֹתֵיהֶ֜ם רָאשֵׁ֧י הָאָב֣וֹת לִפְקוּדֵיהֶ֗ם בְּמִסְפַּ֤ר שֵׁמוֹת֙ לְגֻלְגְּלֹתָ֔ם עֹשֵׂה֙ הַמְּלָאכָ֔ה לַעֲבֹדַ֖ת בֵּ֣ית יְהוָ֑ה מִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וָמָֽעְלָה׃ (כו) כִּ֚י אָמַ֣ר דָּוִ֔יד הֵנִ֛יחַ יְהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל לְעַמּ֑וֹ וַיִּשְׁכֹּ֥ן בִּירוּשָׁלִַ֖ם עַד־ לְעוֹלָֽם׃ (כז) וְגַ֖ם לַלְוִיִּ֑ם אֵין־ לָשֵׂ֧את אֶת־ הַמִּשְׁכָּ֛ן וְאֶת־ כָּל־ כֵּלָ֖יו לַעֲבֹדָתֽוֹ׃ (כח) כִּ֣י בְדִבְרֵ֤י דָוִיד֙ הָאַ֣חֲרֹנִ֔ים הֵ֖מָּה מִסְפַּ֣ר בְּנֵי־ לֵוִ֑י מִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וּלְמָֽעְלָה׃ (כט) כִּ֣י מַעֲמָדָ֞ם לְיַד־ בְּנֵ֣י אַהֲרֹ֗ן לַעֲבֹדַת֙ בֵּ֣ית יְהוָ֔ה עַל־ הַחֲצֵרוֹת֙ וְעַל־ הַלְּשָׁכ֔וֹת וְעַֽל־ טָהֳרַ֖ת לְכָל־ קֹ֑דֶשׁ וּמַֽעֲשֵׂ֔ה עֲבֹדַ֖ת בֵּ֥ית הָאֱלֹהִֽים׃ (ל) וּלְלֶ֨חֶם הַֽמַּעֲרֶ֜כֶת וּלְסֹ֤לֶת לְמִנְחָה֙ וְלִרְקִיקֵ֣י הַמַּצּ֔וֹת וְלַֽמַּחֲבַ֖ת וְלַמֻּרְבָּ֑כֶת וּלְכָל־ מְשׂוּרָ֖ה וּמִדָּֽה׃ (לא) וְלַעֲמֹד֙ בַּבֹּ֣קֶר בַּבֹּ֔קֶר לְהֹד֥וֹת וּלְהַלֵּ֖ל לַיהוָ֑ה וְכֵ֖ן לָעָֽרֶב׃ (לב) וּלְכֹ֨ל הַעֲל֤וֹת עֹלוֹת֙ לַיהוָ֔ה לַשַּׁבָּת֔וֹת לֶחֳדָשִׁ֖ים וְלַמֹּעֲדִ֑ים בְּמִסְפָּ֨ר כְּמִשְׁפָּ֧ט עֲלֵיהֶ֛ם תָּמִ֖יד לִפְנֵ֥י יְהוָֽה׃ (לג) וְשָׁמְר֞וּ אֶת־ מִשְׁמֶ֣רֶת אֹֽהֶל־ מוֹעֵ֗ד וְאֵת֙ מִשְׁמֶ֣רֶת הַקֹּ֔דֶשׁ וּמִשְׁמֶ֕רֶת בְּנֵ֥י אַהֲרֹ֖ן אֲחֵיהֶ֑ם לַעֲבֹדַ֖ת בֵּ֥ית יְהוָֽה׃ (לד) וְלִבְנֵ֥י אַהֲרֹ֖ן מַחְלְקוֹתָ֑ם בְּנֵ֣י אַהֲרֹ֔ן נָדָב֙ וַאֲבִיה֔וּא אֶלְעָזָ֖ר וְאִיתָמָֽר׃ (לה) וַיָּ֨מָת נָדָ֤ב וַאֲבִיהוּא֙ לִפְנֵ֣י אֲבִיהֶ֔ם וּבָנִ֖ים לֹא־ הָי֣וּ לָהֶ֑ם וַֽיְכַהֲנ֔וּ אֶלְעָזָ֖ר וְאִיתָמָֽר׃ (לו) וַיֶּֽחָלְקֵ֣ם דָּוִ֔יד וְצָדוֹק֙ מִן־ בְּנֵ֣י אֶלְעָזָ֔ר וַאֲחִימֶ֖לֶךְ מִן־ בְּנֵ֣י אִיתָמָ֑ר לִפְקֻדָּתָ֖ם בַּעֲבֹדָתָֽם׃ (לז) וַיִּמָּצְא֣וּ בְֽנֵי־ אֶלְעָזָ֡ר רַבִּ֞ים לְרָאשֵׁ֧י הַגְּבָרִ֛ים מִן־ בְּנֵ֥י אִיתָמָ֖ר וַֽיַּחְלְק֑וּם לִבְנֵ֨י אֶלְעָזָ֜ר רָאשִׁ֤ים לְבֵית־ אָבוֹת֙ שִׁשָּׁ֣ה עָשָׂ֔ר וְלִבְנֵ֧י אִיתָמָ֛ר לְבֵ֥ית אֲבוֹתָ֖ם שְׁמוֹנָֽה׃ (לח) וַיַּחְלְק֥וּם בְּגוֹרָל֖וֹת אֵ֣לֶּה עִם־ אֵ֑לֶּה כִּי־ הָי֤וּ שָֽׂרֵי־ קֹ֙דֶשׁ֙ וְשָׂרֵ֣י הָאֱלֹהִ֔ים מִבְּנֵ֥י אֶלְעָזָ֖ר וּבִבְנֵ֥י אִיתָמָֽר׃ (לט) וַֽיִּכְתְּבֵ֡ם שְֽׁמַֽעְיָה֩ בֶן־ נְתַנְאֵ֨ל הַסּוֹפֵ֜ר מִן־ הַלֵּוִ֗י לִפְנֵ֨י הַמֶּ֤לֶךְ וְהַשָּׂרִים֙ וְצָד֣וֹק הַכֹּהֵ֗ן וַאֲחִימֶ֙לֶךְ֙ בֶּן־ אֶבְיָתָ֔ר וְרָאשֵׁי֙ הָֽאָב֔וֹת לַכֹּהֲנִ֖ים וְלַלְוִיִּ֑ם בֵּֽית־ אָ֣ב אֶחָ֗ד אָחֻז֙ לְאֶלְעָזָ֔ר וְאָחֻ֥ז ׀ אָחֻ֖ז לְאִיתָמָֽר׃ (מ) וַיֵּצֵ֞א הַגּוֹרָ֤ל הָרִאשׁוֹן֙ לִיה֣וֹיָרִ֔יב לִֽידַעְיָ֖ה הַשֵּׁנִֽי׃ (מא) לְחָרִם֙ הַשְּׁלִישִׁ֔י לִשְׂעֹרִ֖ים הָרְבִעִֽי׃ (מב) לְמַלְכִּיָּה֙ הַחֲמִישִׁ֔י לְמִיָּמִ֖ן הַשִּׁשִּֽׁי׃ (מג) לְהַקּוֹץ֙ הַשְּׁבִעִ֔י לַאֲבִיָּ֖ה הַשְּׁמִינִֽי׃ (מד) לְיֵשׁ֙וּעַ֙ הַתְּשִׁעִ֔י לִשְׁכַנְיָ֖הוּ הָעֲשִׂרִֽי׃ (מה) לְאֶלְיָשִׁיב֙ עַשְׁתֵּ֣י עָשָׂ֔ר לְיָקִ֖ים שְׁנֵ֥ים עָשָֽׂר׃ (מו) לְחֻפָּה֙ שְׁלֹשָׁ֣ה עָשָׂ֔ר לְיֶֽשֶׁבְאָ֖ב אַרְבָּעָ֥ה עָשָֽׂר׃ (מז) לְבִלְגָּה֙ חֲמִשָּׁ֣ה עָשָׂ֔ר לְאִמֵּ֖ר שִׁשָּׁ֥ה עָשָֽׂר׃ (מח) לְחֵזִיר֙ שִׁבְעָ֣ה עָשָׂ֔ר לְהַפִּצֵּ֖ץ שְׁמוֹנָ֥ה עָשָֽׂר׃ (מט) לִֽפְתַחְיָה֙ תִּשְׁעָ֣ה עָשָׂ֔ר לִֽיחֶזְקֵ֖אל הָעֶשְׂרִֽים׃ (נ) לְיָכִין֙ אֶחָ֣ד וְעֶשְׂרִ֔ים לְגָמ֖וּל שְׁנַ֥יִם וְעֶשְׂרִֽים׃ (נא) לִדְלָיָ֙הוּ֙ שְׁלֹשָׁ֣ה וְעֶשְׂרִ֔ים לְמַֽעַזְיָ֖הוּ אַרְבָּעָ֥ה וְעֶשְׂרִֽים׃ (נב) אֵ֣לֶּה פְקֻדָּתָ֞ם לַעֲבֹדָתָ֗ם לָב֤וֹא לְבֵית־ יְהוָה֙ כְּמִשְׁפָּטָ֔ם בְּיַ֖ד אַהֲרֹ֣ן אֲבִיהֶ֑ם כַּאֲשֶׁ֣ר צִוָּ֔הוּ יְהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (נג) וְלִבְנֵ֥י לֵוִ֖י הַנּוֹתָרִ֑ים לִבְנֵ֤י עַמְרָם֙ שֽׁוּבָאֵ֔ל לִבְנֵ֥י שׁוּבָאֵ֖ל יֶחְדְּיָֽהוּ׃ (נד) לִרְחַבְיָ֑הוּ לִבְנֵ֣י רְחַבְיָ֔הוּ הָרֹ֖אשׁ יִשִּׁיָּֽה׃ (נה) לַיִּצְהָרִ֣י שְׁלֹמ֔וֹת לִבְנֵ֥י שְׁלֹמ֖וֹת יָֽחַת׃ (נו) וּבְנָ֖י יְרִיָּ֑הוּ אֲמַרְיָ֙הוּ֙ הַשֵּׁנִ֔י יַחֲזִיאֵל֙ הַשְּׁלִישִׁ֔י יְקַמְעָ֖ם הָרְבִיעִֽי׃ (נז) בְּנֵ֤י עֻזִּיאֵל֙ מִיכָ֔ה לִבְנֵ֥י מִיכָ֖ה (שמור) שָׁמִֽיר׃ (נח) אֲחִ֤י מִיכָה֙ יִשִּׁיָּ֔ה לִבְנֵ֥י יִשִּׁיָּ֖ה זְכַרְיָֽהוּ׃ (נט) בְּנֵ֤י מְרָרִי֙ מַחְלִ֣י וּמוּשִׁ֔י בְּנֵ֖י יַעֲזִיָּ֥הֽוּ בְנֽוֹ׃ (ס) בְּנֵ֖י מְרָרִ֑י לְיַֽעֲזִיָּ֣הֽוּ בְנ֔וֹ וְשֹׁ֥הַם וְזַכּ֖וּר וְעִבְרִֽי׃ (סא) לְמַחְלִי֙ אֶלְעָזָ֔ר וְלֹא־ הָ֥יָה ל֖וֹ בָּנִֽים׃ (סב) לְקִ֖ישׁ בְּנֵי־ קִ֥ישׁ יְרַחְמְאֵֽל׃ (סג) וּבְנֵ֣י מוּשִׁ֔י מַחְלִ֥י וְעֵ֖דֶר וִירִימ֑וֹת אֵ֛לֶּה בְּנֵ֥י הַלְוִיִּ֖ם לְבֵ֥ית אֲבֹתֵיהֶֽם׃ (סד) וַיַּפִּילוּ֩ גַם־ הֵ֨ם גּוֹרָל֜וֹת לְעֻמַּ֣ת ׀ אֲחֵיהֶ֣ם בְּנֵֽי־ אַהֲרֹ֗ן לִפְנֵ֨י דָוִ֤יד הַמֶּ֙לֶךְ֙ וְצָד֣וֹק וַאֲחִימֶ֔לֶךְ וְרָאשֵׁי֙ הָֽאָב֔וֹת לַכֹּהֲנִ֖ים וְלַלְוִיִּ֑ם אָב֣וֹת הָרֹ֔אשׁ לְעֻמַּ֖ת אָחִ֥יו הַקָּטָֽן׃ (סה) וַיַּבְדֵּ֣ל דָּוִיד֩ וְשָׂרֵ֨י הַצָּבָ֜א לַעֲבֹדָ֗ה לִבְנֵ֤י אָסָף֙ וְהֵימָ֣ן וִֽידוּת֔וּן (הנביאים) הַֽנִּבְּאִ֛ים בְּכִנֹּר֥וֹת בִּנְבָלִ֖ים וּבִמְצִלְתָּ֑יִם וַֽיְהִי֙ מִסְפָּרָ֔ם אַנְשֵׁ֥י מְלָאכָ֖ה לַעֲבֹדָתָֽם׃ (סו) לִבְנֵ֣י אָסָ֗ף זַכּ֧וּר וְיוֹסֵ֛ף וּנְתַנְיָ֥ה וַאֲשַׂרְאֵ֖לָה בְּנֵ֣י אָסָ֑ף עַ֚ל יַד־ אָסָ֔ף הַנִּבָּ֖א עַל־ יְדֵ֥י הַמֶּֽלֶךְ׃ (סז) לִידוּת֑וּן בְּנֵ֣י יְדוּת֡וּן גְּדַלְיָ֡הוּ וּצְרִ֡י וִֽ֠ישַׁעְיָהוּ חֲשַׁבְיָ֨הוּ וּמַתִּתְיָ֜הוּ שִׁשָּׁ֗ה עַל֩ יְדֵ֨י אֲבִיהֶ֤ם יְדוּתוּן֙ בַּכִּנּ֔וֹר הַנִּבָּ֕א עַל־ הֹד֥וֹת וְהַלֵּ֖ל לַיהוָֽה׃ (סח) לְהֵימָ֑ן בְּנֵ֣י הֵימָ֡ן בֻּקִּיָּ֡הוּ מַתַּנְיָ֡הוּ עֻ֠זִּיאֵל שְׁבוּאֵ֨ל וִֽירִימ֜וֹת חֲנַנְיָ֣ה חֲנָ֗נִי אֱלִיאָ֤תָה גִדַּ֙לְתִּי֙ וְרֹמַ֣מְתִּי עֶ֔זֶר יָשְׁבְּקָ֣שָׁה מַלּ֔וֹתִי הוֹתִ֖יר מַחֲזִיאֽוֹת׃ (סט) כָּל־ אֵ֨לֶּה בָנִ֜ים לְהֵימָ֗ן חֹזֵ֥ה הַמֶּ֛לֶךְ בְּדִבְרֵ֥י הָאֱלֹהִ֖ים לְהָרִ֣ים קָ֑רֶן וַיִּתֵּ֨ן הָאֱלֹהִ֜ים לְהֵימָ֗ן בָּנִ֛ים אַרְבָּעָ֥ה עָשָׂ֖ר וּבָנ֥וֹת שָׁלֽוֹשׁ׃ (ע) כָּל־ אֵ֣לֶּה עַל־ יְדֵי֩ אֲבִיהֶ֨ם בַּשִּׁ֜יר בֵּ֣ית יְהוָ֗ה בִּמְצִלְתַּ֙יִם֙ נְבָלִ֣ים וְכִנֹּר֔וֹת לַעֲבֹדַ֖ת בֵּ֣ית הָאֱלֹהִ֑ים עַ֚ל יְדֵ֣י הַמֶּ֔לֶךְ אָסָ֥ף וִידוּת֖וּן וְהֵימָֽן׃ (עא) וַיְהִ֤י מִסְפָּרָם֙ עִם־ אֲחֵיהֶ֔ם מְלֻמְּדֵי־ שִׁ֖יר לַיהוָ֑ה כָּל־ הַ֨מֵּבִ֔ין מָאתַ֖יִם שְׁמוֹנִ֥ים וּשְׁמוֹנָֽה׃ (עב) וַיַּפִּ֜ילוּ גּוֹרָל֣וֹת מִשְׁמֶ֗רֶת לְעֻמַּת֙ כַּקָּטֹ֣ן כַּגָּד֔וֹל מֵבִ֖ין עִם־ תַּלְמִֽיד׃ (עג) וַיֵּצֵ֞א הַגּוֹרָ֧ל הָרִאשׁ֛וֹן לְאָסָ֖ף לְיוֹסֵ֑ף גְּדַלְיָ֙הוּ֙ הַשֵּׁנִ֔י הֽוּא־ וְאֶחָ֥יו וּבָנָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עד) הַשְּׁלִשִׁ֣י זַכּ֔וּר בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עה) הָרְבִיעִי֙ לַיִּצְרִ֔י בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עו) הַחֲמִישִׁ֣י נְתַנְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עז) הַשִּׁשִּׁ֣י בֻקִּיָּ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עח) הַשְּׁבִעִ֣י יְשַׂרְאֵ֔לָה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (עט) הַשְּׁמִינִ֣י יְשַֽׁעְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פ) הַתְּשִׁיעִ֣י מַתַּנְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פא) הָעֲשִׂירִ֣י שִׁמְעִ֔י בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פב) עַשְׁתֵּֽי־ עָשָׂ֣ר עֲזַרְאֵ֔ל בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פג) הַשְּׁנֵ֤ים עָשָׂר֙ לַחֲשַׁבְיָ֔ה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פד) לִשְׁלֹשָׁ֤ה עָשָׂר֙ שֽׁוּבָאֵ֔ל בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פה) לְאַרְבָּעָ֤ה עָשָׂר֙ מַתִּתְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פו) לַחֲמִשָּׁ֤ה עָשָׂר֙ לִֽירֵמ֔וֹת בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פז) לְשִׁשָּׁ֤ה עָשָׂר֙ לַחֲנַנְיָ֔הוּ בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פח) לְשִׁבְעָ֤ה עָשָׂר֙ לְיָשְׁבְּקָ֔שָׁה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (פט) לִשְׁמוֹנָ֤ה עָשָׂר֙ לַחֲנָ֔נִי בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צ) לְתִשְׁעָ֤ה עָשָׂר֙ לְמַלּ֔וֹתִי בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צא) לְעֶשְׂרִים֙ לֶֽאֱלִיָּ֔תָה בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צב) לְאֶחָ֤ד וְעֶשְׂרִים֙ לְהוֹתִ֔יר בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צג) לִשְׁנַ֤יִם וְעֶשְׂרִים֙ לְגִדַּ֔לְתִּי בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צד) לִשְׁלֹשָׁ֤ה וְעֶשְׂרִים֙ לְמַ֣חֲזִיא֔וֹת בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צה) לְאַרְבָּעָ֤ה וְעֶשְׂרִים֙ לְרוֹמַ֣מְתִּי עָ֔זֶר בָּנָ֥יו וְאֶחָ֖יו שְׁנֵ֥ים עָשָֽׂר׃ (צו) לְמַחְלְק֖וֹת לְשֹׁעֲרִ֑ים לַקָּרְחִ֕ים מְשֶֽׁלֶמְיָ֥הוּ בֶן־ קֹרֵ֖א מִן־ בְּנֵ֥י אָסָֽף׃ (צז) וְלִמְשֶֽׁלֶמְיָ֖הוּ בָּנִ֑ים זְכַרְיָ֤הוּ הַבְּכוֹר֙ יְדִֽיעֲאֵ֣ל הַשֵּׁנִ֔י זְבַדְיָ֙הוּ֙ הַשְּׁלִישִׁ֔י יַתְנִיאֵ֖ל הָרְבִיעִֽי׃ (צח) עֵילָ֤ם הַחֲמִישִׁי֙ יְהוֹחָנָ֣ן הַשִּׁשִּׁ֔י אֶלְיְהוֹעֵינַ֖י הַשְּׁבִיעִֽי׃ (צט) וּלְעֹבֵ֥ד אֱדֹ֖ם בָּנִ֑ים שְׁמַֽעְיָ֤ה הַבְּכוֹר֙ יְהוֹזָבָ֣ד הַשֵּׁנִ֔י יוֹאָ֤ח הַשְּׁלִשִׁי֙ וְשָׂכָ֣ר הָרְבִיעִ֔י וּנְתַנְאֵ֖ל הַחֲמִישִֽׁי׃
