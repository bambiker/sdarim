דברי הימים סדר כג

פתיחה באתר · ספר דברי הימים כולו ·

(א) וַיֹּ֣אמֶר אֵלָ֗יו עֲזַרְיָ֧הוּ הַכֹּהֵ֛ן הָרֹ֖אשׁ לְבֵ֣ית צָד֑וֹק וַ֠יֹּאמֶר מֵהָחֵ֨ל הַתְּרוּמָ֜ה לָבִ֣יא בֵית־ יְהוָ֗ה אָכ֨וֹל וְשָׂב֤וֹעַ וְהוֹתֵר֙ עַד־ לָר֔וֹב כִּ֤י יְהוָה֙ בֵּרַ֣ךְ אֶת־ עַמּ֔וֹ וְהַנּוֹתָ֖ר אֶת־ הֶהָמ֥וֹן הַזֶּֽה׃ (ב) וַיֹּ֣אמֶר יְחִזְקִיָּ֗הוּ לְהָכִ֧ין לְשָׁכ֛וֹת בְּבֵ֥ית יְהוָ֖ה וַיָּכִֽינוּ׃ (ג) וַיָּבִ֨יאוּ אֶת־ הַתְּרוּמָ֧ה וְהַֽמַּעֲשֵׂ֛ר וְהַקֳּדָשִׁ֖ים בֶּאֱמוּנָ֑ה וַעֲלֵיהֶ֤ם נָגִיד֙ (כונניהו) כָּֽנַנְיָ֣הוּ הַלֵּוִ֔י וְשִׁמְעִ֥י אָחִ֖יהוּ מִשְׁנֶֽה׃ (ד) וִֽיחִיאֵ֡ל וַ֠עֲזַזְיָהוּ וְנַ֨חַת וַעֲשָׂהאֵ֜ל וִֽירִימ֤וֹת וְיוֹזָבָד֙ וֶאֱלִיאֵ֣ל וְיִסְמַכְיָ֔הוּ וּמַ֖חַת וּבְנָיָ֑הוּ פְּקִידִ֗ים מִיַּ֤ד (כונניהו) כָּֽנַנְיָ֙הוּ֙ וְשִׁמְעִ֣י אָחִ֔יו בְּמִפְקַד֙ יְחִזְקִיָּ֣הוּ הַמֶּ֔לֶךְ וַעֲזַרְיָ֖הוּ נְגִ֥יד בֵּית־ הָאֱלֹהִֽים׃ (ה) וְקוֹרֵ֨א בֶן־ יִמְנָ֤ה הַלֵּוִי֙ הַשּׁוֹעֵ֣ר לַמִּזְרָ֔חָה עַ֖ל נִדְב֣וֹת הָאֱלֹהִ֑ים לָתֵת֙ תְּרוּמַ֣ת יְהוָ֔ה וְקָדְשֵׁ֖י הַקֳּדָשִֽׁים׃ (ו) וְעַל־ יָד֡וֹ עֵ֣דֶן וּ֠מִנְיָמִן וְיֵשׁ֨וּעַ וּֽשְׁמַֽעְיָ֜הוּ אֲמַרְיָ֧הוּ וּשְׁכַנְיָ֛הוּ בְּעָרֵ֥י הַכֹּהֲנִ֖ים בֶּאֱמוּנָ֑ה לָתֵ֤ת לַאֲחֵיהֶם֙ בְּמַחְלְק֔וֹת כַּגָּד֖וֹל כַּקָּטָֽן׃ (ז) מִלְּבַ֞ד הִתְיַחְשָׂ֣ם לִזְכָרִ֗ים מִבֶּ֨ן שָׁל֤וֹשׁ שָׁנִים֙ וּלְמַ֔עְלָה לְכָל־ הַבָּ֥א לְבֵית־ יְהוָ֖ה לִדְבַר־ י֣וֹם בְּיוֹמ֑וֹ לַעֲב֣וֹדָתָ֔ם בְּמִשְׁמְרוֹתָ֖ם כְּמַחְלְקוֹתֵיהֶֽם׃ (ח) וְאֵ֨ת הִתְיַחֵ֤שׂ הַכֹּהֲנִים֙ לְבֵ֣ית אֲבוֹתֵיהֶ֔ם וְהַ֨לְוִיִּ֔ם מִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וּלְמָ֑עְלָה בְּמִשְׁמְרוֹתֵיהֶ֖ם בְּמַחְלְקוֹתֵיהֶֽם׃ (ט) וּלְהִתְיַחֵ֗שׂ בְּכָל־ טַפָּ֧ם נְשֵׁיהֶ֛ם וּבְנֵיהֶ֥ם וּבְנוֹתֵיהֶ֖ם לְכָל־ קָהָ֑ל כִּ֥י בֶאֱמוּנָתָ֖ם יִתְקַדְּשׁוּ־ קֹֽדֶשׁ׃ (י) וְלִבְנֵי֩ אַהֲרֹ֨ן הַכֹּהֲנִ֜ים בִּשְׂדֵ֨י מִגְרַ֤שׁ עָרֵיהֶם֙ בְּכָל־ עִ֣יר וָעִ֔יר אֲנָשִׁ֕ים אֲשֶׁ֥ר נִקְּב֖וּ בְּשֵׁמ֑וֹת לָתֵ֣ת מָנ֗וֹת לְכָל־ זָכָר֙ בַּכֹּ֣הֲנִ֔ים וּלְכָל־ הִתְיַחֵ֖שׂ בַּלְוִיִּֽם׃ (יא) וַיַּ֧עַשׂ כָּזֹ֛את יְחִזְקִיָּ֖הוּ בְּכָל־ יְהוּדָ֑ה וַיַּ֨עַשׂ הַטּ֤וֹב וְהַיָּשָׁר֙ וְהָ֣אֱמֶ֔ת לִפְנֵ֖י יְהוָ֥ה אֱלֹהָֽיו׃ (יב) וּבְכָֽל־ מַעֲשֶׂ֞ה אֲשֶׁר־ הֵחֵ֣ל ׀ בַּעֲבוֹדַ֣ת בֵּית־ הָאֱלֹהִ֗ים וּבַתּוֹרָה֙ וּבַמִּצְוָ֔ה לִדְרֹ֖שׁ לֵֽאלֹהָ֑יו בְּכָל־ לְבָב֥וֹ עָשָׂ֖ה וְהִצְלִֽיחַ׃ (יג) אַחֲרֵ֨י הַדְּבָרִ֤ים וְהָאֱמֶת֙ הָאֵ֔לֶּה בָּ֖א סַנְחֵרִ֣יב מֶֽלֶךְ־ אַשּׁ֑וּר וַיָּבֹ֣א בִֽיהוּדָ֗ה וַיִּ֙חַן֙ עַל־ הֶעָרִ֣ים הַבְּצֻר֔וֹת וַיֹּ֖אמֶר לְבִקְעָ֥ם אֵלָֽיו׃ (יד) וַיַּרְא֙ יְחִזְקִיָּ֔הוּ כִּי־ בָ֖א סַנְחֵרִ֑יב וּפָנָ֕יו לַמִּלְחָמָ֖ה עַל־ יְרוּשָׁלִָֽם׃ (טו) וַיִּוָּעַ֗ץ עִם־ שָׂרָיו֙ וְגִבֹּרָ֔יו לִסְתּוֹם֙ אֶת־ מֵימֵ֣י הָעֲיָנ֔וֹת אֲשֶׁ֖ר מִח֣וּץ לָעִ֑יר וַֽיַּעְזְרֽוּהוּ׃ (טז) וַיִּקָּבְצ֣וּ עַם־ רָ֔ב וַֽיִּסְתְּמוּ֙ אֶת־ כָּל־ הַמַּעְיָנ֔וֹת וְאֶת־ הַנַּ֛חַל הַשּׁוֹטֵ֥ף בְּתוֹךְ־ הָאָ֖רֶץ לֵאמֹ֑ר לָ֤מָּה יָב֙וֹאוּ֙ מַלְכֵ֣י אַשּׁ֔וּר וּמָצְא֖וּ מַ֥יִם רַבִּֽים׃ (יז) וַיִּתְחַזַּ֡ק וַיִּבֶן֩ אֶת־ כָּל־ הַחוֹמָ֨ה הַפְּרוּצָ֜ה וַיַּ֣עַל עַל־ הַמִּגְדָּל֗וֹת וְלַח֙וּצָה֙ הַחוֹמָ֣ה אַחֶ֔רֶת וַיְחַזֵּ֥ק אֶת־ הַמִּלּ֖וֹא עִ֣יר דָּוִ֑יד וַיַּ֥עַשׂ שֶׁ֛לַח לָרֹ֖ב וּמָגִנִּֽים׃ (יח) וַיִּתֵּ֛ן שָׂרֵ֥י מִלְחָמ֖וֹת עַל־ הָעָ֑ם וַיִּקְבְּצֵ֣ם אֵלָ֗יו אֶל־ רְחוֹב֙ שַׁ֣עַר הָעִ֔יר וַיְדַבֵּ֥ר עַל־ לְבָבָ֖ם לֵאמֹֽר׃ (יט) חִזְק֣וּ וְאִמְצ֔וּ אַל־ תִּֽירְא֣וּ וְאַל־ תֵּחַ֗תּוּ מִפְּנֵי֙ מֶ֣לֶךְ אַשּׁ֔וּר וּמִלִּפְנֵ֖י כָּל־ הֶהָמ֣וֹן אֲשֶׁר־ עִמּ֑וֹ כִּֽי־ עִמָּ֥נוּ רַ֖ב מֵעִמּֽוֹ׃ (כ) עִמּוֹ֙ זְר֣וֹעַ בָּשָׂ֔ר וְעִמָּ֜נוּ יְהוָ֤ה אֱלֹהֵ֙ינוּ֙ לְעָזְרֵ֔נוּ וּלְהִלָּחֵ֖ם מִלְחֲמֹתֵ֑נוּ וַיִּסָּמְכ֣וּ הָעָ֔ם עַל־ דִּבְרֵ֖י יְחִזְקִיָּ֥הוּ מֶֽלֶךְ־ יְהוּדָֽה׃ (כא) אַ֣חַר זֶ֗ה שָׁ֠לַח סַנְחֵרִ֨יב מֶֽלֶךְ־ אַשּׁ֤וּר עֲבָדָיו֙ יְר֣וּשָׁלַ֔יְמָה וְהוּא֙ עַל־ לָכִ֔ישׁ וְכָל־ מֶמְשַׁלְתּ֖וֹ עִמּ֑וֹ עַל־ יְחִזְקִיָּ֙הוּ֙ מֶ֣לֶךְ יְהוּדָ֔ה וְעַל־ כָּל־ יְהוּדָ֛ה אֲשֶׁ֥ר בִּירוּשָׁלִַ֖ם לֵאמֹֽר׃ (כב) כֹּ֣ה אָמַ֔ר סַנְחֵרִ֖יב מֶ֣לֶךְ אַשּׁ֑וּר עַל־ מָה֙ אַתֶּ֣ם בֹּטְחִ֔ים וְיֹשְׁבִ֥ים בְּמָצ֖וֹר בִּירוּשָׁלִָֽם׃ (כג) הֲלֹ֤א יְחִזְקִיָּ֙הוּ֙ מַסִּ֣ית אֶתְכֶ֔ם לָתֵ֣ת אֶתְכֶ֔ם לָמ֛וּת בְּרָעָ֥ב וּבְצָמָ֖א לֵאמֹ֑ר יְהוָ֣ה אֱלֹהֵ֔ינוּ יַצִּילֵ֕נוּ מִכַּ֖ף מֶ֥לֶךְ אַשּֽׁוּר׃ (כד) הֲלֹא־ הוּא֙ יְחִזְקִיָּ֔הוּ הֵסִ֥יר אֶת־ בָּמֹתָ֖יו וְאֶת־ מִזְבְּחֹתָ֑יו וַיֹּ֨אמֶר לִֽיהוּדָ֤ה וְלִֽירוּשָׁלִַ֙ם֙ לֵאמֹ֔ר לִפְנֵ֨י מִזְבֵּ֧חַ אֶחָ֛ד תִּֽשְׁתַּחֲו֖וּ וְעָלָ֥יו תַּקְטִֽירוּ׃ (כה) הֲלֹ֣א תֵדְע֗וּ מֶ֤ה עָשִׂ֙יתִי֙ אֲנִ֣י וַאֲבוֹתַ֔י לְכֹ֖ל עַמֵּ֣י הָאֲרָצ֑וֹת הֲיָכ֣וֹל יָֽכְל֗וּ אֱלֹהֵי֙ גּוֹיֵ֣ הָאֲרָצ֔וֹת לְהַצִּ֥יל אֶת־ אַרְצָ֖ם מִיָּדִֽי׃ (כו) מִ֠י בְּֽכָל־ אֱלֹהֵ֞י הַגּוֹיִ֤ם הָאֵ֙לֶּה֙ אֲשֶׁ֣ר הֶחֱרִ֣ימוּ אֲבוֹתַ֔י אֲשֶׁ֣ר יָכ֔וֹל לְהַצִּ֥יל אֶת־ עַמּ֖וֹ מִיָּדִ֑י כִּ֤י יוּכַל֙ אֱלֹ֣הֵיכֶ֔ם לְהַצִּ֥יל אֶתְכֶ֖ם מִיָּדִֽי׃ (כז) וְעַתָּ֡ה אַל־ יַשִּׁיא֩ אֶתְכֶ֨ם חִזְקִיָּ֜הוּ וְאַל־ יַסִּ֨ית אֶתְכֶ֣ם כָּזֹאת֮ וְאַל־ תַּאֲמִ֣ינוּ לוֹ֒ כִּי־ לֹ֣א יוּכַ֗ל כָּל־ אֱל֙וֹהַ֙ כָּל־ גּ֣וֹי וּמַמְלָכָ֔ה לְהַצִּ֥יל עַמּ֛וֹ מִיָּדִ֖י וּמִיַּ֣ד אֲבוֹתָ֑י אַ֚ף כִּ֣י אֱ‍ֽלֹהֵיכֶ֔ם לֹא־ יַצִּ֥ילוּ אֶתְכֶ֖ם מִיָּדִֽי׃ (כח) וְעוֹד֙ דִּבְּר֣וּ עֲבָדָ֔יו עַל־ יְהוָ֖ה הָאֱלֹהִ֑ים וְעַ֖ל יְחִזְקִיָּ֥הוּ עַבְדּֽוֹ׃ (כט) וּסְפָרִ֣ים כָּתַ֔ב לְחָרֵ֕ף לַיהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וְלֵֽאמֹ֨ר עָלָ֜יו לֵאמֹ֗ר כֵּֽאלֹהֵ֞י גּוֹיֵ֤ הָאֲרָצוֹת֙ אֲשֶׁ֨ר לֹא־ הִצִּ֤ילוּ עַמָּם֙ מִיָּדִ֔י כֵּ֣ן לֹֽא־ יַצִּ֞יל אֱלֹהֵ֧י יְחִזְקִיָּ֛הוּ עַמּ֖וֹ מִיָּדִֽי׃ (ל) וַיִּקְרְא֨וּ בְקוֹל־ גָּד֜וֹל יְהוּדִ֗ית עַל־ עַ֤ם יְרוּשָׁלִַ֙ם֙ אֲשֶׁ֣ר עַל־ הַֽחוֹמָ֔ה לְיָֽרְאָ֖ם וּֽלְבַהֲלָ֑ם לְמַ֖עַן יִלְכְּד֥וּ אֶת־ הָעִֽיר׃ (לא) וַֽיְדַבְּר֔וּ אֶל־ אֱלֹהֵ֖י יְרוּשָׁלִָ֑ם כְּעַ֗ל אֱלֹהֵי֙ עַמֵּ֣י הָאָ֔רֶץ מַעֲשֵׂ֖ה יְדֵ֥י הָאָדָֽם׃ (לב) וַיִּתְפַּלֵּ֞ל יְחִזְקִיָּ֣הוּ הַמֶּ֗לֶךְ וִֽישַֽׁעְיָ֧הוּ בֶן־ אָמ֛וֹץ הַנָּבִ֖יא עַל־ זֹ֑את וַֽיִּזְעֲק֖וּ הַשָּׁמָֽיִם׃ (לג) וַיִּשְׁלַ֤ח יְהוָה֙ מַלְאָ֔ךְ וַיַּכְחֵ֞ד כָּל־ גִּבּ֥וֹר חַ֙יִל֙ וְנָגִ֣יד וְשָׂ֔ר בְּמַחֲנֵ֖ה מֶ֣לֶךְ אַשּׁ֑וּר וַיָּשָׁב֩ בְּבֹ֨שֶׁת פָּנִ֜ים לְאַרְצ֗וֹ וַיָּבֹא֙ בֵּ֣ית אֱלֹהָ֔יו (ומיציאו) וּמִֽיצִיאֵ֣י מֵעָ֔יו שָׁ֖ם הִפִּילֻ֥הוּ בֶחָֽרֶב׃ (לד) אֲשֶׁר֩ בָּנָ֨ה בָהֶ֤ם בָּמוֹת֙ וְהֶעֱמִיד֙ הָאֲשֵׁרִ֣ים וְהַפְּסִלִ֔ים לִפְנֵ֖י הִכָּנְע֑וֹ הִנָּ֣ם כְּתוּבִ֔ים עַ֖ל דִּבְרֵ֥י חוֹזָֽי׃ (סה) וַיִּשְׁכַּ֤ב מְנַשֶּׁה֙ עִם־ אֲבֹתָ֔יו וַֽיִּקְבְּרֻ֖הוּ בֵּית֑וֹ וַיִּמְלֹ֛ךְ אָמ֥וֹן בְּנ֖וֹ תַּחְתָּֽיו׃ (סו) בֶּן־ עֶשְׂרִ֧ים וּשְׁתַּ֛יִם שָׁנָ֖ה אָמ֣וֹן בְּמָלְכ֑וֹ וּשְׁתַּ֣יִם שָׁנִ֔ים מָלַ֖ךְ בִּֽירוּשָׁלִָֽם׃ (סז) וַיַּ֤עַשׂ הָרַע֙ בְּעֵינֵ֣י יְהוָ֔ה כַּאֲשֶׁ֥ר עָשָׂ֖ה מְנַשֶּׁ֣ה אָבִ֑יו וּֽלְכָל־ הַפְּסִילִ֗ים אֲשֶׁ֤ר עָשָׂה֙ מְנַשֶּׁ֣ה אָבִ֔יו זִבַּ֥ח אָמ֖וֹן וַיַּֽעַבְדֵֽם׃ (סח) וְלֹ֤א נִכְנַע֙ מִלִּפְנֵ֣י יְהוָ֔ה כְּהִכָּנַ֖ע מְנַשֶּׁ֣ה אָבִ֑יו כִּ֛י ה֥וּא אָמ֖וֹן הִרְבָּ֥ה אַשְׁמָֽה׃ (סט) וַיִּקְשְׁר֤וּ עָלָיו֙ עֲבָדָ֔יו וַיְמִיתֻ֖הוּ בְּבֵיתֽוֹ׃ (ע) וַיַּכּוּ֙ עַם־ הָאָ֔רֶץ אֵ֥ת כָּל־ הַקֹּֽשְׁרִ֖ים עַל־ הַמֶּ֣לֶךְ אָמ֑וֹן וַיַּמְלִ֧יכוּ עַם־ הָאָ֛רֶץ אֶת־ יֹאשִׁיָּ֥הוּ בְנ֖וֹ תַּחְתָּֽיו׃ (עא) בֶּן־ שְׁמוֹנֶ֥ה שָׁנִ֖ים יֹאשִׁיָּ֣הוּ בְמָלְכ֑וֹ וּשְׁלֹשִׁ֤ים וְאַחַת֙ שָׁנָ֔ה מָלַ֖ךְ בִּירוּשָׁלִָֽם׃
