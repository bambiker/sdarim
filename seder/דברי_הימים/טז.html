דברי הימים סדר טז

פתיחה באתר · ספר דברי הימים כולו ·

(א) וּבְהִכָּֽנְע֗וֹ שָׁ֤ב מִמֶּ֙נּוּ֙ אַף־ יְהוָ֔ה וְלֹ֥א לְהַשְׁחִ֖ית לְכָלָ֑ה וְגַם֙ בִּֽיהוּדָ֔ה הָיָ֖ה דְּבָרִ֥ים טוֹבִֽים׃ (ב) וַיִּתְחַזֵּ֞ק הַמֶּ֧לֶךְ רְחַבְעָ֛ם בִּירוּשָׁלִַ֖ם וַיִּמְלֹ֑ךְ כִּ֣י בֶן־ אַרְבָּעִ֣ים וְאַחַ֣ת שָׁנָה֩ רְחַבְעָ֨ם בְּמָלְכ֜וֹ וּֽשֲׁבַ֨ע עֶשְׂרֵ֥ה שָׁנָ֣ה ׀ מָלַ֣ךְ בִּֽירוּשָׁלִַ֗ם הָ֠עִיר אֲשֶׁר־ בָּחַ֨ר יְהוָ֜ה לָשׂ֨וּם אֶת־ שְׁמ֥וֹ שָׁם֙ מִכֹּל֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל וְשֵׁ֣ם אִמּ֔וֹ נַעֲמָ֖ה הָֽעַמֹּנִֽית׃ (ג) וַיַּ֖עַשׂ הָרָ֑ע כִּ֣י לֹ֤א הֵכִין֙ לִבּ֔וֹ לִדְר֖וֹשׁ אֶת־ יְהוָֽה׃ (ד) וְדִבְרֵ֣י רְחַבְעָ֗ם הָרִאשֹׁנִים֙ וְהָאֲ֣חַרוֹנִ֔ים הֲלֹא־ הֵ֨ם כְּתוּבִ֜ים בְּדִבְרֵ֨י שְׁמַֽעְיָ֧ה הַנָּבִ֛יא וְעִדּ֥וֹ הַחֹזֶ֖ה לְהִתְיַחֵ֑שׂ וּמִלְחֲמ֧וֹת רְחַבְעָ֛ם וְיָרָבְעָ֖ם כָּל־ הַיָּמִֽים׃ (ה) וַיִּשְׁכַּ֤ב רְחַבְעָם֙ עִם־ אֲבֹתָ֔יו וַיִּקָּבֵ֖ר בְּעִ֣יר דָּוִ֑יד וַיִּמְלֹ֛ךְ אֲבִיָּ֥ה בְנ֖וֹ תַּחְתָּֽיו׃ (ו) בִּשְׁנַ֛ת שְׁמוֹנֶ֥ה עֶשְׂרֵ֖ה לַמֶּ֣לֶךְ יָרָבְעָ֑ם וַיִּמְלֹ֥ךְ אֲבִיָּ֖ה עַל־ יְהוּדָֽה׃ (ז) שָׁל֣וֹשׁ שָׁנִ֗ים מָלַךְ֙ בִּיר֣וּשָׁלִַ֔ם וְשֵׁ֣ם אִמּ֔וֹ מִיכָיָ֥הוּ בַת־ אוּרִיאֵ֖ל מִן־ גִּבְעָ֑ה וּמִלְחָמָ֥ה הָיְתָ֛ה בֵּ֥ין אֲבִיָּ֖ה וּבֵ֥ין יָרָבְעָֽם׃ (ח) וַיֶּאְסֹ֨ר אֲבִיָּ֜ה אֶת־ הַמִּלְחָמָ֗ה בְּחַ֙יִל֙ גִּבּוֹרֵ֣י מִלְחָמָ֔ה אַרְבַּע־ מֵא֥וֹת אֶ֖לֶף אִ֣ישׁ בָּח֑וּר וְיָרָבְעָ֗ם עָרַ֤ךְ עִמּוֹ֙ מִלְחָמָ֔ה בִּשְׁמוֹנֶ֨ה מֵא֥וֹת אֶ֛לֶף אִ֥ישׁ בָּח֖וּר גִּבּ֥וֹר חָֽיִל׃ (ט) וַיָּ֣קָם אֲבִיָּ֗ה מֵעַל֙ לְהַ֣ר צְמָרַ֔יִם אֲשֶׁ֖ר בְּהַ֣ר אֶפְרָ֑יִם וַיֹּ֕אמֶר שְׁמָע֖וּנִי יָרָבְעָ֥ם וְכָל־ יִשְׂרָאֵֽל׃ (י) הֲלֹ֤א לָכֶם֙ לָדַ֔עַת כִּ֞י יְהוָ֣ה ׀ אֱלֹהֵ֣י יִשְׂרָאֵ֗ל נָתַ֨ן מַמְלָכָ֧ה לְדָוִ֛יד עַל־ יִשְׂרָאֵ֖ל לְעוֹלָ֑ם ל֥וֹ וּלְבָנָ֖יו בְּרִ֥ית מֶֽלַח׃ (יא) וַיָּ֙קָם֙ יָרָבְעָ֣ם בֶּן־ נְבָ֔ט עֶ֖בֶד שְׁלֹמֹ֣ה בֶן־ דָּוִ֑יד וַיִּמְרֹ֖ד עַל־ אֲדֹנָֽיו׃ (יב) וַיִּקָּבְצ֣וּ עָלָ֗יו אֲנָשִׁ֤ים רֵקִים֙ בְּנֵ֣י בְלִיַּ֔עַל וַיִּֽתְאַמְּצ֖וּ עַל־ רְחַבְעָ֣ם בֶּן־ שְׁלֹמֹ֑ה וּרְחַבְעָ֗ם הָ֤יָה נַ֙עַר֙ וְרַךְ־ לֵבָ֔ב וְלֹ֥א הִתְחַזַּ֖ק לִפְנֵיהֶֽם׃ (יג) וְעַתָּ֣ה ׀ אַתֶּ֣ם אֹֽמְרִ֗ים לְהִתְחַזֵּק֙ לִפְנֵי֙ מַמְלֶ֣כֶת יְהוָ֔ה בְּיַ֖ד בְּנֵ֣י דָוִ֑יד וְאַתֶּם֙ הָמ֣וֹן רָ֔ב וְעִמָּכֶם֙ עֶגְלֵ֣י זָהָ֔ב אֲשֶׁ֨ר עָשָׂ֥ה לָכֶ֛ם יָרָבְעָ֖ם לֵאלֹהִֽים׃ (יד) הֲלֹ֤א הִדַּחְתֶּם֙ אֶת־ כֹּהֲנֵ֣י יְהוָ֔ה אֶת־ בְּנֵ֥י אַהֲרֹ֖ן וְהַלְוִיִּ֑ם וַתַּעֲשׂ֨וּ לָכֶ֤ם כֹּהֲנִים֙ כְּעַמֵּ֣י הָאֲרָצ֔וֹת כָּל־ הַבָּ֗א לְמַלֵּ֨א יָד֜וֹ בְּפַ֤ר בֶּן־ בָּקָר֙ וְאֵילִ֣ם שִׁבְעָ֔ה וְהָיָ֥ה כֹהֵ֖ן לְלֹ֥א אֱלֹהִֽים׃ (טו) וַאֲנַ֛חְנוּ יְהוָ֥ה אֱלֹהֵ֖ינוּ וְלֹ֣א עֲזַבְנֻ֑הוּ וְכֹ֨הֲנִ֜ים מְשָׁרְתִ֤ים לַֽיהוָה֙ בְּנֵ֣י אַהֲרֹ֔ן וְהַלְוִיִּ֖ם בַּמְלָֽאכֶת׃ (טז) וּמַקְטִרִ֣ים לַיהוָ֡ה עֹל֣וֹת בַּבֹּֽקֶר־ בַּבֹּ֣קֶר וּבָעֶֽרֶב־ בָּעֶ֣רֶב וּקְטֹֽרֶת־ סַמִּים֩ וּמַעֲרֶ֨כֶת לֶ֜חֶם עַל־ הַשֻּׁלְחָ֣ן הַטָּה֗וֹר וּמְנוֹרַ֨ת הַזָּהָ֤ב וְנֵרֹתֶ֙יהָ֙ לְבָעֵר֙ בָּעֶ֣רֶב בָּעֶ֔רֶב כִּֽי־ שֹׁמְרִ֣ים אֲנַ֔חְנוּ אֶת־ מִשְׁמֶ֖רֶת יְהוָ֣ה אֱלֹהֵ֑ינוּ וְאַתֶּ֖ם עֲזַבְתֶּ֥ם אֹתֽוֹ׃ (יז) וְהִנֵּה֩ עִמָּ֨נוּ בָרֹ֜אשׁ הָאֱלֹהִ֧ים ׀ וְכֹהֲנָ֛יו וַחֲצֹצְר֥וֹת הַתְּרוּעָ֖ה לְהָרִ֣יעַ עֲלֵיכֶ֑ם בְּנֵ֣י יִשְׂרָאֵ֗ל אַל־ תִּלָּֽחֲמ֛וּ עִם־ יְהוָ֥ה אֱלֹהֵֽי־ אֲבֹתֵיכֶ֖ם כִּי־ לֹ֥א תַצְלִֽיחוּ׃ (יח) וְיָֽרָבְעָ֗ם הֵסֵב֙ אֶת־ הַמַּאְרָ֔ב לָב֖וֹא מֵֽאַחֲרֵיהֶ֑ם וַיִּֽהְיוּ֙ לִפְנֵ֣י יְהוּדָ֔ה וְהַמַּאְרָ֖ב מֵאַחֲרֵיהֶֽם׃ (יט) וַיִּפְנ֣וּ יְהוּדָ֗ה וְהִנֵּ֨ה לָהֶ֤ם הַמִּלְחָמָה֙ פָּנִ֣ים וְאָח֔וֹר וַֽיִּצְעֲק֖וּ לַיהוָ֑ה וְהַכֹּ֣הֲנִ֔ים (מחצצרים) מַחְצְרִ֖ים בַּחֲצֹצְרֽוֹת׃ (כ) וַיָּרִ֖יעוּ אִ֣ישׁ יְהוּדָ֑ה וַיְהִ֗י בְּהָרִ֙יעַ֙ אִ֣ישׁ יְהוּדָ֔ה וְהָאֱלֹהִ֗ים נָגַ֤ף אֶת־ יָֽרָבְעָם֙ וְכָל־ יִשְׂרָאֵ֔ל לִפְנֵ֥י אֲבִיָּ֖ה וִיהוּדָֽה׃ (כא) וַיָּנ֥וּסוּ בְנֵי־ יִשְׂרָאֵ֖ל מִפְּנֵ֣י יְהוּדָ֑ה וַיִּתְּנֵ֥ם אֱלֹהִ֖ים בְּיָדָֽם׃ (כב) וַיַּכּ֥וּ בָהֶ֛ם אֲבִיָּ֥ה וְעַמּ֖וֹ מַכָּ֣ה רַבָּ֑ה וַיִּפְּל֤וּ חֲלָלִים֙ מִיִּשְׂרָאֵ֔ל חֲמֵשׁ־ מֵא֥וֹת אֶ֖לֶף אִ֥ישׁ בָּחֽוּר׃ (כג) וַיִּכָּנְע֥וּ בְנֵי־ יִשְׂרָאֵ֖ל בָּעֵ֣ת הַהִ֑יא וַיֶּֽאֶמְצוּ֙ בְּנֵ֣י יְהוּדָ֔ה כִּ֣י נִשְׁעֲנ֔וּ עַל־ יְהוָ֖ה אֱלֹהֵ֥י אֲבוֹתֵיהֶֽם׃ (כד) וַיִּרְדֹּ֣ף אֲבִיָּה֮ אַחֲרֵ֣י יָרָבְעָם֒ וַיִּלְכֹּ֤ד מִמֶּ֙נּוּ֙ עָרִ֔ים אֶת־ בֵּֽית־ אֵל֙ וְאֶת־ בְּנוֹתֶ֔יהָ וְאֶת־ יְשָׁנָ֖ה וְאֶת־ בְּנוֹתֶ֑יהָ וְאֶת־ (עפרון) עֶפְרַ֖יִן וּבְנֹתֶֽיהָ׃ (כה) וְלֹֽא־ עָצַ֧ר כֹּֽחַ־ יָרָבְעָ֛ם ע֖וֹד בִּימֵ֣י אֲבִיָּ֑הוּ וַיִּגְּפֵ֥הוּ יְהוָ֖ה וַיָּמֹֽת׃ (כו) וַיִּתְחַזֵּ֣ק אֲבִיָּ֔הוּ וַיִּ֨שָּׂא־ ל֔וֹ נָשִׁ֖ים אַרְבַּ֣ע עֶשְׂרֵ֑ה וַיּ֗וֹלֶד עֶשְׂרִ֤ים וּשְׁנַ֙יִם֙ בָּנִ֔ים וְשֵׁ֥שׁ עֶשְׂרֵ֖ה בָּנֽוֹת׃ (כז) וְיֶ֙תֶר֙ דִּבְרֵ֣י אֲבִיָּ֔ה וּדְרָכָ֖יו וּדְבָרָ֑יו כְּתוּבִ֕ים בְּמִדְרַ֖שׁ הַנָּבִ֥יא עִדּֽוֹ׃ (כח) וַיִּשְׁכַּ֨ב אֲבִיָּ֜ה עִם־ אֲבֹתָ֗יו וַיִּקְבְּר֤וּ אֹתוֹ֙ בְּעִ֣יר דָּוִ֔יד וַיִּמְלֹ֛ךְ אָסָ֥א בְנ֖וֹ תַּחְתָּ֑יו בְּיָמָ֛יו שָׁקְטָ֥ה הָאָ֖רֶץ עֶ֥שֶׂר שָׁנִֽים׃ (כט) וַיַּ֤עַשׂ אָסָא֙ הַטּ֣וֹב וְהַיָּשָׁ֔ר בְּעֵינֵ֖י יְהוָ֥ה אֱלֹהָֽיו׃ (ל) וַיָּ֛סַר אֶת־ מִזְבְּח֥וֹת הַנֵּכָ֖ר וְהַבָּמ֑וֹת וַיְשַׁבֵּר֙ אֶת־ הַמַּצֵּב֔וֹת וַיְגַדַּ֖ע אֶת־ הָאֲשֵׁרִֽים׃ (לא) וַיֹּ֙אמֶר֙ לִֽיהוּדָ֔ה לִדְר֕וֹשׁ אֶת־ יְהוָ֖ה אֱלֹהֵ֣י אֲבוֹתֵיהֶ֑ם וְלַעֲשׂ֖וֹת הַתּוֹרָ֥ה וְהַמִּצְוָֽה׃ (לב) וַיָּ֙סַר֙ מִכָּל־ עָרֵ֣י יְהוּדָ֔ה אֶת־ הַבָּמ֖וֹת וְאֶת־ הַֽחַמָּנִ֑ים וַתִּשְׁקֹ֥ט הַמַּמְלָכָ֖ה לְפָנָֽיו׃ (לג) וַיִּ֛בֶן עָרֵ֥י מְצוּרָ֖ה בִּיהוּדָ֑ה כִּֽי־ שָׁקְטָ֣ה הָאָ֗רֶץ וְאֵין־ עִמּ֤וֹ מִלְחָמָה֙ בַּשָּׁנִ֣ים הָאֵ֔לֶּה כִּֽי־ הֵנִ֥יחַ יְהוָ֖ה לֽוֹ׃ (לד) וַיֹּ֨אמֶר לִֽיהוּדָ֜ה נִבְנֶ֣ה ׀ אֶת־ הֶעָרִ֣ים הָאֵ֗לֶּה וְנָסֵ֨ב חוֹמָ֣ה וּמִגְדָּלִים֮ דְּלָתַ֣יִם וּבְרִיחִים֒ עוֹדֶ֨נּוּ הָאָ֜רֶץ לְפָנֵ֗ינוּ כִּ֤י דָרַ֙שְׁנוּ֙ אֶת־ יְהוָ֣ה אֱלֹהֵ֔ינוּ דָּרַ֕שְׁנוּ וַיָּ֥נַֽח לָ֖נוּ מִסָּבִ֑יב וַיִּבְנ֖וּ וַיַּצְלִֽיחוּ׃ (לה) וַיְהִ֣י לְאָסָ֗א חַיִל֮ נֹשֵׂ֣א צִנָּ֣ה וָרֹמַח֒ מִֽיהוּדָה֙ שְׁלֹ֣שׁ מֵא֣וֹת אֶ֔לֶף וּמִבִּנְיָמִ֗ן נֹשְׂאֵ֤י מָגֵן֙ וְדֹ֣רְכֵי קֶ֔שֶׁת מָאתַ֥יִם וּשְׁמוֹנִ֖ים אָ֑לֶף כָּל־ אֵ֖לֶּה גִּבּ֥וֹרֵי חָֽיִל׃ (לו) וַיֵּצֵ֨א אֲלֵיהֶ֜ם זֶ֣רַח הַכּוּשִׁ֗י בְּחַ֙יִל֙ אֶ֣לֶף אֲלָפִ֔ים וּמַרְכָּב֖וֹת שְׁלֹ֣שׁ מֵא֑וֹת וַיָּבֹ֖א עַד־ מָרֵשָֽׁה׃ (לז) וַיֵּצֵ֥א אָסָ֖א לְפָנָ֑יו וַיַּֽעַרְכוּ֙ מִלְחָמָ֔ה בְּגֵ֥יא צְפַ֖תָה לְמָרֵשָֽׁה׃ (לח) וַיִּקְרָ֨א אָסָ֜א אֶל־ יְהוָ֣ה אֱלֹהָיו֮ וַיֹּאמַר֒ יְהוָ֗ה אֵֽין־ עִמְּךָ֤ לַעְזוֹר֙ בֵּ֥ין רַב֙ לְאֵ֣ין כֹּ֔חַ עָזְרֵ֜נוּ יְהוָ֤ה אֱלֹהֵ֙ינוּ֙ כִּֽי־ עָלֶ֣יךָ נִשְׁעַ֔נּוּ וּבְשִׁמְךָ֣ בָ֔אנוּ עַל־ הֶהָמ֖וֹן הַזֶּ֑ה יְהוָ֤ה אֱלֹהֵ֙ינוּ֙ אַ֔תָּה אַל־ יַעְצֹ֥ר עִמְּךָ֖ אֱנֽוֹשׁ׃ (לט) וַיִּגֹּ֤ף יְהוָה֙ אֶת־ הַכּוּשִׁ֔ים לִפְנֵ֥י אָסָ֖א וְלִפְנֵ֣י יְהוּדָ֑ה וַיָּנֻ֖סוּ הַכּוּשִֽׁים׃ (מ) וַיִּרְדְּפֵ֨ם אָסָ֜א וְהָעָ֣ם אֲשֶׁר־ עִמּוֹ֮ עַד־ לִגְרָר֒ וַיִּפֹּ֤ל מִכּוּשִׁים֙ לְאֵ֣ין לָהֶ֣ם מִֽחְיָ֔ה כִּֽי־ נִשְׁבְּר֥וּ לִפְנֵֽי־ יְהוָ֖ה וְלִפְנֵ֣י מַחֲנֵ֑הוּ וַיִּשְׂא֥וּ שָׁלָ֖ל הַרְבֵּ֥ה מְאֹֽד׃ (מא) וַיַּכּ֗וּ אֵ֤ת כָּל־ הֶֽעָרִים֙ סְבִיב֣וֹת גְּרָ֔ר כִּי־ הָיָ֥ה פַֽחַד־ יְהוָ֖ה עֲלֵיהֶ֑ם וַיָּבֹ֙זּוּ֙ אֶת־ כָּל־ הֶ֣עָרִ֔ים כִּֽי־ בִזָּ֥ה רַבָּ֖ה הָיְתָ֥ה בָהֶֽם׃ (מב) וְגַם־ אָהֳלֵ֥י מִקְנֶ֖ה הִכּ֑וּ וַיִּשְׁבּ֨וּ צֹ֤אן לָרֹב֙ וּגְמַלִּ֔ים וַיָּשֻׁ֖בוּ יְרוּשָׁלִָֽם׃ (מג) וַעֲזַרְיָ֙הוּ֙ בֶּן־ עוֹדֵ֔ד הָיְתָ֥ה עָלָ֖יו ר֥וּחַ אֱלֹהִֽים׃ (מד) וַיֵּצֵא֮ לִפְנֵ֣י אָסָא֒ וַיֹּ֣אמֶר ל֔וֹ שְׁמָע֕וּנִי אָסָ֖א וְכָל־ יְהוּדָ֣ה וּבִנְיָמִ֑ן יְהוָ֤ה עִמָּכֶם֙ בִּֽהְיֽוֹתְכֶ֣ם עִמּ֔וֹ וְאִֽם־ תִּדְרְשֻׁ֙הוּ֙ יִמָּצֵ֣א לָכֶ֔ם וְאִם־ תַּעַזְבֻ֖הוּ יַעֲזֹ֥ב אֶתְכֶֽם׃ (מה) וְיָמִ֥ים רַבִּ֖ים לְיִשְׂרָאֵ֑ל לְלֹ֣א ׀ אֱלֹהֵ֣י אֱמֶ֗ת וּלְלֹ֛א כֹּהֵ֥ן מוֹרֶ֖ה וּלְלֹ֥א תוֹרָֽה׃ (מו) וַיָּ֙שָׁב֙ בַּצַּר־ ל֔וֹ עַל־ יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל וַיְבַקְשֻׁ֖הוּ וַיִּמָּצֵ֥א לָהֶֽם׃ (מז) וּבָעִתִּ֣ים הָהֵ֔ם אֵ֥ין שָׁל֖וֹם לַיּוֹצֵ֣א וְלַבָּ֑א כִּ֚י מְהוּמֹ֣ת רַבּ֔וֹת עַ֥ל כָּל־ יוֹשְׁבֵ֖י הָאֲרָצֽוֹת׃ (מח) וְכֻתְּת֥וּ גוֹי־ בְּג֖וֹי וְעִ֣יר בְּעִ֑יר כִּֽי־ אֱלֹהִ֥ים הֲמָמָ֖ם בְּכָל־ צָרָֽה׃ (מט) וְאַתֶּ֣ם חִזְק֔וּ וְאַל־ יִרְפּ֖וּ יְדֵיכֶ֑ם כִּ֛י יֵ֥שׁ שָׂכָ֖ר לִפְעֻלַּתְכֶֽם׃ (נ) וְכִשְׁמֹ֨עַ אָסָ֜א הַדְּבָרִ֣ים הָאֵ֗לֶּה וְהַנְּבוּאָה֮ עֹדֵ֣ד הַנָּבִיא֒ הִתְחַזַּ֗ק וַיַּעֲבֵ֤ר הַשִּׁקּוּצִים֙ מִכָּל־ אֶ֤רֶץ יְהוּדָה֙ וּבִנְיָמִ֔ן וּמִן־ הֶ֣עָרִ֔ים אֲשֶׁ֥ר לָכַ֖ד מֵהַ֣ר אֶפְרָ֑יִם וַיְחַדֵּשׁ֙ אֶת־ מִזְבַּ֣ח יְהוָ֔ה אֲשֶׁ֕ר לִפְנֵ֖י אוּלָ֥ם יְהוָֽה׃ (נא) וַיִּקְבֹּ֗ץ אֶת־ כָּל־ יְהוּדָה֙ וּבִנְיָמִ֔ן וְהַגָּרִים֙ עִמָּהֶ֔ם מֵאֶפְרַ֥יִם וּמְנַשֶּׁ֖ה וּמִשִּׁמְע֑וֹן כִּֽי־ נָפְל֨וּ עָלָ֤יו מִיִּשְׂרָאֵל֙ לָרֹ֔ב בִּרְאֹתָ֕ם כִּֽי־ יְהוָ֥ה אֱלֹהָ֖יו עִמּֽוֹ׃ (נב) וַיִּקָּבְצ֥וּ יְרוּשָׁלִַ֖ם בַּחֹ֣דֶשׁ הַשְּׁלִישִׁ֑י לִשְׁנַ֥ת חֲמֵשׁ־ עֶשְׂרֵ֖ה לְמַלְכ֥וּת אָסָֽא׃ (נג) וַיִּזְבְּח֤וּ לַיהוָה֙ בַּיּ֣וֹם הַה֔וּא מִן־ הַשָּׁלָ֖ל הֵבִ֑יאוּ בָּקָר֙ שְׁבַ֣ע מֵא֔וֹת וְצֹ֖אן שִׁבְעַ֥ת אֲלָפִֽים׃ (נד) וַיָּבֹ֣אוּ בַבְּרִ֔ית לִדְר֕וֹשׁ אֶת־ יְהוָ֖ה אֱלֹהֵ֣י אֲבוֹתֵיהֶ֑ם בְּכָל־ לְבָבָ֖ם וּבְכָל־ נַפְשָֽׁם׃ (נה) וְכֹ֨ל אֲשֶׁ֧ר לֹֽא־ יִדְרֹ֛שׁ לַיהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל יוּמָ֑ת לְמִן־ קָטֹן֙ וְעַד־ גָּד֔וֹל לְמֵאִ֖ישׁ וְעַד־ אִשָּֽׁה׃ (נו) וַיִּשָּֽׁבְעוּ֙ לַיהוָ֔ה בְּק֥וֹל גָּד֖וֹל וּבִתְרוּעָ֑ה וּבַחֲצֹצְר֖וֹת וּבְשׁוֹפָרֽוֹת׃
