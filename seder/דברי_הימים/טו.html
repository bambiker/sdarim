דברי הימים סדר טו

פתיחה באתר · ספר דברי הימים כולו ·

(א) וְהֵ֣ם מְבִיאִ֣ים אִ֣ישׁ מִנְחָת֡וֹ כְּלֵ֣י כֶסֶף֩ וּכְלֵ֨י זָהָ֤ב וּשְׂלָמוֹת֙ נֵ֣שֶׁק וּבְשָׂמִ֔ים סוּסִ֖ים וּפְרָדִ֑ים דְּבַר־ שָׁנָ֖ה בְּשָׁנָֽה׃ (ב) וַיְהִ֨י לִשְׁלֹמֹ֜ה אַרְבַּעַת֩ אֲלָפִ֨ים אֻֽרְי֤וֹת סוּסִים֙ וּמַרְכָּב֔וֹת וּשְׁנֵים־ עָשָׂ֥ר אֶ֖לֶף פָּרָשִׁ֑ים וַיַּנִּיחֵם֙ בְּעָרֵ֣י הָרֶ֔כֶב וְעִם־ הַמֶּ֖לֶךְ בִּירוּשָׁלִָֽם׃ (ג) וַיְהִ֥י מוֹשֵׁ֖ל בְּכָל־ הַמְּלָכִ֑ים מִן־ הַנָּהָר֙ וְעַד־ אֶ֣רֶץ פְּלִשְׁתִּ֔ים וְעַ֖ד גְּב֥וּל מִצְרָֽיִם׃ (ד) וַיִּתֵּ֨ן הַמֶּ֧לֶךְ אֶת־ הַכֶּ֛סֶף בִּירוּשָׁלִַ֖ם כָּאֲבָנִ֑ים וְאֵ֣ת הָאֲרָזִ֗ים נָתַ֛ן כַּשִּׁקְמִ֥ים אֲשֶׁר־ בַּשְּׁפֵלָ֖ה לָרֹֽב׃ (ה) וּמוֹצִיאִ֨ים סוּסִ֧ים מִמִּצְרַ֛יִם לִשְׁלֹמֹ֖ה וּמִכָּל־ הָאֲרָצֽוֹת׃ (ו) וּשְׁאָר֙ דִּבְרֵ֣י שְׁלֹמֹ֔ה הָרִאשֹׁנִ֖ים וְהָאֲחֲרוֹנִ֑ים הֲלֹא־ הֵ֣ם כְּתוּבִ֗ים עַל־ דִּבְרֵי֙ נָתָ֣ן הַנָּבִ֔יא וְעַל־ נְבוּאַ֞ת אֲחִיָּ֣ה הַשִּֽׁילוֹנִ֗י וּבַחֲזוֹת֙ (יעדי) יֶעְדּ֣וֹ הַחֹזֶ֔ה עַל־ יָרָבְעָ֖ם בֶּן־ נְבָֽט׃ (ז) וַיִּמְלֹ֨ךְ שְׁלֹמֹ֧ה בִֽירוּשָׁלִַ֛ם עַל־ כָּל־ יִשְׂרָאֵ֖ל אַרְבָּעִ֥ים שָׁנָֽה׃ (ח) וַיִּשְׁכַּ֤ב שְׁלֹמֹה֙ עִם־ אֲבֹתָ֔יו וַֽיִּקְבְּרֻ֔הוּ בְּעִ֖יר דָּוִ֣יד אָבִ֑יו וַיִּמְלֹ֛ךְ רְחַבְעָ֥ם בְּנ֖וֹ תַּחְתָּֽיו׃ (ט) וַיֵּ֥לֶךְ רְחַבְעָ֖ם שְׁכֶ֑מָה כִּ֥י שְׁכֶ֛ם בָּ֥אוּ כָל־ יִשְׂרָאֵ֖ל לְהַמְלִ֥יךְ אֹתֽוֹ׃ (י) וַיְהִ֞י כִּשְׁמֹ֨עַ יָרָבְעָ֤ם בֶּן־ נְבָט֙ וְה֣וּא בְמִצְרַ֔יִם אֲשֶׁ֣ר בָּרַ֔ח מִפְּנֵ֖י שְׁלֹמֹ֣ה הַמֶּ֑לֶךְ וַיָּ֥שָׁב יָרָבְעָ֖ם מִמִּצְרָֽיִם׃ (יא) וַֽיִּשְׁלְחוּ֙ וַיִּקְרְאוּ־ ל֔וֹ וַיָּבֹ֥א יָרָבְעָ֖ם וְכָל־ יִשְׂרָאֵ֑ל וַֽיְדַבְּר֔וּ אֶל־ רְחַבְעָ֖ם לֵאמֹֽר׃ (יב) אָבִ֖יךָ הִקְשָׁ֣ה אֶת־ עֻלֵּ֑נוּ וְעַתָּ֡ה הָקֵל֩ מֵעֲבֹדַ֨ת אָבִ֜יךָ הַקָּשָׁ֗ה וּמֵעֻלּ֧וֹ הַכָּבֵ֛ד אֲשֶׁר־ נָתַ֥ן עָלֵ֖ינוּ וְנַֽעַבְדֶֽךָּ׃ (יג) וַיֹּ֣אמֶר אֲלֵהֶ֗ם ע֛וֹד שְׁלֹ֥שֶׁת יָמִ֖ים וְשׁ֣וּבוּ אֵלָ֑י וַיֵּ֖לֶךְ הָעָֽם׃ (יד) וַיִּוָּעַ֞ץ הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ הַזְּקֵנִים֙ אֲשֶׁר־ הָי֣וּ עֹֽמְדִ֗ים לִפְנֵי֙ שְׁלֹמֹ֣ה אָבִ֔יו בִּֽהְיֹת֥וֹ חַ֖י לֵאמֹ֑ר אֵ֚יךְ אַתֶּ֣ם נֽוֹעָצִ֔ים לְהָשִׁ֥יב לָֽעָם־ הַזֶּ֖ה דָּבָֽר׃ (טו) וַיְדַבְּר֨וּ אֵלָ֜יו לֵאמֹ֗ר אִם־ תִּֽהְיֶ֨ה לְט֜וֹב לְהָעָ֤ם הַזֶּה֙ וּרְצִיתָ֔ם וְדִבַּרְתָּ֥ אֲלֵהֶ֖ם דְּבָרִ֣ים טוֹבִ֑ים וְהָי֥וּ לְךָ֛ עֲבָדִ֖ים כָּל־ הַיָּמִֽים׃ (טז) וַֽיַּעֲזֹ֛ב אֶת־ עֲצַ֥ת הַזְּקֵנִ֖ים אֲשֶׁ֣ר יְעָצֻ֑הוּ וַיִּוָּעַ֗ץ אֶת־ הַיְלָדִים֙ אֲשֶׁ֣ר גָּדְל֣וּ אִתּ֔וֹ הָעֹמְדִ֖ים לְפָנָֽיו׃ (יז) וַיֹּ֣אמֶר אֲלֵהֶ֗ם מָ֚ה אַתֶּ֣ם נֽוֹעָצִ֔ים וְנָשִׁ֥יב דָּבָ֖ר אֶת־ הָעָ֣ם הַזֶּ֑ה אֲשֶׁ֨ר דִּבְּר֤וּ אֵלַי֙ לֵאמֹ֔ר הָקֵל֙ מִן־ הָעֹ֔ל אֲשֶׁר־ נָתַ֥ן אָבִ֖יךָ עָלֵֽינוּ׃ (יח) וַיְדַבְּר֣וּ אִתּ֗וֹ הַיְלָדִים֙ אֲשֶׁ֨ר גָּדְל֣וּ אִתּוֹ֮ לֵאמֹר֒ כֹּֽה־ תֹאמַ֡ר לָעָם֩ אֲשֶׁר־ דִּבְּר֨וּ אֵלֶ֜יךָ לֵאמֹ֗ר אָבִ֙יךָ֙ הִכְבִּ֣יד אֶת־ עֻלֵּ֔נוּ וְאַתָּ֖ה הָקֵ֣ל מֵעָלֵ֑ינוּ כֹּ֚ה תֹּאמַ֣ר אֲלֵהֶ֔ם קָֽטָנִּ֥י עָבָ֖ה מִמָּתְנֵ֥י אָבִֽי׃ (יט) וְעַתָּ֗ה אָבִי֙ הֶעְמִ֤יס עֲלֵיכֶם֙ עֹ֣ל כָּבֵ֔ד וַאֲנִ֖י אֹסִ֣יף עַֽל־ עֻלְּכֶ֑ם אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֖י בָּֽעֲקְרַבִּֽים׃ (כ) וַיָּבֹ֨א יָרָבְעָ֧ם וְכָל־ הָעָ֛ם אֶל־ רְחַבְעָ֖ם בַּיּ֣וֹם הַשְּׁלִשִׁ֑י כַּאֲשֶׁ֨ר דִּבֶּ֤ר הַמֶּ֙לֶךְ֙ לֵאמֹ֔ר שׁ֥וּבוּ אֵלַ֖י בַּיּ֥וֹם הַשְּׁלִשִֽׁי׃ (כא) וַיַּעֲנֵ֥ם הַמֶּ֖לֶךְ קָשָׁ֑ה וַֽיַּעֲזֹב֙ הַמֶּ֣לֶךְ רְחַבְעָ֔ם אֵ֖ת עֲצַ֥ת הַזְּקֵנִֽים׃ (כב) וַיְדַבֵּ֣ר אֲלֵהֶ֗ם כַּעֲצַ֤ת הַיְלָדִים֙ לֵאמֹ֔ר אַכְבִּיד֙ אֶֽת־ עֻלְּכֶ֔ם וַאֲנִ֖י אֹסִ֣יף עָלָ֑יו אָבִ֗י יִסַּ֤ר אֶתְכֶם֙ בַּשּׁוֹטִ֔ים וַאֲנִ֖י בָּעֲקְרַבִּֽים׃ (כג) וְלֹֽא־ שָׁמַ֥ע הַמֶּ֖לֶךְ אֶל־ הָעָ֑ם כִּֽי־ הָיְתָ֤ה נְסִבָּה֙ מֵעִ֣ם הָֽאֱלֹהִ֔ים לְמַעַן֩ הָקִ֨ים יְהוָ֜ה אֶת־ דְּבָר֗וֹ אֲשֶׁ֤ר דִּבֶּר֙ בְּיַד֙ אֲחִיָּ֣הוּ הַשִּֽׁלוֹנִ֔י אֶל־ יָרָבְעָ֖ם בֶּן־ נְבָֽט׃ (כד) וְכָל־ יִשְׂרָאֵ֗ל כִּ֠י לֹא־ שָׁמַ֣ע הַמֶּלֶךְ֮ לָהֶם֒ וַיָּשִׁ֣יבוּ הָעָ֣ם אֶת־ הַמֶּ֣לֶךְ ׀ לֵאמֹ֡ר מַה־ לָּנוּ֩ חֵ֨לֶק בְּדָוִ֜יד וְלֹֽא־ נַחֲלָ֣ה בְּבֶן־ יִשַׁ֗י אִ֤ישׁ לְאֹהָלֶ֙יךָ֙ יִשְׂרָאֵ֔ל עַתָּ֕ה רְאֵ֥ה בֵיתְךָ֖ דָּוִ֑יד וַיֵּ֥לֶךְ כָּל־ יִשְׂרָאֵ֖ל לְאֹהָלָֽיו׃ (כה) וּבְנֵ֣י יִשְׂרָאֵ֔ל הַיֹּֽשְׁבִ֖ים בְּעָרֵ֣י יְהוּדָ֑ה וַיִּמְלֹ֥ךְ עֲלֵיהֶ֖ם רְחַבְעָֽם׃ (כו) וַיִּשְׁלַ֞ח הַמֶּ֣לֶךְ רְחַבְעָ֗ם אֶת־ הֲדֹרָם֙ אֲשֶׁ֣ר עַל־ הַמַּ֔ס וַיִּרְגְּמוּ־ ב֧וֹ בְנֵֽי־ יִשְׂרָאֵ֛ל אֶ֖בֶן וַיָּמֹ֑ת וְהַמֶּ֣לֶךְ רְחַבְעָ֗ם הִתְאַמֵּץ֙ לַֽעֲל֣וֹת בַּמֶּרְכָּבָ֔ה לָנ֖וּס יְרוּשָׁלִָֽם׃ (כז) וַיִּפְשְׁע֤וּ יִשְׂרָאֵל֙ בְּבֵ֣ית דָּוִ֔יד עַ֖ד הַיּ֥וֹם הַזֶּֽה׃ (כח) וַיָּבֹ֣א רְחַבְעָם֮ יְרוּשָׁלִַם֒ וַיַּקְהֵל֩ אֶת־ בֵּ֨ית יְהוּדָ֜ה וּבִנְיָמִ֗ן מֵאָ֨ה וּשְׁמוֹנִ֥ים אֶ֛לֶף בָּח֖וּר עֹשֵׂ֣ה מִלְחָמָ֑ה לְהִלָּחֵם֙ עִם־ יִשְׂרָאֵ֔ל לְהָשִׁ֥יב אֶת־ הַמַּמְלָכָ֖ה לִרְחַבְעָֽם׃ (כט) וַיְהִי֙ דְּבַר־ יְהוָ֔ה אֶל־ שְׁמַֽעְיָ֥הוּ אִישׁ־ הָאֱלֹהִ֖ים לֵאמֹֽר׃ (ל) אֱמֹ֕ר אֶל־ רְחַבְעָ֥ם בֶּן־ שְׁלֹמֹ֖ה מֶ֣לֶךְ יְהוּדָ֑ה וְאֶל֙ כָּל־ יִשְׂרָאֵ֔ל בִּיהוּדָ֥ה וּבִנְיָמִ֖ן לֵאמֹֽר׃ (לא) כֹּ֣ה אָמַ֣ר יְהוָ֡ה לֹא־ תַעֲלוּ֩ וְלֹא־ תִלָּ֨חֲמ֜וּ עִם־ אֲחֵיכֶ֗ם שׁ֚וּבוּ אִ֣ישׁ לְבֵית֔וֹ כִּ֧י מֵֽאִתִּ֛י נִהְיָ֖ה הַדָּבָ֣ר הַזֶּ֑ה וַֽיִּשְׁמְעוּ֙ אֶת־ דִּבְרֵ֣י יְהוָ֔ה וַיָּשֻׁ֖בוּ מִלֶּ֥כֶת אֶל־ יָרָבְעָֽם׃ (לב) וַיֵּ֥שֶׁב רְחַבְעָ֖ם בִּירוּשָׁלִָ֑ם וַיִּ֧בֶן עָרִ֛ים לְמָצ֖וֹר בִּיהוּדָֽה׃ (לג) וַיִּ֧בֶן אֶת־ בֵּֽית־ לֶ֛חֶם וְאֶת־ עֵיטָ֖ם וְאֶת־ תְּקֽוֹעַ׃ (לד) וְאֶת־ בֵּֽית־ צ֥וּר וְאֶת־ שׂוֹכ֖וֹ וְאֶת־ עֲדֻלָּֽם׃ (לה) וְאֶת־ גַּ֥ת וְאֶת־ מָרֵשָׁ֖ה וְאֶת־ זִֽיף׃ (לו) וְאֶת־ אֲדוֹרַ֥יִם וְאֶת־ לָכִ֖ישׁ וְאֶת־ עֲזֵקָֽה׃ (לז) וְאֶת־ צָרְעָה֙ וְאֶת־ אַיָּל֔וֹן וְאֶת־ חֶבְר֔וֹן אֲשֶׁ֥ר בִּיהוּדָ֖ה וּבְבִנְיָמִ֑ן עָרֵ֖י מְצֻרֽוֹת׃ (לח) וַיְחַזֵּ֖ק אֶת־ הַמְּצֻר֑וֹת וַיִּתֵּ֤ן בָּהֶם֙ נְגִידִ֔ים וְאֹצְר֥וֹת מַאֲכָ֖ל וְשֶׁ֥מֶן וָיָֽיִן׃ (לט) וּבְכָל־ עִ֤יר וָעִיר֙ צִנּ֣וֹת וּרְמָחִ֔ים וַֽיְחַזְּקֵ֖ם לְהַרְבֵּ֣ה מְאֹ֑ד וַיְהִי־ ל֖וֹ יְהוּדָ֥ה וּבִנְיָמִֽן׃ (מ) וְהַכֹּהֲנִים֙ וְהַלְוִיִּ֔ם אֲשֶׁ֖ר בְּכָל־ יִשְׂרָאֵ֑ל הִֽתְיַצְּב֥וּ עָלָ֖יו מִכָּל־ גְּבוּלָֽם׃ (מא) כִּֽי־ עָזְב֣וּ הַלְוִיִּ֗ם אֶת־ מִגְרְשֵׁיהֶם֙ וַאֲחֻזָּתָ֔ם וַיֵּלְכ֥וּ לִיהוּדָ֖ה וְלִֽירוּשָׁלִָ֑ם כִּֽי־ הִזְנִיחָ֤ם יָֽרָבְעָם֙ וּבָנָ֔יו מִכַּהֵ֖ן לַיהוָֽה׃ (מב) וַיַּֽעֲמֶד־ לוֹ֙ כֹּֽהֲנִ֔ים לַבָּמ֖וֹת וְלַשְּׂעִירִ֑ים וְלָעֲגָלִ֖ים אֲשֶׁ֥ר עָשָֽׂה׃ (מג) וְאַחֲרֵיהֶ֗ם מִכֹּל֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל הַנֹּֽתְנִים֙ אֶת־ לְבָבָ֔ם לְבַקֵּ֕שׁ אֶת־ יְהוָ֖ה אֱלֹהֵ֣י יִשְׂרָאֵ֑ל בָּ֚אוּ יְר֣וּשָׁלִַ֔ם לִזְבּ֕וֹחַ לַיהוָ֖ה אֱלֹהֵ֥י אֲבוֹתֵיהֶֽם׃ (מד) וַֽיְחַזְּקוּ֙ אֶת־ מַלְכ֣וּת יְהוּדָ֔ה וַֽיְאַמְּצ֛וּ אֶת־ רְחַבְעָ֥ם בֶּן־ שְׁלֹמֹ֖ה לְשָׁנִ֣ים שָׁל֑וֹשׁ כִּ֣י הָֽלְכ֗וּ בְּדֶ֧רֶךְ דָּוִ֛יד וּשְׁלֹמֹ֖ה לְשָׁנִ֥ים שָׁלֽוֹשׁ׃ (מה) וַיִּֽקַּֽח־ ל֤וֹ רְחַבְעָם֙ אִשָּׁ֔ה אֶת־ מָ֣חֲלַ֔ת (בן)־ בַּת־ יְרִימ֖וֹת בֶּן־ דָּוִ֑יד אֲבִיהַ֕יִל בַּת־ אֱלִיאָ֖ב בֶּן־ יִשָֽׁי׃ (מו) וַתֵּ֥לֶד ל֖וֹ בָּנִ֑ים אֶת־ יְע֥וּשׁ וְאֶת־ שְׁמַרְיָ֖ה וְאֶת־ זָֽהַם׃ (מז) וְאַחֲרֶ֣יהָ לָקַ֔ח אֶֽת־ מַעֲכָ֖ה בַּת־ אַבְשָׁל֑וֹם וַתֵּ֣לֶד ל֗וֹ אֶת־ אֲבִיָּה֙ וְאֶת־ עַתַּ֔י וְאֶת־ זִיזָ֖א וְאֶת־ שְׁלֹמִֽית׃ (מח) וַיֶּאֱהַ֨ב רְחַבְעָ֜ם אֶת־ מַעֲכָ֣ה בַת־ אַבְשָׁל֗וֹם מִכָּל־ נָשָׁיו֙ וּפִ֣ילַגְשָׁ֔יו כִּ֠י נָשִׁ֤ים שְׁמוֹנֶֽה־ עֶשְׂרֵה֙ נָשָׂ֔א וּפִֽילַגְשִׁ֖ים שִׁשִּׁ֑ים וַיּ֗וֹלֶד עֶשְׂרִ֧ים וּשְׁמוֹנָ֛ה בָּנִ֖ים וְשִׁשִּׁ֥ים בָּנֽוֹת׃ (מט) וַיַּֽעֲמֵ֨ד לָרֹ֧אשׁ רְחַבְעָ֛ם אֶת־ אֲבִיָּ֥ה בֶֽן־ מַעֲכָ֖ה לְנָגִ֣יד בְּאֶחָ֑יו כִּ֖י לְהַמְלִיכֽוֹ׃ (נ) וַיָּבֶן֩ וַיִּפְרֹ֨ץ מִכָּל־ בָּנָ֜יו לְֽכָל־ אַרְצ֧וֹת יְהוּדָ֣ה וּבִנְיָמִ֗ן לְכֹל֙ עָרֵ֣י הַמְּצֻר֔וֹת וַיִּתֵּ֥ן לָהֶ֛ם הַמָּז֖וֹן לָרֹ֑ב וַיִּשְׁאַ֖ל הֲמ֥וֹן נָשִֽׁים׃ (נא) וַיְהִ֗י כְּהָכִ֞ין מַלְכ֤וּת רְחַבְעָם֙ וּכְחֶזְקָת֔וֹ עָזַ֖ב אֶת־ תּוֹרַ֣ת יְהוָ֑ה וְכָל־ יִשְׂרָאֵ֖ל עִמּֽוֹ׃ (נב) וַיְהִ֞י בַּשָּׁנָ֤ה הַֽחֲמִישִׁית֙ לַמֶּ֣לֶךְ רְחַבְעָ֔ם עָלָ֛ה שִׁישַׁ֥ק מֶֽלֶךְ־ מִצְרַ֖יִם עַל־ יְרוּשָׁלִָ֑ם כִּ֥י מָעֲל֖וּ בַּיהוָֽה׃ (נג) בְּאֶ֤לֶף וּמָאתַ֙יִם֙ רֶ֔כֶב וּבְשִׁשִּׁ֥ים אֶ֖לֶף פָּרָשִׁ֑ים וְאֵ֣ין מִסְפָּ֗ר לָעָ֞ם אֲשֶׁר־ בָּ֤אוּ עִמּוֹ֙ מִמִּצְרַ֔יִם לוּבִ֥ים סֻכִּיִּ֖ים וְכוּשִֽׁים׃ (נד) וַיִּלְכֹּ֛ד אֶת־ עָרֵ֥י הַמְּצֻר֖וֹת אֲשֶׁ֣ר לִֽיהוּדָ֑ה וַיָּבֹ֖א עַד־ יְרוּשָׁלִָֽם׃ (נה) וּֽשְׁמַֽעְיָ֤ה הַנָּבִיא֙ בָּ֣א אֶל־ רְחַבְעָ֔ם וְשָׂרֵ֣י יְהוּדָ֔ה אֲשֶׁר־ נֶאֶסְפ֥וּ אֶל־ יְרוּשָׁלִַ֖ם מִפְּנֵ֣י שִׁישָׁ֑ק וַיֹּ֨אמֶר לָהֶ֜ם כֹּה־ אָמַ֣ר יְהוָ֗ה אַתֶּם֙ עֲזַבְתֶּ֣ם אֹתִ֔י וְאַף־ אֲנִ֛י עָזַ֥בְתִּי אֶתְכֶ֖ם בְּיַד־ שִׁישָֽׁק׃ (נו) וַיִּכָּנְע֥וּ שָׂרֵֽי־ יִשְׂרָאֵ֖ל וְהַמֶּ֑לֶךְ וַיֹּאמְר֖וּ צַדִּ֥יק ׀ יְהוָֽה׃ (נז) וּבִרְא֤וֹת יְהוָה֙ כִּ֣י נִכְנָ֔עוּ הָיָה֩ דְבַר־ יְהוָ֨ה אֶל־ שְׁמַֽעְיָ֧ה ׀ לֵאמֹ֛ר נִכְנְע֖וּ לֹ֣א אַשְׁחִיתֵ֑ם וְנָתַתִּ֨י לָהֶ֤ם כִּמְעַט֙ לִפְלֵיטָ֔ה וְלֹא־ תִתַּ֧ךְ חֲמָתִ֛י בִּירוּשָׁלִַ֖ם בְּיַד־ שִׁישָֽׁק׃ (נח) כִּ֥י יִהְיוּ־ ל֖וֹ לַעֲבָדִ֑ים וְיֵדְעוּ֙ עֲב֣וֹדָתִ֔י וַעֲבוֹדַ֖ת מַמְלְכ֥וֹת הָאֲרָצֽוֹת׃ (נט) וַיַּ֨עַל שִׁישַׁ֥ק מֶֽלֶךְ־ מִצְרַיִם֮ עַל־ יְרוּשָׁלִַם֒ וַיִּקַּ֞ח אֶת־ אֹצְר֣וֹת בֵּית־ יְהוָ֗ה וְאֶת־ אֹֽצְרוֹת֙ בֵּ֣ית הַמֶּ֔לֶךְ אֶת־ הַכֹּ֖ל לָקָ֑ח וַיִּקַּח֙ אֶת־ מָגִנֵּ֣י הַזָּהָ֔ב אֲשֶׁ֥ר עָשָׂ֖ה שְׁלֹמֹֽה׃ (ס) וַיַּ֨עַשׂ הַמֶּ֤לֶךְ רְחַבְעָם֙ תַּחְתֵּיהֶ֔ם מָגִנֵּ֖י נְחֹ֑שֶׁת וְהִפְקִ֗יד עַל־ יַד֙ שָׂרֵ֣י הָרָצִ֔ים הַשֹּׁ֣מְרִ֔ים פֶּ֖תַח בֵּ֥ית הַמֶּֽלֶךְ׃ (סא) וַיְהִ֛י מִדֵּי־ ב֥וֹא הַמֶּ֖לֶךְ בֵּ֣ית יְהוָ֑ה בָּ֤אוּ הָרָצִים֙ וּנְשָׂא֔וּם וֶהֱשִׁב֖וּם אֶל־ תָּ֥א הָרָצִֽים׃
