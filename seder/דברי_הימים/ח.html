דברי הימים סדר ח

פתיחה באתר · ספר דברי הימים כולו ·

(א) חֲזַ֤ק וְנִֽתְחַזְּקָה֙ בְּעַד־ עַמֵּ֔נוּ וּבְעַ֖ד עָרֵ֣י אֱלֹהֵ֑ינוּ וַֽיהוָ֔ה הַטּ֥וֹב בְּעֵינָ֖יו יַעֲשֶֽׂה׃ (ב) וַיִּגַּ֨שׁ יוֹאָ֜ב וְהָעָ֧ם אֲשֶׁר־ עִמּ֛וֹ לִפְנֵ֥י אֲרָ֖ם לַמִּלְחָמָ֑ה וַיָּנ֖וּסוּ מִפָּנָֽיו׃ (ג) וּבְנֵ֨י עַמּ֤וֹן רָאוּ֙ כִּי־ נָ֣ס אֲרָ֔ם וַיָּנ֣וּסוּ גַם־ הֵ֗ם מִפְּנֵי֙ אַבְשַׁ֣י אָחִ֔יו וַיָּבֹ֖אוּ הָעִ֑ירָה וַיָּבֹ֥א יוֹאָ֖ב יְרוּשָׁלִָֽם׃ (ד) וַיַּ֣רְא אֲרָ֗ם כִּ֣י נִגְּפוּ֮ לִפְנֵ֣י יִשְׂרָאֵל֒ וַֽיִּשְׁלְחוּ֙ מַלְאָכִ֔ים וַיּוֹצִ֣יאוּ אֶת־ אֲרָ֔ם אֲשֶׁ֖ר מֵעֵ֣בֶר הַנָּהָ֑ר וְשׁוֹפַ֛ךְ שַׂר־ צְבָ֥א הֲדַדְעֶ֖זֶר לִפְנֵיהֶֽם׃ (ה) וַיֻּגַּ֣ד לְדָוִ֗יד וַיֶּאֱסֹ֤ף אֶת־ כָּל־ יִשְׂרָאֵל֙ וַיַּעֲבֹ֣ר הַיַּרְדֵּ֔ן וַיָּבֹ֣א אֲלֵהֶ֔ם וַֽיַּעֲרֹ֖ךְ אֲלֵהֶ֑ם וַיַּעֲרֹ֨ךְ דָּוִ֜יד לִקְרַ֤את אֲרָם֙ מִלְחָמָ֔ה וַיִּֽלָּחֲמ֖וּ עִמּֽוֹ׃ (ו) וַיָּ֣נָס אֲרָם֮ מִלִּפְנֵ֣י יִשְׂרָאֵל֒ וַיַּהֲרֹ֨ג דָּוִ֜יד מֵאֲרָ֗ם שִׁבְעַ֤ת אֲלָפִים֙ רֶ֔כֶב וְאַרְבָּעִ֥ים אֶ֖לֶף אִ֣ישׁ רַגְלִ֑י וְאֵ֛ת שׁוֹפַ֥ךְ שַֽׂר־ הַצָּבָ֖א הֵמִֽית׃ (ז) וַיִּרְא֞וּ עַבְדֵ֣י הֲדַדְעֶ֗זֶר כִּ֤י נִגְּפוּ֙ לִפְנֵ֣י יִשְׂרָאֵ֔ל וַיַּשְׁלִ֥ימוּ עִם־ דָּוִ֖יד וַיַּֽעַבְדֻ֑הוּ וְלֹא־ אָבָ֣ה אֲרָ֔ם לְהוֹשִׁ֥יעַ אֶת־ בְּנֵי־ עַמּ֖וֹן עֽוֹד׃ (ח) וַיְהִ֡י לְעֵת֩ תְּשׁוּבַ֨ת הַשָּׁנָ֜ה לְעֵ֣ת ׀ צֵ֣את הַמְּלָכִ֗ים וַיִּנְהַ֣ג יוֹאָב֩ אֶת־ חֵ֨יל הַצָּבָ֜א וַיַּשְׁחֵ֣ת ׀ אֶת־ אֶ֣רֶץ בְּנֵֽי־ עַמּ֗וֹן וַיָּבֹא֙ וַיָּ֣צַר אֶת־ רַבָּ֔ה וְדָוִ֖יד יֹשֵׁ֣ב בִּירֽוּשָׁלִָ֑ם וַיַּ֥ךְ יוֹאָ֛ב אֶת־ רַבָּ֖ה וַיֶּֽהֶרְסֶֽהָ׃ (ט) וַיִּקַּ֣ח דָּוִ֣יד אֶת־ עֲטֶֽרֶת־ מַלְכָּם֩ מֵעַ֨ל רֹאשׁ֜וֹ וַֽיִּמְצָאָ֣הּ ׀ מִשְׁקַ֣ל כִּכַּר־ זָהָ֗ב וּבָהּ֙ אֶ֣בֶן יְקָרָ֔ה וַתְּהִ֖י עַל־ רֹ֣אשׁ דָּוִ֑יד וּשְׁלַ֥ל הָעִ֛יר הוֹצִ֖יא הַרְבֵּ֥ה מְאֹֽד׃ (י) וְאֶת־ הָעָ֨ם אֲשֶׁר־ בָּ֜הּ הוֹצִ֗יא וַיָּ֨שַׂר בַּמְּגֵרָ֜ה וּבַחֲרִיצֵ֤י הַבַּרְזֶל֙ וּבַמְּגֵר֔וֹת וְכֵן֙ יַעֲשֶׂ֣ה דָוִ֔יד לְכֹ֖ל עָרֵ֣י בְנֵי־ עַמּ֑וֹן וַיָּ֧שָׁב דָּוִ֛יד וְכָל־ הָעָ֖ם יְרוּשָׁלִָֽם׃ (יא) וַיְהִי֙ אַחֲרֵיכֵ֔ן וַתַּעֲמֹ֧ד מִלְחָמָ֛ה בְּגֶ֖זֶר עִם־ פְּלִשְׁתִּ֑ים אָ֣ז הִכָּ֞ה סִבְּכַ֣י הַחֻֽשָׁתִ֗י אֶת־ סִפַּ֛י מִילִדֵ֥י הָרְפָאִ֖ים וַיִּכָּנֵֽעוּ׃ (יב) וַתְּהִי־ ע֥וֹד מִלְחָמָ֖ה אֶת־ פְּלִשְׁתִּ֑ים וַיַּ֞ךְ אֶלְחָנָ֣ן בֶּן־ (יעור) יָעִ֗יר אֶת־ לַחְמִי֙ אֲחִי֙ גָּלְיָ֣ת הַגִּתִּ֔י וְעֵ֣ץ חֲנִית֔וֹ כִּמְנ֖וֹר אֹרְגִֽים׃ (יג) וַתְּהִי־ ע֥וֹד מִלְחָמָ֖ה בְּגַ֑ת וַיְהִ֣י ׀ אִ֣ישׁ מִדָּ֗ה וְאֶצְבְּעֹתָ֤יו שֵׁשׁ־ וָשֵׁשׁ֙ עֶשְׂרִ֣ים וְאַרְבַּ֔ע וְגַם־ ה֖וּא נוֹלַ֥ד לְהָרָפָֽא׃ (יד) וַיְחָרֵ֖ף אֶת־ יִשְׂרָאֵ֑ל וַיַּכֵּ֙הוּ֙ יְה֣וֹנָתָ֔ן בֶּן־ שִׁמְעָ֖א אֲחִ֥י דָוִֽיד׃ (טו) אֵ֛ל נוּלְּד֥וּ לְהָרָפָ֖א בְּגַ֑ת וַיִּפְּל֥וּ בְיַד־ דָּוִ֖יד וּבְיַד־ עֲבָדָֽיו׃ (טז) וַיַּֽעֲמֹ֥ד שָׂטָ֖ן עַל־ יִשְׂרָאֵ֑ל וַיָּ֙סֶת֙ אֶת־ דָּוִ֔יד לִמְנ֖וֹת אֶת־ יִשְׂרָאֵֽל׃ (יז) וַיֹּ֨אמֶר דָּוִ֤יד אֶל־ יוֹאָב֙ וְאֶל־ שָׂרֵ֣י הָעָ֔ם לְכ֗וּ סִפְרוּ֙ אֶת־ יִשְׂרָאֵ֔ל מִבְּאֵ֥ר שֶׁ֖בַע וְעַד־ דָּ֑ן וְהָבִ֣יאוּ אֵלַ֔י וְאֵדְעָ֖ה אֶת־ מִסְפָּרָֽם׃ (יח) וַיֹּ֣אמֶר יוֹאָ֗ב יוֹסֵף֩ יְהוָ֨ה עַל־ עַמּ֤וֹ ׀ כָּהֵם֙ מֵאָ֣ה פְעָמִ֔ים הֲלֹא֙ אֲדֹנִ֣י הַמֶּ֔לֶךְ כֻּלָּ֥ם לַאדֹנִ֖י לַעֲבָדִ֑ים לָ֣מָּה יְבַקֵּ֥שׁ זֹאת֙ אֲדֹנִ֔י לָ֛מָּה יִהְיֶ֥ה לְאַשְׁמָ֖ה לְיִשְׂרָאֵֽל׃ (יט) וּדְבַר־ הַמֶּ֖לֶךְ חָזַ֣ק עַל־ יוֹאָ֑ב וַיֵּצֵ֣א יוֹאָ֗ב וַיִּתְהַלֵּךְ֙ בְּכָל־ יִשְׂרָאֵ֔ל וַיָּבֹ֖א יְרוּשָׁלִָֽם׃ (כ) וַיִּתֵּ֥ן יוֹאָ֛ב אֶת־ מִסְפַּ֥ר מִפְקַד־ הָעָ֖ם אֶל־ דָּוִ֑יד וַיְהִ֣י כָֽל־ יִשְׂרָאֵ֡ל אֶ֣לֶף אֲלָפִים֩ וּמֵאָ֨ה אֶ֤לֶף אִישׁ֙ שֹׁ֣לֵֽף חֶ֔רֶב וִֽיהוּדָ֕ה אַרְבַּע֩ מֵא֨וֹת וְשִׁבְעִ֥ים אֶ֛לֶף אִ֖ישׁ שֹׁ֥לֵֽף חָֽרֶב׃ (כא) וְלֵוִי֙ וּבִנְיָמִ֔ן לֹ֥א פָקַ֖ד בְּתוֹכָ֑ם כִּֽי־ נִתְעַ֥ב דְּבַר־ הַמֶּ֖לֶךְ אֶת־ יוֹאָֽב׃ (כב) וַיֵּ֙רַע֙ בְּעֵינֵ֣י הָאֱלֹהִ֔ים עַל־ הַדָּבָ֖ר הַזֶּ֑ה וַיַּ֖ךְ אֶת־ יִשְׂרָאֵֽל׃ (כג) וַיֹּ֤אמֶר דָּוִיד֙ אֶל־ הָ֣אֱלֹהִ֔ים חָטָ֣אתִֽי מְאֹ֔ד אֲשֶׁ֥ר עָשִׂ֖יתִי אֶת־ הַדָּבָ֣ר הַזֶּ֑ה וְעַתָּ֗ה הַֽעֲבֶר־ נָא֙ אֶת־ עֲו֣וֹן עַבְדְּךָ֔ כִּ֥י נִסְכַּ֖לְתִּי מְאֹֽד׃ (כד) וַיְדַבֵּ֤ר יְהוָה֙ אֶל־ גָּ֔ד חֹזֵ֥ה דָוִ֖יד לֵאמֹֽר׃ (כה) לֵךְ֩ וְדִבַּרְתָּ֨ אֶל־ דָּוִ֜יד לֵאמֹ֗ר כֹּ֚ה אָמַ֣ר יְהוָ֔ה שָׁל֕וֹשׁ אֲנִ֖י נֹטֶ֣ה עָלֶ֑יךָ בְּחַר־ לְךָ֛ אַחַ֥ת מֵהֵ֖נָּה וְאֶֽעֱשֶׂה־ לָּֽךְ׃ (כו) וַיָּ֥בֹא גָ֖ד אֶל־ דָּוִ֑יד וַיֹּ֥אמֶר ל֛וֹ כֹּֽה־ אָמַ֥ר יְהוָ֖ה קַבֶּל־ לָֽךְ׃ (כז) אִם־ שָׁל֨וֹשׁ שָׁנִ֜ים רָעָ֗ב וְאִם־ שְׁלֹשָׁ֨ה חֳדָשִׁ֜ים נִסְפֶּ֥ה מִפְּנֵי־ צָרֶיךָ֮ וְחֶ֣רֶב אוֹיְבֶ֣ךָ ׀ לְמַשֶּׂגֶת֒ וְאִם־ שְׁלֹ֣שֶׁת יָ֠מִים חֶ֣רֶב יְהוָ֤ה וְדֶ֙בֶר֙ בָּאָ֔רֶץ וּמַלְאַ֣ךְ יְהוָ֔ה מַשְׁחִ֖ית בְּכָל־ גְּב֣וּל יִשְׂרָאֵ֑ל וְעַתָּ֣ה רְאֵ֔ה מָֽה־ אָשִׁ֥יב אֶת־ שֹׁלְחִ֖י דָּבָֽר׃ (כח) וַיֹּ֧אמֶר דָּוִ֛יד אֶל־ גָּ֖ד צַר־ לִ֣י מְאֹ֑ד אֶפְּלָה־ נָּ֣א בְיַד־ יְהוָ֗ה כִּֽי־ רַבִּ֤ים רַחֲמָיו֙ מְאֹ֔ד וּבְיַד־ אָדָ֖ם אַל־ אֶפֹּֽל׃ (כט) וַיִּתֵּ֧ן יְהוָ֛ה דֶּ֖בֶר בְּיִשְׂרָאֵ֑ל וַיִּפֹּל֙ מִיִּשְׂרָאֵ֔ל שִׁבְעִ֥ים אֶ֖לֶף אִֽישׁ׃ (ל) וַיִּשְׁלַח֩ הָאֱלֹהִ֨ים ׀ מַלְאָ֥ךְ ׀ לִֽירוּשָׁלִַם֮ לְהַשְׁחִיתָהּ֒ וּכְהַשְׁחִ֗ית רָאָ֤ה יְהוָה֙ וַיִּנָּ֣חֶם עַל־ הָֽרָעָ֔ה וַיֹּ֨אמֶר לַמַּלְאָ֤ךְ הַמַּשְׁחִית֙ רַ֔ב עַתָּ֖ה הֶ֣רֶף יָדֶ֑ךָ וּמַלְאַ֤ךְ יְהוָה֙ עֹמֵ֔ד עִם־ גֹּ֖רֶן אָרְנָ֥ן הַיְבוּסִֽי׃ (לא) וַיִּשָּׂ֨א דָוִ֜יד אֶת־ עֵינָ֗יו וַיַּ֞רְא אֶת־ מַלְאַ֤ךְ יְהוָה֙ עֹמֵ֗ד בֵּ֤ין הָאָ֙רֶץ֙ וּבֵ֣ין הַשָּׁמַ֔יִם וְחַרְבּ֤וֹ שְׁלוּפָה֙ בְּיָד֔וֹ נְטוּיָ֖ה עַל־ יְרוּשָׁלִָ֑ם וַיִּפֹּ֨ל דָּוִ֧יד וְהַזְּקֵנִ֛ים מְכֻסִּ֥ים בַּשַּׂקִּ֖ים עַל־ פְּנֵיהֶֽם׃ (לב) וַיֹּ֣אמֶר דָּוִ֣יד אֶֽל־ הָאֱלֹהִ֡ים הֲלֹא֩ אֲנִ֨י אָמַ֜רְתִּי לִמְנ֣וֹת בָּעָ֗ם וַאֲנִי־ ה֤וּא אֲשֶׁר־ חָטָ֙אתִי֙ וְהָרֵ֣עַ הֲרֵע֔וֹתִי וְאֵ֥לֶּה הַצֹּ֖אן מֶ֣ה עָשׂ֑וּ יְהוָ֣ה אֱלֹהַ֗י תְּהִ֨י נָ֤א יָֽדְךָ֙ בִּ֚י וּבְבֵ֣ית אָבִ֔י וּֽבְעַמְּךָ֖ לֹ֥א לְמַגֵּפָֽה׃ (לג) וּמַלְאַ֧ךְ יְהוָ֛ה אָמַ֥ר אֶל־ גָּ֖ד לֵאמֹ֣ר לְדָוִ֑יד כִּ֣י ׀ יַעֲלֶ֣ה דָוִ֗יד לְהָקִ֤ים מִזְבֵּ֙חַ֙ לַיהוָ֔ה בְּגֹ֖רֶן אָרְנָ֥ן הַיְבֻסִֽי׃ (לד) וַיַּ֤עַל דָּוִיד֙ בִּדְבַר־ גָּ֔ד אֲשֶׁ֥ר דִּבֶּ֖ר בְּשֵׁ֥ם יְהוָֽה׃ (לה) וַיָּ֣שָׁב אָרְנָ֗ן וַיַּרְא֙ אֶת־ הַמַּלְאָ֔ךְ וְאַרְבַּ֧עַת בָּנָ֛יו עִמּ֖וֹ מִֽתְחַבְּאִ֑ים וְאָרְנָ֖ן דָּ֥שׁ חִטִּֽים׃ (לו) וַיָּבֹ֥א דָוִ֖יד עַד־ אָרְנָ֑ן וַיַּבֵּ֤ט אָרְנָן֙ וַיַּ֣רְא אֶת־ דָּוִ֔יד וַיֵּצֵא֙ מִן־ הַגֹּ֔רֶן וַיִּשְׁתַּ֧חוּ לְדָוִ֛יד אַפַּ֖יִם אָֽרְצָה׃ (לז) וַיֹּ֨אמֶר דָּוִ֜יד אֶל־ אָרְנָ֗ן תְּנָה־ לִּי֙ מְק֣וֹם הַגֹּ֔רֶן וְאֶבְנֶה־ בּ֥וֹ מִזְבֵּ֖חַ לַיהוָ֑ה בְּכֶ֤סֶף מָלֵא֙ תְּנֵ֣הוּ לִ֔י וְתֵעָצַ֥ר הַמַּגֵּפָ֖ה מֵעַ֥ל הָעָֽם׃ (לח) וַיֹּ֨אמֶר אָרְנָ֤ן אֶל־ דָּוִיד֙ קַֽח־ לָ֔ךְ וְיַ֛עַשׂ אֲדֹנִ֥י הַמֶּ֖לֶךְ הַטּ֣וֹב בְּעֵינָ֑יו רְאֵה֩ נָתַ֨תִּי הַבָּקָ֜ר לָֽעֹל֗וֹת וְהַמּוֹרִגִּ֧ים לָעֵצִ֛ים וְהַחִטִּ֥ים לַמִּנְחָ֖ה הַכֹּ֥ל נָתָֽתִּי׃ (לט) וַיֹּ֨אמֶר הַמֶּ֤לֶךְ דָּוִיד֙ לְאָרְנָ֔ן לֹ֕א כִּֽי־ קָנֹ֥ה אֶקְנֶ֖ה בְּכֶ֣סֶף מָלֵ֑א כִּ֠י לֹא־ אֶשָּׂ֤א אֲשֶׁר־ לְךָ֙ לַיהוָ֔ה וְהַעֲל֥וֹת עוֹלָ֖ה חִנָּֽם׃ (מ) וַיִּתֵּ֥ן דָּוִ֛יד לְאָרְנָ֖ן בַּמָּק֑וֹם שִׁקְלֵ֣י זָהָ֔ב מִשְׁקָ֖ל שֵׁ֥שׁ מֵאֽוֹת׃ (מא) וַיִּבֶן֩ שָׁ֨ם דָּוִ֤יד מִזְבֵּ֙חַ֙ לַיהוָ֔ה וַיַּ֥עַל עֹל֖וֹת וּשְׁלָמִ֑ים וַיִּקְרָא֙ אֶל־ יְהוָ֔ה וַיַּֽעֲנֵ֤הוּ בָאֵשׁ֙ מִן־ הַשָּׁמַ֔יִם עַ֖ל מִזְבַּ֥ח הָעֹלָֽה׃ (מב) וַיֹּ֤אמֶר יְהוָה֙ לַמַּלְאָ֔ךְ וַיָּ֥שֶׁב חַרְבּ֖וֹ אֶל־ נְדָנָֽהּ׃ (מג) בָּעֵ֣ת הַהִ֔יא בִּרְא֤וֹת דָּוִיד֙ כִּי־ עָנָ֣הוּ יְהוָ֔ה בְּגֹ֖רֶן אָרְנָ֣ן הַיְבוּסִ֑י וַיִּזְבַּ֖ח שָֽׁם׃ (מד) וּמִשְׁכַּ֣ן יְ֠הוָה אֲשֶׁר־ עָשָׂ֨ה מֹשֶׁ֧ה בַמִּדְבָּ֛ר וּמִזְבַּ֥ח הָעוֹלָ֖ה בָּעֵ֣ת הַהִ֑יא בַּבָּמָ֖ה בְּגִבְעֽוֹן׃ (מה) וְלֹא־ יָכֹ֥ל דָּוִ֛יד לָלֶ֥כֶת לְפָנָ֖יו לִדְרֹ֣שׁ אֱלֹהִ֑ים כִּ֣י נִבְעַ֔ת מִפְּנֵ֕י חֶ֖רֶב מַלְאַ֥ךְ יְהוָֽה׃ (מו) וַיֹּ֣אמֶר דָּוִ֔יד זֶ֣ה ה֔וּא בֵּ֖ית יְהוָ֣ה הָאֱלֹהִ֑ים וְזֶה־ מִּזְבֵּ֥חַ לְעֹלָ֖ה לְיִשְׂרָאֵֽל׃ (מז) וַיֹּ֣אמֶר דָּוִ֔יד לִכְנוֹס֙ אֶת־ הַגֵּרִ֔ים אֲשֶׁ֖ר בְּאֶ֣רֶץ יִשְׂרָאֵ֑ל וַיַּעֲמֵ֣ד חֹֽצְבִ֗ים לַחְצוֹב֙ אַבְנֵ֣י גָזִ֔ית לִבְנ֖וֹת בֵּ֥ית הָאֱלֹהִֽים׃ (מח) וּבַרְזֶ֣ל ׀ לָ֠רֹב לַֽמִּסְמְרִ֞ים לְדַלְת֧וֹת הַשְּׁעָרִ֛ים וְלַֽמְחַבְּר֖וֹת הֵכִ֣ין דָּוִ֑יד וּנְחֹ֥שֶׁת לָרֹ֖ב אֵ֥ין מִשְׁקָֽל׃ (מט) וַעֲצֵ֥י אֲרָזִ֖ים לְאֵ֣ין מִסְפָּ֑ר כִּֽי הֵ֠בִיאוּ הַצִּֽידֹנִ֨ים וְהַצֹּרִ֜ים עֲצֵ֧י אֲרָזִ֛ים לָרֹ֖ב לְדָוִֽיד׃ (נ) וַיֹּ֣אמֶר דָּוִ֗יד שְׁלֹמֹ֣ה בְנִי֮ נַ֣עַר וָרָךְ֒ וְהַבַּ֜יִת לִבְנ֣וֹת לַיהוָ֗ה לְהַגְדִּ֨יל ׀ לְמַ֜עְלָה לְשֵׁ֤ם וּלְתִפְאֶ֙רֶת֙ לְכָל־ הָ֣אֲרָצ֔וֹת אָכִ֥ינָה נָּ֖א ל֑וֹ וַיָּ֧כֶן דָּוִ֛יד לָרֹ֖ב לִפְנֵ֥י מוֹתֽוֹ׃ (נא) וַיִּקְרָ֖א לִשְׁלֹמֹ֣ה בְנ֑וֹ וַיְצַוֵּ֙הוּ֙ לִבְנ֣וֹת בַּ֔יִת לַיהוָ֖ה אֱלֹהֵ֥י יִשְׂרָאֵֽל׃ (נב) וַיֹּ֥אמֶר דָּוִ֖יד לִשְׁלֹמֹ֑ה (בנו) בְּנִ֕י אֲנִי֙ הָיָ֣ה עִם־ לְבָבִ֔י לִבְנ֣וֹת בַּ֔יִת לְשֵׁ֖ם יְהוָ֥ה אֱלֹהָֽי׃ (נג) וַיְהִ֨י עָלַ֤י דְּבַר־ יְהוָה֙ לֵאמֹ֔ר דָּ֤ם לָרֹב֙ שָׁפַ֔כְתָּ וּמִלְחָמ֥וֹת גְּדֹל֖וֹת עָשִׂ֑יתָ לֹֽא־ תִבְנֶ֥ה בַ֙יִת֙ לִשְׁמִ֔י כִּ֚י דָּמִ֣ים רַבִּ֔ים שָׁפַ֥כְתָּ אַ֖רְצָה לְפָנָֽי׃ (נד) הִנֵּה־ בֵ֞ן נוֹלָ֣ד לָ֗ךְ ה֤וּא יִהְיֶה֙ אִ֣ישׁ מְנוּחָ֔ה וַהֲנִח֥וֹתִי ל֛וֹ מִכָּל־ אוֹיְבָ֖יו מִסָּבִ֑יב כִּ֤י שְׁלֹמֹה֙ יִהְיֶ֣ה שְׁמ֔וֹ וְשָׁל֥וֹם וָשֶׁ֛קֶט אֶתֵּ֥ן עַל־ יִשְׂרָאֵ֖ל בְּיָמָֽיו׃ (נה) הֽוּא־ יִבְנֶ֥ה בַ֙יִת֙ לִשְׁמִ֔י וְהוּא֙ יִהְיֶה־ לִּ֣י לְבֵ֔ן וַאֲנִי־ ל֖וֹ לְאָ֑ב וַהֲכִ֨ינוֹתִ֜י כִּסֵּ֧א מַלְכוּת֛וֹ עַל־ יִשְׂרָאֵ֖ל עַד־ עוֹלָֽם׃ (נו) עַתָּ֣ה בְנִ֔י יְהִ֥י יְהוָ֖ה עִמָּ֑ךְ וְהִצְלַחְתָּ֗ וּבָנִ֙יתָ֙ בֵּ֚ית יְהוָ֣ה אֱלֹהֶ֔יךָ כַּאֲשֶׁ֖ר דִּבֶּ֥ר עָלֶֽיךָ׃ (נז) אַ֣ךְ יִֽתֶּן־ לְּךָ֤ יְהוָה֙ שֵׂ֣כֶל וּבִינָ֔ה וִֽיצַוְּךָ֖ עַל־ יִשְׂרָאֵ֑ל וְלִשְׁמ֕וֹר אֶת־ תּוֹרַ֖ת יְהוָ֥ה אֱלֹהֶֽיךָ׃ (נח) אָ֣ז תַּצְלִ֔יחַ אִם־ תִּשְׁמ֗וֹר לַעֲשׂוֹת֙ אֶת־ הַֽחֻקִּ֣ים וְאֶת־ הַמִּשְׁפָּטִ֔ים אֲשֶׁ֨ר צִוָּ֧ה יְהוָ֛ה אֶת־ מֹשֶׁ֖ה עַל־ יִשְׂרָאֵ֑ל חֲזַ֣ק וֶאֱמָ֔ץ אַל־ תִּירָ֖א וְאַל־ תֵּחָֽת׃ (נט) וְהִנֵּ֨ה בְעָנְיִ֜י הֲכִינ֣וֹתִי לְבֵית־ יְהוָ֗ה זָהָ֞ב כִּכָּרִ֤ים מֵֽאָה־ אֶ֙לֶף֙ וְכֶ֗סֶף אֶ֤לֶף אֲלָפִים֙ כִּכָּרִ֔ים וְלַנְּחֹ֤שֶׁת וְלַבַּרְזֶל֙ אֵ֣ין מִשְׁקָ֔ל כִּ֥י לָרֹ֖ב הָיָ֑ה וְעֵצִ֤ים וַאֲבָנִים֙ הֲכִינ֔וֹתִי וַעֲלֵיהֶ֖ם תּוֹסִֽיף׃ (ס) וְעִמְּךָ֤ לָרֹב֙ עֹשֵׂ֣י מְלָאכָ֔ה חֹצְבִ֕ים וְחָרָשֵׁ֥י אֶ֖בֶן וָעֵ֑ץ וְכָל־ חָכָ֖ם בְּכָל־ מְלָאכָֽה׃ (סא) לַזָּהָ֥ב לַכֶּ֛סֶף וְלַנְּחֹ֥שֶׁת וְלַבַּרְזֶ֖ל אֵ֣ין מִסְפָּ֑ר ק֣וּם וַעֲשֵׂ֔ה וִיהִ֥י יְהוָ֖ה עִמָּֽךְ׃ (סב) וַיְצַ֤ו דָּוִיד֙ לְכָל־ שָׂרֵ֣י יִשְׂרָאֵ֔ל לַעְזֹ֖ר לִשְׁלֹמֹ֥ה בְנֽוֹ׃ (סג) הֲלֹ֨א יְהוָ֤ה אֱלֹֽהֵיכֶם֙ עִמָּכֶ֔ם וְהֵנִ֥יחַ לָכֶ֖ם מִסָּבִ֑יב כִּ֣י ׀ נָתַ֣ן בְּיָדִ֗י אֵ֚ת יֹשְׁבֵ֣י הָאָ֔רֶץ וְנִכְבְּשָׁ֥ה הָאָ֛רֶץ לִפְנֵ֥י יְהוָ֖ה וְלִפְנֵ֥י עַמּֽוֹ׃
