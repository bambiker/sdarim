(א) עַמִּיאֵ֤ל הַשִּׁשִּׁי֙ יִשָׂשכָ֣ר הַשְּׁבִיעִ֔י פְּעֻלְּתַ֖י הַשְּׁמִינִ֑י כִּ֥י בֵרֲכ֖וֹ אֱלֹהִֽים׃ (ב) וְלִֽשְׁמַֽעְיָ֤ה בְנוֹ֙ נוֹלַ֣ד בָּנִ֔ים הַמִּמְשָׁלִ֖ים לְבֵ֣ית אֲבִיהֶ֑ם כִּֽי־ גִבּ֥וֹרֵי חַ֖יִל הֵֽמָּה׃ (ג) בְּנֵ֣י שְׁמַֽעְיָ֗ה עָ֠תְנִי וּרְפָאֵ֨ל וְעוֹבֵ֧ד אֶלְזָבָ֛ד אֶחָ֖יו בְּנֵי־ חָ֑יִל אֱלִיה֖וּ וּסְמַכְיָֽהוּ׃ (ד) כָּל־ אֵ֜לֶּה מִבְּנֵ֣י ׀ עֹבֵ֣ד אֱדֹ֗ם הֵ֤מָּה וּבְנֵיהֶם֙ וַאֲחֵיהֶ֔ם אִֽישׁ־ חַ֥יִל בַּכֹּ֖חַ לַעֲבֹדָ֑ה שִׁשִּׁ֥ים וּשְׁנַ֖יִם לְעֹבֵ֥ד אֱדֹֽם׃ (ה) וְלִמְשֶֽׁלֶמְיָ֗הוּ בָּנִ֧ים וְאַחִ֛ים בְּנֵי־ חָ֖יִל שְׁמוֹנָ֥ה עָשָֽׂר׃ (ו) וּלְחֹסָ֥ה מִן־ בְּנֵי־ מְרָרִ֖י בָּנִ֑ים שִׁמְרִ֤י הָרֹאשׁ֙ כִּ֣י לֹא־ הָיָ֣ה בְכ֔וֹר וַיְשִׂימֵ֥הוּ אָבִ֖יהוּ לְרֹֽאשׁ׃ (ז) חִלְקִיָּ֤הוּ הַשֵּׁנִי֙ טְבַלְיָ֣הוּ הַשְּׁלִשִׁ֔י זְכַרְיָ֖הוּ הָרְבִעִ֑י כָּל־ בָּנִ֧ים וְאַחִ֛ים לְחֹסָ֖ה שְׁלֹשָׁ֥ה עָשָֽׂר׃ (ח) לְ֠אֵלֶּה מַחְלְק֨וֹת הַשֹּֽׁעֲרִ֜ים לְרָאשֵׁ֧י הַגְּבָרִ֛ים מִשְׁמָר֖וֹת לְעֻמַּ֣ת אֲחֵיהֶ֑ם לְשָׁרֵ֖ת בְּבֵ֥ית יְהוָֽה׃ (ט) וַיַּפִּ֨ילוּ גוֹרָל֜וֹת כַּקָּטֹ֧ן כַּגָּד֛וֹל לְבֵ֥ית אֲבוֹתָ֖ם לְשַׁ֥עַר וָשָֽׁעַר׃ (י) וַיִּפֹּ֧ל הַגּוֹרָ֛ל מִזְרָ֖חָה לְשֶֽׁלֶמְיָ֑הוּ וּזְכַרְיָ֨הוּ בְנ֜וֹ יוֹעֵ֣ץ ׀ בְּשֶׂ֗כֶל הִפִּ֙ילוּ֙ גּֽוֹרָל֔וֹת וַיֵּצֵ֥א גוֹרָל֖וֹ צָפֽוֹנָה׃ (יא) לְעֹבֵ֥ד אֱדֹ֖ם נֶ֑גְבָּה וּלְבָנָ֖יו בֵּ֥ית הָאֲסֻפִּֽים׃ (יב) לְשֻׁפִּ֤ים וּלְחֹסָה֙ לַֽמַּעֲרָ֔ב עִ֚ם שַׁ֣עַר שַׁלֶּ֔כֶת בַּֽמְסִלָּ֖ה הָעוֹלָ֑ה מִשְׁמָ֖ר לְעֻמַּ֥ת מִשְׁמָֽר׃ (יג) לַמִּזְרָח֮ הַלְוִיִּ֣ם שִׁשָּׁה֒ לַצָּפ֤וֹנָה לַיּוֹם֙ אַרְבָּעָ֔ה לַנֶּ֥גְבָּה לַיּ֖וֹם אַרְבָּעָ֑ה וְלָאֲסֻפִּ֖ים שְׁנַ֥יִם שְׁנָֽיִם׃ (יד) לַפַּרְבָּ֖ר לַֽמַּעֲרָ֑ב אַרְבָּעָה֙ לַֽמְסִלָּ֔ה שְׁנַ֖יִם לַפַּרְבָּֽר׃ (טו) אֵ֗לֶּה מַחְלְקוֹת֙ הַשֹּׁ֣עֲרִ֔ים לִבְנֵ֥י הַקָּרְחִ֖י וְלִבְנֵ֥י מְרָרִֽי׃ (טז) וְֽהַלְוִיִּ֑ם אֲחִיָּ֗ה עַל־ אֽוֹצְרוֹת֙ בֵּ֣ית הָאֱלֹהִ֔ים וּלְאֹֽצְר֖וֹת הַקֳּדָשִֽׁים׃ (יז) בְּנֵ֣י לַ֠עְדָּן בְּנֵ֨י הַגֵּרְשֻׁנִּ֜י לְלַעְדָּ֗ן רָאשֵׁ֧י הָאָב֛וֹת לְלַעְדָּ֥ן הַגֵּרְשֻׁנִּ֖י יְחִיאֵלִֽי׃ (יח) בְּנֵ֖י יְחִֽיאֵלִ֑י זֵתָם֙ וְיוֹאֵ֣ל אָחִ֔יו עַל־ אֹצְר֖וֹת בֵּ֥ית יְהוָֽה׃ (יט) לַֽעַמְרָמִי֙ לַיִּצְהָרִ֔י לַֽחֶבְרוֹנִ֖י לָֽעָזִּיאֵלִֽי׃ (כ) וּשְׁבֻאֵל֙ בֶּן־ גֵּרְשׁ֣וֹם בֶּן־ מֹשֶׁ֔ה נָגִ֖יד עַל־ הָאֹצָרֽוֹת׃ (כא) וְאֶחָ֖יו לֶֽאֱלִיעֶ֑זֶר רְחַבְיָ֨הוּ בְנ֜וֹ וִֽישַׁעְיָ֤הוּ בְנוֹ֙ וְיֹרָ֣ם בְּנ֔וֹ וְזִכְרִ֥י בְנ֖וֹ (ושלמות) וּשְׁלֹמִ֥ית בְּנֽוֹ׃ (כב) ה֧וּא שְׁלֹמ֣וֹת וְאֶחָ֗יו עַ֣ל כָּל־ אֹצְר֤וֹת הַקֳּדָשִׁים֙ אֲשֶׁ֨ר הִקְדִּ֜ישׁ דָּוִ֣יד הַמֶּ֗לֶךְ וְרָאשֵׁ֧י הָאָב֛וֹת לְשָׂרֵֽי־ הָאֲלָפִ֥ים וְהַמֵּא֖וֹת וְשָׂרֵ֥י הַצָּבָֽא׃ (כג) מִן־ הַמִּלְחָמ֥וֹת וּמִן־ הַשָּׁלָ֖ל הִקְדִּ֑ישׁוּ לְחַזֵּ֖ק לְבֵ֥ית יְהוָֽה׃ (כד) וְכֹ֨ל הַֽהִקְדִּ֜ישׁ שְׁמוּאֵ֤ל הָרֹאֶה֙ וְשָׁא֣וּל בֶּן־ קִ֔ישׁ וְאַבְנֵ֣ר בֶּן־ נֵ֔ר וְיוֹאָ֖ב בֶּן־ צְרוּיָ֑ה כֹּ֚ל הַמַּקְדִּ֔ישׁ עַ֥ל יַד־ שְׁלֹמִ֖ית וְאֶחָֽיו׃ (כה) לַיִּצְהָרִ֞י כְּנַנְיָ֣הוּ וּבָנָ֗יו לַמְּלָאכָ֤ה הַחִֽיצוֹנָה֙ עַל־ יִשְׂרָאֵ֔ל לְשֹׁטְרִ֖ים וּלְשֹׁפְטִֽים׃ (כו) לַֽחֶבְרוֹנִ֡י חֲשַׁבְיָהוּ֩ וְאֶחָ֨יו בְּנֵי־ חַ֜יִל אֶ֣לֶף וּשְׁבַע־ מֵא֗וֹת עַ֚ל פְּקֻדַּ֣ת יִשְׂרָאֵ֔ל מֵעֵ֥בֶר לַיַּרְדֵּ֖ן מַעְרָ֑בָה לְכֹל֙ מְלֶ֣אכֶת יְהוָ֔ה וְלַעֲבֹדַ֖ת הַמֶּֽלֶךְ׃ (כז) לַֽחֶבְרוֹנִי֙ יְרִיָּ֣ה הָרֹ֔אשׁ לַֽחֶבְרוֹנִ֥י לְתֹלְדֹתָ֖יו לְאָב֑וֹת בִּשְׁנַ֨ת הָֽאַרְבָּעִ֜ים לְמַלְכ֤וּת דָּוִיד֙ נִדְרָ֔שׁוּ וַיִּמָּצֵ֥א בָהֶ֛ם גִּבּ֥וֹרֵי חַ֖יִל בְּיַעְזֵ֥יר גִּלְעָֽד׃ (כח) וְאֶחָ֣יו בְּנֵי־ חַ֗יִל אַלְפַּ֛יִם וּשְׁבַ֥ע מֵא֖וֹת רָאשֵׁ֣י הָאָב֑וֹת וַיַּפְקִידֵ֞ם דָּוִ֣יד הַמֶּ֗לֶךְ עַל־ הָראוּבֵנִ֤י וְהַגָּדִי֙ וַחֲצִי֙ שֵׁ֣בֶט הַֽמְנַשִּׁ֔י לְכָל־ דְּבַ֥ר הָאֱלֹהִ֖ים וּדְבַ֥ר הַמֶּֽלֶךְ׃ (כט) וּבְנֵ֣י יִשְׂרָאֵ֣ל ׀ לְֽמִסְפָּרָ֡ם רָאשֵׁ֣י הָאָב֣וֹת וְשָׂרֵ֣י הָֽאֲלָפִ֣ים ׀ וְהַמֵּא֡וֹת וְשֹׁטְרֵיהֶם֩ הַמְשָׁרְתִ֨ים אֶת־ הַמֶּ֜לֶךְ לְכֹ֣ל ׀ דְּבַ֣ר הַֽמַּחְלְק֗וֹת הַבָּאָ֤ה וְהַיֹּצֵאת֙ חֹ֣דֶשׁ בְּחֹ֔דֶשׁ לְכֹ֖ל חָדְשֵׁ֣י הַשָּׁנָ֑ה הַֽמַּחֲלֹ֙קֶת֙ הָֽאַחַ֔ת עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (ל) עַ֞ל הַמַּחֲלֹ֤קֶת הָרִֽאשׁוֹנָה֙ לַחֹ֣דֶשׁ הָרִאשׁ֔וֹן יָֽשָׁבְעָ֖ם בֶּן־ זַבְדִּיאֵ֑ל וְעַל֙ מַֽחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (לא) מִן־ בְּנֵי־ פֶ֗רֶץ הָרֹ֛אשׁ לְכָל־ שָׂרֵ֥י הַצְּבָא֖וֹת לַחֹ֥דֶשׁ הָרִאשֽׁוֹן׃ (לב) וְעַ֞ל מַחֲלֹ֣קֶת ׀ הַחֹ֣דֶשׁ הַשֵּׁנִ֗י דּוֹדַ֤י הָאֲחוֹחִי֙ וּמַ֣חֲלֻקְתּ֔וֹ וּמִקְל֖וֹת הַנָּגִ֑יד וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (לג) שַׂ֣ר הַצָּבָ֤א הַשְּׁלִישִׁי֙ לַחֹ֣דֶשׁ הַשְּׁלִישִׁ֔י בְּנָיָ֧הוּ בֶן־ יְהוֹיָדָ֛ע הַכֹּהֵ֖ן רֹ֑אשׁ וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (לד) ה֧וּא בְנָיָ֛הוּ גִּבּ֥וֹר הַשְּׁלֹשִׁ֖ים וְעַל־ הַשְּׁלֹשִׁ֑ים וּמַ֣חֲלֻקְתּ֔וֹ עַמִּיזָבָ֖ד בְּנֽוֹ׃ (לה) הָֽרְבִיעִ֞י לַחֹ֣דֶשׁ הָרְבִיעִ֗י עֲשָׂה־ אֵל֙ אֲחִ֣י יוֹאָ֔ב וּזְבַדְיָ֥ה בְנ֖וֹ אַחֲרָ֑יו וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (לו) הַחַמִישִׁי֙ לַחֹ֣דֶשׁ הַחֲמִישִׁ֔י הַשַּׂ֖ר שַׁמְה֣וּת הַיִּזְרָ֑ח וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (לז) הַשִּׁשִּׁי֙ לַחֹ֣דֶשׁ הַשִּׁשִּׁ֔י עִירָ֥א בֶן־ עִקֵּ֖שׁ הַתְּקוֹעִ֑י וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (לח) הַשְּׁבִיעִי֙ לַחֹ֣דֶשׁ הַשְּׁבִיעִ֔י חֶ֥לֶץ הַפְּלוֹנִ֖י מִן־ בְּנֵ֣י אֶפְרָ֑יִם וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (לט) הַשְּׁמִינִי֙ לַחֹ֣דֶשׁ הַשְּׁמִינִ֔י סִבְּכַ֥י הַחֻשָׁתִ֖י לַזַּרְחִ֑י וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (מ) הַתְּשִׁיעִי֙ לַחֹ֣דֶשׁ הַתְּשִׁיעִ֔י אֲבִיעֶ֥זֶר הָעַנְּתֹתִ֖י (לבנימיני) לַבֵּ֣ן ׀ יְמִינִ֑י וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (מא) הָעֲשִׂירִי֙ לַחֹ֣דֶשׁ הָעֲשִׂירִ֔י מַהְרַ֥י הַנְּטֽוֹפָתִ֖י לַזַּרְחִ֑י וְעַל֙ מַֽחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (מב) עַשְׁתֵּֽי־ עָשָׂר֙ לְעַשְׁתֵּ֣י־ עָשָׂ֣ר הַחֹ֔דֶשׁ בְּנָיָ֥ה הַפִּרְעָתוֹנִ֖י מִן־ בְּנֵ֣י אֶפְרָ֑יִם וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (מג) הַשְּׁנֵ֤ים עָשָׂר֙ לִשְׁנֵ֣ים עָשָׂ֣ר הַחֹ֔דֶשׁ חֶלְדַּ֥י הַנְּטוֹפָתִ֖י לְעָתְנִיאֵ֑ל וְעַל֙ מַחֲלֻקְתּ֔וֹ עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה אָֽלֶף׃ (מד) וְעַל֙ שִׁבְטֵ֣י יִשְׂרָאֵ֔ל לָרֽאוּבֵנִ֣י נָגִ֔יד אֱלִיעֶ֖זֶר בֶּן־ זִכְרִ֑י לַשִּׁ֨מְעוֹנִ֔י שְׁפַטְיָ֖הוּ בֶּֽן־ מַעֲכָֽה׃ (מה) לְלֵוִ֛י חֲשַׁבְיָ֥ה בֶן־ קְמוּאֵ֖ל לְאַהֲרֹ֥ן צָדֽוֹק׃ (מו) לִֽיהוּדָ֕ה אֱלִיה֖וּ מֵאֲחֵ֣י דָוִ֑יד לְיִ֨שָׂשכָ֔ר עָמְרִ֖י בֶּן־ מִיכָאֵֽל׃ (מז) לִזְבוּלֻ֕ן יִֽשְׁמַֽעְיָ֖הוּ בֶּן־ עֹבַדְיָ֑הוּ לְנַ֨פְתָּלִ֔י יְרִימ֖וֹת בֶּן־ עַזְרִיאֵֽל׃ (מח) לִבְנֵ֣י אֶפְרַ֔יִם הוֹשֵׁ֖עַ בֶּן־ עֲזַזְיָ֑הוּ לַחֲצִי֙ שֵׁ֣בֶט מְנַשֶּׁ֔ה יוֹאֵ֖ל בֶּן־ פְּדָיָֽהוּ׃ (מט) לַחֲצִ֤י הַֽמְנַשֶּׁה֙ גִּלְעָ֔דָה יִדּ֖וֹ בֶּן־ זְכַרְיָ֑הוּ לְבִנְיָמִ֔ן יַעֲשִׂיאֵ֖ל בֶּן־ אַבְנֵֽר׃ (נ) לְדָ֕ן עֲזַרְאֵ֖ל בֶּן־ יְרֹחָ֑ם אֵ֕לֶּה שָׂרֵ֖י שִׁבְטֵ֥י יִשְׂרָאֵֽל׃ (נא) וְלֹא־ נָשָׂ֤א דָוִיד֙ מִסְפָּרָ֔ם לְמִבֶּ֛ן עֶשְׂרִ֥ים שָׁנָ֖ה וּלְמָ֑טָּה כִּ֚י אָמַ֣ר יְהוָ֔ה לְהַרְבּ֥וֹת אֶת־ יִשְׂרָאֵ֖ל כְּכוֹכְבֵ֥י הַשָּׁמָֽיִם׃ (נב) יוֹאָ֨ב בֶּן־ צְרוּיָ֜ה הֵחֵ֤ל לִמְנוֹת֙ וְלֹ֣א כִלָּ֔ה וַיְהִ֥י בָזֹ֛את קֶ֖צֶף עַל־ יִשְׂרָאֵ֑ל וְלֹ֤א עָלָה֙ הַמִּסְפָּ֔ר בְּמִסְפַּ֥ר דִּבְרֵֽי־ הַיָּמִ֖ים לַמֶּ֥לֶךְ דָּוִֽיד׃ (נג) וְעַל֙ אֹצְר֣וֹת הַמֶּ֔לֶךְ עַזְמָ֖וֶת בֶּן־ עֲדִיאֵ֑ל וְעַ֣ל הָֽאֹצָר֡וֹת בַּשָּׂדֶ֞ה בֶּעָרִ֤ים וּבַכְּפָרִים֙ וּבַמִּגְדָּל֔וֹת יְהוֹנָתָ֖ן בֶּן־ עֻזִּיָּֽהוּ׃ (נד) וְעַ֗ל עֹשֵׂי֙ מְלֶ֣אכֶת הַשָּׂדֶ֔ה לַעֲבֹדַ֖ת הָאֲדָמָ֑ה עֶזְרִ֖י בֶּן־ כְּלֽוּב׃ (נה) וְעַל־ הַ֨כְּרָמִ֔ים שִׁמְעִ֖י הָרָֽמָתִ֑י וְעַ֤ל שֶׁבַּכְּרָמִים֙ לְאֹצְר֣וֹת הַיַּ֔יִן זַבְדִּ֖י הַשִּׁפְמִֽי׃ (נו) וְעַל־ הַזֵּיתִ֤ים וְהַשִּׁקְמִים֙ אֲשֶׁ֣ר בַּשְּׁפֵלָ֔ה בַּ֥עַל חָנָ֖ן הַגְּדֵרִ֑י וְעַל־ אֹצְר֥וֹת הַשֶּׁ֖מֶן יוֹעָֽשׁ׃ (נז) וְעַל־ הַבָּקָר֙ הָרֹעִ֣ים בַּשָּׁר֔וֹן (שטרי) שִׁרְטַ֖י הַשָּׁרוֹנִ֑י וְעַל־ הַבָּקָר֙ בָּֽעֲמָקִ֔ים שָׁפָ֖ט בֶּן־ עַדְלָֽי׃ (נח) וְעַל־ הַ֨גְּמַלִּ֔ים אוֹבִ֖יל הַיִּשְׁמְעֵלִ֑י וְעַל־ הָ֣אֲתֹנ֔וֹת יֶחְדְּיָ֖הוּ הַמֵּרֹנֹתִֽי׃ (נט) וְעַל־ הַצֹּ֖אן יָזִ֣יז הַֽהַגְרִ֑י כָּל־ אֵ֙לֶּה֙ שָׂרֵ֣י הָרְכ֔וּשׁ אֲשֶׁ֖ר לַמֶּ֥לֶךְ דָּוִֽיד׃ (ס) וִֽיהוֹנָתָ֤ן דּוֹד־ דָּוִיד֙ יוֹעֵ֔ץ אִישׁ־ מֵבִ֥ין וְסוֹפֵ֖ר ה֑וּא וִֽיחִיאֵ֥ל בֶּן־ חַכְמוֹנִ֖י עִם־ בְּנֵ֥י הַמֶּֽלֶךְ׃ (סא) וַאֲחִיתֹ֖פֶל יוֹעֵ֣ץ לַמֶּ֑לֶךְ וְחוּשַׁ֥י הָאַרְכִּ֖י רֵ֥עַ הַמֶּֽלֶךְ׃ (סב) וְאַחֲרֵ֣י אֲחִיתֹ֗פֶל יְהוֹיָדָ֤ע בֶּן־ בְּנָיָ֙הוּ֙ וְאֶבְיָתָ֔ר וְשַׂר־ צָבָ֥א לַמֶּ֖לֶךְ יוֹאָֽב׃ (סג) וַיַּקְהֵ֣ל דָּוִ֣יד אֶת־ כָּל־ שָׂרֵ֣י יִשְׂרָאֵ֡ל שָׂרֵ֣י הַשְּׁבָטִ֣ים וְשָׂרֵ֣י הַמַּחְלְק֣וֹת הַמְשָׁרְתִ֪ים אֶת־ הַמֶּ֟לֶךְ וְשָׂרֵ֣י הָאֲלָפִ֣ים וְשָׂרֵ֣י הַמֵּא֡וֹת וְשָׂרֵ֣י כָל־ רְכוּשׁ־ וּמִקְנֶה֩ ׀ לַמֶּ֨לֶךְ וּלְבָנָ֜יו עִם־ הַסָּרִיסִ֧ים וְהַגִּבּוֹרִ֛ים וּֽלְכָל־ גִּבּ֥וֹר חָ֖יִל אֶל־ יְרוּשָׁלִָֽם׃ (סד) וַיָּ֨קָם דָּוִ֤יד הַמֶּ֙לֶךְ֙ עַל־ רַגְלָ֔יו וַיֹּ֕אמֶר שְׁמָע֖וּנִי אַחַ֣י וְעַמִּ֑י אֲנִ֣י עִם־ לְבָבִ֡י לִבְנוֹת֩ בֵּ֨ית מְנוּחָ֜ה לַאֲר֣וֹן בְּרִית־ יְהוָ֗ה וְלַהֲדֹם֙ רַגְלֵ֣י אֱלֹהֵ֔ינוּ וַהֲכִינ֖וֹתִי לִבְנֽוֹת׃ (סה) וְהָאֱלֹהִים֙ אָ֣מַר לִ֔י לֹא־ תִבְנֶ֥ה בַ֖יִת לִשְׁמִ֑י כִּ֣י אִ֧ישׁ מִלְחָמ֛וֹת אַ֖תָּה וְדָמִ֥ים שָׁפָֽכְתָּ׃ (סו) וַיִּבְחַ֡ר יְהוָ֣ה אֱלֹהֵי֩ יִשְׂרָאֵ֨ל בִּ֜י מִכֹּ֣ל בֵּית־ אָבִ֗י לִהְי֨וֹת לְמֶ֤לֶךְ עַל־ יִשְׂרָאֵל֙ לְעוֹלָ֔ם כִּ֤י בִֽיהוּדָה֙ בָּחַ֣ר לְנָגִ֔יד וּבְבֵ֥ית יְהוּדָ֖ה בֵּ֣ית אָבִ֑י וּבִבְנֵ֣י אָבִ֔י בִּ֣י רָצָ֔ה לְהַמְלִ֖יךְ עַל־ כָּל־ יִשְׂרָאֵֽל׃ (סז) וּמִכָּ֨ל־ בָּנַ֔י כִּ֚י רַבִּ֣ים בָּנִ֔ים נָ֥תַן לִ֖י יְהוָ֑ה וַיִּבְחַר֙ בִּשְׁלֹמֹ֣ה בְנִ֔י לָשֶׁ֗בֶת עַל־ כִּסֵּ֛א מַלְכ֥וּת יְהוָ֖ה עַל־ יִשְׂרָאֵֽל׃ (סח) וַיֹּ֣אמֶר לִ֔י שְׁלֹמֹ֣ה בִנְךָ֔ הֽוּא־ יִבְנֶ֥ה בֵיתִ֖י וַחֲצֵרוֹתָ֑י כִּי־ בָחַ֨רְתִּי ב֥וֹ לִי֙ לְבֵ֔ן וַאֲנִ֖י אֶֽהְיֶה־ לּ֥וֹ לְאָֽב׃ (סט) וַהֲכִינוֹתִ֥י אֶת־ מַלְכוּת֖וֹ עַד־ לְעוֹלָ֑ם אִם־ יֶחֱזַ֗ק לַעֲשׂ֛וֹת מִצְוֺתַ֥י וּמִשְׁפָּטַ֖י כַּיּ֥וֹם הַזֶּֽה׃ (ע) וְ֠עַתָּה לְעֵינֵ֨י כָל־ יִשְׂרָאֵ֤ל קְהַל־ יְהוָה֙ וּבְאָזְנֵ֣י אֱלֹהֵ֔ינוּ שִׁמְר֣וּ וְדִרְשׁ֔וּ כָּל־ מִצְוֺ֖ת יְהוָ֣ה אֱלֹהֵיכֶ֑ם לְמַ֤עַן תִּֽירְשׁוּ֙ אֶת־ הָאָ֣רֶץ הַטּוֹבָ֔ה וְהִנְחַלְתֶּ֛ם לִבְנֵיכֶ֥ם אַחֲרֵיכֶ֖ם עַד־ עוֹלָֽם׃ (עא) וְאַתָּ֣ה שְׁלֹמֹֽה־ בְנִ֡י דַּע֩ אֶת־ אֱלֹהֵ֨י אָבִ֜יךָ וְעָבְדֵ֗הוּ בְּלֵ֤ב שָׁלֵם֙ וּבְנֶ֣פֶשׁ חֲפֵצָ֔ה כִּ֤י כָל־ לְבָבוֹת֙ דּוֹרֵ֣שׁ יְהוָ֔ה וְכָל־ יֵ֥צֶר מַחֲשָׁב֖וֹת מֵבִ֑ין אִֽם־ תִּדְרְשֶׁ֙נּוּ֙ יִמָּ֣צֵא לָ֔ךְ וְאִם־ תַּֽעַזְבֶ֖נּוּ יַזְנִיחֲךָ֥ לָעַֽד׃
