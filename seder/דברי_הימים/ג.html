דברי הימים סדר ג

פתיחה באתר · ספר דברי הימים כולו ·

(א) וְֽאַהֲרֹ֨ן וּבָנָ֜יו מַקְטִירִ֨ים עַל־ מִזְבַּ֤ח הָֽעוֹלָה֙ וְעַל־ מִזְבַּ֣ח הַקְּטֹ֔רֶת לְכֹ֕ל מְלֶ֖אכֶת קֹ֣דֶשׁ הַקֳּדָשִׁ֑ים וּלְכַפֵּר֙ עַל־ יִשְׂרָאֵ֔ל כְּכֹל֙ אֲשֶׁ֣ר צִוָּ֔ה מֹשֶׁ֖ה עֶ֥בֶד הָאֱלֹהִֽים׃ (ב) וְאֵ֖לֶּה בְּנֵ֣י אַהֲרֹ֑ן אֶלְעָזָ֥ר בְּנ֛וֹ פִּֽינְחָ֥ס בְּנ֖וֹ אֲבִישׁ֥וּעַ בְּנֽוֹ׃ (ג) בֻּקִּ֥י בְנ֛וֹ עֻזִּ֥י בְנ֖וֹ זְרַֽחְיָ֥ה בְנֽוֹ׃ (ד) מְרָי֥וֹת בְּנ֛וֹ אֲמַרְיָ֥ה בְנ֖וֹ אֲחִיט֥וּב בְּנֽוֹ׃ (ה) צָד֥וֹק בְּנ֖וֹ אֲחִימַ֥עַץ בְּנֽוֹ׃ (ו) וְאֵ֙לֶּה֙ מוֹשְׁבוֹתָ֔ם לְטִירוֹתָ֖ם בִּגְבוּלָ֑ם לִבְנֵ֤י אַהֲרֹן֙ לְמִשְׁפַּחַ֣ת הַקְּהָתִ֔י כִּ֥י לָהֶ֖ם הָיָ֥ה הַגּוֹרָֽל׃ (ז) וַֽיִּתְּנ֥וּ לָהֶ֛ם אֶת־ חֶבְר֖וֹן בְּאֶ֣רֶץ יְהוּדָ֑ה וְאֶת־ מִגְרָשֶׁ֖יהָ סְבִיבֹתֶֽיהָ׃ (ח) וְאֶת־ שְׂדֵ֥ה הָעִ֖יר וְאֶת־ חֲצֵרֶ֑יהָ נָתְנ֖וּ לְכָלֵ֥ב בֶּן־ יְפֻנֶּֽה׃ (ט) וְלִבְנֵ֣י אַהֲרֹ֗ן נָתְנוּ֙ אֶת־ עָרֵ֣י הַמִּקְלָ֔ט אֶת־ חֶבְר֥וֹן וְאֶת־ לִבְנָ֖ה וְאֶת־ מִגְרָשֶׁ֑יהָ וְאֶת־ יַתִּ֥ר וְאֶֽת־ אֶשְׁתְּמֹ֖עַ וְאֶת־ מִגְרָשֶֽׁיהָ׃ (י) וְאֶת־ חִילֵז֙ וְאֶת־ מִגְרָשֶׁ֔יהָ אֶת־ דְּבִ֖יר וְאֶת־ מִגְרָשֶֽׁיהָ׃ (יא) וְאֶת־ עָשָׁן֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ בֵּ֥ית שֶׁ֖מֶשׁ וְאֶת־ מִגְרָשֶֽׁיהָ׃ (יב) וּמִמַּטֵּ֣ה בִנְיָמִ֗ן אֶת־ גֶּ֤בַע וְאֶת־ מִגְרָשֶׁ֙יהָ֙ וְאֶת־ עָלֶ֣מֶת וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ עֲנָת֖וֹת וְאֶת־ מִגְרָשֶׁ֑יהָ כָּל־ עָרֵיהֶ֛ם שְׁלֹשׁ־ עֶשְׂרֵ֥ה עִ֖יר בְּמִשְׁפְּחוֹתֵיהֶֽם׃ (יג) וְלִבְנֵ֨י קְהָ֜ת הַנּוֹתָרִ֗ים מִמִּשְׁפַּ֣חַת הַמַּטֶּ֡ה מִֽ֠מַּחֲצִית מַטֵּ֨ה חֲצִ֧י מְנַשֶּׁ֛ה בַּגּוֹרָ֖ל עָרִ֥ים עָֽשֶׂר׃ (יד) וְלִבְנֵ֨י גֵרְשׁ֜וֹם לְמִשְׁפְּחוֹתָ֗ם מִמַּטֵּ֣ה יִ֠שָׂשכָר וּמִמַּטֵּ֨ה אָשֵׁ֜ר וּמִמַּטֵּ֣ה נַפְתָּלִ֗י וּמִמַּטֵּ֤ה מְנַשֶּׁה֙ בַּבָּשָׁ֔ן עָרִ֖ים שְׁלֹ֥שׁ עֶשְׂרֵֽה׃ (טו) לִבְנֵ֨י מְרָרִ֜י לְמִשְׁפְּחוֹתָ֗ם מִמַּטֵּ֣ה רְ֠אוּבֵן וּֽמִמַּטֵּה־ גָ֞ד וּמִמַּטֵּ֤ה זְבוּלֻן֙ בַּגּוֹרָ֔ל עָרִ֖ים שְׁתֵּ֥ים עֶשְׂרֵֽה׃ (טז) וַיִּתְּנ֥וּ בְנֵי־ יִשְׂרָאֵ֖ל לַלְוִיִּ֑ם אֶת־ הֶעָרִ֖ים וְאֶת־ מִגְרְשֵׁיהֶֽם׃ (יז) וַיִּתְּנ֣וּ בַגּוֹרָ֗ל מִמַּטֵּ֤ה בְנֵי־ יְהוּדָה֙ וּמִמַּטֵּ֣ה בְנֵי־ שִׁמְע֔וֹן וּמִמַּטֵּ֖ה בְּנֵ֣י בִנְיָמִ֑ן אֵ֚ת הֶעָרִ֣ים הָאֵ֔לֶּה אֲשֶׁר־ יִקְרְא֥וּ אֶתְהֶ֖ם בְּשֵׁמֽוֹת׃ (יח) וּמִֽמִּשְׁפְּח֖וֹת בְּנֵ֣י קְהָ֑ת וַיְהִי֙ עָרֵ֣י גְבוּלָ֔ם מִמַּטֵּ֖ה אֶפְרָֽיִם׃ (יט) וַיִּתְּנ֨וּ לָהֶ֜ם אֶת־ עָרֵ֧י הַמִּקְלָ֛ט אֶת־ שְׁכֶ֥ם וְאֶת־ מִגְרָשֶׁ֖יהָ בְּהַ֣ר אֶפְרָ֑יִם וְאֶת־ גֶּ֖זֶר וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כ) וְאֶֽת־ יָקְמְעָם֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ בֵּ֥ית חוֹר֖וֹן וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כא) וְאֶת־ אַיָּלוֹן֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ גַּת־ רִמּ֖וֹן וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כב) וּמִֽמַּחֲצִית֙ מַטֵּ֣ה מְנַשֶּׁ֔ה אֶת־ עָנֵר֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ בִּלְעָ֖ם וְאֶת־ מִגְרָשֶׁ֑יהָ לְמִשְׁפַּ֥חַת לִבְנֵי־ קְהָ֖ת הַנּוֹתָרִֽים׃ (כג) לִבְנֵי֮ גֵּרְשׁוֹם֒ מִמִּשְׁפַּ֗חַת חֲצִי֙ מַטֵּ֣ה מְנַשֶּׁ֔ה אֶת־ גּוֹלָ֥ן בַּבָּשָׁ֖ן וְאֶת־ מִגְרָשֶׁ֑יהָ וְאֶת־ עַשְׁתָּר֖וֹת וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כד) וּמִמַּטֵּ֣ה יִשָׂשכָ֔ר אֶת־ קֶ֖דֶשׁ וְאֶת־ מִגְרָשֶׁ֑יהָ אֶת־ דָּבְרַ֖ת וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כה) וְאֶת־ רָאמוֹת֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ עָנֵ֖ם וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כו) וּמִמַּטֵּ֣ה אָשֵׁ֔ר אֶת־ מָשָׁ֖ל וְאֶת־ מִגְרָשֶׁ֑יהָ וְאֶת־ עַבְדּ֖וֹן וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כז) וְאֶת־ חוּקֹק֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ רְחֹ֖ב וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כח) וּמִמַּטֵּ֣ה נַפְתָּלִ֗י אֶת־ קֶ֤דֶשׁ בַּגָּלִיל֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ חַמּ֖וֹן וְאֶת־ מִגְרָשֶׁ֑יהָ וְאֶת־ קִרְיָתַ֖יִם וְאֶת־ מִגְרָשֶֽׁיהָ׃ (כט) לִבְנֵ֣י מְרָרִי֮ הַנּוֹתָרִים֒ מִמַּטֵּ֣ה זְבוּלֻ֔ן אֶת־ רִמּוֹנ֖וֹ וְאֶת־ מִגְרָשֶׁ֑יהָ אֶת־ תָּב֖וֹר וְאֶת־ מִגְרָשֶֽׁיהָ׃ (ל) וּמֵעֵ֜בֶר לְיַרְדֵּ֣ן יְרֵחוֹ֮ לְמִזְרַ֣ח הַיַּרְדֵּן֒ מִמַּטֵּ֣ה רְאוּבֵ֔ן אֶת־ בֶּ֥צֶר בַּמִּדְבָּ֖ר וְאֶת־ מִגְרָשֶׁ֑יהָ וְאֶת־ יַ֖הְצָה וְאֶת־ מִגְרָשֶֽׁיהָ׃ (לא) וְאֶת־ קְדֵמוֹת֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ מֵיפַ֖עַת וְאֶת־ מִגְרָשֶֽׁיהָ׃ (לב) וּמִ֨מַּטֵּה־ גָ֔ד אֶת־ רָאמ֥וֹת בַּגִּלְעָ֖ד וְאֶת־ מִגְרָשֶׁ֑יהָ וְאֶֽת־ מַחֲנַ֖יִם וְאֶת־ מִגְרָשֶֽׁיהָ׃ (לג) וְאֶת־ חֶשְׁבּוֹן֙ וְאֶת־ מִגְרָשֶׁ֔יהָ וְאֶת־ יַעְזֵ֖יר וְאֶת־ מִגְרָשֶֽׁיהָ׃ (לד) וְלִבְנֵ֣י יִשָׂשכָ֗ר תּוֹלָ֧ע וּפוּאָ֛ה (ישיב) יָשׁ֥וּב וְשִׁמְר֖וֹן אַרְבָּעָֽה׃ (לה) וּבְנֵ֣י תוֹלָ֗ע עֻזִּ֡י וּרְפָיָ֡ה וִֽ֠ירִיאֵל וְיַחְמַ֨י וְיִבְשָׂ֜ם וּשְׁמוּאֵ֗ל רָאשִׁ֤ים לְבֵית־ אֲבוֹתָם֙ לְתוֹלָ֔ע גִּבּ֥וֹרֵי חַ֖יִל לְתֹלְדוֹתָ֑ם מִסְפָּרָם֙ בִּימֵ֣י דָוִ֔יד עֶשְׂרִֽים־ וּשְׁנַ֥יִם אֶ֖לֶף וְשֵׁ֥שׁ מֵאֽוֹת׃ (לו) וּבְנֵ֥י עֻזִּ֖י יִֽזְרַֽחְיָ֑ה וּבְנֵ֣י יִֽזְרַֽחְיָ֗ה מִֽיכָאֵ֡ל וְ֠עֹבַדְיָה וְיוֹאֵ֧ל יִשִּׁיָּ֛ה חֲמִשָּׁ֖ה רָאשִׁ֥ים כֻּלָּֽם׃ (לז) וַעֲלֵיהֶ֨ם לְתֹלְדוֹתָ֜ם לְבֵ֣ית אֲבוֹתָ֗ם גְּדוּדֵי֙ צְבָ֣א מִלְחָמָ֔ה שְׁלֹשִׁ֥ים וְשִׁשָּׁ֖ה אָ֑לֶף כִּֽי־ הִרְבּ֥וּ נָשִׁ֖ים וּבָנִֽים׃ (לח) וַאֲחֵיהֶ֗ם לְכֹל֙ מִשְׁפְּח֣וֹת יִשָׂשכָ֔ר גִּבּוֹרֵ֖י חֲיָלִ֑ים שְׁמוֹנִ֤ים וְשִׁבְעָה֙ אֶ֔לֶף הִתְיַחְשָׂ֖ם לַכֹּֽל׃ (לט) בִּנְיָמִ֗ן בֶּ֧לַע וָבֶ֛כֶר וִידִֽיעֲאֵ֖ל שְׁלֹשָֽׁה׃ (מ) וּבְנֵ֣י בֶ֗לַע אֶצְבּ֡וֹן וְעֻזִּ֡י וְ֠עֻזִּיאֵל וִירִימ֨וֹת וְעִירִ֜י חֲמִשָּׁ֗ה רָאשֵׁי֙ בֵּ֣ית אָב֔וֹת גִּבּוֹרֵ֖י חֲיָלִ֑ים וְהִתְיַחְשָׂ֗ם עֶשְׂרִ֤ים וּשְׁנַ֙יִם֙ אֶ֔לֶף וּשְׁלֹשִׁ֖ים וְאַרְבָּעָֽה׃ (מא) וּבְנֵ֣י בֶ֗כֶר זְמִירָ֡ה וְיוֹעָ֡שׁ וֶ֠אֱלִיעֶזֶר וְאֶלְיוֹעֵינַ֤י וְעָמְרִי֙ וִירֵמ֣וֹת וַאֲבִיָּ֔ה וַעֲנָת֖וֹת וְעָלָ֑מֶת כָּל־ אֵ֖לֶּה בְּנֵי־ בָֽכֶר׃ (מב) וְהִתְיַחְשָׂ֣ם לְתֹלְדוֹתָ֗ם רָאשֵׁי֙ בֵּ֣ית אֲבוֹתָ֔ם גִּבּוֹרֵ֖י חָ֑יִל עֶשְׂרִ֥ים אֶ֖לֶף וּמָאתָֽיִם׃ (מג) וּבְנֵ֥י יְדִיעֲאֵ֖ל בִּלְהָ֑ן וּבְנֵ֣י בִלְהָ֗ן (יעיש) יְע֡וּשׁ וּ֠בִנְיָמִן וְאֵה֤וּד וּֽכְנַעֲנָה֙ וְזֵיתָ֔ן וְתַרְשִׁ֖ישׁ וַאֲחִישָֽׁחַר׃ (מד) כָּל־ אֵ֜לֶּה בְּנֵ֤י יְדִֽיעֲאֵל֙ לְרָאשֵׁ֣י הָאָב֔וֹת גִּבּוֹרֵ֖י חֲיָלִ֑ים שִׁבְעָֽה־ עָשָׂ֥ר אֶ֙לֶף֙ וּמָאתַ֔יִם יֹצְאֵ֥י צָבָ֖א לַמִּלְחָמָֽה׃ (מה) וְשֻׁפִּ֤ם וְחֻפִּם֙ בְּנֵ֣י עִ֔יר חֻשִׁ֖ם בְּנֵ֥י אַחֵֽר׃ (מו) בְּנֵ֣י נַפְתָּלִ֗י יַחֲצִיאֵ֧ל וְגוּנִ֛י וְיֵ֥צֶר וְשַׁלּ֖וּם בְּנֵ֥י בִלְהָֽה׃ (מז) בְּנֵ֣י מְנַשֶּׁ֔ה אַשְׂרִיאֵ֖ל אֲשֶׁ֣ר יָלָ֑דָה פִּֽילַגְשׁוֹ֙ הָֽאֲרַמִּיָּ֔ה יָלְדָ֕ה אֶת־ מָכִ֖יר אֲבִ֥י עֵ֣שֶׁק אָחִ֑יו אוּלָ֣ם בְּכֹר֔וֹ יְעוּשׁ֙ הַשֵּׁנִ֔י וֶֽאֱלִיפֶ֖לֶט הַשְּׁלִשִֽׁי׃
