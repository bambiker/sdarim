יחזקאל סדר כג

פתיחה באתר · ספר יחזקאל כולו ·

(א) וְיָֽדְעוּ֙ הַגּוֹיִ֔ם כִּ֚י אֲנִ֣י יְהוָ֔ה מְקַדֵּ֖שׁ אֶת־ יִשְׂרָאֵ֑ל בִּהְי֧וֹת מִקְדָּשִׁ֛י בְּתוֹכָ֖ם לְעוֹלָֽם׃ (ב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ג) בֶּן־ אָדָ֗ם שִׂ֤ים פָּנֶ֙יךָ֙ אֶל־ גּוֹג֙ אֶ֣רֶץ הַמָּג֔וֹג נְשִׂ֕יא רֹ֖אשׁ מֶ֣שֶׁךְ וְתֻבָ֑ל וְהִנָּבֵ֖א עָלָֽיו׃ (ד) וְאָ֣מַרְתָּ֔ כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הִנְנִ֤י אֵלֶ֙יךָ֙ גּ֔וֹג נְשִׂ֕יא רֹ֖אשׁ מֶ֥שֶׁךְ וְתֻבָֽל׃ (ה) וְשׁ֣וֹבַבְתִּ֔יךָ וְנָתַתִּ֥י חַחִ֖ים בִּלְחָיֶ֑יךָ וְהוֹצֵאתִי֩ אוֹתְךָ֨ וְאֶת־ כָּל־ חֵילֶ֜ךָ סוּסִ֣ים וּפָרָשִׁ֗ים לְבֻשֵׁ֤י מִכְלוֹל֙ כֻּלָּ֔ם קָהָ֥ל רָב֙ צִנָּ֣ה וּמָגֵ֔ן תֹּפְשֵׂ֥י חֲרָב֖וֹת כֻּלָּֽם׃ (ו) פָּרַ֛ס כּ֥וּשׁ וּפ֖וּט אִתָּ֑ם כֻּלָּ֖ם מָגֵ֥ן וְכוֹבָֽע׃ (ז) גֹּ֚מֶר וְכָל־ אֲגַפֶּ֔יהָ בֵּ֚ית תּֽוֹגַרְמָ֔ה יַרְכְּתֵ֥י צָפ֖וֹן וְאֶת־ כָּל־ אֲגַפָּ֑יו עַמִּ֥ים רַבִּ֖ים אִתָּֽךְ׃ (ח) הִכֹּן֙ וְהָכֵ֣ן לְךָ֔ אַתָּ֕ה וְכָל־ קְהָלֶ֖ךָ הַנִּקְהָלִ֣ים עָלֶ֑יךָ וְהָיִ֥יתָ לָהֶ֖ם לְמִשְׁמָֽר׃ (ט) מִיָּמִ֣ים רַבִּים֮ תִּפָּקֵד֒ בְּאַחֲרִ֨ית הַשָּׁנִ֜ים תָּב֣וֹא ׀ אֶל־ אֶ֣רֶץ ׀ מְשׁוֹבֶ֣בֶת מֵחֶ֗רֶב מְקֻבֶּ֙צֶת֙ מֵעַמִּ֣ים רַבִּ֔ים עַ֚ל הָרֵ֣י יִשְׂרָאֵ֔ל אֲשֶׁר־ הָי֥וּ לְחָרְבָּ֖ה תָּמִ֑יד וְהִיא֙ מֵעַמִּ֣ים הוּצָ֔אָה וְיָשְׁב֥וּ לָבֶ֖טַח כֻּלָּֽם׃ (י) וְעָלִ֙יתָ֙ כַּשֹּׁאָ֣ה תָב֔וֹא כֶּעָנָ֛ן לְכַסּ֥וֹת הָאָ֖רֶץ תִּֽהְיֶ֑ה אַתָּה֙ וְכָל־ אֲגַפֶּ֔יךָ וְעַמִּ֥ים רַבִּ֖ים אוֹתָֽךְ׃ (יא) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא יַעֲל֤וּ דְבָרִים֙ עַל־ לְבָבֶ֔ךָ וְחָשַׁבְתָּ֖ מַחֲשֶׁ֥בֶת רָעָֽה׃ (יב) וְאָמַרְתָּ֗ אֶֽעֱלֶה֙ עַל־ אֶ֣רֶץ פְּרָז֔וֹת אָבוֹא֙ הַשֹּׁ֣קְטִ֔ים יֹשְׁבֵ֖י לָבֶ֑טַח כֻּלָּ֗ם יֹֽשְׁבִים֙ בְּאֵ֣ין חוֹמָ֔ה וּבְרִ֥יחַ וּדְלָתַ֖יִם אֵ֥ין לָהֶֽם׃ (יג) לִשְׁלֹ֥ל שָׁלָ֖ל וְלָבֹ֣ז בַּ֑ז לְהָשִׁ֨יב יָדְךָ֜ עַל־ חֳרָב֣וֹת נוֹשָׁבֹ֗ת וְאֶל־ עַם֙ מְאֻסָּ֣ף מִגּוֹיִ֔ם עֹשֶׂה֙ מִקְנֶ֣ה וְקִנְיָ֔ן יֹשְׁבֵ֖י עַל־ טַבּ֥וּר הָאָֽרֶץ׃ (יד) שְׁבָ֡א וּ֠דְדָן וְסֹחֲרֵ֨י תַרְשִׁ֤ישׁ וְכָל־ כְּפִרֶ֙יהָ֙ יֹאמְר֣וּ לְךָ֔ הֲלִשְׁלֹ֤ל שָׁלָל֙ אַתָּ֣ה בָ֔א הֲלָבֹ֥ז בַּ֖ז הִקְהַ֣לְתָּ קְהָלֶ֑ךָ לָשֵׂ֣את ׀ כֶּ֣סֶף וְזָהָ֗ב לָקַ֙חַת֙ מִקְנֶ֣ה וְקִנְיָ֔ן לִשְׁלֹ֖ל שָׁלָ֥ל גָּדֽוֹל׃ (טו) לָכֵן֙ הִנָּבֵ֣א בֶן־ אָדָ֔ם וְאָמַרְתָּ֣ לְג֔וֹג כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הֲל֣וֹא ׀ בַּיּ֣וֹם הַה֗וּא בְּשֶׁ֨בֶת עַמִּ֧י יִשְׂרָאֵ֛ל לָבֶ֖טַח תֵּדָֽע׃ (טז) וּבָ֤אתָ מִמְּקֽוֹמְךָ֙ מִיַּרְכְּתֵ֣י צָפ֔וֹן אַתָּ֕ה וְעַמִּ֥ים רַבִּ֖ים אִתָּ֑ךְ רֹכְבֵ֤י סוּסִים֙ כֻּלָּ֔ם קָהָ֥ל גָּד֖וֹל וְחַ֥יִל רָֽב׃ (יז) וְעָלִ֙יתָ֙ עַל־ עַמִּ֣י יִשְׂרָאֵ֔ל כֶּֽעָנָ֖ן לְכַסּ֣וֹת הָאָ֑רֶץ בְּאַחֲרִ֨ית הַיָּמִ֜ים תִּֽהְיֶ֗ה וַהֲבִאוֹתִ֙יךָ֙ עַל־ אַרְצִ֔י לְמַעַן֩ דַּ֨עַת הַגּוֹיִ֜ם אֹתִ֗י בְּהִקָּדְשִׁ֥י בְךָ֛ לְעֵינֵיהֶ֖ם גּֽוֹג׃ (יח) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה הַֽאַתָּה־ ה֨וּא אֲשֶׁר־ דִּבַּ֜רְתִּי בְּיָמִ֣ים קַדְמוֹנִ֗ים בְּיַד֙ עֲבָדַי֙ נְבִיאֵ֣י יִשְׂרָאֵ֔ל הַֽנִּבְּאִ֛ים בַּיָּמִ֥ים הָהֵ֖ם שָׁנִ֑ים לְהָבִ֥יא אֹתְךָ֖ עֲלֵיהֶֽם׃ (יט) וְהָיָ֣ה ׀ בַּיּ֣וֹם הַה֗וּא בְּי֨וֹם בּ֥וֹא גוֹג֙ עַל־ אַדְמַ֣ת יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה תַּעֲלֶ֥ה חֲמָתִ֖י בְּאַפִּֽי׃ (כ) וּבְקִנְאָתִ֥י בְאֵשׁ־ עֶבְרָתִ֖י דִּבַּ֑רְתִּי אִם־ לֹ֣א ׀ בַּיּ֣וֹם הַה֗וּא יִֽהְיֶה֙ רַ֣עַשׁ גָּד֔וֹל עַ֖ל אַדְמַ֥ת יִשְׂרָאֵֽל׃ (כא) וְרָעֲשׁ֣וּ מִפָּנַ֡י דְּגֵ֣י הַיָּם֩ וְע֨וֹף הַשָּׁמַ֜יִם וְחַיַּ֣ת הַשָּׂדֶ֗ה וְכָל־ הָרֶ֙מֶשׂ֙ הָרֹמֵ֣שׂ עַל־ הָֽאֲדָמָ֔ה וְכֹל֙ הָֽאָדָ֔ם אֲשֶׁ֖ר עַל־ פְּנֵ֣י הָאֲדָמָ֑ה וְנֶהֶרְס֣וּ הֶהָרִ֗ים וְנָֽפְלוּ֙ הַמַּדְרֵג֔וֹת וְכָל־ חוֹמָ֖ה לָאָ֥רֶץ תִּפּֽוֹל׃ (כב) וְקָרָ֨אתִי עָלָ֤יו לְכָל־ הָרַי֙ חֶ֔רֶב נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה חֶ֥רֶב אִ֖ישׁ בְּאָחִ֥יו תִּֽהְיֶֽה׃ (כג) וְנִשְׁפַּטְתִּ֥י אִתּ֖וֹ בְּדֶ֣בֶר וּבְדָ֑ם וְגֶ֣שֶׁם שׁוֹטֵף֩ וְאַבְנֵ֨י אֶלְגָּבִ֜ישׁ אֵ֣שׁ וְגָפְרִ֗ית אַמְטִ֤יר עָלָיו֙ וְעַל־ אֲגַפָּ֔יו וְעַל־ עַמִּ֥ים רַבִּ֖ים אֲשֶׁ֥ר אִתּֽוֹ׃ (כד) וְהִתְגַּדִּלְתִּי֙ וְהִתְקַדִּשְׁתִּ֔י וְנ֣וֹדַעְתִּ֔י לְעֵינֵ֖י גּוֹיִ֣ם רַבִּ֑ים וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כה) וְאַתָּ֤ה בֶן־ אָדָם֙ הִנָּבֵ֣א עַל־ גּ֔וֹג וְאָ֣מַרְתָּ֔ כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הִנְנִ֤י אֵלֶ֙יךָ֙ גּ֔וֹג נְשִׂ֕יא רֹ֖אשׁ מֶ֥שֶׁךְ וְתֻבָֽל׃ (כו) וְשֹׁבַבְתִּ֙יךָ֙ וְשִׁשֵּׁאתִ֔יךָ וְהַעֲלִיתִ֖יךָ מִיַּרְכְּתֵ֣י צָפ֑וֹן וַהֲבִאוֹתִ֖ךָ עַל־ הָרֵ֥י יִשְׂרָאֵֽל׃ (כז) וְהִכֵּיתִ֥י קַשְׁתְּךָ֖ מִיַּ֣ד שְׂמֹאולֶ֑ךָ וְחִצֶּ֕יךָ מִיַּ֥ד יְמִינְךָ֖ אַפִּֽיל׃ (כח) עַל־ הָרֵ֨י יִשְׂרָאֵ֜ל תִּפּ֗וֹל אַתָּה֙ וְכָל־ אֲגַפֶּ֔יךָ וְעַמִּ֖ים אֲשֶׁ֣ר אִתָּ֑ךְ לְעֵ֨יט צִפּ֧וֹר כָּל־ כָּנָ֛ף וְחַיַּ֥ת הַשָּׂדֶ֖ה נְתַתִּ֥יךָ לְאָכְלָֽה׃ (כט) עַל־ פְּנֵ֥י הַשָּׂדֶ֖ה תִּפּ֑וֹל כִּ֚י אֲנִ֣י דִבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ל) וְשִׁלַּחְתִּי־ אֵ֣שׁ בְּמָג֔וֹג וּבְיֹשְׁבֵ֥י הָאִיִּ֖ים לָבֶ֑טַח וְיָדְע֖וּ כִּי־ אֲנִ֥י יְהוָֽה׃ (לא) וְאֶת־ שֵׁ֨ם קָדְשִׁ֜י אוֹדִ֗יעַ בְּתוֹךְ֙ עַמִּ֣י יִשְׂרָאֵ֔ל וְלֹֽא־ אַחֵ֥ל אֶת־ שֵׁם־ קָדְשִׁ֖י ע֑וֹד וְיָדְע֤וּ הַגּוֹיִם֙ כִּי־ אֲנִ֣י יְהוָ֔ה קָד֖וֹשׁ בְּיִשְׂרָאֵֽל׃ (לב) הִנֵּ֤ה בָאָה֙ וְנִֽהְיָ֔תָה נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה ה֥וּא הַיּ֖וֹם אֲשֶׁ֥ר דִּבַּֽרְתִּי׃ (לג) וְֽיָצְא֞וּ יֹשְׁבֵ֣י ׀ עָרֵ֣י יִשְׂרָאֵ֗ל וּבִעֲר֡וּ וְ֠הִשִּׂיקוּ בְּנֶ֨שֶׁק וּמָגֵ֤ן וְצִנָּה֙ בְּקֶ֣שֶׁת וּבְחִצִּ֔ים וּבְמַקֵּ֥ל יָ֖ד וּבְרֹ֑מַח וּבִעֲר֥וּ בָהֶ֛ם אֵ֖שׁ שֶׁ֥בַע שָׁנִֽים׃ (לד) וְלֹֽא־ יִשְׂא֨וּ עֵצִ֜ים מִן־ הַשָּׂדֶ֗ה וְלֹ֤א יַחְטְבוּ֙ מִן־ הַיְּעָרִ֔ים כִּ֥י בַנֶּ֖שֶׁק יְבַֽעֲרוּ־ אֵ֑שׁ וְשָׁלְל֣וּ אֶת־ שֹׁלְלֵיהֶ֗ם וּבָֽזְזוּ֙ אֶת־ בֹּ֣זְזֵיהֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לה) וְהָיָ֣ה בַיּ֣וֹם הַה֡וּא אֶתֵּ֣ן לְגוֹג֩ ׀ מְקֽוֹם־ שָׁ֨ם קֶ֜בֶר בְּיִשְׂרָאֵ֗ל גֵּ֤י הָעֹֽבְרִים֙ קִדְמַ֣ת הַיָּ֔ם וְחֹסֶ֥מֶת הִ֖יא אֶת־ הָעֹֽבְרִ֑ים וְקָ֣בְרוּ שָׁ֗ם אֶת־ גּוֹג֙ וְאֶת־ כָּל־ הֲמוֹנֹ֔ה וְקָ֣רְא֔וּ גֵּ֖יא הֲמ֥וֹן גּֽוֹג׃ (לו) וּקְבָרוּם֙ בֵּ֣ית יִשְׂרָאֵ֔ל לְמַ֖עַן טַהֵ֣ר אֶת־ הָאָ֑רֶץ שִׁבְעָ֖ה חֳדָשִֽׁים׃ (לז) וְקָֽבְרוּ֙ כָּל־ עַ֣ם הָאָ֔רֶץ וְהָיָ֥ה לָהֶ֖ם לְשֵׁ֑ם י֚וֹם הִכָּ֣בְדִ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לח) וְאַנְשֵׁ֨י תָמִ֤יד יַבְדִּ֙ילוּ֙ עֹבְרִ֣ים בָּאָ֔רֶץ מְקַבְּרִ֣ים אֶת־ הָעֹבְרִ֗ים אֶת־ הַנּוֹתָרִ֛ים עַל־ פְּנֵ֥י הָאָ֖רֶץ לְטַֽהֲרָ֑הּ מִקְצֵ֥ה שִׁבְעָֽה־ חֳדָשִׁ֖ים יַחְקֹֽרוּ׃ (לט) וְעָבְר֤וּ הָעֹֽבְרִים֙ בָּאָ֔רֶץ וְרָאָה֙ עֶ֣צֶם אָדָ֔ם וּבָנָ֥ה אֶצְל֖וֹ צִיּ֑וּן עַ֣ד קָבְר֤וּ אֹתוֹ֙ הַֽמְקַבְּרִ֔ים אֶל־ גֵּ֖יא הֲמ֥וֹן גּֽוֹג׃ (מ) וְגַ֥ם שֶׁם־ עִ֛יר הֲמוֹנָ֖ה וְטִהֲר֥וּ הָאָֽרֶץ׃ (מא) וְאַתָּ֨ה בֶן־ אָדָ֜ם כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה אֱמֹר֩ לְצִפּ֨וֹר כָּל־ כָּנָ֜ף וּלְכֹ֣ל ׀ חַיַּ֣ת הַשָּׂדֶ֗ה הִקָּבְצ֤וּ וָבֹ֙אוּ֙ הֵאָסְפ֣וּ מִסָּבִ֔יב עַל־ זִבְחִ֗י אֲשֶׁ֨ר אֲנִ֜י זֹבֵ֤חַ לָכֶם֙ זֶ֣בַח גָּד֔וֹל עַ֖ל הָרֵ֣י יִשְׂרָאֵ֑ל וַאֲכַלְתֶּ֥ם בָּשָׂ֖ר וּשְׁתִ֥יתֶם דָּֽם׃ (מב) בְּשַׂ֤ר גִּבּוֹרִים֙ תֹּאכֵ֔לוּ וְדַם־ נְשִׂיאֵ֥י הָאָ֖רֶץ תִּשְׁתּ֑וּ אֵילִ֨ים כָּרִ֤ים וְעַתּוּדִים֙ פָּרִ֔ים מְרִיאֵ֥י בָשָׁ֖ן כֻּלָּֽם׃ (מג) וַאֲכַלְתֶּם־ חֵ֣לֶב לְשָׂבְעָ֔ה וּשְׁתִ֥יתֶם דָּ֖ם לְשִׁכָּר֑וֹן מִזִּבְחִ֖י אֲשֶׁר־ זָבַ֥חְתִּי לָכֶֽם׃ (מד) וּשְׂבַעְתֶּ֤ם עַל־ שֻׁלְחָנִי֙ ס֣וּס וָרֶ֔כֶב גִּבּ֖וֹר וְכָל־ אִ֣ישׁ מִלְחָמָ֑ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (מה) וְנָתַתִּ֥י אֶת־ כְּבוֹדִ֖י בַּגּוֹיִ֑ם וְרָא֣וּ כָל־ הַגּוֹיִ֗ם אֶת־ מִשְׁפָּטִי֙ אֲשֶׁ֣ר עָשִׂ֔יתִי וְאֶת־ יָדִ֖י אֲשֶׁר־ שַׂ֥מְתִּי בָהֶֽם׃
