יחזקאל סדר יב

פתיחה באתר · ספר יחזקאל כולו ·

(א) בְּרֵ֣יחַ נִיחֹחַ֮ אֶרְצֶ֣ה אֶתְכֶם֒ בְּהוֹצִיאִ֤י אֶתְכֶם֙ מִן־ הָ֣עַמִּ֔ים וְקִבַּצְתִּ֣י אֶתְכֶ֔ם מִן־ הָ֣אֲרָצ֔וֹת אֲשֶׁ֥ר נְפֹצֹתֶ֖ם בָּ֑ם וְנִקְדַּשְׁתִּ֥י בָכֶ֖ם לְעֵינֵ֥י הַגּוֹיִֽם׃ (ב) וִֽידַעְתֶּם֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה בַּהֲבִיאִ֥י אֶתְכֶ֖ם אֶל־ אַדְמַ֣ת יִשְׂרָאֵ֑ל אֶל־ הָאָ֗רֶץ אֲשֶׁ֤ר נָשָׂ֙אתִי֙ אֶת־ יָדִ֔י לָתֵ֥ת אוֹתָ֖הּ לַאֲבֽוֹתֵיכֶֽם׃ (ג) וּזְכַרְתֶּם־ שָׁ֗ם אֶת־ דַּרְכֵיכֶם֙ וְאֵת֙ כָּל־ עֲלִיל֣וֹתֵיכֶ֔ם אֲשֶׁ֥ר נִטְמֵאתֶ֖ם בָּ֑ם וּנְקֹֽטֹתֶם֙ בִּפְנֵיכֶ֔ם בְּכָל־ רָעוֹתֵיכֶ֖ם אֲשֶׁ֥ר עֲשִׂיתֶֽם׃ (ד) וִֽידַעְתֶּם֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה בַּעֲשׂוֹתִ֥י אִתְּכֶ֖ם לְמַ֣עַן שְׁמִ֑י לֹא֩ כְדַרְכֵיכֶ֨ם הָרָעִ֜ים וְכַעֲלִילֽוֹתֵיכֶ֤ם הַנִּשְׁחָתוֹת֙ בֵּ֣ית יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ה) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ו) בֶּן־ אָדָ֗ם שִׂ֤ים פָּנֶ֙יךָ֙ דֶּ֣רֶךְ תֵּימָ֔נָה וְהַטֵּ֖ף אֶל־ דָּר֑וֹם וְהִנָּבֵ֛א אֶל־ יַ֥עַר הַשָּׂדֶ֖ה נֶֽגֶב׃ (ז) וְאָֽמַרְתָּ֙ לְיַ֣עַר הַנֶּ֔גֶב שְׁמַ֖ע דְּבַר־ יְהוָ֑ה כֹּֽה־ אָמַ֣ר אֲדֹנָ֣י יְהוִ֡ה הִנְנִ֣י מַֽצִּית־ בְּךָ֣ ׀ אֵ֡שׁ וְאָכְלָ֣ה בְךָ֣ כָל־ עֵֽץ־ לַח֩ וְכָל־ עֵ֨ץ יָבֵ֤שׁ לֹֽא־ תִכְבֶּה֙ לַהֶ֣בֶת שַׁלְהֶ֔בֶת וְנִצְרְבוּ־ בָ֥הּ כָּל־ פָּנִ֖ים מִנֶּ֥גֶב צָפֽוֹנָה׃ (ח) וְרָאוּ֙ כָּל־ בָּשָׂ֔ר כִּ֛י אֲנִ֥י יְהוָ֖ה בִּֽעַרְתִּ֑יהָ לֹ֖א תִּכְבֶּֽה׃ (ט) וָאֹמַ֕ר אֲהָ֖הּ אֲדֹנָ֣י יְהוִ֑ה הֵ֚מָּה אֹמְרִ֣ים לִ֔י הֲלֹ֛א מְמַשֵּׁ֥ל מְשָׁלִ֖ים הֽוּא׃ (י) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יא) בֶּן־ אָדָ֗ם שִׂ֤ים פָּנֶ֙יךָ֙ אֶל־ יְר֣וּשָׁלִַ֔ם וְהַטֵּ֖ף אֶל־ מִקְדָּשִׁ֑ים וְהִנָּבֵ֖א אֶל־ אַדְמַ֥ת יִשְׂרָאֵֽל׃ (יב) וְאָמַרְתָּ֞ לְאַדְמַ֣ת יִשְׂרָאֵ֗ל כֹּ֚ה אָמַ֣ר יְהוָ֔ה הִנְנִ֣י אֵלַ֔יִךְ וְהוֹצֵאתִ֥י חַרְבִּ֖י מִתַּעְרָ֑הּ וְהִכְרַתִּ֥י מִמֵּ֖ךְ צַדִּ֥יק וְרָשָֽׁע׃ (יג) יַ֛עַן אֲשֶׁר־ הִכְרַ֥תִּי מִמֵּ֖ךְ צַדִּ֣יק וְרָשָׁ֑ע לָ֠כֵן תֵּצֵ֨א חַרְבִּ֧י מִתַּעְרָ֛הּ אֶל־ כָּל־ בָּשָׂ֖ר מִנֶּ֥גֶב צָפֽוֹן׃ (יד) וְיָֽדְעוּ֙ כָּל־ בָּשָׂ֔ר כִּ֚י אֲנִ֣י יְהוָ֔ה הוֹצֵ֥אתִי חַרְבִּ֖י מִתַּעְרָ֑הּ לֹ֥א תָשׁ֖וּב עֽוֹד׃ (טו) וְאַתָּ֥ה בֶן־ אָדָ֖ם הֵֽאָנַ֑ח בְּשִׁבְר֤וֹן מָתְנַ֙יִם֙ וּבִמְרִיר֔וּת תֵּֽאָנַ֖ח לְעֵינֵיהֶֽם׃ (טז) וְהָיָה֙ כִּֽי־ יֹאמְר֣וּ אֵלֶ֔יךָ עַל־ מָ֖ה אַתָּ֣ה נֶאֱנָ֑ח וְאָמַרְתָּ֡ אֶל־ שְׁמוּעָ֣ה כִֽי־ בָאָ֡ה וְנָמֵ֣ס כָּל־ לֵב֩ וְרָפ֨וּ כָל־ יָדַ֜יִם וְכִהֲתָ֣ה כָל־ ר֗וּחַ וְכָל־ בִּרְכַּ֙יִם֙ תֵּלַ֣כְנָה מַּ֔יִם הִנֵּ֤ה בָאָה֙ וְנִֽהְיָ֔תָה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (יח) בֶּן־ אָדָ֕ם הִנָּבֵא֙ וְאָ֣מַרְתָּ֔ כֹּ֖ה אָמַ֣ר אֲדֹנָ֑י אֱמֹ֕ר חֶ֥רֶב חֶ֛רֶב הוּחַ֖דָּה וְגַם־ מְרוּטָֽה׃ (יט) לְמַ֨עַן טְבֹ֤חַ טֶ֙בַח֙ הוּחַ֔דָּה לְמַעַן־ הֱיֵה־ לָ֥הּ בָּ֖רָק מֹרָ֑טָּה א֣וֹ נָשִׂ֔ישׂ שֵׁ֥בֶט בְּנִ֖י מֹאֶ֥סֶת כָּל־ עֵֽץ׃ (כ) וַיִּתֵּ֥ן אֹתָ֛הּ לְמָרְטָ֖ה לִתְפֹּ֣שׂ בַּכָּ֑ף הִֽיא־ הוּחַ֤דָּה חֶ֙רֶב֙ וְהִ֣יא מֹרָ֔טָּה לָתֵ֥ת אוֹתָ֖הּ בְּיַד־ הוֹרֵֽג׃ (כא) זְעַ֤ק וְהֵילֵל֙ בֶּן־ אָדָ֔ם כִּי־ הִיא֙ הָיתָ֣ה בְעַמִּ֔י הִ֖יא בְּכָל־ נְשִׂיאֵ֣י יִשְׂרָאֵ֑ל מְגוּרֵ֤י אֶל־ חֶ֙רֶב֙ הָי֣וּ אֶת־ עַמִּ֔י לָכֵ֖ן סְפֹ֥ק אֶל־ יָרֵֽךְ׃ (כב) כִּ֣י בֹ֔חַן וּמָ֕ה אִם־ גַּם־ שֵׁ֥בֶט מֹאֶ֖סֶת לֹ֣א יִֽהְיֶ֑ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (כג) וְאַתָּ֣ה בֶן־ אָדָ֔ם הִנָּבֵ֕א וְהַ֖ךְ כַּ֣ף אֶל־ כָּ֑ף וְתִכָּפֵ֞ל חֶ֤רֶב שְׁלִישִׁ֙תָה֙ חֶ֣רֶב חֲלָלִ֔ים הִ֗יא חֶ֚רֶב חָלָ֣ל הַגָּד֔וֹל הַחֹדֶ֖רֶת לָהֶֽם׃ (כד) לְמַ֣עַן ׀ לָמ֣וּג לֵ֗ב וְהַרְבֵּה֙ הַמִּכְשֹׁלִ֔ים עַ֚ל כָּל־ שַׁ֣עֲרֵיהֶ֔ם נָתַ֖תִּי אִבְחַת־ חָ֑רֶב אָ֛ח עֲשׂוּיָ֥ה לְבָרָ֖ק מְעֻטָּ֥ה לְטָֽבַח׃ (כה) הִתְאַחֲדִ֥י הֵימִ֖נִי הָשִׂ֣ימִי הַשְׂמִ֑ילִי אָ֖נָה פָּנַ֥יִךְ מֻעָדֽוֹת׃ (כו) וְגַם־ אֲנִ֗י אַכֶּ֤ה כַפִּי֙ אֶל־ כַּפִּ֔י וַהֲנִחֹתִ֖י חֲמָתִ֑י אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (כז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כח) וְאַתָּ֨ה בֶן־ אָדָ֜ם שִׂים־ לְךָ֣ ׀ שְׁנַ֣יִם דְּרָכִ֗ים לָבוֹא֙ חֶ֣רֶב מֶֽלֶךְ־ בָּבֶ֔ל מֵאֶ֥רֶץ אֶחָ֖ד יֵצְא֣וּ שְׁנֵיהֶ֑ם וְיָ֣ד בָּרֵ֔א בְּרֹ֥אשׁ דֶּֽרֶךְ־ עִ֖יר בָּרֵֽא׃ (כט) דֶּ֣רֶךְ תָּשִׂ֔ים לָב֣וֹא חֶ֔רֶב אֵ֖ת רַבַּ֣ת בְּנֵֽי־ עַמּ֑וֹן וְאֶת־ יְהוּדָ֥ה בִירוּשָׁלִַ֖ם בְּצוּרָֽה׃ (ל) כִּֽי־ עָמַ֨ד מֶלֶךְ־ בָּבֶ֜ל אֶל־ אֵ֣ם הַדֶּ֗רֶךְ בְּרֹ֛אשׁ שְׁנֵ֥י הַדְּרָכִ֖ים לִקְסָם־ קָ֑סֶם קִלְקַ֤ל בַּֽחִצִּים֙ שָׁאַ֣ל בַּתְּרָפִ֔ים רָאָ֖ה בַּכָּבֵֽד׃ (לא) בִּֽימִינ֞וֹ הָיָ֣ה ׀ הַקֶּ֣סֶם יְרוּשָׁלִַ֗ם לָשׂ֤וּם כָּרִים֙ לִפְתֹּ֤חַ פֶּה֙ בְּרֶ֔צַח לְהָרִ֥ים ק֖וֹל בִּתְרוּעָ֑ה לָשׂ֤וּם כָּרִים֙ עַל־ שְׁעָרִ֔ים לִשְׁפֹּ֥ךְ סֹלְלָ֖ה לִבְנ֥וֹת דָּיֵֽק׃ (לב) וְהָיָ֨ה לָהֶ֤ם (כקסום)־ כִּקְסָם־ שָׁוְא֙ בְּעֵ֣ינֵיהֶ֔ם שְׁבֻעֵ֥י שְׁבֻע֖וֹת לָהֶ֑ם וְהֽוּא־ מַזְכִּ֥יר עָוֺ֖ן לְהִתָּפֵֽשׂ׃ (לג) לָכֵ֗ן כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ יַ֗עַן הַזְכַּרְכֶם֙ עֲוֺ֣נְכֶ֔ם בְּהִגָּל֣וֹת פִּשְׁעֵיכֶ֗ם לְהֵֽרָאוֹת֙ חַטֹּ֣אותֵיכֶ֔ם בְּכֹ֖ל עֲלִילֽוֹתֵיכֶ֑ם יַ֚עַן הִזָּ֣כֶרְכֶ֔ם בַּכַּ֖ף תִּתָּפֵֽשׂוּ׃ (לד) וְאַתָּה֙ חָלָ֣ל רָשָׁ֔ע נְשִׂ֖יא יִשְׂרָאֵ֑ל אֲשֶׁר־ בָּ֣א יוֹמ֔וֹ בְּעֵ֖ת עֲוֺ֥ן קֵֽץ׃ (לה) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הָסִיר֙ הַמִּצְנֶ֔פֶת וְהָרִ֖ים הָֽעֲטָרָ֑ה זֹ֣את לֹא־ זֹ֔את הַשָּׁפָ֣לָה הַגְבֵּ֔הַ וְהַגָּבֹ֖הַ הַשְׁפִּֽיל׃ (לו) עַוָּ֥ה עַוָּ֖ה עַוָּ֣ה אֲשִׂימֶ֑נָּה גַּם־ זֹאת֙ לֹ֣א הָיָ֔ה עַד־ בֹּ֛א אֲשֶׁר־ ל֥וֹ הַמִּשְׁפָּ֖ט וּנְתַתִּֽיו׃ (לז) וְאַתָּ֣ה בֶן־ אָדָ֗ם הִנָּבֵ֤א וְאָֽמַרְתָּ֙ כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהֹוִ֔ה אֶל־ בְּנֵ֥י עַמּ֖וֹן וְאֶל־ חֶרְפָּתָ֑ם וְאָמַרְתָּ֗ חֶ֣רֶב חֶ֤רֶב פְּתוּחָה֙ לְטֶ֣בַח מְרוּטָ֔ה לְהָכִ֖יל לְמַ֥עַן בָּרָֽק׃ (לח) בַּחֲז֥וֹת לָךְ֙ שָׁ֔וְא בִּקְסָם־ לָ֖ךְ כָּזָ֑ב לָתֵ֣ת אוֹתָ֗ךְ אֶֽל־ צַוְּארֵי֙ חַֽלְלֵ֣י רְשָׁעִ֔ים אֲשֶׁר־ בָּ֣א יוֹמָ֔ם בְּעֵ֖ת עֲוֺ֥ן קֵֽץ׃ (לט) הָשַׁ֖ב אֶל־ תַּעְרָ֑הּ בִּמְק֧וֹם אֲשֶׁר־ נִבְרֵ֛את בְּאֶ֥רֶץ מְכֻרוֹתַ֖יִךְ אֶשְׁפֹּ֥ט אֹתָֽךְ׃ (מ) וְשָׁפַכְתִּ֤י עָלַ֙יִךְ֙ זַעְמִ֔י בְּאֵ֥שׁ עֶבְרָתִ֖י אָפִ֣יחַ עָלָ֑יִךְ וּנְתַתִּ֗יךְ בְּיַד֙ אֲנָשִׁ֣ים בֹּֽעֲרִ֔ים חָרָשֵׁ֖י מַשְׁחִֽית׃ (מא) לָאֵ֤שׁ תִּֽהְיֶה֙ לְאָכְלָ֔ה דָּמֵ֥ךְ יִהְיֶ֖ה בְּת֣וֹךְ הָאָ֑רֶץ לֹ֣א תִזָּכֵ֔רִי כִּ֛י אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (מב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (מג) וְאַתָּ֣ה בֶן־ אָדָ֔ם הֲתִשְׁפֹּ֥ט הֲתִשְׁפֹּ֖ט אֶת־ עִ֣יר הַדָּמִ֑ים וְה֣וֹדַעְתָּ֔הּ אֵ֖ת כָּל־ תּוֹעֲבוֹתֶֽיהָ׃ (מד) וְאָמַרְתָּ֗ כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה עִ֣יר שֹׁפֶ֥כֶת דָּ֛ם בְּתוֹכָ֖הּ לָב֣וֹא עִתָּ֑הּ וְעָשְׂתָ֧ה גִלּוּלִ֛ים עָלֶ֖יהָ לְטָמְאָֽה׃ (מה) בְּדָמֵ֨ךְ אֲשֶׁר־ שָׁפַ֜כְתְּ אָשַׁ֗מְתְּ וּבְגִלּוּלַ֤יִךְ אֲשֶׁר־ עָשִׂית֙ טָמֵ֔את וַתַּקְרִ֣יבִי יָמַ֔יִךְ וַתָּב֖וֹא עַד־ שְׁנוֹתָ֑יִךְ עַל־ כֵּ֗ן נְתַתִּ֤יךְ חֶרְפָּה֙ לַגּוֹיִ֔ם וְקַלָּסָ֖ה לְכָל־ הָאֲרָצֽוֹת׃ (מו) הַקְּרֹב֛וֹת וְהָרְחֹק֥וֹת מִמֵּ֖ךְ יִתְקַלְּסוּ־ בָ֑ךְ טְמֵאַ֣ת הַשֵּׁ֔ם רַבַּ֖ת הַמְּהוּמָֽה׃ (מז) הִנֵּה֙ נְשִׂיאֵ֣י יִשְׂרָאֵ֔ל אִ֥ישׁ לִזְרֹע֖וֹ הָ֣יוּ בָ֑ךְ לְמַ֖עַן שְׁפָךְ־ דָּֽם׃ (מח) אָ֤ב וָאֵם֙ הֵקַ֣לּוּ בָ֔ךְ לַגֵּ֛ר עָשׂ֥וּ בַעֹ֖שֶׁק בְּתוֹכֵ֑ךְ יָת֥וֹם וְאַלְמָנָ֖ה ה֥וֹנוּ בָֽךְ׃ (מט) קָדָשַׁ֖י בָּזִ֑ית וְאֶת־ שַׁבְּתֹתַ֖י חִלָּֽלְתְּ׃ (נ) אַנְשֵׁ֥י רָכִ֛יל הָ֥יוּ בָ֖ךְ לְמַ֣עַן שְׁפָךְ־ דָּ֑ם וְאֶל־ הֶֽהָרִים֙ אָ֣כְלוּ בָ֔ךְ זִמָּ֖ה עָשׂ֥וּ בְתוֹכֵֽךְ׃ (נא) עֶרְוַת־ אָ֖ב גִּלָּה־ בָ֑ךְ טְמֵאַ֥ת הַנִּדָּ֖ה עִנּוּ־ בָֽךְ׃ (נב) וְאִ֣ישׁ ׀ אֶת־ אֵ֣שֶׁת רֵעֵ֗הוּ עָשָׂה֙ תּֽוֹעֵבָ֔ה וְאִ֥ישׁ אֶת־ כַּלָּת֖וֹ טִמֵּ֣א בְזִמָּ֑ה וְאִ֛ישׁ אֶת־ אֲחֹת֥וֹ בַת־ אָבִ֖יו עִנָּה־ בָֽךְ׃ (נג) שֹׁ֥חַד לָֽקְחוּ־ בָ֖ךְ לְמַ֣עַן שְׁפָךְ־ דָּ֑ם נֶ֧שֶׁךְ וְתַרְבִּ֣ית לָקַ֗חַתְּ וַתְּבַצְּעִ֤י רֵעַ֙יִךְ֙ בַּעֹ֔שֶׁק וְאֹתִ֣י שָׁכַ֔חַתְּ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (נד) וְהִנֵּה֙ הִכֵּ֣יתִי כַפִּ֔י אֶל־ בִּצְעֵ֖ךְ אֲשֶׁ֣ר עָשִׂ֑ית וְעַ֨ל־ דָּמֵ֔ךְ אֲשֶׁ֥ר הָי֖וּ בְּתוֹכֵֽךְ׃ (נה) הֲיַעֲמֹ֤ד לִבֵּךְ֙ אִם־ תֶּחֱזַ֣קְנָה יָדַ֔יִךְ לַיָּמִ֕ים אֲשֶׁ֥ר אֲנִ֖י עֹשֶׂ֣ה אוֹתָ֑ךְ אֲנִ֥י יְהוָ֖ה דִּבַּ֥רְתִּי וְעָשִֽׂיתִי׃ (נו) וַהֲפִיצוֹתִ֤י אוֹתָךְ֙ בַּגּוֹיִ֔ם וְזֵרִיתִ֖יךְ בָּאֲרָצ֑וֹת וַהֲתִמֹּתִ֥י טֻמְאָתֵ֖ךְ מִמֵּֽךְ׃
