יחזקאל סדר טו

40 פסוקים

פתיחה באתר · ספר יחזקאל כולו ·

(א) וְהָיָ֨ה יְחֶזְקֵ֤אל לָכֶם֙ לְמוֹפֵ֔ת כְּכֹ֥ל אֲשֶׁר־ עָשָׂ֖ה תַּעֲשׂ֑וּ בְּבֹאָ֕הּ וִֽידַעְתֶּ֕ם כִּ֥י אֲנִ֖י אֲדֹנָ֥י יְהוִֽה׃ (ב) וְאַתָּ֣ה בֶן־ אָדָ֔ם הֲל֗וֹא בְּי֨וֹם קַחְתִּ֤י מֵהֶם֙ אֶת־ מָ֣עוּזָּ֔ם מְשׂ֖וֹשׂ תִּפְאַרְתָּ֑ם אֶת־ מַחְמַ֤ד עֵֽינֵיהֶם֙ וְאֶת־ מַשָּׂ֣א נַפְשָׁ֔ם בְּנֵיהֶ֖ם וּבְנוֹתֵיהֶֽם׃ (ג) בַּיּ֣וֹם הַה֔וּא יָב֥וֹא הַפָּלִ֖יט אֵלֶ֑יךָ לְהַשְׁמָע֖וּת אָזְנָֽיִם׃ (ד) בַּיּ֣וֹם הַה֗וּא יִפָּ֤תַח פִּ֙יךָ֙ אֶת־ הַפָּלִ֔יט וּתְדַבֵּ֕ר וְלֹ֥א תֵֽאָלֵ֖ם ע֑וֹד וְהָיִ֤יתָ לָהֶם֙ לְמוֹפֵ֔ת וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (ה) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ו) בֶּן־ אָדָ֕ם שִׂ֥ים פָּנֶ֖יךָ אֶל־ בְּנֵ֣י עַמּ֑וֹן וְהִנָּבֵ֖א עֲלֵיהֶֽם׃ (ז) וְאָֽמַרְתָּ֙ לִבְנֵ֣י עַמּ֔וֹן שִׁמְע֖וּ דְּבַר־ אֲדֹנָ֣י יְהוִ֑ה כֹּה־ אָמַ֣ר אֲדֹנָ֣י יְהוִ֡ה יַעַן֩ אָמְרֵ֨ךְ הֶאָ֜ח אֶל־ מִקְדָּשִׁ֣י כִֽי־ נִחָ֗ל וְאֶל־ אַדְמַ֤ת יִשְׂרָאֵל֙ כִּ֣י נָשַׁ֔מָּה וְאֶל־ בֵּ֣ית יְהוּדָ֔ה כִּ֥י הָלְכ֖וּ בַּגּוֹלָֽה׃ (ח) לָכֵ֡ן הִנְנִי֩ נֹתְנָ֨ךְ לִבְנֵי־ קֶ֜דֶם לְמֽוֹרָשָׁ֗ה וְיִשְּׁב֤וּ טִירֽוֹתֵיהֶם֙ בָּ֔ךְ וְנָ֥תְנוּ בָ֖ךְ מִשְׁכְּנֵיהֶ֑ם הֵ֚מָּה יֹאכְל֣וּ פִרְיֵ֔ךְ וְהֵ֖מָּה יִשְׁתּ֥וּ חֲלָבֵֽךְ׃ (ט) וְנָתַתִּ֤י אֶת־ רַבָּה֙ לִנְוֵ֣ה גְמַלִּ֔ים וְאֶת־ בְּנֵ֥י עַמּ֖וֹן לְמִרְבַּץ־ צֹ֑אן וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (י) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֚עַן מַחְאֲךָ֣ יָ֔ד וְרַקְעֲךָ֖ בְּרָ֑גֶל וַתִּשְׂמַ֤ח בְּכָל־ שָֽׁאטְךָ֙ בְּנֶ֔פֶשׁ אֶל־ אַדְמַ֖ת יִשְׂרָאֵֽל׃ (יא) לָכֵ֡ן הִנְנִי֩ נָטִ֨יתִי אֶת־ יָדִ֜י עָלֶ֗יךָ וּנְתַתִּ֤יךָֽ־ (לבג) לְבַז֙ לַגּוֹיִ֔ם וְהִכְרַתִּ֙יךָ֙ מִן־ הָ֣עַמִּ֔ים וְהַאֲבַדְתִּ֖יךָ מִן־ הָאֲרָצ֑וֹת אַשְׁמִ֣ידְךָ֔ וְיָדַעְתָּ֖ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (יב) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה יַ֗עַן אֲמֹ֤ר מוֹאָב֙ וְשֵׂעִ֔יר הִנֵּ֥ה כְּכָֽל־ הַגּוֹיִ֖ם בֵּ֥ית יְהוּדָֽה׃ (יג) לָכֵן֩ הִנְנִ֨י פֹתֵ֜חַ אֶת־ כֶּ֤תֶף מוֹאָב֙ מֵהֶ֣עָרִ֔ים מֵֽעָרָ֖יו מִקָּצֵ֑הוּ צְבִ֗י אֶ֚רֶץ בֵּ֣ית הַיְשִׁימֹ֔ת בַּ֥עַל מְע֖וֹן (וקריתמה) וְקִרְיָתָֽיְמָה׃ (יד) לִבְנֵי־ קֶ֙דֶם֙ עַל־ בְּנֵ֣י עַמּ֔וֹן וּנְתַתִּ֖יהָ לְמֽוֹרָשָׁ֑ה לְמַ֛עַן לֹֽא־ תִזָּכֵ֥ר בְּנֵֽי־ עַמּ֖וֹן בַּגּוֹיִֽם׃ (טו) וּבְמוֹאָ֖ב אֶעֱשֶׂ֣ה שְׁפָטִ֑ים וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (טז) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֣עַן עֲשׂ֥וֹת אֱד֛וֹם בִּנְקֹ֥ם נָקָ֖ם לְבֵ֣ית יְהוּדָ֑ה וַיֶּאְשְׁמ֥וּ אָשׁ֖וֹם וְנִקְּמ֥וּ בָהֶֽם׃ (יז) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה וְנָטִ֤תִי יָדִי֙ עַל־ אֱד֔וֹם וְהִכְרַתִּ֥י מִמֶּ֖נָּה אָדָ֣ם וּבְהֵמָ֑ה וּנְתַתִּ֤יהָ חָרְבָּה֙ מִתֵּימָ֔ן וּדְדָ֖נֶה בַּחֶ֥רֶב יִפֹּֽלוּ׃ (יח) וְנָתַתִּ֨י אֶת־ נִקְמָתִ֜י בֶּאֱד֗וֹם בְּיַד֙ עַמִּ֣י יִשְׂרָאֵ֔ל וְעָשׂ֣וּ בֶאֱד֔וֹם כְּאַפִּ֖י וְכַחֲמָתִ֑י וְיָֽדְעוּ֙ אֶת־ נִקְמָתִ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יט) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֛עַן עֲשׂ֥וֹת פְּלִשְׁתִּ֖ים בִּנְקָמָ֑ה וַיִּנָּקְמ֤וּ נָקָם֙ בִּשְׁאָ֣ט בְּנֶ֔פֶשׁ לְמַשְׁחִ֖ית אֵיבַ֥ת עוֹלָֽם׃ (כ) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֨י נוֹטֶ֤ה יָדִי֙ עַל־ פְּלִשְׁתִּ֔ים וְהִכְרַתִּ֖י אֶת־ כְּרֵתִ֑ים וְהַ֣אֲבַדְתִּ֔י אֶת־ שְׁאֵרִ֖ית ח֥וֹף הַיָּֽם׃ (כא) וְעָשִׂ֤יתִי בָם֙ נְקָמ֣וֹת גְּדֹל֔וֹת בְּתוֹכְח֖וֹת חֵמָ֑ה וְיָֽדְעוּ֙ כִּֽי־ אֲנִ֣י יְהוָ֔ה בְּתִתִּ֥י אֶת־ נִקְמָתִ֖י בָּֽם׃ (כב) וַיְהִ֛י בְּעַשְׁתֵּֽי־ עֶשְׂרֵ֥ה שָׁנָ֖ה בְּאֶחָ֣ד לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כג) בֶּן־ אָדָ֗ם יַ֠עַן אֲשֶׁר־ אָ֨מְרָה צֹּ֤ר עַל־ יְרוּשָׁלִַ֙ם֙ הֶאָ֔ח נִשְׁבְּרָ֛ה דַּלְת֥וֹת הָעַמִּ֖ים נָסֵ֣בָּה אֵלָ֑י אִמָּלְאָ֖ה הָחֳרָֽבָה׃ (כד) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י עָלַ֖יִךְ צֹ֑ר וְהַעֲלֵיתִ֤י עָלַ֙יִךְ֙ גּוֹיִ֣ם רַבִּ֔ים כְּהַעֲל֥וֹת הַיָּ֖ם לְגַלָּֽיו׃ (כה) וְשִׁחֲת֞וּ חֹמ֣וֹת צֹ֗ר וְהָֽרְסוּ֙ מִגְדָּלֶ֔יהָ וְסִֽחֵיתִ֥י עֲפָרָ֖הּ מִמֶּ֑נָּה וְנָתַתִּ֥י אוֹתָ֖הּ לִצְחִ֥יחַ סָֽלַע׃ (כו) מִשְׁטַ֨ח חֲרָמִ֤ים תִּֽהְיֶה֙ בְּת֣וֹךְ הַיָּ֔ם כִּ֚י אֲנִ֣י דִבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה וְהָיְתָ֥ה לְבַ֖ז לַגּוֹיִֽם׃ (כז) וּבְנוֹתֶ֙יהָ֙ אֲשֶׁ֣ר בַּשָּׂדֶ֔ה בַּחֶ֖רֶב תֵּהָרַ֑גְנָה וְיָדְע֖וּ כִּי־ אֲנִ֥י יְהוָֽה׃ (כח) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֧י מֵבִ֣יא אֶל־ צֹ֗ר נְבוּכַדְרֶאצַּ֧ר מֶֽלֶךְ־ בָּבֶ֛ל מִצָּפ֖וֹן מֶ֣לֶךְ מְלָכִ֑ים בְּס֛וּס וּבְרֶ֥כֶב וּבְפָרָשִׁ֖ים וְקָהָ֥ל וְעַם־ רָֽב׃ (כט) בְּנוֹתַ֥יִךְ בַּשָּׂדֶ֖ה בַּחֶ֣רֶב יַהֲרֹ֑ג וְנָתַ֨ן עָלַ֜יִךְ דָּיֵ֗ק וְשָׁפַ֤ךְ עָלַ֙יִךְ֙ סֹֽלְלָ֔ה וְהֵקִ֥ים עָלַ֖יִךְ צִנָּֽה׃ (ל) וּמְחִ֣י קָֽבָלּ֔וֹ יִתֵּ֖ן בְּחֹֽמוֹתָ֑יִךְ וּמִ֨גְדְּלֹתַ֔יִךְ יִתֹּ֖ץ בְּחַרְבוֹתָֽיו׃ (לא) מִשִּׁפְעַ֥ת סוּסָ֖יו יְכַסֵּ֣ךְ אֲבָקָ֑ם מִקּוֹל֩ פָּרַ֨שׁ וְגַלְגַּ֜ל וָרֶ֗כֶב תִּרְעַ֙שְׁנָה֙ חֽוֹמוֹתַ֔יִךְ בְּבֹאוֹ֙ בִּשְׁעָרַ֔יִךְ כִּמְבוֹאֵ֖י עִ֥יר מְבֻקָּעָֽה׃ (לב) בְּפַרְס֣וֹת סוּסָ֔יו יִרְמֹ֖ס אֶת־ כָּל־ חֽוּצוֹתָ֑יִךְ עַמֵּךְ֙ בַּחֶ֣רֶב יַהֲרֹ֔ג וּמַצְּב֥וֹת עֻזֵּ֖ךְ לָאָ֥רֶץ תֵּרֵֽד׃ (לג) וְשָׁלְל֣וּ חֵילֵ֗ךְ וּבָֽזְזוּ֙ רְכֻלָּתֵ֔ךְ וְהָֽרְסוּ֙ חוֹמוֹתַ֔יִךְ וּבָתֵּ֥י חֶמְדָּתֵ֖ךְ יִתֹּ֑צוּ וַאֲבָנַ֤יִךְ וְעֵצַ֙יִךְ֙ וַֽעֲפָרֵ֔ךְ בְּת֥וֹךְ מַ֖יִם יָשִֽׂימוּ׃ (לד) וְהִשְׁבַּתִּ֖י הֲמ֣וֹן שִׁירָ֑יִךְ וְק֣וֹל כִּנּוֹרַ֔יִךְ לֹ֥א יִשָּׁמַ֖ע עֽוֹד׃ (לה) וּנְתַתִּ֞יךְ לִצְחִ֣יחַ סֶ֗לַע מִשְׁטַ֤ח חֲרָמִים֙ תִּֽהְיֶ֔ה לֹ֥א תִבָּנֶ֖ה ע֑וֹד כִּ֣י אֲנִ֤י יְהוָה֙ דִּבַּ֔רְתִּי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לו) כֹּ֥ה אָמַ֛ר אֲדֹנָ֥י יְהוִ֖ה לְצ֑וֹר הֲלֹ֣א ׀ מִקּ֣וֹל מַפַּלְתֵּ֗ךְ בֶּאֱנֹ֨ק חָלָ֜ל בֵּהָ֤רֵֽג הֶ֙רֶג֙ בְּתוֹכֵ֔ךְ יִרְעֲשׁ֖וּ הָאִיִּֽים׃ (לז) וְֽיָרְד֞וּ מֵעַ֣ל כִּסְאוֹתָ֗ם כֹּ֚ל נְשִׂיאֵ֣י הַיָּ֔ם וְהֵסִ֙ירוּ֙ אֶת־ מְעִ֣ילֵיהֶ֔ם וְאֶת־ בִּגְדֵ֥י רִקְמָתָ֖ם יִפְשֹׁ֑טוּ חֲרָד֤וֹת ׀ יִלְבָּ֙שׁוּ֙ עַל־ הָאָ֣רֶץ יֵשֵׁ֔בוּ וְחָֽרְדוּ֙ לִרְגָעִ֔ים וְשָׁמְמ֖וּ עָלָֽיִךְ׃ (לח) וְנָשְׂא֨וּ עָלַ֤יִךְ קִינָה֙ וְאָ֣מְרוּ לָ֔ךְ אֵ֣יךְ אָבַ֔דְתְּ נוֹשֶׁ֖בֶת מִיַּמִּ֑ים הָעִ֣יר הַהֻלָּ֗לָה אֲשֶׁר֩ הָיְתָ֨ה חֲזָקָ֤ה בַיָּם֙ הִ֣יא וְיֹשְׁבֶ֔יהָ אֲשֶׁר־ נָתְנ֥וּ חִתִּיתָ֖ם לְכָל־ יוֹשְׁבֶֽיהָ׃ (לט) עַתָּה֙ יֶחְרְד֣וּ הָֽאִיִּ֔ן י֖וֹם מַפַּלְתֵּ֑ךְ וְנִבְהֲל֛וּ הָאִיִּ֥ים אֲשֶׁר־ בַּיָּ֖ם מִצֵּאתֵֽךְ׃ (מ) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה בְּתִתִּ֤י אֹתָךְ֙ עִ֣יר נֶחֱרֶ֔בֶת כֶּעָרִ֖ים אֲשֶׁ֣ר לֹֽא־ נוֹשָׁ֑בוּ בְּהַעֲל֤וֹת עָלַ֙יִךְ֙ אֶת־ תְּה֔וֹם וְכִסּ֖וּךְ הַמַּ֥יִם הָרַבִּֽים׃
