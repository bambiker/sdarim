יחזקאל סדר י

פתיחה באתר · ספר יחזקאל כולו ·

(א) בְּחֻקּוֹתַ֧י יְהַלֵּ֛ךְ וּמִשְׁפָּטַ֥י שָׁמַ֖ר לַעֲשׂ֣וֹת אֱמֶ֑ת צַדִּ֥יק הוּא֙ חָיֹ֣ה יִֽחְיֶ֔ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ב) וְהוֹלִ֥יד בֵּן־ פָּרִ֖יץ שֹׁפֵ֣ךְ דָּ֑ם וְעָ֣שָׂה אָ֔ח מֵאַחַ֖ד מֵאֵֽלֶּה׃ (ג) וְה֕וּא אֶת־ כָּל־ אֵ֖לֶּה לֹ֣א עָשָׂ֑ה כִּ֣י גַ֤ם אֶל־ הֶֽהָרִים֙ אָכַ֔ל וְאֶת־ אֵ֥שֶׁת רֵעֵ֖הוּ טִמֵּֽא׃ (ד) עָנִ֤י וְאֶבְיוֹן֙ הוֹנָ֔ה גְּזֵל֣וֹת גָּזָ֔ל חֲבֹ֖ל לֹ֣א יָשִׁ֑יב וְאֶל־ הַגִּלּוּלִים֙ נָשָׂ֣א עֵינָ֔יו תּוֹעֵבָ֖ה עָשָֽׂה׃ (ה) בַּנֶּ֧שֶׁךְ נָתַ֛ן וְתַרְבִּ֥ית לָקַ֖ח וָחָ֑י לֹ֣א יִֽחְיֶ֗ה אֵ֣ת כָּל־ הַתּוֹעֵב֤וֹת הָאֵ֙לֶּה֙ עָשָׂ֔ה מ֣וֹת יוּמָ֔ת דָּמָ֖יו בּ֥וֹ יִהְיֶֽה׃ (ו) וְהִנֵּה֙ הוֹלִ֣יד בֵּ֔ן וַיַּ֕רְא אֶת־ כָּל־ חַטֹּ֥את אָבִ֖יו אֲשֶׁ֣ר עָשָׂ֑ה וַיִּרְאֶ֕ה וְלֹ֥א יַעֲשֶׂ֖ה כָּהֵֽן׃ (ז) עַל־ הֶֽהָרִים֙ לֹ֣א אָכָ֔ל וְעֵינָיו֙ לֹ֣א נָשָׂ֔א אֶל־ גִּלּוּלֵ֖י בֵּ֣ית יִשְׂרָאֵ֑ל אֶת־ אֵ֥שֶׁת רֵעֵ֖הוּ לֹ֥א טִמֵּֽא׃ (ח) וְאִישׁ֙ לֹ֣א הוֹנָ֔ה חֲבֹל֙ לֹ֣א חָבָ֔ל וּגְזֵלָ֖ה לֹ֣א גָזָ֑ל לַחְמוֹ֙ לְרָעֵ֣ב נָתָ֔ן וְעֵר֖וֹם כִּסָּה־ בָֽגֶד׃ (ט) מֵעָנִ֞י הֵשִׁ֣יב יָד֗וֹ נֶ֤שֶׁךְ וְתַרְבִּית֙ לֹ֣א לָקָ֔ח מִשְׁפָּטַ֣י עָשָׂ֔ה בְּחֻקּוֹתַ֖י הָלָ֑ךְ ה֗וּא לֹ֥א יָמ֛וּת בַּעֲוֺ֥ן אָבִ֖יו חָיֹ֥ה יִחְיֶֽה׃ (י) אָבִ֞יו כִּֽי־ עָ֣שַׁק עֹ֗שֶׁק גָּזַל֙ גֵּ֣זֶל אָ֔ח וַאֲשֶׁ֥ר לֹא־ ט֛וֹב עָשָׂ֖ה בְּת֣וֹךְ עַמָּ֑יו וְהִנֵּה־ מֵ֖ת בַּעֲוֺנֽוֹ׃ (יא) וַאֲמַרְתֶּ֕ם מַדֻּ֛עַ לֹא־ נָשָׂ֥א הַבֵּ֖ן בַּעֲוֺ֣ן הָאָ֑ב וְהַבֵּ֞ן מִשְׁפָּ֧ט וּצְדָקָ֣ה עָשָׂ֗ה אֵ֣ת כָּל־ חֻקּוֹתַ֥י שָׁמַ֛ר וַיַּעֲשֶׂ֥ה אֹתָ֖ם חָיֹ֥ה יִחְיֶֽה׃ (יב) הַנֶּ֥פֶשׁ הַחֹטֵ֖את הִ֣יא תָמ֑וּת בֵּ֞ן לֹא־ יִשָּׂ֣א ׀ בַּעֲוֺ֣ן הָאָ֗ב וְאָב֙ לֹ֤א יִשָּׂא֙ בַּעֲוֺ֣ן הַבֵּ֔ן צִדְקַ֤ת הַצַּדִּיק֙ עָלָ֣יו תִּֽהְיֶ֔ה וְרִשְׁעַ֥ת (רשע) הָרָשָׁ֖ע עָלָ֥יו תִּֽהְיֶֽה׃ (יג) וְהָרָשָׁ֗ע כִּ֤י יָשׁוּב֙ מִכָּל־ (חטאתו) חַטֹּאתָיו֙ אֲשֶׁ֣ר עָשָׂ֔ה וְשָׁמַר֙ אֶת־ כָּל־ חֻקּוֹתַ֔י וְעָשָׂ֥ה מִשְׁפָּ֖ט וּצְדָקָ֑ה חָיֹ֥ה יִחְיֶ֖ה לֹ֥א יָמֽוּת׃ (יד) כָּל־ פְּשָׁעָיו֙ אֲשֶׁ֣ר עָשָׂ֔ה לֹ֥א יִזָּכְר֖וּ ל֑וֹ בְּצִדְקָת֥וֹ אֲשֶׁר־ עָשָׂ֖ה יִֽחְיֶֽה׃ (טו) הֶחָפֹ֤ץ אֶחְפֹּץ֙ מ֣וֹת רָשָׁ֔ע נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה הֲל֛וֹא בְּשׁוּב֥וֹ מִדְּרָכָ֖יו וְחָיָֽה׃ (טז) וּבְשׁ֨וּב צַדִּ֤יק מִצִּדְקָתוֹ֙ וְעָ֣שָׂה עָ֔וֶל כְּכֹ֨ל הַתּוֹעֵב֜וֹת אֲשֶׁר־ עָשָׂ֧ה הָרָשָׁ֛ע יַעֲשֶׂ֖ה וָחָ֑י כָּל־ (צדקתו) צִדְקֹתָ֤יו אֲשֶׁר־ עָשָׂה֙ לֹ֣א תִזָּכַ֔רְנָה בְּמַעֲל֧וֹ אֲשֶׁר־ מָעַ֛ל וּבְחַטָּאת֥וֹ אֲשֶׁר־ חָטָ֖א בָּ֥ם יָמֽוּת׃ (יז) וַאֲמַרְתֶּ֕ם לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י שִׁמְעוּ־ נָא֙ בֵּ֣ית יִשְׂרָאֵ֔ל הֲדַרְכִּי֙ לֹ֣א יִתָּכֵ֔ן הֲלֹ֥א דַרְכֵיכֶ֖ם לֹ֥א יִתָּכֵֽנוּ׃ (יח) בְּשׁוּב־ צַדִּ֧יק מִצִּדְקָת֛וֹ וְעָ֥שָׂה עָ֖וֶל וּמֵ֣ת עֲלֵיהֶ֑ם בְּעַוְל֥וֹ אֲשֶׁר־ עָשָׂ֖ה יָמֽוּת׃ (יט) וּבְשׁ֣וּב רָשָׁ֗ע מֵֽרִשְׁעָתוֹ֙ אֲשֶׁ֣ר עָשָׂ֔ה וַיַּ֥עַשׂ מִשְׁפָּ֖ט וּצְדָקָ֑ה ה֖וּא אֶת־ נַפְשׁ֥וֹ יְחַיֶּֽה׃ (כ) וַיִּרְאֶ֣ה (וישוב) וַיָּ֔שָׁב מִכָּל־ פְּשָׁעָ֖יו אֲשֶׁ֣ר עָשָׂ֑ה חָי֥וֹ יִחְיֶ֖ה לֹ֥א יָמֽוּת׃ (כא) וְאָֽמְרוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י הַדְּרָכַ֞י לֹ֤א יִתָּֽכְנּוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל הֲלֹ֥א דַרְכֵיכֶ֖ם לֹ֥א יִתָּכֵֽן׃ (כב) לָכֵן֩ אִ֨ישׁ כִּדְרָכָ֜יו אֶשְׁפֹּ֤ט אֶתְכֶם֙ בֵּ֣ית יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה שׁ֤וּבוּ וְהָשִׁ֙יבוּ֙ מִכָּל־ פִּשְׁעֵיכֶ֔ם וְלֹֽא־ יִהְיֶ֥ה לָכֶ֛ם לְמִכְשׁ֖וֹל עָוֺֽן׃ (כג) הַשְׁלִ֣יכוּ מֵעֲלֵיכֶ֗ם אֶת־ כָּל־ פִּשְׁעֵיכֶם֙ אֲשֶׁ֣ר פְּשַׁעְתֶּ֣ם בָּ֔ם וַעֲשׂ֥וּ לָכֶ֛ם לֵ֥ב חָדָ֖שׁ וְר֣וּחַ חֲדָשָׁ֑ה וְלָ֥מָּה תָמֻ֖תוּ בֵּ֥ית יִשְׂרָאֵֽל׃ (כד) כִּ֣י לֹ֤א אֶחְפֹּץ֙ בְּמ֣וֹת הַמֵּ֔ת נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה וְהָשִׁ֖יבוּ וִֽחְיֽוּ׃ (כה) וְאַתָּה֙ שָׂ֣א קִינָ֔ה אֶל־ נְשִׂיאֵ֖י יִשְׂרָאֵֽל׃ (כו) וְאָמַרְתָּ֗ מָ֤ה אִמְּךָ֙ לְבִיָּ֔א בֵּ֥ין אֲרָי֖וֹת רָבָ֑צָה בְּת֥וֹךְ כְּפִרִ֖ים רִבְּתָ֥ה גוּרֶֽיהָ׃ (כז) וַתַּ֛עַל אֶחָ֥ד מִגֻּרֶ֖יהָ כְּפִ֣יר הָיָ֑ה וַיִּלְמַ֥ד לִטְרָף־ טֶ֖רֶף אָדָ֥ם אָכָֽל׃ (כח) וַיִּשְׁמְע֥וּ אֵלָ֛יו גּוֹיִ֖ם בְּשַׁחְתָּ֣ם נִתְפָּ֑שׂ וַיְבִאֻ֥הוּ בַֽחַחִ֖ים אֶל־ אֶ֥רֶץ מִצְרָֽיִם׃ (כט) וַתֵּ֙רֶא֙ כִּ֣י נֽוֹחֲלָ֔ה אָבְדָ֖ה תִּקְוָתָ֑הּ וַתִּקַּ֛ח אֶחָ֥ד מִגֻּרֶ֖יהָ כְּפִ֥יר שָׂמָֽתְהוּ׃ (ל) וַיִּתְהַלֵּ֥ךְ בְּתוֹךְ־ אֲרָי֖וֹת כְּפִ֣יר הָיָ֑ה וַיִּלְמַ֥ד לִטְרָף־ טֶ֖רֶף אָדָ֥ם אָכָֽל׃ (לא) וַיֵּ֙דַע֙ אַלְמְנוֹתָ֔יו וְעָרֵיהֶ֖ם הֶחֱרִ֑יב וַתֵּ֤שַׁם אֶ֙רֶץ֙ וּמְלֹאָ֔הּ מִקּ֖וֹל שַׁאֲגָתֽוֹ׃ (לב) וַיִּתְּנ֨וּ עָלָ֥יו גּוֹיִ֛ם סָבִ֖יב מִמְּדִינ֑וֹת וַֽיִּפְרְשׂ֥וּ עָלָ֛יו רִשְׁתָּ֖ם בְּשַׁחְתָּ֥ם נִתְפָּֽשׂ׃ (לג) וַֽיִּתְּנֻ֤הוּ בַסּוּגַר֙ בַּֽחַחִ֔ים וַיְבִאֻ֖הוּ אֶל־ מֶ֣לֶךְ בָּבֶ֑ל יְבִאֻ֙הוּ֙ בַּמְּצֹד֔וֹת לְמַ֗עַן לֹא־ יִשָּׁמַ֥ע קוֹל֛וֹ ע֖וֹד אֶל־ הָרֵ֥י יִשְׂרָאֵֽל׃ (לד) אִמְּךָ֥ כַגֶּ֛פֶן בְּדָמְךָ֖ עַל־ מַ֣יִם שְׁתוּלָ֑ה פֹּֽרִיָּה֙ וַֽעֲנֵפָ֔ה הָיְתָ֖ה מִמַּ֥יִם רַבִּֽים׃ (לה) וַיִּֽהְיוּ־ לָ֞הּ מַטּ֣וֹת עֹ֗ז אֶל־ שִׁבְטֵי֙ מֹֽשְׁלִ֔ים וַתִּגְבַּ֥הּ קֽוֹמָת֖וֹ עַל־ בֵּ֣ין עֲבֹתִ֑ים וַיֵּרָ֣א בְגָבְה֔וֹ בְּרֹ֖ב דָּלִיֹּתָֽיו׃ (לו) וַתֻּתַּ֤שׁ בְּחֵמָה֙ לָאָ֣רֶץ הֻשְׁלָ֔כָה וְר֥וּחַ הַקָּדִ֖ים הוֹבִ֣ישׁ פִּרְיָ֑הּ הִתְפָּרְק֧וּ וְיָבֵ֛שׁוּ מַטֵּ֥ה עֻזָּ֖הּ אֵ֥שׁ אֲכָלָֽתְהוּ׃ (לז) וְעַתָּ֖ה שְׁתוּלָ֣ה בַמִּדְבָּ֑ר בְּאֶ֖רֶץ צִיָּ֥ה וְצָמָֽא׃ (לח) וַתֵּצֵ֨א אֵ֜שׁ מִמַּטֵּ֤ה בַדֶּ֙יהָ֙ פִּרְיָ֣הּ אָכָ֔לָה וְלֹא־ הָ֥יָה בָ֛הּ מַטֵּה־ עֹ֖ז שֵׁ֣בֶט לִמְשׁ֑וֹל קִ֥ינָה הִ֖יא וַתְּהִ֥י לְקִינָֽה׃
