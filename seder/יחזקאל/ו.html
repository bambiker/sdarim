יחזקאל סדר ו

57 פסוקים

פתיחה באתר · ספר יחזקאל כולו ·

(א) לְמַ֙עַן֙ בְּחֻקֹּתַ֣י יֵלֵ֔כוּ וְאֶת־ מִשְׁפָּטַ֥י יִשְׁמְר֖וּ וְעָשׂ֣וּ אֹתָ֑ם וְהָיוּ־ לִ֣י לְעָ֔ם וַאֲנִ֕י אֶהְיֶ֥ה לָהֶ֖ם לֵאלֹהִֽים׃ (ב) וְאֶל־ לֵ֧ב שִׁקּוּצֵיהֶ֛ם וְתוֹעֲבוֹתֵיהֶ֖ם לִבָּ֣ם הֹלֵ֑ךְ דַּרְכָּם֙ בְּרֹאשָׁ֣ם נָתַ֔תִּי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ג) וַיִּשְׂא֤וּ הַכְּרוּבִים֙ אֶת־ כַּנְפֵיהֶ֔ם וְהָאֽוֹפַנִּ֖ים לְעֻמָּתָ֑ם וּכְב֧וֹד אֱלֹהֵֽי־ יִשְׂרָאֵ֛ל עֲלֵיהֶ֖ם מִלְמָֽעְלָה׃ (ד) וַיַּ֙עַל֙ כְּב֣וֹד יְהוָ֔ה מֵעַ֖ל תּ֣וֹךְ הָעִ֑יר וַֽיַּעֲמֹד֙ עַל־ הָהָ֔ר אֲשֶׁ֖ר מִקֶּ֥דֶם לָעִֽיר׃ (ה) וְר֣וּחַ נְשָׂאַ֗תְנִי וַתְּבִיאֵ֤נִי כַשְׂדִּ֙ימָה֙ אֶל־ הַגּוֹלָ֔ה בַּמַּרְאֶ֖ה בְּר֣וּחַ אֱלֹהִ֑ים וַיַּ֙עַל֙ מֵֽעָלַ֔י הַמַּרְאֶ֖ה אֲשֶׁ֥ר רָאִֽיתִי׃ (ו) וָאֲדַבֵּ֖ר אֶל־ הַגּוֹלָ֑ה אֵ֛ת כָּל־ דִּבְרֵ֥י יְהוָ֖ה אֲשֶׁ֥ר הֶרְאָֽנִי׃ (ז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ח) בֶּן־ אָדָ֕ם בְּת֥וֹךְ בֵּית־ הַמֶּ֖רִי אַתָּ֣ה יֹשֵׁ֑ב אֲשֶׁ֣ר עֵינַיִם֩ לָהֶ֨ם לִרְא֜וֹת וְלֹ֣א רָא֗וּ אָזְנַ֨יִם לָהֶ֤ם לִשְׁמֹ֙עַ֙ וְלֹ֣א שָׁמֵ֔עוּ כִּ֛י בֵּ֥ית מְרִ֖י הֵֽם׃ (ט) וְאַתָּ֣ה בֶן־ אָדָ֗ם עֲשֵׂ֤ה לְךָ֙ כְּלֵ֣י גוֹלָ֔ה וּגְלֵ֥ה יוֹמָ֖ם לְעֵֽינֵיהֶ֑ם וְגָלִ֨יתָ מִמְּקוֹמְךָ֜ אֶל־ מָק֤וֹם אַחֵר֙ לְעֵ֣ינֵיהֶ֔ם אוּלַ֣י יִרְא֔וּ כִּ֛י בֵּ֥ית מְרִ֖י הֵֽמָּה׃ (י) וְהוֹצֵאתָ֨ כֵלֶ֜יךָ כִּכְלֵ֥י גוֹלָ֛ה יוֹמָ֖ם לְעֵֽינֵיהֶ֑ם וְאַתָּ֗ה תֵּצֵ֤א בָעֶ֙רֶב֙ לְעֵ֣ינֵיהֶ֔ם כְּמוֹצָאֵ֖י גּוֹלָֽה׃ (יא) לְעֵינֵיהֶ֖ם חֲתָר־ לְךָ֣ בַקִּ֑יר וְהוֹצֵאתָ֖ בּֽוֹ׃ (יב) לְעֵ֨ינֵיהֶ֜ם עַל־ כָּתֵ֤ף תִּשָּׂא֙ בָּעֲלָטָ֣ה תוֹצִ֔יא פָּנֶ֣יךָ תְכַסֶּ֔ה וְלֹ֥א תִרְאֶ֖ה אֶת־ הָאָ֑רֶץ כִּֽי־ מוֹפֵ֥ת נְתַתִּ֖יךָ לְבֵ֥ית יִשְׂרָאֵֽל׃ (יג) וָאַ֣עַשׂ כֵּן֮ כַּאֲשֶׁ֣ר צֻוֵּיתִי֒ כֵּ֠לַי הוֹצֵ֜אתִי כִּכְלֵ֤י גוֹלָה֙ יוֹמָ֔ם וּבָעֶ֛רֶב חָתַֽרְתִּי־ לִ֥י בַקִּ֖יר בְּיָ֑ד בָּעֲלָטָ֥ה הוֹצֵ֛אתִי עַל־ כָּתֵ֥ף נָשָׂ֖אתִי לְעֵינֵיהֶֽם׃ (יד) וַיְהִ֧י דְבַר־ יְהוָ֛ה אֵלַ֖י בַּבֹּ֥קֶר לֵאמֹֽר׃ (טו) בֶּן־ אָדָ֕ם הֲלֹ֨א אָמְר֥וּ אֵלֶ֛יךָ בֵּ֥ית יִשְׂרָאֵ֖ל בֵּ֣ית הַמֶּ֑רִי מָ֖ה אַתָּ֥ה עֹשֶֽׂה׃ (טז) אֱמֹ֣ר אֲלֵיהֶ֔ם כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהֹוִ֑ה הַנָּשִׂ֞יא הַמַּשָּׂ֤א הַזֶּה֙ בִּיר֣וּשָׁלִַ֔ם וְכָל־ בֵּ֥ית יִשְׂרָאֵ֖ל אֲשֶׁר־ הֵ֥מָּה בְתוֹכָֽם׃ (יז) אֱמֹ֖ר אֲנִ֣י מֽוֹפֶתְכֶ֑ם כַּאֲשֶׁ֣ר עָשִׂ֗יתִי כֵּ֚ן יֵעָשֶׂ֣ה לָהֶ֔ם בַּגּוֹלָ֥ה בַשְּׁבִ֖י יֵלֵֽכוּ׃ (יח) וְהַנָּשִׂ֨יא אֲשֶׁר־ בְּתוֹכָ֜ם אֶל־ כָּתֵ֤ף יִשָּׂא֙ בָּעֲלָטָ֣ה וְיֵצֵ֔א בַּקִּ֥יר יַחְתְּר֖וּ לְה֣וֹצִיא ב֑וֹ פָּנָ֣יו יְכַסֶּ֔ה יַ֗עַן אֲשֶׁ֨ר לֹא־ יִרְאֶ֥ה לַעַ֛יִן ה֖וּא אֶת־ הָאָֽרֶץ׃ (יט) וּפָרַשְׂתִּ֤י אֶת־ רִשְׁתִּי֙ עָלָ֔יו וְנִתְפַּ֖שׂ בִּמְצֽוּדָתִ֑י וְהֵבֵאתִ֨י אֹת֤וֹ בָבֶ֙לָה֙ אֶ֣רֶץ כַּשְׂדִּ֔ים וְאוֹתָ֥הּ לֹֽא־ יִרְאֶ֖ה וְשָׁ֥ם יָמֽוּת׃ (כ) וְכֹל֩ אֲשֶׁ֨ר סְבִיבֹתָ֥יו (עזרה) עֶזְר֛וֹ וְכָל־ אֲגַפָּ֖יו אֱזָרֶ֣ה לְכָל־ ר֑וּחַ וְחֶ֖רֶב אָרִ֥יק אַחֲרֵיהֶֽם׃ (כא) וְיָדְע֖וּ כִּֽי־ אֲנִ֣י יְהוָ֑ה בַּהֲפִיצִ֤י אוֹתָם֙ בַּגּוֹיִ֔ם וְזֵרִיתִ֥י אוֹתָ֖ם בָּאֲרָצֽוֹת׃ (כב) וְהוֹתַרְתִּ֤י מֵהֶם֙ אַנְשֵׁ֣י מִסְפָּ֔ר מֵחֶ֖רֶב מֵרָעָ֣ב וּמִדָּ֑בֶר לְמַ֨עַן יְסַפְּר֜וּ אֶת־ כָּל־ תּוֹעֲבֽוֹתֵיהֶ֗ם בַּגּוֹיִם֙ אֲשֶׁר־ בָּ֣אוּ שָׁ֔ם וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כג) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כד) בֶּן־ אָדָ֕ם לַחְמְךָ֖ בְּרַ֣עַשׁ תֹּאכֵ֑ל וּמֵימֶ֕יךָ בְּרָגְזָ֥ה וּבִדְאָגָ֖ה תִּשְׁתֶּֽה׃ (כה) וְאָמַרְתָּ֣ אֶל־ עַ֣ם הָאָ֡רֶץ כֹּֽה־ אָמַר֩ אֲדֹנָ֨י יְהוִ֜ה לְיוֹשְׁבֵ֤י יְרוּשָׁלִַ֙ם֙ אֶל־ אַדְמַ֣ת יִשְׂרָאֵ֔ל לַחְמָם֙ בִּדְאָגָ֣ה יֹאכֵ֔לוּ וּמֵֽימֵיהֶ֖ם בְּשִׁמָּמ֣וֹן יִשְׁתּ֑וּ לְמַ֜עַן תֵּשַׁ֤ם אַרְצָהּ֙ מִמְּלֹאָ֔הּ מֵחֲמַ֖ס כָּֽל־ הַיֹּשְׁבִ֥ים בָּֽהּ׃ (כו) וְהֶעָרִ֤ים הַנּֽוֹשָׁבוֹת֙ תֶּחֱרַ֔בְנָה וְהָאָ֖רֶץ שְׁמָמָ֣ה תִֽהְיֶ֑ה וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כח) בֶּן־ אָדָ֗ם מָֽה־ הַמָּשָׁ֤ל הַזֶּה֙ לָכֶ֔ם עַל־ אַדְמַ֥ת יִשְׂרָאֵ֖ל לֵאמֹ֑ר יַֽאַרְכוּ֙ הַיָּמִ֔ים וְאָבַ֖ד כָּל־ חָזֽוֹן׃ (כט) לָכֵ֞ן אֱמֹ֣ר אֲלֵיהֶ֗ם כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ הִשְׁבַּ֙תִּי֙ אֶת־ הַמָּשָׁ֣ל הַזֶּ֔ה וְלֹֽא־ יִמְשְׁל֥וּ אֹת֛וֹ ע֖וֹד בְּיִשְׂרָאֵ֑ל כִּ֚י אִם־ דַּבֵּ֣ר אֲלֵיהֶ֔ם קָֽרְבוּ֙ הַיָּמִ֔ים וּדְבַ֖ר כָּל־ חָזֽוֹן׃ (ל) כִּ֠י לֹ֣א יִֽהְיֶ֥ה ע֛וֹד כָּל־ חֲז֥וֹן שָׁ֖וְא וּמִקְסַ֣ם חָלָ֑ק בְּת֖וֹךְ בֵּ֥ית יִשְׂרָאֵֽל׃ (לא) כִּ֣י ׀ אֲנִ֣י יְהוָ֗ה אֲדַבֵּר֙ אֵת֩ אֲשֶׁ֨ר אֲדַבֵּ֤ר דָּבָר֙ וְיֵ֣עָשֶׂ֔ה לֹ֥א תִמָּשֵׁ֖ךְ ע֑וֹד כִּ֣י בִֽימֵיכֶ֞ם בֵּ֣ית הַמֶּ֗רִי אֲדַבֵּ֤ר דָּבָר֙ וַעֲשִׂיתִ֔יו נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לג) בֶּן־ אָדָ֗ם הִנֵּ֤ה בֵֽית־ יִשְׂרָאֵל֙ אֹֽמְרִ֔ים הֶחָז֛וֹן אֲשֶׁר־ ה֥וּא חֹזֶ֖ה לְיָמִ֣ים רַבִּ֑ים וּלְעִתִּ֥ים רְחוֹק֖וֹת ה֥וּא נִבָּֽא׃ (לד) לָכֵ֞ן אֱמֹ֣ר אֲלֵיהֶ֗ם כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה לֹא־ תִמָּשֵׁ֥ךְ ע֖וֹד כָּל־ דְּבָרָ֑י אֲשֶׁ֨ר אֲדַבֵּ֤ר דָּבָר֙ וְיֵ֣עָשֶׂ֔ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לה) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לו) בֶּן־ אָדָ֕ם הִנָּבֵ֛א אֶל־ נְבִיאֵ֥י יִשְׂרָאֵ֖ל הַנִּבָּאִ֑ים וְאָֽמַרְתָּ֙ לִנְבִיאֵ֣י מִלִּבָּ֔ם שִׁמְע֖וּ דְּבַר־ יְהוָֽה׃ (לז) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה ה֖וֹי עַל־ הַנְּבִיאִ֣ים הַנְּבָלִ֑ים אֲשֶׁ֥ר הֹלְכִ֛ים אַחַ֥ר רוּחָ֖ם וּלְבִלְתִּ֥י רָאֽוּ׃ (לח) כְּשֻׁעָלִ֖ים בָּחֳרָב֑וֹת נְבִיאֶ֥יךָ יִשְׂרָאֵ֖ל הָיֽוּ׃ (לט) לֹ֤א עֲלִיתֶם֙ בַּפְּרָצ֔וֹת וַתִּגְדְּר֥וּ גָדֵ֖ר עַל־ בֵּ֣ית יִשְׂרָאֵ֑ל לַעֲמֹ֥ד בַּמִּלְחָמָ֖ה בְּי֥וֹם יְהוָֽה׃ (מ) חָ֤זוּ שָׁוְא֙ וְקֶ֣סֶם כָּזָ֔ב הָאֹֽמְרִים֙ נְאֻם־ יְהוָ֔ה וַֽיהוָ֖ה לֹ֣א שְׁלָחָ֑ם וְיִֽחֲל֖וּ לְקַיֵּ֥ם דָּבָֽר׃ (מא) הֲל֤וֹא מַֽחֲזֵה־ שָׁוְא֙ חֲזִיתֶ֔ם וּמִקְסַ֥ם כָּזָ֖ב אֲמַרְתֶּ֑ם וְאֹֽמְרִים֙ נְאֻם־ יְהוָ֔ה וַאֲנִ֖י לֹ֥א דִבַּֽרְתִּי׃ (מב) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֚עַן דַּבֶּרְכֶ֣ם שָׁ֔וְא וַחֲזִיתֶ֖ם כָּזָ֑ב לָכֵן֙ הִנְנִ֣י אֲלֵיכֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (מג) וְהָיְתָ֣ה יָדִ֗י אֶֽל־ הַנְּבִיאִ֞ים הַחֹזִ֣ים שָׁוְא֮ וְהַקֹּסְמִ֣ים כָּזָב֒ בְּס֧וֹד עַמִּ֣י לֹֽא־ יִהְי֗וּ וּבִכְתָ֤ב בֵּֽית־ יִשְׂרָאֵל֙ לֹ֣א יִכָּתֵ֔בוּ וְאֶל־ אַדְמַ֥ת יִשְׂרָאֵ֖ל לֹ֣א יָבֹ֑אוּ וִידַעְתֶּ֕ם כִּ֥י אֲנִ֖י אֲדֹנָ֥י יְהוִֽה׃ (מד) יַ֣עַן וּבְיַ֜עַן הִטְע֧וּ אֶת־ עַמִּ֛י לֵאמֹ֥ר שָׁל֖וֹם וְאֵ֣ין שָׁל֑וֹם וְהוּא֙ בֹּ֣נֶה חַ֔יִץ וְהִנָּ֛ם טָחִ֥ים אֹת֖וֹ תָּפֵֽל׃ (מה) אֱמֹ֛ר אֶל־ טָחֵ֥י תָפֵ֖ל וְיִפֹּ֑ל הָיָ֣ה ׀ גֶּ֣שֶׁם שׁוֹטֵ֗ף וְאַתֵּ֜נָה אַבְנֵ֤י אֶלְגָּבִישׁ֙ תִּפֹּ֔לְנָה וְר֥וּחַ סְעָר֖וֹת תְּבַקֵּֽעַ׃ (מו) וְהִנֵּ֖ה נָפַ֣ל הַקִּ֑יר הֲלוֹא֙ יֵאָמֵ֣ר אֲלֵיכֶ֔ם אַיֵּ֥ה הַטִּ֖יחַ אֲשֶׁ֥ר טַחְתֶּֽם׃ (מז) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה וּבִקַּעְתִּ֥י רֽוּחַ־ סְעָר֖וֹת בַּֽחֲמָתִ֑י וְגֶ֤שֶׁם שֹׁטֵף֙ בְּאַפִּ֣י יִֽהְיֶ֔ה וְאַבְנֵ֥י אֶלְגָּבִ֖ישׁ בְּחֵמָ֥ה לְכָלָֽה׃ (מח) וְהָ֨רַסְתִּ֜י אֶת־ הַקִּ֨יר אֲשֶׁר־ טַחְתֶּ֥ם תָּפֵ֛ל וְהִגַּעְתִּ֥יהוּ אֶל־ הָאָ֖רֶץ וְנִגְלָ֣ה יְסֹד֑וֹ וְנָֽפְלָה֙ וּכְלִיתֶ֣ם בְּתוֹכָ֔הּ וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (מט) וְכִלֵּיתִ֤י אֶת־ חֲמָתִי֙ בַּקִּ֔יר וּבַטָּחִ֥ים אֹת֖וֹ תָּפֵ֑ל וְאֹמַ֤ר לָכֶם֙ אֵ֣ין הַקִּ֔יר וְאֵ֖ין הַטָּחִ֥ים אֹתֽוֹ׃ (נ) נְבִיאֵ֣י יִשְׂרָאֵ֗ל הַֽנִבְּאִים֙ אֶל־ יְר֣וּשָׁלִַ֔ם וְהַחֹזִ֥ים לָ֖הּ חֲז֣וֹן שָׁלֹ֑ם וְאֵ֣ין שָׁלֹ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהֹוִֽה׃ (נא) וְאַתָּ֣ה בֶן־ אָדָ֗ם שִׂ֤ים פָּנֶ֙יךָ֙ אֶל־ בְּנ֣וֹת עַמְּךָ֔ הַמִּֽתְנַבְּא֖וֹת מִֽלִּבְּהֶ֑ן וְהִנָּבֵ֖א עֲלֵיהֶֽן׃ (נב) וְאָמַרְתָּ֞ כֹּה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה הוֹי֩ לִֽמְתַפְּר֨וֹת כְּסָת֜וֹת עַ֣ל ׀ כָּל־ אַצִּילֵ֣י יָדַ֗י וְעֹשׂ֧וֹת הַמִּסְפָּח֛וֹת עַל־ רֹ֥אשׁ כָּל־ קוֹמָ֖ה לְצוֹדֵ֣ד נְפָשׁ֑וֹת הַנְּפָשׁוֹת֙ תְּצוֹדֵ֣דְנָה לְעַמִּ֔י וּנְפָשׁ֖וֹת לָכֶ֥נָה תְחַיֶּֽינָה׃ (נג) וַתְּחַלֶּלְ֨נָה אֹתִ֜י אֶל־ עַמִּ֗י בְּשַׁעֲלֵ֣י שְׂעֹרִים֮ וּבִפְת֣וֹתֵי לֶחֶם֒ לְהָמִ֤ית נְפָשׁוֹת֙ אֲשֶׁ֣ר לֹֽא־ תְמוּתֶ֔נָה וּלְחַיּ֥וֹת נְפָשׁ֖וֹת אֲשֶׁ֣ר לֹא־ תִֽחְיֶ֑ינָה בְּכַ֨זֶּבְכֶ֔ם לְעַמִּ֖י שֹׁמְעֵ֥י כָזָֽב׃ (נד) לָכֵ֞ן כֹּה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה הִנְנִ֤י אֶל־ כִּסְּתוֹתֵיכֶ֙נָה֙ אֲשֶׁ֣ר אַ֠תֵּנָה מְצֹדְד֨וֹת שָׁ֤ם אֶת־ הַנְּפָשׁוֹת֙ לְפֹ֣רְח֔וֹת וְקָרַעְתִּ֣י אֹתָ֔ם מֵעַ֖ל זְרוֹעֹֽתֵיכֶ֑ם וְשִׁלַּחְתִּי֙ אֶת־ הַנְּפָשׁ֔וֹת אֲשֶׁ֥ר אַתֶּ֛ם מְצֹדְד֥וֹת אֶת־ נְפָשִׁ֖ים לְפֹרְחֹֽת׃ (נה) וְקָרַעְתִּ֞י אֶת־ מִסְפְּחֹֽתֵיכֶ֗ם וְהִצַּלְתִּ֤י אֶת־ עַמִּי֙ מִיֶּדְכֶ֔ן וְלֹֽא־ יִהְי֥וּ ע֛וֹד בְּיֶדְכֶ֖ן לִמְצוּדָ֑ה וִֽידַעְתֶּ֖ן כִּֽי־ אֲנִ֥י יְהוָֽה׃ (נו) יַ֣עַן הַכְא֤וֹת לֵב־ צַדִּיק֙ שֶׁ֔קֶר וַאֲנִ֖י לֹ֣א הִכְאַבְתִּ֑יו וּלְחַזֵּק֙ יְדֵ֣י רָשָׁ֔ע לְבִלְתִּי־ שׁ֛וּב מִדַּרְכּ֥וֹ הָרָ֖ע לְהַחֲיֹתֽוֹ׃ (נז) לָכֵ֗ן שָׁ֚וְא לֹ֣א תֶחֱזֶ֔ינָה וְקֶ֖סֶם לֹא־ תִקְסַ֣מְנָה ע֑וֹד וְהִצַּלְתִּ֤י אֶת־ עַמִּי֙ מִיֶּדְכֶ֔ן וִֽידַעְתֶּ֖ן כִּֽי־ אֲנִ֥י יְהוָֽה׃
