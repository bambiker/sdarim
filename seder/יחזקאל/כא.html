יחזקאל סדר כא

פתיחה באתר · ספר יחזקאל כולו ·

(א) וְנָתַתִּ֥י אוֹתָ֛ם וּסְבִיב֥וֹת גִּבְעָתִ֖י בְּרָכָ֑ה וְהוֹרַדְתִּ֤י הַגֶּ֙שֶׁם֙ בְּעִתּ֔וֹ גִּשְׁמֵ֥י בְרָכָ֖ה יִֽהְיֽוּ׃ (ב) וְנָתַן֩ עֵ֨ץ הַשָּׂדֶ֜ה אֶת־ פִּרְי֗וֹ וְהָאָ֙רֶץ֙ תִּתֵּ֣ן יְבוּלָ֔הּ וְהָי֥וּ עַל־ אַדְמָתָ֖ם לָבֶ֑טַח וְֽיָדְע֞וּ כִּי־ אֲנִ֣י יְהוָ֗ה בְּשִׁבְרִי֙ אֶת־ מֹט֣וֹת עֻלָּ֔ם וְהִ֨צַּלְתִּ֔ים מִיַּ֖ד הָעֹבְדִ֥ים בָּהֶֽם׃ (ג) וְלֹא־ יִהְי֨וּ ע֥וֹד בַּז֙ לַגּוֹיִ֔ם וְחַיַּ֥ת הָאָ֖רֶץ לֹ֣א תֹאכְלֵ֑ם וְיָשְׁב֥וּ לָבֶ֖טַח וְאֵ֥ין מַחֲרִֽיד׃ (ד) וַהֲקִמֹתִ֥י לָהֶ֛ם מַטָּ֖ע לְשֵׁ֑ם וְלֹֽא־ יִהְי֨וּ ע֜וֹד אֲסֻפֵ֤י רָעָב֙ בָּאָ֔רֶץ וְלֹֽא־ יִשְׂא֥וּ ע֖וֹד כְּלִמַּ֥ת הַגּוֹיִֽם׃ (ה) וְיָדְע֗וּ כִּ֣י אֲנִ֧י יְהוָ֛ה אֱלֹהֵיהֶ֖ם אִתָּ֑ם וְהֵ֗מָּה עַמִּי֙ בֵּ֣ית יִשְׂרָאֵ֔ל נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ו) וְאַתֵּ֥ן צֹאנִ֛י צֹ֥אן מַרְעִיתִ֖י אָדָ֣ם אַתֶּ֑ם אֲנִי֙ אֱלֹ֣הֵיכֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ח) בֶּן־ אָדָ֕ם שִׂ֥ים פָּנֶ֖יךָ עַל־ הַ֣ר שֵׂעִ֑יר וְהִנָּבֵ֖א עָלָֽיו׃ (ט) וְאָמַ֣רְתָּ לּ֗וֹ כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה הִנְנִ֥י אֵלֶ֖יךָ הַר־ שֵׂעִ֑יר וְנָטִ֤יתִי יָדִי֙ עָלֶ֔יךָ וּנְתַתִּ֖יךָ שְׁמָמָ֥ה וּמְשַׁמָּֽה׃ (י) עָרֶ֙יךָ֙ חָרְבָּ֣ה אָשִׂ֔ים וְאַתָּ֖ה שְׁמָמָ֣ה תִֽהְיֶ֑ה וְיָדַעְתָּ֖ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (יא) יַ֗עַן הֱי֤וֹת לְךָ֙ אֵיבַ֣ת עוֹלָ֔ם וַתַּגֵּ֥ר אֶת־ בְּנֵֽי־ יִשְׂרָאֵ֖ל עַל־ יְדֵי־ חָ֑רֶב בְּעֵ֣ת אֵידָ֔ם בְּעֵ֖ת עֲוֺ֥ן קֵֽץ׃ (יב) לָכֵ֣ן חַי־ אָ֗נִי נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה כִּֽי־ לְדָ֥ם אֶעֶשְׂךָ֖ וְדָ֣ם יִרְדֲּפֶ֑ךָ אִם־ לֹ֥א דָ֛ם שָׂנֵ֖אתָ וְדָ֥ם יִרְדֲּפֶֽךָ׃ (יג) וְנָֽתַתִּי֙ אֶת־ הַ֣ר שֵׂעִ֔יר לְשִֽׁמְמָ֖ה וּשְׁמָמָ֑ה וְהִכְרַתִּ֥י מִמֶּ֖נּוּ עֹבֵ֥ר וָשָֽׁב׃ (יד) וּמִלֵּאתִ֥י אֶת־ הָרָ֖יו חֲלָלָ֑יו גִּבְעוֹתֶ֤יךָ וְגֵאוֹתֶ֙יךָ֙ וְכָל־ אֲפִיקֶ֔יךָ חַלְלֵי־ חֶ֖רֶב יִפְּל֥וּ בָהֶֽם׃ (טו) שִֽׁמְמ֤וֹת עוֹלָם֙ אֶתֶּנְךָ֔ וְעָרֶ֖יךָ לֹ֣א (תישבנה) תָשֹׁ֑בְנָה וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (טז) יַ֣עַן אֲ֠מָרְךָ אֶת־ שְׁנֵ֨י הַגּוֹיִ֜ם וְאֶת־ שְׁתֵּ֧י הָאֲרָצ֛וֹת לִ֥י תִהְיֶ֖ינָה וִֽירַשְׁנ֑וּהָ וַֽיהוָ֖ה שָׁ֥ם הָיָֽה׃ (יז) לָכֵ֣ן חַי־ אָ֗נִי נְאֻם֮ אֲדֹנָ֣י יְהוִה֒ וְעָשִׂ֗יתִי כְּאַפְּךָ֙ וּכְקִנְאָ֣תְךָ֔ אֲשֶׁ֣ר עָשִׂ֔יתָה מִשִּׂנְאָתֶ֖יךָ בָּ֑ם וְנוֹדַ֥עְתִּי בָ֖ם כַּאֲשֶׁ֥ר אֶשְׁפְּטֶֽךָ׃ (יח) וְֽיָדַעְתָּ֮ כִּֽי־ אֲנִ֣י יְהוָה֒ שָׁמַ֣עְתִּי ׀ אֶת־ כָּל־ נָאָֽצוֹתֶ֗יךָ אֲשֶׁ֥ר אָמַ֛רְתָּ עַל־ הָרֵ֥י יִשְׂרָאֵ֖ל לֵאמֹ֣ר ׀ (שממה) שָׁמֵ֑מוּ לָ֥נוּ נִתְּנ֖וּ לְאָכְלָֽה׃ (יט) וַתַּגְדִּ֤ילוּ עָלַי֙ בְּפִיכֶ֔ם וְהַעְתַּרְתֶּ֥ם עָלַ֖י דִּבְרֵיכֶ֑ם אֲנִ֖י שָׁמָֽעְתִּי׃ (כ) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה כִּשְׂמֹ֙חַ֙ כָּל־ הָאָ֔רֶץ שְׁמָמָ֖ה אֶעֱשֶׂה־ לָּֽךְ׃ (כא) כְּשִׂמְחָ֨תְךָ֜ לְנַחְלַ֧ת בֵּֽית־ יִשְׂרָאֵ֛ל עַ֥ל אֲשֶׁר־ שָׁמֵ֖מָה כֵּ֣ן אֶעֱשֶׂה־ לָּ֑ךְ שְׁמָמָ֨ה תִֽהְיֶ֤ה הַר־ שֵׂעִיר֙ וְכָל־ אֱד֣וֹם כֻּלָּ֔הּ וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כב) וְאַתָּ֣ה בֶן־ אָדָ֔ם הִנָּבֵ֖א אֶל־ הָרֵ֣י יִשְׂרָאֵ֑ל וְאָ֣מַרְתָּ֔ הָרֵי֙ יִשְׂרָאֵ֔ל שִׁמְע֖וּ דְּבַר־ יְהוָֽה׃ (כג) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֣עַן אָמַ֧ר הָאוֹיֵ֛ב עֲלֵיכֶ֖ם הֶאָ֑ח וּבָמ֣וֹת עוֹלָ֔ם לְמֽוֹרָשָׁ֖ה הָ֥יְתָה לָּֽנוּ׃ (כד) לָכֵן֙ הִנָּבֵ֣א וְאָמַרְתָּ֔ כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה יַ֣עַן בְּיַ֡עַן שַׁמּוֹת֩ וְשָׁאֹ֨ף אֶתְכֶ֜ם מִסָּבִ֗יב לִֽהְיוֹתְכֶ֤ם מֽוֹרָשָׁה֙ לִשְׁאֵרִ֣ית הַגּוֹיִ֔ם וַתֵּֽעֲל֛וּ עַל־ שְׂפַ֥ת לָשׁ֖וֹן וְדִבַּת־ עָֽם׃ (כה) לָכֵן֙ הָרֵ֣י יִשְׂרָאֵ֔ל שִׁמְע֖וּ דְּבַר־ אֲדֹנָ֣י יְהוִ֑ה כֹּֽה־ אָמַ֣ר אֲדֹנָ֣י יְ֠הוִה לֶהָרִ֨ים וְלַגְּבָע֜וֹת לָאֲפִיקִ֣ים וְלַגֵּאָי֗וֹת וְלֶחֳרָב֤וֹת הַשֹּֽׁמְמוֹת֙ וְלֶעָרִ֣ים הַנֶּעֱזָב֔וֹת אֲשֶׁ֨ר הָי֤וּ לְבַז֙ וּלְלַ֔עַג לִשְׁאֵרִ֥ית הַגּוֹיִ֖ם אֲשֶׁ֥ר מִסָּבִֽיב׃ (כו) לָכֵ֗ן כֹּֽה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ אִם־ לֹ֠א בְּאֵ֨שׁ קִנְאָתִ֥י דִבַּ֛רְתִּי עַל־ שְׁאֵרִ֥ית הַגּוֹיִ֖ם וְעַל־ אֱד֣וֹם כֻּלָּ֑א אֲשֶׁ֣ר נָתְנֽוּ־ אֶת־ אַרְצִ֣י ׀ לָ֠הֶם לְמ֨וֹרָשָׁ֜ה בְּשִׂמְחַ֤ת כָּל־ לֵבָב֙ בִּשְׁאָ֣ט נֶ֔פֶשׁ לְמַ֥עַן מִגְרָשָׁ֖הּ לָבַֽז׃ (כז) לָכֵ֕ן הִנָּבֵ֖א עַל־ אַדְמַ֣ת יִשְׂרָאֵ֑ל וְאָמַרְתָּ֡ לֶהָרִ֣ים וְ֠לַגְּבָעוֹת לָאֲפִיקִ֨ים וְלַגֵּאָי֜וֹת כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה הִנְנִ֨י בְקִנְאָתִ֤י וּבַחֲמָתִי֙ דִּבַּ֔רְתִּי יַ֛עַן כְּלִמַּ֥ת גּוֹיִ֖ם נְשָׂאתֶֽם׃ (כח) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה אֲנִ֖י נָשָׂ֣אתִי אֶת־ יָדִ֑י אִם־ לֹ֤א הַגּוֹיִם֙ אֲשֶׁ֣ר לָכֶ֣ם מִסָּבִ֔יב הֵ֖מָּה כְּלִמָּתָ֥ם יִשָּֽׂאוּ׃ (כט) וְאַתֶּ֞ם הָרֵ֤י יִשְׂרָאֵל֙ עַנְפְּכֶ֣ם תִּתֵּ֔נוּ וּפֶרְיְכֶ֥ם תִּשְׂא֖וּ לְעַמִּ֣י יִשְׂרָאֵ֑ל כִּ֥י קֵרְב֖וּ לָבֽוֹא׃ (ל) כִּ֖י הִנְנִ֣י אֲלֵיכֶ֑ם וּפָנִ֣יתִי אֲלֵיכֶ֔ם וְנֶעֱבַדְתֶּ֖ם וְנִזְרַעְתֶּֽם׃ (לא) וְהִרְבֵּיתִ֤י עֲלֵיכֶם֙ אָדָ֔ם כָּל־ בֵּ֥ית יִשְׂרָאֵ֖ל כֻּלֹּ֑ה וְנֹֽשְׁבוּ֙ הֶֽעָרִ֔ים וְהֶחֳרָב֖וֹת תִּבָּנֶֽינָה׃ (לב) וְהִרְבֵּיתִ֧י עֲלֵיכֶ֛ם אָדָ֥ם וּבְהֵמָ֖ה וְרָב֣וּ וּפָר֑וּ וְהוֹשַׁבְתִּ֨י אֶתְכֶ֜ם כְּקַדְמֽוֹתֵיכֶ֗ם וְהֵטִֽבֹתִי֙ מֵרִאשֹׁ֣תֵיכֶ֔ם וִֽידַעְתֶּ֖ם כִּֽי־ אֲנִ֥י יְהוָֽה׃ (לג) וְהוֹלַכְתִּי֩ עֲלֵיכֶ֨ם אָדָ֜ם אֶת־ עַמִּ֤י יִשְׂרָאֵל֙ וִֽירֵשׁ֔וּךָ וְהָיִ֥יתָ לָהֶ֖ם לְנַחֲלָ֑ה וְלֹא־ תוֹסִ֥ף ע֖וֹד לְשַׁכְּלָֽם׃ (לד) כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֚עַן אֹמְרִ֣ים לָכֶ֔ם אֹכֶ֥לֶת אָדָ֖ם (אתי) אָ֑תְּ וּמְשַׁכֶּ֥לֶת (גויך) גּוֹיַ֖יִךְ הָיִֽית׃ (לה) לָכֵ֗ן אָדָם֙ לֹא־ תֹ֣אכְלִי ע֔וֹד (וגויך) וְגוֹיַ֖יִךְ לֹ֣א (תכשלי)־ תְשַׁכְּלִי־ ע֑וֹד נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לו) וְלֹא־ אַשְׁמִ֨יעַ אֵלַ֤יִךְ עוֹד֙ כְּלִמַּ֣ת הַגּוֹיִ֔ם וְחֶרְפַּ֥ת עַמִּ֖ים לֹ֣א תִשְׂאִי־ ע֑וֹד (וגויך) וְגוֹיַ֙יִךְ֙ לֹא־ תַכְשִׁ֣לִי ע֔וֹד נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לז) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (לח) בֶּן־ אָדָ֗ם בֵּ֤ית יִשְׂרָאֵל֙ יֹשְׁבִ֣ים עַל־ אַדְמָתָ֔ם וַיְטַמְּא֣וּ אוֹתָ֔הּ בְּדַרְכָּ֖ם וּבַעֲלִֽילוֹתָ֑ם כְּטֻמְאַת֙ הַנִּדָּ֔ה הָיְתָ֥ה דַרְכָּ֖ם לְפָנָֽי׃ (לט) וָאֶשְׁפֹּ֤ךְ חֲמָתִי֙ עֲלֵיהֶ֔ם עַל־ הַדָּ֖ם אֲשֶׁר־ שָׁפְכ֣וּ עַל־ הָאָ֑רֶץ וּבְגִלּוּלֵיהֶ֖ם טִמְּאֽוּהָ׃ (מ) וָאָפִ֤יץ אֹתָם֙ בַּגּוֹיִ֔ם וַיִּזָּר֖וּ בָּאֲרָצ֑וֹת כְּדַרְכָּ֥ם וְכַעֲלִילוֹתָ֖ם שְׁפַטְתִּֽים׃ (מא) וַיָּב֗וֹא אֶל־ הַגּוֹיִם֙ אֲשֶׁר־ בָּ֣אוּ שָׁ֔ם וַֽיְחַלְּל֖וּ אֶת־ שֵׁ֣ם קָדְשִׁ֑י בֶּאֱמֹ֤ר לָהֶם֙ עַם־ יְהוָ֣ה אֵ֔לֶּה וּמֵאַרְצ֖וֹ יָצָֽאוּ׃ (מב) וָאֶחְמֹ֖ל עַל־ שֵׁ֣ם קָדְשִׁ֑י אֲשֶׁ֤ר חִלְּל֙וּהוּ֙ בֵּ֣ית יִשְׂרָאֵ֔ל בַּגּוֹיִ֖ם אֲשֶׁר־ בָּ֥אוּ שָֽׁמָּה׃ (מג) לָכֵ֞ן אֱמֹ֣ר לְבֵֽית־ יִשְׂרָאֵ֗ל כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה לֹ֧א לְמַעַנְכֶ֛ם אֲנִ֥י עֹשֶׂ֖ה בֵּ֣ית יִשְׂרָאֵ֑ל כִּ֤י אִם־ לְשֵׁם־ קָדְשִׁי֙ אֲשֶׁ֣ר חִלַּלְתֶּ֔ם בַּגּוֹיִ֖ם אֲשֶׁר־ בָּ֥אתֶם שָֽׁם׃ (מד) וְקִדַּשְׁתִּ֞י אֶת־ שְׁמִ֣י הַגָּד֗וֹל הַֽמְחֻלָּל֙ בַּגּוֹיִ֔ם אֲשֶׁ֥ר חִלַּלְתֶּ֖ם בְּתוֹכָ֑ם וְיָדְע֨וּ הַגּוֹיִ֜ם כִּי־ אֲנִ֣י יְהוָ֗ה נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה בְּהִקָּדְשִׁ֥י בָכֶ֖ם לְעֵינֵיהֶֽם׃ (מה) וְלָקַחְתִּ֤י אֶתְכֶם֙ מִן־ הַגּוֹיִ֔ם וְקִבַּצְתִּ֥י אֶתְכֶ֖ם מִכָּל־ הָאֲרָצ֑וֹת וְהֵבֵאתִ֥י אֶתְכֶ֖ם אֶל־ אַדְמַתְכֶֽם׃
