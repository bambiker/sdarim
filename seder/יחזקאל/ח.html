יחזקאל סדר ח

פתיחה באתר · ספר יחזקאל כולו ·

(א) וַיֵּ֨צֵא לָ֥ךְ שֵׁ֛ם בַּגּוֹיִ֖ם בְּיָפְיֵ֑ךְ כִּ֣י ׀ כָּלִ֣יל ה֗וּא בַּֽהֲדָרִי֙ אֲשֶׁר־ שַׂ֣מְתִּי עָלַ֔יִךְ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ב) וַתִּבְטְחִ֣י בְיָפְיֵ֔ךְ וַתִּזְנִ֖י עַל־ שְׁמֵ֑ךְ וַתִּשְׁפְּכִ֧י אֶת־ תַּזְנוּתַ֛יִךְ עַל־ כָּל־ עוֹבֵ֖ר לוֹ־ יֶֽהִי׃ (ג) וַתִּקְחִ֣י מִבְּגָדַ֗יִךְ וַתַּֽעֲשִׂי־ לָךְ֙ בָּמ֣וֹת טְלֻא֔וֹת וַתִּזְנִ֖י עֲלֵיהֶ֑ם לֹ֥א בָא֖וֹת וְלֹ֥א יִהְיֶֽה׃ (ד) וַתִּקְחִ֞י כְּלֵ֣י תִפְאַרְתֵּ֗ךְ מִזְּהָבִ֤י וּמִכַּסְפִּי֙ אֲשֶׁ֣ר נָתַ֣תִּי לָ֔ךְ וַתַּעֲשִׂי־ לָ֖ךְ צַלְמֵ֣י זָכָ֑ר וַתִּזְנִי־ בָֽם׃ (ה) וַתִּקְחִ֛י אֶת־ בִּגְדֵ֥י רִקְמָתֵ֖ךְ וַתְּכַסִּ֑ים וְשַׁמְנִי֙ וּקְטָרְתִּ֔י (נתתי) נָתַ֖תְּ לִפְנֵיהֶֽם׃ (ו) וְלַחְמִי֩ אֲשֶׁר־ נָתַ֨תִּי לָ֜ךְ סֹ֣לֶת וָשֶׁ֤מֶן וּדְבַשׁ֙ הֶֽאֱכַלְתִּ֔יךְ וּנְתַתִּ֧יהוּ לִפְנֵיהֶ֛ם לְרֵ֥יחַ נִיחֹ֖חַ וַיֶּ֑הִי נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ז) וַתִּקְחִ֞י אֶת־ בָּנַ֤יִךְ וְאֶת־ בְּנוֹתַ֙יִךְ֙ אֲשֶׁ֣ר יָלַ֣דְתְּ לִ֔י וַתִּזְבָּחִ֥ים לָהֶ֖ם לֶאֱכ֑וֹל הַמְעַ֖ט (מתזנתך) מִתַּזְנוּתָֽיִךְ׃ (ח) וַֽתִּשְׁחֲטִ֖י אֶת־ בָּנָ֑י וַֽתִּתְּנִ֔ים בְּהַעֲבִ֥יר אוֹתָ֖ם לָהֶֽם׃ (ט) וְאֵ֤ת כָּל־ תּוֹעֲבֹתַ֙יִךְ֙ וְתַזְנֻתַ֔יִךְ לֹ֥א (זכרתי) זָכַ֖רְתְּ אֶת־ יְמֵ֣י נְעוּרָ֑יִךְ בִּֽהְיוֹתֵךְ֙ עֵרֹ֣ם וְעֶרְיָ֔ה מִתְבּוֹסֶ֥סֶת בְּדָמֵ֖ךְ הָיִֽית׃ (י) וַיְהִ֕י אַחֲרֵ֖י כָּל־ רָעָתֵ֑ךְ א֣וֹי א֣וֹי לָ֔ךְ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יא) וַתִּבְנִי־ לָ֖ךְ גֶּ֑ב וַתַּעֲשִׂי־ לָ֥ךְ רָמָ֖ה בְּכָל־ רְחֽוֹב׃ (יב) אֶל־ כָּל־ רֹ֣אשׁ דֶּ֗רֶךְ בָּנִית֙ רָֽמָתֵ֔ךְ וַתְּתַֽעֲבִי֙ אֶת־ יָפְיֵ֔ךְ וַתְּפַשְּׂקִ֥י אֶת־ רַגְלַ֖יִךְ לְכָל־ עוֹבֵ֑ר וַתַּרְבִּ֖י אֶת־ (תזנתך) תַּזְנוּתָֽיִךְ׃ (יג) וַתִּזְנִ֧י אֶל־ בְּנֵֽי־ מִצְרַ֛יִם שְׁכֵנַ֖יִךְ גִּדְלֵ֣י בָשָׂ֑ר וַתַּרְבִּ֥י אֶת־ תַּזְנֻתֵ֖ךְ לְהַכְעִיסֵֽנִי׃ (יד) וְהִנֵּ֨ה נָטִ֤יתִי יָדִי֙ עָלַ֔יִךְ וָאֶגְרַ֖ע חֻקֵּ֑ךְ וָאֶתְּנֵ֞ךְ בְּנֶ֤פֶשׁ שֹׂנְאוֹתַ֙יִךְ֙ בְּנ֣וֹת פְּלִשְׁתִּ֔ים הַנִּכְלָמ֖וֹת מִדַּרְכֵּ֥ךְ זִמָּֽה׃ (טו) וַתִּזְנִי֙ אֶל־ בְּנֵ֣י אַשּׁ֔וּר מִבִּלְתִּ֖י שָׂבְעָתֵ֑ךְ וַתִּזְנִ֕ים וְגַ֖ם לֹ֥א שָׂבָֽעַתְּ׃ (טז) וַתַּרְבִּ֧י אֶת־ תַּזְנוּתֵ֛ךְ אֶל־ אֶ֥רֶץ כְּנַ֖עַן כַּשְׂדִּ֑ימָה וְגַם־ בְּזֹ֖את לֹ֥א שָׂבָֽעַתְּ׃ (יז) מָ֤ה אֲמֻלָה֙ לִבָּתֵ֔ךְ נְאֻ֖ם אֲדֹנָ֣י יְהוִ֑ה בַּעֲשׂוֹתֵךְ֙ אֶת־ כָּל־ אֵ֔לֶּה מַעֲשֵׂ֥ה אִשָּֽׁה־ זוֹנָ֖ה שַׁלָּֽטֶת׃ (יח) בִּבְנוֹתַ֤יִךְ גַּבֵּךְ֙ בְּרֹ֣אשׁ כָּל־ דֶּ֔רֶךְ וְרָמָתֵ֥ךְ (עשיתי) עָשִׂ֖ית בְּכָל־ רְח֑וֹב וְלֹא־ (הייתי) הָיִ֥ית כַּזּוֹנָ֖ה לְקַלֵּ֥ס אֶתְנָֽן׃ (יט) הָאִשָּׁ֖ה הַמְּנָאָ֑פֶת תַּ֣חַת אִישָׁ֔הּ תִּקַּ֖ח אֶת־ זָרִֽים׃ (כ) לְכָל־ זֹנ֖וֹת יִתְּנוּ־ נֵ֑דֶה וְאַ֨תְּ נָתַ֤תְּ אֶת־ נְדָנַ֙יִךְ֙ לְכָל־ מְאַֽהֲבַ֔יִךְ וַתִּשְׁחֳדִ֣י אוֹתָ֗ם לָב֥וֹא אֵלַ֛יִךְ מִסָּבִ֖יב בְּתַזְנוּתָֽיִךְ׃ (כא) וַיְהִי־ בָ֨ךְ הֵ֤פֶךְ מִן־ הַנָּשִׁים֙ בְּתַזְנוּתַ֔יִךְ וְאַחֲרַ֖יִךְ לֹ֣א זוּנָּ֑ה וּבְתִתֵּ֣ךְ אֶתְנָ֗ן וְאֶתְנַ֛ן לֹ֥א נִתַּן־ לָ֖ךְ וַתְּהִ֥י לְהֶֽפֶךְ׃ (כב) לָכֵ֣ן זוֹנָ֔ה שִׁמְעִ֖י דְּבַר־ יְהוָֽה׃ (כג) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהֹוִ֗ה יַ֣עַן הִשָּׁפֵ֤ךְ נְחֻשְׁתֵּךְ֙ וַתִּגָּלֶ֣ה עֶרְוָתֵ֔ךְ בְּתַזְנוּתַ֖יִךְ עַל־ מְאַהֲבָ֑יִךְ וְעַל֙ כָּל־ גִּלּוּלֵ֣י תוֹעֲבוֹתַ֔יִךְ וְכִדְמֵ֣י בָנַ֔יִךְ אֲשֶׁ֥ר נָתַ֖תְּ לָהֶֽם׃ (כד) לָ֠כֵן הִנְנִ֨י מְקַבֵּ֤ץ אֶת־ כָּל־ מְאַהֲבַ֙יִךְ֙ אֲשֶׁ֣ר עָרַ֣בְתְּ עֲלֵיהֶ֔ם וְאֵת֙ כָּל־ אֲשֶׁ֣ר אָהַ֔בְתְּ עַ֖ל כָּל־ אֲשֶׁ֣ר שָׂנֵ֑את וְקִבַּצְתִּי֩ אֹתָ֨ם עָלַ֜יִךְ מִסָּבִ֗יב וְגִלֵּיתִ֤י עֶרְוָתֵךְ֙ אֲלֵהֶ֔ם וְרָא֖וּ אֶת־ כָּל־ עֶרְוָתֵֽךְ׃ (כה) וּשְׁפַטְתִּיךְ֙ מִשְׁפְּטֵ֣י נֹאֲפ֔וֹת וְשֹׁפְכֹ֖ת דָּ֑ם וּנְתַתִּ֕יךְ דַּ֥ם חֵמָ֖ה וְקִנְאָֽה׃ (כו) וְנָתַתִּ֨י אוֹתָ֜ךְ בְּיָדָ֗ם וְהָרְס֤וּ גַבֵּךְ֙ וְנִתְּצ֣וּ רָמֹתַ֔יִךְ וְהִפְשִׁ֤יטוּ אוֹתָךְ֙ בְּגָדַ֔יִךְ וְלָקְח֖וּ כְּלֵ֣י תִפְאַרְתֵּ֑ךְ וְהִנִּיח֖וּךְ עֵירֹ֥ם וְעֶרְיָֽה׃ (כז) וְהֶעֱל֤וּ עָלַ֙יִךְ֙ קָהָ֔ל וְרָגְמ֥וּ אוֹתָ֖ךְ בָּאָ֑בֶן וּבִתְּק֖וּךְ בְּחַרְבוֹתָֽם׃ (כח) וְשָׂרְפ֤וּ בָתַּ֙יִךְ֙ בָּאֵ֔שׁ וְעָשׂוּ־ בָ֣ךְ שְׁפָטִ֔ים לְעֵינֵ֖י נָשִׁ֣ים רַבּ֑וֹת וְהִשְׁבַּתִּיךְ֙ מִזּוֹנָ֔ה וְגַם־ אֶתְנַ֖ן לֹ֥א תִתְּנִי־ עֽוֹד׃ (כט) וַהֲנִחֹתִ֤י חֲמָתִי֙ בָּ֔ךְ וְסָ֥רָה קִנְאָתִ֖י מִמֵּ֑ךְ וְשָׁ֣קַטְתִּ֔י וְלֹ֥א אֶכְעַ֖ס עֽוֹד׃ (ל) יַ֗עַן אֲשֶׁ֤ר לֹֽא־ (זכרתי) זָכַרְתְּ֙ אֶת־ יְמֵ֣י נְעוּרַ֔יִךְ וַתִּרְגְּזִי־ לִ֖י בְּכָל־ אֵ֑לֶּה וְגַם־ אֲנִ֨י הֵ֜א דַּרְכֵּ֣ךְ ׀ בְּרֹ֣אשׁ נָתַ֗תִּי נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה וְלֹ֤א (עשיתי) עָשִׂית֙ אֶת־ הַזִּמָּ֔ה עַ֖ל כָּל־ תּוֹעֲבֹתָֽיִךְ׃ (לא) הִנֵּה֙ כָּל־ הַמֹּשֵׁ֔ל עָלַ֥יִךְ יִמְשֹׁ֖ל לֵאמֹ֑ר כְּאִמָּ֖ה בִּתָּֽהּ׃ (לב) בַּת־ אִמֵּ֣ךְ אַ֔תְּ גֹּעֶ֥לֶת אִישָׁ֖הּ וּבָנֶ֑יהָ וַאֲח֨וֹת אֲחוֹתֵ֜ךְ אַ֗תְּ אֲשֶׁ֤ר גָּֽעֲ֙לוּ֙ אַנְשֵׁיהֶ֣ן וּבְנֵיהֶ֔ן אִמְּכֶ֣ן חִתִּ֔ית וַאֲבִיכֶ֖ן אֱמֹרִֽי׃ (לג) וַאֲחוֹתֵ֨ךְ הַגְּדוֹלָ֤ה שֹֽׁמְרוֹן֙ הִ֣יא וּבְנוֹתֶ֔יהָ הַיּוֹשֶׁ֖בֶת עַל־ שְׂמֹאולֵ֑ךְ וַאֲחוֹתֵ֞ךְ הַקְּטַנָּ֣ה מִמֵּ֗ךְ הַיּוֹשֶׁ֙בֶת֙ מִֽימִינֵ֔ךְ סְדֹ֖ם וּבְנוֹתֶֽיהָ׃ (לד) וְלֹ֤א בְדַרְכֵיהֶן֙ הָלַ֔כְתְּ וּבְתוֹעֲבֽוֹתֵיהֶ֖ן (עשיתי) עָשִׂ֑ית כִּמְעַ֣ט קָ֔ט וַתַּשְׁחִ֥תִי מֵהֵ֖ן בְּכָל־ דְּרָכָֽיִךְ׃ (לה) חַי־ אָ֗נִי נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה אִם־ עָֽשְׂתָה֙ סְדֹ֣ם אֲחוֹתֵ֔ךְ הִ֖יא וּבְנוֹתֶ֑יהָ כַּאֲשֶׁ֣ר עָשִׂ֔ית אַ֖תְּ וּבְנוֹתָֽיִךְ׃ (לו) הִנֵּה־ זֶ֣ה הָיָ֔ה עֲוֺ֖ן סְדֹ֣ם אֲחוֹתֵ֑ךְ גָּא֨וֹן שִׂבְעַת־ לֶ֜חֶם וְשַׁלְוַ֣ת הַשְׁקֵ֗ט הָ֤יָה לָהּ֙ וְלִבְנוֹתֶ֔יהָ וְיַד־ עָנִ֥י וְאֶבְי֖וֹן לֹ֥א הֶחֱזִֽיקָה׃ (לז) וַֽתִּגְבְּהֶ֔ינָה וַתַּעֲשֶׂ֥ינָה תוֹעֵבָ֖ה לְפָנָ֑י וָאָסִ֥יר אֶתְהֶ֖ן כַּאֲשֶׁ֥ר רָאִֽיתִי׃ (לח) וְשֹׁ֣מְר֔וֹן כַּחֲצִ֥י חַטֹּאתַ֖יִךְ לֹ֣א חָטָ֑אָה וַתַּרְבִּ֤י אֶת־ תּוֹעֲבוֹתַ֙יִךְ֙ מֵהֵ֔נָּה וַתְּצַדְּקִי֙ אֶת־ (אחותך) אֲחוֹתַ֔יִךְ בְּכָל־ תּוֹעֲבוֹתַ֖יִךְ אֲשֶׁ֥ר (עשיתי) עָשִֽׂית׃ (לט) גַּם־ אַ֣תְּ ׀ שְׂאִ֣י כְלִמָּתֵ֗ךְ אֲשֶׁ֤ר פִּלַּלְתְּ֙ לַֽאֲחוֹתֵ֔ךְ בְּחַטֹּאתַ֛יִךְ אֲשֶׁר־ הִתְעַ֥בְתְּ מֵהֵ֖ן תִּצְדַּ֣קְנָה מִמֵּ֑ךְ וְגַם־ אַ֥תְּ בּ֙וֹשִׁי֙ וּשְׂאִ֣י כְלִמָּתֵ֔ךְ בְּצַדֶּקְתֵּ֖ךְ אַחְיוֹתֵֽךְ׃ (מ) וְשַׁבְתִּי֙ אֶת־ שְׁבִ֣יתְהֶ֔ן אֶת־ (שבית) שְׁב֤וּת סְדֹם֙ וּבְנוֹתֶ֔יהָ וְאֶת־ (שבית) שְׁב֥וּת שֹׁמְר֖וֹן וּבְנוֹתֶ֑יהָ (ושבית) וּשְׁב֥וּת שְׁבִיתַ֖יִךְ בְּתוֹכָֽהְנָה׃ (מא) לְמַ֙עַן֙ תִּשְׂאִ֣י כְלִמָּתֵ֔ךְ וְנִכְלַ֕מְתְּ מִכֹּ֖ל אֲשֶׁ֣ר עָשִׂ֑ית בְּנַחֲמֵ֖ךְ אֹתָֽן׃ (מב) וַאֲחוֹתַ֗יִךְ סְדֹ֤ם וּבְנוֹתֶ֙יהָ֙ תָּשֹׁ֣בְןָ לְקַדְמָתָ֔ן וְשֹֽׁמְרוֹן֙ וּבְנוֹתֶ֔יהָ תָּשֹׁ֖בְןָ לְקַדְמָתָ֑ן וְאַתְּ֙ וּבְנוֹתַ֔יִךְ תְּשֻׁבֶ֖ינָה לְקַדְמַתְכֶֽן׃ (מג) וְל֤וֹא הָֽיְתָה֙ סְדֹ֣ם אֲחוֹתֵ֔ךְ לִשְׁמוּעָ֖ה בְּפִ֑יךְ בְּי֖וֹם גְּאוֹנָֽיִךְ׃ (מד) בְּטֶרֶם֮ תִּגָּלֶ֣ה רָעָתֵךְ֒ כְּמ֗וֹ עֵ֚ת חֶרְפַּ֣ת בְּנוֹת־ אֲרָ֔ם וְכָל־ סְבִיבוֹתֶ֖יהָ בְּנ֣וֹת פְּלִשְׁתִּ֑ים הַשָּׁאט֥וֹת אוֹתָ֖ךְ מִסָּבִֽיב׃ (מה) אֶת־ זִמָּתֵ֥ךְ וְאֶת־ תּוֹעֲבוֹתַ֖יִךְ אַ֣תְּ נְשָׂאתִ֑ים נְאֻ֖ם יְהוָֽה׃ (מו) כִּ֣י כֹ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה (ועשית) וְעָשִׂ֥יתִי אוֹתָ֖ךְ כַּאֲשֶׁ֣ר עָשִׂ֑ית אֲשֶׁר־ בָּזִ֥ית אָלָ֖ה לְהָפֵ֥ר בְּרִֽית׃
