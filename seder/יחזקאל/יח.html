יחזקאל סדר יח

45 פסוקים

פתיחה באתר · ספר יחזקאל כולו ·

(א) בַּיּ֣וֹם הַה֗וּא אַצְמִ֤יחַ קֶ֙רֶן֙ לְבֵ֣ית יִשְׂרָאֵ֔ל וּלְךָ֛ אֶתֵּ֥ן פִּתְחֽוֹן־ פֶּ֖ה בְּתוֹכָ֑ם וְיָדְע֖וּ כִּי־ אֲנִ֥י יְהוָֽה׃ (ב) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ג) בֶּן־ אָדָ֕ם הִנָּבֵא֙ וְאָ֣מַרְתָּ֔ כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הֵילִ֖ילוּ הָ֥הּ לַיּֽוֹם׃ (ד) כִּֽי־ קָר֣וֹב י֔וֹם וְקָר֥וֹב י֖וֹם לַֽיהוָ֑ה י֣וֹם עָנָ֔ן עֵ֥ת גּוֹיִ֖ם יִֽהְיֶֽה׃ (ה) וּבָאָ֥ה חֶ֙רֶב֙ בְּמִצְרַ֔יִם וְהָיְתָ֤ה חַלְחָלָה֙ בְּכ֔וּשׁ בִּנְפֹ֥ל חָלָ֖ל בְּמִצְרָ֑יִם וְלָקְח֣וּ הֲמוֹנָ֔הּ וְנֶהֶרְס֖וּ יְסוֹדֹתֶֽיהָ׃ (ו) כּ֣וּשׁ וּפ֤וּט וְלוּד֙ וְכָל־ הָעֶ֣רֶב וְכ֔וּב וּבְנֵ֖י אֶ֣רֶץ הַבְּרִ֑ית אִתָּ֖ם בַּחֶ֥רֶב יִפֹּֽלוּ׃ (ז) כֹּ֚ה אָמַ֣ר יְהוָ֔ה וְנָֽפְלוּ֙ סֹמְכֵ֣י מִצְרַ֔יִם וְיָרַ֖ד גְּא֣וֹן עֻזָּ֑הּ מִמִּגְדֹּ֣ל סְוֵנֵ֗ה בַּחֶ֙רֶב֙ יִפְּלוּ־ בָ֔הּ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (ח) וְנָשַׁ֕מּוּ בְּת֖וֹךְ אֲרָצ֣וֹת נְשַׁמּ֑וֹת וְעָרָ֕יו בְּתוֹךְ־ עָרִ֥ים נַחֲרָב֖וֹת תִּֽהְיֶֽינָה׃ (ט) וְיָדְע֖וּ כִּֽי־ אֲנִ֣י יְהוָ֑ה בְּתִתִּי־ אֵ֣שׁ בְּמִצְרַ֔יִם וְנִשְׁבְּר֖וּ כָּל־ עֹזְרֶֽיהָ׃ (י) בַּיּ֣וֹם הַה֗וּא יֵצְא֨וּ מַלְאָכִ֤ים מִלְּפָנַי֙ בַּצִּ֔ים לְהַחֲרִ֖יד אֶת־ כּ֣וּשׁ בֶּ֑טַח וְהָיְתָ֨ה חַלְחָלָ֤ה בָהֶם֙ בְּי֣וֹם מִצְרַ֔יִם כִּ֥י הִנֵּ֖ה בָּאָֽה׃ (יא) כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה וְהִשְׁבַּתִּי֙ אֶת־ הֲמ֣וֹן מִצְרַ֔יִם בְּיַ֖ד נְבוּכַדְרֶאצַּ֥ר מֶלֶךְ־ בָּבֶֽל׃ (יב) ה֠וּא וְעַמּ֤וֹ אִתּוֹ֙ עָרִיצֵ֣י גוֹיִ֔ם מֽוּבָאִ֖ים לְשַׁחֵ֣ת הָאָ֑רֶץ וְהֵרִ֤יקוּ חַרְבוֹתָם֙ עַל־ מִצְרַ֔יִם וּמָלְא֥וּ אֶת־ הָאָ֖רֶץ חָלָֽל׃ (יג) וְנָתַתִּ֤י יְאֹרִים֙ חָֽרָבָ֔ה וּמָכַרְתִּ֥י אֶת־ הָאָ֖רֶץ בְּיַד־ רָעִ֑ים וַהֲשִׁמֹּתִ֞י אֶ֤רֶץ וּמְלֹאָהּ֙ בְּיַד־ זָרִ֔ים אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (יד) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה וְהַאֲבַדְתִּ֨י גִלּוּלִ֜ים וְהִשְׁבַּתִּ֤י אֱלִילִים֙ מִנֹּ֔ף וְנָשִׂ֥יא מֵאֶֽרֶץ־ מִצְרַ֖יִם לֹ֣א יִֽהְיֶה־ ע֑וֹד וְנָתַתִּ֥י יִרְאָ֖ה בְּאֶ֥רֶץ מִצְרָֽיִם׃ (טו) וַהֲשִׁמֹּתִי֙ אֶת־ פַּתְר֔וֹס וְנָתַ֥תִּי אֵ֖שׁ בְּצֹ֑עַן וְעָשִׂ֥יתִי שְׁפָטִ֖ים בְּנֹֽא׃ (טז) וְשָׁפַכְתִּ֣י חֲמָתִ֔י עַל־ סִ֖ין מָע֣וֹז מִצְרָ֑יִם וְהִכְרַתִּ֖י אֶת־ הֲמ֥וֹן נֹֽא׃ (יז) וְנָתַ֤תִּי אֵשׁ֙ בְּמִצְרַ֔יִם ח֤וּל (תחיל) תָּחוּל֙ סִ֔ין וְנֹ֖א תִּהְיֶ֣ה לְהִבָּקֵ֑עַ וְנֹ֖ף צָרֵ֥י יוֹמָֽם׃ (יח) בַּח֥וּרֵי אָ֛וֶן וּפִי־ בֶ֖סֶת בַּחֶ֣רֶב יִפֹּ֑לוּ וְהֵ֖נָּה בַּשְּׁבִ֥י תֵלַֽכְנָה׃ (יט) וּבִֽתְחַפְנְחֵס֙ חָשַׂ֣ךְ הַיּ֔וֹם בְּשִׁבְרִי־ שָׁם֙ אֶת־ מֹט֣וֹת מִצְרַ֔יִם וְנִשְׁבַּת־ בָּ֖הּ גְּא֣וֹן עֻזָּ֑הּ הִ֚יא עָנָ֣ן יְכַסֶּ֔נָּה וּבְנוֹתֶ֖יהָ בַּשְּׁבִ֥י תֵלַֽכְנָה׃ (כ) וְעָשִׂ֥יתִי שְׁפָטִ֖ים בְּמִצְרָ֑יִם וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כא) וַיְהִ֗י בְּאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה בָּֽרִאשׁ֖וֹן בְּשִׁבְעָ֣ה לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כב) בֶּן־ אָדָ֕ם אֶת־ זְר֛וֹעַ פַּרְעֹ֥ה מֶֽלֶךְ־ מִצְרַ֖יִם שָׁבָ֑רְתִּי וְהִנֵּ֣ה לֹֽא־ חֻ֠בְּשָׁה לָתֵ֨ת רְפֻא֜וֹת לָשׂ֥וּם חִתּ֛וּל לְחָבְשָׁ֥הּ לְחָזְקָ֖הּ לִתְפֹּ֥שׂ בֶּחָֽרֶב׃ (כג) לָכֵ֞ן כֹּה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה הִנְנִי֙ אֶל־ פַּרְעֹ֣ה מֶֽלֶךְ־ מִצְרַ֔יִם וְשָֽׁבַרְתִּי֙ אֶת־ זְרֹ֣עֹתָ֔יו אֶת־ הַחֲזָקָ֖ה וְאֶת־ הַנִּשְׁבָּ֑רֶת וְהִפַּלְתִּ֥י אֶת־ הַחֶ֖רֶב מִיָּדֽוֹ׃ (כד) וַהֲפִצוֹתִ֥י אֶת־ מִצְרַ֖יִם בַּגּוֹיִ֑ם וְזֵרִיתִ֖ם בָּאֲרָצֽוֹת׃ (כה) וְחִזַּקְתִּ֗י אֶת־ זְרֹעוֹת֙ מֶ֣לֶךְ בָּבֶ֔ל וְנָתַתִּ֥י אֶת־ חַרְבִּ֖י בְּיָד֑וֹ וְשָׁבַרְתִּי֙ אֶת־ זְרֹע֣וֹת פַּרְעֹ֔ה וְנָאַ֛ק נַאֲק֥וֹת חָלָ֖ל לְפָנָֽיו׃ (כו) וְהַחֲזַקְתִּ֗י אֶת־ זְרֹעוֹת֙ מֶ֣לֶךְ בָּבֶ֔ל וּזְרֹע֥וֹת פַּרְעֹ֖ה תִּפֹּ֑לְנָה וְֽיָדְע֞וּ כִּֽי־ אֲנִ֣י יְהוָ֗ה בְּתִתִּ֤י חַרְבִּי֙ בְּיַ֣ד מֶֽלֶךְ־ בָּבֶ֔ל וְנָטָ֥ה אוֹתָ֖הּ אֶל־ אֶ֥רֶץ מִצְרָֽיִם׃ (כז) וַהֲפִצוֹתִ֤י אֶת־ מִצְרַ֙יִם֙ בַּגּוֹיִ֔ם וְזֵרִיתִ֥י אוֹתָ֖ם בָּאֲרָצ֑וֹת וְיָדְע֖וּ כִּֽי־ אֲנִ֥י יְהוָֽה׃ (כח) וַיְהִ֗י בְּאַחַ֤ת עֶשְׂרֵה֙ שָׁנָ֔ה בַּשְּׁלִישִׁ֖י בְּאֶחָ֣ד לַחֹ֑דֶשׁ הָיָ֥ה דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כט) בֶּן־ אָדָ֕ם אֱמֹ֛ר אֶל־ פַּרְעֹ֥ה מֶֽלֶךְ־ מִצְרַ֖יִם וְאֶל־ הֲמוֹנ֑וֹ אֶל־ מִ֖י דָּמִ֥יתָ בְגָדְלֶֽךָ׃ (ל) הִנֵּ֨ה אַשּׁ֜וּר אֶ֣רֶז בַּלְּבָנ֗וֹן יְפֵ֥ה עָנָ֛ף וְחֹ֥רֶשׁ מֵצַ֖ל וּגְבַ֣הּ קוֹמָ֑ה וּבֵ֣ין עֲבֹתִ֔ים הָיְתָ֖ה צַמַּרְתּֽוֹ׃ (לא) מַ֣יִם גִּדְּל֔וּהוּ תְּה֖וֹם רֹֽמְמָ֑תְהוּ אֶת־ נַהֲרֹתֶ֗יהָ הֹלֵךְ֙ סְבִיב֣וֹת מַטָּעָ֔הּ וְאֶת־ תְּעָלֹתֶ֣יהָ שִׁלְחָ֔ה אֶ֖ל כָּל־ עֲצֵ֥י הַשָּׂדֶֽה׃ (לב) עַל־ כֵּן֙ גָּבְהָ֣א קֹמָת֔וֹ מִכֹּ֖ל עֲצֵ֣י הַשָּׂדֶ֑ה וַתִּרְבֶּ֨ינָה סַֽרְעַפֹּתָ֜יו וַתֶּאֱרַ֧כְנָה (פארתו) פֹארֹתָ֛יו מִמַּ֥יִם רַבִּ֖ים בְּשַׁלְּחֽוֹ׃ (לג) בִּסְעַפֹּתָ֤יו קִֽנְנוּ֙ כָּל־ ע֣וֹף הַשָּׁמַ֔יִם וְתַ֤חַת פֹּֽארֹתָיו֙ יָֽלְד֔וּ כֹּ֖ל חַיַּ֣ת הַשָּׂדֶ֑ה וּבְצִלּוֹ֙ יֵֽשְׁב֔וּ כֹּ֖ל גּוֹיִ֥ם רַבִּֽים׃ (לד) וַיְּיִ֣ף בְּגָדְל֔וֹ בְּאֹ֖רֶךְ דָּֽלִיּוֹתָ֑יו כִּֽי־ הָיָ֥ה שָׁרְשׁ֖וֹ אֶל־ מַ֥יִם רַבִּֽים׃ (לה) אֲרָזִ֣ים לֹֽא־ עֲמָמֻהוּ֮ בְּגַן־ אֱלֹהִים֒ בְּרוֹשִׁ֗ים לֹ֤א דָמוּ֙ אֶל־ סְעַפֹּתָ֔יו וְעַרְמֹנִ֥ים לֹֽא־ הָי֖וּ כְּפֹֽארֹתָ֑יו כָּל־ עֵץ֙ בְּגַן־ אֱלֹהִ֔ים לֹא־ דָמָ֥ה אֵלָ֖יו בְּיָפְיֽוֹ׃ (לו) יָפֶ֣ה עֲשִׂיתִ֔יו בְּרֹ֖ב דָּֽלִיּוֹתָ֑יו וַיְקַנְאֻ֙הוּ֙ כָּל־ עֲצֵי־ עֵ֔דֶן אֲשֶׁ֖ר בְּגַ֥ן הָאֱלֹהִֽים׃ (לז) לָכֵ֗ן כֹּ֤ה אָמַר֙ אֲדֹנָ֣י יְהוִ֔ה יַ֕עַן אֲשֶׁ֥ר גָּבַ֖הְתָּ בְּקוֹמָ֑ה וַיִּתֵּ֤ן צַמַּרְתּוֹ֙ אֶל־ בֵּ֣ין עֲבוֹתִ֔ים וְרָ֥ם לְבָב֖וֹ בְּגָבְהֽוֹ׃ (לח) וְאֶ֨תְּנֵ֔הוּ בְּיַ֖ד אֵ֣יל גּוֹיִ֑ם עָשׂ֤וֹ יַֽעֲשֶׂה֙ ל֔וֹ כְּרִשְׁע֖וֹ גֵּרַשְׁתִּֽהוּ׃ (לט) וַיִּכְרְתֻ֧הוּ זָרִ֛ים עָרִיצֵ֥י גוֹיִ֖ם וַֽיִּטְּשֻׁ֑הוּ אֶל־ הֶ֠הָרִים וּבְכָל־ גֵּ֨אָי֜וֹת נָפְל֣וּ דָלִיּוֹתָ֗יו וַתִּשָּׁבַ֤רְנָה פֹֽארֹתָיו֙ בְּכֹל֙ אֲפִיקֵ֣י הָאָ֔רֶץ וַיֵּרְד֧וּ מִצִּלּ֛וֹ כָּל־ עַמֵּ֥י הָאָ֖רֶץ וַֽיִּטְּשֻֽׁהוּ׃ (מ) עַל־ מַפַּלְתּ֥וֹ יִשְׁכְּנ֖וּ כָּל־ ע֣וֹף הַשָּׁמָ֑יִם וְאֶל־ פֹּארֹתָ֣יו הָי֔וּ כֹּ֖ל חַיַּ֥ת הַשָּׂדֶֽה׃ (מא) לְמַ֡עַן אֲשֶׁר֩ לֹא־ יִגְבְּה֨וּ בְקוֹמָתָ֜ם כָּל־ עֲצֵי־ מַ֗יִם וְלֹֽא־ יִתְּנ֤וּ אֶת־ צַמַּרְתָּם֙ אֶל־ בֵּ֣ין עֲבֹתִ֔ים וְלֹֽא־ יַעַמְד֧וּ אֵלֵיהֶ֛ם בְּגָבְהָ֖ם כָּל־ שֹׁ֣תֵי מָ֑יִם כִּֽי־ כֻלָּם֩ נִתְּנ֨וּ לַמָּ֜וֶת אֶל־ אֶ֣רֶץ תַּחְתִּ֗ית בְּת֛וֹךְ בְּנֵ֥י אָדָ֖ם אֶל־ י֥וֹרְדֵי בֽוֹר׃ (מב) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה בְּי֨וֹם רִדְתּ֤וֹ שְׁא֙וֹלָה֙ הֶאֱבַ֜לְתִּי כִּסֵּ֤תִי עָלָיו֙ אֶת־ תְּה֔וֹם וָֽאֶמְנַע֙ נַהֲרוֹתֶ֔יהָ וַיִּכָּלְא֖וּ מַ֣יִם רַבִּ֑ים וָאַקְדִּ֤ר עָלָיו֙ לְבָנ֔וֹן וְכָל־ עֲצֵ֥י הַשָּׂדֶ֖ה עָלָ֥יו עֻלְפֶּֽה׃ (מג) מִקּ֤וֹל מַפַּלְתּוֹ֙ הִרְעַ֣שְׁתִּי גוֹיִ֔ם בְּהוֹרִדִ֥י אֹת֛וֹ שְׁא֖וֹלָה אֶת־ י֣וֹרְדֵי ב֑וֹר וַיִּנָּ֨חֲמ֜וּ בְּאֶ֤רֶץ תַּחְתִּית֙ כָּל־ עֲצֵי־ עֵ֔דֶן מִבְחַ֥ר וְטוֹב־ לְבָנ֖וֹן כָּל־ שֹׁ֥תֵי מָֽיִם׃ (מד) גַּם־ הֵ֗ם אִתּ֛וֹ יָרְד֥וּ שְׁא֖וֹלָה אֶל־ חַלְלֵי־ חָ֑רֶב וּזְרֹע֛וֹ יָשְׁב֥וּ בְצִלּ֖וֹ בְּת֥וֹךְ גּוֹיִֽם׃ (מה) אֶל־ מִ֨י דָמִ֥יתָ כָּ֛כָה בְּכָב֥וֹד וּבְגֹ֖דֶל בַּעֲצֵי־ עֵ֑דֶן וְהוּרַדְתָּ֨ אֶת־ עֲצֵי־ עֵ֜דֶן אֶל־ אֶ֣רֶץ תַּחְתִּ֗ית בְּת֨וֹךְ עֲרֵלִ֤ים תִּשְׁכַּב֙ אֶת־ חַלְלֵי־ חֶ֔רֶב ה֤וּא פַרְעֹה֙ וְכָל־ הֲמוֹנֹ֔ה נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃
