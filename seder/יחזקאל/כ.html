יחזקאל סדר כ

פתיחה באתר · ספר יחזקאל כולו ·

(א) כָּל־ (חטאתו) חַטֹּאתָיו֙ אֲשֶׁ֣ר חָטָ֔א לֹ֥א תִזָּכַ֖רְנָה ל֑וֹ מִשְׁפָּ֧ט וּצְדָקָ֛ה עָשָׂ֖ה חָי֥וֹ יִֽחְיֶֽה׃ (ב) וְאָמְרוּ֙ בְּנֵ֣י עַמְּךָ֔ לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י וְהֵ֖מָּה דַּרְכָּ֥ם לֹֽא־ יִתָּכֵֽן׃ (ג) בְּשׁוּב־ צַדִּ֥יק מִצִּדְקָת֖וֹ וְעָ֣שָׂה עָ֑וֶל וּמֵ֖ת בָּהֶֽם׃ (ד) וּבְשׁ֤וּב רָשָׁע֙ מֵֽרִשְׁעָת֔וֹ וְעָשָׂ֥ה מִשְׁפָּ֖ט וּצְדָקָ֑ה עֲלֵיהֶ֖ם ה֥וּא יִֽחְיֶֽה׃ (ה) וַאֲמַרְתֶּ֕ם לֹ֥א יִתָּכֵ֖ן דֶּ֣רֶךְ אֲדֹנָ֑י אִ֧ישׁ כִּדְרָכָ֛יו אֶשְׁפּ֥וֹט אֶתְכֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל (ו) וַיְהִ֞י בִּשְׁתֵּ֧י עֶשְׂרֵ֣ה שָׁנָ֗ה בָּעֲשִׂרִ֛י בַּחֲמִשָּׁ֥ה לַחֹ֖דֶשׁ לְגָלוּתֵ֑נוּ בָּא־ אֵלַ֨י הַפָּלִ֧יט מִירוּשָׁלִַ֛ם לֵאמֹ֖ר הֻכְּתָ֥ה הָעִֽיר׃ (ז) וְיַד־ יְהוָה֩ הָיְתָ֨ה אֵלַ֜י בָּעֶ֗רֶב לִפְנֵי֙ בּ֣וֹא הַפָּלִ֔יט וַיִּפְתַּ֣ח אֶת־ פִּ֔י עַד־ בּ֥וֹא אֵלַ֖י בַּבֹּ֑קֶר וַיִּפָּ֣תַח פִּ֔י וְלֹ֥א נֶאֱלַ֖מְתִּי עֽוֹד׃ (ח) וַיְהִ֥י דְבַר־ יְהֹוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (ט) בֶּן־ אָדָ֗ם יֹ֠שְׁבֵי הֶחֳרָב֨וֹת הָאֵ֜לֶּה עַל־ אַדְמַ֤ת יִשְׂרָאֵל֙ אֹמְרִ֣ים לֵאמֹ֔ר אֶחָד֙ הָיָ֣ה אַבְרָהָ֔ם וַיִּירַ֖שׁ אֶת־ הָאָ֑רֶץ וַאֲנַ֣חְנוּ רַבִּ֔ים לָ֛נוּ נִתְּנָ֥ה הָאָ֖רֶץ לְמוֹרָשָֽׁה׃ (י) לָכֵן֩ אֱמֹ֨ר אֲלֵיהֶ֜ם כֹּֽה־ אָמַ֣ר ׀ אֲדֹנָ֣י יְהֹוִ֗ה עַל־ הַדָּ֧ם ׀ תֹּאכֵ֛לוּ וְעֵינֵכֶ֛ם תִּשְׂא֥וּ אֶל־ גִּלּוּלֵיכֶ֖ם וְדָ֣ם תִּשְׁפֹּ֑כוּ וְהָאָ֖רֶץ תִּירָֽשׁוּ׃ (יא) עֲמַדְתֶּ֤ם עַֽל־ חַרְבְּכֶם֙ עֲשִׂיתֶ֣ן תּוֹעֵבָ֔ה וְאִ֛ישׁ אֶת־ אֵ֥שֶׁת רֵעֵ֖הוּ טִמֵּאתֶ֑ם וְהָאָ֖רֶץ תִּירָֽשׁוּ׃ (יב) כֹּֽה־ תֹאמַ֨ר אֲלֵהֶ֜ם כֹּה־ אָמַ֨ר אֲדֹנָ֣י יְהוִה֮ חַי־ אָנִי֒ אִם־ לֹ֞א אֲשֶׁ֤ר בֶּֽחֳרָבוֹת֙ בַּחֶ֣רֶב יִפֹּ֔לוּ וַֽאֲשֶׁר֙ עַל־ פְּנֵ֣י הַשָּׂדֶ֔ה לַחַיָּ֥ה נְתַתִּ֖יו לְאָכְל֑וֹ וַאֲשֶׁ֛ר בַּמְּצָד֥וֹת וּבַמְּעָר֖וֹת בַּדֶּ֥בֶר יָמֽוּתוּ׃ (יג) וְנָתַתִּ֤י אֶת־ הָאָ֙רֶץ֙ שְׁמָמָ֣ה וּמְשַׁמָּ֔ה וְנִשְׁבַּ֖ת גְּא֣וֹן עֻזָּ֑הּ וְשָֽׁמְמ֛וּ הָרֵ֥י יִשְׂרָאֵ֖ל מֵאֵ֥ין עוֹבֵֽר׃ (יד) וְיָדְע֖וּ כִּֽי־ אֲנִ֣י יְהוָ֑ה בְּתִתִּ֤י אֶת־ הָאָ֙רֶץ֙ שְׁמָמָ֣ה וּמְשַׁמָּ֔ה עַ֥ל כָּל־ תּוֹעֲבֹתָ֖ם אֲשֶׁ֥ר עָשֽׂוּ׃ (טו) וְאַתָּ֣ה בֶן־ אָדָ֔ם בְּנֵ֣י עַמְּךָ֗ הַנִּדְבָּרִ֤ים בְּךָ֙ אֵ֣צֶל הַקִּיר֔וֹת וּבְפִתְחֵ֖י הַבָּתִּ֑ים וְדִבֶּר־ חַ֣ד אֶת־ אַחַ֗ד אִ֤ישׁ אֶת־ אָחִיו֙ לֵאמֹ֔ר בֹּֽאוּ־ נָ֣א וְשִׁמְע֔וּ מָ֣ה הַדָּבָ֔ר הַיּוֹצֵ֖א מֵאֵ֥ת יְהוָֽה׃ (טז) וְיָב֣וֹאוּ אֵ֠לֶיךָ כִּמְבוֹא־ עָ֞ם וְיֵשְׁב֤וּ לְפָנֶ֙יךָ֙ עַמִּ֔י וְשָֽׁמְעוּ֙ אֶת־ דְּבָרֶ֔יךָ וְאוֹתָ֖ם לֹ֣א יַֽעֲשׂ֑וּ כִּֽי־ עֲגָבִ֤ים בְּפִיהֶם֙ הֵ֣מָּה עֹשִׂ֔ים אַחֲרֵ֥י בִצְעָ֖ם לִבָּ֥ם הֹלֵֽךְ׃ (יז) וְהִנְּךָ֤ לָהֶם֙ כְּשִׁ֣יר עֲגָבִ֔ים יְפֵ֥ה ק֖וֹל וּמֵטִ֣ב נַגֵּ֑ן וְשָֽׁמְעוּ֙ אֶת־ דְּבָרֶ֔יךָ וְעֹשִׂ֥ים אֵינָ֖ם אוֹתָֽם׃ (יח) וּבְבֹאָ֑הּ הִנֵּ֣ה בָאָ֔ה וְיָ֣דְע֔וּ כִּ֥י נָבִ֖יא הָיָ֥ה בְתוֹכָֽם׃ (יט) וַיְהִ֥י דְבַר־ יְהוָ֖ה אֵלַ֥י לֵאמֹֽר׃ (כ) בֶּן־ אָדָ֕ם הִנָּבֵ֖א עַל־ רוֹעֵ֣י יִשְׂרָאֵ֑ל הִנָּבֵ֣א וְאָמַרְתָּ֩ אֲלֵיהֶ֨ם לָרֹעִ֜ים כֹּ֥ה אָמַ֣ר ׀ אֲדֹנָ֣י יְהוִ֗ה ה֤וֹי רֹעֵֽי־ יִשְׂרָאֵל֙ אֲשֶׁ֤ר הָיוּ֙ רֹעִ֣ים אוֹתָ֔ם הֲל֣וֹא הַצֹּ֔אן יִרְע֖וּ הָרֹעִֽים׃ (כא) אֶת־ הַחֵ֤לֶב תֹּאכֵ֙לוּ֙ וְאֶת־ הַצֶּ֣מֶר תִּלְבָּ֔שׁוּ הַבְּרִיאָ֖ה תִּזְבָּ֑חוּ הַצֹּ֖אן לֹ֥א תִרְעֽוּ׃ (כב) אֶֽת־ הַנַּחְלוֹת֩ לֹ֨א חִזַּקְתֶּ֜ם וְאֶת־ הַחוֹלָ֣ה לֹֽא־ רִפֵּאתֶ֗ם וְלַנִּשְׁבֶּ֙רֶת֙ לֹ֣א חֲבַשְׁתֶּ֔ם וְאֶת־ הַנִּדַּ֙חַת֙ לֹ֣א הֲשֵׁבֹתֶ֔ם וְאֶת־ הָאֹבֶ֖דֶת לֹ֣א בִקַּשְׁתֶּ֑ם וּבְחָזְקָ֛ה רְדִיתֶ֥ם אֹתָ֖ם וּבְפָֽרֶךְ׃ (כג) וַתְּפוּצֶ֖ינָה מִבְּלִ֣י רֹעֶ֑ה וַתִּהְיֶ֧ינָה לְאָכְלָ֛ה לְכָל־ חַיַּ֥ת הַשָּׂדֶ֖ה וַתְּפוּצֶֽינָה׃ (כד) יִשְׁגּ֤וּ צֹאנִי֙ בְּכָל־ הֶ֣הָרִ֔ים וְעַ֖ל כָּל־ גִּבְעָ֣ה רָמָ֑ה וְעַ֨ל כָּל־ פְּנֵ֤י הָאָ֙רֶץ֙ נָפֹ֣צוּ צֹאנִ֔י וְאֵ֥ין דּוֹרֵ֖שׁ וְאֵ֥ין מְבַקֵּֽשׁ׃ (כה) לָכֵ֣ן רֹעִ֔ים שִׁמְע֖וּ אֶת־ דְּבַ֥ר יְהוָֽה׃ (כו) חַי־ אָ֜נִי נְאֻ֣ם ׀ אֲדֹנָ֣י יְהוִ֗ה אִם־ לֹ֣א יַ֣עַן הֱיֽוֹת־ צֹאנִ֣י ׀ לָבַ֡ז וַתִּֽהְיֶינָה֩ צֹאנִ֨י לְאָכְלָ֜ה לְכָל־ חַיַּ֤ת הַשָּׂדֶה֙ מֵאֵ֣ין רֹעֶ֔ה וְלֹֽא־ דָרְשׁ֥וּ רֹעַ֖י אֶת־ צֹאנִ֑י וַיִּרְע֤וּ הָֽרֹעִים֙ אוֹתָ֔ם וְאֶת־ צֹאנִ֖י לֹ֥א רָעֽוּ׃ (כז) לָכֵן֙ הָֽרֹעִ֔ים שִׁמְע֖וּ דְּבַר־ יְהוָֽה׃ (כח) כֹּה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה הִנְנִ֨י אֶֽל־ הָרֹעִ֜ים וְֽדָרַשְׁתִּ֧י אֶת־ צֹאנִ֣י מִיָּדָ֗ם וְהִשְׁבַּתִּים֙ מֵרְע֣וֹת צֹ֔אן וְלֹא־ יִרְע֥וּ ע֛וֹד הָרֹעִ֖ים אוֹתָ֑ם וְהִצַּלְתִּ֤י צֹאנִי֙ מִפִּיהֶ֔ם וְלֹֽא־ תִהְיֶ֥יןָ לָהֶ֖ם לְאָכְלָֽה׃ (כט) כִּ֛י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הִנְנִי־ אָ֕נִי וְדָרַשְׁתִּ֥י אֶת־ צֹאנִ֖י וּבִקַּרְתִּֽים׃ (ל) כְּבַקָּרַת֩ רֹעֶ֨ה עֶדְר֜וֹ בְּיוֹם־ הֱיוֹת֤וֹ בְתוֹךְ־ צֹאנוֹ֙ נִפְרָשׁ֔וֹת כֵּ֖ן אֲבַקֵּ֣ר אֶת־ צֹאנִ֑י וְהִצַּלְתִּ֣י אֶתְהֶ֗ם מִכָּל־ הַמְּקוֹמֹת֙ אֲשֶׁ֣ר נָפֹ֣צוּ שָׁ֔ם בְּי֥וֹם עָנָ֖ן וַעֲרָפֶֽל׃ (לא) וְהוֹצֵאתִ֣ים מִן־ הָעַמִּ֗ים וְקִבַּצְתִּים֙ מִן־ הָ֣אֲרָצ֔וֹת וַהֲבִיאֹתִ֖ים אֶל־ אַדְמָתָ֑ם וּרְעִיתִים֙ אֶל־ הָרֵ֣י יִשְׂרָאֵ֔ל בָּאֲפִיקִ֕ים וּבְכֹ֖ל מוֹשְׁבֵ֥י הָאָֽרֶץ׃ (לב) בְּמִרְעֶה־ טּוֹב֙ אֶרְעֶ֣ה אֹתָ֔ם וּבְהָרֵ֥י מְרֽוֹם־ יִשְׂרָאֵ֖ל יִהְיֶ֣ה נְוֵהֶ֑ם שָׁ֤ם תִּרְבַּ֙צְנָה֙ בְּנָ֣וֶה טּ֔וֹב וּמִרְעֶ֥ה שָׁמֵ֛ן תִּרְעֶ֖ינָה אֶל־ הָרֵ֥י יִשְׂרָאֵֽל׃ (לג) אֲנִ֨י אֶרְעֶ֤ה צֹאנִי֙ וַאֲנִ֣י אַרְבִּיצֵ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (לד) אֶת־ הָאֹבֶ֤דֶת אֲבַקֵּשׁ֙ וְאֶת־ הַנִּדַּ֣חַת אָשִׁ֔יב וְלַנִּשְׁבֶּ֣רֶת אֶחֱבֹ֔שׁ וְאֶת־ הַחוֹלָ֖ה אֲחַזֵּ֑ק וְאֶת־ הַשְּׁמֵנָ֧ה וְאֶת־ הַחֲזָקָ֛ה אַשְׁמִ֖יד אֶרְעֶ֥נָּה בְמִשְׁפָּֽט׃ (לה) וְאַתֵּ֣נָה צֹאנִ֔י כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה הִנְנִ֤י שֹׁפֵט֙ בֵּֽין־ שֶׂ֣ה לָשֶׂ֔ה לָאֵילִ֖ים וְלָעַתּוּדִֽים׃ (לו) הַמְעַ֣ט מִכֶּ֗ם הַמִּרְעֶ֤ה הַטּוֹב֙ תִּרְע֔וּ וְיֶ֙תֶר֙ מִרְעֵיכֶ֔ם תִּרְמְס֖וּ בְּרַגְלֵיכֶ֑ם וּמִשְׁקַע־ מַ֣יִם תִּשְׁתּ֔וּ וְאֵת֙ הַנּ֣וֹתָרִ֔ים בְּרַגְלֵיכֶ֖ם תִּרְפֹּשֽׂוּן׃ (לז) וְצֹאנִ֑י מִרְמַ֤ס רַגְלֵיכֶם֙ תִּרְעֶ֔ינָה וּמִרְפַּ֥שׂ רַגְלֵיכֶ֖ם תִּשְׁתֶּֽינָה׃ (לח) לָכֵ֗ן כֹּ֥ה אָמַ֛ר אֲדֹנָ֥י יְהוִ֖ה אֲלֵיהֶ֑ם הִנְנִי־ אָ֕נִי וְשָֽׁפַטְתִּי֙ בֵּֽין־ שֶׂ֣ה בִרְיָ֔ה וּבֵ֥ין שֶׂ֖ה רָזָֽה׃ (לט) יַ֗עַן בְּצַ֤ד וּבְכָתֵף֙ תֶּהְדֹּ֔פוּ וּבְקַרְנֵיכֶ֥ם תְּנַגְּח֖וּ כָּל־ הַנַּחְל֑וֹת עַ֣ד אֲשֶׁ֧ר הֲפִיצוֹתֶ֛ם אוֹתָ֖נָה אֶל־ הַחֽוּצָה׃ (מ) וְהוֹשַׁעְתִּ֣י לְצֹאנִ֔י וְלֹֽא־ תִהְיֶ֥ינָה ע֖וֹד לָבַ֑ז וְשָׁ֣פַטְתִּ֔י בֵּ֥ין שֶׂ֖ה לָשֶֽׂה׃ (מא) וַהֲקִמֹתִ֨י עֲלֵיהֶ֜ם רֹעֶ֤ה אֶחָד֙ וְרָעָ֣ה אֶתְהֶ֔ן אֵ֖ת עַבְדִּ֣י דָוִ֑יד ה֚וּא יִרְעֶ֣ה אֹתָ֔ם וְהֽוּא־ יִהְיֶ֥ה לָהֶ֖ן לְרֹעֶֽה׃ (מב) וַאֲנִ֣י יְהוָ֗ה אֶהְיֶ֤ה לָהֶם֙ לֵֽאלֹהִ֔ים וְעַבְדִּ֥י דָוִ֖ד נָשִׂ֣יא בְתוֹכָ֑ם אֲנִ֥י יְהוָ֖ה דִּבַּֽרְתִּי׃ (מג) וְכָרַתִּ֤י לָהֶם֙ בְּרִ֣ית שָׁל֔וֹם וְהִשְׁבַּתִּ֥י חַיָּֽה־ רָעָ֖ה מִן־ הָאָ֑רֶץ וְיָשְׁב֤וּ בַמִּדְבָּר֙ לָבֶ֔טַח וְיָשְׁנ֖וּ בַּיְּעָרִֽים׃
