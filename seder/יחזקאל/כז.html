יחזקאל סדר כז

פתיחה באתר · ספר יחזקאל כולו ·

(א) וִֽיכַלּ֖וּ אֶת־ הַיָּמִ֑ים וְהָיָה֩ בַיּ֨וֹם הַשְּׁמִינִ֜י וָהָ֗לְאָה יַעֲשׂ֨וּ הַכֹּהֲנִ֤ים עַל־ הַמִּזְבֵּ֙חַ֙ אֶת־ עוֹלֽוֹתֵיכֶם֙ וְאֶת־ שַׁלְמֵיכֶ֔ם וְרָצִ֣אתִי אֶתְכֶ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהֹוִֽה׃ (ב) וַיָּ֣שֶׁב אֹתִ֗י דֶּ֣רֶךְ שַׁ֤עַר הַמִּקְדָּשׁ֙ הַֽחִיצ֔וֹן הַפֹּנֶ֖ה קָדִ֑ים וְה֖וּא סָגֽוּר׃ (ג) וַיֹּ֨אמֶר אֵלַ֜י יְהוָ֗ה הַשַּׁ֣עַר הַזֶּה֩ סָג֨וּר יִהְיֶ֜ה לֹ֣א יִפָּתֵ֗חַ וְאִישׁ֙ לֹא־ יָ֣בֹא ב֔וֹ כִּ֛י יְהוָ֥ה אֱלֹהֵֽי־ יִשְׂרָאֵ֖ל בָּ֣א ב֑וֹ וְהָיָ֖ה סָגֽוּר׃ (ד) אֶֽת־ הַנָּשִׂ֗יא נָ֥שִׂיא ה֛וּא יֵֽשֶׁב־ בּ֥וֹ לֶאֱכָול־ לֶ֖חֶם לִפְנֵ֣י יְהוָ֑ה מִדֶּ֨רֶךְ אֻלָ֤ם הַשַּׁ֙עַר֙ יָב֔וֹא וּמִדַּרְכּ֖וֹ יֵצֵֽא׃ (ה) וַיְבִיאֵ֜נִי דֶּֽרֶךְ־ שַׁ֣עַר הַצָּפוֹן֮ אֶל־ פְּנֵ֣י הַבַּיִת֒ וָאֵ֕רֶא וְהִנֵּ֛ה מָלֵ֥א כְבוֹד־ יְהוָ֖ה אֶת־ בֵּ֣ית יְהוָ֑ה וָאֶפֹּ֖ל אֶל־ פָּנָֽי׃ (ו) וַיֹּ֨אמֶר אֵלַ֜י יְהֹוָ֗ה בֶּן־ אָדָ֡ם שִׂ֣ים לִבְּךָ֩ וּרְאֵ֨ה בְעֵינֶ֜יךָ וּבְאָזְנֶ֣יךָ שְּׁמָ֗ע אֵ֣ת כָּל־ אֲשֶׁ֤ר אֲנִי֙ מְדַבֵּ֣ר אֹתָ֔ךְ לְכָל־ חֻקּ֥וֹת בֵּית־ יְהוָ֖ה וּלְכָל־ (תורתו) תּֽוֹרֹתָ֑יו וְשַׂמְתָּ֤ לִבְּךָ֙ לִמְב֣וֹא הַבַּ֔יִת בְּכֹ֖ל מוֹצָאֵ֥י הַמִּקְדָּֽשׁ׃ (ז) וְאָמַרְתָּ֤ אֶל־ מֶ֙רִי֙ אֶל־ בֵּ֣ית יִשְׂרָאֵ֔ל כֹּ֥ה אָמַ֖ר אֲדֹנָ֣י יְהוִ֑ה רַב־ לָכֶ֛ם מִֽכָּל־ תּוֹעֲבֽוֹתֵיכֶ֖ם בֵּ֥ית יִשְׂרָאֵֽל׃ (ח) בַּהֲבִיאֲכֶ֣ם בְּנֵֽי־ נֵכָ֗ר עַרְלֵי־ לֵב֙ וְעַרְלֵ֣י בָשָׂ֔ר לִהְי֥וֹת בְּמִקְדָּשִׁ֖י לְחַלְּל֣וֹ אֶת־ בֵּיתִ֑י בְּהַקְרִֽיבְכֶ֤ם אֶת־ לַחְמִי֙ חֵ֣לֶב וָדָ֔ם וַיָּפֵ֙רוּ֙ אֶת־ בְּרִיתִ֔י אֶ֖ל כָּל־ תּוֹעֲבוֹתֵיכֶֽם׃ (ט) וְלֹ֥א שְׁמַרְתֶּ֖ם מִשְׁמֶ֣רֶת קָדָשָׁ֑י וַתְּשִׂימ֗וּן לְשֹׁמְרֵ֧י מִשְׁמַרְתִּ֛י בְּמִקְדָּשִׁ֖י לָכֶֽם׃ (י) כֹּה־ אָמַר֮ אֲדֹנָ֣י יְהוִה֒ כָּל־ בֶּן־ נֵכָ֗ר עֶ֤רֶל לֵב֙ וְעֶ֣רֶל בָּשָׂ֔ר לֹ֥א יָב֖וֹא אֶל־ מִקְדָּשִׁ֑י לְכָל־ בֶּן־ נֵכָ֔ר אֲשֶׁ֕ר בְּת֖וֹךְ בְּנֵ֥י יִשְׂרָאֵֽל׃ (יא) כִּ֣י אִם־ הַלְוִיִּ֗ם אֲשֶׁ֤ר רָֽחֲקוּ֙ מֵֽעָלַ֔י בִּתְע֤וֹת יִשְׂרָאֵל֙ אֲשֶׁ֣ר תָּע֣וּ מֵֽעָלַ֔י אַחֲרֵ֖י גִּלּֽוּלֵיהֶ֑ם וְנָשְׂא֖וּ עֲוֺנָֽם׃ (יב) וְהָי֤וּ בְמִקְדָּשִׁי֙ מְשָׁ֣רְתִ֔ים פְּקֻדּוֹת֙ אֶל־ שַׁעֲרֵ֣י הַבַּ֔יִת וּֽמְשָׁרְתִ֖ים אֶת־ הַבָּ֑יִת הֵ֠מָּה יִשְׁחֲט֨וּ אֶת־ הָעֹלָ֤ה וְאֶת־ הַזֶּ֙בַח֙ לָעָ֔ם וְהֵ֛מָּה יַעַמְד֥וּ לִפְנֵיהֶ֖ם לְשָֽׁרְתָֽם׃ (יג) יַ֗עַן אֲשֶׁ֨ר יְשָׁרְת֤וּ אוֹתָם֙ לִפְנֵ֣י גִלּֽוּלֵיהֶ֔ם וְהָי֥וּ לְבֵֽית־ יִשְׂרָאֵ֖ל לְמִכְשׁ֣וֹל עָוֺ֑ן עַל־ כֵּן֩ נָשָׂ֨אתִי יָדִ֜י עֲלֵיהֶ֗ם נְאֻם֙ אֲדֹנָ֣י יְהוִ֔ה וְנָשְׂא֖וּ עֲוֺנָֽם׃ (יד) וְלֹֽא־ יִגְּשׁ֤וּ אֵלַי֙ לְכַהֵ֣ן לִ֔י וְלָגֶ֙שֶׁת֙ עַל־ כָּל־ קָ֣דָשַׁ֔י אֶל־ קָדְשֵׁ֖י הַקְּדָשִׁ֑ים וְנָֽשְׂאוּ֙ כְּלִמָּתָ֔ם וְתוֹעֲבוֹתָ֖ם אֲשֶׁ֥ר עָשֽׂוּ׃ (טו) וְנָתַתִּ֣י אוֹתָ֔ם שֹׁמְרֵ֖י מִשְׁמֶ֣רֶת הַבָּ֑יִת לְכֹל֙ עֲבֹ֣דָת֔וֹ וּלְכֹ֛ל אֲשֶׁ֥ר יֵעָשֶׂ֖ה בּֽוֹ׃ (טז) וְהַכֹּהֲנִ֨ים הַלְוִיִּ֜ם בְּנֵ֣י צָד֗וֹק אֲשֶׁ֨ר שָׁמְר֜וּ אֶת־ מִשְׁמֶ֤רֶת מִקְדָּשִׁי֙ בִּתְע֤וֹת בְּנֵֽי־ יִשְׂרָאֵל֙ מֵֽעָלַ֔י הֵ֛מָּה יִקְרְב֥וּ אֵלַ֖י לְשָֽׁרְתֵ֑נִי וְעָמְד֣וּ לְפָנַ֗י לְהַקְרִ֥יב לִי֙ חֵ֣לֶב וָדָ֔ם נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (יז) הֵ֜מָּה יָבֹ֣אוּ אֶל־ מִקְדָּשִׁ֗י וְהֵ֛מָּה יִקְרְב֥וּ אֶל־ שֻׁלְחָנִ֖י לְשָׁרְתֵ֑נִי וְשָׁמְר֖וּ אֶת־ מִשְׁמַרְתִּֽי׃ (יח) וְהָיָ֗ה בְּבוֹאָם֙ אֶֽל־ שַׁעֲרֵי֙ הֶחָצֵ֣ר הַפְּנִימִ֔ית בִּגְדֵ֥י פִשְׁתִּ֖ים יִלְבָּ֑שׁוּ וְלֹֽא־ יַעֲלֶ֤ה עֲלֵיהֶם֙ צֶ֔מֶר בְּשָֽׁרְתָ֗ם בְּשַׁעֲרֵ֛י הֶחָצֵ֥ר הַפְּנִימִ֖ית וָבָֽיְתָה׃ (יט) פַּאֲרֵ֤י פִשְׁתִּים֙ יִהְי֣וּ עַל־ רֹאשָׁ֔ם וּמִכְנְסֵ֣י פִשְׁתִּ֔ים יִהְי֖וּ עַל־ מָתְנֵיהֶ֑ם לֹ֥א יַחְגְּר֖וּ בַּיָּֽזַע׃ (כ) וּ֠בְצֵאתָם אֶל־ הֶחָצֵ֨ר הַחִיצוֹנָ֜ה אֶל־ הֶחָצֵ֣ר הַחִיצוֹנָה֮ אֶל־ הָעָם֒ יִפְשְׁט֣וּ אֶת־ בִּגְדֵיהֶ֗ם אֲשֶׁר־ הֵ֙מָּה֙ מְשָׁרְתִ֣ם בָּ֔ם וְהִנִּ֥יחוּ אוֹתָ֖ם בְּלִֽשְׁכֹ֣ת הַקֹּ֑דֶשׁ וְלָֽבְשׁוּ֙ בְּגָדִ֣ים אֲחֵרִ֔ים וְלֹֽא־ יְקַדְּשׁ֥וּ אֶת־ הָעָ֖ם בְּבִגְדֵיהֶֽם׃ (כא) וְרֹאשָׁם֙ לֹ֣א יְגַלֵּ֔חוּ וּפֶ֖רַע לֹ֣א יְשַׁלֵּ֑חוּ כָּס֥וֹם יִכְסְמ֖וּ אֶת־ רָאשֵׁיהֶֽם׃ (כב) וְיַ֥יִן לֹֽא־ יִשְׁתּ֖וּ כָּל־ כֹּהֵ֑ן בְּבוֹאָ֖ם אֶל־ הֶחָצֵ֥ר הַפְּנִימִֽית׃ (כג) וְאַלְמָנָה֙ וּגְרוּשָׁ֔ה לֹֽא־ יִקְח֥וּ לָהֶ֖ם לְנָשִׁ֑ים כִּ֣י אִם־ בְּתוּלֹ֗ת מִזֶּ֙רַע֙ בֵּ֣ית יִשְׂרָאֵ֔ל וְהָֽאַלְמָנָה֙ אֲשֶׁ֣ר תִּֽהְיֶ֣ה אַלְמָנָ֔ה מִכֹּהֵ֖ן יִקָּֽחוּ׃ (כד) וְאֶת־ עַמִּ֣י יוֹר֔וּ בֵּ֥ין קֹ֖דֶשׁ לְחֹ֑ל וּבֵין־ טָמֵ֥א לְטָה֖וֹר יוֹדִעֻֽם׃ (כה) וְעַל־ רִ֗יב הֵ֚מָּה יַעַמְד֣וּ (לשפט) לְמִשְׁפָּ֔ט בְּמִשְׁפָּטַ֖י (ושפטהו) יִשְׁפְּט֑וּהוּ וְאֶת־ תּוֹרֹתַ֤י וְאֶת־ חֻקֹּתַי֙ בְּכָל־ מוֹעֲדַ֣י יִשְׁמֹ֔רוּ וְאֶת־ שַׁבְּתוֹתַ֖י יְקַדֵּֽשׁוּ׃ (כו) וְאֶל־ מֵ֣ת אָדָ֔ם לֹ֥א יָב֖וֹא לְטָמְאָ֑ה כִּ֣י אִם־ לְאָ֡ב וּ֠לְאֵם וּלְבֵ֨ן וּלְבַ֜ת לְאָ֗ח וּלְאָח֛וֹת אֲשֶֽׁר־ לֹא־ הָיְתָ֥ה לְאִ֖ישׁ יִטַּמָּֽאוּ׃ (כז) וְאַחֲרֵ֖י טָֽהֳרָת֑וֹ שִׁבְעַ֥ת יָמִ֖ים יִסְפְּרוּ־ לֽוֹ׃ (כח) וּבְיוֹם֩ בֹּא֨וֹ אֶל־ הַקֹּ֜דֶשׁ אֶל־ הֶחָצֵ֤ר הַפְּנִימִית֙ לְשָׁרֵ֣ת בַּקֹּ֔דֶשׁ יַקְרִ֖יב חַטָּאת֑וֹ נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (כט) וְהָיְתָ֤ה לָהֶם֙ לְֽנַחֲלָ֔ה אֲנִ֖י נַֽחֲלָתָ֑ם וַאֲחֻזָּ֗ה לֹֽא־ תִתְּנ֤וּ לָהֶם֙ בְּיִשְׂרָאֵ֔ל אֲנִ֖י אֲחֻזָּתָֽם׃ (ל) הַמִּנְחָה֙ וְהַחַטָּ֣את וְהָאָשָׁ֔ם הֵ֖מָּה יֹֽאכְל֑וּם וְכָל־ חֵ֥רֶם בְּיִשְׂרָאֵ֖ל לָהֶ֥ם יִהְיֶֽה׃ (לא) וְרֵאשִׁית֩ כָּל־ בִּכּ֨וּרֵי כֹ֜ל וְכָל־ תְּר֣וּמַת כֹּ֗ל מִכֹּל֙ תְּרוּמ֣וֹתֵיכֶ֔ם לַכֹּהֲנִ֖ים יִֽהְיֶ֑ה וְרֵאשִׁ֤ית עֲרִסֽוֹתֵיכֶם֙ תִּתְּנ֣וּ לַכֹּהֵ֔ן לְהָנִ֥יחַ בְּרָכָ֖ה אֶל־ בֵּיתֶֽךָ׃ (לב) כָּל־ נְבֵלָה֙ וּטְרֵפָ֔ה מִן־ הָע֖וֹף וּמִן־ הַבְּהֵמָ֑ה לֹ֥א יֹאכְל֖וּ הַכֹּהֲנִֽים׃ (לג) וּבְהַפִּֽילְכֶ֨ם אֶת־ הָאָ֜רֶץ בְּנַחֲלָ֗ה תָּרִימוּ֩ תְרוּמָ֨ה לַיהוָ֥ה ׀ קֹדֶשׁ֮ מִן־ הָאָרֶץ֒ אֹ֗רֶךְ חֲמִשָּׁ֨ה וְעֶשְׂרִ֥ים אֶ֙לֶף֙ אֹ֔רֶךְ וְרֹ֖חַב עֲשָׂ֣רָה אָ֑לֶף קֹדֶשׁ־ ה֥וּא בְכָל־ גְּבוּלָ֖הּ סָבִֽיב׃ (לד) יִהְיֶ֤ה מִזֶּה֙ אֶל־ הַקֹּ֔דֶשׁ חֲמֵ֥שׁ מֵא֛וֹת בַּחֲמֵ֥שׁ מֵא֖וֹת מְרֻבָּ֣ע סָבִ֑יב וַחֲמִשִּׁ֣ים אַמָּ֔ה מִגְרָ֥שׁ ל֖וֹ סָבִֽיב׃ (לה) וּמִן־ הַמִּדָּ֤ה הַזֹּאת֙ תָּמ֔וֹד אֹ֗רֶךְ (חמש) חֲמִשָּׁ֤ה וְעֶשְׂרִים֙ אֶ֔לֶף וְרֹ֖חַב עֲשֶׂ֣רֶת אֲלָפִ֑ים וּבֽוֹ־ יִהְיֶ֥ה הַמִּקְדָּ֖שׁ קֹ֥דֶשׁ קָדָשִֽׁים׃ (לו) קֹ֣דֶשׁ מִן־ הָאָ֜רֶץ ה֗וּא לַכֹּ֨הֲנִ֜ים מְשָׁרְתֵ֤י הַמִּקְדָּשׁ֙ יִֽהְיֶ֔ה הַקְּרֵבִ֖ים לְשָׁרֵ֣ת אֶת־ יְהוָ֑ה וְהָיָ֨ה לָהֶ֤ם מָקוֹם֙ לְבָ֣תִּ֔ים וּמִקְדָּ֖שׁ לַמִּקְדָּֽשׁ׃ (לז) וַחֲמִשָּׁ֨ה וְעֶשְׂרִ֥ים אֶ֙לֶף֙ אֹ֔רֶךְ וַעֲשֶׂ֥רֶת אֲלָפִ֖ים רֹ֑חַב (יהיה) וְֽהָיָ֡ה לַלְוִיִּם֩ מְשָׁרְתֵ֨י הַבַּ֧יִת לָהֶ֛ם לַאֲחֻזָּ֖ה עֶשְׂרִ֥ים לְשָׁכֹֽת׃ (לח) וַאֲחֻזַּ֨ת הָעִ֜יר תִּתְּנ֗וּ חֲמֵ֤שֶׁת אֲלָפִים֙ רֹ֔חַב וְאֹ֗רֶךְ חֲמִשָּׁ֤ה וְעֶשְׂרִים֙ אֶ֔לֶף לְעֻמַּ֖ת תְּרוּמַ֣ת הַקֹּ֑דֶשׁ לְכָל־ בֵּ֥ית יִשְׂרָאֵ֖ל יִהְיֶֽה׃ (לט) וְלַנָּשִׂ֡יא מִזֶּ֣ה וּמִזֶּה֩ לִתְרוּמַ֨ת הַקֹּ֜דֶשׁ וְלַאֲחֻזַּ֣ת הָעִ֗יר אֶל־ פְּנֵ֤י תְרֽוּמַת־ הַקֹּ֙דֶשׁ֙ וְאֶל־ פְּנֵי֙ אֲחֻזַּ֣ת הָעִ֔יר מִפְּאַת־ יָ֣ם יָ֔מָּה וּמִפְּאַת־ קֵ֖דְמָה קָדִ֑ימָה וְאֹ֗רֶךְ לְעֻמּוֹת֙ אַחַ֣ד הַחֲלָקִ֔ים מִגְּב֥וּל יָ֖ם אֶל־ גְּב֥וּל קָדִֽימָה׃ (מ) לָאָ֛רֶץ יִֽהְיֶה־ לּ֥וֹ לַֽאֲחֻזָּ֖ה בְּיִשְׂרָאֵ֑ל וְלֹא־ יוֹנ֨וּ ע֤וֹד נְשִׂיאַי֙ אֶת־ עַמִּ֔י וְהָאָ֛רֶץ יִתְּנ֥וּ לְבֵֽית־ יִשְׂרָאֵ֖ל לְשִׁבְטֵיהֶֽם׃ (מא) כֹּֽה־ אָמַ֞ר אֲדֹנָ֣י יְהוִ֗ה רַב־ לָכֶם֙ נְשִׂיאֵ֣י יִשְׂרָאֵ֔ל חָמָ֤ס וָשֹׁד֙ הָסִ֔ירוּ וּמִשְׁפָּ֥ט וּצְדָקָ֖ה עֲשׂ֑וּ הָרִ֤ימוּ גְרֻשֹֽׁתֵיכֶם֙ מֵעַ֣ל עַמִּ֔י נְאֻ֖ם אֲדֹנָ֥י יְהוִֽה׃ (מב) מֹֽאזְנֵי־ צֶ֧דֶק וְאֵֽיפַת־ צֶ֛דֶק וּבַת־ צֶ֖דֶק יְהִ֥י לָכֶֽם׃ (מג) הָאֵיפָ֣ה וְהַבַּ֗ת תֹּ֤כֶן אֶחָד֙ יִֽהְיֶ֔ה לָשֵׂ֕את מַעְשַׂ֥ר הַחֹ֖מֶר הַבָּ֑ת וַעֲשִׂירִ֤ת הַחֹ֙מֶר֙ הָֽאֵיפָ֔ה אֶל־ הַחֹ֖מֶר יִהְיֶ֥ה מַתְכֻּנְתּֽוֹ׃ (מד) וְהַשֶּׁ֖קֶל עֶשְׂרִ֣ים גֵּרָ֑ה עֶשְׂרִ֨ים שְׁקָלִ֜ים חֲמִשָּׁ֧ה וְעֶשְׂרִ֣ים שְׁקָלִ֗ים עֲשָׂרָ֤ה וַחֲמִשָּׁה֙ שֶׁ֔קֶל הַמָּנֶ֖ה יִֽהְיֶ֥ה לָכֶֽם׃ (מה) זֹ֥את הַתְּרוּמָ֖ה אֲשֶׁ֣ר תָּרִ֑ימוּ שִׁשִּׁ֤ית הָֽאֵיפָה֙ מֵחֹ֣מֶר הַֽחִטִּ֔ים וְשִׁשִּׁיתֶם֙ הָֽאֵיפָ֔ה מֵחֹ֖מֶר הַשְּׂעֹרִֽים׃ (מו) וְחֹ֨ק הַשֶּׁ֜מֶן הַבַּ֣ת הַשֶּׁ֗מֶן מַעְשַׂ֤ר הַבַּת֙ מִן־ הַכֹּ֔ר עֲשֶׂ֥רֶת הַבַּתִּ֖ים חֹ֑מֶר כִּֽי־ עֲשֶׂ֥רֶת הַבַּתִּ֖ים חֹֽמֶר׃
