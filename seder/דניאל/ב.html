דניאל סדר ב

פתיחה באתר · ספר דניאל כולו ·

(א) בֵּאדַ֣יִן דָּ֣קוּ כַחֲדָ֡ה פַּרְזְלָא֩ חַסְפָּ֨א נְחָשָׁ֜א כַּסְפָּ֣א וְדַהֲבָ֗א וַהֲווֹ֙ כְּע֣וּר מִן־ אִדְּרֵי־ קַ֔יִט וּנְשָׂ֤א הִמּוֹן֙ רוּחָ֔א וְכָל־ אֲתַ֖ר לָא־ הִשְׁתֲּכַ֣ח לְה֑וֹן וְאַבְנָ֣א ׀ דִּֽי־ מְחָ֣ת לְצַלְמָ֗א הֲוָ֛ת לְט֥וּר רַ֖ב וּמְלָ֥ת כָּל־ אַרְעָֽא׃ (ב) דְּנָ֣ה חֶלְמָ֔א וּפִשְׁרֵ֖הּ נֵאמַ֥ר קֳדָם־ מַלְכָּֽא׃ (ג) אַ֣נְתְּה מַלְכָּ֔א מֶ֖לֶךְ מַלְכַיָּ֑א דִּ֚י אֱלָ֣הּ שְׁמַיָּ֔א מַלְכוּתָ֥א חִסְנָ֛א וְתָקְפָּ֥א וִֽיקָרָ֖א יְהַב־ לָֽךְ׃ (ד) וּבְכָל־ דִּ֣י (דארין) דָֽיְרִ֣ין בְּֽנֵי־ אֲ֠נָשָׁא חֵיוַ֨ת בָּרָ֤א וְעוֹף־ שְׁמַיָּא֙ יְהַ֣ב בִּידָ֔ךְ וְהַשְׁלְטָ֖ךְ בְּכָלְּה֑וֹן אַנְתְּה־ ה֔וּא רֵאשָׁ֖ה דִּ֥י דַהֲבָֽא׃ (ה) וּבָתְרָ֗ךְ תְּק֛וּם מַלְכ֥וּ אָחֳרִ֖י אֲרַ֣עא מִנָּ֑ךְ וּמַלְכ֨וּ (תליתיא) תְלִיתָאָ֤ה אָחֳרִי֙ דִּ֣י נְחָשָׁ֔א דִּ֥י תִשְׁלַ֖ט בְּכָל־ אַרְעָֽא׃ (ו) וּמַלְכוּ֙ (רביעיה) רְבִ֣יעָאָ֔ה תֶּהֱוֵ֥א תַקִּיפָ֖ה כְּפַרְזְלָ֑א כָּל־ קֳבֵ֗ל דִּ֤י פַרְזְלָא֙ מְהַדֵּ֤ק וְחָשֵׁל֙ כֹּ֔לָּא וּֽכְפַרְזְלָ֛א דִּֽי־ מְרָעַ֥ע כָּל־ אִלֵּ֖ין תַּדִּ֥ק וְתֵרֹֽעַ׃ (ז) וְדִֽי־ חֲזַ֜יְתָה רַגְלַיָּ֣א וְאֶצְבְּעָתָ֗א (מנהון) מִנְּהֵ֞ן חֲסַ֤ף דִּֽי־ פֶחָר֙ (ומנהון) וּמִנְּהֵ֣ין פַּרְזֶ֔ל מַלְכ֤וּ פְלִיגָה֙ תֶּהֱוֵ֔ה וּמִן־ נִצְבְּתָ֥א דִ֥י פַרְזְלָ֖א לֶֽהֱוֵא־ בַ֑הּ כָּל־ קֳבֵל֙ דִּ֣י חֲזַ֔יְתָה פַּ֨רְזְלָ֔א מְעָרַ֖ב בַּחֲסַ֥ף טִינָֽא׃ (ח) וְאֶצְבְּעָת֙ רַגְלַיָּ֔א (מנהון) מִנְּהֵ֥ין פַּרְזֶ֖ל (ומנהון) וּמִנְּהֵ֣ין חֲסַ֑ף מִן־ קְצָ֤ת מַלְכוּתָא֙ תֶּהֱוֵ֣ה תַקִּיפָ֔ה וּמִנַּ֖הּ תֶּהֱוֵ֥ה תְבִירָֽה׃ (ט) (די) וְדִ֣י חֲזַ֗יְתָ פַּרְזְלָא֙ מְעָרַב֙ בַּחֲסַ֣ף טִינָ֔א מִתְעָרְבִ֤ין לֶהֱוֺן֙ בִּזְרַ֣ע אֲנָשָׁ֔א וְלָֽא־ לֶהֱוֺ֥ן דָּבְקִ֖ין דְּנָ֣ה עִם־ דְּנָ֑ה הֵֽא־ כְדִ֣י פַרְזְלָ֔א לָ֥א מִתְעָרַ֖ב עִם־ חַסְפָּֽא׃ (י) וּֽבְיוֹמֵיה֞וֹן דִּ֧י מַלְכַיָּ֣א אִנּ֗וּן יְקִים֩ אֱלָ֨הּ שְׁמַיָּ֤א מַלְכוּ֙ דִּ֤י לְעָלְמִין֙ לָ֣א תִתְחַבַּ֔ל וּמַ֨לְכוּתָ֔ה לְעַ֥ם אָחֳרָ֖ן לָ֣א תִשְׁתְּבִ֑ק תַּדִּ֤ק וְתָסֵיף֙ כָּל־ אִלֵּ֣ין מַלְכְוָתָ֔א וְהִ֖יא תְּק֥וּם לְעָלְמַיָּֽא׃ (יא) כָּל־ קֳבֵ֣ל דִּֽי־ חֲזַ֡יְתָ דִּ֣י מִטּוּרָא֩ אִתְגְּזֶ֨רֶת אֶ֜בֶן דִּי־ לָ֣א בִידַ֗יִן וְ֠הַדֶּקֶת פַּרְזְלָ֨א נְחָשָׁ֤א חַסְפָּא֙ כַּסְפָּ֣א וְדַהֲבָ֔א אֱלָ֥הּ רַב֙ הוֹדַ֣ע לְמַלְכָּ֔א מָ֛ה דִּ֥י לֶהֱוֵ֖א אַחֲרֵ֣י דְנָ֑ה וְיַצִּ֥יב חֶלְמָ֖א וּמְהֵימַ֥ן פִּשְׁרֵֽהּ׃ (יב) בֵּ֠אדַיִן מַלְכָּ֤א נְבֽוּכַדְנֶצַּר֙ נְפַ֣ל עַל־ אַנְפּ֔וֹהִי וּלְדָנִיֵּ֖אל סְגִ֑ד וּמִנְחָה֙ וְנִ֣יחֹחִ֔ין אֲמַ֖ר לְנַסָּ֥כָה לֵֽהּ׃ (יג) עָנֵה֩ מַלְכָּ֨א לְדָנִיֵּ֜אל וְאָמַ֗ר מִן־ קְשֹׁט֙ דִּ֣י אֱלָהֲכ֗וֹן ה֣וּא אֱלָ֧הּ אֱלָהִ֛ין וּמָרֵ֥א מַלְכִ֖ין וְגָלֵ֣ה רָזִ֑ין דִּ֣י יְכֵ֔לְתָּ לְמִגְלֵ֖א רָזָ֥ה דְנָֽה׃ (יד) אֱדַ֨יִן מַלְכָּ֜א לְדָנִיֵּ֣אל רַבִּ֗י וּמַתְּנָ֨ן רַבְרְבָ֤ן שַׂגִּיאָן֙ יְהַב־ לֵ֔הּ וְהַ֨שְׁלְטֵ֔הּ עַ֖ל כָּל־ מְדִינַ֣ת בָּבֶ֑ל וְרַב־ סִגְנִ֔ין עַ֖ל כָּל־ חַכִּימֵ֥י בָבֶֽל׃ (טו) וְדָנִיֵּאל֙ בְּעָ֣א מִן־ מַלְכָּ֔א וּמַנִּ֗י עַ֤ל עֲבִֽידְתָּא֙ דִּ֚י מְדִינַ֣ת בָּבֶ֔ל לְשַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ וְדָנִיֵּ֖אל בִּתְרַ֥ע מַלְכָּֽא׃ (טז) נְבוּכַדְנֶצַּ֣ר מַלְכָּ֗א עֲבַד֙ צְלֵ֣ם דִּֽי־ דְהַ֔ב רוּמֵהּ֙ אַמִּ֣ין שִׁתִּ֔ין פְּתָיֵ֖הּ אַמִּ֣ין שִׁ֑ת אֲקִימֵהּ֙ בְּבִקְעַ֣ת דּוּרָ֔א בִּמְדִינַ֖ת בָּבֶֽל׃ (יז) וּנְבוּכַדְנֶצַּ֣ר מַלְכָּ֡א שְׁלַ֡ח לְמִכְנַ֣שׁ ׀ לַֽאֲחַשְׁדַּרְפְּנַיָּ֡א סִגְנַיָּ֣א וּֽפַחֲוָתָ֡א אֲדַרְגָּזְרַיָּא֩ גְדָ֨בְרַיָּ֤א דְּתָבְרַיָּא֙ תִּפְתָּיֵ֔א וְכֹ֖ל שִׁלְטֹנֵ֣י מְדִֽינָתָ֑א לְמֵתֵא֙ לַחֲנֻכַּ֣ת צַלְמָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֥ר מַלְכָּֽא׃ (יח) בֵּאדַ֡יִן מִֽתְכַּנְּשִׁ֡ין אֲחַשְׁדַּרְפְּנַיָּ֡א סִגְנַיָּ֣א וּֽפַחֲוָתָ֡א אֲדַרְגָּזְרַיָּ֣א גְדָבְרַיָּא֩ דְּתָ֨בְרַיָּ֜א תִּפְתָּיֵ֗א וְכֹל֙ שִׁלְטֹנֵ֣י מְדִֽינָתָ֔א לַחֲנֻכַּ֣ת צַלְמָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֣ר מַלְכָּ֑א (וקאמין) וְקָֽיְמִין֙ לָקֳבֵ֣ל צַלְמָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּֽר׃ (יט) וְכָרוֹזָ֖א קָרֵ֣א בְחָ֑יִל לְכ֤וֹן אָֽמְרִין֙ עַֽמְמַיָּ֔א אֻמַּיָּ֖א וְלִשָּׁנַיָּֽא׃ (כ) בְּעִדָּנָ֡א דִּֽי־ תִשְׁמְע֡וּן קָ֣ל קַרְנָ֣א מַ֠שְׁרוֹקִיתָא (קיתרוס) קַתְר֨וֹס סַבְּכָ֤א פְּסַנְתֵּרִין֙ סוּמְפֹּ֣נְיָ֔ה וְכֹ֖ל זְנֵ֣י זְמָרָ֑א תִּפְּל֤וּן וְתִסְגְּדוּן֙ לְצֶ֣לֶם דַּהֲבָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֥ר מַלְכָּֽא׃ (כא) וּמַן־ דִּי־ לָ֥א יִפֵּ֖ל וְיִסְגֻּ֑ד בַּהּ־ שַׁעֲתָ֣א יִתְרְמֵ֔א לְגֽוֹא־ אַתּ֥וּן נוּרָ֖א יָקִֽדְתָּֽא׃ (כב) כָּל־ קֳבֵ֣ל דְּנָ֡ה בֵּהּ־ זִמְנָ֡א כְּדִ֣י שָֽׁמְעִ֣ין כָּֽל־ עַמְמַיָּ֡א קָ֣ל קַרְנָא֩ מַשְׁר֨וֹקִיתָ֜א (קיתרס) קַתְר֤וֹס שַׂבְּכָא֙ פְּסַנְטֵרִ֔ין וְכֹ֖ל זְנֵ֣י זְמָרָ֑א נָֽפְלִ֨ין כָּֽל־ עַֽמְמַיָּ֜א אֻמַיָּ֣א וְלִשָּׁנַיָּ֗א סָֽגְדִין֙ לְצֶ֣לֶם דַּהֲבָ֔א דִּ֥י הֲקֵ֖ים נְבוּכַדְנֶצַּ֥ר מַלְכָּֽא׃ (כג) כָּל־ קֳבֵ֤ל דְּנָה֙ בֵּהּ־ זִמְנָ֔א קְרִ֖בוּ גֻּבְרִ֣ין כַּשְׂדָּאִ֑ין וַאֲכַ֥לוּ קַרְצֵיה֖וֹן דִּ֥י יְהוּדָיֵֽא׃ (כד) עֲנוֹ֙ וְאָ֣מְרִ֔ין לִנְבוּכַדְנֶצַּ֖ר מַלְכָּ֑א מַלְכָּ֖א לְעָלְמִ֥ין חֱיִֽי׃ (כה) (אנתה) אַ֣נְתְּ מַלְכָּא֮ שָׂ֣מְתָּ טְּעֵם֒ דִּ֣י כָל־ אֱנָ֡שׁ דִּֽי־ יִשְׁמַ֡ע קָ֣ל קַרְנָ֣א מַ֠שְׁרֹקִיתָא (קיתרס) קַתְר֨וֹס שַׂבְּכָ֤א פְסַנְתֵּרִין֙ (וסיפניה) וְסוּפֹּ֣נְיָ֔ה וְכֹ֖ל זְנֵ֣י זְמָרָ֑א יִפֵּ֥ל וְיִסְגֻּ֖ד לְצֶ֥לֶם דַּהֲבָֽא׃ (כו) וּמַן־ דִּי־ לָ֥א יִפֵּ֖ל וְיִסְגֻּ֑ד יִתְרְמֵ֕א לְגֽוֹא־ אַתּ֥וּן נוּרָ֖א יָקִֽדְתָּֽא׃ (כז) אִיתַ֞י גֻּבְרִ֣ין יְהוּדָאיִ֗ן דִּֽי־ מַנִּ֤יתָ יָתְהוֹן֙ עַל־ עֲבִידַת֙ מְדִינַ֣ת בָּבֶ֔ל שַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ גֻּבְרַיָּ֣א אִלֵּ֗ךְ לָא־ שָׂ֨מֽוּ (עליך) עֲלָ֤ךְ מַלְכָּא֙ טְעֵ֔ם (לאלהיך) לֵֽאלָהָךְ֙ לָ֣א פָלְחִ֔ין וּלְצֶ֧לֶם דַּהֲבָ֛א דִּ֥י הֲקֵ֖ימְתָּ לָ֥א סָגְדִֽין׃ (כח) בֵּאדַ֤יִן נְבוּכַדְנֶצַּר֙ בִּרְגַ֣ז וַחֲמָ֔ה אֲמַר֙ לְהַיְתָיָ֔ה לְשַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ בֵּאדַ֙יִן֙ גֻּבְרַיָּ֣א אִלֵּ֔ךְ הֵיתָ֖יוּ קֳדָ֥ם מַלְכָּֽא׃ (כט) עָנֵ֤ה נְבֻֽכַדְנֶצַּר֙ וְאָמַ֣ר לְה֔וֹן הַצְדָּ֕א שַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ לֵֽאלָהַ֗י לָ֤א אִֽיתֵיכוֹן֙ פָּֽלְחִ֔ין וּלְצֶ֧לֶם דַּהֲבָ֛א דִּ֥י הֲקֵ֖ימֶת לָ֥א סָֽגְדִֽין׃ (ל) כְּעַ֞ן הֵ֧ן אִֽיתֵיכ֣וֹן עֲתִידִ֗ין דִּ֣י בְעִדָּנָ֡א דִּֽי־ תִשְׁמְע֡וּן קָ֣ל קַרְנָ֣א מַשְׁרוֹקִיתָ֣א (קיתרס) קַתְר֣וֹס שַׂבְּכָ֡א פְּסַנְתֵּרִין֩ וְסוּמְפֹּ֨נְיָ֜ה וְכֹ֣ל ׀ זְנֵ֣י זְמָרָ֗א תִּפְּל֣וּן וְתִסְגְּדוּן֮ לְצַלְמָ֣א דִֽי־ עַבְדֵת֒ וְהֵן֙ לָ֣א תִסְגְּד֔וּן בַּהּ־ שַׁעֲתָ֣ה תִתְרְמ֔וֹן לְגֽוֹא־ אַתּ֥וּן נוּרָ֖א יָקִֽדְתָּ֑א וּמַן־ ה֣וּא אֱלָ֔הּ דֵּ֥י יְשֵֽׁיזְבִנְכ֖וֹן מִן־ יְדָֽי׃ (לא) עֲנ֗וֹ שַׁדְרַ֤ךְ מֵישַׁךְ֙ וַעֲבֵ֣ד נְג֔וֹ וְאָמְרִ֖ין לְמַלְכָּ֑א נְבֽוּכַדְנֶצַּ֔ר לָֽא־ חַשְׁחִ֨ין אֲנַ֧חְנָה עַל־ דְּנָ֛ה פִּתְגָ֖ם לַהֲתָבוּתָֽךְ׃ (לב) הֵ֣ן אִיתַ֗י אֱלָהַ֙נָא֙ דִּֽי־ אֲנַ֣חְנָא פָֽלְחִ֔ין יָכִ֖ל לְשֵׁיזָבוּתַ֑נָא מִן־ אַתּ֨וּן נוּרָ֧א יָקִֽדְתָּ֛א וּמִן־ יְדָ֥ךְ מַלְכָּ֖א יְשֵׁיזִֽב׃ (לג) וְהֵ֣ן לָ֔א יְדִ֥יעַ לֶהֱוֵא־ לָ֖ךְ מַלְכָּ֑א דִּ֤י (לאלהיך) לֵֽאלָהָךְ֙ לָא־ (איתינא) אִיתַ֣נָא פָֽלְחִ֔ין וּלְצֶ֧לֶם דַּהֲבָ֛א דִּ֥י הֲקֵ֖ימְתָּ לָ֥א נִסְגֻּֽד׃ (לד) בֵּאדַ֨יִן נְבוּכַדְנֶצַּ֜ר הִתְמְלִ֣י חֱמָ֗א וּצְלֵ֤ם אַנְפּ֙וֹהִי֙ (אשתנו) אֶשְׁתַּנִּ֔י עַל־ שַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ עָנֵ֤ה וְאָמַר֙ לְמֵזֵ֣א לְאַתּוּנָ֔א חַ֨ד־ שִׁבְעָ֔ה עַ֛ל דִּ֥י חֲזֵ֖ה לְמֵזְיֵֽהּ׃ (לה) וּלְגֻבְרִ֤ין גִּבָּֽרֵי־ חַ֙יִל֙ דִּ֣י בְחַיְלֵ֔הּ אֲמַר֙ לְכַפָּתָ֔ה לְשַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ לְמִרְמֵ֕א לְאַתּ֥וּן נוּרָ֖א יָקִֽדְתָּֽא׃ (לו) בֵּאדַ֜יִן גֻּבְרַיָּ֣א אִלֵּ֗ךְ כְּפִ֙תוּ֙ בְּסַרְבָּלֵיהוֹן֙ (פטישיהון) פַּטְּשֵׁיה֔וֹן וְכַרְבְּלָתְה֖וֹן וּלְבֻשֵׁיה֑וֹן וּרְמִ֕יו לְגֽוֹא־ אַתּ֥וּן נוּרָ֖א יָקִֽדְתָּֽא׃ (לז) כָּל־ קֳבֵ֣ל דְּנָ֗ה מִן־ דִּ֞י מִלַּ֤ת מַלְכָּא֙ מַחְצְפָ֔ה וְאַתּוּנָ֖א אֵזֵ֣ה יַתִּ֑ירָא גֻּבְרַיָּ֣א אִלֵּ֗ךְ דִּ֤י הַסִּ֙קוּ֙ לְשַׁדְרַ֤ךְ מֵישַׁךְ֙ וַעֲבֵ֣ד נְג֔וֹ קַטִּ֣ל הִמּ֔וֹן שְׁבִיבָ֖א דִּ֥י נוּרָֽא׃ (לח) וְגֻבְרַיָּ֤א אִלֵּךְ֙ תְּלָ֣תֵּה֔וֹן שַׁדְרַ֥ךְ מֵישַׁ֖ךְ וַעֲבֵ֣ד נְג֑וֹ נְפַ֛לוּ לְגֽוֹא־ אַתּוּן־ נוּרָ֥א יָֽקִדְתָּ֖א מְכַפְּתִֽין׃ (לט) אֱדַ֙יִן֙ נְבוּכַדְנֶצַּ֣ר מַלְכָּ֔א תְּוַ֖הּ וְקָ֣ם בְּהִתְבְּהָלָ֑ה עָנֵ֨ה וְאָמַ֜ר לְהַדָּֽבְר֗וֹהִי הֲלָא֩ גֻבְרִ֨ין תְּלָתָ֜א רְמֵ֤ינָא לְגוֹא־ נוּרָא֙ מְכַפְּתִ֔ין עָנַ֤יִן וְאָמְרִין֙ לְמַלְכָּ֔א יַצִּיבָ֖א מַלְכָּֽא׃ (מ) עָנֵ֣ה וְאָמַ֗ר הָֽא־ אֲנָ֨ה חָזֵ֜ה גֻּבְרִ֣ין אַרְבְּעָ֗ה שְׁרַ֙יִן֙ מַהְלְכִ֣ין בְּגֽוֹא־ נוּרָ֔א וַחֲבָ֖ל לָא־ אִיתַ֣י בְּה֑וֹן וְרֵוֵהּ֙ דִּ֣י (רביעיא) רְֽבִיעָאָ֔ה דָּמֵ֖ה לְבַר־ אֱלָהִֽין׃ (מא) בֵּאדַ֜יִן קְרֵ֣ב נְבוּכַדְנֶצַּ֗ר לִתְרַע֮ אַתּ֣וּן נוּרָ֣א יָקִֽדְתָּא֒ עָנֵ֣ה וְאָמַ֗ר שַׁדְרַ֨ךְ מֵישַׁ֧ךְ וַעֲבֵד־ נְג֛וֹ עַבְד֛וֹהִי דִּֽי־ אֱלָהָ֥א (עליא) עִלָּאָ֖ה פֻּ֣קוּ וֶאֱת֑וֹ בֵּאדַ֣יִן נָֽפְקִ֗ין שַׁדְרַ֥ךְ מֵישַׁ֛ךְ וַעֲבֵ֥ד נְג֖וֹ מִן־ גּ֥וֹא נוּרָֽא׃ (מב) וּ֠מִֽתְכַּנְּשִׁין אֲחַשְׁדַּרְפְּנַיָּ֞א סִגְנַיָּ֣א וּפַחֲוָתָא֮ וְהַדָּבְרֵ֣י מַלְכָּא֒ חָזַ֣יִן לְגֻבְרַיָּ֣א אִלֵּ֡ךְ דִּי֩ לָֽא־ שְׁלֵ֨ט נוּרָ֜א בְּגֶשְׁמְה֗וֹן וּשְׂעַ֤ר רֵֽאשְׁהוֹן֙ לָ֣א הִתְחָרַ֔ךְ וְסָרְבָּלֵיה֖וֹן לָ֣א שְׁנ֑וֹ וְרֵ֣יחַ נ֔וּר לָ֥א עֲדָ֖ת בְּהֽוֹן׃ (מג) עָנֵ֨ה נְבֽוּכַדְנֶצַּ֜ר וְאָמַ֗ר בְּרִ֤יךְ אֱלָהֲהוֹן֙ דִּֽי־ שַׁדְרַ֤ךְ מֵישַׁךְ֙ וַעֲבֵ֣ד נְג֔וֹ דִּֽי־ שְׁלַ֤ח מַלְאֲכֵהּ֙ וְשֵׁיזִ֣ב לְעַבְד֔וֹהִי דִּ֥י הִתְרְחִ֖צוּ עֲל֑וֹהִי וּמִלַּ֤ת מַלְכָּא֙ שַׁנִּ֔יו וִיהַ֣בוּ (גשמיהון) גֶשְׁמְה֗וֹן דִּ֠י לָֽא־ יִפְלְח֤וּן וְלָֽא־ יִסְגְּדוּן֙ לְכָל־ אֱלָ֔הּ לָהֵ֖ן לֵאלָֽהֲהֽוֹן׃ (מד) וּמִנִּי֮ שִׂ֣ים טְעֵם֒ דִּי֩ כָל־ עַ֨ם אֻמָּ֜ה וְלִשָּׁ֗ן דִּֽי־ יֵאמַ֤ר (שלה) שָׁלוּ֙ עַ֣ל אֱלָהֲה֗וֹן דִּֽי־ שַׁדְרַ֤ךְ מֵישַׁךְ֙ וַעֲבֵ֣ד נְג֔וֹא הַדָּמִ֣ין יִתְעֲבֵ֔ד וּבַיְתֵ֖הּ נְוָלִ֣י יִשְׁתַּוֵּ֑ה כָּל־ קֳבֵ֗ל דִּ֣י לָ֤א אִיתַי֙ אֱלָ֣ה אָחֳרָ֔ן דִּֽי־ יִכֻּ֥ל לְהַצָּלָ֖ה כִּדְנָֽה׃
