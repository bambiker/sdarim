דניאל סדר א

פתיחה באתר · ספר דניאל כולו ·

(א) בִּשְׁנַ֣ת שָׁל֔וֹשׁ לְמַלְכ֖וּת יְהוֹיָקִ֣ים מֶֽלֶךְ־ יְהוּדָ֑ה בָּ֣א נְבוּכַדְנֶאצַּ֧ר מֶֽלֶךְ־ בָּבֶ֛ל יְרוּשָׁלִַ֖ם וַיָּ֥צַר עָלֶֽיהָ׃ (ב) וַיִּתֵּן֩ אֲדֹנָ֨י בְּיָד֜וֹ אֶת־ יְהוֹיָקִ֣ים מֶֽלֶךְ־ יְהוּדָ֗ה וּמִקְצָת֙ כְּלֵ֣י בֵית־ הָֽאֱלֹהִ֔ים וַיְבִיאֵ֥ם אֶֽרֶץ־ שִׁנְעָ֖ר בֵּ֣ית אֱלֹהָ֑יו וְאֶת־ הַכֵּלִ֣ים הֵבִ֔יא בֵּ֖ית אוֹצַ֥ר אֱלֹהָֽיו׃ (ג) וַיֹּ֣אמֶר הַמֶּ֔לֶךְ לְאַשְׁפְּנַ֖ז רַ֣ב סָרִיסָ֑יו לְהָבִ֞יא מִבְּנֵ֧י יִשְׂרָאֵ֛ל וּמִזֶּ֥רַע הַמְּלוּכָ֖ה וּמִן־ הַֽפַּרְתְּמִֽים׃ (ד) יְלָדִ֣ים אֲשֶׁ֣ר אֵֽין־ בָּהֶ֣ם כָּל־ (מאום) מוּם֩ וְטוֹבֵ֨י מַרְאֶ֜ה וּמַשְׂכִּילִ֣ים בְּכָל־ חָכְמָ֗ה וְיֹ֤דְעֵי דַ֙עַת֙ וּמְבִינֵ֣י מַדָּ֔ע וַאֲשֶׁר֙ כֹּ֣חַ בָּהֶ֔ם לַעֲמֹ֖ד בְּהֵיכַ֣ל הַמֶּ֑לֶךְ וּֽלֲלַמְּדָ֥ם סֵ֖פֶר וּלְשׁ֥וֹן כַּשְׂדִּֽים׃ (ה) וַיְמַן֩ לָהֶ֨ם הַמֶּ֜לֶךְ דְּבַר־ י֣וֹם בְּיוֹמ֗וֹ מִפַּת־ בַּ֤ג הַמֶּ֙לֶךְ֙ וּמִיֵּ֣ין מִשְׁתָּ֔יו וּֽלְגַדְּלָ֖ם שָׁנִ֣ים שָׁל֑וֹשׁ וּמִ֨קְצָתָ֔ם יַֽעַמְד֖וּ לִפְנֵ֥י הַמֶּֽלֶךְ׃ (ו) וַיְהִ֥י בָהֶ֖ם מִבְּנֵ֣י יְהוּדָ֑ה דָּנִיֵּ֣אל חֲנַנְיָ֔ה מִֽישָׁאֵ֖ל וַעֲזַרְיָֽה׃ (ז) וַיָּ֧שֶׂם לָהֶ֛ם שַׂ֥ר הַסָּרִיסִ֖ים שֵׁמ֑וֹת וַיָּ֨שֶׂם לְדָֽנִיֵּ֜אל בֵּ֣לְטְשַׁאצַּ֗ר וְלַֽחֲנַנְיָה֙ שַׁדְרַ֔ךְ וּלְמִֽישָׁאֵ֣ל מֵישַׁ֔ךְ וְלַעֲזַרְיָ֖ה עֲבֵ֥ד נְגֽוֹ׃ (ח) וַיָּ֤שֶׂם דָּנִיֵּאל֙ עַל־ לִבּ֔וֹ אֲשֶׁ֧ר לֹֽא־ יִתְגָּאַ֛ל בְּפַתְבַּ֥ג הַמֶּ֖לֶךְ וּבְיֵ֣ין מִשְׁתָּ֑יו וַיְבַקֵּשׁ֙ מִשַּׂ֣ר הַסָּרִיסִ֔ים אֲשֶׁ֖ר לֹ֥א יִתְגָּאָֽל׃ (ט) וַיִּתֵּ֤ן הָֽאֱלֹהִים֙ אֶת־ דָּ֣נִיֵּ֔אל לְחֶ֖סֶד וּֽלְרַחֲמִ֑ים לִפְנֵ֖י שַׂ֥ר הַסָּרִיסִֽים׃ (י) וַיֹּ֜אמֶר שַׂ֤ר הַסָּרִיסִים֙ לְדָ֣נִיֵּ֔אל יָרֵ֤א אֲנִי֙ אֶת־ אֲדֹנִ֣י הַמֶּ֔לֶךְ אֲשֶׁ֣ר מִנָּ֔ה אֶת־ מַאֲכַלְכֶ֖ם וְאֶת־ מִשְׁתֵּיכֶ֑ם אֲשֶׁ֡ר לָמָּה֩ יִרְאֶ֨ה אֶת־ פְּנֵיכֶ֜ם זֹֽעֲפִ֗ים מִן־ הַיְלָדִים֙ אֲשֶׁ֣ר כְּגִֽילְכֶ֔ם וְחִיַּבְתֶּ֥ם אֶת־ רֹאשִׁ֖י לַמֶּֽלֶךְ׃ (יא) וַיֹּ֥אמֶר דָּנִיֵּ֖אל אֶל־ הַמֶּלְצַ֑ר אֲשֶׁ֤ר מִנָּה֙ שַׂ֣ר הַסָּֽרִיסִ֔ים עַל־ דָּנִיֵּ֣אל חֲנַנְיָ֔ה מִֽישָׁאֵ֖ל וַעֲזַרְיָֽה׃ (יב) נַס־ נָ֥א אֶת־ עֲבָדֶ֖יךָ יָמִ֣ים עֲשָׂרָ֑ה וְיִתְּנוּ־ לָ֜נוּ מִן־ הַזֵּרֹעִ֛ים וְנֹאכְלָ֖ה וּמַ֥יִם וְנִשְׁתֶּֽה׃ (יג) וְיֵרָא֤וּ לְפָנֶ֙יךָ֙ מַרְאֵ֔ינוּ וּמַרְאֵה֙ הַיְלָדִ֔ים הָאֹ֣כְלִ֔ים אֵ֖ת פַּתְבַּ֣ג הַמֶּ֑לֶךְ וְכַאֲשֶׁ֣ר תִּרְאֵ֔ה עֲשֵׂ֖ה עִם־ עֲבָדֶֽיךָ׃ (יד) וַיִּשְׁמַ֥ע לָהֶ֖ם לַדָּבָ֣ר הַזֶּ֑ה וַיְנַסֵּ֖ם יָמִ֥ים עֲשָׂרָֽה׃ (טו) וּמִקְצָת֙ יָמִ֣ים עֲשָׂרָ֔ה נִרְאָ֤ה מַרְאֵיהֶם֙ ט֔וֹב וּבְרִיאֵ֖י בָּשָׂ֑ר מִן־ כָּל־ הַיְלָדִ֔ים הָאֹ֣כְלִ֔ים אֵ֖ת פַּתְבַּ֥ג הַמֶּֽלֶךְ׃ (טז) וַיְהִ֣י הַמֶּלְצַ֗ר נֹשֵׂא֙ אֶת־ פַּתְבָּגָ֔ם וְיֵ֖ין מִשְׁתֵּיהֶ֑ם וְנֹתֵ֥ן לָהֶ֖ם זֵרְעֹנִֽים׃ (יז) וְהַיְלָדִ֤ים הָאֵ֙לֶּה֙ אַרְבַּעְתָּ֔ם נָתַ֨ן לָהֶ֧ם הָֽאֱלֹהִ֛ים מַדָּ֥ע וְהַשְׂכֵּ֖ל בְּכָל־ סֵ֣פֶר וְחָכְמָ֑ה וְדָנִיֵּ֣אל הֵבִ֔ין בְּכָל־ חָז֖וֹן וַחֲלֹמֽוֹת׃ (יח) וּלְמִקְצָת֙ הַיָּמִ֔ים אֲשֶׁר־ אָמַ֥ר הַמֶּ֖לֶךְ לַהֲבִיאָ֑ם וַיְבִיאֵם֙ שַׂ֣ר הַסָּרִיסִ֔ים לִפְנֵ֖י נְבֻכַדְנֶצַּֽר׃ (יט) וַיְדַבֵּ֣ר אִתָּם֮ הַמֶּלֶךְ֒ וְלֹ֤א נִמְצָא֙ מִכֻּלָּ֔ם כְּדָנִיֵּ֣אל חֲנַנְיָ֔ה מִֽישָׁאֵ֖ל וַעֲזַרְיָ֑ה וַיַּֽעַמְד֖וּ לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כ) וְכֹ֗ל דְּבַר֙ חָכְמַ֣ת בִּינָ֔ה אֲשֶׁר־ בִּקֵּ֥שׁ מֵהֶ֖ם הַמֶּ֑לֶךְ וַֽיִּמְצָאֵ֞ם עֶ֣שֶׂר יָד֗וֹת עַ֤ל כָּל־ הַֽחַרְטֻמִּים֙ הָֽאַשָּׁפִ֔ים אֲשֶׁ֖ר בְּכָל־ מַלְכוּתֽוֹ׃ (כא) וַֽיְהִי֙ דָּֽנִיֵּ֔אל עַד־ שְׁנַ֥ת אַחַ֖ת לְכ֥וֹרֶשׁ הַמֶּֽלֶךְ׃ (כב) וּבִשְׁנַ֣ת שְׁתַּ֗יִם לְמַלְכוּת֙ נְבֻֽכַדְנֶצַּ֔ר חָלַ֥ם נְבֻֽכַדְנֶצַּ֖ר חֲלֹמ֑וֹת וַתִּתְפָּ֣עֶם רוּח֔וֹ וּשְׁנָת֖וֹ נִהְיְתָ֥ה עָלָֽיו׃ (כג) וַיֹּ֣אמֶר הַ֠מֶּלֶךְ לִקְרֹ֨א לַֽחַרְטֻמִּ֜ים וְלָֽאַשָּׁפִ֗ים וְלַֽמְכַשְּׁפִים֙ וְלַכַּשְׂדִּ֔ים לְהַגִּ֥יד לַמֶּ֖לֶךְ חֲלֹמֹתָ֑יו וַיָּבֹ֕אוּ וַיַּֽעַמְד֖וּ לִפְנֵ֥י הַמֶּֽלֶךְ׃ (כד) וַיֹּ֧אמֶר לָהֶ֛ם הַמֶּ֖לֶךְ חֲל֣וֹם חָלָ֑מְתִּי וַתִּפָּ֣עֶם רוּחִ֔י לָדַ֖עַת אֶֽת־ הַחֲלֽוֹם׃ (כה) וַֽיְדַבְּר֧וּ הַכַּשְׂדִּ֛ים לַמֶּ֖לֶךְ אֲרָמִ֑ית מַלְכָּא֙ לְעָלְמִ֣ין חֱיִ֔י אֱמַ֥ר חֶלְמָ֛א (לעבדיך) לְעַבְדָ֖ךְ וּפִשְׁרָ֥א נְחַוֵּֽא׃ (כו) עָנֵ֤ה מַלְכָּא֙ וְאָמַ֣ר (לכשדיא) לְכַשְׂדָּאֵ֔י מִלְּתָ֖א מִנִּ֣י אַזְדָּ֑א הֵ֣ן לָ֤א תְהֽוֹדְעוּנַּ֙נִי֙ חֶלְמָ֣א וּפִשְׁרֵ֔הּ הַדָּמִין֙ תִּתְעַבְד֔וּן וּבָתֵּיכ֖וֹן נְוָלִ֥י יִתְּשָׂמֽוּן׃ (כז) וְהֵ֨ן חֶלְמָ֤א וּפִשְׁרֵהּ֙ תְּֽהַחֲוֺ֔ן מַתְּנָ֤ן וּנְבִזְבָּה֙ וִיקָ֣ר שַׂגִּ֔יא תְּקַבְּל֖וּן מִן־ קֳדָמָ֑י לָהֵ֕ן חֶלְמָ֥א וּפִשְׁרֵ֖הּ הַחֲוֺֽנִי׃ (כח) עֲנ֥וֹ תִנְיָנ֖וּת וְאָמְרִ֑ין מַלְכָּ֕א חֶלְמָ֛א יֵאמַ֥ר לְעַבְד֖וֹהִי וּפִשְׁרָ֥ה נְהַחֲוֵֽה׃ (כט) עָנֵ֤ה מַלְכָּא֙ וְאָמַ֔ר מִן־ יַצִּיב֙ יָדַ֣ע אֲנָ֔ה דִּ֥י עִדָּנָ֖א אַנְתּ֣וּן זָבְנִ֑ין כָּל־ קֳבֵל֙ דִּ֣י חֲזֵית֔וֹן דִּ֥י אַזְדָּ֖א מִנִּ֥י מִלְּתָֽא׃ (ל) דִּ֣י הֵן־ חֶלְמָא֩ לָ֨א תְהֽוֹדְעֻנַּ֜נִי חֲדָה־ הִ֣יא דָֽתְכ֗וֹן וּמִלָּ֨ה כִדְבָ֤ה וּשְׁחִיתָה֙ (הזמנתון) הִזְדְּמִנְתּוּן֙ לְמֵאמַ֣ר קָֽדָמַ֔י עַ֛ד דִּ֥י עִדָּנָ֖א יִשְׁתַּנֵּ֑א לָהֵ֗ן חֶלְמָא֙ אֱמַ֣רוּ לִ֔י וְֽאִנְדַּ֕ע דִּ֥י פִשְׁרֵ֖הּ תְּהַחֲוֻנַּֽנִי׃ (לא) עֲנ֨וֹ (כשדיא) כַשְׂדָּאֵ֤י קֳדָם־ מַלְכָּא֙ וְאָ֣מְרִ֔ין לָֽא־ אִיתַ֤י אֲנָשׁ֙ עַל־ יַבֶּשְׁתָּ֔א דִּ֚י מִלַּ֣ת מַלְכָּ֔א יוּכַ֖ל לְהַחֲוָיָ֑ה כָּל־ קֳבֵ֗ל דִּ֚י כָּל־ מֶ֙לֶךְ֙ רַ֣ב וְשַׁלִּ֔יט מִלָּ֤ה כִדְנָה֙ לָ֣א שְׁאֵ֔ל לְכָל־ חַרְטֹּ֖ם וְאָשַׁ֥ף וְכַשְׂדָּֽי׃ (לב) וּמִלְּתָ֨א דִֽי־ מַלְכָּ֤ה שָׁאֵל֙ יַקִּירָ֔ה וְאָחֳרָן֙ לָ֣א אִיתַ֔י דִּ֥י יְחַוִּנַּ֖הּ קֳדָ֣ם מַלְכָּ֑א לָהֵ֣ן אֱלָהִ֔ין דִּ֚י מְדָ֣רְה֔וֹן עִם־ בִּשְׂרָ֖א לָ֥א אִיתֽוֹהִי׃ (לג) כָּל־ קֳבֵ֣ל דְּנָ֔ה מַלְכָּ֕א בְּנַ֖ס וּקְצַ֣ף שַׂגִּ֑יא וַאֲמַר֙ לְה֣וֹבָדָ֔ה לְכֹ֖ל חַכִּימֵ֥י בָבֶֽל׃ (לד) וְדָתָ֣א נֶפְקַ֔ת וְחַכִּֽימַיָּ֖א מִֽתְקַטְּלִ֑ין וּבְע֛וֹ דָּנִיֵּ֥אל וְחַבְר֖וֹהִי לְהִתְקְטָלָֽה׃ (לה) בֵּאדַ֣יִן דָּנִיֵּ֗אל הֲתִיב֙ עֵטָ֣א וּטְעֵ֔ם לְאַרְי֕וֹךְ רַב־ טַבָּחַיָּ֖א דִּ֣י מַלְכָּ֑א דִּ֚י נְפַ֣ק לְקַטָּלָ֔ה לְחַכִּימֵ֖י בָּבֶֽל׃ (לו) עָנֵ֣ה וְאָמַ֗ר לְאַרְיוֹךְ֙ שַׁלִּיטָ֣א דִֽי־ מַלְכָּ֔א עַל־ מָ֥ה דָתָ֛א מְהַחְצְפָ֖ה מִן־ קֳדָ֣ם מַלְכָּ֑א אֱדַ֣יִן מִלְּתָ֔א הוֹדַ֥ע אַרְי֖וֹךְ לְדָנִיֵּֽאל׃ (לז) וְדָ֣נִיֵּ֔אל עַ֖ל וּבְעָ֣ה מִן־ מַלְכָּ֑א דִּ֚י זְמָ֣ן יִנְתֵּן־ לֵ֔הּ וּפִשְׁרָ֖א לְהַֽחֲוָיָ֥ה לְמַלְכָּֽא׃ (לח) אֱדַ֥יִן דָּֽנִיֵּ֖אל לְבַיְתֵ֣הּ אֲזַ֑ל וְ֠לַחֲנַנְיָה מִֽישָׁאֵ֧ל וַעֲזַרְיָ֛ה חַבְר֖וֹהִי מִלְּתָ֥א הוֹדַֽע׃ (לט) וְרַחֲמִ֗ין לְמִבְעֵא֙ מִן־ קֳדָם֙ אֱלָ֣הּ שְׁמַיָּ֔א עַל־ רָזָ֖ה דְּנָ֑ה דִּ֣י לָ֤א יְהֹֽבְדוּן֙ דָּנִיֵּ֣אל וְחַבְר֔וֹהִי עִם־ שְׁאָ֖ר חַכִּימֵ֥י בָבֶֽל׃ (מ) אֱדַ֗יִן לְדָנִיֵּ֛אל בְּחֶזְוָ֥א דִֽי־ לֵילְיָ֖א רָזָ֣ה גֲלִ֑י אֱדַ֙יִן֙ דָּֽנִיֵּ֔אל בָּרִ֖ךְ לֶאֱלָ֥הּ שְׁמַיָּֽא׃ (מא) עָנֵ֤ה דָֽנִיֵּאל֙ וְאָמַ֔ר לֶהֱוֵ֨א שְׁמֵ֤הּ דִּֽי־ אֱלָהָא֙ מְבָרַ֔ךְ מִן־ עָלְמָ֖א וְעַ֣ד־ עָלְמָ֑א דִּ֧י חָכְמְתָ֛א וּגְבוּרְתָ֖א דִּ֥י לֵֽהּ־ הִֽיא׃ (מב) וְ֠הוּא מְהַשְׁנֵ֤א עִדָּנַיָּא֙ וְזִמְנַיָּ֔א מְהַעְדֵּ֥ה מַלְכִ֖ין וּמְהָקֵ֣ים מַלְכִ֑ין יָהֵ֤ב חָכְמְתָא֙ לְחַכִּימִ֔ין וּמַנְדְּעָ֖א לְיָדְעֵ֥י בִינָֽה׃ (מג) ה֛וּא גָּלֵ֥א עַמִּיקָתָ֖א וּמְסַתְּרָתָ֑א יָדַע֙ מָ֣ה בַחֲשׁוֹכָ֔א (ונהירא) וּנְהוֹרָ֖א עִמֵּ֥הּ שְׁרֵֽא׃ (מד) לָ֣ךְ ׀ אֱלָ֣הּ אֲבָהָתִ֗י מְהוֹדֵ֤א וּמְשַׁבַּח֙ אֲנָ֔ה דִּ֧י חָכְמְתָ֛א וּגְבוּרְתָ֖א יְהַ֣בְתְּ לִ֑י וּכְעַ֤ן הֽוֹדַעְתַּ֙נִי֙ דִּֽי־ בְעֵ֣ינָא מִנָּ֔ךְ דִּֽי־ מִלַּ֥ת מַלְכָּ֖א הוֹדַעְתֶּֽנָא׃ (מה) כָּל־ קֳבֵ֣ל דְּנָ֗ה דָּֽנִיֵּאל֙ עַ֣ל עַל־ אַרְי֔וֹךְ דִּ֚י מַנִּ֣י מַלְכָּ֔א לְהוֹבָדָ֖ה לְחַכִּימֵ֣י בָבֶ֑ל אֲזַ֣ל ׀ וְכֵ֣ן אֲמַר־ לֵ֗הּ לְחַכִּימֵ֤י בָבֶל֙ אַל־ תְּהוֹבֵ֔ד הַעֵ֙לְנִי֙ קֳדָ֣ם מַלְכָּ֔א וּפִשְׁרָ֖א לְמַלְכָּ֥א אֲחַוֵּֽא׃ (מו) אֱדַ֤יִן אַרְיוֹךְ֙ בְּהִתְבְּהָלָ֔ה הַנְעֵ֥ל לְדָנִיֵּ֖אל קֳדָ֣ם מַלְכָּ֑א וְכֵ֣ן אֲמַר־ לֵ֗הּ דִּֽי־ הַשְׁכַּ֤חַת גְּבַר֙ מִן־ בְּנֵ֤י גָֽלוּתָא֙ דִּ֣י יְה֔וּד דִּ֥י פִשְׁרָ֖א לְמַלְכָּ֥א יְהוֹדַֽע׃ (מז) עָנֵ֤ה מַלְכָּא֙ וְאָמַ֣ר לְדָנִיֵּ֔אל דִּ֥י שְׁמֵ֖הּ בֵּלְטְשַׁאצַּ֑ר (האיתיך) הַֽאִיתָ֣ךְ כָּהֵ֗ל לְהוֹדָעֻתַ֛נִי חֶלְמָ֥א דִֽי־ חֲזֵ֖ית וּפִשְׁרֵֽהּ׃ (מח) עָנֵ֧ה דָנִיֵּ֛אל קֳדָ֥ם מַלְכָּ֖א וְאָמַ֑ר רָזָה֙ דִּֽי־ מַלְכָּ֣א שָׁאֵ֔ל לָ֧א חַכִּימִ֣ין אָֽשְׁפִ֗ין חַרְטֻמִּין֙ גָּזְרִ֔ין יָכְלִ֖ין לְהַֽחֲוָיָ֥ה לְמַלְכָּֽא׃ (מט) בְּרַ֡ם אִיתַ֞י אֱלָ֤הּ בִּשְׁמַיָּא֙ גָּלֵ֣א רָזִ֔ין וְהוֹדַ֗ע לְמַלְכָּא֙ נְבֽוּכַדְנֶצַּ֔ר מָ֛ה דִּ֥י לֶהֱוֵ֖א בְּאַחֲרִ֣ית יוֹמַיָּ֑א חֶלְמָ֨ךְ וְחֶזְוֵ֥י רֵאשָׁ֛ךְ עַֽל־ מִשְׁכְּבָ֖ךְ דְּנָ֥ה הֽוּא׃ (נ) אַ֣נְתְּה מַלְכָּ֗א רַעְיוֹנָךְ֙ עַל־ מִשְׁכְּבָ֣ךְ סְלִ֔קוּ מָ֛ה דִּ֥י לֶהֱוֵ֖א אַחֲרֵ֣י דְנָ֑ה וְגָלֵ֧א רָזַיָּ֛א הוֹדְעָ֖ךְ מָה־ דִ֥י לֶהֱוֵֽא׃ (נא) וַאֲנָ֗ה לָ֤א בְחָכְמָה֙ דִּֽי־ אִיתַ֥י בִּי֙ מִן־ כָּל־ חַיַּיָּ֔א רָזָ֥א דְנָ֖ה גֱּלִ֣י לִ֑י לָהֵ֗ן עַל־ דִּבְרַת֙ דִּ֤י פִשְׁרָא֙ לְמַלְכָּ֣א יְהוֹדְע֔וּן וְרַעְיוֹנֵ֥י לִבְבָ֖ךְ תִּנְדַּֽע׃ (נב) אַ֣נְתְּה מַלְכָּ֗א חָזֵ֤ה הֲוַ֙יְתָ֙ וַאֲל֨וּ צְלֵ֥ם חַד֙ שַׂגִּ֔יא צַלְמָ֨א דִּכֵּ֥ן רַ֛ב וְזִיוֵ֥הּ יַתִּ֖יר קָאֵ֣ם לְקָבְלָ֑ךְ וְרֵוֵ֖הּ דְּחִֽיל׃ (נג) ה֣וּא צַלְמָ֗א רֵאשֵׁהּ֙ דִּֽי־ דְהַ֣ב טָ֔ב חֲד֥וֹהִי וּדְרָע֖וֹהִי דִּ֣י כְסַ֑ף מְע֥וֹהִי וְיַרְכָתֵ֖הּ דִּ֥י נְחָֽשׁ׃ (נד) שָׁק֖וֹהִי דִּ֣י פַרְזֶ֑ל רַגְל֕וֹהִי (מנהון) מִנְּהֵין֙ דִּ֣י פַרְזֶ֔ל (ומנהון) וּמִנְּהֵ֖ין דִּ֥י חֲסַֽף׃ (נה) חָזֵ֣ה הֲוַ֗יְתָ עַ֠ד דִּ֣י הִתְגְּזֶ֤רֶת אֶ֙בֶן֙ דִּי־ לָ֣א בִידַ֔יִן וּמְחָ֤ת לְצַלְמָא֙ עַל־ רַגְל֔וֹהִי דִּ֥י פַרְזְלָ֖א וְחַסְפָּ֑א וְהַדֵּ֖קֶת הִמּֽוֹן׃
