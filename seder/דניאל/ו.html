לפי החלוקה לפרקים: דניאל ט, ד – י, כ · 44 פסוקים

פתיחה באתר · ספר דניאל כולו ·

(א) וָֽאֶתְפַּֽלְלָ֛ה לַיהוָ֥ה אֱלֹהַ֖י וָאֶתְוַדֶּ֑ה וָאֹֽמְרָ֗ה אָנָּ֤א אֲדֹנָי֙ הָאֵ֤ל הַגָּדוֹל֙ וְהַנּוֹרָ֔א שֹׁמֵ֤ר הַבְּרִית֙ וְֽהַחֶ֔סֶד לְאֹהֲבָ֖יו וּלְשֹׁמְרֵ֥י מִצְוֺתָֽיו׃ (ב) חָטָ֥אנוּ וְעָוִ֖ינוּ (והרשענו) הִרְשַׁ֣עְנוּ וּמָרָ֑דְנוּ וְס֥וֹר מִמִּצְוֺתֶ֖ךָ וּמִמִּשְׁפָּטֶֽיךָ׃ (ג) וְלֹ֤א שָׁמַ֙עְנוּ֙ אֶל־ עֲבָדֶ֣יךָ הַנְּבִיאִ֔ים אֲשֶׁ֤ר דִּבְּרוּ֙ בְּשִׁמְךָ֔ אֶל־ מְלָכֵ֥ינוּ שָׂרֵ֖ינוּ וַאֲבֹתֵ֑ינוּ וְאֶ֖ל כָּל־ עַ֥ם הָאָֽרֶץ׃ (ד) לְךָ֤ אֲדֹנָי֙ הַצְּדָקָ֔ה וְלָ֛נוּ בֹּ֥שֶׁת הַפָּנִ֖ים כַּיּ֣וֹם הַזֶּ֑ה לְאִ֤ישׁ יְהוּדָה֙ וּלְיוֹשְׁבֵ֣י יְרֽוּשָׁלִַ֔ם וּֽלְכָל־ יִשְׂרָאֵ֞ל הַקְּרֹבִ֣ים וְהָרְחֹקִ֗ים בְּכָל־ הָֽאֲרָצוֹת֙ אֲשֶׁ֣ר הִדַּחְתָּ֣ם שָׁ֔ם בְּמַעֲלָ֖ם אֲשֶׁ֥ר מָֽעֲלוּ־ בָֽךְ׃ (ה) יְהוָ֗ה לָ֚נוּ בֹּ֣שֶׁת הַפָּנִ֔ים לִמְלָכֵ֥ינוּ לְשָׂרֵ֖ינוּ וְלַאֲבֹתֵ֑ינוּ אֲשֶׁ֥ר חָטָ֖אנוּ לָֽךְ׃ (ו) לַֽאדֹנָ֣י אֱלֹהֵ֔ינוּ הָרַחֲמִ֖ים וְהַסְּלִח֑וֹת כִּ֥י מָרַ֖דְנוּ בּֽוֹ׃ (ז) וְלֹ֣א שָׁמַ֔עְנוּ בְּק֖וֹל יְהוָ֣ה אֱלֹהֵ֑ינוּ לָלֶ֤כֶת בְּתֽוֹרֹתָיו֙ אֲשֶׁ֣ר נָתַ֣ן לְפָנֵ֔ינוּ בְּיַ֖ד עֲבָדָ֥יו הַנְּבִיאִֽים׃ (ח) וְכָל־ יִשְׂרָאֵ֗ל עָֽבְרוּ֙ אֶת־ תּ֣וֹרָתֶ֔ךָ וְס֕וֹר לְבִלְתִּ֖י שְׁמ֣וֹעַ בְּקֹלֶ֑ךָ וַתִּתַּ֨ךְ עָלֵ֜ינוּ הָאָלָ֣ה וְהַשְּׁבֻעָ֗ה אֲשֶׁ֤ר כְּתוּבָה֙ בְּתוֹרַת֙ מֹשֶׁ֣ה עֶֽבֶד־ הָֽאֱלֹהִ֔ים כִּ֥י חָטָ֖אנוּ לֽוֹ׃ (ט) וַיָּ֜קֶם אֶת־ (דבריו) דְּבָר֣וֹ ׀ אֲשֶׁר־ דִּבֶּ֣ר עָלֵ֗ינוּ וְעַ֤ל שֹֽׁפְטֵ֙ינוּ֙ אֲשֶׁ֣ר שְׁפָט֔וּנוּ לְהָבִ֥יא עָלֵ֖ינוּ רָעָ֣ה גְדֹלָ֑ה אֲשֶׁ֣ר לֹֽא־ נֶעֶשְׂתָ֗ה תַּ֚חַת כָּל־ הַשָּׁמַ֔יִם כַּאֲשֶׁ֥ר נֶעֶשְׂתָ֖ה בִּירוּשָׁלִָֽם׃ (י) כַּאֲשֶׁ֤ר כָּתוּב֙ בְּתוֹרַ֣ת מֹשֶׁ֔ה אֵ֛ת כָּל־ הָרָעָ֥ה הַזֹּ֖את בָּ֣אָה עָלֵ֑ינוּ וְלֹֽא־ חִלִּ֜ינוּ אֶת־ פְּנֵ֣י ׀ יְהוָ֣ה אֱלֹהֵ֗ינוּ לָשׁוּב֙ מֵֽעֲוֺנֵ֔נוּ וּלְהַשְׂכִּ֖יל בַּאֲמִתֶּֽךָ׃ (יא) וַיִּשְׁקֹ֤ד יְהוָה֙ עַל־ הָ֣רָעָ֔ה וַיְבִיאֶ֖הָ עָלֵ֑ינוּ כִּֽי־ צַדִּ֞יק יְהוָ֣ה אֱלֹהֵ֗ינוּ עַל־ כָּל־ מַֽעֲשָׂיו֙ אֲשֶׁ֣ר עָשָׂ֔ה וְלֹ֥א שָׁמַ֖עְנוּ בְּקֹלֽוֹ׃ (יב) וְעַתָּ֣ה ׀ אֲדֹנָ֣י אֱלֹהֵ֗ינוּ אֲשֶׁר֩ הוֹצֵ֨אתָ אֶֽת־ עַמְּךָ֜ מֵאֶ֤רֶץ מִצְרַ֙יִם֙ בְּיָ֣ד חֲזָקָ֔ה וַתַּֽעַשׂ־ לְךָ֥ שֵׁ֖ם כַּיּ֣וֹם הַזֶּ֑ה חָטָ֖אנוּ רָשָֽׁעְנוּ׃ (יג) אֲדֹנָ֗י כְּכָל־ צִדְקֹתֶ֙ךָ֙ יָֽשָׁב־ נָ֤א אַפְּךָ֙ וַחֲמָ֣תְךָ֔ מֵעִֽירְךָ֥ יְרוּשָׁלִַ֖ם הַר־ קָדְשֶׁ֑ךָ כִּ֤י בַחֲטָאֵ֙ינוּ֙ וּבַעֲוֺנ֣וֹת אֲבֹתֵ֔ינוּ יְרוּשָׁלִַ֧ם וְעַמְּךָ֛ לְחֶרְפָּ֖ה לְכָל־ סְבִיבֹתֵֽינוּ׃ (יד) וְעַתָּ֣ה ׀ שְׁמַ֣ע אֱלֹהֵ֗ינוּ אֶל־ תְּפִלַּ֤ת עַבְדְּךָ֙ וְאֶל־ תַּ֣חֲנוּנָ֔יו וְהָאֵ֣ר פָּנֶ֔יךָ עַל־ מִקְדָּשְׁךָ֖ הַשָּׁמֵ֑ם לְמַ֖עַן אֲדֹנָֽי׃ (טו) הַטֵּ֨ה אֱלֹהַ֥י ׀ אָזְנְךָ֮ וּֽשֲׁמָע֒ (פקחה) פְּקַ֣ח עֵינֶ֗יךָ וּרְאֵה֙ שֹֽׁמְמֹתֵ֔ינוּ וְהָעִ֕יר אֲשֶׁר־ נִקְרָ֥א שִׁמְךָ֖ עָלֶ֑יהָ כִּ֣י ׀ לֹ֣א עַל־ צִדְקֹתֵ֗ינוּ אֲנַ֨חְנוּ מַפִּילִ֤ים תַּחֲנוּנֵ֙ינוּ֙ לְפָנֶ֔יךָ כִּ֖י עַל־ רַחֲמֶ֥יךָ הָרַבִּֽים׃ (טז) אֲדֹנָ֤י ׀ שְׁמָ֙עָה֙ אֲדֹנָ֣י ׀ סְלָ֔חָה אֲדֹנָ֛י הַֽקֲשִׁ֥יבָה וַעֲשֵׂ֖ה אַל־ תְּאַחַ֑ר לְמַֽעֲנְךָ֣ אֱלֹהַ֔י כִּֽי־ שִׁמְךָ֣ נִקְרָ֔א עַל־ עִירְךָ֖ וְעַל־ עַמֶּֽךָ׃ (יז) וְע֨וֹד אֲנִ֤י מְדַבֵּר֙ וּמִתְפַּלֵּ֔ל וּמִתְוַדֶּה֙ חַטָּאתִ֔י וְחַטַּ֖את עַמִּ֣י יִשְׂרָאֵ֑ל וּמַפִּ֣יל תְּחִנָּתִ֗י לִפְנֵי֙ יְהוָ֣ה אֱלֹהַ֔י עַ֖ל הַר־ קֹ֥דֶשׁ אֱלֹהָֽי׃ (יח) וְע֛וֹד אֲנִ֥י מְדַבֵּ֖ר בַּתְּפִלָּ֑ה וְהָאִ֣ישׁ גַּבְרִיאֵ֡ל אֲשֶׁר֩ רָאִ֨יתִי בֶחָז֤וֹן בַּתְּחִלָּה֙ מֻעָ֣ף בִּיעָ֔ף נֹגֵ֣עַ אֵלַ֔י כְּעֵ֖ת מִנְחַת־ עָֽרֶב׃ (יט) וַיָּ֖בֶן וַיְדַבֵּ֣ר עִמִּ֑י וַיֹּאמַ֕ר דָּנִיֵּ֕אל עַתָּ֥ה יָצָ֖אתִי לְהַשְׂכִּילְךָ֥ בִינָֽה׃ (כ) בִּתְחִלַּ֨ת תַּחֲנוּנֶ֜יךָ יָצָ֣א דָבָ֗ר וַאֲנִי֙ בָּ֣אתִי לְהַגִּ֔יד כִּ֥י חֲמוּד֖וֹת אָ֑תָּה וּבִין֙ בַּדָּבָ֔ר וְהָבֵ֖ן בַּמַּרְאֶֽה׃ (כא) שָׁבֻעִ֨ים שִׁבְעִ֜ים נֶחְתַּ֥ךְ עַֽל־ עַמְּךָ֣ ׀ וְעַל־ עִ֣יר קָדְשֶׁ֗ךָ לְכַלֵּ֨א הַפֶּ֜שַׁע (ולחתם) וּלְהָתֵ֤ם (חטאות) חַטָּאת֙ וּלְכַפֵּ֣ר עָוֺ֔ן וּלְהָבִ֖יא צֶ֣דֶק עֹֽלָמִ֑ים וְלַחְתֹּם֙ חָז֣וֹן וְנָבִ֔יא וְלִמְשֹׁ֖חַ קֹ֥דֶשׁ קָֽדָשִֽׁים׃ (כב) וְתֵדַ֨ע וְתַשְׂכֵּ֜ל מִן־ מֹצָ֣א דָבָ֗ר לְהָשִׁיב֙ וְלִבְנ֤וֹת יְרֽוּשָׁלִַ֙ם֙ עַד־ מָשִׁ֣יחַ נָגִ֔יד שָׁבֻעִ֖ים שִׁבְעָ֑ה וְשָׁבֻעִ֞ים שִׁשִּׁ֣ים וּשְׁנַ֗יִם תָּשׁוּב֙ וְנִבְנְתָה֙ רְח֣וֹב וְחָר֔וּץ וּבְצ֖וֹק הָעִתִּֽים׃ (כג) וְאַחֲרֵ֤י הַשָּׁבֻעִים֙ שִׁשִּׁ֣ים וּשְׁנַ֔יִם יִכָּרֵ֥ת מָשִׁ֖יחַ וְאֵ֣ין ל֑וֹ וְהָעִ֨יר וְהַקֹּ֜דֶשׁ יַ֠שְׁחִית עַ֣ם נָגִ֤יד הַבָּא֙ וְקִצּ֣וֹ בַשֶּׁ֔טֶף וְעַד֙ קֵ֣ץ מִלְחָמָ֔ה נֶחֱרֶ֖צֶת שֹׁמֵמֽוֹת׃ (כד) וְהִגְבִּ֥יר בְּרִ֛ית לָרַבִּ֖ים שָׁב֣וּעַ אֶחָ֑ד וַחֲצִ֨י הַשָּׁב֜וּעַ יַשְׁבִּ֣ית ׀ זֶ֣בַח וּמִנְחָ֗ה וְעַ֨ל כְּנַ֤ף שִׁקּוּצִים֙ מְשֹׁמֵ֔ם וְעַד־ כָּלָה֙ וְנֶ֣חֱרָצָ֔ה תִּתַּ֖ךְ עַל־ שֹׁמֵֽם׃ (כה) בִּשְׁנַ֣ת שָׁל֗וֹשׁ לְכ֙וֹרֶשׁ֙ מֶ֣לֶךְ פָּרַ֔ס דָּבָר֙ נִגְלָ֣ה לְדָֽנִיֵּ֔אל אֲשֶׁר־ נִקְרָ֥א שְׁמ֖וֹ בֵּלְטְשַׁאצַּ֑ר וֶאֱמֶ֤ת הַדָּבָר֙ וְצָבָ֣א גָד֔וֹל וּבִין֙ אֶת־ הַדָּבָ֔ר וּבִ֥ינָה ל֖וֹ בַּמַּרְאֶֽה׃ (כו) בַּיָּמִ֖ים הָהֵ֑ם אֲנִ֤י דָֽנִיֵּאל֙ הָיִ֣יתִי מִתְאַבֵּ֔ל שְׁלֹשָׁ֥ה שָׁבֻעִ֖ים יָמִֽים׃ (כז) לֶ֣חֶם חֲמֻד֞וֹת לֹ֣א אָכַ֗לְתִּי וּבָשָׂ֥ר וָיַ֛יִן לֹא־ בָ֥א אֶל־ פִּ֖י וְס֣וֹךְ לֹא־ סָ֑כְתִּי עַד־ מְלֹ֕את שְׁלֹ֥שֶׁת שָׁבֻעִ֖ים יָמִֽים׃ (כח) וּבְי֛וֹם עֶשְׂרִ֥ים וְאַרְבָּעָ֖ה לַחֹ֣דֶשׁ הָרִאשׁ֑וֹן וַאֲנִ֗י הָיִ֛יתִי עַ֣ל יַ֧ד הַנָּהָ֛ר הַגָּד֖וֹל ה֥וּא חִדָּֽקֶל׃ (כט) וָאֶשָּׂ֤א אֶת־ עֵינַי֙ וָאֵ֔רֶא וְהִנֵּ֥ה אִישׁ־ אֶחָ֖ד לָב֣וּשׁ בַּדִּ֑ים וּמָתְנָ֥יו חֲגֻרִ֖ים בְּכֶ֥תֶם אוּפָֽז׃ (ל) וּגְוִיָּת֣וֹ כְתַרְשִׁ֗ישׁ וּפָנָ֞יו כְּמַרְאֵ֤ה בָרָק֙ וְעֵינָיו֙ כְּלַפִּ֣ידֵי אֵ֔שׁ וּזְרֹֽעֹתָיו֙ וּמַרְגְּלֹתָ֔יו כְּעֵ֖ין נְחֹ֣שֶׁת קָלָ֑ל וְק֥וֹל דְּבָרָ֖יו כְּק֥וֹל הָמֽוֹן׃ (לא) וְרָאִיתִי֩ אֲנִ֨י דָנִיֵּ֤אל לְבַדִּי֙ אֶת־ הַמַּרְאָ֔ה וְהָאֲנָשִׁים֙ אֲשֶׁ֣ר הָי֣וּ עִמִּ֔י לֹ֥א רָא֖וּ אֶת־ הַמַּרְאָ֑ה אֲבָ֗ל חֲרָדָ֤ה גְדֹלָה֙ נָפְלָ֣ה עֲלֵיהֶ֔ם וַֽיִּבְרְח֖וּ בְּהֵחָבֵֽא׃ (לב) וַאֲנִי֙ נִשְׁאַ֣רְתִּי לְבַדִּ֔י וָֽאֶרְאֶ֗ה אֶת־ הַמַּרְאָ֤ה הַגְּדֹלָה֙ הַזֹּ֔את וְלֹ֥א נִשְׁאַר־ בִּ֖י כֹּ֑ח וְהוֹדִ֗י נֶהְפַּ֤ךְ עָלַי֙ לְמַשְׁחִ֔ית וְלֹ֥א עָצַ֖רְתִּי כֹּֽחַ׃ (לג) וָאֶשְׁמַ֖ע אֶת־ ק֣וֹל דְּבָרָ֑יו וּכְשָׁמְעִי֙ אֶת־ ק֣וֹל דְּבָרָ֔יו וַאֲנִ֗י הָיִ֛יתִי נִרְדָּ֥ם עַל־ פָּנַ֖י וּפָנַ֥י אָֽרְצָה׃ (לד) וְהִנֵּה־ יָ֖ד נָ֣גְעָה בִּ֑י וַתְּנִיעֵ֥נִי עַל־ בִּרְכַּ֖י וְכַפּ֥וֹת יָדָֽי׃ (לה) וַיֹּ֣אמֶר אֵלַ֡י דָּנִיֵּ֣אל אִישׁ־ חֲ֠מֻדוֹת הָבֵ֨ן בַּדְּבָרִ֜ים אֲשֶׁר֩ אָנֹכִ֨י דֹבֵ֤ר אֵלֶ֙יךָ֙ וַעֲמֹ֣ד עַל־ עָמְדֶ֔ךָ כִּ֥י עַתָּ֖ה שֻׁלַּ֣חְתִּי אֵלֶ֑יךָ וּבְדַבְּר֥וֹ עִמִּ֛י אֶת־ הַדָּבָ֥ר הַזֶּ֖ה עָמַ֥דְתִּי מַרְעִֽיד׃ (לו) וַיֹּ֣אמֶר אֵלַי֮ אַל־ תִּירָ֣א דָנִיֵּאל֒ כִּ֣י ׀ מִן־ הַיּ֣וֹם הָרִאשׁ֗וֹן אֲשֶׁ֨ר נָתַ֧תָּ אֶֽת־ לִבְּךָ֛ לְהָבִ֧ין וּלְהִתְעַנּ֛וֹת לִפְנֵ֥י אֱלֹהֶ֖יךָ נִשְׁמְע֣וּ דְבָרֶ֑יךָ וַאֲנִי־ בָ֖אתִי בִּדְבָרֶֽיךָ׃ (לז) וְשַׂ֣ר ׀ מַלְכ֣וּת פָּרַ֗ס עֹמֵ֤ד לְנֶגְדִּי֙ עֶשְׂרִ֣ים וְאֶחָ֣ד י֔וֹם וְהִנֵּ֣ה מִֽיכָאֵ֗ל אַחַ֛ד הַשָּׂרִ֥ים הָרִאשֹׁנִ֖ים בָּ֣א לְעָזְרֵ֑נִי וַאֲנִי֙ נוֹתַ֣רְתִּי שָׁ֔ם אֵ֖צֶל מַלְכֵ֥י פָרָֽס׃ (לח) וּבָ֙אתִי֙ לַהֲבִ֣ינְךָ֔ אֵ֛ת אֲשֶׁר־ יִקְרָ֥ה לְעַמְּךָ֖ בְּאַחֲרִ֣ית הַיָּמִ֑ים כִּי־ ע֥וֹד חָז֖וֹן לַיָּמִֽים׃ (לט) וּבְדַבְּר֣וֹ עִמִּ֔י כַּדְּבָרִ֖ים הָאֵ֑לֶּה נָתַ֧תִּי פָנַ֛י אַ֖רְצָה וְנֶאֱלָֽמְתִּי׃ (מ) וְהִנֵּ֗ה כִּדְמוּת֙ בְּנֵ֣י אָדָ֔ם נֹגֵ֖עַ עַל־ שְׂפָתָ֑י וָאֶפְתַּח־ פִּ֗י וָאֲדַבְּרָה֙ וָאֹֽמְרָה֙ אֶל־ הָעֹמֵ֣ד לְנֶגְדִּ֔י אֲדֹנִ֗י בַּמַּרְאָה֙ נֶהֶפְכ֤וּ צִירַי֙ עָלַ֔י וְלֹ֥א עָצַ֖רְתִּי כֹּֽחַ׃ (מא) וְהֵ֣יךְ יוּכַ֗ל עֶ֤בֶד אֲדֹנִי֙ זֶ֔ה לְדַבֵּ֖ר עִם־ אֲדֹ֣נִי זֶ֑ה וַאֲנִ֤י מֵעַ֙תָּה֙ לֹֽא־ יַעֲמָד־ בִּ֣י כֹ֔חַ וּנְשָׁמָ֖ה לֹ֥א נִשְׁאֲרָה־ בִֽי׃ (מב) וַיֹּ֧סֶף וַיִּגַּע־ בִּ֛י כְּמַרְאֵ֥ה אָדָ֖ם וַֽיְחַזְּקֵֽנִי׃ (מג) וַיֹּ֜אמֶר אַל־ תִּירָ֧א אִישׁ־ חֲמֻד֛וֹת שָׁל֥וֹם לָ֖ךְ חֲזַ֣ק וַחֲזָ֑ק וּֽבְדַבְּר֤וֹ עִמִּי֙ הִתְחַזַּ֔קְתִּי וָאֹֽמְרָ֛ה יְדַבֵּ֥ר אֲדֹנִ֖י כִּ֥י חִזַּקְתָּֽנִי׃ (מד) וַיֹּ֗אמֶר הֲיָדַ֙עְתָּ֙ לָמָּה־ בָּ֣אתִי אֵלֶ֔יךָ וְעַתָּ֣ה אָשׁ֔וּב לְהִלָּחֵ֖ם עִם־ שַׂ֣ר פָּרָ֑ס וַאֲנִ֣י יוֹצֵ֔א וְהִנֵּ֥ה שַׂר־ יָוָ֖ן בָּֽא׃
